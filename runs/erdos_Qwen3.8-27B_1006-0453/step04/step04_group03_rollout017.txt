from scipy.optimize import minimize
import time

_target_sum = [0.0]

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def _proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _sum_constraint(x):
    return x.sum() - _target_sum[0]

def _slsqp_polish(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    _target_sum[0] = n / 2.0
    cons = [{'type': 'eq', 'fun': _sum_constraint}]
    def obj(x):
        x = np.clip(x, 0, 1)
        return _smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return _proj_sum(np.clip(res.x, 0, 1), n / 2.0)

def _coord_descent(h, t_end, tol=1e-13):
    n = len(h)

    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode='full')

                return h, best_c5
            h_old = h[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            idx1 = i + k_vals - (n - 1)
            idx2 = k_vals - (n - 1) - i
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            coef = np.zeros(2 * n - 1)
            coef[m1] = 1.0 - h[idx1[m1]]
            coef[m2] -= h[idx2[m2]]
            a2, b2 = lo, hi
            c2 = b2 - gr * (b2 - a2)
            d2 = a2 + gr * (b2 - a2)
            fc = np.max(corr + c2 * coef)
            fd = np.max(corr + d2 * coef)
            for _ in range(35):

                    b2 = d2; d2 = c2; fd = fc
                    c2 = b2 - gr * (b2 - a2)
                    fc = np.max(corr + c2 * coef)

                    d2 = a2 + gr * (b2 - a2)

                    max_improve = imp
                if c5_new < best_c5:
                    best_c5 = c5_new
        if max_improve < tol:
            break
    return h, best_c5
