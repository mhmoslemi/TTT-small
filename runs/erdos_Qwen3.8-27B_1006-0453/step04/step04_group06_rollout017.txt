import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def power_mean_c5(h, p, n):
    """Power mean approximation to max of correlation * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx  # all values in [0, 1]
    # Power mean of order p
    return np.sum(corr_scaled**p)**(1.0/p)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Water-filling style projection
    if s > target:
        # Need to reduce
        excess = s - target
        # Reduce proportionally where h > 0
        mask = h > 0
        if mask.sum() == 0:
            return h
        scale = 1.0 - excess / s
        h = h * scale
    else:
        # Need to increase
        deficit = target - s
        mask = h < 1
        room = np.where(mask, 1.0 - h, 0)
        room_sum = room.sum()
        if room_sum < 1e-14:
            return h
        h = h + room * (deficit / room_sum)
    return np.clip(h, 0.0, 1.0)

def coord_descent_fast(h, n, t_end, max_sweeps=200, verbose=False):
    """Coordinate descent with full correlation recomputation each sweep."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            x_new = fast_coord_opt(best_h, n, i)
            if abs(x_new - best_h[i]) > 1e-10:
                best_h[i] = x_new
                best_h = project(best_h, n)
                improved = True
        # Recompute c5
        c5 = compute_c5(best_h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
        if not improved:

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        # Scale down
        h = h * (target / s)
    else:
        # Scale up where possible
        room = 1.0 - h
        room_sum = room.sum()
        if room_sum < 1e-14:
            return h
        deficit = target - s
        h = h + room * (deficit / room_sum)
    return np.clip(h, 0.0, 1.0)

def coord_optimize(h, n, i):
    """Optimize coordinate i. Returns new value."""
    h0 = h.copy()
    h0[i] = 0.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    center = n - 1
    a = np.zeros(2*n - 1)
    for k in range(1, n):
        if i + k < n:
            a[center + k] += 1.0 - h[i + k]
        if i - k >= 0:
            a[center + k] -= h[i - k]
        if i - k >= 0:
            a[center - k] += 1.0 - h[i - k]
        if i + k < n:
            a[center - k] -= h[i + k]
    
    def f(x):
        v = corr0 + x * a
        v[center] += x - x*x
        return np.max(v)
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def cd_sweep(h, n, t_end, max_sweeps=100, full_recompute_every=1):
    """Coordinate descent sweeps."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for sweep in range(max_sweeps):

            break
        order = np.random.permutation(n)
        improved = False

                best_h = project(best_h, n)
                improved = True
        c5 = compute_c5(best_h)

def slsqp_power(h, p, n, maxiter=500):

    return np.clip(res.x, 0, 1)
