import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_raw(h):
    """Return max corr before multiplying by dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def binarize(h, n):
    target = n // 2
    # Use threshold with tie-breaking
    bh = (h >= 0.5).astype(float)
    if bh.sum() != target:
        # Adjust
        if bh.sum() > target:
            # Flip some 1s to 0s (those closest to 0.5)
            ones = np.where(bh == 1.0)[0]
            order = np.argsort(-h[ones])  # smallest h values among ones
            for idx in order:
                bh[ones[idx]] = 0.0
                if bh.sum() == target:
                    break
        else:
            zeros = np.where(bh == 0.0)[0]
            order = np.argsort(-h[zeros])  # largest h values among zeros
            for idx in order:
                bh[zeros[idx]] = 1.0
                if bh.sum() == target:
                    break
    return bh

def local_search(h, n, t_end, rng, max_no_improve=50000):
    """Greedy local search on binary sequence."""
    h = h.copy()
    curr = compute_c5_raw(h)
    dx = 2.0 / n
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    no_imp = 0
    while time.time() < t_end:
        if no_imp > max_no_improve:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_val = compute_c5_raw(h)
        if new_val < curr - 1e-15:
            curr = new_val
            # Update lists
            if h[i] == 1.0:
                # i was 0, j was 1 -> i is now 1, j is now 0
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
            else:
                # i was 1, j was 0 -> i is now 0, j is now 1
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            no_imp = 0
        else:
            h[i], h[j] = h[j], h[i]  # revert
            no_imp += 1
    return h, curr * dx

    zeros = np.where(h == 0.0)[0].tolist()

    it = 0

        elapsed = time.time() - t_start
        frac = min(elapsed / total_time, 1.0)
        T = T0 * (1e-5) ** frac  # Exponential cooling over time

        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_raw = compute_c5_raw(h)
        delta = (new_raw - curr_raw) * dx

            curr_raw = new_raw
            curr = new_raw * dx
            # Update lists
            if h[i] == 1.0:
                ones.remove(j); ones.append(i)
                zeros.remove(i); zeros.append(j)
            else:
                ones.remove(i); ones.append(j)

            if new_raw < best_raw:
                best_raw = new_raw
                best_h = h.copy()
                best = new_raw * dx
        else:
            h[i], h[j] = h[j], h[i]
        
        it += 1
    
    return best_h, best

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_raw(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def binarize(h, n):
    target = n // 2
    # Round to nearest, then fix count
    bh = (h >= 0.5).astype(float).copy()
    diff = bh.sum() - target
    if diff > 0:
        # Too many ones: flip the ones with smallest h value
        ones = np.where(bh == 1.0)[0]
        order = np.argsort(h[ones])
        for idx in range(int(diff)):
            bh[ones[order[idx]]] = 0.0
    elif diff < 0:
        # Too few ones: flip the zeros with largest h value
        zeros = np.where(bh == 0.0)[0]
        order = np.argsort(-h[zeros])
        for idx in range(int(-diff)):
            bh[zeros[order[idx]]] = 1.0
    return bh

def local_search(h, n, t_end, rng, max_no_improve=30000):
    h = h.copy()
    dx = 2.0 / n
    curr_raw = compute_c5_raw(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    no_imp = 0
    while time.time() < t_end and no_imp < max_no_improve:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        was_i = h[i]
        h[i], h[j] = h[j], h[i]
        new_raw = compute_c5_raw(h)
        if new_raw < curr_raw - 1e-12:

                ones.remove(i); ones.append(j)
                zeros.remove(j); zeros.append(i)
            else:
                ones.remove(j); ones.append(i)
                zeros.remove(i); zeros.append(j)

        else:
            h[i], h[j] = h[j], h[i]
            no_imp += 1
    return h, curr_raw * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_raw(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def smooth_c5_raw(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_binary(h, n):
    """Project to [0,1] with sum = n/2, preferring binary."""
    vec = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    # Bisection for sum constraint
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(vec + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) / 2, 0, 1)

def binarize(h, n):
    target = n // 2
    bh = (h >= 0.5).astype(float).copy()
    diff = int(bh.sum()) - target
    if diff > 0:
        ones = np.where(bh == 1.0)[0]
        order = np.argsort(h[ones])
        for idx in range(diff):
            bh[ones[order[idx]]] = 0.0
    elif diff < 0:
        zeros = np.where(bh == 0.0)[0]
        order = np.argsort(-h[zeros])
        for idx in range(-diff):
            bh[zeros[order[idx]]] = 1.0
    return bh

        T = T0 * (T_final / T0) ** frac

def continuous_polish(h, n, t_end, T=200):
    """SLSQP continuous optimization."""
    def obj(x):
        return smooth_c5_raw(x, T)
    def con_sum(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_sum}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-14})
        if res.success or compute_c5(np.clip(res.x, 0, 1)) < compute_c5(h):
            return project_binary(res.x, n)
    except:
        pass
    return h
