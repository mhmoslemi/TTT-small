import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h0, t_end, max_sweeps=5000):
    h = h0.astype(float).copy()
    n = len(h)
    dx = 2.0 / n
    nm1 = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    n_gs = 35

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            h_old = h[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            coef = np.zeros(nm1)
            mlo = max(-(n-1), i - n + 1)
            mhi = min(n-1, i)
            if mlo <= mhi:
                ms = np.arange(mlo, mhi + 1)
                coef[ms + n - 1] = 1.0 - h[i - ms]
            mlo2 = max(-(n-1), -i)
            mhi2 = min(n-1, n - 1 - i)
            if mlo2 <= mhi2:
                ms2 = np.arange(mlo2, mhi2 + 1)
                coef[ms2 + n - 1] -= h[i + ms2]
            if np.max(np.abs(coef)) < 1e-15:
                continue
            a, b = lo, hi
            m1 = a + (1.0 - gr) * (b - a)
            m2 = a + gr * (b - a)
            f1 = np.max(corr + m1 * coef)
            f2 = np.max(corr + m2 * coef)
            for _ in range(n_gs):
                if f1 < f2:
                    b = m2; m2 = m1; f2 = f1
                    m1 = a + (1.0 - gr) * (b - a)
                    f1 = np.max(corr + m1 * coef)
                else:
                    a = m1; m1 = m2; f1 = f2
                    m2 = a + gr * (b - a)
                    f2 = np.max(corr + m2 * coef)
            s_new = (a + b) / 2.0
            f_new = np.max(corr + s_new * coef)
            f_cur = np.max(corr)
            if f_new < f_cur - 1e-16:
                h[i] = np.clip(h[i] + s_new, 0.0, 1.0)
                corr = corr + s_new * coef
                best_c5 = np.max(corr) * dx
                improved = True
        if not improved:
            break
    return h, best_c5

def slsqp_opt(h0, T, n, maxiter=800, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h0, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def lbfgsb_opt(h0, T, n, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    tgt = n / 2.0
    P = max(10.0, 100.0 / (1.0 + T))
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(c, T) * dx + P * (x.sum() - tgt) ** 2
    res = minimize(obj, h0, method='L-BFGS-B', bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-9})
    return project_to_sum(np.clip(res.x, 0, 1), tgt)
