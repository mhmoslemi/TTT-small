import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(100):
        s = h.sum()
        d = tgt - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            if r.sum() < 1e-14:
                break
            h += r * (d / r.sum())
        else:
            if s < 1e-14:
                break
            h = np.clip(h * (tgt / s), 0, 1)
    return np.clip(h, 0, 1)

def coord_opt(h, n, i):
    dx = 2.0 / n
    hs = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = hs
    d = np.zeros(2 * n - 1)
    for L in range(-(n - 1), n):
        if L == 0:
            continue
        v = 0.0
        j1 = i - L
        if 0 <= j1 < n:
            v += 1.0 - h[j1]
        j2 = i + L
        if 0 <= j2 < n:
            v -= h[j2]
        d[n - 1 + L] = v
    c0 = cb[n - 1]
    def f1d(t):
        v = cb + t * d
        v[n - 1] = c0 + t * (1.0 - t)
        return np.max(v) * dx
    return minimize_scalar(f1d, bounds=(0, 1), method='bounded',
                           options={'xatol': 1e-12}).x

def slsqp(h, T, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    res = minimize(lambda x: smooth_c5(x, T), h, method='SLSQP',
                   bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def coord_desc(h, n, t_end, rng, maxsw=80):
    bh = h.copy()

            break
        imp = False
        order = rng.permutation(n)

                return bh, bc
            t = coord_opt(bh, n, i)
            if abs(t - bh[i]) > 1e-11:
                bh[i] = t
                bh = proj(bh, n)
                c = compute_c5(bh)
                if c < bc - 1e-14:
                    bc, imp = c, True
        stall = 0 if imp else stall + 1
        if stall >= 3:
            break
    return bh, bc

def bump_pert(h, n, rng, cool):
    p = h.copy()
    for _ in range(rng.integers(1, 4)):
        c = rng.uniform(0, n)
        w = rng.uniform(3, 10 + 15 * cool)
        a = rng.uniform(-1, 1) * (0.03 + 0.1 * cool)
        x = np.arange(n, dtype=float)
        p += a * np.exp(-0.5 * ((x - c) / w) ** 2)
    return proj(p, n)

def hop(h, n, t_end, rng, nits=200):
