def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(T * (corr - m), 0.0)
    exp_s = np.exp(shifted)
    Z = np.sum(exp_s)
    w = exp_s / Z
    val = (m + np.log(Z) / T) * dx
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-14:
            continue
        L = idx - (n - 1)
        lo1, hi1 = max(0, L), min(n - 1, n - 1 + L)
        if lo1 <= hi1:
            js = np.arange(lo1, hi1 + 1)
            grad[js] += wk * (1.0 - h[js - L])
        lo2, hi2 = max(0, -L), min(n - 1, n - 1 - L)
        if lo2 <= hi2:
            js = np.arange(lo2, hi2 + 1)
            grad[js] += wk * (-h[js + L])
    grad *= dx
    return val, grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * ((target - s) / tot)
    else:
        h *= (target / s)
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if s2 < target:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += room * ((target - s2) / tot)
    return h

def slsqp_analytic(h, T, n, maxiter=400, ftol=1e-13):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return smooth_c5_and_grad(x, T)[0]
    def jac(x):
        return smooth_c5_and_grad(x, T)[1]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   jac=jac, options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_opt_fixed(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if 0 <= i - k < n:
            d[n - 1 + k] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            d[n - 1 + k] += (-h[i + k])
        if 0 <= i + k < n:
            d[n - 1 - k] += (1.0 - h[i + k])
        if 0 <= i - k < n:
            d[n - 1 - k] += (-h[i - k])
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=50, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fixed(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
            else:
                best_h[i] = hi
                best_h = project(best_h, n)
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.80,
                                        max_sweeps=50, rng=rng)

    # Phase 5: Another round of top-K + SLSQP
    if time.time() - t0 < budget_s * 0.88:
        try:
            hc, cv = topk_projected_grad(best_h, n, 6, n_iters=50, alpha_init=0.01)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass
        for T in [5000, 20000]:
            if time.time() - t0 > budget_s * 0.92:
                break
            try:
                hc = slsqp_analytic(best_h, T, n, maxiter=500, ftol=1e-14)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5 = hc, cv
            except Exception:
                pass

    # Phase 6: Coordinate descent again
    if time.time() - t0 < budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95,
                                        max_sweeps=30, rng=rng)

    # Phase 7: Try alternative resolutions
    for n_try in [200, 300, 160]:
        if time.time() - t0 > budget_s * 0.985:
            break
        h_interp = interpolate_h(best_h, n_try)
        for T in [500, 5000]:
            if time.time() - t0 > budget_s * 0.99:
                break
            try:
                hc = slsqp_analytic(h_interp, T, n_try, maxiter=300, ftol=1e-13)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5, n = hc, cv, n_try
            except Exception:
                pass
        if len(best_h) == n_try and time.time() - t0 < budget_s * 0.995:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995,
                                            max_sweeps=10, rng=rng)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
