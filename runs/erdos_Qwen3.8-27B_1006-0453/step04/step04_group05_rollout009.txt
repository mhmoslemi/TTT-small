- h[i-k] * (1-h[i]) for the lag -k term (as second factor)

So corr[k] as a function of h[i] involves:
- h[i] * (1-h[i+k]) (linear in h[i])
- (1-h[i]) * h[i-k] = h[i-k] - h[i]*h[i-k] (linear in h[i])

So corr[k] = (terms not involving h[i]) + h[i]*((1-h[i+k]) - h[i-k]) for lag k.

This is linear in h[i]! So we CAN do coordinate descent.

For coordinate descent on h[i]:
- For each lag k, the coefficient of h[i] in corr[k] is: a_k = (1-h[i+k]) - h[i-k] (with appropriate boundary handling)
- The constant term (not involving h[i]) is: b_k = corr[k] - h[i]*a_k (current value minus the h[i] contribution)
- We want to minimize max_k (b_k + h[i]*a_k) subject to 0 ≤ h[i] ≤ 1

This is a 1D problem: minimize max of linear functions of h[i] over [0,1]. The minimum of max of linears is at an intersection point or at a boundary.

- If a_k = 0: need t ≥ corr[k]

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = h.clip(0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def binary_sd(h, N, deadline, max_sweeps=500):
    """Steepest descent on binary h using M-matrix approach."""
    h = h.astype(float).copy()
    nh = N // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > nh:
        h[ones[len(ones)-nh:]] = 0
    elif len(ones) < nh:
        h[zeros[:nh-len(ones)]] = 1
    
    lags = np.arange(2*N-1) - (N-1)
    p_idx = np.arange(N)[:, None]
    lag_idx = lags[None, :]
    total = p_idx + lag_idx
    valid = (total >= 0) & (total < N)
    
    def rebuild_M(h):
        return np.where(valid, 1.0 - h[np.clip(total, 0, N-1)], 0.0)
    
    M = rebuild_M(h)
    best_pk = float(np.max(compute_corr(h)))
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        P = float(c.max())
        if P <= best_pk:
            best_pk = P
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_new = P + 1
        best_pi = -1
        best_qi = -1
        
        for pi in range(len(ones)):
            p = ones[pi]
            base = c - M[p, :]
            # For each zero, compute max(base + M[q, :])
            # Vectorized: M[zeros] has shape (n_zeros, 2N-1)
            # Add base[None, :] to each row
            # Take max along axis 1
            peaks = (M[zeros] + base).max(axis=1)
            mi = np.argmin(peaks)
            if peaks[mi] < best_new:
                best_new = float(peaks[mi])
                best_pi = pi
                best_qi = int(mi)
        
        if best_new >= P - 1e-10:
            break
        
        h[ones[best_pi]] = 0.0
        h[zeros[best_qi]] = 1.0
        M = rebuild_M(h)
    
    return h

def anneal_binary(h, N, n_iters, rng, deadline, T0=2.0, T1=0.01):
    h = h.astype(float).copy()
    nh = N // 2
    best = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    cur_pk = best_pk
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1/T0) ** (it / max(n_iters, 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_pk = float(np.max(compute_corr(h)))
        delta = new_pk - cur_pk
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-6)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best, best_pk * (2.0/N)

def adam_continuous(h, N, deadline, k=30, lr0=0.03, decay=0.995):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-7
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    center = N - 1
    
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        # Gradient: focus on top-k lags
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        wk = 1.0 / k
        for ai in top:
            L = int(ai) - center
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += wk * (1.0 - h[:N-L])
                grad[:N-L] -= wk * h[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += wk * (1.0 - h[M:])
                    grad[M:] -= wk * h[:N-M]
        
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h -= lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        
        if it % 300 == 0:
            lr *= decay
    
    return best_h, best_peak * dx
