    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B with progressive temperature and penalty
    schedules = [
        (30, 50, 300), (100, 200, 300), (500, 1000, 250),
        (2000, 5000, 250), (10000, 20000, 200),
        (50000, 100000, 200), (200000, 500000, 150),
    ]
    for T, mu, mit in schedules:
        if elapsed() > budget_s * 0.30:
            break
        hc = lbfgsb_step(best_h.copy(), T, mu, n, maxiter=mit)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.50, max_sweeps=600)

    # Phase 3: Block coordinate descent
    best_h, best_c5 = block_cd(best_h, n, t0 + budget_s * 0.62, block_sz=3, max_sweeps=100)

    # Phase 4: Multi-restart diversification
    n_restarts = 15
    for r in range(n_restarts):
        if elapsed() > budget_s * 0.82:
            break
        cool = 1.0 - r / float(n_restarts)
        sz = 0.03 + 0.08 * cool
        h_pert = best_h + sz * np.random.randn(n)
        h_pert = project(h_pert, n)
        T_r = 2000 + 80000 * (r / n_restarts)
        mu_r = 5000
        try:
            hc = lbfgsb_step(h_pert, T_r, mu_r, n, maxiter=200)
        except Exception:
            hc = h_pert
        c5v = compute_c5(hc)
        if c5v < best_c5:
            hc, c5v = coord_descent(hc, n, min(t0 + budget_s * 0.82, time.time() + 8), max_sweeps=80)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Intensive CD + block
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=800)
    best_h, best_c5 = block_cd(best_h, n, t0 + budget_s * 0.93, block_sz=4, max_sweeps=80)

    # Phase 6: Final L-BFGS-B polish
    if elapsed() < budget_s * 0.96:
        for T_f in [100000, 500000, 2000000]:
            if elapsed() > budget_s * 0.97:
                break
            try:
                hc = lbfgsb_step(best_h.copy(), T_f, 100000, n, maxiter=500)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Final CD polish
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=300)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
