def project_polytope(h, n, S):
    """Project h onto {0<=x<=1, sum(x)=S} via L2 projection."""
    # Bisection on lambda
    lo, hi = h.min() - S, h.max()  # lambda range
    # Actually: we need sum(clip(h - lambda, 0, 1)) = S
    # As lambda decreases, sum increases.

        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > S:
            lo = mid  # need to increase lambda (decrease sum)
        else:
            hi = mid
    proj = np.clip(h - (lo + hi)/2, 0, 1)
    # Small correction for exactness
    diff = S - proj.sum()
    if abs(diff) > 1e-12:

            tot = room.sum()
            if tot > 1e-14:

        else:

            if tot > 1e-14:
                proj[proj > 0] *= 1 + diff / tot
    return np.clip(proj, 0, 1)

For the problem as stated: h: [0,2] → [0,1], ∫h = 1. The "length" is 2 and the measure is 1, so h has average value 1/2 on [0,2]. The overlap at lag k is ∫h(x)(1-h(x+k))dx.

If h were constant 1/2, the overlap would be ∫(1/2)(1/2)dx = 1/4 for all k (where both are defined). But the domain is [0,2], so for non-zero k, the integral is over a smaller range.

Actually wait - the correlation is computed on the discrete samples h[0..n-1] and (1-h[0..n-1]), with mode='full'. So the lags go from -(n-1) to (n-1), and for lag k, the overlap is between h[i] and (1-h[i+k]) for valid i.

For the continuous version, the integral ∫h(x)(1-h(x+k))dx is over the intersection of [0,2] and [0-k, 2-k], which has measure 2-|k| for |k|≤2.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_and_grad(h, T):
    """Compute smoothed C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    s = np.sum(exps)
    p = exps / s  # softmax weights
    val = (m + np.log(s) / T) * dx
    # Gradient: sum_j p[j] * d(corr[j])/dh
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = p[j]
        if w < 1e-12:
            continue
        k = j - (n - 1)
        # d(corr[j])/dh[m] = (1-h[m-k])*I[0<=m-k<n] - h[m+k]*I[0<=m+k<n]
        # Part 1: (1-h[m-k]) for m where 0<=m-k<n
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - h[lo1-k:hi1+1-k])
        # Part 2: -h[m+k] for m where 0<=m+k<n
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2+k:hi2+1+k]
    grad *= dx
    return val, grad

def project_simplex_box(h, n):
    """Project onto {0<=x<=1, sum(x)=n/2}."""
    S = n / 2.0
    # Bisection for L2 projection
    lo = min(h.min() - 1, -2.0)
    hi = max(h.max() + 1, 2.0)

        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Tiny correction
    diff = S - proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:

            tot = room.sum()
            if tot > 1e-14:

            tot = proj[mask].sum()
            if tot > 1e-14:

def gradient_descent_smooth(h, n, t_end, T_start=50, T_end=50000, lr_start=0.01):

    work_h = best_h.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)

        # Adaptive T
        frac = min(1.0, t / 500.0)

            if 0.0 < old < 1.0:
                res = minimize_scalar(lambda v: cd_obj(v, old, corr, coeff, c0_base, c0_idx),
                                      bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-9})
                if res.fun * dx < best_c5 - 1e-14:
                    best_h[i] = res.x
                    best_c5 = res.fun * dx
                    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def grad_descent_smooth(h, n, t_end, T_start=30.0, T_end=30000.0, lr_start=0.02):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = best_h.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    while time.time() < t_end:
        t += 1
        frac = min(1.0, t / 800.0)
        T_cur = T_start * (T_end / T_start) ** frac
        lr = lr_start / (1.0 + 0.05 * t)
        _, grad = smooth_c5_grad(work_h, T_cur)
        m1 = b1 * m1 + (1 - b1) * grad
        m2 = b2 * m2 + (1 - b2) * grad * grad
        m1h = m1 / (1 - b1 ** t)
        m2h = m2 / (1 - b2 ** t)
        work_h = work_h - lr * m1h / (np.sqrt(m2h) + eps)
        work_h = project_simplex_box(work_h, n)
        if t % 15 == 0 or t <= 5:
            c5v = compute_c5(work_h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = work_h.copy()
        if t % 150 == 0 and time.time() < t_end:
            work_h, c5v = coord_descent(work_h, n, min(time.time() + 8, t_end), max_sweeps=30)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = work_h.copy()
        if t % 300 == 0:
            lr = lr_start * (1.0 + 0.1 * t / 300.0)
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_simplex_box(h_new, n_new)

def basin_hop(work_h, work_c5, n, t_end, max_iters=120):
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.05, 1.0 - 0.7 * it / max_iters)
        step = 0.02 + 0.12 * cool
        h_p = work_h + step * np.random.randn(n)
        h_p = project_simplex_box(h_p, n)
        T_use = 100.0 + 2000.0 * cool
        _, grad = smooth_c5_grad(h_p, T_use)
        h_p = project_simplex_box(h_p - 0.004 * grad * n / 10.0, n)
        c5v = compute_c5(h_p)
        temp = 0.003 * cool + 0.0002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = h_p, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, h_p.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.ones(n_init) * 0.5
        best_h[::2] = 1.0
    n = len(best_h)
    best_h = project_simplex_box(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Quick coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.07, max_sweeps=80)

    # Phase 2: Smoothed gradient descent (Adam)
    best_h, best_c5 = grad_descent_smooth(best_h, n, t0 + budget_s * 0.32, T_start=30, T_end=20000, lr_start=0.025)

    # Phase 3: Coordinate descent refinement
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.42, max_sweeps=150)

    # Phase 4: Multi-resolution exploration
    n_list = [100, 120, 140, 200, 240, 320]
    for n_try in n_list:
        if elapsed() > budget_s * 0.65:
            break
        h_tr = interp_h(best_h, n_try)
        t_sub = min(t0 + budget_s * 0.67, time.time() + budget_s * 0.025)
        h_tr, c5_tr = grad_descent_smooth(h_tr, n_try, t_sub, T_start=50, T_end=10000, lr_start=0.015)
        h_tr, c5_tr = coord_descent(h_tr, n_try, min(t0 + budget_s * 0.67, time.time() + budget_s * 0.015), max_sweeps=40)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 5: Basin hopping
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.82, max_iters=100)

    # Phase 6: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.92, max_sweeps=200)

    # Phase 7: Final gradient polish
    if elapsed() < budget_s * 0.96:
        best_h, best_c5 = grad_descent_smooth(best_h, n, t0 + budget_s * 0.985, T_start=5000, T_end=80000, lr_start=0.005)
        best_h, best_c5 = coord_descent(best_h, n, time.time() + 2, max_sweeps=30)

    best_h = project_simplex_box(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
