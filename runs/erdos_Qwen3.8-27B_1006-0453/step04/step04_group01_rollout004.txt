    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T, lam_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    logits = T * (corr - m)
    logits -= np.max(logits)  # numerical stability
    soft_max = m + np.log(np.sum(np.exp(logits))) / T
    sum_penalty = lam_sum * (h.sum() - n/2.0)**2
    return soft_max * dx + sum_penalty

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(corr, T):
    m = np.max(corr)
    if T < 1e-6:
        return m
    logits = T * (corr - m)
    return m + np.log(np.sum(np.exp(logits))) / T

def obj_func(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_c5(corr, T) * dx
    val += lam * (h.sum() - n/2.0)**2
    return val

def obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    logits = T * (corr - m)
    logits -= np.max(logits)
    exp_l = np.exp(logits)
    w = exp_l / exp_l.sum()
    
    grad = np.zeros(n)
    for i in range(n):
        # Term 1: (1-h[i-k]) for k where 0<=i-k<n
        # k_idx = k + (n-1), k in [i-(n-1), i], k_idx in [i, i+n-1]
        klo = max(0, i)
        khi = min(2*n-2, i+n-1)
        if klo <= khi:
            ki = np.arange(klo, khi+1)
            k = ki - (n-1)
            idx = i - k
            grad[i] += np.dot(w[ki], 1.0 - h[idx])
        # Term 2: -h[i+k] for k where 0<=i+k<n
        # k in [-i, n-1-i], k_idx in [n-1-i, 2n-2-i]
        klo2 = max(0, n-1-i)
        khi2 = min(2*n-2, 2*n-2-i)
        if klo2 <= khi2:
            ki2 = np.arange(klo2, khi2+1)
            k2 = ki2 - (n-1)
            idx2 = i + k2
            grad[i] -= np.dot(w[ki2], h[idx2])
    
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n/2.0)
    return grad

def project(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-13:
        return vc
    lo = -vc.max() - 1.0
    hi = 1.0 - vc.min() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5*(lo+hi), 0.0, 1.0)

def lbfgs_opt(h, T, lam, maxiter=400):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_func, h, jac=obj_grad, method='L-BFGS-B',
                   bounds=bnds, args=(T, lam),
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                            'maxfun': maxiter*3})
    return project(res.x, n)

def coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Optimize h[i] in [0,1]
            def f1d(hi):
                h_try = best_h.copy()
                h_try[i] = hi
                corr = np.correlate(h_try, 1.0 - h_try, mode='full')
                return np.max(corr) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-12})
            hi_new = res.x
            if abs(hi_new - best_h[i]) > 1e-11:
                h_try = best_h.copy()
                h_try[i] = hi_new
                h_try = project(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        no_improve = 0 if improved else no_improve + 1
