    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0)-1.0)/2.0

            a=c; c=d; fc=fd
            d = a+gr*(b-a); fd = np.max(cb+d*coefs)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            if s < 1e-14:
                break
            h *= (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_opt(h, n, i):
    h_i = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_i
    kr = np.arange(-(n - 1), n)
    coefs = np.zeros(2 * n - 1)
    v1 = (i + kr >= 0) & (i + kr < n)
    v2 = (i - kr >= 0) & (i - kr < n)
    coefs[v1] += (1.0 - h[i + kr[v1]])
    coefs[v2] -= h[i - kr[v2]]
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = np.max(cb + c * coefs)
    fd = np.max(cb + d * coefs)
    for _ in range(55):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a); fc = np.max(cb + c * coefs)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a); fd = np.max(cb + d * coefs)
    return (a + b) / 2.0

def coord_sweep(h, n, t_end, max_sweeps=3000, kick_size=5):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stag = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-11:
                best_h[i] = hi
                best_h = project_h(best_h)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if not improved:
            stag += 1
            if stag >= kick_size:
                nk = max(2, n // 20)
                idx = np.random.randint(0, n, size=nk)
                best_h[idx] += 0.02 * np.random.randn(nk)
                best_h = project_h(best_h)
                best_c5 = compute_c5(best_h)
                stag = 0
        else:
            stag = 0
    return best_h, best_c5

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp_opt(h, T, n, maxiter=600, ftol=1e-16):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def basin_hop(h, n, t_end, n_iters=100):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / float(n_iters))
        r = np.random.random()
        if r < 0.25:
            pert = (0.03 + 0.12 * cool) * np.random.randn(n)
        elif r < 0.45:
            f = np.random.randint(1, 8)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.04 + 0.10 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif r < 0.65:
            L = np.random.randint(4, 25)
            s = np.random.randint(0, max(1, n - L))
            pert = np.zeros(n)
            pert[s:s + L] = (0.08 + 0.15 * cool) * np.random.randn(min(L, n - s))
        elif r < 0.8:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = (0.1 + 0.25 * cool) * np.random.randn(int(mask.sum()))
        else:
            L = np.random.randint(5, 30)
            s = np.random.randint(0, max(1, n - L))
            pert = np.zeros(n)
            pert[s:s + L] = 1.0  # flip segment
        h_p = project_h(work_h + pert)
        T = 300 + 8000 * cool
        hc = slsqp_opt(h_p, T, n, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.00008
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h, best_c5 = coord_sweep(best_h, n, t_end, max_sweeps=100)
    return best_h, best_c5

def multi_start_opt(h, n, t_end, n_starts=4):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for s in range(n_starts):
        if time.time() > t_end:
            break
