    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * ((target - s) / tot)
    else:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def softmax_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def coordinate_optimize(h, n, idx):
    """Optimize a single coordinate using 1D minimax."""
    h0 = h.copy()
    h0[idx] = 0.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    dx = 2.0 / n
    shifts = np.arange(2 * n - 1) - (n - 1)
    v1 = (idx + shifts >= 0) & (idx + shifts < n)
    v2 = (idx - shifts >= 0) & (idx - shifts < n)
    A = corr0 + h0[np.clip(idx - shifts, 0, n - 1)] * v2
    B = (1.0 - h0[np.clip(idx + shifts, 0, n - 1)]) * v1 - h0[np.clip(idx - shifts, 0, n - 1)] * v2
    
    def f(x):
        return np.max(A + x * B) * dx
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13, 'maxiter': 300})
    return float(res.x)

    best_c5 = compute_c5(best)
    for sweep in range(max_sweeps):

        changed = False

                changed = True

        c5 = compute_c5(best)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
        elif not changed:
            break
    return best, best_c5

def minimax_phase(h, n, t_end, T_start=100, T_end=50000, n_iter=60):
    """Iterate: compute soft-max weights, optimize h, repeat."""
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        
        # Compute soft-max weights
        dx = 2.0 / n
        corr = np.correlate(best, 1.0 - best, mode='full')
        m = np.max(corr)
        y = np.exp(T * (corr - m))
        y /= y.sum()
        
        # Optimize h to minimize sum_k y_k * corr_k(h)
        # Using L-BFGS-B with projection
        def obj(hm):
            c = np.correlate(hm, 1.0 - hm, mode='full')
            return np.dot(y, c) * dx
        
        # Projected gradient steps
        h_work = best.copy()
        lr = 0.5 / (T * dx)  # learning rate
        for pg in range(50):
            c = np.correlate(h_work, 1.0 - h_work, mode='full')
            # Gradient of sum_k y_k * corr_k
            # For lag l, shift = l-(n-1):
            # d/dh[j] corr_l = (1-h[j+shift])*1{valid1} - h[j-shift]*1{valid2}
            g = np.zeros(n)
            for l in range(2 * n - 1):
                if y[l] < 1e-18:
                    continue
                s = l - (n - 1)
                if s >= 0:
                    g[:n-s] += y[l] * (1.0 - h_work[s:s+n-s])
                    g[s:] += y[l] * (-h_work[:n-s])
                else:
                    s2 = -s
                    g[s2:] += y[l] * (1.0 - h_work[:n-s2])
                    g[:n-s2] += y[l] * (-h_work[s2:])
            g *= dx
            h_work = h_work - lr * g
            h_work = project_h(h_work, n)
        
        c5 = compute_c5(h_work)
        if c5 < best_c5:
            best_c5 = c5
            best = h_work
