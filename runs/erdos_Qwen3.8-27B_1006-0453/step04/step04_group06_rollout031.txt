import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return np.clip(h * target / max(s, 1e-15), 0, 1)
        return np.clip(h + room * (diff / tot), 0, 1)
    else:
        return np.clip(h * (target / max(s, 1e-15)), 0, 1)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = full_corr(h)
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def coord_descent(h, n, t_end, max_sweeps=1000):
    """Fast coordinate descent with incremental correlation updates."""
    best_h = h.copy()
    corr = full_corr(best_h)
    center = n - 1
    dx = 2.0 / n
    best_c5 = corr.max() * dx
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Build delta array: a[k] = change in corr[center+k] per unit change in h[i]
            a = np.zeros(2 * n - 1)
            hi_old = best_h[i]
            for k in range(1, n):
                if i >= k:
                    a[center + k] += (1.0 - best_h[i - k])
                    a[center - k] += best_h[i - k]
                if i + k < n:
                    a[center + k] -= best_h[i + k]
                    a[center - k] += (1.0 - best_h[i + k])
            
            # f(delta) = max(corr + delta*a, with center term being quadratic) * dx
            # center: corr[center] + delta*(1-2*hi_old) - delta^2
            lo = -hi_old
            hi = 1.0 - hi_old
            c0 = corr[center]
            c1 = 1.0 - 2.0 * hi_old
            
            def f1d(d):
                vals = corr + d * a
                vals[center] = c0 + d * c1 - d * d
                return vals.max() * dx
            
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10})
            new_val = hi_old + res.x
            new_val = min(max(new_val, 0.0), 1.0)
            d = new_val - hi_old
            if abs(d) < 1e-12:
                continue
            
            # Update corr incrementally
            corr += d * a
            corr[center] += d * c1 - d * d
            best_h[i] = new_val
            
            # Check sum constraint
            s = best_h.sum()
            if abs(s - n / 2.0) > 1e-10:
                # Project
                best_h = project(best_h, n)
                corr = full_corr(best_h)
            
            new_c5 = corr.max() * dx
            if new_c5 < best_c5 - 1e-13:
                best_c5 = new_c5
                improved = True
    
    return best_h, best_c5

def pair_swap_adopt(h, n, t_end, n_iter=100000, T0=0.05):
    """Annealed pair swaps (maintain sum constraint naturally)."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = full_corr(best_h)
    dx = 2.0 / n
    center = n - 1
    T = T0
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        if it % 2000 == 0:
            T *= 0.95
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Swap: h[i] ↔ h[j]
        hi, hj = best_h[i], best_h[j]
        if abs(hi - hj) < 1e-12:
            continue
        # Compute new corr after swap
        # Change: h[i] goes from hi to hj, h[j] goes from hj to hi
        # delta_i = hj - hi, delta_j = hi - hj = -delta_i
        di = hj - hi
        dj = hi - hj
        # Apply delta_i at position i
        new_corr = corr.copy()
        for k in range(1, n):
            if i >= k:
                new_corr[center + k] += di * (1.0 - best_h[i - k])
                new_corr[center - k] += di * best_h[i - k]
            if i + k < n:
                new_corr[center + k] -= di * best_h[i + k]
                new_corr[center - k] += di * (1.0 - best_h[i + k])
        new_corr[center] += di * (1.0 - 2.0 * hi) - di * di
        # Apply delta_j at position j (using ORIGINAL values, not updated)
        for k in range(1, n):
            if j >= k:
                new_corr[center + k] += dj * (1.0 - best_h[j - k])
                new_corr[center - k] += dj * best_h[j - k]
            if j + k < n:
                new_corr[center + k] -= dj * best_h[j + k] 
                new_corr[center - k] += dj * (1.0 - best_h[j + k])
        new_corr[center] += dj * (1.0 - 2.0 * hj) - dj * dj
        
        new_c5 = new_corr.max() * dx
        accept = new_c5 <= best_c5 or (T > 1e-10 and np.random.rand() < np.exp(-(new_c5 - best_c5) / (T * dx)))
        if accept:
            best_h[i], best_h[j] = hj, hi
            corr = new_corr
            if new_c5 < best_c5:
                best_c5 = new_c5
    
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=1000):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)
