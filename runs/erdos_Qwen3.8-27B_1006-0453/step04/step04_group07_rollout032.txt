def compute_c5(h):

    dx = 2.0 / n

    return np.max(corr) * dx, corr, dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.clip(h, 0, 1)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = corr.max() * dx
    k_vals = np.arange(-(n-1), n)
    rng = np.random
    prev_c5 = best_c5
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        max_delta = 0.0
        for i in order:
            old = best_h[i]
            v1 = (i + k_vals)
            valid1 = (v1 >= 0) & (v1 < n)
            v2 = (i - k_vals)
            valid2 = (v2 >= 0) & (v2 < n)
            coeff = np.zeros(2*n - 1)
            coeff[valid1] = 1.0 - best_h[v1[valid1]]
            coeff[valid2] -= best_h[v2[valid2]]

            delta = best_v - old

                corr += delta * coeff
                best_c5 = best_f * dx

        # Project and recompute at end of sweep
        best_h = project(best_h, n)

        best_c5 = corr.max() * dx

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def min_max_lines(a, b):
    f0 = b.max()
    f1 = (a + b).max()
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = vals.max()
        tol = 1e-12 * max(1.0, abs(mx))
        active = vals >= mx - tol
        if a[active].min() > 0:
            lo = mid
        elif a[active].max() < 0:
            hi = mid
        else:
            break
    v = 0.5 * (lo + hi)
    fv = (a * v + b).max()
    if f0 <= fv:
        return 0.0, f0
    if f1 <= fv:
        return 1.0, f1
    return v, fv

def coord_descent(h, n, t_end, max_sweeps=10000):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = corr.max() * dx
    k_vals = np.arange(-(n - 1), n)
    rng = np.random
    prev_c5 = best_c5
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        max_delta = 0.0
        for i in order:
            old = best_h[i]
            v1 = i + k_vals
            valid1 = (v1 >= 0) & (v1 < n)
            v2 = i - k_vals
            valid2 = (v2 >= 0) & (v2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[valid1] = 1.0 - best_h[v1[valid1]]
            coeff[valid2] -= best_h[v2[valid2]]
            b_vec = corr - old * coeff
            best_v, best_f = min_max_lines(coeff, b_vec)
            delta = best_v - old
            if abs(delta) > 1e-10 and best_f < corr.max() - 1e-12:
                best_h[i] = best_v
                corr += delta * coeff
                max_delta = max(max_delta, abs(delta))
        best_h = project(best_h, n)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        best_c5 = corr.max() * dx
        if abs(prev_c5 - best_c5) < 1e-11 and max_delta < 1e-8:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
        prev_c5 = best_c5
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def structured_perturb(h, n, rng):
    mode = rng.randint(4)
    h_p = h.copy()
    if mode == 0:
        step = 0.05 + 0.2 * rng.rand()
        h_p += step * rng.randn(n)
    elif mode == 1:
        blk = rng.randint(3, n // 4)
        s = rng.randint(0, n - blk)
        h_p[s:s+blk] = 1.0 - h_p[s:s+blk]
    elif mode == 2:
        blk = rng.randint(5, max(6, n // 8))
        s = rng.randint(0, n - blk)
        h_p[s:s+blk] = rng.rand(blk)
    else:
        k1 = rng.randint(0, n)
        k2 = rng.randint(k1 + 1, n)
        swap = h_p[k1:k2].copy()
        h_p[k1:k2] = h_p[k2-1:k2-1-n+k2-k1] if False else swap[::-1]
    return project(h_p, n)

def basin_hop_cd(h, c5, n, t_end, max_iters=200):
    best_h, best_c5 = h.copy(), c5
    work_h, work_c5 = h.copy(), c5
    rng = np.random
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.01, 1.0 - 0.9 * it / max_iters)
        h_p = structured_perturb(work_h, n, rng)
        t_sub = min(time.time() + 3.0, t_end)
        hc, c5v = coord_descent(h_p, n, t_sub, max_sweeps=200)
        temp = 0.002 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)
