    global _g_corr, _g_coeff, _g_c0_idx, _g_c0_base, _g_old
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    c0_idx = n - 1
    _g_c0_idx = c0_idx
    ks = np.arange(-(n - 1), n)
    rng_local = np.random.RandomState(42)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if order == 'random':
            order_idx = rng_local.permutation(n)
        else:
            order_idx = range(n)
        
        for i in order_idx:
            old = best_h[i]
            if old < 1e-15 or old > 1.0 - 1e-15:
                # Can't improve much at boundaries, but still try
                pass
            idx1 = i - ks
            idx2 = i + ks
            v1m = (idx1 >= 0) & (idx1 < n)
            v2m = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1m] += 1.0 - best_h[idx1[v1m]]
            coeff[v2m] -= best_h[idx2[v2m]]
            coeff[c0_idx] = 0.0
            c0_base = corr[c0_idx]
            
            _g_corr = corr
            _g_coeff = coeff
            _g_c0_base = c0_base
            _g_old = old
            
            v0 = _coord_obj(0.0)
            v1r = _coord_obj(1.0)
            res = minimize_scalar(_coord_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            best_v = res.x
            best_f = res.fun
            if v0 < best_f:
                best_v, best_f = 0.0, v0
            if v1r < best_f:
                best_v, best_f = 1.0, v1r

                best_c5 = best_f * dx
                corr += (best_v - old) * coeff
                corr[c0_idx] = c0_base + (best_v - old) * (1.0 - best_v - old)
                improved = True
        if not improved:
            break
    
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_random_binary(160, seed)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from all available starts
    starts = [best_h]
    for s in range(3):
        starts.append(make_random_binary(n, seed + 1000 + s * 7))
    for st in starts:
        t1_end = t0 + budget_s * 0.06
        h_c, c5_c = coord_descent(st, n, t1_end, max_sweeps=200, rand_order=False)
        if c5_c < best_c5:
            best_h, best_c5 = h_c, c5_c

    # Phase 2: Strong basin hopping
    t2 = t0 + budget_s * 0.42
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t2, max_iters=400, seed=seed)

    # Phase 3: Polish
    t3 = t0 + budget_s * 0.48
    best_h, best_c5 = coord_descent(best_h, n, t3, max_sweeps=300, rand_order=True, seed=999)

    # Phase 4: Try alternate resolutions
    for n_try in [200, 240, 320, 128]:
        if time.time() > t0 + budget_s * 0.78:
            break
        h_tr = interp_h(best_h, n_try)
        t_cd = min(t0 + budget_s * 0.80, time.time() + budget_s * 0.03)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t_cd, max_sweeps=200, rand_order=True, seed=555)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try
            t_bh = min(t0 + budget_s * 0.82, time.time() + budget_s * 0.02)
            best_h, best_c5 = basin_hop(best_h, best_c5, n, t_bh, max_iters=40, seed=seed + 77)
            t_po = min(t0 + budget_s * 0.83, time.time() + budget_s * 0.015)
            best_h, best_c5 = coord_descent(best_h, n, t_po, max_sweeps=200, rand_order=True, seed=777)

    # Phase 5: Longer basin hopping at best resolution
    t5 = t0 + budget_s * 0.93
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t5, max_iters=250, seed=seed + 31)

    # Phase 6: Extensive final coordinate descent
    t6 = t0 + budget_s * 0.995
    best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=1000, rand_order=True, seed=123)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
