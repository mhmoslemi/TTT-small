```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def binarize(h):
    """Round to binary with correct sum."""
    n = len(h)
    target = n // 2
    # Sort by value, pick top target as 1s
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def project_continuous(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -np.max(vc) - 1.0
    hi = 1.0 - np.min(vc) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def swap_delta(corr_array, h, i, j, n):

    # k ranges from -(n-1) to (n-1), mapped to index k+(n-1) in corr_array

        ik2 = i - k
        jk = j + k
        jk2 = j - k
        d = 0.0

            d += h[ik2]
        if 0 <= jk < n:
            d -= h[jk]
        if 0 <= jk2 < n:
```

```python
def compute_delta(h, i, j, tau, n):
    """Compute change in c[tau] when swapping h[i] (1→0) and h[j] (0→1)."""
    if tau == 0:
        return 0.0
    ps = set()
    if 0 <= i < n and 0 <= i+tau < n:
        ps.add(i)
    ip = i - tau
    if 0 <= ip < n and 0 <= ip+tau < n:
        ps.add(ip)
    if 0 <= j < n and 0 <= j+tau < n:
        ps.add(j)
    jp = j - tau
    if 0 <= jp < n and 0 <= jp+tau < n:
        ps.add(jp)
    
    delta = 0.0
    for p in ps:
        old_val = h[p] * (1.0 - h[p+tau])
        # new values: h_new[i] = 1-h[i], h_new[j] = 1-h[j]
        hp = (1.0 - h[p]) if p == i or p == j else h[p]
        hpt = (1.0 - h[p+tau]) if p+tau == i or p+tau == j else h[p+tau]
        new_val = hp * (1.0 - hpt)
        delta += new_val - old_val
    return delta
```

```python
    """Vectorized computation of delta in correlation after swap i(1→0), j(0→1)."""
    d = np.zeros(2*n-1)

    idx2 = (i - j) + (n - 1)

    # τ=0 always gives 0 delta for binary
    d[n-1] = 0.0
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def project_continuous(h, n):
    target = n / 2.0
    vec = np.clip(h, 0.0, 1.0)
    if abs(vec.sum() - target) < 1e-10:
        return vec
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0, 1)

def swap_delta_vec(h, i, j, n, ks):
    d = np.zeros(2 * n - 1)
    v = (i + ks >= 0) & (i + ks < n)
    d[v] -= (1.0 - h[i + ks[v]])
    v = (i - ks >= 0) & (i - ks < n)
    d[v] += h[i - ks[v]]
    v = (j + ks >= 0) & (j + ks < n)
    d[v] += (1.0 - h[j + ks[v]])
    v = (j - ks >= 0) & (j - ks < n)
    d[v] -= h[j - ks[v]]
    idx1 = (j - i) + (n - 1)
    if 0 <= idx1 < 2 * n - 1:
        d[idx1] += 1.0
    idx2 = (i - j) + (n - 1)
    if 0 <= idx2 < 2 * n - 1:
        d[idx2] += 1.0
    d[n - 1] = 0.0
    return d

def sa_phase(h, corr, c5, n, ks, t_end, T_start, T_end, t0):

    best_c5 = c5
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    n1 = n // 2
    stuck = 0
    dx = 2.0 / n
    dur = max(t_end - t0, 1.0)
    while time.time() < t_end:
        frac = min((time.time() - t0) / dur, 1.0)
        T = T_start * np.exp(-4.5 * frac) + T_end

        oj = np.random.randint(n1)
        ii = ones[oi]
        jj = zeros[oj]
        d = swap_delta_vec(h, ii, jj, n, ks)
        new_corr = corr + d
        new_max = float(np.max(new_corr))
        new_c5 = new_max * dx
        if new_c5 < c5 or np.random.rand() < np.exp(-(new_c5 - c5) / max(T, 1e-12)):
            h[ii] = 0.0
            h[jj] = 1.0
            ones[oi] = jj
            zeros[oj] = ii
            corr = new_corr
            c5 = new_c5
            stuck = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

                a2, b2 = ones[oi2], zeros[oj2]

def continuous_polish(h, n, t_end):
    from scipy.optimize import minimize
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for Ts in [2000, 8000, 30000]:
        if time.time() > t_end:
            break
        def obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(cc)
            return (m + np.log(np.sum(np.exp(Ts * (cc - m)))) / Ts) * dx
        def eq(x):
            return x.sum() - n / 2.0
        try:
            res = minimize(obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': eq}],
                          options={'maxiter': 400, 'ftol': 1e-16})
            hc = project_continuous(res.x, n)
            cv = compute_c5(hc)

            pass

    if time.time() < t_end:

    h, corr, c5, bh, bc = sa_phase(h, corr, c5, n, ks, t0 + budget_s * 0.50, 0.08, 0.0003, t0)
    if bc < best_c5:
        best_h, best_c5 = bh, bc
    # Phase 2: SA continued lower temp (20%)
    h, corr, c5, bh, bc = sa_phase(h, corr, c5, n, ks, t0 + budget_s * 0.70, 0.02, 0.0001, t0)
    if bc < best_c5:
        best_h, best_c5 = bh, bc
    # Phase 3: Try different n (10%) if time allows
    if time.time() < t0 + budget_s * 0.80:
        n2 = 200
        h2 = np.zeros(n2)
        perm = np.random.permutation(n2)
        h2[perm[:n2 // 2]] = 1.0
        corr2 = compute_corr_full(h2).copy()
        c5_2 = float(np.max(corr2) * 2.0 / n2)
        ks2 = np.arange(-(n2 - 1), n2)
        _, _, _, bh2, bc2 = sa_phase(h2, corr2, c5_2, n2, ks2, t0 + budget_s * 0.80, 0.10, 0.0005, t0)
        if bc2 < best_c5:
            best_h, best_c5, n = bh2, bc2, n2
    # Phase 4: Continuous polish (12%)
    best_h, best_c5 = continuous_polish(best_h, n, t0 + budget_s * 0.97)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
