import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T, k_top):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = np.sort(corr)[-k_top:]
    m = topk[-1]
    val = m + np.log(np.sum(np.exp(T * (topk - m)))) / T
    return val * dx

def smooth_grad(h, T, k_top, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k indices
    topk_idx = np.argsort(corr)[-k_top:]
    topk_vals = corr[topk_idx]
    m = topk_vals[-1]
    exps = np.exp(T * (topk_vals - m))
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    for ii in range(len(topk_idx)):
        k = topk_idx[ii]
        wi = w[ii]
        if wi < 1e-12:
            continue
        lag = k - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if lag < n:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            la = -lag
            if la < n:
                g[la:] += (1.0 - h[la:])
                g[:n - la] -= h[:n - la]
        grad += wi * g
    return dx * grad

def grad_descent(h, n, t_end, T_init=100, T_final=5000, k_top=5, lr=0.01):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_iter = 0
    while time.time() < t_end:
        T = T_init + (T_final - T_init) * min(1.0, n_iter / 200.0)
        g = smooth_grad(best_h, T, k_top, n)
        # Backtracking line search
        alpha = lr
        for _ in range(20):
            h_try = project_h(best_h - alpha * g, n / 2.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                break
            alpha *= 0.5
        n_iter += 1
    return best_h, best_c5

def coord_opt_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n - 1)])
    dx = 2.0 / n
    def f1d(hi):
        return np.max(corr_base + hi * coefs) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200, order='random'):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if order == 'random':
            idx = np.random.permutation(n)
        else:
            idx = np.arange(n)
        for i in idx:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            # Re-project to maintain sum constraint
            h_try = project_h(h_try, n / 2.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_local_search(h, n, t_end, max_iter=50000):
    # Round to nearest and do pair swaps
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_iter = 0
    while time.time() < t_end and n_iter < max_iter:
        # Find positions to swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if best_h[i] < best_h[j]:
            i, j = j, i
        # Swap: put smaller value at i, larger at j (or just swap)
        h_try = best_h.copy()
        h_try[i] = best_h[j]
        h_try[j] = best_h[i]
        h_try = project_h(h_try, n / 2.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try
        n_iter += 1
    return best_h, best_c5

from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, T, k_top, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-k_top:]
    vals = corr[idx]
    m = vals[-1]
    exps = np.exp(T * (vals - m))
    w = exps / exps.sum()

        else:
            a = -lag
            length = n - a
            grad[:length] += wi * (1.0 - h[a:])
            if length > 0:
                grad[a:] -= wi * h[:length]
    return obj, dx * grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, T, k_top, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-k_top:]
    vals = corr[idx]
    m = vals[-1]
    exps = np.exp(T * (vals - m))
    w = exps / exps.sum()
    obj = (m + np.log(np.sum(exps)) / T) * dx

        wi = w[ii]
        if wi < 1e-12:

        lag = k - (n - 1)

            if length > 0:

                grad[:length] -= wi * h[lag:]
        else:
            a = -lag
            length = n - a
            if length > 0:
                grad[:length] += wi * (1.0 - h[a:])
                grad[a:] -= wi * h[:length]
    return obj, dx * grad
