def slsqp_opt(h0, T, n, maxiter=600):

    corr = np.correlate(h0, 1.0 - h0, mode='full')
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return float(m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})

        return h0

def basin_hop(work_h, work_c5, n, t_end, max_iters=400):

def binary_round(h, n):
    order = np.argsort(-h)
    hb = np.zeros(n)
