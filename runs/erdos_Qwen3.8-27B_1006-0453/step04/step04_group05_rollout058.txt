def compute_soft_grad(h, N, beta):
    """Compute gradient of soft-max of correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft-max weights
    corr_max = corr.max()
    weights = np.exp(beta * (corr - corr_max))
    weights /= weights.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for k in range(2 * N - 1):

            m = -lag
            grad[:N-m] += w * (1.0 - h[m:])

def soft_max_gradient(h, N, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*corr))) and the value."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    corr_max = corr.max()
    exp_vals = np.exp(beta * (corr - corr_max))
    w = exp_vals / exp_vals.sum()
    
    # Soft-max value
    f_val = (corr_max + np.log(exp_vals.sum()) / beta)
    
    # Gradient: sum_k w[k] * dcorr_k/dh
    # dcorr_k/dh[i] = (1-h[i-lag])*valid(i-lag) - h[i+lag]*valid(i+lag)
    # where lag = k - (N-1)
    
    grad = np.zeros(N)
    center = N - 1
    
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - center
        if lag >= 0:
            # (1-h[i-lag]) for i in [lag, N-1]
            grad[lag:] += wk * (1.0 - h[:N-lag])
            # -h[i+lag] for i in [0, N-1-lag]
            grad[:N-lag] -= wk * h[:N-lag]
        else:
            m = -lag
            # (1-h[i+m]) for i in [0, N-1-m]
            grad[:N-m] += wk * (1.0 - h[m:])
            # -h[i-m] for i in [m, N-1]
            grad[m:] -= wk * h[m:]
    
    return grad, f_val, corr
