def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-15:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

C_k = Σ_j h[j] * (1 - h[j+k])

When we change h[i] by s (i.e., h[i] → h[i] + s):

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-15:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def min_max_1d(corr, a, lo, hi):
    if hi - lo < 1e-15:
        return 0.0
    s_lo, s_hi = lo, hi

        elif np.min(active) > 1e-12:
            s_hi = s_mid
        else:
            return s_mid
    return (s_lo + s_hi) * 0.5

                    improved = True

    return best_h, best_c5

    dx = 2.0 / n

        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        ec = np.exp(np.clip(T * (c - m), -50, 50))
        return float((m + np.log(ec.sum()) / T)) * dx

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)
