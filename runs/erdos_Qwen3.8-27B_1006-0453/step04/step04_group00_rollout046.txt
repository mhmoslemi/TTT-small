import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n - 1)])
    dx = 2.0 / n
    def f1d(hi):
        return np.max(corr_base + hi * coefs) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def pairwise_refine(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    frac = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
    if len(frac) < 2:
        return best_h, best_c5
    for _ in range(5):
        if time.time() > t_end:
            break
        improved = False
        for a in range(len(frac)):
            for b in range(a + 1, len(frac)):
                if time.time() > t_end:
                    return best_h, best_c5
                i, j = frac[a], frac[b]
                if abs(best_h[i] - best_h[j]) < 1e-9:
                    continue
                h_base = best_h.copy()
                h_base[i] = 0.0
                h_base[j] = 0.0
                corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
                ki = np.arange(-(n-1), n)
                j1i = i + ki; j2i = i - ki
                v1i = (j1i>=0)&(j1i<n); v2i = (j2i>=0)&(j2i<n)
                csi = np.zeros(2*n-1)
                csi[v1i] += (1.0-best_h[np.clip(j1i[v1i],0,n-1)])
                csi[v2i] += -best_h[np.clip(j2i[v2i],0,n-1)]
                j1j = j + ki; j2j = j - ki
                v1j = (j1j>=0)&(j1j<n); v2j = (j2j>=0)&(j2j<n)
                csj = np.zeros(2*n-1)
                csj[v1j] += (1.0-best_h[np.clip(j1j[v1j],0,n-1)])
                csj[v2j] += -best_h[np.clip(j2j[v2j],0,n-1)])
                dx = 2.0/n
                def f2d(z):
                    return np.max(corr_base + z[0]*csi + z[1]*csj)*dx
                res = minimize(f2d, [best_h[i], best_h[j]], method='Nelder-Mead',
                              bounds=[(0,1),(0,1)],
                              options={'maxiter':200,'xatol':1e-12,'fatol':1e-14})
                if res.fun < best_c5 - 1e-14:
                    ht = best_h.copy()
                    ht[i] = res.x[0]; ht[j] = res.x[1]
                    ht = project_h(ht)
                    c5t = compute_c5(ht)
                    if c5t < best_c5:
                        best_c5 = c5t; best_h = ht; improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP annealing (20%)
    t1 = t0 + budget_s * 0.20
    for T in [5, 25, 100, 500, 2000, 10000, 50000, 150000]:
        if time.time() > t1:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=800, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Basin hopping with diverse perturbations (30%)
    t2 = t0 + budget_s * 0.50
    work_h = best_h.copy()
    work_c5 = best_c5
    nit = 0
    while time.time() < t2:
        nit += 1
        cool = max(0.0, 1.0 - nit / 150.0)
        mod = nit % 6
        if mod == 0:
            pert = (0.03 + 0.20*cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 14)
            phase = np.random.uniform(0, 2*np.pi)
            pert = (0.04 + 0.14*cool) * np.sin(2*np.pi*freq*np.arange(n)/n + phase)
        elif mod == 2:
            mask = np.random.random(n) < 0.25
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.25*cool) * np.random.randn(max(1, mask.sum()))
        elif mod == 3:
            half = n//2
            p = (0.03 + 0.10*cool) * np.random.randn(half)
            pert = np.concatenate([p, -p])
        elif mod == 4:
            bsz = np.random.randint(4, 16)
            pert = np.zeros(n)
            for b in range(0, n, bsz):
                sz = min(bsz, n-b)
                pert[b:b+sz] = (0.02 + 0.08*cool)*np.random.randn(sz)
        else:
            pert = 0.008 * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T_bh = 2000 + 30000*cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0008*cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v-work_c5)/temp):
            work_h = hc; work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v; best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5; best_h = work_h

    # Phase 3: Coordinate descent (35%)
    t3 = t0 + budget_s * 0.85
    best_h, best_c5 = coord_descent(best_h, n, t3, max_sweeps=600)

    # Phase 4: Pairwise refinement
    t4 = t0 + budget_s * 0.93
    best_h, best_c5 = pairwise_refine(best_h, n, t4)

    # Phase 5: Final polish
    for T in [100000, 500000]:
        if time.time() > t0 + budget_s * 0.96:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v; best_h = hc

    if time.time() < t0 + budget_s * 0.985:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s*0.99, max_sweeps=80)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
