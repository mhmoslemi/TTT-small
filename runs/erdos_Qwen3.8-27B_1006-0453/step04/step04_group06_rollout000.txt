import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return np.clip(h * (target / s), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    center = n - 1
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-12:
            continue
        k = i - center
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
        else:
            m2 = -k
            grad[m2:] += wi * (1.0 - h[:n - m2])
            grad[:n - m2] -= wi * h[m2:]
    return grad * dx

def _coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip, im = i + k, i - k
        a[center + k] = (1.0 - h[ip] if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
        a[center - k] = (1.0 - h[im] if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
    dx = 2.0 / n
    ct = corr_base[center]
    def f1d(x):
        vals = corr_base + x * a
        vals[center] = ct + x - x * x
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = _coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _lbfgs_wrap(smooth_c5, smooth_c5_grad, h, T, maxiter):
    return minimize(smooth_c5, h, method='L-BFGS-B',
                    jac=smooth_c5_grad, args=(T,),
                    options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxfun': maxiter * 3})

def lbfgs_opt(h, T, n, maxiter=3000):
    res = _lbfgs_wrap(smooth_c5, smooth_c5_grad, h, T, maxiter)
    return project(res.x, n)

        # Quick coord polish after L-BFGS
