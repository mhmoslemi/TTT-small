                    best_corr = compute_corr(best_h)
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, n, T, maxiter=500):
    def obj(x):
        return smooth_max_val(compute_corr(x), T) * (2.0 / n)
    def con_fun(x):
        return x.sum() - n * 0.5
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return proj(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def focused_slsqp(h, n, maxiter=1000):
    corr = compute_corr(h)
    mx = np.max(corr)
    active = np.where(corr >= mx * 0.9999)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    dx = 2.0 / n
    def obj(x):
        c = compute_corr(x)[active]
        return float(np.max(c)) * dx
    def con_fun(x):
        return x.sum() - n * 0.5
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return proj(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov gradient (short)
    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = nesterov_opt(best_h, n, t1, T_init=200, lr_init=0.002)

    # Phase 2: Aggressive coordinate descent
    t2 = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent_fast(best_h, n, t2, max_sweeps=500)

    # Phase 3: SLSQP polish
    for T in [2000, 20000, 100000]:
        if elapsed() > budget_s * 0.42:
            break
        hc = slsqp_polish(best_h, n, T, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Basin hopping with short local optimize
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(80):
        if elapsed() > budget_s * 0.60:
            break
        cool = 1.0 - 0.85 * (it / 80.0)
        step = 0.01 + 0.08 * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = proj(h_pert, n)
        # Quick local: few coord sweeps + gradient
        hc, c5v = coord_descent_fast(h_pert, n, min(t0 + budget_s * 0.60, time.time() + 8), max_sweeps=80)
        hc = slsqp_polish(hc, n, 5000, maxiter=200)
        c5v = compute_c5(hc)
        temp = 0.002 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 5: More coordinate descent
    t5 = t0 + budget_s * 0.78
    best_h, best_c5 = coord_descent_fast(best_h, n, t5, max_sweeps=600)

    # Phase 6: Nesterov with higher T
    t6 = t0 + budget_s * 0.85
    best_h, best_c5 = nesterov_opt(best_h, n, t6, T_init=5000, lr_init=0.0005)

    # Phase 7: Final polish
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.92, max_sweeps=400)
    for T in [50000, 200000]:
        if elapsed() > budget_s * 0.96:
            break
        hc = slsqp_polish(best_h, n, T, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if elapsed() < budget_s * 0.98:
        hc = focused_slsqp(best_h, n, maxiter=1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
