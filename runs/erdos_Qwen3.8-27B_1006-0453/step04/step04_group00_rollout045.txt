        alpha = min(0.4, 0.4 / (np.linalg.norm(momentum) + 0.05))
        alpha *= (0.5 + 0.5 * cool)
        h_new = project_h(h - alpha * momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            h = h_new
        else:
            momentum *= 0.5
            h_new = project_h(h - alpha * 0.3 * grad, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                c5 = c5_new
                h = h_new
            else:
                beta *= 0.93
    return h, c5

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_step(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved += 1
        if improved == 0:
            break
    return best_h, best_c5

def basin_hop(h, n, c5, t_end, n_iters=80):
    work_h = h.copy()
    work_c5 = c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - it / n_iters
        mod = it % 6
        if mod == 0:
            pert = (0.02 + 0.08 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 8)
            phase = np.random.uniform(0, 2 * np.pi)
            xa = np.arange(n)
            pert = (0.015 + 0.06 * cool) * np.sin(2 * np.pi * freq * xa / n + phase)
        elif mod == 2:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.12 * cool) * np.random.randn(mask.sum())
        elif mod == 3:
            bs = np.random.randint(4, 16)
            st = np.random.randint(0, max(1, n - bs))
            pert = np.zeros(n)
            pert[st:st + bs] = (0.02 + 0.05 * cool) * np.random.randn(bs)
        elif mod == 4:
            # Anti-symmetric perturbation
            half = n // 2
            p = (0.02 + 0.06 * cool) * np.random.randn(half)
            pert = np.zeros(n)
            pert[:half] = p
            pert[half:half + len(p)] = -p
        else:
            pert = (0.01 + 0.03 * cool) * np.random.randn(n)
        
        h_pert = project_h(work_h + pert, n)
        # Quick gradient polish from perturbed point
        h_pol, c5_pol = grad_descent(h_pert, n, min(time.time() + 5, t_end), n_iter=30, topk=6)
        if c5_pol > work_c5:
            h_pol, c5_pol = h_pert, compute_c5(h_pert)
        temp = 0.0008 * cool + 0.00005
        if c5_pol < work_c5 or np.random.rand() < np.exp(-(c5_pol - work_c5) / temp):
            work_h = h_pol
            work_c5 = c5_pol
    return work_h, work_c5

def make_start(n, style):
    h = np.zeros(n)
    if style == 0:
        h[:n // 2] = 1.0
    elif style == 1:
        h[::2] = 1.0
    elif style == 2:
        h[np.arange(n) % 3 == 0] = 1.5
    elif style == 3:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * 2 * np.arange(n) / n)
    elif style == 4:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * np.arange(n) / n) + 0.2 * np.sin(2 * np.pi * 5 * np.arange(n) / n)
    else:
        h = np.random.random(n) * 0.6 + 0.2
    return project_h(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    def remaining():
        return budget_s - elapsed()
    
    # Initialize from parent or default
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Gradient descent from parent (budget: 25%)
    end1 = t0 + budget_s * 0.25
    best_h, best_c5 = grad_descent(best_h, n, end1, n_iter=500, topk=12)
    
    # Phase 2: Coordinate descent polish (budget: 20%)
    end2 = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, n, end2, max_sweeps=800)
    
    # Phase 3: Basin hopping on parent n (budget: 25%)
    end3 = t0 + budget_s * 0.70
    bh_h, bh_c5 = basin_hop(best_h, n, best_c5, end3, n_iters=100)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = bh_h.copy()
    # Quick coord descent after basin hopping
    best_h, best_c5 = coord_descent(best_h, n, end3, max_sweeps=300)
    
    # Phase 4: Try alternative sizes and starts (budget: 20%)
    end4 = t0 + budget_s * 0.90
    for alt_n in [120, 200, 240]:
        if remaining() < 60:
            break
        if alt_n == n:
            continue
        for style in range(5):
            if remaining() < 30:
                break
            h_try = make_start(alt_n, style)
            # Quick gradient descent
            h_try, c5_try = grad_descent(h_try, alt_n, min(time.time() + 8, end4), n_iter=100, topk=8)
            # Quick coord descent
            h_try, c5_try = coord_descent(h_try, alt_n, min(time.time() + 15, end4), max_sweeps=200)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                n = alt_n
    
    # Phase 5: Final polish (budget: 8%)
    end5 = t0 + budget_s * 0.98
    best_h, best_c5 = grad_descent(best_h, n, end5, n_iter=200, topk=8)
    best_h, best_c5 = coord_descent(best_h, n, end5, max_sweeps=300)
    
    # Phase 6: Ultra-fine coordinate descent
    if remaining() > 10:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=500)
    
    # Final constraint fix
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0
    s = best_h.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        interior = (best_h > 0.005) & (best_h < 0.995)
        if interior.sum() > 0:
            best_h[interior] += diff / interior.sum()
        else:
            best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
