import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / n)

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] += -wk * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wk * (1.0 - h[:n-kk])
            grad[:n-kk] += -wk * h[kk:]
    return grad * dx

def project(h, n):
    target = n * 0.5
    vc = np.clip(h, 0.0, 1.0)
    diff = target - vc.sum()
    if abs(diff) < 1e-10:
        return vc
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def slsqp(h, T, n, maxit=2000):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:

                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-15})
        return project(r.x, n)
    except:
        return h

    best_c = compute_c5(best)
    stall = 0
    for _ in range(sweeps):

        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def con_eq(x):
    return x.sum() - len(x) * 0.5

def con_eq_jac(x):
    return np.ones(len(x))

def slsqp_opt(h, T, n, maxit=1500):
    try:
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
        r = minimize(smooth_c5, h, args=(T,), jac=lambda x, t=T: smooth_c5_grad(x, t),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-14})
        return project(r.x, n)
    except Exception:
        return h

def fista_opt(h0, T, n, maxit=3000, t_end=None):
    z = h0.copy()
    y = h0.copy()
    tk = 1.0
    L = max(T * 8.0 / n, 10.0)
    for k in range(maxit):
        if t_end is not None and time.time() > t_end:
            break
        grad = smooth_c5_grad(z, T)
        y_new = z - (1.0 / L) * grad
        z_new = project(y_new, n)
        tk_new = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = z_new + ((tk - 1.0) / tk_new) * (z_new - z)
        tk = tk_new
        z = z_new
    return z

def coord_opt(h, n, i):
    def f1d(x):
        h[i] = x
        return compute_c5(h)
    r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return r.x

def cd(h, n, t_end, sweeps=1000):
    best = h.copy()
    best_c = compute_c5(best)
    stall = 0
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best, best_c
            hx = best[i]
            best[i] = coord_opt(best, n, i)
            best = project(best, n)
            c = compute_c5(best)
            if c < best_c - 1e-13:
                best_c = c
                imp = True
        stall = 0 if imp else stall + 1
        if stall >= 4:
            break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def el():
        return time.time() - t0

    n = 160
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = project(np.array(initial_h_values, dtype=float), n)
    else:
        best_h = np.full(n, 0.5)
    best_c = compute_c5(best_h)

    # Phase 1: SLSQP T-sweep from parent (30%)
    for T in [20, 100, 500, 2000, 10000, 40000]:
        if el() > budget_s * 0.30:
            break
        hc = slsqp_opt(best_h, T, n, maxit=2000)
        c = compute_c5(hc)
        if c < best_c:
            best_h, best_c = hc, c

    # Phase 2: FISTA refinement (10%)
    for T in [5000, 20000]:
        if el() > budget_s * 0.40:
            break
        hf = fista_opt(best_h, T, n, maxit=5000, t_end=t0 + budget_s * 0.40)
        c = compute_c5(hf)
        if c < best_c:
            best_h, best_c = hf, c

    # Phase 3: Multi-start diversification (25%)
    work = best_h.copy()
    n_div = 100
    for it in range(n_div):
        if el() > budget_s * 0.65:
            break
        cool = 1.0 - it / float(n_div)
        sc = 0.02 + 0.10 * cool
        hp = project(work + sc * np.random.randn(n), n)
        T_s = 200 + 8000 * cool
        hc = slsqp_opt(hp, T_s, n, maxit=500)
        c = compute_c5(hc)
        temp = 0.002 * cool + 0.0003
        if c < best_c - 1e-10:
            best_h, best_c = hc, c
            work = hc
        elif np.random.rand() < np.exp(-(c - best_c) / max(temp, 1e-9)):
            work = hc

    # Phase 4: Final tight SLSQP + FISTA (15%)
    for T in [50000, 200000, 800000]:
        if el() > budget_s * 0.80:
            break
        hc = slsqp_opt(best_h, T, n, maxit=2500)
        c = compute_c5(hc)
        if c < best_c:
            best_h, best_c = hc, c
    if el() < budget_s * 0.82:
        hf = fista_opt(best_h, 100000, n, maxit=3000, t_end=t0 + budget_s * 0.82)
        c = compute_c5(hf)
        if c < best_c:
            best_h, best_c = hf, c

    # Phase 5: Intensive coordinate descent polish (18%)
    best_h, best_c = cd(best_h, n, t0 + budget_s * 0.98, sweeps=3000)

    # Final SLSQP micro-polish
    if el() < budget_s * 0.99:
        hc = slsqp_opt(best_h, 2000000, n, maxit=500)
        c = compute_c5(hc)
        if c < best_c:
            best_h, best_c = hc, c

    best_h = project(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c, n
