    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()

    if abs(diff) < 1e-14:

    b = (h >= 0.5).astype(float)

        return smooth_c5(x, T)

        return x.sum() - n / 2.0
    res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=300):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):
    b = (h >= 0.5).astype(float)
    ones = int(b.sum())
    target = n // 2
    if ones > target:
        idx = np.where(b == 1)[0]
        rng.shuffle(idx)
        b[idx[:ones - target]] = 0.0
    elif ones < target:
        idx = np.where(b == 0)[0]
        rng.shuffle(idx)
        b[idx[:target - ones]] = 1.0
    return b

def make_random_binary(n, rng):
    b = np.zeros(n)
    b[rng.choice(n, n // 2, replace=False)] = 1.0
    return b

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_obj(x, T, n):
    return smooth_c5(x, T)

def slsqp_con(x, n):
    return x.sum() - n / 2.0

def slsqp_refine(h0, n, T, maxiter=500):
    res = minimize(slsqp_obj, h0, args=(T, n), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    return project(np.clip(res.x, 0, 1), n)

def binary_sa(h0, n, t_end, rng, max_iter=60000, T_start=0.03, T_end=0.0003):
    h = to_binary(h0, n, rng)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    for it in range(max_iter):
        if time.time() > t_end:
            break
        frac = it / max(max_iter, 1)
        T = T_start * (T_end / T_start) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def binary_coord_descent(h, n, t_end, max_sweeps=200):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5_raw = np.max(corr)
    ones = np.where(best_h > 0.5)[0].tolist()
    zeros = np.where(best_h < 0.5)[0].tolist()
    for _ in range(max_sweeps):
        if time.time() > t_end:

        improved = False
        best_swap = None
        best_swap_val = best_c5_raw
        for i in ones:

                for k in range(-(n - 1), n):
                    km = i - k
                    kp = i + k
                    jk = j - k

                    d = 0.0
                    if k != 0:

        if len(oi) == 0:
            continue
        hp[oi], hp[oj] = hp[oj].copy(), hp[oi].copy()
        # Rebalance to exactly n/2 ones
        hp = to_binary(hp, n, rng)
        hr, c5r = binary_sa(hp, n, t5, rng, max_iter=5000, T_start=0.015 * cool, T_end=0.0002)
        if c5r < work_c5 or rng.rand() < np.exp(-(c5r - work_c5) / (0.001 * cool + 0.0001)):
            work_h, work_c5 = hr, c5r
            if work_c5 < best_c5:
                best_c5, best_h = work_c5, work_h.copy()

    # Phase 6: Resolution upgrade (10% time)
    t6 = t0 + budget_s * 0.97
    for n_try in [200, 240, 320]:
        if time.time() > t6:
            break
        h_tr = interp_h(best_h, n_try)
        ht, c5t = binary_coord_descent(to_binary(h_tr, n_try, rng), n_try, t0 + budget_s * 0.975, max_sweeps=50)
        if c5t < best_c5:
            best_c5, best_h, n = c5t, ht, n_try
        h_tr2 = interp_h(best_h, n_try)
        ht2, c5t2 = continuous_refine(h_tr2, n_try, t0 + budget_s * 0.975)
        if c5t2 < best_c5:
            best_c5, best_h, n = c5t2, ht2, n_try

    # Final: project and compute
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
