At the optimum, typically multiple k are active. Let's say the active set is S. Then we want to minimize max_{k in S} q_k(h) subject to constraints.

For a single active k, q_k(h) = sum_j h[j]*(1-h[j+k]) is a quadratic form. Minimizing it subject to 0<=h<=1 and sum=h_target is a QP.

For multiple active k, it's more complex.

Let me try a **QP approach** for the active set:

At each iteration:
1. Compute all correlation values
2. Identify active set S (those within epsilon of the max)
3. Solve: min_h max_{k in S} q_k(h) s.t. 0<=h<=1, sum(h)=n/2
   ≈ min_h (1/|S|) * sum_{k in S} q_k(h) + penalty for those exceeding max
   Or: min_h t s.t. q_k(h) <= t for all k in S, 0<=h<=1, sum(h)=n/2

The QP min t s.t. sum_j h[j](1-h[j+k]) <= t for k in S is a quadratic program (the constraints are quadratic in h, and we're minimizing a linear function).

With cvxpy, this would be:

But this only works if the quadratic forms are convex (for a convex problem). Let's check:
q_k(h) = sum_j h[j] - h[j]*h[j+k]

The Hessian of h[j]*h[j+k] w.r.t. h is a matrix with non-zero entries at (j, j+k) and (j+k, j). This matrix is NOT positive definite in general. So the problem is non-convex.

However, we can still use it as a heuristic: solve the QP relaxation and check if the solution reduces the max.

Actually, let me think differently. The quadratic form q_k(h) = sum_j h[j](1-h[j+k]) can be written as:
= sum_j h[j] - sum_j h[j]h[j+k]
= sum_j h[j] - (1/2)(h^T B_k h + h^T B_k h)  [if B_k is symmetric]

Where (B_k)_{ij} = 1 if j = i+k or i = j+k (for valid indices).

The Hessian of -h^T B_k h / 2 is -B_k, which is indefinite. So the problem is indeed non-convex.

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sh = T * (corr - m)
    sh -= sh.max()
    exps = np.exp(sh)
    s = exps.sum()
    val = (m + np.log(s) / T) * dx
    w = exps / s
    center = n - 1

        if wk < 1e-13:

        else:

            grad[mk:] += wk * (1.0 - h[:n-mk])
            grad[:n-mk] -= wk * h[mk:]

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_c5_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sh = T * (corr - m)
    sh -= sh.max()
    exps = np.exp(sh)
    s = exps.sum()
    val = (m + np.log(s) / T) * dx
    w = exps / s
    center = n - 1
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        wk = w[center + k]
        if wk < 1e-13:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
        else:
            mk = -k
            grad[mk:] += wk * (1.0 - h[:n - mk])
            grad[:n - mk] -= wk * h[mk:]
    grad *= dx / T
    return val, grad

def coord_sweep_once(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    center = n - 1
    improved = False
    for i in range(n):
        if time.time() > t_end:
            break
        save = best_h[i]
        best_h[i] = 0.0
        cb = np.correlate(best_h, 1.0 - best_h, mode='full')
        a = np.zeros(2 * n - 1)
        for k in range(1, n):
            ip, im = i + k, i - k
            if ip < n:
                a[center + k] += 1.0 - best_h[ip]
                a[center - k] += -best_h[ip]
            if im >= 0:
                a[center - k] += 1.0 - best_h[im]
                a[center + k] += -best_h[im]
        best_h[i] = save
        res = minimize_scalar(
            lambda x: np.max(np.concatenate([cb + x * a, np.empty(0)])) if x == 0 else
            np.max(np.where(np.arange(2*n-1)==center, cb+x-x*x, cb+x*a)),
            bounds=(0.0, 1.0), method='bounded',
            options={'xatol': 1e-10, 'maxiter': 40})
        x_opt = res.x
        h_try = best_h.copy()
        h_try[i] = x_opt
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
            improved = True
    return best_h, best_c5, improved

def coord_descent(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while time.time() < t_end:
        out_h, out_c5, improved = coord_sweep_once(best_h, n, t_end)
        best_h, best_c5 = out_h, out_c5
        if not improved:
            break
    return best_h, best_c5

def proj_grad_step(h, T, n):
    val, grad = soft_c5_val_grad(h, T)
    step = 0.8 / max(T * 2.0 / n, 1e-6)
    step = min(step, 0.5)
    for _ in range(20):
        h_new = project_feasible(h - step * grad, n)
        val_new, _ = soft_c5_val_grad(h_new, T)
        if val_new <= val + 1e-13:
            break
        step *= 0.5
    return h_new, val_new

def pgd_loop(h, n, t_end, T_start, T_end, max_iter=8000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        frac = float(it) / max_iter
        T = T_start * (T_end / T_start) ** frac
        T = min(T, 800.0)
        h_new, _ = proj_grad_step(best_h, T, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def make_start(n, seed_val, variant):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0, 2, n, endpoint=False)
    if variant == 0:
        return 0.5 + 0.28 * np.sin(2 * np.pi * x + rng.uniform(0, 6.28))
    elif variant == 1:
        return 0.5 + 0.2 * np.sin(4 * np.pi * x) + 0.12 * np.sin(6 * np.pi * x + rng.uniform(0, 3))
    elif variant == 2:
        return 0.5 + 0.35 * np.cos(2 * np.pi * 3 * x + rng.uniform(0, 6.28))
    else:
        return np.clip(0.5 + 0.2 * rng.randn(n), 0.05, 0.95)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = make_start(160, seed, 0)
    n = len(best_h)
    best_h = project_feasible(best_h, n)

    t1 = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, t1)

    # Phase 2: Projected gradient on soft-max (20%)
    t2 = t0 + budget_s * 0.55
    best_h, best_c5 = pgd_loop(best_h, n, t2, 50.0, 2000.0, 6000)
    best_h = project_feasible(best_h, n)

    t3 = t0 + budget_s * 0.75

    # Phase 4: Multi-start exploration (12%)
    t4 = t0 + budget_s * 0.87
    n_starts = 4
    per_start = (t4 - time.time()) / n_starts
    for s in range(n_starts):
        if time.time() > t4:
            break
        h_try = make_start(n, seed * 7 + s * 13, s % 4)
        h_try = project_feasible(h_try, n)
        sub_end = min(time.time() + per_start * 0.6, t4)

        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Final gradient polish (5%)
    t5 = t0 + budget_s * 0.94
    best_h, best_c5 = pgd_loop(best_h, n, t5, 500.0, 5000.0, 3000)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 6: Final coordinate descent (1%)
    best_h, best_c5 = coord_descent(best_h, n, t_end - 2)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
