def swap_delta(h, a, b, n):
    """Compute delta_c[k] for swap h[a]:1->0, h[b]:0->1"""
    # k ranges from -(n-1) to (n-1)
    k = np.arange(-(n-1), n)
    # term_1: 1-h[a+k] if in range
    ak = a + k
    t1 = np.zeros(2*n-1)
    m1 = (ak >= 0) & (ak < n)
    t1[m1] = 1 - h[ak[m1]]
    # term_2: 1-h[b+k] if in range
    bk = b + k
    t2 = np.zeros(2*n-1)
    m2 = (bk >= 0) & (bk < n)
    t2[m2] = 1 - h[bk[m2]]
    # term_3: h[a-k] if in range
    ak2 = a - k
    t3 = np.zeros(2*n-1)
    m3 = (ak2 >= 0) & (ak2 < n)
    t3[m3] = h[ak2[m3]]
    # term_4: h[b-k] if in range
    bk2 = b - k
    t4 = np.zeros(2*n-1)
    m4 = (bk2 >= 0) & (bk2 < n)
    t4[m4] = h[bk2[m4]]
    # Handle interactions
    # If a+k == b: t1 should use new h[b]=1, so 1-1=0 instead of 1-0=1
    # k where a+k==b: k = b-a
    k_ab = b - a
    if -(n-1) <= k_ab < n:
        idx = k_ab + n - 1
        if 0 <= a + k_ab < n and a + k_ab == b:
            t1[idx] = 0  # was 1 (since h[b] was 0), now should be 0 (h_new[b]=1)
    # If b+k == a: t2 should use new h[a]=0, so 1-0=1 instead of 1-1=0
    k_ba = a - b
    if -(n-1) <= k_ba < n:
        idx = k_ba + n - 1
        if 0 <= b + k_ba < n and b + k_ba == a:
            t2[idx] = 1  # was 0 (since h[a] was 1), now should be 1 (h_new[a]=0)
    # If a-k == b: t3 should use new h[b]=1
    k_ab2 = a - b  # a-k==b => k=a-b
    if -(n-1) <= k_ab2 < n:
        idx = k_ab2 + n - 1
        if 0 <= a - k_ab2 < n and a - k_ab2 == b:
            t3[idx] = 1  # was 0 (h[b] was 0), now 1
    # If b-k == a: t4 should use new h[a]=0
    k_ba2 = b - a  # b-k==a => k=b-a
    if -(n-1) <= k_ba2 < n:

        if 0 <= b - k_ba2 < n and b - k_ba2 == a:
            t4[idx] = 0  # was 1 (h[a] was 1), now 0
    
    return -t1 + t2 + t3 - t4

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    sorted_idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[sorted_idx[:n // 2]] = 1.0
    return hb

def swap_delta_at(h, a, b, k, n):
    dk = 0.0
    ak = a + k
    if 0 <= ak < n:
        val = h[ak]
        if ak == b:
            val = 0.0
        dk -= 1.0 - val
    bk = b + k
    if 0 <= bk < n:
        val = h[bk]
        if bk == a:
            val = 1.0
        dk += 1.0 - val
    ak2 = a - k
    if 0 <= ak2 < n:
        val = h[ak2]
        if ak2 == b:
            val = 1.0
        dk += val
    bk2 = b - k
    if 0 <= bk2 < n:
        val = h[bk2]
        if bk2 == a:
            val = 0.0
        dk -= val
    return dk

def binary_swap_opt(h, n, t_end, max_sweeps=20000):
    h = make_binary(h, n).astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        k_star = np.argmax(corr)
        k_star_lag = k_star - (n - 1)
        best_new = M
        best_a, best_b = -1, -1
        for a in ones:
            for b in zeros:
                dk = swap_delta_at(h, a, b, k_star_lag, n)
                new_star = M + dk
                if new_star < best_new - 1e-12:
                    full_delta = np.zeros(2 * n - 1)
                    ks = np.arange(-(n - 1), n)
                    for ki, kk in enumerate(ks):
                        full_delta[ki] = swap_delta_at(h, a, b, kk, n)
                    new_max = np.max(corr + full_delta)
                    if new_max < best_new:
                        best_new = new_max
                        best_a, best_b = a, b
        if best_a < 0:
            break
        h[best_a] = 0.0
        h[best_b] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        M = np.max(corr)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h, M * (2.0 / n)

def coord_descent_cont(h, n, t_end, max_sweeps=300):
    dx = 2.0 / n
    best_h = h.copy().astype(float)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            coeff = np.zeros(2 * n - 1)
            ks = np.arange(-(n - 1), n)
            for ki, kk in enumerate(ks):
                ak = i + kk
                ak2 = i - kk
                if 0 <= ak < n:
                    coeff[ki] += (1.0 - best_h[ak])
                if 0 <= ak2 < n:
                    coeff[ki] -= best_h[ak2]
            c0_base = corr[n - 1]
            def obj(v):
                cn = corr + (v - old) * coeff
                cn[n - 1] = c0_base + (v - old) * (1.0 - v) - (v - old) * old
                return np.max(cn)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            best_v = res.x if res.fun < obj(0.0) and res.fun < obj(1.0) else (
                0.0 if obj(0.0) < obj(1.0) else 1.0)
            best_f = obj(best_v)
            if best_f * dx < best_c5 - 1e-13:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            break
    return best_h, best_c5

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def perturb_binary(h, n, rng, n_perturb):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_perturb):
        if len(ones) < 2 or len(zeros) < 2:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = h[b], h[a]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h

def thue_morse_n(n, frac):
    tm = np.zeros(n)
    v = 1
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
        v = 1 - v
    cnt = tm.sum()
    target = n // 2
    if cnt > target:
        idx_1 = np.where(tm == 1)[0]
        rng_remove = np.random.choice(len(idx_1), size=int(cnt - target), replace=False)
        tm[idx_1[rng_remove]] = 0
    elif cnt < target:
        idx_0 = np.where(tm == 0)[0]
        rng_add = np.random.choice(len(idx_0), size=int(target - cnt), replace=False)
        tm[idx_0[rng_add]] = 1
    return tm

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    t0 = time.time()

    n_best = 160
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = thue_morse_n(160, 0.5)
    n_best = len(best_h)
    best_h = make_binary(best_h, n_best)
    best_c5 = compute_c5(best_h)

    def remaining_frac():
        return max(0.0, 1.0 - (time.time() - t0) / budget_s)

    # Phase 1: Binary swap optimization at n_best
    t1 = t0 + budget_s * 0.15
    bh, bc5 = binary_swap_opt(best_h, n_best, t1)
    if bc5 < best_c5:
        best_c5, best_h = bc5, bh

    # Phase 2: Contour constraint + SLSQP refinement
    T = 500
    dx = 2.0 / n_best
    def smooth_obj(x, T_val=T):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        return m + np.log(np.sum(np.exp(T_val * (c - m)))) / T_val
    def con_eq(x):
        return x.sum() - n_best / 2.0
    try:
        res = minimize(smooth_obj, best_h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n_best,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': 800, 'ftol': 1e-15})
        hc = project(res.x, n_best)
        c5v = compute_c5(hc)

    best_h = np.clip(best_h, 0.0, 1.0)
