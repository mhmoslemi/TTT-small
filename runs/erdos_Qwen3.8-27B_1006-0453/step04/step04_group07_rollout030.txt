                continue

            coeff[idx1] += (1.0 - best_h[js])

            delta1 = 1.0 - old

            f1 = np.max(corr + delta1 * coeff)

            # Determine best v
            if f0 <= f1:

            else:

            if f0 > f_old and f1 > f_old:

                    continue

            corr += (best_v - old) * coeff

    x_new = np.linspace(0, 2, n_new, endpoint=False)

            break
        ones_l = np.where(h_bin == 1)[0]
        zeros_l = np.where(h_bin == 0)[0]
        if len(ones_l) == 0 or len(zeros_l) == 0:
            break
        cur_max = np.max(corr)
        best_imp = 0
        best_pair = None
        
        # Compute add coefficients for all zeros
        # For speed, compute in batch
        best_pair_improve = 0
        for a in zeros_l:
            # coeff_a: change when h[a] goes 0→1
            ca = np.zeros(2*n-1)
            ca[a - js + nk1] = (1.0 - h_bin[js])
            ca[js - a + nk1] -= h_bin[js]
            for b in ones_l:
                # coeff_b: change when h[b] goes 1→0 = -coeff for 0→1 at b
                # But h_bin[b]=1, so 0→1 change at b would be:
                # (1-h[b-k]) - h[b+k]
                # For 1→0: negative of that
                cb_neg = np.zeros(2*n-1)
                # Actually, let me just compute the combined change
                # new_corr = corr + ca + cb where cb is the 1→0 change at b
                # cb[k] = -(1-h[b-k])*[0<=b-k<n] + h[b+k]*[0<=b+k<n]
                # = -[(1-h[b-js]) at (b-js+nk1)] + [h[b+js] at (js-b+nk1)]... 
                
                # Let me just precompute
                cb = np.zeros(2*n-1)
                cb[b - js + nk1] = -(1.0 - h_bin[js])
                cb[js - b + nk1] = h_bin[js]
                
                nc = corr + ca + cb
                nm = np.max(nc)
                imp = cur_max - nm
                if imp > best_pair_improve:
                    best_pair_improve = imp
                    best_pair = (a, b)
                    best_nc = nc
        
        if best_pair is None:
            break
        
        a, b = best_pair
        h_bin[a] = 1
        h_bin[b] = 0
        corr = best_nc
        best_c5 = np.max(corr) * dx
    
    # Un-round: use h_bin but keep fractional info
    return h_bin, best_c5

def subgradient_step(h, n, step_size, t_end):
    """One subgradient step for the max."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nk1 = n - 1
    js = np.arange(n)
    max_val = np.max(corr)
    active = (corr >= max_val - 1e-10)
    
    # Subgradient: for each i, sum over active k of ∂corr[k]/∂h[i]
    # ∂corr[k+nk1]/∂h[i] = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]
    # This is the same coeff as in coord descent
    g = np.zeros(n)
    for i in range(n):
        coeff_i = np.zeros(2*n-1)
        coeff_i[i - js + nk1] += (1.0 - h[js])
        coeff_i[js - i + nk1] -= h[js]
        g[i] = np.sum(coeff_i[active])
    
    # Update: h -= step_size * g, projected
    h_new = h - step_size * g
    h_new = project(h_new, n)
    return h_new

Hmm, the subgradient step is O(n²) per step (n coordinates × O(n) coeff computation). That's fine.

But I realize the subgradient approach might not be the most effective. Let me focus on what's proven to work:
1. Fast coordinate descent (the main workhorse)
2. Binary pair-flip search
3. Multi-resolution exploration
4. Random restarts + local search

Let me also think about a key insight: the coordinate descent can be made MUCH faster by not doing the golden section search. Instead, just try v=0 and v=1 for each coordinate. This is a "bang-bang" control approach. Since the objective is max of linear functions (in each coordinate), the optimum for each coordinate is at a boundary or at an intersection. Trying just v=0 and v=1 captures the boundary cases, and the intersection cases can be found by a few iterations.

Actually, for the max of linear functions f(v) = max_m (a_m + v*b_m), the minimum over [0,1] is:
- If all b_m ≤ 0: minimum at v=1
- If all b_m ≥ 0: minimum at v=0
- Otherwise: at an intersection of two lines (or at a boundary of the active set)

The intersection of lines (a_i, b_i) and (a_j, b_j) is at v = (a_j - a_i)/(b_i - b_j).

So I can:
1. Find the active set A = {m : a_m + 0*b_m = max} at v=0
2. Find the active set B = {m : a_m + 1*b_m = max} at v=1
3. For each m in A with b_m > 0 and each m' in B with b_m' < 0, compute intersection
4. The optimum is the minimum of f(v) at these intersections and at v=0, v=1

But this is complex. Let me just use the golden section (25 iterations) which is O(25*n) per coordinate. For n=160, that's 4000 operations per coordinate, 640K per sweep. Very fast.

Actually, I just realized: the golden section only triggers when BOTH v=0 and v=1 are worse than current. This should be rare in practice (most coordinates will be pushed to 0 or 1). So the actual cost per sweep is closer to O(n²) (just the max computations).

Key improvements over previous code:
1. Incremental correlation in coord descent (O(n) per step vs O(n²))
2. Binary pair-flip local search (new)
3. More systematic multi-resolution (try more resolutions, more time each)
4. Subgradient steps for escaping local optima (new)
5. Better time management

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def fast_coord_descent(h, n, t_end, max_sweeps=10000):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    nk1 = n - 1
    js = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            if old <= 1e-14 or old >= 1.0 - 1e-14:
                continue
            coeff = np.zeros(2*n-1)
            coeff[i - js + nk1] += (1.0 - best_h[js])
            coeff[js - i + nk1] -= best_h[js]
            f0 = np.max(corr - old * coeff)
            f1 = np.max(corr + (1.0 - old) * coeff)
            f_cur = np.max(corr)
            if f0 >= f_cur - 1e-15 and f1 >= f_cur - 1e-15:
                continue
            if f0 <= f1:
                best_v, best_val = 0.0, f0
            else:
                best_v, best_val = 1.0, f1
            if f0 > f_cur - 1e-15 and f1 > f_cur - 1e-15:
                lo, hi = 0.0, 1.0
                gr = 0.6180339887498949
                amin = corr - old * coeff
                for _ in range(30):
                    x1 = lo + gr * (hi - lo)
                    x2 = hi - gr * (hi - lo)
                    if np.max(amin + x1 * coeff) < np.max(amin + x2 * coeff):
                        hi = x2
                    else:
                        lo = x1
                best_v = (lo + hi) * 0.5
                best_val = np.max(amin + best_v * coeff)
                if best_val >= f_cur - 1e-15:
                    continue
            best_h[i] = best_v
            corr += (best_v - old) * coeff
            nc5 = np.max(corr) * dx
            if nc5 < best_c5:
                best_c5 = nc5
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_pair_flip(h_frac, n, t_end, max_iters=500):
    dx = 2.0 / n
    target = n // 2
    h_bin = (h_frac > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        oi = np.where(h_bin == 1)[0]
        order = np.argsort(h_frac[oi])
        for k in range(n_ones - target):
            h_bin[oi[order[k]]] = 0
    elif n_ones < target:
        zi = np.where(h_bin == 0)[0]
        order = np.argsort(-h_frac[zi])
        for k in range(target - n_ones):
            h_bin[zi[order[k]]] = 1
    nk1 = n - 1
    js = np.arange(n)
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    best_c5 = np.max(corr) * dx
    for it in range(max_iters):
        if time.time() > t_end:
            break
        ones_l = np.where(h_bin == 1)[0]
        zeros_l = np.where(h_bin == 0)[0]
        if len(ones_l) == 0 or len(zeros_l) == 0:
            break
        cur_max = np.max(corr)
        best_imp = 1e-15
        best_pair = None
        best_nc = None
        n_pairs = len(ones_l) * len(zeros_l)
        if n_pairs > 40000:
            idx_pool = np.random.randint(0, n_pairs, 40000)
            ai = idx_pool // len(ones_l)
            bi = idx_pool % len(ones_l)
            a_arr = zeros_l[ai]
            b_arr = ones_l[bi]
        else:
            a_arr = np.repeat(zeros_l, len(ones_l))
            b_arr = np.tile(ones_l, len(zeros_l))
        for a, b in zip(a_arr, b_arr):
            ca = np.zeros(2*n-1)
            ca[a - js + nk1] = (1.0 - h_bin[js])
            ca[js - a + nk1] -= h_bin[js]
            cb = np.zeros(2*n-1)
            cb[b - js + nk1] = -(1.0 - h_bin[js])
            cb[js - b + nk1] = h_bin[js]
            nc = corr + ca + cb
            nm = np.max(nc)
            imp = cur_max - nm
            if imp > best_imp:
                best_imp = imp
                best_pair = (a, b)
                best_nc = nc
        if best_pair is None:
            break
        a, b = best_pair
        h_bin[a] = 1
        h_bin[b] = 0
        corr = best_nc
        best_c5 = np.max(corr) * dx
    return h_bin, best_c5

def smooth_and_opt(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(T*(c-m))))/T) * dx
    def con_fn(x):
        return x.sum() - n/2.0
    res = minimize(obj_fn, h, method='SLSQP', bounds=[(0.0,1.0)]*n,
                   constraints=[{'type':'eq','fun':con_fn}],
                   options={'maxiter':maxiter,'ftol':1e-14})
    return project(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def remaining():
        return budget_s - elapsed()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        best_h[0::2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Fast coord descent (200s)
    t1 = t0 + 200
    best_h, best_c5 = fast_coord_descent(best_h, n, t1, max_sweeps=20000)

    # Phase 2: Binary pair-flip + unround (80s)
    if remaining() > 80:
        t2 = t0 + 280
        h_bin, c5_bin = binary_pair_flip(best_h, n, t2, max_iters=500)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Also try: blend binary with fractional
        h_blend = 0.5 * h_bin + 0.5 * best_h
        h_blend = project(h_blend, n)
        h_blend, c5_blend = fast_coord_descent(h_blend, n, t2+30, max_sweeps=5000)
        if c5_blend < best_c5:
            best_c5 = c5_blend
            best_h = h_blend

    # Phase 3: More coord descent (60s)
    if remaining() > 60:
        t3 = t0 + 340
        best_h, best_c5 = fast_coord_descent(best_h, n, t3, max_sweeps=20000)

    # Phase 4: SLSQP with smoothing (60s)
    if remaining() > 60:
        t4 = t0 + 400
        for T in [50, 200, 800]:
            if elapsed() > t4:
                break
            try:
                hc = smooth_and_opt(best_h, T, n, maxiter=600)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                best_h, best_c5 = fast_coord_descent(best_h, n, min(t4, elapsed()+20), max_sweeps=5000)
            except Exception:
                pass

    # Phase 5: Multi-resolution exploration (350s)
    t5_end = t0 + 750
    resolutions = [100, 120, 140, 180, 200, 240, 280, 320]
    for n_try in resolutions:
        if elapsed() > t5_end:
            break
        time_per = min(40, (t5_end - time.time()))
        if time_per < 5:
            break
        h_tr = interp_h(best_h, n_try)
        # Also try random perturbation at new resolution
        h_pert = h_tr + 0.1 * np.random.randn(n_try)
        h_pert = project(h_pert, n_try)
        for h_start in [h_tr, h_pert]:
            h_try, c5_try = fast_coord_descent(h_start, n_try, t0 + elapsed() + time_per, max_sweeps=10000)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                n = n_try
        # Binary search at this resolution
        if elapsed() < t5_end - 10:
            h_b, c5_b = binary_pair_flip(best_h, n, t0 + elapsed() + 10, max_iters=200)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 6: Basin hopping with fast methods (200s)
    t6_end = t0 + 950
    n = len(best_h)
    work_h = best_h.copy()
    work_c5 = best_c5
    it_hop = 0
    while elapsed() < t6_end:
        it_hop += 1
        cool = max(0.05, 1.0 - 0.7 * it_hop / 80.0)
        step_sz = 0.03 + 0.12 * cool
        h_p = work_h + step_sz * np.random.randn(n)
        h_p = project(h_p, n)
        t_sub = min(elapsed() + 15, t6_end - 5)
        h_try, c5_try = fast_coord_descent(h_p, n, t_sub, max_sweeps=3000)
        temp = 0.001 * cool + 0.0002
        if c5_try < work_c5 or np.random.rand() < np.exp(-(c5_try - work_c5)/temp):
            work_h, work_c5 = h_try, c5_try
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try.copy()

    # Phase 7: Final polish (100s)
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s - 20, max_sweeps=20000)

    # Phase 8: One last binary search + unround
    if elapsed() < budget_s - 15:
        h_b, c5_b = binary_pair_flip(best_h, n, t0 + budget_s - 10, max_iters=100)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
        best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s - 2, max_sweeps=5000)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
