            d[n - 1 + k] += (1.0 - h[i - k])
            d[n - 1 - k] += (-h[i - k])
        if i + k < n:
            d[n - 1 + k] += (-h[i + k])
            d[n - 1 - k] += (1.0 - h[i + k])
    c0 = corr_base[n - 1]
    
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals)
    
    # Try to find minimum using golden section / bounded
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def lbfgsb_opt(h, T, n, maxiter=3000, mu=1e6):
    target = n / 2.0
    def obj(x):
        return smooth_c5(x, T) + mu * (x.sum() - target) ** 2
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='L-BFGS-B', bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=80, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def focus_descent(h, n, t_end, n_rounds=5):
    """Focus on the active constraint and try to reduce it."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1-best_h, mode='full')
        dx = 2.0 / n
        k_active = np.argmax(corr) - (n-1)
        # Find coordinates that most affect corr[k_active]
        # For corr[n-1+k_active], the relevant coordinates are those where
        # h[i] or h[i-k_active] appears
        mask = np.zeros(n, dtype=bool)
        if k_active > 0:
            mask[k_active:] = True
            mask[:n-k_active] = True
        elif k_active < 0:
            mask[:n+k_active] = True
            mask[-k_active:] = True
        else:
            mask[:] = True
        # Optimize in random order within the mask
        idx = np.where(mask)[0]
        rng = np.random.default_rng(r * 1000)
        rng.shuffle(idx)
        for i in idx[:min(len(idx), 50)]:
            if time.time() > t_end:
                break
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-12:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5:
                    best_c5 = c5
    return best_h, best_c5

def smooth_perturb(h, n, rng, scale):
    """Apply smooth (low-frequency) perturbation."""
    h_new = h.copy().astype(float)
    # Add low-frequency noise
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 2*np.pi)
    x = np.arange(n)
    h_new += scale * np.sin(2*np.pi*freq*x/n + phase)
    h_new += scale * 0.5 * rng.standard_normal(n).cumsum() / n
    return project(h_new, n)

def basin_hop(h, n, t_end, rng, n_iters=100):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(n_iters, 1))
        step = 0.06 * (0.2 + 0.8 * cool)
        # Mix smooth and white perturbation
        if rng.random() < 0.4:
            pert = smooth_perturb(work, n, rng, step)
        else:
            pert = work + step * rng.standard_normal(n)
            pert = project(pert, n)
        # Polish
        T = 200 + 800 * cool
        try:
            hc = lbfgsb_opt(pert, T, n, maxiter=500, mu=1e8)
        except Exception:
            hc = pert
        cv = compute_c5(hc)
        temp = 0.003 * cool + 0.0002
        if cv < work_c5 or rng.random() < np.exp(-(cv - work_c5) / max(temp, 1e-10)):
            work, work_c5 = hc, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interp_to(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

**Water-filling / dual approach**:

- The primal is: min_h max_k corr[k]*dx s.t. sum(h)=n/2, 0≤h≤1
- The Lagrangian: L(h, λ) = max_k corr[k]*dx - sum(λ_k * (corr[k]*dx - C)) ... this gets complicated.

Actually, let me think about this differently. The problem can be written as:
min C
s.t. corr[k] ≤ C/dx for all k
     sum(h) = n/2
     0 ≤ h ≤ 1

This is a feasibility problem for a given C. We can binary search on C. For each C, the feasibility is:
Does there exist h with 0≤h≤1, sum(h)=n/2, and sum_j h[j]*(1-h[j+k]) ≤ C/dx for all k?

The constraints are quadratic (non-convex because of h[j]*h[j+k] terms with negative sign). So it's not a convex feasibility problem.

But wait: sum_j h[j]*(1-h[j+k]) = sum_j h[j] - sum_j h[j]*h[j+k]
The first term is linear, the second is bilinear (non-convex). So the constraint is:
sum_j h[j] - sum_j h[j]*h[j+k] ≤ C/dx
i.e., sum_j h[j]*h[j+k] ≥ sum_j h[j] - C/dx

This is a non-convex constraint (the left side is a bilinear form).

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + room * (diff / tot), 0, 1)
        else:
            if s < 1e-14:
                break
            h = np.clip(h * (target / s), 0, 1)
    return h

def coord_opt(h_arr, n, i):
    h = h_arr.copy()
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if i - k >= 0:
            d[n - 1 + k] += (1.0 - h[i - k])
            d[n - 1 - k] += (-h[i - k])
        if i + k < n:
            d[n - 1 + k] += (-h[i + k])
            d[n - 1 - k] += (1.0 - h[i + k])

        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals)

        if i != j:

        h_i = interp_to(best_h, n_try)
        for T in [500, 5000]:
            if time.time() > t0 + budget_s * 0.90:
                break
            try:
                hc = lbfgsb_opt(h_i, T, n_try, maxiter=1500, mu=1e8)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5, n = hc, cv, n_try
            except Exception:
                pass
        if n == n_try:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.92, max_sweeps=20, rng=rng)

    # Phase 7: Final polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=30, rng=rng)
    for T in [30000, 100000]:
        if time.time() > t0 + budget_s * 0.995:
            break
        try:
            hc = lbfgsb_opt(best_h, T, n, maxiter=3000, mu=1e9)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
