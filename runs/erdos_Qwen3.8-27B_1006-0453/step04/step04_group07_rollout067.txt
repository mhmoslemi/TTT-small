def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0

    n = len(best_h)
    target = n / 2.0
    best_h = project_h(best_h, target)
    best_c5, _ = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Heavy coordinate descent (25%)
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.22, target, max_sweeps=50000, eps_reg=0.005)

    # Phase 2: Subgradient escape + refine (12%)
    h2, c5_2 = subgrad_loop(best_h, n, t0 + budget_s * 0.32, target, n_iters=400000, alpha_init=0.08)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.35, target, max_sweeps=10000)

    # Phase 3: Smooth gradient for balanced optimization (15%)
    for T in [40, 150, 600]:
        if elapsed() > budget_s * 0.48:
            break
        h3, c5_3 = smooth_grad_loop(best_h, n, T, min(t0 + budget_s * 0.48, time.time() + 25),
                                     target, n_iters=25000, alpha_init=0.03)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.50, target, max_sweeps=15000, eps_reg=0.01)

    # Phase 4: Basin hopping with structured perturbations (30%)
    work_h = best_h.copy()
    work_c5 = best_c5
    max_hop = 250
    for it in range(max_hop):
        if elapsed() > budget_s * 0.82:
            break
        cool = max(0.08, 1.0 - 0.92 * it / max_hop)
        h_p = structured_perturb(work_h, n, target, rng)
        h_r, c5_r = subgrad_loop(h_p, n, min(t0 + budget_s * 0.82, time.time() + 2.5),
                                  target, n_iters=40000, alpha_init=0.05)
        h_r, c5_r = coord_desc(h_r, n, min(t0 + budget_s * 0.82, time.time() + 2.0),
                                target, max_sweeps=400)
        temp = 0.0008 * cool + 0.00005
        if c5_r < work_c5 or rng.rand() < np.exp(-(c5_r - work_c5) / temp):
            work_h, work_c5 = h_r, c5_r
            if c5_r < best_c5:
                best_h, best_c5 = h_r.copy(), c5_r
        elif it % 50 == 49:
            # Force accept to avoid stalling
            work_h, work_c5 = h_r, c5_r

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Alternate resolutions (8%)
    for n_try in [128, 192, 256]:
        if elapsed() > budget_s * 0.93:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_tr = np.interp(x_new, x_old, best_h)
        h_tr = project_h(h_tr, n_try / 2.0)
        h_tr, c5_tr = coord_desc(h_tr, n_try, min(t0 + budget_s * 0.93, time.time() + 8),
                                  n_try / 2.0, max_sweeps=3000, eps_reg=0.005)
        if c5_tr < best_c5:
            best_h, best_c5, n, target = h_tr, c5_tr, n_try, n_try / 2.0

    # Phase 6: Final coordinate descent polish (7%)
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.99, target, max_sweeps=30000, eps_reg=0.0)

    # Smoke out with subgradient if time
    if elapsed() < budget_s * 0.995:
        h_s, c5_s = subgrad_loop(best_h, n, t0 + budget_s * 0.995, target, n_iters=100000, alpha_init=0.02)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    best_h = project_h(best_h, target)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
