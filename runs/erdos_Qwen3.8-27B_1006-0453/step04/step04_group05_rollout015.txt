    N = len(h)
    g = 0.0
    if 0 <= p + L < N:
        g += (1.0 - h[p + L])
    if 0 <= p - L < N:
        g -= h[p - L]
    return g

    k = min(k_top, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    c = corr[idx].astype(float)
    cmax = c.max()
    w = np.exp((c - cmax) / temp)
    w /= w.sum()
    return cmax, idx, w

max_q max_L (corr[L] - rm[p][L] + ad[q][L]) = max_q (max_L (corr[L] - rm[p][L] + ad[q][L]))

    M = 2 * N - 1

            if 0 <= q - L < N:
                d -= h[q - L]

    return ones, zeros, rm, ad

import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_weighted_grad(h, L_list, w_list):
    N = len(h)
    grad = np.zeros(N)
    for L, wi in zip(L_list, w_list):
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N-L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:N-L]
        else:
            M = -L
            grad[M:] += wi * (1.0 - h[:N-M])
            grad[:N-M] -= wi * h[M:]
    return grad

def continuous_degdescent(h, N, deadline, k_top=10, temp_init=2.0, lr_init=0.015):
    h = h.astype(float).copy()
    target = N * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    dx = 2.0 / N
    temp = temp_init
    lr = lr_init
    it = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak * dx < best_c5:
            best_c5 = peak * dx
            best_h = h.copy()
        k = min(k_top, 2 * N - 1)
        top_idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[top_idx].astype(float)
        c_ref = c_top.max()
        w = np.exp((c_top - c_ref) / temp)
        w /= w.sum()
        L_list = top_idx - (N - 1)
        grad = compute_weighted_grad(h, L_list, w)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-10:
            break
        h_new = project_sum(h - lr * grad, target)
        if float(compute_corr(h_new).max()) < peak:
            h = h_new
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.5
            h_new = project_sum(h - lr * grad, target)
            h = h_new
        it += 1
        if it % 30 == 0:

    return best_h, best_c5

def vec_steepdescent(h, N, deadline, max_sweeps=3000):
    """Vectorized steepest descent for binary h with 1-swap moves."""
    h = h.astype(float).copy()
    M = 2 * N - 1
    corr = compute_corr(h)
    peak = float(corr.max())
    best_peak = peak
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute rm and ad vectors
        rm = np.zeros((len(ones), M))
        for i, p in enumerate(ones):
            L1 = np.arange(-p, N - p)
            L2 = np.arange(p - N + 1, p + 1)
            rm[i, L1 + (N-1)] += -(1.0 - h[p + L1])
            rm[i, L2 + (N-1)] += h[p - L2]
        
        ad = np.zeros((len(zeros), M))
        for i, q in enumerate(zeros):
            L1 = np.arange(-q, N - q)
            L2 = np.arange(q - N + 1, q + 1)
            ad[i, L1 + (N-1)] += (1.0 - h[q + L1])
            ad[i, L2 + (N-1)] += -h[q - L2]
        
        # Find best (p, q) swap
        best_new_peak = peak
        best_pi = -1
        best_qi = -1
        for pi in range(len(ones)):
            base = corr - rm[pi]
            # For each q: new_peak = max(base + ad[q])
            peaks_q = (base[None, :] + ad).max(axis=1)
            qi = np.argmin(peaks_q)
            if peaks_q[qi] < best_new_peak:
                best_new_peak = peaks_q[qi]
                best_pi = pi
                best_qi = qi
        
        if best_pi < 0:
            break  # No improving swap
        
        p = ones[best_pi]

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_weighted_grad(h, L_list, w_list):
    N = len(h)
    grad = np.zeros(N)
    for L, wi in zip(L_list, w_list):
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            grad[:N - L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:N - L]
        else:
            M = -L
            grad[M:] += wi * (1.0 - h[:N - M])
            grad[:N - M] -= wi * h[M:]
    return grad

def continuous_opt(h, N, deadline, k_top=10, temp_init=2.0, lr_init=0.012):
    h = h.astype(float).copy()
    target = N * 0.5
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    dx = 2.0 / N
    temp = temp_init
    lr = lr_init
    it = 0
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak * dx < best_c5:
            best_c5 = peak * dx
            best_h = h.copy()
        k = min(k_top, 2 * N - 1)
        top_idx = np.argpartition(corr, -k)[-k:]
        c_top = corr[top_idx].astype(float)
        c_ref = c_top.max()
        w = np.exp((c_top - c_ref) / temp)
        w /= w.sum()
        L_list = top_idx - (N - 1)
        grad = compute_weighted_grad(h, L_list, w)
        if np.linalg.norm(grad) < 1e-12:
            break
        h_new = project_sum(h - lr * grad, target)
        if float(compute_corr(h_new).max()) < peak:
            h = h_new
            lr = min(lr * 1.04, 0.06)
        else:
            lr *= 0.5
            h_new = project_sum(h - lr * grad, target)
            h = h_new
        it += 1
        if it % 40 == 0:
            temp = max(temp * 0.96, 0.2)
    return best_h, best_c5

def vec_steepdescent(h, N, deadline, max_sweeps=2000):
    h = h.astype(float).copy()
    M = 2 * N - 1
    corr = compute_corr(h)
    peak = float(corr.max())
    best_peak = peak
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rm = np.zeros((len(ones), M))
        for i, p in enumerate(ones):
            L1 = np.arange(-p, N - p)
            L2 = np.arange(p - N + 1, p + 1)
            rm[i, L1 + (N - 1)] += -(1.0 - h[p + L1])
            rm[i, L2 + (N - 1)] += h[p - L2]
        ad = np.zeros((len(zeros), M))
        for i, q in enumerate(zeros):
            L1 = np.arange(-q, N - q)
            L2 = np.arange(q - N + 1, q + 1)
            ad[i, L1 + (N - 1)] += (1.0 - h[q + L1])
            ad[i, L2 + (N - 1)] += -h[q - L2]
        best_new_peak = peak
        best_pi, best_qi = -1, -1
        for pi in range(len(ones)):
            base = corr - rm[pi]
            peaks_q = (base[None, :] + ad).max(axis=1)
            qi = int(np.argmin(peaks_q))
            if peaks_q[qi] < best_new_peak:
                best_new_peak = float(peaks_q[qi])
                best_pi, best_qi = pi, qi
        if best_pi < 0:
            break
        p = ones[best_pi]
        q = zeros[best_qi]
        h[p] = 0.0
        h[q] = 1.0
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
    return h, best_peak * (2.0 / N)

def make_binary(h, N):
    h_b = (h > 0.5).astype(float)
    nh = N // 2
    n_ones = int(h_b.sum())
    if n_ones > nh:
        idx = np.where(h_b == 1)[0]
        h_b[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h_b == 0)[0]
        h_b[idx[:nh - n_ones]] = 1
    return h_b

def interp_resample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def structured_start(N, rng, kind=0):
    h = np.zeros(N)
    if kind == 0:
        h[::2] = 1.0
    elif kind == 1:
        beta = 1.618
        h = (np.floor(np.arange(N) * beta / N * 2.0) % 2 == 0).astype(float)
    elif kind == 2:
        alpha = 0.7071
        frac = (np.arange(N) * alpha) % 1.0
        h = (frac < 0.5).astype(float)
    else:
        h = (rng.random(N) > 0.5).astype(float)
    return make_binary(h, N)

def fast_anneal(h, N, n_iters, rng, T0=2.0, T1=0.002):
    h_b = make_binary(h, N)
    best = h_b.copy()
    corr = compute_corr(h_b)
    best_pk = float(corr.max())
    cur_pk = best_pk
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        h_b[i], h_b[j] = h_b[j], h_b[i]
        new_pk = float(compute_corr(h_b).max())
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.003 and rng.random() < np.exp(-delta / max(T, 1e-6))):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
    return best, best_pk * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Continuous optimization on initial
    d1 = min(deadline - 350, t0 + 60)
    h_c, c5_c = continuous_opt(h0, N0, d1, k_top=10, temp_init=2.0, lr_init=0.012)
    if c5_c < best_c5:
        best_c5, best_h = c5_c, h_c.copy()

    # Phase 2: Binary steepest descent
    h_bin = make_binary(h_c, N0)
    d2 = min(deadline - 200, t0 + 180)
    h_bd, c5_bd = vec_steepdescent(h_bin, N0, d2, max_sweeps=1500)
    if c5_bd < best_c5:

    # Phase 3: Anneal + steepest descent restarts
    h_cur = best_h.copy()
    d3 = deadline - 120
    restart = 0
    while time.time() < d3 and restart < 8:
        ann_t = max(8, min(20, (d3 - time.time()) * 0.3))
        n_ann = int(ann_t * 80)
        h_a, c5_a = fast_anneal(h_cur, N0, n_ann, rng, T0=2.5, T1=0.002)
        sd_dl = min(d3 - 5, time.time() + 40)

        if c5_s < best_c5:

            h_cur = h_s.copy()
        restart += 1

    N_list = []
    for dn in [-60, -40, -20, 20, 40, 60, 80, 120]:
        Nt = N0 + dn
        if Nt >= 200 and Nt <= 800:

    for kind in range(4):

        if c5_s < best_c5:

    for N_try in N_list:
        if time.time() > deadline - 40:
            break

        h_ib = make_binary(h_int, N_try)
        sd_dl = min(deadline - 35, time.time() + 50)
        h_s, c5_s = vec_steepdescent(h_ib, N_try, sd_dl, max_sweeps=800)
        if c5_s < best_c5:

        if time.time() < deadline - 15:

                                         k_top=12, temp_init=1.5, lr_init=0.01)

        hF = project_sum(hF, len(best_h) / 2.0)

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
