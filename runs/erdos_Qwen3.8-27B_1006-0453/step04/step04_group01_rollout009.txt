from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_continuous(h, n):
    """Project to [0,1]^n with sum = n/2."""
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -np.max(vc) - 1.0
    hi = 1.0 - np.min(vc) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, n, rng):
    """Binarize h maintaining sum = n/2."""
    target = n // 2
    h_sorted_idx = np.argsort(h)[::-1]
    bits = np.zeros(n)
    bits[h_sorted_idx[:target]] = 1.0
    return bits

def thue_morse_init(n, rng):
    """Thue-Morse based initialization."""
    length = 1
    while length < n:
        length *= 2
    tm = np.zeros(length, dtype=int)
    for i in range(1, length):
        tm[i] = bin(i).count('1') & 1
    seq = tm[:n].astype(float)
    target = n // 2
    ones = np.where(seq == 1)[0]
    zeros = np.where(seq == 0)[0]
    num_ones = len(ones)
    if num_ones > target:
        rng.shuffle(ones)
        for i in range(num_ones - target):
            seq[ones[i]] = 0
    elif num_ones < target:
        rng.shuffle(zeros)
        for i in range(target - num_ones):
            seq[zeros[i]] = 1
    return seq

    """Golden ratio spaced initialization."""
    phi = (1 + np.sqrt(5)) / 2
    positions = np.floor((np.arange(n) / n) * n * (1 - 1/phi)) % n
    positions = np.sort(positions).astype(int)
    bits = np.zeros(n)
    bits[positions[:n//2]] = 1.0
    # Ensure exactly n/2 ones
    target = n // 2
    if bits.sum() != target:
        ones = np.where(bits == 1)[0]
        zeros = np.where(bits == 0)[0]
        if bits.sum() > target:
            for i in range(int(bits.sum()) - target):
                bits[ones[i]] = 0
        else:
            for i in range(target - int(bits.sum())):
                bits[zeros[i]] = 1
    return bits

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random
    t0 = time.time()

    best_c5 = 0.99
    best_h = None
    best_n = 160

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        best_h = h0.copy()
        best_c5 = compute_c5(best_h)
        best_n = n0

    configs = [
        (160, 'thue', 0.18),
        (160, 'init', 0.18),
        (128, 'thue', 0.14),
        (200, 'rand', 0.14),
        (160, 'rand', 0.10),
    ]

    sa_budget_end = t0 + budget_s * 0.62
    for (n, itype, tfrac) in configs:
        if time.time() > sa_budget_end:
            break
        t_alloc = budget_s * tfrac
        t_end_here = min(time.time() + t_alloc, sa_budget_end)
        if t_end_here - time.time() < 4:
            continue

        if itype == 'thue':
            base = thue_morse_arr(n, rng)
        elif itype == 'init' and best_h is not None and len(best_h) == n:
            base = best_h + 0.1 * rng.randn(n)
        else:
            base = rng.rand(n)

        bits = make_binary(n, base, rng)
        bb, bc = sa_binary(bits, n, t_end_here, T0=0.06, rng=rng)
        if bc < best_c5:
            best_c5 = bc
            best_h = bb
            best_n = n

    # Continuous polish
    if time.time() < t0 + budget_s * 0.92:
        n = best_n
        hp = best_h.copy()
        for T in [500, 3000, 15000]:
            if time.time() > t0 + budget_s * 0.82:
                break
            hc = slsqp_run(hp, n, T, 400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                hp = hc

        cd_end = min(t0 + budget_s * 0.92, time.time() + budget_s * 0.12)
        best_h, best_c5 = cont_cd(best_h, best_n, cd_end, rng)

    # Final SLSQP
    if time.time() < t0 + budget_s * 0.97:
        hc = slsqp_run(best_h.copy(), best_n, 50000, 500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_cont(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
