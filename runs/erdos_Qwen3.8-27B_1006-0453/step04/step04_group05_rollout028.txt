```python
        else:

def make_binary(h, N):
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    nh = N // 2
    if n_ones > nh:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        h_b[idx[order[nh - n_ones:] if False else range(n_ones - nh)]] = 0  # remove smallest

        order = np.argsort(-h[idx])
        h_b[idx[order[:nh - n_ones]]] = 1

def swap_delta(h, p, q, N):
    """Vectorized O(N) delta computation for swap p:1->0, q:0->1."""
    r_lo = max(0, N-1-p, p, N-1-q, q)
    r_hi = min(2*N-2, 2*N-2-p, p+N-1, 2*N-2-q, q+N-1)
    if r_lo > r_hi:
        return np.array([0.0]), r_lo, r_lo
    ks = np.arange(r_lo, r_hi + 1)
    delta = np.zeros(len(ks), dtype=float)
    
    idx1 = p + ks - N + 1
    m1 = (idx1 >= 0) & (idx1 < N)
    if m1.any():
        delta[m1] -= (1.0 - h[idx1[m1]])
    
    idx2 = p - ks + N - 1
    m2 = (idx2 >= 0) & (idx2 < N)
    if m2.any():
        v = h[idx2[m2]].copy()
        ii = idx2[m2]
        v[ii == p] = 0.0
        v[ii == q] = 1.0
        delta[m2] += v
    
    idx3 = q + ks - N + 1
    m3 = (idx3 >= 0) & (idx3 < N)
    if m3.any():
        v = h[idx3[m3]].copy()
        ii = idx3[m3]
        v[ii == p] = 0.0
        v[ii == q] = 1.0
        delta[m3] += (1.0 - v)
    
    idx4 = q - ks + N - 1
    m4 = (idx4 >= 0) & (idx4 < N)

        ii = idx4[m4]
        v[ii == p] = 0.0
        v[ii == q] = 1.0
        delta[m4] -= v
```

Let me think about whether the continuous optimization can help. The problem allows h ∈ [0,1]. The objective is max_k c[k] where c[k] = sum_i h[i](1-h[i+k]). This is a convex function of h (max of linear functions), so we can use subgradient methods. But the non-differentiability at the max makes it tricky.

A better approach for continuous: minimize the log-sum-exp of the correlations (smooth approximation of max).

f(h) = (1/β) * log(sum_k exp(β * c[k]))

As β→∞, this approaches max_k c[k]. The gradient is:
∂f/∂h[i] = (1/β) * sum_k [β * ∂c[k]/∂h[i] * exp(β*c[k])] / sum_k [exp(β*c[k])]
           = sum_k [∂c[k]/∂h[i] * soft_assignment(k)]

where soft_assignment(k) = exp(β*c[k]) / sum_j exp(β*c[j]) is a softmax distribution over lags.

∂c[k]/∂h[i] = (1-h[i+k]) - h[i-k] (with appropriate boundary handling)

This gives a smooth gradient that we can use with any optimizer.

Let me implement this:

```python
def continuous_optimize(h, N, deadline, target, beta_start=10, beta_end=100):
    target = target or N / 2.0
    h = h.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    
    c = compute_corr(h)
    lr = 0.01
    n_steps = 0
    n_lags = 2*N - 1
    
    while time.time() < deadline:
        # Compute smooth objective gradient
        shift = c - c.max()  # numerical stability
        w = np.exp(beta_start * shift)
        w /= w.sum()
        
        grad = np.zeros(N)
        # For each lag k with weight w[k], add w[k] * dc[k]/dh
        # dc[k]/dh[i] = (1-h[i+k-N+1]) * [0<=i+k-N+1<N] - h[i-k+N-1] * [0<=i-k+N+1<N]
        # This is: convolve-like operation
        # Actually: ∂c[k]/∂h[i] = (1-h[i+L]) - h[i-L] where L = k-(N-1)
        # Sum over k: sum_k w[k] * [(1-h[i+k-N+1]) - h[i-k+N-1]]
        # = sum_L (w[L+N-1]) * [(1-h[i+L]) - h[i-L]]
        # = (softmax_convolve_1 - softmax_convolve_0)
        
        # Compute sum_L w[L+N-1] * (1-h[i+L]) for all i
        # This is like a cross-correlation of w with (1-h)
        # sum_L w[L+N-1] * (1-h[i+L]) = corr(w_rev, 1-h) at appropriate offset
        
        # Let me just do it directly for each i (O(N²) but N=380 so it's fine)
        # Actually, let me use the FFT approach
        # correlate(w_centered, (1-h), 'full') gives sum over shifts
        # w_centered[k] = w[k] for k in [0, 2N-2]
        # We want: for each i, sum_k w[k] * (1-h[i+k-N+1]) where 0≤i+k-N+1<N
        # = sum_L w[L+N-1] * (1-h[i+L]) where -i ≤ L < N-i
        # = cross-correlation of w_c with (1-h) evaluated at position i
        
        # Let's use np.correlate
        # np.correlate(a, v, 'full')[m] = sum_n a[n] * v[m-n] (convention-dependent)
        # Actually np.correlate(a, v)[m] = sum_n a[n+m] * v[n] (for 'full' mode)
        
        # I want: out[i] = sum_L w_c[L] * (1-h[i+L]) where w_c[L] = w[L+N-1]
        # This is like a convolution
        # Let me just compute it with the loop since N is small
        
        # For each lag offset L from -(N-1) to N-1:
        # w_idx = L + (N-1), weight = w[w_idx]
        # affects h[i] with (1-h[i+L]) for i in [0, min(N-1, N-1-L)]
        # and h[i] with -h[i-L] for i in [max(0, L), N-1]
        
        for L in range(-(N-1), N):
            wi = w[L + N - 1]
            if wi < 1e-10:
                continue
            if L == 0:
                grad += wi * (2.0 - 2.0 * h)  # (1-h) - h = 1-2h
            elif L > 0:
                # (1-h[i+L]) for i in [0, N-1-L]
                grad[:N-L] += wi * (1.0 - h[L:])
                # -h[i-L] for i in [L, N-1]
                grad[L:] -= wi * h[:N-L]
            else:
                M = -L
                # (1-h[i+L]) = (1-h[i-M]) for i in [M, N-1]
                grad[M:] += wi * (1.0 - h[:N-M])
                # -h[i-L] = -h[i+M] for i in [0, N-1-M]
                grad[:N-M] -= wi * h[M:]
        
        h = h - lr * grad
        h = project_sum(h, target)
        n_steps += 1
        if n_steps % 50 == 0:
            lr *= 0.99
            beta_start *= 1.05
        
        c = compute_corr(h)
        peak = float(c.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
    
    return best_h, best_peak * dx
```
