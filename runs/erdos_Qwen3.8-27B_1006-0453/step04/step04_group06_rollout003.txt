import numpy as np
from scipy.optimize import minimize, minimize_scalar, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    s = np.log(np.sum(np.exp(T * (corr - m))))
    return (m + s / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    center = n - 1
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-10:
            continue
        k = idx - center
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n-k]
        else:
            mk = -k
            grad[mk:] += wi * (1.0 - h[:n-mk])
            grad[:n-mk] -= wi * h[mk:]
    return grad * dx

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.clip(h + mid, 0, 1).sum()
        if s_mid > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) / 2.0, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h

def lbfgs_smooth(h0, T, n, maxiter=500, lam=1e4):
    target = n / 2.0
    def obj(x):
        return smooth_c5(x, T) + lam * (x.sum() - target) ** 2
    def grad(x):
        return smooth_c5_grad(x, T) + 2.0 * lam * (x.sum() - target)
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return project_sum(np.clip(res.x, 0, 1), n)

def coord_opt_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    delta = np.zeros(2 * n - 1)
    for k in range(-i, n - i):
        j = center + k
        idx2 = i + k
        if 0 <= idx2 < n and 0 <= j < 2 * n - 1:
            if j == center:
                delta[j] = 1.0
            else:
                delta[j] = 1.0 - h_base[idx2]
    def f1d(x):
        vals = corr_base + delta * x
        vals[center] -= x * x
        return np.max(vals)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500, random_order=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n) if random_order else range(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            new_val = coord_opt_single(best_h, n, i)
            if abs(new_val - best_h[i]) > 1e-11:
                best_h[i] = new_val
                best_h = project_sum(best_h, n)
                c5_try = compute_c5(best_h)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def fourier_h(params, n, J):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for j in range(J):
        h += params[2*j] * np.cos(np.pi * (j+1) * x)
        h += params[2*j+1] * np.sin(np.pi * (j+1) * x)
    return h

def fourier_objective(params, n, J):
    h = fourier_h(params, n, J)
    pen = 1e5 * np.sum(np.maximum(0.0, -h)**2 + np.maximum(0.0, h - 1.0)**2)
    h = project_sum(np.clip(h, 0, 1), n)
    return compute_c5(h) + pen

def slsqp_opt(h0, T, n, maxiter=800):
    target = n / 2.0
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h0, method='SLSQP', bounds=[(0,1)]*n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_sum(np.clip(res.x, 0, 1), n)
