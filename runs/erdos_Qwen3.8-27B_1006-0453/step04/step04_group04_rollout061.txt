def proj(h, n):

    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * (diff / tot)
    else:
        if s > 1e-14:
            h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent_fast(h, n, t_end, max_sweeps=200):

    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    k0 = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        for i in range(n):
            old = best_h[i]
            # Compute a[k]
            a = np.zeros(2*n-1)
            for k in range(2*n-1):
                lag = k - k0
                if lag == 0:

                vals = corr + (t - old) * a
                vals[k0] = corr[k0] + (t - old) * (1.0 - t - old)
                return vals.max() * dx

            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            m_base = np.delete(corr, n - 1)
            a_base = np.delete(a, n - 1)
            c0 = corr[n - 1]
            f1 = max(m_base.max() + (x1 - old) * a_base.max() if len(a_base) else 0, c0 + (x1 - old) * (1.0 - x1 - old))
            # More careful:
            def fv(t):
                mm = np.max(corr + (t - old) * a)
                qq = c0 + (t - old) * (1.0 - t - old)
                return max(mm, qq)
            f1, f2 = fv(x1), fv(x2)
            for _ in range(28):
                if f1 < f2:
                    hi, x2, f2 = x2, x1, f1
                    x1 = hi - gr * (hi - lo)
                    f1 = fv(x1)
                else:
                    lo, x1, f1 = x1, x2, f2
                    x2 = lo + gr * (hi - lo)
                    f2 = fv(x2)
            t_new = (lo + hi) / 2.0
            if abs(t_new - old) > 1e-7:
                best_h[i] = t_new
                improved = True
        best_h = proj(best_h, n)
        cn = compute_c5(best_h)
        if cn < best_c5:
            best_c5 = cn
        if not improved:
            break
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def binary_sa(h, n, t_end, n_iter=150000):
    idx = np.argsort(h)[::-1]
    cur = np.zeros(n)
    cur[idx[:n // 2]] = 1.0
    best_h = cur.copy()
    best_c5 = compute_c5(cur)
    cur_c5 = best_c5
    T_start, T_end_v = 0.04, 0.0005
    t0 = time.time()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        T = T_start * (T_end_v / T_start) ** (it / float(n_iter))
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        cur[i], cur[j] = cur[j], cur[i]
        nc5 = compute_c5(cur)
        if nc5 < cur_c5 or np.random.rand() < np.exp(-((nc5 - cur_c5) / T)):
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = cur.copy()
        else:
            cur[i], cur[j] = cur[j], cur[i]
    return best_h, best_c5

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def spread_binary(n):
    h = np.zeros(n)
    h[::2] = 1.0
    if n % 2 == 1:
        h[-1] = 0.0
        extra = np.sum(h) < n // 2
        if extra:
            h[-1] = 1.0
    return h

def smooth_interpolate(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return proj(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Initialize candidates
    candidates = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = proj(h0, n0)
        candidates.append(h0)
    candidates.append(random_binary(160, rng))
    candidates.append(spread_binary(160))
    candidates.append(random_binary(180, rng))
    candidates.append(spread_binary(180))

    best_h = candidates[0]
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary CD on each candidate
    for ci in range(len(candidates)):
        if elapsed() > budget_s * 0.12:
            break
        h_c = candidates[ci]
        n_c = len(h_c)
        h_c, c5_c = coord_cd_binary(h_c, n_c, t0 + budget_s * 0.05, max_sweeps=3000)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 2: Global binary CD on best
    if elapsed() < budget_s * 0.25:
        n_best = len(best_h)
        best_h, best_c5 = coord_cd_binary(best_h, n_best, t0 + budget_s * 0.25, max_sweeps=5000)

    # Phase 3: Binary SA
    if elapsed() < budget_s * 0.40:
        n_best = len(best_h)
        h_sa, c5_sa = binary_sa(best_h, n_best, t0 + budget_s * 0.40, n_iter=200000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        # CD after SA
        if elapsed() < budget_s * 0.50:
            best_h, best_c5 = coord_cd_binary(best_h, n_best, t0 + budget_s * 0.50, max_sweeps=3000)

    # Phase 4: Continuous CD (unbinary)
    if elapsed() < budget_s * 0.62:
        n_best = len(best_h)
        best_h, best_c5 = coord_cd_cont(best_h, n_best, t0 + budget_s * 0.62, max_sweeps=60)
        # Binary CD again after unbinary
        if elapsed() < budget_s * 0.70:
            best_h, best_c5 = coord_cd_binary(best_h, n_best, t0 + budget_s * 0.70, max_sweeps=3000)

    # Phase 5: Multi-resolution
    for n_try in [120, 144, 200, 240]:
        if elapsed() > budget_s * 0.82:
            break
        h_try = smooth_interpolate(best_h, n_try)
        h_try, c5_try = coord_cd_binary(h_try, n_try, t0 + budget_s * 0.08, max_sweeps=2000)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        if elapsed() > budget_s * 0.85:
            break

    # Phase 6: Final polish - alternating binary and continuous CD
    n_best = len(best_h)
    for phase in range(4):
        if elapsed() > budget_s * 0.93:
            break
        h_b, c5_b = coord_cd_binary(best_h, n_best, t0 + budget_s * 0.03, max_sweeps=2000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
        h_c, c5_c = coord_cd_cont(best_h, n_best, t0 + budget_s * 0.03, max_sweeps=20)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c

    # Final projection and evaluation
    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
