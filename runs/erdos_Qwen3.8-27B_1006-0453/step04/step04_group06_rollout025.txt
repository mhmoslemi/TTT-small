import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0, 1).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def coord_opt(h, n, i):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    old = h[i]
    shifts = np.arange(2 * n - 1) - center
    a = np.zeros(2 * n - 1)
    ip = i - shifts
    v1 = (ip >= 0) & (ip < n)
    if v1.any():
        a[v1] += 1.0 - h[ip[v1]]
    im = i + shifts
    v2 = (im >= 0) & (im < n)
    if v2.any():
        a[v2] -= h[im[v2]]
    a[center] = 0.0
    
    def f1d(x):
        d = x - old
        vals = corr + d * a
        vals[center] = corr[center] + d * (1.0 - x - old)
        return np.max(vals) * dx
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    T = sa_T
    while time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            x_new = coord_opt(best_h, n, i)
            best_h[i] = x_new
            best_h = project(best_h, n)
            c5_new = compute_c5(best_h)

                        best_c5 = compute_c5(best_h)
        if not improved and T < 1e-10:

        T *= 0.999

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0, 1).sum() < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def coord_opt(h, n, i):
    dx = 2.0 / n
    center = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    old = h[i]
    shifts = np.arange(2 * n - 1) - center
    a = np.zeros(2 * n - 1)
    ip = i - shifts
    v1 = (ip >= 0) & (ip < n)
    if v1.any():
        a[v1] += 1.0 - h[ip[v1]]
    im = i + shifts
    v2 = (im >= 0) & (im < n)
    if v2.any():
        a[v2] -= h[im[v2]]
    a[center] = 0.0
    corr_c = corr[center]
    a_flat = a.copy()
    def f1d(x):
        d = x - old
        vals = corr + d * a_flat
        vals[center] = corr_c + d * (1.0 - x - old)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, t_end, sa_T=0.0, no_improve_limit=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T = sa_T
    no_improve = 0
    while time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            old_val = best_h[i]
            x_new = coord_opt(best_h, n, i)
            best_h[i] = x_new
            c5_new = compute_c5(best_h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
            elif c5_new > best_c5 + 1e-14:
                if T > 1e-10 and np.random.random() < np.exp(-(c5_new - best_c5) / T):
                    best_c5 = c5_new
                    improved = True
                else:
                    best_h[i] = old_val
        best_h = project(best_h, n)
        c5_check = compute_c5(best_h)
        if c5_check < best_c5:
            best_c5 = c5_check
        if not improved:
            no_improve += 1
            if no_improve > no_improve_limit and T < 1e-8:
                break
            if no_improve > no_improve_limit:
                T = max(T * 0.9, best_c5 * 0.002)
                no_improve = 0
        else:
            no_improve = 0
        T *= 0.997
    return best_h, best_c5

def fourier_refit(h, n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    K = 5
    A = np.ones((n, 2 * K + 1))
    for k in range(1, K + 1):
        A[:, 2 * k] = np.cos(2.0 * np.pi * k * x)
        A[:, 2 * k + 1] = np.sin(2.0 * np.pi * k * x)
    coeffs, _, _, _ = np.linalg.lstsq(A, h, rcond=None)
    def f_obj(c):
        val = np.full(n, c[0])
        for k in range(1, K + 1):
            val += c[2 * k] * np.cos(2.0 * np.pi * k * x) + c[2 * k + 1] * np.sin(2.0 * np.pi * k * x)
        return val
    def obj(c):
        return compute_c5(project(f_obj(c), n))
    res = minimize(obj, coeffs, method='Nelder-Mead',
                   options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-12})
    h_out = project(f_obj(res.x), n)
    return h_out, compute_c5(h_out)
