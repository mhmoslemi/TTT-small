import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14: return h
        h += room * (diff / tot)
    else:
        if s < 1e-14: return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_topk(corr, T, k=3):
    """Soft top-k: sum of top k with softmax smoothing."""
    sorted_c = np.sort(corr)[::-1]
    # Use T to smooth the max, take top k
    top = sorted_c[:k]
    shifted = T * (top - top[0])
    shifted = np.clip(shifted, -100, 100)
    weights = np.exp(shifted)
    weights /= weights.sum()
    return np.dot(weights, top)

def grad_topk(h, T, k=3):
    """Gradient of soft top-k of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    # Get top k indices
    top_idx = np.argsort(corr)[::-1][:k]
    top_vals = corr[top_idx]
    shifted = T * (top_vals - top_vals[0])
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        if wi < 1e-10: continue
        k_idx = idx
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            # corr[k] = sum_j h[j]*(1-h[j+lag]) for valid j
            # d/dh[i]: if i+lag < n: (1-h[i+lag]); if i-lag >= 0: -h[i-lag]
            if n - lag > 0:
                grad[:n-lag] += wi * (1.0 - h[lag:n])
                grad[lag:] += wi * (-h[:n-lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += wi * (1.0 - h[:n-nl])
                grad[:n-nl] += wi * (-h[nl:])
    return grad * dx

def coord_sweep(h, n, order=None, t_end=None, tol=1e-14):
    """One full coordinate descent sweep."""
    if order is None:
        order = np.arange(n)
    best_c5 = compute_c5(h)
    improved = 0
    for i in order:
        if t_end is not None and time.time() > t_end:
            break
        v = h[i]
        h[i] = 0.0
        corr_base = np.correlate(h, 1.0 - h, mode='full')
        h[i] = v
        
        dx = 2.0 / n
        coeff = np.zeros(2*n - 1)
        k0 = n - 1
        for k in range(2*n - 1):
            lag = k - k0
            c = 0.0
            if lag == 0:
                pass  # handled separately
            elif lag > 0:
                if i + lag < n: c += (1.0 - h[i+lag])
                if i - lag >= 0: c += (-h[i-lag])
            else:
                nl = -lag
                if i - nl >= 0: c += (1.0 - h[i-nl])
                if i + nl < n: c += (-h[i+nl])
            coeff[k] = c
        
        def f1d(t):
            cv = corr_base + t * coeff
            cv[k0] = corr_base[k0] + t * (1.0 - t)
            return float(np.max(cv))
        
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11, 'maxiter': 50})
        h_new = res.x
        if abs(h_new - v) > 1e-12:
            h[i] = h_new
            h = proj(h, n)
            c5_new = compute_c5(h)
            if c5_new < best_c5 - tol:
                best_c5 = c5_new
                improved += 1
    return h, best_c5, improved

Let me be more specific about the subgradient approach:

The smoothed objective: f_T(h) = (max(corr) + log(sum(exp(T*(corr - max))))/T) * dx

Its gradient is: (1/dx_weighted) * sum_k p_k * grad_corr_k where p_k = exp(T*(corr_k - max))/sum(exp(...))

This is what the previous code does. The issue is that for large T, the gradient becomes very peaked and small steps are needed.

Alternative: use a "top-k average" smoothed max:
f_k(h) = (1/k) * sum of top k values of corr

This is piecewise linear (for k=1, it's the true max; for k>1, it's smoother). Its subgradient is the average of the top-k gradients.

Actually, for the purpose of finding a good local minimum, coordinate descent is the most reliable. Let me focus on making it faster and more effective.

**Speed optimization for coordinate descent:**
- Precompute the correlation once per sweep
- For each coordinate, update the correlation incrementally (O(n) instead of O(n²))

When we change h[i] from v to t (delta = t - v):
- corr[k] changes by delta * coeff[k] for each k
- We can update corr in O(n) per coordinate change

This makes each coordinate update O(n) + O(log n) for the 1D minimization (if we use a simple approach) or O(n) + O(n) for minimize_scalar.

Total per sweep: O(n²) for initial corr + O(n²) for updates = O(n²) per sweep.
For n=180: ~32400 per sweep, very fast.

def coord_sweep_fast(h, n, t_end=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k0 = n - 1
    best_c5_corr = corr.max()  # track in corr units, multiply by dx at end
    improved = 0
    
    for i in range(n):
        if t_end and time.time() > t_end:
            break
        v = h[i]
        # Compute coeff: how each corr[k] changes per unit change in h[i]
        # (relative to current h[i] = v)
        coeff = np.zeros(2*n-1)
        for k in range(2*n-1):
            lag = k - k0
            c = 0.0
            if lag == 0:
                pass  # quadratic, handle separately
            elif lag > 0:
                if i < n - lag: c += (1.0 - h[i+lag])
                if i >= lag: c += (-h[i-lag])
            else:
                nl = -lag
                if i >= nl: c += (1.0 - h[i-nl])
                if i + nl < n: c += (-h[i+nl])
            coeff[k] = c
        
        # Minimize max(corr[k] + (t-v)*coeff[k] for k≠k0, corr[k0] + (t-v)*(1-t) for k=k0)
        # Actually at k0: corr changes by v*(1-v) -> t*(1-t), so delta = t(1-t) - v(1-v) = (t-v)(1-v-t)
        # For k≠k0: delta = (t-v)*coeff[k]
        
        # Define f(t) = max(corr + (t-v)*coeff, with k0 special)
        # This is max of linear functions and one concave quadratic
        # Use golden section / ternary search on [0,1]
        
        def f1d(t):
            # corr[k] + (t-v)*coeff[k] for k≠k0
            # corr[k0] + t*(1-t) - v*(1-v) for k=k0
            vals = corr + (t-v) * coeff
            vals[k0] = corr[k0] + t*(1-t) - v*(1-v)
            return vals.max()
        
        # Use minimize_scalar
        res = minimize_scalar(f1d, bounds=(0, 1), method='bounded', options={'xatol': 1e-10, 'maxiter': 40})
        t_new = res.x
        
        if abs(t_new - v) > 1e-10:
            delta = t_new - v
            h[i] = t_new
            # Update corr incrementally
            corr[k0] += t_new*(1-t_new) - v*(1-v)
            for k in range(2*n-1):
                if k != k0:
                    corr[k] += delta * coeff[k]
            
            # Project (maintain sum constraint)
            h = proj(h, n)
            # Re-compute corr after projection (projection might change other coordinates)
            # This is the expensive part... 
            # Alternative: don't project, just track the sum and fix at end?
            # Actually, projection changes sum to n/2, which requires changing all coords potentially.
            # Better: just do the coordinate change and fix sum at end of sweep.
            
            if f1d(t_new) < best_c5_corr:
                best_c5_corr = f1d(t_new)
                improved += 1
    
    h = proj(h, n)
    # Recompute corr after projection
    corr = np.correlate(h, 1.0 - h, mode='full')
    return h, corr, improved

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def proj(h, n):
    h = np.clip(h, 0, 1).copy()
    s = h.sum()
    d = n / 2.0 - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t > 1e-14:
            h += r * (d / t)
    else:
        if s > 1e-14:
            h *= (1.0 + d / s)
    return np.clip(h, 0, 1)

def coef_idx(h, i, n):
    k0 = n - 1
    c = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - k0
        val = 0.0
        if lag > 0:
            if i < n - lag: val += 1.0 - h[i + lag]
            if i >= lag: val -= h[i - lag]
        elif lag < 0:
            nl = -lag
            if i >= nl: val += 1.0 - h[i - nl]
            if i + nl < n: val -= h[i + nl]
        c[k] = val
    return c

def coord_sweeps(h, n, n_sweeps, t_end=None):
    dx = 2.0 / n
    k0 = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    for s in range(n_sweeps):
        order = np.random.permutation(n)
        for i in order:
            if t_end is not None and time.time() > t_end:
                break
            v = h[i]
            cf = coef_idx(h, i, n)
            def f1d(t):
                vals = corr + (t - v) * cf
                vals[k0] = corr[k0] + t * (1.0 - t) - v * (1.0 - v)
                return vals.max()
            res = minimize_scalar(f1d, bounds=(0, 1), method='bounded',
                                  options={'xatol': 1e-11, 'maxiter': 50})
            tn = res.x
            if abs(tn - v) > 1e-12:
                h[i] = tn
                corr[k0] += tn * (1.0 - tn) - v * (1.0 - v)
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k0] = False
                corr[mask] += (tn - v) * cf[mask]
        h = proj(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h)

def subgrad_step(h, n, T, k_top, step_size):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:k_top]
    top_vals = corr[top_idx]
    w = np.exp(T * (top_vals - top_vals[0]))
    w /= w.sum()
    k0 = n - 1
    grad = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        if wi < 1e-10:
            continue
        lag = idx - k0
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wi * (1.0 - h[lag:])
            grad[lag:] += wi * (-h[:n - lag])
        else:
            nl = -lag
            grad[nl:] += wi * (1.0 - h[:n - nl])
            grad[:n - nl] += wi * (-h[nl:])
    grad *= dx
    h_new = proj(h - step_size * grad, n)
    return h_new

def block_perturb(h, n, intensity):
    L = max(3, int(n * 0.05))

    h_new[i:i + L] = 1.0 - h_new[i:i + L]
