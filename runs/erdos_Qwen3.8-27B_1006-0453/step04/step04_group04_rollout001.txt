    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def slsqp_smooth(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def cons_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': cons_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def paired_move(h, n, L, i, alpha):
    """Apply paired move: h[i]+=alpha, h[i+L]-=alpha (if L>0) or h[i-L]-=alpha (if L<0)."""
    h = h.copy()
    j = i + L
    if 0 <= i < n and 0 <= j < n:
        h[i] += alpha
        h[j] -= alpha
    return h

def paired_moves_sweep(h, n, t_end, K=8):
    """Sweep over active lags doing paired moves."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    for sweep in range(500):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = np.max(corr)
        # Get top K lags
        top_lags = np.argsort(corr)[-K:][::-1]
        improved = False
        for lag_idx in top_lags:
            L = lag_idx - (n - 1)
            if L == 0:
                continue
            if L > 0:
                i_range = range(0, n - L)
            else:
                i_range = range(0, n + L)
            for i in i_range:
                if time.time() > t_end:
                    return best_h, best_c5
                j = i + L
                # Optimal alpha for quadratic
                a = best_h[i] - best_h[j]
                # Also check neighbors for other lags effect
                a1 = 0.0
                if 0 <= i - L < n:
                    a1 += best_h[i - L]
                if 0 <= j + L < n:
                    a1 -= best_h[j + L]
                # alpha* = -(a + a1) / 2, but also try simple gradient direction
                alpha_opt = -(a + a1) / 2.0
                # Clamp to valid range
                alpha_max = min(1.0 - best_h[i], best_h[j])
                alpha_min = max(-best_h[i], 1.0 - best_h[j])
                alpha_opt = np.clip(alpha_opt, alpha_min, alpha_max)
                if abs(alpha_opt) < 1e-10:
                    continue
                # Try the move
                h_try = best_h.copy()
                h_try[i] += alpha_opt
                h_try[j] -= alpha_opt
                h_try = proj(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def coord_descent_fast(h, n, t_end, max_sweeps=200):
    """Fast coordinate descent: for each i, find best h[i] given others fixed."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hz = best_h.copy()
            hz[i] = 0.0
            corr_base = np.correlate(hz, 1.0 - hz, mode='full')
            # Coefficient: how c_k changes with h[i]
            coef = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                lag = k - (n - 1)
                j1 = i + lag
                j2 = i - lag
                c = 0.0
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c += (-best_h[j2])
                coef[k] = c
            k0 = n - 1
            # f(t) = max(corr_base + t*coef, but at k0: corr_base[k0] + t*(1-t))
            def f1d(t):
                cv = corr_base + t * coef
                cv[k0] = corr_base[k0] + t * (1.0 - t)
                return float(np.max(cv)) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            h_try = best_h.copy()
            h_try[i] = res.x
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_h = h_try
                best_c5 = c5_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def focused_active(h, n, maxiter=2000):
    """SLSQP on just the active set of lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val * 0.9999)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    def obj_f(x):
        c = np.correlate(x, 1.0 - x, mode='full')[active]
        return float(np.max(c)) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_f, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return proj(np.clip(res.x, 0, 1), n)

def grad_smooth_c5(h, T):
    """Gradient of smooth C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n - lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += pk * (-h[nl:])
    return grad * dx

def grad_descent(h, n, t_end, lr_init=0.005, T_init=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    T = T_init
    for it in range(5000):
        if time.time() > t_end:
            break
        g = grad_smooth_c5(best_h, T)
        step = lr
        for _ in range(20):
            trial = proj(best_h - step * g, n)
            c5_t = compute_c5(trial)
            if c5_t < best_c5:
                best_h = trial
                best_c5 = c5_t
                T = min(T * 1.04, 20000)
                break
            step *= 0.4
        else:
            lr *= 0.5
            if lr < 1e-11:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        n = len(best_h)
    else:
        n = 180
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0

    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T (10% budget)
    t1 = t0 + budget_s * 0.10
    for T in [50, 200, 800, 3000, 10000, 30000]:
        if time.time() > t1:
            break
        try:
            hc = slsqp_smooth(best_h, T, n, maxiter=600, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5 - 1e-12:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Gradient descent (10% budget)
    t2 = t0 + budget_s * 0.20
    if time.time() < t2:
        best_h, best_c5 = grad_descent(best_h, n, t2, lr_init=0.004, T_init=800)

    # Phase 3: Paired moves (25% budget) - KEY INNOVATION
    t3 = t0 + budget_s * 0.45
    if time.time() < t3:
        best_h, best_c5 = paired_moves_sweep(best_h, n, t3, K=10)

    # Phase 4: Coordinate descent (15% budget)
    t4 = t0 + budget_s * 0.60
    if time.time() < t4:
        best_h, best_c5 = coord_descent_fast(best_h, n, t4, max_sweeps=300)

    # Phase 5: Basin hopping (20% budget)
    t5 = t0 + budget_s * 0.80
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 80
    for it in range(n_bh):
        if time.time() > t5:
            break
        cool = 1.0 - 0.7 * (it / float(n_bh))
        step = 0.01 + 0.08 * cool
        h_pert = work_h + step * np.random.randn(len(work_h))
        h_pert = proj(h_pert, len(work_h))
        T_bh = 1000
        try:
            hc = slsqp_smooth(h_pert, T_bh, len(work_h), maxiter=150, ftol=1e-12)
            c5v = compute_c5(hc)
            temp = 0.002 * cool + 0.0001
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: More paired moves + coord descent (10% budget)
    t6 = t0 + budget_s * 0.92
    if time.time() < t6:
        best_h, best_c5 = paired_moves_sweep(best_h, len(best_h), t0 + budget_s * 0.85, K=12)
        if time.time() < t6:
            best_h, best_c5 = coord_descent_fast(best_h, len(best_h), t6, max_sweeps=100)

    # Phase 7: Final SLSQP polish
    for T in [50000, 100000, 200000]:
        if time.time() > t0 + budget_s * 0.97:
            break
        try:
            hc = slsqp_smooth(best_h, T, len(best_h), maxiter=800, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    try:
        hc = focused_active(best_h, len(best_h), maxiter=1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    except Exception:
        pass

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
