import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):

    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0

def fast_local_search_binary(h, t_end, max_rounds=2000):
    """Local search: try all swaps, take best improving one per round"""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_max = np.max(corr)
    best_c5 = best_max * dx
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]
    
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        # Try a random subset of swaps
        n_tries = min(len(ones) * len(zeros), 500)
        for _ in range(n_tries):
            a = ones[np.random.randint(len(ones))]
            b = zeros[np.random.randint(len(zeros))]
            if a == b:
                continue
            # Swap
            best_h[a] = 0.0
            best_h[b] = 1.0
            new_corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            new_max = np.max(new_corr)

                corr = new_corr

                improved = True
            else:
                best_h[a] = 1.0
                best_h[b] = 0.0
        if not improved:
            break
    return best_h, best_c5

def simulated_annealing_binary(h, t_end, t_start=50, t_final=0.01, alpha=0.999):
    """SA on binary sequences with pair swap moves"""
    n = len(h)
    dx = 2.0 / n
    cur_h = h.copy()
    cur_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
    cur_max = np.max(cur_corr)
    best_h = cur_h.copy()
    best_max = cur_max
    ones = np.where(cur_h > 0.5)[0]
    zeros = np.where(cur_h <= 0.5)[0]
    temp = t_start
    
    while time.time() < t_end and temp > t_final:
        temp *= alpha
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        if a == b:
            continue
        cur_h[a] = 0.0
        cur_h[b] = 1.0
        new_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        new_max = np.max(new_corr)
        delta = new_max - cur_max
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            cur_max = new_max
            cur_corr = new_corr
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h <= 0.5)[0]
            if cur_max < best_max:
                best_max = cur_max
                best_h = cur_h.copy()
        else:
            cur_h[a] = 1.0
            cur_h[b] = 0.0
    
    return best_h, best_max * dx

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)

    return project(h_new, n_new)

def coord_descent_fast(h, n, t_end, max_sweeps=500):
    """Fast coordinate descent for continuous h"""
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n-1), n)
    c0_idx = n - 1
    
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            
            # The objective is max of (corr + (v-old)*coeff) with special c0 term
            # c0 changes by (v-old)*(1-v-old)
            # For a quick 1D minimization:
            # Try 0, 1, and midpoint
            candidates = [0.0, 1.0]
            # Also try where the max peaks
            cn0 = corr + (0 - old) * coeff
            cn0[c0_idx] = c0_base + (0 - old) * (1.0 - old)
            cn1 = corr + (1 - old) * coeff
            cn1[c0_idx] = c0_base + (1 - old) * (1.0 - 1)  # = c0_base
            # Actually (1-v-old) when v=1: (1-1-old) = -old
            cn1[c0_idx] = c0_base + (1 - old) * (-old)
            
            f0 = np.max(cn0)
            f1 = np.max(cn1)
            
            # If current is best, skip
            cn_cur = corr + (old - old) * coeff  # = corr
            cn_cur[c0_idx] = c0_base  # no change
            f_cur = np.max(cn_cur)
            
            best_f = min(f0, f1, f_cur)
            if best_f >= f_cur - 1e-14:
                continue
            if best_f == f0:
                best_v = 0.0
            elif best_f == f1:
                best_v = 1.0
            else:
                continue
            
            best_h[i] = best_v
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_c5 = np.max(corr) * dx
            improved = True
        
        if not improved:
            break
    return best_h, best_c5

def project(h, n):

    target = n / 2.0

        tot = room.sum()

        h = h + room * (diff / tot)
    else:

            return h
        h = h * (1.0 + diff / s)

def coord_opt(x, corr, coeff, c0_idx, old):
    """Find v in [0,1] minimizing max(corr + (v-x)*coeff, with c0 correction)"""
    # For coordinate x:

    # corr_new[c0_idx] = corr[c0_idx] + (v-x)*(1-v-x)

        cn[c0_idx] = corr[c0_idx] + (v - x) * (1.0 - v - x)

    v1 = f(1.0)

    if v1 < best_f:

    if vc < best_f:

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s < 1e-14:
            return h
        h = np.clip(h * (1.0 + diff / s), 0, 1)
    return h

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    c0 = n - 1
    diag_mask = np.zeros(2 * n - 1)
    diag_mask[c0] = 1.0
    m_range = np.arange(2 * n - 1)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = h[i]
            idx = i + m_range - (n - 1)
            valid = (idx >= 0) & (idx < n)
            coeff = np.zeros(2 * n - 1)
            coeff[valid] = 1.0 - 2.0 * h[idx[valid]]
            def obj_v(v):
                cn = corr + (v - old) * coeff
                cn[c0] -= (v - old) ** 2
                return np.max(cn)
            f0 = obj_v(0.0)
            f1 = obj_v(1.0)
            fc = np.max(corr)
            best_f = min(f0, f1, fc)
            if best_f >= fc - 1e-13:
                continue
            res = minimize_scalar(obj_v, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            if res.fun < best_f:
                best_f = res.fun
            if best_f * dx < best_c5 - 1e-15:
                h[i] = res.x
                corr = np.correlate(h, 1.0 - h, mode='full')
                best_c5 = np.max(corr) * dx
                improved = True
        if not improved:
            break
    return h, best_c5

def perturbation_search(h, n, t_end, max_iter=500):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        phase = it / max_iter
        sigma = 0.05 + 0.45 * (1.0 - phase)
        n_pert = max(1, int(3.0 * (1.0 - phase)))
        h_p = best_h.copy()
        pidx = np.random.choice(n, size=n_pert, replace=False)
        h_p[pidx] += sigma * np.random.randn(n_pert)
        h_p = project(h_p, n)
        h_c, c5_c = coord_descent(h_p, n, t_end, max_sweeps=80)
        if c5_c < best_c5 - 1e-14:
            best_c5, best_h = c5_c, h_c
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h), n_new)

def make_structured(n, kind, rng):
    if kind == 0:
        h = np.zeros(n)
        h[::2] = 1.0
        return project(h, n)
    elif kind == 1:
        h = np.zeros(n)
        h[:n//2] = 1.0
        return project(h, n)
    elif kind == 2:
        x = np.linspace(0, 2, n, endpoint=False)
        h = np.clip(0.5 + 0.5 * np.cos(2 * np.pi * rng.randint(1, 6) * x), 0, 1)
        return project(h, n)
    elif kind == 3:
        h = (rng.rand(n) > 0.5).astype(float)
        idx = np.argsort(-h)
        hb = np.zeros(n)
        hb[idx[:n//2]] = 1.0
        return hb
    else:
        x = np.linspace(0, 2, n, endpoint=False)
        freqs = rng.choice([1, 2, 3, 4, 5, 6], size=3)
        phases = rng.uniform(0, 2*np.pi, size=3)
        h = 0.5
        for f, p in zip(freqs, phases):
            h += 0.3 * np.cos(2 * np.pi * f * x + p)
        return project(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[::2] = 1.0
    n = len(best_h)
    if n % 2 != 0:
        n = n - 1
        best_h = best_h[:n]
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: thorough coordinate descent from initial
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.12, max_sweeps=300)

    # Phase 2: perturbation search from best
    best_h, best_c5 = perturbation_search(best_h, n, t0 + budget_s * 0.45, max_iter=300)

    # Phase 3: structured starts at n
    for kind in range(5):
        if elapsed() > budget_s * 0.60:
            break
        h_s = make_structured(n, kind, rng)
        h_c, c5_c = coord_descent(h_s, n, t0 + budget_s * (0.60 + 0.03 * (kind + 1)), max_sweeps=150)
        if c5_c < best_c5 - 1e-14:
            best_c5, best_h = c5_c, h_c

    # Phase 4: different resolutions
    for n_try in [80, 100, 120, 200, 240]:
        if elapsed() > budget_s * 0.85:
            break
        h_try = interp_h(best_h, n_try)
        h_c, c5_c = coord_descent(h_try, n_try, t0 + budget_s * min(0.9 + 0.01 * n_try // 50, 0.92), max_sweeps=200)
        if c5_c < best_c5 - 1e-14:
            best_c5, best_h, n = c5_c, h_c, n_try

    # Phase 5: final perturbation + CD polish
    if elapsed() < budget_s * 0.92:
        best_h, best_c5 = perturbation_search(best_h, n, t0 + budget_s * 0.96, max_iter=150)

    # Final
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
