import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_c5_after_swap(h, i, j, n):
    """Compute C5 after swapping elements i and j."""
    if h[i] == h[j]:
        return compute_c5(h)
    a, b = h[i], h[j]
    # Change in h: Δh[i] = b-a, Δh[j] = a-b
    # Instead of recomputing full correlation, just swap and compute
    h[j], h[i] = h[i], h[j]  # swap
    c5 = compute_c5(h)
    h[j], h[i] = h[i], h[j]  # unswap
    return c5

def swap_local_search(h, n, t_end, max_sweeps=5000):
    """Local search using pair swaps."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        best_swap_c5 = best_c5
        best_i, best_j = -1, -1
        for i in range(n):
            for j in range(i+1, n):
                if abs(best_h[i] - best_h[j]) < 1e-10:
                    continue
                c5 = fast_c5_after_swap(best_h, i, j, n)
                if c5 < best_swap_c5 - 1e-12:
                    best_swap_c5 = c5
                    best_i, best_j = i, j
        if best_i >= 0:
            best_h[best_i], best_h[best_j] = best_h[best_j], best_h[best_i]
            best_c5 = best_swap_c5
            improved = True
        if not improved:
            break
    return best_h, best_c5

            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            def ex_obj(v):
                nc = corr + (v - old) * coeff
                nc[c0_idx] -= (v - old) ** 2
                return np.max(nc)
            res = minimize_scalar(ex_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            bv = res.x
            if ex_obj(0.0) < ex_obj(bv):
                bv = 0.0
            if ex_obj(1.0) < ex_obj(bv):
                bv = 1.0
            if abs(bv - old) < 1e-12:
                continue
            best_h[i] = bv
            nc = np.correlate(best_h, 1.0 - best_h, mode='full')
            nc5 = np.max(nc) * dx
            if nc5 < best_c5 - 1e-14:
                corr = nc
                best_c5 = nc5
                improved = True
            else:
                best_h[i] = old
        if not improved:
            break
    return best_h, best_c5

def sa_swaps(h, n, t_end):
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    t_start = time.time()
    dur = max(t_end - t_start, 1.0)
    while time.time() < t_end:
        frac = min(1.0, (time.time() - t_start) / dur)
        T = 5e-4 * (0.01) ** frac + 5e-5
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or abs(cur_h[i] - cur_h[j]) < 1e-10:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(cur_h)
        d = new_c5 - cur_c5
        if d < 0 or np.random.rand() < np.exp(-d / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
    return best_h, best_c5

def cycle3_local_search(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while time.time() < t_end and best_c5 > 0.380:
        improved = False
        for _ in range(max_sweeps):
            if time.time() > t_end:
                return best_h, best_c5
            best_imp = best_c5
            best_abc = None
            for _trial in range(300):
                abc = np.random.choice(n, 3, replace=False)
                if len(set(best_h[abc].round(6))) < 2:
                    continue
                best_h[abc[0]], best_h[abc[1]], best_h[abc[2]] = (
                    best_h[abc[2]], best_h[abc[0]], best_h[abc[1]])
                c5 = compute_c5(best_h)
                if c5 < best_imp - 1e-14:
                    best_imp = c5
                    best_abc = abc.copy()
                else:
                    best_h[abc[0]], best_h[abc[1]], best_h[abc[2]] = (
                        best_h[abc[2]], best_h[abc[0]], best_h[abc[1]])
            if best_abc is not None:
                a, b, c = best_abc
                best_h[a], best_h[b], best_h[c] = best_h[c], best_h[a], best_h[b]
                best_c5 = best_imp
                improved = True
        if not improved:
            break
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def slsqp_polish(h, n, maxiter=1000):
    dx = 2.0 / n
    T = 8000.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        best_h = np.zeros(n0)
        best_h[np.random.choice(n0, n0 // 2, replace=False)] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def rem(frac):
        return t0 + budget_s * frac

    # Phase 1: Intensive swap local search
    best_h, best_c5 = swap_local_search(best_h, n, rem(0.18), max_sweeps=8000)

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, rem(0.32), max_sweeps=2000)

    # Phase 3: Swap local search again
    best_h, best_c5 = swap_local_search(best_h, n, rem(0.40), max_sweeps=5000)

    # Phase 4: Simulated annealing
    best_h, best_c5 = sa_swaps(best_h, n, rem(0.55))

    # Phase 5: Post-SA refinement
    best_h, best_c5 = swap_local_search(best_h, n, rem(0.62), max_sweeps=5000)
    best_h, best_c5 = coord_descent(best_h, n, rem(0.70), max_sweeps=2000)

    # Phase 6: Cycle-3 moves for extra exploration
    best_h, best_c5 = cycle3_local_search(best_h, n, rem(0.76), max_sweeps=3000)

    # Phase 7: Multi-resolution check
    for n_try in [140, 180, 200]:
        if time.time() > rem(0.80):
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = swap_local_search(h_tr, n_try, rem(0.81), max_sweeps=1000)
        h_tr, c5_tr = coord_descent(h_tr, n_try, rem(0.82), max_sweeps=300)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 8: Final intensive search
    best_h, best_c5 = swap_local_search(best_h, n, rem(0.88), max_sweeps=8000)
    best_h, best_c5 = coord_descent(best_h, n, rem(0.93), max_sweeps=3000)

    # Phase 9: SLSQP polish
    if time.time() < rem(0.97):
        hc = slsqp_polish(best_h, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
        best_h, best_c5 = coord_descent(best_h, n, rem(0.99), max_sweeps=500)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
