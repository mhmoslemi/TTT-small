def smooth_c5_grad(h, T, active_k=None):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if active_k is not None:

        k = k_indices[idx]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_obj(h, T, active_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cs = corr[active_k] if active_k is not None else corr
    m = np.max(cs)
    exps = np.exp(T * (cs - m))
    return (m + np.log(exps.sum()) / T) * dx

def smooth_c5_grad(h, T, active_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if active_k is not None:

        k_idx = active_k
    else:

    tot = exps.sum()
    wts = exps / tot
    grad = np.zeros(n)
    thresh = max(wts.max() * 1e-5, 1e-14)
    for idx in np.where(wts > thresh)[0]:
        w = wts[idx]
        s = k_idx[idx] - (n - 1)
        j1a, j1b = max(0, -s), min(n, n - s)

        j2a, j2b = max(0, s), min(n, n + s)
        if j2b > j2a:
            j2 = np.arange(j2a, j2b)
            grad[j2] -= w * h[j2 - s]
    grad *= dx
    return grad

        return smooth_c5_grad(x, T, active_k)
    def con(x):
        return x.sum() - n / 2.0
    def conjac(x):
        return np.ones(n)
    try:

                       bounds=[(0.0, 1.0)] * n,

                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h.copy()

        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 3: Iterative active-set SLSQP
    for rd in range(12):
        if elapsed() > budget_s * 0.55:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        K = min(12 + 4 * rd, 30)
        top_k = np.argsort(corr)[-K:]
        T_act = 100 + 3000 * (1 - rd / 12.0)
        hc = slsqp_g(best_h, T_act, n, active_k=top_k, maxiter=350, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-15:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.75, max_sweeps=400)

    # Phase 5: Second round of basin hopping (finer)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(20):
        if elapsed() > budget_s * 0.85:
            break
        cool = 1.0 - it / 20.0
        step = 0.015 + 0.05 * cool
        pert = step * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T_bh = 500 + 6000 * cool
        hc = slsqp_g(h_pert, T_bh, n, maxiter=300, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: Final SLSQP polish with high T
    for T in [100, 1000, 10000, 100000]:
        if elapsed() > budget_s * 0.93:
            break
        hc = slsqp_g(best_h, T, n, maxiter=500, ftol=1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-15:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Final coordinate descent
    if elapsed() < budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.985, max_sweeps=200)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
