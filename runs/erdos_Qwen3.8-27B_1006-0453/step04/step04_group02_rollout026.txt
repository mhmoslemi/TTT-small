    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-12:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-12:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-12 and s2 > 1e-12:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    Z = np.sum(exps)
    w = exps / Z
    obj = (m + np.log(Z) / T) * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        o = k - (n - 1)
        if o >= 0:
            if n > o:
                grad[o:n] += wk * (1.0 - h[:n-o])
            if n - o > 0:
                grad[:n-o] -= wk * h[o:]
        else:
            p = -o
            if n - p > 0:
                grad[:n-p] += wk * (1.0 - h[p:])
            if n > p:
                grad[p:n] -= wk * h[:n-p]
    grad *= dx
    return obj, grad

def slsqp_analytic(h, T, n, maxiter=3000, ftol=1e-16):
    def objf(x):
        return smooth_obj_grad(x, T)[0]
    def objfg(x):
        return smooth_obj_grad(x, T)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(objfg, h, jac=True, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    except Exception:
        return project(h, n)
    return project(np.clip(res.x, 0, 1), n)

        order = rng.permutation(n)

import numpy as np
from scipy.optimize import minimize
import time

_N = 1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-13:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-13:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-13 and s2 > 1e-13:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def _sum_con(x):
    return x.sum() - _N / 2.0

def _sum_con_jac(x):
    return np.ones(len(x))

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    Z = exps.sum()
    w = exps / Z
    obj = (m + np.log(Z) / T) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        o = k - (n - 1)
        if o >= 0:
            if n > o:
                grad[o:n] += wk * (1.0 - h[:n - o])
            if n - o > 0:
                grad[:n - o] -= wk * h[o:]
        else:
            p = -o
            if n - p > 0:
                grad[:n - p] += wk * (1.0 - h[p:])
            if n > p:
                grad[p:n] -= wk * h[:n - p]
    grad *= dx
    return obj, grad

def slsqp_solve(h, T, n, maxiter=3000, ftol=1e-17):
    global _N
    _N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    try:
        res = minimize(smooth_obj_grad, h, jac=True, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    except Exception:
        return project(h, n)
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=40, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_save = best_h[i]
            best_h[i] = 0.0
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
            best_h[i] = h_save
            d = np.zeros(2 * n - 1)
            for k in range(1, n):
                if 0 <= i - k < n:
                    d[n - 1 + k] += (1.0 - best_h[i - k])
                    d[n - 1 - k] += (-best_h[i - k])
                if 0 <= i + k < n:
                    d[n - 1 + k] += (-best_h[i + k])
                    d[n - 1 - k] += (1.0 - best_h[i + k])
            c0 = corr_base[n - 1]
            best_t = h_save
            best_val = np.inf
            for t in np.linspace(0.0, 1.0, 33):
                vals = corr_base + t * d
                vals[n - 1] = c0 + t * (1.0 - t)
                v = np.max(vals)
                if v < best_val:
                    best_val = v
                    best_t = t
            if 0 < best_t < 1:
                for t in np.linspace(max(0, best_t - 0.03), min(1, best_t + 0.03), 11):
                    vals = corr_base + t * d
                    vals[n - 1] = c0 + t * (1.0 - t)
                    v = np.max(vals)
                    if v < best_val:
                        best_val = v
                        best_t = t
            if abs(best_t - h_save) > 1e-11:
                best_h[i] = best_t
                best_h = project(best_h, n)
                cv = compute_c5(best_h)
                if cv < best_c5 - 1e-14:
                    best_c5 = cv
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def max_push_subgrad(h, n, t_end, max_iters=2000):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    lr = 0.002
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = np.max(corr)
        active = np.where(corr >= m - 1e-8)[0]
        grad = np.zeros(n)
        for k in active:
            o = k - (n - 1)
            if o >= 0:
                if n > o:
                    grad[o:n] += (1.0 - best_h[:n - o])
                if n - o > 0:
                    grad[:n - o] -= best_h[o:]
            else:
                p = -o
                if n - p > 0:
                    grad[:n - p] += (1.0 - best_h[p:])
                if n > p:
                    grad[p:n] -= best_h[:n - p]
        grad /= len(active)
        gnorm = np.max(np.abs(grad)) + 1e-12
        new_h = project(best_h - lr / gnorm * grad, n)
        cv = compute_c5(new_h)
        if cv < best_c5 - 1e-15:
            best_h = new_h
            best_c5 = cv
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.8
            if lr < 1e-6:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
