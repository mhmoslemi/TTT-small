2. Check if the quadratic constraint is violated at x*
3. If not, done. If yes, we need to find the intersection.

For the 2D LP min t s.t. t ≥ b_k + a_k x, 0 ≤ x ≤ 1:
The optimal solution is at a vertex of the feasible region. The vertices are:
- (x=0, t=max(b_k))
- (x=1, t=max(b_k + a_k))
- Intersection points of pairs of lines (x_ij, t_ij) where b_i + a_i x = b_j + a_j x

The intersection of lines i and j: x = (b_j - b_i)/(a_i - a_j) if a_i ≠ a_j.
t = b_i + a_i * x.

The optimal x is the one that minimizes t. This is equivalent to finding the minimum of the upper envelope of the lines.

def project(h, n):

    target = float(n) / 2.0
    s = h.sum()
    if s < 1e-14:
        h = np.full(n, 0.5)
        return h
    if abs(s - target) < 1e-10:

        if tot < 1e-12:
            h = h * (target / s)
    else:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def gs_min(f, lo, hi, iters=35):
    gr = 0.618033988749895
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f(c), f(d)
    for _ in range(iters):

            c = hi - gr * (hi - lo)
            fc = f(c)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-10:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return np.clip(h * (target / s), 0, 1)
        h = h + room * (diff / tot)
    else:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def gs_min(f, lo, hi, iters):
    gr = 0.618033988749895
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f(c), f(d)
    for _ in range(iters):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def opt_coord(h, corr, i, n, center):
    hi = h[i]
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip, im = i + k, i - k
        a[center + k] = ((1.0 - h[im]) if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
        a[center - k] = ((1.0 - h[ip]) if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
    b = corr - a * hi
    b0 = corr[center] - hi * (1.0 - hi)
    bc = b.copy()
    bc[center] = b0

    def f(x):
        v = bc + a * x
        v[center] = b0 + x - x * x
        return v.max()

    return gs_min(f, 0.0, 1.0, 28)

def cd_loop(h, n, t_end, max_sweeps, gs_iters=28):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    center = n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        changed = False
        for i in range(n):
            x_new = opt_coord(best_h, corr, i, n, center)
            if abs(x_new - best_h[i]) > 1e-10:
                best_h[i] = x_new
                changed = True
        best_h = project(best_h, n)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
    return best_h, best_c5, best_c5

def smooth_grad_step(h, n, T, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    w = np.exp(T * (corr - m))
    w /= w.sum()
    center = n - 1
    grad = np.zeros(n)
    for i in range(n):
        gi = 0.0
        for k in range(1, n):
            ip, im = i + k, i - k
            wk = w[center + k] + w[center - k]
            if wk < 1e-15:
                continue
            d_pos = ((1.0 - h[im]) if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
            d_neg = ((1.0 - h[ip]) if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
            gi += w[center + k] * d_pos + w[center - k] * d_neg
        gi += w[center] * (1.0 - 2.0 * h[i])
        grad[i] = gi
    h_new = h - lr * grad
    return project(h_new, n)

def make_structured_h(n, rng):
    style = rng.randint(5)
    if style == 0:
        x = np.linspace(0, 2, n, endpoint=False)
        h = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x * rng.randint(1, 4) + rng.uniform(0, 6.3)), 0, 1)
        h += 0.05 * np.random.randn(n)
    elif style == 1:
        h = np.zeros(n)
        mask = np.random.choice(n, n // 2, replace=False)
        h[mask] = 1.0
    elif style == 2:
        ph = 0.5 + 1.0 / np.sqrt(5)
        idx = np.arange(n)
        h = (np.floor((idx + 1) * ph) - np.floor(idx * ph)).astype(float)
    elif style == 3:
        p = rng.randint(3, 8)
        block = np.zeros(p)
        block[:p // 2] = 1.0
        h = np.tile(block, n // p + 1)[:n]
        h += 0.08 * np.random.randn(n)
    else:
        h = np.full(n, 0.5)
        h += 0.3 * np.random.randn(n)
    return project(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    best_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)

    if best_h is None:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x), 0, 1)

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Fast coordinate descent (45%)
    t1 = t0 + budget_s * 0.45
    best_h, best_c5, _ = cd_loop(best_h, n, t1, 8000, gs_iters=28)

    # Phase 2: Smoothed gradient + coordinate polish (20%)
    t2 = t0 + budget_s * 0.65
    T_cur = 50.0
    lr = 0.02
    for it in range(2000):
        if time.time() > t2:
            break
        h_sg = smooth_grad_step(best_h, n, T_cur, lr)
        h_sg, c5_sg, _ = cd_loop(h_sg, n, min(time.time() + 0.5, t2), 50, gs_iters=20)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
            T_cur *= 1.1
            lr *= 0.99
        else:
            T_cur = max(10.0, T_cur * 0.95)
            lr *= 0.97
        # Occasionally old-hh perturbation
        if it % 30 == 15:
            h_p = best_h + 0.04 * np.random.randn(n)
            h_p = project(h_p, n)
            h_p, c5_p, _ = cd_loop(h_p, n, min(time.time() + 1.0, t2), 200, gs_iters=22)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p

    # Phase 3: Multi-start structured (15%)
    t3 = t0 + budget_s * 0.80
    for trial in range(12):
        if time.time() > t3:
            break
        h_try = make_structured_h(n, rng)
        h_try, c5_try, _ = cd_loop(h_try, n, min(time.time() + (t3 - t0) / 40 * 3, t3), 2000, gs_iters=25)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 4: Multi-resolution (10%)
    t4 = t0 + budget_s * 0.90
    for n_try in [128, 200, 256, 320]:
        if time.time() > t4:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project(h_interp, n_try)
        h_tr, c5_tr, _ = cd_loop(h_interp, n_try, min(time.time() + 20, t4), 3000, gs_iters=25)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr
            n = n_try

    # Phase 5: Final polish (10%)
    best_h, best_c5, _ = cd_loop(best_h, n, t_end - 2, 5000, gs_iters=30)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
