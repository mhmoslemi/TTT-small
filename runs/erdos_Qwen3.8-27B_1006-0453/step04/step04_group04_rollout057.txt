import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500.0, 500.0)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def make_obj_con(n):
    def obj_fun(x, T):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    return obj_fun, con_fun

def slsqp_smooth(h, T, n, maxiter=500):
    obj_f, con_f = make_obj_con(n)
    res = minimize(obj_f, h, args=(T,), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_f}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def coord_single_sweep(h, n):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_corr = 2 * n - 1
    for i in range(n):
        h_save = best_h[i]
        hz = best_h.copy()
        hz[i] = 0.0
        corr_base = np.correlate(hz, 1.0 - hz, mode='full')
        coef = np.zeros(n_corr)
        for k in range(n_corr):
            lag = k - (n - 1)
            c = 0.0
            j1 = i + lag
            j2 = i - lag
            if 0 <= j1 < n:
                c += (1.0 - best_h[j1])
            if 0 <= j2 < n:
                c += (-best_h[j2])
            coef[k] = c
        k0 = n - 1
        base_k0 = corr_base[k0]
        def f1d(t):
            cv = corr_base + t * coef
            cv[k0] = base_k0 + t * (1.0 - t)
            return float(np.max(cv)) * dx
        try:
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            h_try = best_h.copy()
            h_try[i] = res.x
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_h = h_try
                best_c5 = c5_try
        except Exception:
            pass
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        best_h, best_c5 = coord_single_sweep(best_h, n)
    return best_h, best_c5

def structured_perturb(h, n, rng):
    h_p = h.copy()
    op = rng.randint(5)
    if op == 0:
        bsize = int(rng.randint(4, 14))
        start = int(rng.randint(0, n - bsize))
        h_p[start:start+bsize] = 1.0 - h_p[start:start+bsize]
    elif op == 1:
        bsize = int(rng.randint(6, 18))
        start = int(rng.randint(0, max(1, n - bsize)))
        shift = int(rng.randint(-4, 5))
        seg = h_p[start:start+bsize]
        h_p[start:start+bsize] = np.roll(seg, shift)
    elif op == 2:
        r = int(rng.randint(2, 6))
        idx = int(rng.randint(r, n - r))
        win = h_p[idx-r:idx+r+1].copy()
        ker = np.ones(2*r+1)/(2*r+1)
        h_p[idx-r:idx+r+1] = np.convolve(win, ker, 'same')
    elif op == 3:
        bsize = int(rng.randint(12, 35))
        start = int(rng.randint(0, n - bsize))
        h_p[start:start+bsize] += 0.08 * rng.randn(bsize)
    else:
        bsize = int(rng.randint(5, 20))
        start = int(rng.randint(0, n - bsize))
        h_p[start:start+bsize] = np.clip(h_p[start:start+bsize] + 0.15*(rng.rand(bsize)-0.5)*2, 0, 1)
    return proj(h_p, n)

def basin_hop(h, n, t_end, n_iters=150):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = best_h.copy()
    work_c5 = best_c5
    rng = np.random.RandomState(int(time.time()*1000) % 100000)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.88 * (it / float(n_iters))
        h_pert = structured_perturb(work_h, n, rng)
        T_bh = 800 + 8000 * cool
        try:
            hc = slsqp_smooth(h_pert, T_bh, n, maxiter=250)
            c5v = compute_c5(hc)
            temp = 0.005 * cool + 0.0004
            if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass
    return best_h, best_c5

def block_slsqp(h, n, t_end, bs=6, max_passes=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for p in range(max_passes):
        if time.time() > t_end:
            break
        improved = False
        for start in range(0, n - bs + 1, bs):
            if time.time() > t_end:
                return best_h, best_c5
            idx = list(range(start, start+bs))
            h0 = best_h[idx].copy()
            fixed = best_h.copy()
            def blk_obj(xb):
                ht = fixed.copy()
                ht[idx] = xb
                ht = np.clip(ht, 0.0, 1.0)
                return smooth_c5(ht, 4000)
            def blk_con(xb):
                return xb.sum() - h0.sum()
            try:
                res = minimize(blk_obj, h0, method='SLSQP',
                             bounds=[(0.0,1.0)]*bs,
                             constraints=[{'type':'eq','fun':blk_con}],
                             options={'maxiter':80,'ftol':1e-14})
                ht = best_h.copy()
                ht[idx] = np.clip(res.x, 0, 1)
                ht = proj(ht, n)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-14:
                    best_h = ht
                    best_c5 = c5t
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with progressive T
    t1 = t0 + 40
    for T in [200, 1000, 4000, 15000, 50000]:
        if time.time() > t1:
            break
        try:
            hc = slsqp_smooth(best_h, T, n, maxiter=600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Single-coordinate descent (many sweeps)
    t2 = t0 + 200
    if time.time() < t2:
        best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=3000)

    # Phase 3: Basin hopping with structured perturbations
    t3 = t0 + 500
    if time.time() < t3:
        best_h, best_c5 = basin_hop(best_h, n, t3, n_iters=120)

    # Phase 4: Block SLSQP
    t4 = t0 + 650
    if time.time() < t4:
        best_h, best_c5 = block_slsqp(best_h, n, t4, bs=5, max_passes=60)

    # Phase 5: More coordinate descent
    t5 = t0 + 850
    if time.time() < t5:
        best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=3000)

    # Phase 6: Second basin hop round
    t6 = t0 + 1000
    if time.time() < t6:
        best_h, best_c5 = basin_hop(best_h, n, t6, n_iters=60)

    # Phase 7: Final high-T SLSQP + coord
    t7 = t0 + budget_s - 5
    if time.time() < t7:
        for T in [80000, 200000]:
            if time.time() > t7:
                break
            try:
                hc = slsqp_smooth(best_h, T, n, maxiter=2000)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass
    if time.time() < t0 + budget_s - 2:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s - 2, max_sweeps=500)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
