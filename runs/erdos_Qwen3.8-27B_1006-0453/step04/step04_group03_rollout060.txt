def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def coord_descent(h, n, t_end, n_sweeps=200, random_order=True):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n) if random_order else np.arange(n)
        for i in order:
            if (i & 63) == 0 and time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            coef = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                j1 = i - k
                j2 = i + k
                c = 0.0
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[k + n - 1] = c
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            def f1d(s):
                return np.max(corr + s * coef)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-11})
            s_new = res.x
            if abs(s_new) > 1e-12:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def fourier_perturb(h, n, K, scale):
    x = np.arange(n)
    pert = np.zeros(n)
    for k in range(1, K + 1):
        a = np.random.randn() * scale
        b = np.random.randn() * scale
        pert += a * np.cos(2.0 * np.pi * k * x / n) + b * np.sin(2.0 * np.pi * k * x / n)
    return h + pert

def block_perturb(h, n, scale):
    block_len = int(np.random.randint(n // 8, n // 3 + 1))
    start = np.random.randint(0, n - block_len + 1)
    h_p = h.copy()
    h_p[start:start + block_len] += scale * np.random.randn(block_len)
    return h_p

def symmetric_perturb(h, n, cool):
    h_p = h.copy()
    n_pairs = int(n * 0.05 * cool + 3)
    for _ in range(n_pairs):
        idx = np.random.randint(0, n // 2)
        jdx = n - 1 - idx
        delta = (0.02 + 0.1 * cool) * np.random.randn()
        h_p[idx] += delta
        h_p[jdx] -= delta
    return h_p

def level_set_perturb(h, thresh):
    mask = (h > thresh).astype(float)
    return 0.7 * h + 0.3 * mask

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(224)
        best_h[:112] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial CD polish (8% budget)
    t1 = t0 + budget_s * 0.08
    best_h, best_c5 = coord_descent(best_h, n, t1, n_sweeps=40)

    # Phase 2: Basin hopping (42% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 120
    for it in range(n_bh):
        if elapsed() > budget_s * 0.50:
            break
        cool = max(1.0 - 0.96 * (it / float(n_bh)), 0.02)
        r = np.random.rand()
        time_pts = 1.5 + 2.0 * cool
        if r < 0.25:
            step = 0.02 + 0.14 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.45:
            K = int(2 + 6 * cool)
            h_p = fourier_perturb(work_h, n, K, 0.015 + 0.07 * cool)
        elif r < 0.60:
            h_p = block_perturb(work_h, n, 0.02 + 0.1 * cool)
        elif r < 0.80:
            h_p = symmetric_perturb(work_h, n, cool)
        else:
            h_p = level_set_perturb(work_h, np.random.uniform(0.25, 0.75))

        h_p = project_to_sum(h_p, n / 2)
        try:
            hc, c5v = coord_descent(h_p, n, time.time() + time_pts, n_sweeps=int(8 + 12 * cool))
        except Exception:
            continue
        temp = 0.0025 * cool + 0.00004
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 3: SLSQP with T-schedule (15% budget)
    for T in [200, 1000, 5000, 20000, 80000]:
        if elapsed() > budget_s * 0.65:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=600, ftol=1e-13)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Phase 4: Multi-resolution (15% budget)
    for n_try in [192, 256, 288, 320, 128]:
        if n_try == n or elapsed() > budget_s * 0.80:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2)
        hc, c5v = coord_descent(h_i, n_try, time.time() + 4.0, n_sweeps=25)
        if c5v < best_c5:
            best_c5, best_h, n = c5v, hc, n_try

    # Phase 5: Final intensive CD (12% budget)
    t5 = t0 + budget_s * 0.92
    best_h, best_c5 = coord_descent(best_h, n, t5, n_sweeps=300, random_order=False)

    # Phase 6: High-T SLSQP polish (5% budget)
    if elapsed() < budget_s * 0.97:
        try:
            hc = slsqp_polish(best_h, 100000, n, maxiter=1000, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Final micro-CD
    if elapsed() < budget_s * 0.995:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, n_sweeps=40)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
