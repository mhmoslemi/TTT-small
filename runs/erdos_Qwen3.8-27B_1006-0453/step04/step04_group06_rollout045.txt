import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        h = h * (target / s)
        h = np.clip(h, 0, 1)
    else:
        h = h * (target / max(s, 1e-15))
        h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if diff > 1e-14:
        order = np.argsort(h)
        for i in order:
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
            if diff < 1e-14:
                break
    elif diff < -1e-14:
        order = np.argsort(-h)
        for i in order:
            a = min(-diff, h[i])
            h[i] -= a
            diff += a
            if diff > -1e-14:
                break
    return h

def smooth_max(c, T):
    m = c.max()
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def slsqp_opt(h, T, n, maxiter=600):
    def obj(x):
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T)
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        e = np.exp(T * (c - c.max()))
        w = e / e.sum()
        g = np.zeros(2 * n - 1)
        g[:] = w
        g *= T
        # gradient of corr wrt x
        grad_x = np.zeros(n)
        for idx in range(2 * n - 1):
            k = idx - (n - 1)
            gi = g[idx]
            if gi < 1e-16:
                continue
            if k == 0:
                grad_x += gi * (1.0 - 2.0 * x)
            elif k > 0:
                grad_x[:n - k] += gi * (1.0 - x[k:])
                grad_x[k:] -= gi * x[:n - k]
            else:
                m2 = -k
                grad_x[m2:] += gi * (1.0 - x[:n - m2])
                grad_x[:n - m2] -= gi * x[m2:]
        return obj(x), grad_x
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_grad, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def coord_sweep(h, n, t_end, max_sweeps=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    center = n - 1
    dx = 2.0 / n
    sweep = 0
    stagnant = 0
    while sweep < max_sweeps and time.time() < t_end:
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            h0 = best.copy(); h0[i] = 0.0
            c0 = np.correlate(h0, 1.0 - h0, mode='full')
            h1 = best.copy(); h1[i] = 1.0
            c1 = np.correlate(h1, 1.0 - h1, mode='full')
            d = c1 - c0
            c0c = c0[center]
            def f1(x):
                c = c0 + x * d
                c[center] = c0c + x - x * x
                return np.max(c) * dx
            r = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            best[i] = r.x
            best = project(best, n)
        c5_now = compute_c5(best)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            improved = True
        stagnant = stagnant + 1 if not improved else 0
        if stagnant >= 15:
            break
        sweep += 1
    return best, best_c5

def interp_h(h_old, n_old, n_new):
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(xn, xo, h_old, period=2.0)

def binary_perturb(h, n, alpha):
    b = (h > 0.5).astype(float)
    b = project(b, n)
    return project(alpha * h + (1 - alpha) * b, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x), 0, 1)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    candidates = [(best_h.copy(), best_c5, n)]

    def time_left(frac):
        return time.time() < t0 + budget_s * frac

    # Phase 1: Coordinate descent
    cd_end = t0 + budget_s * 0.22
    best_h, best_c5 = coord_sweep(best_h, n, cd_end, max_sweeps=200)
    candidates.append((best_h.copy(), best_c5, n))

    # Phase 2: SLSQP with increasing T
    for T in [30, 80, 200, 500, 1500, 5000]:
        if not time_left(0.40):
            break
        try:
            hc = slsqp_opt(best_h.copy(), T, n, maxiter=800)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_c5 = cv
                best_h = hc
                candidates.append((best_h.copy(), best_c5, n))
        except Exception:
            pass

    # Phase 3: Perturb + coordinate descent (escape local minima)
    for trial in range(8):
        if not time_left(0.58):
            break
        alpha = 0.7 - 0.04 * trial
        hp = binary_perturb(best_h, n, alpha)
        hp += 0.03 * np.random.randn(n)
        hp = project(hp, n)
        hc, cv = coord_sweep(hp, n, min(t0 + budget_s * 0.58, time.time() + 25), max_sweeps=40)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc
            candidates.append((best_h.copy(), best_c5, n))

    # Phase 4: Multi-resolution
    for n_try in [200, 240, 320]:
        if not time_left(0.76):
            break
        h_interp = interp_h(best_h, n, n_try)
        h_interp = project(h_interp, n_try)
        for T in [200, 800, 3000]:
            if not time_left(0.76):
                break
            try:
                hc = slsqp_opt(h_interp, T, n_try, maxiter=500)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_c5 = cv
                    best_h = hc
                    n = n_try
                    candidates.append((best_h.copy(), best_c5, n))
            except Exception:
                pass
        hc2, cv2 = coord_sweep(best_h, n, min(t0 + budget_s * 0.76, time.time() + 20), max_sweeps=50)
        if cv2 < best_c5:
            best_c5 = cv2
            best_h = hc2
            n = len(best_h)
            candidates.append((best_h.copy(), best_c5, n))

    # Phase 5: High-T SLSQP polish
    for T in [8000, 20000, 60000]:
        if not time_left(0.92):
            break
        try:
            hc = slsqp_opt(best_h.copy(), T, n, maxiter=1000)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_c5 = cv
                best_h = hc
                candidates.append((best_h.copy(), best_c5, n))
        except Exception:
            pass

    # Phase 6: Final coordinate descent
    best_h, best_c5 = coord_sweep(best_h, n, t_end - 5, max_sweeps=150)
    candidates.append((best_h.copy(), best_c5, n))

    # Return the best candidate
    best_h, best_c5, best_n = min(candidates, key=lambda t: t[1])
    best_h = project(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
