from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_full(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def coef_for_i(h, i):
    """Compute the coefficient vector for changing h[i]."""
    n = len(h)
    c = np.zeros(2 * n - 1)
    for k in range(-(n-1), n):
        j1 = i - k  # index in h for h[i-k] term
        j2 = i + k  # index for h[i+k] term
        val = 0.0
        if 0 <= j1 < n:
            val += (1.0 - h[j1])
        if 0 <= j2 < n:
            val -= h[j2]
        c[k + n - 1] = val
    return c

def ternary_min_piecewise_linear(corr, coef, lo, hi, n_iter=30):
    """Minimize f(s) = max(corr + s*coef) over [lo, hi]. Convex piecewise linear."""
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr + m1 * coef)
        f2 = np.max(corr + m2 * coef)
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def nesterov_subgrad(h, t_end, lr_init=0.01, momentum=0.9):
    """Nesterov accelerated subgradient method."""
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(x)
    lr = lr_init
    
    t = 0
    while time.time() < t_end:
        t += 1
        lr = lr_init / (1.0 + 0.01 * t)
        
        # Subgradient: pick the lag k* that achieves max correlation
        corr = corr_full(x)
        k_star = np.argmax(corr)
        
        # Subgradient of corr[k_star] w.r.t. h[i]
        k_shift = k_star - (n - 1)
        subgrad = np.zeros(n)
        for i in range(n):
            j1 = i - k_shift
            j2 = i + k_shift
            val = 0.0
            if 0 <= j2 < n:

def slsqp_smooth(h, T, n, maxiter=500, ftol=1e-13):
    """SLSQP with smooth max approximation."""
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sm = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return sm * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(np.clip(res.x, 0, 1), n/2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_full(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def coef_for_i(h, i):
    n = len(h)
    c = np.zeros(2 * n - 1)
    lo1 = max(-i, -(n-1))
    hi1 = min(n-1-i, n-1)
    if lo1 <= hi1:
        c[lo1+n-1:hi1+n-1+1] += (1.0 - h[i+lo1:i+hi1+1])
    lo2 = max(i-n+1, -(n-1))
    hi2 = min(i, n-1)
    if lo2 <= hi2:
        c[lo2+n-1:hi2+n-1+1] -= h[i-hi2:i-lo2+1][::-1]
    return c

def ternary_pl(corr, coef, lo, hi, n_iter=28):
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr + m1 * coef) <= np.max(corr + m2 * coef):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def fast_coord_descent(h, t_end, max_sweeps=2000):
    n = len(h)
    best_h = h.copy()
    corr = corr_full(best_h)
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            lo = -best_h[i]
            hi = 1.0 - best_h[i]
            if hi - lo < 1e-13:
                continue
            c = coef_for_i(best_h, i)
            s_new = ternary_pl(corr, c, lo, hi, 24)
            if abs(s_new) > 1e-12:
                best_h[i] += s_new
                corr += s_new * c
                c5_new = np.max(corr) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def nesterov_subgrad(h, t_end, lr_init=0.015, mom=0.88):
    n = len(h)
    x = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(x)
    t = 0
    while time.time() < t_end:
        t += 1
        lr = lr_init / (1.0 + 0.005 * t)
        corr = corr_full(x)
        k_star = np.argmax(corr)
        k_shift = k_star - (n - 1)
        subgrad = np.zeros(n)
        for i in range(n):
            val = 0.0
            j2 = i + k_shift
            if 0 <= j2 < n:
                val += (1.0 - x[j2])
            j1 = i - k_shift
            if 0 <= j1 < n:
                val -= x[j1]
            subgrad[i] = val
        v = mom * v - lr * subgrad
        x_new = project_sum(x + v, n / 2.0)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new.copy()
        x = x_new
    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=600, ftol=1e-13):
    dx = 2.0 / n
    def obj(x):
        xc = np.clip(x, 0, 1)
        cr = np.correlate(xc, 1.0 - xc, mode='full')
        m = np.max(cr)
        return (m + np.log(np.sum(np.exp(T * (cr - m)))) / T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_sum(np.clip(res.x, 0, 1), n / 2.0)
    except Exception:
        return h

        best_h, best_c5 = fast_coord_descent(best_h, min(elapsed() + 0.5, t3), max_sweeps=50)
