import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx, m

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
    return h

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wm = w[j]
        if wm < 1e-10:
            continue
        mlag = j - (n - 1)
        if mlag >= 0:
            grad[:n - mlag] += wm * (1.0 - h[mlag:])
            grad[mlag:] -= wm * h[:n - mlag]
        else:
            p = -mlag
            grad[p:] += wm * (1.0 - h[:n - p])
            grad[:n - p] -= wm * h[p:]
    return grad * dx

def subgrad_opt(h0, n, T_schedule, lr0, steps, momentum=0.85, t_end=None):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_sched = len(T_schedule)
    steps_per = steps // n_sched
    for si, T in enumerate(T_schedule):
        for s in range(steps_per):
            if t_end and time.time() > t_end:
                return best_h, best_c5
            grad = grad_smooth(h, T)
            lr = lr0 * (1.0 / (1.0 + 0.5 * s / steps_per))
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            h = h_new
    return best_h, best_c5

def coord_opt_fixed(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += (1.0 - h[i - k])
        if 0 <= i + k < n:
            dk += (-h[i + k])
        d[n - 1 + k] = dk
        dk2 = 0.0
        if 0 <= i + k < n:
            dk2 += (1.0 - h[i + k])
        if 0 <= i - k < n:
            dk2 += (-h[i - k])
        d[n - 1 - k] = dk2
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=50, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fixed(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5_val(x, T)[0]
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def structural_init(n, rng, A=0.3):
    """Create structured initializations."""
    h = np.zeros(n)
    mode = rng.integers(0, 4)
    if mode == 0:
        freq = rng.choice([1, 2, 3, 4, 5, 7])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + A * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 1:
        # Block structure
        block = int(rng.integers(3, 15))
        h = np.tile(0.5 + A * rng.choice([-1, 1], size=(n // block + 1,)), n)[:n]
    elif mode == 2:
        # Two-frequency
        f1 = rng.integers(1, 6)
        f2 = rng.integers(6, 15)
        h = 0.5 + 0.2 * np.cos(2 * np.pi * f1 * np.arange(n) / n) + 0.1 * np.cos(2 * np.pi * f2 * np.arange(n) / n)
    else:
        h = rng.uniform(0, 1, n)
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
    return h

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wm = w[j]
        if wm < 1e-11:
            continue
        mlag = j - (n - 1)
        if mlag >= 0:
            grad[:n - mlag] += wm * (1.0 - h[mlag:])
            grad[mlag:] -= wm * h[:n - mlag]
        else:
            p = -mlag
            grad[p:] += wm * (1.0 - h[:n - p])
            grad[:n - p] -= wm * h[p:]
    return grad * dx

def subgrad_opt(h0, n, T_list, lr0, total_steps, momentum, t_end):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    steps_per_T = max(1, total_steps // len(T_list))
    for T in T_list:
        for s in range(steps_per_T):
            if time.time() > t_end:
                return best_h, best_c5
            grad = grad_smooth(h, T)
            lr = lr0 / (1.0 + 0.3 * s / steps_per_T)
            v = momentum * v - lr * grad
            h_new = project(h + v, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            h = h_new
    return best_h, best_c5

def coord_opt_fixed(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += (1.0 - h[i - k])
        if 0 <= i + k < n:
            dk += (-h[i + k])
        d[n - 1 + k] = dk
        d[n - 1 - k] = dk
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=50, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fixed(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=600):
    def obj(x):
        return smooth_c5_val(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def structural_init(n, rng):
    h = np.zeros(n)
    mode = rng.integers(0, 5)
    if mode == 0:
        freq = int(rng.integers(1, 8))
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 1:
        block = int(rng.integers(4, 20))
        pat = 0.5 + 0.3 * rng.choice([-1, 1], size=n // block + 2)
        h = np.tile(pat, n)[:n]
    elif mode == 2:
        f1, f2 = int(rng.integers(1, 5)), int(rng.integers(6, 16))
        h = 0.5 + 0.22 * np.cos(2 * np.pi * f1 * np.arange(n) / n) + 0.15 * np.cos(2 * np.pi * f2 * np.arange(n) / n + rng.uniform(0, 6))
    elif mode == 3:
        h = 0.5 + 0.3 * rng.standard_normal(n)
    else:
        h = rng.uniform(0.2, 0.8, n)
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish
    t1 = t0 + budget_s * 0.10
    for T in [50, 300, 2000]:
        if time.time() > t1:

        try:
            hc = slsqp_opt(best_h, T, n, maxiter=700)
            cv = compute_c5(hc)
            if cv < best_c5:

        except Exception:
            pass

    # Phase 2: Subgradient with increasing T (heavy ball)
    t2 = t0 + budget_s * 0.35
    T_sched = [80, 200, 500, 1500, 5000, 15000, 50000]
    best_h, best_c5 = subgrad_opt(best_h, n, T_sched, lr0=0.3, total_steps=50000, momentum=0.82, t_end=t2)

    t3 = t0 + budget_s * 0.50

        h_try, c5_try = subgrad_opt(h_try, n, [500, 3000, 12000], lr0=0.2, total_steps=20000, momentum=0.8, t_end=t4)
        if c5_try < best_c5:

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
