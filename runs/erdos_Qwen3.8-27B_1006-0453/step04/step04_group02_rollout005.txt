```python
def subgradient_step(h, n, step_size):
    corr = np.correlate(h, 1-h, mode='full')
    k_max = np.argmax(corr)
    # Compute d(corr[k_max])/dh
    k = k_max - (n-1)  # actual lag
    grad = np.zeros(n)
    # d(corr[k])/dh[i] = (1-h[i-k])*I[valid] - h[i+k]*I[valid]
    for i in range(n):
        # First term: h[i] appears as h[n+k] when i-k is valid
        if 0 <= i-k < n and 0 <= i-k < n:
            # Check if i-k is in the valid range for the sum
            j = i - k
            if 0 <= j < n and 0 <= j+k < n:
                grad[i] += (1 - h[j])
        # Second term: h[i] appears as 1-h[j] when j=i
        j = i
        if 0 <= j < n and 0 <= j+k < n:
            grad[i] -= h[j+k]
    # Projected gradient step
    h_new = h - step_size * grad
    return project(h_new, n)
```

```python
def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    p = np.exp(T * (corr - m))
    p /= p.sum()
    grad = np.zeros(n)
    # k >= 0: k_idx from n-1 to 2n-2, k = k_idx - (n-1)
    for j in range(1, n):  # j = k, from 1 to n-1
        pi = p[n-1+j]
        grad[j:n] += pi * (1.0 - h[0:n-j])
        grad[0:n-j] -= pi * h[j:n]
    # k < 0: k_idx from 0 to n-2, m = -k = n-1-k_idx
    for j in range(1, n):  # j = m = -k
        pi = p[n-1-j]
        grad[0:n-j] += pi * (1.0 - h[j:n])
        grad[j:n] -= pi * h[0:n-j]
    return grad * dx
```

The problem is min_h max_k corr[k] * dx

The constraint corr[k] * dx ≤ t can be written as inner product: <A_k h + b_k, 1> ≤ t where A_k is a linear operator (approximately, since corr[0] is quadratic).

For k ≠ 0, corr[k] is linear in h... wait no, it's bilinear: h[i] * (1-h[i+k]) = h[i] - h[i]*h[i+k].

So it's not linear. The smooth approximation via log-sum-exp is the right way.

Let me try a different angle: **Alternating optimization**

Fix the "weights" w on the simplex (sum w_k = 1, w_k ≥ 0) and minimize:
f(h) = sum_k w_k * corr[k] * dx

This is a quadratic in h (since corr[0] has the -h² term). The optimal h can be found in closed form (or via a simple solver).

Then update w to put more weight on the largest corr[k].

This is the **alternating method for minimax** (also known as the "weight method" or "multiplicative weights").

Algorithm:
1. Initialize w uniformly
2. Repeat:
   a. Given w, minimize g_h(w) = sum_k w_k * (dx/2) * corr_k(h) over h (with constraints)
      - This is a quadratic (or near-quadratic) optimization
   b. Update w_k proportional to exp(eta * corr_k(h)) (multiplicative weights)

For step (a), the objective is:
g(h) = sum_k w_k * dx/2 * sum_{valid i} h[i]*(1-h[i+k])
     = dx/2 * sum_i h[i] * (1 - sum_k w_k * h[i+k_valid])
       ... this is getting complicated.

Actually for k=0: term is w_0 * sum h[i]*(1-h[i]) = w_0 * (n/2 - sum h[i]²)
For k≠0: term is w_k * sum_{valid} h[i]*(1-h[i+k]) = w_k * (sum h[i] - sum h[i]*h[i+k])
                                          = w_k * (n/2 - sum h[i]*h[i+k])

So total: g(h) = dx/2 * [w_0*(n/2 - sum h²) + sum_{k≠0} w_k*(n/2 - sum h[i]*h[i+k])]
            = dx/2 * [n/2 * sum w_k - w_0*sum h² - sum_{k≠0} w_k * sum h[i]*h[i+k]]
            = dx/2 * [n/2 - w_0*sum h² - sum_{k≠0} w_k * C_k(h)]

where C_k(h) = sum_{valid i} h[i]*h[i+k]

Minimizing g(h) = maximizing w_0*sum h² + sum_{k≠0} w_k * C_k(h) subject to sum h = n/2, 0≤h≤1.

This is a QUADRATIC program! And it's maximize, so the Hessian needs to be negative semidefinite for it to be convex (in the min sense).

The quadratic form is: Q(h) = w_0*sum h² + sum_{k≠0} w_k * sum_{valid} h[i]*h[i+k]
     = h^T Q h + linear terms...

Actually sum h[i]*h[i+k] for fixed k is not a standard quadratic form (it's a shifted dot product). But we can write it as h^T A_k h where (A_k)_{ij} = 1 if j-i = k or i-j = k (for the symmetric part) and appropriate boundary handling.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / len(h))

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    p = np.exp(T * (corr - m))
    p /= p.sum()
    grad = np.zeros(n)
    for j in range(1, n):
        pj = p[n - 1 + j]
        if pj > 1e-15:
            grad[j:n] += pj * (1.0 - h[0:n-j])
            grad[0:n-j] -= pj * h[j:n]
        pj2 = p[n - 1 - j]
        if pj2 > 1e-15:
            grad[0:n-j] += pj2 * (1.0 - h[j:n])
            grad[j:n] -= pj2 * h[0:n-j]
    return grad * dx

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    for _ in range(10):
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            h = np.clip(h + diff / n, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
    h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, T, n, maxiter=800, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def lbfgsb_opt(h, T, n, maxiter=2000, ftol=1e-16, gtol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def obj_grad(x):
        return smooth_c5_grad(x, T)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def coord_opt_fixed(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if 0 <= i - k < n:
            d[n - 1 + k] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            d[n - 1 + k] += (-h[i + k])
        if 0 <= i + k < n:
            d[n - 1 - k] += (1.0 - h[i + k])
        if 0 <= i - k < n:
            d[n - 1 - k] += (-h[i - k])
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=80, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fixed(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-13:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def subgrad_opt(h, n, t_end, n_iter=5000, eta0=0.02, decay=0.999):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(work, 1.0 - work, mode='full')
        k_max = np.argmax(corr)
        k = k_max - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            if k < n:
                grad[k:n] += (1.0 - work[0:n-k])
            if n - k > 0:
                grad[0:n-k] -= work[k:n]
        else:
            m = -k
            if n - m > 0:
                grad[0:n-m] += (1.0 - work[m:n])
            if m < n:
                grad[m:n] -= work[0:n-m]
        eta = eta0 * (decay ** it)
        work = work - eta * grad
        work = project(work, n)
        c5 = compute_c5(work)
        if c5 < best_c5 - 1e-15:
            best_h, best_c5 = work.copy(), c5
    return best_h, best_c5

def lbfgsb_smooth(h, T, n, t_end):
    dx = 2.0 / n
    def obj(x):
        return smooth_c5(x, T) * n  # scale so grad matches sum constraint
    def gfunc(x):
        return smooth_c5_grad(x, T) * n
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=gfunc, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-15,
                            'maxls': 50, 'iprint': -1})
    hc = project(np.clip(res.x, 0, 1), n)
    return hc

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def multi_start_opt(seed, n, t_end, rng):
    best_h = None
    best_c5 = float('inf')
    n_starts = 5
    per_start = (t_end - time.time()) / n_starts
    for s in range(n_starts):
        if per_start < 10:
            break
        t_sub = time.time() + per_start
        h0 = rng.random(n)
        h0 = project(h0, n)
        c5_0 = compute_c5(h0)
        if best_h is not None and c5_0 > best_c5 + 0.05:
            h0 = interpolate_h(best_h, n)
        for T in [50, 300, 2000]:
            if time.time() > t_sub - 5:
                break
            try:
                hc = lbfgsb_smooth(h0, T, n, t_sub)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5 = hc, cv
                h0 = hc
            except Exception:
                pass
        if time.time() < t_sub - 3:
            try:
                hc = slsqp_opt(h0, 5000, n, maxiter=500, ftol=1e-14)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5 = hc, cv
            except Exception:
                pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0
```
