Minimizing max_k (N/2 - A(k)) is equivalent to minimizing max_k sum_i h[i]*(1-h[i+k])

This is a max of linear functions (in the "flattened" sense), so it's a convex-over-linear problem. We can use bisection: is there h with all overlaps ≤ T? This is a feasibility problem.

But with the binary constraint, it's still NP-hard in general.

Let me just focus on practical improvements to the optimization:

1. **Segment-based local search**: Instead of only swapping two bits, try:
   - Flip a segment (complement h[a:b])
   - Reverse a segment
   - Swap two segments

2. **Multi-start with diverse structures**:
   - Quadratic residue pattern
   - Sturmian / mechanical word
   - Random
   - Perturbed initial

3. **L-BFGS with projected gradient**:
   - Use scipy.optimize.minimize with method='L-BFGS-B'
   - Objective: soft-max approximation of C₅
   - Handle sum constraint by optimizing over N-1 free variables (fix last to satisfy sum)
   - Or use penalty method: add λ*(sum(h) - N/2)² to objective

4. **Iterated refinement**:
   - Binary → continuous L-BFGS → round → binary local search → repeat

def compute_c5(h, N):

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid

def lbfgs_opt(h0, N, deadline, K=150, beta=8.0, max_iter=3000):
    target = N / 2.0
    center = N - 1
    best_c5 = compute_c5(h0, N)
    best_h = h0.copy()

    def fg(x):
        hx = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        if K >= len(corr):
            sel = np.arange(len(corr))
        else:
            sel = np.argpartition(corr, -K)[-K:]
        cv = corr[sel]
        m = cv.max()
        w = np.exp(beta * (cv - m))
        w /= w.sum()
        obj = float(w.dot(cv)) * (2.0 / N)
        s = float(hx.sum())
        obj += 50.0 * (s - target) ** 2
        grad = np.zeros(N)
        for i in range(len(sel)):
            L = sel[i] - center
            wi = w[i]
            if L == 0:
                grad += wi * (1.0 - 2.0 * hx)
            elif L > 0:
                grad[:N - L] += wi * (1.0 - hx[L:])
                grad[L:] -= wi * hx[:N - L]
            else:
                M = -L
                grad[M:] += wi * (1.0 - hx[:N - M])
                grad[:N - M] -= wi * hx[M:]
        grad *= (2.0 / N)
        grad += 100.0 * (s - target)
        return obj, grad

    h_cur = h0.astype(float).copy()
    while time.time() < deadline:
        try:
            res = minimize(fg, h_cur, method='L-BFGS-B',
                           bounds=[(0, 1)] * N,
                           options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-9,
                                    'maxfun': 2500, 'disp': False})
            h_opt = project_sum(res.x, target)
            c5_opt = compute_c5(h_opt, N)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_cur = h_opt.copy()
            else:
                break
        except Exception:
            break
    return best_h, best_c5

    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / n_iters)
        pi = ones[rng.randint(len(ones))]
        qi = zeros[rng.randint(len(zeros))]
        h[pi], h[qi] = h[qi], h[pi]
        cn = np.correlate(h, 1.0 - h, mode='full')
        npk = float(cn.max())
        delta = npk - cur_peak
        if delta < 0 or (T > 0.004 and rng.random() < np.exp(-delta / T)):
            cur_peak = npk
            if npk < peak:
                peak = npk
                c5v = npk * (2.0 / N)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[pi], h[qi] = h[qi], h[pi]
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float).copy()
    N0 = len(best_h)
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary steepest descent from initial
    d1 = t0 + 120
    h_bin = binary_fix(best_h, N0)
    h1, c5_1 = steepest_descent(h_bin, N0, d1, rng, n_try=50)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Anneal + descent iterations
    d2 = t0 + 400
    h_cur = best_h.copy()
    for restart in range(15):
        if time.time() > d2:
            break
        rem = d2 - time.time()
        n_ann = max(2000, int(rem * 15))
        h_a, c5_a = anneal_binary(h_cur, N0, n_ann, rng, T0=3.0, T1=0.002)
        h_s, c5_s = steepest_descent(h_a, N0, min(d2 - 5, time.time() + 25), rng, n_try=50)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s.copy()
            h_cur = h_s.copy()

    # Phase 3: L-BFGS continuous optimization
    d3 = t0 + 650
    h_cont = best_h.astype(float)
    h_cont += rng.randn(N0) * 0.05
    h_cont = project_sum(h_cont, N0 / 2.0)
    h3, c5_3 = lbfgs_opt(h_cont, N0, d3, K=150, beta=8.0)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 4: Iterate binary -> continuous
    d4 = t0 + 850
    for it in range(5):
        if time.time() > d4:
            break
        h_b = binary_fix(best_h, N0)
        h_s, c5_s = steepest_descent(h_b, N0, min(d4 - 10, time.time() + 30), rng, n_try=50)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s
        h_c = best_h.astype(float) + rng.randn(N0) * 0.03
        h_c = project_sum(h_c, N0 / 2.0)
        h_l, c5_l = lbfgs_opt(h_c, N0, min(d4 - 5, time.time() + 30), K=180, beta=10.0)
        if c5_l < best_c5:
            best_c5, best_h = c5_l, h_l

    # Phase 5: Try different N (upsample)
    d5 = deadline
    for N_try in [N0 + 20, N0 + 40, N0 + 60]:
        if time.time() > d5 - 60 or N_try < 200:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = binary_fix(h_int, N_try)
        h_s, c5_s = steepest_descent(h_int, N_try, d5 - 40, rng, n_try=40)
        c5_s = compute_c5(h_s, N_try)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s
        if time.time() < d5 - 10:
            h_c = h_s.astype(float) + rng.randn(N_try) * 0.03
            h_c = project_sum(h_c, N_try / 2.0)
            h_l, c5_l = lbfgs_opt(h_c, N_try, d5 - 5, K=150, beta=8.0)
            if c5_l < best_c5:
                best_c5, best_h = c5_l, h_l

    # Final polish
    if time.time() < deadline - 3:
        h_f = best_h.astype(float) + rng.randn(len(best_h)) * 0.02
        h_f = project_sum(h_f, len(best_h) / 2.0)
        h_f, c5_f = lbfgs_opt(h_f, len(best_h), deadline - 1, K=200, beta=12.0)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
