def exact_1d_bisect(corr, coef, lo, hi):
    # Find delta in [lo, hi] minimizing max(corr + delta*coef)
    tol = 1e-12
    for _ in range(60):  # enough for double precision
        mid = (lo + hi) / 2
        vals = corr + mid * coef
        mx = vals.max()
        active = vals >= mx - 1e-10  # lags at the max (within tol)
        max_coef = coef[active].max()
        min_coef = coef[active].min()
        if max_coef > tol:
            lo = mid
        elif min_coef < -tol:
            hi = mid
        else:
            return mid
    return (lo + hi) / 2

        best_d = bisect_1d(corr, coef, lo, hi)
        if abs(best_d) > 1e-13:
            h[i] = np.clip(h_old + best_d, 0.0, 1.0)
            corr += best_d * coef
            improved = True
    return improved

def anti_periodic_init(n):
    """h[i + n/2] = 1 - h[i], sum auto = n/2"""
    h = np.zeros(n)
    half = n // 2
    for i in range(half):
        v = 0.5 + 0.4 * np.sin(2.0 * np.pi * i / half * 3.0)
        h[i] = np.clip(v, 0.05, 0.95)
        h[i + half] = 1.0 - h[i]
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(128)
        best_h[:64] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent (40% budget)
    t1 = t0 + budget_s * 0.40
    corr = compute_corr(best_h)
    while time.time() < t1:
        if not cd_sweep(best_h, corr, t1):
            break
    best_c5 = compute_c5(best_h)

    # Phase 2: Basin hopping with diverse moves (35% budget)
    t2 = t0 + budget_s * 0.75
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iters = 0
    while time.time() < t2:
        n_iters += 1
        cool = max(1.0 - n_iters / 300.0, 0.03)
        r = np.random.rand()
        if r < 0.35:
            step = 0.03 + 0.15 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.55:
            h_p = work_h.copy()
            i1 = np.random.randint(n)
            i2 = np.random.randint(n)
            if i1 != i2:
                d = np.random.uniform(0.05, 0.4) * cool
                h_p[i1] += d
                h_p[i2] -= d
        elif r < 0.72:
            # Anti-periodic mix
            h_ap = anti_periodic_init(n)
            h_p = 0.4 * h_ap + 0.6 * work_h
        else:
            # Block perturb
            h_p = work_h.copy()
            blk = np.random.randint(2, max(3, n // 6))
            st = np.random.randint(0, n - blk)
            sh = np.random.uniform(-0.25, 0.25) * cool
            h_p[st:st + blk] += sh
        h_p = project_to_sum(h_p, n / 2.0)
        h_c = h_p.copy()
        corr_c = compute_corr(h_c)
        for _ in range(4):
            if time.time() > t2:
                break
            cd_sweep(h_c, corr_c, time.time() + 8)
        c5_new = compute_c5(h_c)
        temp = 0.002 * cool + 0.00005
        if c5_new < work_c5 or np.random.rand() < np.exp(-(c5_new - work_c5) / max(temp, 1e-10)):
            work_h, work_c5 = h_c, c5_new
            if work_c5 < best_c5:
                best_h, best_c5 = work_h.copy(), work_c5

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 3: Multi-resolution try (10% budget)
    t3 = t0 + budget_s * 0.90
    for n_try in [96, 128, 160, 192]:
        if n_try == n or time.time() > t3:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2.0)
        corr_i = compute_corr(h_i)
        c5_i = compute_c5(h_i)
        t_sub = min(time.time() + budget_s * 0.03, t3)
        while time.time() < t_sub:
            if not cd_sweep(h_i, corr_i, t_sub):
                break
        c5_i = compute_c5(h_i)
        if c5_i < best_c5:
            best_h, best_c5, n = h_i, c5_i, n_try

    # Phase 4: Final intensive CD (10% budget)
    t4 = t0 + budget_s * 0.995
    corr = compute_corr(best_h)
    while time.time() < t4:
        if not cd_sweep(best_h, corr, t4):
            break

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
