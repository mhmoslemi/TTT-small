import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=1000):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def slsqp_step(h, T, n, maxiter=300):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def fast_coord_descent(h, n, t_end, max_sweeps=20000, random_order=True):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * 2.0 / n)
    k_range = np.arange(-(n - 1), n)
    nc = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        changed = 0
        order = np.random.permutation(n) if random_order else np.arange(n)
        for i in order:
            coefs = np.zeros(nc)
            im = i - k_range
            ip = i + k_range
            v1 = (im >= 0) & (im < n)
            v2 = (ip >= 0) & (ip < n)
            coefs[v1] += (1.0 - h[im[v1]])
            coefs[v2] -= h[ip[v2]]
            lo, hi = 0.0, 1.0
            cur_max = np.max(corr)
            f_at_0 = cur_max
            f_at_1 = np.max(corr + coefs)
            if f_at_0 <= min(f_at_0, f_at_1) and abs(f_at_1 - f_at_0) < 1e-12:
                continue
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(corr + m1 * coefs) <= np.max(corr + m2 * coefs):
                    hi = m2
                else:
                    lo = m1
            d = (lo + hi) * 0.5
            if abs(d - h[i]) > 1e-11:
                h[i] = d
                corr += d * coefs
                changed += 1
        h = project_h(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
        if changed <= 1:
            break
    return h, best_c5

def block_perturb(h, n, scale):
    h_new = h.copy()
    w = np.random.randint(3, min(15, n // 4) + 1)
    start = np.random.randint(0, n - w)
    h_new[start:start + w] += scale * np.random.randn(w)
    return project_h(h_new)

def swap_perturb(h, n):
    h_new = h.copy()
    i, j = np.random.choice(n, 2, replace=False)
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return project_h(h_new)
