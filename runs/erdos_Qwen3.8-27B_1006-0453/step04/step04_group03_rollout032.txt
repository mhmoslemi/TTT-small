```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = 1.0 - h > 1e-12
            if not m.any():
                break
            h[m] = np.minimum(1.0, h[m] + (1.0 - h[m]) * (diff / (1.0 - h[m])[m].sum()))
        else:
            m = h > 1e-12
            if not m.any():
                break
            h[m] = np.maximum(0.0, h[m] * max(1.0 + diff / h[m].sum(), 0.0))
    return np.clip(h, 0.0, 1.0)

def smooth_max(v, T):
    m = v.max()
    return m + np.log(np.sum(np.exp(T * (v - m)))) / T

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def slsqp_step(h, n, T, maxiter=500, ftol=1e-13):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def coord_descent(h, n, t_end, n_sweeps=300):
    dx = 2.0 / n
    corr = corr_full(h)
    bc5 = corr.max() * dx
    bh = h.copy()

        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return bh, bc5
            ho = bh[i]
            if ho < 1e-12 or ho > 1.0 - 1e-12:
                continue
            lags = np.arange(-(n - 1), n)
            ip = i + lags
            ino = i - lags
            A = np.zeros(2 * n - 1)
            ma = (ip >= 0) & (ip < n)
            A[ma] = 1.0 - bh[ip[ma]]
            B = np.zeros(2 * n - 1)
            mb = (ino >= 0) & (ino < n)
            B[mb] = bh[ino[mb]]
            eff = A - B
            pm = np.abs(eff) > 1e-15
            if not pm.any():
                continue
            pi = np.where(pm)[0]
            unaff_mask = ~pm
            umax = corr[unaff_mask].max() * dx if unaff_mask.any() else -1e18
            lo_d, hi_d = -ho, 1.0 - ho
            def f1d(d):
                return max((corr[pi] + d * eff[pi]).max() * dx, umax)
            res = minimize_scalar(f1d, bounds=(lo_d, hi_d), method='bounded',
                                  options={'xatol': 1e-11})
            if abs(res.x) > 1e-12 and res.fun < bc5 - 1e-14:
                bh[i] = ho + res.x
                corr[pi] += res.x * eff[pi]
                bc5 = res.fun
                imp = True
        if not imp:
            break
    return bh, bc5

def fourier_perturb(h, n, rng, n_modes=3, scale=0.05):
    F = np.fft.rfft(h)
    idx = np.arange(len(F))
    # Perturb random frequency components (skip DC)
    selections = rng.choice(1, len(F) - 1, size=min(n_modes, len(F) - 1), replace=False) + 1
    for k in selections:
        F[k] += scale * rng.randn() * np.exp(1j * rng.uniform(0, 2 * np.pi))
    if len(F) > 1:
        for k in selections:
            F[-k] if -k > 0 else None  # handle conjugate
        # Actually for real signal, we need to handle conjugate pairs
        # rfft gives 0 to n//2, and the constraint is automatic with irfft
    h_new = np.fft.irfft(F, n=n)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = 1.0 - h > 1e-12
            if not m.any():
                break
            h[m] = np.minimum(1.0, h[m] + (1.0 - h[m]) * (diff / (1.0 - h[m])[m].sum()))
        else:
            m = h > 1e-12
            if not m.any():
                break
            h[m] = np.maximum(0.0, h[m] * max(1.0 + diff / h[m].sum(), 0.0))
    return np.clip(h, 0.0, 1.0)

def smooth_max(v, T):
    m = v.max()
    return m + np.log(np.sum(np.exp(T * (v - m)))) / T

def slsqp_obj(x, T, dx):
    x = np.clip(x, 0, 1)
    return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx

def slsqp_sum_con(x, target):
    return x.sum() - target

def slsqp_step(h, n, T, maxiter=500, ftol=1e-13):
    dx = 2.0 / n
    res = minimize(slsqp_obj, h, args=(T, dx), method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': slsqp_sum_con, 'args': (n / 2.0,)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

    return np.correlate(h, 1.0 - h, mode='full').astype(float)

            ho = bh[i]
            if ho < 1e-12 or ho > 1.0 - 1e-12:

            ip = i + lags
            ino = i - lags
            A = np.zeros(2 * n - 1)
            ma = (ip >= 0) & (ip < n)
            A[ma] = 1.0 - bh[ip[ma]]
            B = np.zeros(2 * n - 1)
            mb = (ino >= 0) & (ino < n)
            B[mb] = bh[ino[mb]]
            eff = A - B
            pm = np.abs(eff) > 1e-14
            if not pm.any():

            pi = np.where(pm)[0]
            unaff = ~pm
            umax = corr[unaff].max() * dx if unaff.any() else -1e18
            lo_d = -ho
            hi_d = 1.0 - ho
            d_vals = np.linspace(lo_d, hi_d, 11)

            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2)
        try:
            hc = slsqp_step(h_i, n_try, 400, maxiter=300, ftol=1e-13)
            hc, c5v = coord_descent(hc, n_try, time.time() + 4.0, n_sweeps=15)
            if c5v < best_c5:
                best_c5, best_h, n = c5v, hc.copy(), n_try
        except Exception:
            pass

    # Phase 5: Final polish (20%)
    t5 = t0 + budget_s * 0.94
    best_h, best_c5 = coord_descent(best_h, n, t5, n_sweeps=800)

    for T in [15000, 80000]:
        if time.time() > t0 + budget_s * 0.97:
            break
        try:
            hc = slsqp_step(best_h, n, T, maxiter=800, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, n_sweeps=200)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
