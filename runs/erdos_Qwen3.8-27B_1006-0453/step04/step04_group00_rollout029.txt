import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_topk(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(-corr, k)[:k]
    c = corr[idx]
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx

def project_h(h):
    n = len(h)
    target = n / 2.0
    for _ in range(20):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                return np.clip(h, 0, 1)
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                return np.clip(h, 0, 1)
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def lbfgs_opt(h, T, k, mu, maxiter):
    n = len(h)
    def obj(x):
        return smooth_topk(x, T, k) + mu * (x.sum() - n / 2.0) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-13})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=20000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stagnant = 0
    s_range = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            best_h[i] = 0.0
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            idx1 = i + s_range
            idx2 = i - s_range
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - best_h[idx1[v1]])
            coefs[v2] += (-best_h[idx2[v2]])
            def f1d(x):
                return np.max(corr + x * coefs)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-15})
            best_h[i] = res.x
            best_h = project_h(best_h)
        c5_try = compute_c5(best_h)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            stagnant = 0
        else:
            stagnant += 1
        if stagnant >= 25:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, amp_base, T_base, temp_base):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        amp = amp_base * (0.5 + cool)
        mod = it % 3
        if mod == 0:
            pert = amp * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 7)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = amp * 0.8 * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        else:
            mask = np.random.random(n) < 0.25
            pert = np.zeros(n)
            pert[mask] = amp * 1.5 * np.random.randn(mask.sum())
        h_pert = project_h(work_h + pert)
        T_val = T_base * (1 + 5 * cool)
        k_val = max(3, 6 - int(3 * cool))
        mu_val = 300 * (1 + 3 * cool)
        hc = lbfgs_opt(h_pert, T_val, k_val, mu_val, maxiter=300)
        c5v = compute_c5(hc)
        temp = temp_base * (cool + 0.1)
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
            work_h, work_c5 = hc, c5v
    return work_h, work_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B initialization with multi-scale
    for T, k, mu in [(30, 5, 80), (100, 8, 150), (500, 10, 300), (2000, 6, 500)]:
        if elapsed() > budget_s * 0.06:
            break
        hc = lbfgs_opt(best_h, T, k, mu, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Phase 2: Basin hopping (broad exploration)
    bh_h, bh_c5 = basin_hop(best_h, n, t0 + budget_s * 0.18, 40,
                            0.10, 500, 0.004)
    if bh_c5 < best_c5:
        best_h, best_c5 = bh_h, bh_c5

    # Phase 3: Main coordinate descent (bulk of time)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.62, max_sweeps=50000)

    # Phase 4: Second basin hopping (finer)
    bh_h, bh_c5 = basin_hop(best_h, n, t0 + budget_s * 0.72, 25,
                            0.04, 800, 0.002)
    if bh_c5 < best_c5:
        best_h, best_c5 = bh_h, bh_c5
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82, max_sweeps=50000)

    # Phase 5: Third basin hopping (very fine)
    bh_h, bh_c5 = basin_hop(best_h, n, t0 + budget_s * 0.86, 15,
                            0.02, 1500, 0.001)
    if bh_c5 < best_c5:
        best_h, best_c5 = bh_h, bh_c5

    # Phase 6: Final long coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=50000)

    # Phase 7: Final L-BFGS polish + micro CD
    if elapsed() < budget_s * 0.975:
        hc = lbfgs_opt(best_h, 5000, 4, 800, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=10000)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
