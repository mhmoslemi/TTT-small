import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, T, n):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c = corr * dx
    m = c.max()
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def project(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-13:
        return vc
    lo, hi = vc.min() - 2.0, 2.0 - vc.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=500):
    def obj(x):
        return smooth_obj(x, T, n)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_grad(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_grad}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(res.x, n)

def coord_descent(h, n, t_end, max_sweeps=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            def f1d(hi):
                best_h[i] = hi
                c = compute_c5(best_h)
                best_h[i] = saved
                return c
            saved = best_h[i]
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            hi_new = res.x
            if abs(hi_new - saved) < 1e-11:
                continue
            best_h[i] = hi_new
            h_try = project(best_h.copy(), n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
            else:
                best_h[i] = saved
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= 4:
            break
    return best_h, best_c5
