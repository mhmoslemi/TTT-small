    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / len(h))

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * ((target - s) / tot)
        else:
            frac = target / s if s > 0 else 0
            h = np.clip(h * frac, 0, 1)
    # Final exact fix
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

            self_idx = np.where(aff == 2*i)[0]

            mask[m_lo:m_hi+1] = False
            unaff_max = np.max(corr[mask]) if mask.any() else -np.inf

                v = base + d * a_coef
                if len(self_idx) > 0:
                    v = v.copy()
                    v[self_idx[0]] = base[self_idx[0]] + d * self_a - d*d
                mv = np.max(v)
                return mv if mv > unaff_max else unaff_max

            grid = np.linspace(0, 1, 40)
            vals = np.array([fval(t) for t in grid])
            bi = np.argmin(vals)

                corr = new_corr

        else:

        if n == n_try and time.time() < t0 + budget_s * 0.92:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.93,
                                            rng, max_sweeps=40)

    # Phase 6: Try alternative starting points and quick optimizations
    if time.time() < t0 + budget_s * 0.95:
        for alt_seed in [seed + 1, seed + 7, seed + 13]:
            if time.time() > t0 + budget_s * 0.95:
                break
            r2 = np.random.default_rng(alt_seed)
            h_alt = make_initial(n, r2)
            for T in [500, 5000]:
                if time.time() > t0 + budget_s * 0.95:
                    break
                h_alt = slsqp_polish(h_alt, n, T, maxiter=400)
            h_alt, c5_alt = coord_descent(h_alt, n, t0 + budget_s * 0.955, rng,
                                          max_sweeps=20)
            if c5_alt < best_c5:
                best_h, best_c5 = h_alt, c5_alt

    # Phase 7: Final polish
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.975, rng,
                                        max_sweeps=30)
    if time.time() < t0 + budget_s * 0.995:
        for T in [20000, 100000]:
            if time.time() > t0 + budget_s * 0.995:
                break
            hc = slsqp_polish(best_h, n, T, maxiter=1500)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
