```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            return h
        h[mask] += diff / cnt
    else:
        mask = h > 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            return h
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def soft_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    ev = np.exp(T * (corr - m))
    p = ev / ev.sum()
    grad = np.zeros(n)
    for i in range(n):
        # sum over lags where 0 <= i+l < n: p[n-1+l]*(1-h[i+l])
        # l range: max(-i, -(n-1)) to min(n-1-i, n-1)
        lo1 = max(-i, -(n-1))
        hi1 = min(n-1-i, n-1)
        if lo1 <= hi1:
            lags = np.arange(lo1, hi1+1)
            grad[i] += np.dot(p[n-1+lags], 1.0 - h[i+lags])
        lo2 = max(-(n-1), i-n+1)
        hi2 = min(n-1, i)
        if lo2 <= hi2:
            lags = np.arange(lo2, hi2+1)
            grad[i] -= np.dot(p[n-1+lags], h[i-lags])
    grad *= dx
    return grad

def grad_desc(h, n, t_end, T_start=100, T_end=5000, n_steps=200):
    h = h.copy()
    for step in range(n_steps):
        if time.time() > t_end:
            break
        frac = step / max(n_steps - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        g = soft_grad(h, T)
        # Backtracking line search
        best_alpha = 0.01
        best_f = soft_obj(h, T)
        for alpha in [0.1, 0.05, 0.01, 0.005, 0.001]:
            h_new = project(h - alpha * g, n)
            f_new = soft_obj(h_new, T)
            if f_new < best_f:
                best_alpha = alpha
                best_f = f_new
                h = h_new
                break
    return h

def coord_sweep(h, n, t_end, rng, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_save = best_h[i]
            best_h[i] = 0.0
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
            best_h[i] = h_save
            d = np.zeros(2*n-1)
            for k in range(-n+1, n):
                idx = n - 1 + k
                dk = 0.0
                jk = i - k
                if 0 <= jk < n:
                    dk += (1.0 - best_h[jk])
                j2 = i + k
                if 0 <= j2 < n:
                    dk += (-best_h[j2])
                d[idx] = dk
            def obj(t):
                return np.max(corr_base + t * d) * (2.0/n)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
            if abs(res.x - h_save) > 1e-12:
                best_h[i] = res.x
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def anneal(h, n, t_end, rng, n_iters=500):
    best = h.copy()
    best_c5 = compute_c5(best)
    work = h.copy()
    work_c5 = best_c5
    T0 = 0.05
    T1 = 0.001
    sigma0 = 0.15
    sigma1 = 0.02
    for it in range(n_iters):
        if time.time() > t_end:
            break
        frac = it / max(n_iters - 1, 1)
        T_sa = T0 * (T1 / T0) ** frac
        sigma = sigma0 * (sigma1 / sigma0) ** frac
        # Propose
        nv = rng.standard_normal(n)
        nv -= nv.mean()
        pert = project(work + sigma * nv, n)
        pert_c5 = compute_c5(pert)
        if pert_c5 < work_c5 or rng.random() < np.exp(-min((pert_c5 - work_c5) / T_sa, 50)):
            work, work_c5 = pert, pert_c5
            if work_c5 < best_c5:
                best, best_c5 = work.copy(), work_c5
        # Gradient polish every 30 steps
        if it % 30 == 29:
            try:
                gd = grad_desc(work, n, time.time() + 2.0, T_start=500, T_end=2000, n_steps=15)
                gdc5 = compute_c5(gd)
                if gdc5 < work_c5:
                    work, work_c5 = gd, gdc5
                    if work_c5 < best_c5:
                        best, best_c5 = work.copy(), work_c5
            except Exception:
                pass

        if it % 100 == 99 and frac < 0.8:
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            return h
        h[mask] += diff / cnt
    else:
        mask = h > 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            return h
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def soft_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    ev = np.exp(T * (corr - m))
    p = ev / ev.sum()
    grad = np.zeros(n)
    for i in range(n):
        lo1 = max(-i, -(n-1))
        hi1 = min(n-1-i, n-1)
        if lo1 <= hi1:
            lags = np.arange(lo1, hi1+1)
            grad[i] += np.dot(p[n-1+lags], 1.0 - h[i+lags])
        lo2 = max(-(n-1), i-n+1)
        hi2 = min(n-1, i)
        if lo2 <= hi2:
            lags2 = np.arange(lo2, hi2+1)
            grad[i] -= np.dot(p[n-1+lags2], h[i-lags2])
    grad *= dx
    return grad

def grad_desc(h, n, t_end, T_start=100, T_end=5000, n_steps=150):
    h = h.copy()
    for step in range(n_steps):
        if time.time() > t_end:
            break
        frac = step / max(n_steps - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        g = soft_grad(h, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        best_alpha = 0.0
        best_f = soft_obj(h, T)
        alpha0 = 2.0 / (T * gnorm + 1e-8)
        for alpha in [alpha0, alpha0*0.5, alpha0*0.25, alpha0*0.1, 0.01, 0.005, 0.001]:
            h_new = project(h - alpha * g, n)
            f_new = soft_obj(h_new, T)
            if f_new < best_f:
                best_alpha = alpha
                best_f = f_new
                h = h_new
                break
    return h

def coord_sweep(h, n, t_end, rng, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_save = best_h[i]
            best_h[i] = 0.0
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
            best_h[i] = h_save
            d = np.zeros(2*n-1)
            for k in range(-n+1, n):
                idx = n - 1 + k
                dk = 0.0
                jk = i - k
                if 0 <= jk < n:
                    dk += (1.0 - best_h[jk])
                j2 = i + k
                if 0 <= j2 < n:
                    dk += (-best_h[j2])
                d[idx] = dk
            def obj(t):
                return np.max(corr_base + t * d) * dx
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                   options={'xatol': 1e-14})
            if abs(res.x - h_save) > 1e-12:
                best_h[i] = res.x
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def anneal(h, n, t_end, rng, n_iters=500):
    best = h.copy()
    best_c5 = compute_c5(best)
    work = h.copy()
    work_c5 = best_c5
    T0 = 0.04
    T1 = 0.0005
    s0 = 0.12
    s1 = 0.015
    for it in range(n_iters):
        if time.time() > t_end:
            break

        T_sa = T0 * (T1 / T0) ** frac
        sigma = s0 * (s1 / s0) ** frac
        mode = rng.random()
        if mode < 0.5:
            nv = rng.standard_normal(n)
            nv -= nv.mean()
            pert = project(work + sigma * nv, n)
        elif mode < 0.75:

            idx = rng.choice(n, size=n_mod, replace=False)
            delta = rng.uniform(-sigma*3, sigma*3, n_mod)
            pert = work.copy()
            pert[idx] += delta
            pert = project(pert, n)
        else:
            corr = np.correlate(work, 1.0 - work, mode='full')
            top_k = np.argsort(corr)[-5:]
            k_star = top_k[rng.integers(0, len(top_k))] - (n-1)
            pert = work.copy()
            if 0 < k_star < n:

                        pert[j] -= sigma
            else:
                nv = rng.standard_normal(n)
                nv -= nv.mean()
                pert = project(work + sigma * nv, n)
            pert = project(pert, n)
        pc5 = compute_c5(pert)
        if pc5 < work_c5 or rng.random() < np.exp(-min((pc5 - work_c5) / T_sa, 50)):
            work, work_c5 = pert, pc5
```
