import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smaxobj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s))) / T) * dx

def smaxgrad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(T * (corr - m), -500, 500)
    e = np.exp(s)
    p = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            cnt = n - lag
            if cnt > 0:
                grad[:cnt] += pk * (1.0 - h[lag:lag + cnt])
                grad[lag:] += pk * (-h[:cnt])
        else:
            nl = -lag
            cnt = n - nl
            if cnt > 0:
                grad[nl:] += pk * (1.0 - h[:cnt])
                grad[:cnt] += pk * (-h[nl:])
    return grad * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lbfgs_opt(h, T, n, t_end, maxiter=3000):
    def f(x):
        return smaxobj(x, T)
    def fprime(x):
        return smaxgrad(x, T)
    bnds = [(0.0, 1.0)] * n
    res = minimize(f, h, jac=fprime, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-16,
                            'maxls': 40, 'maxfun': 20000})
    return proj(res.x, n)

def coord_single(h, n, i):
    h_save = h[i]
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k] = c
    k0 = n - 1
    def f1d(t):
        cv = corr_base + t * coef
        cv[k0] = corr_base[k0] + t * (1.0 - t)
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binarize_try(h, n):
    c5 = compute_c5(h)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    if h_bin.sum() != n // 2:
        excess = h_bin.sum() - n // 2
        if excess > 0:
            thresh = np.sort(h)[::-1][int(n/2)]
            h_frac = np.where(h > thresh, 1.0, 0.0)
        else:
            thresh = np.sort(h)[int(n/2)]
            h_frac = np.where(h >= thresh, 1.0, 0.0)
            if h_frac.sum() > n // 2:
                idx = np.where(h_frac == 1.0)[0]
                to_zero = idx[np.argsort(h[idx])[:h_frac.sum() - int(n/2)]]
                h_frac[to_zero] = 0.0
        h_bin = h_frac
    c5_bin = compute_c5(h_bin)
    if c5_bin < c5:
        return h_bin, c5_bin
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 180
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing T
    for T in [50, 300, 1500, 8000, 40000]:
        if time.time() > t0 + budget_s * 0.10:
            break
        hc = lbfgs_opt(best_h, T, n, t0 + budget_s * 0.10, maxiter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent
    end_cd = t0 + budget_s * 0.20
    if time.time() < end_cd:
        best_h, best_c5 = coord_descent(best_h, n, end_cd, max_sweeps=200)

    # Phase 3: Binarization attempt
    best_h, best_c5 = binarize_try(best_h, n)

    # Phase 4: Basin hopping with L-BFGS-B
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 400
    bh_end = t0 + budget_s * 0.55
    for it in range(n_bh):
        if time.time() > bh_end:
            break
        cool = 1.0 - 0.85 * (it / float(n_bh))
        step = 0.015 + 0.15 * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = proj(h_pert, n)
        T_bh = 2000
        hc = lbfgs_opt(h_pert, T_bh, n, t0 + budget_s * 0.55, maxiter=400)
        c5v = compute_c5(hc)
        temp = 0.004 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Coordinate descent refinement
    end_cd2 = t0 + budget_s * 0.68
    if time.time() < end_cd2:
        best_h, best_c5 = coord_descent(best_h, n, end_cd2, max_sweeps=200)

    # Phase 6: Binarize again
    best_h, best_c5 = binarize_try(best_h, n)

    # Phase 7: Try different resolutions
    for n_try in [90, 100, 120, 140, 160]:
        if time.time() > t0 + budget_s * 0.84:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = proj(h_interp, n_try)
        for T in [500, 5000, 50000]:
            if time.time() > t0 + budget_s * 0.86:
                break
            hc = lbfgs_opt(h_interp, T, n_try, t0 + budget_s * 0.86, maxiter=1500)
            c5v = compute_c5(hc)
            hc, c5v = binarize_try(hc, n_try)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                n = n_try

    # Phase 8: Final L-BFGS-B with very high T + coordinate descent
    T_final = 100000
    if time.time() < t0 + budget_s * 0.93:
        hc = lbfgs_opt(best_h, T_final, n, t0 + budget_s * 0.93, maxiter=3000)
        c5v = compute_c5(hc)

    end_final = t0 + budget_s * 0.97
    if time.time() < end_final:
        best_h, best_c5 = coord_descent(best_h, n, end_final, max_sweeps=100)

    # Final projection and evaluation
    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
