                                'jac':lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    if res.success or res.fun < smooth_c5(h, T):
        return project(np.clip(res.x, 0, 1), n)
    return h

def coord_opt(h, n, i):
    # Optimize h[i] given all other h fixed
    dx = 2.0 / n
    h_save = h[i]
    # c_k as function of t = h[i]:
    # c_{n-1+k} (k>=1): contribution is (1-t)*[i+k<n] + (-h[i-k])*... wait
    # Let me compute: if h[i] = t, then
    # corr[n-1] = sum_j h[j](1-h[j]) → term at j=i: t(1-t)
    # corr[n-1+k] = sum_{j=0}^{n-1-k} h[j](1-h[j+k])
    #   if i < n-k: term j=i: t*(1-h[i+k])
    #   if i >= k and i-k < n-k: term j=i-k: h[i-k]*(1-t)
    # corr[n-1-k] = sum_{j=k}^{n-1} h[j](1-h[j-k])
    #   if i >= k: term j=i: t*(1-h[i-k])
    #   if i < n-k: term j=i+k: h[i+k]*(1-t)
    
    # So corr[n-1+k](t) = base_plus[k] + t*(1-h[i+k])*[i<n-k] - h[i-k]*t*[k<=i<n-k+i]
    # Let me just compute the coefficient of t and the constant for each k.
    
    # For k=0: corr[n-1](t) = base0 - 2*t^2 + t  where base0 = sum_{j≠i} h[j](1-h[j])
    #   Actually: sum_j h[j](1-h[j]) = sum_{j≠i} h[j](1-h[j]) + t(1-t)
    #   = (corr[n-1] - h_save*(1-h_save)) + t - t^2
    
    # For k>=1: corr[n-1+k](t) = A_k + t * B_k
    #   A_k = sum_{j≠i, j<n-k, j+k≠i} h[j](1-h[j+k]) + (if i>=k and i-k<n-k and i-k≠i) h[i-k]*(1-h_save)
    #   This is getting complicated. Let me just evaluate at t=0 and t=1 and interpolate.
    
    # Actually, each corr[n-1±k] is linear in t (for k>=1) or quadratic (for k=0).
    # So the max of (2n-1) functions, each linear or quadratic in t, can be minimized
    # by finding the lower envelope.
    
    # Simpler: just use minimize_scalar on the exact C5 as function of t.
    h[i] = 0.0
    corr_at_0 = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = 1.0
    corr_at_1 = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    
    # For k=0: corr[n-1](t) = corr_at_0[n-1] + t * (corr_at_1[n-1] - corr_at_0[n-1])
    #   But this is only linear if we linearly interpolate. Actually:
    #   corr[n-1](t) = sum_j h[j](1-h[j]) with h[i]=t
    #   = (sum_{j≠i} h[j](1-h[j])) + t(1-t)
    #   corr_at_0[n-1] = sum_{j≠i} h[j](1-h[j]) + 0 = sum_{j≠i} h[j](1-h[j])
    #   corr_at_1[n-1] = sum_{j≠i} h[j](1-h[j]) + 0 = same? No:
    #   When h[i]=0: term is 0*(1-0) = 0
    #   When h[i]=1: term is 1*(1-1) = 0
    #   So corr_at_0[n-1] = corr_at_1[n-1]! The k=0 term is t(1-t) added.
    
    # For k>=1: corr is linear in t:
    #   corr[n-1+k](t) = corr_at_0[n-1+k] + t * (corr_at_1[n-1+k] - corr_at_0[n-1+k])
    #   Because the only t-dependent terms are linear (t appears once in each product).
    
    # So C5(t) = max of {quadratic at k=0, linears at k≠0} * dx
    # The k=0 term: c0(t) = corr_at_0[n-1] + t - t^2
    # Other terms: c_k(t) = corr_at_0[m] + t * (corr_at_1[m] - corr_at_0[m])
    
    def c5_of_t(t):
        if t == 0:
            corr = corr_at_0.copy()
        elif t == 1:
            corr = corr_at_1.copy()
        else:
            corr = corr_at_0 + t * (corr_at_1 - corr_at_0)
        corr[n-1] = corr_at_0[n-1] + t - t*t  # exact quadratic
        return np.max(corr) * dx
    
    res = minimize_scalar(c5_of_t, bounds=(0, 1), method='bounded', options={'xatol': 1e-14})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=50, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()

                    improved = True

    return best_h, best_c5

def perturb_refine(h, n, rng, t_end, T=500, step=0.05):

    best = h.copy()

        temp = 0.003 * cool + 0.0002

            work, work_c5 = hc, cv
        if work_c5 < best_c5 - 1e-14:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

    x_new = np.linspace(0, 2, n_new, endpoint=False)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    g = w[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = w[n - 1 + k]
        wm = w[n - 1 - k]
        if wp < 1e-14 and wm < 1e-14:
            continue
        g[:n - k] += wp * (1.0 - h[k:])
        g[k:] -= wp * h[:-k]
        g[k:] += wm * (1.0 - h[:-k])
        g[:n - k] -= wm * h[k:]
    return g * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = h < 1.0
            if not free.any():
                break
            room = (1.0 - h[free]).sum()
            if room < 1e-15:
                break
            frac = min(1.0, diff / room)
            h[free] += (1.0 - h[free]) * frac
        else:
            free = h > 0.0
            if not free.any():
                break
            tot = h[free].sum()
            if tot < 1e-15:
                break
            frac = min(1.0, (-diff) / tot)
            h[free] -= h[free] * frac
    h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_solve(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def jac(x):
        return smooth_c5_grad(x, T)
    def con(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    res = minimize(obj, h, jac=jac, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    if hasattr(res, 'x') and res.x is not None:
        cand = project(np.clip(res.x, 0, 1), n)
        if compute_c5(cand) < compute_c5(h):
            return cand
    return h

def coord_opt(h, n, i):
    dx = 2.0 / n
    hs = h[i]
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = hs
    d = c1 - c0
    c0mid = c0[n - 1]
    def f(t):
        v = c0mid + t - t * t
        rest_max = np.max(c0 + t * d)
        rest_max = max(rest_max, 0.0)
        return max(v, rest_max) * dx
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=60, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-12:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def basin_hop(h, n, t_end, rng, n_max=300, step0=0.05):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_max):
        if time.time() > t_end:
            break
        cool = max(0.05, 1.0 - 0.9 * it / max(n_max, 1))
        step = step0 * cool
        pert = work + step * rng.standard_normal(n)
        pert = project(pert, n)
        try:
            hc = slsqp_solve(pert, 400, n, maxiter=200)
            cv = compute_c5(hc)
        except Exception:
            hc, cv = pert, compute_c5(pert)
        temp = 0.003 * cool + 0.0002
        if cv < work_c5 or rng.random() < np.exp(-(cv - work_c5) / temp):
            work, work_c5 = hc, cv
        if work_c5 < best_c5 - 1e-14:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interp_h(h_old, n_new):
    xo = np.linspace(0, 2, len(h_old), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(xn, xo, h_old), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with analytical gradient, adaptive T
    for T in [5, 30, 150, 800, 4000, 20000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        hc = slsqp_solve(best_h, T, n, maxiter=600)
        cv = compute_c5(hc)
        if cv < best_c5 - 1e-15:
            best_h, best_c5 = hc, cv

    # Phase 2: Coordinate descent
    t2 = t0 + budget_s * 0.28
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=50, rng=rng)

    # Phase 3: SLSQP polish at high T
    for T in [10000, 50000]:
        if time.time() - t0 > budget_s * 0.38:
            break
        hc = slsqp_solve(best_h, T, n, maxiter=800)
        cv = compute_c5(hc)
        if cv < best_c5 - 1e-15:
            best_h, best_c5 = hc, cv

    # Phase 4: Basin hopping with SLSQP inner
    t4 = t0 + budget_s * 0.62
    best_h, best_c5 = basin_hop(best_h, n, t4, rng, n_max=200, step0=0.04)

    # Phase 5: Coordinate descent after global search
    t5 = t0 + budget_s * 0.78
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=60, rng=rng)

    # Phase 6: Multi-resolution
    for n_try in [120, 160, 200, 320, 400]:
        if time.time() - t0 > budget_s * 0.90:
            break
        hi = interp_h(best_h, n_try)
        for T in [300, 5000]:
            if time.time() - t0 > budget_s * 0.93:
                break
            hc = slsqp_solve(hi, T, n_try, maxiter=500)
            cv = compute_c5(hc)
            if cv < best_c5 - 1e-14:
                best_h, best_c5, n = hc, cv, n_try
                hi = interp_h(best_h, n_try)
        if n == n_try:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95,
                                            max_sweeps=20, rng=rng)

    # Phase 7: Final coordinate descent + SLSQP
    if time.time() - t0 < budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96,
                                        max_sweeps=40, rng=rng)
    if time.time() - t0 < budget_s * 0.99:
        hc = slsqp_solve(best_h, 80000, n, maxiter=1200)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
