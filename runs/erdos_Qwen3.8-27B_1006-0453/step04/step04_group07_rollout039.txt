def binary_push(h, n, threshold=0.15):
    """Push values close to 0 or 1 to exactly 0 or 1."""
    h2 = h.copy()
    h2[h2 < threshold] = 0.0
    h2[h2 > 1 - threshold] = 1.0
    return project(h2, n)

def basin_hop_v2(h, c5, n, t_end, max_iters=300):

        cool = max(0.03, 1.0 - 0.9 * frac)
        # Diverse perturbation types
        rtype = np.random.rand()

            step = 0.02 + 0.15 * cool
            h_p = work_h + step * np.random.randn(n)
        elif rtype < 0.5:

            block_size = np.random.randint(3, 15)
            for _ in range(np.random.randint(1, 4)):

                h_p = work_h + 0.3 * cool * np.random.randn(block_size)
                h_p[start:start+block_size] += 0.3 * cool * np.random.randn(block_size)
            h_p = work_h.copy()
            for _ in range(np.random.randint(1, 4)):
                start = np.random.randint(0, n - block_size)
                h_p[start:start+block_size] += (0.05 + 0.2*cool) * np.random.randn(block_size)
        elif rtype < 0.7:
            # Flip entries
            nflip = np.random.randint(2, max(3, n // 20))
            idx = np.random.choice(n, nflip, replace=False)
            h_p = work_h.copy()
            h_p[idx] = 1.0 - h_p[idx]
        elif rtype < 0.85:
            # Shift
            shift = np.random.randint(1, 4) * np.random.choice([-1, 1])
            h_p = np.roll(work_h, shift)
            h_p += 0.01 * np.random.randn(n)
        else:
            # Random restart toward binary
            h_p = work_h.copy()
            h_p[np.random.rand(n) < 0.1] = np.random.rand(np.sum(np.random.rand(n) < 0.1))
        
        h_p = project(h_p, n)
        
        # Quick SLSQP
        T_use = int(100 + 2000 * cool)
        try:
            hc = minimize(lambda x: smooth_c5(x, T_use, n), h_p,

                         options={'maxiter': 200, 'ftol': 1e-12})
            hc = project(np.clip(hc.x, 0, 1), n)
            c5v = compute_c5(hc)
        except Exception:
            continue
        
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    return best_h, best_c5

def slsqp_opt(h0, T, n, maxiter=500, ftol=1e-14):
    res = minimize(lambda x: smooth_c5(x, T, n), h0,
                   method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def interp_h(h, n_new):

    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = np.float64(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(np.float64(T) * (corr - m)))) / T
    return val * dx

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def coord_obj(v, corr, old, coeff, c0_idx, c0_base):
    cn = np.add(corr, (v - old) * coeff, out=corr.copy())
    cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
    return np.max(cn)

def slsqp_opt(h0, T, n, maxiter=500, ftol=1e-14):
    res = minimize(smooth_c5, h0, args=(T, n),
                   method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

            break

            idx2 = i + ks

            v2 = (idx2 >= 0) & (idx2 < n)

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def coord_obj(v, corr, old, coeff, c0_idx, c0_base):
    cn = corr.copy()
    cn += (v - old) * coeff
    cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
    return np.max(cn)

def slsqp_opt(h0, T, n, maxiter=500, ftol=1e-14):
    res = minimize(smooth_c5, h0, args=(T, n),
                   method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def exact_coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            v0f = coord_obj(0.0, corr, old, coeff, c0_idx, c0_base)
            v1f = coord_obj(1.0, corr, old, coeff, c0_idx, c0_base)
            res = minimize_scalar(coord_obj, bounds=(0.0, 1.0),
                                  args=(corr, old, coeff, c0_idx, c0_base),
                                  method='bounded', options={'xatol': 1e-10})
            best_v, best_f = res.x, res.fun
            if v0f < best_f:
                best_v, best_f = 0.0, v0f
            if v1f < best_f:
                best_v, best_f = 1.0, v1f
            if best_f * dx < best_c5 - 1e-15:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            break
    return best_h, best_c5

def pair_descent(h, n, t_end, max_sweeps=100):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            j = (i + 1) % n
            old_i, old_j = best_h[i], best_h[j]
            di = np.zeros(2*n-1); dj = np.zeros(2*n-1)
            ii1 = i - ks; ii2 = i + ks
            jj1 = j - ks; jj2 = j + ks
            m1i = (ii1>=0)&(ii1<n); m2i = (ii2>=0)&(ii2<n)
            m1j = (jj1>=0)&(jj1<n); m2j = (jj2>=0)&(jj2<n)
            di[m1i] += 1.0-best_h[ii1[m1i]]; di[m2i] -= best_h[ii2[m2i]]
            dj[m1j] += 1.0-best_h[jj1[m1j]]; dj[m2j] -= best_h[jj2[m2j]]
            c0b = corr[c0_idx]
            for vi, vj in [(0,0),(1,1),(0,1),(1,0)]:
                cn = corr + (vi-old_i)*di + (vj-old_j)*dj
                cn[c0_idx] = c0b+(vi-old_i)*(1-vi-old_i)+(vj-old_j)*(1-vj-old_j)
                f = np.max(cn)*dx
                if f < best_c5 - 1e-15:
                    best_h[i], best_h[j] = vi, vj
                    best_c5 = f
                    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)

    x_new = np.linspace(0, 2, n_new, endpoint=False)

def basin_hop_v2(h, c5, n, t_end, max_iters=400):

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    return x.sum() - n / 2.0

    cn = corr.copy()
    cn += (v - old) * coeff
    cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
    return np.max(cn)
