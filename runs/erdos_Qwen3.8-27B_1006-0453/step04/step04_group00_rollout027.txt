def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e8:
        return m * dx
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def subgrad_top(h, alpha, n_topk, momentum_v=None, beta=0.5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if momentum_v is not None:
        grad = beta * momentum_v + (1 - beta) * grad
    h_new = h - alpha * grad
    return project_h(h_new), grad

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):

        improved = False

            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):

                jp = i + k
                jm = i - k

    else:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-n_pen:]
        val += flat_pen * np.std(top)
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def subgrad_top(h, alpha, n_topk, mom_v=None, beta=0.4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if mom_v is not None:
        grad = beta * mom_v + (1.0 - beta) * grad
    return project_h(h - alpha * grad), grad

def slsqp(h, T, n, maxiter=800, ftol=1e-15, flat_pen=0.0):
    try:
        res = minimize(lambda x: smooth_c5(x, T, flat_pen), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=200, n_top=20):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi, use_targeted=False):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if use_targeted and r < 0.3:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-5:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n-k] += (work_h[k:n] - 0.5)
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n+k] - 0.5)
            pert *= scale * 0.4 * cool
        elif r < 0.4:
            f = np.random.randint(1, 12)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.7 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.6:
            mask = np.random.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = scale * 1.5 * cool * np.random.randn(max(int(mask.sum()), 1))
        elif r < 0.8:
            blen = np.random.randint(max(n // 10, 2), max(n // 3, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs+blen] = scale * cool * np.random.randn(blen)
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp(h_pert, T, n, 400, 1e-14)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.00015
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_at_n(h_init, n, t_end, tag=""):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # SLSQP warmup
    if t_end - time.time() > 40:
        for T in [100, 1000, 10000, 100000]:
            if t_end - time.time() < 20:
                break
            hc = slsqp(best_h, T, n, 500, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Momentum subgradient + SLSQP
    if t_end - time.time() > 50:
        mom_v = np.zeros(n)
        for it in range(40):
            if t_end - time.time() < 15:
                break
            cool = 1.0 - it / 40.0
            alpha = 0.002 + 0.005 * cool
            n_topk = 5 + int(12 * cool)
            hs, g = subgrad_top(best_h, alpha, n_topk, mom_v, beta=0.4)
            mom_v = g
            T = 500 + 5000 * cool
            hc = slsqp(hs, T, n, 350, 1e-14, flat_pen=0.05 * cool)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Basin hopping with targeted perturbations
    if t_end - time.time() > 60:
        bh_h, bh_c5 = basin_hop(best_h, n, t_end - (t_end - time.time()) * 0.3,
                                40, 0.12, 500, 5000, use_targeted=True)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Coordinate descent
    if t_end - time.time() > 30:
        best_h, best_c5 = coord_descent(best_h, n, t_end - (t_end - time.time()) * 0.2,
                                        100, 20)
    # Fine basin hopping
    if t_end - time.time() > 30:
        bh_h, bh_c5 = basin_hop(best_h, n, t_end - (t_end - time.time()) * 0.4,
                                25, 0.04, 2000, 8000, use_targeted=False)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Final coord descent + polish
    if t_end - time.time() > 15:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 5, 60, 30)
    if t_end - time.time() > 10:
        hc = slsqp(best_h, 1000000, n, 800, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(160)
        init_h[:80] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Main optimization at parent n
    t_phase1_end = t0 + budget_s * 0.55
    best_h, best_c5 = optimize_at_n(best_h, n, t_phase1_end)

    # Try a second resolution if time allows
    if t_end - time.time() > budget_s * 0.35:
        n2 = 200
        h2 = np.zeros(n2)
        h2[:n2 // 2] = 1.0
        # Interpolate from best solution for better init
        h2_interp = np.interp(np.linspace(0, 2, n2, endpoint=False),
                              np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2_interp)
        t_phase2_end = t0 + budget_s * 0.82
        h2_opt, c5_2 = optimize_at_n(h2, n2, t_phase2_end)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt
            n = n2

    # Try a third resolution (finer)
    if t_end - time.time() > budget_s * 0.22 and n != 240:
        n3 = 240
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = project_h(h3)
        t_phase3_end = t0 + budget_s * 0.95
        h3_opt, c5_3 = optimize_at_n(h3, n3, t_phase3_end)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3_opt
            n = n3

    # Final polish at best n
    if t_end - time.time() > 10:
        hc = slsqp(best_h, 5000000, n, 1500, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if t_end - time.time() > 8:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, 40, 30)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
