import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h0, T, n, maxiter=400):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    n_corr = 2 * n - 1
    j_lags = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            va = (i - j_lags >= 0) & (i - j_lags < n)
            vb = (i + j_lags >= 0) & (i + j_lags < n)
            coef = np.zeros(n_corr)
            coef[va] += 1.0 - best_h[i - j_lags[va]]
            coef[vb] -= best_h[i + j_lags[vb]]
            cur_max = np.max(corr)
            active = corr >= cur_max - 1e-12
            if active.any():
                ac = coef[active]
                if ac.size > 0 and ac.min() > -1e-12 and ac.max() < 1e-12:
                    continue
            def obj_fn(v):
                return np.max(corr + (v - old) * coef)
            fv0, fv1 = obj_fn(0.0), obj_fn(1.0)
            best_v, best_f = (0.0, fv0) if fv0 <= fv1 else (1.0, fv1)
            res = minimize_scalar(obj_fn, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            if res.fun < best_f - 1e-15:
                best_v, best_f = res.x, res.fun
            if best_f * dx < best_c5 - 1e-14:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = corr + (best_v - old) * coef
                improved = True
        if not improved:
            break
    return best_h, best_c5

def structured_perturb(h, n, rng, strength=0.08):
    h_p = h.copy()
    mode = rng.randint(3)
    if mode == 0:
        h_p += strength * rng.randn(n)
    elif mode == 1:
        center = rng.randint(0, n)
        width = int(n * rng.uniform(0.1, 0.3))
        x = np.arange(n)
        bump = np.exp(-((x - center) ** 2) / (2.0 * width * width))
        h_p += strength * bump * (0.5 + 0.5 * rng.random(n))
        h_p -= strength * 0.5 * bump
    else:
        h_p += strength * np.sin(2.0 * np.pi * rng.uniform(1, 4) * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    return project(h_p, n)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent (35% time)
    t1 = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=2000)

    # Phase 2: SLSQP soft-max refinement (10% time)
    t2 = t0 + budget_s * 0.45
    for T in [50, 300, 1500]:
        if time.time() > t2:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 3: Coordinate descent after SLSQP (10% time)
    t3 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, n, t3, max_sweeps=1500)

    # Phase 4: Multi-restart with structured perturbations (30% time)
    t4 = t0 + budget_s * 0.85
    n_restarts = 0
    while time.time() < t4 and n_restarts < 200:
        n_restarts += 1
        cool = max(0.1, 1.0 - 0.9 * n_restarts / 200.0)
        strength = 0.02 + 0.12 * cool
        h_p = structured_perturb(best_h, n, rng, strength)
        c5p = compute_c5(h_p)
        # Quick coordinate descent from perturbed point
        h_c, c5c = coord_descent(h_p, n, min(time.time() + 1.5, t4), max_sweeps=120)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c

    # Phase 5: Try alternate resolutions (8% time)
    t5 = t0 + budget_s * 0.93
    for n_try in [100, 120, 200]:
        if time.time() > t5:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, min(time.time() + 8.0, t5), max_sweeps=300)
        if c5_tr < best_c5:

    if time.time() < t6 + 5.0:

        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s, max_sweeps=300)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
