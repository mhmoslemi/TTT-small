    best_h = h.copy()
    best_corr = corr.copy()
    best_c5 = best_corr.max() * (2.0/n)
    center = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            x_old = best_h[i]
            # Compute sensitivity
            s = np.zeros(2*n-1)
            for k in range(1, n):
                ip = i + k
                im = i - k
                if ip < n:
                    s[center+k] += (1.0 - best_h[ip])
                if im >= 0:
                    s[center+k] -= best_h[im]
                if im >= 0:
                    s[center-k] += (1.0 - best_h[im])
                if ip < n:
                    s[center-k] -= best_h[ip]
            
            # 1D optimization
            ct = best_corr[center]
            cb = best_corr.copy()
            cb[center] = 0  # handle separately
            
            def f1d(x):
                vals = cb + (x - x_old) * s
                vals[center] = ct + (x - x_old) * (1.0 - x - x_old)
                return vals.max()
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            x_new = res.x
            
            if abs(x_new - x_old) > 1e-12:
                delta = x_new - x_old
                best_corr += delta * s
                best_corr[center] = ct + delta * (1.0 - x_new - x_old)
                best_h[i] = x_new
                improved = True
        
        # Re-verify and fix
        new_corr = compute_corr(best_h)
        new_c5 = new_corr.max() * (2.0/n)
        if new_c5 < best_c5 - 1e-14:

    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    M = corr.max()
    exp_T = np.exp(T * (corr - M))
    alpha = exp_T / exp_T.sum()
    
    center = n - 1
    grad = np.zeros(n)
    
    # For each lag k, alpha[center+k] and alpha[center-k] contribute
    for k in range(0, n):
        if k == 0:
            w = alpha[center]
            grad += w * (1.0 - 2.0 * h)
        else:
            w_pos = alpha[center + k]
            w_neg = alpha[center - k]
            # corr[center+k] = sum_j h[j]*(1-h[j+k]) for valid j
            # d/dh[i]: (1-h[i+k]) if i+k valid, minus h[i-k] if i-k valid
            if w_pos != 0:
                grad[:n-k] += w_pos * (1.0 - h[k:])
                grad[:n-k] -= 0  # h[i-k] not relevant... 
                # Actually: for j=i, term is h[i]*(1-h[i+k])
                # d/dh[i]: (1-h[i+k])
                # For j=i-k, term is h[i-k]*(1-h[i])

                # d corr[center+k]/dh[i] = (1-h[i+k]) * 1_{i+k < n} - h[i-k] * 1_{i-k >= 0}
                # For i in [0, n-1-k]: (1-h[i+k])
                # For i in [k, ...]: -h[i-k]

                if n - k > 0:

                if k < n:
                    grad[k:] -= h[:n-k]  # from second term

    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    if s > target:
        h = h * (target / s)
    else:
        h = h * (target / s)
    h = np.clip(h, 0, 1)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += room * diff / tot
        else:
            h = h * (target / h.sum())
            h = np.clip(h, 0, 1)
    return h

def fast_cd(h, n, t_end, max_sweeps=300):

    center = n - 1
    dx = 2.0 / n
    best_c5 = corr.max() * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break

        for i in range(n):
            x0 = h[i]
            s = np.zeros(2 * n - 1)
            for k in range(1, n):
                ip = i + k
                im = i - k
                if ip < n:
                    s[center + k] += 1.0 - h[ip]
                    s[center - k] -= h[ip]
                if im >= 0:
                    s[center + k] -= h[im]
                    s[center - k] += 1.0 - h[im]
            ct = corr[center]
            def f1d(x):
                v = corr + (x - x0) * s
                v[center] = ct + (x - x0) * (1.0 - x - x0)
                return v.max()
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            x_new = res.x
            if abs(x_new - x0) > 1e-12:
                delta = x_new - x0
                corr += delta * s
                corr[center] = ct + delta * (1.0 - x_new - x0)
                h[i] = x_new

        if not improved:

    best_c5 = corr.max() * dx

def slsqp_smooth(h, n, T, maxiter=500):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        M = corr.max()
        e = np.exp(T * (corr - M))
        return (M + np.log(e.sum()) / T) * 2.0 / n
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        center = n - 1
        M = corr.max()
        e = np.exp(T * (corr - M))
        alpha = e / e.sum()
        g = alpha[center] * (1.0 - 2.0 * x)

            an = alpha[center - k]
            if ap > 1e-15:

                g[k:] -= ap * x[:n-k]
            if an > 1e-15:
                g[k:] += an * (1.0 - x[:n-k])
                g[:n-k] -= an * x[k:]
        return g * 2.0 / n

             'jac': lambda x: np.ones(n)}]

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def compute_c5(h):

    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def project(h, n):

    co = np.correlate(x, 1.0 - x, mode='full')
    M = co.max()
    e = np.exp(T * (co - M))
    return (M + np.log(e.sum()) / T) * (2.0 / n)

    ce = n - 1
    M = co.max()
    e = np.exp(T * (co - M))
    a = e / e.sum()
    g = a[ce] * (1.0 - 2.0 * x)
    for k in range(1, n):
        ap, an = a[ce + k], a[ce - k]
        if ap > 1e-12:
            g[:n-k] += ap * (1.0 - x[k:])
            g[k:] -= ap * x[:n-k]
        if an > 1e-12:
            g[k:] += an * (1.0 - x[:n-k])
            g[:n-k] -= an * x[k:]
    return g * (2.0 / n)
