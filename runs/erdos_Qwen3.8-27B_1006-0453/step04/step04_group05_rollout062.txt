def grad_smooth_max(h, N, alpha, top_k=200):
    """Efficient gradient using top-k lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    # Use top-k for efficiency
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c_top = corr[idx]
    max_top = c_top.max()
    exp_top = np.exp(alpha * (c_top - max_top))
    w = exp_top / exp_top.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for ai in range(len(idx)):
        k = idx[ai]
        L = k - center
        wi = w[ai]
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0 and L < N:
            # d/dh[i]: (1-h[i+L]) for i in [0, N-1-L] minus h[i-L] for i in [L, N-1]
            grad[:N-L] += wi * (1.0 - h[L:])
            grad[L:] -= wi * h[:N-L]
        elif L < 0 and L > -N:
            M = -L
            grad[:N-M] += wi * (1.0 - h[M:])
            grad[M:] -= wi * h[:N-M]
    return grad, max_c

import numpy as np
import time

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_weighted(h, N, weights):
    grad = np.zeros(N)
    center = N - 1
    for k, wk in weights:
        L = int(k) - center
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif 0 < L < N:
            grad[:N-L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:N-L]
        elif -N < L < 0:
            M = -L
            grad[:N-M] += wk * (1.0 - h[M:])
            grad[M:] -= wk * h[:N-M]
    return grad

def smooth_max_weights(corr, alpha, top_k):
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c = corr[idx]
    mc = c.max()
    w = np.exp(alpha * (c - mc))
    w /= w.sum()
    pairs = []
    for i in range(len(idx)):
        if w[i] > 1e-6:
            pairs.append((idx[i], w[i]))
    return pairs

def adam_cts(h, N, target, deadline, alpha=10.0, top_k=150, lr0=0.02, decay=0.996, b1=0.9, b2=0.999):
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_peak = float(corr.max())
    dx = 2.0 / N
    lr = lr0
    eps = 1e-8
    m_ = np.zeros(N)
    v_ = np.zeros(N)
    it = 0
    while time.time() < deadline:
        it += 1
        if it % 30 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            pk = float(corr.max())
            if pk < best_peak:
                best_peak = pk
                best_h = h.copy()
            lr *= decay
        a = alpha * (1.0 + 0.3 * min(it / 1000.0, 2.0))
        corr = np.correlate(h, 1.0 - h, mode='full')
        pairs = smooth_max_weights(corr, a, top_k)
        grad = grad_weighted(h, N, pairs)
        m_ = b1 * m_ + (1.0 - b1) * grad
        v_ = b2 * v_ + (1.0 - b2) * (grad * grad)
        it_ = it
        mh = m_ / (1.0 - b1 ** it_)
        vh = v_ / (1.0 - b2 ** it_)
        h = project_box_sum(h - lr * mh / (np.sqrt(vh) + eps), target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    pk = float(corr.max())
    if pk < best_peak:
        best_peak = pk
        best_h = h.copy()
    return best_h, best_peak * dx

def greedy_bin(h, N, deadline):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        h[np.where(h == 1)[0][nh:]] = 0
    elif n1 < nh:
        h[np.where(h == 0)[0][:nh - n1]] = 1
    stuck = 0
    while time.time() < deadline and stuck < 30:
        corr = np.correlate(h, 1.0 - h, mode='full')
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        pairs = smooth_max_weights(corr, 12.0, 80)
        grad = grad_weighted(h, N, pairs)
        p = ones[np.argmax(grad[ones])]
        q = zeros[np.argmin(grad[zeros])]
        h[p], h[q] = 0.0, 1.0
        np2 = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        if np2 < P:
            stuck = 0
        else:
            h[p], h[q] = 1.0, 0.0
            stuck += 1
            if stuck > 5:
                for _ in range(30):
                    p2 = ones[np.random.randint(len(ones))]
                    q2 = zeros[np.random.randint(len(zeros))]
                    h[p2], h[q2] = 0.0, 1.0
                    np3 = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                    if np3 < P:
                        stuck = 0
                        break
                    h[p2], h[q2] = 1.0, 0.0
            if stuck >= 30:
                break
    return h

def multi_round(h, N, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, N)
    rng_local = np.random.RandomState(7)
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 15:
            break
        hb = greedy_bin(best_h, N, time.time() + min(20, remaining * 0.25))
        c5b = compute_c5(hb, N)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb.copy()
        d_adam = time.time() + min(50, remaining * 0.45)
        noise = rng_local.randn(N) * 0.04
        hc = project_box_sum(best_h + noise, N / 2.0)
        ha, c5a = adam_cts(hc, N, N / 2.0, d_adam, alpha=8.0, top_k=180, lr0=0.015, decay=0.996)
        if c5a < best_c5:
            best_c5, best_h = c5a, ha.copy()
        d_adam2 = time.time() + min(30, remaining * 0.2)
        if time.time() < d_adam2:
            ha2, c5a2 = adam_cts(best_h + rng_local.randn(N) * 0.02, N, N / 2.0, d_adam2, alpha=15.0, top_k=100, lr0=0.008, decay=0.997)
            if c5a2 < best_c5:
                best_c5, best_h = c5a2, ha2.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary refinement of initial (2 min)
    d1 = t0 + min(120, budget_s * 0.12)
    hb = greedy_bin(h0, N0, d1)
    c5b = compute_c5(hb, N0)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 2: Continuous optimization from multiple inits (4 min)
    d2 = t0 + min(400, budget_s * 0.4)
    inits = [best_h, (best_h + np.random.randn(N0) * 0.06)]
    for hc_init in inits:
        if time.time() > d2 - 60:
            break
        hc = project_box_sum(hc_init, N0 / 2.0)
        ha, c5a = adam_cts(hc, N0, N0 / 2.0, d2 - 60, alpha=10.0, top_k=200, lr0=0.02, decay=0.996)
        if c5a < best_c5:
            best_c5, best_h = c5a, ha.copy()

    # Phase 3: Multi-round alternating (3 min)
    d3 = t0 + min(680, budget_s * 0.68)
    best_h, best_c5 = multi_round(best_h, N0, d3 - 60)

    # Phase 4: Try nearby N values
    d4 = deadline - 20
    for dN in [2, -2, 4, -4, 6]:
        if time.time() > d4 - 40:
            break
        Nt = N0 + dN
        if Nt < 200 or Nt % 2 != 0:
            continue
        x_old = np.linspace(0.0, 2.0, N0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, Nt, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = project_box_sum(hi, Nt / 2.0)
        hib = greedy_bin(hi, Nt, time.time() + 15)
        c5t = compute_c5(hib, Nt)
        if c5t < best_c5:
            best_c5, best_h = c5t, hib
        if time.time() < d4 - 30:
            ha2, c5a2 = adam_cts(hib + np.random.randn(Nt) * 0.04, Nt, Nt / 2.0, d4 - 25, alpha=10.0, top_k=180, lr0=0.015, decay=0.996)
            if c5a2 < best_c5:
                best_c5, best_h = c5a2, ha2

    # Phase 5: Final fine polish
    if time.time() < deadline - 5:
        hf = project_box_sum(best_h + np.random.randn(len(best_h)) * 0.015, len(best_h) / 2.0)
        hf, c5f = adam_cts(hf, len(best_h), len(best_h) / 2.0, deadline - 2, alpha=20.0, top_k=200, lr0=0.005, decay=0.998)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
