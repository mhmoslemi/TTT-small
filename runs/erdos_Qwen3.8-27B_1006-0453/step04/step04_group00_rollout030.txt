import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    # Project onto sum constraint while maintaining [0,1]
    if s > target:
        # Scale down
        excess = s - target
        h_pos = h > 0
        if h_pos.sum() == 0:
            return h
        ratio = 1.0 - excess / s
        h = h * ratio
    else:
        deficit = target - s
        h_neg = h < 1.0
        if h_neg.sum() == 0:
            return h
        h = h + (1.0 - h) * (deficit / (n - s))
    return np.clip(h, 0.0, 1.0)

def compute_grad_soft(h, T):
    """Compute gradient of soft-max smoothed C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Compute softmax weights
    exponents = T * (corr - m)
    # Stabilize
    exponents = np.clip(exponents, -50, 50)
    p = np.exp(exponents)
    p = p / p.sum()
    # grad[i] = dx * sum_s p[s] * ((1-h[i-s])*[valid] - h[i+s]*[valid])
    # s = m_idx - (n-1) for m_idx in range(2n-1)

    # sum_m p[m] * (1-h[i - (m-(n-1))]) with validity
    # = sum_m p[m] * (1-h[i+m-n+1])*[0<=i+m-n+1<n]
    # Let's compute this via convolution-like operations
    grad = np.zeros(n)

        s = m_idx - (n - 1)
        if abs(p[m_idx]) < 1e-15:

        # (1-h[i-s]) valid when 0<=i-s<n → s<=i<n+s

            # -h[i+s] for i=0..n-1-s

            # -h[i+s] = -h[i-|s|] for i=|s|..n-1
            grad[-s:] -= p[m_idx] * h[:-s]
        else:
            # s=0: (1-h[i]) - h[i] = 1-2h[i]
            grad += p[m_idx] * (1.0 - 2.0 * h)
    return dx * grad

def soft_c5_value(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exponents = np.clip(T * (corr - m), -50, 50)
    return dx * (m + np.log(np.sum(np.exp(exponents))) / T)

def proj_grad_descent(h, n, t_end, max_iters=5000, T_start=100, T_end=5000):
    """Projected gradient descent on smooth objective."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_curr = h.copy()
    momentum = np.zeros(n)
    beta = 0.7
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = it / max_iters
        T = T_start * (T_end / T_start) ** frac
        T = max(T, 1.0)
        g = compute_grad_soft(h_curr, T)
        momentum = beta * momentum + g
        # Adaptive step: use gradient magnitude
        gm = np.max(np.abs(momentum))
        if gm < 1e-10:

        alpha = 0.01 / gm  # normalize step
        h_new = h_curr - alpha * momentum
        h_new = project_h(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c5

When h[i] changes from old to v (delta = v - old):
- h[i] appears as h[j] when j=i: adds (v-old)*(1-h[i-s]) if i-s valid and i-s≠i (i.e., s≠0)

Note: if i-s == i (s=0) or i+s == i (s=0), we excluded those.

What if (i-s) == (i+s)? Same thing, s=0.

Could h[i+s] involve h[i]? Only if i+s==i, i.e., s=0. Excluded.

For s=0:

Change: (v)(1-v) - old*(1-old) = v-v² - old + old² = (v-old) - (v²-old²) = delta - delta*(v+old)
= delta*(1 - v - old)

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    for _ in range(10):
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        else:
            if s < 1e-14:
                break
            h = h * ((target) / s)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    return h

def coord_descent(h, n, t_end, max_sweeps=1000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * 2.0 / n)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                v = 0.0
                js = i - s
                if 0 <= js < n:
                    v += 1.0 - best_h[js]
                jp = i + s
                if 0 <= jp < n:
                    v -= best_h[jp]
                lin[mp] = v
                # For -s
                v2 = 0.0
                js2 = i + s
                if 0 <= js2 < n:
                    v2 += 1.0 - best_h[js2]
                jp2 = i - s
                if 0 <= jp2 < n:
                    v2 -= best_h[jp2]
                lin[mn] = v2
            lin[n-1] = 1.0 - 2.0 * old
            # 1D minimization
            def f1d(v):

                val = corr + d * lin
                val[n-1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            if abs(v_new - old) < 1e-12:

            delta = v_new - old
            best_h[i] = v_new

            corr[n-1] -= delta * delta

            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:

                best_h[i] = old
                corr = corr - delta * lin
                corr[n-1] += delta * delta
        if not improved:

    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def grad_soft_h(h, T, n):
    """Gradient of soft-max smoothed C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exponents = np.clip(T * (corr - m), -60, 60)
    p = np.exp(exponents)
    p /= p.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = p[m_idx]
        if w < 1e-14:
            continue
        s = m_idx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n-a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n-a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

def proj_grad_h(h, n, t_end, max_iters=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    mom = np.zeros(n)

        frac = min(1.0, it / max_iters)
        T = 50 * (200.0 / 50.0) ** frac

        if c5_new < best_c5:

            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c5

def basin_hop_cd(h, n, t_end, n_hops=200, seed=42):

    best_h = h.copy()

                best_h = h_opt.copy()

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 3 == 2 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                if i <= n - 1 - s:
                    lin[mp] -= best_h[i + s]
                if i <= n - 1 - s:
                    lin[mn] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[mn] -= best_h[i - s]
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def grad_soft_h(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exponents = np.clip(T * (corr - m), -60.0, 60.0)
    p = np.exp(exponents)
    p /= p.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = p[m_idx]
        if w < 1e-14:
            continue
        s = m_idx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

def proj_grad_h(h, n, t_end, max_iters=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    mom = np.zeros(n)
    beta = 0.7
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = min(1.0, it / max_iters)
        T = 50.0 * 40.0 ** frac
        g = grad_soft_h(h_c, T, n)
        mom = beta * mom + g
        gm = np.max(np.abs(mom))
        if gm < 1e-13:
            break
        alpha = 0.4 / gm
        h_new = project_h(h_c - alpha * mom, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c5

def basin_hop_cd(h, n, t_end, n_hops=200, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.015 + 0.06 * cool
        mode = hop % 5
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 12)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.25
            pert = np.zeros(n)
            pert[mask] = step * 2 * rng.randn(mask.sum())
        elif mode == 3:
            idx = rng.choice(n, size=max(4, n // 10), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * rng.randn(len(idx))
        else:
            pert = step * 0.5 * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        hop_end = min(t_end, time.time() + max(2.0, 8.0 * cool))
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=80, seed=seed + hop)
        temp = 0.004 * cool + 0.0004
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent
    t1 = t0 + budget_s * 0.22
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=3000, seed=seed)

    # Phase 2: Gradient descent + coord polish
    if time.time() < t0 + budget_s * 0.38:
        h_g, c5_g = proj_grad_h(best_h, n, t0 + budget_s * 0.35, max_iters=2000)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.42, max_sweeps=1000, seed=seed + 1)

    # Phase 3: Basin hopping (main exploration)
    if time.time() < t0 + budget_s * 0.72:
        best_h, best_c5 = basin_hop_cd(best_h, n, t0 + budget_s * 0.72, n_hops=400, seed=seed + 10)

    # Phase 4: Final coordinate descent
    if time.time() < t0 + budget_s * 0.90:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=2000, seed=seed + 100)

    # Phase 5: Second gradient pass + coord
    if time.time() < t0 + budget_s * 0.94:
        h_g2, c5_g2 = proj_grad_h(best_h, n, t0 + budget_s * 0.93, max_iters=800)
        if c5_g2 < best_c5:
            best_h, best_c5 = h_g2, c5_g2

    # Phase 6: Last coord polish
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=500, seed=seed + 200)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
