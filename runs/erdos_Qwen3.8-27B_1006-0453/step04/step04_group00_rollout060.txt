    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Thorough coordinate descent
    best_h, best_c5 = fast_coord_descent(best_h, n, max_sweeps=400, t_end=t0 + budget_s * 0.25)

    # Phase 2: Basin hopping with diverse perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 0
    while elapsed() < budget_s * 0.55:
        n_iter += 1
        cool = max(0.1, 1.0 - n_iter / 80.0)
        strat = n_iter % 6
        if strat == 0:
            pert = (0.03 + 0.10 * cool) * rng.standard_normal(n)
        elif strat == 1:
            freq = rng.randint(1, 15)
            phase = rng.uniform(0, 2 * np.pi)
            pert = (0.02 + 0.07 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        elif strat == 2:
            pert = targeted_perturb(work_h, n, cool, rng)
        elif strat == 3:
            bs = rng.randint(4, 25)
            st = rng.randint(0, n - bs)
            pert = np.zeros(n)
            pert[st:st + bs] = (0.04 + 0.10 * cool) * rng.standard_normal(bs)
        elif strat == 4:
            # Try reversal
            h_rev = work_h[::-1].copy()
            h_rev = project_h(h_rev, n)
            h_new, c5_new = fast_coord_descent(h_rev, n, max_sweeps=80, t_end=t0 + budget_s * 0.55)
            if c5_new < work_c5:
                work_h = h_new
                work_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            continue
        else:
            pert = (0.015 + 0.04 * cool) * rng.standard_normal(n)

        h_pert = project_h(work_h + pert, n)
        h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=120, t_end=t0 + budget_s * 0.55)
        temp = 0.0008 * cool + 0.00005
        if c5_new < work_c5 or (c5_new - work_c5 < 0.004 and rng.rand() < np.exp(-(c5_new - work_c5) / temp)):
            work_h = h_new
            work_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 3: SLSQP soft-max polish (multiple temperatures)
    for T in [200, 1000, 5000, 20000, 100000, 500000]:
        if elapsed() > budget_s * 0.78:
            break
        h_s = slsqp_soft(best_h, T, n, maxiter=800)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
        # Also follow up with coord descent
        h_cd, c5_cd = fast_coord_descent(h_s, n, max_sweeps=100, t_end=t0 + budget_s * 0.78)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 4: Second round of fine basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(30):
        if elapsed() > budget_s * 0.88:
            break
        step = 0.01 + 0.03 * (1.0 - it / 30.0)
        h_pert = project_h(work_h + step * rng.standard_normal(n), n)
        h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=80, t_end=t0 + budget_s * 0.88)
        if c5_new < work_c5:
            work_h = h_new
            work_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Final long coordinate descent
    best_h, best_c5 = fast_coord_descent(best_h, n, max_sweeps=300, t_end=t0 + budget_s * 0.96)

    # Phase 6: Ultra-high-T SLSQP
    for T in [2000000, 10000000]:
        if elapsed() > budget_s * 0.99:
            break
        h_s = slsqp_soft(best_h, T, n, maxiter=1000)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Final polish
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = fast_coord_descent(best_h, n, max_sweeps=100, t_end=t0 + budget_s * 0.99)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
