import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    target = n / 2.0
    vec = np.clip(h, 0.0, 1.0)
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    # Binary search for clipping offset
    lo, hi = -max(vec.max(), 1.0) - 1.0, 1.0 - min(vec.min(), 0.0) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo+hi)*0.5, 0, 1)

def corr_delta(h, i, delta, n):
    """Compute change in corr array when h[i] changes by delta."""
    corr = compute_corr(h)  # This is O(n^2), too slow
    # Better: just compute the change
    dcorr = np.zeros(2*n-1, dtype=float)
    # k_idx = k + (n-1), where k is the shift
    # For shift k: term at x=i contributes delta*(1-h[i+k]) if valid
    #               term at x=i-k contributes -delta*h[i-k] if valid
    for k in range(-(n-1), n):
        k_idx = k + n - 1
        # x = i, need i+k in [0, n-1]
        if 0 <= i+k < n:
            dcorr[k_idx] += delta * (1.0 - h[i+k])
        # x = i-k, need i-k in [0, n-1]
        if 0 <= i-k < n:
            dcorr[k_idx] -= delta * h[i-k]
    return dcorr

new_max = max(corr + F[j,:] - F[i,:])

This is O(n) per pair (just a numpy max of a 319-element array).

Precomputation: O(n * 2n) = O(n²) = 25k ops. Done once per sweep (since h changes).
Pair evaluations: for "steepest descent", n/2 × n/2 = 6400 pairs × O(n) = 6400 × 319 ≈ 2M ops.
Total per sweep: ~2M ops ≈ 2-5ms in numpy. Can do 200k+ sweeps in 1000s.

This is very fast! Let me implement this.

But wait - after a swap, F changes for all positions (not just i and j). Specifically, F(a,k) changes for all a that interact with i or j. Let me think...

F(a,k) = (1-h(a+k)) - h(a-k)
If h[i] changes:
- F(a,k) changes when a+k=i (a=i-k) or a-k=i (a=i+k)
- So F changes for a = i±k for all k, i.e., all a in [0,n-1] (since k ranges over all valid shifts)

So after any change, ALL F values change. Precomputation must be redone each iteration. That's O(n²) per iteration.

Per iteration: O(n²) precompute + O(n³) pair eval = O(n³) total.
For n=160: 4M ops per iteration. At 100ns/op: 0.4ms per iteration.
In 1000s: 2.5M iterations. Way more than enough.

Actually, let me reconsider. The "steepest descent" approach (evaluate all pairs, take best) is O(n³) per step. But "first improvement" or "random sampling" is O(n²) per step (just try a random pair). And we can batch the random samples.

Let me just go with a practical implementation:
1. Precompute F matrix (n × (2n-1))
2. Each iteration: sample K random pairs, evaluate each in O(n), take best
3. If improvement: update h, corr, F
4. Repeat

With K=50 samples per iteration, each iteration is O(n + K*n) = O(K*n) ≈ 50*319 ≈ 16k ops. At 100ns/op: 1.6ms. 600k iterations in 1000s.

OK let me write the final code now. I'll keep it clean and efficient.

Let me also add a "group flip" idea: instead of flipping one pair, flip multiple pairs at once. This can escape local optima faster.

And for the continuous case, I'll use a "2D pair" optimization: for pair (i,j), find the optimal split point using minimize_scalar on the 1D function.

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    target = n / 2.0
    vec = np.clip(h, 0.0, 1.0)
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0, 1)

def build_F(h, n):
    """F[i, k_idx] = (1-h[i+k]) - h[i-k], k_idx = k+(n-1). 0 if out of range."""
    shifts = np.arange(-(n-1), n, dtype=np.int64)
    F = np.zeros((n, 2*n-1))
    for i in range(n):
        v1 = (i + shifts >= 0) & (i + shifts < n)
        v2 = (i - shifts >= 0) & (i - shifts < n)
        F[i, v1] += 1.0 - h[i + shifts[v1]]
        F[i, v2] -= h[i - shifts[v2]]
    return F

def pair_flip_steepest(h, n, t_end, max_iter=500000, n_samples=60):
    """Steepest-descent pair-flip with incremental corr."""
    shifts = np.arange(-(n-1), n, dtype=np.int64)
    corr = compute_corr(h)
    dx_val = 2.0 / n
    best_c5 = np.max(corr) * dx_val
    F = build_F(h, n)
    iters = 0
    no_improve = 0
    while iters < max_iter and time.time() < t_end:
        iters += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_new_max = np.max(corr)
        best_dc = None
        best_pair = None
        ns = min(n_samples, len(ones) * len(zeros))
        idx_o = np.random.choice(len(ones), size=min(len(ones), ns+10), replace=False)
        idx_z = np.random.choice(len(zeros), size=min(len(zeros), ns+10), replace=False)
        for a_i in idx_o:
            for b_i in idx_z[:ns // max(1, len(idx_o))]:
                a = ones[a_i] if a_i < len(ones) else ones[-1]
                b = zeros[b_i] if b_i < len(zeros) else zeros[-1]
                if a == b:
                    continue
                dc = F[b] - F[a]
                nm = np.max(corr + dc)
                if nm < best_new_max - 1e-6:
                    best_new_max = nm
                    best_dc = dc
                    best_pair = (a, b)
        if best_pair is not None:
            a, b = best_pair
            corr += best_dc
            h[a] = 0.0
            h[b] = 1.0
            F = build_F(h, n)
            best_c5 = best_new_max * dx_val
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break
    return h, compute_c5(h)

def pair_flip_continuous(h, n, t_end, max_sweeps=200):
    """Continuous pair optimization with 1D minmax per pair."""
    corr = compute_corr(h)
    dx_val = 2.0 / n
    best_c5 = np.max(corr) * dx_val
    F = build_F(h, n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return h, best_c5
            j = order[(order == i).argmax() + 1] if (order == i).argmax() + 1 < n else (i + 1) % n
            if j == i:
                j = (i + 1) % n
            # Optimize delta: h[i]+δ, h[j]-δ
            # corr_new(k) = corr(k) + δ*(F[i,k] - F[j,k])
            G = F[i] - F[j]
            def obj(d):
                return np.max(corr + d * G)
            res = minimize_scalar(obj, bounds=(-min(h[i], h[j]), min(1-h[i], 1-h[j])),
                                  method='bounded', options={'xatol': 1e-8})
            d = res.x
            if abs(d) > 1e-6:
                new_max = obj(d)
                if new_max < best_c5 / dx_val - 1e-8:
                    h[i] += d
                    h[j] -= d
                    h = np.clip(h, 0, 1)
                    diff = n/2.0 - h.sum()
                    h += diff / n
                    h = np.clip(h, 0, 1)
                    h = project(h, n)
                    corr = compute_corr(h)
                    F = build_F(h, n)
                    best_c5 = np.max(corr) * dx_val
                    improved = True
        if not improved:
            break
    return h, best_c5

def nested_grad(h, n, t_end, T, lr, momentum):
    """Nesterov accelerated projected gradient on smooth max."""
    corr = compute_corr(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    y = h.copy()
    for it in range(5000):
        if time.time() > t_end:
            break
        c = compute_corr(y)
        m = np.max(c)
        w = np.exp(T * (c - m))
        w /= w.sum()
        shifts = np.arange(-(n-1), n, dtype=np.int64)
        grad = np.zeros(n)
        for i in range(n):
            v1 = (i + shifts >= 0) & (i + shifts < n)
            v2 = (i - shifts >= 0) & (i - shifts < n)
            grad[i] = np.sum(w[v1] * (1 - y[i + shifts[v1]])) - np.sum(w[v2] * y[i - shifts[v2]])
        grad *= 2.0 / n
        t_curr = (1 + np.sqrt(1 + 4 * momentum * momentum * it * it)) / 2
        theta = (it - 1) / t_curr if it > 0 else 0
        y_new = y + (1 + theta) * v
        v_new = (y_new - y) / (1 + theta) - lr * grad
        y = y_new + theta * v_new
        y = project(y, n)
        c5v = compute_c5(y)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = y.copy()
        v = v_new
    return best_h, best_c5

def golden_init(n):
    alpha = (1.0 + 5.0**0.5) * 0.5
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * alpha) % 1.0 < 0.5 else 0.0
    return project(h, n)

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        p = 0
        while v:
            p ^= v & 1
            v >>= 1
        h[i] = 1.0 if p else 0.0
    return project(h, n)
