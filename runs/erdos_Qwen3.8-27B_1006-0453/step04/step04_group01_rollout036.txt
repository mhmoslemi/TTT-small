```python
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    current_ones = int(h_bin.sum())
    if current_ones == target:
        return h_bin
    if current_ones > target:
        # Need to flip some 1s to 0s
        excess = current_ones - target
        ones = np.where(h_bin == 1)[0]
        # Flip the ones with largest h values (most "1-like")
        # Actually for binary, just flip random ones
        to_flip = np.random.choice(ones, excess, replace=False)
        h_bin[to_flip] = 0
    else:
        deficit = target - current_ones
        zeros = np.where(h_bin == 0)[0]
        to_flip = np.random.choice(zeros, deficit, replace=False)
        h_bin[to_flip] = 1
    return h_bin

def project_continuous(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -np.max(vc) - 1.0
    hi = 1.0 - np.min(vc) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=600, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_continuous(res.x, n)

def coord_descent(h, t_end, max_sweeps=500, n=None):
    if n is None:
        n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):

        improved = False
        order = np.random.permutation(n)

            candidates = [max(0, min(1, c)) for c in candidates]

            if no_improve >= 3:
```

```python
def delta_corr(h, i, j, n):
    # h[i] and h[j] are being swapped (h[i]=1, h[j]=0 or vice versa)
    # Δcorr[k] for k = -(n-1), ..., n-1
    # Maps to index j_idx = k + (n-1) in the full correlation
    delta = np.zeros(2*n - 1)
    # For each k, delta[k + n - 1] = h[i+k] - h[j+k] + h[i-k] - h[j-k]
    # Valid where indices are in range
    for k in range(-(n-1), n):
        val = 0.0
        if 0 <= i+k < n: val += h[i+k]
        if 0 <= j+k < n: val -= h[j+k]
        if 0 <= i-k < n: val += h[i-k]
        if 0 <= j-k < n: val -= h[j-k]
        delta[k + n - 1] = val
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_continuous(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-12:
        return vc
    lo = -np.max(vc) - 1.0
    hi = 1.0 - np.min(vc) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def to_binary_fix_sum(h, n):
    """Convert to binary with exactly n//2 ones."""
    target = n // 2
    h_bin = (h > 0.5).astype(np.float64)
    n_ones = int(h_bin.sum())
    if n_ones == target:
        return h_bin
    if n_ones > target:
        excess = n_ones - target
        ones_idx = np.where(h_bin == 1)[0]
        to_flip = np.random.choice(ones_idx, excess, replace=False)
        h_bin[to_flip] = 0.0
    else:
        deficit = target - n_ones
        zeros_idx = np.where(h_bin == 0)[0]
        to_flip = np.random.choice(zeros_idx, deficit, replace=False)
        h_bin[to_flip] = 1.0
    return h_bin

def delta_corr_vec(h, i, j, n):
    """Compute change in full correlation when swapping binary h[i] and h[j]."""
    k = np.arange(-(n - 1), n, dtype=np.int64)
    idx1 = i + k
    idx2 = j + k
    idx3 = i - k
    idx4 = j - k
    delta = np.zeros(2 * n - 1)
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    m3 = (idx3 >= 0) & (idx3 < n)
    m4 = (idx4 >= 0) & (idx4 < n)
    if m1.any(): delta[m1] += h[idx1[m1]]
    if m2.any(): delta[m2] -= h[idx2[m2]]
    if m3.any(): delta[m3] += h[idx3[m3]]
    if m4.any(): delta[m4] -= h[idx4[m4]]
    return delta

def sa_binary(h, t_end, n, T0=0.05, T_f=1e-7, max_stuck=50000):
    """Simulated annealing with binary 2-swap moves."""
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5
    best_corr = corr.copy()
    
    t_start = time.time()
    n_iter = 0
    stuck = 0
    
    # Pre-compute k array
    k_arr = np.arange(-(n-1), n, dtype=np.int64)
    
    while time.time() < t_end:
        n_iter += 1
        # Temperature (time-based)
        frac = (time.time() - t_start) / max(t_end - t_start, 1e-9)
        T = T0 * (T_f / T0) ** frac
        if T < 1e-9:
            T = 1e-9
        
        # Pick random swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        
        # Compute delta
        idx1 = i + k_arr
        idx2 = j + k_arr
        idx3 = i - k_arr
        idx4 = j - k_arr
        delta = np.zeros(2 * n - 1)
        m1 = (idx1 >= 0) & (idx1 < n)
        m2 = (idx2 >= 0) & (idx2 < n)
        m3 = (idx3 >= 0) & (idx3 < n)
        m4 = (idx4 >= 0) & (idx4 < n)
        delta[m1] += h[idx1[m1]]
        delta[m2] -= h[idx2[m2]]
        delta[m3] += h[idx3[m3]]
        delta[m4] -= h[idx4[m4]]
        
        new_corr = corr + delta
        new_c5 = np.max(new_corr) * dx
        
        # SA acceptance
        if new_c5 <= c5:
            accept = True
        else:
            accept = np.random.rand() < np.exp(-(new_c5 - c5) / T)
        
        if accept:
            h[i], h[j] = h[j], h[i]
            corr = new_corr
            c5 = new_c5
            stuck = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_corr = corr.copy()
        else:
            stuck += 1
    
    return best_h, best_c5

def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_continuous(res.x, n)

def local_cd(h, t_end, n, max_sweeps=200):
    """Light coordinate descent for continuous h."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            # Try 0, 1, and midpoint
            for val in (0.0, 1.0, 1.0 - old):
                if abs(val - old) < 1e-9:
                    continue
                h_try = best_h.copy()
                h_try[i] = val
                h_try = project_continuous(h_try, n)
                c = compute_c5(h_try)
                if c < best_c5 - 1e-14:
                    best_h = h_try
                    best_c5 = c
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_cont(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    s = vc.sum()
    if abs(s - target) < 1e-12:
        return vc
    lo, hi = -1.0, 1.0

        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

    hb = (h > 0.5).astype(np.float64)

        idx = np.where(hb == 1)[0]

    elif no < target:
        idx = np.where(hb == 0)[0]

    return hb

    h = h0.copy()
    corr = compute_corr(h)
    c5 = np.max(corr) * dx

        if h[i] == h[j]:

        h[i], h[j] = h[j], h[i]
        nc = compute_corr(h)
        n5 = np.max(nc) * dx

        else:
            corr, c5 = nc, n5
            if c5 < bc5:

    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m))) / T)) * dx
```

```python
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        ones = np.where(bh == 1)[0]
        zeros = np.where(bh == 0)[0]
        for i in ones[:]:
            if time.time() > t_end:
                return bh, bc5
            for j in zeros[:]:
                bh[i], bh[j] = bh[j], bh[i]
                c = compute_c5(bh)
                if c < bc5 - 1e-14:
                    bc5, imp = c, True
                else:
                    bh[i], bh[j] = bh[j], bh[i]
        if not imp:
            break
    return bh, bc5
```
