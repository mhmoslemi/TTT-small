import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        h = h * (target / s)
    else:
        h = h + (target - s) / n
    return np.clip(h, 0.0, 1.0)

def corr_grad_k(h, n, k):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n-k] += (1.0 - h[k:n])
        g[k:] -= h[:n-k]
    else:
        m = -k
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

def smooth_obj_and_grad(x, T, n):
    dx = 2.0 / n
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = T * (corr - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    log_sum = mx + np.log(exps.sum())
    obj = (m + log_sum / T) * dx
    log_w = shifted - log_sum
    w = np.exp(log_w)
    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - (n - 1)
        g += w[idx] * corr_grad_k(h, n, k)
    g *= dx
    return obj, g

def coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        a[center + k] = (1.0 - h[ip] if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
        a[center - k] = (1.0 - h[im] if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
    dx = 2.0 / n
    ct = corr_base[center]
    def f1d(x):
        v = corr_base + x * a
        v[center] = ct + x - x*x
        return np.max(v) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def cd_sweep(h, n, order=None):
    if order is None:
        order = np.random.permutation(n)
    for i in order:
        h[i] = coord_opt(h, n, i)
        h = project(h, n)
    return h

    def obj(x):
        return smooth_obj_and_grad(x, T, n)[0]
    def jac(x):
        return smooth_obj_and_grad(x, T, n)[1]
    def con_fun(x):
        return x.sum() - n/2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   jac=jac, options={'maxiter': maxiter, 'ftol': 1e-18})
    return project(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        h = h * (target / s)
    else:
        h = h + (target - s) / n
    return np.clip(h, 0.0, 1.0)

def corr_grad_k(h, n, k):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n-k] += (1.0 - h[k:n])
        g[k:] -= h[:n-k]
    else:
        m = -k
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    return g

def smooth_obj_grad(x, T, n):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    sh = T * (corr - m)
    mx = sh.max()
    exps = np.exp(sh - mx)
    log_s = mx + np.log(exps.sum())
    obj = (m + log_s / T) * dx
    log_w = sh - log_s
    w = np.exp(log_w)
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - (n - 1)
        g += w[idx] * corr_grad_k(h, n, k)
    g *= dx
    return obj, g

def coord_opt(h, n, i):
    h_b = h.copy()
    h_b[i] = 0.0
    cb = np.correlate(h_b, 1.0 - h_b, mode='full')
    c = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip, im = i + k, i - k
        a[c+k] = (1.0 - h[ip] if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
        a[c-k] = (1.0 - h[im] if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
    dx = 2.0 / n
    ct = cb[c]
    def f1d(x):
        v = cb + x * a
        v[c] = ct + x - x * x
        return np.max(v) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

    order = np.random.permutation(n)
    for i in order:
        h[i] = coord_opt(h, n, i)
        h = project(h, n)
    return h

        return smooth_obj_grad(x, T, n)[0]
    def jac(x):
        return smooth_obj_grad(x, T, n)[1]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   jac=jac, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project(np.clip(res.x, 0, 1), n)
