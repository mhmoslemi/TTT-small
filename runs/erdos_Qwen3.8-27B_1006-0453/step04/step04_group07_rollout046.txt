def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = corr_arr(best_h)
    best_c5 = np.max(corr) * dx
    c0_idx = n - 1
    ks = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            def obj(v):
                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                return np.max(cn)
            v0 = obj(0.0)
            v1r = obj(1.0)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            best_v = res.x
            best_f = res.fun
            if v0 < best_f:
                best_v, best_f = 0.0, v0
            if v1r < best_f:
                best_v, best_f = 1.0, v1r
            if best_f * dx < best_c5 - 1e-13:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = corr_arr(best_h)
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_sa(h_start, n, t_end, max_iters=800000):
    h = (h_start > 0.5).astype(float)
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    T_start, T_end = 0.03, 0.0005
    for it in range(max_iters):
        if time.time() > t_end:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)
        i1 = ones_list[np.random.randint(len(ones_list))]
        i0 = zeros_list[np.random.randint(len(zeros_list))]
        h[i1] = 0.0
        h[i0] = 1.0
        c5_new = compute_c5(h)
        if c5_new <= c5_cur or np.random.rand() < np.exp(-(c5_new - c5_cur) / T):
            c5_cur = c5_new
            ones_list.remove(i1)
            ones_list.append(i0)
            zeros_list.remove(i0)
            zeros_list.append(i1)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    return h_best, c5_best

def smooth_c5(h, T):
    dx = 2.0 / len(h)
    corr = corr_arr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp_opt(h0, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h), n_new)

def itls(h_start, n, t_end, max_restarts=20):
    """Iterated Local Search with increasing perturbation."""
    best_h, best_c5 = h_start.copy(), compute_c5(h_start)
    for r in range(max_restarts):
        if time.time() > t_end:
            break
        scale = 0.02 + 0.15 * (r / max_rest())
        h_p = best_h + scale * np.random.randn(n)
        h_p = project(h_p, n)
        h_c, c5_c = coord_descent(h_p, n, t0_local + (t_end - t0_local) * 0.3, max_sweeps=100)
        if c5_c < best_c5:
            best_h, best_c5 = h_c, c5_c
    return best_h, best_c5

def max_rest():
    return 20.0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[::2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Binary SA - explore discrete landscape aggressively
    h_bin = (best_h > 0.5).astype(float)
    h_b, c5_b = binary_sa(h_bin, n, t0 + budget_s * 0.12, max_iters=600000)
    if c5_b < best_c5:
        best_h, best_c5 = project(h_b, n), c5_b

    # Phase 2: Coordinate descent on best
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.35, max_sweeps=3000)

    # Phase 3: Iterated local search (perturb + coordinate descent)
    for r in range(15):
        if elapsed() > budget_s * 0.52:
            break
        scale = 0.03 + 0.12 * (r / 15.0)
        h_p = best_h + scale * np.random.randn(n)
        h_p = project(h_p, n)
        h_c, c5_c = coord_descent(h_p, n, t0 + budget_s * 0.52, max_sweeps=150)
        if c5_c < best_c5:
            best_h, best_c5 = h_c, c5_c

    # Phase 4: SLSQP with smooth max at varying T
    for T in [200, 800, 3000, 10000]:
        if elapsed() > budget_s * 0.65:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 5: More coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.75, max_sweeps=2000)

    # Phase 6: Binary SA on current best (shorter, for fine-tuning)
    h_bin2 = (best_h > 0.5).astype(float)
    h_b2, c5_b2 = binary_sa(h_bin2, n, t0 + budget_s * 0.82, max_iters=300000)
    if c5_b2 < best_c5:
        best_h, best_c5 = h_bin2, c5_b2
        best_h, best_c5 = coord_descent(project(best_h, n), n, t0 + budget_s * 0.88, max_sweeps=500)

    # Phase 7: Try different resolutions
    for n_try in [120, 200, 240, 320]:
        if elapsed() > budget_s * 0.92:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t0 + budget_s * 0.93, max_sweeps=200)
        if c5_tr < best_c5:
            best_h, best_c5, n = h_tr, c5_tr, n_try

    # Phase 8: Final polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=500)
    if elapsed() < budget_s * 0.98:
        try:
            hc = slsqp_opt(best_h, 50000, n, maxiter=1500, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
