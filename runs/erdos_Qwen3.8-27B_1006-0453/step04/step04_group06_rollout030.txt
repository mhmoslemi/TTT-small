def coord_opt_grid(h, n, i, M=200):
    """Optimize coordinate i using grid + local refinement."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    dx = 2.0 / n
    
    # Compute effect vector a
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        if ip < n:
            a[center + k] += (1.0 - h[ip])
        if im >= 0:
            a[center + k] -= h[im]
        if im >= 0:
            a[center - k] += (1.0 - h[im])
        if ip < n:
            a[center - k] -= h[ip]
    
    # Grid evaluation
    grid = np.linspace(0.0, 1.0, M)
    # Correlation at each grid point:
    # c[m] = corr_base + grid[m] * a  (for k != 0)
    # c[m][center] = corr_base[center] + grid[m] - grid[m]^2
    base = corr_base.copy()

    # Shape: (M, 2n-1)
    corr_matrix = (corr_base[None, :] + grid[:, None] * a[None, :])
    corr_matrix[:, center] = base[center] + grid - grid**2
    
    obj = corr_matrix.max(axis=1) * dx
    best_m = np.argmin(obj)
    x_best = grid[best_m]
    
    # Local refinement around best grid point
    lo = grid[max(0, best_m - 1)]
    hi = grid[min(M-1, best_m + 1)]
    def f1d(x):
        vals = corr_base + x * a
        vals[center] = base[center] + x - x*x
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded', options={'xatol': 1e-12})
    return res.x

def slsqp_opt(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def obj_grad(x):

        m = corr.max()

        weights /= weights.sum()

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        h.fill(0.5)
        return h
    if abs(s - target) < 1e-11:
        return h
    if s > target:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    else:
        room = 1.0 - h
        rt = room.sum()
        if rt < 1e-12:
            h *= target / s
        else:
            h += room * (target - s) / rt
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    orig = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = orig
    center = n - 1
    dx = 2.0 / n
    ks = np.arange(1, n)
    ip = i + ks
    im = i - ks
    ip_ok = ip < n
    im_ok = im >= 0
    ip_safe = np.clip(ip, 0, n - 1)
    im_safe = np.clip(im, 0, n - 1)
    a = np.zeros(2 * n - 1)
    a[center + ks] = np.where(ip_ok, 1.0 - h[ip_safe], 0.0) - np.where(im_ok, h[im_safe], 0.0)
    a[center - ks] = np.where(im_ok, 1.0 - h[im_safe], 0.0) - np.where(ip_ok, h[ip_safe], 0.0)
    c0 = corr_base[center]
    M = 60
    grid = np.linspace(0.0, 1.0, M)
    lines = corr_base[None, :] + grid[:, None] * a[None, :]
    lines[:, center] = c0 + grid - grid * grid
    obj = lines.max(axis=1) * dx
    bi = int(np.argmin(obj))
    lo = grid[max(0, bi - 2)]
    hi = grid[min(M - 1, bi + 2)]
    if hi - lo < 1e-10:
        return float(grid[bi])
    def f1(x):
        v = corr_base + x * a
        v[center] = c0 + x - x * x
        return float(np.max(v) * dx)
    res = minimize_scalar(f1, bounds=(lo, hi), method='bounded', options={'xatol': 1e-12})
    return float(res.x)

def fast_cd(h, n, t_end, max_sweeps=10000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            xn = coord_opt(best_h, n, i)
            if abs(xn - best_h[i]) > 1e-11:
                best_h[i] = xn
        best_h = project(best_h, n)
        c5n = compute_c5(best_h)
        if c5n < best_c5 - 1e-12:
            best_c5 = c5n
            stall = 0
        else:
            stall += 1
        if stall >= 10:
            break
    return best_h, best_c5

def slsqp_step(h, T, n, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return float((m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx)
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        w = np.exp(T * (c - m))
        w /= w.sum()
        g = np.zeros(n)
        ctr = n - 1
        for idx in range(2 * n - 1):
            wi = w[idx]
            if wi < 1e-14:
                continue
            k = idx - ctr
            if k == 0:
                g += wi * (1.0 - 2.0 * x)
            elif k > 0:
                g[:n-k] += wi * (1.0 - x[k:])
                g[k:] -= wi * x[:n-k]
            else:
                m2 = -k
                g[m2:] += wi * (1.0 - x[:n-m2])
                g[:n-m2] -= wi * x[m2:]
        return g * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        xv = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.15 * np.cos(np.pi * xv) + 0.08 * np.cos(2 * np.pi * xv), 0, 1)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent (35%)
    best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.35)

    # Phase 2: SLSQP with increasing T (15%)
    slsqp_end = t0 + budget_s * 0.50
    for T in [30, 100, 500, 2000]:
        if time.time() > slsqp_end:
            break
        hc = slsqp_step(best_h, T, n, 250)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Structured perturbation + re-optimize (30%)
    pert_end = t0 + budget_s * 0.80
    xv = np.linspace(0, 2, n, endpoint=False)
    for it in range(30):
        if time.time() > pert_end:
            break
        m1 = np.random.randint(1, 6)
        m2 = np.random.randint(2, 9)
        p1 = np.random.uniform(0, 2 * np.pi)
        p2 = np.random.uniform(0, 2 * np.pi)
        eps = 0.04 + 0.04 * np.random.random()
        hp = best_h + eps * (np.sin(m1 * np.pi * xv + p1) + 0.5 * np.sin(m2 * np.pi * xv + p2))
        hp = project(hp, n)
        cd_time = min(budget_s * 0.025, pert_end - time.time())
        if cd_time < 3:
            continue
        hc, c5v = fast_cd(hp, n, time.time() + cd_time, max_sweeps=2000)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Final CD + SLSQP polish (15%)
    best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.93)
    if time.time() < t_end - 15:
        hc = slsqp_step(best_h, 10000, n, 300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = fast_cd(best_h, n, t_end - 2, max_sweeps=1000)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
