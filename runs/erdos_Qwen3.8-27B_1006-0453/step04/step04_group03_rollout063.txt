from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:

    else:
        if s < 1e-14:
            return np.clip(h, 0, 1)
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def compute_grad_at_i(h, i, n):
    """Compute d corr / d h[i] as a vector of length 2n-1."""
    k_vals = np.arange(-(n-1), n)
    idx_pos = i + k_vals
    idx_neg = i - k_vals
    mask1 = (idx_pos >= 0) & (idx_pos < n)
    mask2 = (idx_neg >= 0) & (idx_neg < n)
    coef = np.zeros(2*n-1)
    coef[mask1] += (1.0 - h[idx_pos[mask1]])
    coef[mask2] -= h[idx_neg[mask2]]
    return coef

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.clip(h, 0, 1)
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

    m = np.max(vals)

                hc = slsqp_polish(h_i, T, n_try, 500, 1e-13)
                hc, c5v = coord_descent(hc, n_try, time.time() + 4.0, n_sweeps=50)
                if c5v < best_c5:
                    best_c5, best_h, n = c5v, hc, n_try
            except Exception:
                pass

    # Phase 5: Final ultra-precise polish (15% budget)
    t_p5 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent(best_h, n, t_p5, n_sweeps=500)

    for T in [30000, 150000]:
        if elapsed() > budget_s * 0.995:
            break
        try:
            hc = slsqp_polish(best_h, T, n, 1500, 1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.999, n_sweeps=200)
    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
