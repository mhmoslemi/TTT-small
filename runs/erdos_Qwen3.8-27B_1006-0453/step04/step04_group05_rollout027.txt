```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def comp_c5(h):
    N = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / N)

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    d_lags = idx - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    pmd = p_idx - d_vals
    vpd = (pd >= 0) & (pd < N)
    vpmd = (pmd >= 0) & (pmd < N)
    hpd = np.where(vpd, h[np.clip(pd, 0, N - 1)], 0.0)
    hpmd = np.where(vpmd, h[np.clip(pmd, 0, N - 1)], 0.0)
    return -(1.0 - hpd) * vpd.astype(float) + hpmd * vpmd.astype(float)

def steepest_bin(h, N, deadline, max_sw=400, rng=None):
    h = h.astype(float).copy()
    stuck = 0
    for _ in range(max_sw):
        if time.time() > deadline:
            break
        D = compute_D(h, N)
        c = compute_corr(h)
        P = float(c.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if not len(ones) or not len(zeros):
            break
        best_pk = P
        best_pq = None
        for p in ones:
            base = c - D[p, :]
            cands = base[None, :] + D[zeros, :]
            npk = cands.max(axis=1)
            mi = int(np.argmin(npk))
            v = float(npk[mi])
            if v < best_pk - 1e-12:
                best_pk = v
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            stuck += 1
            if stuck > 3 and rng is not None:
                for _ in range(4):
                    p = ones[rng.randint(len(ones))]
                    q = zeros[rng.randint(len(zeros))]
                    h[p] = 0.0
                    h[q] = 1.0
            else:
                break
        else:
            stuck = 0
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return h

def grad_topk(h, N, k):
    corr = compute_corr(h)
    Nc = len(corr)
    kc = min(k, Nc)
    top = np.argpartition(corr, -kc)[-kc:]
    w = 1.0 / kc
    g = np.zeros(N)
    center = N - 1
    for m in top:
        lag = m - center
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        elif lag > 0:
            g[:N - lag] += w * (1.0 - h[lag:])
            g[lag:] -= w * h[:N - lag]
        else:
            L = -lag
            g[:N - L] += w * (1.0 - h[L:])
            g[L:] -= w * h[:N - L]
    return g, corr

def cont_opt(h, N, target, deadline, k=30, lr0=0.02, beta=0.7, rng=None):
    if rng is None:

    best = h.copy()

    cur_lr = lr0
    no_imp = 0
    it = 0

        g, corr = grad_topk(h, N, k)
        pk = float(corr.max())
        if pk < best_pk - 1e-11:
            best_pk = pk
            best = h.copy()
            no_imp = 0
        else:
            no_imp += 1

        npp = float(np.max(compute_corr(h_new)))
