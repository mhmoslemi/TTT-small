def project(h, n):

    diff = target - h.sum()

        return h

        if tot < 1e-12:

    else:
        s = h.sum()
        if s < 1e-12:
            return np.full(n, target / n)
        h = h * (target / s)

def eval_f1d(x, cb, ct, a, center):
    vals = cb + x * a
    vals = vals.copy()
    vals[center] = ct + x - x * x
    return np.max(vals)

def min_1d_f1d(cb, ct, a, center, lo=0.0, hi=1.0, tol=1e-11):
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = eval_f1d(x1, cb, ct, a, center)
    f2 = eval_f1d(x2, cb, ct, a, center)
    for _ in range(60):
        if hi - lo < tol:

        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = eval_f1d(x1, cb, ct, a, center)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = eval_f1d(x2, cb, ct, a, center)
    return (lo + hi) / 2.0

def coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    center = n - 1
    a = np.zeros(2 * n - 1)
    for k in range(1, n):
        ip = i + k
        im = i - k
        val_p = (1.0 - h[ip]) if ip < n else 0.0
        val_m = h[im] if im >= 0 else 0.0
        a[center + k] = val_p - val_m
        val_p2 = (1.0 - h[im]) if im >= 0 else 0.0
        val_m2 = h[ip] if ip < n else 0.0
        a[center - k] = val_p2 - val_m2

def corr_grad_correct(h, n, k):
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        m = -k
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def subgradient_step(h, n, step_size, T):
    corr = get_corr(h)
    m = corr.max()
    weights = np.exp(T * (corr - m))
    weights /= weights.sum()
    center = n - 1
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] < 1e-8:
            continue
        k = idx - center
        grad += weights[idx] * corr_grad_correct(h, n, k)
    h_new = h - step_size * grad
    return project(h_new, n)
