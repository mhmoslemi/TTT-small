import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        return np.clip(h * (target / max(s, 1e-15)), 0.0, 1.0)
    room = 1.0 - h
    tot = room.sum()
    if tot < 1e-13:
        return h
    return np.clip(h + room * ((target - s) / tot), 0.0, 1.0)

def fast_cd(h, n, t_end, n_sweeps=100000, seed=0):
    rng = np.random.RandomState(seed)
    best_h = h.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float).copy()
    best_c5 = float(np.max(corr)) * dx
    shifts = np.arange(-(n - 1), n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    NGS = 28
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            h_old = best_h[i]
            pos_idx = i + shifts
            neg_idx = i - shifts
            vp = (pos_idx >= 0) & (pos_idx < n)
            vn = (neg_idx >= 0) & (neg_idx < n)
            hp = np.where(vp, best_h[np.clip(pos_idx, 0, n - 1)], 0.0)
            hn = np.where(vn, best_h[np.clip(neg_idx, 0, n - 1)], 0.0)
            coef = (1.0 - hp) - hn
            
            lo, hi = -h_old, 1.0 - h_old
            if hi - lo < 1e-12:
                continue
            # Golden section
            a, b = lo, hi
            c_pt = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            fc = float(np.max(corr + c_pt * coef))
            fd = float(np.max(corr + d_pt * coef))
            for _ in range(NGS):
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b - gr * (b - a)
                    fc = float(np.max(corr + c_pt * coef))
                else:
                    a = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a + gr * (b - a)
                    fd = float(np.max(corr + d_pt * coef))
            delta = (a + b) * 0.5
            # Check endpoints
            f0 = float(np.max(corr))
            flo = float(np.max(corr + lo * coef))
            fhi = float(np.max(corr + hi * coef))
            fmid = float(np.max(corr + delta * coef))
            if flo < fmid and flo < fhi and flo < f0:
                delta = lo
            elif fhi < fmid and fhi < flo and fhi < f0:
                delta = hi
            if abs(delta) > 1e-12:
                best_h[i] = np.clip(best_h[i] + delta, 0.0, 1.0)
                delta_actual = best_h[i] - h_old
                if abs(delta_actual) > 1e-13:
                    corr += delta_actual * coef
                    c5_new = float(np.max(corr)) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter=500, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + float(np.log(np.sum(np.exp(T * (c - m)))))) / T * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

        else:

        h_p = project_to_sum(h_p, n / 2)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

    bc5 = float(np.max(corr)) * dx
    sh = np.arange(-(n - 1), n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    NGS = 28
    for _ in range(n_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in rng.permutation(n):
            ho = bh[i]
            pi = i + sh
            ni = i - sh
            vp = (pi >= 0) & (pi < n)
            vn = (ni >= 0) & (ni < n)
            hp = np.where(vp, bh[np.clip(pi, 0, n - 1)], 0.0)
            hn = np.where(vn, bh[np.clip(ni, 0, n - 1)], 0.0)
            cf = (1.0 - hp) - hn
            lo, hi = -ho, 1.0 - ho
            if hi - lo < 1e-12:
                continue
            a, b = lo, hi
            cp = b - gr * (b - a)
            dp = a + gr * (b - a)
            fc = float(np.max(corr + cp * cf))
            fd = float(np.max(corr + dp * cf))
            for _ in range(NGS):
                if fc < fd:
                    b = dp
                    dp = cp
                    fd = fc
                    cp = b - gr * (b - a)
                    fc = float(np.max(corr + cp * cf))
                else:
                    a = cp
                    cp = dp
                    fc = fd
                    dp = a + gr * (b - a)
                    fd = float(np.max(corr + dp * cf))
            d = (a + b) * 0.5
            f0 = float(np.max(corr))
            fL = float(np.max(corr + lo * cf))
            fH = float(np.max(corr + hi * cf))
            fM = float(np.max(corr + d * cf))
            if fL < fM and fL < fH and fL < f0:
                d = lo
            elif fH < fM and fH < fL and fH < f0:
                d = hi
            if abs(d) > 1e-12:
                nv = float(np.clip(bh[i] + d, 0.0, 1.0))
                da = nv - ho
                if abs(da) > 1e-13:
                    bh[i] = nv
                    corr += da * cf
                    c5n = float(np.max(corr)) * dx
                    if c5n < bc5:
                        bc5 = c5n
                        imp = True
        if not imp:
            break
    return bh, bc5

def slsqp(h, T, n, miter=400, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(c))
        return (m + float(np.log(np.sum(np.exp(T * (c - m)))))) / T * dx
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: float(x.sum()) - n / 2.0}],
                   options={'maxiter': miter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

The problem is to minimize max_k ∫₀² h(x)(1-h(x+k)) dx over h ∈ [0,1] with ∫h=1.

If h were the indicator of a set A ⊂ [0,2] with |A|=1, then the expression becomes ∫_A (1 - 1_{A+k}(x)) dx = |A| - |A ∩ (A+k)| = 1 - |A ∩ (A+k)|.

So we want to minimize max_k (1 - |A ∩ (A+k)|), which is equivalent to maximizing min_k |A ∩ (A+k)|.

For a measurable set A ⊂ [0,2] with |A|=1, we want to maximize the minimum overlap |A ∩ (A+k)| over all shifts k.

This is related to the "horrible sets" problem in additive combinatorics!

For A = [0,1], |A ∩ (A+k)| = max(0, 1-|k|) for k in [-1,1] and 0 otherwise. So the minimum is 0.

For A to have positive overlap with ALL shifts, A needs to be "thick" in some sense. But |A|=1 in [0,2], so A can cover at most half of [0,2].

The optimal A should be a set that maximizes the minimum autocorrelation. This is related to "barrier" or "separating" sets.
