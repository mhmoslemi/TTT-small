    return (lo + hi) * 0.5

def coord_descent(h, n, t_end, max_sweeps=1000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end or no_improve >= 5:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            v_opt = coord_opt_1d(best_h, n, i)
            if abs(v_opt - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = v_opt
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
    return best_h, best_c5

def subgrad_step(h, n, step):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    thresh = m - 0.02 * n
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    center = n - 1
    grad = np.zeros(n)
    for mi in active:
        k = mi - center
        if k == 0:
            grad += (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:] -= h[:n - k]
        else:
            mk = -k
            grad[mk:] += (1.0 - h[:n - mk])
            grad[:n - mk] -= h[mk:]
    grad /= len(active)
    return project(h - step * grad, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 160
        x = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x), 0, 1)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent warmup
    t1 = t0 + budget_s * 0.20
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=500)

    # Phase 2: SLSQP with smooth max, increasing T
    t2 = t0 + budget_s * 0.38
    for T in [30, 100, 300, 1000, 3000]:
        if time.time() > t2:
            break
        try:
            hc = slsqp_smooth(best_h.copy(), T, n, maxiter=800)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 3: Alternate subgradient + coordinate descent
    t3 = t0 + budget_s * 0.58
    step = 0.05
    for it in range(2000):
        if time.time() > t3:
            break
        h_sg = subgrad_step(best_h, n, step)
        # Quick 1-sweep coordinate polish
        hc = h_sg.copy()
        for i in range(n):
            if time.time() > t3:
                break
            vc = coord_opt_1d(hc, n, i)
            hc[i] = vc
        hc = project(hc, n)
        c5v = compute_c5(hc)
        if c5v < best_c5 - 1e-13:
            best_c5 = c5v
            best_h = hc
            step = min(step * 1.03, 0.2)
        else:
            step *= 0.97
        # Periodic random restart
        if it % 20 == 10:
            for amp in [0.03, 0.08, 0.15]:
                h_p = best_h + amp * np.random.randn(n)
                h_p = project(h_p, n)
                hc2, c5p = coord_descent(h_p, n, min(time.time() + 5, t3), max_sweeps=30)
                if c5p < best_c5:
                    best_c5 = c5p
                    best_h = hc2

    # Phase 4: Extended coordinate descent
    t4 = t0 + budget_s * 0.78
    best_h, best_c5 = coord_descent(best_h, n, t4, max_sweeps=2000)

    # Phase 5: Multi-resolution
    t5 = t0 + budget_s * 0.90
    for n_try in [128, 192, 256, 320]:
        if time.time() > t5:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h, period=2.0)
        hi = project(hi, n_try)
        c5i = compute_c5(hi)
        hc, c5v = coord_descent(hi, n_try, min(time.time() + budget_s * 0.04, t5), max_sweeps=80)
        if c5v < c5i:
            c5i, hi = c5v, hc
        # Also try SLSQP at this resolution
        try:
            hs = slsqp_smooth(hi.copy(), 500, n_try, maxiter=400)
            c5s = compute_c5(hs)
            if c5s < c5i:
                c5i, hi = c5s, hs
        except Exception:
            pass
        if c5i < best_c5:
            best_c5 = c5i
            best_h = hi
            n = n_try

    # Phase 6: Final SLSQP polish
    t6 = t_end - 30
    for T in [5000, 20000, 80000]:
        if time.time() > t6:
            break
        try:
            hc = slsqp_smooth(best_h.copy(), T, n, maxiter=1500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 7: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t_end - 2, max_sweeps=500)
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
