    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.9 * (it / max(n_iters, 1))
        step = base_step * (0.3 + 0.7 * cool)
        pert = project(work + step * rng.standard_normal(n), n)
        try:
            hc = fast_cd(pert, n, min(time.time() + budget_per_basin, t_end), max_sweeps=30, rng=rng)
        except NameError:
            hc = fast_cd(pert, n, t_end, max_sweeps=20, rng=rng)
        cv = compute_c5(hc)
        temp = 0.003 * cool + 0.0002
        if cv < work_c5 or rng.random() < np.exp(-(cv - work_c5) / max(temp, 1e-8)):
            work, work_c5 = hc, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h_old), n_new)

budget_per_basin = 8.0

def run(seed=42, budget_s=1100, **kwargs):
    global budget_per_basin
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    budget_per_basin = max(3.0, budget_s / 200.0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 120
        best_h = np.zeros(n0)
        best_h[:n0 // 2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Fast coordinate descent
    t1 = t0 + budget_s * 0.20
    best_h, best_c5 = fast_cd(best_h, n, t1, max_sweeps=200, rng=rng)

    # Phase 2: SLSQP polish
    for T in [50, 300, 1200, 5000]:
        if time.time() > t0 + budget_s * 0.32:
            break
        hc = slsqp_polish(best_h, T, n, maxiter=600)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv
            best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.34, max_sweeps=50, rng=rng)

    # Phase 3: Subgradient global search
    if time.time() < t0 + budget_s * 0.50:
        best_h, best_c5 = subgrad_global(best_h, n, t0 + budget_s * 0.50, rng)
        best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.55, max_sweeps=80, rng=rng)

    # Phase 4: Basin hopping
    if time.time() < t0 + budget_s * 0.72:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.72, rng, n_iters=80, base_step=0.03)
        best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.75, max_sweeps=100, rng=rng)

    # Phase 5: Multi-resolution
    for n_try in [160, 200, 240, 280, 320]:
        if time.time() > t0 + budget_s * 0.88:
            break
        if n_try == n:
            continue
        h_interp = interpolate_h(best_h, n_try)
        h_interp, c5_new = fast_cd(h_interp, n_try, t0 + budget_s * 0.89, max_sweeps=40, rng=rng)
        if c5_new < best_c5:
            for T in [500, 3000]:
                if time.time() > t0 + budget_s * 0.90:
                    break
                hc = slsqp_polish(h_interp, T, n_try, maxiter=500)
                cv = compute_c5(hc)
                if cv < c5_new:
                    h_interp, c5_new = hc, cv
            h_interp, c5_new = fast_cd(h_interp, n_try, t0 + budget_s * 0.91, max_sweeps=50, rng=rng)
            if c5_new < best_c5:
                best_h, best_c5, n = h_interp, c5_new, n_try

    # Phase 6: Final polish
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.97, max_sweeps=200, rng=rng)
    for T in [10000, 40000]:
        if time.time() > t0 + budget_s * 0.99:
            break
        hc = slsqp_polish(best_h, T, n, maxiter=1000)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
