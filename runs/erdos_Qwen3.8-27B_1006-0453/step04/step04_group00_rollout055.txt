import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fw_linear_step(g, n):
    """Solve min g^T h s.t. 0<=h_i<=1, sum h_i = n/2"""
    target = n / 2.0
    h = np.zeros(n)
    idx = np.argsort(g)
    remaining = target
    for i in idx:
        if remaining >= 1.0 - 1e-14:
            h[i] = 1.0
            remaining -= 1.0
        elif remaining > 1e-14:
            h[i] = remaining
            remaining = 0.0
            break
    return h

def subgrad_topk(h, topk=5):
    """Compute subgradient of max_k corr[k]*dx using top-k averaging"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-topk:]
    grad = np.zeros(n)
    for m in topk_idx:
        lag = m - (n - 1)
        if lag == 0:
            # d/dh[j] of sum_i h[i](1-h[i]) = sum_i (1-2h[i]) at lag 0
            grad += (1.0 - 2.0 * h)
        else:
            j = np.arange(n)
            j1 = j - lag
            j2 = j + lag
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            grad += (1.0 - h[np.clip(j1, 0, n-1)]) * v1.astype(float)
            grad -= h[np.clip(j2, 0, n-1)] * v2.astype(float)
    return grad * dx / topk

def slsqp_sum_topk(h, n, K, maxiter=1000, ftol=1e-15):
    """SLSQP minimizing sum of top-K correlations"""
    def obj(x):
        c = compute_corr(x)
        topk = np.sort(c)[-K:]
        return topk.sum() * (2.0 / n)
    
    def obj_grad(x):
        c = compute_corr(x)
        topk_idx = np.argsort(c)[-K:]
        g = np.zeros(n)
        for m in topk_idx:
            lag = m - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * x)
            else:
                j = np.arange(n)
                j1 = j - lag
                j2 = j + lag
                v1 = (j1 >= 0) & (j1 < n)
                v2 = (j2 >= 0) & (j2 < n)
                g += (1.0 - x[np.clip(j1, 0, n-1)]) * v1.astype(float)
                g -= x[np.clip(j2, 0, n-1)] * v2.astype(float)
        return g * (2.0 / n)
    
    def con(x):
        return x.sum() - n / 2.0
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or compute_c5(res.x) < compute_c5(h):
            return project_h(res.x)
    except Exception:
        pass
    return h

def coord_opt_fast(h, n, i):
    """Optimize h[i] given all other coordinates fixed"""
    # corr[m] = sum_j h[j]*(1-h[j-lag_m]) where lag_m = m-(n-1)
    # d corr[m]/d h[i] = (1-h[i-lag_m]) - h[i+lag_m] (with validity)
    base_h = h.copy()
    base_h[i] = 0.0
    corr_base = compute_corr(base_h)
    
    coef = np.zeros(2*n-1)
    for m in range(2*n-1):
        lag = m - (n-1)
        # d/dh[i] of term h[i]*(1-h[i-lag]) = 1-h[i-lag] (if i-lag valid)
        # d/dh[i] of term h[i+lag]*(1-h[i]) = -(h[i+lag]) (if i+lag valid)
        j1 = i - lag
        j2 = i + lag
        if 0 <= j1 < n:
            coef[m] += (1.0 - h[j1])
        if 0 <= j2 < n:
            coef[m] -= h[j2]
    
    dx = 2.0 / n
    def f1d(hi):
        return np.max(corr_base + hi * coef) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_fast(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:

            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fw_linear_step(g, n):
    target = n / 2.0
    h = np.zeros(n)
    idx = np.argsort(g)
    remaining = target
    for i in idx:
        if remaining >= 1.0 - 1e-14:
            h[i] = 1.0
            remaining -= 1.0
        elif remaining > 1e-14:
            h[i] = remaining
            remaining = 0.0
            break
    return h

            j2 = j + lag

            lag = m - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * x)
            else:
                j = np.arange(n)
                j1 = j - lag
                j2 = j + lag

                g += (1.0 - x[np.clip(j1, 0, n-1)]) * v1.astype(float)
                g -= x[np.clip(j2, 0, n-1)] * v2.astype(float)
        return g * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        if compute_c5(res.x) < compute_c5(h):
            return project_h(res.x)
    except Exception:
        pass
    return h

def coord_opt_fast(h, n, i):
    base_h = h.copy()
    base_h[i] = 0.0
    corr_base = compute_corr(base_h)
    coef = np.zeros(2 * n - 1)

        j2 = i + lag

            coef[m] -= h[j2]

        return np.max(corr_base + hi * coef) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
