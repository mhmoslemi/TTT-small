**Idea: Lagrangian / Dual approach**
The problem is: min_h max_k f_k(h) s.t. g(h) = 0, 0≤h≤1

The dual function is: min_h max_k f_k(h) + λ*g(h) + sum μ_i*h_i + sum ν_i*(1-h_i)

For a fixed λ, this is a QP (since f_k is quadratic in h). We can solve it with a QP solver. Then update λ using subgradient.

But this is essentially what SLSQP does internally. And it's slow for large n.

**Idea: Game-theoretic / No-regret approach**
View this as a game: Player 1 chooses h, Player 2 chooses the shift k. Player 1 wants to minimize, Player 2 wants to maximize.

Using no-regret algorithms (like Hedge or multiplicative weights):
- Player 2 maintains a distribution over shifts: q_k
- Player 1 best-responds: h = argmin_h sum_k q_k * f_k(h)
- Player 2 updates: q_k ∝ q_k * exp(eta * f_k(h))

This is the "Mirror Descent" or "Follow the Regularized Leader" approach. It converges to a saddle point.

But the best response for Player 1 is a QP (minimize sum of weighted quadratics), which can be solved with projected gradient.

This might be better than the soft-max approach because it naturally handles the max structure.

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def c5_grad(h, n, T, top_k=20):
    corr = compute_corr(h)
    m = corr.max()
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    center = n - 1
    act = np.argsort(-w)[:top_k]
    grad = np.zeros(n)
    for a in act:
        wa = w[a]
        if wa < 1e-10:
            break
        k = a - center
        if k == 0:
            grad += wa * (1.0 - 2.0 * h)
        elif 0 < k < n:
            grad[k:] += wa * (1.0 - h[:n - k])
            grad[:n - k] -= wa * h[k:]
        elif -n < k < 0:
            m2 = -k
            grad[:n - m2] += wa * (1.0 - h[m2:])
            grad[m2:] -= wa * h[:n - m2]
    return grad

def proj_grad_opt(h, n, t_end, T_init=80.0, lr_init=0.04, max_iter=50000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T = T_init
    lr = lr_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        g = c5_grad(best_h, n, T, top_k=20)
        h_new = project(best_h - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new
            stall = 0
            lr = min(lr * 1.015, 0.15)
            T *= 0.9999
        else:
            stall += 1
            lr *= 0.93
            if stall > 80:
                stall = 0
                lr = lr_init * np.random.uniform(0.5, 2.0)
                T = max(10.0, T_init * np.random.uniform(0.2, 3.0))
                best_h = project(best_h + 0.015 * np.random.randn(n), n)
    return best_h, best_c5

def coord_desc(h, n, t_end, max_sweeps=300, xatol=1e-11):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    center = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            best_h[i] = 0.0
            cb = compute_corr(best_h)
            a = np.zeros(2 * n - 1)
            for k in range(1, n):
                c = 0.0
                if i - k >= 0:
                    c += 1.0 - best_h[i - k]
                if i + k < n:
                    c -= best_h[i + k]
                a[center + k] = c
                a[center - k] = c
            best_h[i] = old
            ct = cb[center]
            def f(t):
                v = cb + t * a
                v[center] = ct + t - t * t
                return np.max(v) * dx
            res = minimize_scalar(f, bounds=(0.0, 1.0), method="bounded", options={"xatol": xatol})
            t_new = res.x
            if abs(t_new - old) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def interp_n(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project(h_new, n_new)
