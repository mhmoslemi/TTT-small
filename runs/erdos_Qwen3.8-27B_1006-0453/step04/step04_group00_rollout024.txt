    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    if v1.any():
        coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    if v2.any():
        coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    dx = 2.0 / n
    res = minimize_scalar(_coord_obj, bounds=(0.0, 1.0), method='bounded',
                          args=(corr_base, coefs, dx), options={'xatol': 1e-13})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        active_p = np.argsort(corr)[-min(15, 2 * n - 1):]
        active_coords = set()
        for p in active_p:
            s = p - (n - 1)
            lo = max(0, -s)
            hi = min(n, n - s)
            for i in range(lo, hi):
                active_coords.add(i)
            lo2 = max(0, s)
            hi2 = min(n, n + s)
            for i in range(lo2, hi2):
                active_coords.add(i)
        order = list(active_coords)
        np.random.shuffle(order)
        rest = [i for i in range(n) if i not in active_coords]
        np.random.shuffle(rest)
        order = order + rest
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = exact_coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, n_iter, t_end):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(n_iter, 1))
        mod = it % 5
        if mod == 0:
            pert = (0.03 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 8)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.10 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.4
            pert = np.zeros(n)
            pert[mask] = (0.08 + 0.15 * cool) * np.random.randn(mask.sum())
        elif mod == 3:
            b = max(1, np.random.randint(n // 12, n // 4))
            s = np.random.randint(-b, b + 1)
            pert = (np.roll(work_h, s) - work_h) * 0.5
        else:
            pert = (0.02 + 0.06 * cool) * np.random.randn(n)
        h_p = project_h(work_h + pert, n)
        T = 300 + 10000 * cool
        hc = slsqp_polish(h_p, T, n, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    el = lambda: time.time() - t0

    # Phase 1: SLSQP annealing on parent
    for T in [10, 50, 200, 1000, 5000, 20000, 80000]:
        if el() > budget_s * 0.10:
            break
        hc = slsqp_polish(best_h, T, n, maxiter=1200, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Momentum gradient exploration
    for T in [200, 2000, 20000]:
        if el() > budget_s * 0.22:
            break
        h_m, c5_m = momentum_step(best_h, T, n, lr=0.05, beta=0.9,
                                  t_end=t0 + budget_s * 0.22, n_steps=300)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m
        hc = slsqp_polish(best_h, int(T * 5), n, maxiter=500, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Basin hopping (coarse)
    bh_h, bh_c5 = basin_hop(best_h, n, 55, t0 + budget_s * 0.40)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = bh_h

    # Phase 4: SLSQP re-polish
    for T in [5000, 50000, 300000]:
        if el() > budget_s * 0.50:
            break
        hc = slsqp_polish(best_h, T, n, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Coordinate descent (exact, active-set focused)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.68, max_sweeps=180)

    # Phase 6: Basin hopping round 2 (fine)
    bh_h, bh_c5 = basin_hop(best_h, n, 35, t0 + budget_s * 0.82)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = bh_h

    # Phase 7: Ultra-high-T SLSQP polish
    for T in [200000, 2000000, 10000000]:
        if el() > budget_s * 0.91:
            break
        hc = slsqp_polish(best_h, T, n, maxiter=3000, ftol=1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 8: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, max_sweeps=60)

    # Phase 9: Try alternate grid size if time remains
    if el() < budget_s * 0.94:
        alt_n = n + 40
        xs = np.linspace(0, 2, n)
        xa = np.linspace(0, 2, alt_n)
        h_wrap = np.tile(best_h, 3)
        h_alt_raw = np.interp(xa, xs, h_wrap)
        h_alt = project_h(h_alt_raw, alt_n)
        hc = slsqp_polish(h_alt, 5000, alt_n, maxiter=500, ftol=1e-14)
        hc = slsqp_polish(hc, 100000, alt_n, maxiter=800, ftol=1e-16)
        hc, c5v2 = coord_descent(hc, alt_n, min(t0 + budget_s * 0.95, time.time() + 30), max_sweeps=20)
        hc = project_h(hc, alt_n)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            n = alt_n

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
