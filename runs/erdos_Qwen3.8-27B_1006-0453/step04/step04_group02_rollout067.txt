```python
def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1e-6:
        return m * dx
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        s2 = s if s > 1e-12 else 1e-12
        h = h - h * (diff / s2)
        h = np.clip(h, 0.0, 1.0)
        s3 = h.sum()
        diff3 = target - s3
        if diff3 > 1e-12:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-12:
                h = h + room * (diff3 / tot)
    return h

def coord_descent(h, n, t_end, max_sweeps=200, rng=None, tol=1e-14):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_one(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-11:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - tol:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 2:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def sin_perturb(h, n, rng, amp=0.05):
    """Add a sinusoidal perturbation."""
    freq = rng.uniform(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n, endpoint=False)
    pert = h + amp * np.sin(2 * np.pi * freq * x + phase)
    return project(pert, n)

def block_perturb(h, n, rng, amp=0.04):
    """Perturb a random block of h."""
    block_size = rng.integers(n // 10, n // 3)
    start = rng.integers(0, n - block_size)
    pert = h.copy()
    pert[start:start+block_size] += amp * rng.standard_normal(block_size)
    return project(pert, n)

def active_set_step(h, n, t_end, n_steps=100, lr=0.005, top_k=5):
    """Gradient step on the active set (top-k correlations)."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for step in range(n_steps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Find top-k shifts
        top_idx = np.argsort(corr)[-top_k:]
        # Compute gradient for each active shift
        # d(corr[j])/d(h[i]) = (1-h[i+j-n+1])*[valid] - h[i-j+n-1]*[valid]
        grad = np.zeros(n)
        for j in top_idx:
            for i in range(n):
                a = i + j - (n - 1)
                b = i - j + (n - 1)
                g = 0.0
                if 0 <= a < n:
                    g += (1.0 - best_h[a])
                if 0 <= b < n:
                    g += (-best_h[b])
                grad[i] += g
        grad /= top_k
        # Normalize
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        # Step
        new_h = best_h - lr * grad / gn
        new_h = project(new_h, n)
        c5 = compute_c5(new_h)
        if c5 < best_c5:
            best_h = new_h
            best_c5 = c5
        else:
            lr *= 0.7
    return best_h, best_c5
```

The general formula gives: corr_base[n-1] + t*(1-h[i]) where h[i]=h_save.

So the special handling `vals[n-1] = c0 + t*(1-t)` is correct and necessary. Good.

```python
def update_corr(corr, h, n, i, old_val, new_val):
    """Update corr when h[i] changes from old_val to new_val."""
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return
    for j in range(2 * n - 1):
        a = i + j - (n - 1)
        b = i - j + (n - 1)
        change = 0.0
        if 0 <= a < n and a != i:
            change += delta * (1 - h[a])
        if 0 <= b < n and b != i:
            change += (-delta) * h[b]
        if j == n - 1:
            # Quadratic term: new_val*(1-new_val) - old_val*(1-old_val)
            change = new_val*(1-new_val) - old_val*(1-old_val)
            # But we also need to account for the "other" i terms... 
            # Actually for j=n-1: corr[n-1] = sum_m h[m]*(1-h[m])
            # Only the m=i term changes: new_val*(1-new_val) - old_val*(1-old_val)

            # But for j=n-1, a=i and b=i, so a==i and b==i, both skipped.
            # So we only have the quadratic change.

```

```python
def update_corr_at(corr, h, n, i, old_val, new_val):

    if abs(delta) < 1e-15:
        return corr
    for j in range(2 * n - 1):

            if 0 <= b < n:
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        s2 = max(s, 1e-12)
        h = h - h * (diff / s2)
        h = np.clip(h, 0.0, 1.0)
        s3 = h.sum()
        diff3 = target - s3
        if diff3 > 1e-12:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-12:
                h = h + room * (diff3 / tot)
        elif diff3 < -1e-12 and s3 > 1e-12:
            h = np.clip(h * (target / s3), 0.0, 1.0)
    return h

def coord_opt_one(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    dx = 2.0 / n
    d = np.zeros(2 * n - 1)
    for j in range(2 * n - 1):
        a = i + j - (n - 1)
        b = i - j + (n - 1)
        val = 0.0
        if 0 <= a < n:
            val += (1.0 - h[a])
        if 0 <= b < n and j != n - 1:
            val += (-h[b])
        d[j] = val
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200, rng=None, stall_patience=3):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_one(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-11:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= stall_patience:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=400, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def sin_perturb(h, n, rng, amp=0.04):
    freq = rng.uniform(1.0, 10.0)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n, endpoint=False)
    pert = h + amp * np.sin(2 * np.pi * freq * x + phase)
    return project(pert, n)

def block_perturb(h, n, rng, amp=0.03):
    bs = int(rng.integers(n // 12, n // 4))
    start = int(rng.integers(0, n - bs))
    pert = h.copy()
    pert[start:start + bs] += amp * rng.standard_normal(bs)
    return project(pert, n)

def active_step(h, n, t_end, n_steps=200, lr=0.008, top_k=5):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_lr = lr
    for step in range(n_steps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        j0 = n - 1
        for j in top_idx:
            for i in range(n):
                a = i + j - j0
                b = i - j + j0
                g = 0.0
                if 0 <= a < n:
                    g += (1.0 - best_h[a])
                if 0 <= b < n and j != j0:
                    g += (-best_h[b])
                if j == j0:
                    g = 1.0 - 2.0 * best_h[i]
                grad[i] += g
        grad /= float(top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        new_h = best_h - cur_lr * grad / gn
        new_h = project(new_h, n)
        c5 = compute_c5(new_h)
        if c5 < best_c5 - 1e-16:
            best_h = new_h
            best_c5 = c5
            cur_lr = min(cur_lr * 1.05, lr * 2)
        else:
            cur_lr *= 0.6
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish (12%)
    t1 = t0 + budget_s * 0.12
    for T in [30, 200, 1000, 4000]:
        if time.time() > t1:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=300, ftol=1e-14)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 2: Coordinate descent with FIXED bug (18%)
    t2 = t0 + budget_s * 0.30
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=120, rng=rng)

    # Phase 3: Multi-resolution (35%)
    t3 = t0 + budget_s * 0.65
    n_tries = [200, 240, 280, 320]
    if n not in n_tries:
        n_tries = [n] + n_tries
    for n_try in n_tries:
        if time.time() > t3:
            break
        h_int = interpolate_h(best_h, n_try)
        for T in [300, 3000]:
            if time.time() > t3:
                break
            try:
                hc = slsqp_opt(h_int, T, n_try, maxiter=300, ftol=1e-14)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5, n = hc, cv, n_try
                    h_int = interpolate_h(best_h, n_try)
            except Exception:
                pass
        if time.time() < t3:
            best_h2, best_c5_2 = coord_descent(h_int if n_try != n else best_h, n_try, min(t3, time.time() + 15), max_sweeps=40, rng=rng)
            if best_c5_2 < best_c5:
                best_h, best_c5, n = best_h2, best_c5_2, n_try

    # Phase 4: Basin hopping with structured perturbations (15%)
    t4 = t0 + budget_s * 0.80
    work_h = best_h.copy()
    n_w = len(work_h)
    n_iters = min(60, int((t4 - time.time()) / 2.0))
    for it in range(n_iters):
        if time.time() > t4:
            break
        cool = 1.0 - 0.8 * (it / max(n_iters, 1))
        amp = 0.02 + 0.03 * cool
        ptype = it % 4
        if ptype == 0:
            pert = work_h + amp * rng.standard_normal(n_w)
        elif ptype == 1:
            pert = sin_perturb(work_h, n_w, rng, amp)
```
