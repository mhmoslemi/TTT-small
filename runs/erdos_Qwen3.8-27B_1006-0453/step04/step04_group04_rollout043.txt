```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    """Project to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > 0 and s < target:
        # Scale up, but clip at 1
        # Use iterative approach
        for _ in range(100):
            h = np.clip(h + (target - h.sum()) * 0.5, 0, 1)
            if abs(h.sum() - target) < 1e-13:
                break
    elif s > target:
        for _ in range(100):
            h = np.clip(h - (h.sum() - target) * 0.5, 0, 1)
            if abs(h.sum() - target) < 1e-13:
                break
    return h

def compute_eff_off(h, n):
    """Precompute eff_off[i] for all i: effect on corr when h[i] goes 1→0."""
    n_lags = 2 * n - 1
    offsets = np.arange(n_lags) - (n - 1)
    eff = np.zeros((n, n_lags))
    for i in range(n):
        ip = i + offsets
        im = i - offsets
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        eff[i, vp] = -(1.0 - h[ip[vp]])
        eff[i, vm] += h[im[vm]]
    return eff

def binary_swap_search(h, n, t_end, max_sweeps=200):

    s = h.sum()
    if s != n/2:
        # Adjust
        order = np.argsort(-h)

        idx_sorted = np.argsort(-h)
        h = np.zeros(n)
        h[idx_sorted[:n//2]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    n_lags = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        eff = compute_eff_off(h, n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_pair = None
        best_new_corr = None
        
        for i in ones:
            # For all j in zeros: new_corr = corr + eff[i] - eff[j]
            diffs = eff[i][None, :] - eff[zeros]  # shape (n_z, n_lags)
            new_corrs = corr[None, :] + diffs
            new_maxes = np.max(new_corrs, axis=1)
            idx_min = np.argmin(new_maxes)

        h[i] = 0
        h[j] = 1
        corr = best_new_corr

    return h, best_c5

def coord_descent_continuous(h, n, t_end, max_sweeps=50):
    """Coordinate descent for continuous h."""
    h = h.copy()
    n_lags = 2 * n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    k_offset = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return h, best_c5
            # Compute corr as function of h[i] = t
            # corr_k(t) = corr_k(base with h[i]=0) + t * coef_k
            h_save = h[i]
            h[i] = 0.0
            corr_base = np.correlate(h, 1.0 - h, mode='full')
            
            # Coefficient for t
            coef = np.zeros(n_lags)
            for k_idx in range(n_lags):
                lag = k_idx - k_offset
                c = 0.0
                j1 = i + lag
                j2 = i - lag
                if 0 <= j1 < n:
                    c += (1.0 - h[j1])
                if 0 <= j2 < n:
                    c += (-h[j2])
                coef[k_idx] = c
            
            # Minimize max_k (corr_base[k] + t * coef[k]) over t in [0,1]
            def f(t):
                return np.max(corr_base + t * coef)
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-14})
            new_val = res.x
            h[i] = new_val
            
            # Recompute corr

            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                improved = True
            else:
                h[i] = h_save  # revert
                corr = np.correlate(h, 1.0 - h, mode='full')
        
        if not improved:
            break
    
    return h, best_c5

def solve_lp_erdos(n, timeout=120):
    """Solve LP relaxation for the Erdős problem."""
    import cvxpy as cp
    t_start = time.time()
    
    h = cp.Variable(n)
    C = cp.Variable()
    cons = [h >= 0, h <= 1, cp.sum(h) == n/2, C >= 0]
    
    n_lags = 2 * n - 1
    w_vars = []
    for l_idx in range(n_lags):
        actual_lag = l_idx - (n - 1)
        m = n - abs(actual_lag)
        w = cp.Variable(m)
        w_vars.append(w)
        if actual_lag >= 0:
            js = np.arange(0, m)
        else:
            js = np.arange(-actual_lag, n)
        cons.append(w >= 0)
        cons.append(w <= h[js])
        cons.append(w <= h[js + actual_lag])
        cons.append(w >= h[js] + h[js + actual_lag] - 1)
    
    for l_idx in range(n_lags):
        actual_lag = l_idx - (n - 1)
        m = n - abs(actual_lag)
        if actual_lag >= 0:
            js = np.arange(0, m)
        else:
            js = np.arange(-actual_lag, n)
        S_l = cp.sum(h[js])
        W_l = cp.sum(w_vars[l_idx])
        cons.append(S_l - W_l <= C)
    
    prob = cp.Problem(cp.Minimize(C), cons)
    
    # Try solvers in order
    for solver in [cp.GLOP, cp.CBC, cp.CPLEX]:
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return h.value, C.value
        except Exception:
            continue
    
    return None, None
```

One more thought: for the coordinate descent on continuous h, I'm doing O(n²) work per coordinate (computing corr_base and coef). But I can be smarter:

When changing h[i] from h[i] to t:
- corr_k_new = corr_k_old + (t - h[i_old]) * [(1-h[i+k]) - h[i-k]] (for valid indices)

This is O(1) per k (just a few floating point ops), so O(n_lags) = O(2n-1) per coordinate. Total per sweep: O(n * 2n) = O(n²). For n=180: 32400 ops per sweep. Very fast!

Let me use this incremental approach instead of recomputing the full correlation:

```python
def coord_step(h, i, n, corr):
    """Compute optimal h[i] given current corr. Returns new h[i] and update to corr."""
    n_lags = 2 * n - 1
    k_offset = n - 1
    old_val = h[i]
    
    # Correlation as linear function of t = h[i]:
    # corr_k(t) = corr_k(0) + t * coef_k
    # where corr_k(0) = corr_k - old_val * coef_k (since current corr has h[i]=old_val)
    
    coef = np.zeros(n_lags)
    for k_idx in range(n_lags):
        lag = k_idx - k_offset
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k_idx] = c
    
    corr_at_0 = corr - old_val * coef
    
    # Minimize max_k (corr_at_0[k] + t * coef[k]) for t in [0,1]
    def f(t):
        return np.max(corr_at_0 + t * coef)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    new_val = res.x
    delta = new_val - old_val
    corr += delta * coef
    return new_val, corr
```

This is O(n_lags) per coordinate = O(2n-1) per coordinate. Total per sweep: O(n * 2n) = O(2n²). For n=180: ~65k ops per sweep. Extremely fast.

But wait, the coef computation inside the loop is a Python for-loop over n_lags values. That's 359 iterations with a few ops each. In Python, that's maybe 10-20 μs per coefficient, so ~5ms per coordinate, ~1ms per sweep. Totally fine.

```python
offsets = np.arange(n_lags) - (n-1)  # precomputed

vp = (ip >= 0) & (ip < n)
vm = (im >= 0) & (im < n)
coef = np.zeros(n_lags)
coef[vp] = (1.0 - h[ip[vp]])
coef[vm] += (-h[im[vm]])
```
