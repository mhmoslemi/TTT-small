import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def cndescent(h, n, t_end, max_sweeps=5000):
    dx = 2.0 / n
    best_h = h.copy().astype(float)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    best_c5 = float(np.max(corr) * dx)
    c0 = n - 1
    shifts = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        for i in range(n):
            old = best_h[i]
            j1 = i + shifts
            j2 = i - shifts
            m1 = (j1 >= 0) & (j1 < n)
            m2 = (j2 >= 0) & (j2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[m1] += 1.0 - best_h[j1[m1]]
            coeff[m2] -= best_h[j2[m2]]
            cands = [0.0, 1.0, 0.5]
            top2 = np.argpartition(corr, -2)[-2:][::-1]
            a, b = int(top2[0]), int(top2[1])
            dc = coeff[a] - coeff[b]
            if abs(dc) > 1e-12:
                v_eq = old + (corr[b] - corr[a]) / dc
                if 0.0 < v_eq < 1.0:
                    cands.append(v_eq)
                v_eq2 = old + 1.0 / dc if dc > 0 else old - 1.0 / dc
                if 0.0 < v_eq2 < 1.0:
                    cands.append(v_eq2)
            best_v = old
            best_f = float(np.max(corr))
            for v in cands:
                if abs(v - old) < 1e-10:
                    continue
                dv = v - old
                cn = corr + dv * coeff
                cn[c0] = corr[c0] + dv * (1.0 - v - old)
                fv = float(np.max(cn))
                if fv < best_f - 1e-14:
                    best_f = fv
                    best_v = v
            if best_v != old:
                dv = best_v - old
                corr += dv * coeff
                corr[c0] -= dv * dv
                best_h[i] = best_v
                best_c5 = best_f * dx
                improved += 1
        if improved == 0:
            break
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def perturb(h, n, rng, scale):
    h_p = h.copy()
    op = rng.randint(6)
    if op == 0:
        h_p += scale * rng.randn(n)
    elif op == 1:
        blen = rng.randint(6, max(7, n // 5))
        bs = rng.randint(0, n - blen)
        h_p[bs:bs + blen] = 1.0 - h_p[bs:bs + blen]
    elif op == 2:
        blen = rng.randint(4, max(5, n // 4))
        bs = rng.randint(0, n - blen)
        h_p[bs:bs + blen] += scale * 2.0 * rng.randn(blen)
    elif op == 3:
        idx = rng.randint(n)
        h_p[idx] = rng.uniform(0.0, 1.0)
    elif op == 4:
        freq = rng.randint(1, max(2, n // 3))
        phase = rng.uniform(0, 2 * np.pi)
        h_p += scale * 0.6 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    else:
        idx1 = rng.randint(n)
        h_p[idx1] = 1.0 - h_p[idx1]
        idx2 = rng.randint(n)
        h_p[idx2] = 1.0 - h_p[idx2]
    return project(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 160
        best_h = np.ones(n_init) * 0.5
        for i in range(0, n_init, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Initial coordinate descent
    t1 = min(t0 + budget_s * 0.18, t0 + 180)
    best_h, best_c5 = cndescent(best_h, n, t1, max_sweeps=10000)

    # Phase 2: Annealed perturb-and-refine
    t2_end = t0 + budget_s * 0.68
    work_h = best_h.copy()
    work_c5 = best_c5
    no_improve = 0
    max_no_improve = 60
    for it in range(20000):
        if time.time() > t2_end:
            break
        progress = (time.time() - t0) / (t2_end - t0)
        T_anneal = 0.003 * (1.0 - progress) + 0.0002
        scale = 0.12 * (1.0 - progress) + 0.02

        h_p = perturb(work_h, n, rng, scale)
        t_sweep_min = min(200, max(80, int(300 * (1.0 - 0.5 * progress))))
        t_sweep_max = min(500, t_sweep_min + 200)
        t_alloc = min(4.0, max(1.0, (t2_end - time.time()) / 200.0))
        h_r, c5_r = cndescent(h_p, n, time.time() + t_alloc, max_sweeps=t_sweep_max)

        delta = c5_r - work_c5
        accept = c5_r < work_c5 or (delta < 0.005 and rng.rand() < np.exp(-delta / T_anneal))
        if accept:
            work_h, work_c5 = h_r, c5_r
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # Restart from best with larger perturbation
            work_h = perturb(best_h, n, rng, 0.2 + 0.3 * rng.rand())
            work_c5 = compute_c5(work_h)
            h_r, c5_r = cndescent(work_h, n, time.time() + 2.0, max_sweeps=300)
            work_h, work_c5 = h_r, c5_r
            no_improve = 0

    t3_end = t0 + budget_s * 0.85
    for n_try in [128, 200, 240, 320, 400]:
        if time.time() > t3_end or n_try == n:
            continue
        h_tr = interp_h(best_h, n_try)
        t_alloc = min(8.0, max(2.0, (t3_end - time.time()) / 3.0))
        h_tr, c5_tr = cndescent(h_tr, n_try, time.time() + t_alloc, max_sweeps=4000)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 4: Final intensive polishing
    t4_end = t0 + budget_s * 0.97
    best_h, best_c5 = cndescent(best_h, n, t4_end, max_sweeps=20000)

        for _ in range(20):
            if time.time() > t0 + budget_s * 0.995:
                break

            h_try, c5_try = cndescent(h_try, n, time.time() + 1.0, max_sweeps=200)
            if c5_try < best_c5:
                best_c5, best_h = c5_try, h_try

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
