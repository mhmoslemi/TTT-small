import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_lse(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def slsqp_polish(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max_lse(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    constr = {'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=constr, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def coord_descent(h, n, t_end, max_sweeps=5000, xatol=1e-10):
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        any_improve = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            lo = n - 1 - i
            hi = 2 * n - 2 - i
            coef = 1.0 - 2.0 * best_h
            # Outside window max
            if lo > 0 and hi < 2*n-2:
                max_out = max(np.max(corr[:lo]), np.max(corr[hi+1:]))
            elif lo > 0:
                max_out = np.max(corr[:lo])
            elif hi < 2*n-2:
                max_out = np.max(corr[hi+1:])
            else:
                max_out = -1e30
            
            f_lo = -h_old
            f_hi = 1.0 - h_old
            if f_hi - f_lo < 1e-14:
                continue
            
            def f1d(s):
                return max(max_out, np.max(corr[lo:hi+1] + s * coef))
            
            res = minimize_scalar(f1d, bounds=(f_lo, f_hi), method='bounded',
                                  options={'xatol': xatol})
            s_new = res.x
            if abs(s_new) > 1e-10:
                best_h[i] += s_new
                corr[lo:hi+1] += s_new * coef
                c5_new = max(max_out, np.max(corr)) * dx
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    any_improve = True
        if not any_improve:
            break
    return best_h, best_c5

def frank_wolfe_step(h, corr, T, n):
    """One FW step using smooth max subgradient."""
    weights = np.exp(T * (corr - np.max(corr)))
    weights /= weights.sum()
    # Subgradient of smooth_max_lse(corr, T) w.r.t. h
    # d(corr[m])/d(h[i]) = (1-h[neighbor]) for h[i] as first factor
    #                      + h[where i is the second factor index] * (-1)
    # = for m in [n-1-i, 2n-2-i]: 1 - 2*h[i+m-n+1]
    # But weighted by weights[m]
    
    grad = np.zeros(n)
    for i in range(n):
        lo = n - 1 - i
        hi = 2 * n - 2 - i
        w = weights[lo:hi+1]
        grad[i] = np.sum(w * (1.0 - 2.0 * h))
    
    # FW: find s in polytope minimizing grad·s
    # Optimal: s[j]=1 for n/2 smallest grad[j], s[j]=0 otherwise
    idx = np.argsort(grad)
    s = np.zeros(n)
    s[idx[:n//2]] = 1.0
    
    # Line search
    def f_line(t):
        h_t = (1-t)*h + t*s
        h_t = np.clip(h_t, 0, 1)
        c = np.correlate(h_t, 1-h_t, mode='full')
        return smooth_max_lse(c, T)
    
    res = minimize_scalar(f_line, bounds=(0, 1), method='bounded', options={'xatol': 1e-4})
    t_opt = res.x
    h_new = (1-t_opt)*h + t_opt*s
    return np.clip(h_new, 0, 1)

    dx = 2.0/n

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

_cd_state = {'corr': None, 'coef': None, 'maxout': 0.0}

def _cd_f1d(s):
    return max(_cd_state['maxout'], np.max(_cd_state['corr'] + s * _cd_state['coef']))

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def proj_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) < 1e-15:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-15:
            return h
        h = h + r * (d / t)
    else:
        s = h.sum()
        if s < 1e-15:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def cd_optimize(h, t_end, max_sweeps, xatol):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    dx = 2.0 / n
    c5 = corr.max() * dx
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                break
            lo = n - 1 - i
            hi = 2 * n - 2 - i
            coef = 1.0 - 2.0 * h
            if lo > 0:
                if hi < 2*n - 2:
                    mo = max(np.max(corr[:lo]), np.max(corr[hi+1:]))
                else:
                    mo = np.max(corr[:lo])
            else:
                if hi < 2*n - 2:
                    mo = np.max(corr[hi+1:])
                else:
                    mo = -1e30
            a_lo = -h[i]
            a_hi = 1.0 - h[i]
            if a_hi - a_lo < 1e-14:
                continue
            _cd_state['corr'] = corr[lo:hi+1]
            _cd_state['coef'] = coef
            _cd_state['maxout'] = mo
            s = minimize_scalar(_cd_f1d, bounds=(a_lo, a_hi), method='bounded',
                                options={'xatol': xatol}).x
            if abs(s) > 1e-12:
                h[i] += s
                corr[lo:hi+1] += s * coef
                nv = max(mo, corr.max()) * dx
                if nv < c5 - 1e-14:
                    c5, imp = nv, True
        if not imp:
            break
    corr = np.correlate(h, 1.0 - h, mode='full')
    return h, float(np.max(corr)) * dx

def sgrad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    w = np.exp(T * (corr - mx))
    w /= w.sum()
    return dx * np.correlate(w, 1.0 - 2.0*h, mode='full')[n-1:2*n-1]

def sobj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    return (mx + np.log(np.sum(np.exp(T*(corr - mx)))) / T) * dx

def pgd_optimize(h, T, lr_init, max_steps, t_end):
    n = len(h)
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for _ in range(max_steps):
        if time.time() > t_end:
            break
        g = sgrad(h, T)
        h_new = proj_sum(h - lr * g, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best = h_new.copy()
            best_c5 = c5_new
        if c5_new <= sobj(h, T) + 1e-12:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best, best_c5

def slsqp_opt(h, T, maxiter, ftol):
    n = len(h)
    dx = 2.0 / n
    res = minimize(sobj, h, jac=function_sgrad_default, method='SLSQP',
                   bounds=[(0.0, 1.0)]*n,
                   constraints=[{'type': 'eq', 'fun': function_equity,
                                 'jac': function_ones}],
                   args=(T,), options={'maxiter': maxiter, 'ftol': ftol})
    return proj_sum(np.clip(res.x, 0, 1), n/2)

_slq_T = 100
_slq_n = 200

def function_sgrad_default(x, T=None):
    if T is None:
        T = _slq_T
    return sgrad(np.clip(x, 0, 1), T)

def function_equity(x):
    return x.sum() - _slq_n / 2.0

def function_ones(x):
    return np.ones(len(x))

def perturb_h(h, rng, mode):
    n = len(h)
    if mode == 0:
        return h + 0.04 * rng.standard_normal(n)
    elif mode == 1:
        sh = rng.randint(-8, 9)
        return np.roll(h, sh)
    elif mode == 2:
        m = rng.integers(5, max(6, n//4))
        st = rng.integers(0, n - m)
        h2 = h.copy()
        blk = h[st:st+m].copy()
        h2[st:st+m] = 0.5
        dst = (st + rng.randint(1, n-1)) % n
        h2[dst:dst+m] = blk
        return h2
    else:
        mask = rng.random(n) < 0.15
        h2 = h.copy()
        h2[mask] += 0.06 * rng.standard_normal(mask.sum())
        return h2
