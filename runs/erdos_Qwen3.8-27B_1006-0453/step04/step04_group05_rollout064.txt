def smooth_gradient(h, N, top_k=30, temp=0.05):
    corr = np.correlate(h, 1.0-h, mode='full')
    max_c = corr.max()
    # Softmax weights over all lags (or top_k)
    # Use top_k for efficiency
    idx = np.argpartition(-corr, top_k)[:top_k]
    c_top = corr[idx]
    w = np.exp((c_top - c_top.max()) / (temp * max(1e-6, c_top.max())))
    w /= w.sum()
    
    grad = np.zeros(N)
    center = N - 1
    for j, k in enumerate(idx):
        L = k - center
        wj = w[j]
        if L == 0:
            grad += wj * (1.0 - 2.0*h)
        elif L > 0:
            # (1-h[p-L]) for p>=L: indices L..N-1
            grad[L:] += wj * (1.0 - h[:N-L])
            # -h[p+L] for p<=N-1-L: indices 0..N-1-L
            grad[:N-L] -= wj * h[L:]
        else:
            M = -L
            # (1-h[p+M]) for p<=N-1-M: indices 0..N-1-M
            grad[:N-M] += wj * (1.0 - h[M:])
            # -h[p-M] for p>=M: indices M..N-1
            grad[M:] -= wj * h[:N-M]
    return grad

import numpy as np

def compute_c5(h, N):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, N, top_k=30, temp=0.03):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(-corr, min(top_k, 2*N-1))[:top_k]
    c_top = corr[idx]
    scale = temp * max(1e-8, abs(c_top.max()))
    w = np.exp((c_top - c_top.max()) / scale)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for j, k in enumerate(idx):
        L = k - center
        wj = w[j]
        if L == 0:
            grad += wj * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += wj * (1.0 - h[:N-L])
            grad[:N-L] -= wj * h[L:]
        else:
            M = -L
            grad[:N-M] += wj * (1.0 - h[M:])
            grad[M:] -= wj * h[:N-M]
    return grad

def adam_steps(h, N, target, n_steps, top_k=30, temp=0.03, lr0=0.03, decay=0.995):
    best_h = h.copy()
    best_peak = float(np.correlate(h, 1.0-h, mode='full').max())
    lr = lr0
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N); v = np.zeros(N)
    for it in range(1, n_steps+1):
        grad = smooth_grad(h, N, top_k, temp)
        peak = float(np.correlate(h, 1.0-h, mode='full').max())
        if peak < best_peak:
            best_peak = peak; best_h = h.copy()
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        mh = m/(1-b1**it); vh = v/(1-b2**it)
        h = h - lr*mh/(np.sqrt(vh)+eps)
        h = project_sum(h, target)
        if it % 80 == 0:
            lr *= decay
    return best_h, best_peak*(2.0/N)

    v2 = np.where((pp>=0)&(pp<N), h[np.clip(pp,0,N-1)], 0.0)
    return v1 - v2
