    """Minimize max_m (a[m] + b[m]*s) over s in [lo, hi]. a, b are arrays."""
    # Check boundaries
    f_lo = np.max(a + lo * b)
    f_hi = np.max(a + hi * b)

    # At s=lo: if all active (a[j]+lo*b[j] == f_lo) have b[j] >= 0, then f is increasing → min at lo
    active_lo = np.abs(a + lo*b - f_lo) < 1e-12

    if np.all(b[active_hi] <= 1e-12):
        return hi

    for _ in range(40):

        f_mid = np.max(a + mid * b)

            lo = mid  # f is decreasing, move right
        else:
            break  # 0 is in subgradient

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def min_1d_max_linear(a, b, lo, hi):
    f_lo = np.max(a + lo * b)
    f_hi = np.max(a + hi * b)
    active = np.abs(a + lo * b - f_lo) < 1e-12
    if np.all(b[active] >= -1e-13):
        return lo
    active = np.abs(a + hi * b - f_hi) < 1e-12
    if np.all(b[active] <= 1e-13):
        return hi
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        fm = np.max(a + mid * b)
        act = np.abs(a + mid * b - fm) < 1e-11
        if b[act].min() > 1e-14:
            hi = mid
        elif b[act].max() < -1e-14:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, time_limit, n_sweeps=500):
    best_h = h.astype(float).copy()
    dx = 2.0 / n
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    t_vals = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        if time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() > time_limit:
                return best_h, best_c5
            h_old = best_h[i]
            first = i + t_vals
            second = i - t_vals
            vf = (first >= 0) & (first < n)
            vs = (second >= 0) & (second < n)
            coef = np.zeros(2 * n - 1)
            coef[vf] += (1.0 - best_h[first[vf]])
            coef[vs] -= best_h[second[vs]]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:

            s_new = min_1d_max_linear(corr, coef, lo, hi)

                    improved = True
                elif c5_new > best_c5 + 1e-10:

                    corr -= s_new * coef
        if not improved:
            break

def slsqp_smooth(h, T, n, maxiter=800, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_sum(np.clip(res.x, 0, 1), n / 2.0)
    except Exception:
        return h.copy()

def fourier_start(n, K=5):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    amps = 0.25 / np.sqrt(np.arange(1, K + 1))
    for k in range(1, K + 1):
        h += amps[k-1] * np.cos(k * np.pi * x + 0.3 * k)
    h = np.clip(h, 0, 1)
    return project_sum(h, n / 2.0)

        hc = slsqp_smooth(best_h, T, n, maxiter=600, ftol=1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Phase 3: CD again (15% budget)
    t3 = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, n, t3, n_sweeps=800)

    # Phase 4: Random restarts + CD (20% budget)
    t4 = t0 + budget_s * 0.70
    for restart in range(30):
        if elapsed() > t4:
            break
        cool = max(1.0 - restart / 30.0, 0.1)
        if restart % 3 == 0:
            h_p = best_h + 0.05 * cool * np.random.randn(n)
        elif restart % 3 == 1:
            shift = int(np.random.uniform(-0.03, 0.03) * n)
            h_p = 0.6 * np.roll(best_h, shift) + 0.4 * best_h
        else:
            mask = np.random.rand(n) < 0.2
            h_p = best_h.copy()
            h_p[mask] = np.clip(h_p[mask] + 0.1 * cool * np.random.randn(mask.sum()), 0, 1)
        h_p = project_sum(h_p, n / 2)
        hc, c5v = coord_descent(h_p, n, time.time() + 3.0, n_sweeps=50)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Phase 5: Multi-n exploration (15% budget)
    t5 = t0 + budget_s * 0.85
    for n_try in [128, 192, 256]:
        if elapsed() > t5:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_sum(h_i, n_try / 2)
        hc, c5v = coord_descent(h_i, n_try, time.time() + 5.0, n_sweeps=100)
        hc = slsqp_smooth(hc, 5000, n_try, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        hc, c5v = coord_descent(hc, n_try, time.time() + 3.0, n_sweeps=100)
        if c5v < best_c5:
            best_h, best_c5, n = hc, c5v, n_try

    # Phase 6: Final heavy CD + SLSQP (10% budget)
    t6 = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, n, t6, n_sweeps=500)
    if elapsed() < budget_s * 0.98:
        for T in [10000, 50000]:
            hc = slsqp_smooth(best_h, T, n, maxiter=1000, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v

    # Final tiny CD
    if elapsed() < budget_s * 0.995:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, n_sweeps=100)

    best_h = project_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
