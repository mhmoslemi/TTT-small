from scipy.optimize import minimize, minimize_scalar, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.asarray(h, dtype=float)
    def do_clip(tau):
        return np.clip(h - tau, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if do_clip(mid).sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = do_clip((lo + hi) * 0.5)
    diff = target - h_p.sum()
    if abs(diff) > 1e-13:
        free = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if free.sum() > 0:
            h_p = h_p.copy()
            h_p[free] += diff / free.sum()
            np.clip(h_p, 0.0, 1.0, out=h_p)
    return h_p

def fourier_to_h(params, n):
    """Convert Fourier params [a1,b1,...,aK,bK] to h on grid of size n."""
    K = len(params) // 2
    x = 2.0 * np.arange(n) / n  # x_i in [0, 2)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        h += params[2*(k-1)] * np.cos(np.pi * k * x)
        h += params[2*(k-1)+1] * np.sin(np.pi * k * x)
    return h

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def slsqp_polish(h, T, n, maxiter=1000, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

    best_h = h.copy().astype(float)
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        # Random order for diversity
        idx = np.random.permutation(n)
        for i in idx:
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            # Compute effect of changing h[i] by s on each corr[k]
            # corr[k] changes by: s*(1-h[i-k]) - s*h[i+k] (with bounds)
            coef = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                c = 0.0
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    c += (1.0 - best_h[j1])
                if 0 <= j2 < n:
                    c -= best_h[j2]
                coef[k + n - 1] = c
            # Minimize max(corr + s * coef) over s in [-h_old, 1-h_old]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:

            def f1d(s):
                return np.max(corr + s * coef)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',

            if abs(s_new) > 1e-12:

                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def coord_descent(h, n, t_end, n_sweeps=200):
    best_h = np.asarray(h, dtype=float).copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    k_idx = np.arange(-(n-1), n + 1)  # length 2n-1
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        idx = np.random.permutation(n)
        for i in idx:
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            j1 = i - k_idx  # where (1-h) contributes
            j2 = i + k_idx  # where -h contributes
            mask1 = (j1 >= 0) & (j1 < n)
            mask2 = (j2 >= 0) & (j2 < n)
            c1 = np.where(mask1, 1.0 - best_h[np.clip(j1, 0, n-1)], 0.0)
            c2 = np.where(mask2, best_h[np.clip(j2, 0, n-1)], 0.0)
            coef = c1 - c2
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            # Minimize max(corr + s*coef) over s
            # This is equivalent to finding s that minimizes the max of 2n-1 linear functions
            # Use golden section
            def f1d(s):
                return np.max(corr + s * coef)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10})
            s_new = res.x
            new_val = h_old + s_new
            new_val = min(max(new_val, 0.0), 1.0)
            actual_s = new_val - h_old
            if abs(actual_s) > 1e-13:
                best_h[i] = new_val
                corr += actual_s * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h_p.sum()
    if abs(diff) > 1e-13:
        free = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if free.sum() > 0:
            h_p[free] += diff / free.sum()
            np.clip(h_p, 0, 1, out=h_p)
    return h_p

def fourier_to_h(params, n):
    K = len(params) // 2
    x = 2.0 * np.arange(n) / n
    h = np.full(n, 0.5)
    for k in range(K):
        h += params[2*k] * np.cos(np.pi * (k+1) * x)
        h += params[2*k+1] * np.sin(np.pi * (k+1) * x)
    return h

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def slsqp_polish(h, T, n, maxiter=500, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    def grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        w = T * np.exp(T * (corr - np.max(corr)))
        w /= np.sum(w)
        g = np.zeros(n)
        # grad of corr[k] w.r.t. h[j]
        # ... this is complex, let SLSQP do finite difference
        return None
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def coord_descent(h, n, t_end, n_sweeps=200):
    best_h = np.asarray(h, dtype=float).copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    k_idx = np.arange(-(n-1), (n-1) + 1, dtype=int)  # length 2n-1
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        idx = np.random.permutation(n)
        for i in idx:
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            j1 = i - k_idx
            j2 = i + k_idx
            mask1 = (j1 >= 0) & (j1 < n)
            mask2 = (j2 >= 0) & (j2 < n)
            c1 = np.zeros(2*n - 1)
            c1[mask1] = 1.0 - best_h[j1[mask1]]
            c2 = np.zeros(2*n - 1)
            c2[mask2] = best_h[j2[mask2]]
            coef = c1 - c2
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            # Minimize max(corr + s*coef) for s in [lo, hi]
            # Quick check: is the current point (s=0) optimal?
            # At s=0, the active constraint is corr at argmax
            # Check if moving in either direction increases the max
            val0 = np.max(corr)
            eps = 1e-8
            vals = np.array([lo, hi, 0.0, lo*0.5, hi*0.5])
            vals = vals[(vals >= lo - 1e-12) & (vals <= hi + 1e-12)]
            obj_vals = np.array([np.max(corr + v * coef) for v in vals])
            if obj_vals.min() >= val0 - 1e-12:
                continue  # no improvement possible
            res = minimize_scalar(lambda s: np.max(corr + s * coef),
                                  bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10})
            s_new = res.x

- s = hi, or
- A point where two of the linear functions cross and both are at the maximum

The idea:

2. Update: h ← Project(h - α * g + β * momentum)
3. Use diminishing step size α_t = α_0 / (1 + t/τ)

∂L_k/∂h[j] = (1-h[j-k])_valid - h[j+k]_valid

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_slp = {}
_cd = {}

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def project_to_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h_p.sum()
    if abs(diff) > 1e-13:
        free = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if free.sum() > 0:
            h_p[free] += diff / free.sum()
            np.clip(h_p, 0, 1, out=h_p)
    return h_p

def fourier_to_h(params, n):
    K = len(params) // 2
    x = 2.0 * np.arange(n) / n
    h = np.full(n, 0.5)
    for k in range(K):
        h += params[2 * k] * np.cos(np.pi * (k + 1) * x)
        h += params[2 * k + 1] * np.sin(np.pi * (k + 1) * x)
    return h

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def _slsqp_obj(x):
    n, T = _slp['n'], _slp['T']
    xc = np.clip(x, 0, 1)
    return smooth_max(np.correlate(xc, 1.0 - xc, mode='full'), T) * (2.0 / n)

def _slsqp_con(x):
    return x.sum() - _slp['n'] / 2.0

def slsqp_polish(h, T, n, maxiter=500, ftol=1e-14):
    _slp['n'], _slp['T'] = n, T
    res = minimize(_slsqp_obj, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def _cd_obj(s):
    return np.max(_cd['corr'] + s * _cd['coef'])

def coord_descent(h, n, t_end, n_sweeps=200):
    best_h = np.asarray(h, dtype=float).copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    k_idx = np.arange(-(n - 1), n, dtype=np.int64)
    length = 2 * n - 1
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        idx = np.random.permutation(n)
        for i in idx:
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            j1 = i - k_idx
            j2 = i + k_idx
            m1 = (j1 >= 0) & (j1 < n)
            m2 = (j2 >= 0) & (j2 < n)
            c1 = np.zeros(length)
            c1[m1] = 1.0 - best_h[j1[m1]]
            c2 = np.zeros(length)
            c2[m2] = best_h[j2[m2]]
            coef = c1 - c2
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            _cd['corr'] = corr
            _cd['coef'] = coef
            res = minimize_scalar(_cd_obj, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10})
            new_val = min(max(h_old + res.x, 0.0), 1.0)
            actual_s = new_val - h_old
            if abs(actual_s) > 1e-13:
                best_h[i] = new_val
                corr += actual_s * coef
                c5_new = float(np.max(corr)) * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def _fourier_obj(p):
    h_t = fourier_to_h(p, _slp['n'])
    return compute_c5(project_to_sum(h_t, _slp['n'] / 2.0))

        except Exception:
            pass

    # Phase 2: Fourier global exploration (35%)
    t_p2 = t0 + budget_s * 0.40
    _slp['n'] = n
    for K in [3, 4, 5, 6, 8, 10]:
        if elapsed() > t_p2:
            break
        n_params = 2 * K
        n_trials = max(3, int(8 * (t_p2 - time.time()) / (K * 3)))
        n_trials = min(n_trials, 10)
        for trial in range(n_trials):
            if time.time() > t_p2:
                break
            if trial == 0:
                p0 = np.zeros(n_params)
                for k in range(K):
                    p0[2 * k] = 0.25 * np.cos(2 * np.pi * k / max(K, 1) * 0.6)
                    p0[2 * k + 1] = 0.25 * np.sin(2 * np.pi * k / max(K, 1) * 0.6)
            elif trial == 1:
                p0 = np.random.uniform(-0.4, 0.4, n_params)
                p0[0], p0[1] = 0.3, 0.1
            else:

            bounds_f = [(-1.0, 1.0)] * n_params
            try:
                res_f = minimize(_fourier_obj, p0, method='Nelder-Mead',
                                 bounds=bounds_f,
                                 options={'maxiter': 150, 'xatol': 0.005, 'fatol': 5e-6, 'adaptive': True})
                h_cand = fourier_to_h(res_f.x, n)

            except Exception:

    for it in range(n_bh):
        if time.time() > t_p3:
            break

        r = np.random.rand()
        if r < 0.35:
            step = 0.015 + 0.12 * cool
            h_p = work_h + step * np.random.randn(n)
        elif r < 0.65:

        else:

            mask = np.random.rand(n) < 0.35
            h_p = work_h.copy()

        h_p = project_to_sum(h_p, n / 2.0)
        try:
            T_bh = int(150 + 1500 * cool)
            hc = slsqp_polish(h_p, T_bh, n, 200, 1e-12)
            hc, c5v = coord_descent(hc, n, time.time() + 0.8, 10)
            temp = 0.0025 * cool + 0.0001
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h, work_c5 = hc, c5v
                if c5v < best_c5:

        except Exception:

    if work_c5 < best_c5:

    x_old = np.linspace(0, 2, n, endpoint=False)
