```python
    """
    h: array of length N
    p: array of length 2N-1 (probability weights for each lag)
    Returns gradient of smooth max overlap w.r.t. h
    """
    grad = np.zeros(N)
    # For each j, grad[j] = sum_L p[L+N-1] * (1-h[j-L]) [if valid] - sum_L p[L+N-1] * h[j+L] [if valid]

        for L in range(L_min2, L_max2+1):
```

So conv1[j+N-1] = sum_k (1-h)[k] * p[j+N-1-k]

And I want sum_m p[m] * (1-h)[j-m+N-1]
Let k = j-m+N-1: m = j+N-1-k
= sum_k (1-h)[k] * p[j+N-1-k]

Yes! They match. So term1[j] = conv1[j+N-1] where conv1 = np.convolve(p, 1-h, 'full').

Great, so the gradient computation can be done with two convolutions/correlations, which is O(N log N) with FFT or O(N^2) with direct. For N=380, direct is fine.

```python
def lse_adam(h, N, target, deadline, rng, beta_start=2.0, beta_end=50.0, lr0=0.03, k_iters=...):
    h = h.copy()
    best_h = h.copy()
    best_peak = compute_corr_peak(h)
    
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    
    while time.time() < deadline:
        # Compute correlations
        c = np.correlate(h, 1.0-h, 'full')
        peak = float(c.max())
        
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        # Compute softmax weights
        beta = beta_start * (beta_end/beta_start) ** (it / max(1, total_iters))
        c_shift = c - c.max()  # for numerical stability
        p = np.exp(beta * c_shift)
        p /= p.sum()
        
        # Compute gradient using convolutions
        grad = compute_grad(h, p, N)
        
        # Adam update
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        
        # Periodic learning rate decay
        if it % 200 == 0:
            lr *= 0.95
    
    return best_h, best_peak * (2.0/N)
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)*0.5, 0.0, 1.0)

def compute_grad(h, p, N):
    """Compute gradient of log-sum-exp of correlations w.r.t. h"""
    # p has length 2N-1
    # term1[j] = sum_k (1-h)[k] * p[j+N-1-k] = conv(p, 1-h)[j+N-1]
    conv1 = np.convolve(p, 1.0 - h, 'full')
    term1 = conv1[N-1:2*N-1]
    # term2[j] = sum_k h[k] * p[k+N-1-j] = correlate(h, p)[N-1+j]
    corr2 = np.correlate(h, p, 'full')
    term2 = corr2[N-1:2*N-1]
    return term1 - term2

def lse_adam(h, N, target, deadline, rng, beta0=3.0, beta1=80.0, lr0=0.025):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    lr = lr0
    while time.time() < deadline:
        c = compute_corr(h)
        peak = float(c.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        # Beta schedule
        frac = min(1.0, it / 2000.0)
        beta = beta0 + (beta1 - beta0) * frac
        # Softmax weights
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        p = exp_c / exp_c.sum()
        # Gradient
        grad = compute_grad(h, p, N)
        # Adam
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 100 == 0:
            lr *= 0.97
    return best_h, best_peak * dx

def adjust_ones(h, N):
    nh = N // 2
    n_ones = int(h.sum())
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh-n_ones]] = 1
    return h

def binary_steepest(h, N, deadline):
    h = h.astype(float).copy()
    adjust_ones(h, N)
    c = compute_corr(h)
    P = float(c.max())
    # Precompute D
    idx = np.arange(2*N-1)
    d_lags = idx - (N-1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N-1).astype(int)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N-1).astype(int)], 0.0)
    D = -(1.0-h_pd)*valid_pd.astype(float) + h_pmd*valid_pmd.astype(float)
    
    for _ in range(1000):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_peak = P
        best_pq = None
        for p in ones:
            base = c - D[p, :]
            cands = base[None, :] + D[zeros, :]
            new_peaks = cands.max(axis=1)
            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_peak - 1e-12:
                best_peak = float(new_peaks[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            break
        p, q = best_pq
        h[p], h[q] = h[q], h[p]
        # Update c and D
        c = compute_corr(h)
        P = float(c.max())
        # Update D rows for p and q
        h_pd_new = np.where(valid_pd, h[np.clip(pd, 0, N-1).astype(int)], 0.0)
        h_pmd_new = np.where(valid_pmd, h[np.clip(p_md, 0, N-1).astype(int)], 0.0)
        D = -(1.0-h_pd_new)*valid_pd.astype(float) + h_pmd_new*valid_pmd.astype(float)
    return h

    c5 = compute_c5(h_bin, N)

    # Phase 2: Continuous LSE-Adam (main workhorse)

    h_cont = project_sum(h_cont, target)
    h_cont, c5 = lse_adam(h_cont, N, target, d2, rng, beta0=2.0, beta1=100.0, lr0=0.02)
    if c5 < best_c5:

    # Phase 3: Second LSE-Adam with different params

        h2 = best_h + rng.randn(N) * 0.04
        h2 = project_sum(h2, target)
        h2, c5 = lse_adam(h2, N, target, d3, rng, beta0=10.0, beta1=200.0, lr0=0.008)
        if c5 < best_c5:
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, p, N):
    conv1 = np.convolve(p, 1.0 - h, 'full')
    corr2 = np.correlate(h, p, 'full')
    term1 = conv1[N-1:2*N-1]
    term2 = corr2[N-1:2*N-1]
    return term1 - term2

def lse_adam(h, N, target, deadline, rng, beta0=3.0, beta1=120.0, lr0=0.02, decay_every=80, decay=0.96):
    h = h.astype(float).copy()
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    lr = lr0
    while time.time() < deadline:
        c = compute_corr(h)
        peak = float(c.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        frac = min(1.0, it / 3000.0)
        beta = beta0 + (beta1 - beta0) * frac * frac
        c_max = c.max()
        exp_c = np.exp(beta * (c - c_max))
        p = exp_c / exp_c.sum()
        grad = compute_grad(h, p, N)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % decay_every == 0:
            lr *= decay
    return best_h, best_peak * dx

def adjust_ones(h, N):
    nh = N // 2
    n_ones = int(np.round(h.sum()))
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    return h

def build_D(h, N):
    idx = np.arange(2 * N - 1)
    d_lags = idx - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N-1).astype(int)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N-1).astype(int)], 0.0)
    D = -(1.0 - h_pd) * valid_pd.astype(float) + h_pmd * valid_pmd.astype(float)
    return D, valid_pd, valid_pmd, pd, p_md

def binary_steepest(h, N, deadline, max_sweeps=500):
    h = h.astype(float).copy()
    adjust_ones(h, N)
    c = compute_corr(h)
    P = float(c.max())
    D, vp, vpm, pd, pmd = build_D(h, N)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_peak = P
        best_pq = None
        for pi in range(len(ones)):
            p = ones[pi]
            base = c - D[p, :]
            cands = base[None, :] + D[zeros, :]
            new_peaks = cands.max(axis=1)
            mi = int(np.argmin(new_peaks))
            if new_peaks[mi] < best_peak - 1e-12:
                best_peak = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        p, q = best_pq
        h[p], h[q] = h[q], h[p]
        c = compute_corr(h)
        P = float(c.max())
        h_pd = np.where(vp, h[np.clip(pd, 0, N-1).astype(int)], 0.0)
        h_pmd = np.where(vpm, h[np.clip(pmd, 0, N-1).astype(int)], 0.0)
        D = -(1.0 - h_pd) * vp.astype(float) + h_pmd * vpm.astype(float)
    return h

def binary_anneal(h, N, n_iters, rng, T0=3.0, T1=0.004):
    h = (h > 0.5).astype(float)
    adjust_ones(h, N)
    best = h.copy()
    c = compute_corr(h)
    best_pk = float(c.max())
    cur_pk = best_pk
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / n_iters)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        ones[i], zeros[j] = zeros[j], ones[i]
        c = compute_corr(h)
        new_pk = float(c.max())
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.003 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            ones[i], zeros[j] = zeros[j], ones[i]
    return best, best_pk * (2.0 / N)

def make_pattern(N, kind, rng):
    if kind == 'alt':
        h = np.zeros(N)
        h[::2] = 1.0
    elif kind == 'rand':
        h = (rng.random(N) > 0.5).astype(float)
    else:
        h = np.zeros(N)
        h[:N//2] = 1.0
    adjust_ones(h, N)
    return h
```
