  - If d < 0: j-d in [0,N) means j in [0, N-1+d] (since -d > 0, j < N-d = N+|d|, but j < N always)
    Actually j-d in [0,N) means 0 <= j-d < N, so d <= j < N+d. Since d < 0, this is 0 <= j < N+d (since j >= 0 > d is automatic).
    So for j in [0, N-1+d]: value = 1-h[j-d] = 1-h[j+|d|]
    grad[0:N+d] += w[d+N-1] * (1 - h[|d|:N])  where |d| = -d

- Term 2: -h[j+d]*I[0<=j+d<N]
  - If d >= 0: j+d in [0,N) means j in [0, N-1-d]. 
    grad[0:N-d] -= w[d+N-1] * h[d:N]
  - If d < 0: j+d in [0,N) means j in [-d, N-1] (since j > -d = |d|, and j < N-|d| for j+d < N... wait)
    j+d in [0,N): 0 <= j+d < N, so -d <= j < N-d. Since d<0, -d>0, so j in [|d|, N-1].
    grad[|d|:N] -= w[d+N-1] * h[0:N+d]

Let me code this up:

```python
def compute_grad(h, w):
    N = len(h)
    grad = np.zeros(N)
    for d in range(-(N-1), N):
        wd = w[d + N - 1]
        if wd < 1e-15:
            continue
        if d >= 0:
            # Term 1: j in [d, N-1], value = 1-h[j-d]
            if N - d > 0:
                grad[d:] += wd * (1.0 - h[:N-d])
            # Term 2: j in [0, N-1-d], value = -h[j+d]
            if N - d > 0:
                grad[:N-d] -= wd * h[d:]
        else:
            m = -d  # m > 0
            # Term 1: j in [0, N-1-m], value = 1-h[j+m]
            if N - m > 0:
                grad[:N-m] += wd * (1.0 - h[m:])
            # Term 2: j in [m, N-1], value = -h[j-m]
            if N - m > 0:
                grad[m:] -= wd * h[:N-m]
    return grad
```

import numpy as np
import time

def compute_c5(h, N):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def softmax_weights(corr, tau):
    tc = tau * corr
    tc -= tc.max()
    e = np.exp(tc)
    s = e.sum()
    return e / s if s > 0 else np.ones_like(corr) / len(corr)

def compute_grad(h, w):
    N = len(h)
    grad = np.zeros(N)
    for d in range(-(N - 1), N):
        wd = w[d + N - 1]
        if wd < 1e-14:
            continue
        if d >= 0:
            if N - d > 0:
                grad[d:] += wd * (1.0 - h[:N - d])
                grad[:N - d] -= wd * h[d:]
        else:
            m = -d
            if N - m > 0:
                grad[:N - m] += wd * (1.0 - h[m:])
                grad[m:] -= wd * h[:N - m]
    return grad

def adam_softmax(h, N, target, deadline, tau0=3.0, tau1=60.0, lr0=0.03, decay=0.997, k_skip=0):
    best_h = h.copy()
    best_peak = float(np.correlate(h, 1.0 - h, mode='full').max())
    dx = 2.0 / N
    lr = lr0
    m = np.zeros(N)
    v = np.zeros(N)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    it = 0
    n_total = max(1, int((deadline - time.time()) * 8))
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        frac = min(1.0, it / n_total)
        tau = tau0 * (tau1 / tau0) ** frac
        w = softmax_weights(corr, tau)
        grad = compute_grad(h, w)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        it += 1
        mh = m / (1.0 - beta1 ** it)
        vh = v / (1.0 - beta2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 50 == 0:
            lr *= decay
    return best_h, best_peak * dx

def binary_sd(h, N, max_sweeps=300, deadline=None):
    h = h.astype(float).copy()
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_peak = P
        best_pq = None
        for p in ones:
            base = corr.copy()
            d = p - (N - 1)
            if d >= 0:
                base[:N - d] += (1.0 - h[d:]) * 0  # placeholder
                # Effect of setting h[p]=0: corr changes
                # d(corr[k])/d(h[p]) for k=N-1+dd: (1-h[p-dd])*I[0<=p-dd<N] - h[p+dd]*I[0<=p+dd<N]
                pass
            # Compute effect of flipping h[p] from 1 to 0 on corr
            # For lag dd: delta_c = -(1-h[p-dd])*I[0<=p-dd<N] + h[p+dd]*I[0<=p+dd<N]
            # Then flipping h[q] from 0 to 1: delta_c = +(1-h[q-dd])*I[0<=q-dd<N] - h[q+dd]*I[0<=q+dd<N]
            # New corr = corr - D_p + D_q where D_p is effect of unsetting p, D_q effect of setting q
            # Effect of unsetting p (1->0): -[(1-h[p-dd])*I_p_minus + (-h[p+dd])*I_p_plus] = h[p+dd]*I_p_plus - (1-h[p-dd])*I_p_minus
            # Wait, let me recompute.
            # corr[N-1+dd] = sum_n h[n]*(1-h[n-dd]) * I[valid]
            # Setting h[p]=0: remove term h[p]*(1-h[p-dd])*I[0<=p-dd<N]
            #   AND for each n such that n-dd=p, i.e., n=p+dd: remove term h[p+dd]*(1-h[p])*I[0<=p+dd<N]
            #   = change in corr[N-1+dd] = -(1-h[p-dd])*I[0<=p-dd<N] - h[p+dd]*I[0<=p+dd<N]*(1-0)
            #   Wait no: the second term: originally h[p+dd]*(1-h[p]) with h[p]=1, so it was 0.
            #   After h[p]=0: h[p+dd]*(1-0) = h[p+dd]. So change = +h[p+dd]*I[0<=p+dd<N]
            #   And first term: originally h[p]*(1-h[p-dd]) = 1*(1-h[p-dd]). After h[p]=0: 0. Change = -(1-h[p-dd])*I[0<=p-dd<N]
            # Total change for unset p: DD_p[dd] = -(1-h[p-dd])*I[0<=p-dd<N] + h[p+dd]*I[0<=p+dd<N]
            # For set q (0->1): change is opposite: DD_q[dd] = +(1-h[q-dd])*I[0<=q-dd<N] - h[q+dd]*I[0<=q+dd<N]
            # But wait, after unsetting p, h changes. For simplicity, compute with original h then adjust.
            # Actually for the swap (unset p, set q), the combined effect on corr[N-1+dd]:
            #   -h_orig[p]*(1-h_orig[p-dd])*I[0<=p-dd<N]  (lose p's contribution as h[n])
            #   +h_orig[p+dd]*(1-0)*I[0<=p+dd<N]  (gain from n=p+dd now seeing h[p]=0)
            #   +h_orig[q]*(1-0)*I[0<=q-dd<N]  (gain q as h[n], h[q] was 0 now 1... wait h[q] is 0 originally)
            #   Hmm this is getting complicated. Let me just use the D matrix approach.
            break  # Fallback to simpler method
        # Use D matrix approach instead
        # D[j, dd] = effect of h[j] on corr[N-1+dd]
        # d(corr[N-1+dd])/d(h[j]) = (1-h[j-dd])*I[0<=j-dd<N] - h[j+dd]*I[0<=j+dd<N]
        D = np.zeros((N, 2 * N - 1))
        for j in range(N):
            for dd in range(-(N - 1), N):
                val = 0.0
                if 0 <= j - dd < N:
                    val += (1.0 - h[j - dd])
                if 0 <= j + dd < N:
                    val -= h[j + dd]
                D[j, dd + N - 1] = val
        for p in ones:
            base = corr - D[p, :]
            candidates = base[None, :] + D[zeros, :]
            new_peaks = candidates.max(axis=1)
            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_peak:
                best_peak = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def binary_sd_fast(h, N, max_sweeps=300, deadline=None):
    h = h.astype(float).copy()
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_peak = P
        best_pq = None
        # Compute D for all positions at once
        # D[j, dd+N-1] = (1-h[j-dd])*I[0<=j-dd<N] - h[j+dd]*I[0<=j+dd<N]
        D = np.zeros((N, 2 * N - 1))
        for dd in range(-(N - 1), N):
            wd = dd + N - 1
            if dd >= 0:
                if N - dd > 0:
                    D[dd:, wd] += (1.0 - h[:N - dd])
                    D[:N - dd, wd] -= h[dd:]
            else:
                m = -dd
                if N - m > 0:
                    D[:N - m, wd] += (1.0 - h[m:])
                    D[m:, wd] -= h[:N - m]
        for p in ones:
            base = corr - D[p, :]
            cands = base[None, :] + D[zeros, :]
            npeaks = cands.max(axis=1)
            mi = np.argmin(npeaks)
            if npeaks[mi] < best_peak:
                best_peak = float(npeaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def anneal_binary(h, N, n_iters, rng, T0=2.0, T1=0.01, deadline=None):
    h = h.astype(float).copy()
    best = h.copy()
    corr_b = np.correlate(h, 1.0 - h, mode='full')
    best_pk = float(corr_b.max())
    cur_pk = best_pk
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / n_iters)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_pk = float(corr_new.max())
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.005 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
        # Update sets every 100 iters
        if it % 100 == 0 and it > 0:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    return best, best_pk * (2.0 / N)

def make_binary(N, h_cont, rng):
    """Binarize continuous h to have exactly N/2 ones."""
    n1 = N // 2
    sort_idx = np.argsort(-h_cont)
    h_b = np.zeros(N)
    h_b[sort_idx[:n1]] = 1.0
    return h_b
