def compute_c5(h, N):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * (2.0 / N)

    h = np.clip(h, 0.0, 1.0)
    diff = float(h.sum()) - target
    if abs(diff) < 1e-10:
        return h
    lo, hi = -1.2, 1.2

            lo = mid
        else:
            hi = mid

def make_G(h, N):
    """G[p,m] = d(c[m])/d(h[p]) = 1 - h[p+lag] - h[p-lag]"""
    nx = 2 * N - 1

    c = np.correlate(h, 1.0 - h, mode='full')

        zeros = np.where(h == 0)[0]

            mi = int(np.argmin(new_peaks))
            if float(new_peaks[mi]) < best_new:
