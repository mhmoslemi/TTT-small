def binary_local_search(h, n, t_end, rng):

            improved = False
            for i in range(n):

                    break

                cv = compute_c5(b)
                if cv < best_c5 - 1e-14:

                    improved = True
                else:
                    b[i] = old
                    b = project(b, n)

                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5 = hc, cv
            except Exception:
                pass
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        v1 = idx + k - n + 1
        m1 = (v1 >= 0) & (v1 < n)
        if m1.any():
            grad[m1] -= wk * h[v1[m1]]
        v2 = idx - k + n - 1
        m2 = (v2 >= 0) & (v2 < n)
        if m2.any():
            grad[m2] -= wk * h[v2[m2]]
    grad *= dx
    grad += dx
    return grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (target / max(s, 1e-14))
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if abs(diff2) > 1e-13:
            if diff2 > 0:
                room = 1.0 - h
                tot = room.sum()
                h = h + room * (diff2 / max(tot, 1e-14))
            else:
                h = h * (target / max(s2, 1e-14))
                h = np.clip(h, 0.0, 1.0)
    return h

def nesterov_gradient(h, n, T, t_end, n_steps=300, lr_init=0.02, beta=0.9):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    mom = np.zeros(n)
    for step in range(n_steps):
        if time.time() > t_end:
            break
        g = smooth_grad(h, T)
        mom = beta * mom + (1 - beta) * g
        lr = lr_init / (1.0 + step * 0.01)
        h_new = h - lr * mom
        h_new = project(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new
        if c5_new < compute_c5(h):
            h = h_new
        else:
            lr_init *= 0.8
            h = h_new
        if step % 50 == 0:
            T = min(T * 1.5, 20000)
    return best_h, best_c5

def coord_opt(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    corr_base = get_corr(h).copy()
    h[i] = h_save
    dx = 2.0 / n
    d = np.zeros(2 * n - 1)
    v1 = i + np.arange(2 * n - 1) - n + 1
    m1 = (v1 >= 0) & (v1 < n)
    d[m1] += (1.0 - h[v1[m1]])
    v2 = i - np.arange(2 * n - 1) + n - 1
    m2 = (v2 >= 0) & (v2 < n)
    d[m2] -= h[v2[m2]]
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t - t * t
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                         options={'xatol': 1e-14})
    return res.x

def active_coord_descent(h, n, t_end, max_sweeps=60, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = get_corr(best_h)
        cmax = np.max(corr)
        thresh = cmax * 0.995
        active_mask = corr > thresh
        coord_wt = np.zeros(n)
        for k in range(2 * n - 1):
            if not active_mask[k]:
                continue
            v1 = np.arange(n) + k - n + 1
            m1 = (v1 >= 0) & (v1 < n)
            coord_wt[m1] += 1
            v2 = np.arange(n) - k + n + 1 - 2
            v2 = np.arange(n) - k + n - 1
            m2 = (v2 >= 0) & (v2 < n)
            coord_wt[m2] += 1
        probs = (coord_wt + 0.5).astype(float)
        probs /= probs.sum()
        order = rng.choice(n, size=n, replace=False, p=probs)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord
            coord = coord_opt(best_h, n, i)
            if abs(coord - best_h[i]) > 1e-12:
                best_h[i] = coord
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def binary_local_search(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for thresh in [0.35, 0.5, 0.65]:
        if time.time() > t_end:
            break
        b = (best_h > thresh).astype(float)
        b = project(b, n)
        improved = True
        n_iter = 0
        while improved and time.time() < t_end and n_iter < 5:
            improved = False
            n_iter += 1
            for i in range(n):
                if time.time() > t_end:
                    break
                old = b[i]
                b[i] = 1.0 - old
                b = project(b, n)
                cv = compute_c5(b)
                if cv < best_c5 - 1e-14:
                    best_h, best_c5 = b.copy(), cv
                    improved = True
                else:
                    b[i] = old
                    b = project(b, n)
    # Continuous refinement
    if time.time() < t_end:
        best_h, best_c5 = active_coord_descent(best_h, n, min(t_end, time.time() + 30),
                                                max_sweeps=15, rng=rng)
    return best_h, best_c5

def perturbation_search(h, n, t_end, rng, n_trials=8):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for trial in range(n_trials):
        if time.time() > t_end:
            break
        scale = 0.03 * (1.0 + 0.5 * rng.random())
        mask = rng.random(n) < 0.3
        pert = best_h.copy()
        pert[mask] += scale * rng.standard_normal(mask.sum())
        pert = np.clip(pert, 0, 1)
        pert = project(pert, n)
        h2, c2 = nesterov_gradient(pert, n, T=500, t_end=min(t_end, time.time() + 15),
                                    n_steps=80, lr_init=0.01)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 240
        best_h = np.zeros(n0)
        best_h[:n0 // 2] = 1.0

    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov gradient with increasing T (12%)
    t1 = t0 + budget_s * 0.12
    for T in [100, 500, 2000]:
        if elapsed() > t1:
            break
        h2, c2 = nesterov_gradient(best_h, n, T, t1, n_steps=150, lr_init=0.015)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 2: Active coordinate descent (20%)
    t2 = t0 + budget_s * 0.32
    best_h, best_c5 = active_coord_descent(best_h, n, t2, max_sweeps=80, rng=rng)

    # Phase 3: More gradient refinement (10%)
    t3 = t0 + budget_s * 0.42
    h2, c2 = nesterov_gradient(best_h, n, T=5000, t_end=t3, n_steps=100, lr_init=0.008)
    if c2 < best_c5:
        best_h, best_c5 = h2, c2
    best_h, best_c5 = active_coord_descent(best_h, n, t3, max_sweeps=20, rng=rng)

    # Phase 4: Binary local search (15%)
    t4 = t0 + budget_s * 0.57
    best_h, best_c5 = binary_local_search(best_h, n, t4, rng)

    # Phase 5: Perturbation search (15%)
    t5 = t0 + budget_s * 0.72
    best_h, best_c5 = perturbation_search(best_h, n, t5, rng, n_trials=10)

    # Phase 6: Coordinate descent polish (10%)
    t6 = t0 + budget_s * 0.82
    best_h, best_c5 = active_coord_descent(best_h, n, t6, max_sweeps=50, rng=rng)

    # Phase 7: Multi-resolution attempt (10%)
    for n_try in [200, 288]:
        if elapsed() > budget_s * 0.90:
            break
        h_interp = interpolate_h(best_h, n_try)
        h2, c2 = nesterov_gradient(h_interp, n_try, T=1000, t_end=min(t0 + budget_s * 0.93, time.time() + 20),
                                    n_steps=80, lr_init=0.01)
        if c2 < best_c5:
            best_h, best_c5, n = h2, c2, n_try
        h2, c2 = active_coord_descent(h_interp, n_try, min(t0 + budget_s * 0.95, time.time() + 15),
                                       max_sweeps=20, rng=rng)
        if c2 < best_c5:
            best_h, best_c5, n = h2, c2, n_try

    # Phase 8: Final coordinate descent (8%)
    t8 = t0 + budget_s * 0.98
    best_h, best_c5 = active_coord_descent(best_h, n, t8, max_sweeps=30, rng=rng)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
