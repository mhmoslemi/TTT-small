```python
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 1e-6:
        return m * dx
    val = m + np.logaddexp.reduce(T*(corr - m)) / T  # not quite right
    # Better: m + log(sum(exp(T*(corr-m))))/T
    shifted = T * (corr - m)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
    return h

    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h.copy(), method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return project(np.clip(res.x, 0, 1), n)

    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save

    d = np.zeros(2 * n - 1)

    for k in range(1, n):

        if i >= k:

        if i >= k:
            d[n-1-k] += 1.0 - h[i-k]

    # Actually corr[n-1] when h[i]=t: the contribution is t*(1-t) instead of 0
    # corr_base[n-1] already has h[i]=0, so the actual corr[n-1] = corr_base[n-1] + t*(1-t)
    # And for k≠0: corr[k] = corr_base[k] + t*d[k]
    
    c0_base = corr_base[n-1]
    
    def f1d(t):
        vals = corr_base + t * d
        vals[n-1] = c0_base + t * (1.0 - t)
        return np.max(vals) * dx
    
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return res.x
```

This coordinate descent is O(n) per coordinate (for the d computation) plus O(n) for the 1D optimization (which evaluates f1d O(log(1/eps)) times, each O(n)). So total per coordinate is O(n log(1/eps)) ≈ O(n). For all n coordinates, one sweep is O(n²). For n=240, that's about 57600 operations, very fast.

Actually, I can precompute the d vector more efficiently. Notice that:
d[n-1+k] = (1-h[i+k])*[i+k<n] - h[i-k]*[i>=k] for the "first factor" contribution
           + ... wait, I already computed it correctly above.

Let me think about the 1D optimization. The function f1d(t) = max(corr_base + t*d, with special k=0 term) is piecewise linear (from the k≠0 terms) plus one parabola (from k=0). The maximum of piecewise linear + parabola is a convex piecewise function. We can find its minimum by:
1. Finding all the "break points" where the argmax changes
2. Evaluating f at each break point and at the vertex of the parabola

The break points are where two correlation values cross: corr_base[a] + t*d[a] = corr_base[b] + t*d[b], i.e., t = (corr_base[b]-corr_base[a])/(d[a]-d[b]).

There are O(n²) such break points, but we only need to check the ones in [0,1]. This is getting complex.

A simpler approach: since f1d is convex (max of affine functions + convex parabola), we can use a ternary search or golden section search. But `minimize_scalar` with bounded method already does this.

The issue is the number of f1d evaluations. With xatol=1e-14, the bounded method (Brent) might do 100+ evaluations. Each evaluation is O(n) (computing max of 2n-1 values). So 100*240 = 24000 operations per coordinate. For 240 coordinates, that's 5.76M per sweep. In Python with numpy, this is maybe 0.5-1 seconds per sweep.

We can do maybe 10-20 sweeps in the time budget. That should be sufficient for convergence.

Let me now write the complete solution:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def project(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return h

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt(h, n, i):
    h_sa = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_sa
    d = np.zeros(2 * n - 1)
    d[n-1] = 1.0 - 2.0 * h_sa
    for k in range(1, n):
        if i + k < n:
            d[n-1+k] += 1.0 - h[i+k]
            d[n-1-k] -= h[i+k]
        if i >= k:
            d[n-1+k] -= h[i-k]
            d[n-1-k] += 1.0 - h[i-k]
    c0b = cb[n-1]
    def f1d(t):
        v = cb + t * d
        v[n-1] = c0b + t * (1.0 - t)
        return np.max(v) * (2.0 / n)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-14})
    return res.x

def coord_descent(h, n, t_end, rng, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-13:
                best_h[i] = hi
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                break
    return best_h, best_c5
