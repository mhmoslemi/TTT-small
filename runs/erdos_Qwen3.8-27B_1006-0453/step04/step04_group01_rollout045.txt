import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_continuous(h, n):
    target = n / 2.0
    vec = np.clip(h.astype(float), 0.0, 1.0)
    if abs(vec.sum() - target) < 1e-14:
        return vec
    lo = -float(np.max(vec)) - 1.0
    hi = 1.0 - float(np.min(vec)) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(vec + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, n, rng):
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    s = int(h_bin.sum())
    if s > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(len(ones), s - target, replace=False)
        h_bin[ones[idx]] = 0
    elif s < target:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(len(zeros), target - s, replace=False)
        h_bin[zeros[idx]] = 1
    return h_bin

def random_binary(n, rng):
    h = np.zeros(n)
    ones_positions = rng.choice(n, n // 2, replace=False)
    h[ones_positions] = 1.0
    return h

def sa_binary(h_init, n, t_end, rng, T0=0.003, Tmin=0.00002, move_types='swap'):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    t_start = time.time()
    total = max(t_end - t_start, 1.0)
    stagnant = 0
    
    iterations = 0
    while time.time() < t_end - 2.0:
        elapsed_frac = min((time.time() - t_start) / total, 1.0)
        T = T0 * (Tmin / T0) ** elapsed_frac
        
        # Choose move
        move = rng.random()
        h_new = h.copy()
        
        if move < 0.7:
            # Single swap: pick i with h=1, j with h=0, swap

            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i], h_new[j] = h_new[j], h_new[i]
        elif move < 0.9:
            # Double swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) >= 2 and len(zeros) >= 2:
                idx1 = rng.choice(len(ones), 2, replace=False)
                idx2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[idx1[0]]] = 0
                h_new[ones[idx1[1]]] = 0
                h_new[zeros[idx2[0]]] = 1
                h_new[zeros[idx2[1]]] = 1
                # Restore if we made a mess
            else:
                h_new = h.copy()
        else:

            h_new = h.copy()
            n_flips = rng.randint(3, 8)

                h_new[ones_idx[to_flip]] = 0

                zero_idx = np.where(h_new == 0)[0]

def coord_descent_continuous(h, n, t_end, rng):

    best_c5 = compute_c5(best_h)
    for sweep in range(10000):
        if time.time() > t_end:

                h1[i] = m1

def slsqp_opt(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx
    
    def con(x):
        return x.sum() - n / 2.0
    
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_continuous(res.x, n)
    except Exception:
        return project_continuous(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_continuous(h, n):
    target = n / 2.0
    vec = np.clip(h.astype(float), 0.0, 1.0)
    if abs(vec.sum() - target) < 1e-14:
        return vec
    lo = -float(np.max(vec)) - 1.0
    hi = 1.0 - float(np.min(vec)) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(vec + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, n, rng):
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    s = int(h_bin.sum())
    if s > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(len(ones), s - target, replace=False)
        h_bin[ones[idx]] = 0.0
    elif s < target:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(len(zeros), target - s, replace=False)
        h_bin[zeros[idx]] = 1.0
    return h_bin

def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def sa_binary(h_init, n, t_end, rng, T0=0.003, Tmin=0.00002):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    t_start = time.time()
    total_dur = max(t_end - t_start, 1.0)
    stagnant = 0
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]

    while time.time() < t_end - 1.0:
        el = min((time.time() - t_start) / total_dur, 1.0)
        T = T0 * (Tmin / T0) ** el
        r = rng.random()
        h_new = h.copy()

        if r < 0.65:
            i = ones_cache[rng.integers(len(ones_cache))]
            j = zeros_cache[rng.integers(len(zeros_cache))]
            h_new[i], h_new[j] = 0.0, 1.0
        elif r < 0.85:
            if len(ones_cache) >= 2 and len(zeros_cache) >= 2:
                oi = rng.choice(len(ones_cache), 2, replace=False)
                zi = rng.choice(len(zeros_cache), 2, replace=False)
                h_new[ones_cache[oi[0]]] = 0.0
                h_new[ones_cache[oi[1]]] = 0.0
                h_new[zeros_cache[zi[0]]] = 1.0
                h_new[zeros_cache[zi[1]]] = 1.0
            else:
                continue
        else:
            nf = rng.integers(3, 9)
            pos = rng.choice(n, nf, replace=False)
            h_new[pos] = 1.0 - h_new[pos]
            s = int(h_new.sum())
            if s > n // 2:
                oi2 = np.where(h_new == 1)[0]
                h_new[oi2[rng.choice(len(oi2), s - n // 2, replace=False)]] = 0.0
            elif s < n // 2:
                zi2 = np.where(h_new == 0)[0]
                h_new[zi2[rng.choice(len(zi2), n // 2 - s, replace=False)]] = 1.0

        new_c5 = compute_c5(h_new)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            h = h_new
            cur_c5 = new_c5
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
        if stagnant > 3000:
            stagnant = 0

    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        val = m + float(np.log(np.sum(np.exp(T * (corr - m))))) / T
        return val * dx
    def con(x):
        return float(x.sum() - n / 2.0)
    cons = [{'type': 'eq', 'fun': con}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_continuous(res.x, n)
    except Exception:
        return project_continuous(h, n)

def coord_descent(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    while time.time() < t_end:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            lo, hi = 0.0, 1.0
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                h1 = best_h.copy(); h1[i] = m1; h1 = project_continuous(h1, n)
                h2 = best_h.copy(); h2[i] = m2; h2 = project_continuous(h2, n)
                if compute_c5(h1) < compute_c5(h2):
                    hi = m2
                else:
                    lo = m1
            nv = (lo + hi) / 2.0
            ht = best_h.copy(); ht[i] = nv; ht = project_continuous(ht, n)
            cv = compute_c5(ht)
            if cv < best_c5 - 1e-15:
                best_c5 = cv
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(160)
        init_h[:80] = 1.0

    n = len(init_h)
    best_h = init_h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary SA from initial (20%)
    t1 = t0 + budget_s * 0.20
    h_bin = to_binary(init_h, n, rng)
    h1, c1 = sa_binary(h_bin, n, t1, rng, T0=0.004, Tmin=0.00003)
    if c1 < best_c5:
        best_c5, best_h = c1, h1

    # Phase 2: Diverse starts with SA (25%)
    t2 = t0 + budget_s * 0.45
    trial_ns = [n, n+40, n+80, n-40, n+120]
    trial_ns = sorted(set(nn for nn in trial_ns if nn >= 80))
    for nn in trial_ns:
        if time.time() > t2:
            break
        sub_end = min(time.time() + budget_s * 0.055, t2)
        for trial in range(2):
            if time.time() > sub_end:
                break
            hr = random_binary(nn, rng)
            hb, cb = sa_binary(hr, nn, sub_end, rng, T0=0.003, Tmin=0.00002)
            if cb < best_c5:
                best_c5, best_h, n = cb, hb, nn

    # Phase 3: Focused SA from best (20%)
    t3 = t0 + budget_s * 0.70
    h_best_bin = to_binary(best_h, n, rng)
    h3, c3 = sa_binary(h_best_bin, n, t3, rng, T0=0.0015, Tmin=0.00001)
    if c3 < best_c5:
        best_c5, best_h = c3, h3

    # Phase 4: Continuous SLSQP (15%)
    t4 = t0 + budget_s * 0.85
    h_cont = project_continuous(best_h.astype(float), n)
    for T in [200, 1000, 5000, 20000]:
        if time.time() > t4:
            break
        hc = slsqp_opt(h_cont.copy(), T, n, maxiter=400)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5, best_h, h_cont = cv, hc, hc

    # Phase 5: CD polish (12%)
    t5 = t0 + budget_s * 0.98
    best_h, best_c5 = coord_descent(best_h, n, t5, rng)

    best_h = project_continuous(best_h, n)
    best_c5 = compute_c5(best_h)
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, rng)
        best_h = project_continuous(best_h, n)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
