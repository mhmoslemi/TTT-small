import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h * (target / s), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    h_old = h[i]
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_old
    center = n - 1
    coeff = np.zeros(2 * n - 1)
    for k in range(-n + 1, n):
        p = center + k
        c = 0.0
        if k == 0:
            c = 1.0
        elif k > 0:
            if k <= i < n:
                c += 1.0 - h[i - k]
            if 0 <= i < n - k:
                c -= h[i + k]
        else:
            m2 = -k
            if 0 <= i < n - m2:
                c += 1.0 - h[i + m2]
            if m2 <= i < n:
                c -= h[i - m2]
        coeff[p] = c
    def f1d(x):
        vals = corr0 + x * coeff
        vals[center] += x - x * x
        return np.max(vals)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=9999):

    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):

        improved = False
