    corr = compute_corr(h)
    scaled = alpha * corr
    mx = np.max(scaled)
    exps = np.exp(scaled - mx)
    wts = exps / np.sum(exps)

    smax = (mx + np.log(np.sum(exps))) / alpha

            grad[v1] += w * (1.0 - h[np.floor(j[v1] - m).astype(int)])
            v2 = j < N - m
            grad[v2] -= w * h[np.floor(j[v2] + m).astype(int)]
    
    return smax, grad * alpha

def adam_smooth(h, N, target, deadline, alpha0=100.0, alpha1=5000.0, lr=0.01, iters=5000):
    best = h.copy()
    best_val = float(np.max(compute_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    for it in range(iters):
        if time.time() > deadline:
            break
        t += 1
        alpha = alpha0 * (alpha1 / alpha0) ** (it / iters)
        smax, g = smooth_max_and_grad(h, N, alpha)
        # true max
        true_max = float(np.max(compute_corr(h)))
        if true_max < best_val:
            best_val = true_max
            best = h.copy()
        mh = m = b1 * m + (1 - b1) * g
        vh = v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best, best_val * (2.0 / N)

        # D[p, L]: change in c[L] when h[p] goes 1->0
        # = -(h[p+lag] + 1 - h[p-lag]) for valid indices
        center = N - 1
        # For each one p, compute effect on all lags
        best_improve = 0
        best_pq = None
        corr_cur = compute_corr(h)
        for p in ones:
            # Effect of h[p]: 1->0 on each lag L
            lag_arr = np.arange(2*N-1) - center  # -(N-1) to N-1
            # h[p+lag] valid: 0<=p+lag<N
            # 1-h[p-lag] valid: 0<=p-lag<N
            valid_plus = (p + lag_arr >= 0) & (p + lag_arr < N)
            valid_minus = (p - lag_arr >= 0) & (p - lag_arr < N)
            hp = np.where(valid_plus, h[np.clip(p + lag_arr, 0, N-1)], 0.0)
            hm = np.where(valid_minus, h[np.clip(p - lag_arr, 0, N-1)], 0.0)
            # Δc[L] for h[p]: 1->0 = -hp - (1-hm) where valid, else 0
            delta_p = np.zeros(2*N-1)
            delta_p[valid_plus] = -hp[valid_plus]
            delta_p[valid_minus] -= (1.0 - hm[valid_minus])
            # For zeros: Δc[L] for h[q]: 0->1 = (1-h[q-lag]) - h[q+lag]
            # Target = corr_cur - delta_p + delta_q for each q
            # new_max = max of (corr_cur - delta_p + delta_q)
            for q in zeros:
                valid_plus_q = (q + lag_arr >= 0) & (q + lag_arr < N)
                valid_minus_q = (q - lag_arr >= 0) & (q - lag_arr < N)
                hq = np.where(valid_plus_q, h[np.clip(q + lag_arr, 0, N-1)], 0.0)
                hqm = np.where(valid_minus_q, h[np.clip(q - lag_arr, 0, N-1)], 0.0)
                delta_q = np.zeros(2*N-1)
                delta_q[valid_minus_q] = (1.0 - hqm[valid_minus_q])
                delta_q[valid_plus_q] -= hq[valid_plus_q]
                new_corr = corr_cur - delta_p + delta_q
                new_max = float(np.max(new_corr))
                if cur_max - new_max > best_improve:
                    best_improve = cur_max - new_max
                    best_pq = (p, q)
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        cur_max = float(np.max(compute_corr(h)))
    return h

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binary_local_search(h, N, max_sweeps=500, deadline=None):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n1]] = 1
    center = N - 1
    lag_arr = np.arange(2 * N - 1) - center
    for _ in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        c = compute_corr(h)
        P = float(c.max())
        best_peak = P
        best_pq = None
        for p in ones:
            vp = (p + lag_arr >= 0) & (p + lag_arr < N)
            vm = (p - lag_arr >= 0) & (p - lag_arr < N)
            delta_p = np.zeros(2 * N - 1)
            delta_p[vp] -= h[np.clip(p + lag_arr, 0, N - 1)][vp]
            delta_p[vm] -= (1.0 - h[np.clip(p - lag_arr, 0, N - 1)][vm])
            base = c - delta_p
            qp = zeros[:, None] + lag_arr[None, :]
            qm = zeros[:, None] - lag_arr[None, :]
            vqp = (qp >= 0) & (qp < N)
            vqm = (qm >= 0) & (qm < N)
            hqp = np.where(vqp, h[np.clip(qp, 0, N - 1)], 0.0)
            hqm = np.where(vqm, h[np.clip(qm, 0, N - 1)], 0.0)
            delta_q = vqm * (1.0 - hqm) - vqp * hqp
            new_peaks = (base[None, :] + delta_q).max(axis=1)
            mi = int(np.argmin(new_peaks))
            if new_peaks[mi] < best_peak - 1e-12:
                best_peak = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def anneal_binary(h, N, n_iters, rng, T0=2.0, T1=0.002, deadline=None):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n1]] = 1
    best = h.copy()
    best_pk = float(np.max(compute_corr(h)))
    cur_pk = best_pk
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / n_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_pk = float(np.max(compute_corr(h)))
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.003 and rng.random() < np.exp(-delta / T)):
            cur_pk = new_pk
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_pk * (2.0 / N)

def gradient_smooth_max(h, N, alpha):
    corr = compute_corr(h)
    scaled = alpha * corr
    mx = np.max(scaled)
    exps = np.exp(scaled - mx)
    wts = exps / np.sum(exps)
    grad = np.zeros(N)
    center = N - 1
    lag_arr = np.arange(2 * N - 1) - center
    for k in range(2 * N - 1):
        lag = lag_arr[k]
        w = wts[k]
        if w < 1e-13:
            continue
        if lag >= 0:
            idx1 = np.arange(lag, N)
            grad[idx1] += w * (1.0 - h[idx1 - lag])
            idx2 = np.arange(0, N - lag)
            grad[idx2] -= w * h[idx2 + lag]
        else:
            m = -lag
            idx1 = np.arange(m, N)
            grad[idx1] += w * (1.0 - h[idx1 - m])
            idx2 = np.arange(0, N - m)
            grad[idx2] -= w * h[idx2 + m]
    return grad * alpha

def adam_continuous(h, N, target, deadline, iters=3000, alphas=None, lr=0.008):
    if alphas is None:
        alphas = np.geomspace(50, 3000, iters)
    best = h.copy()
    best_val = float(np.max(compute_corr(h)))
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.8, 0.999, 1e-8
    t = 0
    for it in range(iters):
        if time.time() > deadline:
            break
        t += 1
        alpha = alphas[min(it, len(alphas) - 1)]
        g = gradient_smooth_max(h, N, alpha)
        true_max = float(np.max(compute_corr(h)))
        if true_max < best_val:
            best_val = true_max
            best = h.copy()
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
    return best, best_val * (2.0 / N)

def make_perturbed(h, N, rng, strength=0.1):
    hp = h.astype(float) + rng.randn(N) * strength
    hp = project_sum(hp, N / 2.0)
    hb = (hp > 0.5).astype(float)
    n1 = int(hb.sum())
    nh = N // 2
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        hb[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(hb == 0)[0]
        hb[idx[:nh - n1]] = 1
    return hb

def interpolate_rebin(h, N_src, N_dst):
    x_old = np.linspace(0, 2, N_src, endpoint=False)
    x_new = np.linspace(0, 2, N_dst, endpoint=False)
    hi = np.interp(x_new, x_old, h)
    hi = (hi > 0.5).astype(float)
    n1 = int(hi.sum())
    nh = N_dst // 2
    if n1 > nh:
        idx = np.where(hi == 1)[0]
        hi[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(hi == 0)[0]
        hi[idx[:nh - n1]] = 1
    return hi
