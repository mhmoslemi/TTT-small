import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full')

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0).copy()

def coord_descent(h, t_end, n_sweeps=2000):
    n = len(h)
    best_h = h.copy()
    dx = 2.0 / n
    corr = compute_corr(best_h)
    best_c5 = float(np.max(corr) * dx)
    k_range = np.arange(-(n - 1), n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            j1 = i - k_range
            j2 = i + k_range
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(2 * n - 1)
            if v2.any():
                coef[v2] += (1.0 - best_h[j2[v2]])
            if v1.any():
                coef[v1] -= best_h[j1[v1]]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            a, b = lo, hi
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            fc = float(np.max(corr + c_ * coef))
            fd = float(np.max(corr + d_ * coef))
            for _ in range(45):
                if b - a < 1e-13:
                    break
                if fc < fd:
                    b = d_; d_ = c_; fd = fc
                    c_ = b - gr * (b - a)
                    fc = float(np.max(corr + c_ * coef))
                else:
                    a = c_; c_ = d_; fc = fd
                    d_ = a + gr * (b - a)
                    fd = float(np.max(corr + d_ * coef))
            s_new = (a + b) / 2.0
            if abs(s_new) > 1e-12:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = float(np.max(corr) * dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        m = np.max(x) * 0  # placeholder
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(corr)
        return (mx + np.log(np.sum(np.exp(T * (corr - mx)))) / T) * dx
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(corr)
        w = np.exp(T * (corr - mx))
        w /= w.sum()
        grad = np.zeros(2 * n - 1)
        grad[:] = w * T * dx
        k_range = np.arange(-(n - 1), n)
        g = np.zeros(n)
        for i in range(n):
            j1 = i - k_range
            j2 = i + k_range
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            if v2.any():
                g[i] += np.sum(grad[v2] * (1.0 - x[j2[v2]]))
            if v1.any():
                g[i] -= np.sum(grad[v1] * x[j1[v1]])
        return g
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.success or res.x is not None:
        return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)
    return h

def interp_cyclic(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old).astype(float)
    x_new = np.linspace(0, n_old - 1, n_new, endpoint=False)
    h_extended = np.concatenate([h, h])
    h_new = np.interp(x_new, x_old, np.interp(np.mod(x_new, n_old), np.arange(n_old), h))
    return project_to_sum(h_new, n_new / 2.0)

def get_active_lags(corr, frac=0.5):
    mx = np.max(corr)
    thresh = mx - (1.0 - frac) * (np.max(corr) - np.min(corr)) * 0.1
    return np.where(corr >= thresh)[0]

def subgradient_direction(h, corr, n, n_active=5):
    k_range = np.arange(-(n - 1), n)
    mx = np.max(corr)
    idx = np.argsort(corr)[::-1][:n_active]
    g = np.zeros(n)
    for m_idx in idx:
        k = int(m_idx) - (n - 1)
        weights = np.exp(-0.5 * (corr[m_idx] - mx) * 20)
        for i in range(n):
            j2 = i + k
            j1 = i - k
            if 0 <= j2 < n:
                g[i] += weights * (1.0 - h[j2])
            if 0 <= j1 < n:
                g[i] -= weights * h[j1]
    norm = np.linalg.norm(g)
    if norm > 1e-12:
        g /= norm
    return -g
