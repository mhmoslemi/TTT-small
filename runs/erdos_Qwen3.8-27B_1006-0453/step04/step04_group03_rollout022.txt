        else:
            shift = rng.integers(1, max(2, n // 50))
            h_p = 0.6 * np.roll(work_h, shift) + 0.4 * work_h
        h_p = project_to_sum(h_p, n / 2)
        # Quick LP coord descent on perturbed
        t_sub = time.time() + max(0.5, budget_s * 0.03 * cool)
        try:
            hc, c5v = lp_coord_descent(h_p, n, t_sub, n_sweeps=300)
        except Exception:
            continue
        temp = 0.003 * cool + 0.00005
        if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = work_h.copy(), work_c5

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Final SLSQP + LP polish (10% budget)
    for T in [2000, 10000, 50000]:
        if elapsed() > budget_s * 0.92:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=800, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    t_final = t0 + budget_s * 0.99
    best_h, best_c5 = lp_coord_descent(best_h, n, t_final, n_sweeps=3000)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
