            h_p = work_h.copy()
            h_p[idx] += step * rng.randn(n_chg)
        else:
            shift = int(rng.uniform(-0.03, 0.03) * n)
            h_p = 0.8 * np.roll(work_h, shift) + 0.2 * work_h
        h_p = project_to_sum(h_p, n / 2.0)
        corr_p = compute_corr(h_p)
        hc, c5v = cd_inner(h_p, corr_p, n, time.time() + 2.5, max_sweeps=25)
        temp = 0.003 * cool + 0.00005
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = hc.copy(), c5v

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Multi-resolution exploration (13% budget)
    for n_try in [128, 160, 192, 256, 320]:
        if n_try == n or time.time() > t0 + budget_s * 0.88:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h, kind='cubic')
        h_i = project_to_sum(h_i, n_try / 2.0)
        corr_i = compute_corr(h_i)
        hc, c5v = cd_inner(h_i, corr_i, n_try, time.time() + 5.0, max_sweeps=150)
        if c5v < best_c5:
            best_h, best_c5, n = hc, c5v, n_try

    # Phase 6: Final aggressive CD (10% budget)
    corr = compute_corr(best_h)
    best_h, best_c5 = cd_inner(best_h, corr, n, t0 + budget_s * 0.985, max_sweeps=10000)

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
