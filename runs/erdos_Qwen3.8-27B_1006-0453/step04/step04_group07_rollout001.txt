            # m ranges from -(n-1) to (n-1), indexed by c0+m in corr

            # eff[c0] = (1-h[i]) - h[i] = 1-2*old (linear approx)

            # For the max, we binary search or use bounded minimization
            # The function is max of linear functions (one quadratic), so it's convex in v
            # Use bounded scalar minimization

                if v == old:
                    return best_c5 / dx  # current value
                updates = (v - old) * eff
                new_corr = corr + updates
                new_corr[c0] = corr[c0] + (v - old) * (1.0 - old - v)
                return np.max(new_corr)
            
            # Quick check at boundaries
            f0 = f(0.0)
            f1 = f(1.0)
            # Try bounded minimization
            try:
                res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-9})
                bv, bf = res.x, res.fun
            except:
                bv, bf = old, f(old)
            if f0 < bf: bv, bf = 0.0, f0
            if f1 < bf: bv, bf = 1.0, f1
            
            if bf < best_c5/dx - 1e-13:

                delta = bv - old
                corr += delta * eff
                corr[c0] -= delta * delta  # quadratic correction
                # Recompute max
                best_c5 = np.max(corr) * dx
                best_h[i] = bv
                improved = True
        if not improved:

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def fast_coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    bh = h.copy()
    corr = np.correlate(bh, 1.0 - bh, mode='full').copy()
    bc5 = np.max(corr) * dx
    c0 = n - 1
    ms = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = bh[i]
            jm = i - ms
            jp = i + ms
            eff = np.zeros(2 * n - 1)
            vf = (jm >= 0) & (jm < n)
            vs = (jp >= 0) & (jp < n)
            eff[vf] += 1.0 - bh[jm[vf]]
            eff[vs] -= bh[jp[vs]]
            f0 = np.max(corr + (0.0 - old) * eff)
            f0 = np.max(np.concatenate([corr[0:c0] + (0.0 - old) * eff[0:c0],
                                         [corr[c0] + (0.0 - old) * (1.0 - old - 0.0)],
                                         corr[c0+1:] + (0.0 - old) * eff[c0+1:]]))
            f1 = np.max(np.concatenate([corr[0:c0] + (1.0 - old) * eff[0:c0],
                                         [corr[c0] + (1.0 - old) * (1.0 - old - 1.0)],
                                         corr[c0+1:] + (1.0 - old) * eff[c0+1:]]))
            def fobj(v):
                vc = corr + (v - old) * eff
                vc[c0] = corr[c0] + (v - old) * (1.0 - old - v)
                return np.max(vc)
            best_f = min(f0, f1)
            best_v = 0.0 if f0 < f1 else 1.0
            try:
                res = minimize_scalar(fobj, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-10})
                if res.fun < best_f:
                    best_f, best_v = res.fun, res.x
            except Exception:
                pass
            if best_f * dx < bc5 - 1e-13:
                delta = best_v - old
                corr += delta * eff
                corr[c0] -= delta * delta
                bc5 = np.max(corr) * dx
                bh[i] = best_v
                improved = True
        if not improved:
            break
    return bh, bc5

def binary_local_search(h, n, t_end, max_rounds=200):
    dx = 2.0 / n
    bh = np.where(h > 0.5, 1.0, 0.0)
    bh = project(bh, n)
    corr = np.correlate(bh, 1.0 - bh, mode='full').copy()
    bc5 = np.max(corr) * dx
    c0 = n - 1
    ms = np.arange(-(n - 1), n)
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            new_v = 1.0 - bh[i]
            old = bh[i]
            delta = new_v - old
            jm = i - ms
            jp = i + ms
            eff = np.zeros(2 * n - 1)
            vf = (jm >= 0) & (jm < n)
            vs = (jp >= 0) & (jp < n)
            eff[vf] += 1.0 - bh[jm[vf]]
            eff[vs] -= bh[jp[vs]]
            nc = corr + delta * eff
            nc[c0] += -delta * delta
            nf = np.max(nc)
            if nf < np.max(corr) - 1e-12:
                bh[i] = new_v
                corr = nc.copy()
                bc5 = nf * dx
                improved = True
        if not improved:
            break
    return bh, bc5

def basin_hop(h, c5, n, t_end, max_iters=300):
    best_h, best_c5 = h.copy(), c5
    work_h, work_c5 = h.copy(), c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = it / max_iters
        cool = max(0.03, 1.0 - 0.9 * frac)
        mode = np.random.randint(4)
        if mode == 0:
            step = 0.03 + 0.15 * cool
            hp = work_h + step * np.random.randn(n)
        elif mode == 1:
            bstart = np.random.randint(n)
            blen = np.random.randint(3, max(4, n // 4))
            hp = work_h.copy()
            hp[bstart:(bstart + blen) % n] += (np.random.rand(blen) - 0.5) * 0.6 * cool
        elif mode == 2:
            hp = work_h.copy()
            k = np.random.randint(2, max(3, n // 6))
            idx = np.random.choice(n, k, replace=False)
            hp[idx] = np.random.rand(k)
        else:
            hp = np.where(work_h > 0.5, 1.0, 0.0)
            k = np.random.randint(2, max(3, n // 5))
            idx = np.random.choice(n, k, replace=False)
            hp[idx] = 1.0 - hp[idx]
        hp = project(hp, n)
        # Quick coord descent on perturbed
        try:
            hc, c5v = fast_coord_descent(hp, n, time.time() + 8, max_sweeps=30)
        except Exception:
            continue
        temp = 0.0004 + 0.003 * cool
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = hc.copy(), c5v
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5
    return best_h, best_c5

def slsqp_opt(h0, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h), n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Initial coordinate descent
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.12, max_sweeps=400)

    # Phase 2: Binary local search
    bh, bc5b = binary_local_search(best_h, n, t0 + budget_s * 0.18, max_rounds=100)
    if bc5b < best_c5:

    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.22, max_sweeps=300)

    # Phase 3: SLSQP refinement
    for T in [50, 300, 2000]:
        if elapsed() > budget_s * 0.32:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.38, max_sweeps=300)

    # Phase 4: Basin hopping (main exploration)
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.72, max_iters=250)
