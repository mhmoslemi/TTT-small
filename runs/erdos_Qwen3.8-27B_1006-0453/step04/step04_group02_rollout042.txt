    return np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n

    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full') * dx
    m = c.max()
    return m + np.log(np.sum(np.exp(T*(c-m)))) / T

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        if room.sum() > 1e-14:
            h += room * (target - s) / room.sum()
    else:
        h *= target / max(s, 1e-14)
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    return m + np.log(np.sum(np.exp(T * (c - m)))) / T

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        if room.sum() > 1e-14:
            h += room * (target - s) / room.sum()
    else:
        h *= target / max(s, 1e-14)
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if s2 < target and (1.0 - h).sum() > 1e-14:
            h += (1.0 - h) * (target - s2) / (1.0 - h).sum()
    return np.clip(h, 0.0, 1.0)

def grad_sm(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = c.max()
    w = np.exp(T * (c - m))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(-(n - 1), n):
        j = n - 1 + k
        if w[j] < 1e-10:
            continue
        if k == 0:
            g += w[j] * dx * (1.0 - 2.0 * h)
        elif k > 0:
            t = np.zeros(n)
            t[k:] += 1.0 - h[:n - k]
            t[:n - k] -= h[k:]
            g += w[j] * dx * t
        else:
            k2 = -k
            t = np.zeros(n)
            t[:n - k2] += 1.0 - h[k2:]
            t[k2:] -= h[:n - k2]
            g += w[j] * dx * t
    return g

def coord_opt(h, n, i, corr):
    hs = h[i]
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        v = (1.0 - h[i - k]) if 0 <= i - k < n else 0.0
        v -= h[i + k] if 0 <= i + k < n else 0.0
        d[n - 1 + k] = v
        v2 = (1.0 - h[i + k]) if 0 <= i + k < n else 0.0
        v2 -= h[i - k] if 0 <= i - k < n else 0.0
        d[n - 1 - k] = v2
    c0 = corr[n - 1]
    def f(t):
        v = corr + (t - hs) * d
        v[n - 1] = c0 + t * (1.0 - t) - hs * (1.0 - hs)
        return v.max()
    r = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return r.x

def cd_fast(h, n, t_end, rng, nsweep=100):
    best = h.copy()
    corr = np.correlate(best, 1.0 - best, mode='full').copy()
    bc = corr.max() * 2.0 / n
    stall = 0
    for _ in range(nsweep):
        if time.time() > t_end:
            break
        imp = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best, bc
            t = coord_opt(best, n, i, corr)
            if abs(t - best[i]) > 1e-13:
                best[i] = t
                corr += (t - best[i] + (t - t)) * 0  # will update below
                # Recompute d effect: corr changed by (t-hs)*d + central fix
                # Actually let's just recompute corr periodically for safety
                pass
        # Full recompute at end of sweep
        best = project(best, n)
        corr = np.correlate(best, 1.0 - best, mode='full').copy()
        c5 = corr.max() * 2.0 / n
        if c5 < bc - 1e-15:
            bc, imp = c5, True
        else:
            imp = False
        stall = 0 if imp else stall + 1
        if stall >= 5:
            break
    return best, bc

def cd_acc(h, n, t_end, rng, nsweep=100):
    """Accurate coordinate descent with incremental corr update."""
    dx = 2.0 / n
    best = h.copy()
    corr = np.correlate(best, 1.0 - best, mode='full').copy()
    bc = corr.max() * dx
    stall = 0
    for sw in range(nsweep):

            break
        imp = False

                v = corr + (t - hs) * d
                v[n - 1] = c0 + t * (1.0 - t) - hs * (1.0 - hs)

            t = r.x

        else:
            imp = False
        stall = 0 if imp else stall + 1
        if stall >= 5:

    return best, bc

def subgrad(h, n, t_end, T0=30, T1=8000, niter=600):
    best, bc = h.copy(), compute_c5(h)
    x = h.copy()
    lr = 0.3
    for it in range(niter):
        if time.time() > t_end:
            break
        T = T0 + (T1 - T0) * it / max(niter - 1, 1)
        g = grad_sm(x, T)
        g -= g.mean()
        for _ in range(15):
            x2 = project(x - lr * lr * g, n)
            c5 = compute_c5(x2)
            x1 = compute_c5(x)
            if c5 < x1 - 1e-12:
                x = x2
                if c5 < bc:
                    best, bc = x.copy(), c5
                lr = min(lr * 1.05, 1.5)
                break
            lr *= 0.75
    return best, bc

def slsqp(h, n, T, maxit=400):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': con}],
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return project(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def basin(h, n, t_end, rng, niter=30):
    best, bc = h.copy(), compute_c5(h)
    x, xc5 = h.copy(), compute_c5(h)
    for it in range(niter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.95 * it / max(niter, 1)
        step = 0.1 * cool + 0.01
        x2 = project(x + step * rng.standard_normal(n), n)
        x2, c5 = cd_acc(x2, n, min(time.time() + 5, t_end), rng, nsweep=6)
        temp = 0.004 * cool + 0.0003
        if c5 < xc5 or rng.random() < np.exp(-(c5 - xc5) / temp):
            x, xc5 = x2, c5
        if c5 < bc:
            best, bc = x.copy(), c5
    return best, bc

def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h_old), n_new)
