def softmax_objective(c, T):
    """Compute (1/T)*log(sum exp(T*c/dx_norm)) - actually just the numerics"""
    # c is the correlation array, T is temperature
    # We want to minimize max_d c[d] * dx, where dx = 2/N
    # So minimize max_d c[d] (the dx factor is constant)
    c_max = c.max()
    return c_max + np.log(np.sum(np.exp(T*(c - c_max)))) / T

def softmax_grad(h, N, T, topk=50):
    """Gradient of softmax objective w.r.t. h."""
    c = compute_corr(h)
    c_max = c.max()
    weights = np.exp(T * (c - c_max))
    weights /= weights.sum()
    # Only need top k lags for gradient
    active = np.where(weights > weights.max() / topk)[0]
    # grad of c[d] w.r.t. h:
    # c[N-1+dl] = sum_n h[n]*(1-h[n+dl]) for dl = d-(N-1)
    # d/dh[i] c[N-1+dl] = (1-h[i+dl])_valid - h[i-dl]_valid
    # (the -h[i]*h[i+dl] term: derivative w.r.t. h[i] gives -h[i+dl] if i+dl valid,
    #  and the h[i]*(1-h[i+dl]) term gives (1-h[i+dl]) if i+dl valid)
    # Wait: c[dl] = sum_n h[n]*(1-h[n+dl])
    # d/dh[i]: from term n=i: (1-h[i+dl]) if i+dl in [0,N-1]
    #          from term n=i-dl: h[i-dl]*(-1) if i-dl in [0,N-1] (since d/dh[n+dl] (1-h[n+dl]) = -1)
    # So: d/dh[i] c[N-1+dl] = (1-h[i+dl])_valid - h[i-dl]_valid
    
    grad = np.zeros(N)
    for d in active:
        dl = d - (N - 1)
        w = weights[d]
        # Contribution from (1-h[i+dl]): i+dl in [0,N-1] → i in [-dl, N-1-dl]
        lo1, hi1 = max(0, -dl), min(N, N - dl)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1+dl:hi1+dl])
        # Contribution from -h[i-dl]: i-dl in [0,N-1] → i in [dl, N-1+dl]
        lo2, hi2 = max(0, dl), min(N, N + dl)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2-dl:hi2-dl]
    return grad, c

def compute_corr(h):

def project_sum(h, target):

        return h
    lo, hi = -2.0, 2.0

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def build_AB(h, N):
    """Build A and B matrices for fast swap evaluation."""
    # A[p, k] where k = 0..2N-2 corresponds to lag d = k-(N-1)
    # A[p,d] = -(1-h[p+d])_valid + h[p-d]_valid
    # B[q,d] = (1-h[q+d])_valid - h[q-d]_valid
    
    idx = np.arange(2 * N - 1)
    d = idx - (N - 1)  # d ranges from -(N-1) to N-1
    
    p_idx = np.arange(N)[:, None]  # (N, 1)
    d_row = d[None, :]  # (1, 2N-1)
    
    pd = p_idx + d_row  # (N, 2N-1): p+d
    pmd = p_idx - d_row  # (N, 2N-1): p-d
    
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (pmd >= 0) & (pmd < N)
    
    h_clip = np.clip(pd, 0, N-1)
    h_clip_md = np.clip(pmd, 0, N-1)
    
    h_pd = np.where(valid_pd, h[h_clip], 0.0)
    h_pmd = np.where(valid_pmd, h[h_clip_md], 0.0)
    
    A = -(1.0 - h_pd) * valid_pd.astype(float) + h_pmd * valid_pmd.astype(float)
    B = (1.0 - h_pd) * valid_pd.astype(float) - h_pmd * valid_pmd.astype(float)
    
    return A, B

def steepest_descent_fast(h, N, max_sweeps=200, deadline=None):

    c = compute_corr(h)
    peak = float(c.max())

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_new_peak = peak
        best_pq = None
        
        for p in ones:
            base = c[None, :] + A[p, :][None, :]  # (1, 2N-1)
            candidates = base + B[zeros, :]  # (|zeros|, 2N-1)
            new_peaks = candidates.max(axis=1)
            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_new_peak:
                best_new_peak = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        
        if best_pq is None:
            break
        
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
        c = compute_corr(h)
        peak = float(c.max())
    
    return h

def soft_gradient(h, N, T=20.0, topk=30):
    """Compute gradient of soft-max of correlation."""
    c = compute_corr(h)
    c_max = c.max()
    logit = T * (c - c_max)
    logit -= logit.max()
    weights = np.exp(logit)
    weights /= weights.sum()
    
    # Only use top k lags
    if len(weights) > topk:
        top_idx = np.argpartition(weights, -topk)[-topk:]
        w = np.zeros_like(weights)
        w[top_idx] = weights[top_idx]
        w /= w.sum()
    else:
        w = weights
    
    # Gradient of c[d] w.r.t. h[i]:
    # ∂c[N-1+d]/∂h[i] = (1-h[i+d])_valid - h[i-d]_valid
    grad = np.zeros(N)
    for d in np.where(w > 0)[0]:
        dl = d - (N - 1)
        w_d = w[d]
        # (1-h[i+dl])_valid: i+dl in [0,N-1] → i in [max(0,-dl), min(N, N-dl))
        lo1, hi1 = max(0, -dl), min(N, N - dl)
        if hi1 > lo1:
            grad[lo1:hi1] += w_d * (1.0 - h[lo1+dl:hi1+dl])
        # -h[i-dl]_valid: i-dl in [0,N-1] → i in [max(0,dl), min(N, N+dl))
        lo2, hi2 = max(0, dl), min(N, N + dl)
        if hi2 > lo2:
            grad[lo2:hi2] -= w_d * h[lo2-dl:hi2-dl]
    
    return grad, c

    h = h.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_peak = float(compute_corr(h).max())
    
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    it = 0
    
    while time.time() < deadline:
        grad, c = soft_gradient(h, N, T=T, topk=k)
        peak = float(c.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        
        if it % 500 == 0:
            lr *= 0.98
    
    return best_h, best_peak * (2.0 / N)

                    cand = base + B[zeros, :] + B[zeros, :][None, :]
                    # For 2-swap we need pairs of zeros, too expensive. Skip.
                    break
        if best_move is None:
            break
        p, q = best_move[0]
        h[p] = 0.0
        h[q] = 1.0
        c = compute_corr(h)
        peak = float(c.max())
    return h

def anneal_binary(h, N, n_iters, rng, T0=4.0, T1=0.003, deadline=None):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[idx[nh:]] = 0
        else:
            idx = np.where(h == 0)[0]
            h[idx[:nh - n1]] = 1
    c = compute_corr(h)
    cur_peak = float(c.max())
    best = h.copy()
    best_peak = cur_peak
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_iters))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c = compute_corr(h)
        new_peak = float(c.max())
        delta = new_peak - cur_peak
        if delta < 0 or (T > 0.004 and rng.random() < np.exp(-delta / (T * 0.1))):
            cur_peak = new_peak
            if new_peak < best_peak:
                best_peak = new_peak
                best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best, best_peak * (2.0 / N)

def soft_grad(h, N, T=25.0, topk=40):
    c = compute_corr(h)
    c_max = c.max()
    logit = T * (c - c_max)
    logit -= logit.max()
    w = np.exp(logit)
    w /= w.sum()
    active = np.where(w > w.max() / topk * 0.1)[0]
    wa = w[active]
    wa /= wa.sum()
    grad = np.zeros(N)
    center = N - 1
    for d in active:
        dl = d - center
        wd = wa[active == d][0] if len(active) > 1 else wa[0]
        lo1, hi1 = max(0, -dl), min(N, N - dl)
        if hi1 > lo1:
            grad[lo1:hi1] += w[d] * (1.0 - h[lo1 + dl:hi1 + dl])
        lo2, hi2 = max(0, dl), min(N, N + dl)
        if hi2 > lo2:
            grad[lo2:hi2] -= w[d] * h[lo2 - dl:hi2 - dl]
    return grad

def continuous_opt(h, N, target, deadline, lr0=0.04, T=30.0, topk=40):
    h = h.astype(float).copy()
    h = project_sum(h, target)
    best_h = h.copy()
    best_peak = float(compute_corr(h).max())
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    it = 0
    while time.time() < deadline:
        grad = soft_grad(h, N, T=T, topk=topk)
        c = compute_corr(h)
        peak = float(c.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = project_sum(h_new, target)
        if np.all(np.isfinite(h_new)):
            h = h_new
        if it % 300 == 0:
            lr *= 0.985
    return best_h, best_peak * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)
    N0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    N = N0
    nh = N // 2
    # Phase 1: Binarize + steepest descent
    h_bin = (h0 > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 > nh:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[:nh - n1]] = 1
    d1 = deadline - 700
    h_s = steepest_swap(h_bin, N, max_sweeps=400, deadline=d1)
    c5_s = compute_c5(h_s, N)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    # Phase 2: Iterated local search (anneal + steepest)
    d2 = deadline - 250
    h_cur = best_h.copy()
    for restart in range(40):
        if time.time() > d2 - 5:
            break
        rem = d2 - time.time() - 10
        if rem < 15:
            break
        n_pert = rng.randint(1, 7)
        h_p = h_cur.copy()
        ones = np.where(h_p == 1)[0]
        zeros = np.where(h_p == 0)[0]
        for _ in range(n_pert):
            if len(ones) < 1 or len(zeros) < 1:
                break
            i, j = int(rng.choice(ones)), int(rng.choice(zeros))
            h_p[i], h_p[j] = h_p[j], h_p[i]
            ones = np.where(h_p == 1)[0]
            zeros = np.where(h_p == 0)[0]
        h_a, c5_a = anneal_binary(h_p, N, int(min(rem * 0.35, 3000)), rng, T0=3.0, T1=0.004, deadline=d2 - 30)
        h_sd = steepest_swap(h_a, N, max_sweeps=300, deadline=d2 - 5)
        c5_sd = compute_c5(h_sd, N)
        if c5_sd < best_c5:
            best_c5, best_h = c5_sd, h_sd.copy()
            h_cur = h_sd.copy()
    # Phase 3: Continuous soft-max optimization
    if time.time() < deadline - 80:
        h_c = best_h.astype(float) + rng.randn(N) * 0.06
        h_c = project_sum(h_c, nh)
        d3 = deadline - 50
        h_co, c5_co = continuous_opt(h_c, N, nh, d3, lr0=0.035, T=25.0, topk=40)
        if c5_co < best_c5:
            best_c5, best_h = c5_co, h_co.copy()
    # Phase 4: Different N values
    for N_try in [N0 - 20, N0 + 20, N0 - 40, N0 + 40, N0 - 60, N0 + 60]:
        if time.time() > deadline - 50 or N_try < 100:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = (hi > 0.5).astype(float)
        n1t = int(hi.sum())
        nht = N_try // 2
        if n1t > nht:
            idx = np.where(hi == 1)[0]
            hi[idx[nht:]] = 0
        elif n1t < nht:
            idx = np.where(hi == 0)[0]
            hi[idx[:nht - n1t]] = 1
        h_sd = steepest_swap(hi, N_try, max_sweeps=200, deadline=deadline - 25)
        c5_sd = compute_c5(h_sd, N_try)
        if c5_sd < best_c5:
            best_c5, best_h = c5_sd, h_sd.copy()
        if time.time() < deadline - 15:
            hi2 = h_sd.astype(float) + rng.randn(N_try) * 0.04
            hi2 = project_sum(hi2, N_try / 2.0)
            h_co2, c5_co2 = continuous_opt(hi2, N_try, N_try / 2.0, deadline - 10, lr0=0.025, T=30.0, topk=30)
            if c5_co2 < best_c5:
                best_c5, best_h = c5_co2, h_co2.copy()
    # Phase 5: Final polish
    if time.time() < deadline - 4:
        hF = best_h.astype(float) + rng.randn(len(best_h)) * 0.02
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = continuous_opt(hF, len(best_h), len(best_h) / 2.0, deadline - 1, lr0=0.012, T=50.0, topk=50)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
