import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_fast(h, n, i, dx):
    """Optimize h[i] while keeping others fixed. O(n) computation."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range  # where h_base's i-th element contributes to corr_base at lag k
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n - 1)])
    # C5(h_i) = max(corr_base + h_i * coefs) * dx
    # This is a piecewise linear function of h_i, minimized at a "kink point"
    # The kink points are where corr_base + h_i * coefs changes which index is max
    # For bounded h_i in [0,1], just do minimization
    def f1d(hi):
        return np.max(corr_base + hi * coefs)
    # Use bounded scalar minimization
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        order = np.random.permutation(n)
        for i in order:
            hi_new = coord_opt_fast(best_h, n, i, dx)
            if abs(hi_new - best_h[i]) > 1e-10:
                h_try = best_h.copy()
                h_try[i] = hi_new
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - tol:

                    best_h = h_try
                    improved += 1
        if improved == 0:
            no_improve += 1
            if no_improve >= 3:
                break
        else:

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_fast(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n - 1)])
    def f1d(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_fast(best_h, n, i)
            if abs(hi_new - best_h[i]) > 1e-11:
                h_try = best_h.copy()
                h_try[i] = hi_new
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved += 1
        if improved == 0:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
    return best_h, best_c5

def subgrad_step(h, alpha, n_topk=4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    grad = np.zeros(n)
    idx = np.arange(n)
    for m in topk_idx:
        k = m - (n - 1)
        jk = idx - k
        jpk = idx + k
        v1 = (jk >= 0) & (jk < n)
        v2 = (jpk >= 0) & (jpk < n)
        grad += (1.0 - h[jk[v1]]).astype(np.float64) * v1
        grad -= h[jpk[v2]].astype(np.float64) * v2
    grad = grad / n_topk * dx
    h_new = h - alpha * grad
    return project_h(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing smoothness
    for T in [50, 300, 1500, 8000, 30000]:
        if time.time() > deadline - budget_s * 0.15:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=1200, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Intensive coordinate descent (main workhorse)
    best_h, best_c5 = coord_descent(best_h, n, deadline - budget_s * 0.35, max_sweeps=2000)

    # Phase 3: Subgradient + SLSQP refinement cycle
    for it in range(25):
        if time.time() > deadline - budget_s * 0.25:
            break
        cool = 1.0 - it / 25.0
        alpha = 0.003 + 0.005 * cool
        ntk = 3 if cool > 0.5 else 5
        hs = subgrad_step(best_h, alpha, ntk)
        T_s = 3000 + 10000 * cool
        hc = slsqp_opt(hs, T_s, n, maxiter=500, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        # Short coordinate descent after each subgrad step
        if time.time() < deadline - budget_s * 0.20:
            best_h, best_c5 = coord_descent(best_h, n, min(time.time() + 3, deadline - budget_s * 0.18), max_sweeps=30)

    # Phase 4: Basin hopping with diverse perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    n_trials = 50
    for it in range(n_trials):
        if time.time() > deadline - budget_s * 0.12:
            break
        cool = 1.0 - 0.9 * (it / n_trials)
        r = np.random.rand()
        if r < 0.25:
            pert = (0.008 + 0.04 * cool) * np.random.randn(n)
        elif r < 0.50:
            n_pert = max(4, int(n * 0.08 * (0.3 + cool)))
            idx_p = np.random.choice(n, n_pert, replace=False)
            pert = np.zeros(n)
            pert[idx_p] = (0.04 + 0.12 * cool) * np.random.randn(n_pert)
        elif r < 0.70:
            freq = np.random.randint(1, 6)
            phase = np.random.uniform(0, 2 * np.pi)
            pert = (0.015 + 0.06 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        elif r < 0.85:
            # Target perturbation: push near the worst lag region
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            worst_lag = int(np.argmax(corr)) - (n - 1)
            center = n // 2 - worst_lag // 2
            sigma = n * 0.15
            mask = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
            pert = mask * (0.03 + 0.08 * cool) * np.random.randn(n)
        else:
            block = max(5, n // 10)
            start = np.random.randint(0, max(1, n - block))
            pert = np.zeros(n)
            pert[start:start + block] = (0.04 + 0.1 * cool) * np.random.randn(block)

        h_pert = project_h(work_h + pert)
        T_bh = 2000 + 8000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
            # Quick coord descent after accepting
            if time.time() < deadline - budget_s * 0.10:
                work_h, work_c5 = coord_descent(work_h, n, min(time.time() + 1.5, deadline - budget_s * 0.08), max_sweeps=15)
                if work_c5 < best_c5:
                    best_c5 = work_c5
                    best_h = work_h

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Final heavy coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, deadline - 3, max_sweeps=1000)

    # Phase 6: Ultra-high-T SLSQP final polish
    if time.time() < deadline - 1:
        for T in [50000, 200000]:
            if time.time() > deadline - 1:
                break
            hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Final coordinate descent
    if time.time() < deadline - 0.5:
        best_h, best_c5 = coord_descent(best_h, n, deadline - 0.1, max_sweeps=200)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
