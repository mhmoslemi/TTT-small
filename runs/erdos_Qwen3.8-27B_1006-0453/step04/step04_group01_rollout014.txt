                    if 0 <= j - s < n:
                        v -= best_h[j - s]
                    d[k] = v
            c0 = corr[n - 1]
            oj = old_j
            oj2 = oj * oj
            def f1d(t):
                vals = corr + (t - oj) * d
                vals[n - 1] = c0 - oj + oj2 + t - t * t
                return np.max(vals)
            try:
                res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-14})
                tb = res.x
            except Exception:
                tb = oj
            chg = abs(tb - oj)
            if chg > 1e-14:
                best_h[j] = tb
                max_change = max(max_change, chg)
        best_h = project(best_h, n)
        if max_change < 1e-12:
            break
    return best_h, compute_c5(best_h)

def pair_swap_cd(h, n, t_end, max_trials=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_trials):
        if time.time() > t_end:
            break
        i, j = np.random.choice(n, 2, replace=False)
        h_try = best_h.copy()
        h_try[i], h_try[j] = best_h[j], best_h[i]
        h_try = project(h_try, n)
        h_try, c5_try = fast_cd(h_try, n, min(t_end, time.time() + budget_remaining()), max_sweeps=200, order='fwd')
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global budget_remaining
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def budget_remaining():
        return max(0, (1 - elapsed() / budget_s)) * 3.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T (8% time)
    for T in [30, 100, 500, 2000, 8000]:
        if elapsed() > budget_s * 0.08:
            break
        hc = slsqp_quick(best_h.copy(), T, n, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Intensive coordinate descent - forward (40% time)
    phase2_end = t0 + budget_s * 0.48
    best_h, best_c5 = fast_cd(best_h, n, phase2_end, max_sweeps=100000, order='fwd')

    # Phase 3: CD in different order to break symmetry (10%)
    phase3_end = t0 + budget_s * 0.58
    best_h, best_c5 = fast_cd(best_h, n, phase3_end, max_sweeps=50000, order='rev')
    best_h, best_c5 = fast_cd(best_h, n, phase3_end + budget_s * 0.05, max_sweeps=20000, order='rnd')

    # Phase 4: Diversification via perturbation (20%)
    phase4_end = t0 + budget_s * 0.78
    for it in range(30):
        if elapsed() > phase4_end:
            break
        cool = 1.0 - it / 30.0
        scale = 0.03 + 0.12 * cool
        h_pert = best_h + scale * np.random.randn(n)
        h_pert = project(h_pert, n)
        # Quick SLSQP
        T_s = 500 + 5000 * cool
        h_opt = slsqp_quick(h_pert, T_s, n, maxiter=150)
        # CD polish
        sub_end = min(phase4_end, time.time() + budget_s * 0.008)
        h_opt, c5_opt = fast_cd(h_opt, n, sub_end, max_sweeps=3000, order='fwd')
        # Accept if better or via simulated annealing
        temp = 0.0005 * cool + 1e-5
        if c5_opt < best_c5 or (c5_opt > best_c5 and np.random.rand() < np.exp(-(c5_opt - best_c5) / max(temp, 1e-9))):
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 5: Pair swap diversification (5%)
    phase5_end = t0 + budget_s * 0.83
    best_h, best_c5 = pair_swap_cd(best_h, n, phase5_end, max_trials=100)

    # Phase 6: Final intensive CD polish (17%)
    final_end = t0 + budget_s * 0.99
    for order in ['fwd', 'rev', 'rnd']:
        if elapsed() > final_end:
            break
        sub = min(final_end, time.time() + budget_s * 0.05)
        best_h, best_c5 = fast_cd(best_h, n, sub, max_sweeps=50000, order=order)

    # Final SLSQP polish
    if elapsed() < budget_s * 0.98:
        for T in [10000, 50000]:
            if elapsed() > budget_s * 0.99:
                break
            hc = slsqp_quick(best_h.copy(), T, n, maxiter=800)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
