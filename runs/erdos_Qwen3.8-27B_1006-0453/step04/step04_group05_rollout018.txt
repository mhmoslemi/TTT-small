import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_w(c, T):
    cs = c - c.max()
    e = np.exp(cs / T)
    return e / e.sum()

def compute_D(h, N):
    L = np.arange(2 * N - 1) - (N - 1)
    D = np.zeros((N, 2 * N - 1))
    for p in range(N):
        pf = np.clip(p + L, 0, N - 1)
        ps = np.clip(p - L, 0, N - 1)
        vf = ((p + L) >= 0).astype(float) * ((p + L) < N).astype(float)
        vs = ((p - L) >= 0).astype(float) * ((p - L) < N).astype(float)
        D[p] = (1.0 - h[pf]) * vf - h[ps] * vs
    return D

def compute_grad(h, w, N):
    center = N - 1
    s = np.zeros(2 * N - 1)
    s[center] = 2.0 * w[center]
    for L in range(1, N):
        v = w[center + L] + w[center - L]
        s[center + L] = v
        s[center - L] = v
    C = np.full(N, w[center])
    pp = np.zeros(N)
    np = np.zeros(N)
    for k in range(1, N):
        pp[k] = pp[k-1] + w[center + k]
        np[k] = np[k-1] + w[center - k]
    for j in range(N):
        if N - 1 - j > 0:
            C[j] += pp[N - 1 - j]
        if j > 0:
            C[j] += np[j]
    Sh = np.zeros(N)
    for L in range(-(N-1), N):
        sL = s[center + L]
        if abs(sL) < 1e-16:
            continue
        if L >= 0:
            Sh[:N-L] += sL * h[L:]
        else:
            Sh[-L:] += sL * h[:-L]
    return C - Sh

def fista_run(h, N, deadline, T0=0.4, T1=0.008):
    target = N / 2.0
    h = project_sum(h.astype(float).copy(), target)
    best_h = h.copy()
    best_val = float(compute_corr(h).max())
    h_y = h.copy()
    t_prev = 1.0
    lr = 0.005
    n_it = 0
    while time.time() < deadline:
        n_it += 1
        T = max(T1, T0 * np.exp(-n_it / 400.0))
        corr_y = compute_corr(h_y)
        w = softmax_w(corr_y, T)
        grad = compute_grad(h_y, w, N)
        h_new = h_y - lr * grad
        h_new = project_sum(h_new, target)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        h_y = h_new + ((t_prev - 1.0) / t_new) * (h_new - h)
        h_y = project_sum(h_y, target)
        h = h_new
        t_prev = t_new
        if n_it % 30 == 0:
            val = float(compute_corr(h).max())
            if val < best_val:
                best_val = val
                best_h = h.copy()
        if n_it % 500 == 0:
            lr *= 0.7
    val = float(compute_corr(h).max())
    if val < best_val:
        best_h = h.copy()
        best_val = val
    return best_h, best_val * (2.0 / N)

def binary_search(h, N, deadline, max_iters=5000):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.RandomState(1).shuffle(idx)
        h[idx[:n1-nh]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh-n1]] = 1
    D = compute_D(h, N)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_pq = None
        best_peak = P
        for p in ones:
            base = corr - D[p]
            cands = base[None, :] + D[zeros]
            npeaks = cands.max(axis=1)
            mi = int(np.argmin(npeaks))
            if npeaks[mi] < best_peak:
                best_peak = float(npeaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        p, q = best_pq
        h[p] = 0.0
        h[q] = 1.0
        D = compute_D(h, N)
    return h

def two_for_two(h, N, deadline, n_tries=3000, seed=0):
    h = h.astype(float).copy()
    rng = np.random.RandomState(seed)
    D = compute_D(h, N)
    corr = compute_corr(h)
    P = float(corr.max())
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_tries):
        if time.time() > deadline:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        p1, p2 = rng.choice(ones, 2, replace=False)
        q1, q2 = rng.choice(zeros, 2, replace=False)
        delta = -D[p1] - D[p2] + D[q1] + D[q2]
        new_P = float((corr + delta).max())
        if new_P < P:
            h[p1] = 0; h[p2] = 0
            h[q1] = 1; h[q2] = 1
            P = new_P
            corr = corr + delta
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            D = compute_D(h, N)
    return h

def anneal_binary(h, N, n_iters, rng, T0=2.0, T1=0.1):
    h = h.astype(float).copy()
    corr = compute_corr(h)
    cur_p = float(corr.max())
    best_h = h.copy()
    best_p = cur_p
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_corr = compute_corr(h)
        new_p = float(new_corr.max())
        d = new_p - cur_p
        if d < 0 or (T > 0.05 and rng.random() < np.exp(-d / T)):
            cur_p = new_p
            corr = new_corr
            if new_p < best_p:
                best_p = new_p
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_p * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N)

    # Phase 1: Binary steepest descent from initial
    h_bin = (h0 > 0.5).astype(float)
    d1 = min(deadline - 300, time.time() + 120)
    h_s = binary_search(h_bin, N, d1, max_iters=3000)
    c5_s = compute_c5(h_s, N)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: 2-for-2 swaps
    d2 = min(deadline - 250, time.time() + 100)
    h_22 = two_for_two(best_h, N, d2, n_tries=5000, seed=seed+1)
    c5_22 = compute_c5(h_22, N)
    if c5_22 < best_c5:
        best_c5, best_h = c5_22, h_22.copy()

    # Phase 3: Anneal + binary search iterations
    d3 = deadline - 120
    h_cur = best_h.copy()
    for restart in range(15):
        if time.time() > d3:
            break
        remaining = d3 - time.time()
        if remaining < 20:
            break
        n_ann = int(min(remaining * 0.3, 8.0) * 120)
        h_a, c5_a = anneal_binary(h_cur, N, n_ann, rng, T0=2.0, T1=0.08)
        d_sd = d3 - 10
        h_s2 = binary_search(h_a, N, d_sd, max_iters=2000)
        c5_s2 = compute_c5(h_s2, N)
        if c5_s2 < best_c5:

            h_cur = h_s2.copy()
        # Also try 2-for-2 on the new result
