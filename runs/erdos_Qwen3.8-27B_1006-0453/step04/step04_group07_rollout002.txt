def block_perturb(h, n, rng):
    bh = h.copy()
    length = rng.randint(4, 25)
    start = rng.randint(0, n - length)
    mode = rng.randint(3)
    if mode == 0:
        bh[start:start+length] = 1.0 - bh[start:start+length]
    elif mode == 1:
        shift = rng.randint(-length, length + 1)
        if shift != 0:
            tmp = bh[start:start+length].copy()
            end = start + length
            if shift > 0:
                if end + shift <= n:
                    s2, e2 = start + shift, start + shift + length
                    bh[start:e2] = np.concatenate([bh[start+s2:end], tmp])
                else:
                    bh[:] = h[:]
            else:
                s2, e2 = start + shift, start
                bh[s2:e2+length] = np.concatenate([tmp, bh[s2:start]])
    else:
        noise = rng.normal(0, 0.15, length)
        bh[start:start+length] = np.clip(bh[start:start+length] + noise, 0, 1)
    return project(bh, n)

def basin_hop(base_h, base_c5, n, t_end, max_iters=500):

    rng = np.random.RandomState(int(time.time() * 1000) % 100000)

        if c5v < best_c5:
            best_c5, best_h = c5v, hc.copy()
            base_h = hc.copy()
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def make_random_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = rng.randint(2, 12)
        val = rng.choice([0.0, 1.0], p=[0.4, 0.6])
        end = min(pos + blen, n)
        h[pos:end] = val
        pos = end
    return project(h, n)

def make_symmetric_start(n, seed):
    rng = np.random.RandomState(seed)
    half = n // 2
    h_half = rng.choice([0.0, 1.0], size=half, p=[0.4, 0.6])
    h = np.zeros(n)
    h[:half] = h_half
    h[half:] = h_half[::-1]
    return project(h, n)
