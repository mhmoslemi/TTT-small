import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    d = target - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-14:
            return h
        h = h + r * (d / t)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp_obj(x, T, n):
    dx = 2.0 / n
    x = np.clip(x, 0.0, 1.0)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx

def slsqp(h, T, maxiter, ftol):
    n = len(h)
    res = minimize(slsqp_obj, h, args=(T, n), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': sum_eq}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)

def coord_descent(h, t_end, max_sweeps):
    n = len(h)
    dx = 2.0 / n
    bh = project_to_sum(h, n / 2.0).astype(float).copy()
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    best = float(corr.max())
    ks = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return bh, best * dx
            ho = bh[i]
            cf = np.zeros(2 * n - 1)
            j1 = i + ks
            m1 = (j1 >= 0) & (j1 < n)
            j2 = i - ks
            m2 = (j2 >= 0) & (j2 < n)
            if m1.any():
                cf[m1] += 1.0 - bh[j1[m1]]
            if m2.any():
                cf[m2] -= bh[j2[m2]]
            lo, hi = -ho, 1.0 - ho
            if hi - lo < 1e-13:
                continue
            a, b = lo, hi
            for _ in range(20):
                u = a + (b - a) * 0.382
                v = a + (b - a) * 0.618
                if np.max(corr + u * cf) < np.max(corr + v * cf):
                    b = v
                else:
                    a = u
            sn = 0.5 * (a + b)
            if abs(sn) > 1e-10:
                bh[i] = ho + sn
                corr += sn * cf
                nb = corr.max()
                if nb < best:
                    best = float(nb)
                    imp = True
        bh = project_to_sum(bh, n / 2.0)
        corr = np.correlate(bh, 1.0 - bh, mode='full')
        best = float(corr.max())
        if not imp:
            break
    return bh, best * dx

def sinusoid(n, fr, ph):
    x = np.arange(n, dtype=float)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * fr * x / n + ph)
