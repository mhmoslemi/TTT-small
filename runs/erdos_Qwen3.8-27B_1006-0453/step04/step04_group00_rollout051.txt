    dx = 2.0 / n

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            if h.sum() < 1e-14:
                break
            h = h * (1.0 + diff / h.sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_opt_single(h, i, n):
    h_base = h.copy()
    h_base[i] = 0.0

    s_off = n - 1

    valid1 = (j1 >= 0) & (j1 < n)

    valid2 = (j2 >= 0) & (j2 < n)

        grad[idx] += d1 + d2
