            return h

            hc = slsqp_smooth(h_i, T, n_try, maxiter=350, ftol=1e-13)
            cd_end = min(time.time() + 4.0, t3 - 1.0)
            hc, c5v = cd(hc, n_try, cd_end, max_sweeps=40)
            if c5v < best_c5:
                best_h, best_c5, n = hc, c5v, n_try

    # Phase 4: Second short basin hop from best (10% budget)
    t4 = t0 + budget_s * 0.88
    work_h2 = best_h.copy()
    bh2 = 0
    while time.time() < t4:
        bh2 += 1
        cool = max(1.0 - 0.9 * (bh2 / 80.0), 0.05)
        h_p = gen_perturbation(work_h2, n, rng, cool)
        h_p = project_to_sum(h_p, n / 2)
        hc = slsqp_smooth(h_p, max(100, int(500 * cool)), n, maxiter=200, ftol=1e-12)
        hc, c5v = cd(hc, n, min(time.time() + 2.0, t4 - 0.3), max_sweeps=20, random_order=True)
        if c5v < work_h2 is not None and c5v < best_c5:
            best_h, best_c5 = hc.copy(), c5v
        if c5v < compute_c5(work_h2):
            work_h2 = hc

    # Phase 5: Final aggressive refinement (12% budget)
    t5 = t0 + budget_s * 0.985
    best_h, best_c5 = cd(best_h, n, t5, max_sweeps=250, random_order=False)
    for T in [10000, 50000, 200000]:
        if time.time() > t5:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=1000, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v
    best_h, best_c5 = cd(best_h, n, t0 + budget_s * 0.995, max_sweeps=25)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
