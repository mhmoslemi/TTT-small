import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = np.clip(h * (target / max(s, 1e-14)), 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
    return np.clip(h, 0.0, 1.0)

def coord_opt_fixed(h, n, i):
    h[i] = 0.0
    corr_zero = np.correlate(h, 1.0 - h, mode='full').astype(float)
    h[i] = 1.0
    corr_one = np.correlate(h, 1.0 - h, mode='full').astype(float)
    h[i] = min(1.0, max(0.0, h[i]))
    h[i] = 0.5  # temporary
    h_save = 0.5
    corr_diff = corr_one - corr_zero
    dx = 2.0 / n
    def f1d(t):
        vals = corr_zero + t * corr_diff
        vals[n - 1] += t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return res.x

def slsqp_opt(h, T, n, maxiter=600):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project(np.clip(res.x, 0, 1), n)

        improved = False
        if priority_mode and sweep % 10 == 0:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            m = np.max(corr)
            weights = np.zeros(n)
            for i in range(n):

        else:
            order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt_fixed(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-13:
                continue
            best_h[i] = hi
            best_h = project(best_h, n)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def minimax_shake(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(300):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = np.max(corr)
        thresh = m - 0.02 * m
        hot_lags = np.where(corr > thresh)[0]
        if len(hot_lags) == 0:
            break
        lag = hot_lags[rng.integers(len(hot_lags))]
        k = lag - (n - 1)
        if k >= 0:
            valid = np.arange(0, n - k)
        else:
            valid = np.arange(-k, n)
        if len(valid) == 0:
            continue
        contrib = np.zeros(n)
        for i in valid:
            j = i + k
            if 0 <= j < n:
                contrib[i] = best_h[i] * (1.0 - best_h[j])
        top_i = np.argmax(contrib)
        if contrib[top_i] < 1e-6:
            continue
        old_val = best_h[top_i]
        for delta in [0.05, -0.05, 0.02, -0.02, 0.08, -0.08]:
            test = best_h.copy()
            test[top_i] = min(1.0, max(0.0, old_val + delta))
            test = project(test, n)
            cv = compute_c5(test)
            if cv < best_c5 - 1e-14:
                best_h, best_c5 = test, cv
                break
        if cv < best_c5:
            best_h, best_c5 = minimax_shake_local(best_h, n, t_end, rng, top_i)
    return best_h, best_c5

def minimax_shake_local(h, n, t_end, rng, center):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lo = max(0, center - 5)
    hi = min(n, center + 6)
    for _ in range(30):
        if time.time() > t_end:
            break
        for i in range(lo, hi):
            new_val = coord_opt_fixed(best_h, n, i)
            if abs(new_val - best_h[i]) > 1e-12:
                old_c = best_c5
                best_h[i] = new_val
                best_h = project(best_h, n)
                best_c5 = compute_c5(best_h)
                if best_c5 > old_c + 1e-10:
                    best_h[i] = coord_opt_fixed(best_h, n, i)
                    best_h = project(best_h, n)
                    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def pair_descent(h, n, t_end, rng, n_pairs=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        for scale in [0.03, 0.01, 0.06]:
            hi = best_h[i] + rng.choice([-1.0, 1.0]) * scale
            hj = best_h[j] + rng.choice([-1.0, 1.0]) * scale
            if 0 <= hi <= 1 and 0 <= hj <= 1:
                test = best_h.copy()
                test[i], test[j] = hi, hj
                test = project(test, n)
                cv = compute_c5(test)
                if cv < best_c5 - 1e-14:
                    best_h, best_c5 = test, cv
                    break
    return best_h, best_c5

def basin_hop(h, n, t_end, rng, n_iters=60, base_step=0.04):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(n_iters, 1))
        step = base_step * (0.2 + 0.8 * cool)
        pert = work + step * rng.standard_normal(n)
        pert = project(pert, n)
        hc, cv = coord_descent(pert, n, min(time.time() + 8, t_end - 5), max_sweeps=15, rng=rng)
        temp = 0.003 * cool + 0.0002
        if cv < work_c5 or rng.random() < np.exp(-min((cv - work_c5) / temp, 40)):
            work, work_c5 = hc, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish (12%)
    t1 = t0 + budget_s * 0.12
    for T in [100, 500, 2000, 10000, 50000]:
        if time.time() > t1:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=500)
            cv = compute_c5(hc)
            if cv < best_c5 - 1e-15:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 2: Priority coordinate descent (25%)
    t2 = t0 + budget_s * 0.37
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=300, rng=rng, priority_mode=True)

    # Phase 3: Minimax shake (20%)
    t3 = t0 + budget_s * 0.57
    best_h, best_c5 = minimax_shake(best_h, n, t3, rng)

    # Phase 4: Pairwise descent (13%)
    t4 = t0 + budget_s * 0.70
    best_h, best_c5 = pair_descent(best_h, n, t4, rng, n_pairs=5000)

    # Phase 5: Basinhopping with CD local (20%)
    t5 = t0 + budget_s * 0.88
    best_h, best_c5 = basin_hop(best_h, n, t5, rng, n_iters=50, base_step=0.035)

    # Phase 6: Final CD polish (7%)
    t6 = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=200, rng=rng, priority_mode=True)

    # Phase 7: Final SLSQP at very high T
    if time.time() < t0 + budget_s * 0.995:
        for T in [200000, 1000000]:
            if time.time() > t0 + budget_s * 0.995:
                break
            try:
                hc = slsqp_opt(best_h, T, n, maxiter=300)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5 = hc, cv
            except Exception:
                pass
        if time.time() < t0 + budget_s * 0.999:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.999, max_sweeps=30, rng=rng)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
