```python
def minmax_1d(a, b, lo, hi):
    # Minimize max_j (a[j] + b[j]*x) over x in [lo, hi]
    # a = corr, b = coef, shape (2n-1,)
    # Returns optimal x and f(x)
    # Breakpoints: x_j = -a[j]/b[j] where b[j] != 0
    nz = b != 0
    if not np.any(nz):
        # Constant function
        return (lo+hi)/2, np.max(a)
    xbr = -a[nz] / b[nz]
    xbr = np.clip(xbr, lo, hi)
    xbr = np.unique(xbr)  # remove duplicates
    # Candidates: breakpoints + endpoints
    candidates = np.concatenate([[lo, hi], xbr])
    # Evaluate f at each candidate
    vals = a[None, :] + candidates[:, None] * b[None, :]
    obj = np.max(vals, axis=1)
    idx = np.argmin(obj)
    return candidates[idx], obj[idx]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def minmax_1d(corr, coef, lo, hi):
    """Minimize max_k(corr[k]+delta*coef[k]) for delta in [lo,hi] via bisection."""
    if hi - lo < 1e-14:
        return lo
    iters = min(50, max(25, int(np.ceil(np.log2((hi - lo) / 1e-13)))))
    for _ in range(iters):
        mid = 0.5 * (lo + hi)
        vals = corr + mid * coef
        imax = int(np.argmax(vals))
        if coef[imax] > 1e-16:
            hi = mid
        elif coef[imax] < -1e-16:
            lo = mid
        else:
            break
    # Final check at a few points
    cands = np.array([lo, hi, 0.5*(lo+hi)])
    vals = corr[None, :] + cands[:, None] * coef[None, :]
    obj = np.max(vals, axis=1)
    idx = int(np.argmin(obj))
    return cands[idx]

def coord_descent(h, t_end, max_sweeps=5000, verbose=False):
    """Fast coordinate descent for the Erdős overlap problem."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = compute_corr(best_h)
    m_range = np.arange(-(n - 1), n)
    sweep = 0
    last_best = best_c5
    stall = 0
    while sweep < max_sweeps:
        t_check = time.time()
        if t_check > t_end:
            break
        sweep += 1
        any_imp = False
        for i in range(n):
            h_old = best_h[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-15:
                continue
            # Compute coefficient vector
            valid1 = (i + m_range >= 0) & (i + m_range < n)
            valid2 = (i - m_range >= 0) & (i - m_range < n)
            coef = np.zeros(2 * n - 1)
            coef[valid1] += (1.0 - best_h[i + m_range[valid1]])
            coef[valid2] -= best_h[i - m_range[valid2]]
            # 1D minmax
            delta = minmax_1d(corr, coef, lo, hi)
            if abs(delta) > 1e-14:
                new_c5_val = np.max(corr + delta * coef) * dx
                if new_c5_val < best_c5 - 1e-16:
                    best_h[i] = h_old + delta
                    corr += delta * coef
                    best_c5 = new_c5_val
                    any_imp = True
        if not any_imp:
            break
        if best_c5 < last_best - 1e-9:
            last_best = best_c5
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break
    return best_h, best_c5

def smooth_max(vals, T):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def slsqp_polish(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(np.correlate(x, 1.0 - x, mode='full'), T) * dx
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        weights = np.exp(T * (corr - m))
        weights /= weights.sum()
        n_pts = len(x)
        m_range = np.arange(-(n_pts-1), n_pts)
        grad = np.zeros(n_pts)
        for i in range(n_pts):
            valid1 = (i + m_range >= 0) & (i + m_range < n_pts)
            valid2 = (i - m_range >= 0) & (i - m_range < n_pts)
            c1 = (1.0 - x[i + m_range[valid1]])
            c2 = x[i - m_range[valid2]]
            coef_i = np.zeros(2*n_pts-1)
            coef_i[valid1] += c1
            coef_i[valid2] -= c2
            grad[i] = np.dot(weights, coef_i) * dx
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=[(0, 1)] * n, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def block_init(n, nblocks):
    """Create a piecewise-constant block initial condition."""
    h = np.zeros(n)
    block_size = n // nblocks
    for j in range(nblocks):
        start = j * block_size
        end = start + block_size if j < nblocks - 1 else n
        h[start:end] = 0.5 + 0.45 * np.cos(2.0 * np.pi * j / nblocks)
    return project_to_sum(h, n / 2.0)

def perturb_h(h, rng, style):
    """Apply a perturbation to h based on style (int)."""
    n = len(h)
    if style == 0:
        return h + 0.05 * rng.standard_normal(n)
    elif style == 1:
        # Random block shift
        seg = rng.integers(0, n // 3)
        start = rng.integers(0, n - seg)
        h2 = h.copy()
        h2[start:start+seg] = h[start+seg:start+2*seg] if start+2*seg <= n else h[start+seg:]
        return h2
    elif style == 2:
        # Sinusoidal modulation
        f = rng.uniform(2.0, 8.0) / n * 2.0
        amp = 0.15
        x = np.arange(n)
        return h + amp * np.sin(2.0 * np.pi * f * x / 2.0) * rng.uniform(0.3, 1.0)
    elif style == 3:
        # Random swap of values
        idx1 = rng.integers(0, n)
        idx2 = rng.integers(0, n)
        h2 = h.copy()
        h2[idx1], h2[idx2] = h[idx2], h2[idx1]
        return h2
    else:
        # Local Gaussian bump
        center = rng.integers(n // 4, 3 * n // 4)
        width = rng.integers(n // 16, n // 6)
        bump = 0.1 * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        return h + bump * rng.uniform(-1, 1)

                    best_c5, best_h = c5v, hc.copy()
        except Exception:
            pass

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 4: Multi-resolution search
    t_p4 = t0 + budget_s * 0.75
    for n_try in [128, 160, 192, 256, 288]:
        if elapsed() > t_p4:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2)
        hc, c5v = coord_descent(h_i, time.time() + budget_s * 0.03, max_sweeps=500)
        if c5v < best_c5:
            best_c5, best_h, n = c5v, hc, n_try

    # Phase 5: Heavy coordinate descent on best
    t_p5 = t0 + budget_s * 0.93
    best_h, best_c5 = coord_descent(best_h, t_p5, max_sweeps=10000)

    # Phase 6: Final SLSQP polish with very high T
    for T in [10000, 50000]:
        if elapsed() > t0 + budget_s * 0.97:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=800, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Final tiny coord pass
    best_h, best_c5 = coord_descent(best_h, t0 + budget_s * 0.99, max_sweeps=200)
    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
