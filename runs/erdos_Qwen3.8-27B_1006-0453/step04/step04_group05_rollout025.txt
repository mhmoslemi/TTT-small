```python
def adam_polish(h, N, deadline, k=20, lr0=0.01, decay=0.995):
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    center = N - 1
    
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    
    while time.time() < deadline:
        corr = compute_corr(h)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        # Top-k lags
        topk = np.argpartition(corr, -k)[-k:]
        
        # Compute gradient as weighted average of gradients at top-k lags
        grad = np.zeros(N)
        for ai in topk:
            L = ai - center
            g = grad_at_lag(h, L, N)
            grad += g
        grad /= k
        
        # Adam update
        t += 1
        m = b1 * m + (1-b1) * grad
        v = b2 * v + (1-b2) * grad**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        
        if t % 100 == 0:
            lr *= decay
    
    return best_h, best_peak * dx
```

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    idx = np.arange(2 * N - 1)
    d_lags = idx - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)

    B_part = h_pmd * valid_pmd.astype(float)

def grad_at_lag(h, L, N):
    g = np.zeros(N)
    if L >= 0:
        if L == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            n = N - L
            g[:n] += (1.0 - h[L:N])
            g[L:] -= h[:n]
    else:
        M = -L
        n = N - M
        g[M:] += (1.0 - h[:n])
        g[:n] -= h[M:N]
    return g
