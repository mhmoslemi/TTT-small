```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    d_lags = idx - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N - 1)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N - 1)], 0.0)
    A_part = -(1.0 - h_pd) * valid_pd.astype(float)
    B_part = h_pmd * valid_pmd.astype(float)
    D = A_part + B_part
    D[:, N-1] = 0.0  # Fix lag-0
    return D

def steepest_descent(h, N, deadline, max_sweeps=1000, plateau_limit=30):
    h = h.astype(float).copy()
    plateau = 0
    best = h.copy()
    best_peak = float(compute_corr(h).max())
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        P = float(c.max())
        if P < best_peak - 1e-10:
            best_peak = P
            best = h.copy()
            plateau = 0
        else:
            plateau += 1
        if plateau > plateau_limit:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        D = compute_D(h, N)
        best_step_peak = P
        best_pq = None
        # Vectorized: for each p in ones, find best q in zeros
        # new_corr = c + D[p,:] - D[q,:]
        # We want min over (p,q) of max_k (c[k] + D[p,k] - D[q,k])
        
        # For computational efficiency, process in batches
        n_ones = len(ones)
        n_zeros = len(zeros)
        D_ones = D[ones, :]  # (n_ones, 2N-1)
        D_zeros = D[zeros, :]  # (n_zeros, 2N-1)
        
        # For each p (row in D_ones), compute:
        #   peaks[p, q] = max_k (c[k] + D_ones[p,k] - D_zeros[q,k])
        # This is (n_ones x n_zeros x (2N-1)) which is too big for N=380
        # 190 * 190 * 759 = 27M floats = 216 MB. Too much!
        
        # Instead, loop over p:
        for ip in range(n_ones):
            base = c + D_ones[ip, :]  # (2N-1,)
            # peaks for all q: max_k (base[k] - D_zeros[q,k])
            # = max of (base - D_zeros.T) along axis 1
            # D_zeros is (n_zeros, 2N-1), so base[None,:] - D_zeros is (n_zeros, 2N-1)
            candidates = base[None, :] - D_zeros  # (n_zeros, 2N-1)
            new_peaks = candidates.max(axis=1)  # (n_zeros,)
            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_step_peak:
                best_step_peak = float(new_peaks[mi])
                best_pq = (ones[ip], zeros[mi])
        
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return best, best_peak
```

```python
            break

        top_k = np.argpartition(corr, -K)[-K:]

        # d(corr[k])/dh[p] = (1-h[p+L])*[p+L valid] - h[p-L]*[p-L valid] (for L>0)
        #                    = (1-h[p-L])*[p-L valid] + h[p+L]*[p+L valid] (for L<0)
```

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    d_lags = idx - (N - 1)
    p_idx = np.arange(N)[:, None]
    d_vals = d_lags[None, :]
    pd = p_idx + d_vals
    p_md = p_idx - d_vals
    valid_pd = (pd >= 0) & (pd < N)
    valid_pmd = (p_md >= 0) & (p_md < N)
    h_pd = np.where(valid_pd, h[np.clip(pd, 0, N - 1)], 0.0)
    h_pmd = np.where(valid_pmd, h[np.clip(p_md, 0, N - 1)], 0.0)
    D = -(1.0 - h_pd) * valid_pd.astype(float) + h_pmd * valid_pmd.astype(float)
    D[:, N - 1] = 0.0
    return D

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def binary_sd(h, N, deadline, max_sweeps=5000, patience=40):
    h = h.astype(float).copy()
    best = h.copy()
    best_pk = float(compute_corr(h).max())
    no_improve = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        P = float(c.max())
        if P < best_pk - 1e-12:
            best_pk = P
            best = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        D = compute_D(h, N)
        D_o = D[ones]
        D_z = D[zeros]
        best_step = P
        bp = -1
        bq = -1
        for ip in range(len(ones)):
            base = c + D_o[ip]
            peaks = (base[None, :] - D_z).max(axis=1)
            mi = int(np.argmin(peaks))
            if peaks[mi] < best_step:
                best_step = float(peaks[mi])
                bp = ip
                bq = mi
        if bp < 0:
            break
        h[ones[bp]] = 0.0
        h[zeros[bq]] = 1.0
    return best, best_pk

def anneal_binary(h, N, n_iters, rng, T0=3.0, T1=0.003):
    h = h.astype(float).copy()
    best = h.copy()
    best_pk = float(compute_corr(h).max())
    cur_pk = best_pk
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / max(1, n_iters))
        ip = rng.randint(len(ones))
        iq = rng.randint(len(zeros))
        p, q = ones[ip], zeros[iq]
        h[p], h[q] = h[q], h[p]
        new_pk = float(compute_corr(h).max())
        delta = new_pk - cur_pk
        if delta < 0 or (T > 0.01 and rng.random() < np.exp(-min(delta / T, 50.0))):
            cur_pk = new_pk
            ones[ip] = q
            zeros[iq] = p
            if new_pk < best_pk:
                best_pk = new_pk
                best = h.copy()
        else:
            h[p], h[q] = h[q], h[p]
    return best, best_pk

def continuous_grad(h, N, deadline, K=30, lr0=0.04, lr_decay=0.9995):
    h = h.astype(float).copy()
    target = N / 2.0
    best = h.copy()
    best_pk = float(compute_corr(h).max())
    lr = lr0
    it = 0
    while time.time() < deadline:
        c = compute_corr(h)
        P = float(c.max())
        if P < best_pk - 1e-12:
            best_pk = P
            best = h.copy()
        if K < len(c):
            top = np.argpartition(c, -K)[-K:]
        else:
            top = np.arange(len(c))
        D = compute_D(h, N)
        grad = D[:, top].sum(axis=1)
        grad = grad - np.mean(grad)
        h = h + lr * grad
        h = project_sum(h, target)
        it += 1
        if it % 500 == 0:
            lr *= lr_decay
            if lr < 1e-5:
                break
    return best, best_pk

def make_binary(N, pattern, rng):
    if pattern == 'alt':
        h = np.zeros(N)
        h[::2] = 1.0
    elif pattern == 'block':
        h = np.zeros(N)
        h[:N // 2] = 1.0
    else:
        h = (rng.random(N) > 0.5).astype(float)
    n_ones = int(h.sum())
    nh = N // 2
    if n_ones > nh:
        idx = np.where(h == 1)[0]
        h[idx[nh:]] = 0
    elif n_ones < nh:
        idx = np.where(h == 0)[0]
        h[idx[:nh - n_ones]] = 1
    return h
```
