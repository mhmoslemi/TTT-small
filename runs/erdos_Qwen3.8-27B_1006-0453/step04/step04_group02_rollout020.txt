        return np.clip(h - lam, 0, 1).sum()

    # We want sum_clipped(lam) = target
    lo = h.min() - 1  # sum = n (all 1s)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    v = np.zeros(n)  # momentum
    alpha = 0.01  # initial step size
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        obj, grad = smooth_c5_grad(y, T)
        
        # Backtracking line search
        t = alpha
        y_new = project_constrained(y - t * grad, n)
        obj_new, _ = smooth_c5_grad(y_new, T)
        
        backtracks = 0
        while obj_new > obj + 1e-4 * t * np.dot(grad, grad) and backtracks < 20:
            t *= 0.5
            y_new = project_constrained(y - t * grad, n)
            obj_new, _ = smooth_c5_grad(y_new, T)
            backtracks += 1
            if t < 1e-10:
                break
        
        alpha = max(t / 0.8, 1e-12)  # update step size
        alpha = min(alpha, 0.1)
        
        # Nesterov update
        beta = 2.0 / (it + 3)
        y = y_new + beta * (y_new - y)
        y = project_constrained(y, n)
        
        c5 = compute_c5(y)
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()
    
    return best_h, best_c5

            #   where the max is over all k (with the n-1 entry handled specially)
            
            # This is a 1D problem. Use minimize_scalar.
            def f1d(t):
                vals = base + t * d
                vals[n-1] = base[n-1] + t - t*t
                return np.max(vals)
            
            # Quick grid to find approximate minimum
            ts = np.linspace(0, 1, 50)
            fs = np.array([f1d(t) for t in ts])
            i_min = np.argmin(fs)
            
            # Refine
            lo = ts[max(0, i_min-1)]
            hi_t = ts[min(49, i_min+1)]
            for _ in range(50):
                m1 = lo + (hi_t-lo)/3
                m2 = hi_t - (hi_t-lo)/3
                if f1d(m1) < f1d(m2):
                    hi_t = m2
                else:
                    lo = m1
            new_t = (lo + hi_t) / 2
            
            if abs(new_t - hi) > 1e-12:
                # Update corr incrementally
                corr += (new_t - hi) * d
                corr[n-1] += (new_t - hi) * (1 - new_t - hi)
                best_h[i] = new_t
                improved = True
        
        best_h = project(best_h, n)
        # Recompute corr after projection (since projection changes values)
        corr_check = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Check if our incremental corr is still valid
        # (it should be, unless projection changed things significantly)
        corr = corr_check
        
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            stall = 0
        else:
            stall += 1
            if stall >= 10:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    target = n / 2.0
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_c5_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    et = np.exp(T * (corr - m))
    Z = et.sum()
    obj = (m + np.log(Z) / T) * dx
    w = et / Z
    W = w + w[::-1]
    conv = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        wv = W[idx]
        if wv > 1e-15:
            lo = max(0, k)
            hi = min(n, n + k)
            conv[lo:hi] += wv * h[lo - k:hi - k]
    grad = dx * (1.0 - 2.0 * conv)
    return obj, grad

def f1d_eval(base, d, n, t):
    vals = base + t * d
    vals[n - 1] = base[n - 1] + t - t * t
    return np.max(vals)

def fast_coord_descent(h, n, t_end, max_sweeps=15000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    stall = 0
    ts_grid = np.linspace(0.0, 1.0, 40)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        for i in range(n):
            hi = best_h[i]
            d = np.zeros(2 * n - 1)
            for k in range(1, n):
                ip = i - k
                iq = i + k
                v = 0.0
                if 0 <= ip < n:
                    v += 1.0 - best_h[ip]
                if 0 <= iq < n:
                    v += -best_h[iq]
                d[n - 1 + k] = v
                v = 0.0
                if 0 <= iq < n:
                    v += 1.0 - best_h[iq]
                if 0 <= ip < n:
                    v += -best_h[ip]
                d[n - 1 - k] = v
            base = (corr - hi * d).copy()
            base[n - 1] = corr[n - 1] - hi * (1.0 - hi)
            fs = np.array([f1d_eval(base, d, n, t) for t in ts_grid])
            i_min = np.argmin(fs)
            lo_b = ts_grid[max(0, i_min - 1)]
            hi_b = ts_grid[min(39, i_min + 1)]
            for _ in range(45):
                m1 = lo_b + (hi_b - lo_b) / 3.0
                m2 = hi_b - (hi_b - lo_b) / 3.0
                if f1d_eval(base, d, n, m1) < f1d_eval(base, d, n, m2):
                    hi_b = m2
                else:
                    lo_b = m1
            new_t = (lo_b + hi_b) / 2.0
            if abs(new_t - hi) > 1e-13:
                corr += (new_t - hi) * d
                corr[n - 1] += (new_t - hi) * (1.0 - new_t - hi)
                best_h[i] = new_t
        best_h = project(best_h, n)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            stall = 0
        else:
            stall += 1
            if stall >= 15:
                break
    return best_h, best_c5

def grad_polish(h, n, t_end, max_iters=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for T in [200, 800, 3000, 12000, 50000]:
        if time.time() > t_end:
            break
        y = best_h.copy()
        step = 0.05
        for it in range(max_iters):
            if time.time() > t_end:
                break
            obj, grad = smooth_c5_grad(y, T)
            gnorm = np.sqrt(np.dot(grad, grad))
            if gnorm < 1e-12:
                break
            g = grad / gnorm
            t = step
            accepted = False
            for _ in range(30):
                y_new = project(y - t * grad, n)
                obj_new, _ = smooth_c5_grad(y_new, T)
                if obj_new <= obj + 1e-7 * t * gnorm * gnorm:
                    y = y_new
                    accepted = True
                    break
                t *= 0.5
            if accepted:
                step = min(t * 1.8, 0.5)
            else:
                step *= 0.5
            c5 = compute_c5(y)
            if c5 < best_c5:
                best_c5 = c5
                best_h = y.copy()
    return best_h, best_c5

def perturb_refine(h, n, t_end, rng, n_pert=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for p in range(n_pert):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.7 * (p / float(n_pert))
        scale = 0.03 * cool + 0.005
        hp = h + scale * rng.standard_normal(n)
        hp = project(hp, n)
        hp, cp = fast_coord_descent(hp, n, min(time.time() + 15, t_end), max_sweeps=2000)
        if cp < best_c5:
            best_h, best_c5 = hp, cp
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0.0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

    t2 = t0 + budget_s * 0.55

    t4 = t0 + budget_s * 0.82

        ha = interpolate_h(best_h, n_alt)
        ha, ca = fast_coord_descent(ha, n_alt, min(time.time() + 20, t0 + budget_s * 0.92), max_sweeps=3000)

        best_h, best_c5 = grad_polish(best_h, n, t0 + budget_s * 0.99, max_iters=500)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
