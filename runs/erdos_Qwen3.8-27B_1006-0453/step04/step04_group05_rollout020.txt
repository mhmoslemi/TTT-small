    vLm = (pLm >= 0) & (pLm < N)
    hL = np.where(vL, h[np.clip(pL, 0, N - 1)], 0.0)
    hLm = np.where(vLm, h[np.clip(pLm, 0, N - 1)], 0.0)
    R = -np.where(vL, 1.0 - hL, 0.0)
    A = np.where(vL, 1.0 - hL, 0.0) + np.where(vLm, hLm, 0.0)
    return R, A

def binary_sd(h, N, rng, max_sweeps=300, deadline=None, sample_sz=30):
    h = h.astype(float).copy()
    for _ in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        corr = compute_corr(h)
        P = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        R, A = build_RA(h, N)
        n_p = min(len(ones), sample_sz)
        n_q = min(len(zeros), sample_sz)
        p_sub = ones if n_p == len(ones) else ones[rng.choice(len(ones), n_p, replace=False)]
        q_sub = zeros if n_q == len(zeros) else zeros[rng.choice(len(zeros), n_q, replace=False)]
        best_pk, best_pq = P, None
        for p in p_sub:
            base = corr + R[p]
            cands = base[None, :] + A[q_sub]
            peaks = cands.max(axis=1)
            mi = np.argmin(peaks)
            if peaks[mi] < best_pk:
                best_pk = peaks[mi]
                best_pq = (p, q_sub[mi])
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def lse_optimize(h, N, deadline, beta=50.0):
    """Smooth optimization using projected subgradient with LSE."""
    target = N / 2.0
    h = h.astype(float).copy()
    corr = compute_corr(h)
    center = N - 1
    best_h = h.copy()
    best_c5 = float(corr.max()) * (2.0 / N)
    
    beta_s = max(1, int((deadline - time.time()) * 8))
    lr = 0.05
    decay = 0.995
    for it in range(beta_s):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        w = np.exp(corr * beta - np.logaddexp.reduce(corr * beta))  # softmax weights
        grad = np.zeros(N)
        idx = np.arange(2 * N - 1)
        L_all = idx - center
        for k in range(2 * N - 1):
            wk = w[k]
            if wk < 1e-8:
                continue
            L = int(L_all[k])
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif L > 0 and L < N:
                grad[:N - L] += wk * (1.0 - h[L:])
                grad[L:] -= wk * h[:N - L]
            elif L < 0 and -L < N:
                M = -L
                grad[M:] += wk * (1.0 - h[:N - M])
                grad[:N - M] -= wk * h[M:]
        h = project_sum(h - lr * grad, target)
        if it % 50 == 0:
            lr *= decay
        c5 = float(compute_corr(h).max()) * (2.0 / N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def round_and_refine(h, N, rng, deadline):
    hb = (h > 0.5).astype(float)
    n_ones = int(hb.sum())
    nh = N // 2
    if n_ones > nh:
        idx = np.where(hb == 1)[0]
        hb[idx[rng.choice(n_ones, n_ones - nh, replace=False)]] = 0
    elif n_ones < nh:
        idx = np.where(hb == 0)[0]
        hb[idx[rng.choice(N - n_ones, nh - n_ones, replace=False)]] = 1
    hb = binary_sd(hb, N, rng, max_sweeps=200, deadline=deadline, sample_sz=30)
    return hb, compute_c5(hb, N)
