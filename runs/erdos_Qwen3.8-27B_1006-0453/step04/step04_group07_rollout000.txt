```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
        return np.clip(h, 0.0, 1.0)
    else:
        if s < 1e-14:
            return h
        scale = target / s
        h = h * scale
        return np.clip(h, 0.0, 1.0)

    bh = h.copy()
    corr = np.correlate(bh, 1.0 - bh, mode='full')
```

```python
                fm = obj_val(mid)
                if fm < best_f:
                    lo, hi = (0.0, 0.5) if best_v == 1.0 else (0.5, 1.0)
                    try:
                        res = minimize_scalar(obj_val, bounds=(lo, hi), method='bounded', options={'xatol': 1e-7})
                        if res.fun < best_f:
                            best_f, best_v = float(res.fun), float(res.x)
                    except:
                        pass
            
            if best_f * dx < best_c5 - 1e-13:
                bh[i] = best_v
                best_c5 = best_f * dx
                # Update corr incrementally
                if best_v != old:
                    corr += (best_v - old) * coeff
                    corr[c0] = c0_base + (best_v - old) * (1.0 - best_v - old)
                improved = True
        if not improved:
            break
    return bh, best_c5
```

This is much faster because we update corr incrementally (O(n) per coordinate) instead of recomputing (O(n²)).

Now for the global search, instead of SLSQP-based basin hopping, I'll use:
1. Random perturbations (Gaussian, binary flips, block shifts)
2. Fast coordinate descent as local optimizer
3. Accept/reject based on improvement (or occasional worsening for escaping)

Let me also add a "subgradient" step as another tool:
- Find all active shifts (corr[k] close to max)
- Compute subgradient of max at active set
- Take a step in negative subgradient direction
- Project

Let me write the full solution now. I'll aim for:
- Phase 1: Quick coord descent from initial (30s)
- Phase 2: Basin hopping with fast local opt (400s) - many iterations since each is fast
- Phase 3: More coord descent (30s)
- Phase 4: Try different resolutions (100s)
- Phase 5: Final refinement (50s)
- Buffer: remaining time

Let me also think about what perturbations to use in basin hopping:
1. Gaussian noise: h_p = project(h + sigma * randn(n))
2. Binary flip: pick k random positions, flip them (1-h[i])
3. Block shift: pick a random block of length L, shift it by a random offset
4. Swap: pick two random positions, swap their values

For the "fast local optimization" after perturbation, I'll use just a few sweeps of coordinate descent (10-30 sweeps).

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent_fast(h, n, t_end, max_sweeps=200):
    dx = 2.0 / n
    bh = h.copy()
    corr = np.array(np.correlate(bh, 1.0 - bh, mode='full'), dtype=float)
    best_c5 = float(np.max(corr)) * dx
    c0 = n - 1
    ks = np.arange(-(n - 1), n)
    ja_cache = (i + ks for i in range(n))
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = bh[i]
            ja = i + ks
            jb = i - ks
            va = (ja >= 0) & (ja < n)
            vb = (jb >= 0) & (jb < n)
            coeff = np.zeros(2 * n - 1)
            coeff[va] += (1.0 - bh[ja[va]])
            coeff[vb] -= bh[jb[vb]]
            c0_base = corr[c0]
            
            def obj_val(v):
                if v == old:
                    return float(np.max(corr))
                cn = corr + (v - old) * coeff
                cn[c0] = c0_base + (v - old) * (1.0 - v - old)
                return float(np.max(cn))
            
            f0 = obj_val(0.0)
            f1 = obj_val(1.0)
            best_f = min(f0, f1)
            best_v = 0.0 if f0 <= f1 else 1.0
            fo = float(np.max(corr))
            if fo < best_f:
                best_f, best_v = fo, old
            
            if 0.0 < best_v < 1.0:
                lo, hi = max(0.0, best_v - 0.4), min(1.0, best_v + 0.4)
            elif best_v == 0.0:
                lo, hi = 0.0, 0.5
            else:
                lo, hi = 0.5, 1.0
            try:
                res = minimize_scalar(obj_val, bounds=(lo, hi), method='bounded', options={'xatol': 1e-7})
                if res.fun < best_f - 1e-15:
                    best_f, best_v = float(res.fun), float(res.x)
            except:
                pass
            
            if best_f * dx < best_c5 - 1e-13:
                bh[i] = best_v
                best_c5 = best_f * dx
                delta = best_v - old
                corr += delta * coeff
                corr[c0] = c0_base + delta * (1.0 - best_v - old)
                improved = True
        if not improved:
            break
    return bh, best_c5

def subgradient_step(bh, n, corr, c5_val, lr=0.05):
    dx = 2.0 / n
    max_corr = np.max(corr)
    active = corr > max_corr - 0.1 * dx
    n_active = np.sum(active)
    if n_active == 0:
        return bh, corr, c5_val
    c0 = n - 1
    grad = np.zeros(n)
    ks = np.arange(-(n - 1), n)
    for i in range(n):
        ja = i + ks
        jb = i - ks
        va = (ja >= 0) & (ja < n)
        vb = (jb >= 0) & (jb < n)
        g = np.zeros(2 * n - 1)
        g[va] += (1.0 - bh[ja[va]])
        g[vb] -= bh[jb[vb]]
        g[c0] = 1.0 - 2.0 * bh[i]
        grad[i] = np.sum(g[active]) / n_active
    bh_new = bh - lr * grad
    bh_new = project(bh_new, n)
    corr_new = np.array(np.correlate(bh_new, 1.0 - bh_new, mode='full'), dtype=float)
    c5_new = float(np.max(corr_new)) * dx
    return bh_new, corr_new, c5_new

def basin_hop_fast(bh, c5_val, n, t_end, max_iters=500):
    dx = 2.0 / n
    best_h, best_c5 = bh.copy(), c5_val
    work_h, work_c5 = bh.copy(), c5_val
    work_corr = np.array(np.correlate(work_h, 1.0 - work_h, mode='full'), dtype=float)
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.1, 1.0 - 0.9 * it / max_iters)
        r = np.random.rand()
        
        if r < 0.4:
            sigma = 0.05 + 0.2 * cool
            h_p = work_h + sigma * np.random.randn(n)
        elif r < 0.7:
            h_p = work_h.copy()
            n_flips = max(1, int(np.random.randint(1, n // 8)))
            idx = np.random.choice(n, size=n_flips, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        else:
            h_p = work_h.copy()
            blk_len = np.random.randint(3, max(4, n // 10) + 1)
            start = np.random.randint(0, n - blk_len)
            shift = np.random.randint(-5, 6)
            block = h_p[start:start + blk_len].copy()
            new_start = start + shift
            new_end = new_start + blk_len
            if new_start < 0 or new_end > n:
                continue
            h_p[start:start + blk_len] = h_p[new_start:new_end]
            h_p[new_start:new_end] = block
            h_p[start:start + blk_len] = block
        
        h_p = project(h_p, n)
        hc, c5v = coord_descent_fast(h_p, n, t_end, max_sweeps=30)
        
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h, work_c5, work_corr = hc, c5v, np.array(np.correlate(hc, 1.0 - hc, mode='full'), dtype=float)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
        
        if it % 50 == 49:
            bh_sg, corr_sg, c5_sg = subgradient_step(work_h, n, work_corr, work_c5, lr=0.03 * cool)
            hc2, c5v2 = coord_descent_fast(bh_sg, n, t_end, max_sweeps=20)
            if c5v2 < work_c5:
                work_h, work_c5 = hc2, c5v2
                if c5v2 < best_c5:
                    best_c5, best_h = c5v2, hc2.copy()
    
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    
    t = time.time() - t0
    
    # Phase 1: Initial coordinate descent
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.12, max_sweeps=300)
    
    t = time.time() - t0
    
    # Phase 2: Basin hopping with fast local optimization
    best_h, best_c5 = basin_hop_fast(best_h, best_c5, n, t0 + budget_s * 0.50, max_iters=600)
    
    t = time.time() - t0
    
    # Phase 3: Final coordinate descent
    best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.58, max_sweeps=400)
    
    t = time.time() - t0
    
    # Phase 4: Second round basin hopping (different perturbations focus)
    np.random.seed(seed + 100)
    best_h, best_c5 = basin_hop_fast(best_h, best_c5, n, t0 + budget_s * 0.72, max_iters=400)
    
    t = time.time() - t0
    
    # Phase 5: Try different resolutions
    for n_try in [120, 140, 200, 240, 280, 320]:
        if time.time() - t0 > budget_s * 0.88:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent_fast(h_tr, n_try, t0 + budget_s * 0.90, max_sweeps=80)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try
    
    t = time.time() - t0
    
    # Phase 6: Subgradient + coord descent polishing
    if time.time() - t0 < budget_s * 0.92:
        corr = np.array(np.correlate(best_h, 1.0 - best_h, mode='full'), dtype=float)
        for lr in [0.05, 0.03, 0.02, 0.01]:
            if time.time() - t0 > budget_s * 0.94:
                break
            bh2, corr2, c52 = subgradient_step(best_h, n, corr, best_c5, lr=lr)
            bh2, c52 = coord_descent_fast(bh2, n, t0 + budget_s * 0.945, max_sweeps=50)
            if c52 < best_c5:
                best_c5, best_h = c52, bh2
                corr = corr2
    
    # Phase 7: Final polish
    if time.time() - t0 < budget_s * 0.96:
        best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.98, max_sweeps=500)
    
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
