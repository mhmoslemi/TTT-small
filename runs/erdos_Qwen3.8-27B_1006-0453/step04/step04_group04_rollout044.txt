import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -60, 60)
    exps = np.exp(shifted)
    s = exps.sum()
    p = exps / s
    fval = (m + np.log(s) / T) * dx
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if p[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        w = p[k_idx] * dx
        if lag >= 0:
            nn = n - lag
            if nn > 0:
                grad[:nn] += w * (1.0 - h[lag:n])
                grad[lag:] += w * (-h[:nn])
        else:
            nl = -lag
            nn = n - nl
            if nn > 0:
                grad[nl:] += w * (1.0 - h[:nn])
                grad[:nn] += w * (-h[nl:])
    return fval, grad

def proj_exact(y, target):
    n = len(y)
    lo = y.min() - target - 0.5
    hi = y.max() - target + 0.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(y - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_loop(h, n, t_end, T_start=80, T_max=80000, lr0=0.008, mu=0.85):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    x = h.copy()
    v = np.zeros(n)
    T = T_start
    lr = lr0
    stall = 0
    for step in range(50000):
        if time.time() > t_end:
            break
        fval, grad = smooth_c5_grad(x, T)
        v = mu * v - lr * grad
        x_new = proj_exact(x + v, n / 2.0)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5 - 1e-14:
            best_h = x_new.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
        x = x_new
        if step % 50 == 0:
            T = min(T * 1.04, T_max)
            lr = max(lr * 0.996, 1e-6)
        if stall > 2000:
            break
    return best_h, best_c5

def coord_opt_i(h, n, i):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    hi_old = h[i]
    a = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        a[k] = c
    base = corr - hi_old * a
    candidates = [0.0, 1.0]
    top10 = np.argsort(-np.abs(a))[:12]
    for idx in top10:
        if abs(a[idx]) > 1e-14:
            tc = -base[idx] / a[idx]
            if 0.01 < tc < 0.99:
                candidates.append(tc)
    best_t, best_v = hi_old, float('inf')
    for t in candidates:
        t = np.clip(t, 0, 1)
        v = np.max(base + t * a) * dx
        if v < best_v:
            best_v, best_t = v, t
    res = minimize_scalar(lambda t: np.max(base + t * a) * dx,
                          bounds=(max(0.0, best_t - 0.15), min(1.0, best_t + 0.15)),
                          method='bounded', options={'xatol': 1e-13})
    return np.clip(res.x, 0, 1)

def coord_sweep(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            t = coord_opt_i(best_h, n, i)
            if abs(t - best_h[i]) > 1e-11:
                best_h[i] = t
                best_h = proj_exact(best_h, n / 2.0)
                c5v = compute_c5(best_h)
                if c5v < best_c5 - 1e-15:
                    best_c5 = c5v
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def active_set_slsqp(h, n, t_end):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    active = np.where(corr >= mx * 0.99998)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')[active]
        return np.max(c) * dx
    def con_x(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': con_x}],
                   options={'maxiter': 800, 'ftol': 1e-16})
    return proj_exact(np.clip(res.x, 0, 1), n / 2.0)
