def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h.copy()
    if s > target:
        h = h * (target / s)
    else:
        room = (1.0 - h).sum()
        if room > 1e-14:
            h = h + (1.0 - h) * ((target - s) / room)
        else:
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def smooth_val(corr, T):
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val

    corr_full = compute_corr(best_h)
    best_c5 = float(np.max(corr_full)) * (2.0 / n)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):

            break
        improved = False
        order = np.random.permutation(n)

            if time.time() > t_end:
                return best_h, best_c5
            hi_old = best_h[i]

            k1 = np.arange(n-1-i, 2*n-1-i)

            k2 = np.arange(i, i+n)

        return float(m + np.log(np.sum(np.exp(T*(corr-m))))/T) * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h.copy()
    if s > target:
        h = h * (target / s)
    else:
        room = (1.0 - h).sum()
        if room > 1e-14:
            h = h + (1.0 - h) * ((target - s) / room)
        else:
            h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def remove_and_coef(corr_full, h, i, n):
    hi = h[i]
    corr_base = corr_full.copy()
    idx1 = slice(i, i + n)
    corr_base[idx1] -= hi * (1.0 - h)[::-1]
    idx2 = slice(n - 1 - i, 2 * n - 1 - i)
    corr_base[idx2] -= h * (1.0 - hi)
    corr_base[n - 1] += hi * (1.0 - hi)
    coefs = np.zeros(2 * n - 1)
    coefs[idx1] += (1.0 - h)[::-1]
    coefs[idx2] -= h
    return corr_base, coefs

                          options={'xatol': 1e-11})

    corr_full = compute_corr(best_h)
    best_c5 = float(np.max(corr_full)) * (2.0 / n)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:

def slsqp_polish(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return float(m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def con(x):
        return x.sum() - n / 2.0

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))

        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0003
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = work_h.copy(), work_c5
            work_h, work_c5 = coord_descent_fast(work_h, n,
                                                 min(time.time() + 3, t2),
                                                 max_sweeps=80)
            if work_c5 < best_c5:
                best_h, best_c5 = work_h.copy(), work_c5
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 3: Second coord descent round
    t3 = t0 + budget_s * 0.85
    best_h, best_c5 = coord_descent_fast(best_h, n, t3, max_sweeps=6000)

    # Phase 4: Fine basin hopping
    t4 = t0 + budget_s * 0.92
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(30):
        if time.time() > t4:
            break
        cool = 1.0 - 0.7 * (it / 30.0)
        amp = 0.008 + 0.03 * cool
        h_pert = work_h + amp * rng.randn(n)
        h_pert = project_h(h_pert)
        hc = slsqp_polish(h_pert, 3000 + 5000 * cool, n, maxiter=200)
        c5v = compute_c5(hc)
        if c5v < work_c5:
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = work_h.copy(), work_c5
            work_h, work_c5 = coord_descent_fast(work_h, n,
                                                 min(time.time() + 2, t4),
                                                 max_sweeps=60)
            if work_c5 < best_c5:
                best_h, best_c5 = work_h.copy(), work_c5
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Final coord polish
    if time.time() < t0 + budget_s * 0.96:
        best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.96,
                                             max_sweeps=4000)

    # Phase 6: Ultra-high-T SLSQP
    if time.time() < t0 + budget_s * 0.99:
        hc = slsqp_polish(best_h, 200000, n, maxiter=1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
