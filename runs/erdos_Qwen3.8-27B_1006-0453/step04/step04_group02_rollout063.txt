import time

    dx = 2.0 / n

    return float(np.max(corr) * dx)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def grad_smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(T * (corr - m))
    ws = weights.sum()
    if ws < 1e-300:
        weights = np.ones_like(corr) / len(corr)
    else:
        weights /= ws
    grad = np.zeros(n)

        w = weights[l + n - 1]
        if w < 1e-16:

                grad[l:] += w * (1.0 - h[:n-l])
                grad[:n-l] -= w * h[l:]

            ml = -l
            if ml < n:
                grad[:n-ml] += w * (1.0 - h[ml:])
                grad[ml:] -= w * h[:n-ml]

            if cnt == 0:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    s = w.sum()
    if s > 1e-300:
        w /= s
    else:
        w = np.ones(len(corr)) / len(corr)
    g = np.zeros(n)
    for lag in range(-(n - 1), n):
        wi = w[lag + n - 1]
        if wi < 1e-16:
            continue
        if lag >= 0:
            g[lag:] += wi * (1.0 - h[:n - lag])
            g[:n - lag] -= wi * h[lag:]
        else:
            ml = -lag
            g[:n - ml] += wi * (1.0 - h[ml:])
            g[ml:] -= wi * h[:n - ml]
    return g * dx

def grad_proj(h, T):
    g = grad_smooth(h, T)
    return g - g.mean()

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(100):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_opt(h, T, n, iters=3000, gtol=1e-12):
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(smooth_c5, h, jac=grad_proj, args=(T,), method='L-BFGS-B',
                     bounds=bnds, options={'maxiter': iters, 'gtol': gtol, 'ftol': 1e-16})
        return project(r.x, n)
    except Exception:
        return h

def f1d_min(t, args_list):
    corr_base, dv, dx, n = args_list
    vals = corr_base + t * dv
    vals[n - 1] = corr_base[n - 1] + t * (1.0 - t)
    return np.max(vals) * dx

def coord_desc(h, n, t_end, sweeps=40, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    bh = h.copy()
    bc5 = compute_c5(bh)
    dx = 2.0 / n
    stall = 0
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return bh, bc5
            hs = bh[i]
            bh[i] = 0.0
            cb = np.correlate(bh, 1.0 - bh, mode='full').copy()
            bh[i] = hs
            dv = np.zeros(2 * n - 1)
            for k in range(1, n):
                v1 = (1.0 - bh[i - k]) if 0 <= i - k < n else 0.0
                v2 = -bh[i + k] if 0 <= i + k < n else 0.0
                dv[n - 1 + k] += v1 + v2
                v3 = (1.0 - bh[i + k]) if 0 <= i + k < n else 0.0
                v4 = -bh[i - k] if 0 <= i - k < n else 0.0
                dv[n - 1 - k] += v3 + v4
            r = minimize_scalar(f1d_min, bounds=(0.0, 1.0), method='bounded',
                                args=(cb, dv, dx, n), options={'xatol': 1e-12})
            if abs(r.x - hs) > 1e-12:
                bh[i] = r.x
                bh = project(bh, n)
                c = compute_c5(bh)
                if c < bc5 - 1e-14:
                    bc5 = c
                    imp = True
        if imp:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return bh, bc5

def interp_h(h, n_new):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(xn, xo, h), n_new)

def diverse_starts(best_h, n, rng):
    cands = []
    cands.append(project(best_h + 0.04 * rng.standard_normal(n), n))
    hb = np.zeros(n)

    h2 = np.zeros(n)
    h2[n // 4:n * 3 // 4] = 1.0
    cands.append(project(h2 + 0.06 * rng.standard_normal(n), n))
    return cands

        if cv < best_c5:

    # Phase 4: Diverse restarts
    if elapsed() < budget_s * 0.62:
        cands = diverse_starts(best_h, n, rng)
        for ci in range(len(cands)):
            if elapsed() > budget_s * 0.62:

            for T in [200, 3000]:
