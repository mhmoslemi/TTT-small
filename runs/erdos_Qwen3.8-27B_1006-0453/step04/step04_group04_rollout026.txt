```python
    corr = np.correlate(h, 1.0 - h, mode='full')

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    grad = np.zeros(n)

        if pk < 1e-14:
```

```python
def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 0)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n-lag] += pk * (1.0 - h[lag:])
            grad[lag:] -= pk * h[:n-lag]
        else:
            nl = -lag
            grad[nl:] += pk * (1.0 - h[:n-nl])
            grad[:n-nl] -= pk * h[nl:]
    return grad * dx / T
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 0)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n-lag] += pk * (1.0 - h[lag:])
            grad[lag:] -= pk * h[:n-lag]
        else:
            nl = -lag
            grad[nl:] += pk * (1.0 - h[:n-nl])
            grad[:n-nl] -= pk * h[nl:]
    return grad * dx / T

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lbgfs_opt(h0, n, T, mu, maxiter=400):
    def obj(x):
        return softmax_obj(x, T) + mu * (x.sum() - n / 2.0) ** 2
    def obj_grad(x):
        g = softmax_grad(x, T)
        return g + 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
    res = minimize(obj, h0, jac=obj_grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    return proj(res.x, n)

def coord_opt(h, n, i):
    h_save = h[i]
    def f1d(t):
        h[i] = t
        c = compute_c5(h)
        h[i] = h_save
        return c
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) > 1e-12:
                best_h[i] = hi_new
                best_h = proj(best_h, n)
                c5_try = compute_c5(best_h)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def pair_swap_opt(h, n, i, j):
    hi_s, hj_s = h[i], h[j]
    best_t = (hi_s, hj_s)
    best_c5 = compute_c5(h)
    for ti in [0.0, 0.5, 1.0]:
        for tj in [0.0, 0.5, 1.0]:
            h[i], h[j] = ti, tj
            h = proj(h, n)
            c5v = compute_c5(h)
            if c5v < best_c5 - 1e-15:
                best_c5 = c5v
                best_t = (ti, tj)
    h[i], h[j] = best_t
    h = proj(h, n)
    return h, best_c5

def block_perturb(h, n, magnitude=0.08):
    hp = h.copy()
    bstart = np.random.randint(0, max(1, n - 6))
    blen = np.random.randint(4, max(5, n // 4))
    blen = min(blen, n - bstart)
    hp[bstart:bstart + blen] += magnitude * np.sin(np.pi * np.arange(blen) / blen)
    hp = proj(hp, n)
    return hp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 180
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0

    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing T and penalty
    for T, mu in [(100, 1.0), (500, 2.0), (2000, 5.0), (8000, 10.0)]:
        if time.time() > t0 + budget_s * 0.06:
            break
        try:
            hc = lbgfs_opt(best_h, n, T, mu, maxiter=300)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Aggressive coordinate descent
    t_cd_end = t0 + budget_s * 0.18
    best_h, best_c5 = coord_descent(best_h, n, t_cd_end, max_sweeps=800)

    # Phase 3: Basin hopping with structured perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 60
    for it in range(n_bh):
        if time.time() > t0 + budget_s * 0.55:
            break
        cool = 1.0 - 0.85 * (it / float(n_bh))
        amp = 0.015 + 0.10 * cool
        h_pert = block_perturb(work_h, n, amp)
        # Add some random small noise
        h_pert += amp * 0.3 * np.random.randn(n)
        h_pert = proj(h_pert, n)
        try:
            T_bh = 2000 if cool > 0.3 else 8000
            hc = lbgfs_opt(h_pert, n, T_bh, 5.0, maxiter=150)
            # Quick coord descent polish
            hc, c5h = coord_descent(hc, n, time.time() + 3.0, max_sweeps=50)
            c5v = compute_c5(hc)
            temp = 0.002 * cool + 0.0003
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Coordinate descent again
    if time.time() < t0 + budget_s * 0.68:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.68, max_sweeps=600)

    # Phase 5: Try different resolutions
    for n_try in [160, 200, 220]:
        if time.time() > t0 + budget_s * 0.82:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = proj(h_interp, n_try)
        try:
            hc = lbgfs_opt(h_interp, n_try, 3000, 5.0, maxiter=300)
            hc, c5h = coord_descent(hc, n_try, time.time() + 8.0, max_sweeps=200)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                n = n_try
        except Exception:
            pass

    # Phase 6: Binary push + re-optimize
    if time.time() < t0 + budget_s * 0.90:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = proj(h_bin, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5 + 0.005:
            try:
                h_b = lbgfs_opt(h_bin, n, 5000, 5.0, maxiter=200)
                h_b, c5_b = coord_descent(h_b, n, time.time() + 5.0, max_sweeps=100)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b
            except Exception:
                pass

    # Phase 7: Final coordinate descent polish
    if time.time() < t0 + budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=400)

    # Phase 8: Very high T L-BFGS polish
    if time.time() < t0 + budget_s * 0.99:
        try:
            hc = lbgfs_opt(best_h, n, 20000, 10.0, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
