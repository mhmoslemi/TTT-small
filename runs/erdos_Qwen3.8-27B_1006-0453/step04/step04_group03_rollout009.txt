    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

                                 np.random.RandomState(n_try))
        if c5v < results[1]:
            results = (h_i, c5v, n_try)
    return results

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=np.float64)
    else:
        best_h = np.zeros(224)
        best_h[:112] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with increasing T and penalty (10% budget)
    t1_end = t0 + budget_s * 0.10
    for T, lam in [(100, 200.0), (1000, 2000.0), (10000, 20000.0)]:
        if elapsed() > budget_s * 0.10:
            break
        hc = lbfgs_polish(best_h, T, n, lam, maxiter=1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Phase 2: SLSQP with high T (10% budget)
    t2_end = t0 + budget_s * 0.20
    for T in [5000, 20000]:
        if elapsed() > budget_s * 0.20:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=800, ftol=1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 3: Heavy coordinate descent (30% budget)
    t_cd = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, n, t_cd, max_sweeps=8000, rng=rng)

    # Phase 4: Basin hopping with small perturbations (25% budget)
    work_h = best_h.copy()
    work_c5 = best_c5
    n_perturb = 400
    for it in range(n_perturb):
        if elapsed() > budget_s * 0.75:
            break
        cool = max(1.0 - 0.98 * (it / float(n_perturb)), 0.02)
        sigma = 0.005 * cool + 0.0005
        if it % 10 == 5:
            sigma *= 3.0
        h_p = work_h + sigma * rng.randn(n)
        h_p = project_to_sum(h_p, n / 2.0)
        hc, c5v = coord_descent(h_p, n, time.time() + 0.8, 15, rng)
        temp = 0.001 * cool + 1e-7
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-15)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = hc.copy(), c5v

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Try other grid sizes (10% budget)
    t_oth = t0 + budget_s * 0.85
    if elapsed() < budget_s * 0.85:
        h_new, c5_new, n_new = try_other_n(best_h, n, best_c5, t_oth)
        if c5_new < best_c5:
            best_h, best_c5, n = h_new, c5_new, n_new

    # Phase 6: Final SLSQP + coordinate descent (10% budget)
    for T in [50000, 200000]:
        if elapsed() > budget_s * 0.95:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=1500, ftol=1e-16)
            hc, c5v = coord_descent(hc, n, time.time() + 3.0, 100, rng)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Final coordinate descent
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, 500, rng)

    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
