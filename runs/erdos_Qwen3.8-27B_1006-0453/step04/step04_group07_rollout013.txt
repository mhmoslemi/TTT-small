        frac = it / max_iter
        top_k = max(3, int(5 + (top_k_max - 5) * frac))
        corr = compute_corrs(best_h)
        top_idx = np.argpartition(corr, min(top_k, n_out-1))[-top_k:]
        top_vals = corr[top_idx]
        spread = top_vals - top_vals.min()
        w = np.exp(30.0 * spread / (spread.max() + 1e-12))
        w /= w.sum()

        grad = np.zeros(n)
        for ii in range(top_k):
            k = all_ks[top_idx[ii]]
            grad += w[ii] * grad_corr_shift(best_h, int(k))

        lr = 0.008 * (1.0 - 0.5 * frac)
        gn = np.linalg.norm(grad)
        if gn > 1e-10:
            grad /= gn
        cand = project_feasible(best_h - lr * grad + 0.1 * lr * np.random.randn(n) * (1.0 - frac), n)
        c5 = np.max(compute_corrs(cand)) * dx
        if c5 < best_c5:
            best_h = cand
            best_c5 = c5
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corrs(best_h)
    best_c5 = np.max(corr) * dx
    n_out = 2 * n - 1
    ks = np.arange(-(n-1), n)
    arange_n = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            shifts = ks
            vp = (i + shifts >= 0) & (i + shifts < n)
            vn = (i - shifts >= 0) & (i - shifts < n)
            coeff = np.zeros(n_out)
            coeff[vp] += (1.0 - best_h[i + shifts[vp]])
            coeff[vn] -= best_h[i - shifts[vn]]
            base = corr.copy()
            def obj(v):
                return np.max(base + (v - old) * coeff)
            f0 = obj(0.0)
            f1 = obj(1.0)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            bv, bf = res.x, res.fun
            if f0 < bf: bv, bf = 0.0, f0
            if f1 < bf: bv, bf = 1.0, f1
            if bf < np.max(corr) - 1e-13:
                best_h[i] = bv
                corr = compute_corrs(best_h)
                improved = True
        if not improved:
            break
    best_c5 = np.max(compute_corrs(best_h)) * dx
    return best_h, best_c5

def cosine_start(n, a1, a2, a3):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + a1*np.cos(np.pi*x) + a2*np.cos(2*np.pi*x) + a3*np.cos(3*np.pi*x)
    return project_feasible(h, n)

def smooth_perturb(h, n, n_modes, amp, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    pert = np.zeros(n)
    for m in range(n_modes):
        freq = 1 + rng.integers(0, 3)
        phase = rng.uniform(0, 2*np.pi)
        a = amp / np.sqrt(m + 1)
        pert += a * np.sin(np.pi * freq * x + phase)
    return project_feasible(h + pert, n)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def try_candidate(h, n, t_end, top_k=15, sg_iters=3000, cd_sweeps=200):
    dx = 2.0 / n
    h = project_feasible(h, n)
    c5 = np.max(compute_corrs(h)) * dx
    t_sg = min(t_end, time.time() + (t_end - time.time()) * 0.4)
    h, c5 = subgradient_opt(h, n, t_sg, top_k_max=top_k, max_iter=sg_iters)
    t_cd = min(t_end, time.time() + (t_end - time.time()) * 0.6)
    h, c5 = coord_descent(h, n, t_cd, max_sweeps=cd_sweeps)
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient on initial
    t1 = t0 + budget_s * 0.15
    h1, c5_1 = subgradient_opt(best_h, n, t1, top_k_max=30, max_iter=15000)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Coordinate descent
    t2 = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=400)

    # Phase 3: Subgradient + CD refinement
    t3 = t0 + budget_s * 0.42
    h3, c5_3 = subgradient_opt(best_h, n, t3, top_k_max=20, max_iter=10000)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3
    t3b = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, n, t3b, max_sweeps=300)

    # Phase 4: Diverse restarts
    t4_end = t0 + budget_s * 0.72
    starts = []
    # Cosine-based starts
    for (a1, a2, a3) in [(0.3,0.1,0.0),(0.4,0.0,0.05),(0.2,0.2,0.1),
                          (0.35,0.15,0.0),(0.25,0.25,0.05),(0.4,0.1,0.1)]:
        starts.append(cosine_start(n, a1, a2, a3))
    # Perturbed best
    for amp in [0.03, 0.06, 0.10]:
        starts.append(smooth_perturb(best_h, n, 4, amp, rng))
    # Shifted best (circular by n//4)
    sh = np.roll(best_h, n//4)
    starts.append(project_feasible(sh, n))

    n_starts = len(starts)
    time_per_start = (t4_end - time.time()) / max(n_starts, 1)
    for si, h_s in enumerate(starts):
        if time.time() > t4_end - 1:
            break
        t_s = time.time() + max(0, time_per_start * 0.9)
        h_s, c5_s = try_candidate(h_s, n, t_s, top_k=12, sg_iters=2000, cd_sweeps=100)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 5: Polishing coordinate descent
    t5 = t0 + budget_s * 0.82
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=500)

    # Phase 6: Multi-resolution
    t6_end = t0 + budget_s * 0.93
    for n_try in [120, 200, 240, 320]:
        if time.time() > t6_end - 5:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = try_candidate(h_tr, n_try, t6_end, top_k=10,
                                     sg_iters=2000, cd_sweeps=150)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 7: Final coordinate descent
    if elapsed() < budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=400)

    best_h = project_feasible(best_h, len(best_h))
    # Verify sum constraint exactly
    target = len(best_h) / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - best_h
            best_h += room * (diff / room.sum())
        else:
            best_h *= (1.0 + diff / best_h.sum())
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
