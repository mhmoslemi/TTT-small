        h[a] = 0.0
        h[b] = 1.0
        corr = compute_corr(h)
    return h

def perturb_binary(h, n, n_swaps):
    h = h.copy()
    for _ in range(n_swaps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
    return h

def sa_binary(h, n, t_end):
    h = h.copy()
    corr = compute_corr(h)
    c5 = int(np.max(corr))
    best_c5 = c5
    best_h = h.copy()
    T = 2.0
    start = time.time()
    
    while time.time() < t_end and T > 0.01:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        nc = compute_corr(h)
        nc5 = int(np.max(nc))
        d = nc5 - c5
        if d <= 0 or np.random.rand() < np.exp(-d / T):
            corr = nc
            c5 = nc5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[b] = 0.0
            h[a] = 1.0
        T *= 0.999995
        
        if time.time() - start > 0.5:
            # Periodic local search
            h = ls_binary(h, n, time.time() + 0.1)
            corr = compute_corr(h)
            c5 = int(np.max(corr))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            start = time.time()
    
    return best_h, best_c5

def coord_descent_cont(h, n, t_end, max_sweeps=200):
    """Continuous coordinate descent."""
    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            
            def obj(v):
                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                return np.max(cn)
            
            v0 = obj(0.0)
            v1r = obj(1.0)
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            best_v = res.x
            best_f = res.fun
            if v0 < best_f:
                best_v, best_f = 0.0, v0
            if v1r < best_f:
                best_v, best_f = 1.0, v1r
            if best_f * dx < best_c5 - 1e-14:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = compute_corr(best_h)
                improved = True
        if not improved:
            break
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h), n_new)
