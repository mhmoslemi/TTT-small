def weighted_corr_grad(h, weights, n):
    """Compute gradient of sum_k weights[k] * corr[k]"""
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for k_idx, w in enumerate(weights):
        if w < 1e-12:
            continue
        lag = k_idx - center
        # r(lag) = sum_i h[i](1-h[i+lag])
        if lag >= 0:
            for i in range(n - lag):
                # h[i] appears as first term: contributes w*(1-h[i+lag])
                # h[i+lag] appears as second term: contributes -w*h[i]
                grad[i] += w * (1.0 - h[i + lag])
                grad[i + lag] += -w * h[i]
        else:
            l = -lag
            for i in range(l, n):
                grad[i] += w * (1.0 - h[i - l])
                grad[i - l] += -w * h[i]
    return grad * dx

- corr(k) = number of i where h[i]=1 and h[i+k]=0 (for valid i)

The sum of corr(k) over all positive k = sum_i h[i] * (number of zeros after position i)
For a "random" 0/1 pattern with 90 ones and 90 zeros, E[corr(k)] ≈ 90 * 90/(180-k) for each k.
For k=1: E[corr(1)] ≈ 90 * 90/179 ≈ 45.3
For k=90: E[corr(90)] ≈ 90 * 90/90 = 90... wait that can't be right.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -400, 400)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / T) * dx

def slsqp_smooth(h, T, n, maxiter=600):
    def obj(x):
        return smooth_c5(x, T)
    def con_f(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_f}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def slsqp_lag(h, target_idx, n, maxiter=800):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return c[target_idx] * dx
    def con_f(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_f}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    u = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = u
    n2 = 2 * n - 1
    center = n - 1
    coef = np.zeros(n2)
    for k in range(n2):
        lag = k - center
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k] = c
    dx = 2.0 / n
    def f1d(t):
        cv = cb + t * coef
        cv[center] = cv[center] - t * t
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=400):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            new_v = coord_opt(best_h, n, i)
            if abs(new_v - best_h[i]) > 1e-11:
                ht = best_h.copy()
                ht[i] = new_v
                ht = proj(ht, n)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-14:
                    best_h = ht
                    best_c5 = c5t
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def alt_worst_lag(h, n, t_end, n_iters=40):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        idx = int(np.argmax(corr))
        try:
            hc = slsqp_lag(best_h, idx, n, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    return best_h, best_c5

def weighted_grad_step(h, n, t_end, K=12, lr_init=0.008, max_iter=3000):
    dx = 2.0 / n
    center = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    stall = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = float(np.max(corr))
        top_k = np.argsort(corr)[-K:]
        vals = corr[top_k]
        w = np.maximum(vals - m * 0.99, 0.01)
        w = w / w.sum()
        grad = np.zeros(n)
        for kk, wi in zip(top_k, w):
            lag = kk - center
            if lag == 0:
                grad += wi * (1.0 - 2.0 * best_h)
            elif lag > 0:
                grad[:n - lag] += wi * (1.0 - best_h[lag:])
                grad[lag:] += -wi * best_h[:n - lag]
            else:
                l = -lag
                grad[l:] += wi * (1.0 - best_h[:-l])
                grad[:n - l] += -wi * best_h[l:]
        grad *= dx
        step = lr
        improved = False
        for _ in range(25):
            cand = proj(best_h - step * grad, n)
            c5n = compute_c5(cand)
            if c5n < best_c5 - 1e-15:
                best_h = cand
                best_c5 = c5n
                stall = 0
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.6
            stall += 1
            if stall > 80 or lr < 1e-13:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T
    for T in [30, 150, 800, 4000, 20000]:
        if time.time() > t_end * 0.08:
            break
        try:
            hc = slsqp_smooth(best_h, T, n, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Alternating worst-lag minimization
    if time.time() < t_end * 0.25:
        best_h, best_c5 = alt_worst_lag(best_h, n, t_end * 0.25, n_iters=30)

    # Phase 3: Weighted gradient descent on active lags
    if time.time() < t_end * 0.40:
        best_h, best_c5 = weighted_grad_step(best_h, n, t_end * 0.40, K=10, lr_init=0.01, max_iter=4000)

    # Phase 4: Coordinate descent
    if time.time() < t_end * 0.55:
        best_h, best_c5 = coord_descent(best_h, n, t_end * 0.55, max_sweeps=400)

    # Phase 5: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 60
    for it in range(n_bh):
        if time.time() > t_end * 0.75:
            break
        cool = 1.0 - 0.85 * (it / float(n_bh))
        step_size = 0.01 + 0.08 * cool
        hp = work_h + step_size * np.random.randn(n)
        hp = proj(hp, n)
        try:
            hc = slsqp_smooth(hp, 2000, n, maxiter=250)
            c5v = compute_c5(hc)
            temp = 0.0015 * cool + 0.00005
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: More coordinate descent + alt worst lag
    if time.time() < t_end * 0.85:
        best_h, best_c5 = coord_descent(best_h, n, t_end * 0.80, max_sweeps=300)
    if time.time() < t_end * 0.90:
        best_h, best_c5 = alt_worst_lag(best_h, n, t_end * 0.90, n_iters=15)

    # Phase 7: Try different resolutions
    for n_try in [160, 200]:
        if time.time() > t_end * 0.96:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        hi = np.interp(x_new, x_old, best_h)
        hi = proj(hi, n_try)
        for T in [500, 5000]:
            if time.time() > t_end * 0.97:
                break
            try:
                hc = slsqp_smooth(hi, T, n_try, maxiter=500)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
                    n = n_try
            except Exception:
                pass

    # Phase 8: Final high-T SLSQP + coord
    if time.time() < t_end * 0.99:
        for T in [50000, 200000]:
            if time.time() > t_end * 0.99:
                break
            try:
                hc = slsqp_smooth(best_h, T, n, maxiter=800)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass
    best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=100)

    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
