import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(vals, T):
    m = np.max(vals)
    if T < 1e-6:
        return m
    return m + float(np.log(np.sum(np.exp(T * (vals - m)))) / T)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, n_sweeps=200, verbose=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            k = np.arange(-(n-1), n)
            j1 = i - k
            j2 = i + k
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(2*n-1)
            coef[v1] += 1.0 - best_h[j1[v1]]
            coef[v2] += -best_h[j2[v2]]
            
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            
            def f1d(s):
                return np.max(corr + s * coef)
            
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10})
            s_new = res.x
            if abs(s_new) > 1e-12:
                best_h[i] += s_new
                corr = corr + s_new * coef
                c5_new = float(np.max(corr)) * dx
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved += 1
        if improved == 0:
            break
    return best_h, best_c5

def nesterov_grad(h, n, t_end, T=5000, lr_init=0.01, max_iters=5000):

    v = np.zeros(n)
    mu = 0.9
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    target = n / 2.0
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        x = project_to_sum(x, target)
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr * dx
        
        # Subgradient of smooth_max
        exps = np.exp(T * (vals - np.max(vals)))
        weights = exps / exps.sum()
        # Gradient of each corr[k] w.r.t. x
        # d corr[k] / dx[j] = (1-x[j-k]) - x[j+k] (with validity)
        # This is O(n²) to compute fully. Use FFT instead?
        # For n=224, O(n²) = 50K, which is fast.
        
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            for kk in range(-(n-1), n):
                w = weights[kk + n - 1]
                if w < 1e-10:
                    continue
                jk1 = j - kk

                c = 0.0
                if 0 <= jk1 < n:
                    c += 1.0 - x[jk1]
                if 0 <= jk2 < n:
                    c -= x[jk2]
                g += w * c
            grad[j] = g * dx
        
        # Nesterov step
        x_new = project_to_sum(x - lr * grad, target)
        # Actually Nesterov: look-ahead
        x_look = project_to_sum(x - lr * (grad + (x - x_prev) / lr - grad) if False else ...)

def compute_grad_smooth(x, n, dx, weights):
    # weights has shape (2n-1), indexed by k+n-1
    # grad[j] = dx * sum_k weights[k+n-1] * ((1-x[j-k]) - x[j+k])
    k = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each j, sum over k:
    # term1[j] = sum_k weights * (1 - x[j-k]) * [0<=j-k<n]
    # term2[j] = sum_k weights * x[j+k] * [0<=j+k<n]
    
    j = np.arange(n)  # shape (n,)
    # j-k has shape (n, 2n-1)
    jm = j[:, None] - k[None, :]  # j - k
    jp = j[:, None] + k[None, :]  # j + k
    
    valid_m = (jm >= 0) & (jm < n)  # (n, 2n-1)
    valid_p = (jp >= 0) & (jp < n)
    
    # term1[j] = sum_k weights[k] * (1 - x[j-k]) * valid_m
    # = sum_k weights[k] * valid_m - sum_k weights[k] * x[j-k] * valid_m
    w = weights  # (2n-1,)
    
    term1 = np.sum(w[None, :] * valid_m, axis=1)
    # For x[j-k] part:
    xm = np.where(valid_m, x[np.clip(jm, 0, n-1)], 0.0)  # (n, 2n-1)
    term1_x = np.sum(w[None, :] * xm, axis=1)
    
    term2 = np.sum(w[None, :] * np.where(valid_p, x[np.clip(jp, 0, n-1)], 0.0), axis=1)
    
    return dx * (term1 - term1_x - term2)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(vals, T):
    m = np.max(vals)
    return m + float(np.log(np.sum(np.exp(T * (vals - m)))) / T)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def slsqp_polish(h, T, n, maxiter=500, ftol=1e-13):
    dx = 2.0 / n
    def obj(x):
        return smooth_max(corr_full(np.clip(x, 0, 1)), T) * dx
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        vals = corr_full(x) * dx
        m = np.max(vals)
        exps = np.exp(T * (vals - m))
        w = exps / exps.sum()
        # gradient
        grad = np.zeros(n)
        k = np.arange(-(n-1), n)
        j = np.arange(n)
        jm = j[:, None] - k[None, :]
        jp = j[:, None] + k[None, :]
        vm = (jm >= 0) & (jm < n)
        vp = (jp >= 0) & (jp < n)
        # term from (1-x[j-k]):
        xm = np.where(vm, x[np.clip(jm, 0, n-1)], 0.0)
        t1 = np.sum(w[None, :] * vm - w[None, :] * xm, axis=1)
        # term from x[j+k]:
        xp = np.where(vp, x[np.clip(jp, 0, n-1)], 0.0)
        t2 = np.sum(w[None, :] * xp, axis=1)
        return dx * (t1 - t2)
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0,
                                 'jac':lambda x: np.ones(n)}],
                   options={'maxiter':maxiter,'ftol':ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n/2.0)

    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = corr_full(best_h)
    k_vals = np.arange(-(n-1), n)
    j_idx = np.arange(n)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        order = np.random.permutation(n)
        improved = 0
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            jm = i - k_vals
            jp = i + k_vals
            vm = (jm >= 0) & (jm < n)
            vp = (jp >= 0) & (jp < n)
            coef = np.zeros(2*n-1)
            coef[vm] += 1.0 - best_h[jm[vm]]
            coef[vp] += -best_h[jp[vp]]
            
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-14:
                continue
            
            def f1d(s):
                return np.max(corr + s * coef)
            
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10})
            s_new = float(res.x)
            if abs(s_new) > 1e-12:
                best_h[i] += s_new
                corr += s_new * coef
                c5_new = float(np.max(corr)) * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved += 1
        if improved == 0:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(vals, T):
    m = np.max(vals)
    return m + float(np.log(np.sum(np.exp(T * (vals - m)))) / T)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def slsqp_grad(h, T, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        return smooth_max(corr_full(x), T) * dx
    def obj_g(x):
        x = np.clip(x, 0, 1)
        vals = corr_full(x) * dx
        m = np.max(vals)
        exps = np.exp(T * (vals - m))
        w = exps / exps.sum()
        k_arr = np.arange(-(n-1), n)
        j_arr = np.arange(n)
        jm = j_arr[:, None] - k_arr[None, :]
        jp = j_arr[:, None] + k_arr[None, :]
        vm = (jm >= 0) & (jm < n)
        vp = (jp >= 0) & (jp < n)
        xm = np.where(vm, x[np.clip(jm, 0, n-1)], 0.0)
        xp = np.where(vp, x[np.clip(jp, 0, n-1)], 0.0)
        t1 = np.sum(w[None, :] * vm.astype(float) - w[None, :] * xm, axis=1)
        t2 = np.sum(w[None, :] * xp, axis=1)
        return dx * (t1 - t2)
    c_fn = lambda x: x.sum() - n / 2.0
    c_jac = lambda x: np.ones(n)
    res = minimize(obj, h, jac=obj_g, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':c_fn,'jac':c_jac}],
                   options={'maxiter':maxiter,'ftol':ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n/2.0)

def coord_descent(h, n, t_end, n_sweeps=200):
    bh = h.copy()
    bc5 = compute_c5(bh)
    dx = 2.0 / n
    corr = corr_full(bh).copy()
    k_vals = np.arange(-(n-1), n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        order = np.random.permutation(n)
        improved = 0
        for i in order:
            if time.time() > t_end:
                return bh, bc5
            ho = bh[i]
            jm = i - k_vals
            jp = i + k_vals
            vm = (jm >= 0) & (jm < n)
            vp = (jp >= 0) & (jp < n)
            coef = np.zeros(2*n-1)
            coef[vm] += 1.0 - bh[jm[vm]]
            coef[vp] += -bh[jp[vp]]
            lo, hi = -ho, 1.0 - ho
            if hi - lo < 1e-14:
                continue
            def f1d(s):
                return np.max(corr + s * coef)
            rs = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                 options={'xatol': 1e-10})
            sn = float(rs.x)
            if abs(sn) > 1e-12:
                bh[i] += sn
                corr += sn * coef
                c5n = float(np.max(corr)) * dx
                if c5n < bc5 - 1e-15:
                    bc5 = c5n
                    improved += 1
        if improved == 0:
            break
    return bh, bc5

def lagrangian_opt(h, n, t_end, n_outer=100, n_inner=40):
    dx = 2.0 / n
    bh = h.copy()
    bc5 = compute_c5(bh)
    lw = np.ones(2*n-1) / (2*n-1)
    j_arr = np.arange(n)
    k_arr = np.arange(-(n-1), n)
    jm = j_arr[:, None] - k_arr[None, :]
    jp = j_arr[:, None] + k_arr[None, :]
    vm = ((jm >= 0) & (jm < n)).astype(float)
    vp = ((jp >= 0) & (jp < n)).astype(float)
    idx_m = np.clip(jm, 0, n-1)
    idx_p = np.clip(jp, 0, n-1)
    hi_vals = lw[n-1:]
    lo_rev = lw[:n][::-1]
    for outer in range(n_outer):
        if time.time() > t_end: break
        hc = bh.copy()
        for inn in range(n_inner):
            if time.time() > t_end: break
            mo_h = 1.0 - hc
            t1 = np.einsum('j,ij,i->j', mo_h, vm, hi_vals)
            xm_val = np.where(vm > 0, hc[np.broadcast_to(idx_m.reshape(n, -1), (n, 2*n-1))], 0.0)
            t1x = np.sum(hi_vals[None, :] * vm * mo_h[np.broadcast_to(idx_m.reshape(n,-1),(n,2*n-1)], 0) if False else np.zeros(n), axis=1)
            # Simpler: direct computation
            t1 = np.zeros(n)
            t2 = np.zeros(n)
            for kp in range(n):
                if kp < n:
                    wv = hi_vals[kp]
                    if wv > 1e-15:
                        t1[:n-kp] += wv * mo_h[kp:]
            for m in range(n):
                wv = lo_rev[m]
                if wv > 1e-15:
                    t2[m:] += wv * hc[:n-m]
            grad = dx * (t1 - t2)
            lr = 0.5 / (np.max(np.abs(grad)) + 1e-10)
            hc = project_to_sum(hc - lr * grad, n/2)
            c5c = compute_c5(hc)
            if c5c < bc5:
                bc5 = c5c
                bh = hc.copy()
        bh = hc
        corr_v = corr_full(bh) * dx
        eta = 0.02
        lw = np.exp(-corr_v / eta)
        lw /= lw.sum()
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(128)
        best_h[:64] = 1.0
    n = len(best_h)
    best_h = project_to_sum(best_h, n/2)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial CD (60s)
    t1 = t0 + 60
    best_h, best_c5 = coord_descent(best_h, n, t1, 80)

    # Phase 2: SLSQP with gradient (90s)
    for T in [1000, 5000]:
        if time.time() > t0 + 150: break
        try:
            hc = slsqp_grad(best_h, T, n, 600, 1e-13)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 3: Lagrangian dual optimization (120s)
    t3 = t0 + 270
    try:
        hl, cl = lagrangian_opt(best_h, n, t3, 100, 35)
        if cl < best_c5:
            best_h, best_c5 = hl, cl
    except Exception:
        pass

    # Phase 4: Basin hopping (55% budget ≈ 605s)
    work_h = best_h.copy()
    work_c5 = best_c5
    hop_end = t0 + budget_s * 0.65
    n_hops = 0
    while time.time() < hop_end:
        n_hops += 1
        cool = max(1.0 - (n_hops / 1500.0), 0.02)
        r = np.random.rand()
        if r < 0.25:
            h_p = work_h + (0.12 * cool + 0.06) * np.random.randn(n)
        elif r < 0.45:
            h_p = work_h.copy()
            bs = np.random.randint(0, max(1, n - 15))
            bl = np.random.randint(5, min(20, n - bs))
            h_p[bs:bs+bl] += (0.25 * cool + 0.1) * np.random.randn(bl)
        elif r < 0.65:
            h_p = work_h.copy()
            corr_v = corr_full(work_h)
            k_max = int(np.argmax(corr_v)) - (n-1)
            mask = np.zeros(n, dtype=bool)
            for ii in range(n):
                if 0 <= ii + k_max < n:
                    mask[ii] = True
            idx = np.where(mask)[0]
            if len(idx) > 0:
                h_p[idx] += (0.15 * cool + 0.05) * np.random.randn(len(idx))
        elif r < 0.85:
            h_p = work_h.copy()
            bs = np.random.randint(0, max(1, n - 12))
            bl = np.random.randint(4, min(14, n - bs))
            h_p[bs:bs+bl] = 1.0 - h_p[bs:bs+bl]
        else:
            h_p = work_h.copy()
            for _ in range(np.random.randint(3, 8)):
                a = np.random.randint(0, n - 5)
                b = a + np.random.randint(5, min(20, n - a))
                h_p[a:b], h_p[b:min(b+20, n)] = h_p[b:min(b+20, n)].copy(), h_p[a:b].copy()
        h_p = project_to_sum(h_p, n/2)
        hc, c5v = coord_descent(h_p, n, time.time() + 2.5, 8)
        temp = 0.004 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Heavy_CD + final SLSQP (25% budget)

    best_h, best_c5 = coord_descent(best_h, n, t5, 200)

            pass

    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.997, 60)

    best_h = project_to_sum(best_h, len(best_h) / 2)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
