def compute_c5(h):
    n = len(h)
    corr = np.correlate(h.astype(float), 1.0-h, mode='full')
    return float(np.max(corr) * 2.0/n)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    d = target - s
    if d > 0:
        r = (1.0 - h).sum()
        if r < 1e-13: return h
        h += (1.0-h) * (d/r)
    else:
        if s < 1e-13: return h
        h *= target/s
    return np.clip(h, 0.0, 1.0)

def _golden_min(corr, coef, lo, hi):
    gr = (np.sqrt(5.0)-1.0)/2.0
    x1, x2 = hi-gr*(hi-lo), lo+gr*(hi-lo)
    f1, f2 = float(np.max(corr+x1*coef)), float(np.max(corr+x2*coef))
    for _ in range(50):

            x1 = hi-gr*(hi-lo); f1 = float(np.max(corr+x1*coef))

            x2 = lo+gr*(hi-lo); f2 = float(np.max(corr+x2*coef))
    return (lo+hi)*0.5

from scipy.special import logsumexp

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    d = target - s
    if d > 0:
        r = (1.0 - h).sum()
        if r < 1e-13:
            return h
        h += (1.0 - h) * (d / r)
    else:
        if s < 1e-13:
            return h
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def _golden_min(corr, coef, lo, hi):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = float(np.max(corr + x1 * coef))
    f2 = float(np.max(corr + x2 * coef))
    for _ in range(55):
        if f1 < f2:
            hi, x2, f2 = x2, x1, f1
            x1 = hi - gr * (hi - lo)
            f1 = float(np.max(corr + x1 * coef))
        else:
            lo, x1, f1 = x1, x2, f2
            x2 = lo + gr * (hi - lo)
            f2 = float(np.max(corr + x2 * coef))
    return (lo + hi) * 0.5

def coord_descent(h, t_end, max_sweeps=10000):
    n = len(h)
    dx = 2.0 / n
    bh = h.copy().astype(float)
    corr = np.correlate(bh, 1.0 - bh, mode='full').astype(float)
    bc5 = float(np.max(corr) * dx)
    shifts = np.arange(-(n - 1), n)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = 0
        for i in range(n):
            if time.time() > t_end:
                return bh, bc5
            ho = bh[i]
            i1 = i - shifts
            i2 = i + shifts
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            cf = np.zeros(2 * n - 1)
            if v1.any():
                cf[v1] += 1.0 - bh[i1[v1]]
            if v2.any():

            lo = -ho
            hiv = 1.0 - ho
            if hiv - lo < 1e-14:

                    imp += 1
        if imp == 0:

def _sm_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    v = np.clip(_g_T * (corr - m), -50.0, 50.0)
    return float(m + logsumexp(v) / _g_T) * _g_dx

def _sm_con(x):
    return float(x.sum() - _g_target)

def slsqp_polish(h, T, maxiter=500, ftol=1e-14):
    global _g_T, _g_dx, _g_target
    n = len(h)

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sm_con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)
    except Exception:
        return h
