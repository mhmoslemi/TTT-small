    """Project to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h.copy(), 0.0, 1.0)

        return h

        if s > 1e-12:

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += room * (diff / tot)
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h *= (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_val(corr, T):
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return m + np.log(np.exp(shifted).sum()) / T

def soft_max_grad(corr, h, T, n):
    dx = 2.0 / n
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            cnt = n - lag
            if cnt > 0:
                grad[:cnt] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:cnt])
        else:
            nl = -lag
            cnt = n - nl
            if cnt > 0:
                grad[nl:] += pk * (1.0 - h[:cnt])
                grad[:cnt] += pk * (-h[nl:])
    return grad * dx

def coord_step(h, n, i):
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    k0 = n - 1
    dx = 2.0 / n
    coeff = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        k = lag + n - 1
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if lag != 0 and 0 <= j2 < n:
            c += (-h[j2])
        coeff[k] = c
    def f1d(t):
        vals = corr0 + t * coeff
        vals[k0] = corr0[k0] + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    h[i] = res.x
    return h[i]

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            t_new = coord_step(best_h, n, i)
            if abs(t_new - best_h[i]) > 1e-9:
                best_h[i] = t_new
                best_h = proj(best_h, n)
                c5_t = compute_c5(best_h)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_smooth(h, n, T, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_val(corr, T) * dx
    def obj_g(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_grad(corr, x, T, n)
    def con_f(x):
        return x.sum() - n / 2.0
    def con_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=obj_g, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

def grad_refine(h, n, t_end, T=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    lr = 0.003
    stag = 0
    while time.time() < t_end:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        g = soft_max_grad(corr, best_h, T, n)
        step = lr
        improved = False
        for _ in range(20):
            trial = proj(best_h - step * g, n)
            c5_t = compute_c5(trial)
            if c5_t < best_c5 - 1e-14:
                best_h = trial
                best_c5 = c5_t
                improved = True
                T = min(T * 1.04, 60000)
                stag = 0
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
            stag += 1
            if lr < 1e-11 or stag > 40:
                break
    return best_h, best_c5

def targeted_perturb(h, n, t_end, n_iter=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / float(n_iter)
        n_pert = np.random.choice([2, 3, 4, 5, 6], p=[0.2, 0.3, 0.25, 0.15, 0.1])
        idx = np.random.choice(n, size=n_pert, replace=False)
        h_try = best_h.copy()
        for i in idx:
            h_try[i] = np.random.rand()
        h_try = proj(h_try, n)
        t_sub = min(time.time() + (t_end - time.time()) / max(n_iter - it, 1) * 0.7, t_end)
        h_try, c5_try = coord_descent(h_try, n, t_sub, max_sweeps=15)
        temp = 0.0008 * cool + 0.00005
        if c5_try < best_c5 or (c5_try > best_c5 and np.random.rand() < np.exp(-(c5_try - best_c5) / max(temp, 1e-10))):
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5

def snap_binary(h, n, tol=0.02):
    h = h.copy()
    h[h < tol] = 0.0
    h[h > 1.0 - tol] = 1.0
    return proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
