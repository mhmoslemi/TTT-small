    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, t_end, max_sweeps=2000):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i + ks
            idx2 = i - ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            def obj(v):
                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + v * (1.0 - v) - old * (1.0 - old)
                return np.max(cn) * dx
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})

The objective is C₅ = max_k c_k where c_k = Σ_i h[i](1-h[i+k])*dx.

At the optimum, the active lags (those achieving the max) should all be equal. Call this value C*. 

We can set up a feasibility problem: find h such that c_k ≤ C* for all k, ∫h = 1, 0≤h≤1.
Then minimize C* via bisection.

But the constraint c_k ≤ C* is:
Σ_i h[i](1-h[i+k])*dx ≤ C*
Σ_i h[i]*dx - Σ_i h[i]*h[i+k]*dx ≤ C*

This is quadratic in h (the h[i]*h[i+k] term), so it's not a simple linear constraint.
