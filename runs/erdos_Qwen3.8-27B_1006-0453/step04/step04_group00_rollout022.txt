                corr_full = np.correlate(best_h, 1.0 - best_h, mode='full')
                c5_try = float(np.max(corr_full) * 2.0 / n)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    improved = True
                else:
                    best_h[i] = h_i
                    corr_full = np.correlate(best_h, 1.0 - best_h, mode='full')
        if not improved:
            break
    return best_h, best_c5

def remez_equate(h, n, t_end):
    h_work = h.copy()
    best_c5 = compute_c5(h_work)
    while time.time() < t_end:
        corr = np.correlate(h_work, 1.0 - h_work, mode='full')
        mx = np.max(corr)
        thresh = mx - 0.05 * mx
        active = np.where(corr > thresh)[0]
        if len(active) < 2:
            break
        active_sum = corr[active].sum() / len(active)
        T_r = 50.0 + 200.0 * len(active)
        h_r = slsqp_opt(h_work, T_r, n, maxiter=300, ftol=1e-13)
        c5v = compute_c5(h_r)
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            h_work = h_r
        else:
            break
    return h_work, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    parent_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_h = project_h(parent_h)

    best_h = None
    best_c5 = 1.0
    best_n = 160

    if parent_h is not None:
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = len(parent_h)

    n_list = [80, 120, 160, 200, 240]
    if best_n not in n_list:
        n_list.append(best_n)
    n_list = sorted(set(n_list))

    for n in n_list:
        if time.time() - t0 > budget_s * 0.18:
            break
        inits = []
        if parent_h is not None:
            if len(parent_h) == n:
                inits.append(parent_h.copy())
            else:
                old_n = len(parent_h)
                new_x = np.linspace(0, old_n - 1, n)
                h_interp = np.interp(new_x, np.arange(old_n), parent_h)
                inits.append(project_h(h_interp))
        inits.append(gen_binary_init(n, seed + n * 7))
        inits.append(gen_periodic_init(n, seed + 3000 + n))
        inits.append(gen_fourier_init(n, seed + 5000 + n))
        for init in inits:
            if time.time() - t0 > budget_s * 0.18:
                break
            hw = init.copy()
            for T in [30, 100, 500, 2000, 8000, 30000]:
                if time.time() - t0 > budget_s * 0.18:
                    break
                hw = slsqp_opt(hw, T, n, maxiter=350, ftol=1e-13)
                c5v = compute_c5(hw)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hw.copy()
                    best_n = n

    n = best_n
    work_h = best_h.copy()
    work_c5 = best_c5

    for it in range(50):
        if time.time() - t0 > budget_s * 0.48:
            break
        cool = 1.0 - 0.9 * (it / 50.0)
        mod = it % 5
        if mod == 0:
            pert = (0.03 + 0.10 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.02 + 0.08 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = (0.06 + 0.12 * cool) * np.random.randn(int(mask.sum()))
        elif mod == 3:
            b1 = np.random.randint(0, n - 8)
            b2 = np.random.randint(0, n - 8)
            L = np.random.randint(3, min(8, n // 5))
            pert = np.zeros(n)
            pert[b1:b1+L] = work_h[b2:b2+L] - work_h[b1:b1+L]
            pert[b2:b2+L] = work_h[b1:b1+L] - work_h[b2:b2+L]
        else:
            pert = (0.015 + 0.05 * cool) * np.random.randn(n)
        h_p = project_h(work_h + pert)
        T_bh = 200 + 6000 * cool
        hc = slsqp_opt(h_p, T_bh, n, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0002
        if c5v < work_c5 or (np.random.rand() < np.exp(-(c5v - work_c5) / temp) and c5v < work_c5 + 0.004):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = work_h.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    best_h, best_c5 = coord_refine(best_h, len(best_h), t0 + budget_s * 0.65, max_sweeps=80)

    best_h, best_c5 = remez_equate(best_h, len(best_h), t0 + budget_s * 0.72)

    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(30):
        if time.time() - t0 > budget_s * 0.82:
            break
        cool = 1.0 - 0.8 * (it / 30.0)
        step = 0.01 + 0.04 * cool
        h_p = project_h(work_h + step * np.random.randn(n))
        T_bh = 2000 + 10000 * cool
        hc = slsqp_opt(h_p, T_bh, n, maxiter=350, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0005 * cool + 0.0001
        if c5v < work_c5 or (np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = work_h.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    best_h, best_c5 = coord_refine(best_h, len(best_h), t0 + budget_s * 0.92, max_sweeps=100)

    if time.time() - t0 < budget_s * 0.96:
        for T in [100000, 500000]:
            if time.time() - t0 > budget_s * 0.97:
                break
            hc = slsqp_opt(best_h, T, len(best_h), maxiter=1500, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    if time.time() - t0 < budget_s * 0.98:
        best_h, best_c5 = coord_refine(best_h, len(best_h), t0 + budget_s * 0.985, max_sweeps=40)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
