def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

    dx = 2.0 / n
    # Precompute coefficients
    # out[k] changes by (h_i_new - h_i_old) * coef[k]
    # where coef[k] = (1-h[i+k-n+1]) - h[i+k-n+1] for valid indices
    # = 1 - 2*h[...] for valid, else partial
    
    k_range = np.arange(-(n - 1), n)  # k - (n-1) ranges from -(n-1) to n-1
    
    # Contribution 1: j=i, k such that i+k-(n-1) is in [0,n-1]
    idx1 = i + k_range - (n - 1)  # = i + k - n + 1
    valid1 = (idx1 >= 0) & (idx1 < n)
    
    # Contribution 2: j+k-(n-1)=i, i.e., j = i+k-(n-1), need j in [0,n-1]
    idx2 = i + k_range - (n - 1)  # same index! j = i + k - n + 1
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    # coef[k] = (1-h[idx1])*valid1 - h[idx2]*valid2 (where both use same index)
    idx_c = np.clip(idx1, 0, n - 1)
    coef = (1.0 - h[idx_c]) * valid1 - h[idx_c] * valid2
    coef = coef.astype(np.float64)

    # = max_k ((corr[k] - h[i]*coef[k]) + t * coef[k]) * dx
    base = corr - h[i] * coef

        return np.max(base + t * coef) * dx

                          options={'xatol': 1e-13})
    return res.x, res.fun / dx  # return value and unnormalized f

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n

        T = max(T_min, T * 0.97)
        improved = False
        order = np.random.permutation(n)

                return best_h, best_c5
            hi_new, f_new_norm = fast_coord_opt(corr, best_h, i, n)
            # f_new_norm is the max correlation (not multiplied by dx)
            if abs(hi_new - best_h[i]) < 1e-14:

            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            if h_try[i] != hi_new:  # projection changed the value

                f_try_norm = c5_try / dx
            else:
                f_try_norm = f_new_norm
            
            if f_try_norm < max_corr - 1e-14:

                max_corr = f_try_norm
                if max_corr * dx < best_c5:
                    best_c5 = max_corr * dx
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                max_corr = np.max(corr)
                improved = True
            elif f_try_norm > max_corr:

                if np.random.rand() < np.exp(-(f_try_norm - max_corr) / T):

                    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                    max_corr = np.max(corr)
                    improved = True
        if not improved and T < T_min * 2:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def fast_coord_step(corr, h, i, n):
