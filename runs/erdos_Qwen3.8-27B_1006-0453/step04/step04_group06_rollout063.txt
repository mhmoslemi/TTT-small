        return np.max(vals)

Let me also think about the SLSQP gradient computation:

The smooth objective is: f_T(h) = (1/T) log(sum_k exp(T * c_k(h))) (plus offset for numerical stability)

The gradient is: ∂f_T/∂h[i] = sum_k (c_k'(i) * exp(T*c_k)) / sum_k exp(T*c_k)

where c_k'(i) is the derivative of c_k with respect to h[i].

c_k = sum_j h[j](1-h[j+k])
∂c_k/∂h[i] = (1-h[i+k]) - h[i-k] (for the two affected terms, with boundary handling)

    n = len(h)
    return np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    diff = target - s
    if diff > 0:
        room = (1.0 - h)
        tot = room.sum()
        if tot < 1e-12:
            return np.clip(h * target / max(s, 1e-12), 0, 1)
        h += room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h *= target / s
    return np.clip(h, 0.0, 1.0)

def golden1d(f, lo=0.0, hi=1.0, iters=50):
    gr = (np.sqrt(5.0) - 1.0) * 0.5

    d = lo + gr*(hi-lo)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return np.clip(h * target / max(s, 1e-12), 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def golden1d(f, lo=0.0, hi=1.0, iters=50):
    gr = (np.sqrt(5.0) - 1.0) * 0.5
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc, fd = f(c), f(d)
    for _ in range(iters):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def cd_sweep(h, n, t_end, order=None):
    center = n - 1
    best_c5 = compute_c5(h)
    if order is None:
        order = np.random.permutation(n)
    improved = False
    for i in order:
        if time.time() > t_end:
            break
        hi_save = h[i]
        h[i] = 0.0
        cb = np.correlate(h, 1.0 - h, mode='full')
        h[i] = hi_save
        
        a = np.zeros(2 * n - 1)
        for k in range(1, n):
            ip, im = i + k, i - k
            a[center + k] = ((1.0 - h[ip]) if ip < n else 0.0) - (h[im] if im >= 0 else 0.0)
            a[center - k] = ((1.0 - h[im]) if im >= 0 else 0.0) - (h[ip] if ip < n else 0.0)
        
        ct = cb[center]
        x_opt = golden1d(lambda x: max(np.max(cb + x * a), ct + x - x * x))
        h[i] = x_opt
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            improved = True
    return h, best_c5, improved

def smooth_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / len(h))

def smooth_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    weights = np.exp(T * (corr - m))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    center = n - 1
    for idx in range(len(corr)):
        k = idx - center
        if k == 0:
            grad += weights[idx] * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += weights[idx] * (1.0 - h[k:])
            grad[k:] -= weights[idx] * h[:n-k]
        else:
            mk = -k
            grad[mk:] += weights[idx] * (1.0 - h[:n-mk])
            grad[:n-mk] -= weights[idx] * h[mk:]
    return grad * (2.0 / n)

def slsqp_opt(h, T, n, maxiter=300):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(smooth_obj, h, jac=smooth_grad, args=(T,),
                   method='SLSQP', bounds=[(0,1)]*n, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
