```python
        h = h * (1.0 + diff / s)

    grad = np.zeros(n)

        shift = k - (n-1)  # this is the lag
        if abs(p[k]) < 1e-15:
            continue
        # (1-h[m-shift]) for valid m-shift: m from max(0,shift) to min(n-1, n-1+shift)
        # -h[m+shift] for valid m+shift: m from max(0,-shift) to min(n-1, n-1-shift)
        lo1 = max(0, shift)
        hi1 = min(n-1, n-1+shift)
        if lo1 <= hi1:
            m_idx = np.arange(lo1, hi1+1)
            grad[m_idx] += p[k] * (1.0 - h[m_idx - shift])
        lo2 = max(0, -shift)
        hi2 = min(n-1, n-1-shift)
        if lo2 <= hi2:
            m_idx = np.arange(lo2, hi2+1)
            grad[m_idx] -= p[k] * h[m_idx + shift]
    return grad * dx
```

```python
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, T, n):
    """Return (objective, gradient) for smoothed C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    obj_val = (m + np.log(np.sum(exps)) / T) * dx
    # Gradient
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        s = k - (n - 1)
        # First term: +pk * (1 - h[m-s]) for valid m
        lo1 = max(0, s)
        hi1 = min(n - 1, n - 1 + s)
        if lo1 <= hi1:
            g[lo1:hi1+1] += pk * (1.0 - h[lo1 - s:hi1 - s + 1])
        # Second term: -pk * h[m+s] for valid m
        lo2 = max(0, -s)
        hi2 = min(n - 1, n - 1 - s)
        if lo2 <= hi2:
            g[lo2:hi2+1] -= pk * h[lo2 + s:hi2 + s + 1]
    g *= dx
    return obj_val, g

def slsqp_opt(h0, T, n, maxiter=200):
    def obj(x):
        v, _ = smooth_obj_grad(x, T, n)
        return v
    def grad(x):
        _, g = smooth_obj_grad(x, T, n)
        return g
    def con_fun(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    c0_idx = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            ks = np.arange(-(n - 1), n)
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            # Minimize max(corr + (v-old)*coeff, with c0 adjusted quadratically)
            def obj_fn(v):
                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                return np.max(cn)
            res = minimize_scalar(obj_fn, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            best_v = res.x
            best_f = res.fun
            # Check endpoints
            f0 = obj_fn(0.0)
            f1 = obj_fn(1.0)
            if f0 < best_f:
                best_v, best_f = 0.0, f0
            if f1 < best_f:
                best_v, best_f = 1.0, f1
            if best_f * dx < best_c5 - 1e-13:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_local_search(h_bin, n, t_end, max_rounds=1000):

    dx = 2.0 / n

        improved = False
        zeros = np.where(h_b < 0.5)[0]

        best_val = best_c5
        best_pair = None

                # At t where t+s=i: h changes 0->1, contrib: +(1-h[t])

                # At t=i: (1-h[t]) changes 1->0, contrib: -h[i+s]
                # At t=j: (1-h[t]) changes 0->1, contrib: +h[j+s]
                delta = np.zeros(2*n-1)
                s_vals = np.arange(2*n-1) - (n-1)

                # Actually t ranges 0..n-1-s for s>=0, or -s..n-1 for s<0
                # Let me just do it simply:
                for kk in range(2*n-1):

                    if 0 <= i + s < n and i >= 0:

                    if 0 <= i - s < n:
                        d += (1.0 - h_b[i - s])

            h_b[i] = 1.0
            h_b[j] = 0.0
            corr = corr + delta
            best_c5 = best_val
            improved = True
        if not improved:
```

```python
            ik_m = i - shifts  # i-s for each shift
            valid_p = (ik_p >= 0) & (ik_p < n)
            valid_m = (ik_m >= 0) & (ik_m < n)
            const = np.zeros(2*n-1)
            const[valid_p] -= h_b[ik_p[valid_p]]
            const[valid_m] += (1.0 - h_b[ik_m[valid_m]])
            
            # For each j in ones:
            # delta[k] = const[k] + h[j+s_k] (if valid) - (1-h[j-s_k]) (if valid)
            for j in ones:
                jk_p = j + shifts
                jk_m = j - shifts
                valid_jp = (jk_p >= 0) & (jk_p < n)
                valid_jm = (jk_m >= 0) & (jk_m < n)
                var = np.zeros(2*n-1)
                var[valid_jp] += h_b[jk_p[valid_jp]]
                var[valid_jm] -= (1.0 - h_b[jk_m[valid_jm]])
                new_corr = corr + const + var
                v = np.max(new_corr) * dx
                if v < best_val - 1e-14:
                    best_val = v
                    best_pair = (i, j)
                    best_delta = (const + var).copy()
        if best_pair is not None:
            i, j = best_pair
            h_b[i] = 1.0
            h_b[j] = 0.0
            corr = corr + best_delta
            best_c5 = best_val
        else:
            break
    return h_b, best_c5
```

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def smooth_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    p = np.exp(T * (corr - m))
    p /= p.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-16:
            continue
        s = k - (n - 1)
        lo1, hi1 = max(0, s), min(n - 1, n - 1 + s)
        if lo1 <= hi1:
            g[lo1:hi1+1] += p[k] * (1.0 - h[lo1-s:hi1-s+1])
        lo2, hi2 = max(0, -s), min(n - 1, n - 1 - s)
        if lo2 <= hi2:
            g[lo2:hi2+1] -= p[k] * h[lo2+s:hi2+s+1]
    return g * dx

def slsqp_opt(h0, T, n, maxiter=200):
    def obj(x):
        return smooth_obj(x, T, n)
    def jac(x):
        return smooth_grad(x, T, n)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h0, jac=jac, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    c0_idx = n - 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            ks = np.arange(-(n - 1), n)
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            # Vectorized 1D minimization
            vs = np.linspace(0, 1, 201)
            cn = corr[None, :] + (vs[:, None] - old) * coeff[None, :]
            cn[:, c0_idx] = c0_base + (vs - old) * (1.0 - vs - old)
            vals = cn.max(axis=1) * dx
            bi = vals.argmin()
            best_f = vals[bi]
            if best_f < best_c5 - 1e-13:
                best_h[i] = vs[bi]
                best_c5 = best_f
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_local_search(h_bin, n, t_end, max_rounds=500):
    dx = 2.0 / n
    h_b = h_bin.copy().astype(float)
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = np.max(corr) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    for _ in range(max_rounds):
        if time.time() > t_end:
            break
        zeros = np.where(h_b < 0.5)[0]
        ones = np.where(h_b >= 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_val = best_c5
        best_ij = None
        for i in zeros:
            ik_p = i + shifts
            ik_m = i - shifts
            vp = (ik_p >= 0) & (ik_p < n)
            vm = (ik_m >= 0) & (ik_m < n)
            const = np.zeros(2 * n - 1)
            const[vp] -= h_b[ik_p[vp]]
            const[vm] += 1.0 - h_b[ik_m[vm]]
            jp = ones[:, None] + shifts[None, :]
            jm = ones[:, None] - shifts[None, :]
            vjp = (jp >= 0) & (jp < n)
            vjm = (jm >= 0) & (jm < n)
            h_jp = np.where(vjp, h_b[np.clip(jp, 0, n-1)], 0.0)
            h_jm = np.where(vjm, h_b[np.clip(jm, 0, n-1)], 0.0)
            var = h_jp - (1.0 - h_jm) * vjm
            maxs = (corr[None, :] + const[None, :] + var).max(axis=1) * dx
            bi = maxs.argmin()
            if maxs[bi] < best_val - 1e-14:
                best_val = maxs[bi]
                best_ij = (i, ones[bi])
                best_delta = const + var[bi]
        if best_ij is None:
            break
        i, j = best_ij
        h_b[i] = 1.0
        h_b[j] = 0.0
        corr = corr + best_delta
        best_c5 = best_val
    return h_b, best_c5

def basin_hop(work_h, work_c5, n, t_end, max_iters=200):
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.03, 1.0 - 0.9 * it / max_iters)
        step = 0.02 + 0.2 * cool
        h_p = work_h + step * np.random.randn(n)
        h_p = project(h_p, n)
        T_use = int(200 + 4000 * cool)
        try:
            hc = slsqp_opt(h_p, T_use, n, maxiter=200)
            c5v = compute_c5(hc)
        except Exception:
            continue
        # Quick coord descent
        hc, c5v = coord_descent(hc, n, min(time.time() + 2, t_end), max_sweeps=30)
        temp = 0.003 * cool + 0.0003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)
