import numpy as np
import time

def compute_c5(h, N):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / N)

def project_sum(h, target):
    """Project h onto [0,1]^N with sum = target using bisection."""
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(corr, temperature):
    """Compute gradient of log-sum-exp of corr wrt h values."""
    # corr has length 2N-1, center at N-1
    c = corr - corr.max()
    c = np.exp(c / temperature)
    weights = c / c.sum()
    return weights  # These are the weights for each lag

    center = N - 1

        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(corr.max())
        if peak < best_peak:

            best_h = h.copy()
        
        # Smooth max weights
        c_shifted = corr - corr.max()
        w = np.exp(c_shifted / temp)
