```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(np.float64), 1.0 - h, mode='full').astype(np.float64)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def minimax_1d(a, b, lo, hi):
    """Find s in [lo,hi] minimizing max_j(a_j + s*b_j) via subgradient bisection."""
    for _ in range(100):
        mid = (lo + hi) * 0.5
        vals = a + mid * b
        v = vals.max()
        active = vals >= v - 1e-13
        bg = b[active]
        if bg.min() <= 0.0 and bg.max() >= 0.0:
            break
        elif bg.max() < 0.0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def get_coefs(h, i, n):
    """Vectorized computation of coefficient vector for coordinate i."""
    m = 2 * n - 1
    coef = np.zeros(m)
    nm1 = n - 1
    lo1 = max(0, nm1 - i)
    hi1 = min(m, n + i)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        coef[idx] += 1.0 - h[i + idx - nm1]
    lo2 = max(0, i)
    hi2 = min(m, i + n)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        coef[idx] -= h[i - idx + nm1]
    return coef

def coord_descent(h, t_end, n_sweeps=2000, order='forward', rng=None):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if order == 'forward':
            indices = range(n)
        elif order == 'reverse':
            indices = range(n - 1, -1, -1)
        else:
            indices = rng.permutation(n)
        for i in indices:
            if time.time() > t_end:
                return best_h, best_c5
            h_old = best_h[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-12:
                continue
            coef = get_coefs(best_h, i, n)
            s_new = minimax_1d(corr, coef, lo, hi)
            if abs(s_new) > 1e-11:
                best_h[i] = h_old + s_new
                corr += s_new * coef
                c5_new = np.max(corr) * dx
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, T, n, maxiter=800):
    dx = 2.0 / n
    def objective(x):
        xc = np.clip(x, 0, 1)
        c = np.correlate(xc, 1.0 - xc, mode='full')
        m = c.max()
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(objective, h, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=np.float64)
    else:
        best_h = np.zeros(128)
        best_h[:64] = 1.0

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive coordinate descent, multiple orderings (25% budget)
    t1 = t0 + budget_s * 0.25
    for order in ['forward', 'reverse', 'forward', 'random', 'reverse']:
        if elapsed() > t1:
            break
        best_h, best_c5 = coord_descent(best_h, t1, n_sweeps=5000, order=order, rng=rng)

    # Phase 2: SLSQP with increasing T (10% budget)
    t2 = t0 + budget_s * 0.35
    for T in [50, 200, 1000, 5000, 20000]:
        if elapsed() > t2:
            break
        try:
            hc = slsqp_polish(best_h, T, n, maxiter=600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    # Phase 3: More coordinate descent after SLSQP (15% budget)
    t3 = t0 + budget_s * 0.50
    for order in ['random', 'forward', 'reverse', 'random']:
        if elapsed() > t3:
            break
        best_h, best_c5 = coord_descent(best_h, t3, n_sweeps=5000, order=order, rng=rng)

    # Phase 4: Basin hopping with small perturbations (20% budget)
    t4 = t0 + budget_s * 0.70
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 100
    for it in range(n_bh):
        if elapsed() > t4:
            break
        cool = max(1.0 - 0.97 * (it / float(n_bh)), 0.02)
        r = rng.rand()
        if r < 0.30:
            step = 0.005 + 0.04 * cool
            h_p = work_h + step * rng.randn(n)
        elif r < 0.60:
            mask = rng.rand(n) < 0.15
            cnt = max(1, mask.sum())
            step = 0.01 + 0.05 * cool
            h_p = work_h.copy()
            h_p[mask] += step * rng.randn(cnt)
        elif r < 0.80:
            # Pair flip: swap two coordinates
            idx = rng.choice(n, 2, replace=False)
            h_p = work_h.copy()
            h_p[idx[0]], h_p[idx[1]] = work_h[idx[1]], work_h[idx[0]]
        else:
            shift = rng.randint(-2, 3)
            h_p = 0.5 * np.roll(work_h, shift) + 0.5 * work_h
        h_p = project_to_sum(h_p, n / 2.0)
        hc, c5v = coord_descent(h_p, time.time() + 2.5, n_sweeps=30, order='random', rng=rng)
        temp = 0.001 * cool + 0.00005
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = hc.copy(), c5v

    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Multi-resolution (10% budget)
    t5 = t0 + budget_s * 0.80
    for n_try in [256, 288]:
        if n_try == n or elapsed() > t5:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2.0)
        hc, c5v = coord_descent(h_i, time.time() + 12.0, n_sweeps=800, order='forward')
        if c5v < best_c5:
            best_h, best_c5, n = hc, c5v, n_try

    # Phase 6: Final intensive coordinate descent (15% budget)
    t6 = t0 + budget_s * 0.95
    for order in ['forward', 'reverse', 'random', 'forward']:
        if elapsed() > t6:
            break
        best_h, best_c5 = coord_descent(best_h, t6, n_sweeps=8000, order=order, rng=rng)

    # Phase 7: Final SLSQP + tiny coord pass
    if elapsed() < t0 + budget_s * 0.97:
        try:
            hc = slsqp_polish(best_h, 50000, n, maxiter=1000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass

    if elapsed() < t0 + budget_s * 0.995:
        best_h, best_c5 = coord_descent(best_h, t0 + budget_s * 0.998, n_sweeps=200, order='forward')

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
