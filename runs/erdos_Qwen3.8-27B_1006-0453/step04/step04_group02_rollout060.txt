import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_simple(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s > target:
        mask = h > 0
        h[mask] *= target / max(s, 1e-15)
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if s2 < target:
            deficit = target - s2
            mask2 = h < 1.0
            room = (1.0 - h)[mask2].sum()
            if room > 1e-15:
                h[mask2] += (1.0 - h[mask2]) * (deficit / room)
                h = np.clip(h, 0.0, 1.0)
    else:
        deficit = target - s
        mask = h < 1.0
        room = (1.0 - h)[mask].sum()
        if room > 1e-15:
            h[mask] += (1.0 - h[mask]) * (deficit / room)
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_simple(h_new, n_new)

def fast_coord_descent(h, n, t_end, max_sweeps=200, rng=None):

    h = h.copy()
    dx = 2.0 / n

    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = corr.max() * dx
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        any_change = False
        for i in order:

            idx_l = i - ks

def subsgrad_polish(h, n, t_end, n_steps=200, alpha0=0.02):
    """Subgradient steps to polish after coordinate descent."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = np.correlate(h, 1-h, mode='full').max() * dx
    best_h = h.copy()
    
    for step in range(n_steps):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1-h, mode='full')
        c5 = corr.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        k_star = np.argmax(corr)
        k = k_star - (n - 1)  # actual lag
        
        if k == 0:
            g = 1.0 - 2.0 * h
        elif k > 0:
            # grad of corr[mid+k] = sum_j h[j+k]*(1-h[j]) wrt h[i]
            g = np.zeros(n)
            # h[i] in h[j+k] part: i = j+k → j = i-k, valid if i-k >= 0
            idx = np.arange(k, n)
            g[idx] += (1.0 - h[idx - k])
            # h[i] in (1-h[j]) part: j = i, i+k < n
            idx2 = np.arange(0, n - k)
            g[idx2] -= h[idx2 + k]
        else:
            k = -k
            g = np.zeros(n)
            # grad of corr[mid-k] = sum_j h[j]*(1-h[j+k]) wrt h[i]
            idx = np.arange(0, n - k)
            g[idx] += (1.0 - h[idx + k])
            idx2 = np.arange(k, n)
            g[idx2] -= h[idx2 - k]
        
        alpha = alpha0 / (1.0 + step * 0.05)
        h = h - alpha * g
        h = project_simple(h, n)
    
    return best_h, best_c5

def smooth_c5_obj(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp_refine(h, n, T, t_end, maxiter=500):
    from scipy.optimize import minimize
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        T2 = T
        exps = np.exp(T2 * (corr - corr.max()))
        weights = exps / exps.sum()
        # gradient of smoothed max
        # d(corr[mid+k])/dx[i] for k>0: (1-x[i+k])*[i+k<n] - x[i-k]*[i-k>=0]
        grad = np.zeros(n)
        for k in range(1, n):
            w_k = weights[n-1+k] + weights[n-1-k]
            if w_k < 1e-15:
                continue
            idx_r = np.arange(k, n)
            grad[idx_r] += w_k * (1.0 - x[idx_r - k])
            idx_l = np.arange(0, n - k)
            grad[idx_l] -= w_k * x[idx_l + k]
        # k=0
        w0 = weights[n-1]
        if w0 > 1e-15:
            grad += w0 * (1.0 - 2.0 * x)
        return grad
    
    def coneq(x):
        return x.sum() - n / 2.0
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_simple(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_simple(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        mask = h > 0
        if mask.any():
            h[mask] *= target / s
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if s2 < target:
            d = target - s2
            m2 = h < 1.0
            r = (1.0 - h)[m2].sum()
            if r > 1e-15:
                h[m2] += (1.0 - h[m2]) * (d / r)
                h = np.clip(h, 0.0, 1.0)
    else:
        d = target - s
        m2 = h < 1.0
        r = (1.0 - h)[m2].sum()
        if r > 1e-15:
            h[m2] += (1.0 - h[m2]) * (d / r)
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_simple(h_new, n_new)

def fast_coord_descent(h, n, t_end, max_sweeps=300, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    h = h.copy()
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = corr.max() * (2.0 / n)
    stall = 0
    ks = np.arange(1, n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        for i in order:
            v = h[i]
            if v < -1e-14 or v > 1.0 + 1e-14:
                continue
            v = np.clip(v, 0.0, 1.0)
            idx_m = i - ks
            idx_p = i + ks
            mask_m = idx_m >= 0
            mask_p = idx_p < n
            d_right = np.zeros(n - 1)
            d_left = np.zeros(n - 1)
            if mask_m.any():
                d_right[mask_m] += 1.0 - h[idx_m[mask_m]]
                d_left[mask_m] -= h[idx_m[mask_m]]
            if mask_p.any():
                d_right[mask_p] -= h[idx_p[mask_p]]
                d_left[mask_p] += 1.0 - h[idx_p[mask_p]]
            full_d = np.empty(2 * n - 1)
            full_d[:mid] = d_left[::-1]
            full_d[mid] = 0.0
            full_d[mid + 1:] = d_right
            lo, hi = -v, 1.0 - v
            if lo >= hi - 1e-14:
                continue
            c_mid = corr[mid]
            def f1d(delta):
                vals = corr + delta * full_d
                vals[mid] = c_mid + delta * (1.0 - v - delta)
                return vals.max()
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-13, 'maxiter': 60})
            od = res.x
            if abs(od) < 1e-13:
                continue
            h[i] = v + od
            corr += od * full_d
            corr[mid] += od * (1.0 - v - od)
        h = project_simple(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * (2.0 / n)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            stall = 0
        else:
            stall += 1
            if stall >= 10:
                break
    return h, best_c5

def smoothed_grad(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * x)
    for k in range(1, n):
        wp = w[n - 1 + k]
        wm = w[n - 1 - k]
        if wp > 1e-14:
            ip = np.arange(k, n)
            grad[ip] += wp * (1.0 - x[ip - k])
            iq = np.arange(0, n - k)
            grad[iq] -= wp * x[iq + k]
        if wm > 1e-14:
            ip = np.arange(0, n - k)
            grad[ip] += wm * (1.0 - x[ip + k])
            iq = np.arange(k, n)
            grad[iq] -= wm * x[iq - k]
    return grad

def proj_grad_descent(h, n, t_end, T=500, max_iter=2000, alpha0=0.01):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        if it % 200 == 199:
            alpha *= 0.9
        g = smoothed_grad(h, n, T)
        h_new = h - alpha * g
        h_new = project_simple(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def slsqp_refine(h, n, T, t_end, maxiter=400):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)
    def objg(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        exps = np.exp(T * (corr - m))
        w = exps / exps.sum()
        g = np.zeros(n)
        w0 = w[n-1]
        if w0 > 1e-15:
            g += w0 * (1.0 - 2.0 * x)
        for k in range(1, n):
            wp, wm = w[n-1+k], w[n-1-k]
            if wp > 1e-14:
                ip = np.arange(k, n)
                g[ip] += wp * (1.0 - x[ip-k])
                iq = np.arange(0, n-k)
                g[iq] -= wp * x[iq+k]
            if wm > 1e-14:
                ip = np.arange(0, n-k)
                g[ip] += wm * (1.0 - x[ip+k])
                iq = np.arange(k, n)
                g[iq] -= wm * x[iq-k]
        return g * (2.0 / n)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_simple(np.clip(res.x, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(240)
        best_h[120:] = 0.0
    n = len(best_h)
    best_h = project_simple(best_h, n)
    best_c5 = compute_c5(best_h)
    # Phase 1: Fast coordinate descent
    t1 = t0 + budget_s * 0.20
    best_h, best_c5 = fast_coord_descent(best_h, n, t1, max_sweeps=300, rng=rng)
    # Phase 2: SLSQP with gradient
    for T in [100, 500, 2000]:
        if time.time() > t0 + budget_s * 0.32:
            break
        try:
            hc = slsqp_refine(best_h, n, T, t0 + budget_s * 0.35, maxiter=300)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass
    # Phase 3: Projected gradient
    if time.time() < t0 + budget_s * 0.42:
        best_h, best_c5 = proj_grad_descent(best_h, n, t0 + budget_s * 0.42, T=800, max_iter=3000, alpha0=0.005)
    # Phase 4: More coordinate descent
    if time.time() < t0 + budget_s * 0.52:
        best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.52, max_sweeps=300, rng=rng)
    # Phase 5: Multi-resolution
    for n_try in [168, 200, 288, 336, 400]:
        if time.time() > t0 + budget_s * 0.80:
            break
        h_interp = interpolate_h(best_h, n_try)
        h_cur, c5_cur = fast_coord_descent(h_interp, n_try, t0 + budget_s * 0.82, max_sweeps=150, rng=rng)
        if time.time() < t0 + budget_s * 0.85:

            except Exception:
                pass
        c5_cur = compute_c5(h_cur)
