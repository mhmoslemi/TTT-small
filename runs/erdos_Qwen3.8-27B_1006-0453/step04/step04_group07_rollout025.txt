import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def binarize(h, n):
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:n // 2]] = 1.0
    return b

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = h + room * (diff / tot)
    else:
        if s > 1e-14:
            h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def local_search(h, n, t_end, max_passes=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_passes):
        if time.time() > t_end:
            break
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        best_delta = 1e-15
        best_pair = None
        for a in ones:
            for b in zeros:
                best_h[a] = 0.0
                best_h[b] = 1.0
                c5_new = compute_c5(best_h)
                best_h[a] = 1.0
                best_h[b] = 0.0
                d = best_c5 - c5_new
                if d > best_delta:
                    best_delta = d
                    best_pair = (a, b)
        if best_pair is None:
            break
        best_h[best_pair[0]] = 0.0
        best_h[best_pair[1]] = 1.0
        best_c5 -= best_delta
    return best_h, best_c5

    h_work = h0.copy()
    work_c5 = compute_c5(h_work)
    best_h = h0.copy()

    while time.time() < t_end:
        if stall > stall_limit and n_kicks_used < max_kicks:
            n_kicks_used += 1
            for _ in range(kick_sz):
                ones = np.where(h_work == 1)[0]
                zeros = np.where(h_work == 0)[0]
                a = ones[np.random.randint(len(ones))]
                b = zeros[np.random.randint(len(zeros))]
                h_work[a] = 0.0
                h_work[b] = 1.0
            work_c5 = compute_c5(h_work)
            stall = 0
            stall_limit = int(stall_limit * 1.3)
            continue
        progress = (n_kicks_used * 1.0 + stall / max(stall_limit, 1)) / 20.0
        T = T0 * (T1 / T0) ** min(1.0, progress)
        ones = np.where(h_work == 1)[0]
        zeros = np.where(h_work == 0)[0]
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        h_work[a] = 0.0
        h_work[b] = 1.0
        c5_new = compute_c5(h_work)
        if c5_new <= work_c5:
            work_c5 = c5_new
            stall = 0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_work.copy()
        else:

                work_c5 = c5_new
            else:
                h_work[a] = 1.0
                h_work[b] = 0.0
            stall += 1
    return best_h, best_c5

def interp_to_width(h, n_new):
    n_old = len(h)
    x_new = np.linspace(0, n_old, n_new, endpoint=False)
    h_new = np.interp(x_new, np.arange(n_old), h)
    return project(h_new, n_new)

def smooth_obj(x, n, T):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    dx = 2.0 / n
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def sum_con(x, n):
    return x.sum() - n / 2.0

def continuous_polish(h_bin, n, t_end):
    h = h_bin.astype(float).copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: sum_con(x, n)}]

        if time.time() > t_end:
            break

                return smooth_obj(x, n, T)
            res = minimize(obj_wrapper, h_cur, method='SLSQP', bounds=bounds,
                           options={'maxiter': 150, 'ftol': 1e-13})

            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_new
        except Exception:
            pass
    return h, best_c5
