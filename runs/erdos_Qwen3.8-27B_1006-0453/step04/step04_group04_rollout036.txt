import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.sum(np.exp(np.clip(T * (corr - m), -500, 500)))
    return (m + np.log(s) / T) * dx

def slsqp_run(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_max_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt_single(h, n, i):
    """Optimize coordinate i with all others fixed."""
    dx = 2.0 / n
    hz = h.copy()
    val = hz[i]
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    # Linear coefficient for h[i] in corr
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k] = c
    k0 = n - 1
    def f1(t):
        cv = corr_base + t * coef
        cv[k0] = corr_base[k0] + t * (1.0 - t)  # quadratic at lag 0
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def binary_project(h):
    """Project h to binary with exactly n/2 ones."""
    n = len(h)
    idx = np.argsort(-h)  # descending
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def swap_local_search(h, n, t_end, max_iter=5000):
    """Binary swap local search."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]

        h_try = best_h.copy()

            best_h = h_try
            ones = np.where(best_h > 0.5)[0]

                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.sum(np.exp(np.clip(T * (corr - m), -500, 500)))
    return (m + np.log(s) / T) * dx

def slsqp_run(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    dx = 2.0 / n
    hz = h.copy()
    val = hz[i]
    hz[i] = 0.0
    base = np.correlate(hz, 1.0 - hz, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        if 0 <= i + lag < n:
            c += 1.0 - h[i + lag]
        if 0 <= i - lag < n:
            c += -h[i - lag]
        coef[k] = c
    k0 = n - 1
    def f1(t):
        cv = base + t * coef
        cv[k0] += t * (1.0 - t)
        return float(np.max(cv)) * dx
    return minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                           options={'xatol': 1e-12}).x

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):

            break
        improved = False
        order_arr = np.random.permutation(n) if order == 'random' else np.arange(n)
        for i in order_arr:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi
            h_try = proj(h_try, n)
            c5 = compute_c5(h_try)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def to_binary(h):
    n = len(h)
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def binary_swap_search(h, n, t_end, max_iter=10000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    ones = np.where(best_h > 0.5)[0]
    zeros = np.where(best_h <= 0.5)[0]
    npat = 0

        if len(ones) < 2 or len(zeros) < 2:

        j = zeros[np.random.randint(len(zeros))]

        h_try[i], h_try[j] = 0.0, 1.0
        c5 = compute_c5(h_try)
        if c5 < best_c5 - 1e-14:

            best_h = h_try
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h <= 0.5)[0]
        else:
            npat += 1
            if npat > len(ones) * len(zeros) * 2:
                break
    return best_h, best_c5

    m = np.max(corr)

                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.sum(np.exp(np.clip(T * (corr - m), -500, 500)))
    return (m + np.log(s) / T) * dx

def slsqp_run(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    dx = 2.0 / n
    hz = h.copy()
    hz[i] = 0.0
    base = np.correlate(hz, 1.0 - hz, mode='full')
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        if 0 <= i + lag < n:
            c += 1.0 - h[i + lag]
        if 0 <= i - lag < n:
            c += -h[i - lag]
        coef[k] = c
    k0 = n - 1
    def f1(t):
        cv = base + t * coef
        cv[k0] = base[k0] + t - t * t
        return float(np.max(cv)) * dx
    return minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                           options={'xatol': 1e-12}).x

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order_arr = np.random.permutation(n)
        for i in order_arr:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_opt(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi
            h_try = proj(h_try, n)
            c5 = compute_c5(h_try)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def to_binary(h):
    n = len(h)
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def binary_swap_search(h, n, t_end, max_iter=12000):

                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return proj(np.clip(res.x, 0, 1), n)
