    """Simulated annealing for binary sequence optimization."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    curr_e = int(np.max(corr))
    best_h = h.copy()
    best_e = curr_e
    
    t_start = time.time()
    total_time = max(t_end - t_start, 1.0)
    
    while time.time() < t_end:
        frac = min((time.time() - t_start) / total_time, 1.0)
        # Cooling: T goes from 8 to 0.05
        T = 8.0 * (0.05 / 8.0) ** frac
        
        # Pick random 1 and 0
        i = rng.randint(n)
        while h[i] == 0:
            i = rng.randint(n)
        j = rng.randint(n)
        while h[j] == 1:
            j = rng.randint(n)
        if i == j:
            continue
        
        # Swap
        h[i] = 0
        h[j] = 1
        
        corr = np.correlate(h, 1-h, mode='full')
        new_e = int(np.max(corr))
        
        if new_e <= curr_e or (rng.random() < np.exp(-(new_e - curr_e) / max(T, 1e-10))):
            curr_e = new_e
            if new_e < best_e:
                best_e = new_e
                best_h = h.copy()
        else:
            # Undo
            h[i] = 1
            h[j] = 0
    
    return best_h, best_e * dx

        bin_c5 = compute_c5(bin_init.astype(float))
        if bin_c5 < best_c5:
            best_c5 = bin_c5
            best_h = bin_init.astype(float)

    for ci, (n, frac) in enumerate(configs_clean):
        t_phase_end = t0 + budget_s * (sum(f for _, f in configs_clean[:ci]) + frac)
        t_phase_end = min(t_phase_end, t0 + budget_s * 0.94)
        if time.time() > t_phase_end:
            break

        starts = []
        if n == n0 and init is not None:
            starts.append(to_binary(init, n))
        n_random = max(3, int(5 * frac))
        for s in range(n_random):
            starts.append(random_binary(n, rng))

        phase_time = max(t_phase_end - time.time(), 1.0)
        per_start = phase_time / len(starts)

        for si, start in enumerate(starts):
            st_end = time.time() + per_start
            st_end = min(st_end, t_phase_end - 1.0)
            if time.time() >= t_phase_end:
                break

            h_res, c5_res = sa_binary(start, n, st_end, rng)
            # Greedy polish
            gp_end = min(st_end + per_start * 0.4, t_phase_end - 0.5)
            if time.time() < gp_end:
                h_res, c5_res = greedy_polish(h_res, n, gp_end, rng)

            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n

    # Final intensive polish on best solution
    if best_h is not None and time.time() < t0 + budget_s * 0.97:
        remaining = t0 + budget_s * 0.99 - time.time()
        if remaining > 2.0:
            h_pol, c5_pol = greedy_polish(best_h, best_n, t0 + budget_s * 0.99, rng)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    if best_h is None:
        best_h = random_binary(n0, rng).astype(float)
        best_c5 = compute_c5(best_h)
        best_n = n0

    return best_h.astype(float), best_c5, best_n
