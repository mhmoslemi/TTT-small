                improved = True

            break
    
    # Verify
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, n):
    target = n / 2.0
    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + diff * room / tot
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + diff * h / tot
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        for i in range(n):
            if diff > 0 and h[i] < 1.0:
                adj = min(diff, 1.0 - h[i])
                h[i] += adj
                diff -= adj
            elif diff < 0 and h[i] > 0.0:
                adj = max(diff, -h[i])
                h[i] += adj
                diff -= adj
            if abs(diff) < 1e-12:
                break
    return h

def coord_descent(h_init, n, t_end, max_sweeps=5000):
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    c0 = n - 1
    ks = np.arange(-(n - 1), n)
    sweep = 0
    while time.time() < t_end:
        improved = False
        for i in range(n):
            old = h[i]
            vf = (i - ks >= 0) & (i - ks < n)
            vs = (i + ks >= 0) & (i + ks < n)
            coeff = np.zeros(2 * n - 1)
            coeff[vf] += 1.0 - h[i - ks[vf]]
            coeff[vs] -= h[i + ks[vs]]
            coeff[c0] = 0.0
            c_at_0 = corr[c0] - old * (1.0 - old)
            f0 = max((corr - coeff * old).max(), c_at_0)
            c_at_1 = corr[c0] - old * (1.0 - old)
            f1 = max((corr + coeff * (1.0 - old)).max(), c_at_1)
            if f0 * dx < best_c5 - 1e-14:
                best_v, best_f = 0.0, f0
            elif f1 * dx < best_c5 - 1e-14:
                best_v, best_f = 1.0, f1
            else:
                def obj(v):
                    vals = corr + coeff * (v - old)
                    vals[c0] = corr[c0] + (v - old) * (1.0 - v - old)
                    return vals.max()
                res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                     options={'xatol': 1e-11, 'maxiter': 60})
                best_v, best_f = res.x, res.fun
                if f0 < best_f:
                    best_v, best_f = 0.0, f0
                if f1 < best_f:
                    best_v, best_f = 1.0, f1
            if best_f * dx < best_c5 - 1e-14:
                h[i] = best_v
                best_c5 = best_f * dx
                corr += coeff * (best_v - old)
                corr[c0] += (best_v - old) * (1.0 - best_v - old)
                improved = True
        sweep += 1
        if sweep % 50 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
    best_c5 = compute_c5(h)
    return h, best_c5

def binary_round(h, n, t_end):
    h = h.copy()
    target = n / 2.0
    best_c5 = compute_c5(h)
    while time.time() < t_end:
        improved = False
        for i in range(n):
            for v in [0.0, 1.0]:
                if abs(h[i] - v) < 1e-8:
                    continue
                h[i] = v
                h = project_sum(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                    break
                h[i] = h[i]  # undo
                h = project_sum(h, n)
        if not improved:
            break
    return h, best_c5

def basin_hop(work_h, work_c5, n, t_end, rng):
    best_h, best_c5 = work_h.copy(), work_c5
    it = 0
    total_iters = 400
    while time.time() < t_end:
        it += 1
        cool = max(0.03, 1.0 - 0.7 * it / total_iters)
        step = 0.03 + 0.25 * cool
        h_p = work_h + step * rng.standard_normal(n)
        h_p = project_sum(h_p, n)
        hc, c5v = coord_descent(h_p, n, min(time.time() + 15, t_end), max_sweeps=300)
        temp = 0.003 * cool + 0.0001
        if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) / n_old * 2.0
    x_new = np.arange(n_new, dtype=float) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h)
    return project_sum(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[0::2] = 1.0
    n = len(best_h)
    best_h = project_sum(best_h, n)
    best_c5 = compute_c5(best_h)

    elapsed = lambda: time.time() - t0

    # Phase 1: Heavy coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.20, max_sweeps=5000)

    # Phase 2: Binary rounding
    best_h, best_c5 = binary_round(best_h, n, t0 + budget_s * 0.28)

    # Phase 3: More coordinate descent after rounding
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.38, max_sweeps=5000)

    # Phase 4: Basin hopping
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.65, rng)

    # Phase 5: Final coordinate descent + rounding
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.75, max_sweeps=5000)
    best_h, best_c5 = binary_round(best_h, n, t0 + budget_s * 0.78)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82, max_sweeps=3000)

    # Phase 6: Try different resolutions
    for n_try in [200, 240, 120, 280]:
        if elapsed() > budget_s * 0.90:
            break
        h_tr = interp_h(best_h, n_try)
        h_tr, c5_tr = coord_descent(h_tr, n_try, min(t0 + elapsed() + 10, t0 + budget_s * 0.905), max_sweeps=500)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 7: Final polish
    if elapsed() < budget_s * 0.93:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, max_sweeps=3000)
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = binary_round(best_h, n, t0 + budget_s * 0.98)
    best_h = project_sum(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
