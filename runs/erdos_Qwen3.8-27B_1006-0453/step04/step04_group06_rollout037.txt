    h = h0.copy()
    h_prev = h0.copy()
    t_k = 1.0

    best_obj = smooth_obj(h, T)
    best_h = h.copy()

        if t_end is not None and time.time() > t_end:
            break
        grad = smooth_grad(h, T)
        # Backtracking

            h_try = project(h - step * grad, target_sum)
            if smooth_obj(h_try, T) <= smooth_obj(h, T) - 0.5 * step * np.dot(grad, h_try - h):
                break

        h_new = h_try

        h = project(h, target_sum)
        h_prev = h_new
        t_k = t_next
        
        obj = smooth_obj(h, T)
        if obj < best_obj - 1e-12:

            best_h = h.copy()

            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def coord_opt_fast(h, n, i):
    """Optimize single coordinate i. Returns optimal value."""
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save
    center = n - 1
    dx = 2.0 / n
    a = np.zeros(2*n - 1)
    for k in range(1, n):
        ip, im = i+k, i-k
        av = (1.0 - h[ip]) if ip < n else 0.0
        bv = h[im] if im >= 0 else 0.0
        a[center+k] = av - bv
        a[center-k] = bv - av
    ct = corr_base[center]
    # f(x) = max(max_k(corr_base[k] + x*a[k]), ct + x - x^2) * dx
    # This is convex in x (max of linear + concave quadratic... wait)
    # Actually it's max of linears and a concave quadratic, which is NOT convex.
    # But it's still 1D, so minimize_scalar works.
    def f1d(x):
        vals = corr_base + x * a
        vals[center] += 0  # will be overwritten
        vals[center] = ct + x - x*x
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
    return res.x

def cd_sweep(h, n, order=None, t_end=None):
    """One sweep of coordinate descent."""
    if order is None:
        order = np.random.permutation(n)
    for i in order:
        if t_end is not None and time.time() > t_end:
            break
        h[i] = coord_opt_fast(h, n, i)
    return project(h, n)

def multi_cd(h, n, t_end, max_sweeps=200):
    """Multiple sweeps of coordinate descent."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        h_try = best_h.copy()
        h_try = cd_sweep(h_try, n, t_end=t_end)
        c5 = compute_c5(h_try)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_try
        else:
            # No improvement, try a few more sweeps with different order
            improved = False
            for _ in range(3):
                if time.time() > t_end:
                    return best_h, best_c5
                h_try2 = best_h.copy()
                h_try2 = cd_sweep(h_try2, n, t_end=t_end)
                c5_2 = compute_c5(h_try2)
                if c5_2 < best_c5 - 1e-14:
                    best_c5 = c5_2
                    best_h = h_try2
                    improved = True
            if not improved:
                break
    return best_h, best_c5

def perturb_and_cd(h, n, sigma, t_end):

    h_pert = h + sigma * np.random.randn(n)
    h_pert = project(h_pert, n)
    return multi_cd(h_pert, n, t_end, max_sweeps=30)

def fista_smoothed(h0, T, n, max_iter=3000, t_end=None):
    """FISTA-like on smoothed objective (no convexity guarantee)."""
    h = h0.copy()
    hkm1 = h0.copy()
    t_k = 1.0
    step = 0.05
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        if t_end is not None and time.time() > t_end:
            break
        # Compute smoothed gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        shifted = T * (corr - m)
        exp_s = np.exp(shifted - shifted.max())
        w = exp_s / exp_s.sum()
        dx = 2.0 / n
        grad = w[n-1] * (1.0 - 2.0 * h)
        for k in range(1, n):
            wp, wm = w[n-1+k], w[n-1-k]
            if wp + wm < 1e-12:
                continue
            if wp > 1e-12:
                grad[:n-k] += wp * (1.0 - h[k:])
                grad[k:] -= wp * h[:n-k]
            if wm > 1e-12:
                grad[k:] += wm * (1.0 - h[:n-k])
                grad[:n-k] -= wm * h[k:]
        grad *= dx
        
        # Gradient step
        h_new = project(h - step * grad, n)
        # FISTA update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = project(h_new + ((t_k - 1.0) / t_next) * (h_new - hkm1), n)
        hkm1 = h_new
        t_k = t_next
        
        # Adaptive step
        if it % 50 == 49:
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
                step = min(step * 1.02, 0.2)
            elif it > 100:
                step *= 0.9
    
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        shifted = T * (corr - m)
        exp_s = np.exp(shifted - shifted.max())
        w = exp_s / exp_s.sum()
        g = w[n-1] * (1.0 - 2.0 * x)
        for k in range(1, n):
            wp, wm = w[n-1+k], w[n-1-k]
            if wp + wm < 1e-12:
                continue
            if wp > 1e-12:
                g[:n-k] += wp * (1.0 - x[k:])
                g[k:] -= wp * x[:n-k]
            if wm > 1e-12:
                g[k:] += wm * (1.0 - x[:n-k])
                g[:n-k] -= wm * x[k:]
        return g * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    from scipy.optimize import minimize
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=[(0.0, 1.0)]*n,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

    while time.time() < exp_end and attempt < 40:
        attempt += 1
        if attempt % 3 == 0:
            hp, cp = block_perturb_cd(best_h, n, min(time.time() + 12, exp_end))
        else:
            sigma = 0.015 + 0.05 * np.random.rand()
            hp, cp = perturb_cd(best_h, n, sigma, min(time.time() + 12, exp_end))
        if cp < best_c5:
            best_c5 = cp
            best_h = hp

    # Phase 5: Try different resolution
    for n_try in [192, 200]:
        if time.time() > t0 + budget_s * 0.78:
            break
        ht, ct = try_resolution(best_h, n, n_try, min(time.time() + 20, t0 + budget_s * 0.78))
        if ct < best_c5:
            best_c5 = ct
            best_h = ht
            n = n_try

    # Phase 6: Fine perturbations
    fine_end = t0 + budget_s * 0.88
    for _ in range(20):
        if time.time() > fine_end:
            break
        sigma = 0.003 + 0.012 * np.random.rand()
        hp, cp = perturb_cd(best_h, n, sigma, min(time.time() + 10, fine_end))
        if cp < best_c5:
            best_c5 = cp
            best_h = hp

    # Phase 7: Final SLSQP + CD
    for T in [5000, 20000]:
        if time.time() > t0 + budget_s * 0.94:
            break
        try:
            hs = slsqp_opt(best_h.copy(), T, n, maxiter=300)
            cs = compute_c5(hs)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs
        except Exception:
            pass

    # Phase 8: Final CD polish
    best_h, best_c5 = multi_cd(best_h, n, t_end - 3, max_sweeps=500)

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
