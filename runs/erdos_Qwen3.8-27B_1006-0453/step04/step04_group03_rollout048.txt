def compute_coef(h, i, n):
    coef = np.zeros(2*n-1)
    # For m in range(-(n-1), n):
    # coef[n-1+m] = (1-h[i+m])*[0<=i+m<n] - h[i-m]*[0<=i-m<n]
    
    # The first term: (1-h[i+m]) for m where 0<=i+m<n, i.e., -i <= m < n-i
    # The second term: h[i-m] for m where 0<=i-m<n, i.e., i-n < m <= i
    
    m_vals = np.arange(-(n-1), n)
    # First term
    ip = i + m_vals  # i+m
    mask1 = (ip >= 0) & (ip < n)
    coef[n-1+m_vals[mask1]] += (1.0 - h[ip[mask1]])
    # Second term
    im = i - m_vals  # i-m
    mask2 = (im >= 0) & (im < n)
    coef[n-1+m_vals[mask2]] -= h[im[mask2]]
    
    return coef
