def compute_grad(h, w, dx, n):
    # w has length 2n-1, indexed from -(n-1) to n-1
    # grad_i = dx * (sum_s w[s]*(1-h[i-s]) - sum_s w[s]*h[i+s])
    grad = np.zeros(n)
    shifts = np.arange(-(n-1), n)
    # Vectorized: for each shift s, add contribution to all valid i
    for s in shifts:
        # First term: (1-h[i-s]) for i in [max(0,-s), min(n, n-s))
        # i-s in [0,n-1] means i in [s, s+n-1] intersect [0, n-1]
        i_lo = max(0, s)
        i_hi = min(n, s + n)  # exclusive
        if i_lo < i_hi:
            grad[i_lo:i_hi] += w[s + n - 1] * (1.0 - h[i_lo - s:i_hi - s])
        # Second term: h[i+s] for i in [max(0,-s), min(n, n-s))
        i_lo = max(0, -s)
        i_hi = min(n, n - s)  # exclusive
        if i_lo < i_hi:
            grad[i_lo:i_hi] -= w[s + n - 1] * h[i_lo + s:i_hi + s]
    return grad * dx

def project(h, n):

        h = np.clip(h, 0.0, 1.0)

    if s > target:

        for i in order:
            adj = min(excess, h[i])
            h[i] -= adj
            excess -= adj

        g += 2.0 * pen * (x.sum() - n / 2.0)
