def ga_binary(n, pop_size, generations, t_end, seed):
    rng = np.random.default_rng(seed)
    # Initialize population
    pop = []
    fitness = []
    for _ in range(pop_size):
        h = np.zeros(n)
        ones = rng.choice(n, size=n//2, replace=False)
        h[ones] = 1.0
        c5, _ = compute_c5(h)
        pop.append(h)
        fitness.append(c5)
    
    best_c5 = min(fitness)
    best_h = pop[fitness.index(best_c5)].copy()
    
    for gen in range(generations):
        if time.time() > t_end:
            break
        new_pop = []
        new_fit = []
        for _ in range(pop_size):
            # Tournament selection
            idx1 = tournament(fitness, rng, k=5)
            idx2 = tournament(fitness, rng, k=5)
            p1, p2 = pop[idx1], pop[idx2]
            
            # Crossover (1-point)
            if rng.random() < 0.8:
                child = p1.copy()
                if n > 1:
                    cut = rng.integers(1, n)
                    child[cut:] = p2[cut:]
            else:
                child = p1.copy()
            
            # Fix count to n/2
            num_ones = int(np.sum(child == 1))
            if num_ones > n//2:
                excess = num_ones - n//2
                one_pos = np.where(child == 1)[0]
                for i in rng.choice(one_pos, size=excess, replace=False):
                    child[i] = 0
            elif num_ones < n//2:
                deficit = n//2 - num_ones
                zero_pos = np.where(child == 0)[0]
                for i in rng.choice(zero_pos, size=deficit, replace=False):
                    child[i] = 1
            
            # Mutation: swap
            mut_rate = 0.1
            for _ in range(int(mut_rate * n)):
                if rng.random() < 1.0:
                    ones_pos = np.where(child == 1)[0]
                    zeros_pos = np.where(child == 0)[0]
                    if len(ones_pos) > 0 and len(zeros_pos) > 0:
                        i = rng.choice(ones_pos)
                        j = rng.choice(zeros_pos)
                        child[i], child[j] = 0, 1
            
            c5, _ = compute_c5(child)
            new_pop.append(child)
            new_fit.append(c5)
        
        # Elitism
        # Replace worst with best of new
        pop = new_pop
        fitness = new_fit
        
        bi = np.argmin(fitness)
        if fitness[bi] < best_c5:
            best_c5 = fitness[bi]
            best_h = pop[bi].copy()
    
    return best_h, best_c5

            best_v = old
            best_f = max_corr
            if f0 < best_f:
                best_f, best_v = f0, 0.0
            if f1 < best_f:
                best_f, best_v = f1, 1.0

            if 0.05 < old < 0.95 and best_f < max_corr * 1.005:
                def obj(v):
                    return np.max(corr + (v - old) * c)
                try:
                    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                          options={'xatol': 1e-9})
                    if res.fun < best_f:

                except Exception:
                    pass

                h[i] = best_v
                max_corr = np.max(corr)

        if not improved:
            break
    
    return h, float(max_corr) * dx

def ga_binary(n, pop_size, t_end, rng, start_h=None):
    # Build initial population
    pop = []
    if start_h is not None:
        pop.append(start_h.copy())
    while len(pop) < pop_size:
        pop.append(make_binary(n, rng))
    
    fits = [compute_c5(pop[i]) for i in range(pop_size)]
    best_idx = int(np.argmin(fits))
    best_h = pop[best_idx].copy()
    best_c5 = fits[best_idx]
    
    gen = 0
    while time.time() < t_end:
        gen += 1
        new_pop = []
        new_fits = []
        
        for p in range(pop_size):
            # Tournament selection (k=5)
            cands = rng.integers(0, pop_size, size=5)
            s1 = int(cands[np.argmin([fits[c] for c in cands])])
            cands = rng.integers(0, pop_size, size=5)
            s2 = int(cands[np.argmin([fits[c] for c in cands])])
            
            child = pop[s1].copy()
            
            # Crossover
            if rng.random() < 0.7 and n > 2:
                cut = int(rng.integers(1, n))
                child[cut:] = pop[s2][cut:]
            
            # Fix count to n//2
            n1 = int(np.sum(child))
            tgt = n // 2
            if n1 > tgt:
                excess = n1 - tgt
                op = np.where(child > 0.5)[0]
                zp = np.where(child < 0.5)[0]
                sw_o = rng.choice(op, size=excess, replace=False)
                sw_z = rng.choice(zp, size=excess, replace=False)
                child[sw_o] = 0
                child[sw_z] = 1
            elif n1 < tgt:
                deficit = tgt - n1
                zp = np.where(child < 0.5)[0]
                op = np.where(child > 0.5)[0]
                sw_z = rng.choice(zp, size=deficit, replace=False)
                sw_o = rng.choice(op, size=deficit, replace=False)
                child[sw_z] = 1
                child[sw_o] = 0
            
            # Mutation: 1-3 random swaps
            n_mut = int(rng.integers(1, 4))
            for _ in range(n_mut):
                op = np.where(child > 0.5)[0]
                zp = np.where(child < 0.5)[0]
                if len(op) > 0 and len(zp) > 0:
                    child[int(rng.choice(op))], child[int(rng.choice(zp))] = 0, 1
            
            new_pop.append(child)
            new_fits.append(compute_c5(child))
        
        bi = int(np.argmin(new_fits))
        if new_fits[bi] < best_c5:
            best_c5 = new_fits[bi]
            best_h = new_pop[bi].copy()
        
        # Elitism
        wi = int(np.argmax(new_fits))
        new_pop[wi] = pop[best_idx].copy()
        new_fits[wi] = fits[best_idx]
        
        pop = new_pop
        fits = new_fits
        best_idx = int(np.argmin(fits))
    
    return best_h, best_c5

def antiperiodic_project(h, n):
    """Project h onto the anti-periodic subspace: h[i+n/2] = 1-h[i]."""
    n2 = n // 2
    h_ap = h.copy()
    for i in range(n2):
        avg = (h[i] + (1.0 - h[i + n2])) / 2.0
        h_ap[i] = avg
        h_ap[i + n2] = 1.0 - avg
    return project(h_ap, n)

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def softmax_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def lbfgs_opt(h, n, T, t_end, maxiter=200):
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)
    
    res = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8,
                            'maxfun': maxiter * 3})
    hc = project(np.clip(res.x, 0, 1), n)
    return hc, compute_c5(hc)
