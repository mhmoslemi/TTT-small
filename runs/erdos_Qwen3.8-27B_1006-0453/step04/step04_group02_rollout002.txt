    for k in range(1, n):

        b = n - 1 - k

            d[a] += (-h[i + k])
        if 0 <= i + k < n:

        if 0 <= i - k < n:

        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP polish
    for T in [50, 200, 800]:
        if elapsed() > budget_s * 0.08:
            break
        try:
            hc = slsqp_polish(best_h, n, T, maxiter=300)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 2: Coordinate descent (main)
    best_h, best_c5 = coord_sweep(best_h, n, t0 + budget_s * 0.35, rng, max_sweeps=200)

    # Phase 3: Pair descent
    if elapsed() < budget_s * 0.55:
        best_h, best_c5 = pair_sweep(best_h, n, t0 + budget_s * 0.50, rng, n_pairs=18000)

    # Phase 4: Basin hopping (brief)
    if elapsed() < budget_s * 0.65:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.65, rng,
                                    n_iters=50, T_smooth=300)

    # Phase 5: More coordinate descent
    if elapsed() < budget_s * 0.80:
        best_h, best_c5 = coord_sweep(best_h, n, t0 + budget_s * 0.80, rng, max_sweeps=150)

    # Phase 6: Multi-resolution
    for n_try in [200, 220, 260, 280, 300, 320, 360]:
        if elapsed() > budget_s * 0.90:
            break
        h_interp = interp_h(best_h, n_try)
        t_sub = min(t0 + budget_s * 0.90, time.time() + 12)
        h_interp, c5_sub = coord_sweep(h_interp, n_try, t_sub, rng, max_sweeps=30)
        if elapsed() > budget_s * 0.90:
            break
        try:
            h_interp = slsqp_polish(h_interp, n_try, 500, maxiter=150)
        except Exception:
            pass
        cv = compute_c5(h_interp)
        if cv < best_c5:
            best_h, best_c5, n = h_interp, cv, n_try
            t_sub2 = min(t0 + budget_s * 0.90, time.time() + 8)
            best_h, best_c5 = coord_sweep(best_h, n, t_sub2, rng, max_sweeps=20)

    # Phase 7: Final polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_sweep(best_h, n, t0 + budget_s * 0.97, rng, max_sweeps=50)
    if elapsed() < budget_s * 0.99:
        try:
            hc = slsqp_polish(best_h, n, 5000, maxiter=200)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
