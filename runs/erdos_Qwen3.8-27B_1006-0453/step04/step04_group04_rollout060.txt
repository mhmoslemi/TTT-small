    # Phase 3: Intensive coordinate descent
    if rem() > 200:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.38, max_sweeps=800)

    # Phase 4: Basin hopping at multiple scales
    if rem() > 180:
        work_h = best_h.copy()
        work_c5 = best_c5
        n_iter = 50
        for it in range(n_iter):
            if rem() < 25:
                break
            cool = max(1.0 - 0.92 * (it / float(n_iter)), 0.05)
            sz = 0.005 + 0.10 * cool
            hp = work_h + sz * np.random.randn(n)
            hp = proj(hp, n)
            try:
                T_bh = 1500 + 5000 * (1.0 - cool)
                hc = slsqp(hp, T_bh, n, 250)
                c5v = compute_c5(hc)
                temp = 0.0015 * cool + 0.0001
                if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
                    work_h, work_c5 = hc, c5v
                    if c5v < best_c5:
                        best_c5, best_h = c5v, hc
            except Exception:
                pass

    # Phase 5: Second round coordinate descent + subgradient
    if rem() > 120:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.62, max_sweeps=500)
    if rem() > 80:
        best_h, best_c5 = subgrad_opt(best_h, n, t0 + budget_s * 0.70)

    # Phase 6: Try alternative resolutions
    for n_try in [120, 140, 100, 150]:
        if rem() < 70:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_in = np.interp(x_new, x_old, best_h)
        h_in = proj(h_in, n_try)
        try:
            hc = slsqp(h_in, 5000, n_try, 600)
            c5v = compute_c5(hc)
            t_end_cd = t0 + min(budget_s * 0.74, time.time() + rem() * 0.4)
            hc, c5v2 = coord_descent(hc, n_try, t_end_cd, max_sweeps=400)
            if c5v2 < best_c5:
                best_c5, best_h, n = c5v2, hc, n_try
        except Exception:
            pass

    # Phase 7: Final polish
    n = len(best_h)
    if rem() > 60:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=400)
    if rem() > 40:
        best_h, best_c5 = subgrad_opt(best_h, n, t0 + budget_s * 0.95, max_it=10000)
    for T in [50000, 200000]:
        if rem() < 8:
            break
        try:
            hc = slsqp(best_h, T, n, 800)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        except Exception:
            pass

    # Final coordinate tweak
    if rem() > 10:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=100)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
