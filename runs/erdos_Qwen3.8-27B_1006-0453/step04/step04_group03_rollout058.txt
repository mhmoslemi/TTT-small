    return float(np.max(compute_corr(h))) * (2.0 / len(h))

    diff = target - s
    if abs(diff) < 1e-15:
        return h

            return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h, mode='full').astype(float)

def compute_c5(h):
    return float(np.max(compute_corr(h))) * (2.0 / len(h))

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(c, d, s_lo, s_hi):
    for _ in range(45):
        s = 0.5 * (s_lo + s_hi)
        vals = c + s * d
        m = vals.max()
        active = vals > m - 1e-11
        dm = d[active]
        if dm.size == 0:
            break
        if dm.min() >= -1e-15:
            s_hi = s
        elif dm.max() <= 1e-15:
            s_lo = s
        else:
            break
    return 0.5 * (s_lo + s_hi)

def fast_cd(h, t_end, max_sweeps=10000):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * dx)
    nr = 2 * n - 1
    r_arr = np.arange(nr)
    nm1 = n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.arange(n) if sweep < 3 else np.random.permutation(n)
        for i in range(n):
            idx = int(order[i])
            ho = h[idx]
            j1 = idx - r_arr + nm1
            v1 = (j1 >= 0) & (j1 < n)
            j2 = idx + r_arr - nm1
            v2 = (j2 >= 0) & (j2 < n)
            d = np.zeros(nr)
            if v1.any():
                d[v1] = 1.0 - h[j1[v1]]
            if v2.any():
                d[v2] -= h[j2[v2]]
            slo, shi = -ho, 1.0 - ho
            if shi - slo < 1e-14:
                continue
            sopt = solve_1d_minimax(corr, d, slo, shi)
            if abs(sopt) > 1e-13:
                h[idx] += sopt
                corr += sopt * d
                c5n = float(np.max(corr) * dx)
                if c5n < best_c5:
                    best_c5 = c5n
                    improved = True
        if not improved:
            break
    return h, best_c5

def slsqp_opt(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return (m + np.log(np.sum(np.exp(T * (c - m))) + 1e-30) / T) * dx
    target = n / 2.0
    def cons_fun(x):
        return x.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)
    except Exception:
        return h.copy()
