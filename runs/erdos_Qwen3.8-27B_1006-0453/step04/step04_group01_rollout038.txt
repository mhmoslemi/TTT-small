    grad = np.zeros(n)
    # For each shift k, the contribution to grad[j] is:
    # w[n-1+k] * ((1-h[j+k])*I(0<=j+k<n) - h[j-k]*I(0<=j-k<n))

                grad[:n-k] += wk * (1.0 - h[k:])  # (1-h[j+k]) for j=0..n-k-1
            if k > 0:

    return grad * dx

                if kk < n:
                    g[kk:] += wk * (1.0 - x[:n-kk])
                g[:n-kk] -= wk * x[kk:]
        return val, g * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_grad, h, jac=True, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def project(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-13:
        return vc
    lo, hi = -1.5, 1.5
    orig = h.copy()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(orig + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(orig + (lo + hi) / 2.0, 0.0, 1.0)

def compute_g_vec(h, j, n):
    g = np.zeros(2 * n - 1)
    s_lo = max(-j, -(n - 1))
    s_hi = min(n - 1 - j, n - 1)
    if s_lo <= s_hi:
        idx = np.arange(n - 1 + s_lo, n - s_hi + n - 1)
        hidx = np.arange(j + s_lo, j + s_hi + 1)
        g[idx] += 1.0 - h[hidx]
    s_lo2 = max(j - n + 1, -(n - 1))
    s_hi2 = min(j, n - 1)
    if s_lo2 <= s_hi2:
        idx2 = np.arange(n - 1 + s_lo2, n - 1 + s_hi2 + 1)
        hvals = h[j - s_hi2 : j - s_lo2 + 1][::-1]
        g[idx2] -= hvals
    return g

def minimax_1d(C, g, v):
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(C + (m1 - v) * g) <= np.max(C + (m2 - v) * g):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def fast_cd(h, n, t_end, max_sweeps=100000):
    dx = 2.0 / n
    fc = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = fc.max() * dx
    best_h = h.copy()
    stale = 0
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for j in range(n):
            v = h[j]
            g = compute_g_vec(h, j, n)
            nv = minimax_1d(fc, g, v)
            if abs(nv - v) > 5e-12:
                d = nv - v
                h[j] = nv
                fc += d * g
                c5 = fc.max() * dx
                if c5 < best_c5 - 3e-15:
                    best_c5 = c5
                    best_h = h.copy()
                    imp = True
        h = project(h, n)
        fc = np.correlate(h, 1.0 - h, mode='full')
        c5 = fc.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if not imp:
            stale += 1
        else:
            stale = 0
        if stale >= 6:
            break
    return best_h, best_c5

def slsqp_smooth(h, n, T, maxiter=800):
    dx = 2.0 / n
    target = n / 2.0
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        e = np.exp(T * (c - m))
        s = e.sum()
        w = e / s
        val = (m + np.log(s) / T) * dx
        g = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = w[k_idx]
            if wk < 1e-14:
                continue
            k = k_idx - (n - 1)
            if k >= 0:
                if k < n:
                    g[:n - k] += wk * (1.0 - x[k:])
                if k > 0:
                    g[k:] -= wk * x[:n - k]
            else:
                kk = -k
                if kk < n:
                    g[kk:] += wk * (1.0 - x[:n - kk])
                g[:n - kk] -= wk * x[kk:]
        return val, g * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj_grad, h, jac=True, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP smooth (12%)
    for T in [200, 2000, 15000]:
        if elapsed() > budget_s * 0.12:
            break
        try:
            hc = slsqp_smooth(best_h.copy(), n, T, maxiter=1200)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Main fast coordinate descent (50%)
    best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.62, max_sweeps=200000)

    # Phase 3: SLSQP + CD alternation (20%)
    for it in range(12):
        if elapsed() > budget_s * 0.82:
            break
        T = 3000 + 5000 * it
        try:
            hc = slsqp_smooth(best_h.copy(), n, T, maxiter=400)
        except Exception:
            hc = best_h.copy()
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h = hc
            best_c5 = c5v
        best_h, best_c5 = fast_cd(best_h, n, min(t0 + budget_s * 0.82, time.time() + budget_s * 0.012), max_sweeps=8000)

    # Phase 4: Diversification (12%)
    for it in range(6):
        if elapsed() > budget_s * 0.94:
            break
        scale = 0.03 + 0.03 * it
        h_p = best_h + scale * np.random.randn(n)
        h_p = project(h_p, n)
        h_p, c5v = fast_cd(h_p, n, min(t0 + budget_s * 0.94, time.time() + budget_s * 0.018), max_sweeps=15000)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_p

    # Phase 5: Final CD polish (6%)
    best_h, best_c5 = fast_cd(best_h, n, t0 + budget_s * 0.99, max_sweeps=50000)

    # Last SLSQP polish
    if elapsed() < budget_s * 0.97:
        try:
            hc = slsqp_smooth(best_h.copy(), n, 50000, maxiter=600)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h = hc
                best_c5 = c5v
        except Exception:
            pass

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
