```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def make_binary(h):
    n = len(h)
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n//2]] = 1.0
    return h_b

def sa_binary(h, t_end, max_iter=2000000):
    n = len(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    c5 = corr.max() * dx
    best_c5 = c5
    best_h = h.copy()
    
    T_init = 0.4
    T_end = 0.005
    min_iter = 100000
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        if it < min_iter:
            T = T_init
        else:
            progress = (it - min_iter) / max(1, max_iter - min_iter)
            T = T_init * (T_end / T_init) ** progress
            if T < T_end:
                T = T_end
        
        # Pick random swap
        a = np.random.randint(n)
        while h[a] != 1:
            a = np.random.randint(n)
        b = np.random.randint(n)
        while h[b] != 0:
            b = np.random.randint(n)
        
        # Compute new corr
        new_corr = corr.copy()
        new_corr[n-1-a : 2*n-1-a] -= (1-h)
        new_corr[a+n-1 : a-1 : -1] += h
        new_corr[n-1-b : 2*n-1-b] += (1-h)
        new_corr[b+n-1 : b-1 : -1] -= h
        
        new_c5 = new_corr.max() * dx
        delta = new_c5 - c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h[a] = 0
            h[b] = 1
            corr = new_corr
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    ks = np.arange(-(n - 1), n)
    c0_idx = n - 1
    
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0_base = corr[c0_idx]
            
            def obj(v):
                cn = corr + (v - old) * coeff
                cn[c0_idx] = c0_base + (v - old) * (1.0 - v - old)
                return np.max(cn)
            
            res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            best_v = res.x
            best_f = res.fun
            v0 = obj(0.0)
            if v0 < best_f:
                best_v, best_f = 0.0, v0
            v1r = obj(1.0)
            if v1r < best_f:
                best_v, best_f = 1.0, v1r
            
            if best_f * dx < best_c5 - 1e-13:
                best_h[i] = best_v
                best_c5 = best_f * dx
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            break
    
    return best_h, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def make_binary(h):
    n = len(h)
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n // 2]] = 1.0
    return h_b

def apply_swap(corr, h, a, b, n):
    new_corr = corr.copy()
    new_corr[n - 1 - a:2 * n - 1 - a] -= (1 - h)
    new_corr[a + n - 1:a - 1:-1] += h
    new_corr[n - 1 - b:2 * n - 1 - b] += (1 - h)
    new_corr[b + n - 1:b - 1:-1] -= h
    return new_corr

    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)

def smooth_slsqp(h0, T, n, maxiter=500):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * (2.0 / n)
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0
```
