import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) >= target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, N, L):
    g = np.zeros(N)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:N-L] += 1.0 - h[L:]
        g[L:] += -h[:N-L]
    else:
        M = -L
        g[M:] += 1.0 - h[:N-M]
        g[:N-M] += -h[M:]
    return g

def adam_optimize(h, N, deadline, temp0, temp1, top_k, lr0, lr_decay):
    h = h.astype(float).copy()
    target = float(N) / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr0
    while time.time() < deadline:
        t += 1
        frac = min(1.0, t / 3000.0)
        temp = temp0 * (temp1 / temp0) ** frac
        
        corr = compute_corr(h)
        c5 = float(corr.max()) * (2.0 / N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        k = min(top_k, len(corr))
        top_idx = np.argpartition(corr, -k)[-k:]
        vals = corr[top_idx]
        w = np.exp((vals - vals.max()) / max(temp, 1e-6))
        w /= w.sum()
        
        grad = np.zeros(N)
        center = N - 1
        for i in range(k):
            L = int(top_idx[i]) - center
            grad += w[i] * grad_lag(h, N, L)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_box_sum(h, target)
        
        if t % 500 == 0:
            lr *= lr_decay
    return best_h, best_c5

def anneal_binary(hb, N, n_steps, rng, T0, T1, deadline):
    best = hb.copy()
    best_c5 = compute_c5(best, N)
    cur_c5 = best_c5
    for it in range(n_steps):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(1, n_steps))
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        hb[i], hb[j] = 0.0, 1.0
        new_c5 = compute_c5(hb, N)
        d = new_c5 - cur_c5
        if d < 0 or (T > 0.002 and rng.random() < np.exp(-d / max(T, 1e-10))):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = hb.copy()
        else:
            hb[i], hb[j] = 1.0, 0.0
    return best, best_c5

def scale_to_n(h, N_new):

    x_old = np.linspace(0, 2, N_old, endpoint=False)

    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, N_new / 2.0)
