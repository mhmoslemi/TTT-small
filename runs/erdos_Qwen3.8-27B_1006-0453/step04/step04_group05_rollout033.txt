def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def compute_A(h, N):
    L = np.arange(-(N-1), N)
    i = np.arange(N)[:, None]
    il = i + L[None, :]
    im = i - L[None, :]
    valid_plus = (il >= 0) & (il < N)
    valid_minus = (im >= 0) & (im < N)
    h_plus = np.where(valid_plus, h[np.clip(il, 0, N-1)], 0.0)
    h_minus = np.where(valid_minus, h[np.clip(im, 0, N-1)], 0.0)
    return h_minus - valid_plus.astype(float) * (1.0 - h_plus)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid

            break

            best_h = h.copy()
