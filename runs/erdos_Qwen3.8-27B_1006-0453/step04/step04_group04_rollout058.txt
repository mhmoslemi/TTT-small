        h_[:n-1] = u
        h_[n-1] = target - u.sum()

        h_clipped = np.clip(h_, 0, 1)
        val = smooth_c5(h_clipped, T)
        # Penalty for h_[n-1] out of bounds
        viol = 0.0
        if h_[n-1] > 1:

        elif h_[n-1] < 0:

        # Penalty for other elements out of bounds (shouldn't happen with L-BFGS-B bounds)
        return val + M * viol
    
    def grad(u):

        h_clipped = np.clip(h_, 0, 1)
        g_full = smooth_grad(h_clipped, T)
        # Chain rule: df/du[k] = sum_j dh[j]/du[k] * g_full[j]
        # dh[j]/du[k] = 1 if j==k (j < n-1), -1 if j==n-1

        elif h_[n-1] < 0:
            g_u += 2 * M * h_[n-1] * (-1)
        return g_u
    
    def obj_and_grad(u):
        return obj(u), grad(u)
    
    bnds = [(0.0, 1.0)] * (n-1)
    res = minimize(obj, u, jac=grad, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'gtol': 1e-10, 'maxls': 50})

    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k] = c
    k0 = n - 1
    def f1d(t):
        cv = corr_base + t * coef
        cv[k0] = corr_base[k0] + t * (1.0 - t)
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def symmetric_opt(h_init, n, t_end):
    nh = n // 2
    u0 = h_init[:nh].copy()
    target_nh = nh / 2.0
    def u_to_h(u):
        h_ = np.empty(n)
        h_[:nh] = u
        h_[nh:] = 1.0 - u
        return h_
    def obj(s):
        u = nh * sigmoid_vec(s)
        return smooth_c5(u_to_h(u), 2000)
    def sigmoid_vec(s):
        return 1.0 / (1.0 + np.exp(-s))
    bnds = [(-10, 10)] * nh
    try:
        res = minimize(obj, u0 - 0.5, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': 2000, 'gtol': 1e-10})
        u_out = nh * sigmoid_vec(res.x)
        h_out = u_to_h(u_out)
        return proj(h_out, n)
    except Exception:
        return h_init

def sigmoid_vec(s):
    return 1.0 / (1.0 + np.exp(-np.clip(s, -50, 50)))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS with increasing T (fast global-ish descent)
    for T in [50, 200, 1000, 5000, 20000, 100000]:
        if elapsed() > budget_s * 0.12:
            break
        hc = labfgs(best_h, T, n, maxiter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: SLSQP with gradient (fine-tune)
    for T in [5000, 50000]:
        if elapsed() > budget_s * 0.20:
            break
        hc = slsqp_g(best_h, T, n, maxiter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Coordinate descent
    if elapsed() < budget_s * 0.35:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.35, max_sweeps=300)

    # Phase 4: Basin hopping with targeted perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 120
    for it in range(n_bh):
        if elapsed() > budget_s * 0.55:
            break
        cool = 1.0 - 0.85 * (it / float(n_bh))
        step = 0.03 + 0.15 * cool
        noise = np.random.randn(n)
        # Targeted component: bias toward reducing worst lag
        g_dir = smooth_grad(work_h, 2000)
        g_norm = np.linalg.norm(g_dir)
        if g_norm > 1e-12:
            noise = 0.6 * noise + 0.4 * (-g_dir / g_norm * n)
        h_pert = work_h + step * noise
        h_pert = proj(h_pert, n)
        T_bh = 5000
        hc = labfgs(h_pert, T_bh, n, maxiter=800)
        c5v = compute_c5(hc)
        temp = 0.002 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Coord descent polish
    if elapsed() < budget_s * 0.65:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.65, max_sweeps=300)

    # Phase 6: Symmetric reduction search
    if elapsed() < budget_s * 0.75 and n % 2 == 0:
        h_sym = symmetric_opt(best_h, n, t0 + budget_s * 0.75)
        h_sym = proj(h_sym, n)
        c5_sym = compute_c5(h_sym)
        if c5_sym < best_c5:
            best_c5 = c5_sym
            best_h = h_sym
        # Also try optimizing the symmetric result
        hc = labfgs(h_sym, 20000, n, maxiter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Try other resolutions
    for n_try in [160, 200, 240]:
        if elapsed() > budget_s * 0.88:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = proj(h_interp, n_try)
        for T in [3000, 30000]:
            if elapsed() > budget_s * 0.90:
                break
            hc = labfgs(h_interp, T, n_try, maxiter=2000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                n = n_try

    # Phase 8: Final polish - coord descent + high-T L-BFGS
    if elapsed() < budget_s * 0.94:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, max_sweeps=200)
    for T in [50000, 200000]:
        if elapsed() > budget_s * 0.98:
            break
        hc = labfgs(best_h, T, n, maxiter=3000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final coord polish
    if elapsed() < time.time() + min(10, budget_s - elapsed() - 1):
        t_fin = min(time.time() + 15, t0 + budget_s - 1)
        best_h, best_c5 = coord_descent(best_h, n, t_fin, max_sweeps=100)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
