        if it % 15 == 14 and best_c5 < 0.3809:
            try:
                hc2 = slsqp_polish(best_h, 2000, n, maxiter=300, ftol=1e-13)
                c5v2 = compute_c5(hc2)
                if c5v2 < best_c5:
                    best_c5, best_h = c5v2, hc2
                    work_h, work_c5 = best_h.copy(), best_c5
            except Exception:
                pass
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h.copy()

    # Phase 4: Multi-resolution (10%)
    for n_try in [256, 192]:
        if elapsed() > budget_s * 0.85:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_i = np.interp(x_new, x_old, best_h)
        h_i = project_to_sum(h_i, n_try / 2.0)
        h_i, c5i = cd_fast(h_i, time.time() + budget_s * 0.04, sweeps=60)
        if c5i < best_c5:
            best_c5, best_h, n = c5i, h_i, n_try

    # Phase 5: Final polish (15%)
    t5_end = t0 + budget_s * 0.97
    best_h, best_c5 = cd_fast(best_h, t5_end, sweeps=400)
    try:
        hc = slsqp_polish(best_h, 10000, n, maxiter=800, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
    except Exception:
        pass
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = cd_fast(best_h, t0 + budget_s * 0.99, sweeps=100)

    best_h = project_to_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
