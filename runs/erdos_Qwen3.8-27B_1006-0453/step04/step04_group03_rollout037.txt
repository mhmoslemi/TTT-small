def coordinate_descent(h, t_end, n_sweeps=500):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            h0 = best_h[i]
            # Compute a_k = (1-h[i+k]) - h[i-k] for all k
            a = np.zeros(2*n-1)
            for k in range(-(n-1), n):
                ki = k + (n-1)
                val = 0.0
                if 0 <= i+k < n:
                    val += (1.0 - best_h[i+k])
                if 0 <= i-k < n:
                    val += -best_h[i-k]
                a[ki] = val
            
            # Minimize max_k (corr[k] + s * a[k]) for s in [-h0, 1-h0]
            # This is a 1D piecewise linear function
            lo = -h0
            hi = 1.0 - h0
            if hi - lo < 1e-14:
                continue
            
            # Quick check: try s=0 (no change)
            current_max = np.max(corr)
            
            # Try a few candidate s values:
            # 0, lo, hi, and breakpoints of top correlations
            candidates = [0.0, lo, hi, -0.5, 0.5]
            # Add breakpoints from top-10 correlations
            top_k = np.argsort(corr)[-20:]
            for kk in top_k:
                for ll in top_k:
                    if kk != ll:
                        denom = a[kk] - a[ll]
                        if abs(denom) > 1e-14:
                            s_bp = (corr[ll] - corr[kk]) / denom
                            if lo <= s_bp <= hi:
                                candidates.append(s_bp)
            
            best_s = 0.0
            best_val = current_max
            for s in candidates:
                s = max(lo, min(hi, s))
                val = np.max(corr + s * a)
                if val < best_val - 1e-13:
                    best_val = val
                    best_s = s

                # Actually the piecewise-linear function is convex, so

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h.astype(np.float64), 1.0 - h, mode='full').astype(np.float64)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fix_binary_count(h, target):
    n = len(h)
    h = (h >= 0.5).astype(np.float64)
    n1 = int(h.sum())
    if n1 > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n1 - target]] = 0
    elif n1 < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n1]] = 1
    return h

def binary_local_search(h_init, t_end, max_passes=2000):
    n = len(h_init)
    h = fix_binary_count(h_init, n // 2)
    corr = compute_corr(h)
    best_val = float(np.max(corr))
    dx = 2.0 / n
    best_c5 = best_val * dx
    n_corr = 2 * n - 1
    for sweep in range(max_passes):
        if time.time() > t_end:
            break
        improved = False
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        for ii in range(len(one_idx)):
            if time.time() > t_end:
                return h, best_c5
            i = one_idx[ii]
            ci = np.zeros(n_corr)
            for k in range(-(n-1), n):
                ki = k + n - 1
                v = 0.0
                if 0 <= i + k < n:
                    v -= (1.0 - h[i + k])
                if 0 <= i - k < n:
                    v += h[i - k]
                ci[ki] = v
            cj_list = []
            for j in zero_idx:
                cj = np.zeros(n_corr)
                for k in range(-(n-1), n):
                    ki = k + n - 1
                    v = 0.0
                    if 0 <= j + k < n:
                        v += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        v -= h[j - k]
                    cj[ki] = v
                cj_list.append(cj)
            cj_mat = np.array(cj_list)
            delta = ci[np.newaxis, :] + cj_mat
            new_corr_mat = corr[np.newaxis, :] + delta
            new_max = new_corr_mat.max(axis=1)
            best_jj = int(np.argmin(new_max))
            if new_max[best_jj] < best_val - 1e-12:
                j = zero_idx[best_jj]
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                best_val = float(np.max(corr))
                best_c5 = best_val * dx
                improved = True
                one_idx = np.where(h == 1)[0]
                zero_idx = np.where(h == 0)[0]
                break
        if not improved:
            break
    return h, best_c5

def coord_descent_fast(h, t_end, n_sweeps=1000):
    n = len(h)
    dx = 2.0 / n
    bh = h.copy()
    corr = compute_corr(bh)
    best_c5 = float(np.max(corr)) * dx
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return bh, best_c5
            h0 = bh[i]
            a = np.zeros(2 * n - 1)
            for k in range(-(n-1), n):
                ki = k + n - 1
                v = 0.0
                if 0 <= i + k < n:
                    v += (1.0 - bh[i + k])
                if 0 <= i - k < n:
                    v -= bh[i - k]
                a[ki] = v
            lo = -h0
            hi = 1.0 - h0
            if hi - lo < 1e-14:
                continue
            cur_max = float(np.max(corr))
            top_k = np.argsort(corr)[-16:]
            cands = [lo, hi, 0.0]
            for kk in top_k:
                for ll in top_k:
                    if kk != ll:
                        d = a[kk] - a[ll]
                        if abs(d) > 1e-14:
                            s_bp = (corr[ll] - corr[kk]) / d
                            if lo <= s_bp <= hi:
                                cands.append(s_bp)
            best_s = 0.0
            best_v = cur_max
            for s in cands:
                s = max(lo, min(hi, s))
                v = float(np.max(corr + s * a))
                if v < best_v - 1e-14:
                    best_v = v
                    best_s = s
            if best_v < cur_max - 1e-14:
                bh[i] = h0 + best_s
                corr += best_s * a
                if float(np.max(corr)) * dx < best_c5:
                    best_c5 = float(np.max(corr)) * dx
                    improved = True
        if not improved:
            break
    return bh, best_c5

def slsqp_smooth(h, T, n, maxiter=500, ftol=1e-13):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_to_sum(np.clip(res.x, 0, 1), n / 2.0)

def fourier_init(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.ones(n) * 0.5
    for freq in range(1, 8):
        amp = rng.uniform(0.05, 0.25)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(freq * np.pi * x + phase)
    h = project_to_sum(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        best_h = np.zeros(224)

    n = len(best_h)
    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary local search on n=224 (25% budget)
    t1 = t0 + budget_s * 0.25
    h_bin = (best_h >= 0.5).astype(np.float64)
    h_bin = fix_binary_count(h_bin, n // 2)
    h_bin, c5_bin = binary_local_search(h_bin, t1, max_passes=3000)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin.copy(), c5_bin

    # Phase 2: Continuous coordinate descent (20% budget)
    t2 = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent_fast(best_h, t2, n_sweeps=2000)

    # Phase 3: Try n=384 with multiple inits (30% budget)
    t3 = t0 + budget_s * 0.75
    n_new = 384
    inits = []
    # Interpolate best
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, best_h)
    inits.append(fix_binary_count(h_interp, n_new // 2))
    # Fourier-based
    inits.append(fix_binary_count(fourier_init(n_new, seed=seed), n_new // 2))
    inits.append(fix_binary_count(fourier_init(n_new, seed=seed+100), n_new // 2))
    # Random
    hr = np.empty(n_new)
    hr[:n_new//2] = 1.0
    hr[n_new//2:] = 0.0
    np.random.shuffle(hr)
    inits.append(hr.copy())

    for init_i, h_init in enumerate(inits):
        if time.time() > t3:
            break
        t_sub = min(t3, t0 + budget_s * 0.15)
        h_b, c5_b = binary_local_search(h_init, t_sub, max_passes=800)
        if time.time() > t3 - 2:
            break
        t_cd = min(t3, time.time() + 15.0)

    t4 = t0 + budget_s * 0.97

            if c5v < best_c5:
