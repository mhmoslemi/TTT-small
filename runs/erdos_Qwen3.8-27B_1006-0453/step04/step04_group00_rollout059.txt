```python
def pair_exchange(h, i, j, n):
    """Try moving δ from h[i] to h[j] (or vice versa) to minimize C5."""
    # h[i] -> h[i]+δ, h[j] -> h[j]-δ (or opposite)
    # Find optimal δ by minimizing max of affected correlations
    # Affected correlations: those involving index i or j
    # This is expensive, so just do a 1D search
    
    def f(d):
        h_new = h.copy()
        h_new[i] = max(0, min(1, h[i] + d))
        h_new[j] = max(0, min(1, h[j] - d))
        # Need to maintain sum: if clipping happened, adjust
        diff = n/2 - h_new.sum()
        if abs(diff) > 1e-12:
            # Adjust one of them
            room_i = 1 - h_new[i]
            room_j = 1 - h_new[j]
            if diff > 0 and room_i > 0:
                h_new[i] += diff
            elif diff > 0 and room_j > 0:
                h_new[j] += diff
            elif diff < 0:
                h_new[i] += diff
                h_new[j] -= diff
            h_new = np.clip(h_new, 0, 1)
        c = get_corr(h_new)
        return np.max(c) * (2.0/n)
    
    # Search d in range
    d_max = min(h[i], 1 - h[j])  # max we can transfer i->j
    d_min = max(-h[j], h[i] - 1)  # min (transferring j->i)
    res = minimize_scalar(f, bounds=(d_min, d_max), method='bounded', options={'xatol': 1e-10})
    return res.x
```

```python
def compute_grad_soft(h, w, n):
    """Compute gradient of Σ_k w[k]*corr[k] w.r.t. h."""
    dx = 2.0/n
    A = 1.0 - h  # shape (n,)
    B = h         # shape (n,)
    
    # For corr[k] = Σ_i h[i]*(1-h[i-k+n-1])
    # d/dh[j] corr[k] = (1-h[j-k+n-1]) - h[j+k-n+1] (with validity)
    #                  = A[j-k+n-1] - B[j+k-n+1]
    
    # First term: Σ_k w[k] * A[j-k+n-1] for all j
    # This is a cross-correlation: (w padded) ⊛ A
    # Let's compute it as: for each j, sum over k where j-k+n-1 is in [0,n-1]
    
    # Vectorized: create indices
    j = np.arange(n)[:, None]  # (n, 1)
    k = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    
    idx1 = j - k + n - 1  # (n, 2n-1)
    idx2 = j + k - n + 1  # (n, 2n-1)
    
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    
    A_i1 = np.where(v1, A[np.clip(idx1, 0, n-1)], 0.0)
    B_i2 = np.where(v2, B[np.clip(idx2, 0, n-1)], 0.0)
    
    grad = (A_i1 * w[None, :] - B_i2 * w[None, :]).sum(axis=1)
    return grad * dx
```

```python
            hi_new = coord_opt_single(best_h, i, n)
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5


def multi_freq_perturb(h, n, magnitude, rng):
    x = np.arange(n)
    pert = np.zeros(n)
    n_freqs = int(rng.integers(2, 6))
    for _ in range(n_freqs):
        freq = int(rng.integers(1, max(2, n // 3)))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = magnitude / np.sqrt(n_freqs)
        pert += amp * np.sin(2 * np.pi * freq * x / n + phase)
    return pert


def basin_hop(h, n, t_end, n_iter=50, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    work_h = h.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(1, n_iter))
        mag = 0.03 + 0.12 * cool
        h_pert = work_h + multi_freq_perturb(work_h, n, mag, rng)
        h_pert = project_h(h_pert, n)
        hc, c5v = momentum_opt(h_pert, n, t_end - 5, lr_init=0.002, mu=0.7, T=400, max_iter=40)
        temp = 0.003 * cool + 0.0003
        if c5v < work_c5 or float(rng.random()) < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    return best_h, best_c5


def slsqp_polish(h, n, T, maxiter=500):
    def obj(x):
        return soft_max_val(x, T)

    def con(x):
        return x.sum() - n / 2.0

    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Momentum subgradient (broad exploration)
    h1, c5_1 = momentum_opt(best_h, n, t0 + budget_s * 0.10, lr_init=0.005, mu=0.85, T=100, max_iter=400)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: SLSQP soft-max refinement (increasing T)
    for T in [500, 2000, 8000]:
        if elapsed() > budget_s * 0.22:
            break
        hc = slsqp_polish(best_h, n, T, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Basin hopping (structured multi-frequency perturbations)
    h3, c5_3 = basin_hop(best_h, n, t0 + budget_s * 0.42, n_iter=45, rng=rng)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Phase 4: Coordinate descent (precise local search)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.68, max_sweeps=150, rng=rng)

    # Phase 5: Basin hopping round 2 (finer)
    h5, c5_5 = basin_hop(best_h, n, t0 + budget_s * 0.80, n_iter=30, rng=rng)
    if c5_5 < best_c5:
        best_h, best_c5 = h5, c5_5

    # Phase 6: Second coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.91, max_sweeps=80, rng=rng)

    # Phase 7: Final SLSQP with very high T (focus on true max)
    for T in [30000, 100000, 400000]:
        if elapsed() > budget_s * 0.96:
            break
        hc = slsqp_polish(best_h, n, T, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 8: Ultra-fine coordinate descent
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.985, max_sweeps=40, rng=rng)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
