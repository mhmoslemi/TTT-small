    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

    s = h.sum()

        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h.copy()

    diff2 = target - s2
    if diff2 > 1e-15:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-15:

    n = len(x)
    dx = 2.0 / n
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

    return x.sum() - len(x) / 2.0

def slsqp(h, T, n, maxiter=2000):

    try:

                       bounds=bnds, constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project(np.clip(res.x, 0, 1), n)

def subgrad_step(h, n, alpha):
    corr = compute_corr(h)
    k_star = np.argmax(corr)
    # gradient of corr[k_star] w.r.t. h
    grad = np.zeros(n)

    # h[i] appears as h[j] when j=i: (1-h)[i - k_star + n - 1] = 1 - h[i - k_star + n - 1] if valid
    # h[i] appears as (1-h)[j-k+n-1] when j-k+n-1 = i, i.e., j = i + k_star - n + 1: contribution -h[j]
    offset = k_star - (n - 1)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        h = h - h * (diff / max(s, 1e-15))
        h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    diff2 = target - s2
    if diff2 > 1e-15:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-15:
            h = h + room * (diff2 / tot)
    elif diff2 < -1e-15 and s2 > 1e-15:
        h = np.clip(h * (target / s2), 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def soft_obj(x, T):
    corr = compute_corr(x)
    n = len(x)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * (2.0 / n)

def con_eq(x):
    return x.sum() - len(x) / 2.0

def slsqp(h, T, n, maxiter=2000):
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(soft_obj, h, args=(T,), method='SLSQP',
                       bounds=bnds, constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

def f1d_eval(t, corr0, d, c0, mid):
    vals = corr0 + t * d
    vals[mid] = c0 + t * (1.0 - t)
    return np.max(vals)

def coord_opt(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    corr0 = compute_corr(h).copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if i >= k:
            dk += 1.0 - h[i - k]
        if i + k < n:
            dk += -h[i + k]
        d[n - 1 + k] = dk
        dk2 = 0.0
        if i + k < n:
            dk2 += 1.0 - h[i + k]
        if i >= k:
            dk2 += -h[i - k]
        d[n - 1 - k] = dk2
    c0 = corr0[n - 1]
    res = minimize_scalar(f1d_eval, args=(corr0, d, c0, n - 1),
                          bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-15})
    return res.x

def coord_descent(h, n, t_end, rng, max_sweeps=200):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            t_opt = coord_opt(h, n, i)
            if abs(t_opt - h[i]) < 5e-14:
                continue
            h[i] = t_opt
            h = project(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def pair_descent(h, n, t_end, rng, n_pairs=6000):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        found = False
        for scale in [0.02, 0.008, 0.003]:
            di = rng.choice([-1.0, 1.0]) * scale
            dj = rng.choice([-1.0, 1.0]) * scale
            test = h.copy()
            test[i] = np.clip(test[i] + di, 0, 1)
            test[j] = np.clip(test[j] + dj, 0, 1)
            test = project(test, n)
            cv = compute_c5(test)
            if cv < best_c5 - 1e-15:
                best_c5 = cv
                best_h = test.copy()
                h = test
                found = True
                break
    return best_h, best_c5

def basin_hop(h, n, t_end, rng, n_iters=80, base_step=0.025):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.8 * (it / max(n_iters, 1))
        step = base_step * (0.2 + 0.8 * cool)
        pert = work + step * rng.standard_normal(n)
        pert = project(pert, n)
        T_use = 300 + 200 * cool
        hc = slsqp(pert, T_use, n, maxiter=250)
        cv = compute_c5(hc)
        temp = 0.002 * cool + 0.00015
        if cv < work_c5 or rng.random() < np.exp(-max(0.0, cv - work_c5) / max(temp, 1e-10)):
            work, work_c5 = hc, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interp_res(h, n_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
