    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = 1.0

    n = len(best_h)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T (global smoothing)
    for T in [10, 50, 200, 800]:
        if elapsed() > budget_s * 0.12:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=1200, ftol=1e-15)
            cv = compute_c5(hc)
            if cv < best_c5 - 1e-15:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 2: Coordinate descent (main workhorse)
    t_cd1 = t0 + budget_s * 0.38
    best_h, best_c5 = coord_sweep(best_h, n, t_cd1, rng, max_sweeps=300, stall_limit=5)

    # Phase 3: Multi-start perturbation
    if elapsed() < budget_s * 0.52:
        best_h, best_c5 = multi_start_search(best_h, n, t0 + budget_s * 0.52, rng,
                                             n_starts=5, T_val=300)

    # Phase 4: Basin hopping
    if elapsed() < budget_s * 0.70:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.70, rng,
                                    n_iters=120, base_step=0.03, T_smooth=500)

    # Phase 5: Coordinate descent again (refine after basin hopping)
    t_cd2 = t0 + budget_s * 0.85
    best_h, best_c5 = coord_sweep(best_h, n, t_cd2, rng, max_sweeps=300, stall_limit=5)

    # Phase 6: Try a different resolution
    for n_try in [300, 360, 192]:
        if elapsed() > budget_s * 0.94:
            break
        if n_try == n:
            continue
        h_interp = interpolate_h(best_h, n_try)
        try:
            hc = slsqp_opt(h_interp, 500, n_try, maxiter=800, ftol=1e-14)
            cv = compute_c5(hc)
            if cv < best_c5 - 1e-6:
                best_h, best_c5, n = hc, cv, n_try
                best_h, best_c5 = coord_sweep(best_h, n, t0 + budget_s * 0.96, rng,
                                              max_sweeps=80, stall_limit=4)
        except Exception:
            pass

    # Phase 7: Final SLSQP polishing with very high T
    for T in [5000, 20000, 50000]:
        if elapsed() > budget_s * 0.985:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-16)
            cv = compute_c5(hc)
            if cv < best_c5 - 1e-15:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    # Phase 8: Final coordinate descent
    if elapsed() < budget_s * 0.995:
        best_h, best_c5 = coord_sweep(best_h, n, t0 + budget_s * 0.998, rng,
                                      max_sweeps=100, stall_limit=3)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
