Actually, let me just use the previous approach with minimize_scalar but with tighter tolerance. Or even better, since g(t) is convex (max of convex functions), we can use ternary search or binary search on the subgradient.

The subgradient of g(t) at a point t is the convex hull of {coef[k] : k in active(t)} ∪ {1-2t} (for the quadratic term). g is minimized when 0 ∈ subgradient, i.e., when there exist active k's and possibly the quadratic such that their coef values straddle 0.

This is getting complex. Let me just use a practical approach:
- Sample t at fine points (e.g., 1000 points in [0,1])
- Evaluate g(t) at each
- Find the minimum
- Refine with a few iterations of golden section

This is O(1000 * (2n-1)) per coordinate, which for n=180 is 1000*359 ≈ 359K operations. For 180 coordinates, that's 65M operations per sweep. At ~1 GFLOP/s, that's about 0.065 seconds per sweep. Very fast!

Actually, I can be smarter. For each coordinate i, I precompute the coefficients coef[k] for all k. Then g(t) = max_k (corr_base[k] + t*coef[k]) with the quadratic correction. I can vectorize the evaluation over t.

```python
def coord_exact(h, n, i):
    # Compute corr with h[i]=0
    hz = h.copy()
    hz[i] = 0.0
    corr_base = corr_full(hz)
    dx = 2.0/n
    
    # Compute coef[k] = contribution of t=h[i] to corr[k]
    coef = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = k - (n-1)
        j1 = i + lag
        j2 = i - lag
        c = 0.0
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k] = c
    
    # Quadratic correction for k=n-1
    # corr[n-1] = corr_base[n-1] + t*(1-t) when h[i]=t (since h[i]*(1-h[i]) = t - t^2)
    # But wait: when hz[i]=0, corr_base[n-1] already has -h[i]*(1-h[i]) = 0 contributed
    # Actually: corr_base[n-1] = Σ_j hz[j]*(1-hz[j]) = Σ_{j≠i} h[j]*(1-h[j])
    # With h[i]=t: corr[n-1] = corr_base[n-1] + t*(1-t)
    # And the linear formula gives: corr_base[n-1] + t * coef[n-1]
    # where coef[n-1] = (1-h[i]) * [i+(n-1-(n-1)) < n] + (-h[i-(n-1-(n-1))]) * [...]
    # lag=0: j1=i, j2=i
    # coef[n-1] = (1-h[i]) + (-h[i]) = 1 - 2*h[i]
    # But the true contribution is t*(1-t) = t - t^2
    # Linear formula gives: t*(1-2*h[i])
    # Difference: t - t^2 - t*(1-2*h[i]) = t - t^2 - t + 2*t*h[i] = t*(2*h[i] - t)
    # Hmm, this is because we're using h[i] in the coefficient, not t.
    
    # Let me redo: with h[i]=t and all others fixed at their current values:
    # corr_{n-1} = Σ_j h[j](1-h[j]) with h[i]=t
    #            = (Σ_{j≠i} h[j](1-h[j])) + t(1-t)
    #            = corr_base[n-1] + t(1-t)  [since hz[i]=0, so corr_base doesn't include i]
    
    # For k ≠ n-1:
    # corr_k = corr_base[k] + t * (contribution of position i to corr_k)
    # The contribution: looking at the formula corr_k = Σ h[a](1-h[b]) where (a,b) depends on k
    # h[i] appears as h[a] or h[b] in specific terms.
    
    # When h[i]=0 (i.e., hz), corr_base[k] = sum over all terms NOT involving h[i]
    # When h[i]=t, the additional terms are those where i participates.
    
    # For lag = k-(n-1) ≥ 0: corr = Σ_{j=0}^{n-1-lag} h[j](1-h[j+lag])
    #   h[i] appears as h[j] if j=i (i.e., i ≤ n-1-lag): term = t*(1-h[i+lag])
    #   h[i] appears as h[j+lag] if j+lag=i (j=i-lag, i.e., lag ≤ i ≤ n-1): term = h[i-lag]*(1-t)
    # Additional contribution to corr: t*(1-h[i+lag]) + h[i-lag]*(1-t) - [had 0 before]
    #   Wait, when h[i]=0: the terms are 0*(1-h[i+lag]) + h[i-lag]*(1-0) = h[i-lag]
    #   When h[i]=t: t*(1-h[i+lag]) + h[i-lag]*(1-t)
    #   Difference: t*(1-h[i+lag]) + h[i-lag] - h[i-lag]*t - h[i-lag] = t*(1-h[i+lag]-h[i-lag])
    #   = t * (1 - h[i+lag] - h[i-lag])
    #   This matches coef[k] computed above (with the fixed h values, not t).
    
    # So for k ≠ n-1: corr_k(t) = corr_base[k] + t * coef[k]  ✓
    # For k = n-1: corr_{n-1}(t) = corr_base[n-1] + t*(1-t)
    #   But our formula gives: corr_base[n-1] + t * coef[n-1] where coef[n-1] = (1-h[i]) - h[i] = 1-2h[i]
    #   True: corr_base[n-1] + t - t^2
    #   Formula: corr_base[n-1] + t*(1-2h[i])
    #   These are equal only if t = h[i] (since t - t^2 = t*(1-t) and t*(1-2h[i]) = t*(1-h[i])-t*h[i])
    #   At t = h[i]: t*(1-t) = t*(1-h[i])  ✓ since t=h[i]
    #   But in general they differ. The correction is:
    #   t*(1-t) - t*(1-2h[i]) = t - t^2 - t + 2t*h[i] = t*(2h[i] - t)
    
    # So: corr_{n-1}(t) = corr_base[n-1] + t*coef[n-1] + t*(2*h[i] - t)
    #                       = corr_base[n-1] + t*(1-2h[i]) + t*2h[i] - t^2
    #                       = corr_base[n-1] + t - t^2  ✓
    
    # OK so the formula in the previous code is correct:
    # cv = corr_base + t * coef
    # cv[n-1] = corr_base[n-1] + t * (1 - t)
    
    # Now to find min of max(cv) over t in [0,1]:
    # cv[k] = corr_base[k] + t*coef[k] for k ≠ n-1
    # cv[n-1] = corr_base[n-1] + t - t^2
    
    # This is max of (2n-2) linears and 1 quadratic (opening downward) on [0,1].
    # The max is convex, so we can minimize it.
    
    # Practical approach: evaluate on a fine grid + refine
    t_grid = np.linspace(0, 1, 200)
    vals = np.zeros(200)
    for idx, t in enumerate(t_grid):
        cv = corr_base + t * coef
        cv[n-1] = corr_base[n-1] + t * (1 - t)
        vals[idx] = np.max(cv)
    t_best = t_grid[np.argmin(vals)]
    
    # Refine with golden section
    a, b = t_grid[max(0, np.argmin(vals)-1)], t_grid[min(199, np.argmin(vals)+1)]
    for _ in range(50):
        m1 = a + (b-a)*0.382
        m2 = a + (b-a)*0.618
        def f(t):
            cv = corr_base.copy() + t * coef
            cv[n-1] = corr_base[n-1] + t * (1-t)
            return np.max(cv)
        if f(m1) < f(m2):
            b = m2
        else:
            a = m1
    return (a+b)/2
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, T):
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / T

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n - lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += pk * (-h[nl:])
    return grad * dx

def project_box_sum(h, s):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        hc = np.clip(h - mid, 0.0, 1.0)
        if hc.sum() > s:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def projected_grad(h, T, lr, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    target = n / 2.0
    for _ in range(50000):
        if time.time() > t_end:
            break
        g = grad_smooth(best_h, T)
        step = lr
        ok = False
        for _ in range(25):
            trial = project_box_sum(best_h - step * g, target)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-16:
                best_h = trial
                best_c5 = c5t
                ok = True
                break
            step *= 0.5
        if not ok:
            lr *= 0.8
            if lr < 1e-14:
                break
        T = min(T * 1.01, 80000)
    return best_h, best_c5

def active_subgrad_step(h, n, lr, t_end, n_iter):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    target = n / 2.0
    dx = 2.0 / n
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        corr = corr_full(best_h)
        mx = np.max(corr)
        active = corr >= mx - 1e-10 * mx
        cnt = np.sum(active)
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            lag = k - (n - 1)
            if lag >= 0:
                grad[:n - lag] += (1.0 - best_h[lag:n])
                grad[lag:] += (-best_h[:n - lag])
            else:
                nl = -lag
                grad[nl:] += (1.0 - best_h[:n - nl])
                grad[:n - nl] += (-best_h[nl:])
        grad /= cnt
        grad *= dx
        step = lr
        ok = False
        for _ in range(20):
            trial = project_box_sum(best_h - step * grad, target)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-16:
                best_h = trial
                best_c5 = c5t
                ok = True
                break
            step *= 0.5
        if not ok:
            lr *= 0.9
            if lr < 1e-14:
                break
    return best_h, best_c5

def coord_one(h, n, i):
    hz = h.copy()
    hz[i] = 0.0
    corr_base = corr_full(hz)
    k0 = n - 1
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k] = c
    t_grid = np.linspace(0, 1, 500)
    vals = corr_base[None, :] + t_grid[:, None] * coef[None, :]
    vals[:, k0] = corr_base[k0] + t_grid * (1.0 - t_grid)
    maxes = vals.max(axis=1)
    idx = np.argmin(maxes)
    a = t_grid[max(0, idx - 1)]
    b = t_grid[min(499, idx + 1)]
    for _ in range(40):
        m1 = a + (b - a) * 0.381966
        m2 = a + (b - a) * 0.618034
        v1 = np.max(corr_base + m1 * coef)
        v1 = max(v1, corr_base[k0] + m1 * (1.0 - m1))
        v2 = np.max(corr_base + m2 * coef)
        v2 = max(v2, corr_base[k0] + m2 * (1.0 - m2))
        if v1 < v2:
            b = m2
        else:
            a = m1
    return 0.5 * (a + b)

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    target = n / 2.0
    rng = np.random.default_rng(12345)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            t_new = coord_one(best_h, n, i)
            if abs(t_new - best_h[i]) < 1e-10:
                continue
            trial = best_h.copy()
            trial[i] = t_new
            trial = project_box_sum(trial, target)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = trial
                improved = True
        if not improved:
            break
    return best_h, best_c5

def multi_start(h, n, t_end, n_starts=8):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    target = n / 2.0
    rng = np.random.default_rng(99)
    for s in range(n_starts):
        if time.time() > t_end:
            break
        amp = 0.15 * (1.0 - s / n_starts) + 0.03
        h_p = best_h + amp * rng.standard_normal(n)
        h_p = project_box_sum(h_p, target)
        h_r, c5_r = projected_grad(h_p, T=200, lr=0.01, n=n, t_end=min(time.time() + 120, t_end))
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
        if time.time() < t_end - 60:
            h_r2, c5_r2 = coord_descent(h_r, n, t_end=min(time.time() + 120, t_end), max_sweeps=80)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    target = n / 2.0
    best_h = project_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient with increasing T
    phase1_end = t0 + budget_s * 0.20
    best_h, best_c5 = projected_grad(best_h, T=50, lr=0.015, n=n, t_end=phase1_end)

    # Phase 2: Active subgradient
    phase2_end = t0 + budget_s * 0.30
    best_h, best_c5 = active_subgrad_step(best_h, n, lr=0.005, t_end=phase2_end, n_iter=20000)

    # Phase 3: Coordinate descent
    phase3_end = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, n, t_end=phase3_end, max_sweeps=200)

    # Phase 4: Multi-start
    phase4_end = t0 + budget_s * 0.72
    best_h, best_c5 = multi_start(best_h, n, t_end=phase4_end, n_starts=6)

    # Phase 5: More coordinate descent
    phase5_end = t0 + budget_s * 0.85
    best_h, best_c5 = coord_descent(best_h, n, t_end=phase5_end, max_sweeps=400)

    # Phase 6: Final projected gradient polish with very high T
    phase6_end = t0 + budget_s * 0.93
    best_h, best_c5 = projected_grad(best_h, T=10000, lr=0.001, n=n, t_end=phase6_end)

    # Phase 7: Final coord descent
    best_h, best_c5 = coord_descent(best_h, n, t_end=t0 + budget_s * 0.98, max_sweeps=200)

    # Final projection and evaluation
    best_h = project_box_sum(best_h, target)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
