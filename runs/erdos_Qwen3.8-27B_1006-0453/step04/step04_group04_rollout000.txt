def paired_coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    swap = n // 2
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            j = (i + swap) % n
            if j == i:
                continue
            lo = max(0.0, best_h[i] + best_h[j] - 1.0)
            hi = min(1.0, best_h[i] + best_h[j])
            if hi - lo < 1e-12:
                continue
            def f1d(t):
                hz = best_h.copy()
                hz[i] = t
                hz[j] = best_h[i] + best_h[j] - t
                c = np.correlate(hz, 1.0 - hz, mode='full')
                return float(np.max(c))
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-12, 'maxiter': 150})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try[j] = best_h[i] + best_h[j] - t_new
            h_try = np.clip(h_try, 0.0, 1.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def softmax_slsqp(h, n, T, maxiter=600):
    def obj(x):
        return softmax_obj(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return proj(np.clip(res.x, 0, 1), n)

def adaptive_T(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sorted_c = np.sort(corr)[-8:]
    spread = sorted_c[0] - sorted_c[-1]
    if spread < 1e-10:
        return 20000.0
    return min(max(50.0 / spread, 50.0), 100000.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n = len(best_h)
    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Adaptive softmax SLSQP
    for round_idx in range(6):
        if elapsed() > budget_s * 0.06:
            break
        T = adaptive_T(best_h)
        T = max(T, 100.0 * (round_idx + 1))
        try:
            hc = softmax_slsqp(best_h, n, T, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 2: Intensive coordinate descent
    if elapsed() < budget_s * 0.30:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.30,
                                        max_sweeps=800, xatol=1e-14)

    # Phase 3: Paired coordinate descent
    if elapsed() < budget_s * 0.40:
        best_h, best_c5 = paired_coord_descent(best_h, n, t0 + budget_s * 0.40,
                                               max_sweeps=150)

    # Phase 4: More softmax at high T
    for T in [5000, 20000, 60000]:
        if elapsed() > budget_s * 0.48:
            break
        try:
            hc = softmax_slsqp(best_h, n, T, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass

    # Phase 5: Coordinate descent again
    if elapsed() < budget_s * 0.60:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.60,
                                        max_sweeps=1000, xatol=1e-15)

    # Phase 6: Basin hopping (few iters)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(12):
        if elapsed() > budget_s * 0.72:
            break
        cool = 1.0 - 0.7 * (it / 12.0)
        step = 0.03 + 0.08 * cool
        h_pert = work_h + step * np.random.randn(n)
        h_pert = proj(h_pert, n)
        try:
            hc = softmax_slsqp(h_pert, n, 2000, maxiter=150)
            c5v = compute_c5(hc)
            temp = 0.002 * cool + 0.0001
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-12)):
                work_h = hc
                work_c5 = c5v
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
        except Exception:
            pass
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 7: Try n=200
    if elapsed() < budget_s * 0.82:
        n2 = 200
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = proj(h_interp, n2)
        for T in [3000, 30000]:
            if elapsed() > budget_s * 0.84:
                break
            try:
                hc2 = softmax_slsqp(h_interp, n2, T, maxiter=500)
                c5v2 = compute_c5(hc2)
                if c5v2 < best_c5:
                    hc2, c5v2 = coord_descent(hc2, n2, t0 + budget_s * 0.85,
                                              max_sweeps=300, xatol=1e-14)
                    if c5v2 < best_c5:
                        best_c5 = c5v2
                        best_h = hc2
                        n = n2
            except Exception:
                pass

    # Phase 8: Final intensive coordinate descent
    if elapsed() < budget_s * 0.94:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94,
                                        max_sweeps=2000, xatol=1e-15)

    # Phase 9: Final paired + polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = paired_coord_descent(best_h, n, t0 + budget_s * 0.97,
                                               max_sweeps=100)

    # Final: ultra-tight coordinate sweep
    for _ in range(3):
        if elapsed() > budget_s * 0.99:
            break
        improved = False
        for i in range(n):
            t_new = coord_opt(best_h, n, i, xatol=1e-15)
            if abs(t_new - best_h[i]) > 1e-14:
                best_h[i] = t_new
                best_h = proj(best_h, n)
                improved = True
        c5v = compute_c5(best_h)
        if c5v < best_c5:
            best_c5 = c5v
        if not improved:
            break

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
