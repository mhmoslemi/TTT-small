        best_h, best_c5 = basin_hop_structured(best_h, n, t0 + budget_s * 0.72, rng, n_iters=200)

    # Phase 6: Post-hopping coordinate descent
    if time.time() - t0 < budget_s * 0.82:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82, max_sweeps=50, rng=rng)

    # Phase 7: Multi-resolution refinement
    for n_try in [280, 320]:
        if time.time() - t0 > budget_s * 0.92:
            break
        h_interp = interpolate_h(best_h, n_try)
        for T in [500, 3000]:
            if time.time() - t0 > budget_s * 0.94:
                break
            try:
                hc = slsqp_opt(h_interp, T, n_try, maxiter=1200, ftol=1e-15)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5, n = hc, cv, n_try
                    h_interp = hc.copy()
            except Exception:
                pass
        if n == n_try:
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, max_sweeps=25, rng=rng)

    # Phase 8: Final precision polish
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=30, rng=rng)
    for T in [20000, 80000]:
        if time.time() - t0 > budget_s * 0.995:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-16)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
