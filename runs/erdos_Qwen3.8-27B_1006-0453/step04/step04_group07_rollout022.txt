                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def structured_perturb(h, n, rng, method):
    h_p = h.copy()
    if method == 0:
        m = rng.randint(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.25)
        x = np.linspace(0, 2, n, endpoint=False)
        h_p += amp * np.sin(2 * np.pi * m * x + phase)
    elif method == 1:
        s = rng.randint(1, n // 4 + 1)
        h_p = np.roll(h, s)
    elif method == 2:
        b = max(2, n // 10)
        i = rng.randint(0, n - 2 * b)
        j = i + b
        h_p[i:i + b] = h[i + b:i + 2 * b]
        h_p[i + b:i + 2 * b] = h[i:i + b]
    elif method == 3:
        k = rng.randint(1, n // 3)
        i1 = rng.randint(0, n - k)
        i2 = rng.randint(0, n - k)
        if abs(i1 - i2) >= k:
            tmp = h_p[i1:i1 + k].copy()
            h_p[i1:i1 + k] = h_p[i2:i2 + k]
            h_p[i2:i2 + k] = tmp
    else:
        h_p += rng.uniform(0.03, 0.18) * rng.randn(n)
    return project(h_p, n)

def basin_hop(h, c5, n, t_end, max_iters=300, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h, best_c5 = h.copy(), c5
    work_h, work_c5 = h.copy(), c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.03, 1.0 - 0.92 * it / max_iters)
        method = rng.randint(5)
        h_p = structured_perturb(work_h, n, rng, method)
        hc, c5v = coord_descent(h_p, n, min(time.time() + 4.0, t_end), max_sweeps=60)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.ones(160) * 0.5
        for i in range(0, 160, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.18, max_sweeps=3000)

    # Phase 2: Smoothed SLSQP with increasing T
    for T in [50, 200, 800, 3000]:
        if time.time() > t0 + budget_s * 0.28:
            break
        try:
            hc = slsqp_smooth(best_h, T, n, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.29, max_sweeps=200)
        except Exception:
            pass

    # Phase 3: Structured basin hopping
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.62, max_iters=500, rng=rng)

    # Phase 4: Multi-resolution exploration
    for n_try in [100, 128, 144, 192, 256]:
        if time.time() > t0 + budget_s * 0.78:
            break
        h_tr = interp_h(best_h, n_try)
        t_sub = min(t0 + budget_s * 0.79, time.time() + budget_s * 0.025)
        h_tr, c5_tr = coord_descent(h_tr, n_try, t_sub, max_sweeps=800)
        if c5_tr < best_c5:
            best_c5, best_h, n = c5_tr, h_tr, n_try

    # Phase 5: Second basin hop with different seed
    rng2 = np.random.RandomState(seed + 7)
    if time.time() < t0 + budget_s * 0.88:
        h2, c5_2 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.88, max_iters=200, rng=rng2)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2

    # Phase 6: Final coordinate descent + SLSQP
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, max_sweeps=3000)
    if time.time() < t0 + budget_s * 0.97:
        for T in [5000, 15000]:
            if time.time() > t0 + budget_s * 0.97:
                break
            try:
                hc = slsqp_smooth(best_h, T, n, maxiter=800)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
                best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.975, max_sweeps=500)
            except Exception:
                pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
