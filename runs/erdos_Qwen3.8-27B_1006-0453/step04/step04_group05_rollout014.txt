        if L == 0:
            grad += wi * (1.0 - 2.0*h)
        elif L > 0:
            n = N - L
            grad[L:] += wi * (1.0 - h[:n])
            grad[:n] -= wi * h[L:]
        else:
            M = -L
            if M < N:
                n = N - M
                grad[:n] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:n]
    return grad, float(corr.max()), top_idx

def adam_optimize(h, N, target, deadline, k=50, lr=0.03, beta_start=5.0, beta_end=80.0):
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    total_iters = 3000
    while time.time() < deadline:
        prog = min(1.0, it / total_iters)
        cur_beta = beta_start + (beta_end - beta_start) * prog
        cur_k = k
        if it > total_iters // 2:
            cur_k = max(10, k // 2)
        grad, peak, _ = grad_topk_softmax(h, N, cur_k, cur_beta)
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        it += 1
        mh = m / (1-b1**it)
        vh = v / (1-b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 2000 == 0:
            lr *= 0.95
    return best_h, best_peak * dx

def binary_local_search(h, N, deadline, max_sweeps=200):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h==1)[0]
        h[idx[nh:]] = 0
    elif n1 < nh:
        idx = np.where(h==0)[0]
        h[idx[:nh-n1]] = 1
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        P = float(corr.max())
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        center = N-1
        d_lags = np.arange(2*N-1) - center
        best_new_P = P
        best_pq = None
        for p in ones:
            # A_p[k] = change in corr[k] when h[p] goes 1->0
            # = -(1-h[p+L]) + h[p-L] (with bounds)
            jp = p + d_lags  # p+L for all lags
            jm = p - d_lags  # p-L for all lags
            valid_p = (jp >= 0) & (jp < N)
            valid_m = (jm >= 0) & (jm < N)
            A_p = np.zeros(2*N-1)
            A_p[valid_p] -= (1.0 - h[np.clip(jp, 0, N-1)][valid_p])
            A_p[valid_m] += h[np.clip(jm, 0, N-1)][valid_m]
            base = corr + A_p
            # For each q: B_q[k] = (1-h[q+L]) - h[q-L]
            # new_peak(q) = max(base + B_q)
            # We want argmin_q max_k(base[k] + B_q[k])
            # Approximate: compute for all q's
            jq = zeros[:,None] + d_lags[None,:]
            jmm = zeros[:,None] - d_lags[None,:]
            valid_qp = (jq >= 0) & (jq < N)
            valid_qm = (jmm >= 0) & (jmm < N)
            h_qp = np.where(valid_qp, h[np.clip(jq, 0, N-1)], 0.0)
            h_qm = np.where(valid_qm, h[np.clip(jmm, 0, N-1)], 0.0)
            B = np.zeros_like(base)  # placeholder, shape (len(zeros), 2N-1)
            B = valid_qp * (1.0 - h_qp) - valid_qm * h_qm
            new_peaks = np.max(base[None,:] + B, axis=1)
            mi = np.argmin(new_peaks)
            if new_peaks[mi] < best_new_P:
                best_new_P = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_topk(h, N, k, beta):
    corr = compute_corr(h)
    center = N - 1
    k = min(k, 2 * N - 1)
    top_idx = np.argpartition(corr, -k)[-k:]
    tv = corr[top_idx].copy()
    tv -= tv.max()
    w = np.exp(beta * tv)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(k) / k
    else:
        w /= ws
    grad = np.zeros(N)
    for i in range(k):
        wi = w[i]
        if wi < 1e-12:
            continue
        L = top_idx[i] - center
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L > 0:
            n = N - L
            grad[L:] += wi * (1.0 - h[:n])
            grad[:n] -= wi * h[L:]
        else:
            M = -L
            if M < N:
