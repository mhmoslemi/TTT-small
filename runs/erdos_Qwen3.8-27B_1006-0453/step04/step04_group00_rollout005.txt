```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n=None):
    if n is None:
        n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        excess = s - target
        # Scale down non-zero entries
        mask = h > 1e-15
        if mask.sum() == 0:
            return h
        h[mask] -= excess * (h[mask] / h[mask].sum())
        h = np.clip(h, 0.0, 1.0)
    else:
        deficit = target - s
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h += deficit * room / tot
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

    # Minimize max_k (base[k] + x*g[k]) over x in [0,1]
    base = corr - h[i] * g
    # Use bisection on the 1D convex piecewise-linear function
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = base + mid * g
        mx = np.max(vals)
        # gradient: sign of max g[k] among active
        active = np.abs(vals - mx) < 1e-12
        slope = np.max(g[active]) - np.min(g[active])
        # Actually for convex PWL, subgradient at mid is:
        # if multiple k achieve max: subgradient is convex hull of {g[k]: k active}
        # For 1D: if slope > 0, decrease x; if slope < 0, increase x
        # Simpler: subgradient = g[argmax(vals)] (one of the active)
        # But better: if max of active g > 0, go left; if min of active g < 0, go right;
        # if all active g are same sign, go that way
        if np.min(g[active]) > 1e-10:
            hi = mid
        elif np.max(g[active]) < -1e-10:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0
```

Wait, the bisection logic isn't quite right. For a convex piecewise-linear function f(x) = max_k (a_k + b_k·x), the subgradient at x is the set {b_k : k active}. The function is decreasing where the subgradient contains negative values and increasing where it contains positive values. The minimum is where 0 is in the subgradient set.

Better approach: just evaluate f at a grid of points or use ternary search (since f is convex):

```python
def find_1d_min(base, g, n_iter=50):
    """Minimize max_k(base[k] + x*g[k]) over x in [0,1]"""
    def f(x):
        return np.max(base + x * g)
    # Ternary search (works for convex functions)
    lo, hi = 0.0, 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        excess = s - target
        mask = h > 1e-14
        if mask.sum() == 0:
            return h
        h[mask] -= excess * (h[mask] / h[mask].sum())
        h = np.clip(h, 0.0, 1.0)
    else:
        deficit = target - s
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += deficit * room / tot
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, T, n, maxiter=400, ftol=1e-12):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return float((m + np.log(np.sum(np.exp(T*(c-m))))/T) * dx)
    def con(x):
        return x.sum() - n/2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter':maxiter,'ftol':ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def find_1d_min(base, g):
    def f(x):
        return np.max(base + x * g)
    lo, hi = 0.0, 1.0
    for _ in range(45):
        m1 = lo + (hi - lo)/3.0
        m2 = hi - (hi - lo)/3.0
        if f(m1) <= f(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def fast_coord_descent(h, n, t_end, max_sweeps=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = compute_corr(best_h)
    dx = 2.0 / n
    stagnation = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            g = np.zeros(2*n - 1)
            for k in range(2*n - 1):
                m = k - n + 1
                j1 = i + m
                j2 = i - m
                c1 = (1.0 - best_h[j1]) if 0 <= j1 < n else 0.0
                c2 = best_h[j2] if 0 <= j2 < n else 0.0
                g[k] = c1 - c2
            base = corr - best_h[i] * g
            x_new = find_1d_min(base, g)
            if abs(x_new - best_h[i]) > 1e-12:
                delta = x_new - best_h[i]
                best_h[i] = x_new
                corr += delta * g
                c5_try = np.max(corr) * dx
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    improved = True
                else:
                    best_h[i] -= delta
                    corr -= delta * g
        if not improved:
            stagnation += 1
            if stagnation > 3:
                break
        else:
            stagnation = 0
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def make_binary_start(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def make_sine_start(n, seed, freq):
    rng = np.random.RandomState(seed)
    phase = rng.uniform(0, 2*np.pi)
    x = np.arange(n)
    h = 0.5 + 0.5*np.sin(2*np.pi*freq*x/n + phase)
    return project_h(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_binary_start(160, seed)
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP annealing from parent (20% time)
    for T in [50, 200, 1000, 5000, 20000]:
        if elapsed() > budget_s * 0.18:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=600, ftol=1e-13)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Fast coordinate descent polish (10% time)
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.28, max_sweeps=2000)

    # Phase 2: Basin hopping with diverse perturbations (35% time)
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 60
    max_bh = int(budget_s * 0.63)
    for it in range(n_bh):
        if elapsed() > max_bh:
            break
        cool = 1.0 - 0.85*(it / n_bh)
        mod = it % 5
        if mod == 0:
            pert = (0.04 + 0.12*cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(2, 12)
            phase = np.random.uniform(0, 2*np.pi)
            pert = (0.03 + 0.10*cool) * np.sin(2*np.pi*freq*np.arange(n)/n + phase)
        elif mod == 2:
            mask = np.random.random(n) < 0.25
            pert = np.zeros(n)
            pert[mask] = (0.08 + 0.15*cool) * np.random.randn(mask.sum())
        elif mod == 3:
            # Swap-based perturbation
            pert = np.zeros(n)
            nswap = max(2, n//20)
            for _ in range(nswap):
                a, b = np.random.randint(0, n, 2)
                pert[a] = work_h[b] - work_h[a]
                pert[b] = work_h[a] - work_h[b]
        else:
            pert = (0.02 + 0.06*cool) * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T_bh = 200 + 3000*cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=300, ftol=1e-12)
        # Quick coord descent
        hc, c5v = fast_coord_descent(hc, n, t0 + budget_s*0.63, max_sweeps=200)
        temp = 0.0015*cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5)/temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 3: Extended fast coordinate descent (15% time)
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.80, max_sweeps=5000)

    # Phase 4: Try different n values (10% time)
    for n_try in [80, 128]:
        if elapsed() > budget_s * 0.90:
            break
        if n_try == n:
            continue
        # Downsample or resample
        if n_try < n:
            step = n // n_try
            h_try = best_h[:n_try*step].reshape(n_try, step).mean(axis=1)
        else:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]))
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        # Optimize this
        for T in [500, 5000]:
            if elapsed() > budget_s * 0.90:
                break
            hc = slsqp_opt(h_try, T, n_try, maxiter=300, ftol=1e-12)
            hc, c5_try = fast_coord_descent(hc, n_try, t0 + budget_s*0.90, max_sweeps=500)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            n = n_try

    # Phase 5: Final polish (10% time)
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.94, max_sweeps=3000)
    for T in [50000, 200000]:
        if elapsed() > budget_s * 0.97:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=800, ftol=1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final coordinate descent
    if elapsed() < budget_s * 0.985:
        best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=2000)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
