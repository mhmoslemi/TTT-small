```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # weights
    ex = np.exp(T * (corr - m))
    w = ex / ex.sum()  # softmax weights
    # Now compute weighted gradient
    grad = np.zeros(n)
    # k=0 contribution
    grad += w[n-1] * (1.0 - 2.0 * h)
    # k>0 contributions
    for k in range(1, n):
        wk = w[n-1+k]
        wm = w[n-1-k]
        # For lag +k: grad += wk * [(1-h[i-k]) for i=k..n-1] - wk * [h[i+k] for i=0..n-1-k]
        if wk > 1e-15:
            grad[k:] += wk * (1.0 - h[:-k])
            grad[:n-k] -= wk * h[k:]
        # For lag -k: grad += wm * [(1-h[i+k]) for i=0..n-1-k] - wm * [h[i-k] for i=k..n-1]
        if wm > 1e-15:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:-k]
    return grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        s2 = h.sum()
        if s2 < 1e-14:
            return h
        h = h - h * (diff / s2)
        h = np.clip(h, 0.0, 1.0)
        s3 = h.sum()
        diff3 = target - s3
        if diff3 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff3 / tot)
        elif diff3 < -1e-14 and s3 > 1e-14:
            h = np.clip(h * (target / s3), 0.0, 1.0)
    return h

def grad_desc(h, n, T, t_end, lr_init=0.01, n_max_iter=5000):
    """Projected gradient descent with analytical gradient"""
    best = h.copy()
    best_c5 = compute_c5(best)
    h_work = h.copy()
    lr = lr_init
    for it in range(n_max_iter):
        if time.time() > t_end:
            break
        g = smooth_grad(h_work, T)
        # Subtract mean for sum constraint
        g = g - g.mean()
        h_new = h_work - lr * g
        h_new = project(h_new, n)
        # Line search (simple backtracking)
        f_new = smooth_obj(h_new, T)
        f_old = smooth_obj(h_work, T)
        if f_new > f_old:
            # Backtrack
            for _ in range(10):
                lr *= 0.5
                h_new = h_work - lr * g
                h_new = project(h_new, n)
                f_new = smooth_obj(h_new, T)
                if f_new <= f_old:
                    break
        h_work = h_new
        c5 = compute_c5(h_work)
        if c5 < best_c5 - 1e-12:
            best = h_work.copy()
            best_c5 = c5
    return best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ex = np.exp(T * (corr - m))
    w = ex / ex.sum()
    grad = np.zeros(n)
    w0 = w[n - 1]
    grad += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k]
        wm = w[n - 1 - k]
        if wk > 1e-12:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        if wm > 1e-12:
            grad[:n - k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n - k]
    return grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h - h * (diff / s)
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-12:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-12 and s2 > 1e-14:
            h = np.clip(h * (target / s2), 0.0, 1.0)
    return h

def grad_descent(h, n, t_end, T0=50.0, T1=20000.0, lr0=0.05):
    best = h.copy()
    best_c5 = compute_c5(best)
    h_w = h.copy()
    lr = lr0
    n_iter = 0
    for it in range(20000):
        if time.time() > t_end:
            break
        frac = min(1.0, (it + 1) / 5000.0)
        T = T0 * (T1 / T0) ** frac
        g = smooth_grad(h_w, T)
        g = g - g.mean()
        f_old = smooth_obj(h_w, T)
        step = lr * (1.0 - 0.8 * frac)
        for _ in range(15):
            h_try = project(h_w - step * g, n)
            f_try = smooth_obj(h_try, T)
            if f_try <= f_old + 1e-15:
                break
            step *= 0.5
        h_w = h_try
        if it % 20 == 0:
            c5 = compute_c5(h_w)
            if c5 < best_c5 - 1e-13:
                best = h_w.copy()
                best_c5 = c5
    c5 = compute_c5(h_w)
    if c5 < best_c5:
        best, best_c5 = h_w.copy(), c5
    return best, best_c5

def coord_descent(h, n, t_end, max_sweeps=80, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_save = best_h[i]
            best_h[i] = 0.0
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
            best_h[i] = h_save
            d = np.zeros(2 * n - 1)
            for k in range(1, n):
                if k <= i <= n - 1:
                    d[n - 1 + k] += (1.0 - best_h[i - k])
                d[n - 1 - k] += (1.0 - best_h[i + k]) if i + k < n else 0.0
                if i + k < n:
                    d[n - 1 + k] += (-best_h[i + k])
                if i - k >= 0:
                    d[n - 1 - k] += (-best_h[i - k])
            c0 = corr_base[n - 1]
            def f1d(t):
                vals = corr_base + t * d
                vals[n - 1] = c0 + t * (1.0 - t)
                return np.max(vals) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-14})
            new_val = res.x
            if abs(new_val - h_save) > 1e-10:
                best_h[i] = new_val
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_obj(x, T)
    res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_kill = t0 + budget_s * 0.97

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradient descent with increasing T
    h1, c51 = grad_descent(best_h, n, t0 + budget_s * 0.08, T0=20, T1=5000, lr0=0.04)
    if c51 < best_c5:
        best_h, best_c5 = h1, c51

    # Phase 2: Coordinate descent
    h2, c52 = coord_descent(best_h, n, t0 + budget_s * 0.22, max_sweeps=80, rng=rng)
    if c52 < best_c5:
        best_h, best_c5 = h2, c52

    # Phase 3: Multi-start basin hopping
    n_starts = 8
    t_basin_end = t0 + budget_s * 0.60
    per_start = (t_basin_end - t0) / n_starts
    for si in range(n_starts):
        if time.time() > t_basin_end:
            break
        t_sub = t0 + (si + 1) * per_start
        freq = rng.integers(2, 12)
        amp = rng.uniform(0.03, 0.12) * (1.0 - 0.08 * min(si, 5))
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n, endpoint=False)
        pert = best_h + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
        pert = project(pert, n)
        # Gradient polish
        hc, c5c = grad_descent(pert, n, min(t_sub, time.time() + 0.01),
                                T0=30, T1=8000, lr0=0.03)
        # Coordinate descent
        hc, c5c = coord_descent(hc, n, t_sub, max_sweeps=30, rng=rng)
        if c5c < best_c5:
            best_h, best_c5 = hc, c5c

    # Phase 4: SLSQP polish on best
    if elapsed() < budget_s * 0.72:
        for T in [300, 2000, 10000]:
            if elapsed() > budget_s * 0.72:
                break
            try:
                hc = slsqp_opt(best_h, T, n, maxiter=400, ftol=1e-14)
                cv = compute_c5(hc)
                if cv < best_c5:
                    best_h, best_c5 = hc, cv
            except Exception:
                pass

    # Phase 5: More coordinate descent
    if elapsed() < budget_s * 0.82:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82,
                                         max_sweeps=60, rng=rng)

    # Phase 6: Multi-resolution
    for n_try in [n - 40, n - 20, n + 20, n + 40]:
        if elapsed() > budget_s * 0.92:
            break
        if n_try < 80:
            continue
        h_interp = interpolate_h(best_h, n_try)
        hc, c5c = grad_descent(h_interp, n_try, t0 + budget_s * 0.88, T0=50, T1=10000, lr0=0.03)
        hc, c5c = coord_descent(hc, n_try, t0 + budget_s * 0.91, max_sweeps=25, rng=rng)
        if c5c < best_c5:
            best_h, best_c5, n = hc, c5c, n_try

    # Phase 7: Final polish
    if elapsed() < budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95,
                                         max_sweeps=40, rng=rng)
    if elapsed() < budget_s * 0.97:
        try:
            hc = slsqp_opt(best_h, 50000, n, maxiter=1000, ftol=1e-15)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv
        except Exception:
            pass

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
