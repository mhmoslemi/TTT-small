def soft_grad(c, h, N, T):
    center = N - 1
    max_c = float(np.max(c))
    ex = np.exp(T * (c - max_c))
    sw = ex / ex.sum()
    # Only keep significant weights
    thresh = 1e-8
    active = sw > thresh
    if not np.any(active):
        return np.zeros(N)
    grad = np.zeros(N)
    for k in np.where(active)[0]:
        L = k - center
        w = sw[k]
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            n = N - L
            grad[:n] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n]
        else:
            M = -L
            n = N - M
            grad[:n] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:n]
    return grad / sw[active].sum()

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_grad(c, h, N, T):
    center = N - 1
    max_c = float(np.max(c))
    ex = np.exp(T * (c - max_c))
    sw = ex / ex.sum()
    thresh = 1e-10
    active = sw > thresh
    n_act = int(np.sum(active))
    if n_act == 0:
        return np.zeros(N)
    sw_sum = float(sw[active].sum())
    grad = np.zeros(N)
    ks = np.where(active)[0]
    for k in ks:
        L = k - center
        w = sw[k]
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            n = N - L
            grad[:n] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n]
        else:
            M = -L
            n = N - M
            grad[:n] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:n]
    return grad / sw_sum

            base = c - d_p
            qL = zeros[:, None] + Ls[None, :]
            qM = zeros[:, None] - Ls[None, :]
            vqL = (qL >= 0) & (qL < N)

            hqL = np.where(vqL, h[np.clip(qL, 0, N - 1)], 0.0)
            hqM = np.where(vqM, h[np.clip(qM, 0, N - 1)], 0.0)
            d_q = (1.0 - hqL) * vqL.astype(float) - hqM * vqM.astype(float)
            new_peaks = (base[None, :] + d_q).max(axis=1)
            mi = int(np.argmin(new_peaks))

                best_delta = float(new_peaks[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is None:
            break
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h
