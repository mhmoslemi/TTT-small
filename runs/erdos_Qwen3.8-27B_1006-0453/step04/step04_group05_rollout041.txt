This gives the BEST swap for reducing lag L in O(N) time!

For the max-norm problem, I want to minimize max_k c[k]. A good heuristic is to focus on the top-K lags and find swaps that reduce them.

Here's my refined algorithm:

Phase 1: Binary steepest-like descent
- Repeat:
  - Find top-K lags (K=5-10)
  - For each top lag, find the best swap (i,j) that reduces it
  - Among the K candidate swaps, pick the one that most reduces the overall max
  - Apply it
  - Stop if no improvement

Phase 2: Simulated annealing
- Use the result of Phase 1
- Random swaps with acceptance criterion
- Cooling schedule

Phase 3: Continuous refinement
- Adam with vectorized gradient

Phase 4: Multi-N search
- Try N±20, N±40

One more thought: for the continuous optimization, the gradient computation for top-K lags:

For lag L, ∂c(L)/∂h[p] = (1-h[p-L])*[valid] - h[p+L]*[valid]

This is the same as the "term1" and "term2" for position p at lag L.

For K lags, the gradient is:
grad[p] = (1/K) * Σ_{k in top_K} [(1-h[p-L_k])*[valid_p1(k)] - h[p+L_k]*[valid_p2(k)]]

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, N):
    return float(np.max(compute_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    h_b = (h > 0.5).astype(float)
    nh = N // 2
    n_ones = int(h_b.sum())
    if n_ones > nh:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        for t in range(n_ones - nh):
            h_b[idx[order[t]]] = 0.0
    elif n_ones < nh:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        for t in range(nh - n_ones):
            h_b[idx[order[t]]] = 1.0
    return h_b

def swap_delta(h, i, j, N):
    center = N - 1
    L = 2 * N - 1
    delta = np.zeros(L)
    ks1 = np.arange(i, i + N)
    delta[ks1] -= (1.0 - h[i + center - ks1])
    ks2 = np.arange(N - 1 - i, 2 * N - 1 - i)
    delta[ks2] += h[i + ks2 - center]
    ks3 = np.arange(j, j + N)
    delta[ks3] += (1.0 - h[j + center - ks3])
    ks4 = np.arange(N - 1 - j, 2 * N - 1 - j)
    delta[ks4] -= h[j + ks4 - center]
    return delta

def best_swap_for_lag(h, L_lag, N, ones, zeros):
    center = N - 1
    k = L_lag + center
    s1i = (max(0, L_lag) <= i <= min(N-1, N-1+L_lag)) for i in [0]
    f_vals = np.zeros(N)
    g_vals = np.zeros(N)
    # f(i) for h[i]=1: contribution to reducing c(L) from i:1->0
    for i in ones:
        t1 = (max(0, L_lag) <= i <= min(N-1, N-1+L_lag)) and (0 <= i - L_lag < N)
        t2 = (max(0, L_lag) <= i + L_lag <= min(N-1, N-1+L_lag)) and (0 <= i + L_lag < N)
        val = 0.0
        if t1:
            val -= (1.0 - h[i - L_lag])
        if t2:
            val += h[i + L_lag]
        f_vals[i] = val
    for j in zeros:
        t1 = (max(0, L_lag) <= j <= min(N-1, N-1+L_lag)) and (0 <= j - L_lag < N)
        t2 = (max(0, L_lag) <= j + L_lag <= min(N-1, N-1+L_lag)) and (0 <= j + L_lag < N)
        val = 0.0
        if t1:
            val += (1.0 - h[j - L_lag])
        if t2:
            val -= h[j + L_lag]
        g_vals[j] = val
    fi = f_vals[ones]
    gj = g_vals[zeros]
    mi = np.argmin(fi)
    mj = np.argmin(gj)
    return int(ones[mi]), int(zeros[mj]), float(fi[mi] + gj[mj])

def steepest_binary(h, N, deadline, max_rounds=5000):
    h = h.astype(float).copy()
    nh = N // 2
    corr = compute_corr(h)
    best_peak = float(corr.max())
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    center = N - 1
    topk = 5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        top_lags = np.argpartition(corr, -topk)[-topk:]
        top_lags = top_lags[np.argsort(corr[top_lags])[::-1]]
        best_new_peak = best_peak
        best_pair = None
        for kl in top_lags[:3]:
            L_lag = int(kl) - center
            i, j, single_del = best_swap_for_lag(h, L_lag, N, ones, zeros)
            delta = swap_delta(h, i, j, N)
            new_peak = float((corr + delta).max())
            if new_peak < best_new_peak:
                best_new_peak = new_peak
                best_pair = (i, j, delta)
        if best_pair is None:
            break
        i, j, delta = best_pair
        h[i], h[j] = 0.0, 1.0
        corr = corr + delta
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if best_new_peak < best_peak:
            best_peak = best_new_peak
            best_h = h.copy()
    return best_h, best_peak * (2.0 / N)

def anneal_binary(h, N, n_iters, rng, deadline, T0=2.0, T1=0.002):
    h = h.astype(float).copy()
    corr = compute_corr(h)
    cur_peak = float(corr.max())
    best_peak = cur_peak
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        pi = ones[rng.randint(len(ones))]
        pj = zeros[rng.randint(len(zeros))]
        if pi == pj:
            continue
        delta = swap_delta(h, pi, pj, N)
        new_peak = float((corr + delta).max())
        d = new_peak - cur_peak
        if d < 0 or (T > 0.01 and rng.random() < np.exp(-d / T)):
            h[pi], h[pj] = 0.0, 1.0
            corr = corr + delta
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_h = h.copy()
        else:
            pass
    return best_h, best_peak * (2.0 / N)

def grad_topk_lags(h, N, K=30):
    corr = compute_corr(h)
    top_idx = np.argpartition(corr, -K)[-K:]
    center = N - 1
    grad = np.zeros(N)
    for ki in top_idx:
        L_lag = int(ki) - center
        for p in range(N):
            t1 = (max(0, L_lag) <= p <= min(N-1, N-1+L_lag)) and (0 <= p - L_lag < N)
            t2 = (max(0, L_lag) <= p + L_lag <= min(N-1, N-1+L_lag)) and (0 <= p + L_lag < N)
            g = 0.0
            if t1:
                g += (1.0 - h[p - L_lag])
            if t2:
                g -= h[p + L_lag]
            grad[p] += g
    return grad / len(top_idx), corr

def grad_topk_fast(h, N, K=30):
    corr = compute_corr(h)
    top_idx = np.argpartition(corr, -K)[-K:]
    center = N - 1
    grad = np.zeros(N)
    for ki in top_idx:
        L_lag = int(ki) - center
        if L_lag == 0:
            grad += (1.0 - 2.0 * h)
        elif L_lag > 0:
            grad[L_lag:] += (1.0 - h[:N - L_lag])
            grad[:N - L_lag] -= h[L_lag:]
        else:
            s = -L_lag
            grad[:N - s] += (1.0 - h[s:])
            grad[s:] -= h[:N - s]
    return grad / len(top_idx), corr

def adam_continuous(h, N, deadline, K=40, lr0=0.03, decay=0.995):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_peak = float(np.max(compute_corr(h)))
    dx = 2.0 / N
    m = np.zeros(N)
    v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    it = 0
    lr = lr0
    while time.time() < deadline:
        grad, corr = grad_topk_fast(h, N, K)
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        it += 1
        mh = m / (1.0 - b1 ** it)
        vh = v / (1.0 - b2 ** it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        if it % 300 == 0:
            lr *= decay
    return best_h, best_peak * dx

def periodic_init(N, period, rng):
    h = np.zeros(N)
    block = (rng.random(period) > 0.5).astype(float)
    n_ones_block = int(block.sum())
    if n_ones_block == 0:
        block[rng.randint(period)] = 1.0
        n_ones_block = 1
    n_full = N // period
    rem = N % period
    h[:n_full * period] = np.tile(block, n_full)
    if rem > 0:
        h[n_full * period:] = block[:rem]
    return make_binary(h, N)
