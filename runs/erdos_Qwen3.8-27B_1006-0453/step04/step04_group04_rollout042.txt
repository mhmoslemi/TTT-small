import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def soft_grad(h, T):

    dx = 2.0 / n

    m = corr.max()
    shifted = np.clip(T * (corr - m), -500, 500)

    p = exps / exps.sum()

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_AS = {'lags': None, 'n': None}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def sum_eq(x):
    return x.sum() - len(x) / 2.0

def corr_at_lag(h, lag):
    n = len(h)
    if lag >= 0:
        r = n - lag
        return float(np.dot(h[:r], 1.0 - h[lag:])) if r > 0 else 0.0
    nl = -lag
    r = n - nl
    return float(np.dot(h[nl:], 1.0 - h[:r])) if r > 0 else 0.0

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def soft_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            r = n - lag
            if r > 0:
                grad[:r] += pk * (1.0 - h[lag:])
                grad[lag:] -= pk * h[:r]
        else:
            nl = -lag
            r = n - nl
            if r > 0:
                grad[nl:] += pk * (1.0 - h[:r])
                grad[:r] -= pk * h[nl:]
    return grad * dx

def slsqp_smooth(h, T, n, maxiter=500, ftol=1e-14):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    res = minimize(soft_obj, h, args=(T,), method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(res.x, n)

def as_obj(x):
    return x[-1]

def as_sum_con(x):
    return x[:-1].sum() - _AS['n'] / 2.0

def as_ineq_con(x):
    h = x[:-1]
    t = x[-1]
    n = _AS['n']
    out = np.zeros(len(_AS['lags']))
    for i, k in enumerate(_AS['lags']):
        out[i] = t - corr_at_lag(h, k - (n - 1))
    return out

def active_set_opt(h, n, top_m=20, maxiter=3000, ftol=1e-16):

    top_idx = np.argsort(corr)[-top_m:]

    _AS['n'] = n

    cons = [{'type': 'eq', 'fun': as_sum_con}, {'type': 'ineq', 'fun': as_ineq_con}]
    res = minimize(as_obj, x0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_h(res.x[:-1], n)
