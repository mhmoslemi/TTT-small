from scipy.optimize import minimize, minimize_scalar
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    et = np.exp(T * (corr - m))
    s = et.sum()
    p = et / s
    val = (m + np.log(s) / T) * dx
    grad = np.zeros(n)
    center = n - 1
    for k_idx in range(2 * n - 1):
        w = p[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - center
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] -= w * h[k:]
            grad[k:] += w * (1.0 - h[:n-k])
        else:
            mm = -k
            grad[:n-mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n-mm]
    return val, grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        # need to reduce: scale down the non-zero entries or subtract from non-bound entries
        above = h > 0
        below = h < 1
        free = above & below
        diff = target - s
        if diff < 0:
            # reduce
            room = h[free]
            tot = room.sum()
            if tot > 1e-14:
                h[free] -= room * (-diff / tot)
            else:
                h = h * (target / s)
        else:
            room = 1.0 - h[free]
            tot = room.sum()
            if tot > 1e-14:
                h[free] += room * (diff / tot)
            else:
                h = h * (target / s)

        below = h < 1

For lag k (where k can be negative, 0, or positive):
corr[k] = sum of h[j+k] * (1-h[j]) over valid j

When h[i] changes by δ (from 0 to δ since we set it to 0 first):
- If i appears as h[j+k] term: j = i-k, and the term changes by δ * (1-h[i-k])
- If i appears as h[j] term: j = i, and the term changes by h[i+k] * (-δ) = -δ * h[i+k]

So the change in corr[k] = δ * [(1-h[i-k]) * valid1 - h[i*k] * valid2]

where valid1 depends on whether j=i-k is a valid index for lag k, and valid2 similarly.

For k > 0: valid j range is [0, n-1-k]
- valid1: 0 <= i-k <= n-1-k → k <= i <= n-1
- valid2: 0 <= i <= n-1-k → 0 <= i <= n-1-k

For k < 0 (let m = -k): valid j range is [m, n-1]
- valid1: m <= i-m <= n-1 → 0 <= i <= n-1-m
  Wait, j = i-k = i+m. Need m <= i+m <= n-1, i.e., 0 <= i <= n-1-m.
- valid2: m <= i <= n-1

For k = 0: valid j range is [0, n-1]
- valid1: 0 <= i <= n-1 (always true)
- valid2: 0 <= i <= n-1 (always true)
- But the k=0 term is special: h[i] * (1-h[i]), so the change is δ(1-δ) not δ*1

So the slope (change per unit δ) for the k=0 case is 1-δ, which varies with δ. That's why it's quadratic.

For k ≠ 0, the change is linear in δ.

So: corr_new[k] = corr_old[k] + δ * s[k] for k ≠ 0
corr_new[0] = corr_old[0] + δ - δ² (where corr_old was computed with h[i]=0)

And the optimal δ minimizes max_k corr_new[k] * dx.

                return best_h, best_c5

                    if 0 <= i <= n - 1 - k:

                        s[ki] += (1.0 - best_h[i + m])
                    if m <= i <= n - 1:
                        s[ki] -= best_h[i - m]

                cb_new = corr + delta * s
                cb_new[center] += (x - x*x) - (old - old*old)
                return np.max(cb_new) * dx

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    et = np.exp(T * (corr - m))
    s = et.sum()
    p = et / s
    val = (m + np.log(max(s, 1e-300)) / T) * dx
    grad = np.zeros(n)
    center = n - 1
    for ki in range(2 * n - 1):
        w = p[ki]
        if w < 1e-14:
            continue
        k = ki - center
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] -= w * h[k:]
            grad[k:] += w * (1.0 - h[:n - k])
        else:
            mm = -k
            grad[:n - mm] += w * (1.0 - h[mm:])
            grad[mm:] -= w * h[:n - mm]
    return val, grad * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        free = h < 1.0 - 1e-10
        room = 1.0 - h[free]
        tot = room.sum()
        if tot > 1e-14:
            h[free] += room * (diff / tot)
    else:
        free = h > 1e-10
        room = h[free]
        tot = room.sum()
        if tot > 1e-14:
            h[free] -= room * (-diff / tot)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h_init, T, n, maxiter=500):
    def obj(x):
        v, _ = smooth_obj_grad(x, T)
        return v
    def objg(x):
        _, g = smooth_obj_grad(x, T)
        return g
    def cf(x):
        return x.sum() - n / 2.0
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, jac=objg, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h_init.copy()

def compute_slope(h, n, i):
    center = n - 1
    s = np.zeros(2 * n - 1)
    for k in range(1, n):
        if k <= i:
            s[center + k] += 1.0 - h[i - k]
            s[center - k] -= h[i - k]
        if i + k < n:
            s[center + k] -= h[i + k]
            s[center - k] += 1.0 - h[i + k]
    return s

def coord_sweep(h, n, t_stop, max_sweeps=200):
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = corr.max() * dx
    center = n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_stop:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_stop:
                return best_h, compute_c5(best_h)
            old = best_h[i]
            s = compute_slope(best_h, n, i)
            def f1d(x):
                d = x - old
                cb = corr + d * s
                cb[center] = corr[center] + x - x * x - old + old * old
                return np.max(cb) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            x_new = res.x
            if abs(x_new - old) > 1e-12:
                d = x_new - old
                corr += d * s
                corr[center] += d + (old * old - x_new * x_new)
                best_h[i] = x_new
                new_c5 = corr.max() * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                improved = True
        if improved:
            best_h = project(best_h, n)
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_c5 = corr.max() * dx
        else:
            break
    return best_h, best_c5

    # Phase 2: Extended coordinate descent
    best_h, best_c5 = coord_sweep(best_h, n, t0 + budget_s * 0.55, max_sweeps=400)
