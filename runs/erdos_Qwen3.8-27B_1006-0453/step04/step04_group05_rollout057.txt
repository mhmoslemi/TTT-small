        #   At q: was 1, now 0. For lag L>0: corr[L] -= (1-h[q-L]) - h[q+L] if applicable

    w = np.exp(beta * (ct - ct.max()))

            g += wi * (1.0 - 2.0 * h)
        else:
            g[L:] += wi * (1.0 - h[:N - L])
            g[:N - L] -= wi * h[L:]
    return g

    h_b2 = make_binary(best_h, N0)
    d3a = t0 + 250
    h_b2, c5_b2 = bin_descent(h_b2, N0, d3a, rng)
    if c5_b2 < best_c5:
        best_c5, best_h = c5_b2, h_b2.copy()
    h_a2 = best_h.astype(float) + rng.randn(N0) * 0.04
    h_a2 = project_sum(h_a2, N0 / 2.0)
    h_ad2, c5_ad2 = adam_opt(h_a2, N0, N0 / 2.0, d3, k=50, beta0=5.0, beta1=22.0, lr0=0.02)
    if c5_ad2 < best_c5:
        best_c5, best_h = c5_ad2, h_ad2.copy()

    # Phase 4: Long Adam with slow decay (150s)
    d4 = t0 + 450
    h_a3 = best_h.astype(float) + rng.randn(N0) * 0.03
    h_a3 = project_sum(h_a3, N0 / 2.0)
    h_ad3, c5_ad3 = adam_opt(h_a3, N0, N0 / 2.0, d4, k=60, beta0=8.0, beta1=25.0, lr0=0.012)
    if c5_ad3 < best_c5:
        best_c5, best_h = c5_ad3, h_ad3.copy()

    # Phase 5: Binary refinement + Adam (100s)
    d5 = t0 + 550
    h_b3 = make_binary(best_h, N0)
    d5a = t0 + 500
    h_b3, c5_b3 = bin_descent(h_b3, N0, d5a, rng, max_fails=50)
    if c5_b3 < best_c5:
        best_c5, best_h = c5_b3, h_b3.copy()
    h_a4 = best_h.astype(float) + rng.randn(N0) * 0.02
    h_a4 = project_sum(h_a4, N0 / 2.0)
    h_ad4, c5_ad4 = adam_opt(h_a4, N0, N0 / 2.0, d5, k=60, beta0=10.0, beta1=30.0, lr0=0.008)
    if c5_ad4 < best_c5:
        best_c5, best_h = c5_ad4, h_ad4.copy()

    # Phase 6: Try slightly different N values
    for N_try in [N0 + 2, N0 + 6, N0 - 4]:
        if time.time() > deadline - 120 or N_try < 200:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_it_b = make_binary(h_interp, N_try)
        d_bt = min(deadline - 60, time.time() + 50)
        h_it_b, c5_bt = bin_descent(h_it_b, N_try, d_bt, rng, max_fails=20)
        if c5_bt < best_c5:
            best_c5, best_h = c5_bt, h_it_b.copy()
        if time.time() < deadline - 40:
            h_it_a = h_it_b.astype(float) + rng.randn(N_try) * 0.04
            h_it_a = project_sum(h_it_a, N_try / 2.0)
            h_it_ad, c5_bt2 = adam_opt(h_it_a, N_try, N_try / 2.0, deadline - 40,
                                       k=40, beta0=5.0, beta1=20.0, lr0=0.025)
            if c5_bt2 < best_c5:
                best_c5, best_h = c5_bt2, h_it_ad.copy()

    # Phase 7: Alternative structured starts
    if time.time() < deadline - 60:
        for pat in range(2):
            if time.time() > deadline - 60:
                break
            if pat == 0:
                h_alt = np.zeros(N0)
                h_alt[::2] = 1.0
            else:
                h_alt = (rng.random(N0) > 0.5).astype(float)
            h_alt = make_binary(h_alt, N0)
            d_alt = min(deadline - 30, time.time() + 40)
            h_alt, c5_alt = bin_descent(h_alt, N0, d_alt, rng, max_fails=15)
            if c5_alt < best_c5:
                best_c5, best_h = c5_alt, h_alt.copy()

    # Phase 8: Final precision Adam
    if time.time() < deadline - 3:
        hF = best_h.astype(float) + rng.randn(len(best_h)) * 0.015
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = adam_opt(hF, len(best_h), len(best_h) / 2.0, deadline - 1,
                           k=80, beta0=12.0, beta1=35.0, lr0=0.005)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
