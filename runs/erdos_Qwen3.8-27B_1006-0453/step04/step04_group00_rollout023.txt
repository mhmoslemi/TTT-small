def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:

            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=800):

    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 30})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_1d(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def leveling(h, n, max_iter=50):
    """Try to equalize top correlations by adjusting boundary points."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_iter):
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top = np.argsort(corr)[-4:]
        bottom = np.argsort(corr)[:4]
        top_val = np.mean(corr[top])
        bot_val = np.mean(corr[bottom])
        if top_val - bot_val < 1e-6:
            break
        # Slightly reduce top, increase spread at bottom lags
        # Find i that most affects top lags
        for ki in top:
            lag = ki - (n-1)
            # Adjust h at random position that affects this lag
            for _ in range(3):
                i = np.random.randint(n)
                if 0 <= i+lag < n or 0 <= i-lag < n:
                    h_try = best_h.copy()
                    h_try[i] = best_h[i] + 0.01 * np.random.choice([-1, 1])
                    h_try = project_h(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        best_h = h_try
                        best_c5 = c5_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Try alternative initializations
    alts = []
    # Thue-Morse
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    alts.append(project_h(tm, n))
    # Random binary
    for _ in range(4):
        rb = np.zeros(n)
        rb[np.random.choice(n, n//2, replace=False)] = 1.0
        alts.append(rb)
    # Alternating blocks
    ab = np.zeros(n)
    block = 4
    val = 1.0
    pos = 0
    while pos < n:
        ab[pos:min(pos+block, n)] = val
        pos += block
        val = 1.0 - val
        block = max(2, block - 1) if block > 2 else 4
    alts.append(project_h(ab, n))

    for alt in alts:
        if elapsed() > 20:
            break
        hc = slsqp_smooth(alt, 5000, n, maxiter=150)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 1: SLSQP annealing
    T_list = [30000, 10000, 4000, 1500, 600, 250, 100, 40, 15, 5]
    for T in T_list:
        if elapsed() > budget_s * 0.22:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Subgradient + SLSQP
    for it in range(25):
        if elapsed() > budget_s * 0.42:
            break
        cool = 1.0 - it / 25.0
        alpha = 0.004 + 0.008 * cool
        k_top = 4 + int(6 * cool)
        hs = subgrad_topk(best_h, n, k_top, alpha)
        T = 100 + 1500 * cool
        hc = slsqp_smooth(hs, T, n, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(45):
        if elapsed() > budget_s * 0.58:
            break
        cool = max(0.1, 1.0 - it / 45.0)
        amp = 0.02 + 0.08 * cool
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        T_bh = 400 + 2500 * cool
        hc = slsqp_smooth(h_pert, T_bh, n, maxiter=350)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.78, max_sweeps=120)

    # Phase 5: Leveling
    best_h, best_c5 = leveling(best_h, n, max_iter=30)

    # Phase 6: Fine basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(25):
        if elapsed() > budget_s * 0.87:
            break
        cool = 1.0 - 0.9 * it / 25.0
        amp = 0.015 + 0.03 * cool
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        T_bh = 800 + 2000 * cool
        hc = slsqp_smooth(h_pert, T_bh, n, maxiter=300)
        c5v = compute_c5(hc)
        temp = 0.0004 * cool + 0.00003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 7: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, max_sweeps=80)

    # Phase 8: Ultra-high T SLSQP
    for T in [500000, 2000000]:
        if elapsed() > budget_s * 0.98:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final coordinate descent
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=30)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
