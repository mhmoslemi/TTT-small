For a binary h with n/2 ones: at lag ℓ, the correlation counts positions where h[i]=1 and h[i+ℓ]=0 (or the reverse, depending on sign). The total number of valid positions is n-|ℓ|. If the ones are randomly placed, the expected count is (n/2) * (n/2) / (n-|ℓ|) ≈ n/4 for small |ℓ|.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_sweep(h, n, t_end, max_sweeps=2000):
    h = h.copy()
    dx = 2.0 / n
    k0 = n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_c5 = compute_c5(h)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            h[i] = 0.0
            corr0 = compute_corr(h)
            h[i] = 1.0
            corr1 = compute_corr(h)
            slope = corr1 - corr0
            cv0 = corr0.copy()
            cv0[k0] = 0.0
            def f1d(t):
                v = cv0 + t * slope
                v[k0] += t * (1.0 - t)
                return float(np.max(v))
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = f1d(c)
            fd = f1d(d)
            for _ in range(35):
                if fc < fd:
                    b = d; d = c; fd = fc
                    c = b - gr * (b - a); fc = f1d(c)
                else:
                    a = c; c = d; fc = fd
                    d = a + gr * (b - a); fd = f1d(d)
            t_opt = (a + b) / 2.0
            h[i] = t_opt
        h = proj(h, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            improved = True
        if not improved:
            break
    return h, best_c5

def subgrad_step(h, n, T, lr):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -400, 400)
    exps = np.exp(shifted)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n - lag])
        else:
            nl = -lag
            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += pk * (-h[nl:])
    return grad * dx

def momentum_descent(h, n, t_end, lr_init=0.005, T0=200, T_end=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    momentum = 0.85
    lr = lr_init
    T = T0
    for it in range(5000):
        if time.time() > t_end:
            break
        T = min(T * 1.05, T_end)
        g = subgrad_step(best_h, n, T, lr)
        v = momentum * v - lr * g
        trial = proj(best_h + v, n)
        c5_t = compute_c5(trial)
        if c5_t < best_c5:
            best_h = trial
            best_c5 = c5_t
        else:
            lr *= 0.9
            v *= 0.5
            if lr < 1e-9:
                break
    return best_h, best_c5

def gen_binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return proj(h, n)

def gen_sine_start(n, freq):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.45 * np.sin(np.pi * freq * x)
    return proj(np.clip(h, 0, 1), n)

def gen_thue_start(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = t.astype(float)
    ones = h.sum()
    target = n // 2
    if ones < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:int(target - ones)]] = 1.0
    elif ones > target:
        ones_idx = np.where(h == 1)[0]
        rng.shuffle(ones_idx)
        h[ones_idx[:int(ones - target)]] = 0.0
    return proj(h, n)

def optimize_from(h, n, t_end):
    h, c5 = coord_sweep(h, n, min(t_end, time.time() + 8), max_sweeps=200)
    if time.time() < t_end - 3:
        h, c5 = momentum_descent(h, n, time.time() + 3)
    h, c5 = coord_sweep(h, n, t_end, max_sweeps=500)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s * 0.97

    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(180)
        best_h[:90] = 1.0
    n0 = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize initial at its resolution
    best_h, best_c5 = optimize_from(best_h, n0, t0 + budget_s * 0.40)

    # Phase 2: Try n=160
    if time.time() < t_end - budget_s * 0.3:
        for n_try in [160, 200, 150]:
            t_s = time.time()
            t_e = min(t_end - budget_s * 0.15, t_s + budget_s * 0.12)
            if t_s >= t_e:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_int = np.interp(x_new, x_old, best_h)
            h_int = proj(h_int, n_try)
            hc, c5v = optimize_from(h_int, n_try, t_e)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 3: Diversified restarts at best resolution
    n_best = len(best_h)
    t_restart_end = min(t_end - budget_s * 0.1, t0 + budget_s * 0.80)
    n_restarts = 6
    per_restart = (t_restart_end - time.time()) / max(n_restarts, 1)
    for r in range(n_restarts):
        if time.time() > t_restart_end:
            break
        t_r_end = time.time() + max(per_restart, 2.0)
        # Generate perturbed start
        if r % 3 == 0:
            h_start = best_h + 0.15 * rng.randn(n_best)
        elif r % 3 == 1:
            h_start = gen_binary_start(n_best, rng)
        else:
            h_start = gen_thue_start(n_best, rng)
        h_start = proj(h_start, n_best)
        hc, c5v = optimize_from(h_start, n_best, min(t_r_end, t_restart_end))
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Final polish
    if time.time() < t_end - 1:
        best_h, best_c5 = coord_sweep(best_h, len(best_h), t_end, max_sweeps=1000)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
