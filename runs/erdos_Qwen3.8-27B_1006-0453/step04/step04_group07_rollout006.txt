from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def topK_gradient(h, K):
    """Compute gradient of average of top-K correlations."""
    n = len(h)
    corr = get_corr(h)
    # Top K indices in the full correlation
    top_idx = np.argsort(corr)[-K:]
    # Convert to shift: shift = idx - (n-1)
    shifts = top_idx - (n - 1)
    grad = np.zeros(n)
    for s in shifts:
        if s > 0:
            # (1-h[j-s]) for j=s..n-1 → positions s..n-1
            grad[s:] += (1.0 - h[:n-s])
            # -h[j+s] for j=0..n-1-s → positions 0..n-1-s
            grad[:n-s] -= h[s:]
        elif s < 0:
            t = -s
            # (1-h[j+t]) for j=0..n-1-t → positions 0..n-1-t
            grad[:n-t] += (1.0 - h[t:])
            # -h[j-t] for j=t..n-1 → positions t..n-1
            grad[t:] -= h[:n-t]
        else:
            # s=0: (1-h[j]) - h[j] = 1-2h[j]
            grad += (1.0 - 2.0 * h)
    grad /= K
    return grad

def subgradient_descent(h, n, t_end, K=50, steps=5000):
    """Project subgradient descent on top-K correlations."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    lr = 0.01  # initial learning rate
    for step in range(steps):
        if time.time() > t_end:
            break
        grad = topK_gradient(best_h, K)
        # Adaptive step size
        lr = 0.01 * (1.0 + step) ** (-0.5)
        h_new = project(best_h - lr * grad * dx, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        else:

            lr *= 0.5
            h_new = project(h_new, n)  # just recalculate
            best_h = project(best_h - lr * grad * dx, n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, T, n):
    dx = 2.0 / n
    corr = get_corr(x)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return float(val * dx)

def coord_descent(h, n, t_end, max_sweeps=5000, tol=1e-14):

    best_h = h.copy()

    best_c5 = float(np.max(corr)) * dx
    c0 = n - 1

            break

            old = best_h[i]
            idx1 = i - ks
            idx2 = i + ks
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(2 * n - 1)
            coeff[v1] += 1.0 - best_h[idx1[v1]]
            coeff[v2] -= best_h[idx2[v2]]
            c0b = corr[c0]
            def ofn(v):
                cn = corr + (v - old) * coeff
                cn[c0] = c0b + (v - old) * (1.0 - v - old)
                return float(np.max(cn))
            f0 = ofn(0.0)
            f1 = ofn(1.0)

            fo = ofn(old)

            res = minimize_scalar(ofn, bounds=(0.0, 1.0), method='bounded',

                best_c5 = best_f * dx
                corr = get_corr(best_h)
                improved = True
        if not improved:
            break
    return best_h, best_c5

    w = np.exp(T * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    shifts = np.arange(-(n - 1), n)
    for idx_s in range(2 * n - 1):
        ws = w[idx_s]
        if ws < 1e-12:
            continue
        s = shifts[idx_s]
        if s > 0:
            grad[s:] += ws * (1.0 - h[:n - s])
            grad[:n - s] -= ws * h[s:]
        elif s < 0:
            t = -s
            grad[:n - t] += ws * (1.0 - h[t:])
            grad[t:] -= ws * h[:n - t]
        else:
            grad += ws * (1.0 - 2.0 * h)
    return grad
