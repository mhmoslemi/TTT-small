def compute_c5(h, N):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, N):
    nh = N // 2
    h_b = np.zeros(N)
    idx = np.argsort(h)[::-1]
    h_b[idx[:nh]] = 1.0
    return h_b

def swap_delta(h, p, q, N):
    """Compute change in correlation when h[p] 1->0 and h[q] 0->1"""
    m = np.arange(2 * N - 1)
    k = m - N + 1
    delta = np.zeros(2 * N - 1)
    kp = k + p
    vp1 = (kp >= 0) & (kp < N)
    km_p = p - k
    vp2 = (km_p >= 0) & (km_p < N)
    delta[vp1] -= (1.0 - h[kp[vp1]])
    delta[vp2] += h[km_p[vp2]]
    kq = k + q
    vq1 = (kq >= 0) & (kq < N)
    km_q = q - k
    vq2 = (km_q >= 0) & (km_q < N)
    delta[vq1] += (1.0 - h[kq[vq1]])
    delta[vq2] -= h[km_q[vq2]]
    return delta

            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        else:

                break
    
    return best_h, best_peak

def continuous_adam(h, N, deadline, tau0=50.0, tau1=200.0, lr0=0.03, topk=20, rng=None):
    """Adam optimization with soft-max/smooth max objective"""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_peak = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(N)
    v = np.zeros(N)
    it = 0
    lr = lr0
    
    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak = float(corr.max())
        if peak < best_peak:
            best_peak = peak
            best_h = h.copy()
        
        it += 1
        # Adaptive temperature
        tau = tau0 + (tau1 - tau0) * min(1.0, it / 5000)
        
        # Compute gradient using top-k lags (approximate soft-max)
        idx_top = np.argpartition(corr, -topk)[-topk:]
        # Use soft-max weight within top-k
        c_top = corr[idx_top]
        w = np.exp(tau * (c_top - c_top.max()))
        w /= w.sum()
        
        grad = np.ones(N)  # The "1" part of the gradient
        center = N - 1
        for ii in range(topk):
            m_idx = idx_top[ii]
            wm = w[ii]
            if wm < 1e-10:
                continue
            k = m_idx - center
            if k == 0:
                grad -= 2.0 * wm * h
            elif k > 0:
                grad[:N-k] -= wm * h[k:]
                grad[k:] -= wm * h[:N-k]
            else:
                M = -k
                grad[:N-M] -= wm * h[M:]
                grad[M:] -= wm * h[:N-M]
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**it)
        vh = v / (1 - b2**it)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_sum(h, target)
        
        if it % 500 == 0:
            lr *= 0.95
    
    return best_h, best_peak * (2.0 / N)
