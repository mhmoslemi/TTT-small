            xn = coord_opt_fast(best_h, n, idx)
            h_try = best_h.copy()
            h_try[idx] = xn
            h_try = project(h_try, n)
            c5t = compute_c5(h_try)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=600):
    def obj(x):
        return smooth_c5_grad(x, T)[0]
    def grad(x):
        return smooth_c5_grad(x, T)[1]
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def gradient_step(h, n, T, lr):
    _, g = smooth_c5_grad(h, T)
    h_new = h - lr * g
    return project(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        xx = np.linspace(0, 2, n0, endpoint=False)
        best_h = np.clip(0.5 + 0.2*np.sin(np.pi*xx) + 0.08*np.sin(2*np.pi*xx), 0, 1)
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s*0.20, max_sweeps=300)

    # Phase 2: SLSQP annealing
    for T in [20, 50, 150, 400, 1200, 4000, 15000]:
        if time.time() > t0 + budget_s*0.45:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        # brief CD after each SLSQP
        best_h, best_c5 = coord_descent(best_h, n, min(time.time()+2, t0+budget_s*0.45), max_sweeps=30)

    # Phase 3: gradient descent + CD cycles with perturbation
    T_g = 200
    lr = 0.02
    for it in range(40):
        if time.time() > t0 + budget_s*0.65:
            break
        h_cur = best_h.copy()
        for _ in range(5):
            h_cur = gradient_step(h_cur, n, T_g, lr)
        h_cur, c5_cur = coord_descent(h_cur, n, min(time.time()+3, t0+budget_s*0.65), max_sweeps=40)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur
            lr *= 1.1
        else:
            lr *= 0.9
        # random restart perturbation every 8 iterations
        if it % 8 == 4:
            h_p = best_h + 0.04 * np.random.randn(n)
            h_p = project(h_p, n)
            h_p, c5_p = coord_descent(h_p, n, min(time.time()+4, t0+budget_s*0.65), max_sweeps=50)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p

    # Phase 4: high-T SLSQP polish
    for T in [30000, 100000, 300000]:
        if time.time() > t0 + budget_s*0.85:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: final CD polish
    best_h, best_c5 = coord_descent(best_h, n, t_end - 5, max_sweeps=400)

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
