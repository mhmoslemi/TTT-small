    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def smooth_max_corr(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    if T < 1e-6:
        return m * dx, None
    exps = np.exp(T * (corr - m))
    s = exps.sum()
    w = exps / s
    return (m + np.log(s) / T) * dx, w

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for i in range(n):
        gi = 0.0
        for k in range(2 * n - 1):
            l = k - n + 1
            j1 = i - l
            j2 = i + l
            if 0 <= j1 < n:
                gi += w[k] * (1.0 - h[j1])
            if 0 <= j2 < n:
                gi -= w[k] * h[j2]
        grad[i] = dx * gi
    return grad

        val, _ = smooth_max_corr(x, T)
        return val
    def grad(x):
        return smooth_grad(x, T)

        return x.sum() - n / 2.0

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    s = exps.sum()
    return (m + np.log(s) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for i in range(n):
        gi = 0.0
        for l in range(-(n - 1), n):
            wk = w[n - 1 + l]
            if wk < 1e-12:
                continue
            j1 = i - l
            j2 = i + l
            if 0 <= j1 < n:
                gi += wk * (1.0 - h[j1])
            if 0 <= j2 < n:
                gi -= wk * h[j2]
        grad[i] = dx * gi
    return grad

def slsqp_step(h, T, n, maxiter=400):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(smooth_obj, h, args=(T,), jac=lambda x: smooth_grad(x, T),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h.copy()

def coord_descent(h, n, t_end, max_sweeps, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            d = np.zeros(2 * n - 1)
            for l in range(1, n):
                v = n - 1 + l
                if i - l >= 0:
                    d[v] += (1.0 - best_h[i - l])
                if i + l < n:
                    d[v] -= best_h[i + l]
                u = n - 1 - l
                if i + l < n:
                    d[u] += (1.0 - best_h[i + l])
                if i - l >= 0:
                    d[u] -= best_h[i - l]
            ci = best_h[i]
            lo, hi = -ci, 1.0 - ci
            if lo == hi:
                continue
            def f1d(delta, _corr=corr, _d=d, _ci=ci, _dx=dx, _n=n):
                vals = _corr + delta * _d
                vals[_n - 1] = _corr[_n - 1] + delta * (1.0 - 2.0 * _ci) - delta * delta
                return float(np.max(vals) * _dx)
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-11})
            if res.x is not None and abs(res.x) > 1e-12:
                best_h[i] = ci + res.x
                best_h = np.clip(best_h, 0, 1)
                best_h = project(best_h, n)
                c5 = compute_c5(best_h)
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def perturb_reopt(h, n, t_end, rng, n_pert=30, scale=0.03):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_pert):
        if time.time() > t_end:
            break
        p = best_h + scale * (time.time() - 0)  # placeholder
        p = best_h + scale * rng.standard_normal(n)
        p = project(p, n)
        hc = slsqp_step(p, 2000, n, maxiter=200)
        hc = coord_descent(hc, n, min(time.time() + 2.0, t_end), 5, rng)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing T
    for T in [20, 100, 500, 2000, 10000, 50000]:
        if time.time() - t0 > budget_s * 0.20:
            break
        hc = slsqp_step(best_h, T, n, maxiter=500)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    # Phase 2: Coordinate descent
    t_cd = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, n, t_cd, max_sweeps=150, rng=rng)

    # Phase 3: SLSQP again with very high T
    if time.time() - t0 < budget_s * 0.55:
        for T in [50000, 200000]:
            hc = slsqp_step(best_h, T, n, maxiter=400)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5 = hc, cv

    # Phase 4: More coordinate descent
    t_cd2 = t0 + budget_s * 0.68
    best_h, best_c5 = coord_descent(best_h, n, t_cd2, max_sweeps=100, rng=rng)

    # Phase 5: Perturbation + re-optimization
    t_pert = t0 + budget_s * 0.82
    best_h, best_c5 = perturb_reopt(best_h, n, t_pert, rng, n_pert=25, scale=0.02)

    # Phase 6: Multi-resolution
    for n_try in [160, 200, 280, 320]:
        if time.time() - t0 > budget_s * 0.93:
            break
        h_interp = interpolate_h(best_h, n_try)
        for T in [500, 5000, 50000]:
            if time.time() - t0 > budget_s * 0.95:
                break
            hc = slsqp_step(h_interp, T, n_try, maxiter=300)
            hc = coord_descent(hc, n_try, min(time.time() + 3.0, t0 + budget_s * 0.96), 10, rng)
            cv = compute_c5(hc)
            if cv < best_c5:
                best_h, best_c5, n = hc, cv, n_try
                h_interp = interpolate_h(best_h, n_try)

    # Phase 7: Final polish
    t_final = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t_final, max_sweeps=30, rng=rng)
    if time.time() - t0 < budget_s * 0.99:
        hc = slsqp_step(best_h, 100000, n, maxiter=500)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_h, best_c5 = hc, cv

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
