import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_smooth_max(corr, T):
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val

def project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return np.clip(h, 0, 1)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_smooth_match(h, T, n_topk=50):
    """Gradient of softmax-smoothed max of corr values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    w /= w.sum()
    # Only use top-k for efficiency
    if n_topk < len(w):
        idx = np.argsort(w)[-n_topk:]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        w2 /= w2.sum()
        w = w2
    
    # d(corr[m])/d(h[j]) = (1-h[j+k])*[j in S] - h[j-k]*[j-k in S]
    # where k = m - n + 1, S depends on k
    # For efficiency, only compute for nonzero w
    nz = np.nonzero(w > 1e-15)[0]
    grad = np.zeros(n)
    j = np.arange(n)
    for m_idx in nz:
        k = m_idx - n + 1
        wm = w[m_idx]
        # First term: (1-h[j+k]) for j where 0<=j<n and 0<=j+k<n
        valid1 = (j + k >= 0) & (j + k < n)
        # Second term: h[j-k] for j-k where 0<=j-k<n and 0<=(j-k)+k<n => 0<=j-k<n and 0<=j<n
        valid2 = (j - k >= 0) & (j - k < n)
        grad[valid1] += wm * (1.0 - h[j[valid1] + k])
        grad[valid2] -= wm * h[j[valid2] - k]
    return grad * dx

def nesterov_step(h, lr, momentum, T, n_topk=50):
    n = len(h)
    grad = grad_smooth_match(h, T, n_topk)
    h_new = h - lr * grad
    return project_to_feasible(h_new, n)

def coord_optimize_1d(h, n, i):
    """Optimize coordinate i to minimize max of (A_m + B_m * x)."""
    n_full = 2 * n - 1
    dx = 2.0 / n
    # Compute A_m and B_m
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_range = np.arange(-(n - 1), n)
    j = i
    B = np.zeros(n_full)
    A = corr.copy()
    m_idx = np.arange(n_full)
    k = m_idx - n + 1
    valid1 = (j + k >= 0) & (j + k < n)
    valid2 = (j - k >= 0) & (j - k < n)
    B[valid1] += (1.0 - h[j + k[valid1]])
    B[valid2] -= h[j - k[valid2]]
    A = corr - h[i] * B  # corr at h[i]=0 would be A + 0*B = A
    
    # Minimize max_m (A_m + B_m * x) for x in [0,1]
    # Binary search on target t
    def max_val(x):
        return np.max(A + B * x)
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if max_val(mid) > max_val(0):
            hi = mid
        else:
            lo = mid
        # Actually this isn't quite right for the general case
    # Better: use scipy minimize_scalar on the convex 1D function
    res = minimize_scalar(lambda x: float(np.max(A + B * x)), bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coordinate_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_optimize_1d(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_to_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return np.clip(h, 0, 1)
        h = h + room*(diff/tot)
    else:
        h = h*(1.0 + diff/s) if s > 1e-15 else h
    return np.clip(h, 0.0, 1.0)

def smooth_max(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T*(corr-m))))/T

def sqp_step(h, T, n, maxiter=500, ftol=1e-14, line_search=True):
    n_full = 2*n - 1
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    k_range = np.arange(-(n-1), n)
    j = np.arange(n)
    # Build B: B[m, i] = d(corr[m])/d(h[i])
    B = np.zeros((n_full, n))
    for m in range(n_full):
        k = k_range[m]
        v1 = (j+k>=0) & (j+k<n)
        v2 = (j-k>=0) & (j-k<n)
        B[m, v1] = 1.0 - h[j[v1]+k]
        B[m, v2] -= h[j[v2]-k]
    a = corr - B @ h  # a_m = corr_m - sum_i B_{m,i} h_i, so q_m = a_m + B_m @ h
    def obj(x):
        q = a + B @ x
        mm = np.max(q)
        return (mm + np.log(np.sum(np.exp(T*(q-mm))))/T)*dx
    def obj_grad(x):
        q = a + B @ x
        mm = np.max(q)
        w = np.exp(T*(q-mm))
        w /= w.sum()
        return (B.T @ w)*dx
    def con(x):
        return x.sum() - n/2.0
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':con}],
                       options={'maxiter':maxiter,'ftol':ftol})
        h_new = project_h(res.x, n)
        # Line search
        if line_search:
            c5_old = compute_c5(h)
            c5_new = compute_c5(h_new)
            if c5_new > c5_old:
                for alpha in [0.5, 0.25, 0.1, 0.5, 0.25]:
                    h_try = project_h(h + alpha*(h_new-h), n)
                    if compute_c5(h_try) < c5_old:
                        h_new = h_try
                        break
                else:
                    h_new = h
        return h_new
    except:
        return h

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    elif s > 1e-15:
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def build_B(h, n):
    n_full = 2 * n - 1
    B = np.zeros((n_full, n))
    j = np.arange(n)
    for m in range(n_full):
        k = m - n + 1
        v1 = (j + k >= 0) & (j + k < n)
        v2 = (j - k >= 0) & (j - k < n)
        B[m, v1] = 1.0 - h[j[v1] + k]
        B[m, v2] -= h[j[v2] - k]
    return B

def sqp_step(h, T, n, maxiter=400, ftol=1e-14):
    n_full = 2 * n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    B = build_B(h, n)
    a = corr - B @ h

    def obj(x):
        q = a + B @ x
        mm = np.max(q)
        return (mm + np.log(np.sum(np.exp(T * (q - mm)))) / T) * dx

    def obj_grad(x):
        q = a + B @ x
        mm = np.max(q)
        w = np.exp(T * (q - mm))
        w /= w.sum()
        return (B.T @ w) * dx

    def con(x):
        return x.sum() - n / 2.0

    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_h(res.x, n)
        c5_old = compute_c5(h)
        c5_new = compute_c5(h_new)
        if c5_new > c5_old:
            for alpha in [0.7, 0.5, 0.3, 0.1]:
                h_try = project_h(h + alpha * (h_new - h), n)
                if compute_c5(h_try) < c5_old:
                    return h_try
            return h
        return h_new
    except Exception:
        return h

def nesterov_pg(h, T, n, n_steps=200, lr0=0.01, momentum=0.9):
    dx = 2.0 / n
    h_curr = h.copy()
    v = np.zeros(n)
    best = h_curr.copy()
    best_c5 = compute_c5(best)
    for it in range(n_steps):
        lr = lr0 / (1.0 + 0.01 * it)
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        mm = np.max(corr)
        w = np.exp(T * (corr - mm))
        w /= w.sum()
        nz = np.nonzero(w > 1e-12)[0]
        B_nz = build_B(h_curr, n)[nz]
        grad = (B_nz.T @ w[nz]) * dx
        v = momentum * v - lr * grad
        h_curr = project_h(h_curr + v, n)
        c5v = compute_c5(h_curr)
        if c5v < best_c5:
            best_c5 = c5v
            best = h_curr.copy()
    return best, best_c5

def coord_descent_fast(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved_any = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            corr_cur = np.correlate(best_h, 1.0 - best_h, mode='full')
            A = corr_cur - best_h[i] * 0  # baseline
            k_range = np.arange(-(n - 1), n)
            k_full = k_range
            B_i = np.zeros(2 * n - 1)
            v1 = (i + k_full >= 0) & (i + k_full < n)
            v2 = (i - k_full >= 0) & (i - k_full < n)
            B_i[v1] = 1.0 - best_h[i + k_full[v1]]
            B_i[v2] -= best_h[i - k_full[v2]]
            A_i = corr_cur - best_h[i] * B_i
            def f1d(x):

                                  options={'xatol': 1e-11})
