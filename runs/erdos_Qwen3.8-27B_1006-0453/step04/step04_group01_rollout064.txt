            else:

    # corr[k_max] = sum_i h[i] * (1-h[i+lag])
    # d/dh[j] = (1-h[j+lag]) - h[j-lag] (when valid)

        jp = j + lag
        jm = j - lag
        if 0 <= jp < n:
            g += (1.0 - h[jp])
        if 0 <= jm < n:
            g -= h[jm]
        grad[j] = g
    h_new = h - alpha * grad * dx
    return project(h_new, n)
