import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, T, n, mu=50.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    wsum = w.sum()
    w /= wsum
    val = (m + np.log(wsum) / T) * dx
    if mu > 0:
        val += mu * (h.sum() - n * 0.5) ** 2
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        s = i - (n - 1)
        wi = w[i]
        if wi < 1e-14:
            continue
        j_lo = max(0, s)
        j_hi = min(n - 1, s + n - 1)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wi * (1.0 - h[j_lo - s:j_hi - s + 1])
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= wi * h[j_lo2 + s:j_hi2 + s + 1]
    grad *= dx
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n * 0.5)
    return val, grad

def project(h, n):
    target = n / 2.0
    vec = np.array(h, dtype=float)
    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-10:
        return vc
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def lbfgs_opt(h0, T, n, mu=50.0, maxiter=3000):
    def obj(x):
        v, g = smooth_obj_grad(x, T, n, mu)
        return v
    def objg(x):
        return smooth_obj_grad(x, T, n, mu)
    bnds = [(0.0, 1.0)] * n
    res = minimize(objg, h0, jac=True, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    return res.x

def coord_opt(h, n, i):
    best_val = np.inf
    best_hi = h[i]
    # Golden section or use the fact that f(hi) is a 1D function
    # For speed, use minimize_scalar
    n_local = n
    def f1d(hi):
        h_try = h.copy()
        h_try[i] = hi
        corr = np.correlate(h_try, 1.0 - h_try, mode='full')
        return np.max(corr) * (2.0 / n_local)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 100})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return best_h, best_c5

        best_h = np.zeros(160)

    best_c5 = compute_c5(best_h)
    
    def elapsed():
        return time.time() - t0

    T_list = [20, 80, 300, 1000, 4000, 15000]

        except Exception:

    # Phase 3: Multi-restart with perturbation + re-optimize (30%)
    t_restart = t0 + budget_s * 0.80

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, T, n, mu=50.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    wsum = w.sum()
    w /= wsum
    val = (m + np.log(wsum) / T) * dx
    if mu > 0:
        val += mu * (h.sum() - n * 0.5) ** 2
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        s = i - (n - 1)
        wi = w[i]
        if wi < 1e-14:
            continue
        j1lo = max(0, s)
        j1hi = min(n - 1, s + n - 1)
        if j1lo <= j1hi:
            grad[j1lo:j1hi + 1] += wi * (1.0 - h[j1lo - s:j1hi - s + 1])
        j2lo = max(0, -s)
        j2hi = min(n - 1, n - 1 - s)
        if j2lo <= j2hi:
            grad[j2lo:j2hi + 1] -= wi * h[j2lo + s:j2hi + s + 1]
    grad *= dx
    if mu > 0:
        grad += 2.0 * mu * (h.sum() - n * 0.5)
    return val, grad

def project(h, n):
    target = n * 0.5
    vec = np.array(h, dtype=float)
    vc = np.clip(vec, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-10:
        return vc
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(vec + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vec + (lo + hi) * 0.5, 0.0, 1.0)

def lbfgs_opt(h0, T, n, mu=100.0, maxiter=5000):
    def objg(x):
        return smooth_obj_grad(x, T, n, mu)
    bnds = [(0.0, 1.0)] * n
    res = minimize(objg, h0, jac=True, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 20000})
    return res.x

def coord_opt(h, n, i):
    def f1d(hi):
        h_try = h.copy()
        h_try[i] = hi
        corr = np.correlate(h_try, 1.0 - h_try, mode='full')
        return np.max(corr) * (2.0 / n)
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13, 'maxiter': 80})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=500, tolerance=1e-14):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - tolerance:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

                best_h = hc
        except Exception:

            h_pert = np.convolve(h_pert, k5, mode='same')
        else:

        h_pert = project(h_pert, n)

            hc = h_pert
        hc = project(hc, n)

                pass

    # Final CD sweep

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
