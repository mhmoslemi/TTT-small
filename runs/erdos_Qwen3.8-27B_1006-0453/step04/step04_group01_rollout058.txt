from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_binary(h, target_ones):
    """Project to binary: exactly target_ones ones, rest zeros."""
    h_int = (h >= 0.5).astype(float)
    n_ones = int(h_int.sum())
    if n_ones == target_ones:
        return h_int
    # Need to flip
    if n_ones > target_ones:
        # Change some 1s to 0s
        ones_idx = np.where(h_int == 1)[0]
        # Sort by original value descending (keep the ones closest to 1)
        order = np.argsort(-h[ones_idx])
        flip_count = n_ones - target_ones
        for idx in order[:flip_count]:
            h_int[ones_idx[idx]] = 0
    else:
        zeros_idx = np.where(h_int == 0)[0]
        order = np.argsort(-h[zeros_idx])
        flip_count = target_ones - n_ones
        for idx in order[:flip_count]:
            h_int[zeros_idx[idx]] = 1
    return h_int

def project_continuous(h, n):
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    if abs(vc.sum() - target) < 1e-14:
        return vc
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def project_c(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    out = np.where(h >= 0.5, 1.0, 0.0)
    n1 = int(out.sum())
    if n1 > target:
        idx = np.where(out == 1)[0]
        idx = idx[np.argsort(-h[idx])]
        out[idx[:n1 - target]] = 0.0
    elif n1 < target:
        idx = np.where(out == 0)[0]
        idx = idx[np.argsort(h[idx])]
        out[idx[:target - n1]] = 1.0
    return out

def binary_sa(h, n, t_end, seed, T0=0.003, T1=3e-6):
    rng = np.random.RandomState(seed)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur == 1)[0].copy()
    zeros = np.where(cur == 0)[0].copy()
    t0 = time.time()
    while time.time() < t_end:
        frac = min((time.time() - t0) / max(t_end - t0, 1e-9), 1.0)
        T = T0 * (T1 / T0) ** frac
        pi = rng.randint(len(ones))
        pj = rng.randint(len(zeros))
        i, j = ones[pi], zeros[pj]
        cur[i] = 0.0
        cur[j] = 1.0
        nc = compute_c5(cur)
        if nc <= cur_c5 or rng.rand() < np.exp(-(nc - cur_c5) / max(T, 1e-15)):
            cur_c5 = nc
            ones[pi] = j
            zeros[pj] = i
            if nc < best_c5:
                best_c5 = nc
                best = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
    return best, best_c5

def continuous_cd(h, n, t_end, seed, max_sweeps=20000):
    rng = np.random.RandomState(seed)
    best = project_c(h.copy(), n)
    best_c5 = compute_c5(best)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best, best_c5
            orig = best[i]
            cands = [c for c in (0.0, 0.5, 1.0) if abs(c - orig) > 1e-6]
            if len(cands) < 2:
                cands = [0.25, 0.75]
            best_val = best_c5
            best_trial = None
            for v in cands:
                trial = best.copy()
                trial[i] = v
                trial = project_c(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_val - 1e-14:
                    best_val = c5t
                    best_trial = trial
            if best_trial is not None:
                best = best_trial
                best_c5 = best_val
                improved = True
        stall = 0 if improved else stall + 1
        if stall >= 10:
            break
    return best, best_c5

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init = np.array(initial_h_values, dtype=float)
    else:
        init = make_random_binary(160, seed)

    best_h = init.copy()
    best_n = len(best_h)
    best_c5 = compute_c5(best_h)

    def rem():
        return max(budget_s - (time.time() - t_start), 0.0)

    # Phase 1: Multi-n binary SA (55% budget)
    n_list = [best_n]
    for cand in [120, 140, 150, 160, 170, 180, 200, 240]:
        if cand != best_n and cand not in n_list:
            n_list.append(cand)
    n_list = n_list[:7]

    phase1_budget = budget_s * 0.55
    per_n = phase1_budget / len(n_list)
    rng_main = np.random.RandomState(seed)

    for nn in n_list:
        if rem() < 15:
            break
        t_n = time.time() + min(per_n, rem() - 5)
        if nn == best_n:
            h_n = binarize(best_h, nn)
        else:
            h_n = make_random_binary(nn, seed + nn * 31)

        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            best_n = nn

    # Phase 2: Intensive CD on best (20% of what remains)
    if rem() > 30:
        t2 = time.time() + rem() * 0.45
        best_h, best_c5 = continuous_cd(best_h, best_n, t2, seed + 4242)

        if c5_sa2 < best_c5:
            best_h = h_sa2
            best_c5 = c5_sa2
        h_cd2, c5_cd2 = continuous_cd(h_sa2, best_n, t3 + 5, seed + 55555)
        if c5_cd2 < best_c5:
            best_h = h_cd2
            best_c5 = c5_cd2

    # Phase 4: Final polish CD
    if rem() > 10:
        t4 = time.time() + rem() * 0.9
        best_h, best_c5 = continuous_cd(best_h, best_n, t4, seed + 12345)

    best_h = project_c(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
