import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def active_slsqp(h, n, K=5, maxiter=800):
    dx = 2.0 / n
    corr = compute_corr(h)
    top_k = np.argsort(corr)[-K:]
    def obj(x):
        c = compute_corr(x)[top_k]
        return np.max(c) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return proj(np.clip(res.x, 0, 1), n)

def softmax_slsqp(h, n, T, maxiter=800):
    dx = 2.0 / n
    def obj(x):
        c = compute_corr(x)
        m = np.max(c)
        sh = np.clip(T * (c - m), -200, 200)
        e = np.exp(sh)
        return (m + np.log(e.sum()) / T) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return proj(np.clip(res.x, 0, 1), n)

def coord_opt(h, n, i):
    h_save = h[i]
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full')
    dx = 2.0 / n
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        j1 = i + lag
        j2 = i - lag
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coef[k] = c
    k0 = n - 1
    def f1d(t):
        cv = corr_base + t * coef
        cv[k0] = corr_base[k0] + t * (1.0 - t)
        return float(np.max(cv)) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=150):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = proj(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_opt(h, n, t_end, K=5, lr_init=0.08):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    lr = lr_init
    no_improve = 0
    while time.time() < t_end:
        corr = compute_corr(best_h)
        cur_c5 = np.max(corr) * dx
        if cur_c5 < best_c5 - 1e-15:
            best_c5 = cur_c5
            no_improve = 0
        else:
            no_improve += 1
        active = np.argsort(corr)[-K:]
        T_sm = max(50.0, 300.0 - no_improve * 2.0)
        m = np.max(corr[active])
        sh = np.clip(T_sm * (corr[active] - m), -100, 100)
        exps = np.exp(sh)
        p = exps / max(exps.sum(), 1e-30)
        g = np.zeros(n)
        for idx, k in enumerate(active):
            pk = p[idx]
            if pk < 1e-12:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                if n - lag > 0:
                    g[:n - lag] += pk * (1.0 - best_h[lag:n])
                    g[lag:] += pk * (-best_h[:n - lag])
            else:
                nl = -lag
                if n - nl > 0:
                    g[nl:] += pk * (1.0 - best_h[:n - nl])
                    g[:n - nl] += pk * (-best_h[nl:])
        step = lr
        improved = False
        for _ in range(20):
            trial = proj(best_h - step * g, n)
            c5_t = compute_c5(trial)
            if c5_t < best_c5 - 1e-15:
                best_h = trial
                best_c5 = c5_t
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.7
            if lr < 1e-12:
                break
    return best_h, best_c5

def binary_anneal(h_start, n, t_end, n_iters=80000):

    target = n // 2
    ones = np.where(h == 1.0)[0]
    if len(ones) > target:
        to_flip = np.random.choice(ones, len(ones) - target, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < target:
        zeros = np.where(h == 0.0)[0]
        to_flip = np.random.choice(zeros, target - len(ones), replace=False)
        h[to_flip] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5
    T_init = 0.008
    T_min = 0.00005
    for it in range(n_iters):
        if time.time() > t_end:
            break
        T = T_init * (T_min / T_init) ** (it / float(n_iters))
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        a = ones_idx[np.random.randint(len(ones_idx))]
        b = zeros_idx[np.random.randint(len(zeros_idx))]
        h[a] = 0.0
        h[b] = 1.0
        c5_new = compute_c5(h)
        if c5_new <= cur_c5 or np.random.rand() < np.exp(-(c5_new - cur_c5) / max(T, 1e-15)):
            cur_c5 = c5_new
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
    return best_h, best_c5
