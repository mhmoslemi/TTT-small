Now for the coordinate descent, let me make it faster. The key is that when I change h[i], the correlation changes in a specific way. Let me precompute the "influence" of h[i] on each lag:

For lag k, c_k depends on h[i] through:
- h[i] in the first factor: contributes (1-h[i+k]) to c_k
- h[i] in the second factor (1-h[i-k]): contributes -h[i-k] to c_k (when i-k is the first index)

So if I change h[i] by δ:
Δc_k = δ * (1-h[i+k]) - δ * h[i-k] (for valid indices)

This is the same as before. For coordinate descent, I need to find h[i] that minimizes max_k (c_k + δ * influence_k[i]).

Let me precompute for each i:
influence[i, k] = (1-h[i+k])*1{valid} - h[i-k]*1{valid}

Then c_k(h with h[i] changed to v) = c_k + (v - h[i]) * influence[i, k]

And I want to minimize max_k [c_k + (v-h[i])*influence[i,k]] * dx over v ∈ [0,1].

This is a 1D min-max problem. I can solve it with scipy.optimize.minimize_scalar on the smooth version, or with a direct approach.

Actually, for a single coordinate, this is:
min_v max_k [c_k + (v-h[i])*a_k] * dx

where a_k = influence[i, k]. This is a convex function in v (max of linear functions). I can solve it with a simple approach:
- The function is piecewise linear in v
- The minimum occurs where the "upper envelope" is flat
- I can use minimize_scalar with bounded method

Let me just use minimize_scalar for simplicity (it's fast for 1D).

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def smooth_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = dx * np.ones(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = lags[k_idx]
        js, je = max(0, -k), min(n, n - k)
        if js < je:
            grad[js:je] -= dx * wk * h[js + k:je + k]
        js2, je2 = max(0, k), min(n, n + k)
        if js2 < je2:
            grad[js2:je2] -= dx * wk * h[js2 - k:je2 - k]
    return grad

def project_h(h, n=None):
    if n is None:
        n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * ((target - s) / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=800):
    dx = 2.0 / n
    def obj(x):
        return smooth_c5(x, T)
    def jac(x):
        return smooth_grad(x, T, n)
    ones = np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target(n), 'jac': lambda x: ones}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

def target(n):
    return n / 2.0

def true_subgrad(h, alpha, n_topk=8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-n_topk:]
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in lags[idx]:
        g = np.ones(n)
        js, je = max(0, -int(k)), min(n, n - int(k))
        if js < je:
            g[js:je] -= h[js + int(k):je + int(k)]
        js2, je2 = max(0, int(k)), min(n, n + int(k))
        if js2 < je2:
            g[js2:je2] -= h[js2 - int(k):je2 - int(k)]
        grad += g
    return project_h(h - alpha * grad / n_topk * dx)

def coord_opt_point(h, i, n):
    dx = 2.0 / n
    h_i = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    lags = np.arange(2 * n - 1) - (n - 1)
    # influence of changing h[i] from h_i to v:
    # c_k changes by (v - h_i) * [1-h[i+k] - h[i-k]] for valid indices
    infl = np.zeros(2 * n - 1)
    for k_idx in range(2 * n - 1):
        k = int(lags[k_idx])
        val = 0.0
        j1 = i + k
        if 0 <= j1 < n:
            val += (1.0 - h[j1])
        j2 = i - k
        if 0 <= j2 < n:
            val -= h[j2]
        infl[k_idx] = val
    def f1d(v):
        new_corr = corr + (v - h_i) * infl
        return np.max(new_corr) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def coordinate_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    improved_any = True
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if not improved_any:
            break
        improved_any = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            v_new = coord_opt_point(best_h, i, n)
            if abs(v_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = v_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved_any = True
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def smooth_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    w = exps / exps.sum()
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = dx * np.ones(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-14:
            continue
        k = int(lags[ki])
        js, je = max(0, -k), min(n, n - k)
        if js < je:
            grad[js:je] -= dx * wk * h[js + k:je + k]
        js2, je2 = max(0, k), min(n, n + k)
        if js2 < je2:
            grad[js2:je2] -= dx * wk * h[js2 - k:je2 - k]
    return grad

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * ((target - s) / tot)
    else:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=800):
    target = n / 2.0
    ones = np.ones(n)
    def obj(x):
        return smooth_c5(x, T)
    def jac(x):
        return smooth_grad(x, T, n)
    def con(x):
        return x.sum() - target
    def con_jac(x):
        return ones
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def true_subgrad(h, alpha, n_topk=8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-n_topk:]
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in lags[idx]:
        ki = int(k)
        g = np.ones(n)
        js, je = max(0, -ki), min(n, n - ki)
        if js < je:
            g[js:je] -= h[js + ki:je + ki]
        js2, je2 = max(0, ki), min(n, n + ki)
        if js2 < je2:
            g[js2:je2] -= h[js2 - ki:je2 - ki]
        grad += g
    return project_h(h - alpha * grad / n_topk * dx)

def coord_opt_point(h, i, n):
    dx = 2.0 / n
    h_i = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    lags = np.arange(2 * n - 1) - (n - 1)
    infl = np.empty(2 * n - 1)
    for ki in range(2 * n - 1):
        k = int(lags[ki])
        val = 0.0
        j1 = i + k
        if 0 <= j1 < n:
            val += (1.0 - h[j1])
        j2 = i - k
        if 0 <= j2 < n:
            val -= h[j2]
        infl[ki] = val
    def f1d(v):
        return np.max(corr + (v - h_i) * infl) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x

def coordinate_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            v_new = coord_opt_point(best_h, i, n)
            if abs(v_new - best_h[i]) < 1e-12:
                continue
            ht = best_h.copy()
            ht[i] = v_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def nesterov_steps(h, n, T, nsteps=200, t_end=None):
    dx = 2.0 / n
    x = h.copy()
    v = np.zeros(n)
    beta = 0.9
    best = x.copy()
    best_c = compute_c5(x)
    for t in range(nsteps):
        if t_end is not None and time.time() > t_end:
            break
        lr = 0.02 * (1.0 + t / nsteps) ** 0.5
        g = smooth_grad(x, T, n)
        v = beta * v + lr * g
        x_new = project_h(x - v)
        c = compute_c5(x_new)
        if c < best_c:
            best_c = c
            best = x_new.copy()
        x = x_new
    return best, best_c
