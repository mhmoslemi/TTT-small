def smooth_c5(h, T):

    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

The second term is the "autocorrelation" at lag m.

So C₅ = max_m [tail_sum(m) - autocorr(m)] * dx

where tail_sum(m) = Σ_{j=m}^{n-1} h(j) and autocorr(m) = Σ_{i=0}^{n-1-m} h(i)h(i+m).

This is an interesting reformulation. The tail_sum(m) is determined by h, and autocorr(m) is also determined by h. We want to minimize the maximum of (tail_sum(m) - autocorr(m)) over m.

For m=0: tail_sum(0) - autocorr(0) = Σ h(i) - Σ h(i)² = n/2 - Σ h(i)²
This is maximized when h is as "spread out" as possible (all h(i) = 0.5), giving n/2 - n/4 = n/4. With dx: (n/4)(2/n) = 1/2. Hmm, that's the maximum possible value for m=0.

For m=n-1: tail_sum(n-1) - autocorr(n-1) = h(n-1) - h(0)h(n-1) = h(n-1)(1-h(0))
This is at most 0.25 (when h(0)=0.5, h(n-1)=1 or vice versa). With dx: 0.25 * 2/n = 0.5/n. Very small.

So the interesting shifts are in the middle range.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    dx = 2.0 / n
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_ternary(h, n, i, niter=60):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(vm1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(vm2[valid2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr_base + m1 * coefs)
        f2 = np.max(corr_base + m2 * coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_opt_pair(h, n, i, j, niter=40):
    h_base = h.copy()
    h_base[i] = 0.0
    h_base[j] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1_i = i - m_vals; vm2_i = i + m_vals
    vm1_j = j - m_vals; vm2_j = j + m_vals
    v1i = (vm1_i >= 0) & (vm1_i < n); v2i = (vm2_i >= 0) & (vm2_i < n)
    v1j = (vm1_j >= 0) & (vm1_j < n); v2j = (vm2_j >= 0) & (vm2_j < n)
    ci = np.zeros(2*n-1); cj = np.zeros(2*n-1)
    ci[v1i] += (1.0 - h[np.clip(vm1_i[v1i], 0, n-1)])
    ci[v2i] += (-h[np.clip(vm2_i[v2i], 0, n-1)])
    cj[v1j] += (1.0 - h[np.clip(vm1_j[v1j], 0, n-1)])
    cj[v2j] += (-h[np.clip(vm2_j[v2j], 0, n-1)])
    def f2d(xi, xj):
        return np.max(corr_base + xi * ci + xj * cj)
    lo_i, hi_i, lo_j, hi_j = 0.0, 1.0, 0.0, 1.0
    for _ in range(niter):
        mi1 = lo_i + (hi_i-lo_i)/3; mi2 = hi_i - (hi_i-lo_i)/3
        mj1 = lo_j + (hi_j-lo_j)/3; mj2 = hi_j - (hi_j-lo_j)/3
        f1 = f2d(mi1, mj1); f2 = f2d(mi2, mj2)
        f3 = f2d(mi1, mj2); f4 = f2d(mi2, mj1)
        best_v = min(f1,f2,f3,f4)
        if best_v == f1: hi_i=mi2; hi_j=mj2
        elif best_v == f2: lo_i=mi1; lo_j=mj1
        elif best_v == f3: hi_i=mi2; lo_j=mj1
        else: lo_i=mi1; hi_j=mj2
    return (lo_i+hi_i)/2.0, (lo_j+hi_j)/2.0

def coord_descent(h, n, t_end, max_sweeps=100, use_pairs=False):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_ternary(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if use_pairs and sweep % 3 == 2 and time.time() < t_end:
            for _ in range(n // 4):
                if time.time() > t_end:
                    break
                i, j = np.random.permutation(n)[:2]
                hi, hj = coord_opt_pair(best_h, n, i, j)
                if abs(hi - best_h[i]) < 1e-12 and abs(hj - best_h[j]) < 1e-12:
                    continue
                h_try = best_h.copy()
                h_try[i] = hi; h_try[j] = hj
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5
