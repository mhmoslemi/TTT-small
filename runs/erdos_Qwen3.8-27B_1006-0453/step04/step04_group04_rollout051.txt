                improved = True
        if not improved:
            break
    return best, best_c5

But actually I can be smarter: for a single coordinate change, I can compute the correlation change analytically in O(n) instead of O(n²).

When h[i] changes from 0 to t (with all other h fixed):
corr_old[k] = sum_{valid} h[a] * (1-h[b])  (where {a,b} depends on lag)
corr_new[k] = corr_old[k] + t * (1 - h[b_i(k)]) - t * h[a_i(k)] - t² * [i appears in both a and b]

The self-interaction (t²) only happens at lag=0 where i appears both as h[i] and (1-h[i]):
At lag 0: term = h[i]*(1-h[i]) = t*(1-t) = t - t²
So the change is t - t² - 0 = t - t² (from 0 to t).

For other lags, the change is linear in t.

So: corr_new[k] = corr_old[k] + t * L[k] + (k==n-1 ? -t² : 0)
where L[k] is the linear coefficient (= M[i,k] from before, but with h[i]=0).

def coord_opt_fast(h, n, i):
    # Compute correlation with h[i] = 0
    hz = h.copy()
    hz[i] = 0.0
    corr0 = get_corr(hz)
    # Linear coefficient for h[i] (M[i,:] with h[i]=0)
    # This is the same as M[i,:] when h[i]=0
    L = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = k - (n-1)
        val = 0.0
        if lag >= 0:
            if lag <= i <= n-1:
                val += (1.0 - hz[i-lag])  # hz[i]=0 but this is the OTHER position
            if 0 <= i <= n-1-lag:
                val += -hz[i+lag]
        else:
            nl = -lag
            if nl <= i <= n-1:
                if i+nl < n:
                    val += (1.0 - hz[i+nl])
            if i >= 2*nl:
                val += -hz[i-nl]
        L[k] = val
    # The corrected correlation: corr(t) = corr0 + t*L, except at k=n-1 where we add -t²
    k0 = n-1
    def f1(t):
        cv = corr0 + t * L
        cv[k0] -= t * t
        return cv.max()
    res = minimize_scalar(lambda t: f1(t), bounds=(0,1), method='bounded', options={'xatol':1e-10})
    return res.x

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < target:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * ((target - s) / tot)
    else:
        h *= (target / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, T, maxiter=800, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        c = get_corr(np.clip(x, 0, 1))
        m = c.max()
        sh = np.clip(T * (c - m), -400, 400)
        return (m + np.log(np.exp(sh).sum()) / T) * dx
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':con}],
                   options={'maxiter':maxiter, 'ftol':ftol})
    return proj(np.clip(res.x, 0, 1), n)

def slsqp_active(h, n, active, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        return float(np.max(get_corr(np.clip(x,0,1))[active])) * dx
    def con(x):
        return x.sum() - n / 2.0
    res = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':con}],
                   options={'maxiter':maxiter, 'ftol':1e-16})
    return proj(np.clip(res.x, 0, 1), n)

def find_active(h, tol=1e-8):
    c = get_corr(h)
    mx = c.max()
    return np.where(c >= mx - tol)[0]

def coord_fast_opt(h, n, i):
    full_corr = get_corr(h)
    k0 = n - 1
    L = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = k - (n-1)
        val = 0.0
        if lag >= 0:
            if lag <= i <= n-1:
                val += (1.0 - h[i-lag])
            if 0 <= i <= n-1-lag:
                val += -h[i+lag]
        else:
            nl = -lag
            if nl <= i and i+nl < n:
                val += (1.0 - h[i+nl])
            if i >= 2*nl:
                val += -h[i-nl]
        L[k] = val
    hi = h[i]
    corr0 = full_corr - hi * L
    corr0[k0] += hi * hi
    def f1(t):
        cv = corr0 + t * L
        cv[k0] -= t * t
        return cv.max()
    res = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded', options={'xatol':1e-10})
    return res.x, res.fun

def coord_desc(h, n, t_end, max_sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best, best_c5
            t_new, f_val = coord_fast_opt(best, n, i)
            if f_val * dx < best_c5 - 1e-14:
                best = best.copy()
                best[i] = t_new
                best = proj(best, n)
                best_c5 = compute_c5(best)
                improved = True
        if not improved:
            break
    return best, best_c5

def binary_swap(h, n, t_end, max_sweeps=30):
    best = h.copy()
    order = np.argsort(best)[::-1]
    best[order[:n//2]] = 1.0
    best[order[n//2:]] = 0.0
    best_c5 = compute_c5(best)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        M = np.zeros((n, 2*n-1))
        for j in range(n):

                val = 0.0

                        val += (1.0 - best[j-lag])

                        val += -best[j+lag]

                    nl = -lag
                    if nl <= j and j+nl < n:
                        val += (1.0 - best[j+nl])
                    if j >= 2*nl:
                        val += -best[j-nl]
                M[j,k] = val
        corr = get_corr(best)
        mx = corr.max()
        ones = np.where(best == 1.0)[0]
        zeros = np.where(best == 0.0)[0]
