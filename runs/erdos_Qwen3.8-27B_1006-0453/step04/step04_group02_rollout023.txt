```python
def smooth_c5(h, T):

        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
```

**Idea: Track which lags are "active" (near the maximum) and focus optimization on coordinates that affect those lags.**

**Idea: Use the dual problem.** The problem min_h max_k corr[k]*dx can be reformulated. The optimal solution satisfies: there exist multipliers λ_k ≥ 0, sum λ_k = 1, such that h minimizes sum_k λ_k * corr[k] * dx. This is a bilinear (quadratic) program. We can alternate:
- Given λ: minimize the quadratic sum_k λ_k * corr[k] (this is a QP, or can be solved by coordinate descent)
- Given h: set λ_k proportional to "how close" corr[k] is to the max

This is the method of Lagrange multipliers / minimax duality. Let me implement this!

```python
def minimax_iterate(h, n, t_end, max_iters=50):
    """Alternating: update h for fixed weights, update weights based on h."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T = 100  # Temperature for weight computation
    
    for iteration in range(max_iters):
        if time.time() > t_end:
            break
        # Compute weights based on current max
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = np.max(corr)
        # Soft max weights
        weights = np.exp(T * (corr - m))
        weights /= weights.sum()
        
        # Optimize h to minimize weighted sum of correlations
        # This is equivalent to minimizing a quadratic: sum_k w_k * corr_k
        # Use SLSQP with the weighted objective
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return np.sum(weights * c) * (2.0 / n)
        
        def coneq(x):
            return x.sum() - n / 2.0
        
        bnds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': coneq}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bnds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-12})
            new_h = project(np.clip(res.x, 0, 1), n)
        except:
            new_h = best_h
        
        cv = compute_c5(new_h)
        if cv < best_c5:
            best_h, best_c5 = new_h, cv
        
        # Increase T to focus on max
        T *= 2.0
        if T > 100000:
            T = 100000
    
    return best_h, best_c5
```

```python
            # Specifically:

            # where delta_m depends linearly (and quadratically for m=n-1) on t

            best_h[i] = 0.0

            # corr[n-1+k] = corr_base[n-1+k] + ... 
            # When we set h[i]=t:
            # corr[k>0] gets: (1-h[i-k])*t (from h[i]*(1-h[i-k])) + h[i+k]*(h[i_old]-t)... 
            # This is getting complicated. Let me just use the same approach as before.

                    d[n-1-k] += (1.0 - best_h[i+k])
                if 0 <= i + k < n:
                    d[n-1+k] += (-best_h[i+k])
                    d[n-1-k] += (-best_h[i-k])
            
            c0 = corr_base[n-1]

            # Minimize 1D quadratic (approximately)
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T < 10:
        return m * dx
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        ss = max(s, 1e-14)
        h = np.clip(h * (target / ss), 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if diff2 > 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
        elif diff2 < -1e-14:
            ss2 = max(h.sum(), 1e-14)
            h = np.clip(h * (target / ss2), 0.0, 1.0)
    return h

def coord_opt_exact(h, n, i):
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = h_save
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        if 0 <= i - k < n:
            d[n - 1 + k] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            d[n - 1 + k] += (-h[i + k])
        if 0 <= i + k < n:
            d[n - 1 - k] += (1.0 - h[i + k])
        if 0 <= i - k < n:
            d[n - 1 - k] += (-h[i - k])
    c0 = corr_base[n - 1]
    def f1d(t):
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return np.max(vals) * dx
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return res.x

def intensive_cd(h, n, t_end, max_sweeps=200, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        changed = False
        for i in order:
            if time.time() > t_end:
                best_h = project(best_h, n)
                return best_h, compute_c5(best_h)
            hi = coord_opt_exact(best_h, n, i)
            if abs(hi - best_h[i]) > 1e-13:
                best_h[i] = hi
                changed = True
        best_h = project(best_h, n)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
            if stall >= 4:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project(np.clip(res.x, 0, 1), n)

def weighted_slsqp_step(h, n, weights, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return np.dot(weights, c) * dx
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    return project(np.clip(res.x, 0, 1), n)

def minimax_alternating(h, n, t_end, max_iters=40):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    T = 50.0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = np.max(corr)
        w = np.exp(T * (corr - m))
        w /= w.sum()
        try:
            new_h = weighted_slsqp_step(best_h, n, w, maxiter=400)
        except Exception:
            new_h = best_h
        cv = compute_c5(new_h)
        if cv < best_c5:
            best_h, best_c5 = new_h, cv
            best_h = project(best_h, n)
        T = min(T * 1.8, 50000.0)
    return best_h, best_c5

def subgradient_step(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    active = np.where(corr > m - 0.05 * dx)[0]
    grad = np.zeros(n)
    for am in active:
        k = am - (n - 1)
        weight = 1.0 / len(active)
        for i in range(n):
            if k > 0:
                if 0 <= i - k < n:
                    grad[i] += weight * (1.0 - h[i - k])
                if 0 <= i + k < n:
                    grad[i] += weight * (-h[i + k])
            elif k < 0:
                kk = -k
                if 0 <= i + kk < n:
                    grad[i] += weight * (1.0 - h[i + kk])
                if 0 <= i - kk < n:
                    grad[i] += weight * (-h[i - kk])
            else:
                grad[i] += weight * (1.0 - 2.0 * h[i]) * (-1.0)
    new_h = h + alpha * grad
    new_h = np.clip(new_h, 0.0, 1.0)
    new_h = project(new_h, n)
    return new_h

def basin_hop_v2(h, n, t_end, rng, n_iters=25, base_step=0.025):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.92 * (it / max(n_iters, 1))
        step = base_step * (0.15 + 0.85 * cool)
        pert = work + step * rng.standard_normal(n)
        pert = np.clip(pert, 0, 1)
        pert = project(pert, n)
        hc = intensive_cd(pert, n, min(t_end, time.time() + 12.0), max_sweeps=40, rng=rng)
        cv = compute_c5(hc)
        temp = 0.0025 * cool + 0.0002
        if cv < work_c5 or rng.random() < np.exp(-(cv - work_c5) / temp):
            work, work_c5 = hc, cv
        if work_c5 < best_c5:
            best, best_c5 = work.copy(), work_c5
    return best, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)
```
