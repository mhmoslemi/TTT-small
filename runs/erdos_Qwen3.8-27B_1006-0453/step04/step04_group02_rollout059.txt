def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Bisection: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    if s > target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        h_min = np.clip(h - mid, 0.0, 1.0)
        if h_min.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    h = np.clip(h - lam, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if diff > 1e-10:
        room = 1.0 - h
        if room.sum() > 1e-14:
            h += room * (diff / room.sum())
    elif diff < -1e-10:
        s2 = h.sum()
        if s2 > 1e-14:
            idx = np.where(h > 0)[0]
            h[idx] += diff * h[idx] / s2
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-10:
        room = 1.0 - h
        if room.sum() > 1e-14:
            h += room * (diff / room.sum())
    elif diff < -1e-10:
        s2 = h.sum()
        if s2 > 1e-14:
            idx = np.where(h > 1e-14)[0]
            h[idx] += diff * h[idx] / h[idx].sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    ss = exps.sum()
    w = exps / ss
    val = (m + np.log(ss) / T) * dx
    FT = np.convolve(1.0 - h, w, 'full')[n - 1:n - 1 + n]
    ST = np.correlate(w, h, 'full')[n - 1::-1][:n]
    grad = (FT - ST) * dx
    return val, grad

def coord_opt(h, n, i):
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full').copy()
    ij = h[i]
    h[i] = ij
    d = np.zeros(2 * n - 1)
    for k in range(1, n):
        dk = 0.0
        if 0 <= i - k < n:
            dk += 1.0 - h[i - k]
        if 0 <= i + k < n:
            dk -= h[i + k]
        d[n - 1 + k] = dk
        d[n - 1 - k] = dk
    c0 = corr_base[n - 1]
    dx = 2.0 / n

    def f1d(t):
        t = max(0.0, min(1.0, t))
        vals = corr_base + t * d
        vals[n - 1] = c0 + t * (1.0 - t)
        return float(np.max(vals) * dx)

    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=50, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            new = coord_opt(best_h, n, i)
            if abs(new - best_h[i]) > 1e-11:
                best_h[i] = new
                best_h = project_box_sum(best_h, n * 0.5)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return best_h, best_c5

def grad_step(h, n, T, lr, t_end, n_steps):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    y = h.copy()
    v = np.zeros(n)
    beta = 0.85
    for it in range(n_steps):
        if time.time() > t_end:
            break
        val, grad = smooth_obj_grad(y, T)
        v = beta * v - lr * grad
        step = 1.0 / (1.0 + 0.1 * it)
        y_new = y + v * step
        y_new = project_box_sum(y_new, n * 0.5)
        c5 = compute_c5(y_new)
        if c5 < best_c5:
            best_h, best_c5 = y_new.copy(), c5
        y = y_new
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_box_sum(np.interp(x_new, x_old, h_old), n_new * 0.5)

def perturb_reoptimize(h, n, rng, t_check, step_sizes, t_opt):
    best_h, best_c5 = h.copy(), compute_c5(h)
    for ss in step_sizes:
        if time.time() > t_check:
            break
        pert = h + ss * rng.standard_normal(n)
        pert = project_box_sum(pert, n * 0.5)
        hc, c5 = coord_descent(pert, n, t_opt, max_sweeps=25, rng=rng)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(240)
        best_h[:120] = 1.0
    n = len(best_h)
    best_h = project_box_sum(best_h, n * 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial gradient stepping
    for T in [200, 1000, 5000]:
        if elapsed() > budget_s * 0.10:
            break
        hw, cw = grad_step(best_h, n, T, lr=0.02/n, t_end=t0+budget_s*0.04, n_steps=300)
        if cw < best_c5:
            best_h, best_c5 = hw, cw

    # Phase 2: Coordinate descent to local optimum
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.25, max_sweeps=50, rng=rng)

    # Phase 3: ILS - perturb and re-optimize with multiple strategies
    t_ils_end = t0 + budget_s * 0.70
    n_trials = 0
    while elapsed() < t_ils_end - budget_s * 0.05:
        n_trials += 1
        trial_time = min(budget_s * 0.04, t_ils_end - elapsed())
        r = rng.random()
        if r < 0.35:
            # Small Gaussian perturb
            ss = 0.03 + 0.05 * rng.random()
            pert = best_h + ss * rng.standard_normal(n)
        elif r < 0.60:
            # Coordinate flip (set some to 0 or 1)
            k = rng.integers(3, 15)
            ids = rng.choice(n, k, replace=False)
            pert = best_h.copy()
            pert[ids] = rng.uniform(0, 1, k)
        elif r < 0.80:
            # Gradient-informed direction
            _, grad = smooth_obj_grad(best_h, n, 5000)
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-10:
                direction = -grad / g_norm
            else:
                direction = rng.standard_normal(n)
            ss = 0.05 + 0.04 * rng.random()
            pert = best_h + ss * direction
        else:
            # Larger perturbation
            ss = 0.08 + 0.06 * rng.random()
            pert = best_h + ss * rng.standard_normal(n)

        pert = project_box_sum(pert, n * 0.5)
        t_opt = t0 + elapsed() + trial_time * 0.6
        hc, c5 = coord_descent(pert, n, t_opt, max_sweeps=20, rng=rng)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
            # Short gradient polish after improvement
            if elapsed() < t_ils_end - budget_s * 0.03:
                hw, cw = grad_step(best_h, n, 10000, lr=0.01/n, t_end=t0+elapsed()+budget_s*0.015, n_steps=100)
                if cw < best_c5:
                    best_h, best_c5 = hw, cw
                    best_h, best_c5 = coord_descent(best_h, n, t0+elapsed()+budget_s*0.015, max_sweeps=10, rng=rng)

    # Phase 4: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82, max_sweeps=50, rng=rng)

    # Phase 5: Multi-resolution
    for n_try in [200, 280, 360, 480]:
        if elapsed() > budget_s * 0.92:
            break
        h_interp = interpolate_h(best_h, n_try)
        c5_int = compute_c5(h_interp)
        hc, c5_cd = coord_descent(h_interp, n_try, t0 + budget_s * 0.015, max_sweeps=20, rng=rng)
        if c5_cd < best_c5:
            best_h, best_c5, n = hc, c5_cd, n_try
            # Polish at new resolution
            if elapsed() < budget_s * 0.94:
                best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, max_sweeps=15, rng=rng)

    # Phase 6: Final polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=40, rng=rng)

    best_h = project_box_sum(best_h, len(best_h) * 0.5)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
