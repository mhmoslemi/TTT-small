            if k == 0:
                grad[i] += (1.0 - 2.0 * h[i])
            else:
                ip, im = i + k, i - k
                if 0 <= ip < n:
                    grad[i] += (1.0 - h[ip])
                if 0 <= im < n:
                    grad[i] -= h[im]
    grad /= len(active)
    h_new = h - alpha * grad
    h_new = project(h_new, n)
    return h_new

def block_shuffle(h, n, rng):
    """Shuffle a random contiguous block"""
    blen = rng.randint(n // 8, n // 3)
    pos = rng.randint(0, n - blen)
    h2 = h.copy()
    h2[pos:pos+blen] = rng.permutation(h2[pos:pos+blen])
    return h2

def basin_hop2(h, c5, n, t_end, max_iters=200):
    best_h, best_c5 = h.copy(), c5
    work_h, work_c5 = h.copy(), c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.03, 1.0 - 0.9 * it / max_iters)
        rng = np.random.RandomState(it + 12345)
        # Different perturbation strategies
        strat = it % 3
        if strat == 0:
            step = 0.03 + 0.15 * cool
            hp = best_h + step * np.random.randn(n)
        elif strat == 1:
            hp = block_shuffle(best_h, n, np.random)
            hp = hp + 0.05 * cool * np.random.randn(n)
        else:
            # Swap two random pairs
            hp = best_h.copy()
            for _ in range(3 + int(10 * cool)):
                i, j = np.random.choice(n, 2, replace=False)
                hp[i], hp[j] = hp[j], hp[i]
        hp = project(hp, n)
        hc, c5v = coord_descent(hp, n, min(t_end, time.time() + 3.0), max_sweeps=60)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_h, best_c5 = hc.copy(), c5v
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(x_new, x_old, h), n_new)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp_opt(h0, T, n, maxiter=400):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)
    except:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        best_h = np.zeros(n0)
        for i in range(0, n0, 2):
            best_h[i] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive coordinate descent
    t_end = t0 + budget_s * 0.20
    best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=800)

    # Phase 2: SLSQP with T-schedule
    if time.time() < t0 + budget_s * 0.35:
        for T in [50, 200, 1000, 5000]:
            if time.time() > t0 + budget_s * 0.35:
                break
            hc = slsqp_opt(best_h, T, n, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5, best_h = c5v, hc

    # Phase 3: Coordinate descent after SLSQP
    t_end = t0 + budget_s * 0.45
    best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=500)

    # Phase 4: Subgradient polishing
    if time.time() < t0 + budget_s * 0.52:
        alpha = 0.01
        for _ in range(100):
            if time.time() > t0 + budget_s * 0.52:
                break
            hs = subgrad_step(best_h, n, alpha)
            c5s = compute_c5(hs)
            if c5s < best_c5:
                best_c5, best_h = c5s, hs
            else:
                alpha *= 0.9
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.52, max_sweeps=100)

    # Phase 5: Basin hopping (more aggressive)
    t_end = t0 + budget_s * 0.78
    best_h, best_c5 = basin_hop2(best_h, best_c5, n, t_end, max_iters=400)

    # Phase 6: Final coordinate descent + SLSQP
    t_end = t0 + budget_s * 0.88
    best_h, best_c5 = coord_descent(best_h, n, t_end, max_sweeps=600)
    if time.time() < t0 + budget_s * 0.95:
        for T in [10000, 50000]:
            if time.time() > t0 + budget_s * 0.95:
                break
            hc = slsqp_opt(best_h, T, n, maxiter=1000)
