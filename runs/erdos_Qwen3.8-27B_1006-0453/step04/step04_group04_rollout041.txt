from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def exact_coord(h, i):
    n = len(h)
    dx = 2.0 / n
    hz = h.copy()
    hz[i] = 0.0
    corr_base = np.correlate(hz, 1.0 - hz, mode='full').copy()
    k0 = n - 1
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - k0
        c = 0.0
        jf = i - lag
        if 0 <= jf < n:
            c += 1.0 - h[jf]
        js = i + lag
        if 0 <= js < n:
            c -= h[js]
        coef[k] = c
    def f1d(t):
        if t < 1e-15:
            return float(corr_base.max()) * dx
        cv = corr_base + t * coef
        cv[k0] -= t * t
        return float(cv.max()) * dx
    r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                        options={'xatol': 1e-12, 'maxiter': 200})
    return max(0.0, min(1.0, r.x))

def coord_desc(h, n, t_end, sweeps=400, seed=0):

            bh[i] = t
            bh = proj(bh, n)
            c = compute_c5(bh)
            if c < bc - 1e-14:

                imp = True
        if not imp:

    else:  # Random walk step

        h[i] += 0.1 * rng.choice([-1, 1])

        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.99, sweeps=100, seed=271)

    best_h = proj(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
