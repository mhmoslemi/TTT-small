import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def smooth_grad(h, T):
    """Gradient of smooth C5 w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_c = np.exp(T * (corr - m))
    w = exp_c / np.sum(exp_c)
    
    # Only use top 30 weights for efficiency
    top_idx = np.argsort(w)[-30:]
    w_top = w[top_idx] / np.sum(w[top_idx])
    
    grad = np.zeros(n)
    i = np.arange(n)
    for idx in range(len(top_idx)):
        k = (top_idx[idx] - (n - 1))  # actual lag
        wt = w_top[idx]
        j1 = i - k
        j2 = i + k
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        grad += wt * ((1.0 - h[np.clip(j1, 0, n-1)]) * v1 - h[np.clip(j2, 0, n-1)] * v2)
    
    return grad * dx

def coord_opt_single(h, n, i):
    """Optimize h[i] holding others fixed."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n-1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2*n-1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n-1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n-1)])
    dx = 2.0 / n
    
    def f1d(hi):
        return np.max(corr_base + hi * coefs) * dx
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_c = np.exp(T * (corr - m))
    s = np.sum(exp_c)
    val = (m + np.log(s) / T) * dx
    w = exp_c / s
    top30 = np.argsort(w)[-30:]
    w_top = w[top30] / np.sum(w[top30])
    grad = np.zeros(n)
    i = np.arange(n)
    for idx in range(len(top30)):
        k = top30[idx] - (n - 1)
        wt = w_top[idx]
        j1 = i - k
        j2 = i + k
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        grad += wt * ((1.0 - h[np.clip(j1, 0, n-1)]) * v1 - h[np.clip(j2, 0, n-1)] * v2)
    grad *= dx
    return val, grad

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return h
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h + diff / n
        if h.min() < 0:
            h = np.clip(h, 0.0, 1.0)
            s2 = h.sum()
            if abs(s2 - target) > 1e-10:
                h = h + (target - s2) / (n - np.sum(h > 0.99))
                h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def momentum_opt(h, t_end, lr0=0.02, T=300.0, max_iter=5000):
    n = len(h)
    vel = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        lr = lr0 / (1.0 + 0.002 * it)
        _, grad = smooth_c5_and_grad(h, T)
        vel = beta * vel - lr * grad
        h_new = h + vel
        h_new = project_h(h_new)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        if it % 200 == 0:
            T += 50
    return best_h, best_c5

def coord_opt_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    valid1 = (j1 >= 0) & (j1 < n)
    valid2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(j1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(j2[valid2], 0, n - 1)])
    dx = 2.0 / n

    def f1d(hi):
        return np.max(corr_base + hi * coefs) * dx

    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)

                return best_h, best_c5
            hi_new = coord_opt_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:

            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5
