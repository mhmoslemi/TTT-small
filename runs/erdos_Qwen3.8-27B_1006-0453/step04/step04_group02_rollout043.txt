import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h - h * (diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def fast_coord(h, n, t_end, max_sweeps=200, stall_lim=5, rng=None, grid_pts=100):
    if rng is None:
        rng = np.random.default_rng(42)
    dx = 2.0 / n
    work = h.astype(float).copy()
    corr = np.correlate(work, 1.0 - work, mode='full').astype(float).copy()
    best_c5 = float(np.max(corr) * dx)
    best_h = work.copy()
    stall = 0
    ts_grid = np.linspace(0.0, 1.0, grid_pts)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() > t_end:
                break
            old = work[j]
            if old < 1e-14 or old > 1.0 - 1e-14:
                continue
            dvec = np.zeros(2 * n - 1)
            i_lo = max(0, j - n + 1)
            i_hi = min(j, n - 1)
            if i_lo <= i_hi:
                iv = np.arange(i_lo, i_hi + 1)
                dvec[n - 1 + (j - iv)] += (1.0 - work[iv])
            k_lo = max(0, -j)
            k_hi = min(n - 1 - j, n - 1)
            if k_lo <= k_hi:
                kv = np.arange(k_lo, k_hi + 1)
                dvec[n - 1 + kv] += (-work[j + kv])
            
            bc0 = corr[n - 1]
            # Grid evaluation
            c_grid = corr[None, :] + (ts_grid[:, None] - old) * dvec[None, :]
            c_grid[:, n - 1] = bc0 + (ts_grid - old) * (1.0 - old - ts_grid)
            vals = np.max(c_grid, axis=1) * dx
            bi = int(np.argmin(vals))
            nt = float(ts_grid[bi])
            
            # Refine: bisection in [ts_grid[bi-1], ts_grid[bi+1]]
            if bi > 0:
                lo = float(ts_grid[bi - 1])
            else:
                lo = 0.0
            if bi < grid_pts - 1:
                hi = float(ts_grid[bi + 1])
            else:
                hi = 1.0
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                d1, d2 = m1 - old, m2 - old
                c1 = corr + d1 * dvec; c1[n-1] = bc0 + d1*(1.0-old-m1)
                c2 = corr + d2 * dvec; c2[n-1] = bc0 + d2*(1.0-old-m2)
                v1, v2 = float(np.max(c1)*dx), float(np.max(c2)*dx)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            nt = 0.5 * (lo + hi)
            
            if abs(nt - old) < 1e-13:
                continue
            d = nt - old
            corr += d * dvec
            corr[n - 1] = bc0 + d * (1.0 - old - nt)
            work[j] = nt
            nc5 = float(np.max(corr) * dx)
            if nc5 < best_c5 - 1e-15:

                best_h = work.copy()
                improved = True
        if improved:

        else:
            stall += 1

    work = project(work, n)
    c5f = compute_c5(work)
    if c5f < best_c5:

    return best_h, best_c5

def basin_hop(h, n, t_end, rng, n_iter=50, base_step=0.03):
    work = h.copy()
    wc5 = compute_c5(work)
    best, bc5 = h.copy(), wc5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * it / max(n_iter, 1)
        step = base_step * (0.2 + 0.8 * cool)
        pert = project(work + step * rng.standard_normal(n), n)
        hc, hc5 = fast_coord(pert, n, min(t_end, time.time() + 4.0),
                              max_sweeps=30, stall_lim=3, rng=rng, grid_pts=80)
        temp = 0.003 * cool + 0.0002
        if hc5 < wc5 or rng.random() < np.exp(-(hc5 - wc5) / temp):
            work, wc5 = hc, hc5
        if wc5 < bc5:
            best, bc5 = work.copy(), wc5
    return best, bc5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:

    n = len(best_h)

    best_c5 = compute_c5(best_h)
    
    # Phase 1: Fast coordinate descent on initial solution
    t1 = t0 + budget_s * 0.20
    best_h, best_c5 = fast_coord(best_h, n, t1, max_sweeps=150, stall_lim=5, rng=rng)
    
    # Phase 2: Basin hopping
    if elapsed() < budget_s * 0.55:
        t2 = t0 + budget_s * 0.55
        best_h, best_c5 = basin_hop(best_h, n, t2, rng, n_iter=40, base_step=0.03)
    
    # Phase 3: More coordinate descent
    if elapsed() < budget_s * 0.75:
        t3 = t0 + budget_s * 0.75
        best_h, best_c5 = fast_coord(best_h, n, t3, max_sweeps=200, stall_lim=5, rng=rng, grid_pts=150)
    
    # Phase 4: Try different resolutions
    for n_try in [160, 200, 320, 400]:
        if elapsed() > budget_s * 0.88:

        h_interp = interp_to(best_h, n_try)
        t_try = min(t0 + budget_s * 0.90, time.time() + 8.0)
        hc, hc5 = fast_coord(h_interp, n_try, t_try, max_sweeps=80, stall_lim=4, rng=rng)
        if hc5 < best_c5:
            best_h, best_c5, n = hc, hc5, n_try
    
    # Phase 5: Final polish
    if elapsed() < budget_s * 0.97:
        t5 = t0 + budget_s * 0.97
        best_h, best_c5 = fast_coord(best_h, n, t5, max_sweeps=300, stall_lim=3, rng=rng, grid_pts=200)

        t6 = t0 + budget_s * 0.99
        best_h, best_c5 = basin_hop(best_h, n, t6, rng, n_iter=20, base_step=0.015)
    
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

            if tot < 1e-15:
