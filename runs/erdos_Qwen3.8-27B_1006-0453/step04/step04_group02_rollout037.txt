    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        h = np.clip(h * (target / s), 0, 1)
        diff = target - h.sum()
        if diff > 1e-13:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-15:
                h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-15:
            h = np.clip(h + room * (target - s) / tot, 0, 1)
    return h

def _smooth_val(corr, T, dx):
    m = np.max(corr)
    e = np.clip(T * (corr - m), -500, 500)
    ls = np.log(np.sum(np.exp(e)))
    return (m + ls / T) * dx

def _smooth_weights(corr, T, dx):
    m = np.max(corr)
    e = np.clip(T * (corr - m), -500, 500)
    ls = np.log(np.sum(np.exp(e)))
    return np.exp(e - ls) / T * dx

def sfn(x, T, n):
    return _smooth_val(np.correlate(x, 1.0 - x, mode='full'), T, 2.0 / n)

def sjac(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    w = _smooth_weights(corr, T, dx)
    g = np.zeros(n)
    g += w[n-1] * (1.0 - 2.0 * x)
    for k in range(1, n):
        wk = w[n-1+k]; wk2 = w[n-1-k]
        if wk < 1e-14 and wk2 < 1e-14:
            continue
        if wk >= 1e-14:
            g[:n-k] += wk * (1.0 - x[k:])
            g[k:] -= wk * x[:n-k]
        if wk2 >= 1e-14:
            g[k:] += wk2 * (1.0 - x[:n-k])
            g[:n-k] -= wk2 * x[k:]
    return g

def slsqp_opt(h, T, n, maxiter=2000):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(sfn, h, jac=sjac, args=(T, n), method='SLSQP',
                       bounds=bnds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def cstep(h, n, i):
    hs = h[i]; h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full').copy()
    h[i] = hs
    d = np.zeros(2*n-1)
    for k in range(1, n):
        ip, im = i+k, i-k
        if ip < n:
            d[n-1+k] += 1.0 - h[ip]; d[n-1-k] -= h[ip]
        if im >= 0:
            d[n-1+k] -= h[im]; d[n-1-k] += 1.0 - h[im]
    c0 = cb[n-1]
    dx = 2.0 / n
    idx0 = np.arange(2*n-1) == n-1
    res = minimize_scalar(lambda t: np.max(np.where(idx0, c0+t-t*t, cb+t*d)) * dx,
                          bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
    return res.x

def coord_desc(h, n, t_end, max_sw=80, rng=None):
    if rng is None:
        rng = np.random.default_rng(7)
    bh = h.copy(); bc5 = compute_c5(bh); stall = 0
    for _ in range(max_sw):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return bh, bc5
            t = cstep(bh, n, i)
            if abs(t - bh[i]) > 1e-13:
                bh[i] = t; bh = project(bh, n); improved = True
        c5 = compute_c5(bh)
        if c5 < bc5 - 1e-14:
            bc5 = c5; stall = 0
        else:
            if not improved:
                stall += 1
                if stall >= 4:
                    break
    return bh, bc5

def equalize(h, n, n_iter=300, t_end=None):
    bh = h.copy(); bc5 = compute_c5(bh)
    alphas = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    for _ in range(n_iter):
        if t_end and time.time() > t_end:
            break
        corr = np.correlate(bh, 1.0 - bh, mode='full')
        mx = np.max(corr)
        active = np.where(corr >= mx - 0.005)[0]
        if len(active) < 2:
            break
        g = np.zeros(n)
        for k in active:
            lag = k - (n-1)
            if lag == 0:
                g += 1.0 - 2.0 * bh
            elif lag > 0:
                g[:n-lag] += 1.0 - bh[lag:]; g[lag:] -= bh[:n-lag]
            else:
                l = -lag; g[l:] += 1.0 - bh[:n-l]; g[:n-l] -= bh[l:]
        g /= len(active)
        for a in alphas:
            ht = project(bh - a * g, n)
            c5 = compute_c5(ht)
            if c5 < bc5 - 1e-15:
                bh, bc5 = ht, c5; break
    return bh, bc5

def interp_h(h, n_new):
    xo = np.linspace(0, 2, len(h), endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return project(np.interp(xn, xo, h), n_new)
