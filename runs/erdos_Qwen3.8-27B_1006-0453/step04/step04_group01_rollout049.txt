import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_feasible(h, n):
    """Clip to [0,1] and adjust to sum = n/2."""
    target = n / 2.0
    vc = np.clip(h, 0.0, 1.0)
    diff = target - vc.sum()
    if abs(diff) < 1e-14:
        return vc
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(vc + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(vc + (lo + hi) * 0.5, 0.0, 1.0)

def coord_grad_vec(h, i):
    """Compute d_k for all lags k when changing h[i]. O(n)."""
    n = len(h)
    d = np.zeros(2 * n - 1)
    for k_range in range(-(n-1), n):
        idx1 = i - k_range  # for (1-h[i-k])
        idx2 = i + k_range  # for h[i+k]
        val = 0.0
        if 0 <= idx1 < n:
            val += (1.0 - h[idx1])
        if 0 <= idx2 < n:
            val -= h[idx2]
        d[k_range + n - 1] = val
    return d

def optimize_coord(corr, h, i, d, old_ai):
    """
    Find optimal h[i] given current corr values and gradient direction d.
    c_k(new) = corr[k] + (a - old_ai) * d[k]
    Minimize max_k c_k(new) * dx over a in [0,1].
    """
    n = len(h)
    dx = 2.0 / n
    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr + (m1 - old_ai) * d) * dx
        f2 = np.max(corr + (m2 - old_ai) * d) * dx
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5, np.max(corr + ((lo+hi)*0.5 - old_ai) * d) * dx

def coord_descent_fast(h, n, t_end, max_sweeps=10000, anneal_temp=0.005):
    """Fast coordinate descent with O(n) per step."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr_best = compute_corr_full(best_h)
    no_improve = 0
    sweep = 0
    while time.time() < t_end and sweep < max_sweeps:
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old_ai = best_h[i]
            d = coord_grad_vec(best_h, i)
            new_ai, new_c5 = optimize_coord(corr_best, best_h, i, d, old_ai)
            
            # Simulated annealing: accept worse with small probability
            accept = new_c5 < best_c5
            if not accept and anneal_temp > 0:
                delta = new_c5 - best_c5
                if np.random.rand() < np.exp(-delta / anneal_temp):
                    accept = True
            
            if accept and abs(new_ai - old_ai) > 1e-12:
                best_h[i] = new_ai
                # Update corr incrementally
                corr_best += (new_ai - old_ai) * d
                best_h = project_to_feasible(best_h, n)
                # Recompute corr after projection (projection may have changed other coords)
                corr_best = compute_corr_full(best_h)
                best_c5 = compute_c5(best_h)
                improved = True
                no_improve = 0
            elif not improved:
                no_improve += 1
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= 5:
            break
        sweep += 1
        # Cool down annealing
        anneal_temp = max(anneal_temp * 0.97, 0.001)
    return best_h, best_c5
