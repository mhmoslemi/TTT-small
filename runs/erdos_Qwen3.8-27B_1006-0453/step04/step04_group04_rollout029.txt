def coord_optimize_exact(h, n, j):
    """Optimize h[j] exactly given all other h fixed."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    old_h = h[j]
    
    # Compute B_k: coefficient of t in corr_new[k]
    m = len(corr)  # 2n-1
    B = np.zeros(m)
    for k in range(m):
        lag = k - (n - 1)
        bk = 0.0
        j1 = j + lag  # h[j] appears as first factor, (1-h[j+lag])
        if 0 <= j1 < n:
            bk += (1.0 - h[j1])
        j2 = j - lag  # h[j] appears as second factor, (1-h[j]) * h[j-lag]
        if 0 <= j2 < n:

        return np.max(A + t * B) * dx
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    return res.x

def exact_coord_descent(h, n, t_end, max_sweeps=500):
    h = h.copy()
    dx = 2.0 / n
    m = 2 * n - 1
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time.time() > t_end:
                return h, best_c5
            
            old = h[j]
            
            # Compute B
            B = np.zeros(m)
            for k in range(m):
                lag = k - (n - 1)
                bk = 0.0
                j1 = j + lag
                if 0 <= j1 < n:
                    bk += (1.0 - h[j1])
                j2 = j - lag
                if 0 <= j2 < n:
                    bk -= h[j2]
                B[k] = bk
            
            A = corr - old * B
            
            # Minimize max(A + t*B) over t in [0,1]
            # Convex 1D: use minimize_scalar
            def f(t):
                return np.max(A + t * B)
            
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-13})
            t_opt = res.x
            
            if abs(t_opt - old) > 1e-12:
                corr += (t_opt - old) * B
                h[j] = t_opt
                improved = True
        
        h = proj(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
        if not improved:
            break
    
    return h, best_c5

        hc = focused_slsqp(best_h, n, maxiter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Final coordinate descent
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = exact_coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=2000)

    best_h = proj(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
