def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_to_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        return np.clip(h + room * (diff / tot), 0.0, 1.0)
    else:
        if s < 1e-14:
            return h.copy()
        return np.clip(h * (1.0 + diff / s), 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_and_prob(vals, T):
    m = np.max(vals)
    shifted = np.clip(T * (vals - m), -500, 500)
    exps = np.exp(shifted)
    s = exps.sum()
    sm = m + np.log(s) / T
    p = exps / s
    return sm, p

def slsqp_smooth(h, T, maxiter=500, tol=1e-12):
    h = project_to_sum(h, len(h) / 2.0)
    n = len(h)
    dx = 2.0 / n
    km1 = n - 1
    
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = get_corr(x)
        sm, _ = smooth_max_and_prob(corr, T)
        return sm * dx
    
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        corr = get_corr(x)
        sm, p = smooth_max_and_prob(corr, T)
        # grad[i] = dx * sum_k p[k] * (dcorr[k]/dh[i])
        # dcorr[k]/dh[i] = (1-h[i+k-km1])*[valid] - h[i-k+km1]*[valid]
        grad = np.zeros(n)
        for i in range(n):

                j1 = i + k - km1

                j2 = i - k + km1
                if 0 <= j2 < n:
                    g -= pk * x[j2]
            grad[i] = dx * g
        return sm * dx, grad

def coord_descent(h_in, t_end, n_sweeps=5000):
    h = h_in.astype(float).copy()
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    km1 = n - 1
    corr = get_corr(h)
    best_c5 = float(np.max(corr) * dx)
    
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        any_improve = False
        for i in range(n):
            if time.time() > t_end:
                return h, best_c5
            h_old = h[i]
            lo = -h_old
            hi = 1.0 - h_old
            if hi - lo < 1e-15:
                continue

            coef = np.zeros(m)
            for k in range(-km1, n):
                c = 0.0
                j1 = i - k

                j2 = i + k
                if 0 <= j2 < n:
                    c -= h[j2]
                coef[k + km1] = c
            
            # Minimize max(corr + s*coef) over s in [lo, hi]

            f0 = np.max(corr)
            best_s = 0.0
            best_f = f0
            
            f_lo = np.max(corr + lo * coef)
            if f_lo < best_f:
                best_f, best_s = f_lo, lo
            f_hi = np.max(corr + hi * coef)
            if f_hi < best_f:
                best_f, best_s = f_hi, hi

                            f_c = np.max(corr + s_c * coef)
                            if f_c < best_f:
                                best_f, best_s = f_c, s_c
            
            if abs(best_s) > 1e-13:

                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    any_improve = True
        if not any_improve:
            break
