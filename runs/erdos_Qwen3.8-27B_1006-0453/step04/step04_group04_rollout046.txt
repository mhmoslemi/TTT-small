    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    val = (m + np.log(np.sum(exps)) / T) * dx
    
    # Gradient
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # h[i] * (1-h[i+lag]) for i in [0, n-1-lag)
            # d/dh[j] = (1-h[j+lag]) for j in [0, n-1-lag)
            #          + (-h[j-lag]) for j in [lag, n)
            if lag == 0:

            else:
                grad[:n-lag] += pk * (1.0 - h[lag:n])
                grad[lag:] += pk * (-h[:n-lag])
        else:
            nl = -lag
            if nl == 0:
                grad += pk * (1.0 - 2.0 * h)
            else:
                grad[nl:] += pk * (1.0 - h[:n-nl])
                grad[:n-nl] += pk * (-h[nl:])
    grad *= dx
    return val, grad

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

            if n - nl > 0:
                grad[nl:] += pk * (1.0 - h[:n - nl])
                grad[:n - nl] += pk * (-h[nl:])
    return grad * dx

def constrained_obj(x, T, n, lam=1e5):
    val = smooth_c5_val(x, T, n)
    s = x.sum() - n / 2.0
    return val + lam * s * s

def constrained_grad(x, T, n, lam=1e5):
    g = smooth_c5_grad(x, T, n)
    s = x.sum() - n / 2.0
    g += 2.0 * lam * s * np.ones(n)
    return g

def proj(h, n):

Actually, for coordinate descent, I need to evaluate the objective at a single changed coordinate. This is:
- Save h[i]
- Set h[i] = t
- Project (adjusts other coordinates to maintain sum)
- Compute C5

But projection changes other coordinates, which makes the 1D problem more complex. 

Alternative: Don't project in the individual step. Instead, just evaluate c5 for h with only h[i] changed, but then the sum constraint is violated.

Let me simplify: for coordinate descent, I'll just try setting h[i] to 0 or 1 (binary coordinate descent) plus a few intermediate values. This is fast and often effective for this type of problem.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

# Module-level state for objective functions (avoiding closures/lambdas)
_T_glob = 100.0
_n_glob = 180
_lam_glob = 1e5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _obj(x):
    T, n, lam = _T_glob, _n_glob, _lam_glob
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    log_s = np.log(np.sum(np.exp(shifted)))
    val = (m + log_s / T) * dx
    s = x.sum() - n / 2.0
    return val + lam * s * s

def _grad(x):
    T, n, lam = _T_glob, _n_glob, _lam_glob
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500, 500)
    exps = np.exp(shifted)
    tot = np.sum(exps)
    p = exps / tot
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += pk * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[:n - lag] += pk * (1.0 - x[lag:])
            grad[lag:] += pk * (-x[:n - lag])
        else:
            nl = -lag
            grad[nl:] += pk * (1.0 - x[:n - nl])
            grad[:n - nl] += pk * (-x[nl:])
    grad *= dx
    s = x.sum() - n / 2.0
    grad += 2.0 * lam * s
    return grad

def proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lbfgs_step(h, T, n, maxiter=300):

    _T_glob = T
    _n_glob = n
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9})
    return proj(np.clip(res.x, 0.0, 1.0), n)

    best_c5 = compute_c5(best_h)
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    t = 0
    T = T_init
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    stale = 0
    while time.time() < t_end:
        t += 1
        global _T_glob, _n_glob
        _T_glob = T
        _n_glob = n
        g = _grad(best_h)
        m_acc = beta1 * m_acc + (1 - beta1) * g
        v_acc = beta2 * v_acc + (1 - beta2) * g * g
        m_hat = m_acc / (1.0 - beta1 ** t)
        v_hat = v_acc / (1.0 - beta2 ** t)

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    vals_to_try = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_save = best_h[i]
            best_local = h_save
            best_local_c5 = best_c5
            h_try = best_h.copy()
            for v in vals_to_try:
                h_try[i] = v
                h_try = proj(h_try, n)
                c5_t = compute_c5(h_try)
                if c5_t < best_local_c5:
                    best_local_c5 = c5_t
                    best_local = v
            if best_local != h_save:
                best_h[i] = best_local
                best_h = proj(best_h, n)
                if best_local_c5 < best_c5:
                    best_c5 = best_local_c5
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def structured_perturb(h, n, strength):
    h_p = h.copy()
    r = np.random.random()
    if r < 0.4:
        # Flip a contiguous block
        start = np.random.randint(0, n - 5)
        length = np.random.randint(5, max(6, n // 4))
        end = min(start + length, n)
        h_p[start:end] = 1.0 - h_p[start:end]
    elif r < 0.7:
        # Add a smooth Gaussian bump
        center = np.random.randint(0, n)
        sigma = np.random.randint(5, max(6, n // 5))
        x = np.arange(n)
        bump = strength * np.exp(-0.5 * ((x - center) / float(sigma)) ** 2)
        h_p += bump * (np.random.sign(np.random.random(n)) - 0.5) * 2
    else:
        # Swap two segments
        a = np.random.randint(0, n - 10)
        b = a + np.random.randint(10, n - a)
        la = np.random.randint(5, min(15, n - a))
        lb = np.random.randint(5, min(15, n - b))
        tmp = h_p[a:a+la].copy()
        h_p[a:a+la] = h_p[b:b+lb][:la]
        h_p[b:b+lb][:la] = tmp
    return proj(np.clip(h_p, 0.0, 1.0), n)
