import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-15:
                break
            h = h + room * (diff / room.sum())
        else:
            if s < 1e-15:
                break
            h = h - h * (abs(diff) / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 5000:
        return m * dx
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        return smooth_c5(x, T)
    def coneq(x):
        return x.sum() - n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': coneq}]
    try:

                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(np.clip(res.x, 0, 1), n)

                return np.max(v) * dx
