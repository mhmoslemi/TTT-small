def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, n, t_end, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = best_h[i]
            # coeff[m] = d(corr[m])/d(h[i]) for m in [i, i+n-1]
            mm = np.arange(i, i + n)
            kk = mm - i
            coeff = np.zeros(2 * n - 1)
            coeff[mm] = 1.0 - 2.0 * best_h[kk]
            
            c0_max = np.max(corr)

                cn_max = c0_max
                # Only need to check if (v-old)*coeff[m] pushes corr[m] above c0_max
                # For efficiency, just compute max directly
                cn = corr + (v - old) * coeff
                return np.max(cn)
            
            # Quick check: try 0 and 1
            f0 = eval_v(0.0)
            f1 = eval_v(1.0)
            fold = eval_v(old)
            
            best_v, best_f = old, fold
            if f0 < best_f:
                best_v, best_f = 0.0, f0
            if f1 < best_f:
                best_v, best_f = 1.0, f1
            
            # If not at boundary, try minimize_scalar
            if 0.0 < best_v < 1.0:
                try:
                    res = minimize_scalar(eval_v, bounds=(0.0, 1.0), method='bounded',
                                          options={'xatol': 1e-10, 'maxiter': 30})
                    if res.fun < best_f:
                        best_v, best_f = res.x, res.fun
                except:
                    pass

                corr += (best_v - old) * coeff
                best_c5 = best_f * dx
                improved = True
        if not improved:
            break

    best_h = h.copy()
    corr = compute_corr(best_h)
    best_c5 = np.max(corr) * dx
    rng = np.random.default_rng(12345)

            break

        j = rng.integers(n)
        if i == j:
            continue

        di = best_h[i]
        dj = best_h[j]

        # coeff_i: effect of changing h[i]
        mm_i = np.arange(i, i + n)
        coeff_i = np.zeros(2 * n - 1)
        coeff_i[mm_i] = 1.0 - 2.0 * best_h[mm_i - i]

        coeff_j = np.zeros(2 * n - 1)
        coeff_j[mm_j] = 1.0 - 2.0 * best_h[mm_j - j]

def binary_snap(h, n):
    """Try snapping to binary and see if it helps."""
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Adjust to get exactly n/2 ones
    s = h_bin.sum()
    if s != n / 2:

                h_bin[idx] = 1.0
                s += 1
        else:

                h_bin[idx] = 0.0
                s -= 1

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)) + 1e-300)) / T
    return val * dx

def slsqp_opt(h0, T, n, maxiter=300):
    def obj(x):
        return smooth_c5(x, T)
    def con_fun(x):
        return x.sum() - n / 2.0

        res = minimize(obj, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project(np.clip(res.x, 0, 1), n)
    except:
        return h0.copy()

        temp = 0.003 * cool + 0.0003

            if c5v < best_c5:

    if work_c5 < best_c5:

    return best_h, best_c5

        mm_j = np.arange(j, j + n)
        coeff_j = np.zeros(2 * n - 1)
        coeff_j[mm_j] = 1.0 - 2.0 * best_h[mm_j - j]
        new_corr = corr + (dj - di) * coeff_i + (di - dj) * coeff_j
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5 - 1e-15:
            best_h[i], best_h[j] = dj, di
            corr = new_corr
            best_c5 = new_c5
    return best_h, best_c5

def binary_snap(h, n):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    s = h_bin.sum()
    target = n / 2
    if s != target:
        order = np.argsort(h)
        if s < target:
            for idx in order[::-1]:
                if s == target:
                    break
                h_bin[idx] = 1.0
                s += 1
        else:
            for idx in order:
                if s == target:
                    break
                h_bin[idx] = 0.0
                s -= 1
    return h_bin

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)) + 1e-300)) / T
    return val * dx

def slsqp_opt(h0, T, n, maxiter=300):
    res = minimize(lambda x: smooth_c5(x, T), h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(np.clip(res.x, 0, 1), n)

def basin_hopping(work_h, work_c5, n, t_end, max_iters=200, seed=777):
    rng = np.random.default_rng(seed)
    best_h, best_c5 = work_h.copy(), work_c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = max(0.08, 1.0 - 0.75 * it / max_iters)
        step = 0.03 + 0.15 * cool
        h_p = work_h + step * rng.standard_normal(n)
        h_p = project(h_p, n)
        t_local = min(time.time() + 3.0, t_end)
        hc, c5v = fast_coord_descent(h_p, n, t_local, max_sweeps=40)
        temp = 0.003 * cool + 0.0002
        if c5v < work_c5 or rng.random() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc.copy()
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5
    return best_h, best_c5

def interp_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[::2] = 1.0
    n = len(best_h)
    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate descent
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.08, max_sweeps=300)

    # Phase 2: Binary snap + optimize
    h_bin = binary_snap(best_h, n)
    hc, c5b = fast_coord_descent(h_bin, n, t0 + budget_s * 0.14, max_sweeps=150)
    if c5b < best_c5:
        best_h, best_c5 = hc, c5b

    # Phase 3: Heavy coordinate descent
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.35, max_sweeps=5000)

    # Phase 4: Swap search
    best_h, best_c5 = swap_search(best_h, n, t0 + budget_s * 0.48, max_iters=80000, seed=seed)
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.52, max_sweeps=200)

    # Phase 5: SLSQP smoothed
    for T in [50, 200, 800, 3000, 10000]:
        if time.time() > t0 + budget_s * 0.64:
            break
        try:
            hc = slsqp_opt(best_h, T, n, maxiter=400)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
            best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.64, max_sweeps=80)
        except Exception:
            pass

    # Phase 6: Basin hopping
    best_h, best_c5 = basin_hopping(best_h, best_c5, n, t0 + budget_s * 0.82, max_iters=200, seed=seed)

    # Phase 7: Final polish
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=1000)
    best_h, best_c5 = swap_search(best_h, n, t0 + budget_s * 0.94, max_iters=30000, seed=seed + 1)
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=300)

    # Phase 8: Try alternate resolutions
    if time.time() < t0 + budget_s * 0.94:
        for n_try in [120, 200, 240]:
            if time.time() > t0 + budget_s * 0.955:
                break
            h_tr = interp_h(best_h, n_try)
            h_tr, c5_tr = fast_coord_descent(h_tr, n_try, t0 + budget_s * 0.96, max_sweeps=100)
            if c5_tr < best_c5:
                best_h, best_c5, n = h_tr, c5_tr, n_try

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
