    # Negative shift (index n-1-s):
    #   h[i] appears as: h[i+s]*(1-h[i]) if i+s<n, and (1-h[i-s])*h[i] if i-s>=0

def golden_min(f, lo, hi, tol=1e-10, max_iter=100):
    """Minimize f on [lo, hi] using golden section search"""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):

            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= (s - target) / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * ((target - s) / tot)
            h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            rem = diff
            for idx in order:
                a = min(rem, room[idx])
                h[idx] += a
                rem -= a
                if rem < 1e-13:
                    break
        else:
            order = np.argsort(h)
            rem = -diff
            for idx in order:
                a = min(rem, h[idx])
                h[idx] -= a
                rem -= a
                if rem < 1e-13:
                    break
    return h

def lin_for_i(h, n, i):
    lin = np.zeros(2 * n - 1)
    for s in range(1, n):
        mp = s + n - 1
        mn = n - 1 - s
        cp = 0.0
        cn = 0.0
        if i + s < n:
            cp += 1.0 - h[i + s]
            cn -= h[i + s]
        if i - s >= 0:
            cp -= h[i - s]
            cn += 1.0 - h[i - s]
        lin[mp] += cp
        lin[mn] += cn
    lin[n - 1] = 1.0 - 2.0 * h[i]
    return lin

def f1d_val(corr, lin, n1, old, v):
    d = v - old
    if abs(d) < 1e-15:
        return float(np.max(corr))
    vals = corr + d * lin
    vals[n1] -= d * d
    return float(np.max(vals))

def opt_1d(corr, lin, n1, old, dx):
    top4 = np.argsort(corr)[-4:]
    cands = [0.0, 1.0, old]
    for ii in range(len(top4)):
        for jj in range(ii + 1, len(top4)):
            ki, kj = int(top4[ii]), int(top4[jj])
            bi, bj = lin[ki], lin[kj]
            if abs(bi - bj) > 1e-12:
                vc = (corr[kj] - corr[ki]) / (bi - bj) + old
                if 0.0 <= vc <= 1.0:
                    cands.append(float(vc))
    ti = int(np.argmax(corr))
    if abs(lin[ti]) > 1e-12:
        for st in (0.02, 0.05, 0.1, 0.2, 0.4):
            vt = old - st * np.sign(lin[ti])
            if 0.0 <= vt <= 1.0:
                cands.append(float(vt))
    best_v = old
    best_val = f1d_val(corr, lin, n1, old, old)
    for v in cands:
        val = f1d_val(corr, lin, n1, old, v)
        if val < best_val:
            best_val = val
            best_v = v
    if 0 < best_v < 1:
        half = 0.04
        for _ in range(7):
            lo = max(0.0, best_v - half)
            hi = min(1.0, best_v + half)
            mids = np.linspace(lo, hi, 7)
            for v in mids:
                val = f1d_val(corr, lin, n1, old, float(v))
                if val < best_val:
                    best_val = val
                    best_v = float(v)
            half *= 0.4
    return best_v, best_val * dx

def cd_refine(h, n, t_end, max_sweeps, seed):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    n1 = n - 1
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep < 4 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = lin_for_i(best_h, n, i)
            bv, bval = opt_1d(corr, lin, n1, old, dx)
            d = bv - old
            if abs(d) > 1e-12 and bval < best_c5 - 1e-13:
                best_h[i] = bv
                corr += d * lin
                corr[n1] -= d * d
                nc = float(np.max(corr) * dx)
                if nc < best_c5:
                    best_c5 = nc
                    improved = True
                else:
                    best_h[i] = old
                    corr -= d * lin
                    corr[n1] += d * d
        if not improved and sweep >= 8:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def anneal(h, n, t_end, n_iters, seed, T0, T1):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() > t_end:
            break
        frac = it / max(n_iters, 1)
        T = T0 * (T1 / T0) ** frac
        saved = cur_h.copy()
        r = rng.rand()
        step = 0.01 + 0.08 * (1.0 - frac)
        if r < 0.45:
            idx = rng.randint(n)
            cur_h[idx] += rng.randn() * step
            cur_h = np.clip(cur_h, 0, 1)
            cur_h = project_h(cur_h, n)
        elif r < 0.70:
            ii, jj = rng.choice(n, 2, replace=False)
            cur_h[ii], cur_h[jj] = cur_h[jj], cur_h[ii]
        elif r < 0.85:
            w = rng.randint(2, max(3, n // 12))
            st = rng.randint(0, n - w)
            cur_h[st:st + w] = 1.0 - cur_h[st:st + w]
            cur_h = project_h(cur_h, n)
        else:
            k = rng.randint(3, 8)
            sel = rng.choice(n, k, replace=False)
            cur_h[sel] += rng.randn(k) * step * 0.6
            cur_h = np.clip(cur_h, 0, 1)
            cur_h = project_h(cur_h, n)
        nc = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        nc5 = float(np.max(nc) * dx)
        old_c5 = compute_c5(saved)
        if nc5 < old_c5 or rng.rand() < np.exp(-(nc5 - old_c5) / max(T, 1e-6)):
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = cur_h.copy()
        else:
            cur_h = saved
    return best_h, best_c5
