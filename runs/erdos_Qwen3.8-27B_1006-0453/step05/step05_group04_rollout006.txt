    corr = np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):

def cd_one_sweep(h, n, dx, corr, c5_val, order):
    """One sweep of coordinate descent. Returns updated h, corr, c5."""
    best_c5 = c5_val
    for i in order:
        old = h[i]
        if old == 0.0 or old == 1.0:
            # Still try to move from boundary
            pass
        lin = np.zeros(2 * n - 1)

        for s in range(1, n):

            mn = n - 1 - s
            if s <= i:
                other_p = i - s
                lin[mp] += (1.0 - h[other_p])
            if i + s < n:
                other_m = i + s
                lin[mp] -= h[other_m]
                lin[mn] += (1.0 - h[other_m])
            if s <= i:
                lin[mn] -= h[i - s]

            val = corr + d * lin

            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.02 + 0.08 * cool
        mode = hop % 6
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = step * 2.5 * rng.randn(mask.sum())
        elif mode == 3:
            idx = rng.choice(n, size=max(5, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 1.5 * rng.randn(len(idx))
        elif mode == 4:
            blk = rng.randint(0, n - 12)
            pert = np.zeros(n)
            pert[blk:blk+12] = step * rng.randn(12)
        else:
            i1 = rng.randint(n)
            i2 = rng.randint(n)
            pert = np.zeros(n)
            pert[i1] = step * 2.0 * rng.randn()
            pert[i2] = step * 2.0 * rng.randn()
        hp = project_h(cur_h + pert, n)
        hopt, c5opt = coord_descent(hp, n, min(t_end, time.time() + 3.0 + 5.0*cool), 60, seed + hop)
        temp = 0.005 * cool + 0.0003
        if c5opt < cur_c5 or (c5opt > cur_c5 and rng.rand() < np.exp(-(c5opt - cur_c5) / temp)):
            cur_h, cur_c5 = hopt, c5opt
            if c5opt < best_c5:
                best_c5, best_h = c5opt, hopt.copy()
    return best_h, best_c5

def make_structural_initial(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n
    f1, f2 = rng.randint(3, 12), rng.randint(13, 40)
    ph1, ph2 = rng.uniform(0, 2*np.pi), rng.uniform(0, 2*np.pi)
    h = 0.5 + 0.35*np.sin(2*np.pi*f1*x/2 + ph1) + 0.15*np.sin(2*np.pi*f2*x/2 + ph2)
    return project_h(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: CD polish from start
    t1 = t0 + budget_s * 0.12
    best_h, best_c5 = coord_descent(best_h, n, t1, 500, seed)

    # Phase 2: Block optimization
    t2 = t0 + budget_s * 0.28
    h_b, c5_b = block_cd(best_h, n, t2, 10, 40)
    if c5_b < best_c5:
        best_h, best_c5 = h_b, c5_b

    # Phase 3: Subgradient
    t3 = t0 + budget_s * 0.38
    h_g, c5_g = subgrad_phase(best_h, n, t3, 1500)
    if c5_g < best_c5:
        best_h, best_c5 = h_g, c5_g
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.42, 400, seed + 1)

    # Phase 4: Basin hopping (main exploration)
    t4 = t0 + budget_s * 0.65
    best_h, best_c5 = basin_hop(best_h, n, t4, 350, seed + 10)

    # Phase 5: Block refine
    t5 = t0 + budget_s * 0.75
    h_b2, c5_b2 = block_cd(best_h, n, t5, 8, 50)
    if c5_b2 < best_c5:
        best_h, best_c5 = h_b2, c5_b2

    # Phase 6: More basin hops with different structure
    t6 = t0 + budget_s * 0.85
    best_h, best_c5 = basin_hop(best_h, n, t6, 200, seed + 100)

    # Phase 7: Try structural starts
    t7 = t0 + budget_s * 0.92
    for trial in range(3):
        if time.time() > t7:
            break
        hs = make_structural_initial(n, seed + 200 + trial * 7)
        hs, c5s = coord_descent(hs, n, min(t7, time.time() + 15), 80, seed + 300 + trial)
        if c5s < best_c5:
            best_h, best_c5 = hs, c5s

    # Phase 8: Final CD + subgrad + CD
    t8 = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, n, t8, 500, seed + 400)
    if time.time() < t0 + budget_s * 0.97:
        h_g2, c5_g2 = subgrad_phase(best_h, n, t0 + budget_s * 0.97, 500)
        if c5_g2 < best_c5:
            best_h, best_c5 = h_g2, c5_g2

    # Phase 9: Last polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, 300, seed + 500)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
