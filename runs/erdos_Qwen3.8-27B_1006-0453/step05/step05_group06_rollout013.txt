import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-15:
                h += room * (diff / tot)
        else:
            s = h.sum()
            if s > 1e-15:
                h *= (1.0 + diff / s)
        np.clip(h, 0.0, 1.0, out=h)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-10:
            if diff2 > 0:
                room = 1.0 - h
                if room.sum() > 1e-15:
                    h += room * (diff2 / room.sum())
            else:
                if h.sum() > 1e-15:
                    h *= (1.0 + diff2 / h.sum())
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_c(h, w):
    n = len(h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    grad = np.zeros(n)
    for m in range(m_full):
        wi = w[m]
        if wi < 1e-16:
            continue
        lg = lag_arr[m]
        if lg >= 0:
            grad[:n - lg] += wi * (1.0 - h[lg:])
            if lg > 0:
                grad[lg:] -= wi * h[:n - lg]
            else:
                grad[:] -= wi * h
        else:
            p = -lg
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] -= wi * h[p:]
    return grad

def obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= Tc.max()
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = _grad_c(h, w) * dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_round(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def _f(x):
        return obj_grad(x, T, lam)[0]
    def _g(x):
        return obj_grad(x, T, lam)[1]
    try:
        res = minimize(_f, x0, jac=_g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_h(res.x)
    except Exception:
        return x0

def subgrad_active(h, step, n_act):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[-n_act:]
    w = np.zeros(m_full)
    w[top] = 1.0 / n_act
    g = _grad_c(h, w)
    h_new = project_h(h - step * dx * g)
    return h_new

def fast_coord(h, t_end, max_sweeps):
    n = len(h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    lag_arr = np.arange(m_full) - (n - 1)
    best = h.copy()
    best_c5 = compute_c5(best)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            A = np.zeros(m_full)
            for m in range(m_full):
                lg = lag_arr[m]
                v = 0.0
                ip = i + lg
                if 0 <= ip < n:
                    v += 1.0 - best[ip]
                im = i - lg
                if 0 <= im < n:
                    v -= best[im]
                A[m] = v
            lo = -best[i]
            hi = 1.0 - best[i]
            if hi - lo < 1e-14:
                continue
            corr = np.correlate(best, 1.0 - best, mode='full')
            def _f1d(d):
                return np.max(corr + d * A)
            res = minimize_scalar(_f1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-11})
            d = res.x
            if abs(d) < 1e-12:
                continue
            old = best[i]
            best[i] = old + d
            best = project_h(best)
            c5 = compute_c5(best)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
            else:
                best[i] = old
                best = project_h(best)
        if not improved:
            break
    return best, best_c5

def block_optimize(best_h, n, n_blocks, t_end):
    best_c5 = compute_c5(best_h)
    for _ in range(50):
        if time.time() > t_end:
            break
        bounds_b = np.linspace(0, n, n_blocks + 1) + np.random.uniform(-0.3, 0.3, n_blocks - 1) * (n / n_blocks)
        bounds_b = np.clip(np.sort(bounds_b), 1, n - 1)
        bounds_b[0] = 0
        bounds_b[-1] = n
        vals = np.random.uniform(0.1, 0.9, n_blocks)
        for inner in range(200):
            if time.time() > t_end:
                break
            lengths = np.diff(bounds_b)
            h_tmp = np.zeros(n)
            for j in range(n_blocks):
                h_tmp[int(bounds_b[j]):int(bounds_b[j+1])] = vals[j]
            h_tmp = project_h(h_tmp)
            c5 = compute_c5(h_tmp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_tmp.copy()
            # Perturb
            vals += np.random.randn(n_blocks) * 0.05
            vals = np.clip(vals, 0, 1)
    return best_h, best_c5
