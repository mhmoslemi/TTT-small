            lin[im] -= h[i - s]
        if i + s < n:
            lin[ip] -= h[i + s]
            lin[im] += (1.0 - h[i + s])
    return lin

def solve_1d(corr, lin, lo, hi):
    n_idx = len(corr) // 2
    def f(d):
        v = corr + d * lin
        v[n_idx] -= d * d
        return float(np.max(v))
    if hi - lo < 1e-13:
        return 0.5 * (lo + hi)
    pts = np.linspace(lo, hi, 10)
    # Add parabola vertex
    pv = lin[n_idx] * 0.5
    if lo < pv < hi:
        pts = np.concatenate([pts, [pv]])
    best_d = 0.0
    best_f = f(0.0)
    for d in pts:
        dd = max(lo, min(hi, float(d)))
        fd = f(dd)
        if fd < best_f:
            best_f = fd
            best_d = dd
    span = (hi - lo) / 25.0
    lo2 = max(lo, best_d - span)
    hi2 = min(hi, best_d + span)
    if hi2 > lo2:
        pts2 = np.linspace(lo2, hi2, 15)
        for d in pts2:
            dd = float(d)
            fd = f(dd)
            if fd < best_f:
                best_f = fd
                best_d = dd
    return best_d

def coord_descent(h, n, t_end, max_sweeps=5000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    no_imp = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if sweep % 7 == 5:
            order = rng.permutation(n)
        elif sweep % 3 == 2:
            order = np.arange(n - 1, -1, -1)
        else:
            order = np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = compute_lin(best_h, i)
            d_new = solve_1d(corr, lin, -old, 1.0 - old)
            d = d_new - old
            if abs(d) < 1e-13:
                continue
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5 - 1e-14:
                best_h[i] = d_new
                best_c5 = c5_new
                improved = True
            else:
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            no_imp += 1
            if no_imp >= 4:
                break
        else:
            no_imp = 0
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def subgradient_step(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exponents = np.clip(T * (corr - m), -50.0, 50.0)
    p = np.exp(exponents)
    p /= p.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = p[k_idx]
        if w < 1e-14:
            continue
        s = k_idx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

def grad_descent(h, n, t_end, max_iters=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    hc = h.copy()
    mom = np.zeros(n)
    beta = 0.75
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = min(1.0, it / max(max_iters, 1))
        T = 30.0 * 50.0 ** frac
        g = subgradient_step(hc, n, T)
        mom = beta * mom + g
        gm = np.max(np.abs(mom))
        if gm < 1e-14:
            break
        alpha = 0.35 / gm
        hn = project_h(hc - alpha * mom, n)
        c5n = compute_c5(hn)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = hn.copy()
        hc = hn
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.01 + 0.05 * cool
        mode = hop % 6
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 10)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            blk = n // 4
            start = rng.randint(0, n - blk)
            pert = np.zeros(n)
            pert[start:start+blk] = step * 1.8 * rng.randn(blk)
        elif mode == 3:
            blk = rng.randint(n // 8, n // 3)
            start = rng.randint(0, max(1, n - blk))
            pert = np.zeros(n)
            pert[start:start+blk] = step * 1.5 * rng.randn(blk)
        elif mode == 4:
            idx = rng.choice(n, size=max(8, n // 6), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * rng.randn(len(idx))
        else:
            f1 = rng.randint(1, 6)
            f2 = rng.randint(7, 20)
            ph = rng.uniform(0, 2 * np.pi)
            ar = np.arange(n) / n
            pert = step * (np.sin(2*np.pi*f1*ar + ph) + 0.5*np.sin(2*np.pi*f2*ar))
        hp = project_h(cur_h + pert, n)
        hop_time = max(1.5, 6.0 * cool)
        h_opt, c5_opt = coord_descent(hp, n, min(t_end, time.time() + hop_time),
                                       max_sweeps=60, seed=seed + hop)
        temp = 0.003 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def downsample(h, n_old, n_new):
    if n_new == n_old:
        return h.copy()
    idx = np.round(np.linspace(0, n_old - 1, n_new)).astype(int)
    return h[idx]

def upsample(h, n_old, n_new):
    old = np.linspace(0, 2.0, n_old, endpoint=False)
    new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(new, old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent (50% of budget)
    t1 = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=8000, seed=seed)

    # Phase 2: Subgradient + coord polish (12%)
    t2 = t0 + budget_s * 0.62
    if time.time() < t2:
        hg, cg = grad_descent(best_h, n, t2 * 0.7, max_iters=1500)
        if cg < best_c5:
            best_h, best_c5 = hg, cg
        best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=2000, seed=seed + 1)

    # Phase 3: Basin hopping (18%)
    t3 = t0 + budget_s * 0.80
    if time.time() < t3:
        best_h, best_c5 = basin_hop(best_h, n, t3, n_hops=250, seed=seed + 10)

    # Phase 4: Final coordinate descent (8%)
    t4 = t0 + budget_s * 0.88
    if time.time() < t4:
        best_h, best_c5 = coord_descent(best_h, n, t4, max_sweeps=3000, seed=seed + 100)

    # Phase 5: Try n=200 (4%)
    t5 = t0 + budget_s * 0.92
    if time.time() < t5:
        h200 = upsample(best_h, n, 200)
        h200 = project_h(h200, 200)
        best_h200, best_c5_200 = coord_descent(h200, 200, t5, max_sweeps=1500, seed=seed + 200)
        if best_c5_200 < best_c5:
            best_h = best_h200
            best_c5 = best_c5_200
            n = 200

    # Phase 6: Final polish (8%)
    t6 = t0 + budget_s * 0.995
    if time.time() < t6:
        best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=2000, seed=seed + 300)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
