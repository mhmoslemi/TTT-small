def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h += room * (diff / tot)
    else:
        s = h.sum()
        if s < 1e-12:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    """Optimize h[i] given all others fixed. Returns optimal value."""
    old = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full')
    # coefficient: corr[t] = cb + t * coef
    lags = np.arange(-(n-1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coef = np.zeros(2*n - 1)
    coef[v1] = 1.0 - h[j1[v1]]
    coef[v2] -= h[j2[v2]]
    h[i] = old
    
    def obj(t):
        return np.max(cb + t * coef)
    
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 40})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):

            break
        sum_before = best_h.sum()
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            best_h[i] = coord_opt(best_h, n, i)
        best_h = project_h(best_h, n)
        c5 = compute_c5(best_h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter=40, base_temp=0.002):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5

        if time.time() > t_end:
            break

            cur_c5 = c5_opt
            if c5_opt < best_c5:

                best_c5 = c5_opt
