```python
def project_h(h):

    h = np.clip(h, 0.0, 1.0)

        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(h - lam, 0.0, 1.0)
    # Fix any residual
    diff = target - h.sum()

        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:

            h = np.clip(h, 0.0, 1.0)

def softmax_grad(h, T, lam=100.0):

    dx = 2.0 / n
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = dx * (mc + np.log(exps.sum()) / T)

    grad = np.zeros(n)
    lag_arr = np.arange(m) - n + 1

                grad[p:] += -wi * h[:n-p]

        return softmax_grad(x, T, lam)[1]
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= Tc.max()
    exps = np.exp(Tc)
    w = exps / exps.sum()
    val = dx * (mc + np.log(exps.sum()) / T) + lam * (h.sum() - n/2.0)**2
    return val, w, corr

def softmax_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    val, w, corr = softmax_obj(h, T, lam)
    grad = np.zeros(n)
    lag_arr = np.arange(m) - n + 1
    for idx in range(m):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lg = lag_arr[idx]
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n/2.0)
    return val, grad

    x0 = project_h(h0)
    def f(x):
        return softmax_obj(x, T, lam)[0]

    try:

                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return x0

                lin[idx] += (1.0 - best_h[off])
                j2 = i - idx + n - 1
```

**What if I try a fundamentally different approach?**

Idea: **Subgradient method with Polyak averaging**

The min-max problem: minimize F(h) = max_k f_k(h) subject to h ∈ S (feasible set)

Subgradient method:
- h_{t+1} = P_S(h_t - η_t * g_t) where g_t is a subgradient of F at h_t
- Use Polyak step: η_t = (F(h_t) - F*) / ||g_t||²  (but F* is unknown)
- Use diminishing step: η_t = c / sqrt(t)
- Return average: h_avg = (1/T) ∑_{t=1}^T h_t

For our problem, a subgradient at h is:
g = dx * ∂f_{k*}/∂h where k* = argmax_k f_k(h)

Or more generally, g = dx * ∑_k w_k * ∂f_k/∂h where w_k ≥ 0, ∑w_k = 1, w_k > 0 only for active k's.

The advantage of averaging: even if individual iterates oscillate, the average converges to a good point.

Let me implement this:


**Idea: Use the structure of the correlation to identify "bottleneck" pairs**

For a given shift k, corr[k] = ∑_i h[i](1-h[i+k]). The pairs (i, i+k) that contribute most to corr[k] are those where h[i] is large and h[i+k] is small (or vice versa for negative shifts).

To reduce corr[k], we want to "swap" values: make h[i] and h[i+k] more similar.

**Algorithm: Pair swapping for active shifts**
1. For each active shift k (top 5-10 by correlation):
   - Find the pair (i, i+k) that contributes most to corr[k]
   - Try to swap/adjust h[i] and h[i+k]
   - Check if this reduces the overall max correlation
