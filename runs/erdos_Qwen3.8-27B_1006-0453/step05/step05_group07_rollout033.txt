import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    # Project onto {0<=h<=1, sum(h)=n/2}
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    # Water-filling projection
    if s > tgt:
        # Need to decrease some values
        # Scale down values > 0
        pos = h > 0
        if pos.sum() == 0:
            return h
        scale = tgt / s
        h_new = h * scale
        h_new = np.clip(h_new, 0, 1)
        # If still not exact, adjust
        d = tgt - h_new.sum()
        if abs(d) > 1e-12:
            mid = (h_new > 0.05) & (h_new < 0.95)
            if mid.sum() > 0:
                h_new[mid] += d / mid.sum()
                h_new = np.clip(h_new, 0, 1)
        return h_new
    else:
        # Need to increase some values
        space = 1.0 - h
        t = space.sum()
        if t < 1e-13:
            return h
        h_new = h + space * ((tgt - s) / t)
        h_new = np.clip(h_new, 0, 1)
        d = tgt - h_new.sum()
        if abs(d) > 1e-12:
            mid = (h_new > 0.05) & (h_new < 0.95)
            if mid.sum() > 0:
                h_new[mid] += d / mid.sum()
                h_new = np.clip(h_new, 0, 1)
        return h_new

def coord_desc_fast(h, n, t_end, sweeps=200, n_candidates=100):
    """Fast coordinate descent with vectorized 1D subproblems."""
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    ts = np.linspace(0, 1, n_candidates)
    order = np.arange(n)
    
    for sweep in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            idx_i = order[i % n]
            old = best[idx_i]
            # Compute coefficient vector for changing h[idx_i]
            # coef[k] = (1 - h[idx_i+k-n+1]) * valid - h[idx_i-k+n-1] * valid
            # Valid: 0 <= idx_i+k-n+1 < n for first term
            #        0 <= idx_i-k+n-1 < n for second term
            k_range = np.arange(mf)
            j1 = idx_i + k_range - n + 1  # first factor index
            j2 = idx_i - k_range + n - 1  # second factor index
            valid1 = (j1 >= 0) & (j1 < n)
            valid2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(mf)
            coef[valid1] += (1.0 - best[np.clip(j1, 0, n-1)][valid1])
            coef[valid2] -= best[np.clip(j2, 0, n-1)][valid2]
            
            # New correlations if h[idx_i] = t: corr + (t - old) * coef
            # Evaluate max over k for each candidate t
            # corr + (t - old) * coef = (corr - old * coef) + t * coef
            base = corr - old * coef
            # For each candidate t: max_k(base + t * coef)
            # Vectorized: ts[:,None] * coef[None,:] + base[None,:]
            vals = ts[:, None] * coef[None, :] + base[None, :]
            max_vals = vals.max(axis=1)
            best_t = ts[np.argmin(max_vals)]
            
            if abs(best_t - old) > 1e-10:
                trial = best.copy()
                trial[idx_i] = best_t
                trial = proj_h(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_c5 - 1e-14:
                    best = trial
                    best_c5 = c5t
                    corr = np.correlate(best, 1.0 - best, mode='full')
                    imp = True
        if not imp:
            break
        order = np.random.permutation(n)
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_T_box = [1.0]

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    d = tgt - h.sum()
    if abs(d) < 1e-14:
        return h
    interior = (h > 0.01) & (h < 0.99)
    nm = interior.sum()
    if nm == 0:
        interior = np.ones(n, dtype=bool)
        nm = n
    h[interior] += d / nm
    h = np.clip(h, 0.0, 1.0)
    d2 = tgt - h.sum()
    if abs(d2) > 1e-12:
        m2 = (h > 0.05) & (h < 0.95)
        if m2.sum() > 0:
            h[m2] += d2 / m2.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _compute_C(h, n):
    mf = 2 * n - 1
    i = np.arange(n)[:, None]
    k = np.arange(mf)[None, :]
    j1 = i + k - n + 1
    j2 = i - k + n - 1
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    hj1 = np.where(v1, h[np.clip(j1, 0, n)], 0.0)
    hj2 = np.where(v2, h[np.clip(j2, 0, n)], 0.0)
    C = (1.0 - hj1) * v1 - hj2 * v2
    return C, v1, v2, j1, j2

def _sobj(x):
    T = _T_box[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n

def _sgrad(x):
    T = _T_box[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    C, _, _, _, _ = _compute_C(x, n)
    return (2.0 / n) * (C @ w)

def lbfgs_step(h, T, n, iters=300):
    _T_box[0] = T
    try:
        r = minimize(h, _sobj, _sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except:
        return h

def coord_desc_fast(h_in, n, t_end, sweeps=500):

    mf = 2 * n - 1
    n_cand = 64

            k = np.arange(mf)
            j1 = idx + k - n + 1
            j2 = idx - k + n - 1
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(mf)
            coef[v1] = 1.0 - best[j1[v1]]
            coef[v2] -= best[j2[v2]]
            base = corr - old * coef
            mat = ts[:, None] * coef[None, :] + base[None, :]
            bi = int(np.argmin(mat.max(axis=1)))

def binary_cd(h_bin, n, t_end, max_iter=200000):
    h = h_bin.copy()
    ones = np.sum(h)
    if ones != n // 2:
        if ones > n // 2:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:int(ones) - n//2]] = 0
        else:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:n//2 - int(ones)]] = 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    T_temp = 0.03
    for it in range(max_iter):
        if time.time() > t_end:
            break
        # Pair swap: pick i with h=1, j with h=0
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        h[i], h[j] = h[j], h[i]
        nc = np.correlate(h, 1.0 - h, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T_temp):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T_temp = max(0.0003, T_temp * 0.9999)
    return best_h, best_c5
