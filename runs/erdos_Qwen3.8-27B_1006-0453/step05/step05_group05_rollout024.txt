```python
    grad = np.zeros(n)
    m_vals = np.arange(-(n-1), n)
    one_minus_h = 1.0 - h
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        L = m_vals[k]
        # (1-h) shifted by -L: contributes to positions i where 0<=i+L<n
        # i.e., i in [-L, n-1-L] if L<=0, or [0, n-1-L] if L>0
        start = max(0, -L)
        end = min(n, n - L)
        if end > start:
            grad[start:end] += w[k] * one_minus_h[start+L:end+L]
        # h shifted by L: contributes to positions i where 0<=i-L<n
        # i in [L, L+n-1], intersect [0,n-1]
        start2 = max(0, L)
        end2 = min(n, n + L)
        if end2 > start2:
            grad[start2:end2] -= w[k] * h[start2-L:end2-L]
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=2000, ftol=1e-15):
    def obj(x):
        return smooth_max_val(get_corr(x), T) * (2.0 / n)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def corr_coefs(h, n, i):
    coefs = np.zeros(2 * n - 1)
    m = np.arange(-(n - 1), n)
    v1 = i + m
    m1 = (v1 >= 0) & (v1 < n)
    coefs[m1] += (1.0 - h[v1[m1]])
    v2 = i - m
    m2 = (v2 >= 0) & (v2 < n)
    coefs[m2] -= h[v2[m2]]
    return coefs

def ternary_1d(corr0, co, niter=45):
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr0 + m1 * co) < np.max(corr0 + m2 * co):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=1000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = get_corr(best_h).copy()
    no_improve = 0
    while time.time() < t_end and no_improve < 5:
        corr = get_corr(best_h)
        improved = False
        order = np.random.permutation(n)
        for i in order:
            co = corr_coefs(best_h, n, i)
            corr0 = corr - best_h[i] * co
            v_new = ternary_1d(corr0, co)
            dv = v_new - best_h[i]
            if abs(dv) < 1e-13:
                continue
            best_h[i] = v_new
            corr += dv * co
            improved = True
        if improved:
            best_h = project_h(best_h)
            new_c5 = compute_c5(best_h)
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

_G_T = 100.0
_G_n = 160

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _obj(x):
    return smooth_max_val(get_corr(x), _G_T) * (2.0 / _G_n)

def _con(x):
    return x.sum() - _G_n / 2.0

def slsqp_opt(h, T, n, maxiter=2000, ftol=1e-15):
    global _G_T, _G_n
    _G_T, _G_n = T, n
    try:
        res = minimize(_obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def corr_coefs(h, n, i):
    coefs = np.zeros(2 * n - 1)
    m = np.arange(-(n - 1), n)
    v1 = i + m
    m1 = (v1 >= 0) & (v1 < n)
    coefs[m1] += (1.0 - h[v1[m1]])
    v2 = i - m
    m2 = (v2 >= 0) & (v2 < n)
    coefs[m2] -= h[v2[m2]]
    return coefs

def ternary_1d(corr0, co, niter=45):
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr0 + m1 * co) < np.max(corr0 + m2 * co):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

    best_c5 = compute_c5(best_h)
    no_improve = 0

        improved = False

        for i in order:
            co = corr_coefs(best_h, n, i)
            corr0 = corr - best_h[i] * co
            v_new = ternary_1d(corr0, co)
