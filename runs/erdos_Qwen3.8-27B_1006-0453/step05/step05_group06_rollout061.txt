    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_step(h, i, n):
    """Do one coordinate descent step on coordinate i."""
    old_hi = h[i]
    h[i] = 0.0
    base = np.correlate(h, 1.0 - h, mode='full')
    b = np.zeros(2 * n - 1)
    # Compute b vectorized
    sv = np.arange(1, n)
    tp = n - 1 + sv
    m1 = sv <= i
    if m1.any():
        b[tp[m1]] += (1.0 - h[i - sv[m1]])
    m2 = sv <= (n - 1 - i)
    if m2.any():
        b[tp[m2]] += (-h[i + sv[m2]])
    tn = n - 1 - sv
    m3 = sv <= (n - 1 - i)
    if m3.any():

    m4 = sv <= i
    if m4.any():
        b[tn[m4]] += (1.0 - h[i - sv[m4]])

    h[i] = old_hi

        v = base + x * b
        v[n - 1] -= x * x
        return np.max(v)
    
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-13})
    return res.x
