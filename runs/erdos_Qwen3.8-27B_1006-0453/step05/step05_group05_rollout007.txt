def smooth_obj_grad(h, T):
    """Compute smoothed objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    Tcorr = T * corr
    m = np.max(Tcorr)
    exps = np.exp(Tcorr - m)
    logsum = m + np.log(np.sum(exps))
    p = exps / np.sum(exps)  # softmax weights, length 2n-1
    obj = logsum / T * dx
    
    # Gradient: g[i] = dx * sum_l p[l] * dcorr[l]/dh[i]
    # dcorr[l]/dh[i] where l = 0,...,2n-2 corresponds to lag l-(n-1)
    lags = np.arange(2*n-1) - (n-1)  # -(n-1), ..., 0, ..., (n-1)

    # Build gradient efficiently
    g = np.zeros(n)
    for l in range(2*n-1):
        lag = lags[l]
        w = dx * p[l]
        if w < 1e-15:
            continue
        if lag >= 0:

            idx = np.arange(n - lag)
            g[idx] += w * (1.0 - h[idx + lag])
            # -h[i-lag] for i = lag, ..., n-1
            idx2 = np.arange(lag, n)

            idx2 = np.arange(n - lag_abs)
            g[idx2] += w * (-h[idx2 + lag_abs])

    m = np.max(Tcorr)

            g[ell:]   += w * (-h[:n-ell])             # -h[i-ell], i=ell..n-1

        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=20)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
