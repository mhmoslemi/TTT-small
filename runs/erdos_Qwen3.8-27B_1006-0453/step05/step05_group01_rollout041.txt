```python
def minimize_1d_linear_max(base, coefs, lo, hi):
    # f(t) = max(base + t*coefs), convex in t
    for _ in range(50):  # 50 iterations of ternary search gives ~1e-10 precision
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(base + m1 * coefs)
        f2 = np.max(base + m2 * coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2
```

```python
from scipy.optimize import minimize, minimize_scalar, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T, flat_pen=0.0, n_pen=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e8:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-n_pen:]
        val += flat_pen * np.std(top)
    return val * dx
```

```python
def project_h(h):

    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = corr.max()
    if T > 1e8:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    if bimode_pen > 0:
        val += bimode_pen * np.sum(h * (1 - h))
    return val * dx

                       bounds=[(0,1)]*n,
                       constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0}],
                       options={'maxiter':maxiter, 'ftol':ftol})
```

```python
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, T, bpen):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0-x, mode='full')
    m = corr.max()
    if T > 1e8:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(T*(corr-m))))/T
    if bpen > 0:
        val += bpen * np.sum(x*(1-x))
    return val * dx

                       constraints=[{'type':'eq','fun':lambda x: x.sum()-float(n)/2.0}],
                       options={'maxiter':maxiter,'ftol':ftol})
```

            ht = bh.copy()
            ht[i] = tn
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < bc5 - 1e-15:
                bc5 = c5t
                bh = ht
                corr = np.correlate(bh, 1.0-bh, mode='full')
                imp = True
        if not imp:
            break
    return bh, bc5
