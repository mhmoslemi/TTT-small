    max_c = np.max(corr)
    active = np.where(corr > max_c - max_c*tol)[0]

    grads = []
    for idx in active:
        m = idx - (n-1)
        g = np.zeros(n)
        for i in range(n):
            g_i = 0.0
            im = i - m
            ip = i + m
            if 0 <= im < n:
                g_i += (1.0 - h[im])
            if 0 <= ip < n:
                g_i += (-h[ip])
            g[i] = g_i

    d = -np.mean(grads, axis=0)

    # Actually, just project the step

        else: st = 0

def balance_lags(h, n, n_active=5, steps=30):
    """Try to equalize the top lags to reduce max."""
    bh = h.copy()
    for s in range(steps):
        corr = get_corr(bh)
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        c5 = np.max(corr)
        # Compute gradient w.r.t. each coordinate for the max
        # Use subgradient: average of active lag gradients
        g = np.zeros(n)
        for idx in active:
            m = idx - (n-1)
            for i in range(n):
                gm, gp = i-m, i+m
                gi = 0.0
                if 0 <= gm < n: gi += 1.0 - bh[gm]
                if 0 <= gp < n: gi += -bh[gp]
                g[i] += gi
        g /= len(active)
        # Projected gradient descent step
        lr = 0.01 * (1.0 - s/float(steps))
        nh = project_h(bh - lr*g)
        if compute_c5(nh) < compute_c5(bh):
            bh = nh
    return bh

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    tgt, s = n / 2.0, h.sum()
    d = tgt - s
    if abs(d) < 1e-15:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-15:
            return h
        h += r * (d / t)
    else:
        if s < 1e-15:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i, nit=45):
    hb = h.copy()
    hb[i] = 0.0
    cb = get_corr(hb)
    m = np.arange(-(n - 1), n)
    a, b = i - m, i + m
    va, vb = (a >= 0) & (a < n), (b >= 0) & (b < n)
    cf = np.zeros(2 * n - 1)
    cf[va] += 1.0 - h[a[va]]
    cf[vb] += -h[b[vb]]
    lo, hi = 0.0, 1.0
    for _ in range(nit):
        t1, t2 = lo + (hi - lo) / 3.0, hi - (hi - lo) / 3.0
        if np.max(cb + t1 * cf) < np.max(cb + t2 * cf):
            hi = t2
        else:
            lo = t1
    return (lo + hi) * 0.5

        else:

            im, ip = a_arr - m, a_arr + m
