def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    return np.real(np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(np.flip(1.0 - h), L)))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())

        return h

            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

        idx = np.random.choice(zeros, nh - n1, replace=False)
        h[idx] = 1
    return h

def compute_delta1(h, N, shifts):
    K = len(shifts)
    D = np.zeros((N, K))
    for k in range(K):
        s = int(shifts[k])
        if s == 0:

        elif s > 0:

            D[s:, k] += h[:N - s]
        else:
            s2 = -s
            D[s2:, k] = -(1.0 - h[:N - s2])
            D[:N - s2, k] += h[s2:]
    return D
