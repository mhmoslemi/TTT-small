def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=300, topk=32):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        k = min(topk, len(corr))
        top = np.partition(corr, -k)[-k:]
        m = top[0]
        sh = T * (top - m)
        max_exp = np.max(sh)
        if max_exp > 500:
            return (m + max_exp / T) * (2.0 / n)
        return (m + np.log(np.sum(np.exp(sh - max_exp)) + np.exp(-max_exp)) / T) * (2.0 / n)
    
    def cons_eq(x):
        return x.sum() - n / 2.0
    
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    for sweep in range(max_sweeps):

                coeff = np.zeros(2*n-1)
                j_plus = i + lags
                vp = (j_plus >= 0) & (j_plus < n)
                coeff[vp] = 1.0 - best_h[j_plus[vp]]
                j_minus = i - lags
                vm = (j_minus >= 0) & (j_minus < n)
                coeff[vm] -= best_h[j_minus[vm]]

                    nc[n-1] = corr[n-1] + v*(1.0-v) - c0_val
                    return np.max(nc)

                                      options={'xatol': 1e-10, 'maxiter': 25})

            if abs(delta) < 1e-10:
                continue

            best_h[i] = v_new
            sum_h += delta
            c0_val = old * (1.0 - old)
            corr += delta * coeff
            corr[n-1] += v_new*(1.0-v_new) - c0_val
            
            c5_try = float(np.max(corr) * 2.0 / n)
            # Note: sum_h might drift, but we'll fix at sweep end
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                improved = True
        
        # End of sweep: fix constraint
        diff = n/2.0 - sum_h
        if abs(diff) > 1e-10:
            best_h = project_h(best_h, n)
            sum_h = best_h.sum()
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_c5 = float(np.max(corr) * 2.0 / n)
        
        if not improved:
            break
    return best_h, best_c5

def thue_morse(n):
    # t(i) = parity of popcount(i)

For the Erdős problem, we want h to be "anti-correlated" with its shifts of (1-h). This is related to having low autocorrelation of h and low cross-correlation of h with 1-h.

The Thue-Morse sequence has the property that it's "balanced" - in every prefix, the number of 1s and 0s differ by at most 1. This might help.

But for the Erdős problem specifically, we need:
∫ h(x)(1-h(x+k)) dx to be small for ALL k.

This means h(x) and 1-h(x+k) should have low overlap for all shifts k.

If h is binary (0 or 1), then h(x)(1-h(x+k)) = 1 only when h(x)=1 and h(x+k)=0. So we want: for all k, the number of positions where h has a 1 followed k positions later by a 0 (or vice versa) to be small.

This is related to the "discrepancy" problem in combinatorics. The Thue-Morse and related sequences are designed to minimize such discrepancies.

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=300, topk=40):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        k = min(topk, len(corr))
        top = np.partition(corr, -k)[-k:]
        m = top[0]
        sh = T * (top - m)
        mx = np.max(sh)
        if mx > 500:
            return (m + mx / T) * dx
        return (m + np.log(np.sum(np.exp(sh - mx)) + np.exp(-mx)) / T) * dx
    def cons_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

                return np.max(nc)
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11, 'maxiter': 35})
            v_new = float(res.x)
            delta = v_new - old
            if abs(delta) < 1e-10:
                continue
            best_h[i] = v_new
            sum_h += delta
            corr += delta * coeff
            corr[n - 1] += v_new * (1.0 - v_new) - c0v

def perturb_struct(h, n, amp):
    hnew = h.copy()
    r = np.random.random()
    if r < 0.45:
        sl = int(np.random.choice([2, 3, 4, 6, 8, 12, 16, 24]))
        sl = min(sl, n - 2)
        st = np.random.randint(0, n - sl)
        sign = np.random.choice([-1, 1])
        hnew[st:st + sl] = np.clip(hnew[st:st + sl] + sign * amp * np.random.rand(sl), 0, 1)
    elif r < 0.75:

    else:
        p = np.random.randint(4, n // 3)
        hnew[:p] = hnew[p:2 * p]
        hnew[p:2 * p] = hnew[:p]
    return project_h(hnew, n)

def level_top(h, n, n_iter=12):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iter):
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-6:]
        for ki in top_idx[:4]:
            lag = ki - (n - 1)
            is_ = max(0, -lag)
            ie = min(n, n - lag) if lag > 0 else n
            if is_ >= ie:
                continue
            grads = np.zeros(n)
            for ii in range(is_, ie):
                g = (1.0 - best_h[ii + lag])
                if 0 <= ii - lag < n and lag != 0:
                    g -= best_h[ii - lag]
                grads[ii] = g
