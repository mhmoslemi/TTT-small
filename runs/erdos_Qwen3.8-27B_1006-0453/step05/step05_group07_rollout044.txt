import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj_h(h, n):
    tgt = n * 0.5
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(x, T):
    c = corr_of(x)
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def smooth_grad(x, T):
    n = len(x)
    c = corr_of(x)
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    mf = 2 * n - 1
    for k in range(mf):
        wk = w[k]
        if wk < 1e-15:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0 / n)

def lbfgs_step(h, T, n, iters=300):
    try:
        r = minimize(h, smooth_obj, smooth_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, args=(T,),
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-11})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=100, topk=8):
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = corr_of(best)
    mf = 2 * n - 1
    top_lags = np.argsort(-corr)[:topk]
    for _sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        scores = np.zeros(n)
        for k in top_lags:
            off = k - (n - 1)
            mas = np.abs(off)
            scores[:n - mas] += 1.0
            scores[mas:] += 1.0
        order = np.argsort(-scores)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            base = corr - old * coef
            def f1d(v):
                return float(np.max(base + v * coef))
            r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-14})
            nv = float(r.x)
            if abs(nv - old) < 1e-14:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr = corr_of(trial)
                if _sw % 8 == 7:
                    top_lags = np.argsort(-corr)[:topk]
                imp = True
        if not imp:
            break
    return best, best_c5

def binary_sa(h, n, t_end, max_it=250000):
    hb = (h > 0.5).astype(float)
    ones = np.where(hb == 1)[0]
    if len(ones) > n // 2:
        hb[np.random.choice(ones, len(ones) - n//2, replace=False)] = 0
    elif len(ones) < n // 2:
        zb = np.where(hb == 0)[0]
        hb[np.random.choice(zb, n//2 - len(ones), replace=False)] = 1
    dx = 2.0 / n
    corr = corr_of(hb)
    best_c5 = float(np.max(corr) * dx)
    best_h = hb.copy()
    T = 0.018
    for _ in range(max_it):
        if time.time() > t_end:
            break

        j = np.random.randint(n)
        if hb[i] == hb[j]:
            continue
        hb[i], hb[j] = hb[j], hb[i]
        nc = corr_of(hb)

            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = hb.copy()
        else:
            hb[i], hb[j] = hb[j], hb[i]
        T = max(0.0008, T * 0.99992)
    return best_h, best_c5
