import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_effect(h, N):
    L = 2 * N - 1
    center = N - 1
    pos = np.arange(N)[:, None]
    lag = (np.arange(L) - center)[None, :]
    idx1 = pos + lag
    v1 = (idx1 >= 0) & (idx1 < N)
    t1 = np.where(v1, 1.0 - h[np.clip(idx1, 0, N-1)], 0.0)
    idx2 = pos - lag
    v2 = (idx2 >= 0) & (idx2 < N)
    t2 = np.where(v2, h[np.clip(idx2, 0, N-1)], 0.0)
    return t1 - t2

def fix_count(h, nh):
    n1 = int((h > 0.5).sum())
    if n1 == nh:
        return (h > 0.5).astype(float)
    hb = (h > 0.5).astype(float)
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        rm = idx[np.random.choice(len(idx), n1-nh, replace=False)]
        hb[rm] = 0
    else:
        idx = np.where(hb == 0)[0]
        add = idx[np.random.choice(len(idx), nh-n1, replace=False)]
        hb[add] = 1
    return hb

def binary_anneal(h, N, deadline, T0=0.8, Tcool=0.996, max_sweeps=5000):
    h = fix_count(h, N // 2)
    best_c = float(np.max(fft_corr(h))) * (2.0/N)
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    T = T0
    stall = 0
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr)) * (2.0/N)
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        eff = compute_effect(h, N)
        # Try 1-3 random swaps
        for _ in range(2):
            if len(ones) == 0 or len(zeros) == 0:
                break
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            new_corr = corr - eff[p] + eff[q]
            new_P = float(np.max(new_corr)) * (2.0/N)
            delta = new_P - P
            if delta < 0 or np.random.random() < min(1.0, np.exp(-delta / max(T, 0.005))):
                h[p], h[q] = 0.0, 1.0
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                corr = new_corr
                P = new_P
        # Greedy best-improve swap
        if stall > 20:
            nsamp = min(len(ones), 120)
            p_s = np.random.choice(ones, nsamp, replace=False)
            best_d = P
            best_pq = None
            for p in p_s:
                base = corr - eff[p]
                cm = (base[None,:] + eff[zeros]).max(axis=1)
                mi = int(np.argmin(cm))
                if cm[mi] < best_d * N / 2.0:  # comparing in raw units
                    best_d = cm[mi]
                    best_pq = (p, zeros[mi])
            if best_pq is not None:
                h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                stall = 0
        T *= Tcool
    return best_h, best_c

def nesterov_poly(h, N, deadline, k=80, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        ct = corr[top]
        w = np.exp((ct - ct.max()) * 3.0)
        w /= w.sum()
        center = N - 1
        grad = np.zeros(N)
        for idx in range(len(top)):
            L = int(top[idx] - center)
            a = w[idx]
            if L == 0:
                grad += a * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += a * (1.0 - y[:N-L])
                grad[:N-L] -= a * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N-M] += a * (1.0 - y[M:])
                    grad[M:] -= a * y[:N-M]
        grad /= w.sum()
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.985
            beta = min(0.94, 0.85 + it * 0.0001)
    return best_h, best_pk * dx

def genetic_pop(h_best, N, deadline, pop_size=25, gens=80):
    """Simple genetic algorithm for binary strings."""
    nh = N // 2
    pop = np.zeros((pop_size, N))
    pop[0] = fix_count(h_best, nh)
    for i in range(1, pop_size):
        r = np.random.random(N)
        pop[i] = (r < 0.5).astype(float)
        # Fix count
        n1 = int(pop[i].sum())
        if n1 > nh:
            idx = np.where(pop[i]==1)[0]
            pop[i][idx[np.random.choice(len(idx), n1-nh, replace=False)]] = 0
        elif n1 < nh:
            idx = np.where(pop[i]==0)[0]
            pop[i][idx[np.random.choice(len(idx), nh-n1, replace=False)]] = 1
    # Evaluate
    scores = np.array([float(np.max(fft_corr(pop[i]))) * (2.0/N) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_c = scores[best_idx]
    best_h = pop[best_idx].copy()
    for g in range(gens):
        if time.time() > deadline:
            break
        # Select top half
        order = np.argsort(scores)
        parents = pop[order[:pop_size//2]]
        # Create offspring
        for i in range(pop_size//2, pop_size):
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]
            mask = np.random.random(N) < 0.5
            child = np.where(mask, p1, p2)
            # Mutate: flip a few bits
            n_mut = np.random.randint(2, 6)
            flip_idx = np.random.choice(N, n_mut, replace=False)
            child[flip_idx] = 1.0 - child[flip_idx]
            # Fix count
            child = fix_count(child, nh)
            pop[i] = child
            pop[i+pop_size//2] = pop[i] if i + pop_size//2 < pop_size else 0  # fill
        # Simpler: just replace bottom half
        for i in range(pop_size//2):
            p1 = pop[order[i]]
            p2 = pop[order[np.random.randint(pop_size//2)]]
            mask = np.random.random(N) < 0.5
            child = np.where(mask, p1, p2).copy()
            n_mut = np.random.randint(2, 5)
            fi = np.random.choice(N, n_mut, replace=False)
            child[fi] = 1.0 - child[fi]
            pop[pop_size//2 + i] = fix_count(child, nh)
        # Re-evaluate offspring
        for i in range(pop_size//2, pop_size):
            scores[i] = float(np.max(fft_corr(pop[i]))) * (2.0/N)
        new_best = int(np.argmin(scores))
        if scores[new_best] < best_c:
            best_c = scores[new_best]
            best_h = pop[new_best].copy()
    return best_h, best_c

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_effect(h, N):
    L = 2 * N - 1
    center = N - 1
    pos = np.arange(N)[:, None]
    lag = (np.arange(L) - center)[None, :]
    idx1 = pos + lag
    v1 = (idx1 >= 0) & (idx1 < N)
    t1 = np.where(v1, 1.0 - h[np.clip(idx1, 0, N - 1)], 0.0)
    idx2 = pos - lag
    v2 = (idx2 >= 0) & (idx2 < N)
    t2 = np.where(v2, h[np.clip(idx2, 0, N - 1)], 0.0)
    return t1 - t2

def fix_count(h, nh):
    hb = (h > 0.5).astype(float)
    n1 = int(hb.sum())
    if n1 == nh:
        return hb
    if n1 > nh:
        idx = np.where(hb == 1)[0]
        rem = idx[np.random.choice(len(idx), n1 - nh, replace=False)]
        hb[rem] = 0
    else:
        idx = np.where(hb == 0)[0]
        add = idx[np.random.choice(len(idx), nh - n1, replace=False)]
        hb[add] = 1
    return hb

def binary_anneal(h, N, deadline, T0=0.8, Tcool=0.996, max_sweeps=10000):
    h = fix_count(h, N // 2)
    best_c = float(np.max(fft_corr(h))) * (2.0 / N)
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    T = T0
    stall = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(np.max(corr))
        Pn = P * (2.0 / N)
        if Pn < best_c - 1e-13:
            best_c = Pn
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        eff = compute_effect(h, N)
        did_swap = False
        for _ in range(3):
            if len(ones) == 0 or len(zeros) == 0:
                break
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            nc = corr - eff[p] + eff[q]
            nP = float(np.max(nc))
            delta = nP - P
            if delta < 0 or np.random.random() < min(1.0, np.exp(-delta / max(T, 0.004))):
                h[p], h[q] = 0.0, 1.0
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                corr, P = nc, nP
                did_swap = True
        if stall > 15 and len(ones) > 10:
            ns = min(len(ones), 140)
            ps = np.random.choice(ones, ns, replace=False)
            bd = P
            bpq = None
            for p in ps:

                cm = (base[None, :] + eff[zeros]).max(axis=1)
                mi = int(np.argmin(cm))
                if cm[mi] < bd:

                    bpq = (int(p), int(zeros[mi]))
            if bpq is not None:
                delta = bd - P
                if delta < 0 or np.random.random() < min(1.0, np.exp(-delta / max(T, 0.004))):
                    h[bpq[0]], h[bpq[1]] = 0.0, 1.0
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    did_swap = True
                    stall = 0
        T *= Tcool
    return best_h, best_c

def nesterov_soft(h, N, deadline, k=80, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    it = 0
    center = N - 1
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        ct = corr[top]
        w = np.exp((ct - ct.max()) * 4.0)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones_like(ct) / k
            ws = 1.0
        else:
            w /= ws
        grad = np.zeros(N)
        for ii in range(len(top)):
            L = int(top[ii] - center)
            a = w[ii]
            if L == 0:
                grad += a * (1.0 - 2.0 * y)
            elif L > 0:
                grad[:N - L] += a * (1.0 - y[L:])
                grad[L:] -= a * y[:N - L]
            else:
                M = -L
                grad[M:] += a * (1.0 - y[:N - M])
                grad[:N - M] -= a * y[M:]
        h_new = project_sum(y - lr * grad / ws, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 400 == 0:
            lr *= 0.982
            beta = min(0.93, 0.85 + it * 0.0001)
    return best_h, best_pk * dx

def genetic_pop(h_best, N, deadline, pop_sz=30, gens=120):
    nh = N // 2
    pop = np.zeros((pop_sz, N))
    pop[0] = fix_count(h_best, nh)
    for i in range(1, pop_sz):
        r = np.random.random(N)
        pop[i] = fix_count(r, nh)
    scores = np.array([float(np.max(fft_corr(pop[i]))) * (2.0 / N) for i in range(pop_sz)])
    bi = int(np.argmin(scores))
    best_c = scores[bi]
    best_h = pop[bi].copy()
    half = pop_sz // 2
    for _ in range(gens):
        if time.time() > deadline:
            break
        order = np.argsort(scores)
        for i in range(half):
            p1 = pop[order[np.random.randint(half)]]
            p2 = pop[order[np.random.randint(half)]]
            mask = np.random.random(N) < 0.5
            child = np.where(mask, p1, p2).copy()
            nm = np.random.randint(2, 6)
            fi = np.random.choice(N, nm, replace=False)
            child[fi] = 1.0 - child[fi]
            pop[half + i] = fix_count(child, nh)
        for i in range(half, pop_sz):
            scores[i] = float(np.max(fft_corr(pop[i]))) * (2.0 / N)
        nb = int(np.argmin(scores))
        if scores[nb] < best_c:
            best_c = scores[nb]
            best_h = pop[nb].copy()
    return best_h, best_c
