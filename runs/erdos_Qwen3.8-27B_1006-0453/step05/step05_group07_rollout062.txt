```python
        s = h.sum()
        if abs(s - tgt) < 1e-13:
            break
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        m = (h > 0.005) & (h < 0.995)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def softobj(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    if T < 1e7:
        val = m + np.log(np.sum(np.exp(T * (c - m)))) / T
    else:
        val = m
    return val * 2.0 / n

def softgrad(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    mf = len(c)
    g = np.zeros(n)
    if T < 1e7:
        e = np.exp(T * (c - m))
        w = e / e.sum()
    else:
        w = np.zeros(mf)
        w[c == m] = 1.0
        w /= w.sum()
    for k in range(mf):
        wk = w[k]
        if wk < 1e-15:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - h[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * h[idx]
    return g * (2.0 / n)

def lbfgs(h, T, n, iters=400):
    try:
        r = minimize(h.copy(), softobj, softgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, args=(T,),
                     options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-12})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        corr = np.correlate(best, 1.0 - best, mode='full')
        for ii in range(n):
            if time.time() > t_end:
                return best, best_c5
            i = order[ii]
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            base = corr - old * coef
            # Minimize max(base + v*coef) over v in [0,1]
            # This is a 1D convex problem (max of linear functions)
            # Use ternary/golden search or scipy
            def f1d(v):
                return float(np.max(base + v * coef))
            r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-14})
            nv = float(r.x)
            if abs(nv - old) < 1e-12:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5
```

**Different idea: Use a penalty/augmented Lagrangian approach**

The problem is min max_k f_k(h) where f_k(h) = ∫ h(x)(1-h(x+k)) dx.

We can rewrite as: min_h,t t subject to f_k(h) <= t for all k.

The Lagrangian is: L(h, t, λ) = t + sum_k λ_k (f_k(h) - t) = (1 - sum λ_k) t + sum λ_k f_k(h)

At the optimum, sum λ_k = 1, so L reduces to sum λ_k f_k(h) = weighted average of the correlations.

The dual problem: max_λ inf_h sum λ_k f_k(h) subject to sum λ_k = 1, λ_k >= 0.

This is a min-max game. We can use a multiplicative weights / Follow-the-Leader approach:
1. Start with uniform λ
2. For fixed λ, minimize sum λ_k f_k(h) over h (this is a smooth optimization)
3. Update λ: λ_k ← λ_k * exp(f_k(h) / T) normalized

This is essentially what the log-sum-exp soft-max is doing, but the iterative pjl (multiplicative weights) approach might converge differently.

Actually, the log-sum-exp IS the soft-max of the max, and L-BFGS-B on the log-sum-exp should be equivalent to roughly the same thing. So this isn't really different.

**Another idea: Use subgradient methods on the true max**

The subdifferential of max_k f_k at h is the convex hull of the gradients of the active f_k's. We can use:
- Find all k where f_k is within ε of the max (active set)
- Average their gradients
- Take a projected gradient step

This avoids the T parameter and directly optimizes the max.

```python
def subgrad_max(h, n, eps=1e-4, lr=0.005, steps=500):
```

Actually, for each k, the gradient ∂f_k/∂h is a vector of length n. Computing the sum over active k's requires iterating over k. For n=160, there are 319 shifts, and for each we do O(n) work, so total O(n^2) per gradient step. That's fine.

Let me vectorize the gradient computation for a single k:

For shift m (where m = k - (n-1)):
g[j] = (1-h[j+m]) * 1[0<=j+m<n] - h[j-m] * 1[0<=j-m<n]

This can be computed as:
- First term: for j=0 to n-1, j+m ranges from m to n-1+m. Valid when 0<=j+m<n, i.e., max(0,-m) <= j <= min(n-1, n-1-m).
- Second term: for j=0 to n-1, j-m ranges from -m to n-1-m. Valid when 0<=j-m<n, i.e., max(0,m) <= j <= min(n-1, n-1+m).

```python
def grad_single_shift(h, n, m):
    g = np.zeros(n)
    # Term 1: (1-h[j+m]) for valid j
    j_lo = max(0, -m)
    j_hi = min(n, n - m)
    if j_lo < j_hi:
        g[j_lo:j_hi] += (1.0 - h[j_lo+m:j_hi+m])
    # Term 2: -h[j-m] for valid j
    j_lo2 = max(0, m)
    j_hi2 = min(n, n + m)
    if j_lo2 < j_hi2:
        g[j_lo2:j_hi2] -= h[j_lo2-m:j_hi2-m]
    return g
```
