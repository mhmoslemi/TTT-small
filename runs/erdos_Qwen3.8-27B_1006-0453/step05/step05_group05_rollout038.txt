```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-12:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (d / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def smoothed_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    se = e.sum()
    w = e / se
    smax = m + np.log(se) / T
    g = np.zeros(n)
    for s in range(-(n-1), n):
        wk = w[s + n - 1]
        if wk < 1e-12:
            continue
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            g[lo1:hi1] += wk * (1.0 - h[lo1+s:hi1+s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            g[lo2:hi2] -= wk * h[lo2-s:hi2-s]
    sh = h.sum()
    val = smax * dx + lam * (sh - n/2.0)**2
    grad = g * dx + 2.0 * lam * (sh - n/2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, n, maxiter=500):
    res = minimize(smoothed_obj_grad, h0, jac=True, method='L-BFGS-B',
                   bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    return project_h(np.clip(res.x, 0, 1))

def coord_opt(h, n, i, niter=50):
    hb = h.copy(); hb[i] = 0.0
    cb = np.correlate(hb, 1.0-hb, mode='full')
    mv = np.arange(-(n-1), n)
    v1 = i - mv; v2 = i + mv
    ok1 = (v1>=0)&(v1<n); ok2 = (v2>=0)&(v2<n)
    c = np.zeros(2*n-1)
    c[ok1] += (1.0 - h[np.clip(v1[ok1],0,n-1)])
    c[ok2] += (-h[np.clip(v2[ok2],0,n-1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi-lo)/3.0
        m2 = hi - (hi-lo)/3.0
        if np.max(cb + m1*c) < np.max(cb + m2*c):
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2.0

def coord_descent(h, n, t_end, max_sweeps=50):
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-12:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (d / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)


def smoothed_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    se = e.sum()
    w = e / se
    smax = m + np.log(se) / T
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        wk = w[s + n - 1]
        if wk < 1e-13:
            continue
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            g[lo1:hi1] += wk * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            g[lo2:hi2] -= wk * h[lo2 - s:hi2 - s]
    sh = h.sum()
    val = smax * dx + lam * (sh - n / 2.0) ** 2
    grad = g * dx + 2.0 * lam * (sh - n / 2.0)
    return val, grad


def lbfgs_opt(h0, T, lam, maxiter=400):
    n = len(h0)
    try:
        res = minimize(smoothed_obj_grad, h0, args=(T, lam), jac=True,
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return project_h(h0.copy())


def coord_opt(h, n, i, niter=50):
    hb = h.copy()
    hb[i] = 0.0
    cb = np.correlate(hb, 1.0 - hb, mode='full')
    mv = np.arange(-(n - 1), n)
    v1 = i - mv
    v2 = i + mv
    ok1 = (v1 >= 0) & (v1 < n)
    ok2 = (v2 >= 0) & (v2 < n)
    c = np.zeros(2 * n - 1)
    c[ok1] += (1.0 - h[np.clip(v1[ok1], 0, n - 1)])
    c[ok2] += (-h[np.clip(v2[ok2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(cb + m1 * c) < np.max(cb + m2 * c):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def coord_descent(h, n, t_end, max_sweeps=60):
    best = h.copy()
    best_c5 = compute_c5(best)
    for sw in range(max_sweeps):
        if time.time() > t_end:

            if c5v < best_c5 - 1e-15:

                best = ht
                imp = True
        if not imp:
            break
    return best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-12:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (d / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)


def smoothed_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    se = e.sum()
    w = e / se
    smax = m + np.log(se) / T
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        wk = w[s + n - 1]
        if wk < 1e-13:
            continue
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            g[lo1:hi1] += wk * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            g[lo2:hi2] -= wk * h[lo2 - s:hi2 - s]
    sh = h.sum()
    val = smax * dx + lam * (sh - n / 2.0) ** 2
    grad = g * dx + 2.0 * lam * (sh - n / 2.0)
    return val, grad


def lbfgs_opt(h0, T, lam, maxiter=400):
    n = len(h0)
    try:
        res = minimize(smoothed_obj_grad, h0, args=(T, lam), jac=True,
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return project_h(h0.copy())


def coord_opt(h, n, i, niter=50):
    hb = h.copy()
    hb[i] = 0.0
    cb = np.correlate(hb, 1.0 - hb, mode='full')
    mv = np.arange(-(n - 1), n)
    v1 = i - mv
    v2 = i + mv
    ok1 = (v1 >= 0) & (v1 < n)
    ok2 = (v2 >= 0) & (v2 < n)
    c = np.zeros(2 * n - 1)
    c[ok1] += (1.0 - h[np.clip(v1[ok1], 0, n - 1)])
    c[ok2] += (-h[np.clip(v2[ok2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(cb + m1 * c) < np.max(cb + m2 * c):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def coord_descent(h, n, t_end, max_sweeps=60):
    best = h.copy()
    best_c5 = compute_c5(best)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best, best_c5
            hv = coord_opt(best, n, i)
            if abs(hv - best[i]) < 1e-13:
                continue
            ht = best.copy()
            ht[i] = hv
            ht = project_h(ht)
            c5v = compute_c5(ht)
            if c5v < best_c5 - 1e-15:
                best_c5 = c5v
                best = ht
                imp = True
        if not imp:
            break
    return best, best_c5


def make_periodic(n, period, ones_per_period):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < ones_per_period:
            h[i] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_periodic(160, 8, 4)

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B multi-temperature annealing
    for T in [60.0, 20.0, 7.0, 3.0, 1.2, 0.5, 0.2, 0.08]:
        if time.time() - t0 > budget_s * 0.28:
            break
        hc = lbfgs_opt(best_h.copy(), T, 50.0, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Basin hopping with diverse perturbations
    w_h = best_h.copy()
    w_c5 = best_c5
    n_bh = 150
    for it in range(n_bh):
        if time.time() - t0 > budget_s * 0.58:
            break
        cool = max(0.03, 1.0 - 1.0 * it / n_bh)
        mod = it % 7
        if mod == 0:
            p = (0.02 + 0.07 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 6.28)
            p = (0.02 + 0.06 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            p = np.zeros(n)
            mask = np.random.random(n) < 0.2
            p[mask] = (0.05 + 0.10 * cool) * np.random.randn(int(mask.sum()))
        elif mod == 3:
            bl = max(2, n // 10)
            b = np.random.randint(0, max(1, n // bl))
            p = np.zeros(n)
            end = min(n, (b + 1) * bl)
            p[b * bl:end] = (0.03 + 0.07 * cool) * np.random.randn(end - b * bl)
        elif mod == 4:
            p = np.zeros(n)
            for _ in range(4):
                a, b2 = np.random.choice(n, 2, replace=False)
                d = (best_h[b2] - best_h[a]) * (0.3 + 0.7 * cool)
                p[a] += d
                p[b2] -= d
        elif mod == 5:
            p = (0.01 + 0.04 * cool) * np.random.randn(n)
        else:
            mask = np.random.random(n) < 0.08
            p = np.zeros(n)
            p[mask] = (0.06 + 0.12 * cool) * np.sign(np.random.randn(int(mask.sum())))

        hp = project_h(w_h + p)
        T_bh = 5.0 + 60.0 * cool
        hc = lbfgs_opt(hp, T_bh, 50.0, maxiter=250)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.00005
        if c5v < w_c5 or np.random.rand() < np.exp(-(c5v - w_c5) / temp):
            w_h, w_c5 = hc, c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if w_c5 < best_c5:
        best_c5, best_h = w_c5, w_h

    # Phase 3: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82, max_sweeps=60)

    # Phase 4: Low-T L-BFGS-B polish
    for T in [1.0, 0.3, 0.1]:
        if time.time() - t0 > budget_s * 0.90:
            break
        hc = lbfgs_opt(best_h.copy(), T, 50.0, maxiter=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Final coordinate descent
    if time.time() - t0 < budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=30)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
