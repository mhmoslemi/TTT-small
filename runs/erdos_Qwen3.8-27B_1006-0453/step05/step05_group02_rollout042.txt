        else:

def get_coefs(h, n, i):
    """Get coefficient vector for coordinate i in each correlation."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    coefs = np.zeros(2 * n - 1)
    k_range = np.arange(-(n-1), n)
    j1 = i + k_range  # for (1-h[i+k]) term
    j2 = i - k_range  # for -h[i-k] term
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n-1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n-1)])
    return corr_base, coefs

def eval_1d(hi, corr_base, coefs, dx):
    return float(np.max(corr_base + hi * coefs) * dx)

def coord_optimize(h, i, n):
    corr_base, coefs = get_coefs(h, n, i)
    dx = 2.0 / n
    res = minimize_scalar(eval_1d, args=(corr_base, coefs, dx),
                          bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-14})
    return float(res.x)

1. Do a "clean" coordinate descent cycle: for each i, optimize h[i] given all others fixed
2. After a full cycle, project once
3. Repeat

But the projection changes the sum, which invalidates the previous optimizations. So we need to be more careful.

Alternative: use "pair moves" that preserve the sum exactly:
- Pick two indices i, j
- Move δ from h[i] to h[j]: h[i] -= δ, h[j] += δ
- This preserves the sum exactly
- Optimize over δ (and the choice of i,j)

This avoids the projection issue entirely!

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        h += room * (diff / tot) if tot > 1e-14 else diff / n
    else:
        h *= (target / s) if s > 1e-14 else 0.5
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-14:
            continue
        k = ki - (n - 1)
        if k >= 0:
            grad[:n-k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n-k]
        else:
            a = -k
            grad[a:] += wi * (1.0 - h[:n-a])
            grad[:n-a] -= wi * h[a:]
    return grad * dx

def cons_eq(x, n):
    return x.sum() - n / 2.0

def cons_jac(x, n):
    return np.ones(n)

def slsqp_solve(h, T, n, maxiter, ftol):
    try:
        bnds = [(0.0, 1.0)] * n
        res = minimize(smooth_c5, h, args=(T,), jac=smooth_grad,
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': cons_eq,
                                      'jac': cons_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def pair_move(h, n, i, j, dx):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-14:
        return
    si, sj = h[i], h[j]
    def f(t):
        h[i] = t
        h[j] = s - t
        c = np.correlate(h, 1.0 - h, mode='full')
        return np.max(c) * dx
    try:
        r = minimize_scalar(f, bounds=(lo, hi), method='bounded',
                            options={'xatol': 1e-13})
        h[i] = r.x
        h[j] = s - r.x
    except Exception:
        h[i], h[j] = si, sj

def greedy_pair(h, n, dx, t_end, max_passes, rng):
    best_c5 = compute_c5(h)
    for _ in range(max_passes):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return h, best_c5
            for j in range(i + 1, n):
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-14:
                    continue
                pair_move(h, n, i, j, dx)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    improved = True
                else:
                    # revert
                    h[i], h[j] = np.clip(h[i], 0, 1), h[j]
                    # restore from saved
                    pass
        if not improved:
            break
    return h, best_c5

def sa_pair_search(h, n, dx, t_end, n_iter, T0, alpha, rng):
    c5_curr = compute_c5(h)
    best_in_sa = c5_curr
    best_snapshot = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-14:
            continue
        si, sj = h[i], h[j]
        def f(t):
            h[i] = t
            h[j] = s - t
            c = np.correlate(h, 1.0 - h, mode='full')
            return np.max(c) * dx
        try:
            r = minimize_scalar(f, bounds=(lo, hi), method='bounded',
                                options={'xatol': 1e-13})
            h[i] = r.x
            h[j] = s - r.x
        except Exception:
            h[i], h[j] = si, sj
            continue
        c5_new = compute_c5(h)
        cool = 1.0 - 0.85 * (it / n_iter)
        T_cur = T0 * (alpha ** it) * cool + 1e-7
        if c5_new < c5_curr or rng.rand() < np.exp(-(c5_new - c5_curr) / T_cur):
            c5_curr = c5_new
            if c5_new < best_in_sa:
                best_in_sa = c5_new
                best_snapshot = h.copy()
        else:
            h[i], h[j] = si, sj
    return best_snapshot, best_in_sa

def coord_descent(h, n, dx, t_end, max_sweeps):
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return h, best_c5
            h_base = h.copy()
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            k_range = np.arange(-(n-1), n)
            j1 = i + k_range
            j2 = i - k_range
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n-1)])
            coefs[v2] += (-h[np.clip(j2[v2], 0, n-1)])
            def f_eval(t):
                return float(np.max(corr_base + t * coefs) * dx)
            try:
                r = minimize_scalar(f_eval, bounds=(0.0, 1.0), method='bounded',
                                    options={'xatol': 1e-14})
                h_new = h.copy()
                h_new[i] = r.x
                h_new = project_h(h_new)
                c5_try = compute_c5(h_new)
            except Exception:
                continue
            if c5_try < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_try
                imp = True
        if not imp:
            break
    return h, best_c5

    t1 = t0 + budget_s * 0.18
    for T in [30, 100, 500, 2000, 10000, 50000, 200000, 800000]:
        if time.time() > t1:

        c5v = compute_c5(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Greedy pair local search
    t2 = t0 + budget_s * 0.32
    h, c5_local = greedy_pair(best_h.copy(), n, dx, t2, 200, rng)
    if c5_local < best_c5:
        best_c5 = c5_local
        best_h = h.copy()

    t3 = t0 + budget_s * 0.72

        h_sa, c5_sa = sa_pair_search(best_h.copy(), n, dx, t_seg_end,
