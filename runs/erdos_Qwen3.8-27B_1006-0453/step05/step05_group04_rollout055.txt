import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def f1d(v, old, c_base, lin, n, dx):
    d = v - old
    val = c_base + d * lin
    val[n - 1] -= d * d
    return float(np.max(val) * dx)

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 3 == 2 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return project_h(best_h, n), compute_c5(best_h)
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                if i <= n - 1 - s:
                    lin[mp] -= best_h[i + s]
                    lin[mn] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[mn] -= best_h[i - s]
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            res = minimize_scalar(f1d, args=(old, c_base, lin, n, dx),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

    blend = rng.uniform(0, 0.3)
    h = (1-blend)*h + blend*0.5*(h1+h2)
