import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

_CB = [np.zeros(1)]
_CC = [np.zeros(1)]

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h *= (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t > 1e-14:
            h += rm * ((tgt - s) / t)
        else:
            h *= (tgt / max(s, 1e-14))
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        idx = np.where((h > 0.001) & (h < 0.999))[0]
        if len(idx) > 0:
            h[idx] += d / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def _f1d(v):
    return float(np.max(_CB[0] + v * _CC[0]))

def coord_descent(h, n, t_end, max_sweeps=300):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            d_coeff = np.zeros(mf)
            ks = np.arange(mf)
            j1 = i - ks + n - 1
            m1 = (j1 >= 0) & (j1 < n)
            if m1.any():
                d_coeff[m1] += (1.0 - best[j1[m1]])
            j2 = i + ks - n + 1
            m2 = (j2 >= 0) & (j2 < n)
            if m2.any():
                d_coeff[m2] -= best[j2[m2]]
            _CB[0] = corr - old * d_coeff
            _CC[0] = d_coeff
            res = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            nv = float(res.x)
            if abs(nv - old) < 1e-12:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def binary_swap_search(h, n, t_end, max_iters=50000):
    """Simulated annealing with swap moves on binary h."""
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    ones_to_fix = n // 2
    if len(ones) > ones_to_fix:
        rm = np.random.choice(ones, len(ones) - ones_to_fix, replace=False)
        h_b[rm] = 0
    elif len(ones) < ones_to_fix:
        zeros = np.where(h_b == 0)[0]
        ch = np.random.choice(zeros, ones_to_fix - len(ones), replace=False)
        h_b[ch] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T = 0.002
    T_min = 0.0001
    for it in range(max_iters):
        if time.time() > t_end:
            break
        # Pick a random 1 and a random 0 to swap
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_b[i], h_b[j] = 0.0, 1.0
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0, 0.0
        T = max(T_min, T * 0.99999)
    return best_h, best_c5
