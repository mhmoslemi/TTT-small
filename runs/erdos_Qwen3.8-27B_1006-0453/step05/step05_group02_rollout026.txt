from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

                       options={'maxiter': maxiter, 'ftol': ftol})

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation (12% budget)
    t1 = t0 + budget_s * 0.12
    for T in [50, 200, 800, 3000, 12000, 50000, 200000, 800000]:
        if time.time() > t1:
            break
        hc = slsqp_step(best_h, T, n, maxiter=600, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Intensive coordinate descent (48% budget)
    t2 = t0 + budget_s * 0.60
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=500)

    # Phase 3: Pair descent (18% budget)
    t3 = t0 + budget_s * 0.78
    best_h, best_c5 = pair_descent(best_h, n, t3, n_pairs_per_sweep=60, max_sweeps=200)

    # Phase 4: Basin hopping with SLSQP (10% budget)
    t4 = t0 + budget_s * 0.88
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 25
    for it in range(n_bh):
        if time.time() > t4:
            break
        cool = 1.0 - 0.8 * (it / n_bh)
        pert = make_pert(n, cool, rng)
        h_pert = project_h(work_h + pert)
        T_bh = 500 + 15000 * cool
        hc = slsqp_step(h_pert, T_bh, n, maxiter=400, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 5: Final coordinate descent + SLSQP polish (10% budget)
    t5 = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=200)

    if time.time() < t0 + budget_s * 0.985:
        for T in [2000000, 8000000]:
            if time.time() > t0 + budget_s * 0.985:
                break
            hc = slsqp_step(best_h, T, n, maxiter=1500, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
