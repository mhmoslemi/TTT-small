def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:

        room = 1.0 - h
        tot = room.sum()

            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
