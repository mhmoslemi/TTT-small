import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_h(h):

    if abs(diff) < 1e-14:

            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    """Compute correlation and return top-k indices"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-k:][::-1]
    return corr, idx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr, top_idx = corr_top(best_h, n_top)
        top_shifts = top_idx - (n - 1)
        # Compute gradient coefficients for top shifts
        # For each position i, d(corr_k)/d(h[i]) ≈ (1-h[i+k]) - h[i-k]
        # But we only need it for the 1D minimization

            # new_corr_k = base_k + h[i] * coef_k  (approximately linear in h[i])

            # = corr_k + (t - h[i]) * (1-h[i+k]-h[i-k])
