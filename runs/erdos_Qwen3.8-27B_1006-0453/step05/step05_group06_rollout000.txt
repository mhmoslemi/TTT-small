def active_lag_step(h, te, n_steps=50):
    """Reduce the max by finding a direction that decreases all active lags."""
    n = len(h)
    dx = 2.0/n
    best = h.copy()
    bc = compute_c5(best)
    for _ in range(n_steps):
        if time.time() > te: break
        corr = np.correlate(best, 1-best, mode='full')
        mc = corr.max() * dx
        # Active lags: within 1% of max
        active = np.where(corr * dx >= mc * 0.999)[0]
        if len(active) == 0:
            active = np.where(corr == corr.max())[0]
        # Compute gradients for active lags
        grads = np.zeros((len(active), n))
        for ii, idx in enumerate(active):
            lg = idx - (n-1)
            g = np.zeros(n)
            if lg >= 0:
                g[lg:] += (1.0 - best[:n-lg])
                g[:n-lg] -= best[lg:]
            else:
                p = -lg
                g[:n-p] += (1.0 - best[p:])
                g[p:] -= best[:n-p]
            grads[ii] = g * dx
        # Find direction minimizing max of active lags (linearized)
        # Minimize max_i (g_i · d) subject to ||d|| <= 1, 0 <= best+d <= 1
        # Approximate: minimize average of active lags
        d = -grads.sum(axis=0)
        d = d / (np.linalg.norm(d) + 1e-15)
        # Try a small step
        for alpha in [0.01, 0.005, 0.001, 0.0005]:
            ht = project_h(best + alpha * d)
            c = compute_c5(ht)
            if c < bc:
                bc = c
                best = ht
                break
    return best, bc

            hp[si:si+bsz] = 1.0 - hp[si:si+bsz]
            hp = project_h(hp)
        T_bh = max(500, int(3000 * cool))
        hw = lbfgs(hp, T_bh, 50.0, 800, p5)
        cw = compute_c5(hw)
        temp = 0.0002 * cool + 0.00003
        if cw < wc or np.random.rand() < np.exp(-(cw - wc) / temp):
            wh, wc = hw, cw
            if cw < bc:
                bc, best = cw, hw

    # Phase 6: Active-lag refinement (10%)
    p6 = t0 + budget_s * 0.88
    best, bc = active_step(best, p6, nsteps=200)

    # Phase 7: Final polish
    if time.time() < te * 0.85:
        best, bc = block_cd(best, te * 0.85, bs=2, nsweeps=150)
    if time.time() < te * 0.95:
        hw = lbfgs(best, 500000, 100.0, 5000, te * 0.95)
        cw = compute_c5(hw)
        if cw < bc:
            bc, best = cw, hw
    if time.time() < te:
        best, bc = active_step(best, te, nsteps=100)
    if time.time() < te:
        hw = lbfgs(best, 1000000, 100.0, 3000, te)
        cw = compute_c5(hw)
        if cw < bc:
            bc, best = cw, hw

    best = project_h(best)
    bc = compute_c5(best)
    return best, bc, n
