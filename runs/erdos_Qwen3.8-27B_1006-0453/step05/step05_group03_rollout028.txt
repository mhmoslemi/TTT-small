        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -100, 100)
    obj = (m + np.log(np.sum(np.exp(sh))) / T) * dx
    return obj, corr, m

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -100, 100)
    exp_sh = np.exp(sh)

        m = np.max(corr)

        return (m + np.log(np.sum(np.exp(sh))) / T) * dx

        return x.sum() - n / 2.0
    try:

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],

def coord_opt(h, n, i, t_end=None):
    """Optimize coordinate i."""
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')

    coefs[v2] += (-h[j2[v2]])

        return np.max(corr_base + hi * coefs)
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -100, 100)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx, corr, m

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -100, 100)
    ew = np.exp(sh)
    w = ew / ew.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-12:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad[:] += wj * (1.0 - 2.0 * h)
            else:
                grad[lag:n] += wj * (1.0 - h[:n-lag])
                grad[:n-lag] += wj * (-h[lag:])
        else:
            p = -lag
            grad[:n-p] += wj * (1.0 - h[p:])
            grad[p:] += wj * (-h[:n-p])
    return grad * dx

def gd_step(h, T, lr, n):
    g = grad_smooth(h, T)
    obj0, _, _ = smooth_obj(h, T)
    for _ in range(15):
        h_new = project_h(h - lr * g, n)
        obj_new, _, _ = smooth_obj(h_new, T)
        if obj_new < obj0:
            return h_new, obj_new
        lr *= 0.5
    return h, obj0

def slsqp_polish(h, T, n, maxiter=250):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    lags = np.arange(-(n-1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[j1[v1]])
    coefs[v2] += (-h[j2[v2]])
    def f(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 30})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def top_lag_repair(h, n, t_end, max_rounds=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for rd in range(max_rounds):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_ids = np.argsort(corr)[-6:]
        for ki in top_ids:
            lag = ki - (n - 1)
            for _ in range(5):
                i = np.random.randint(n)

                h_try[i] = best_h[i] + 0.03 * np.random.choice([-1.0, 1.0])

                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
    return best_h, best_c5

def make_periodic(hp, n, d):
    h = np.tile(hp, n // d)
    return project_h(h, n)
