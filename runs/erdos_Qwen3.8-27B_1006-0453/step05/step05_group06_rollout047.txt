import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def thue_morse_h(n):
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ 1.0
    return t

def softmax_val_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= Tc.max()
    exps = np.exp(Tc)
    Z = exps.sum()
    w = exps / Z
    val = dx * (mc + np.log(Z) / T) + lam * (h.sum() - n / 2.0) ** 2
    m_full = 2 * n - 1
    grad = np.zeros(n)
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lg = idx - (n - 1)
        if lg == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        elif lg > 0:
            grad[lg:] += wi * (1.0 - h[:n - lg])
            grad[:n - lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] += -wi * h[p:]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol, t_end):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x):
        return softmax_val_grad(x, T, lam)[0]
    def grad(x):
        return softmax_val_grad(x, T, lam)[1]
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10, 'maxfun': maxiter * 3})
        return project_h(res.x)
    except Exception:
        return x0

def binary_hill_climb(h_start, max_rounds, trials_per_round, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    h_bin = np.where(h_start > 0.5, 1.0, 0.0)
    adj = n / 2.0 - h_bin.sum()
    if adj > 0:
        idx = np.argsort(h_start)[:int(adj)]
        h_bin[idx] = 1.0
    elif adj < 0:
        idx = np.argsort(-h_start)[:int(-adj)]
        h_bin[idx] = 0.0
    best_c5 = compute_c5(h_bin)
    for rd in range(max_rounds):
        improved = False
        for _ in range(trials_per_round):
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = compute_c5(h_bin)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            break
    return h_bin, best_c5

def coord_descent(h, t_end, max_sweeps, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                break
            old = best_h[i]
            h0_full = np.correlate(best_h, 1.0 - best_h, mode='full')
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            cb = h0_full.copy()
            lag_arr = np.arange(m_full) - (n - 1)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] += 1.0 - 2.0 * best_h[i]
                    quad[idx] -= 2.0
                elif lg > 0:
                    if i < n - lg:
                        lin[idx] += (1.0 - best_h[i + lg])
                    if i >= lg:
                        lin[idx] += -best_h[i - lg]
                else:
                    p = -lg
                    if i >= p:
                        lin[idx] += (1.0 - best_h[i - p])
                    if i < n - p:
                        lin[idx] += -best_h[i + p]
            cb_adj = cb - old * lin - old * old * quad
            def f1d(x):
                return np.max(cb_adj + x * lin + x * x * quad)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            hi_new = res.x
            if abs(hi_new - old) < 1e-13:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5
