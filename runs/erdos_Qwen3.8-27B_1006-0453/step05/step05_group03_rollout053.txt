```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def gradient_topk(h, n, k=8):
    """Compute gradient of sum of top-k correlations w.r.t. h."""
    corr = compute_corr(h)
    idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    lags = idx - (n - 1)
    for lag in lags:
        # d corr[lag] / dh[j] = (1-h[j+lag])*[valid] - h[j-lag]*[valid]
        if lag >= 0:
            valid_j = np.arange(0, n - lag)
            grad[valid_j] += (1.0 - h[lag:lag+len(valid_j)])
            valid_j2 = np.arange(lag, n)
            grad[valid_j2] += (-h[0:n-lag])
        else:
            L = -lag
            valid_j = np.arange(L, n)
            grad[valid_j] += (1.0 - h[0:n-L])
            valid_j2 = np.arange(0, n - L)
            grad[valid_j2] += (-h[L:L+len(valid_j2)])
    return grad, c5

def proj_subgrad(h, n, t_end, k=8, iters=2000):
    """Projected subgradient with adaptive step size."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = 0.05
    for it in range(iters):
        if time.time() > t_end:
            break
        grad, c5_cur = gradient_topk(best_h, n, k)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        # Adaptive step
        h_try = project_h(best_h - alpha * grad / gnorm * 0.1, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            alpha = min(alpha * 1.05, 0.5)
        else:
            alpha *= 0.7
        # Occasionally increase k
        if it % 100 == 0:
            k = min(8 + it // 100, 16)
    return best_h, best_c5

def coord_opt_fast(h, n, i):
    """Fast 1D coordinate optimization for index i."""
    corr = compute_corr(h)
    old = h[i]
    h[i] = 0.0
    corr_base = compute_corr(h)
    h[i] = old
    
    # Compute coefficients: corr_new = corr_base + h[i] * coef
    coefs = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        lag = k - (n - 1)
        j1 = i + lag
        j2 = i - lag
        c = 0.0
        if 0 <= j1 < n:
            c += (1.0 - h[j1])
        if 0 <= j2 < n:
            c += (-h[j2])
        coefs[k] = c
    
    # Find h[i] that minimizes max(corr_base + hi * coefs)
    # This is a 1D piecewise linear minimization
    def f(hi):
        return np.max(corr_base + hi * coefs)
    
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 40})
    return res.x

def coord_descent_fast(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_fast(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5
```

```python
def project_h(h, n):

    if abs(diff) < 1e-14:
        return h

        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)

def get_coefs(h, n, i):
    """Compute linear coefficients for changing h[i]."""
    lags = np.arange(-(n-1), n)
    i_minus_L = i - lags
    i_plus_L = i + lags
    valid1 = (i_minus_L >= 0) & (i_minus_L < n)
    valid2 = (i_plus_L >= 0) & (i_plus_L < n)
    coefs = np.zeros(2*n - 1)
    coefs[valid1] += (1.0 - h[i_minus_L[valid1]])
    coefs[valid2] += (-h[i_plus_L[valid2]])
    return coefs

    """Optimize h[i] given current correlations. Returns new value."""
    coefs = get_coefs(h, n, i)
```

For f(v) = max_k (b_k + a_k * v) where b, a are fixed vectors:
- f is convex, piecewise linear
- The breakpoints are at v where b_i + a_i*v = b_j + a_j*v, i.e., v = (b_j - b_i)/(a_i - a_j)
- The minimum is at the smallest v where the slope changes from negative to positive
- Or at v=0 or v=1 if the function is monotone

A practical approach:
1. Compute f(0) and f(1)
2. If f is decreasing at 0 (slope < 0 for all active k) and increasing at 1, the minimum is interior
3. Binary search for the point where slope changes sign

The slope of f at v (where differentiable) is a_k* where k is the active (max) index.
The subdifferential at v is [min_{k in active}(a_k), max_{k in active}(a_k)] where active = {k: b_k + a_k*v = max}.

So the minimum is at v* where min_active ≤ 0 ≤ max_active.

Binary search:
- If min_active > 0 at current v, decrease v (function is increasing)
- If max_active < 0 at current v, increase v (function is decreasing)
- Otherwise v is optimal

This converges in O(log(1/ε) * n) time, which is fast.

```python
def pwlin_min(b, a, lo=0.0, hi=1.0):
    """Minimize max_k(b_k + a_k*v) over v in [lo, hi]."""
    for _ in range(60):
        v = 0.5 * (lo + hi)
        vals = b + v * a
        idx_max = np.argmax(vals)
        # At the max, check subdifferential
        active_mask = np.isclose(vals, vals[idx_max], atol=1e-14)
        a_active = a[active_mask]
        a_min = a_active.min()
        a_max = a_active.max()
        if a_min > 1e-15:
            hi = v  # function increasing, go lower
        elif a_max < -1e-15:
            lo = v  # function decreasing, go higher
        else:
            break  # v is optimal
    return 0.5 * (lo + hi)
```

```python
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc

    # Phase 3: Subgradient (10%)
    t3_end = t0 + budget_s*0.26
    best_h, best_c5 = subgrad_loop(best_h, n, t3_end, iters=8000)

    # Phase 4: Coordinate descent (25%)
    t4_end = t0 + budget_s*0.51
    best_h, best_c5 = coord_descent(best_h, n, t4_end, max_sweeps=500)

    # Phase 5: Basin hopping (25%)
    t5_end = t0 + budget_s*0.76
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(80):
        if time.time() > t5_end:
            break
        cool = max(0.05, 1.0 - it/80.0)
        amp = 0.015 + 0.07*cool
        hp = project_h(work_h + amp*rng.randn(n), n)
        hc, cv = subgrad_loop(hp, n, time.time()+15, iters=200)
        hc, cv = coord_descent(hc, n, time.time()+40, max_sweeps=20)
        temp = 0.0004*cool + 0.00003
        if cv < work_c5 or rng.rand() < np.exp(-(cv-work_c5)/temp):
            work_h = hc
            work_c5 = cv
            if cv < best_c5:
                best_c5 = cv
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: More coordinate descent (12%)
    t6_end = t0 + budget_s*0.88
    best_h, best_c5 = coord_descent(best_h, n, t6_end, max_sweeps=500)

    # Phase 7: Fine perturbation + coord (8%)
    t7_end = t0 + budget_s*0.96
    for it in range(30):
        if time.time() > t7_end:
            break
        hp = project_h(best_h + 0.008*rng.randn(n), n)
        hc, cv = coord_descent(hp, n, time.time()+10, max_sweeps=8)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc

    # Phase 8: Final SLSQP + coord (4%)
    if time.time() < t0 + budget_s*0.98:
        hc = smooth_max_slsqp(best_h, n, 200000, maxiter=400)
        cv = compute_c5(hc)
        if cv < best_c5:
            best_c5 = cv
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t0+budget_s*0.99, max_sweeps=50)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
