The key optimization is: instead of recomputing the full correlation each time, we can compute c0 = full_correlation - contribution_of_h[i].

The full correlation with current h is `corr_full`. When we set h[i]=0:
- For each lag l, the change is:
  - l=0: -h[i]*(1-h[i]) → c0[n-1] = corr_full[n-1] - h[i]*(1-h[i])
  - l>0: remove h[i]*(1-h[i-l]) if i≥l and i≤n-1, and h[i+l]*(1-h[i]) if i+l≤n-1
    → c0[n-1+l] = corr_full[n-1+l] - h[i]*(1-h[i-l]) - h[i+l]*(1-h[i]) (with appropriate bounds)
  - l<0, p=-l: remove h[i]*(1-h[i-p]) if i≥p and i≤n-1, and h[i+p]*(1-h[i]) if i+p≤n-1
    → c0[n-1-p] = corr_full[n-1-p] - h[i]*(1-h[i-p]) - h[i+p]*(1-h[i]) (with appropriate bounds)

So c0 can be computed in O(n) instead of O(n²)! This is the key speedup.

The first part is the max of linear functions (convex in x).
The second part is a concave quadratic in x.

- The max of linear functions changes its "active" piece at breakpoints
- Between breakpoints, f is either a line or a quadratic (minus x²)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h + h * (diff / s)  # scale down
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x):
        return smooth_max_obj_grad(x, T, lam)[0]
    def grad(x):
        return smooth_max_obj_grad(x, T, lam)[1]
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12,
                                'maxls': 40, 'maxfun': 10000})
        return project_h(res.x)
    except Exception:
        return x0

def vec_coord_descent(h, t_end, max_sweeps, rng):
    n = len(h)
    m = 2 * n - 1
    best_h = project_h(h.copy())
    best_c5 = compute_c5(best_h)
    corr_full = np.correlate(best_h, 1.0 - best_h, mode='full')
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            vi = best_h[i]
            if vi < 1e-12:
                continue
            # Compute c0 and lin in O(n)
            c0 = corr_full.copy()
            lin = np.zeros(m)
            lin[n-1] = 1.0
            c0[n-1] -= vi * (1.0 - vi)
            
            m1 = min(i, n - 1)
            m2 = min(n - 1 - i, n - 1)
            
            if m1 > 0:
                seg1 = h[i-1:i-m1-1:-1]  # h[i-1], h[i-2], ..., h[i-m1]
                lin[n:n+m1] += (1.0 - seg1)
                c0[n:n+m1] -= vi * (1.0 - seg1)
                ks_neg1 = n - 2 - np.arange(m1)  # n-2, n-3, ..., n-1-m1
                lin[ks_neg1] += (1.0 - seg1)
                c0[ks_neg1] -= vi * (1.0 - seg1)
            
            if m2 > 0:
                seg2 = h[i+1:i+m2+1]  # h[i+1], ..., h[i+m2]
                lin[n:n+m2] += -seg2
                c0[n:n+m2] -= seg2 * (1.0 - vi)
                ks_neg2 = n - 2 - np.arange(m2)
                lin[ks_neg2] += -seg2
                c0[ks_neg2] -= seg2 * (1.0 - vi)
            
            # 1D minimax: f(x) = max(c0 + x*lin) with c0[n-1] += x*lin[n-1] - x^2
            # Simplify: f(x) = max(max_{k!=n-1}(c0[k]+x*lin[k]), c0[n-1]+x-x^2)
            
            # Use golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
            a, b = 0.0, 1.0
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            fc = c0.copy()
            fc += c_ * lin
            fc[n-1] -= c_ * c_
            fc = np.max(fc)
            fd = c0.copy()
            fd += d_ * lin
            fd[n-1] -= d_ * d_
            fd = np.max(fd)
            
            for _ in range(25):
                if fc < fd:
                    b = d_
                    d_ = c_
                    fd = fc
                    c_ = b - gr * (b - a)
                    fc = c0.copy(); fc += c_ * lin; fc[n-1] -= c_ * c_
                    fc = np.max(fc)
                else:
                    a = c_
                    c_ = d_
                    fc = fd
                    d_ = a + gr * (b - a)
                    fd = c0.copy(); fd += d_ * lin; fd[n-1] -= d_ * d_
                    fd = np.max(fd)
            
            x_new = 0.5 * (a + b)
            
            if abs(x_new - vi) < 1e-11:
                continue
            
            best_h[i] = x_new
            best_h = project_h(best_h)
            c5_new = compute_c5(best_h)
            
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
                corr_full = np.correlate(best_h, 1.0 - best_h, mode='full')
            else:
                best_h[i] = vi
                best_h = project_h(best_h)
        
        if not improved:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0.copy())
    def _obj(x):
        return smooth_max_obj_grad(x, T, lam)[0]
    def _grad(x):
        return smooth_max_obj_grad(x, T, lam)[1]
    try:
        res = minimize(_obj, x0, jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12,
                                'maxls': 40, 'maxfun': 10000})
        return project_h(res.x)
    except Exception:
        return x0

def vec_coord_descent(h, t_end, max_sweeps, rng):
    n = len(h)
    m = 2 * n - 1
    best_h = project_h(h.copy())
    best_c5 = compute_c5(best_h)
    corr_full = np.correlate(best_h, 1.0 - best_h, mode='full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            vi = best_h[i]
            if vi < 1e-12:
                continue
            c0 = corr_full.copy()
            lin = np.zeros(m)
            lin[n-1] = 1.0
            c0[n-1] -= vi * (1.0 - vi)
            m1 = min(i, n - 1)
            m2 = min(n - 1 - i, n - 1)
            if m1 > 0:
                seg1 = best_h[i-1:i-m1-1:-1]
                lin[n:n+m1] += (1.0 - seg1)
                c0[n:n+m1] -= vi * (1.0 - seg1)
                ks1 = n - 2 - np.arange(m1)
                lin[ks1] += (1.0 - seg1)
                c0[ks1] -= vi * (1.0 - seg1)
            if m2 > 0:
                seg2 = best_h[i+1:i+m2+1]
                lin[n:n+m2] += -seg2
                c0[n:n+m2] -= seg2 * (1.0 - vi)
                ks2 = n - 2 - np.arange(m2)
                lin[ks2] += -seg2
                c0[ks2] -= seg2 * (1.0 - vi)
            a, b = 0.0, 1.0
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            tmp = c0 + c_ * lin
            tmp = tmp.copy()
            tmp[n-1] -= c_*c_
            fc = np.max(tmp)
            tmp = c0 + d_ * lin
            tmp = tmp.copy()
            tmp[n-1] -= d_*d_
            fd = np.max(tmp)
            for _ in range(24):
                if fc < fd:
                    b = d_; d_ = c_; fd = fc
                    c_ = b - gr*(b-a)
                    tmp = c0 + c_*lin; tmp = tmp.copy(); tmp[n-1] -= c_*c_
                    fc = np.max(tmp)
                else:
                    a = c_; c_ = d_; fc = fd
                    d_ = a + gr*(b-a)
                    tmp = c0 + d_*lin; tmp = tmp.copy(); tmp[n-1] -= d_*d_
                    fd = np.max(tmp)
            x_new = 0.5*(a+b)
            if abs(x_new - vi) < 1e-11:
                continue
            best_h[i] = x_new
            best_h = project_h(best_h)
            c5_new = compute_c5(best_h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
                corr_full = np.correlate(best_h, 1.0-best_h, mode='full')
            else:
                best_h[i] = vi
                best_h = project_h(best_h)
        if not improved:
            break
    return best_h, best_c5
