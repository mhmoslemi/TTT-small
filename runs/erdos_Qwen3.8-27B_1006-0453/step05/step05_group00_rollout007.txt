def swap_delta_ones(h, N):
    """For each position p, compute the vector of changes to corr if h[p] 1->0."""
    # corr[k] = sum_i h[i+k-(N-1)] * (1-h[i])
    # Change from h[p]: 1->0:
    #   Term where h[p] is the "h" factor: i+k-(N-1)=p, i=p-k+N-1
    #     change: -(1-h[p-k+N-1]) for valid p-k+N-1
    #   Term where h[p] is in (1-h[i]): i=p
    #     change: +h[p+k-(N-1)] for valid p+k-(N-1) (since 1-h[p] goes from 0 to 1)
    # 
    # delta_p[k] = -(1-h[p-k+N-1]) * [0<=p-k+N-1<N] + h[p+k-(N-1)] * [0<=p+k-(N-1)<N]
    
    L = 2*N - 1  # length of corr
    delta = np.zeros((N, L))
    for p in range(N):
        # First term: -(1-h[p-k+N-1]) for k where 0<=p-k+N-1<N
        # p-k+N-1 >= 0 => k <= p+N-1
        # p-k+N-1 < N => k > p-1 => k >= p
        # So k ranges from p to min(p+N-1, L-1) = p to p+N-1 (since L-1=2N-2 and p<N)
        k_start = p
        k_end = min(p + N, L)
        i_vals = p - np.arange(k_start, k_end) + N - 1
        delta[p, k_start:k_end] -= (1 - h[i_vals])
        
        # Second term: +h[p+k-(N-1)] for k where 0<=p+k-(N-1)<N
        # p+k-N+1 >= 0 => k >= N-1-p
        # p+k-N+1 < N => k < 2N-1-p => k <= 2N-2-p = L-1-p
        k_start2 = max(0, N-1-p)
        k_end2 = min(L-1-p+1, L)
        if k_start2 < k_end2:
            i_vals2 = p + np.arange(k_start2, k_end2) - N + 1
            delta[p, k_start2:k_end2] += h[i_vals2]
    return delta

def swap_delta_zeros(h, N):
    """For each position q, compute the vector of changes to corr if h[q] 0->1."""
    # Change from h[q]: 0->1:
    #   Term where h[q] is the "h" factor: i+k-(N-1)=q, i=q-k+N-1
    #     change: +(1-h[q-k+N-1]) for valid i
    #   Term where h[q] is in (1-h[i]): i=q
    #     change: -h[q+k-(N-1)] for valid q+k-(N-1) (since 1-h[q] goes from 1 to 0)
    L = 2*N - 1
    delta = np.zeros((N, L))
    for q in range(N):
        k_start = q
        k_end = min(q + N, L)
        i_vals = q - np.arange(k_start, k_end) + N - 1
        delta[q, k_start:k_end] += (1 - h[i_vals])
        
        k_start2 = max(0, N-1-q)
        k_end2 = min(L-1-q+1, L)
        if k_start2 < k_end2:
            i_vals2 = q + np.arange(k_start2, k_end2) - N + 1
            delta[q, k_start2:k_end2] -= h[i_vals2]
    return delta

Now, let me also think about the continuous Nesterov approach. The previous code's Nesterov might not be as effective because:
1. The objective is non-smooth (max of many functions)
2. The feasible set is a hyperplane intersection with a hypercube

A better continuous approach might be:

Actually, for the continuous case, the objective is:
f(h) = max_k corr[k](h)

where corr[k](h) = sum_i h[i+k-(N-1)] * (1-h[i]) for valid indices.

This is a DC (difference of convex) function! corr[k] = sum_i h[i+k-(N-1)] - sum_i h[i+k-(N-1)]*h[i]

The first term is linear in h, the second is quadratic (and concave in each pair). So corr[k] is concave... wait no.

corr[k] = sum_i h[a_i] * (1 - h[b_i]) where a_i and b_i are the appropriate indices.
= sum_i h[a_i] - sum_i h[a_i]*h[b_i]

The first sum is linear. The second sum involves products of pairs. If a_i ≠ b_i, then h[a_i]*h[b_i] is not convex or concave in general (it's indefinite). If a_i = b_i, then h[a_i]² is convex.

So the overall function is not simply convex or concave. The max of these is even more complex.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_balanced_binary(h, N):
    nh = N // 2
    idx = np.argsort(-h)
    out = np.zeros(N)
    out[idx[:nh]] = 1.0
    return out

def compute_d1(h, N, p):
    """Delta in corr when h[p] goes from 1 to 0."""
    L = 2 * N - 1
    d = np.zeros(L)
    ks = p
    ke = min(p + N, L)
    iv = p - np.arange(ks, ke) + N - 1
    d[ks:ke] -= (1 - h[iv])
    ks2 = max(0, N - 1 - p)
    ke2 = min(L - p, L)
    if ks2 < ke2:
        iv2 = p + np.arange(ks2, ke2) - N + 1
        d[ks2:ke2] += h[iv2]
    return d

def compute_d0(h, N, q):
    """Delta in corr when h[q] goes from 0 to 1."""
    L = 2 * N - 1
    d = np.zeros(L)
    ks = q
    ke = min(q + N, L)
    iv = q - np.arange(ks, ke) + N - 1
    d[ks:ke] += (1 - h[iv])
    ks2 = max(0, N - 1 - q)
    ke2 = min(L - q, L)
    if ks2 < ke2:
        iv2 = q + np.arange(ks2, ke2) - N + 1
        d[ks2:ke2] -= h[iv2]
    return d

def binary_local_search(h, N, deadline, max_sweeps=1000, n_samp_ones=60, n_samp_zeros=60):
    h = h.astype(float).copy()
    nh = N // 2
    best_c = float(np.max(fft_corr(h)))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
        
        # Perturbation if stuck
        if no_improve > 50 and np.random.random() < 0.3:
            for _ in range(3):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h[p], h[q] = 0, 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_improve = 0
            continue
        
        # Sample
        n_so = min(n_samp_ones, len(ones))
        n_sz = min(n_samp_zeros, len(zeros))
        samp_ones = np.random.choice(ones, n_so, replace=False)
        samp_zeros = np.random.choice(zeros, n_sz, replace=False)
        
        best_new_P = P
        best_pq = None
        
        for p in samp_ones:
            d1 = compute_d1(h, N, p)
            base = corr + d1
            for q in samp_zeros:
                if q == p:
                    continue
                d0 = compute_d0(h, N, q)
                new_P = float(np.max(base + d0))
                if new_P < best_new_P:
                    best_new_P = new_P
                    best_pq = (int(p), int(q))
        
        if best_pq is None:
            continue
        
        h[best_pq[0]], h[best_pq[1]] = 0, 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return h, best_c

def nesterov_continuous(h, N, deadline, k=100, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Subgradient from top-k peaks
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            Lshift = int(ai - center)
            if Lshift == 0:
                grad += (1.0 - 2.0 * y)
            elif Lshift > 0:
                grad[Lshift:] += (1.0 - y[:N - Lshift])
                grad[:N - Lshift] -= y[Lshift:]
            else:
                M = -Lshift
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= k
        
        h_new = np.clip(y - lr * grad, 0, 1)
        # Project to sum constraint
        s = h_new.sum()
        if s != target:
            # Simple projection
            h_new = h_new - (s - target) / N
            h_new = np.clip(h_new, 0, 1)
            # Iterate a few times
            for _ in range(50):
                s = h_new.sum()
                if abs(s - target) < 1e-10:
                    break
                h_new = h_new - (s - target) / N
                h_new = np.clip(h_new, 0, 1)
        
        m = beta * (h_new - h)
        h = h_new
        lr *= 0.9995
        
    return best_h, best_pk * dx
