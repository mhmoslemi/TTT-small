import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

_SM_T = [1.0]
_SM_N = [1]
_SH = [np.zeros(1)]
_SC = [np.zeros(1)]
_TOPK = [[]]

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.05) & (h < 0.95)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _sm_obj(x):
    T = _SM_T[0]
    n = _SM_N[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if T < 2:
        return m * 2.0 / n
    return (m + np.log(np.sum(np.exp(np.clip(T * (c - m), -500, 500)))) / T) * 2.0 / n

def _sm_grad(x):
    T = _SM_T[0]
    n = _SM_N[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if T < 2:
        idx = np.argmax(c)
        return _single_shift_grad(x, idx, n)
    e = np.exp(np.clip(T * (c - m), -500, 500))
    w = e / e.sum()
    return _shift_grad(x, w, n)

def _single_shift_grad(x, idx, n):
    grad = np.zeros(n)
    lo1 = max(0, n - 1 - idx)
    hi1 = min(n, 2 * n - 1 - idx)
    if lo1 < hi1:
        ii = np.arange(lo1, hi1) + idx - n + 1
        grad[lo1:hi1] += 1.0 - x[ii]
    lo2 = max(0, idx - n + 1)
    hi2 = min(n, idx + 1)
    if lo2 < hi2:
        ii = np.arange(lo2, hi2) - idx + n - 1
        grad[lo2:hi2] -= x[ii]
    return grad * 2.0 / n

def _shift_grad(x, w, n):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            ii = np.arange(lo1, hi1) + k - n + 1
            grad[lo1:hi1] += w[k] * (1.0 - x[ii])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            ii = np.arange(lo2, hi2) - k + n - 1
            grad[lo2:hi2] -= w[k] * x[ii]
    return grad * 2.0 / n

def _topk_obj(x):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.mean(c[_TOPK[0]])) * 2.0 / n

def _topk_grad(x):
    n = len(x)
    grad = np.zeros(n)
    for k in _TOPK[0]:
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            ii = np.arange(lo1, hi1) + k - n + 1

            ii = np.arange(lo2, hi2) - k + n - 1
            grad[lo2:hi2] -= x[ii]

def lbfgs_opt(h, T, n, iters=300):

    _SM_N[0] = n
    try:
        r = minimize(h, _sm_obj, _sm_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

def topk_opt(h, n, iters=200):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_shifts = np.argsort(c)[-8:].tolist()
    _TOPK[0] = k_shifts
    try:
        r = minimize(h, _topk_obj, _topk_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

def _f1d(hi):
    return float(np.max(_SH[0] + hi * _SC[0]))

def coord_desc(h, n, t_end, sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            _SH[0] = corr - old * coef
            _SC[0] = coef
            r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-12})
            nv = float(r.x)
            if abs(nv - old) < 1e-13:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

        idx = np.random.choice(ones, len(ones) - n // 2, replace=False)
        h_b[idx] = 0
    elif len(ones) < n // 2:
        idx = np.random.choice(zeros, n // 2 - len(ones), replace=False)
        h_b[idx] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T = 0.03
    for it in range(max_iters):
        if time.time() > t_end:
            break
        found = False
        for _a in range(10):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if h_b[i] != h_b[j]:
                found = True
                break
        if not found:
            continue
        h_b[i], h_b[j] = h_b[j], h_b[i]
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 <= best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        if it % 8000 == 0:
            T = max(0.0002, T * 0.82)
    return best_h, best_c5

def multi_start_opt(h_start, n, t_start, t_end):
    h = proj_h(h_start, n)
    h = lbfgs_opt(h, 300.0, n, iters=200)
    h, c5 = coord_desc(h, n, min(t_end, time.time() + (t_end - t_start) * 0.3), sweeps=20)
    h, c5 = coord_desc(h, n, t_end, sweeps=30)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: T-annealed L-BFGS on parent
    for T in [50, 200, 800, 3000, 12000, 50000]:
        if time.time() - t0 > budget_s * 0.08:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.15, sweeps=80)

    # Phase 3: Top-k focused optimization
    htk = topk_opt(best_h, n, iters=300)
    c5tk = compute_c5(htk)
    if c5tk < best_c5:
        best_h, best_c5 = htk, c5tk
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.20, sweeps=30)

    # Phase 4: Binary SA
    h_bin, c5_bin = bin_sa(best_h, n, t0 + budget_s * 0.35)
    if c5_bin < best_c5:
        best_h = lbfgs_opt(h_bin, 3000.0, n, iters=400)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.42, sweeps=40)

    # Phase 5: Multi-start with diverse initializations
    starts = []
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = (i & 1) ^ tm[i >> 1] if i > 0 else 1.0
    starts.append(tm.copy())
    arr = np.zeros(n)
    arr[:n // 2] = 1.0
    np.random.shuffle(arr)
    starts.append(arr.copy())
    arr2 = np.zeros(n)
    arr2[::2] = 1.0
    np.random.permutation(arr2)
    starts.append(arr2.copy())
    x = np.arange(n) / n
    sg = 0.5 + 0.5 * np.sin(2 * np.pi * 5 * x + 0.3)
    starts.append((sg > 0.5).astype(float))

    for start in starts:
        if time.time() - t0 > budget_s * 0.52:
            break
        s = proj_h(start, n)
        s, c5s = multi_start_opt(s, n, t0, t0 + budget_s * 0.52)
        if c5s < best_c5:
            best_h, best_c5 = s, c5s

    # Phase 6: Basin hopping
    work = best_h.copy()
    work_c5 = compute_c5(work)
    n_iters = 120
    for it in range(n_iters):
        if time.time() - t0 > budget_s * 0.70:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 4
        if mod == 0:
            pert = (0.02 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(2, 25)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.02 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.25 + 0.35 * cool)
            pert = np.zeros(n)
            if mask.sum() > 0:
                pert[mask] = (0.05 + 0.2 * cool) * np.random.randn(mask.sum())
        else:
            pert = np.zeros(n)
            nfl = max(3, int(4 + 15 * cool))
            idx = np.random.choice(n, nfl, replace=False)
            pert[idx] = np.where(best_h[idx] > 0.5, -0.4, 0.4)
        hp = proj_h(work + pert, n)
        T_val = 500.0 + 10000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=200)
        ht, c5t = coord_desc(ht, n, min(t0 + budget_s * 0.70, time.time() + 10), sweeps=8)
        temp = 0.0005 * cool + 0.00002
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 7: Higher resolution
    for n_hi in [200, 240, 280]:
        if time.time() - t0 > budget_s * 0.84:
            break
        try:
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_hi, endpoint=False) + 1.0 / n_hi
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = proj_h(h_hi, n_hi)
            h_hi = lbfgs_opt(h_hi, 3000.0, n_hi, iters=300)
            h_hi, c5_hi = coord_desc(h_hi, n_hi, t0 + budget_s * 0.88, sweeps=25)
            if c5_hi < best_c5:
                best_h, best_c5, n = h_hi, c5_hi, n_hi
        except Exception:
            pass

    # Phase 8: Final polish - top-k + high-T LBFGS + coord desc
    if time.time() - t0 < budget_s * 0.92:
        htk2 = topk_opt(best_h, n, iters=300)
        c5hk = compute_c5(htk2)
        if c5hk < best_c5:
            best_h, best_c5 = htk2, c5hk
    if time.time() - t0 < budget_s * 0.95:
        best_h = lbfgs_opt(best_h, 200000.0, n, iters=500)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.98:
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.99, sweeps=50)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
