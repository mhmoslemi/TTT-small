def nesterov_opt(h, N, deadline, rng, k=100, lr0=0.02):

        # Softmax gradient
        top = np.argpartition(corr, -k)[-k:]
        w = np.zeros(len(corr))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h):
    N = len(h)
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def make_binary(h, N, rng):
    h = (h.astype(float) >= 0.5).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[nh:]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:nh - n1]]] = 1
    return h

def full_binary_search(h, N, deadline, rng, stuck_limit=300):
    h = make_binary(h, N, rng)
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    tabu_set = set()
    tabu_queue = []
    tabu_max = 30

    while time.time() < deadline:
        if no_improve > stuck_limit:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1

        D = compute_D(h, N)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        best_new_P = P
        best_p = -1
        best_q = -1

        # Full evaluation over all ones
        for ii in range(len(ones)):
            p = int(ones[ii])
            if p in tabu_set:
                continue
            base_p = c - D[p, :]
            vals = (base_p[None, :] - D[zeros, :]).max(axis=1)
            mi = int(np.argmin(vals))
            if vals[mi] < best_new_P:
                best_new_P = float(vals[mi])
                best_p = p
                best_q = int(zeros[mi])

        if best_p < 0:
            break

        h[best_p] = 0.0
        h[best_q] = 1.0

        tabu_queue.append(best_p)
        tabu_set.add(best_p)
        if len(tabu_queue) > tabu_max:
            old = tabu_queue.pop(0)
            tabu_set.discard(old)

    return h, best_c

def nesterov_opt(h, N, deadline, rng, k=80, lr0=0.018):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.89

    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        top = np.argpartition(corr, -k)[-k:]
        w = np.exp(corr[top] - corr[top].max())
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for j, ai in enumerate(top):
            L = int(ai - center)
            wi = w[j]
            if L == 0:
                grad += (1.0 - 2.0 * y) * wi
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L]) * wi
                grad[:N - L] -= y[L:] * wi
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:]) * wi
                    grad[M:] -= y[:N - M] * wi
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new

    return best_h, best_pk * dx
