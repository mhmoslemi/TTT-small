import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h.astype(np.float64), L)
    G = np.fft.fft(np.flip(1.0 - h.astype(np.float64)), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(np.float64) + hp * vp.astype(np.float64)

        n1 = int(h.sum())
        if n1 > nh:
            idx = np.where(h == 1)[0]
            idx = idx[np.random.choice(len(idx), n1-nh, replace=False)]
            h[idx] = 0
        elif n1 < nh:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.choice(len(idx), nh-n1, replace=False)]
            h[idx] = 1
    
    c = fft_corr(h)
    best_c = float(c.max())
    D = compute_D(h, N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_c - 1e-15:
            best_c = P

        for p in ones:

            h[p] = 0.0
            h[q] = 1.0

                h[q] = 0.0

                if not found:
                    break
            else:
                break

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h.astype(np.float64), L)
    G = np.fft.fft(np.flip(1.0 - h.astype(np.float64)), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(np.float64) + hp * vp.astype(np.float64)

def fix_count(h, N):
    h = (h > 0.5).astype(np.float64)
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            to_flip = idx[np.random.choice(len(idx), n1 - nh, replace=False)]
            h[to_flip] = 0
        else:
            idx = np.where(h == 0)[0]
            to_flip = idx[np.random.choice(len(idx), nh - n1, replace=False)]
            h[to_flip] = 1
    return h

def binary_search(h, N, deadline, max_sweeps=1000, T_anneal=0.0, n2s_trials=30):
    h = fix_count(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    D = compute_D(h, N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        best_new_P = P
        best_pq = None
        for p in ones:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P - 1e-14:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is not None:
            p, q = best_pq
            h[p] = 0.0; h[q] = 1.0
            c = fft_corr(h)
            D = compute_D(h, N)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            best_c = min(best_c, P)
        else:
            found = False
            if len(ones) >= 2 and len(zeros) >= 2:
                for _t in range(n2s_trials):
                    p1, p2 = np.random.choice(ones, 2, replace=False)
                    q1, q2 = np.random.choice(zeros, 2, replace=False)
                    h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
                    nc = fft_corr(h)
                    P2 = float(nc.max())
                    if P2 < P - 1e-14:
                        c = nc; D = compute_D(h, N)
                        ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                        best_c = min(best_c, P2)
                        found = True
                        break
                    else:
                        h[p1] = 1; h[p2] = 1; h[q1] = 0; h[q2] = 0
            if not found and T_anneal > 0.001:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p] = 0; h[q] = 1
                nc = fft_corr(h)
                P2 = float(nc.max())
                if P2 < P or np.random.random() < np.exp(-(P2 - P) / T_anneal):
                    c = nc; D = compute_D(h, N)
                    ones = np.where(h == 1)[0]; zeros = np.where(h == 0)[0]
                    best_c = min(best_c, P2)
                    T_anneal *= 0.995
                else:
                    h[p] = 1; h[q] = 0
            if not found and T_anneal <= 0.001:
                break
    return h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=60, lr0=0.02):
    h = h.astype(np.float64).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.96, 0.88 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)

def kronecker_h(N, golden):
    a = (1.0 + np.sqrt(5.0)) / 2.0
    frac = (np.arange(N, dtype=np.float64) * (a - 1.0)) % 1.0
    h = (frac < 0.5).astype(np.float64)
    return h
