```python
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        lo, hi = -3.0, 3.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp(h, T, n, maxiter=800):
    def obj(x):
        return smooth_obj(x, T)
    def con_f(x):
        return x.sum() - n / 2.0
    def con_j(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con_f, 'jac': con_j}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return proj_box_sum(np.clip(r.x, 0, 1), n / 2.0)
    except:
        return h

def fast_coord(h, n, t_end, max_sweeps=2000):
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    m_arr = np.arange(-(n-1), n)
    
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        any_imp = False
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            j1 = i + m_arr
            j2 = i - m_arr
            coef = np.zeros(2*n-1)
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coef[v1] += 1.0 - best[j1[v1]]
            coef[v2] -= best[j2[v2]]
            lo, hi = -best[i], 1.0 - best[i]
            if hi - lo < 1e-12:
                continue
            try:
                r = minimize_scalar(lambda d: np.max(corr + d * coef),
                                    bounds=(lo, hi), method='bounded',
                                    options={'xatol': 1e-10})
                d = float(r.x)
            except:
                continue
            if abs(d) < 1e-11:
                continue
            new_v = np.clip(best[i] + d, 0, 1)
            d = new_v - best[i]
            if abs(d) < 1e-12:
                continue
            best[i] = new_v
            corr += d * coef
            any_imp = True
        best = proj_box_sum(best, target)
        c5n = compute_c5(best)
        if c5n < best_c5 - 1e-13:
            best_c5 = c5n
            any_imp = True
        elif not any_imp:
            break
    return best, best_c5

def binary_local(h, n, t_end, max_iters=5000):
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    rng = np.random.RandomState(123)

        if time.time() > t_end:
            break

        j = sorted_idx[rng.randint(0, min(20, n))]

            continue
        # Try flipping i up and j down (or vice versa)

            h_try[i] += di
            h_try[j] += dj

def basin_hop(h, n, t_end, rng, n_iter=30):
    best = h.copy()
    best_c5 = compute_c5(h)
    work = h.copy()

            pert[st:st+bs] = (0.03 + 0.1 * cool) * rng.standard_normal(bs)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    for _ in range(30):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        lo, hi = -3.0, 3.0
        for _ in range(40):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0.0, 1.0).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    return h

def _smooth_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def _con_fun(x, n):
    return x.sum() - n * 0.5

def _con_jac(x, n):
    return np.ones(n)

def slsqp(h, T, n, maxiter=800):
    try:
        r = minimize(_smooth_obj, h, args=(T,), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return proj_box_sum(np.clip(r.x, 0, 1), n * 0.5)
    except Exception:
        return h

def _solve_minimax(corr, coef, lo, hi):
    for _ in range(50):
        mid = (lo + hi) * 0.5
        g = coef[np.argmax(corr + mid * coef)]
        if g > 1e-14:
            hi = mid
        elif g < -1e-14:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def fast_coord(h, n, t_end, max_sweeps=3000):
    target = n * 0.5
    best = h.copy()
    best_c5 = compute_c5(best)
    m_arr = np.arange(-(n - 1), n)
    no_improve = 0
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            j1 = i + m_arr
            j2 = i - m_arr
            coef = np.zeros(2 * n - 1)
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coef[v1] += 1.0 - best[j1[v1]]
            coef[v2] -= best[j2[v2]]
            lo = -best[i]
            hi = 1.0 - best[i]
            if hi - lo < 1e-12:
                continue
            d = _solve_minimax(corr, coef, lo, hi)
            if abs(d) < 1e-11:
                continue
            nv = best[i] + d
            if nv < 0.0:
                nv = 0.0
            elif nv > 1.0:
                nv = 1.0
            d = nv - best[i]
            if abs(d) < 1e-12:
                continue
            best[i] = nv
            corr += d * coef
            imp = True
        best = proj_box_sum(best, target)
        c5n = compute_c5(best)
        if c5n < best_c5 - 1e-13:
            best_c5 = c5n
            no_improve = 0
        else:
            no_improve += 1
        if not imp and no_improve > 5:
            break
        if no_improve > 200:
            break
    return best, best_c5

def binary_local(h, n, t_end, max_iters=20000, seed=77):
    target = n * 0.5
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for ni, nj in [(1.0, 0.0), (0.0, 1.0)]:
            h_try = best.copy()
            h_try[i] = ni
            h_try[j] = nj
            h_try = proj_box_sum(h_try, target)
            c5t = compute_c5(h_try)
            if c5t < best_c5 - 1e-12:
                best_c5 = c5t
                best = h_try
                break
    return best, best_c5

def basin_hop(h, n, t_end, rng, n_iter=50):
    best = h.copy()
    best_c5 = compute_c5(h)
    work = h.copy()
    work_c5 = compute_c5(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.01, 1.0 - 0.85 * it / n_iter)
        corr = np.correlate(work, 1.0 - work, mode='full')
        cm = corr.max()
        crit = np.where(corr > 0.92 * cm)[0]
        if len(crit) > 0 and rng.rand() < 0.4:
            kl = crit[rng.randint(len(crit))]
            lag = kl - (n - 1)
            freq = max(1, abs(lag))
            amp = 0.04 + 0.12 * cool
            pert = amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 6.28))
        else:
            bs = rng.randint(3, max(4, n // 8))
            st = rng.randint(0, n - bs)
            pert = np.zeros(n)
            pert[st:st + bs] = (0.03 + 0.1 * cool) * rng.standard_normal(bs)
        h_p = proj_box_sum(work + pert, n * 0.5)
        h_o = fast_coord(h_p, n, t_end, max_sweeps=60)
        c5v = compute_c5(h_o)
        temp = 0.001 * cool + 0.0002
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work = h_o
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best = h_o
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

        if time.time() - t0 > budget_s * 0.07:

        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Fast coordinate descent
    best_h, best_c5 = fast_coord(best_h, n, t0 + budget_s * 0.32, 5000)

    # Phase 3: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.55, rng, 45)

    # Phase 4: Fast coordinate descent again
    best_h, best_c5 = fast_coord(best_h, n, t0 + budget_s * 0.70, 3000)

    # Phase 5: Binary local search
    best_h, best_c5 = binary_local(best_h, n, t0 + budget_s * 0.80, 20000)

    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.89, rng, 25)

    # Phase 7: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.94:

        c5v = compute_c5(hc)

            best_c5 = c5v
            best_h = hc

    # Phase 8: Final coordinate descent
    best_h, best_c5 = fast_coord(best_h, n, t0 + budget_s * 0.99, 500)

    best_h = proj_box_sum(best_h, n * 0.5)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
