def to_binary(h, N):

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def corr_max(h):
    return float(np.max(fft_corr(h)))

def c5_from_max(h, cmax):
    return cmax * (2.0 / len(h))

def make_binary(h, N):
    nh = N // 2
    ranks = np.argsort(h)[::-1]  # descending
    hb = np.zeros(N)
    hb[ranks[:nh]] = 1.0
    return hb

def make_D_full(h, N):
    shifts = np.arange(2*N-1) - (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N-1)
    pp_c = np.clip(pp, 0, N-1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(np.float64) + hp * vp.astype(np.float64)
