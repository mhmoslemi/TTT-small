import numpy as np
from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        h = h * (target / s)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (target - s) / tot
    return np.clip(h, 0.0, 1.0)

def corr_grad_h(h, idx, n):
    """Gradient of corr[idx] with respect to h."""
    lg = idx - n + 1
    grad = np.zeros(n)
    if lg >= 0:
        if lg < n:
            grad[:n-lg] += (1.0 - h[lg:])
            grad[lg:] += (-h[:n-lg])
    else:
        p = -lg
        if p < n:
            grad[p:] += (1.0 - h[:n-p])
            grad[:n-p] += (-h[p:])
    return grad

def obj_soft(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - np.max(corr)
    exps = np.exp(T * shifted)
    sm = np.max(corr) + np.log(exps.sum()) / T
    return dx * sm + lam * (h.sum() - n/2.0)**2

    try:

    n = len(h0)
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full')

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        h = h * (target / s)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (target - s) / tot
    return np.clip(h, 0.0, 1.0)

def corr_grad_h(h, idx, n):
    lg = idx - n + 1
    grad = np.zeros(n)
    if lg >= 0:
        if lg < n:
            grad[:n - lg] += (1.0 - h[lg:])
            grad[lg:] += (-h[:n - lg])
    else:
        p = -lg
        if p < n:
            grad[p:] += (1.0 - h[:n - p])
            grad[:n - p] += (-h[p:])
    return grad

def obj_soft(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sh = corr - np.max(corr)
    return dx * (np.max(corr) + np.log(np.exp(sh * T).sum()) / T) + lam * (h.sum() - n / 2.0) ** 2

def obj_soft_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sh = corr - np.max(corr)
    e = np.exp(T * sh)
    w = e / e.sum()
    sm = np.max(corr) + np.log(e.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    for idx in range(m_full):
        if w[idx] < 1e-13:
            continue
        grad += w[idx] * corr_grad_h(h, idx, n)
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    try:
        res = minimize(obj_soft, x0, jac=lambda x: obj_soft_grad(x, T, lam)[1],
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return x0

def slp_step(h0, n_cuts=None, tolerance=0.0):
    n = len(h0)
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    m_full = 2 * n - 1
    c5_cur = np.max(corr) * dx
    if n_cuts is None:
        idxs = np.arange(m_full)
    else:
        idxs = np.argsort(corr)[::-1][:n_cuts]
    hvar = cp.Variable(n)
    tvar = cp.Variable()
    cons = [hvar >= 0, hvar <= 1, cp.sum(hvar) == n / 2]
    for idx in idxs:
        c_k = corr[idx] * dx
        g = corr_grad_h(h0, idx, n)
        cons.append(c_k + dx * (g @ (hvar - h0)) <= tvar + tolerance)
    prob = cp.Problem(cp.Minimize(tvar), cons)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return h0, c5_cur
    if hvar.value is None:
        return h0, c5_cur
    nh = project_h(np.array(hvar.value).flatten())
    nc = compute_c5(nh)
    if nc < c5_cur + 0.0005:
        return nh, nc
    return h0, c5_cur

def coord_desc(h, t_end, max_sweeps):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    dx = 2.0 / n
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        free = np.where((best_h > 1e-6) & (best_h < 1.0 - 1e-6))[0]
        idxs = free[np.random.permutation(len(free))] if len(free) > 0 else np.random.permutation(n)
        for i in idxs:
            cb = np.correlate(best_h, 1.0 - best_h, mode='full')
            d_i = np.zeros(m_full)
            for idx in range(m_full):
                lg = idx - n + 1
                if lg == 0:
                    d_i[idx] = 1.0 - 2.0 * best_h[i]
                elif 0 < lg < n:
                    if i >= lg:
                        d_i[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        d_i[idx] += (-best_h[i + lg])
                else:
                    p = -lg
                    if p < n:
                        if i + p < n:
                            d_i[idx] += (1.0 - best_h[i + p])
                        if i >= p:
                            d_i[idx] += (-best_h[i - p])
            def f1d(x):
                return np.max(cb + (x - best_h[i]) * d_i) * dx
            try:
                r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-11})
            except Exception:
                continue
            ht = best_h.copy()
            ht[i] = r.x
            ht = project_h(ht)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-13:
                best_c5, best_h, improved = ct, ht, True
        if not improved:
            break
    return best_h, best_c5

def multistage_h(h0, n, seed=0):
    rng = np.random.RandomState(seed)
    cands = []
    sc = 0.02 + 0.04 * (seed % 3)
    cands.append(project_h(h0 + sc * rng.randn(n)))
    lo = int(rng.rand() * (n - 16))
    hi = lo + int(rng.rand() * 10 + 8)
    hb = h0.copy()
    tmp = hb[lo:lo + 4].copy()
    hb[lo:lo + 4] = hb[hi:hi + 4]
    hb[hi:hi + 4] = tmp
    cands.append(project_h(hb))
    freq = rng.choice([2, 3, 5, 7, 11])
    phase = rng.rand() * 2 * np.pi
    cands.append(project_h(h0 + 0.03 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)))
    return cands

def run(seed=42, budget_s=1100, **kwargs):
