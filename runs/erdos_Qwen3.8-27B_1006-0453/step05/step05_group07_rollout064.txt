    if s > tgt:

        h *= tgt / s
    else:

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

_T = [1.0]

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h *= tgt / s
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h *= tgt / max(s, 1e-13)
        else:
            h += rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.005) & (h < 0.995)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(x):
    T = _T[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n

def _sgrad(x):
    T = _T[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    mf = 2 * n - 1
    for k in range(mf):
        wk = w[k]
        if wk < 1e-11:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=300):
    _T[0] = T
    try:
        r = minimize(h, _sobj, _sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16,
                              'gtol': 1e-10, 'maxls': 40})
        return proj_h(r.x, n)
    except Exception:
        return h

_CB = [np.zeros(1)]
_CC = [np.zeros(1)]

def _f1d(hi):
    return float(np.max(_CB[0] + hi * _CC[0]))

def _coef_i(h, i, n):
    mf = 2 * n - 1
    coef = np.zeros(mf)
    for lag in range(-(n - 1), n):
        k = lag + n - 1
        d = abs(lag)
        if lag >= 0:
            if i <= n - 1 - d:
                coef[k] += (1.0 - h[i + d])
            if i >= d:
                coef[k] -= h[i - d]
        else:
            if i >= d:
                coef[k] += (1.0 - h[i - d])
            if i <= n - 1 - d:
                coef[k] -= h[i + d]
    return coef

def coord_desc(h, n, t_end, sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr_cache = np.correlate(best, 1.0 - best, mode='full')
        order = np.random.permutation(n)
        for ii in range(n):
            if time.time() > t_end:
                return best, best_c5
            i = order[ii]
            old = best[i]
            coef = _coef_i(best, i, n)
            _CB[0] = corr_cache - old * coef
            _CC[0] = coef
            r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-13})
            nv = float(r.x)
            if abs(nv - old) < 1e-12:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                corr_cache = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5
