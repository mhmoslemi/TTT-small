import numpy as np

def corr_fft(h):
    N = len(h)
    L = 2 * N - 1
    return np.fft.irfft(np.fft.rfft(h, L) * np.conj(np.fft.rfft(1-h, L)), L)[:L]

def c5_of(h):
    return float(np.max(np.correlate(h, 1-h, 'full'))) * 2.0 / len(h)

def make_D(h, N):
    p = np.arange(N)[:, None]
    s = (np.arange(2*N-1) - N + 1)[None, :]
    a, b = p - s, p + s
    va, vb = (a >= 0) & (a < N), (b >= 0) & (b < N)
    ac, bc = np.clip(a, 0, N-1), np.clip(b, 0, N-1)
    return -(1 - h[ac]) * va + h[bc] * vb

    h = np.asarray(h, float)
    if h.size != N:

    t = N // 2

                if np.clip(h2 - mid, 0, 1).sum() > tgt: lo = mid
                else: hi = mid
            h2 = np.clip(h2 - (lo + hi) / 2, 0, 1)
        m = beta * (h2 - h)
        h = h2
        it += 1
        if it % 300 == 0:
