import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        # Scale down non-max elements
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

        p[idx] = (0.08 + 0.25 * cool) * rng.standard_normal(np_)

        p[:half] = 0.1 * cool * rng.standard_normal(half)

def fast_coord_descent(h, n, t_end, max_sweeps=200):
    """Fast coordinate descent using precomputed coefficients."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)  # offset for each correlation index
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            # Precompute for coordinate i
            h0 = best_h.copy()
            old_val = h0[i]
            h0[i] = 0.0
            corr_base = np.correlate(h0, 1.0 - h0, mode='full')
            
            # Coefficients
            j1 = i + s_arr
            j2 = i - s_arr
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - best_h[np.clip(j1[v1], 0, n - 1)])
            coefs[v2] += (-best_h[np.clip(j2[v2], 0, n - 1)])
            
            # Evaluate objective at delta
            def obj_eval(d):
                vals = corr_base + d * coefs
                vals[n - 1] = corr_base[n - 1] + d * (1.0 - d)
                return np.max(vals)
            
            try:
                res = minimize_scalar(obj_eval, bounds=(0.0, 1.0),
                                      method='bounded', options={'xatol': 1e-10})
                new_val = float(res.x)
            except:
                new_val = old_val
            
            if abs(new_val - old_val) < 1e-10:
                continue
            
            best_h[i] = new_val
            # Recompute c5 (this is the expensive part, but we do it once per accepted change)
            # Actually, we can check quickly:
            h0[i] = new_val
            vals = np.correlate(best_h, 1.0 - best_h, mode='full')
            c5_new = float(np.max(vals) * dx)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                imp = True
            # else: revert? No, we accepted the 1D optimum, which should be fine
            # Actually let's keep it even if c5 doesn't improve globally (it might help future coords)
    
        # After each sweep, ensure feasibility
        best_h = project_h(best_h)
        c5_check = compute_c5(best_h)
        if c5_check < best_c5:
            best_c5 = c5_check
        
        if not imp:
            break
    
    return best_h, best_c5

Wait, I realize the issue. The fast coordinate descent still computes the full correlation at the start of each coordinate optimization (for corr_base). That's O(n²) per coordinate, so O(n³) per sweep total. For n=160, that's 4M operations per sweep.

I can avoid this! The key insight: if I maintain the full correlation vector and update it when I change a coordinate, I don't need to recompute from scratch.

When h[j] changes from old to new (Δ = new - old):
For each shift k (corresponding to offset s = k - (n-1)):
- The term h[j]*(1-h[j-s]) changes by Δ*(1-h[j-s]) if j-s is valid and j-s ≠ j
- The term h[j+s]*(1-h[j]) changes by -h[j+s]*Δ if j+s is valid and j+s ≠ j
- If j-s = j (s=0): the term h[j]*(1-h[j]) changes by new*(1-new) - old*(1-old)

So I can maintain the correlation vector incrementally! This makes each coordinate update O(n) instead of O(n²).

            old = best_h[i]

            # corr[k] = corr_base + (d - old) * coef[k] + quadratic_term
            # Where corr_base = corr with h[i] replaced by 0
            # Actually, let's just compute the linear + quadratic decomposition directly.
            
            # corr[k] as function of best_h[i] = d:
            # The terms involving h[i]:
            # 1. i as first index: d * (1 - h[i-s]) if i-s valid and i-s != i
            # 2. i as second index (i-s = i, i.e., s=0): handled separately
            # 3. i-s as first index, i as second: h[i-s] * (1-d) if i-s valid and i-s != i
            
            # Let me just compute coefs and the quadratic correction:
            # For s != 0:
            #   term1: d * (1 - h[i-s]) if i-s in [0,n)
            #   term2: h[i+s] * (1-d) if i+s in [0,n)
            #   Total contribution to corr[k] from these terms: d*(1-h[i-s]) + h[i+s]*(1-d)
            #   = d - d*h[i-s] + h[i+s] - d*h[i+s]
            #   = (d - old) * (1 - h[i-s] - h[i+s]) + old*(1-h[i-s]) + h[i+s]*(1-old)
            #   So the "base" (current corr) already has old*(1-h[i-s]) + h[i+s]*(1-old)
            #   The change is (d-old)*(1-h[i-s]-h[i+s])
            
            # For s = 0 (k = n-1):
            #   term: h[i]*(1-h[i]) = d*(1-d)
            #   Change: d*(1-d) - old*(1-old) = (d-old)*(1-old-d) (approximately linear)
            #   Exact: (d-old) - (d²-old²) = (d-old)(1-d-old)
            
            # So for all k:
            # corr_new[k] = corr[k] + (d-old) * coefs[k]
            # where coefs[k] = (1-h[i-s]-h[i+s]) for s≠0, and (1-d-old) for s=0
            # But for s=0, the coefficient depends on d! So it's quadratic.
            
            # Let me define:
            # For s≠0: coefs is constant
            # For s=0: corr[n-1]_new = corr[n-1] + d*(1-d) - old*(1-old)
            
            coefs = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                s = s_arr[k]
                if s == 0:
                    coefs[k] = 0  # handle separately
                else:
                    c = 0.0
                    if 0 <= i - s < n:
                        c += (1.0 - best_h[i - s])
                    if 0 <= i + s < n:
                        c += (-best_h[i + s])
                    coefs[k] = c
            
            # Now optimize d in [0,1] to minimize max_k corr_new[k]
            # corr_new[k] = corr[k] + (d-old)*coefs[k] for k≠n-1
            # corr_new[n-1] = corr[n-1] + d*(1-d) - old*(1-old)
            
            def obj(d):
                vals = corr + (d - old) * coefs
                vals[n - 1] = corr[n - 1] + d * (1.0 - d) - old * (1.0 - old)
                return np.max(vals)
            
            try:
                res = minimize_scalar(obj, bounds=(0.0, 1.0),
                                      method='bounded', options={'xatol': 1e-10})
                new_d = float(res.x)
            except:
                new_d = old
            
            if abs(new_d - old) < 1e-10:
                continue
            
            # Update corr incrementally
            delta = new_d - old
            corr += delta * coefs
            corr[n - 1] += new_d * (1.0 - new_d) - old * (1.0 - old)
            
            best_h[i] = new_d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                imp = True
        
        if not imp:
            break
    
    return best_h, best_c5, corr

def optimize_coord(corr, coefs, old, n):
    """Find optimal d for one coordinate."""
    # The objective is max_k [corr[k] + (d-old)*coefs[k]] with quad correction at k=n-1
    # This is a piecewise-linear (plus one quadratic) function of d
    # Find its minimum on [0,1]
    
    # Breakpoints occur where two lines cross:
    # corr[k1] + (d-old)*coefs[k1] = corr[k2] + (d-old)*coefs[k2]
    # d = old + (corr[k2] - corr[k1]) / (coefs[k1] - coefs[k2])
    # There could be O(n²) breakpoints, but we don't need all of them.
    
    # Simpler: just evaluate on a grid and refine
    grid = np.linspace(0.0, 1.0, 33)
    vals = np.empty(len(grid))
    for idx, d in enumerate(grid):
        v = corr + (d - old) * coefs
        v[n - 1] = corr[n - 1] + d * (1.0 - d) - old * (1.0 - old)
        vals[idx] = np.max(v)
    
    best_idx = np.argmin(vals)
    best_d = grid[best_idx]
    
    # Refine: binary search on the grid to narrow down
    if best_idx > 0 and best_idx < len(grid) - 1:
        lo = grid[max(0, best_idx - 1)]
        hi = grid[min(len(grid) - 1, best_idx + 1)]
        # Golden section search
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(30):
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            v1 = np.max(corr + (x1 - old) * coefs)  # approximate (skip quad correction for speed)
            v2 = np.max(corr + (x2 - old) * coefs)
            if v1 < v2:
                hi = x2
            else:
                lo = x1
        best_d = (lo + hi) / 2
    elif best_idx == 0:
        # Minimum at d=0
        best_d = 0.0
    else:
        best_d = 1.0
    
    return best_d
