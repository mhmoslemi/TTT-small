def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    if abs(diff) < 1e-14:

    if diff > 0:

        h = h + room * (diff / tot)

            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def topk_obj(h, top_k, power):
    """Minimize the 'power-weighted' sum of top-k correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Sort and take top k
    idx = np.argpartition(corr, -top_k)[-top_k:]
    top_vals = corr[idx]
    # Power-weighted sum (approximates max as power -> infinity)
    val = dx * np.sum(top_vals ** power)
    return val

    """Gradient of the top-k power-weighted objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    top_vals = corr[idx]
    # Weights: proportional to top_vals^(power-1)
    w = top_vals ** (power - 1)
    w = w / w.sum()  # normalize

    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    n1 = n - 1
    for j in range(m_full):
        k = j - n1
        wi = w[j]
        if wi < 1e-14:
            continue
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        else:
            p = abs(k)
            if p < n:
                grad[p:n] += wi * (1.0 - h[0:n-p])
                grad[0:n-p] += -wi * h[p:n]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_refine(h0, T, lam, maxiter):
    n = len(h0)
    x0 = project_h(h0)
    def fobj(x):
        return lse_obj_and_grad(x, T, lam)[0]
    def fgrad(x):
        return lse_obj_and_grad(x, T, lam)[1]
    try:
        res = minimize(fobj, x0, jac=fgrad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
        return project_h(res.x)
    except Exception:
        return x0

def subgrad_polish(h, n, n_steps, step_scale):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    eta0 = step_scale * 2.0 / n
    for s in range(n_steps):
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_star = int(np.argmax(corr))
        k = k_star - (n - 1)
        g = np.zeros(n)
        if k == 0:
            g[:] = 1.0 - 2.0 * best_h
        else:
            p = abs(k)
            if p < n:
                g[p:n] += (1.0 - best_h[0:n-p])
                g[0:n-p] += -best_h[p:n]
        eta = eta0 / (1.0 + s * 0.1)
        cand = best_h - eta * g
        cand = project_h(cand)
        c5v = compute_c5(cand)
        if c5v < best_c5:
            best_h = cand
            best_c5 = c5v
        else:
            eta0 *= 0.8
    return best_h, best_c5

def coord_descent(h_in, n, t_end, max_sweeps):
    best_h = h_in.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    n1 = n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h_i = best_h[i]
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            delta = np.zeros(m_full)
            for k in range(1, n):
                d = 0.0
                if i >= k:
                    d += (1.0 - best_h[i - k])
                if i + k < n:
                    d += -best_h[i + k]
                delta[n1 + k] = d
                delta[n1 - k] = d
            def f1d(t):
                ct = corr_base + (t - h_i) * delta
                ct[n1] = corr_base[n1] + (t - h_i) - (t * t - h_i * h_i)
                return np.max(ct)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            hi_new = res.x
            c5t = res.fun * (2.0 / n)
            if c5t < best_c5 - 1e-13:
                best_h[i] = hi_new
                best_c5 = c5t
                improved = True
        if not improved and sweep > 5:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Progressive T ramp L-BFGS-B (20% budget)
    p1_end = t0 + budget_s * 0.20
    for T in [80, 300, 1200, 5000, 15000]:
        if elapsed() > budget_s * 0.6:
            break
        hi = lbfgs_refine(best_h, T, 500.0, 8000)
        c5v = compute_c5(hi)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hi
        if elapsed() > p1_end:
            break

    # Phase 2: Subgradient polish (5% budget)
    if elapsed() < t0 + budget_s * 0.30:
        p2_end = min(t0 + budget_s * 0.30, t0 + budget_s * 0.45)
        best_h, best_c5 = subgrad_polish(best_h, n, 200, 0.5)

    # Phase 3: Coordinate descent (25% budget)
    p3_end = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, n, p3_end, 2000)

    # Phase 4: High-T L-BFGS-B (20% budget)
    p4_end = t0 + budget_s * 0.70
    for T in [20000, 60000, 150000, 400000]:
        if elapsed() > p4_end:
            break
        hi = lbfgs_refine(best_h, T, 800.0, 6000)
        c5v = compute_c5(hi)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hi

    # Phase 5: Structured basin hopping (15% budget)
    p5_end = t0 + budget_s * 0.85
    work_h = best_h.copy()
    work_c5 = best_c5
    freqs = [1, 2, 3, 5, 8, 13, 21]
    for it in range(40):
        if elapsed() > p5_end:
            break
        cool = max(1.0 - 0.95 * (it / 40.0), 0.05)
        sc = 0.005 + 0.04 * cool
        pert = np.random.randn(n) * sc
        fr = freqs[np.random.randint(len(freqs))]
        amp = 0.02 * cool
        pert += amp * np.sin(2 * np.pi * fr * np.arange(n) / n + np.random.rand() * 6.28)
        h_p = project_h(work_h + pert)
        T_bh = int(2000 * cool + 500)
        hc = lbfgs_refine(h_p, T_bh, 400.0, 3000)
        c5v = compute_c5(hc)
        temp = 0.0003 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        else:
            temp *= 0.95

    # Phase 6: Final coordinate descent + subgrad (10% budget)
    if elapsed() < t0 + budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.95, 800)

    if elapsed() < t0 + budget_s * 0.97:
        best_h, best_c5 = subgrad_polish(best_h, n, 150, 0.3)

    # Final L-BFGS-B polish
    if elapsed() < t0 + budget_s * 0.99:
        hi = lbfgs_refine(best_h, 500000, 1000.0, 5000)
        c5v = compute_c5(hi)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hi

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
