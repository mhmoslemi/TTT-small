        corr_sub = corr

    val = m + np.log(np.sum(np.exp(T * (corr_sub - m)))) / T
    n = len(h)
    dx = 2.0 / n
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)

            pert[chosen] = (0.015 + 0.06 * cool) * np.random.randn(len(chosen))
        elif mod == 1:
            x = np.arange(n) / n
            f1 = np.random.randint(1, 12)
            f2 = np.random.randint(12, 30)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.012 + 0.045 * cool) * (np.sin(2*np.pi*f1*x + ph) + 0.5*np.sin(2*np.pi*f2*x))
        elif mod == 2:
            # Binary snap
            h_snap = snap_binary(work_h, thresh=0.02 + 0.08 * cool)
            hc = slsqp_opt(h_snap, 3000 + 20000 * cool, n, maxiter=600, ftol=1e-14)
            c5v = compute_c5(hc)
            temp = 0.0004 * cool + 0.00002
            if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
                work_h, work_c5 = hc, c5v
                if c5v < best_c5:
                    best_c5, best_h = c5v, hc
            continue
        elif mod == 3:
            blk = np.random.randint(n // 12, n // 4)
            b = np.random.randint(0, max(1, n - blk))
            pert = np.zeros(n)
            pert[b:b + blk] = (0.02 + 0.06 * cool) * np.random.randn(blk)
        elif mod == 4:
            # Paired mass transfer
            pert = np.zeros(n)
            for _ in range(6):
                a, b = np.random.choice(n, 2, replace=False)
                d = (work_h[b] - work_h[a]) * (0.3 + 0.7 * cool)
                pert[a] += d
                pert[b] -= d
        elif mod == 5:
            pert = (0.008 + 0.03 * cool) * np.random.randn(n)
        else:
            mask = np.random.random(n) < 0.1
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.1 * cool) * np.sign(np.random.randn(mask.sum()))

        h_pert = project_h(work_h + pert)
        T_bh = 300 + 30000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=500, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0005 * cool + 0.00003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc

    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h

    # Phase 4: SLSQP high-T polish (7% time)
    t4 = t0 + budget_s * 0.85
    for T in [100000, 500000, 2000000]:
        if time.time() > t4:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=3000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 5: Final coordinate descent (10% time)
    t5 = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=600)

    # Phase 6: Ultimate polish
    if el() < budget_s * 0.985:
        hc = slsqp_opt(best_h, 5000000, n, maxiter=5000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
        h_snap = snap_binary(best_h, thresh=0.015)
        hc2 = slsqp_opt(h_snap, 1000000, n, maxiter=3000, ftol=1e-16)
        c5v2 = compute_c5(hc2)
        if c5v2 < best_c5:
            best_c5, best_h = c5v2, hc2
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=100)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
