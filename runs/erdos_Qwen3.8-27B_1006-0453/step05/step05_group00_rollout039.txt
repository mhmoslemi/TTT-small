def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0

            lo = mid
        else:

    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    # D[i, k] = change in corr[k] when h[i] goes 1→0
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]  # (N, 1)
    sv = shifts[None, :]  # (1, 2N-1)
    pm = pos - sv  # (N, 2N-1)
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def local_search(h, N, deadline, max_sweeps=1000, full_eval=True):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        # Fix sum
        if n1 > nh:
            m = np.where(h >= 1)[0]
            if len(m) >= n1 - nh:
                m = m[np.argsort(h[m])[::-1]][:n1 - nh]  # reduce largest
                h[m] = 0
            else:
                m_all = np.where(h > 0)[0]
                m_all = m_all[np.argsort(h[m_all])[::-1]][:n1 - nh]
                h[m_all] = 0
        else:
            m = np.where(h <= 0)[0]
            if len(m) >= nh - n1:
                m = m[np.argsort(-h[m])][:nh - n1]
                h[m] = 1
            else:
                m_all = np.where(h < 1)[0]
                m_all = m_all[np.argsort(h[m_all])][:nh - n1]
                h[m_all] = 1
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-12:
            best_c = P
            best_h = h.copy()

        zeros = np.where(h == 0)[0]

        else:

            p_list = np.random.choice(ones, n_sample, replace=False) if len(ones) > n_sample else ones
            for p in p_list:

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    r = np.fft.irfft(H * G, L)
    return r[:L]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)
