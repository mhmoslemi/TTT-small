def compute_swap_D(h, N):
    """Compute D[p, k] = change in corr[k] when h[p] goes from 1 to 0."""
    idx = np.arange(2 * N - 1)

    # Term i=p-shift where i+shift=p: was 1*(1-h[p-shift]), now 0. Change: -(1-h[p-shift])

    shifts = shifts.astype(float)

    D -= np.where(vm, 1.0 - h[pm_c], 0.0)
    
    return D

D[p, k] where k indexes the full correlation (0 to 2N-2), corresponding to shift = k - (N-1).

If h[p] changes from 1 to 0:
- In corr[k] = ∑_i h[i+shift](1-h[i]):
  - Term i=p: was h[p+shift]*(1-1) = 0, now h[p+shift]*(1-0) = h[p+shift]. Δ = +h[p+shift] if 0≤p+shift<N
  - Term i=p-shift (if valid): was 1*(1-h[p-shift]), now 0*(1-h[p-shift]). Δ = -(1-h[p-shift]) if 0≤p-shift<N

So D[p,k] = h[p+shift_k] * [0≤p+shift_k<N] - (1-h[p-shift_k]) * [0≤p-shift_k<N]

For the swap (p: 1→0, q: 0→1):
Δcorr[k] = D[p,k] + (-D_for_0_to_1[q,k])

What's D_for_0_to_1[q,k]? If h[q] changes from 0 to 1:
- Term i=q: was h[q+shift]*(1-0) = h[q+shift], now h[q+shift]*(1-1) = 0. Δ = -h[q+shift]
- Term i=q-shift: was 0*(1-h[q-shift]) = 0, now 1*(1-h[q-shift]). Δ = +(1-h[q-shift])

So D_0_to_1[q,k] = -h[q+shift] + (1-h[q-shift]) = -D_1_to_0[q,k]

Where D_1_to_0[q,k] = h[q+shift] - (1-h[q-shift]) (same formula as D[p,k] with p→q).

So Δcorr[k] = D[p,k] - D[q,k] where D is computed with the CURRENT h (before the swap).

This is what the code does. Good.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv
    pm = pos - sv
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    D = np.zeros((N, 2 * N - 1))
    D += np.where(vp, h[np.clip(pp, 0, N - 1)], 0.0)
    D -= np.where(vm, 1.0 - h[np.clip(pm, 0, N - 1)], 0.0)
    return D

def local_search(h, N, deadline, kick_every=30, kick_n=4):
    h = h.astype(float).copy()
    nh = N // 2
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            h[np.random.choice(idx, n1 - nh, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[np.random.choice(idx, nh - n1, replace=False)] = 1
    best_c = float(np.max(fft_corr(h)))
    stag = 0
    while time.time() < deadline:
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            stag = 0
        else:
            stag += 1
        if stag >= kick_every:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            kn = min(kick_n, len(ones), len(zeros))
            if kn > 0:
                h[np.random.choice(ones, kn, replace=False)] = 0.0
                h[np.random.choice(zeros, kn, replace=False)] = 1.0
            stag = 0
            if stag >= 200:
                break
            continue
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        best_NP = P
        best_pq = None
        for p in ones:
            base = c + D[p, :]
            cm = (base[None, :] - D[zeros, :]).max(axis=1)
            mi = int(np.argmin(cm))
            if cm[mi] < best_NP - 1e-12:
                best_NP = float(cm[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            stag += 2
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return h, best_c

def nesterov_opt(h, N, deadline, top_k=30, lr0=0.02, mom0=0.85):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h))) * (2.0 / N)
    m = np.zeros(N)
    lr = lr0
    beta = mom0
    ctr = N - 1
    pat = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr)) * (2.0 / N)
        if pk < best_pk - 1e-9:
            best_pk = pk
            best_h = y.copy()
            pat = 0
        else:
            pat += 1
            if pat > 400:
                break
            if pat % 80 == 0:
                lr *= 0.96
        ti = np.argpartition(corr, -top_k)[-top_k:]
        g = np.zeros(N)
        for ai in ti:
            L = int(ai - ctr)
            if L == 0:
                g += (1.0 - 2.0 * y)
            elif L > 0:
                g[L:] += (1.0 - y[:N - L])
                g[:N - L] -= y[L:]
            else:
                M = -L
                g[:N - M] += (1.0 - y[M:])
                g[M:] -= y[:N - M]
        g /= top_k
        h_new = project_sum(y - lr * g, target)
        m = beta * (h_new - h)
        h = h_new
    return best_h, best_pk
