            _CB[0] = corr - old * coef
            _CC[0] = coef
            r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-14, 'maxiter': 60})
            nv = float(r.x)
            if abs(nv - old) < 1e-13:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def pair_swap_best(h, n, t_end, iters=50000):
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode='full')
    dx = 2.0 / n
    ones = np.where(best > 0.5)[0]
    zeros = np.where(best <= 0.5)[0]
    for _ in range(iters):
        if time.time() > t_end:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        best[i], best[j] = best[j], best[i]
        nc = np.correlate(best, 1.0 - best, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 - 1e-15:
            best_c5 = nc5
            corr = nc
        else:
            best[i], best[j] = best[j], best[i]
    return best, best_c5

def basin_hop_fast(h, n, t_end, n_iters=300):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / max(n_iters, 1))
        mod = it % 4
        if mod == 0:
            pert = (0.03 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(2, 20)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.3 + 0.4 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.2 * cool) * np.random.randn(mask.sum())
        else:
            k = np.random.randint(2, 8)
            idx = np.random.randint(0, n - k)
            pert = np.zeros(n)
            pert[idx:idx+k] = (0.1 + 0.2 * cool) * np.random.randn(k)
        hp = proj_h(work + pert, n)
        ht = lbfgs_opt(hp, 2000.0 + 3000.0 * cool, n, iters=150)
        ht, c5t = fast_coord(ht, n, min(time.time() + 2.0, t_end), sweeps=30)
        temp = 0.0006 * cool + 0.00003
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht
    return best, best_c5

def golden_init(n):
    phi = (1 + np.sqrt(5)) / 2
    h = np.zeros(n)
    for i in range(n):
        frac = (i * 0.618033988749895) % 1.0
        if frac < 0.5:
            h[i] = 1.0
    h = proj_h(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Also prepare a golden-ratio structured start
    h_gold = golden_init(n)
    h_gold = lbfgs_opt(h_gold, 500.0, n, iters=200)
    h_gold, c5g = fast_coord(h_gold, n, t0 + budget_s * 0.05, sweeps=100)
    if c5g < best_c5:
        best_h = h_gold
        best_c5 = c5g

    # Phase 1: L-BFGS with increasing T
    for T in [100, 500, 2000, 10000, 50000, 200000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Fast coordinate descent (many sweeps)
    best_h, best_c5 = fast_coord(best_h, n, t0 + budget_s * 0.30, sweeps=500)

    # Phase 3: Basin hopping with fast local opt
    best_h, best_c5 = basin_hop_fast(best_h, n, t0 + budget_s * 0.70, n_iters=400)

    # Phase 4: More coordinate descent
    best_h, best_c5 = fast_coord(best_h, n, t0 + budget_s * 0.80, sweeps=300)

    # Phase 5: L-BFGS refinement at high T
    if time.time() - t0 < budget_s * 0.88:
        for T in [50000, 200000, 800000]:
            if time.time() - t0 > budget_s * 0.88:
                break
            h_new = lbfgs_opt(best_h, T, n, iters=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 6: Final coordinate descent
    best_h, best_c5 = fast_coord(best_h, n, t0 + budget_s * 0.96, sweeps=400)

    # Phase 7: Try n=200 if time permits
    if time.time() - t0 < budget_s * 0.90:
        n2 = 200
        try:
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
            h2 = np.interp(x_new, x_old, best_h)
            h2 = proj_h(h2, n2)
            h2 = lbfgs_opt(h2, 2000.0, n2, iters=300)
            h2, c5_2 = fast_coord(h2, n2, t0 + budget_s * 0.95, sweeps=100)
            if c5_2 < best_c5:
                best_h = h2
                best_c5 = c5_2
                n = n2
        except Exception:
            pass

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
