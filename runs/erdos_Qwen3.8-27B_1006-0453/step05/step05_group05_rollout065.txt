    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2*n-1):
            lag = k - (n-1)
            # term 1: w[k] * (1-h[j+lag]) if j+lag in [0,n-1]

            idx2 = j - lag
            if 0 <= idx2 < n:
                s -= w[k] * h[idx2]
        grad[j] = s
    return (2.0 / n) * grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    dx = 2.0 / len(h)
    return val * dx

def compute_grad(h, T):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    
    lags = np.arange(-(n-1), n)
    J = np.arange(n)[:, None]
    L = lags[None, :]
    idx1 = J + L
    idx2 = J - L
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    A = ((1.0 - h[np.clip(idx1, 0, n-1)]) * valid1 * w[None, :]).sum(axis=1)
    B = (h[np.clip(idx2, 0, n-1)] * valid2 * w[None, :]).sum(axis=1)
    
    return (2.0 / n) * (A - B)

def lbg(log_T):
    T = np.exp(log_T)
    return T

def lbfgs_step(h, T, n, penalty=1e8, maxiter=500):
    def obj(x):
        val = smooth_c5(x, T)
        s = x.sum() - n / 2.0
        return val + penalty * s * s
    
    def jac(x):
        g = compute_grad(x, T)
        s = x.sum() - n / 2.0
        return g + 2 * penalty * s
    
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                      bounds=[(0, 1)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt(h, n, i, corr_base, coefs, niter=50):
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr_base + m1 * coefs)
        f2 = np.max(corr_base + m2 * coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def get_coefs(h, n, i):
    corr_base = get_corr(np.where(np.arange(n) == i, 0.0, h))
    m_vals = np.arange(-(n-1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2*n-1)
    coefs[valid1] += (1.0 - h[np.clip(vm1[valid1], 0, n-1)])
    coefs[valid2] += (-h[np.clip(vm2[valid2], 0, n-1)])
    return corr_base, coefs

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            corr_base, coefs = get_coefs(best_h, n, i)
            hi_new = coord_opt(best_h, n, i, corr_base, coefs)
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / len(h))

def compute_grad(h, T):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    lags = np.arange(-(n-1), n)
    J = np.arange(n)[:, None]
    L = lags[None, :]
    idx1 = J + L
    idx2 = J - L
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    A = ((1.0 - h[np.clip(idx1, 0, n-1)]).astype(float) * v1 * w[None, :]).sum(axis=1)
    B = (h[np.clip(idx2, 0, n-1)].astype(float) * v2 * w[None, :]).sum(axis=1)
    return (2.0 / n) * (A - B)

def lbfgs_opt(h, T, n, penalty, maxiter):
    def obj(x):
        v = smooth_c5(x, T)
        s = x.sum() - n / 2.0
        return v + penalty * s * s
    def jac(x):
        g = compute_grad(x, T)
        s = x.sum() - n / 2.0
        return g + 2.0 * penalty * s
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_h(np.clip(res.x, 0.0, 1.0))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps, tern_iter=50):
    best_h = h.copy()
    corr = get_corr(best_h)
    best_c5 = float(np.max(corr) * 2.0 / n)
    lags = np.arange(-(n-1), n)
    cm1 = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            vm1 = i - lags
            vm2 = i + lags
            v1 = (vm1 >= 0) & (vm1 < n)
            v2 = (vm2 >= 0) & (vm2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - best_h[np.clip(vm1[v1], 0, n-1)])
            coefs[v2] += (-best_h[np.clip(vm2[v2], 0, n-1)])
            cv = best_h[i]
            tmp = corr - cv * coefs
            lo, hi = 0.0, 1.0
            for _ in range(tern_iter):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(tmp + m1 * coefs) < np.max(tmp + m2 * coefs):
                    hi = m2
                else:
                    lo = m1
            new_val = (lo + hi) / 2.0
            if abs(new_val - cv) < 1e-13:
                continue
            delta = new_val - cv
            corr += delta * coefs
            best_h[i] = new_val
            nm = float(np.max(corr))
            if nm * cm1 < best_c5:
                best_c5 = nm * cm1
                improved = True
        best_h = project_h(best_h)
        corr = get_corr(best_h)
        best_c5 = float(np.max(corr) * cm1)
        if not improved:
            break
    return best_h, best_c5

def coord_descent_pairs(h, n, t_end, max_sweeps, tern_iter=40):
    best_h = h.copy()
    corr = get_corr(best_h)
    best_c5 = float(np.max(corr) * 2.0 / n)
    lags = np.arange(-(n-1), n)
    cm1 = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for _ in range(n // 3):
            if time.time() > t_end:
                return best_h, best_c5
            i, j = np.random.choice(n, 2, replace=False)
            ci = np.zeros(2*n-1); cj = np.zeros(2*n-1)
            for idx in [i, j]:
                vm1 = idx - lags; vm2 = idx + lags
                v1 = (vm1 >= 0) & (vm1 < n); v2 = (vm2 >= 0) & (vm2 < n)
                c = np.zeros(2*n-1)
                c[v1] += (1.0 - best_h[np.clip(vm1[v1], 0, n-1)])
                c[v2] += (-best_h[np.clip(vm2[v2], 0, n-1)])
                if idx == i:
                    ci = c
                else:
                    cj = c
            base = corr - best_h[i]*ci - best_h[j]*cj
            lo_i, hi_i, lo_j, hi_j = 0.0, 1.0, 0.0, 1.0
            for _ in range(tern_iter):
                mi1 = lo_i + (hi_i-lo_i)/3; mi2 = hi_i - (hi_i-lo_i)/3
                mj1 = lo_j + (hi_j-lo_j)/3; mj2 = hi_j - (hi_j-lo_j)/3
                f1 = np.max(base + mi1*ci + mj1*cj)
                f2 = np.max(base + mi2*ci + mj2*cj)
                f3 = np.max(base + mi1*ci + mj2*cj)
                f4 = np.max(base + mi2*ci + mj1*cj)
                bv = min(f1, f2, f3, f4)
                if bv == f1: hi_i = mi2; hi_j = mj2
                elif bv == f2: lo_i = mi1; lo_j = mj1
                elif bv == f3: hi_i = mi2; lo_j = mj1
                else: lo_i = mi1; hi_j = mj2
            new_i = (lo_i+hi_i)/2.0; new_j = (lo_j+hi_j)/2.0
            if abs(new_i - best_h[i]) < 1e-12 and abs(new_j - best_h[j]) < 1e-12:
                continue
            corr += (new_i - best_h[i])*ci + (new_j - best_h[j])*cj
            best_h[i] = new_i; best_h[j] = new_j
            nm = float(np.max(corr))
            if nm * cm1 < best_c5:
                best_c5 = nm * cm1
                improved = True
        best_h = project_h(best_h)
        corr = get_corr(best_h)
        best_c5 = float(np.max(corr) * cm1)
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160); best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B warmup with increasing T
    for T in [30, 200, 1000, 5000, 30000]:
        if elapsed() > budget_s * 0.10:
            break
        pen = 1e6 * max(1.0, smooth_c5(best_h, T))
        hc = lbfgs_opt(best_h, T, n, pen, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent (long)
    t2 = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=200)

    # Phase 3: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 100
    for it in range(n_bh):
        if elapsed() > budget_s * 0.62:
            break
        cool = 1.0 - 0.92 * (it / n_bh)
        mod = it % 8
        if mod == 0:
            pert = (0.015 + 0.06 * cool) * np.random.randn(n)
        elif mod == 1:
            fr = np.random.randint(1, 12)
            ph = np.random.uniform(0, 6.28)
            pert = (0.02 + 0.04 * cool) * np.sin(2*np.pi*fr*np.arange(n)/n + ph)
        elif mod == 2:
            mk = np.random.random(n) < 0.10
            pert = np.zeros(n); pert[mk] = (0.04 + 0.10*cool) * np.random.randn(mk.sum())
        elif mod == 3:
            bl = max(2, n//8); b = np.random.randint(0, max(1, n//bl))
            pert = np.zeros(n); pert[b*bl:min(n,(b+1)*bl)] = (0.02+0.05*cool)*np.random.randn(bl)
        elif mod == 4:
            pert = (0.008 + 0.025*cool) * np.random.randn(n)
        elif mod == 5:
            mk = np.random.random(n) < 0.06
            pert = np.zeros(n); pert[mk] = (0.03+0.10*cool)*np.random.choice([-1,1],size=mk.sum())
        elif mod == 6:
            f1=np.random.randint(1,6); f2=np.random.randint(6,15)
            pert=(0.01+0.03*cool)*(np.sin(2*np.pi*f1*np.arange(n)/n)+np.sin(2*np.pi*f2*np.arange(n)/n+0.5))
        else:
            a,b = np.random.choice(n,2,replace=False)
            pert = np.zeros(n); d=(work_h[b]-work_h[a])*(0.2+0.5*cool)
            pert[a]+=d; pert[b]-=d
        h_p = project_h(work_h + pert)
        T_bh = 300 + 6000*cool
        pen = 1e7 * max(1.0, smooth_c5(h_p, T_bh))
        hc = lbfgs_opt(h_p, T_bh, n, pen, maxiter=300)
        c5v = compute_c5(hc)
        temp = 0.0004 * cool + 0.00003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v-work_c5)/temp):
            work_h = hc; work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v; best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5; best_h = work_h

    # Phase 4: High-T L-BFGS
    for T in [50000, 200000, 800000]:
        if elapsed() > budget_s * 0.72:
            break
        pen = 1e8 * max(1.0, smooth_c5(best_h, T))

        c5v = compute_c5(hc)

            best_c5 = c5v; best_h = hc

    # Phase 5: Extended coordinate descent with pairs
    t5 = t0 + budget_s * 0.82
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=200)
    t5b = t0 + budget_s * 0.88

    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=50)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
