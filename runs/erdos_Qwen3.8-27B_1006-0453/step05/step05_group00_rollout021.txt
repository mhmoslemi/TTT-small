    return np.real(np.fft.ifft(H * G))

    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm + hp * vp

def ensure_sum(h, N):
    nh = N // 2
    h = h.astype(float).copy()
    h_bin = (h > 0.5).astype(float)
    n1 = int(h_bin.sum())
    if n1 == nh:
        return h_bin
    rng = np.random.RandomState(12345 + N)
    if n1 > nh:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, n1 - nh, replace=False)
        h_bin[idx] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, nh - n1, replace=False)
        h_bin[idx] = 1
    return h_bin

def binary_search(h, N, deadline, max_sweeps=500, sample_frac=1.0, perturb_every=0, verbose=False):
    h = h.astype(float).copy()
    if int(h.sum()) != N // 2:
        h = ensure_sum(h, N)
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-13:
            best_c = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        Dz = D[zeros, :]
        if sample_frac < 1.0 and len(ones) > 40:
            n_s = max(40, int(len(ones) * sample_frac))
            p_list = np.random.choice(ones, n_s, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - Dz
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if perturb_every > 0 and stall > perturb_every:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
                stall = 0
            else:
                break
        else:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
    return best_h, best_c * (2.0 / N)

def nesterov_opt(h, N, deadline, k=50, lr0=0.018):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 250 == 0:
            lr *= 0.992
            beta = min(0.97, 0.85 + it * 0.0001)
    return best_h, best_pk * dx

def upsample(h_old, N_old, N_new):
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Thorough full binary search at N0
    h_bin = ensure_sum(h0, N0)
    d1 = min(deadline - 500, t0 + 60)
    h1, c5_1 = binary_search(h_bin, N0, d1, max_sweeps=800, sample_frac=1.0)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Upscale to 2*N0 and binary search (multi-restart)
    N1 = 2 * N0
    for restart in range(3):
        if time.time() > deadline - 350:
            break
        if restart == 0:
            h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
        else:
            h_up = ensure_sum(upsample(best_h, len(best_h), N1), N1)
            # Perturb: flip 2-3 bits
            n_flip = 2 + restart
            ones = np.where(h_up == 1)[0]
            zeros = np.where(h_up == 0)[0]
            for _ in range(n_flip):
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h_up[p], h_up[q] = 0.0, 1.0
        d2 = min(deadline - 250 + restart * 30, time.time() + 120)
        h2, c5_2 = binary_search(h_up, N1, d2, max_sweeps=300, sample_frac=0.65)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Continuous Nesterov at N0
    if time.time() < deadline - 200:
        h_resamp = upsample(best_h, len(best_h), N0) if len(best_h) != N0 else best_h.copy()
        hc = h_resamp + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        d3 = min(deadline - 150, time.time() + 80)
        h3, c5_3 = nesterov_opt(hc, N0, d3, k=60, lr0=0.015)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Upscale to N=1520 and binary search
    N2 = 4 * N0
    if N2 <= 1600 and time.time() < deadline - 150:
        h_up2 = ensure_sum(upsample(best_h, len(best_h), N2), N2)
        d4 = min(deadline - 80, time.time() + 80)
        h4, c5_4 = binary_search(h_up2, N2, d4, max_sweeps=150, sample_frac=0.4)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Full binary search at current best resolution with perturbation
    N_cur = len(best_h)
    if time.time() < deadline - 60:
        h_bin5 = ensure_sum(best_h, N_cur)
        d5 = min(deadline - 40, time.time() + 70)
        sf = 1.0 if N_cur <= 400 else 0.7
        h5, c5_5 = binary_search(h_bin5, N_cur, d5, max_sweeps=400, sample_frac=sf, perturb_every=40)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Second Nesterov pass at N1
    if time.time() < deadline - 40:
        N_nest = min(N1, max(N0, len(best_h)))
        h_res = upsample(best_h, len(best_h), N_nest) if len(best_h) != N_nest else best_h.copy()
        hc2 = h_res + np.random.randn(N_nest) * 0.03
        hc2 = project_sum(hc2, N_nest / 2.0)
        d6 = min(deadline - 20, time.time() + 50)
        h6, c5_6 = nesterov_opt(hc2, N_nest, d6, k=70, lr0=0.01)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Final binary polish
    if time.time() < deadline - 2:
        N_fin = len(best_h)
        hF = ensure_sum(best_h, N_fin)
        sf_fin = 1.0 if N_fin <= 500 else 0.6
        hF2, c5F = binary_search(hF, N_fin, deadline - 0.5, max_sweeps=300, sample_frac=sf_fin)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
