    else:
        if s < 1e-14: return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt(h, n, i):
    """Optimize h[i] given all other h fixed."""
    old = h[i]
    h2 = h.copy()
    h2[i] = 0.0
    corr0 = np.correlate(h2, 1.0 - h2, mode='full')
    h2[i] = old
    # corr = corr0 + h[i] * delta where delta[k] depends on i and lag
    # delta[k] = contribution of h[i] to corr[k]
    # corr[k] = sum_j h[j](1-h[j+l]) where l = k-(n-1)
    # h[i] contributes (1-h[i+l]) if i+l valid, and -h[i-l] if i-l valid
    lags = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    for m, l in enumerate(lags):
        if 0 <= i+l < n:
            delta[m] += (1.0 - h[i+l])
        if 0 <= i-l < n:
            delta[m] -= h[i-l]
    # f(t) = max(corr0 + t*delta) * dx
    # Use minimize_scalar
    vals = corr0 + delta  # corr when h[i]=1
    # Actually: f(t) = max(corr0 + t*delta) * dx
    res = minimize_scalar(lambda t: np.max(corr0 + t * delta),
                          bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end: break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11: continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved: break
    return best_h, best_c5

def basin_hopping(h, n, c5, iters, t_end, amp_base=0.05, T_base=2000, cool_rate=0.92):
    work_h = h.copy()
    work_c5 = c5
    best_h = h.copy()
    best_c5 = c5
    for it in range(iters):
        if time.time() > t_end: break
        cool = max(0.05, cool_rate ** it)
        amp = amp_base * cool + 0.01
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        T = T_base * cool + 100
        hc = slsqp_smooth(h_pert, T, n, maxiter=300)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    return best_h, best_c5

def swap_search(h, n, c5, n_swaps=200, t_end=None):
    """Try swapping h[i] and 1-h[j] (or small adjustments)."""
    best_h = h.copy()
    best_c5 = c5
    for _ in range(n_swaps):
        if t_end and time.time() > t_end: break
        i, j = np.random.choice(n, 2, replace=False)
        h_try = best_h.copy()
        # Swap: set h[i] toward 1-h[j], h[j] toward 1-h[i]
        # Or just: h[i] = 1-h[i], h[j] = 1-h[j]
        h_try[i] = 1.0 - best_h[i]
        h_try[j] = 1.0 - best_h[j]
        h_try = project_h(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5
