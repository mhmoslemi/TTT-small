import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def objective_fn(h, T):
    corr = get_corr(h)
    n = len(h)
    dx = 2.0 / n
    return smooth_max(corr, T) * dx

def slsqp_step(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return objective_fn(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_minimax_1d(h_base, coefs, active_idx, lo=0.0, hi=1.0, niter=50):
    """Minimize max over active lags of (h_base[k] + x * coefs[k])"""
    # This is min_x max_k (a_k + x*b_k) for k in active_idx
    # = min_x max_k (b_k * x + a_k)
    # The min of max of linear functions can be found by bisection on the value
    # For a given value v, check if exists x in [lo,hi] s.t. a_k + x*b_k <= v for all k
    # This is a feasibility problem: for each k, x*b_k <= v - a_k
    # If b_k > 0: x <= (v-a_k)/b_k
    # If b_k < 0: x >= (v-a_k)/b_k  
    # If b_k == 0: need v >= a_k
    a = h_base[active_idx]
    b = coefs[active_idx]
    
    # Bisection on v (the achievable max value)
    # Lower bound: max of minimum possible values
    v_lo = max(np.where(b > 0, a + b*lo, np.where(b < 0, a + b*hi, a)))
    # Actually let me just do ternary/bisection on x directly for the max
    
    def f(x):
        return np.max(a + x * b)
    
    # Ternary search
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = f(m1)
        f2 = f(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def get_coord_coefs(h, n, i):
    """Get coefficients for coordinate i in each lag"""
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(vm1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(vm2[valid2], 0, n - 1)])
    return coefs

def coord_descent_active(h, n, t_end, max_sweeps=200, n_active=5):

        improved = False
        corr = get_corr(best_h)

        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = get_corr(h_base)
            coefs = get_coord_coefs(best_h, n, i)
            
            hi_new = coord_minimax_1d(corr_base, coefs, top_lags)
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)

            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def get_coefs(h, n, i):
    coefs = np.zeros(2 * n - 1)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
    return coefs

def minimax_1d(a, b, niter=40):
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a + m1 * b) < np.max(a + m2 * b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def slsqp_step(h, T, n, maxiter=500, ftol=1e-14):
    def obj(x):
        corr = get_corr(x)
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m) + 1))) / T - 1.0/T
        return val * (2.0 / n)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=200, n_active=4):
    best = h.copy()
    best_c5 = compute_c5(best)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = get_corr(best)
        top = np.argsort(corr)[-n_active:]
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                break
            coefs = get_coefs(best, n, i)
            h_old = best[i]
            corr_base = corr - h_old * coefs
            x_new = minimax_1d(corr_base[top], coefs[top])
            if abs(x_new - h_old) < 1e-10:

            h_try = best.copy()

                best = h_try
                corr = get_corr(best)
                top = np.argsort(corr)[-n_active:]
                improved = True
        if not improved:
            break
    return project_h(best), compute_c5(project_h(best))

            pert = (0.015 + 0.06 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif pt == 2:
            mask = np.random.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.12 * cool) * np.random.randn(mask.sum())
        elif pt == 3:
            blk = max(3, n // 10)
            b = np.random.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            pert[b*blk:(b+1)*blk] = (0.025 + 0.06 * cool) * np.random.randn(blk)
        elif pt == 4:
            a, b2 = np.random.choice(n, 2, replace=False)
            pert = np.zeros(n)
            d = (work_h[b2] - work_h[a]) * (0.3 + 0.6 * cool)
            pert[a] += d
            pert[b2] -= d
        elif pt == 5:
            f1, f2 = np.random.randint(1, 8), np.random.randint(8, 22)
            pert = (0.01 + 0.04 * cool) * (np.sin(2*np.pi*f1*np.arange(n)/n) +
                                           np.sin(2*np.pi*f2*np.arange(n)/n + 1.1))
        else:
            mask = np.random.random(n) < 0.06
            pert = np.zeros(n)
            pert[mask] = np.clip((0.05 + 0.15 * cool) * np.random.randn(mask.sum()), -0.3, 0.3)
        h_p = project_h(work_h + pert)
        T_bh = 300 + 8000 * cool
        hc = slsqp_step(h_p, T_bh, n, maxiter=300)
        hc, c5v = coord_descent(hc, n, min(t3, time.time() + budget_s * 0.04),
                                max_sweeps=40, n_active=4, use_pairs=False)
        temp = 0.0006 * cool + 0.00003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h

    # Phase 4: Second coord descent pass (15%)
    t4 = t0 + budget_s * 0.78
    best_h, best_c5 = coord_descent(best_h, n, t4, max_sweeps=400, n_active=6, use_pairs=True)

    # Phase 5: Fine basin hopping (10%)
    t5 = t0 + budget_s * 0.88
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(40):
        if elapsed() > t5:
            break
        cool = 1.0 - 0.8 * (it / 40.0)
        step = 0.008 + 0.025 * cool
        h_p = project_h(work_h + step * np.random.randn(n))
        T_bh = 2000 + 20000 * cool
        hc = slsqp_step(h_p, T_bh, n, maxiter=400)
        hc, c5v = coord_descent(hc, n, min(t5, time.time() + budget_s * 0.03),
                                max_sweeps=30, n_active=5, use_pairs=False)
        temp = 0.0003 * cool + 0.00002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h

    # Phase 6: Final polish (12%)
    hc = slsqp_step(best_h, 100000, n, maxiter=3000, ftol=1e-15)
    c5v = compute_c5(hc)
    if c5v < best_c5:
        best_c5, best_h = c5v, hc
    if elapsed() < budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97,
                                        max_sweeps=300, n_active=7, use_pairs=True)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
