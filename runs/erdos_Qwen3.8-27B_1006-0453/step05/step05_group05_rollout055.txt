    dx = 2.0 / n

    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * (2.0 / len(h))

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if diff > 1e-14:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h += room * (diff / tot)
    elif diff < -1e-14:
        if s > 1e-14:
            h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_gradient(h, T):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    dx_val = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        else:
            grad[:n + k] += wk * (1.0 - h[-k:])
            grad[-k:] -= wk * h[:n + k]
    return grad * dx_val

def proj_grad(h, T, max_iter=5000, lr_init=0.01, t_end=None):
    best_h = h.copy()
    best_f = smooth_obj(h, T)
    f_cur = best_f
    lr = lr_init

def block_slsqp(h, n, t_end, n_blocks=500, block_size=8):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    idx = n - 1
    for _ in range(n_blocks):
        if time.time() > t_end:
            break
        start = np.random.randint(0, n - block_size + 1)
        positions = list(range(start, start + block_size))
        x0 = best_h[positions].copy()
        fixed_h = best_h.copy()

        def obj_block(x):
            fixed_h[positions] = np.clip(x, 0, 1)
            corr = get_corr(fixed_h)
            m = np.max(corr)
            return (m + np.log(np.sum(np.exp(500 * (corr - m)))) / 500.0) * (2.0 / n)

        def con_block(x):
            total = best_h.sum() - x0.sum() + x.sum()
            return total - n / 2.0

        try:
            res = minimize(obj_block, x0, method='SLSQP',
                           bounds=[(0.0, 1.0)] * block_size,
                           constraints=[{'type': 'eq', 'fun': con_block}],
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.fun < obj_block(x0):

                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:

                    best_h = h_try
        except Exception:
            pass
    return best_h, best_c5

def full_slsqp(h, T, n, maxiter=1500):

def coord_opt_single(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    coefs = np.zeros(2 * n - 1)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr_base + m1 * coefs)
        f2 = np.max(corr_base + m2 * coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:

            h_try[i] = hi_new

            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

        h0, c5v = coord_descent(h0, n, t_cd_end, max_sweeps=30)
        if start_idx == 0 and c5v < best_c5:
            best_c5 = c5v
            best_h = h0.copy()

        # Phase D: Basin hopping
        t_bh_end = t_phase_end
        h0, c5v = basin_hopping(h0, c5v, n, t_bh_end, n_iter=80)
        if start_idx == 0 and c5v < best_c5:
            best_c5 = c5v
            best_h = h0.copy()

    # Final polish on best solution found
    n = len(best_h)
    if time.time() < t0 + budget_s * 0.90:
        for T in [50000, 200000, 800000]:
            if time.time() > t0 + budget_s * 0.90:
                break
            hc = full_slsqp(best_h, T, n, maxiter=2000)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    if time.time() < t0 + budget_s * 0.95:
        best_h, best_c5_final = proj_grad(best_h, 500000, max_iter=2000, lr_init=1e-5,
                                           t_end=t0 + budget_s * 0.95)
        c5v = compute_c5(best_h)
        if c5v < best_c5:
            best_c5 = c5v

    if time.time() < t0 + budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=30)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
