def grad_soft(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ex = np.clip(T * (corr - m), -50.0, 50.0)
    p = np.exp(ex)
    p /= p.sum()
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        w = p[mi]
        if w < 1e-15:
            continue
        s = mi - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad
