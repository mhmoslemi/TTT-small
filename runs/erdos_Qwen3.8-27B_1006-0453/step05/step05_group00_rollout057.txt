def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

        else:

def nesterov_lse(h, N, deadline, k=80, lr0=0.015, lse_beta=5.0):
    """Nesterov with log-sum-exp penalty instead of top-k."""
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    center = N - 1
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Gradient using top-k lags
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.97, 0.88 + it * 0.0003)
    
    return best_h, best_pk * dx

def try_multiple_N(h_best, N_best, deadline, candidates=None):
    """Try different N values by resampling."""
    if candidates is None:
        candidates = [N_best - 10, N_best, N_best + 10, N_best + 20, N_best + 40]
    
    x_best = np.linspace(0, 2, N_best, endpoint=False)
    results = []
    for N_try in candidates:
        if time.time() > deadline - 10:
            break
        if N_try < 100:
            continue
        if N_try % 2 != 0:
            N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, h_best)
        h_bt = (h_interp > 0.5).astype(float)
        nh = N_try // 2
        n1 = int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            idx = idx[np.random.permulation(len(idx))][nh:]
            h_bt[idx] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh - n1]
            h_bt[idx] = 1
        results.append((compute_c5(h_bt, N_try), h_bt.copy(), N_try))

def nesterov_lse(h, N, deadline, lr0=0.01, beta_mom=0.9, lse_beta=20.0):
    """Continuous optimization with log-sum-exp smooth max."""
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_score = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    it = 0
    center = N - 1
    
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        score = float(corr.max())
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        # Log-sum-exp gradient
        corr_shifted = corr - corr.max()
        weights = np.exp(lse_beta * corr_shifted)
        weights /= weights.sum()  # softmax-like weights
        
        # Gradient: sum over lags of weights[lag] * d(corr[lag])/d(h)
        grad = np.zeros(N)
        # For efficiency, only use lags with significant weight
        top_idx = np.where(weights > 0.01 / N)[0]
        n_active = len(top_idx)
        if n_active == 0:
            top_idx = npargmax(corr)
            n_active = 1
        
        for ai in top_idx:
            w = weights[ai]
            L = int(ai - center)
            if L == 0:
                grad += w * (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += w * (1.0 - y[:N - L])
                grad[:N - L] -= w * y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += w * (1.0 - y[M:])
                    grad[M:] -= w * y[:N - M]
        
        h_new = project_sum(y - lr * grad, target)
        m = beta_mom * (h_new - h)
        h = h_new
        it += 1
        if it % 50 == 0:
            lr *= 0.99
            lse_beta = max(5.0, lse_beta * 0.98)  # gradually sharpen
    
    return best_h, best_score * (2.0 / N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

    lo, hi = -1.0, 1.0

        else:
