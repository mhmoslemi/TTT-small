    return float(np.max(corr) * dx)

def project_h(h):
    """Project to [0,1]^n with sum = n/2 using iterative water-filling."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (1.0 + diff / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_max(h, T):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = 2 * n - 1
    Tc = T * (corr - np.max(corr))
    w = np.exp(Tc)
    w /= w.sum()
    grad = np.zeros(n)
    lag_arr = np.arange(m) - n + 1
    for idx in range(m):
        if w[idx] < 1e-14:
            continue
        lg = lag_arr[idx]
        wi = w[idx] * dx
        if lg >= 0:

            if p < n:

            if p > 0:

            p = -lg
            if p < n:
                grad[p:] += wi * (1.0 - h[:n-p])
                grad[:n-p] -= wi * h[p:]

    x = h.copy()
    v = np.zeros(len(h))
    for step in range(n_steps):
        if time.time() > t_end:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(40):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if d > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (d / tot)
        else:
            if s < 1e-14:
                break
            h *= (1.0 + d / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = 2 * n - 1
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    lag_arr = np.arange(m) - n + 1
    for idx in range(m):
        wi = w[idx]
        if wi < 1e-15:
            continue
        lg = lag_arr[idx]
        if lg >= 0:
            p = lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wi * h[:n-p]
        else:
            p = -lg
            if p < n:
                grad[p:] += wi * (1.0 - h[:n-p])
                grad[:n-p] -= wi * h[p:]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def f(x):
        return obj_grad(x, T, lam)[0]
    def g(x):
        return obj_grad(x, T, lam)[1]
    try:
        res = minimize(f, x0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return x0

def block_swap(h, rng):
    """Swap two contiguous random blocks."""
    n = len(h)
    if n < 4:
        return h.copy()
    min_len = max(2, n // 32)
    max_len = max(min_len + 1, n // 4)
    lt = min_len + rng.integers(0, max_len - min_len)
    lb = lt + rng.integers(0, max(1, n - 2 * lt))
    lt2 = min_len + rng.integers(0, max_len - min_len)
    lb2 = lt2 + rng.integers(0, max(1, n - lt - lt2))
    if lb + lt <= lb2 or lb2 + lt2 <= lb:
        h2 = h.copy()
        h2[lb:lb+lt], h2[lb2:lb2+lt2] = h[lb2:lb2+lt2], h[lb:lb+lt]
        return project_h(h2)
    return h.copy()

def block_shift(h, rng):
    """Shift a block by a few positions."""
    n = len(h)
    blen = max(2, n // 20) + rng.integers(0, n // 5)
    start = rng.integers(0, max(1, n - blen - 4))
    shift = rng.integers(-3, 4)
    if shift == 0:
        return h.copy()
    s = max(0, start + shift)
    e = min(n, start + blen + shift)
    if e - s < blen // 2:
        return h.copy()
    h2 = h.copy()
    tmp = h2[s:e].copy()
    end_orig = start + blen
    h2[start:end_orig] = h2[end_orig + shift if shift > 0 else s:start+blen+shift if shift < 0 else start:end_orig]
    # Simpler: just copy the block to new location
    h2 = h.copy()
    block = h[start:start+blen].copy()
    ns = start + shift
    ne = ns + blen
    if ns < 0 or ne > n:
        return h.copy()
    h2[ns:ne] = block
    # Fill the gap
    if shift > 0:
        h2[ns:start] = h2[start:start+(start-ns)]
    else:
        h2[start+blen+shift:start+blen] = h2[start+blen:start+blen+(-shift)]
    return project_h(h2)

def coord_descent(h, t_end, max_sweeps=500):
    n = len(h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h0 = best_h.copy()
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            def f1d(x):
                return np.max(cb + x * lin + x * x * quad)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5
