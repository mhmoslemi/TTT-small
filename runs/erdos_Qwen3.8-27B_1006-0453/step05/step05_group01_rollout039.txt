import numpy as np
from scipy.optimize import minimize, minimize_scalar
from scipy.linalg import solve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T, target_C=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if target_C is not None and m > target_C:
        # Heavy penalty for exceeding target
        pen = 1000.0 * (m - target_C) ** 2
    else:
        pen = 0.0
    if T > 1e6:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val + pen

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h + diff/n, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.clip(h + diff/n, 0, 1)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, target_C=None, maxiter=1000):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(lambda x: smooth_obj(x, T, target_C), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if T > 1e8:
        return m
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h + diff/n, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.clip(h + diff/n, 0, 1)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=800):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(lambda x: smooth_obj(x, T), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

def solve_1d_minimax(consts, coefs):
    """Solve min_{t in [0,1]} max_j (consts[j] + coefs[j]*t)"""
    if len(coefs) == 0:
        return 0.0
    if np.max(np.abs(coefs)) < 1e-15:
        return 0.0
    cands = [0.0, 1.0]
    m = len(coefs)
    # Find all pairwise intersections in [0,1]
    for a in range(m):
        for b in range(a+1, m):
            dc = coefs[b] - coefs[a]
            if abs(dc) < 1e-15:
                continue
            t_int = (consts[a] - consts[b]) / dc
            if -1e-10 <= t_int <= 1.0 + 1e-10:
                cands.append(np.clip(t_int, 0, 1))
    # Also add points where individual lines are at their "knee"
    # (not needed for linear functions, but for numerical safety)
    best_t = 0.0
    best_val = np.inf
    for tc in cands:
        val = np.max(consts + tc * coefs)
        if val < best_val:
            best_val = val
            best_t = tc
    return best_t, best_val

def coord_descent_vec(h, n, t_end, max_sweeps=200, n_top=40):
    """Vectorized coordinate descent over top shifts."""

def solve_1d_minimax_fast(consts, coefs):
    n = len(coefs)
    if n == 0:
        return 0.0
    # All pairwise intersections
    dc = coefs[None, :] - coefs[:, None]  # n x n
    da = consts[:, None] - consts[None, :]  # n x n
    # t_int = da / dc where |dc| > eps
    mask = np.abs(dc) > 1e-14
    with np.errstate(divide='ignore', invalid='ignore'):
        t_ints = np.where(mask, da / dc, -1)
    valid = mask & (t_ints >= 0) & (t_ints <= 1)
    candidates = t_ints[valid]
    candidates = np.concatenate([candidates, [0.0, 1.0]])
    # Find best
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        val = np.max(consts + t * coefs)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if T > 1e8:
        return m
    return float(m + np.log(np.sum(np.exp(T * (corr - m)))) / T)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h + diff / n, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.clip(h + diff / n, 0, 1)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=800):
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(lambda x: smooth_obj(x, T), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def solve_1d_minimax(consts, coefs):
    n = len(coefs)
    if n == 0 or np.max(np.abs(coefs)) < 1e-14:
        return 0.0
    dc = coefs[None, :] - coefs[:, None]
    da = consts[:, None] - consts[None, :]
    mask = np.abs(dc) > 1e-14
    with np.errstate(divide='ignore', invalid='ignore'):

    cands = t_ints[valid]
    cands = np.concatenate([cands, np.array([0.0, 1.0])])
    cands = np.clip(cands, 0.0, 1.0)
    vals = np.array([np.max(consts + t * coefs) for t in cands])
    return float(cands[np.argmin(vals)])

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        nt = min(n_top, 2 * n - 1)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        top_idx = np.argsort(corr)[-nt:]
        top_c = corr[top_idx].copy()
        top_k = ks[top_idx]
        improved = False

            coefs = np.zeros(nt)
            idx_m = np.where(vm)[0]
            idx_p = np.where(vp)[0]
            coefs[idx_m] += (1.0 - best_h[i - top_k[idx_m]])
            coefs[idx_p] -= best_h[i + top_k[idx_p]]
            consts = top_c - best_h[i] * coefs
            t_opt = solve_1d_minimax(consts, coefs)

    return best_h, best_c5
