```python
    except:
        return h
```

```python
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(corr)[-K:]
    top_vals = corr[top_idx]
    obj = np.sum(top_vals**2) / K
    # Gradient
    shifts = top_idx - (n - 1)
    grad = np.zeros(n)
    for t in range(K):

        ip = np.arange(n) + s

        vm = (im >= 0) & (im < n)
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            s2 = h.sum()
            if s2 < 1e-14:
                break
            h += diff * h / s2
        h = np.clip(h, 0, 1)
    return h

def get_shift_coefs(h, shifts, n):
    coefs = np.zeros((len(shifts), n))
    idx = np.arange(n)
    for t, s in enumerate(shifts):
        ip = idx + s
        im = idx - s
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        coefs[t, vp] += (1.0 - h[ip[vp]])
        coefs[t, vm] -= h[im[vm]]
    return coefs

def coord_descent(h, n, t_end, n_top=60, max_sweeps=300, reid_interval=8):
    best_h = h.copy()
    best_c5, corr = compute_c5(best_h)
    top_idx = np.argsort(corr)[-n_top:]
    shifts = (top_idx - (n - 1)).astype(int)
    coefs = get_shift_coefs(best_h, shifts, n)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if sweep % reid_interval == 0:
            _, corr = compute_c5(best_h)
            top_idx = np.argsort(corr)[-n_top:]
            shifts = (top_idx - (n - 1)).astype(int)
            coefs = get_shift_coefs(best_h, shifts, n)
        
        order = np.random.permutation(n)
        improved = 0
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            base = corr[top_idx] - best_h[i] * coefs[:, i]
            res = minimize_scalar(
                lambda t: np.max(base + t * coefs[:, i]),
                bounds=(0.0, 1.0), method='bounded',
                options={'xatol': 8e-11})
            if abs(res.x - best_h[i]) < 5e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = res.x
            h_try = project_h(h_try)
            c_try, new_corr = compute_c5(h_try)
            if c_try < best_c5 - 5e-16:
                best_c5 = c_try
                best_h = h_try
                corr = new_corr
                improved += 1
        if improved == 0:
            break
    return best_h, best_c5

def smooth_topk_obj(h, top_idx, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    vals = corr[top_idx]
    m = vals.max()
    return m + T * np.log(np.sum(np.exp((vals - m) / T)))

def projected_grad_step(h, n, t_end, K=40, n_steps=200, lr_init=0.004):
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    mom = np.zeros(n)
    beta = 0.85
    
    for step in range(n_steps):
        if time.time() > t_end:
            break
        cool = 1.0 - step / max(n_steps, 1)
        lr = lr_init * (0.15 + 0.85 * cool)
        _, corr = compute_c5(h)
        top_idx = np.argsort(corr)[-K:]
        shifts = (top_idx - (n - 1)).astype(int)
        vals = corr[top_idx]
        w = np.exp((vals - vals.max()) / (0.005 + 0.02 * cool))
        w /= w.sum()
        coefs = get_shift_coefs(h, shifts, n)
        grad = (w @ coefs) * (2.0 / n)
        mom = beta * mom + (1 - beta) * grad
        h_new = project_h(h - lr * mom)
        c_try, _ = compute_c5(h_new)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_new
        h = h_new
    return best_h, best_c5

def small_perturb_reopt(h, n, t_end, scale=0.03, n_cd=80, n_top=60):
    h_p = project_h(h + scale * np.random.randn(n))
    hc, c5c = coord_descent(h_p, n, t_end, n_top=n_top, max_sweeps=n_cd)
    return hc, c5c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Quick projected gradient to shape the solution
    t1 = min(t0 + budget_s * 0.12, t_end - 30)
    if t1 > time.time():
        hg, cg = projected_grad_step(best_h, n, t1, K=50, n_steps=150, lr_init=0.005)
        if cg < best_c5:
            best_c5, best_h = cg, hg

    # Phase 2: Main coordinate descent (bulk of budget)
    t2 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, n, t2, n_top=60, max_sweeps=400, reid_interval=6)

    # Phase 3: Interleaved grad + coord
    t3 = t0 + budget_s * 0.72
    if t3 > time.time():
        for round in range(4):
            if time.time() > t3:
                break
            t_sub = min(time.time() + 4, t3)
            hg, cg = projected_grad_step(best_h, n, t_sub, K=40, n_steps=40, lr_init=0.003)
            if cg < best_c5:
                best_c5, best_h = cg, hg
            best_h, best_c5 = coord_descent(best_h, n, min(time.time() + 8, t3),
                                            n_top=70, max_sweeps=60, reid_interval=5)

    # Phase 4: Random restarts
    t4 = t0 + budget_s * 0.88
    for trial in range(4):
        if time.time() > t4:
            break
        scale = 0.04 * (1.0 - trial * 0.15)
        hc, c5c = small_perturb_reopt(best_h, n, min(time.time() + (t4 - time.time()) / (4 - trial), t4),
                                       scale=scale, n_cd=80, n_top=60)
        if c5c < best_c5:
            best_c5, best_h = c5c, hc

    # Phase 5: Final long coordinate descent with more shifts
    t5 = t_end - 15
    if t5 > time.time():
        best_h, best_c5 = coord_descent(best_h, n, t5, n_top=100, max_sweeps=300, reid_interval=5)

    # Phase 6: Final micro-polish
    if time.time() < t_end - 3:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 2, n_top=120, max_sweeps=100, reid_interval=4)

    best_h = project_h(best_h)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
