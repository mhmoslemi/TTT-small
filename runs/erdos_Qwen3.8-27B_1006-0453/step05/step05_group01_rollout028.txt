    h1, c5_1 = lp_iterate(best_h, n, n_iter=500, damping=0.65, t_end=t_p1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Alternative starts
    starts = []
    starts.append(1.0 - best_h)
    starts.append(best_h[::-1])
    starts.append(best_h + 0.15 * np.random.randn(n))
    starts.append(np.tile(np.array([0.6, 0.4, 0.7, 0.3, 0.5, 0.5]), n // 6))
    if n % 6:
        starts[-1] = np.resize(starts[-1], n)
    starts.append(project_h(best_h[np.random.permutation(n)]))
    for si in range(len(starts)):
        if time.time() > t_end - budget_s * 0.35:
            break
        h_s = project_h(starts[si].copy())
        t_sub = min(time.time() + budget_s * 0.035, t_end - budget_s * 0.3)
        h_try, c5_try = lp_iterate(h_s, n, n_iter=150, damping=0.7, t_end=t_sub)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try

    # Phase 3: SLSQP polish (multiple T)
    for T in [30, 100, 500, 2000]:
        if time.time() > t_end - 80:
            break
        h_pol = slsqp_smooth(best_h, n, T, 700)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5, best_h = c5_pol, h_pol

    # Phase 4: Coordinate descent
    if time.time() < t_end - 50:
        t_cd = min(time.time() + budget_s * 0.08, t_end - 25)
        best_h, best_c5 = coord_descent(best_h, n, t_cd, 70, 20)

    # Phase 5: Try finer grid
    if time.time() < t_end - 150 and n < 400:
        n2 = min(2 * n, 400)
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        t_sub2 = min(time.time() + 50, t_end - 80)
        h2_lp, c5_2 = lp_iterate(h2, n2, n_iter=100, damping=0.6, t_end=t_sub2)
        if time.time() < t_end - 40:
            h2_s = slsqp_smooth(h2_lp, n2, 200, 500)
            c5_2s = compute_c5(h2_s)
            if c5_2s < c5_2:
                h2_lp, c5_2 = h2_s, c5_2s
        if c5_2 < best_c5:
            best_c5, best_h, n = c5_2, h2_lp, n2

    # Phase 6: Another LP round + final polish
    if time.time() < t_end - 40:
        h6, c5_6 = lp_iterate(best_h, n, n_iter=80, damping=0.75, t_end=t_end - 30)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6
    if time.time() < t_end - 15:
        h_fin = slsqp_smooth(best_h, n, 5000, 800)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin
    if time.time() < t_end - 5:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, 25, 25)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
