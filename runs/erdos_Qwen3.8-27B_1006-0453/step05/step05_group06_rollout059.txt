from scipy.optimize import linprog
import numpy as np

def sca_step(h0, n):
    # Compute linearized overlap constraints
    # overlap(s) ≈ sum_{j=s}^{n-1} h[j] - sum_{j=s}^{n-1}(h0[j]*h[j-s] + h[j]*h0[j-s] - h0[j]*h0[j-s])
    #           = sum_{j=s}^{n-1} h[j] - [sum_j h0[j]*h[j-s] + sum_j h[j]*h0[j-s] - const_s]
    
    # Let's write overlap(s) as a linear function of h:
    # overlap(s) = sum_{j=s}^{n-1} h[j] - sum_{i=0}^{n-1-s} [h0[i+s]*h[i] + h[i+s]*h0[i] - h0[i+s]*h0[i]]
    #            = sum_{j=s}^{n-1} h[j] - sum_{i=0}^{n-1-s} h0[i+s]*h[i] - sum_{i=0}^{n-1-s} h[i+s]*h0[i] + sum_{i=0}^{n-1-s} h0[i+s]*h0[i]
    
    # Coefficient of h[j] in overlap(s):
    # From first term: +1 if j >= s
    # From second term: -h0[i+s] where i = j, so -h0[j+s] if j < n-s (i.e., j+s < n)
    # From third term: -h0[i] where i+s = j, i.e., i = j-s, so -h0[j-s] if j >= s
    # From fourth term: constant
    
    # So coeff of h[j] in overlap(s):
    # c_j(s) = (1 if j>=s else 0) - (h0[j+s] if j < n-s else 0) - (h0[j-s] if j>=s else 0)
    # const(s) = sum_{i=0}^{n-1-s} h0[i+s]*h0[i]
    
    # overlap(s) = sum_j c_j(s) * h[j] + const(s)
    
    # We want: overlap(s) * dx <= t  =>  sum_j c_j(s)*h[j] <= t/dx - const(s)
    
    # LP: minimize t
    # subject to: c(s)·h - t/dx <= -const(s) for all s
    #             sum h = n/2
    #             0 <= h <= 1
    
    n_var = n + 1  # h[0..n-1] and t
    c = np.zeros(n_var)
    c[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((n, n_var))
    b_ub = np.zeros(n)
    
    for s in range(n):
        const_s = 0.0
        for i in range(n - s):
            const_s += h0[i+s] * h0[i]
        for j in range(n):
            cj = 0.0
            if j >= s:
                cj += 1.0
            if j + s < n:
                cj -= h0[j+s]
            if j >= s:
                cj -= h0[j-s]
            A_ub[s, j] = cj
        A_ub[s, -1] = -1.0
        b_ub[s] = -const_s  # c(s)·h - t <= -const(s), so c(s)·h + (-1)*t <= -const(s)
    
    # Equality: sum h = n/2
    A_eq = np.zeros((1, n_var))
    for j in range(n):
        A_eq[0, j] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return h0, np.inf

def compute_c5(h):

def project_h(h):

def softmax_obj_grad(h, T, lam):

import numpy as np
from scipy.optimize import minimize, linprog, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    ov = np.zeros(n)
    for s in range(n):
        ov[s] = np.dot(h[s:], 1.0 - h[:n - s])
    return ov * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def softmax_vals(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n - lg])
                grad[:n - lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n - p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    def _obj(x):
        return softmax_vals(x, T, lam)[0]
    def _grad(x):
        return softmax_vals(x, T, lam)[1]
    try:
        res = minimize(_obj, x0, jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-13})
        return project_h(res.x)
    except Exception:
        return x0

def sca_lp(h0):
    n = len(h0)
    n_var = n + 1
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    A_ub = np.zeros((n, n_var))
    b_ub = np.zeros(n)
    for s in range(n):
        const_s = 0.0
        for i in range(n - s):
            const_s += h0[i + s] * h0[i]
        for j in range(n):
            cj = 0.0
            if j >= s:
                cj += 1.0 - h0[j - s]
            if j + s < n:
                cj -= h0[j + s]
            A_ub[s, j] = cj
        A_ub[s, -1] = -1.0
        b_ub[s] = -const_s
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n].copy(), res.x[-1]
    return h0.copy(), np.inf

def coord_descent_v2(h, t_end, max_sweeps=300):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        current_ov = compute_overlaps(best_h)
        improved = False
        for j in range(n):
            slopes = np.zeros(n)
            ms1 = min(j, n - 1)
            if ms1 >= 0:
                slopes[:ms1 + 1] += (1.0 - best_h[j - ms1:j + 1][::-1])
            ms2 = n - 1 - j
            if ms2 >= 0:
                slopes[:ms2 + 1] -= best_h[j:j + ms2 + 1]
            hj_save = best_h[j]
            def f1d(x):
                return float(np.max(current_ov + slopes * (x - hj_save)))
            try:
                r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                    options={'xatol': 1e-11})
            except Exception:
                continue
            x_new = r.x
            if abs(x_new - hj_save) < 1e-11:
                continue
            ht = best_h.copy()
            ht[j] = x_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                current_ov = compute_overlaps(best_h)
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS warmup
    p1_end = t0 + budget_s * 0.10
    for T in [500, 2000, 8000]:
        if time.time() > p1_end:
            break
        hc = lbfgs_opt(best_h, T, 500.0, 3000, 1e-18)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: SCA with LP
    p2_end = t0 + budget_s * 0.30
    h_sca = best_h.copy()
    for it in range(60):
        if time.time() > p2_end:
            break
        h_lp, t_lp = sca_lp(h_sca)
        if not np.isfinite(t_lp):
            break
        c5_cur = compute_c5(h_sca)
        improved_sca = False
        for alpha in [0.7, 0.4, 1.0, 0.2]:
            h_new = project_h((1 - alpha) * h_sca + alpha * h_lp)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_sca = h_new
                c5_cur = c5_new
                improved_sca = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_sca.copy()
                break
        if not improved_sca:
            break

    # Phase 3: L-BFGS refinement with high T
    p3_end = t0 + budget_s * 0.52
    for T in [8000, 50000, 300000]:
        if time.time() > p3_end:
            break
        hc = lbfgs_opt(best_h, T, 800.0, 6000, 1e-18)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Focused coordinate descent
    p4_end = t0 + budget_s * 0.72
    best_h, best_c5 = coord_descent_v2(best_h, p4_end, max_sweeps=400)

    # Phase 5: Basin hopping
    p5_end = t0 + budget_s * 0.88
    for it in range(50):
        if time.time() > p5_end:
            break
        cool = 1.0 - 0.9 * (it / 50.0)
        sc = 0.003 + 0.015 * cool
        h_p = project_h(best_h + sc * np.random.randn(n))
        T_bh = int(8000 * cool + 800)
        hc = lbfgs_opt(h_p, T_bh, 800.0, 2000, 1e-17)
        c5v = compute_c5(hc)
        temp = 0.0002 * cool + 0.00003
        if c5v < best_c5 or np.random.rand() < np.exp(-(c5v - best_c5) / max(temp, 1e-12)):
            best_h = hc
            best_c5 = c5v

    # Phase 6: Final SCA + L-BFGS + coord polish
    p6_end = t0 + budget_s * 0.97
    h_fin = best_h.copy()
    for it in range(40):
        if time.time() > p6_end:
            break
        h_lp, t_lp = sca_lp(h_fin)
        if not np.isfinite(t_lp):
            break
        c5_cur = compute_c5(h_fin)
        done = False
        for alpha in [0.5, 0.3, 1.0]:
            h_new = project_h((1 - alpha) * h_fin + alpha * h_lp)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_fin = h_new
                done = True
                break
        if not done:
            break
    c5v = compute_c5(h_fin)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h_fin

    hc = lbfgs_opt(best_h, 500000, 1000.0, 6000, 1e-18)
    c5v = compute_c5(hc)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc

    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent_v2(best_h, t0 + budget_s * 0.99, max_sweeps=200)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
