            corr[n-1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new <= best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n-1] += d * d
        if not improved:
            break
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def grad_focused(h, n, top_k=8):
    """Subgradient focusing on top-k active shifts."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    thresh = c_max - 0.02 * (c_max - np.min(corr)) + 1e-12
    active = np.where(corr >= thresh)[0]
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    grad = np.zeros(n)
    for m_idx in active:
        wi = w[m_idx]
        s = m_idx - (n - 1)
        if s > 0:
            grad[:n-s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n-s]
        elif s < 0:
            a = -s
            grad[a:] += wi * (1.0 - h[:n-a])
            grad[:n-a] -= wi * h[a:]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return dx * grad

def proj_grad(h, n, t_end, max_iters=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    mom = np.zeros(n)
    beta = 0.6
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = min(1.0, it / max(1, max_iters))
        g = grad_focused(h_c, n, top_k=max(4, int(12 * (1 - frac))))
        mom = beta * mom + g
        gm = np.max(np.abs(mom))
        if gm < 1e-14:
            break
        alpha = 0.3 / (gm + 1e-10)
        h_new = project_h(h_c - alpha * mom)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c5

def make_binary_pattern(n, rng):
    """Generate a random balanced binary pattern."""
    h = np.zeros(n)
    ones = np.sort(rng.choice(n, size=n//2, replace=False))
    h[ones] = 1.0
    return h

def upsample(h, factor):
    """Upsample by repeating values."""
    return np.repeat(h, factor)

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - hop / max(n_hops, 1))
        step = 0.01 + 0.08 * cool
        mode = hop % 7
        pert = np.zeros(n)
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            start = rng.randint(0, max(1, n - 10))
            length = rng.randint(3, max(4, n // 5))
            length = min(length, n - start)
            pert[start:start+length] = step * rng.choice([-1, 1], size=length).astype(float)
        elif mode == 3:
            idx = rng.choice(n, size=max(4, n // 8), replace=False)
            pert[idx] = step * 1.5 * rng.randn(len(idx))
        elif mode == 4:
            # Block swap: flip a block to its complement
            start = rng.randint(0, max(1, n - 8))
            length = rng.randint(4, max(5, n // 6))
            length = min(length, n - start)
            pert[start:start+length] = 1.0 - 2.0 * cur_h[start:start+length]
            pert[start:start+length] *= min(1.0, step * 3)
        elif mode == 5:
            # Rotational perturbation
            shift = rng.randint(1, max(2, n // 10))
            pert = step * 0.3 * (np.roll(cur_h, -shift) - cur_h)
        else:
            # Gaussian mixture perturbation
            n_centers = rng.randint(2, 6)
            centers = rng.randint(0, n, size=n_centers)
            widths = rng.uniform(2, n//4, size=n_centers)
            for c, w in zip(centers, widths):
                pert += step * rng.randn(1) * np.exp(-0.5 * ((np.arange(n) - c) / w)**2)
        h_pert = project_h(cur_h + pert)
        hop_time = max(1.5, 6.0 * cool)
        hop_end = min(t_end, time.time() + hop_time)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=100, seed=seed + hop)
        temp = 0.003 * cool + 0.0002
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h, cur_c5 = h_opt, c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def multi_n_search(t_end, seed=42):
    """Try multiple n values and return best (h, c5, n)."""
    rng = np.random.RandomState(seed)
    best_h, best_c5, best_n = None, 1.0, 160
    n_list = [80, 100, 120, 140, 200, 240]
    per_n_time = max(3.0, (t_end - time.time()) / len(n_list) * 0.7)
    for n in n_list:
        t_n = time.time() + per_n_time
        if time.time() > t_end:
            break
        # Try 3 random binary starts
        for try_i in range(3):
            if time.time() > t_n:
                break
            h0 = make_binary_pattern(n, rng)
            t_cd = min(t_n, time.time() + per_n_time / 4)
            h_opt, c5_opt = coord_descent(h0, n, t_cd, max_sweeps=300, seed=seed + try_i)
            if c5_opt < best_c5:
                best_c5, best_h, best_n = c5_opt, h_opt.copy(), n
    return best_h, best_c5, best_n

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    # Phase 1: Load initial and quick polish
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = make_binary_pattern(160, np.random.RandomState(seed))
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 2: Quick CD + gradient on initial
    t_phase2 = t0 + budget_s * 0.15
    best_h, best_c5 = coord_descent(best_h, n, t_phase2, max_sweeps=1500, seed=seed)

    # Phase 3: Multi-n exploration
    t_phase3 = t0 + budget_s * 0.35
    h_mn, c5_mn, n_mn = multi_n_search(t_phase3, seed=seed + 1000)
    if h_mn is not None and c5_mn < best_c5:
        best_h, best_c5, n = h_mn, c5_mn, n_mn

    # Gradient polish on best
    if time.time() < t0 + budget_s * 0.40:
        hg, c5g = proj_grad(best_h, n, t0 + budget_s * 0.39, max_iters=1000)
        if c5g < best_c5:
            best_h, best_c5 = hg, c5g
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.42, max_sweeps=500, seed=seed+1)

    # Phase 4: Basin hopping (main search)
    t_phase4 = t0 + budget_s * 0.70
    best_h, best_c5 = basin_hop(best_h, n, t_phase4, n_hops=250, seed=seed + 50)

    # Phase 5: Intensive final optimization
    if time.time() < t0 + budget_s * 0.85:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.83, max_sweeps=5000, seed=seed + 200)
        hg2, c5g2 = proj_grad(best_h, n, t0 + budget_s * 0.85, max_iters=500)
        if c5g2 < best_c5:
            best_h, best_c5 = hg2, c5g2

    # Phase 6: Second basin hop from improved point
    if time.time() < t0 + budget_s * 0.92:
        best_h2, best_c5_2 = basin_hop(best_h, n, t0 + budget_s * 0.92, n_hops=100, seed=seed + 500)
        if best_c5_2 < best_c5:
            best_h, best_c5 = best_h2, best_c5_2

    # Phase 7: Final intensive CD
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=8000, seed=seed + 999)

    # Final polish
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
