def golden_min(f, a, b, tol=1e-12, max_iter=200):
    """Golden section minimization of f on [a,b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return 0.5 * (a + b)

    t1 = t0 + budget_s * 0.25
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=5000, seed=seed)
    
    # Phase 2: Basin hopping
    t2 = t0 + budget_s * 0.65
    h2, c5_2 = basin_hop(best_h, n, t2, seed=seed+1)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2
    
    # Phase 3: Final polish
    t3 = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, n, t3, max_sweeps=5000, seed=seed+100)
    
    return best_h, best_c5, n

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(50):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + room * (d / tot)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h * (target / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_1d(v, c_base, lin, old, n, dx):
    dval = v - old
    val = c_base + dval * lin
    val[n - 1] -= dval * dval
    return float(np.max(val) * dx)

def golden_min_1d(a, b, c_base, lin, old, n, dx, tol=1e-13):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = eval_1d(c, c_base, lin, old, n, dx)
    fd = eval_1d(d, c_base, lin, old, n, dx)
    for _ in range(80):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_1d(c, c_base, lin, old, n, dx)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = eval_1d(d, c_base, lin, old, n, dx)
    return 0.5 * (a + b)

def coord_descent(h, n, t_end, max_sweeps=5000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if sweep % 4 == 2:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                    lin[mn] -= (1.0 - best_h[i - s])
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                    lin[mn] += best_h[i + s]
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            v_new = golden_min_1d(0.0, 1.0, c_base, lin, old, n, dx)
            dval = v_new - old
            if abs(dval) < 1e-14:
                continue
            best_h[i] = v_new
            corr += dval * lin
            corr[n - 1] -= dval * dval
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= dval * lin
                corr[n - 1] += dval * dval
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def perturb_h(h, n, mode, rng, amplitude):
    h_p = h.copy()
    if mode == 0:
        h_p += amplitude * rng.randn(n)
    elif mode == 1:
        f = rng.randint(1, 15)
        ph = rng.uniform(0, 2 * np.pi)
        h_p += amplitude * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif mode == 2:
        idx = rng.choice(n, size=max(3, n // 8), replace=False)
        h_p[idx] += amplitude * 1.5 * rng.randn(len(idx))
    elif mode == 3:
        lo = rng.randint(0, n - 10)
        hi = lo + rng.randint(4, 12)
        h_p[lo:hi] = 1.0 - h_p[lo:hi]
    elif mode == 4:
        k = rng.randint(1, 6)
        h_p = np.roll(h_p, k)
    else:
        i1 = rng.randint(0, n)
        i2 = rng.randint(0, n)
        tmp = h_p[i1]
        h_p[i1] = h_p[i2]
        h_p[i2] = tmp
        h_p += amplitude * 0.3 * rng.randn(n)
    return project_h(h_p, n)

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - float(hop) / max(n_hops, 1)
        amp = 0.01 + 0.08 * cool
        mode = hop % 6
        h_p = perturb_h(cur_h, n, mode, rng, amp)
        hop_time = max(1.5, 6.0 * cool + 1.0)
        t_hop = min(t_end, time.time() + hop_time)
        h_opt, c5_opt = coord_descent(h_p, n, t_hop, max_sweeps=150, seed=seed + hop * 7)
        temp = 0.003 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5

def grad_step(h, n, t_end, max_iters=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    mom = np.zeros(n)
    beta = 0.75
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = min(1.0, float(it) / max(max_iters, 1))
        T = 30.0 * 50.0 ** frac
        corr = np.correlate(h_c, 1.0 - h_c, mode='full')
        m = np.max(corr)
        exponents = np.clip(T * (corr - m), -60.0, 60.0)
        p = np.exp(exponents)
        ps = p.sum()
        if ps < 1e-15:
            break
        p /= ps
        grad = np.zeros(n)
        for m_idx in range(2 * n - 1):
            w = p[m_idx]
            if w < 1e-14:
                continue
            s = m_idx - (n - 1)
            if s > 0:
                grad[s:] += w * (1.0 - h_c[:n - s])
                grad[:n - s] -= w * h_c[s:]
            elif s < 0:
                a = -s
                grad[:n - a] += w * (1.0 - h_c[a:])
                grad[a:] -= w * h_c[:n - a]
            else:
                grad += w * (1.0 - 2.0 * h_c)
        g = dx * grad
        mom = beta * mom + g
        gm = np.max(np.abs(mom))
        if gm < 1e-14:
            break
        alpha = 0.5 / gm
        h_new = project_h(h_c - alpha * mom, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c5

def try_other_n(h_src, n_src, n_new, t_end, seed):
    x_old = np.linspace(0, 2, n_src + 1)
    y_old = np.concatenate([h_src, [h_src[-1]]])
    x_new = np.linspace(0, 2, n_new + 1)
    y_new = np.interp(x_new[:-1], x_old, y_old)
    y_new = project_h(y_new, n_new)
    h_opt, c5_opt = coord_descent(y_new, n_new, t_end, max_sweeps=2000, seed=seed)
    return h_opt, c5_opt, n_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
