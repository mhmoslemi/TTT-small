    """Optimize coordinate i by finding min of max of top lags."""
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    
    # Compute coefficient for each lag
    coefs = np.zeros(2*n - 1)
    # h[i] contributes to corr at various lags
    # For lag ell: h[i]*(1-h[i+ell]) + h[i-ell]*(1-h[i])
    # First term: (1-h[i+ell]) if i+ell valid
    # Second term: -h[i-ell] if i-ell valid
    for ell in range(-(n-1), n):
        c = 0.0
        if 0 <= i + ell < n:
            c += (1.0 - h[i + ell])
        if 0 <= i - ell < n:
            c += (-h[i - ell])
        coefs[ell + n - 1] = c
    
    # f(t) = max_k (corr_base[k] + t * coefs[k])
    # Find min over t in [0,1]
    # Use top 30 lags for efficiency
    top_idx = np.argsort(corr_base)[-30:]
    vals = corr_base[top_idx]
    coef_top = coefs[top_idx]
    
    # Candidate points: 0, 1, and breakpoints
    candidates = [0.0, 1.0]
    for a in range(len(vals)):
        for b in range(a+1, len(vals)):
            if abs(coef_top[a] - coef_top[b]) < 1e-15:
                continue
            t = (vals[b] - vals[a]) / (coef_top[a] - coef_top[b])
            if 0 <= t <= 1:
                candidates.append(t)
    
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        f = np.max(vals + t * coef_top)
        # But we should check against ALL lags, not just top 30
        f_full = np.max(corr_base + t * coefs)
        if f_full < best_val:
            best_val = f_full
            best_t = t
    
    return best_t

        best_c5 = bc5
        best_h = bh.copy()
    for it in range(max_iter):
        if time.time() > t_end:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        bh[i] = 0.0
        bh[j] = 1.0
        nc = compute_c5(bh)
        if nc < bc5:
            bc5 = nc
            ones = np.where(bh == 1.0)[0]
            zeros = np.where(bh == 0.0)[0]
            if bc5 < best_c5:
                best_c5 = bc5
                best_h = bh.copy()
        else:
            bh[i] = 1.0
            bh[j] = 0.0
    return best_h, best_c5

def coord_opt_smart(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    coefs = np.zeros(2 * n - 1)
    for ell in range(-(n - 1), n):
        c = 0.0
        if 0 <= i + ell < n:
            c += (1.0 - h[i + ell])
        if 0 <= i - ell < n:
            c += (-h[i - ell])
        coefs[ell + n - 1] = c
    top_idx = np.argsort(corr_base)[-15:]
    vals = corr_base[top_idx].copy()
    ctop = coefs[top_idx].copy()
    candidates = [0.0, 1.0]
    for a in range(len(vals)):
        for b in range(a + 1, len(vals)):
            denom = ctop[a] - ctop[b]
            if abs(denom) < 1e-15:
                continue
            t = (vals[b] - vals[a]) / denom
            if -0.01 <= t <= 1.01:
                candidates.append(np.clip(t, 0.0, 1.0))
    best_t = 0.0
    best_v = np.inf
    for t in candidates:
        fv = np.max(corr_base + t * coefs)
        if fv < best_v:
            best_v = fv
            best_t = t
    return best_t

def coord_descent(h, n, t_end, max_sweeps=60):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_lags = np.argsort(corr)[-10:]
        impact = np.zeros(n)
        for ki in top_lags:
            lag = ki - (n - 1)
            impact += np.abs(grad_corr_lag(best_h, n, lag))
        order = np.argsort(-impact)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_smart(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = project_h(ht, n)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    alts = []
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    alts.append(project_h(tm, n))
    for _ in range(3):
        rb = np.zeros(n)
        rb[np.random.choice(n, n // 2, replace=False)] = 1.0
        alts.append(rb)
    for alt in alts:
        if elapsed() > 15:
            break
        hc = slsqp_smooth(alt, 5000, n, maxiter=100)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h, best_c5 = subgrad_search(best_h, n, t0 + budget_s * 0.10)

    T_list = [50000, 20000, 8000, 3000, 1000, 400, 150, 50, 20]
    for T in T_list:
        if elapsed() > budget_s * 0.25:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=350)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    _, _bc = binary_local_search(best_h, n, t0 + budget_s * 0.35, max_iter=3000)
    if _bc < best_c5:
        best_c5 = _bc
        best_h = project_h((best_h >= 0.5).astype(float), n)
        best_h = slsqp_smooth(best_h, 500, n, maxiter=200)
        best_c5 = compute_c5(best_h)

    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(40):
        if elapsed() > budget_s * 0.55:
            break
        cool = max(0.1, 1.0 - it / 40.0)
        amp = 0.015 + 0.06 * cool
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        T_bh = 300 + 3000 * cool
        hc = slsqp_smooth(h_pert, T_bh, n, maxiter=300)
        c5v = compute_c5(hc)
        temp = 0.0005 * cool + 0.00003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.75, max_sweeps=80)

    for T in [100, 40, 15, 5]:
        if elapsed() > budget_s * 0.82:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=40)

    for T in [200000, 500000]:
        if elapsed() > budget_s * 0.95:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=20)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
