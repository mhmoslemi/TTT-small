    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_exact(h, n, j):
    h_base = h.copy()
    h_base[j] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_off = n - 1
    coef = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        idx1 = j + k - k_off
        if 0 <= idx1 < n:
            coef[k] += (1.0 - h[idx1])
        idx2 = j - k + k_off
        if 0 <= idx2 < n:
            coef[k] -= h[idx2]
    # Find x in [0,1] minimizing max_k(corr_base[k] + x*coef[k])
    candidates = [0.0, 1.0, h[j]]
    top0 = np.argsort(corr_base)[-10:][::-1]
    top1 = np.argsort(corr_base + coef)[-10:][::-1]
    for i in top0:
        for j2 in top1:
            if i == j2:
                continue
            dc = coef[i] - coef[j2]
            if abs(dc) < 1e-15:
                continue
            xc = (corr_base[j2] - corr_base[i]) / dc
            if -0.01 <= xc <= 1.01:
                candidates.append(float(np.clip(xc, 0.0, 1.0)))
    best_x = h[j]
    best_val = 1e18
    for x in candidates:
        x = float(np.clip(x, 0.0, 1.0))
        val = float(np.max(corr_base + x * coef))
        if val < best_val:
            best_val = val
            best_x = x
    return best_x
