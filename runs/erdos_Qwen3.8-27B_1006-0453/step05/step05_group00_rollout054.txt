def fft_corr_full(h):

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def c5_val(h):
    return float(np.max(fft_corr(h))) * (2.0 / len(h))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_D(h, N):
    """D[i, m] = change in corr[m] if h[i] goes 0->1"""
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    ifl = pos + sv
    irl = pos - sv
    vl = (ifl >= 0) & (ifl < N)
    vr = (irl >= 0) & (irl < N)
    hfl = np.where(vl, h[np.clip(ifl, 0, N - 1)], 0.0)
    hrl = np.where(vr, h[np.clip(irl, 0, N - 1)], 0.0)
    return (1.0 - hfl) * vl.astype(np.float64) - hrl * vr.astype(np.float64)

def swap_search(h, N, deadline, max_sweeps=2000, n_samp=130):

        x_dst = np.linspace(0, 2, N_try, endpoint=False) + 1.0 / N_try
        h_interp = np.interp(x_dst, x_src, best_h)
        h_bt = (h_interp > 0.5).astype(np.float64)
        nh = N_try // 2
        n1 = int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            h_bt[idx[np.random.permutation(len(idx))[nh:]]] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            h_bt[idx[:nh - n1]] = 1
        d_r = min(deadline - 10, time.time() + 55)
        hr, cr = swap_search(h_bt, N_try, d_r, max_sweeps=500, n_samp=100)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    # Phase 5: multi-restart with perturbation
    for _ in range(4):
        if time.time() > deadline - 25:
            break
        h_r = (best_h > 0.5).astype(np.float64)
        npert = max(4, len(h_r) // 25)
        for _ in range(npert):
            ones = np.where(h_r == 1)[0]
            zeros = np.where(h_r == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                p = ones[np.random.randint(len(ones))]
                q = zeros[np.random.randint(len(zeros))]
                h_r[p] = 0.0
                h_r[q] = 1.0
        d_re = min(deadline - 5, time.time() + 35)
        hr, cr = swap_search(h_r, len(h_r), d_re, max_sweeps=400, n_samp=90)
        if cr < best_c5:
            best_c5, best_h = cr, hr

    # Phase 6: final Nesterov polish
    if time.time() < deadline - 5:
        hF = best_h + np.random.randn(len(best_h)) * 0.04
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, cF = nesterov_cont(hF, len(best_h), deadline - 2, lr0=0.015, beta0=0.85)
        if cF < best_c5:
            best_c5, best_h = cF, hF
        else:
            hF = (best_h > 0.5).astype(np.float64)
            hF, cF = swap_search(hF, len(hF), deadline - 1, max_sweeps=200, n_samp=100)
            if cF < best_c5:
                best_c5, best_h = cF, hF

    final_c5 = c5_val(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
