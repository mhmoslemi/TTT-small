import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def c5_smooth(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = T * dx * corr
    vm = vals.max()
    e = np.exp(vals - vm)
    return (vm + np.log(e.sum())) / T

def c5_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = T * dx * corr
    vm = vals.max()
    ew = np.exp(vals - vm)
    w = ew / ew.sum()
    j = np.arange(n)[:, None]
    k = np.arange(-(n-1), n)[None, :]
    jm = j - k
    jp = j + k
    m1 = (jm >= 0) & (jm < n)
    m2 = (jp >= 0) & (jp < n)
    h1 = np.where(m1, 1.0 - h[np.clip(jm, 0, n-1)], 0.0)
    h2 = np.where(m2, -h[np.clip(jp, 0, n-1)], 0.0)
    return ((h1 + h2) @ w) * dx

def slsqp_pass(h, T, n, maxiter=500):
    def obj(x):
        return c5_smooth(x, T, n)
    def jac(x):
        return c5_grad(x, T, n)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return proj(np.clip(res.x, 0, 1))
    except Exception:
        return h

def cd_optimize(h, n, t_end, n_sweeps=100, n_grid=600):
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = compute_c5(h)
    k_range = np.arange(-(n-1), n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for j in range(n):
            if time.time() > t_end:
                return h, best_c5
            old = h[j]
            if old <= 1e-12 or old >= 1.0 - 1e-12:
                continue
            jm = j - k_range
            jp = j + k_range
            m1 = (jm >= 0) & (jm < n)
            m2 = (jp >= 0) & (jp < n)
            a = np.zeros(2 * n - 1)
            a[m1] += 1.0 - h[jm[m1]]
            a[m2] -= h[jp[m2]]
            t_grid = np.linspace(0.0, 1.0, n_grid)
            deltas = t_grid - old
            new_corr = corr[None, :] + deltas[:, None] * a[None, :]
            new_corr[:, n-1] = corr[n-1] + deltas * (1.0 - old - t_grid)
            obj_vals = new_corr.max(axis=1)
            t_new = t_grid[obj_vals.argmin()]
            if abs(t_new - old) > 1e-9:
                h[j] = t_new
                s = h.sum()
                diff = target - s
                if diff > 0:
                    room = 1.0 - h
                    h = h + room * (diff / room.sum())
                elif s > 0:
                    h = h * (1.0 + diff / s)
                h = np.clip(h, 0, 1)
                corr = np.correlate(h, 1.0 - h, mode='full')
                improved = True
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
        if not improved:
            break
    return h, best_c5

def pair_cd(h, n, t_end, n_sweeps=30, n_grid=400):
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = compute_c5(h)
    k_range = np.arange(-(n-1), n)
    t_grid = np.linspace(0.0, 1.0, n_grid)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        for j in range(0, n, 2):
            if time.time() > t_end:
                return h, best_c5
            m = (j + 1) % n
            vj, vm = h[j], h[m]
            if vj <= 1e-12 or vj >= 1-1e-12:
                continue
            jm_j = j - k_range
            jp_j = j + k_range
            m1j = (jm_j >= 0) & (jm_j < n)
            m2j = (jp_j >= 0) & (jp_j < n)
            aj = np.zeros(2*n-1)
            aj[m1j] += 1.0 - h[jm_j[m1j]]
            aj[m2j] -= h[jp_j[m2j]]
            jm_m = m - k_range
            jp_m = m + k_range
            m1m = (jm_m >= 0) & (jm_m < n)
            m2m = (jp_m >= 0) & (jp_m < n)
            am = np.zeros(2*n-1)
            am[m1m] += 1.0 - h[jm_m[m1m]]
            am[m2m] -= h[jp_m[m2m]]
            d = aj - am
            lo = max(-vj, vm - 1.0)
            hi = min(1.0 - vj, vm)
            if hi - lo < 1e-12:
                continue
            deltas = np.linspace(lo, hi, n_grid)
            new_corr = corr[None, :] + deltas[:, None] * d[None, :]
            quad_j = deltas * (1.0 - vj - (vj + deltas))
            quad_m = -deltas * (1.0 - vm - (vm - deltas))
            new_corr[:, n-1] = corr[n-1] + quad_j + quad_m
            obj_vals = new_corr.max(axis=1)
            db_opt = deltas[obj_vals.argmin()]
            if abs(db_opt) > 1e-10:
                h[j] = np.clip(vj + db_opt, 0, 1)
                h[m] = np.clip(vm - db_opt, 0, 1)
                corr = np.correlate(h, 1.0 - h, mode='full')
    h = proj(h)
    best_c5 = compute_c5(h)
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n0 = 200
        h = np.zeros(n0)
        h[:n0//2] = 1.0
    n = len(h)
    h = proj(h)
    best_c5 = compute_c5(h)
    # Phase 1: SLSQP warmup with annealing
    for T in [200, 1000, 5000, 20000]:
        if t_end - time.time() < 30:
            break
        hc = slsqp_pass(h, T, n, 350)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            h = hc
    # Phase 2: Coordinate descent (coarse)
    t_cd1 = t_end - (t_end - t0) * 0.25
    h, best_c5 = cd_optimize(h, n, t_cd1, n_sweeps=60, n_grid=500)
    # Phase 3: SLSQP refinement
    for T in [50000, 200000, 1000000]:
        if t_end - time.time() < 25:
            break
        hc = slsqp_pass(h, T, n, 500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            h = hc
    # Phase 4: Pair coordinate descent
    t_pc = t_end - (t_end - t0) * 0.12
    h, best_c5 = pair_cd(h, n, t_pc, n_sweeps=40, n_grid=500)
    # Phase 5: Fine coordinate descent
    t_cd2 = t_end - (t_end - t0) * 0.05
    h, best_c5 = cd_optimize(h, n, t_cd2, n_sweeps=80, n_grid=1000)
    # Phase 6: Final SLSQP polish
    if t_end - time.time() > 20:
        hc = slsqp_pass(h, 5000000, n, 1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            h = hc
    # Phase 7: Final fine CD
    if t_end - time.time() > 10:
        h, best_c5 = cd_optimize(h, n, t_end - 2, n_sweeps=40, n_grid=1200)
    h = proj(h)
    best_c5 = compute_c5(h)
    return h, best_c5, len(h)
