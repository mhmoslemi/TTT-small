                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-12,
                              'maxfun': iters * 2})
        return proj_h(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def _f1d(hi):
    return float(np.max(_CB[0] + hi * _CC[0]))

def coord_desc(h, n, t_end, sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    corr = np.correlate(best, 1.0 - best, mode='full')
    active_m = min(30, mf)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            j2 = i
            for k in range(mf):
                p = i + k - n + 1
                if 0 <= p < n:
                    coef[k] += (1.0 - best[p])
                q = i - k + n - 1
                if 0 <= q < n:
                    coef[k] -= best[q]
            _CB[0] = corr - old * coef
            _CC[0] = coef
            r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-13})
            nv = float(np.clip(r.x, 0.0, 1.0))
            if abs(nv - old) < 1e-14:
                continue
            best[i] = nv
            delta = nv - old
            corr = _CB[0] + nv * coef
            best_c5 = float(corr.max() * 2.0 / n)
            proj_diff = 0.5 * n - best.sum()
            if abs(proj_diff) > 1e-12:
                best = proj_h(best, n)
                corr = np.correlate(best, 1.0 - best, mode='full')
                best_c5 = float(corr.max() * 2.0 / n)
            imp = True
        if not imp:
            break
    return best, best_c5

def basin_hop(h, n, t_end, n_iters=300):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        frac = it / max(1, n_iters)
        cool = 1.0 - frac
        mod = it % 4
        if mod == 0:
            scale = 0.015 + 0.08 * cool
            pert = scale * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(2, 20)
            ph = np.random.uniform(0, 2 * np.pi)
            amp = 0.01 + 0.06 * cool
            pert = amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.15 + 0.25 * cool)
            pert = np.zeros(n)
            np.random.randn(mask.sum())
            pert[mask] = (0.03 + 0.10 * cool) * np.random.randn(mask.sum())
        else:
            idx = np.random.choice(n, size=max(2, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = (0.04 + 0.12 * cool) * np.random.randn(len(idx))
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 8000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.00005 + 0.0005 * cool
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def continuous_sa(h, n, t_end, max_iters=50000):
    best = h.copy()
    best_c5 = compute_c5(best)
    work = h.copy()
    work_c5 = best_c5
    T = 0.002
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        idx = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * 0.1
        new_val = np.clip(work[idx] + delta, 0.0, 1.0)
        if abs(new_val - work[idx]) < 1e-14:
            continue
        work[idx] = new_val
        c5t = compute_c5(work)
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / T):
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = work.copy()
        else:
            work[idx] = new_val - delta
        T = max(0.00005, T * 0.99995)
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive L-BFGS with warm starts
    T_schedule = [100, 500, 2000, 8000, 30000, 100000, 300000]
    for T in T_schedule:
        if time.time() - t0 > budget_s * 0.10:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Coordinate descent
    t_cd1 = t0 + budget_s * 0.20
    best_h, best_c5 = coord_desc(best_h, n, t_cd1, sweeps=120)

    # Phase 3: Basin hopping (extended)
    t_bh = t0 + budget_s * 0.55
    best_h, best_c5 = basin_hop(best_h, n, t_bh, n_iters=400)

    # Phase 4: Coordinate descent after basin hopping
    t_cd2 = t0 + budget_s * 0.65
    best_h, best_c5 = coord_desc(best_h, n, t_cd2, sweeps=80)

    # Phase 5: Continuous SA for fine-tuning
    t_sa = t0 + budget_s * 0.75
    best_h, best_c5 = continuous_sa(best_h, n, t_sa, max_iters=30000)

    # Phase 6: More basin hopping from improved point
    t_bh2 = t0 + budget_s * 0.82
    best_h, best_c5 = basin_hop(best_h, n, t_bh2, n_iters=150)

    # Phase 7: Final coordinate descent
    t_cd3 = t0 + budget_s * 0.88
    best_h, best_c5 = coord_desc(best_h, n, t_cd3, sweeps=60)

    # Phase 8: Try lower resolution n=120
    if time.time() - t0 < budget_s * 0.90:
        try:
            n_lo = 120
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_lo = np.linspace(0, 2, n_lo, endpoint=False) + 1.0 / n_lo
            h_lo = np.interp(x_lo, x_old, best_h)
            h_lo = proj_h(h_lo, n_lo)
            h_lo = lbfgs_opt(h_lo, 100000.0, n_lo, iters=500)
            c5_lo = compute_c5(h_lo)
            h_lo, c5_lo = coord_desc(h_lo, n_lo, t0 + budget_s * 0.93, sweeps=40)
            if c5_lo < best_c5:
                best_h = h_lo
                best_c5 = c5_lo
        except Exception:
            pass

    # Phase 9: Final ultra-high-T L-BFGS polish
    if time.time() - t0 < budget_s * 0.96:
        n_final = len(best_h)
        best_h = lbfgs_opt(best_h, 500000.0, n_final, iters=1000)
        best_c5 = compute_c5(best_h)

    # Phase 10: Final coordinate descent
    if time.time() - t0 < budget_s * 0.99:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t0 + budget_s * 0.995, sweeps=20)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
