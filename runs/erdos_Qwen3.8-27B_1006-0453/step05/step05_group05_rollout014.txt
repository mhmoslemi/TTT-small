from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m))) + 1e-300) / T
    return val * (2.0 / len(h))

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def lbfgs_pen(h, T, n, maxiter=5000):
    pen = 1e8
    def obj(x):
        return smooth_obj(x, T) + pen * (x.sum() - n * 0.5) ** 2
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-11, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

def slsqp_opt(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - n * 0.5
    try:

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

def coord_opt(h, n, i, niter=45):
    h[i] = 0.0
    cb = get_corr(h)
    h[i] = 1.0
    cf = get_corr(h)
    coef = cf - cb
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(cb + m1 * coef) < np.max(cb + m2 * coef):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=60, rng=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    if rng is None:
        rng = np.random

        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            ni = coord_opt(best_h, n, i)
            if abs(ni - best_h[i]) < 1e-14:

            h_try = best_h.copy()
            h_try[i] = ni
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5
