```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def coord_opt_1d(h, n, i, niter=50):
    """Optimize h[i] with others fixed. Returns optimal value."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    # Effect of h[i] = v on corr: (v - 0) * C[m] = v * C[m]
    # where C[m] accounts for the two ways i enters
    m_vals = np.arange(-(n-1), n)
    # i appears as j in term j*(1-h[j+m]): contributes v*(1-h[i+m])
    # i appears as j+m, i.e., j=i-m: contributes h[i-m]*(1-v)
    im = i + m_vals  # i+m
    im1 = i - m_vals  # i-m = j where j+m=i
    v1 = (im >= 0) & (im < n)
    v2 = (im1 >= 0) & (im1 < n)
    C = np.zeros(2*n-1)
    C[v1] += (1.0 - h[np.clip(im[v1], 0, n-1)])
    C[v2] += (-h[np.clip(im1[v2], 0, n-1)])
    # corr(v) = corr_base + v * C
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi-lo)/3
        m2 = hi - (hi-lo)/3
        if np.max(corr_base + m1*C) < np.max(corr_base + m2*C):
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2.0
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = float(n) / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-13:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-13:
            return h
        h = h * (1.0 + diff / s)

    corr = get_corr(h)
    m = np.max(corr)
```

But wait, I can optimize: if I maintain the full correlation array and update it incrementally when changing one coordinate, I save the O(n²) per-coordinate cost.

When h[i] changes from old to new:
corr_new[lag] = corr_old[lag] + (new-old) * [(1-h[i+lag])_{valid} - h[i-lag]_{valid}]

So I can maintain corr and update it in O(n) per coordinate change. This makes each coordinate update O(n) instead of O(n²).

```python
def coord_desc_fast(h, n, t_end, sweeps=500):

    corr = get_corr(bh)
    bc = corr.max() * 2.0/n
    
    for sw in range(sweeps):

        imp = True
        while imp:
            imp = False
            for i in np.random.permutation(n):
                if time.time() > t_end: return bh, bc
                old = bh[i]
                # Compute C for this coordinate
                mv = np.arange(-(n-1), n)
                ia = i + mv; ib = i - mv
                va = (ia>=0)&(ia<n); vb = (ib>=0)&(ib<n)
                C = np.zeros(2*n-1)
                C[va] += 1.0 - bh[np.clip(ia[va],0,n-1)]
                C[vb] -= bh[np.clip(ib[vb],0,n-1)]
                # Minimize max(corr + (v-old)*C) over v in [0,1]
                lo, hi = 0.0, 1.0
                for _ in range(45):
                    m1 = lo+(hi-lo)/3; m2 = hi-(hi-lo)/3
                    if np.max(corr+(m1-old)*C) < np.max(corr+(m2-old)*C): hi=m2
                    else: lo=m1
                nv = (lo+hi)/2
                if abs(nv-old) < 1e-11: continue
                # Update corr
                corr += (nv-old)*C
                bh[i] = nv
                # Re-project
                s = bh.sum()
                tgt = float(n)/2
                if abs(s-tgt) > 1e-10:
                    # Simple projection
                    bh = project_h(bh, n)
                    corr = get_corr(bh)  # Recompute (expensive but rare)
                cm = corr.max() * 2.0/n
                if cm < bc - 1e-14:
                    bc = cm; imp = True
    return bh, bc
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = float(n) / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    d = target - s
    if abs(d) < 1e-12:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-12:
            return h
        return np.clip(h + r * (d / t), 0, 1)
    if s < 1e-12:
        return h
    return np.clip(h * (1.0 + d / s), 0, 1)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = get_corr(h)
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx

def optimize_coord(h, n, i):
    old = h[i]
    hb = h.copy()
    hb[i] = 0.0
    cb = get_corr(hb)
    mv = np.arange(-(n - 1), n)
    ia = i + mv
    ib = i - mv
    va = (ia >= 0) & (ia < n)
    vb = (ib >= 0) & (ib < n)
    C = np.zeros(2 * n - 1)
    C[va] += 1.0 - h[np.clip(ia[va], 0, n - 1)]
    C[vb] -= h[np.clip(ib[vb], 0, n - 1)]
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(cb + m1 * C) < np.max(cb + m2 * C):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_desc(h, n, t_end, sweeps=500, npairs=12):
    bh = h.copy()
    bc = compute_c5(bh)
    for sw in range(sweeps):
        if time.time() > t_end:
            return bh, bc
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return bh, bc
            nv = optimize_coord(bh, n, i)
            if abs(nv - bh[i]) < 1e-11:
                continue
            ht = bh.copy()
            ht[i] = nv
            ht = project_h(ht, n)
            c = compute_c5(ht)
            if c < bc - 1e-13:
                bh = ht
                bc = c
                imp = True
        if not imp:
            for _ in range(npairs):
                if time.time() > t_end:
                    return bh, bc
                i, j = np.random.choice(n, 2, replace=False)
                nv_i = optimize_coord(bh, n, i)
                bh[i] = nv_i
                nv_j = optimize_coord(bh, n, j)
                bh[j] = nv_j
                bh = project_h(bh, n)
                c = compute_c5(bh)
                if c < bc:
                    bc = c
                    imp = True
            if not imp:
                break
    return bh, bc

def slsqp_step(h, n, T, maxiter):
    def obj(x):
        return smooth_obj(x, T)
    def con(x):
        return x.sum() - float(n) / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def slsqp_anneal(h, n, t_end, T_list):
    bh = h.copy()
    bc = compute_c5(bh)
    for T in T_list:
        if time.time() > t_end:
            break
        hc = slsqp_step(bh, n, T, maxiter=2000)
        c5 = compute_c5(hc)
        if c5 < bc:
            bh = hc
            bc = c5
    return bh, bc

def basin_hop(h, n, t_end, n_iter=80):
    bh = h.copy()
    bc = compute_c5(bh)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(n_iter, 1))
        mod = it % 6
        if mod == 0:
            pert = (0.015 + 0.06 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 12)
            ph = np.random.uniform(0, 6.28)
            pert = (0.01 + 0.05 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mk = np.random.random(n) < 0.12
            pert = np.zeros(n)
            pert[mk] = (0.03 + 0.1 * cool) * np.random.randn(mk.sum())
        elif mod == 3:
            blk = max(1, n // 6)
            b = np.random.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            pert[b * blk:(b + 1) * blk] = (0.02 + 0.04 * cool) * np.random.randn(blk)
        elif mod == 4:
            pert = np.zeros(n)
            for _ in range(3):
                a, b2 = np.random.choice(n, 2, replace=False)
                d = (bh[b2] - bh[a]) * (0.3 + 0.7 * cool)
                pert[a] += d
                pert[b2] -= d
        else:
            pert = (0.008 + 0.025 * cool) * np.random.randn(n)
        hp = project_h(bh + pert, n)
        T_bh = 500 + 20000 * cool
        hc = slsqp_step(hp, n, T_bh, maxiter=400)
        c5 = compute_c5(hc)
        temp = 0.0006 * cool + 0.00003
        if c5 < bc or np.random.rand() < np.exp(-(c5 - bc) / temp):
            bh = hc
            bc = c5
    return bh, bc

def binary_local_search(h, n, t_end, n_iter=300):
    bh = np.round(h).copy()
    bh = project_h(bh, n)
    bc = compute_c5(bh)
    best_b = bh.copy()
    best_bc = bc
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i, j = np.random.choice(n, 2, replace=False)
        if abs(bh[i] - bh[j]) < 0.1:
            continue
        bh[i], bh[j] = bh[j], bh[i]
        bh = project_h(bh, n)
        c = compute_c5(bh)
        if c > best_bc * 1.001:
            bh[i], bh[j] = bh[j], bh[i]
            bh = project_h(bh, n)
        else:
            if c < best_bc:
                best_bc = c
                best_b = bh.copy()
    return best_b, best_bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP annealing
    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = slsqp_anneal(best_h, n, t1, [50, 500, 5000, 50000])

    # Phase 2: Coordinate descent
    t2 = t0 + budget_s * 0.40
    best_h, best_c5 = coord_desc(best_h, n, t2, sweeps=400, npairs=15)

    # Phase 3: SLSQP round 2
    t3 = t0 + budget_s * 0.50
    best_h, best_c5 = slsqp_anneal(best_h, n, t3, [200, 2000, 20000])

    # Phase 4: More coordinate descent
    t4 = t0 + budget_s * 0.65
    best_h, best_c5 = coord_desc(best_h, n, t4, sweeps=400, npairs=15)

    # Phase 5: Binary local search (explore binary basin)
    t5 = t0 + budget_s * 0.72
    hb, cb = binary_local_search(best_h, n, t5, n_iter=400)
    if cb < best_c5:
        best_c5 = cb
        best_h = hb

    # Phase 6: Coordinate descent from best
    t6 = t0 + budget_s * 0.78
    best_h, best_c5 = coord_desc(best_h, n, t6, sweeps=300, npairs=10)

    # Phase 7: Basin hopping
    t7 = t0 + budget_s * 0.90
    best_h, best_c5 = basin_hop(best_h, n, t7, n_iter=60)

    # Phase 8: Final coordinate descent
    t8 = t0 + budget_s * 0.96
    best_h, best_c5 = coord_desc(best_h, n, t8, sweeps=300, npairs=10)

    # Phase 9: Final SLSQP polish
    if time.time() < t0 + budget_s * 0.99:
        hc = slsqp_step(best_h, n, 200000, maxiter=3000)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
