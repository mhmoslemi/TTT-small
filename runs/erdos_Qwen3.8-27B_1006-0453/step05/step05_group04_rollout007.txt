The subgradient of max_k corr_k * dx at h is:
sg = dx * (1/|S|) * sum_{k ∈ S} ∇_h corr_k

where S is the set of k's that achieve the max (or are close to it).

def subgradient_step(h, n, t_end, max_iters=5000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h_c, 1.0 - h_c, mode='full')
        m = np.max(corr)
        dx = 2.0 / n
        # Find lags close to max
        threshold = m - 0.01 * m  # top 1%
        active = np.where(corr >= threshold)[0]
        # Compute subgradient
        sg = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s > 0:
                sg[s:] += (1.0 - h_c[:n-s])
                sg[:n-s] -= h_c[s:]
            elif s < 0:
                a = -s
                sg[:n-a] += (1.0 - h_c[a:])
                sg[a:] -= h_c[:n-a]
            else:
                sg += (1.0 - 2.0 * h_c)
        sg /= len(active)
        # Step size (Polyak-like)
        alpha = 1.0 / (it + 10)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                    lin[mn] -= best_h[i - s]
                if i <= n - 1 - s:
                    lin[mp] -= best_h[i + s]
                    lin[mn] += (1.0 - best_h[i + s])
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-12:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=500, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        frac = 1.0 - hop / max(n_hops, 1)
        step = 0.01 + 0.05 * frac
        mode = hop % 7
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 16)
            ph = rng.uniform(0, 2*np.pi)
            pert = step * np.sin(2*np.pi*f*np.arange(n)/n + ph)
        elif mode == 2:
            # Block flip
            a = rng.randint(0, n - 4)
            b = min(n, a + rng.randint(2, 12))
            pert = np.zeros(n)
            pert[a:b] = rng.uniform(-1, 1, b-a)
        elif mode == 3:
            # Swap pairs
            pert = np.zeros(n)
            for _ in range(rng.randint(2, 8)):
                i, j = rng.choice(n, 2, replace=False)
                pert[i] = cur_h[j] - cur_h[i]
                pert[j] = cur_h[i] - cur_h[j]
        elif mode == 4:
            # Targeted: find max corr lag, adjust h in that region
            corr = np.correlate(cur_h, 1-cur_h, mode='full')
            k_max = np.argmax(corr)
            s = k_max - (n-1)
            pert = np.zeros(n)
            idx = list(range(max(0, -s), min(n, n-s))) if s >= 0 else list(range(max(0, s), min(n, n+s)))
            pert[idx] = -step * 0.5 * np.sign(corr[k_max]) * rng.uniform(0.5, 1.0, len(idx))
        elif mode == 5:
            # Large perturbation (escape)
            pert = step * 2 * rng.randn(n)
        else:
            # Smooth low-freq perturbation
            f = rng.randint(1, 5)
            ph = rng.uniform(0, 2*np.pi)
            pert = step * np.cos(2*np.pi*f*np.arange(n)/n + ph)
        
        h_pert = project_h(cur_h + pert, n)
        # Short local optimization
        local_budget = max(1.0, 5.0 * frac)
        hop_end = min(t_end, time.time() + local_budget)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=50, seed=seed+hop)
        
        # Accept/reject
        temp = 0.003 * frac + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt-cur_c5)/temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                # Reset cur to best occasionally
                if rng.rand() < 0.1:
                    cur_h = best_h.copy()
                    cur_c5 = best_c5
    return best_h, best_c5
