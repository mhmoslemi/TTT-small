import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = float(n) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def subgrad(h, n, topk):
    dx = 2.0 / n
    corr = get_corr(h)
    top_idx = np.argsort(corr)[-topk:]
    shifts = top_idx - (n - 1)
    grad = np.zeros(n)
    for k in shifts:
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
        else:
            kk = -k
            grad[:n-kk] += (1.0 - h[kk:])
            grad[kk:] -= h[:n-kk]
    grad /= len(shifts)
    grad *= dx
    return grad

def slsqp(h, n, T, maxiter=500, flat_w=0.0, t_end=None):
    dx = 2.0 / n
    def obj(x):
        if t_end is not None and time.time() > t_end:
            return 1e10
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if T < 1e8:
            val = m + np.log(np.sum(np.exp(T * (corr - m) - np.log(len(corr))))) / T
        else:
            val = m
        val *= dx
        if flat_w > 0:
            top = np.sort(corr)[-8:]
            val += flat_w * dx * (top[-1] - top[0])
        return val
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descend(h, n, t_end, max_sweeps=200, n_top=15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, T_lo=500, T_hi=10000):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    rng = np.random
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / max(n_iter, 1))
        r = rng.random()
        if r < 0.35:
            # Targeted: perturb along worst shift direction
            corr = get_corr(work_h)
            top_idx = np.argsort(corr)[-4:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0:
                    pert[k:] += (work_h[:n-k] - 0.5)
                else:
                    kk = -k
                    pert[:n-kk] += (work_h[kk:] - 0.5)
            pert *= scale * 0.3 * (0.3 + 0.7 * cool)
        elif r < 0.55:
            # Frequency perturbation
            f = rng.randint(1, 15)
            p = rng.uniform(0, 2 * np.pi)
            pert = scale * 0.6 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.75:
            # Local block perturbation
            blen = rng.randint(max(n // 12, 2), max(n // 4, 3))
            bs = rng.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs+blen] = scale * cool * rng.randn(blen)
        else:
            # Global noise
            pert = scale * cool * rng.randn(n)
        h_pert = project_h(work_h + pert, n)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp(h_pert, n, T, 300, 0.02 * cool, t_end)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and rng.random() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

    t_end = t0 + budget_s

            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
        else:
            kk = -k
            grad[:n - kk] += (1.0 - h[kk:])
            grad[kk:] -= h[:n - kk]
    grad /= len(shifts)
    grad *= dx
    return grad

def slsqp_opt(h, n, T, maxiter=500, flat_w=0.0, t_end=None):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if T < 1e7:
            exps = np.exp(T * (corr - m))
            val = m + np.log(np.sum(exps)) / T
        else:
            val = m
        val *= dx
        if flat_w > 0.0:
            top = np.sort(corr)[-8:]
            val += flat_w * dx * (top[-1] - top[0])
        return val
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq',
                                     'fun': lambda x: x.sum() - n / 2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_descend(h, n, t_end, max_sweeps=200, n_top=15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: float(np.max(base + t * coefs)),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, T_lo=500, T_hi=20000):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / max(float(n_iter), 1.0))
        r = np.random.random()
        if r < 0.35:
            corr = get_corr(work_h)
            top_idx = np.argsort(corr)[-4:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0:
                    pert[k:] += (work_h[:n - k] - 0.5)
                else:
                    kk = -k
                    pert[:n - kk] += (work_h[kk:] - 0.5)
            pert *= scale * 0.3 * (0.2 + 0.8 * cool)
        elif r < 0.55:
            f = np.random.randint(1, 16)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.6 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.72:
            blen = np.random.randint(max(n // 12, 2), max(n // 4, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * cool * np.random.randn(blen)
        elif r < 0.85:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.2 * cool * np.random.randn(max(int(mask.sum()), 1))
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert, n)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp_opt(h_pert, n, T, 250, 0.015 * cool)
        c5v = compute_c5(hc)
        temp = 0.0006 * cool + 0.00008
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def smoothing_restart(h, n, t_end, strength):
    kernel = np.array([0.15, 0.35, 0.35, 0.15], dtype=float)
    h_smooth = np.convolve(h, kernel, mode='same')
    h_mixed = (1.0 - strength) * h + strength * h_smooth
    h_mixed = project_h(h_mixed, n)
    hc = slsqp_opt(h_mixed, n, 5000, 400, 0.01)
    return hc, compute_c5(hc)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 200
        init_h = np.full(n_init, 0.5)
        init_h += 0.15 * np.sin(2 * np.pi * 2 * np.arange(n_init) / n_init)

    n = len(init_h)
    best_h = project_h(init_h.copy(), n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive SLSQP warmup
    t_ph1 = t0 + budget_s * 0.22
    for T in [30, 100, 500, 2000, 10000]:
        if time.time() > t_ph1:
            break
        hc = slsqp_opt(best_h, n, T, 600, 0.04)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Subgradient with momentum + SLSQP
    t_ph2 = t0 + budget_s * 0.42
    mom = np.zeros(n)
    for it in range(80):
        if time.time() > t_ph2:
            break
        cool = 1.0 - it / 80.0
        alpha = 0.0008 + 0.0035 * cool
        topk = 4 + int(10 * cool)
        g = subgrad(best_h, n, topk)
        mom = 0.5 * mom + 0.5 * g
        h_s = project_h(best_h - alpha * mom, n)
        T = 1000 + 30000 * cool
        flat = 0.02 * cool
        hc = slsqp_opt(h_s, n, T, 350, flat)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        elif it % 15 == 14 and cool < 0.8:
            hc2, c5_2 = smoothing_restart(best_h, n, t_ph2, 0.3)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = hc2

    # Phase 3: Medium basin hopping
    t_ph3 = t0 + budget_s * 0.65
    bh_h, bh_c5 = basin_hop(best_h, n, t_ph3, 60, 0.10, 1000, 20000)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = bh_h

    # Phase 4: Coordinate descent (coarse)
    t_ph4 = t0 + budget_s * 0.78
    best_h, best_c5 = coord_descend(best_h, n, t_ph4, 120, 18)

    # Phase 5: Fine basin hopping
    t_ph5 = t0 + budget_s * 0.88
    bh_h, bh_c5 = basin_hop(best_h, n, t_ph5, 35, 0.035, 5000, 60000)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = bh_h

    # Phase 6: Final SLSQP with high T
    if time.time() < t_end - 30:
        hc = slsqp_opt(best_h, n, 200000, 1200, 0.0)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Final coordinate descent
    if time.time() < t_end - 2:
        best_h, best_c5 = coord_descend(best_h, n, t_end - 0.5, 80, 25)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
