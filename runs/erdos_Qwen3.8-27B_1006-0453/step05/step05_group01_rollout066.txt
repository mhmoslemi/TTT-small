def subgrad_top_vec(h, T, k_top):
    """Compute top-k weighted gradient and smoothed C5"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top k shifts
    top_idx = np.argsort(corr)[-k_top:]
    shifts = top_idx - (n - 1)
    
    # Compute weights (softmax-like)
    top_corr = corr[top_idx]
    exp_vals = np.exp(T * (top_corr - top_corr.max()))
    weights = exp_vals / exp_vals.sum()
    
    # Compute gradient: weighted sum of per-shift gradients
    grad = np.zeros(n)
    for w, s in zip(weights, shifts):
        if s >= 0:
            # i from 0 to n-1-s: grad[i] += w * (1-h[i+s])
            grad[:n-s] += w * (1.0 - h[s:])
            # i from s to n-1: grad[i] += w * (-h[i-s])
            grad[s:] += w * (-h[:n-s])
        else:
            r = -s
            # i from r to n-1: grad[i] += w * (1-h[i-r])
            grad[r:] += w * (1.0 - h[:-r])
            # i from 0 to n-1-r: grad[i] += w * (-h[i+r])
            grad[:n-r] += w * (-h[r:])
    
    # Smoothed C5 value
    smooth_val = corr.max() + np.log(exp_vals.sum()) / T
    return smooth_val * dx, grad * dx

    diff = target - s

    if diff > 0:

def smooth_c5_and_grad(h, T, k_top):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    smooth_val = (m + np.log(exp_vals.sum()) / T) * dx

        else:
            r = -s
            grad[r:] += w * (1.0 - h[:-r])

            grad[r:] += wi * (1.0 - h[:-r])

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_grad(h, T, k_top):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k_top:]
    shifts = top_idx - (n - 1)
    tc = corr[top_idx]
    m = tc.max()
    ev = np.exp(T * (tc - m))
    w = ev / ev.sum()
    val = (m + np.log(ev.sum()) / T) * dx
    grad = np.zeros(n)
    for wi, s in zip(w, shifts):
        if s >= 0:
            grad[:n-s] += wi * (1.0 - h[s:])
            if n - s > 0:
                grad[s:] += wi * (-h[:n-s])
        else:
            r = -s
            grad[r:] += wi * (1.0 - h[:-r])
            if n - r > 0:
                grad[:n-r] += wi * (-h[r:])
    grad *= dx
    return val, grad

def adam_opt(h, n, t_end, iters=5000, lr=0.004, T_s=3000, T_e=80, k_s=25, k_e=4):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    for it in range(iters):
        if time.time() > t_end:
            break
        frac = it / max(iters - 1, 1)
        T = T_s * (T_e / T_s) ** frac
        k = max(3, int(k_s + (k_e - k_s) * frac))
        lr_i = lr * (1.0 - 0.65 * frac)
        val, grad = smooth_c5_grad(h, T, k)
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad * grad
        t_adj = it + 1
        m_hat = m_v / (1 - b1 ** t_adj)
        v_hat = v_v / (1 - b2 ** t_adj)
        step = lr_i * m_hat / (np.sqrt(v_hat) + eps)
        gnorm = np.linalg.norm(step)
        if gnorm > 0.5:
            step *= 0.5 / gnorm
        h = project_h(h - step)
        if it % 500 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def multi_adam(h, n, t_end, n_rounds=5):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for r in range(n_rounds):
        frac_r = r / max(n_rounds - 1, 1)
        t_sub = min(t_end, time.time() + (t_end - time.time()) / (n_rounds - r))
        init = best_h + 0.005 * (1.0 - frac_r) * np.random.randn(n)
        init = project_h(init)
        h_r, c5_r = adam_opt(init, n, t_sub,
                             iters=4000,
                             lr=0.003 * (1.0 - 0.3 * frac_r),
                             T_s=2000 * (1.0 - 0.5 * frac_r),
                             T_e=50, k_s=20 - 5 * r, k_e=3)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
    return best_h, best_c5

def coord_descent(h, n, t_end, sweeps=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            cur = best_h[i]
            trials = [0.0, 1.0, 1.0 - cur, cur * 0.5, cur + (1.0 - cur) * 0.3]
            for trial in trials:
                if trial < -1e-12 or trial > 1 + 1e-12 or abs(trial - cur) < 1e-10:
                    continue
                h_try = best_h.copy()
                h_try[i] = min(1.0, max(0.0, trial))
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, n, T=1e6, maxiter=500):
    try:
        def obj(x):
            c = np.correlate(np.clip(x, 0, 1), 1.0 - np.clip(x, 0, 1), mode='full')
            m = c.max()
            return m + np.log(np.sum(np.exp(T * (c - m)))) / T
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        init_h = np.zeros(n0)

    n = len(init_h)

    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-round Adam (primary optimization)
    t1 = t0 + budget_s * 0.45
    best_h, best_c5 = multi_adam(best_h, n, t1, n_rounds=6)

    # Phase 2: SLSQP polish
    if t_end - time.time() > 60:
        hp = slsqp_polish(best_h, n, T=5e5, maxiter=400)
        c5p = compute_c5(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp

    # Phase 3: Coordinate descent
    t3 = t0 + budget_s * 0.70
    best_h, best_c5 = coord_descent(best_h, n, t3, sweeps=60)

    # Phase 4: Perturbed Adam restart
    if t_end - time.time() > 60:
        h4 = best_h + 0.003 * np.random.randn(n)
        h4 = project_h(h4)
        t4 = min(t_end - 30, time.time() + (t_end - time.time()) * 0.35)
        h4, c5_4 = adam_opt(h4, n, t4, iters=3000, lr=0.002,
                            T_s=1500, T_e=40, k_s=15, k_e=3)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
        best_h, best_c5 = coord_descent(best_h, n, min(t_end - 10, time.time() + 15), sweeps=40)

    # Phase 5: Try alternate n values
    if t_end - time.time() > budget_s * 0.18:
        for n_alt in [160, 240, 300]:
            if t_end - time.time() < budget_s * 0.07:
                break
            if n_alt == n:
                continue
            xg = np.linspace(0, 2, n_alt, endpoint=False)
            ha = np.interp(xg, np.linspace(0, 2, n, endpoint=False), best_h)
            ha = project_h(ha)
            t_sub = min(t_end - 30, time.time() + (t_end - time.time()) * 0.35)
            ha, c5_a = adam_opt(ha, n_alt, t_sub, iters=2500, lr=0.003,
                                T_s=2000, T_e=60, k_s=18, k_e=4)
            if t_end - time.time() > 20:
                ha, c5_a = coord_descent(ha, n_alt, t_end - 15, sweeps=30)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = ha
                n = n_alt

    # Final SLSQP + coord polish
    if t_end - time.time() > 25:
        hp = slsqp_polish(best_h, n, T=1e7, maxiter=300)
        c5p = compute_c5(hp)
        if c5p < best_c5:

            best_h = hp

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
