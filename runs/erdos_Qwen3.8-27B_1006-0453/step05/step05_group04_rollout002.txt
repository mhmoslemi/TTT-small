    """Evaluate Chebyshev series at points x in [-1,1]."""

        t_next = 2.0 * x * t_curr - t_prev
        result += coeffs[k] * t_next
        t_prev = t_curr
        t_curr = t_next
    return result

            break

            h = np.clip(h * (target / s), 0.0, 1.0)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + room * ((target - s) / tot), 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 3 == 2 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                if i <= n - 1 - s:
                    lin[mp] -= best_h[i + s]
                    lin[mn] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[mn] -= best_h[i - s]
            lin[n - 1] = 1.0 - 2.0 * old
            cb = corr.copy()
            n1 = n - 1
            def f1d(v):
                d = v - old
                val = cb + d * lin
                val[n1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved and sweep > 5:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def _de_callback(t_end):
    def cb(xk, convergence):
        if time.time() > t_end:
            raise _Timeout()
        return False
    return cb

def de_full(h_init, n, t_end, seed):
    def obj(x):
        return compute_c5(project_h(x, n))
    bounds = [(0.0, 1.0)] * n
    try:
        res = differential_evolution(obj, bounds, seed=seed, maxiter=500,
                                      tol=1e-9, popsize=25, mutation=(0.3, 0.8),
                                      recombination=0.7, polish=False,
                                      init='sobol', updating='deferred',
                                      callback=_de_callback(t_end))
        if res.fun < np.inf:
            return project_h(res.x, n)
    except _Timeout:
        pass
    except Exception:
        pass
    return h_init

def basin_hop(h, n, t_end, seed):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    n_hops = 60
    for hop in range(n_hops):
        if time.time() >= t_end:
            break
        cool = 1.0 - hop / n_hops
        step = 0.03 + 0.08 * cool
        mode = hop % 5
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.uniform(1.0, 6.0)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * 0.6 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            bsz = rng.randint(4, 40)
            st = rng.randint(0, max(1, n - bsz))
            pert = np.zeros(n)
            pert[st:st + bsz] = step * 1.5 * rng.choice([-1, 1], size=bsz)
        elif mode == 3:
            idx = rng.choice(n, size=max(3, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 2.0 * rng.randn(len(idx))
        else:
            pert = step * 0.4 * (rng.randn(n) - rng.randn(n)[::-1])
        h_pert = project_h(cur_h + pert, n)
        hop_end = min(t_end, time.time() + max(1.5, 6.0 * cool))
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=80,
                                       seed=seed + 2000 + hop)
        temp = 0.004 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h, cur_c5 = h_opt, c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def multi_start_cd(h0, n, t_end, seed):
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(seed)
    n_starts = 6
    t_per = (t_end - time.time()) / max(1, n_starts)
    for s in range(n_starts):
        t_s = time.time() + t_per
        if time.time() >= t_s:
            break
        if s == 0:
            hc = h0.copy()
        elif s == 1:
            f = rng.uniform(2, 8)
            x = np.linspace(0, 2, n)
            hc = 0.5 + 0.35 * np.sin(2 * np.pi * f * x / 2 + rng.uniform(0, 6.28))
        elif s == 2:
            bsz = rng.randint(10, 40)
            hc = np.zeros(n)
            for b in range(0, n, bsz):
                if (b // bsz) % 2 == 0:
                    hc[b:min(b + bsz, n)] = 1.0
        elif s == 3:
            hc = 0.5 + 0.25 * rng.randn(n)
        elif s == 4:
            hc = h0.copy()
            hc += 0.15 * rng.randn(n)
        else:
            f = rng.uniform(0.5, 3.0)
            x = np.linspace(0, 2, n)
            hc = 0.5 + 0.3 * np.cos(2 * np.pi * f * x / 2) ** 3
        hc = project_h(hc, n)
        h_opt, c5_opt = coord_descent(hc, n, t_s, max_sweeps=800, seed=seed + s)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = 160
    if len(best_h) != n:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n)
        best_h = np.interp(x_new, x_old, best_h, left=0, right=0)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Polish initial (10%)
    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=800, seed=seed)

    # Phase 2: Differential Evolution (global search, 35%)
    t2 = t0 + budget_s * 0.45
    h_de = de_full(best_h, n, t2, seed=seed)
    c5_de = compute_c5(h_de)
    if c5_de < best_c5:
        best_h, best_c5 = h_de, c5_de
    # Polish DE result
    if time.time() < t0 + budget_s * 0.50:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.50,
                                         max_sweeps=600, seed=seed + 1)

    # Phase 3: Multi-start coordinate descent (15%)
    t3 = t0 + budget_s * 0.65
    h_ms, c5_ms = multi_start_cd(best_h, n, t3, seed=seed + 50)
    if c5_ms < best_c5:
        best_h, best_c5 = h_ms, c5_ms

    # Phase 4: Basin hopping with large jumps (20%)
    t4 = t0 + budget_s * 0.85
    best_h, best_c5 = basin_hop(best_h, n, t4, seed=seed + 100)

    # Phase 5: Second DE around best (10%)
    t5 = t0 + budget_s * 0.95
    h_de2 = de_full(best_h, n, t5, seed=seed + 500)
    c5_de2 = compute_c5(h_de2)
    if c5_de2 < best_c5:
        best_h, best_c5 = h_de2, c5_de2

    # Phase 6: Final coordinate descent polish (5%)
    t6 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=3000, seed=seed + 999)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
