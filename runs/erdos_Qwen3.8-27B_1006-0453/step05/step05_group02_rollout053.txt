    """Build linear coefficients and quadratic matrix for the inner QP."""
    S = 2*n - 1
    c = np.zeros(n)  # linear coefficients
    Q = np.zeros((n, n))  # quadratic matrix (symmetric)
    
    for s_idx in range(S):
        ps = p[s_idx]
        if ps < 1e-15:
            continue
        shift = s_idx - (n - 1)

        if shift >= 0:

            j = i + shift
            Q[i, j] -= ps
            Q[j, i] -= ps  # symmetric
    
    # Make Q symmetric (it should already be, but just in case)
    Q = 0.5 * (Q + Q.T)
    return c, Q

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        scale = target / s
        h = h * scale
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            h = h * (target / max(s, 1e-14))
        else:
            h = h + room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def inner_gradient(h, p, n):
    """Gradient of sum_s p_s * F_s(h) w.r.t. h, where F_s = corr[s]."""
    grad = np.zeros(n)
    dx = 2.0 / n
    for s_idx in range(2 * n - 1):
        ps = p[s_idx]
        if abs(ps) < 1e-16:
            continue
        shift = s_idx - (n - 1)
        if shift >= 0:
            # j as "i": j in [0, n-1-shift]
            idx = np.arange(0, n - shift)
            grad[idx] += ps * (1.0 - h[idx + shift])
            # j as "i+shift": j in [shift, n-1]
            idx2 = np.arange(shift, n)
            grad[idx2] -= ps * h[idx2 - shift]
        else:
            a = -shift
            # j as "i": j in [a, n-1]
            idx = np.arange(a, n)
            grad[idx] += ps * (1.0 - h[idx - a])  # h[j+shift] = h[j-a]
            # j as "i+shift" = i-a: j in [0, n-1-a]
            idx2 = np.arange(0, n - a)
            grad[idx2] -= ps * h[idx2 + a]  # h[j-shift] = h[j+a]
    return grad

def minimax_solve(h, n, t_end, max_outer=50, lr_outer=0.5, inner_steps=30):
    """Minimax outer loop with projected gradient inner."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    p = np.ones(2 * n - 1) / (2 * n - 1)
    
    for outer in range(max_outer):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        c5_raw = np.max(corr)
        
        # Update p: emphasize worst lags
        # Use softmax with temperature
        T_p = 5.0 + 20.0 * (outer / max_outer)  # increasing temp for exploration
        p_new = p * np.exp(lr_outer * (corr - c5_raw) / max(c5_raw, 1e-6))
        p_new = np.clip(p_new, 1e-10, None)
        p = p_new / p_new.sum()
        
        # Inner: projected gradient on g(h) = sum_s p_s * F_s(h)
        h_curr = best_h.copy()
        step_size = 0.05 / (np.sqrt(sum(p**2)) + 1e-10)
        
        for inner in range(inner_steps):
            if time.time() > t_end:
                break
            g = inner_gradient(h_curr, p, n)
            h_curr = h_curr - step_size * g
            h_curr = project_h(h_curr)
        
        c5_new = compute_c5(h_curr)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_curr.copy()
    
    return best_h, best_c5

def coord_single(h, n, i):
    """Optimize h[i] given all other coordinates fixed."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    dx = 2.0 / n
    # corr = corr_base + h_i * coef
    # Find coef: how corr changes with h[i]
    S = 2 * n - 1
    coef = np.zeros(S)
    for s_idx in range(S):
        shift = s_idx - (n - 1)
        if shift >= 0:
            if 0 <= i < n - shift:
                coef[s_idx] += (1.0 - h[i + shift])  # h[i] as first factor
            if shift <= i < n:
                coef[s_idx] += (-h[i - shift])  # h[i] as second factor (h[i-shift]*h[i])
        else:
            a = -shift
            if a <= i < n:
                coef[s_idx] += (1.0 - h[i - a])  # h[i] as first, h[i-a] as second
            if 0 <= i < n - a:
                coef[s_idx] += (-h[i + a])  # h[i] as second factor
    new_corr = corr_base + h[i] * coef
    # Minimize max(new_corr) * dx: find h[i] in [0,1]
    # This is: minimize max_s (corr_base[s] + h[i]*coef[s])
    # = minimize max_s (corr_base[s] + x*coef[s]) for x in [0,1]
    # Try: the optimum is where the top coefficients are balanced
    # Simple approach: try a few points

    best_x = h[i]
    best_val = np.max(corr_base + h[i] * coef)
    for x in x_candidates:
        val = np.max(corr_base + x * coef)
        if val < best_val:
            best_val = val
            best_x = x
    # Refine with ternary search
    for _ in range(20):
        m1 = best_x - 1e-3
        m2 = best_x + 1e-3
        if m1 < 0: m1 = 0
        if m2 > 1: m2 = 1
        v1 = np.max(corr_base + m1 * coef)
        v2 = np.max(corr_base + m2 * coef)
        if v1 < v2:
            best_x = (best_x + m2) / 2 if m2 > best_x else m1
            best_x = min(max(best_x, 0), 1)
        else:
            best_x = (best_x + m1) / 2 if m1 < best_x else m2
            best_x = min(max(best_x, 0), 1)
    return best_x

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            x = coord_single(best_h, n, i)
            if abs(x - best_h[i]) > 1e-10:
                h_try = best_h.copy()
                h_try[i] = x
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    imp = True
        if not imp:
            break
    return best_h, best_c5

def swap_move(h, n, rng, t_end):
    """Try a random swap of two positions."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(500):

            break
        i, j = rng.choice(n, 2, replace=False)

        h_try = best_h.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]

        if c5_try < best_c5 - 1e-14:

    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        h = h * (target / s)
    else:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            h = h * (target / max(s, 1e-14))
        else:
            h = h + room * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def inner_gradient(h, p, n):
    grad = np.zeros(n)
    for s_idx in range(2 * n - 1):
        ps = p[s_idx]
        if abs(ps) < 1e-16:
            continue
        shift = s_idx - (n - 1)
        if shift >= 0:
            idx = np.arange(0, n - shift)
            grad[idx] += ps * (1.0 - h[idx + shift])
            idx2 = np.arange(shift, n)
            grad[idx2] -= ps * h[idx2 - shift]
        else:
            a = -shift
            idx = np.arange(a, n)
            grad[idx] += ps * (1.0 - h[idx - a])
            idx2 = np.arange(0, n - a)
            grad[idx2] -= ps * h[idx2 + a]
    return grad

def minimax_solve(h, n, t_end, max_outer=80, lr_outer=1.0, inner_steps=25, step_base=0.15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    S = 2 * n - 1
    p = np.ones(S) / S
    for outer in range(max_outer):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        c5_raw = np.max(corr)
        T_p = 3.0 + 15.0 * (outer / max(1, max_outer - 1))
        p_new = p * np.exp(lr_outer * (corr - c5_raw) / max(T_p * c5_raw, 1e-8))
        p_new = np.clip(p_new, 1e-12, None)
        p = p_new / p_new.sum()
        h_curr = best_h.copy()
        step = step_base / (np.linalg.norm(p, 1) + 0.1)
        step = min(step, 0.08)
        for inner in range(inner_steps):
            if time.time() > t_end:
                break
            g = inner_gradient(h_curr, p, n)
            h_curr = h_curr - step * g
            h_curr = project_h(h_curr)
            step *= 0.97
        c5_new = compute_c5(h_curr)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_curr.copy()
    return best_h, best_c5

def coord_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    dx = 2.0 / n
    S = 2 * n - 1
    coef = np.zeros(S)
    for s_idx in range(S):
        shift = s_idx - (n - 1)
        if shift >= 0:
            if 0 <= i < n - shift:
                coef[s_idx] += (1.0 - h[i + shift])
            if shift <= i < n:
                coef[s_idx] += (-h[i - shift])
        else:
            a = -shift
            if a <= i < n:
                coef[s_idx] += (1.0 - h[i - a])
            if 0 <= i < n - a:
                coef[s_idx] += (-h[i + a])
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = np.max(corr_base + m1 * coef)
        v2 = np.max(corr_base + m2 * coef)
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=60):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            x = coord_single(best_h, n, i)
            if abs(x - best_h[i]) > 1e-10:
                h_try = best_h.copy()
                h_try[i] = x
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    imp = True
        if not imp:
            break
    return best_h, best_c5

def swap_search(h, n, rng, t_end, n_trials=800):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_trials):
        if time.time() > t_end:
            break
        i, j = rng.choice(n, 2, replace=False)
        if abs(best_h[i] - best_h[j]) < 1e-5:
            continue
        h_try = best_h.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def slsqq_polish(h, n, T=5000, maxiter=800):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -60.0, 60.0)
        return float((m + np.log(np.sum(np.exp(sh))) / T) * dx)
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.zeros(160)
        h[:80] = 1.0
    n = len(h)
    h = project_h(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # Phase 1: Minimax (25%)
    t1 = t0 + budget_s * 0.25
    best_h, best_c5 = minimax_solve(best_h, n, t1, max_outer=120, lr_outer=0.8, inner_steps=20, step_base=0.12)
    # Phase 2: Coordinate descent (15%)
    t2 = t0 + budget_s * 0.40
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=80)
    # Phase 3: Swap search (10%)
    t3 = t0 + budget_s * 0.50
    best_h, best_c5 = swap_search(best_h, n, rng, t3, n_trials=600)
    # Phase 4: Aggressive minimax (20%)
    t4 = t0 + budget_s * 0.70
    best_h, best_c5 = minimax_solve(best_h, n, t4, max_outer=200, lr_outer=2.0, inner_steps=12, step_base=0.06)
    # Phase 5: SLSQP polish (10%)
    t5 = t0 + budget_s * 0.80
    if time.time() < t5:
        for T in [3000, 20000, 150000]:
            if time.time() > t5:
                break
            hc = slsqq_polish(best_h, n, T=T, maxiter=500)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Phase 6: Final coord descent (10%)
    t6 = t0 + budget_s * 0.92
    best_h, best_c5 = coord_descent(best_h, n, t6, max_sweeps=100)
    # Phase 7: Final swap + minimax micro (7%)
    t7 = t0 + budget_s * 0.97
    best_h, best_c5 = swap_search(best_h, n, rng, t7, n_trials=400)
    if time.time() < t7 + budget_s * 0.02:
        best_h, best_c5 = minimax_solve(best_h, n, t0 + budget_s * 0.99, max_outer=30, lr_outer=3.0, inner_steps=8, step_base=0.03)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
