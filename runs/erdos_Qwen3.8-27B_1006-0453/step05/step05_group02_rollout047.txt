import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def solve_coord_lp(corr_base, coefs):
    a = corr_base
    b = coefs
    pos = b > 1e-12
    neg = b < -1e-12
    zero = ~(pos | neg)
    if np.any(zero) and np.max(a[zero]) > 1e10:
        return 0.5
    v_lo = float(np.max(a + np.minimum(b, 0.0)))
    v_hi = float(min(np.max(a), np.max(a + b)))
    if v_hi <= v_lo:
        v_hi = v_lo + 1e-10
    for _ in range(45):
        v = (v_lo + v_hi) / 2.0
        t_min = 0.0
        t_max = 1.0
        if np.any(pos):
            t_max = min(t_max, float(np.min((v - a[pos]) / b[pos])))
        if np.any(neg):
            t_min = max(t_min, float(np.max((v - a[neg]) / b[neg])))
        if (np.any(zero) and np.max(a[zero]) > v) or t_min > t_max:
            v_lo = v
        else:
            v_hi = v
    v = (v_lo + v_hi) / 2.0
    t_min = 0.0
    t_max = 1.0
    if np.any(pos):
        t_max = min(t_max, float(np.min((v - a[pos]) / b[pos])))
    if np.any(neg):
        t_min = max(t_min, float(np.max((v - a[neg]) / b[neg])))
    return float(np.clip((t_min + t_max) / 2.0, 0.0, 1.0))

def coord_update(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h_base[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h_base[np.clip(j2[v2], 0, n - 1)])
    return solve_coord_lp(corr_base, coefs)

def coord_descent(h, n, t_end, max_sweeps=500, rng=None):
    if rng is None:
        rng = np.random
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    patience = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_update(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp = True
                patience = 0
            else:
                patience += 1
        if not imp:
            break
    return best_h, best_c5

def slsqp_step(h, T, n, maxiter=800):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def make_pert_active(h, n, corr, rng, cool):
    top_k = 3
    shift_idx = np.argsort(corr)[::-1][:top_k]
    scale = 0.02 + 0.10 * cool
    p = np.zeros(n)
    for si in shift_idx:
        k = int(si - (n - 1))
        if k == 0:
            continue
        valid_i = np.arange(max(0, -k), min(n, n - abs(k)))
        if len(valid_i) < 2:
            continue
        sub = rng.permutation(valid_i)[:min(12, len(valid_i))]
        p[sub] += scale * rng.standard_normal(len(sub))
    idx = rng.randint(n)
    p[idx] += 0.05 * cool * rng.standard_normal()
    return p

    # Phase 2: Heavy coordinate descent

    work_h = best_h.copy()

    n_iter = 30
    for it in range(n_iter):
