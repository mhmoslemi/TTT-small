def project_h(h):
    n = len(h)

    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)

        return (0.015 + 0.07 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        mask = np.random.random(n) < 0.18
        pert = np.zeros(n)
        pert[mask] = (0.04 + 0.12 * cool) * np.random.randn(max(1, mask.sum()))
        return pert
    elif mod == 3:
        pert = np.zeros(n)
        for _ in range(5):
            a, b = np.random.permutation(n)[:2]
            d = (base_h[b] - base_h[a]) * (0.4 + 0.6 * cool)
            pert[a] += d
            pert[b] -= d
        return pert
    elif mod == 4:
        blk = max(1, n // 10)
        b1 = np.random.randint(0, max(1, n // blk))
        b2 = np.random.randint(0, max(1, n // blk))
        if b1 == b2:
            b2 = (b2 + 1) % max(1, n // blk)
        pert = np.zeros(n)
        pert[b1*blk:(b1+1)*blk] = (base_h[b2*blk:(b2+1)*blk] - base_h[b1*blk:(b1+1)*blk]) * cool
        return pert
    else:
        mask = np.random.random(n) < 0.06
        pert = np.zeros(n)
        pert[mask] = (0.05 + 0.1 * cool) * np.sign(np.random.randn(max(1, mask.sum())))
        return pert

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP smooth-max with increasing T
    for T in [80, 800, 8000]:
        if time.time() - t0 > budget_s * 0.10:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=1200, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Quick coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.20, max_sweeps=18, use_pairs=False)

    # Phase 3: Fourier L-BFGS-B global optimization
    for K in [3, 5, 7]:
        if time.time() - t0 > budget_s * 0.38:
            break
        coeffs = fit_fourier(best_h, n, K)
        n_params = 2 * K + 1
        bounds = [(-0.3, 1.3)] + [(-0.6, 0.6)] * (2 * K)
        for trial in range(4):
            if time.time() - t0 > budget_s * 0.38:
                break
            if trial == 0:
                x0 = coeffs.copy()
            elif trial == 1:
                x0 = coeffs + 0.05 * np.random.randn(n_params)
            elif trial == 2:
                x0 = coeffs + np.random.uniform(-0.15, 0.15, n_params)
            else:
                x0 = np.random.uniform(-0.2, 1.2, n_params)
                x0[0] = 0.5
            try:
                res = minimize(fourier_obj, x0, args=(n, K, 800),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 3000, 'gtol': 1e-10, 'maxls': 50})
                h_f = fourier_eval(res.x, n, K)
                c5v = compute_c5(h_f)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_f.copy()
                hc = slsqp_opt(h_f, 3000, n, maxiter=600, ftol=1e-15)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc
            except Exception:
                pass

    # Phase 4: Coordinate descent after Fourier
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.45, max_sweeps=15, use_pairs=False)

    # Phase 5: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter_bh = 55
    for it in range(n_iter_bh):
        if time.time() - t0 > budget_s * 0.68:
            break
        cool = max(0.1, 1.0 - 0.85 * (it / n_iter_bh))
        pert = make_perturb(n, it, cool, best_h)
        h_pert = project_h(work_h + pert)
        T_bh = 200 + 6000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=350, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.00004
        if c5v < work_c5 or (c5v - work_c5 > -0.01 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: SLSQP high-T polish
    for T in [30000, 150000]:
        if time.time() - t0 > budget_s * 0.78:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Extended coordinate descent with pairs
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.93, max_sweeps=120, use_pairs=True)

    # Phase 8: Final SLSQP + coord
    if time.time() - t0 < budget_s * 0.95:
        hc = slsqp_opt(best_h, 400000, n, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.98, max_sweeps=20, use_pairs=False)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
