import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N, rng):
    nh = N // 2
    # Add tiny noise to break ties
    h_n = h + rng.random(N) * 1e-6
    idx = np.argsort(-h_n)
    out = np.zeros(N)
    out[idx[:nh]] = 1.0
    return out

def project_cont(h, target):
    h = np.clip(h, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5*(lo+hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def nesterov_step(h, N, lr, beta, m, top_k=80):
    """One Nesterov step. Returns new h, momentum m, peak value."""
    y = h + beta * m
    corr = fft_corr(y)
    pk = float(corr.max())
    center = N - 1
    
    # Weighted gradient using top shifts
    active = np.argsort(-corr)[:top_k]
    # Smooth weights: w[s] = (corr[s] - corr[active[-1]]) / (corr[active[0]] - corr[active[-1]] + eps)
    w = corr[active].copy()
    w_min = w[-1]
    w = (w - w_min) / (w[0] - w_min + 1e-12)
    
    grad = np.zeros(N)
    for i, s in enumerate(active):
        st = int(s) - center
        wi = w[i]
        if st == 0:
            grad += wi * (1.0 - 2.0 * y)
        elif st > 0:
            grad[:N-st] += wi * (1.0 - y[st:])
            grad[st:] -= wi * y[:N-st]
        else:
            M = -st
            grad[:N-M] += wi * (1.0 - y[M:])
            grad[M:] -= wi * y[:N-M]
    grad /= top_k
    
    h_new = project_cont(y - lr * grad, N / 2.0)
    m_new = beta * (h_new - h)
    return h_new, m_new, pk, corr

def optimize_nesterov(h, N, deadline, lr0=0.015, top_k=80):
    h = h.astype(float).copy()
    m = np.zeros(N)
    beta = 0.85
    lr = lr0
    best = h.copy()
    best_pk = float(fft_corr(h).max())
    
    for it in range(10000):
        if time.time() > deadline:
            break
        h, m, pk, _ = nesterov_step(h, N, lr, beta, m, top_k)
        if pk < best_pk:
            best_pk = pk
            best = h.copy()
        if it % 300 == 0:
            lr *= 0.995
            beta = min(0.95, 0.85 + it * 0.0001)
    
    return best, best_pk * (2.0/N)

            break

        else:

This is getting way too verbose. Let me just use a simpler approach: precompute the D matrix once per h configuration, and use it for all swaps.

Actually, the simplest efficient approach: for each p, compute the "removal effect" as a vector of length 2N-1. This can be done in O(N) per p (just a few slices into h). Then for finding the best q, we need max over zeros of (corr + delta_p - delta_q).max().

But computing delta_q for all zeros is O(N * 2N) which is what we want to avoid.

Alternative: for a fixed p, the new max after adding q is:
max_s (corr[s] + delta_p[s] - delta_q[s])

where delta_q[s] is the "addition effect" of h[q]: 0→1, which is -delta_remove[q].

So: max_s (corr[s] + delta_p[s] + delta_remove[q][s])

For a fixed p, this is: max over q in zeros of max_s (base[s] + delta_remove[q][s])
where base[s] = corr[s] + delta_p[s].

This is exactly what the previous code does with the D matrix! D[q, :] is delta_remove[q].

So the algorithm is:
1. Compute D matrix (N × (2N-1)): O(N²)
2. For sampled p: base = corr + D[p]
3. For each zero q: new_max = max(base - D[q]) → O(2N) per q
4. Pick best (p, q) pair
5. Apply swap
6. Repeat

Total per sweep: O(sample × zeros × 2N) = O(100 × 190 × 760) ≈ 14M for N=380.
With 400 sweeps: 5.7B. At ~10ns per op (numpy vectorized), that's ~57s. OK.

Now, for 2-flip moves: swap (p1, p2) ones with (q1, q2) zeros.
The change is D[p1] + D[p2] - D[q1] - D[q2] + corrections.
The corrections are small (at most a few entries change by ±1).
For the 2-flip, we can approximate: new_corr ≈ corr + D[p1] + D[p2] - D[q1] - D[q2]
And find the best quadruple. But that's O(ones² × zeros² × 2N) which is too expensive.

Instead: for 2-flip, we can try:
- For a fixed pair (p1, p2): base2 = corr + D[p1] + D[p2]
- Find best pair (q1, q2): min over q1≠q2 of max(base2 - D[q1] - D[q2])
- This is still O(zeros² × 2N) per pair.

Too expensive. Let's just do: for 2-flip, randomly try a few pairs and accept if it helps.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def make_binary(h, N, rng):
    nh = N // 2
    idx = np.argsort(-(h + rng.random(N) * 1e-8))
    out = np.zeros(N)
    out[idx[:nh]] = 1.0
    return out

def project_cont(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def compute_D(h, N):
    """D[p, s] = change in corr[s] when h[p] flips 1->0."""
    C = N - 1
    deltas = np.zeros((N, 2 * N - 1))
    s_range = np.arange(2 * N - 1)
    for p in range(N):
        # Direct: i=p, need 0<=p-(s-C)<N, i.e. s in [p, p+C]
        s0, s1 = p, min(p + C, 2 * N - 2)
        h_idx = p + C - s_range[s0:s1 + 1]
        m = (h_idx >= 0) & (h_idx < N)
        deltas[p, s0:s1 + 1][m] -= (1.0 - h[np.clip(h_idx[m], 0, N - 1)])
        # Indirect: i-s_C=p, i=p+(s-C), need 0<=i<N, s in [C-p, C+N-1-p]
        s0b, s1b = max(C - p, 0), min(C + N - 1 - p, 2 * N - 2)
        h_idx = p + s_range[s0b:s1b + 1] - C
        m = (h_idx >= 0) & (h_idx < N)
        deltas[p, s0b:s1b + 1][m] += h[np.clip(h_idx[m], 0, N - 1)]
    return deltas

def binary_search(h, N, deadline, rng, max_sweeps=400, sample=120, T_anneal=0.0):
    h = h.astype(float).copy()
    if int(round(h.sum())) != N // 2:
        h = make_binary(h, N, rng)
    best_c = float(fft_corr(h).max())
    D = compute_D(h, N)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80 and T_anneal <= 0:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if len(ones) > sample:
            p_list = rng.choice(ones, sample, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = corr + D[int(p)]
            cand = base[None, :] - D[zeros]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if T_anneal > 0 and rng.random() < T_anneal:
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
            else:
                p = ones[rng.randint(len(ones))]
                q = zeros[rng.randint(len(zeros))]
        else:
            p, q = best_pq
        h[p], h[q] = 0.0, 1.0
        D = compute_D(h, N)
        if T_anneal > 0:
            T_anneal *= 0.997
    return h, best_c

def optimize_nesterov(h, N, deadline, lr0=0.012, top_k=100):
    h = h.astype(float).copy()
    m = np.zeros(N)
    beta = 0.88
    lr = lr0
    best = h.copy()
    best_pk = float(fft_corr(h).max())
    C = N - 1
    for it in range(20000):
        if time.time() > deadline:
            break
        y = h + beta * m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best = h.copy()
        active = np.argsort(-corr)[:top_k]
        w = corr[active].copy()
        w = (w - w[-1]) / (w[0] - w[-1] + 1e-12)
        grad = np.zeros(N)
        for i, s in enumerate(active):
            st = int(s) - C
            wi = w[i]
            if st == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif st > 0:
                grad[:N - st] += wi * (1.0 - y[st:])
                grad[st:] -= wi * y[:N - st]
            else:
                M = -st
                grad[:N - M] += wi * (1.0 - y[M:])
                grad[M:] -= wi * y[:N - M]
        grad /= top_k
        h_new = project_cont(y - lr * grad, N / 2.0)
        m = beta * (h_new - h)
        h = h_new
        if it % 250 == 0:
            lr *= 0.994
            beta = min(0.96, 0.88 + it * 0.00015)
    return best, best_pk * (2.0 / N)
