def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Intensive binary search at N0 with soft scoring
    d1 = min(deadline - 400, t0 + 220)
    h_bin = (h0 > 0.5).astype(float)
    h_s, _ = binary_search_v2(h_bin, N0, d1, max_sweeps=900, sample_n=200)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Nesterov to explore continuous landscape
    if time.time() < deadline - 250:
        d2 = min(deadline - 230, time.time() + 100)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k=60, lr0=0.015)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary search from improved continuous
    if time.time() < deadline - 130:
        d3 = min(deadline - 110, time.time() + 130)
        h_bin3 = (best_h > 0.5).astype(float)
        h_s3, _ = binary_search_v2(h_bin3, N0, d3, max_sweeps=700, sample_n=220, anneal_start=200)
        c5_3 = compute_c5(h_s3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Upscale to 2x resolution (760 points)
    N_up = N0 * 2
    if time.time() < deadline - 60:
        x_best = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, N_up, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        h_bt = fix_sum_binary(h_bt, N_up)
        d4 = min(deadline - 25, time.time() + 90)
        h_rt, _ = binary_search_v2(h_bt, N_up, d4, max_sweeps=400, sample_n=150)
        c5_rt = compute_c5(h_rt, N_up)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 5: Final Nesterov polish at best resolution
    Nf = len(best_h)
    if time.time() < deadline - 5:
        hF = best_h.astype(float) + np.random.randn(Nf) * 0.02
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = nesterov_opt(hF, Nf, deadline - 2, k=80, lr0=0.01)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 6: Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = (best_h > 0.5).astype(float)
        hF2 = fix_sum_binary(hF2, len(hF2))
        hF2, _ = binary_search_v2(hF2, len(hF2), deadline - 0.5, max_sweeps=50, sample_n=150)
        c5F2 = compute_c5(hF2, len(hF2))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
