    tgt = n * 0.5
    s = h.sum()

        return h
    if s > tgt:

    else:

        if t < 1e-13:
            return np.full(n, 0.5)

    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:

        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)

def solve_lp(n):
    """Solve McCormick LP relaxation for the Erdos problem."""
    h = cp.Variable(n)
    t = cp.Variable()
    # For each shift k, we need sum h_i - sum y_{i,i+k} <= t
    # y_{i,i+k} represents h_i * h_{i+k}
    # Collect all valid (i, i+k) pairs
    pairs = []  # (i, j, k) where j = i+k
    for k in range(-(n-1), n):
        if k == 0:
            continue
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                pairs.append((i, j, k))
    # Group by k
    from collections import defaultdict
    k_groups = defaultdict(list)
    for i, j, k in pairs:
        k_groups[k].append((i, j))
    
    # Create y variables
    y_vars = {}
    for idx, (i, j) in enumerate(pairs):
        y_vars[(i, j)] = cp.Variable()
    
    # Objective: minimize t
    # Constraints
    constraints = [
        cp.sum(h) == n / 2.0,
        h >= 0,
        h <= 1,
        t >= 0
    ]
    
    for k, pair_list in k_groups.items():
        # c_k = sum of h_i for valid i - sum of y_{i,i+k}
        # We need: sum_{(i,j) in pair_list} h_i - sum y_{i,j} <= t * n / 2
        # Actually: c_k * (2/n) ≤ t, so c_k ≤ t * n / 2
        # c_k = sum_i h[i]*(1-h[i+k]) for valid i
        # = sum_i h[i] - sum_i h[i]*h[i+k]
        sum_h = 0
        sum_y = 0
        for (i, j) in pair_list:
            sum_h += h[i]
            sum_y += y_vars[(i, j)]
            constraints.append(y_vars[(i, j)] <= h[i])
            constraints.append(y_vars[(i, j)] <= h[j])
            constraints.append(y_vars[(i, j)] >= h[i] + h[j] - 1)
            constraints.append(y_vars[(i, j)] >= 0)
        constraints.append(sum_h - sum_y <= t * n / 2.0)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.GLOP)  # or CBC, CLARABEL
    if h.value is None:
        return np.full(n, 0.5)
    h_val = np.array(h.value).flatten()
    return proj_h(h_val, n)

        m = c.max()

        return (m + float(np.log(float(e.sum()))) / T) * 2.0 / n
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()

        w = e / e.sum()

            wk = w[r]

                continue

            ip = np.arange(n) + e_r
            im = np.arange(n) - e_r
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            g[vm] += wk * (1.0 - x[im[vm]])
            g[vp] -= wk * x[ip[vp]]
        return g * 2.0 / n

            im = i - ell

            coeff[vm] -= best[im[vm]]

from scipy.optimize import minimize_scalar, minimize

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h.copy()
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        tr = rm.sum()
        if tr < 1e-14:
            return np.full(n, 0.5)
        h = h + rm * ((tgt - s) / tr)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        idx = np.where((h > 0.005) & (h < 0.995))[0]
        if len(idx) > 0:
            h[idx] += d / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(np.clip(T * (c - m), -400, 400))
    return (m + float(np.log(float(e.sum()))) / T) * 2.0 / len(x)

def _smooth_grad(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(np.clip(T * (c - m), -400, 400))
    w = e / e.sum()
    nf = 2 * n - 1
    ell = np.arange(nf) - (n - 1)
    g = np.zeros(n)
    for r in range(nf):
        wk = w[r]
        if wk < 1e-15:
            continue
        er = ell[r]
        ip = np.arange(n) + er
        im = np.arange(n) - er
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        g[vm] += wk * (1.0 - x[im[vm]])
        g[vp] -= wk * x[ip[vp]]
    return g * 2.0 / n

                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12})
        return proj_h(r.x, n)
    except Exception:
        return h0

    ell = np.arange(nf) - (n - 1)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = 0
        corr = np.correlate(best, 1.0 - best, mode='full')
        for i in np.random.permutation(n):
            if time.time() > t_end:
                break
            old = best[i]
            ip = i + ell
            im = i - ell
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            coeff = np.zeros(nf)
            coeff[vp] = 1.0 - best[ip[vp]]
            coeff[vm] -= best[im[vm]]
            base = corr - old * coeff
            def f1d(v):
                return float(np.max(base + v * coeff))
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': xatol})
