Now, the minimize_scalar on max of affine functions is a piecewise-linear minimization. The minimum occurs at a breakpoint where two lines cross, or at a boundary. minimize_scalar with 'bounded' should handle this fine.

Let me also add the binary swap idea: instead of changing one coordinate, swap two binary coordinates (change h[i] from 0 to 1 and h[j] from 1 to 0). This maintains the sum automatically.

For the swap: Δcorr = (1-h[i])·coefs_i + h[j]·coefs_j (wait, that's not right either).

Actually, for a swap: h[i] goes 0→1 (Δ=+1) and h[j] goes 1→0 (Δ=-1).
Δcorr = 1·coefs_i + (-1)·coefs_j = coefs_i - coefs_j

So I can precompute all coefs and then evaluate swaps in O(1) per pair (just looking at max of corr + coefs_i - coefs_j).

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def get_coefs_all(h, n):
    """Precompute coefficient vectors for all coordinates."""
    coefs = np.zeros((n, 2*n-1))
    for i in range(n):
        old = h[i]
        h[i] = 0.0
        c0 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = old
        coefs[i] = c1 - c0
    return coefs

def coord_sweep(h, n, t_end, max_sweeps=500):
    """Exact coordinate descent."""
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_max = float(np.max(corr))

            break
        improved = False

                best_h = np.clip(best_h, 0, 1)
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                best_max = float(np.max(corr))
                return best_h, best_max * dx
            old = best_h[i]
            best_h[i] = 0.0
            c0 = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_h[i] = 1.0
            c1 = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_h[i] = old
            coefs = c1 - c0
            
            def f(v):
                return np.max(corr + (v - old) * coefs)

                                  options={'xatol': 1e-9, 'maxiter': 80})
            v_new = res.x
            if abs(v_new - old) > 1e-10:

                corr = corr + (v_new - old) * coefs
                new_max = float(np.max(corr))

                    improved = True

        best_h = project_h(best_h, n)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        best_max = float(np.max(corr))
        if not improved:
            break
    
    return best_h, best_max * dx

def swap_sa(h, n, t_end, temp_init=0.01, temp_final=0.001, max_iter=5000):
    """Binary swap simulated annealing."""
    dx = 2.0 / n
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_max = float(np.max(corr))
    cur_max = best_max

    zeros = np.where(best_h < 0.5)[0]

        if time.time() > t_end:
            break
        T = temp_init * (temp_final / temp_init) ** (it / max_iter)
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = np.random.choice(zeros)
        j = np.random.choice(ones)
        
        # Compute effect of swap
        old_hi = best_h[i]
        old_hj = best_h[j]
        best_h[i] = 1.0 - old_hi
        best_h[j] = 1.0 - old_hj
        new_corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        new_max = float(np.max(new_corr))
        
        if new_max <= cur_max + T * dx:
            cur_max = new_max
            # Update ones/zeros
            if old_hi < 0.5:  # was 0, now 1
                zeros = zeros[zeros != i]
                ones = np.append(ones, i)
            else:
                ones = ones[ones != i]
                zeros = np.append(zeros, i)
            if old_hj > 0.5:  # was 1, now 0
                ones = ones[ones != j]
                zeros = np.append(zeros, j)
            else:
                zeros = zeros[zeros != j]
                ones = np.append(ones, j)
            if cur_max < best_max:
                best_max = cur_max
        else:
            best_h[i] = old_hi
            best_h[j] = old_hj
    
    best_h = project_h(best_h, n)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_max = float(np.max(corr))
    return best_h, best_max * dx

        best_h = work_h

    # Phase 5: SA coordinate descent (slightly stochastic)
    t5 = t0 + budget_s * 0.92
    best_h, best_c5 = coord_sweep(best_h, n, t5, max_sweeps=300, sa_temp=0.002)
    # Re-deterministic polish
    best_h, best_c5 = coord_sweep(best_h, n, t5, max_sweeps=200)

    # Phase 6: Final polish
    t6 = t0 + budget_s * 0.99
    best_h, best_c5 = coord_sweep(best_h, n, t6, max_sweeps=200)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
