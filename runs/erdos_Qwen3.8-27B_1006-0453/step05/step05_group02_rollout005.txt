def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    m = np.max(corr)
    # Softmax weights
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    p = exps / Z  # length 2n-1, indexed 0..2n-2, corresponding to k=-(n-1)..(n-1)
    
    # Objective value
    val = (m + np.log(Z) / T) * dx
    
    # Gradient
    # p is indexed 0..2n-2, where index i corresponds to lag k = i - (n-1)
    # grad[j] = dx * sum_k p(k) * [(1-h(j-k))*1(0<=j-k<n) - h(j+k)*1(0<=j+k<n)]
    
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0  # sum over valid (j-k)
        s2 = 0.0  # sum over valid (j+k)
        for i in range(2*n-1):
            k = i - (n-1)
            a = j - k  # for (1-h[j-k])
            b = j + k  # for h[j+k]
            if 0 <= a < n:
                s1 += p[i] * (1.0 - h[a])
            if 0 <= b < n:
                s2 += p[i] * h[b]
        grad[j] = dx * (s1 - s2)
    
    return val, grad

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    p = exps / Z
    val = (m + np.log(Z) / T) * dx
    # Gradient computation
    js = np.arange(n)[:, None]
    ks = np.arange(-(n-1), n)[None, :]
    a = js - ks
    b = js + ks
    mask_a = (a >= 0) & (a < n)
    mask_b = (b >= 0) & (b < n)
    one_minus_h = 1.0 - h
    h_safe = h[np.clip(b, 0, n-1)]
    omh_safe = one_minus_h[np.clip(a, 0, n-1)]
    term1 = np.sum(p[None, :] * mask_a * omh_safe, axis=1)
    term2 = np.sum(p[None, :] * mask_b * h_safe, axis=1)
    grad = dx * (term1 - term2)
    return val, grad

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def constraint_jac(x, n):
    return np.ones(n)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_grad(h, T, n, maxiter=500):
    try:
        res = minimize(smooth_c5_obj, h, args=(T,),
                       jac=lambda x: smooth_c5_grad(x, T)[1],
                       method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,), 'jac': constraint_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    dx = 2.0 / n
    try:
        res = minimize_scalar(lambda hi: np.max(corr_base + hi * coefs) * dx,
                              bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def upscale(h, factor=2):
    n = len(h)
    h2 = np.zeros(n * factor)
    for i in range(n):
        h2[i*factor:(i+1)*factor] = h[i]
    return h2

from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    p = exps / Z
    val = (m + np.log(Z) / T) * dx
    js = np.arange(n)[:, None]
    ks = np.arange(-(n - 1), n)[None, :]
    a = js - ks
    b = js + ks
    mask_a = (a >= 0) & (a < n)
    mask_b = (b >= 0) & (b < n)
    omh = 1.0 - h
    term1 = np.sum(p[None, :] * mask_a * omh[np.clip(a, 0, n - 1)], axis=1)
    term2 = np.sum(p[None, :] * mask_b * h[np.clip(b, 0, n - 1)], axis=1)
    return val, dx * (term1 - term2)

        return h

    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)

            hc2 = slsqp_grad(h2, T, n2, maxiter=500)
            c5v2 = compute_c5(hc2)
            if c5v2 < c5_2:
                c5_2 = c5v2
                h2 = hc2
        h2, c5_2 = coord_descent(h2, n2, t0 + budget_s * 0.76, max_sweeps=80, rng=rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            n = n2

    # Phase 6: More basin hopping on best
    n = len(best_h)
    best_h, best_c5 = basin_hop(best_h, best_c5, n, t0 + budget_s * 0.88, rng,
                                ni=25, cool_base=0.6, t_scale=6000, maxiter=500, temp_scale=0.0004)

    # Phase 7: Final polish
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, max_sweeps=200, rng=rng)
    for T in [300000, 1000000]:
        if time.time() - t0 > budget_s * 0.97:
            break
        hc = slsqp_grad(best_h, T, n, maxiter=1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    if time.time() - t0 < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=60, rng=rng)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
