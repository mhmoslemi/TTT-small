```python
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = fast_lin(best_h, i, n)
            # Minimize max(corr + d*lin) * dx over d ∈ [-old, 1-old]
            # Also corr[n-1] has an extra -d^2 term
            # Actually: corr[n-1] = sum h[j](1-h[j]), when we change h[i] by d:
            # new corr[n-1] = old_corr[n-1] + d*(1-2*old) - d^2
            # For other lags: linear in d
            # So we minimize max(
            #   max_{m≠n-1} (corr[m] + d*lin[m]),
            #   corr[n-1] + d*lin[n-1] - d^2
            # ) * dx
            # lin[n-1] = 1-2*old, and the -d^2 term is separate
            # Actually: the change at lag 0 is d*(1-2*old) - d^2
            # But lin[n-1] already = 1-2*old, so:
            # new_corr[n-1] = corr[n-1] + d*lin[n-1] - d^2
            # And for m ≠ n-1: new_corr[m] = corr[m] + d*lin[m]
            
            def f1d(d):
                c_new = corr + d * lin
                c_new[n-1] -= d * d
                return float(np.max(c_new) * dx)
            
            res = minimize_scalar(f1d, bounds=(-old, 1.0-old), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 50})
            d = float(res.x)
            if abs(d) < 1e-12:
                continue
            best_h[i] += d
            corr += d * lin
            corr[n-1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n-1] += d * d
        if not improved:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5
```

Let me also try a completely different approach: **subgradient method with restarts**.

The idea: C₅(h) = max_k c_k(h) where c_k is the k-th correlation value.
Subgradient at h: g = dx * Σ_k p_k * ∇c_k where p_k ∝ exp(T*c_k) (softmax).

```python
def grad_descent(h_init, n, t_end, max_iters=5000, seed=42):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        # Softmax weights
        T = 20.0 + 100.0 * (it / max_iters)
        exponents = T * (corr - m)
        exponents = np.clip(exponents, -50, 0)
        p = np.exp(exponents)
        p /= p.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for m_idx in range(2 * n - 1):
            w = p[m_idx]
            if w < 1e-12:
                continue
            s = m_idx - (n - 1)
            if s > 0:
                grad[s:] += w * (1.0 - h[:n-s])
                grad[:n-s] -= w * h[s:]
            elif s < 0:
                a = -s
                grad[:n-a] += w * (1.0 - h[a:])
                grad[a:] -= w * h[:n-a]
            else:
                grad += w * (1.0 - 2.0 * h)
        grad *= 2.0 / n
        
        # Line search
        alpha = 0.5
        for ls in range(20):
            h_new = project_h(h - alpha * grad, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 or ls == 0:
                break
            alpha *= 0.5
        else:
            break
        
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return best_h, best_c5
```
