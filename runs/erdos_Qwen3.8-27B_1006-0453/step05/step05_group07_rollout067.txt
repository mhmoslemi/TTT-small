Actually, I want to think about one more thing: the `minimize_scalar` call in coordinate descent. The function `max(base + v*coef)` is piecewise linear in v. The minimum is at the intersection of two adjacent line segments, or at a boundary.

- At v=0: f(0) = max(base)
- At v=1: f(1) = max(base + coef)
- The minimum is where the "active" set changes

```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

_T = [1.0]
_CB = [np.zeros(1)]
_CC = [np.zeros(1)]

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = 0.5*n
    for _ in range(30):
        s = h.sum()
        if abs(s-tgt) < 1e-12: break
        if s > tgt:
            h *= tgt/max(s,1e-15)
        else:
            rm = 1.0-h
            t = rm.sum()
            if t > 1e-15: h += rm*((tgt-s)/t)
        h = np.clip(h, 0.0, 1.0)
    d = tgt-h.sum()
    if abs(d) > 1e-10:
        m = (h>0.05)&(h<0.95)
        if m.sum()>0:
            h[m] += d/m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse(n):
    t = np.zeros(n)
    for i in range(1,n):
        t[i] = 1.0 if bin(i).count('1')&1 else 0.0
    return t

def _f1d(v):
    return float(np.max(_CB[0]+v*_CC[0]))

def sm_obj(x):
    T=_T[0]
    c=np.correlate(x,1.0-x,mode='full')
    m=c.max(); n=len(x)
    return (m+np.log(np.sum(np.exp(T*(c-m))))/T)*2.0/n

def sm_grad(x):
    T=_T[0]; n=len(x); mf=2*n-1
    c=np.correlate(x,1.0-x,mode='full')
    m=c.max(); e=np.exp(T*(c-m)); w=e/e.sum()
    g=np.zeros(n)
    for k in range(mf):
        wk=w[k]
        if wk<1e-14: continue
        lo1,hi1=max(0,n-1-k),min(n,2*n-1-k)
        if lo1<hi1: g[lo1:hi1]+=wk*(1.0-x[np.arange(lo1,hi1)+k-n+1])
        lo2,hi2=max(0,k-n+1),min(n,k+1)
        if lo2<hi2: g[lo2:hi2]-=wk*x[np.arange(lo2,hi2)-k+n-1]
    return g*2.0/n

    try:

                   options={'maxiter':iters,'ftol':1e-18,'gtol':1e-10})
        return proj_h(r.x,n)
    except: return h

def coord_desc(h,n,t_end,sweeps=200):

    for sw in range(sweeps):
        if time.time()>t_end: break
        imp=False
        cb=np.correlate(best,1.0-best,mode='full')
        if sw%3==0: order=np.arange(n)
        elif sw%3==1: order=np.arange(n-1,-1,-1)
        else: order=np.random.permutation(n)
        for i in order:
            if time.time()>t_end: return best,bc5
            old=best[i]
            coef=np.zeros(mf)
            for k in range(mf):

                if 0<=a<n: coef[k]+=1.0-best[a]
                b=i-k+n-1
                if 0<=b<n: coef[k]-=best[b]
            _CB[0]=cb-old*coef
            _CC[0]=coef
            r=minimize_scalar(_f1d,bounds=(0.0,1.0),method='bounded',options={'xatol':1e-12})
            nv=float(r.x)
```

Actually, I want to add one more idea: **subgradient descent with the active set**.

After coord desc, identify the top-5 correlations. Then do a few steps of projected subgradient descent on just those 5, with a decreasing step size. This can help fine-tune the solution when the soft-max has already converged.

The subgradient of max(c_{k1}, ..., c_{k5}) at h is the sum of the gradients of the active c_k's (weighted by the softmax weights).

The gradient of c_k with respect to h[i]:
- If i appears as first factor: (1-h[j]) where j = i+k-n+1 (if valid)
- If i appears as second factor: -h[j] where j = i-k+n-1 (if valid)

This is the same as what we compute in the soft-max gradient, but only for the active lags.

Let me add this as an optimization step.

Actually, this is essentially what coord desc does (one coordinate at a time). The subgradient method updates all coordinates simultaneously. Let me try it.

```python
def active_set_step(h, n, t_end, iters=50):
    best = h.copy()
    bc5 = compute_c5(best)
    alpha = 0.01
    for _ in range(iters):
        if time.time() > t_end: break
        corr = np.correlate(best, 1.0-best, mode='full')
        cmax = corr.max()
        # Find top-5 correlations
        idx = np.argsort(corr)[-5:]
        # Compute subgradient
        g = np.zeros(n)
        for k in idx:
            w = np.exp((corr[k]-cmax)) / np.sum(np.exp(corr[idx]-cmax))
            for i in range(n):
                a = i + k - n + 1
                if 0 <= a < n: g[i] += w * (1.0-best[a])
                b = i - k + n - 1
                if 0 <= b < n: g[i] -= w * best[b]
        g *= 2.0/n
        # Projected gradient step (descending = minimize)
        trial = proj_h(best - alpha*g, n)
        c5t = compute_c5(trial)
        if c5t < bc5:
            best = trial
            bc5 = c5t
        else:
            alpha *= 0.8
    return best, bc5
```
