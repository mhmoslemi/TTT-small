def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    s = h.sum()
    if s < 1e-15:
        h[:] = 0.5; return h
    if s > tgt:
        h *= tgt / s
    else:
        room = 1.0 - h
        if room.sum() < 1e-15:
            h *= tgt / max(s, 1e-15)
        else:
            h += room * ((tgt - s) / room.sum())
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if mask.any():

    mx = c.max()

    return (mx + np.log(np.sum(np.exp(T * (c - mx)))) / T) * 2.0 / n
