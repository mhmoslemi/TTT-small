def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = compute_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(np.clip(T * (corr - m), -50.0, 50.0)))) / T
    return val * (2.0 / len(h))

def grad_smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(np.clip(T * (corr - m), -50.0, 50.0))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-13:
            continue
        lag = k - (n - 1)
        lo_a = max(0, -lag)
        hi_a = min(n, n - lag)
        if hi_a > lo_a:
            ia = np.arange(lo_a, hi_a)
            grad[ia] += w[k] * (1.0 - h[ia + lag])
        lo_b = max(0, lag)
        hi_b = min(n, n + lag)
        if hi_b > lo_b:
            ib = np.arange(lo_b, hi_b)
            grad[ib] -= w[k] * h[ib - lag]
    return grad * dx

    n = len(h)

    t = n / 2.0

        if r.sum() < 1e-14:
            return h
        h = h + r * (d / r.sum())
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
