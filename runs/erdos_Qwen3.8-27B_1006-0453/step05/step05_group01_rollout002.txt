from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, T, lam=0.0):
    """Log-sum-exp smoothed objective + flat penalty on top values"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    if lam > 0:
        # penalty for spread in top correlations
        top_k = min(5, len(corr))
        top = np.sort(corr)[-top_k:]
        val += lam * np.std(top)
    return val * dx

def compute_subgrad(h, T, top_k=10):
    """Compute informative subgradient focused on active shifts"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft weights for top shifts
    m = corr.max()
    w = np.exp(T * (corr - m))
    w = w / w.sum()
    # Focus on top_k for efficiency
    top_idx = np.argsort(w)[-top_k:]
    w = np.zeros(len(corr))
    w[top_idx] = np.exp(T * (corr[top_idx] - m))
    w = w / w.sum()
    
    shifts = np.arange(len(corr)) - (n - 1)
    # Only use top_k shifts for subgradient computation
    active_shifts = shifts[top_idx]
    active_w = w[top_idx]
    
    i_idx = np.arange(n)[:, None]
    s_idx = active_shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    
    # grad for each active shift
    grads = ((1.0 - h[jpc]) * v1 - h[jmc] * v2) * dx  # shape (n, top_k)
    # Weighted average
    grad = grads.T @ active_w  # shape (n,)
    return grad, top_idx, shifts

def projected_line_search(h, grad, T, max_iter=50):
    """Backtracking line search with projection"""
    n = len(h)
    f0 = soft_max_obj(h, T)
    alpha = 0.01
    for _ in range(max_iter):
        h_new = project_h(h - alpha * grad)
        f_new = soft_max_obj(h_new, T)
        if f_new < f0 - 1e-4 * alpha * np.dot(grad, h - h):  # Armijo
            return h_new, alpha
        alpha *= 0.5
    return h, 0.0

def grad_optimize(h, n, t_end, T_start=5000, T_end=50, n_iter=200):
    """Gradient-based optimization with annealing and momentum"""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        T = T_start * (T_end / T_start) ** (1 - cool)  # geometric cooling
        
        grad, top_idx, _ = compute_subgrad(h, T, top_k=min(15, n//4))
        
        # Nesterov momentum
        momentum = 0.7 * momentum + 0.3 * grad
        grad_eff = momentum
        
        # Adaptive step size
        alpha = 0.005 + 0.02 * cool
        h_new = project_h(h - alpha * grad_eff)
        
        # Check improvement

            alpha *= 0.5
            h_new = project_h(h - alpha * grad_eff)
        
        c5_try = compute_c5(h_new)
        c5_cur = compute_c5(h)
        
        # Accept if better or with probability (annealing)

        if c5_try < c5_cur or np.random.rand() < np.exp(-(c5_try - c5_cur) / max(temp, 1e-8)):
            h = h_new
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_new.copy()
    
    return best_h, best_c5

def slsqp_refine(h, T, n, maxiter=500):
    """SLSQP refinement with smooth objective"""
    try:
        res = minimize(
            lambda x: soft_max_obj(np.clip(x, 0, 1), T),
            h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

def coord_descent(h, n, t_end, max_sweeps=100, n_top=15):
    """Coordinate descent on the max-overlap"""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        perm = np.random.permutation(n)
        
        for i in perm:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp, jm = i + k, i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = np.array([corr[ti] - best_h[i] * coefs[t] for t, ti in enumerate(top_idx)])
            # Maximize min of (base + t*coefs) over top shifts, or minimize max
            res = minimize_scalar(
                lambda t: np.max(base + t * coefs),
                bounds=(0.0, 1.0), method='bounded',
                options={'xatol': 1e-12}
            )
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

    active_s = shifts[top_idx]
    ii = np.arange(n)[:, None]
    ss = active_s[None, :]
    jp = ii + ss
    jm = ii - ss
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grads = ((1.0 - h[jpc]) * v1 - h[jmc] * v2) * dx
    grad = grads.T @ w_top
    return grad, top_idx

def slsqp_solve(h, T, n, maxiter=600, lam=0.0):
    try:
        res = minimize(lambda x: soft_obj(np.clip(x, 0, 1), T, lam),
                       h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def grad_anneal(h, n, t_end, n_iter=None):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    n_iter = n_iter or int((t_end - time.time()) / 0.3)
    n_iter = max(n_iter, 30)
    mom = np.zeros(n)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        T = 5000 * (60 / 5000) ** (1.0 - cool)
        tk = max(5, int(15 * cool + 3))
        grad, _ = subgrad_active(cur_h, T, tk)
        mom = 0.65 * mom + 0.35 * grad
        alpha = 0.003 + 0.015 * cool
        h_try = project_h(cur_h - alpha * mom)
        if np.allclose(h_try, cur_h, atol=1e-12):
            alpha *= 0.5
            h_try = project_h(cur_h - alpha * mom)
        c5_try = compute_c5(h_try)
        c5_cur = compute_c5(cur_h)
        temp_acc = 0.0008 * cool + 0.00005
        if c5_try < c5_cur or (np.isfinite(c5_try - c5_cur) and np.random.rand() < np.exp(-(c5_try - c5_cur) / temp_acc)):
            cur_h = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = cur_h.copy()
        else:
            mom *= 0.5
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=120, n_top=20):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = (top_idx - (n - 1)).astype(int)
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            base = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                b = corr[top_idx[t]]
                jp, jm = i + k, i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                    b -= 0.0
                coefs[t] = c
                base[t] = corr[top_idx[t]] - best_h[i] * c
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved and sweep > 5:
            break
    return best_h, best_c5

def targeted_perturb(h, n, scale, n_shifts=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_shifts:]
    shifts = (top_idx - (n - 1)).astype(int)
    pert = np.zeros(n)
    np.random.seed(np.random.randint(100000))
    for k in shifts:
        if k >= 0 and n - k > 0:
            pert[:n - k] += (h[k:n] - 0.5) * scale
        elif k < 0 and n + k > 0:
            pert[-k:n] += (h[:n + k] - 0.5) * scale
    return project_h(h + np.random.randn(n) * scale * 0.3)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(200)
        h[:100] = 1.0
    n = len(h)
    h = project_h(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: SLSQP warmup (multiple T)
    t_p1 = t0 + budget_s * 0.18
    for T in [500, 3000, 15000, 60000]:
        if time.time() > t_p1:
            break
        hc = slsqp_solve(h, T, n, 500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h, h = c5v, hc, hc

    # Phase 2: Gradient annealing
    t_p2 = t0 + budget_s * 0.50
    h_g, c5_g = grad_anneal(h, n, t_p2)
    if c5_g < best_c5:
        best_c5, best_h, h = c5_g, h_g, h_g

    # Phase 3: High-T SLSQP (near-exact max)
    t_p3 = t0 + budget_s * 0.68
    for T in [200000, 1000000, 5000000]:
        if time.time() > t_p3:
            break
        hc = slsqp_solve(h, T, n, 700)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h, h = c5v, hc, hc

    # Phase 4: Coordinate descent (main refinement)
    t_p4 = t0 + budget_s * 0.85
    h_cd, c5_cd = coord_descent(h, n, t_p4, 100, 25)
    if c5_cd < best_c5:
        best_c5, best_h, h = c5_cd, h_cd, h_cd

    # Phase 5: Perturb + re-optimize (escape local minima)
    t_p5 = t0 + budget_s * 0.93
    for attempt in range(8):
        if time.time() > t_p5:
            break
        r = np.random.random()
        if r < 0.4:
            h_p = targeted_perturb(h, n, 0.04 * (1 - attempt / 8.0), 5)
        elif r < 0.7:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = 0.08 * np.random.randn(max(int(mask.sum()), 1))
            h_p = project_h(h + pert)
        else:
            f = np.random.randint(2, 15)
            p = np.random.uniform(0, 2 * np.pi)
            h_p = project_h(h + 0.03 * np.sin(2 * np.pi * f * np.arange(n) / n + p))
        hc = slsqp_solve(h_p, 50000, n, 400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h, h = c5v, hc, hc
            h_cd2, c5_cd2 = coord_descent(h, n, t_p5, 40, 25)
            if c5_cd2 < best_c5:
                best_c5, best_h, h = c5_cd2, h_cd2, h_cd2

    # Phase 6: Final polish
    if time.time() < t_end - 5:
        hc = slsqp_solve(h, 20000000, n, 1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
    if time.time() < t_end - 2:
        h_f, c5_f = coord_descent(best_h, n, t_end - 0.5, 50, 30)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
