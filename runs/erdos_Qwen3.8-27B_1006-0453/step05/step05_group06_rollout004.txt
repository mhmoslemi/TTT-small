from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        frac = diff / tot
        h = np.clip(h + room * frac, 0.0, 1.0)
    else:
        if s < 1e-12:
            return h
        h = np.clip(h * (target / s), 0.0, 1.0)
    # Final correction for clipping
    s2 = h.sum()
    diff2 = target - s2
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 0:
                h = np.clip(h + room * (diff2 / tot), 0.0, 1.0)
        else:
            s3 = h.sum()
            if s3 > 0:
                h = np.clip(h * (target / s3), 0.0, 1.0)
    return h

def smooth_max_val_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = corr.max()
    Tc = T * (corr - mc) - T.max()
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    p_abs = np.abs(np.arange(m_full) - (n - 1))
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-12:
            continue
        p = p_abs[idx]
        if p >= n:
            continue
        # d1: (1-h[i+p]) for i in [0, n-p)
        # d2: -h[i-p] for i in [p, n)
        d = np.zeros(n)
        d[:n-p] += 1.0 - h[p:]
        d[p:] += -h[:n-p]
        grad += wi * d
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol, t_end):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x):
        return smooth_max_val_grad(x, T, lam)[0]
    def grad(x):
        return smooth_max_val_grad(x, T, lam)[1]
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12,
                                'maxls': 50})
        return project_h(res.x)
    except Exception:
        return x0

    m_full = 2 * n - 1
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Random order for diversity
        order = np.random.permutation(n)
        for i in order:
            old = best_h[i]
            # Compute lin and quad for variable i
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lag = idx - (n - 1)
                p = abs(lag)
                if p >= n:
                    continue
                if p == 0:
                    lin[idx] = 1.0 - 2.0 * old
                    quad[idx] = -1.0
                else:
                    if i + p < n:
                        lin[idx] += 1.0 - best_h[i + p]
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            # 1D optimization: minimize max_m [corr[m] + d*lin[m] + d^2*quad[m]]
            # where d = new - old, new in [0,1], so d in [-old, 1-old]
            dlo, dhi = -old, 1.0 - old
            if dhi - dlo < 1e-14:
                continue

            d_grid = np.linspace(dlo, dhi, grid_size)
            # Vectorized: F[k,m] = corr[m] + d_grid[k]*lin[m] + d_grid[k]^2*quad[m]
            F = corr[None, :] + d_grid[:, None] * lin[None, :] + (d_grid * d_grid)[:, None] * quad[None, :]
            f_vals = F.max(axis=1)
            kmin = f_vals.argmin()
            d_best = d_grid[kmin]
            # Refine with golden section in a small window
            w = 2.0 * (dhi - dlo) / grid_size
            wlo = max(dlo, d_best - w)
            whi = min(dhi, d_best + w)
            if whi - wlo > 1e-14:
                def f1d(d):
                    return float(np.max(corr + d * lin + d * d * quad))
                res = minimize_scalar(f1d, bounds=(wlo, whi), method='bounded',
                                      options={'xatol': 1e-12})

            best_h[i] = new_val
            # Check improvement

            if tc5 < best_c5 - 1e-14:
                best_c5 = tc5
                improved = True
            else:
                best_h[i] = old  # revert
        if not improved:
            break
    return best_h, best_c5

    kmin = fvals.argmin()
    d_best = d_grid[kmin]
    w = 2.0 * (dhi - dlo) / grid_sz
    wlo = max(dlo, d_best - w)
    whi = min(dhi, d_best + w)
    if whi - wlo > 1e-12:
        def f1d(d):
            return float(np.max(corr + d * lin + d * d * quad))
        try:
            res = minimize_scalar(f1d, bounds=(wlo, whi), method='bounded', options={'xatol': 1e-13})
            d_best = res.x
        except Exception:
            pass
    return d_best

def coord_descent(h, t_end, max_sweeps=500, grid_sz=250):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            d_best = coord1d_solve(corr, i, old, best_h, n, grid_sz)
            if abs(d_best) < 1e-13:
                continue
            best_h[i] = old + d_best
            tc5 = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * (2.0 / n))
            if tc5 < best_c5 - 1e-14:
                best_c5 = tc5
                improved = True
            else:
                best_h[i] = old
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS warm (8%)
    p1 = t0 + budget_s * 0.08
    for T in [200, 1000, 5000]:
        if time.time() > p1:
            break
        hc = lbfgs_opt(best_h, T, 800.0, 1500, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: Coordinate descent (30%)
    p2 = t0 + budget_s * 0.38
    best_h, best_c5 = coord_descent(best_h, p2, max_sweeps=150, grid_sz=200)

    # Phase 3: L-BFGS refinement (12%)
    p3 = t0 + budget_s * 0.50
    for T in [10000, 50000, 200000]:
        if time.time() > p3:
            break
        hc = lbfgs_opt(best_h, T, 800.0, 2500, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 4: Thorough coordinate descent (25%)
    p4 = t0 + budget_s * 0.75
    best_h, best_c5 = coord_descent(best_h, p4, max_sweeps=300, grid_sz=350)

    # Phase 5: Basin hopping with small perturbations (12%)
    p5 = t0 + budget_s * 0.87
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(50):
        if time.time() > p5:
            break
        cool = 1.0 - 0.9 * (it / 50.0)
        sc = 0.003 + 0.025 * cool
        h_p = project_h(work_h + sc * np.random.randn(n))
        T_b = 1000 * cool + 500
        hc = lbfgs_opt(h_p, T_b, 800.0, 800, 1e-15)
        c5v = compute_c5(hc)
        temp = 0.0002 * cool + 0.00003
        if c5v < work_c5 or (np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
        # Quick coord polish after each hop
        if time.time() < p5 - budget_s * 0.02:
            hc2, c5v2 = coord_descent(work_h, min(p5, time.time() + budget_s * 0.04), max_sweeps=30, grid_sz=150)
            if c5v2 < work_c5:
                work_h, work_c5 = hc2, c5v2
                if c5v2 < best_c5:
                    best_c5, best_h = c5v2, hc2

    # Phase 6: Final polish (10%)
    p6 = t0 + budget_s * 0.96
    best_h, best_c5 = coord_descent(best_h, p6, max_sweeps=200, grid_sz=400)

    if time.time() < t0 + budget_s * 0.99:
        hc = lbfgs_opt(best_h, 100000, 1500.0, 2000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
