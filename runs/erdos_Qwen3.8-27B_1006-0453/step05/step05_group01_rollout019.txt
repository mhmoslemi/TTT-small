import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T, flat_pen, n_pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e8:
        val = m
    else:
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-n_pen:]
        val += flat_pen * np.std(top)
    return val * dx

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def slsqp(h, T, n, maxiter, ftol, flat_pen):
    try:
        res = minimize(smooth_c5, h, method='SLSQP',
                       args=(T, flat_pen, 3),
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def cd_obj(t, base, coefs):
    return np.max(base + t * coefs)

def coord_descent(h, n, t_end, max_sweeps, n_top):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(cd_obj, bounds=(0.0, 1.0),
                                  args=(base, coefs), method='bounded',
                                  options={'xatol': 1e-12})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break

        r = np.random.random()

            top_idx = np.argsort(corr)[-8:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:

                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n+k] - 0.5) * 0.5
            pert *= scale * cool
        elif r < 0.55:
            pert = np.zeros(n)
            for _ in range(3):
                f = np.random.randint(1, 15)
                p = np.random.uniform(0, 2 * np.pi)
                pert += np.sin(2 * np.pi * f * np.arange(n) / n + p)

        elif r < 0.75:

            work_c5 = c5v
    return work_h, work_c5
