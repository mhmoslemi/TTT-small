**New idea: Water-filling / equalization**

At the optimum, we want R(k) to be as flat as possible (all lags at the same value). This is similar to the "flatness" criterion in signal processing. We can use a "water-filling" approach:
- Start with current h
- Find the lag with minimum R(k)
- Try to increase it by adjusting h
- This naturally equalizes the autocorrelation

In practice, this means: the subgradient of "min_k R(k)" points in the direction of the lag with minimum R(k). So we should take a step that increases R at the minimum lags.

∂R(k)/∂h[i] involves the correlation structure. Specifically:
R(k) = Σ_j h[j]h[j+k] (for k≥0, with appropriate bounds)

∂R(k)/∂h[i] = h[i-k] + h[i+k] (for valid indices)

So the gradient of min_k R(k) is a weighted average of the gradients at the minimum lags.

from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            h *= target / s
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += room * (target - s) / tot
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=5000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.astype(float).copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 5 == 4 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                m = n - 1 + s
                if i + s < n:
                    lin[m] += 1.0 - best_h[i + s]
                if i - s >= 0:
                    lin[m] -= best_h[i - s]
                m2 = n - 1 - s
                if i + s < n:
                    lin[m2] += 1.0 - best_h[i + s]
                if i - s >= 0:
                    lin[m2] -= best_h[i - s]
            lin[n - 1] = 1.0 - 2.0 * old
            def obj(d):
                v = corr + d * lin
                v[n - 1] -= d * d
                return np.max(v)
            res = minimize_scalar(obj, bounds=(-old, 1.0 - old), method='bounded',
                                  options={'xatol': 1e-11})
            d_new = float(res.x)
            if abs(d_new) < 1e-13:
                continue
            tv = corr + d_new * lin
            tv[n - 1] -= d_new * d_new
            if np.max(tv) < np.max(corr) - 1e-15:
                best_h[i] = old + d_new
                corr = tv
                best_c5 = float(np.max(corr) * dx)
                improved = True
        if not improved and sweep > 2:
            break
    best_h = project_h(best_h, n)
    return best_h, compute_c5(best_h)

def maxmin_acorr_grad(h, n, t_end, max_iters=2000):
    best_h = h.astype(float).copy()
    best_c5 = compute_c5(best_h)
    h_c = best_h.copy()
    for it in range(max_iters):
        if time.time() > t_end:
            break
        acorr = np.correlate(h_c, h_c, mode='full')
        min_val = np.min(acorr)
        T = 5.0 + 5.0 * (it / max_iters)
        w = np.exp(-T * (acorr - min_val))
        w /= w.sum()
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] < 1e-10:
                continue
            s = m - (n - 1)
            if s == 0:
                grad += 2.0 * w[m] * h_c
            elif s > 0:
                grad[:n-s] += w[m] * h_c[s:]
                grad[s:] += w[m] * h_c[:n-s]
            else:
                a = -s
                grad[:n-a] += w[m] * h_c[a:]
                grad[a:] += w[m] * h_c[:n-a]
        gm = np.max(np.abs(grad))
        if gm < 1e-14:

        step = 0.3 / gm

            best_h = h_new.copy()
        h_c = h_new

def block_cd(h, n, t_end, block_size=4, max_iters=500, seed=42):

    best_h = h.astype(float).copy()
    best_c5 = compute_c5(best_h)

        i0 = rng.randint(0, n - block_size)

        def obj(x):
            h_test = best_h.copy()
            h_test[start:end] = x

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.astype(float).copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.02 + 0.05 * cool
        mode = hop % 6
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(2, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            # Block flip
            bsize = rng.randint(5, 20)
            bstart = rng.randint(0, n - bsize)
            pert = np.zeros(n)
            pert[bstart:bstart+bsize] = (rng.random(bsize) - 0.5) * 2 * step * 2
        elif mode == 3:
            # Swap two blocks
            bsize = rng.randint(3, 12)
            i1 = rng.randint(0, n - bsize)
            i2 = rng.randint(0, n - bsize)
            pert = np.zeros(n)
            pert[i1:i1+bsize] = best_h[i2:i2+bsize] - best_h[i1:i1+bsize]
            pert[i2:i2+bsize] = best_h[i1:i1+bsize] - best_h[i2:i2+bsize]
        elif mode == 4:
            # Sparse random
            n_sp = max(5, n // 8)
            idx = rng.choice(n, size=n_sp, replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 1.5 * rng.randn(n_sp)
        else:
            # Circular shift by 1-3
            shift = rng.randint(1, 4) * (1 if rng.rand() < 0.5 else -1)
            pert = np.roll(best_h, shift) - best_h
        h_pert = project_h(cur_h + pert, n)

        hop_end = min(t_end, time.time() + hop_time)

        temp = 0.003 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):

            if c5_opt < best_c5:

                best_h = h_opt.copy()
