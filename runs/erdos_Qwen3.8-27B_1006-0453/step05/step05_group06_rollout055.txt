    """Project h to [0,1]^n with sum = n/2."""
    n = len(h)
    h = h.copy()

            tot = room.sum()

        else:

            tot = h.sum()
            if tot < 1e-14:

            h = h * (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

    n = len(h)

            best_c5 = c5
