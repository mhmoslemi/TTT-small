import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def gold_min_1d(f, lo=0.0, hi=1.0, niter=35):
    """Minimize convex 1D function using golden section"""
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    c = hi - (hi - lo) / gr
    d = lo + (hi - lo) / gr
    fc, fd = f(c), f(d)
    for _ in range(niter):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - (hi - lo) / gr
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + (hi - lo) / gr
            fd = f(d)
    return (lo + hi) / 2.0

    h[i] = 0.0
    cb = corr_full(h)
    h[i] = h_save
    
    m_vals = np.arange(-(n-1), n)
    coefs = np.zeros(2*n-1)
    # h[i] in H part: m ranges such that i+m in [0,n-1]
    # corr[idx] where idx = n-1+m
    lo_m = 0
    hi_m = n - 1 - i  # i + m <= n-1

    # h[i] in (1-H) part: j+m = i, j in [0,n-1]

    # m ranges from i-(n-1) to i, clamped to [-(n-1), n-1]
    lo_m2 = max(-(n-1), i - (n-1))
    hi_m2 = min(n-1, i)
    for m in range(lo_m2, hi_m2 + 1):
        j = i - m
        if 0 <= j < n:
            coefs[n-1+m] -= h[j]
    
    def obj(x):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def gold_min_1d(f, lo=0.0, hi=1.0, niter=40):
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    c = hi - (hi - lo) / gr
    d = lo + (hi - lo) / gr
    fc, fd = f(c), f(d)
    for _ in range(niter):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - (hi - lo) / gr
            fc = f(c)
        else:
            lo, c, fc = c, d, fd
            d = lo + (hi - lo) / gr
            fd = f(d)
    return (lo + hi) / 2.0

def coord_opt_single(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    cb = corr_full(h)
    h[i] = h_save
    coefs = np.zeros(2 * n - 1)
    lo_m = 0
    hi_m = n - 1 - i
    for m in range(lo_m, hi_m + 1):
        coefs[n - 1 + m] += (1.0 - h[i + m])
    lo_m2 = max(-(n - 1), 0)
    hi_m2 = min(n - 1, i)
    for m in range(lo_m2, hi_m2 + 1):
        j = i - m
        if 0 <= j < n:
            coefs[n - 1 + m] -= h[j]
    def obj(x):
        return np.max(cb + x * coefs)
    return gold_min_1d(obj, 0.0, 1.0, 38)

def coord_descent(h, n, t_end, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            x_new = coord_opt_single(best_h, n, i)
            if abs(x_new - best_h[i]) < 1e-11:
                continue
            best_h[i] = x_new
            best_h = project_h(best_h)
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=400, ftol=1e-14):
    def obj(x):
        c = corr_full(x)
        return float(smooth_max(c, T) * (2.0 / n))
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_step(h, n, step_size=0.002):
    c = corr_full(h)
    k_best = np.argmax(c)
    m = k_best - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i + m < n:
            grad[i] += (1.0 - h[i + m])
        j = i - m
        if 0 <= j < n:
            grad[i] -= h[j]
    h_new = h - step_size * grad
    h_new = project_h(h_new)
    return h_new
