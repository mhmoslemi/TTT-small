from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i, niter=40):
    """Optimize single coordinate h[i] using ternary search on max of lines."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n-1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2*n-1)
    coefs[v2] += (1.0 - h[np.clip(vm2[v2], 0, n-1)])
    coefs[v1] += (-h[np.clip(vm1[v1], 0, n-1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi-lo)/3.0
        m2 = hi - (hi-lo)/3.0
        if np.max(corr_base + m1*coefs) < np.max(corr_base + m2*coefs):
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2.0

def active_grad(h, n, tol=0.01):
    """Compute averaged gradient over active set."""
    corr = get_corr(h)
    cmax = np.max(corr)
    active = corr > cmax - tol
    m_vals = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        m = m_vals[k]
        # d c_k / d h[i] = (1-h[i+m]) - h[i-m] for valid
        im = np.arange(n) + m
        im_valid = (im >= 0) & (im < n)
        grad[im_valid] += (1.0 - h[np.clip(im[im_valid], 0, n-1)])
        im_neg = np.arange(n) - m
        im_neg_valid = (im_neg >= 0) & (im_neg < n)
        grad[im_neg_valid] -= h[np.clip(im_neg[im_neg_valid], 0, n-1)]
    grad /= active.sum()
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        h = h + room * (diff / tot) if tot > 1e-14 else h
    elif s > 1e-14:
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i, niter=40):
    h[i] = 0.0
    corr_base = get_corr(h)
    h[i] += 1.0  # restore for coefficient computation
    m_vals = np.arange(-(n-1), n)
    vm2 = i + m_vals
    vm1 = i - m_vals
    v2 = (vm2 >= 0) & (vm2 < n)
    v1 = (vm1 >= 0) & (vm1 < n)
    coefs = np.zeros(2*n-1)
    coefs[v2] += (1.0 - h[np.clip(vm2[v2], 0, n-1)])
    coefs[v1] += (-h[np.clip(vm1[v1], 0, n-1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi-lo)/3.0
        m2 = hi - (hi-lo)/3.0
        if np.max(corr_base + m1*coefs) < np.max(corr_base + m2*coefs):
            hi = m2
        else:
            lo = m1
    h[i] += 0.0  # no-op to keep h unchanged
    return (lo+hi)/2.0

def coord_descent(h, n, t_end, max_sweeps=50, niter_cd=40, stop_tol=1e-14):

                best_h = h_try
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return best_h, best_c5

    corr = get_corr(h)
    cmax = np.max(corr)
    active = corr > cmax - tol_frac * cmax
    m_vals = np.arange(-(n-1), n)
    grad = np.zeros(n)
    idxs = np.where(active)[0]
    for k in idxs:
        m = m_vals[k]
        im = np.arange(n) + m
        vm = (im >= 0) & (im < n)
        grad[vm] += (1.0 - h[np.clip(im[vm], 0, n-1)])
        imn = np.arange(n) - m
        vn = (imn >= 0) & (imn < n)
        grad[vn] -= h[np.clip(imn[vn], 0, n-1)]
    if len(idxs) > 0:
        grad /= len(idxs)
    return project_h(h - alpha * grad)
