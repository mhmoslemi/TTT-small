import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_step(h, T, n, maxiter=1000, ftol=1e-15):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def binary_local_search(h_binary, n, t_end, max_iters=50000):
    """Local search on binary pattern: flip one bit at a time"""
    best = h_binary.copy()
    best_c5 = compute_c5(best)
    improved = True
    while improved and time.time() < t_end:
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            # Try flipping bit i

            h_try = project_h(h_try)

            if c5_try < best_c5 - 1e-14:
                best = h_try
                best_c5 = c5_try
                improved = True
    return best, best_c5

def swap_local_search(h, n, t_end, max_iters=50000):
    """Try swapping values at two positions"""
    best = h.copy()
    best_c5 = compute_c5(best)
    improved = True
    while improved and time.time() < t_end:
        improved = False
        for _ in range(max_iters):
            if time.time() > t_end:
                break
            i, j = np.random.choice(n, 2, replace=False)
            if abs(best[i] - best[j]) < 0.3:  # Only swap if significantly different
                continue
            h_try = best.copy()
            h_try[i], h_try[j] = best[j], best[i]
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best = h_try
                best_c5 = c5_try
                improved = True
    return best, best_c5

def coord_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    dx = 2.0 / n
    
    def eval_1d(hi):
        return float(np.max(corr_base + hi * coefs) * dx)
    
    try:
        res = minimize_scalar(eval_1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)

            if time.time() > t_end:

                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_single(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    dx = 2.0 / n
    
    def obj(hi):
        return float(np.max(corr_base + hi * coefs) * dx)
    
    try:
        res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()

    for sweep in range(max_sweeps):

                return best_h, best_c5

                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5
