        h[i] = t_opt
    return h

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        h_try = coord_sweep(best_h, n)
        h_try = project_h(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try <= best_c5 + 1e-14:
            best_c5 = c5_try
            best_h = h_try
        else:
            best_h = project_h(best_h, n)
            best_c5 = compute_c5(best_h)
            break
    return best_h, best_c5

def make_perturbation(n, cool, rng):
    mod = rng.randint(0, 5)
    if mod == 0:
        bs = rng.randint(4, max(5, n // 5))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.03 + 0.12 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 12))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.03 + 0.08 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(3, max(4, n // 8))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.08 + 0.25 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.02 + 0.05 * cool) * rng.standard_normal(n)
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.08 * cool * rng.standard_normal(half)
        p[half:] = -0.08 * cool * rng.standard_normal(n - half)
    return p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation (8% time)
    for T in [100, 1000, 10000, 100000]:
        if elapsed() > budget_s * 0.07:
            break
        hc = slsqp_step(best_h, T, n, maxiter=500, ftol=1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent to convergence (5% time)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.12, max_sweeps=200)

    # Phase 3: Basin hopping (70% time)
    work_h = best_h.copy()
    work_c5 = best_c5
    bh_end = t0 + budget_s * 0.82
    it_count = 0
    no_improve = 0
    while time.time() < bh_end:
        it_count += 1
        cool = max(0.0, 1.0 - 0.3 * it_count / 200.0)
        pert = make_perturbation(n, cool, rng)
        h_pert = project_h(work_h + pert, n)
        T_bh = 500 + 8000 * cool
        hc = slsqp_step(h_pert, T_bh, n, maxiter=400, ftol=1e-14)
        hc = coord_sweep(hc, n)
        hc = project_h(hc, n)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0002
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # Every 8 iterations, refine best with coord descent
        if it_count % 8 == 0:
            best_h, best_c5 = coord_descent(best_h, n, min(time.time() + 3, bh_end), max_sweeps=20)
            work_h = best_h.copy()
            work_c5 = best_c5
        # If stuck, bigger perturbation
        if no_improve > 30:
            big_pert = 0.3 * cool * rng.standard_normal(n)
            h_big = project_h(best_h + big_pert, n)
            work_h = h_big
            work_c5 = compute_c5(h_big)
            no_improve = 0

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Final coordinate descent (10% time)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=300)

    # Phase 5: High-T SLSQP polish (5% time)
    for T in [500000, 2000000]:
        if elapsed() > budget_s * 0.96:
            break
        hc = slsqp_step(best_h, T, n, maxiter=1500, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final coord descent
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=50)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
