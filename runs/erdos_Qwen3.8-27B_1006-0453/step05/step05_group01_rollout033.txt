import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_val(h, T):
    corr = corr_full(h)
    m = np.max(corr)
    if T > 1e7:
        return m
    return (m + np.log(np.sum(np.exp(T * (corr - m))))) / T

    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    idx_list = np.where(w > 1e-5)[0]
    i_idx = np.arange(n)
    for ki in idx_list:
        ks = ki - (n - 1)
        wk = w[ki]
        if ks >= 0:
            vp = i_idx + ks < n
            grad[vp] += wk * (1.0 - h[i_idx[vp] + ks])
            vm = i_idx >= ks
            grad[vm] -= wk * h[i_idx[vm] - ks]
        else:
            kk = -ks
            vp = i_idx + kk < n
            grad[vp] += wk * (1.0 - h[i_idx[vp] + kk])
            vm = i_idx >= kk
            grad[vm] -= wk * h[i_idx[vm] - kk]
    return grad * dx

            x = x_new

        if c5v < best_f - 1e-12:
            best_f = c5v
            best_x = x.copy()
    return best_x, best_f

        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_val(h, T):
    corr = corr_full(h)
    m = np.max(corr)
    if T > 1e7:
        return m
    return (m + np.log(np.sum(np.exp(T * (corr - m))))) / T

def grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    w = np.exp(T * (corr - m))
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    idx_list = np.where(w > 1e-5)[0]
    i_idx = np.arange(n)
    for ki in idx_list:
        ks = ki - (n - 1)
        wk = w[ki]
        if ks >= 0:
            vp = i_idx + ks < n
            grad[vp] += wk * (1.0 - h[i_idx[vp] + ks])
            vm = i_idx >= ks
            grad[vm] -= wk * h[i_idx[vm] - ks]
        else:
            kk = -ks
            vp = i_idx + kk < n
            grad[vp] += wk * (1.0 - h[i_idx[vp] + kk])
            vm = i_idx >= kk
            grad[vm] -= wk * h[i_idx[vm] - kk]
    return grad * dx

def momentum_opt(h, n, n_outer, t_end):
    x = h.copy()
    v = np.zeros(n)
    best_x = x.copy()
    best_f = compute_c5(x)
    for outer in range(n_outer):
        if time.time() > t_end:
            break
        frac = outer / max(n_outer - 1, 1)
        T = 300 + 4000 * frac
        lr = 0.005 * (1.0 - 0.6 * frac)
        n_inner = 25
        for inner in range(n_inner):
            if time.time() > t_end:
                return best_x, best_f
            g = grad_smooth(x, T)
            v = 0.82 * v - lr * g
            x = project_h(x + v, n)
        c5v = compute_c5(x)
        if c5v < best_f - 1e-12:
            best_f = c5v
            best_x = x.copy()
    return best_x, best_f

def coord_descent(h, n, t_end, n_top=50, max_sweeps=400):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = corr_full(best_h)
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            b_k = np.zeros(n_top)
            for t in range(n_top):
                ks = top_shifts[t]
                b = 0.0
                if ks >= 0:
                    if i <= n - 1 - ks:
                        b += (1.0 - best_h[i + ks])
                    if i >= ks:
                        b -= best_h[i - ks]
                else:
                    kk = -ks
                    if i >= kk:
                        b += (1.0 - best_h[i - kk])
                    if i <= n - 1 - kk:
                        b -= best_h[i + kk]
                b_k[t] = b
            a_k = corr[top_idx] - best_h[i] * b_k
            res = minimize_scalar(lambda t: np.max(a_k + t * b_k),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            t_new = res.x
            if abs(t_new - best_h[i]) > 1e-11:
                best_h[i] = t_new
                improved = True
        if improved:
            best_h = project_h(best_h, n)
            c5_new = compute_c5(best_h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                stall = 0
            else:
                stall += 1
            if stall > 8:
                break
        else:
            break
    return best_h, best_c5

def basin_refine(h, n, t_end, n_tries=15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for attempt in range(n_tries):
        if time.time() > t_end:
            break
        cool = 1.0 - attempt / max(n_tries, 1)
        scale = 0.015 + 0.06 * cool
        r = np.random.random()
        if r < 0.4:
            pert = scale * cool * np.random.randn(n)
        elif r < 0.7:
            f = np.random.randint(2, 15)
            p = np.random.uniform(0, 2.0 * np.pi)
            pert = scale * 0.6 * cool * np.sin(2.0 * np.pi * f * np.arange(n) / n + p)
        else:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.2 * cool * np.random.randn(max(int(mask.sum()), 1))
        h_pert = project_h(best_h + pert, n)
        t_sub = min(time.time() + 8.0, t_end - 2.0)
        h_try, c5_try = coord_descent(h_pert, n, t_sub, n_top=35, max_sweeps=40)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def optimize_phase(h, n, t_end):
    best_h = project_h(h, n)
    best_c5 = compute_c5(best_h)
    remaining = t_end - time.time()
    if remaining > 80:
        t_sub = time.time() + remaining * 0.25
        mh, mc = momentum_opt(best_h, n, 25, t_sub)
        if mc < best_c5:
            best_c5 = mc
            best_h = mh
    remaining = t_end - time.time()
    if remaining > 40:
        t_sub = time.time() + remaining * 0.45
        ch, cc = coord_descent(best_h, n, t_sub, n_top=50, max_sweeps=500)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch
    remaining = t_end - time.time()
    if remaining > 50:
        t_sub = time.time() + remaining * 0.55
        bh, bc = basin_refine(best_h, n, t_sub, n_tries=12)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh
    remaining = t_end - time.time()
    if remaining > 20:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 3, n_top=60, max_sweeps=300)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 200
        init_h = 0.5 + 0.35 * np.sin(np.pi * np.arange(n_init) / n_init)

    n = len(init_h)
    best_h = project_h(init_h, n)
    best_c5 = compute_c5(best_h)

    t_phase1 = t0 + budget_s * 0.52
    best_h, best_c5 = optimize_phase(best_h, n, t_phase1)

    if t_end - time.time() > budget_s * 0.32:
        n2 = n + 100
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_h(h2, n2)
        t_phase2 = t0 + budget_s * 0.82
        h2_opt, c5_2 = optimize_phase(h2, n2, t_phase2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt
            n = n2

    if t_end - time.time() > budget_s * 0.18 and n < 600:
        n3 = min(n + 100, 600)
        if n3 > n:
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n3, endpoint=False)
            h3 = np.interp(x_new, x_old, best_h)
            h3 = project_h(h3, n3)
            t_phase3 = t_end - 12
            h3_opt, c5_3 = optimize_phase(h3, n3, t_phase3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3_opt
                n = n3

    if t_end - time.time() > 12:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 2, n_top=60, max_sweeps=150)

    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
