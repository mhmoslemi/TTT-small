        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = np.clip(h * (tgt / max(s, 1e-13)), 0, 1)
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        m = (h > 0.05) & (h < 0.95)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_sweep(h, n, n_samples=500):
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    center = n - 1
    mf = 2 * n - 1
    for i in range(n):
        old = h[i]
        h[i] = 0.0
        c0 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = old
        delta = c1 - c0
        ts = np.linspace(0.0, 1.0, n_samples)
        vals = c0[None, :] + ts[:, None] * delta[None, :]
        vals[:, center] += ts * (1.0 - ts)
        c5_arr = vals.max(axis=1) * dx
        idx = np.argmin(c5_arr)
        new_val = ts[idx]
        if 0 < idx < n_samples - 1:
            tl, tr = float(ts[idx-1]), float(ts[idx+1])
            ts2 = np.linspace(tl, tr, 300)
            vals2 = c0[None, :] + ts2[:, None] * delta[None, :]
            vals2[:, center] += ts2 * (1.0 - ts2)
            c5_arr2 = vals2.max(axis=1) * dx
            new_val = float(ts2[np.argmin(c5_arr2)])
        if abs(new_val - old) > 1e-13:
            h[i] = new_val
            h = proj_h(h, n)
    return h, compute_c5(h)

def _smooth_obj(x, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def lbfgs_step(h, T, n, iters=200):
    _T0 = [T]
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return (m + np.log(np.sum(np.exp(_T0[0] * (c - m)))) / _T0[0]) * 2.0 / n
    try:
        r = minimize(h, _obj, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-8})
        return proj_h(r.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS warm-start with moderate smoothing
    p1_end = t0 + budget_s * 0.06
    for T in [100, 500, 2000]:
        if time.time() > p1_end:
            break
        h_new = lbfgs_step(best_h, T, n, iters=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Intensive coordinate descent
    p2_end = t0 + budget_s * 0.38
    sweep_count = 0
    while time.time() < p2_end:
        best_h, new_c5 = coord_sweep(best_h, n)
        if new_c5 < best_c5:
            best_c5 = new_c5
        sweep_count += 1

    # Phase 3: Basin hopping with coordinate refinement
    p3_end = t0 + budget_s * 0.78
    for it in range(400):
        if time.time() > p3_end:
            break
        cool = max(0.0, 1.0 - it / 400.0)
        mode = it % 4
        hp = best_h.copy()
        if mode == 0:
            mask = np.random.random(n) < (0.3 + 0.4 * cool)
            hp[mask] += (0.03 + 0.12 * cool) * np.random.randn(int(mask.sum()))
        elif mode == 1:
            f = np.random.randint(2, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.10 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
            hp += pert
        elif mode == 2:
            for _ in range(5):
                a, b = np.random.choice(n, 2, replace=False)
                hp[a], hp[b] = hp[b], hp[a]
        else:
            m1 = np.random.choice(n, n // 4, replace=False)
            hp[m1] += (0.05 + 0.15 * cool) * np.random.randn(len(m1))
        hp = proj_h(hp, n)
        hp, c5_hp = coord_sweep(hp, n)
        temp = 0.00008 * cool + 5e-6
        if c5_hp < best_c5:
            best_h, best_c5 = hp, c5_hp
        elif np.random.rand() < np.exp(-(c5_hp - best_c5) / temp):
            best_h = hp

    # Phase 4: Final polishing - many tight coordinate sweeps
    p4_end = t0 + budget_s * 0.96
    while time.time() < p4_end:
        best_h, new_c5 = coord_sweep(best_h, n, n_samples=700)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            break

    # Final tiny L-BFGS polish
    if time.time() < t0 + budget_s * 0.99:
        h_fin = lbfgs_step(best_h, 20000, n, iters=100)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
