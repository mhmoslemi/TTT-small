    corr = np.correlate(h, 1.0 - h, mode='full')
    m_full = 2 * n - 1
    if top_k is not None and top_k < m_full:
        idx = np.argpartition(corr, m_full - top_k)[m_full - top_k:]
        ct = corr[idx]
        Tc = T * (ct - ct.max())
        Tc -= Tc.max()
        exps = np.exp(Tc)
        ws = exps / exps.sum()
        val = dx * (ct.max() + np.log(exps.sum()) / T) + lam * (h.sum() - n / 2.0) ** 2
        w = np.zeros(m_full)
        w[idx] = ws
    else:
        mc = corr.max()
        Tc = T * (corr - mc)
        Tc -= Tc.max()
        exps = np.exp(Tc)
        w = exps / exps.sum()
        val = dx * (mc + np.log(exps.sum()) / T) + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    lag_arr = np.arange(m_full) - n + 1
    for idx2 in range(m_full):
        wi = w[idx2]
        if wi < 1e-15:
            continue
        lg = lag_arr[idx2]
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n - lg])
            if lg > 0:
                grad[:n - lg] += -wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n - p] += wi * (1.0 - h[p:])
            if p > 0:
                grad[p:] += -wi * h[:n - p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol, top_k=None):
    x0 = project_h(h0)
    def obj_fn(x):
        return smooth_obj_grad(x, T, lam, top_k)[0]
    def grad_fn(x):
        return smooth_obj_grad(x, T, lam, top_k)[1]
    try:
        res = minimize(obj_fn, x0, jac=grad_fn, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * len(x0),
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-11})
        return project_h(res.x)
    except Exception:
        return x0

def coord_descent(h, t_end, max_sweeps):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old_val = best_h[i]
            h_at0 = best_h.copy()
            h_at0[i] = 0.0
            cb = np.correlate(h_at0, 1.0 - h_at0, mode='full')
            lin = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            f0 = np.max(cb)
            f1 = np.max(cb + lin)
            if f0 <= f1 + 1e-15:
                res_x = 0.0
            elif f1 < f0 - 1e-15:
                res_x = 1.0
            else:
                def f1d(x):
                    return np.max(cb + x * lin)
                res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-12})
                res_x = res.x
            if abs(res_x - old_val) < 1e-14:
                continue
            ht = best_h.copy()
            ht[i] = res_x
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def local_search_01(h, t_end, max_iters):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    s = int(h_bin.sum())
    if s > target:
        idx_ones = np.where(h_bin == 1.0)[0]
        np.random.shuffle(idx_ones)
        for j in range(s - target):
            h_bin[idx_ones[j]] = 0.0
    elif s < target:
        idx_zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[idx_zeros])
        for j in range(target - s):
            h_bin[idx_zeros[order[j]]] = 1.0
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    T_sa = 0.005
    for it in range(max_iters):
        if time.time() > t_end:
            break
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        improved = False
        T_sa *= 0.995
        for _ in range(80):
            oi = ones[np.random.randint(len(ones))]
            zi = zeros[np.random.randint(len(zeros))]
            ht = best_h.copy()
            ht[oi] = 0.0
            ht[zi] = 1.0
            c5t = compute_c5(ht)
            delta = c5t - best_c5
            if delta < -1e-14:
                best_h = ht
                best_c5 = c5t
                improved = True
            elif T_sa > 1e-8 and np.random.rand() < np.exp(-delta / T_sa):
                best_h = ht
                best_c5 = c5t
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B warm-up (12% budget)
    p1_end = t0 + budget_s * 0.12
    for T in [20, 80, 300, 1000]:
        if time.time() > p1_end:
            break
        hc = lbfgs_opt(best_h, T, 200.0, 1500, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Top-k focused optimization (18% budget)
    p2_end = t0 + budget_s * 0.30
    for T in [800, 2000, 5000]:
        if time.time() > p2_end:
            break
        for tk in [8, 15, 30, None]:
            if time.time() > p2_end:
                break
            hc = lbfgs_opt(best_h, T, 200.0, 1200, 1e-16, top_k=tk)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 3: 0-1 local search diversification (15% budget)
    p3_end = t0 + budget_s * 0.45
    h01, c5_01 = local_search_01(best_h, p3_end, 300)
    if c5_01 < best_c5:
        best_c5 = c5_01
        best_h = h01.copy()
    h_rand = project_h(0.5 + 0.35 * np.random.randn(n))
    if time.time() < p3_end:
        h01b, c5_01b = local_search_01(h_rand, p3_end, 150)
        if c5_01b < best_c5:
            best_c5 = c5_01b
            best_h = h01b.copy()
    # L-BFGS-B from 0-1 solution to refine
    if time.time() < p3_end:
        hc = lbfgs_opt(best_h, 1000, 200.0, 1000, 1e-16, top_k=15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Coordinate descent (20% budget)
    p4_end = t0 + budget_s * 0.65
    best_h, best_c5 = coord_descent(best_h, p4_end, 300)

    # Phase 5: High-T L-BFGS-B refinement (15% budget)
    p5_end = t0 + budget_s * 0.80
    for T in [5000, 15000, 50000, 150000]:
        if time.time() > p5_end:
            break
        hc = lbfgs_opt(best_h, T, 400.0, 2500, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        hc2 = lbfgs_opt(best_h, T, 400.0, 800, 1e-17, top_k=10)
        c5v2 = compute_c5(hc2)
        if c5v2 < best_c5:
            best_c5 = c5v2
            best_h = hc2

    # Phase 6: Basin hopping (10% budget)
    p6_end = t0 + budget_s * 0.90
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(40):
        if time.time() > p6_end:
            break
        cool = 1.0 - 0.9 * (it / 40.0)
        sc = 0.005 + 0.04 * cool
        h_p = project_h(work_h + sc * np.random.randn(n))
        T_bh = int(2000 * cool) + 300
        hc = lbfgs_opt(h_p, T_bh, 300.0, 800, 1e-16, top_k=15)
        c5v = compute_c5(hc)
        temp = 0.0003 * cool + 0.00005
        if c5v < work_c5 or (temp > 1e-8 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 7: Final coordinate descent + L-BFGS-B polish (8% budget)
    p7_end = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, p7_end, 150)

    if time.time() < t0 + budget_s * 0.99:
        hc = lbfgs_opt(best_h, 200000, 400.0, 1500, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        hc2 = lbfgs_opt(best_h, 50000, 400.0, 500, 1e-17, top_k=8)
        c5v2 = compute_c5(hc2)
        if c5v2 < best_c5:
            best_c5 = c5v2
            best_h = hc2

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
