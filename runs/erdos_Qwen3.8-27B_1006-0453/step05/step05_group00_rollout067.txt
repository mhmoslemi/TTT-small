```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_val(h):
    N = len(h)
    return float(compute_corr(h).max()) * 2.0 / N

def project_bbox_sum(h, target):
    """Project onto {x: 0<=x_i<=1, sum(x)=target}."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        if np.clip(h-mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def grad_at_lag(h, N, s):
    """Gradient of c_s w.r.t. h."""
    g = np.zeros(N)
    # c_s = sum_i h[i](1-h[i+s])
    # d/dh[i]: (1-h[i+s]) if 0<=i+s<N, minus h[i-s] if 0<=i-s<N
    valid1 = np.arange(N) + s
    m1 = (valid1 >= 0) & (valid1 < N)
    g[m1] += (1.0 - h[valid1[m1]])
    valid2 = np.arange(N) - s
    m2 = (valid2 >= 0) & (valid2 < N)
    g[m2] -= h[valid2[m2]]
    return g

def subgradient_opt(h, N, deadline, lr0=0.1, decay=0.999):
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_bbox_sum(h, target)
    corr = compute_corr(h)
    best_c = float(corr.max()) * 2.0/N
    best_h = h.copy()
    lr = lr0
    beta = 0.7
    m = np.zeros(N)
    
    while time.time() < deadline:
        corr = compute_corr(h)
        max_val = float(corr.max())
        c5 = max_val * 2.0/N
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        
        # Find top lags (within 1 of max)
        threshold = max_val - 2.0  # within 2 units
        top_idx = np.where(corr >= threshold)[0]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(corr)])
        
        # Average subgradient
        g = np.zeros(N)
        for idx in top_idx:
            s = int(idx) - (N - 1)
            g += grad_at_lag(h, N, s)
        g /= len(top_idx)
        
        # Nesterov step
        y = h + m
        g_y = np.zeros(N)
        corr_y = compute_corr(y)
        max_y = float(corr_y.max())
        top_y = np.where(corr_y >= max_y - 2.0)[0]
        for idx in top_y:
            s = int(idx) - (N - 1)
            g_y += grad_at_lag(y, N, s)
        g_y /= len(top_y)
        
        h_new = project_bbox_sum(h - lr * g_y, target)
        m = beta * (h_new - h)
        h = h_new
        lr *= decay
    
    return best_h, best_c
```

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_val(h):
    N = len(h)
    return float(compute_corr(h).max()) * 2.0 / N

def project_bbox_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gen_binary(N, rng):
    """Random binary vector with N/2 ones."""
    h = np.zeros(N)
    idx = rng.choice(N, N//2, replace=False)
    h[idx] = 1.0
    return h

def anneal_binary(h, N, deadline, T0=1.0, T_min=0.01, rng=None):
    """Simulated annealing with single swap moves."""
    if rng is None:
        rng = np.random
    h = h.copy()
    corr = compute_corr(h)
    cur_max = float(corr.max())
    best_max = cur_max
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = T0
    n_iter = 0
    patience = 0
    
    while time.time() < deadline:
        # Pick random swap
        p = int(rng.choice(ones))
        q = int(rng.choice(zeros))
        h[p] = 0.0
        h[q] = 1.0
        new_corr = compute_corr(h)
        new_max = float(new_corr.max())
        
        if new_max <= cur_max:
            # Accept
            cur_max = new_max
            corr = new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            # Annealing acceptance
            delta = new_max - cur_max
            if rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_max = new_max
                corr = new_corr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                # Reject
                h[p] = 1.0
                h[q] = 0.0
        
        n_iter += 1
        T = max(T * 0.9997, T_min)
        # Occasional reheat if stuck
        if patience > 5000:
            T = min(T * 3, T0)
            patience = 0
    
    return best_h, best_max

def subgrad_opt(h, N, deadline, lr0=5.0):
    """Subgradient optimization for continuous h."""
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_bbox_sum(h, target)
    corr = compute_corr(h)
    best_c = float(compute_corr(h).max()) * 2.0 / N
    best_h = h.copy()
    lr = lr0
    m = np.zeros(N)
    beta = 0.5
    
    it = 0
    while time.time() < deadline:
        if it % 100 == 0:
            corr = compute_corr(h)
            c5 = float(corr.max()) * 2.0 / N
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
        
        # Nesterov extrapolation
        y = h + beta * m
        
        # Find active lags
        corr_y = compute_corr(y)
        max_y = float(corr_y.max())
        top = np.where(corr_y >= max_y - 3.0)[0]
        
        # Subgradient
        g = np.zeros(N)
        for idx in top:
            s = int(idx) - (N - 1)
            i_idx = np.arange(N)
            # (1 - y[i+s]) for valid i+s
            vs = i_idx + s
            m1 = (vs >= 0) & (vs < N)
            g[m1] += (1.0 - y[vs[m1]])
            # -y[i-s] for valid i-s
            vs2 = i_idx - s
            m2 = (vs2 >= 0) & (vs2 < N)
            g[m2] -= y[vs2[m2]]
        g /= max(len(top), 1)
        
        # Update
        h_new = project_bbox_sum(h - lr * g, target)
        m = h_new - h
        h = h_new
        it += 1
        if it % 500 == 0:
            lr *= 0.95
            beta = min(0.8, 0.3 + it * 0.0001)
    
    # Final eval
    final_c = float(compute_corr(h).max()) * 2.0 / N
    if final_c < best_c:
        best_c = final_c
        best_h = h.copy()
    
    return best_h, best_c
