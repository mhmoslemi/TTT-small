```python
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    sum_exp = np.sum(exp_s)
    log_sum = max_s + np.log(sum_exp)
    val = (m + log_sum / T) * dx
    weights = exp_s / sum_exp
    
    ks = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for p in range(n):
        mask1 = (p + ks >= 0) & (p + ks < n)
        if mask1.any():

        mask2 = (p - ks >= 0) & (p - ks < n)
        if mask2.any():
            grad[p] -= np.dot(weights[mask2], h[p - ks[mask2]])
    grad *= dx
    return val, grad

        gp = 200.0 * (x.sum() - n / 2.0)
        return g + gp
    try:

                       bounds=[(0.0, 1.0)] * n,
```

```python
def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-14: return h
        h = h + r * (d / t)
    else:
        if s < 1e-14: return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def _svgrad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    sh = T * (c - m)
    ms = np.max(sh)
    es = np.exp(sh - ms)
    se = es.sum()
    val = (m + (ms + np.log(se)) / T) * dx
    w = es / se
    ks = np.arange(-(n-1), n)
    g = np.empty(n)
    for p in range(n):
        m1 = (p + ks >= 0) & (p + ks < n)
        m2 = (p - ks >= 0) & (p - ks < n)
        gp = 0.0
        if m1.any():
            gp += np.dot(w[m1], 1.0 - h[p + ks[m1]])
        if m2.any():
            gp -= np.dot(w[m2], h[p - ks[m2]])
        g[p] = gp * dx
    return val, g

def _objp(x, T, n, mu):
    v, _ = _svgrad(x, T)
    return v + mu * (x.sum() - n * 0.5) ** 2

def _gradp(x, T, n, mu):
    _, g = _svgrad(x, T)
    return g + 2.0 * mu * (x.sum() - n * 0.5)

    try:

                     method='SLSQP', bounds=[(0,1)]*n,
                     options={'maxiter': mi, 'ftol': 1e-16})
        return project_h(np.clip(r.x, 0, 1))
    except Exception:
        return h

                     method='L-BFGS-B', bounds=[(0,1)]*n,
                     options={'maxiter': mi, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_h(np.clip(r.x, 0, 1))
    except Exception:
        return h

def _coord1(h, n, i):
    ks = np.arange(-(n-1), n)
    hb = h.copy(); hb[i] = 0.0
    cb = np.correlate(hb, 1.0 - hb, mode='full')
    k1 = i + ks; k2 = i - ks
    v1 = (k1 >= 0) & (k1 < n); v2 = (k2 >= 0) & (k2 < n)
    cf = np.zeros(2 * n - 1)
    cf[v1] += 1.0 - h[np.clip(k1[v1], 0, n-1)]
    cf[v2] += -h[np.clip(k2[v2], 0, n-1)]
    lo, hi = 0.0, 1.0

        b = hi - (hi - lo) / 3.0
        if np.max(cb + a * cf) < np.max(cb + b * cf):
            hi = b
        else:
            lo = a
    return 0.5 * (lo + hi)

            if ct < bc - 1e-15:
                bc = ct; bh = ht; imp = True
        if not imp: break
    return bh, bc
```

```python
def _svgrad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    sh = T * (c - m)
    ms = np.max(sh)
    es = np.exp(sh - ms)
    se = es.sum()
    val = (m + (ms + np.log(se)) / T) * dx
    w = es / se
    ks = np.arange(-(n-1), n)
    g = np.empty(n)
    for p in range(n):
        m1 = (p - ks >= 0) & (p - ks < n)  # p-k_j valid
        m2 = (p + ks >= 0) & (p + ks < n)  # p+k_j valid
        gp = 1.0
        if m1.any():
            gp -= np.dot(w[m1], h[p - ks[m1]])
        if m2.any():
            gp -= np.dot(w[m2], h[p + ks[m2]])
        g[p] = gp * dx
    return val, g
```
