Wait: effect[i] is the change when h[i] goes from 1 to 0. So:
- Removing p (1->0): c changes by effect[p]
- Adding q (0->1): c changes by -effect[q] (opposite)

We want max(c_new) < P (current max of c).

This is: for all k, c[k] + effect[p,k] - effect[q,k] < P
i.e., effect[p,k] - effect[q,k] < P - c[k] for all k

The most constraining k is where P - c[k] is smallest (i.e., c[k] is near P).

Actually, for the binary search, let me use a simpler but effective approach:

1. Compute corr = fft_corr(h)
2. Find the top 30 critical lags
3. For each one position p, compute the "savings" at those 30 lags if we flip p to 0
4. For each zero position q, compute the "addition" at those 30 lags if we flip q to 1
5. Find the pair (p,q) where the max of (corr[top] + savings_p + addition_q) is minimized
6. If this max < current P, make the move

The savings for flipping p from 1 to 0 at lag k:
- We lose the term h[p]*(1-h[p+k]) at lag k: save (1-h[p+k]) if 0≤p+k<N
- We lose the term h[p-k]*(1-h[p]) at lag k: but (1-h[p]) goes from 0 to 1, so we GAIN h[p-k]
- Net: -(1-h[p+k])*I(0≤p+k<N) + h[p-k]*I(0≤p-k<N)

The addition for flipping q from 0 to 1 at lag k:
- We gain h[q]*(1-h[q+k]) at lag k: add (1-h[q+k]) if 0≤q+k<N
- We lose h[q-k]*(1-h[q]) at lag k: (1-h[q]) goes from 1 to 0, so we LOSE h[q-k]
- Net: (1-h[q+k])*I(0≤q+k<N) - h[q-k]*I(0≤q-k<N)

So: c_new[k] = c[k] + savings_p[k] + addition_q[k]
Where savings_p[k] = -(1-h[p+k])*I(0≤p+k<N) + h[p-k]*I(0≤p-k<N)
And addition_q[k] = (1-h[q+k])*I(0≤q+k<N) - h[q-k]*I(0≤q+k<N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2*N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0/N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, T):
    N = len(h)
    corr = fft_corr(h)
    cm = float(corr.max())
    w = np.exp(T * (corr - cm))
    w = w / float(w.sum())
    idx = np.arange(N)[:, None]
    lags = np.arange(2*N-1)[None, :] - (N-1)
    m1 = idx + lags
    v1 = (m1 >= 0) & (m1 < N)
    m1c = np.clip(m1, 0, N-1)
    t1 = (w[None, :] * (1.0 - h[m1c]) * v1).sum(axis=1)
    m2 = idx - lags
    v2 = (m2 >= 0) & (m2 < N)
    m2c = np.clip(m2, 0, N-1)
    t2 = (w[None, :] * h[m2c] * v2).sum(axis=1)
    return t1 - t2, float(corr.max())

def nesterov_smooth(h, N, T_start, T_end, deadline, lr0=0.03, n_iter=2000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta = 0.9
    lr = lr0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max(n_iter-1, 1))
        y = np.clip(h + m, 0, 1)
        grad, pk = smooth_grad(y, T)
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 200 == 0 and it > 0:
            lr *= 0.97
            beta = min(0.98, 0.9 + it * 0.0001)
    return best_h, best_pk * (2.0/N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, T):
    N = len(h)
    corr = fft_corr(h)
    cm = float(corr.max())
    w = np.exp(T * (corr - cm))
    w = w / float(w.sum())
    idx = np.arange(N)[:, None]
    lags = np.arange(2 * N - 1)[None, :] - (N - 1)
    m1 = idx + lags
    v1 = (m1 >= 0) & (m1 < N)
    m1c = np.clip(m1, 0, N - 1)
    t1 = (w[None, :] * (1.0 - h[m1c]) * v1).sum(axis=1)
    m2 = idx - lags
    v2 = (m2 >= 0) & (m2 < N)
    m2c = np.clip(m2, 0, N - 1)
    t2 = (w[None, :] * h[m2c] * v2).sum(axis=1)
    return t1 - t2, float(corr.max())

def nesterov_smooth(h, N, T_start, T_end, deadline, lr0=0.025, n_iter=2000):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    beta = 0.9
    lr = lr0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        y = np.clip(h + m, 0, 1)
        grad, pk = smooth_grad(y, T)
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        if it % 150 == 0 and it > 0:
            lr *= 0.96
            beta = min(0.98, 0.88 + it * 0.0002)
    return best_h, best_pk * (2.0 / N)

def binary_search_fast(h, N, deadline, max_sweeps=5000, n_top=40):
    h = (h > 0.5).astype(float)
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        order = np.argsort(np.random.random(len(idx)))
        h[idx[order[nh:]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        order = np.argsort(np.random.random(len(idx)))
        h[idx[order[:nh - n1]]] = 1
    best_c = float(np.max(fft_corr(h)))
    best_h = h.copy()
    top_n = min(n_top, 2 * N - 1)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max())
        if P < best_c - 1e-15:
            best_c = P
            best_h = h.copy()
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        top_idx = np.argpartition(corr, -top_n)[-top_n:]
        top_lags = (top_idx - (N - 1)).astype(int)
        top_vals = corr[top_idx].astype(float)
        p_arr = ones[:, None].astype(int)
        k_arr = top_lags[None, :].astype(int)
        m1 = p_arr + k_arr
        v1 = (m1 >= 0) & (m1 < N)
        m1c = np.clip(m1, 0, N - 1)
        m2 = p_arr - k_arr
        v2 = (m2 >= 0) & (m2 < N)
        m2c = np.clip(m2, 0, N - 1)
        savings = (-(1.0 - h[m1c].astype(float)) * v1 + h[m2c].astype(float) * v2)
        q_arr = zeros[:, None].astype(int)
        m3 = q_arr + k_arr
        v3 = (m3 >= 0) & (m3 < N)
        m3c = np.clip(m3, 0, N - 1)
        m4 = q_arr - k_arr
        v4 = (m4 >= 0) & (m4 < N)
        m4c = np.clip(m4, 0, N - 1)
        additions = ((1.0 - h[m3c].astype(float)) * v3 - h[m4c].astype(float) * v4)
        c_new = top_vals[None, None, :] + savings[:, None, :] + additions[None, :, :]
        new_max = c_new.max(axis=2)
        min_val = float(new_max.min())
        if min_val < P - 1e-12:
            bi, bj = np.unravel_index(np.argmin(new_max), new_max.shape)
            h[ones[bi]], h[zeros[bj]] = 0.0, 1.0
        elif sweep % 100 == 99:
            bi = np.random.randint(len(ones))
            bj = np.random.randint(len(zeros))
            h[ones[bi]], h[zeros[bj]] = 0.0, 1.0
            if sweep % 500 == 499:
                bi2 = np.random.randint(len(ones))
                bj2 = np.random.randint(len(zeros))
                if ones[bi2] != ones[bi]:
                    h[ones[bi2]], h[zeros[bj2]] = 0.0, 1.0
    return h, best_c * (2.0 / N), best_h
