```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if diff > 1e-14:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    elif diff < -1e-14:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
```

```python
    dx = 2.0 / n
```

```python
def _eval_1d_sym(corr, a, n, old, d):
    """Evaluate the max correlation after symmetric update by d."""
    # corr is the current correlation array
    # a is the coefficient vector (for position p, single coordinate)
    # The update is: corr_new[k] = corr[k] + 2*d*a[k] for k != n-1
    #                corr_new[n-1] = corr[n-1] + 2*d*(1-2*old) - 2*d*d
    v = corr + 2.0 * d * a
    v[n - 1] = corr[n - 1] + 2.0 * d * (1.0 - 2.0 * old) - 2.0 * d * d
    return float(np.max(v))
```
