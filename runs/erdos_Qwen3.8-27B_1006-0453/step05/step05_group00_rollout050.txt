import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def ensure_binary(h, N):
    h = h.astype(float).copy()
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 != nh:
        h = (h > 0.5).astype(float)
        n1 = int(h.sum())
        if n1 > nh:
            ones = np.where(h == 1)[0]
            keep = np.random.choice(ones, nh, replace=False)
            h[:] = 0.0
            h[keep] = 1.0
        else:
            zeros = np.where(h == 0)[0]
            add = np.random.choice(zeros, nh - n1, replace=False)
            h[add] = 1.0
    return h

def vec_binary_search(h, N, deadline, max_sweeps=2000, sample_n=200):
    h = ensure_binary(h, N)
    best_c = float(np.max(fft_corr(h)))
    no_imp = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 60:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                for _ in range(4):
                    p = ones[np.random.randint(len(ones))]
                    q = zeros[np.random.randint(len(zeros))]
                    h[p], h[q] = h[q], h[p]
                no_imp = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        # Focus sampling on peak contributors
        peak_idx = int(np.argmax(c))
        peak_shift = peak_idx - (N - 1)
        if abs(peak_shift) > 0:
            contrib = np.zeros(N)
            for i in range(N):
                if h[i] == 1:
                    j1 = i - peak_shift
                    j2 = i + peak_shift
                    if 0 <= j1 < N and h[j1] == 0:
                        contrib[i] += 1
                    if 0 <= j2 < N and h[j2] == 0:
                        contrib[i] += 1
            cands = np.where((contrib > 0) & (h == 1))[0]
            if len(cands) > 0 and len(cands) < len(ones):
                p_list = cands if len(cands) <= sample_n else np.random.choice(cands, sample_n, replace=False)
            else:
                p_list = ones if len(ones) <= sample_n else np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones if len(ones) <= sample_n else np.random.choice(ones, sample_n, replace=False)
        # Vectorized swap evaluation
        Dp = D[p_list]  # (n_p, 2N-1)
        Dz = D[zeros]  # (n_z, 2N-1)
        base = c[None, :] - Dp  # (n_p, 2N-1)
        cand = base[:, None, :] - Dz[None, :, :]  # (n_p, n_z, 2N-1)
        npk = cand.max(axis=2)  # (n_p, n_z)
        flat_min = float(npk.min())
        if flat_min < P - 1e-12:
            pi, qi = np.unravel_index(int(npk.argmin()), npk.shape)
            h[p_list[pi]] = 0.0
            h[zeros[qi]] = 1.0
        else:
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p], h[q] = h[q], h[p]
    return h, best_c

def nesterov_opt(h, N, deadline, k=100, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-10:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        top = top[corr[top] > corr.max() * 0.7]
        if len(top) == 0:
            top = np.array([int(np.argmax(corr))])
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= max(len(top), 1)
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.0001)
    return best_h, best_pk * dx
