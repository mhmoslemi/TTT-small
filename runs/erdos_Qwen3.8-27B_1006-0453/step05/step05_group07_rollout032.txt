    if abs(s-tgt) < 1e-13:

    if s > tgt:

    elif len(ones) < k:

    # h[j]: 1→0
    d_lo = max(-(n-1), -j)
    d_hi = min(n-1, n-1-j)
    for d in range(d_lo, d_hi+1):
        if d == 0: continue
        corr[d+n-1] -= (1.0 - h[j+d])
    d_lo2 = max(-(n-1), j-(n-1))
    d_hi2 = min(n-1, j)
    for d in range(d_lo2, d_hi2+1):
        if d == 0: continue
        corr[d+n-1] += h[j-d]
    h[j] = 0

def swap_sa(h0, n, t_end, max_iters=2000000, T0=0.08, T1=0.002):
    dx = 2.0/n
    h = h0.copy()
    corr = corr_from_h(h, n)
    cur_c5 = float(np.max(corr)*dx)
    best_c5 = cur_c5
    best_h = h.copy()
    ncorr = 2*n-1
    ones = np.where(h==1)[0]
    zeros = np.where(h==0)[0]
    T = T0
    cool_rate = (T1/T0)**(1.0/max(1,max_iters))
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        apply_swap(corr, h, i, j, n)
        nc5 = float(np.max(corr)*dx)
        
        if nc5 < cur_c5 or np.random.rand() < np.exp(-(nc5-cur_c5)/T):
            cur_c5 = nc5
            # Update ones and zeros
            # i was 1, now 0; j was 0, now 1
            # Just rebuild occasionally for simplicity
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            unswap(corr, h, i, j, n)
        
        T *= cool_rate
        if T < T1:
            T = T1
    
    return best_h, best_c5

def coord_desc(h, n, t_end, sweeps=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2*n - 1
    dx = 2.0/n
    
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr_cache = np.correlate(best, 1.0-best, mode='full')
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            # first factor contributions
            for d in range(max(0,-i), min(n-i, n)):
                if d == 0: continue
                coef[d+n-1] += (1.0-best[i+d])
            # second factor: these are negative in the objective (they subtract)
            for d in range(max(i-(n-1),-(n-1)), min(i+1, n)):
                if d == 0: continue
                coef[d+n-1] -= best[i-d]
            
            base = corr_cache - old * coef
            def f1d(v):
                return float(np.max(base + v*coef))
            r = minimize_scalar(f1d, bounds=(0.0,1.0), method='bounded',
                                options={'xatol':1e-11})
            nv = float(r.x)
            if abs(nv-old) < 1e-10:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = float(np.max(np.correlate(trial, 1-trial, mode='full'))*dx)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                corr_cache = np.correlate(best, 1.0-best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def thue_morse_bin(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = (i & 1) + (t[i>>1] if i>>1 < n else 0)
        t[i] &= 1
    # Ensure exactly n/2 ones
    # Thue-Morse of length n/2 (if n is power of 2) has exactly n/4 ones
    # For general n, we need to adjust
    t = t.astype(float)
    ones = np.where(t==1)[0]
    k = n//2
    if len(ones) > k:
        np.random.shuffle(ones)
        t[ones[k:]] = 0
    elif len(ones) < k:
        zeros = np.where(t==0)[0]
        np.random.shuffle(zeros)
        t[zeros[:k-len(ones)]] = 1
    return t

def mod_construction(n, a, b):
    """h[i] = 1 if (i*a + b) mod n < n/2, else 0."""
    h = np.zeros(n)
    idx = np.arange(n)
    h[((idx*a + b) % n) < n//2] = 1.0
    # Adjust to have exactly n/2 ones
    if h.sum() != n//2:
        h_b = binarize(h, n)
        return h_b
    return h
