```python
from scipy.optimize import minimize_scalar, minimize
import time

_glob_base = [None]
_glob_coef = [None]

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        mask = h > 0
        if mask.sum() == 0:
            h[:] = 0.5
            return h
        excess = s - tgt
        # Reduce proportionally among positive entries
        h[mask] -= excess * h[mask] / h[mask].sum()
        h = np.clip(h, 0, 1)
    else:
        deficit = tgt - s
        mask = h < 1
        if mask.sum() == 0:
            h[:] = 0.5
            return h
        h[mask] += deficit * (1 - h[mask]) / (1 - h[mask]).sum()
        h = np.clip(h, 0, 1)
    # Fix residual
    for _ in range(3):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1
        else:
            mask = h > 0
        if mask.sum() == 0:
            break
        step = d / mask.sum()
        h[mask] += step
        h = np.clip(h, 0, 1)
    return h

    c = _glob_coef[0]
    return float(np.max(b + v * c))

        if time.time() > t_end:
            break
        imp = False

            # Actually for d=0: term is h[i]*(1-h[i]), coeff is (1-h[i]) + (-h[i]) but it's h[i]-h[i]^2
            # For coordinate descent, we handle this by evaluating directly
            idx1 = i + d[valid1]
            idx2 = i - d[valid2]
            coef[valid1] += (1.0 - best[idx1])
            coef[valid2] -= best[idx2]

def softmax_obj(h, T):
    c = compute_corr(h)
    m = c.max()
    n = len(h)
    val = (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n
    return val

def softmax_grad(h, T):
    c = compute_corr(h)
    n = len(h)
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    mf = 2*n - 1
    for k in range(mf):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n-1)
        # coef of h[i] in c[k] for all i
        # c[k] = sum_j h[j]*(1-h[j+d])
        # dCoef[j] = (1-h[j+d])_valid - h[j-d]_valid
        valid1 = (np.arange(n) + d >= 0) & (np.arange(n) + d < n)
        valid2 = (np.arange(n) - d >= 0) & (np.arange(n) - d < n)
        dc = np.zeros(n)
        dc[valid1] += (1.0 - h[np.arange(n)[valid1] + d])
        dc[valid2] -= h[np.arange(n)[valid2] - d]
        g += wk * dc
    return g * 2.0 / n

                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except:
        return h

    idx = rng.choice(n, n//2, replace=False)

    h = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2
    positions = np.floor(np.arange(n//2) * phi).astype(int) % n

        if p not in seen:
            seen.add(p)
            unique_pos.append(p)

    return h
```

- c_new[k] = c[k] + δ*(coef[i,k] - coef[j,k])
- Minimize max_k c_new[k] over δ

This is a 1D optimization! And it naturally preserves the sum constraint.

Let me implement this:

```python
def two_coord_descent(h, n, t_end, max_sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2*n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = compute_corr(best)
        # Random pairs
        for _ in range(n):  # n random pairs per sweep
            if time.time() > t_end:
                return best, best_c5
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            # coef diff: c_new[k] = c[k] + delta*(coef_i[k] - coef_j[k])
            # where coef_i[k] is the coefficient of h[i] in c[k]
            # (ignoring the quadratic term at k=n-1 for simplicity)
            di = np.arange(mf) - (n-1)
            v1i = (i + di >= 0) & (i + di < n)
            v2i = (i - di >= 0) & (i - di < n)
            ci = np.zeros(mf)
            ci[v1i] += (1.0 - best[i + di[v1i]])
            ci[v2i] -= best[i - di[v2i]]
            
            v1j = (j + di >= 0) & (j + di < n)
            v2j = (j - di >= 0) & (j - di < n)
            cj = np.zeros(mf)
            cj[v1j] += (1.0 - best[j + di[v1j]])
            cj[v2j] -= best[j - di[v2j]]
            
            diff = ci - cj
            # Minimize max(corr + delta*diff) over delta such that h[i]+delta in [0,1] and h[j]-delta in [0,1]
            delta_max = min(1.0 - best[i], best[j])
            delta_min = max(-(1.0 - best[j]), -best[i])
            
            _glob_base[0] = corr
            _glob_coef[0] = diff
            r = minimize_scalar(_cd_obj, bounds=(delta_min, delta_max), method='bounded',
                               options={'xatol': 1e-10, 'maxiter': 50})
            d_opt = float(r.x)
            if abs(d_opt) < 1e-12:
                continue
            trial = best.copy()
            trial[i] += d_opt
            trial[j] -= d_opt
            trial = np.clip(trial, 0, 1)
            # Check sum is still OK (it should be by construction)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                corr = compute_corr(best)
                imp = True
        if not imp:
            break
    return best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

_glob_base = [np.zeros(1)]
_glob_coef = [np.zeros(1)]

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        mask = h > 0
        if mask.sum() == 0:
            h[:] = 0.5
            return h
        h[mask] -= (s - tgt) * h[mask] / h[mask].sum()
    else:
        mask = h < 1
        if mask.sum() == 0:
            h[:] = 0.5
            return h
        h[mask] += (tgt - s) * (1 - h[mask]) / (1 - h[mask]).sum()
    h = np.clip(h, 0, 1)
    for _ in range(5):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h < 0.999) if d > 0 else (h > 0.001)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0, 1)
    return h

def _cd_obj(v):
    return float(np.max(_glob_base[0] + v * _glob_coef[0]))

def _coeff(h, i, n):
    mf = 2 * n - 1
    d = np.arange(mf) - (n - 1)
    ci = np.zeros(mf)
    v1 = (i + d >= 0) & (i + d < n)
    v2 = (i - d >= 0) & (i - d < n)
    ci[v1] += (1.0 - h[i + d[v1]])
    ci[v2] -= h[i - d[v2]]
    return ci

def coord_descent(h, n, t_end, max_sweeps=200, two_coord_frac=0.5):
    rng = np.random
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = compute_corr(best)
        # Single coordinate passes
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            ci = _coeff(best, i, n)
            _glob_base[0] = corr - old * ci
            _glob_coef[0] = ci
            r = minimize_scalar(_cd_obj, bounds=(0.0, 1.0), method='bounded',
                               options={'xatol': 1e-11, 'maxiter': 40})
            nv = float(r.x)
            if abs(nv - old) < 1e-13:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            nc = compute_corr(trial)
            c5t = float(np.max(nc) * 2.0 / n)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                corr = nc
                imp = True
        # Two-coordinate steps
        for _ in range(int(n * two_coord_frac)):
            if time.time() > t_end:
                return best, best_c5
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            ci = _coeff(best, i, n)
            cj = _coeff(best, j, n)
            diff = ci - cj
            d_lo = max(-(1.0 - best[j]), -best[i])
            d_hi = min(1.0 - best[i], best[j])
            if d_hi - d_lo < 1e-13:
                continue
            _glob_base[0] = corr
            _glob_coef[0] = diff
            r = minimize_scalar(_cd_obj, bounds=(d_lo, d_hi), method='bounded',
                               options={'xatol': 1e-11, 'maxiter': 40})
            d_opt = float(r.x)
            if abs(d_opt) < 1e-12:
                continue
            trial = best.copy()
            trial[i] = best[i] + d_opt
            trial[j] = best[j] - d_opt
            nc = compute_corr(trial)
            c5t = float(np.max(nc) * 2.0 / n)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                corr = nc
                imp = True
        if not imp:
            break
    return best, best_c5

def _softmax_obj(h, T):
    c = compute_corr(h)
    m = c.max()
    n = len(h)
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n

def _softmax_grad(h, T):
    c = compute_corr(h)
    n = len(h)
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        idx = np.arange(n)
        v1 = (idx + d >= 0) & (idx + d < n)
        v2 = (idx - d >= 0) & (idx - d < n)
        dc = np.zeros(n)
        dc[v1] += (1.0 - h[idx[v1] + d])
        dc[v2] -= h[idx[v2] - d]
        g += wk * dc
    return g * 2.0 / n

def lbfgs_polish(h, T, n, iters=300):
    try:
        r = minimize(h, lambda x: _softmax_obj(x, T), lambda x: _softmax_grad(x, T),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-11})
        return proj_h(r.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.12, max_sweeps=300)

    # Phase 2: L-BFGS with moderate T
    if time.time() < t0 + budget_s * 0.22:
        h_lb = lbfgs_polish(best_h, 2000.0, n, iters=300)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_h, best_c5 = h_lb, c5_lb
            best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.25, max_sweeps=100)

    # Phase 3: Basin hopping with targeted perturbations
    t_basin_end = t0 + budget_s * 0.55
    work = best_h.copy()
    work_c5 = best_c5
    n_iters = 0
    while time.time() < t_basin_end and n_iters < 150:
        n_iters += 1
        cool = max(0.0, 1.0 - n_iters / 150.0)
        mod = n_iters % 4
        if mod == 0:
            mask = np.random.random(n) < (0.15 + 0.25 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.03 + 0.12 * cool) * np.random.randn(mask.sum())
        elif mod == 1:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.02 + 0.06 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            # Target worst shifts: flip a few positions
            corr = compute_corr(work)
            worst_k = np.argsort(corr)[-5:]
            pert = np.zeros(n)
            for _ in range(3):
                i = np.random.randint(0, n)
                pert[i] = (0.05 + 0.1 * cool) * np.random.choice([-1, 1])
        else:
            # Binary swap perturbation
            h_bin = (work > 0.5).astype(float)
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            pert = np.zeros(n)
            if len(ones) > 0 and len(zeros) > 0:
                a = np.random.choice(ones)
                b = np.random.choice(zeros)
                pert[a] = -0.5
                pert[b] = 0.5
        hp = proj_h(work + pert, n)
        T_val = 500.0 + 5000.0 * cool
        ht = lbfgs_polish(hp, T_val, n, iters=150)
        ht, c5t = coord_descent(ht, n, min(time.time() + 2, t_basin_end), max_sweeps=20)
        temp = 0.0005 * cool + 0.00002
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_h = ht.copy()
                best_c5 = c5t

    # Phase 4: Intensive coordinate descent on best
    if time.time() < t0 + budget_s * 0.75:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.78, max_sweeps=400)

    # Phase 5: L-BFGS with high T for polishing
    if time.time() < t0 + budget_s * 0.88:
        h_lb = lbfgs_polish(best_h, 50000.0, n, iters=500)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_h, best_c5 = h_lb, c5_lb

    # Phase 6: Another round of intensive coordinate descent
    if time.time() < t0 + budget_s * 0.95:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=300)

    # Phase 7: Try higher resolution
    if time.time() < t0 + budget_s * 0.92:
        n_hi = min(n + 40, 240)
        try:
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_hi, endpoint=False) + 1.0 / n_hi
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = proj_h(h_hi, n_hi)
            h_hi, c5_hi = coord_descent(h_hi, n_hi, t0 + budget_s * 0.96, max_sweeps=200)
            if c5_hi < best_c5:
                best_h, best_c5, n = h_hi, c5_hi, n_hi
        except Exception:
            pass

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
