import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    diff2 = target - s2
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = np.clip(h + room * (diff2 / tot), 0, 1)
        else:
            h = np.clip(h * (1.0 + diff2 / max(h.sum(), 1e-14)), 0, 1)
    return h

def coord_minimal(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    corr_base = get_corr(h0)
    shifts = np.arange(2 * n - 1) - (n - 1)
    ia = i + shifts
    ib = i - shifts
    va = (ia >= 0) & (ia < n)
    vb = (ib >= 0) & (ib < n)
    ca = np.where(va, 1.0 - h[np.clip(ia, 0, n - 1)], 0.0)
    cb = np.where(vb, h[np.clip(ib, 0, n - 1)], 0.0)
    coef = ca - cb
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = corr_base + mid * coef
        m = vals.max()
        active = vals >= m - 1e-12
        mn = coef[active].min()
        mx = coef[active].max()
        if mn <= 0.0 and mx >= 0.0:
            return mid
        elif mx < 0.0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def coord_sweep(h, n, order):
    for i in order:
        t_new = coord_minimal(h, n, i)
        h[i] = t_new
    return project_h(h)

def smooth_max_val(h, n, T):
    dx = 2.0 / n
    corr = get_corr(h) * dx
    m = corr.max()
    ev = np.exp(T * (corr - m))
    return m + np.log(ev.sum()) / T

def smooth_max_grad(h, n, T):
    dx = 2.0 / n
    corr = get_corr(h) * dx
    m = corr.max()
    ev = np.exp(T * (corr - m))
    W = ev.sum()
    w = ev / W
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        ia = i + shifts
        ib = i - shifts
        va = (ia >= 0) & (ia < n)
        vb = (ib >= 0) & (ib < n)
        ca = np.where(va, 1.0 - h[np.clip(ia, 0, n - 1)], 0.0)
        cb = np.where(vb, h[np.clip(ib, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ca - cb)
    return grad

def slsqp_smooth(h, n, T, maxiter=800):
    def obj(x):
        return smooth_max_val(x, n, T)
    def objg(x):
        return obj(x), smooth_max_grad(x, n, T)
    def con(x):
        return x.sum() - n / 2.0
    def con_g(x):
        return np.ones(n)
    try:
        res = minimize(objg, h, method='SLSQP', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_g}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent (30 sweeps)
    for sw in range(30):
        if tl() < 80:
            break
        order = np.random.permutation(n)
        best_h = coord_sweep(best_h, n, order)
        c5v = compute_c5(best_h)
        if c5v < best_c5:
            best_c5 = c5v

    # Phase 2: SLSQP smooth max annealing
    for T in [20, 50, 100, 300, 800, 2000]:
        if tl() < 60:
            break
        hc = slsqp_smooth(best_h, n, T, maxiter=600)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: More coordinate descent
    for sw in range(50):
        if tl() < 60:
            break
        order = np.random.permutation(n)
        best_h = coord_sweep(best_h, n, order)
        c5v = compute_c5(best_h)
        if c5v < best_c5:
            best_c5 = c5v

    # Phase 4: Basin hopping with diverse perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 60
    for it in range(n_bh):
        if tl() < 40:
            break
        cool = 1.0 - 0.85 * (it / float(n_bh))
        mod = it % 6
        if mod == 0:
            pert = (0.03 + 0.09 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 12)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.02 + 0.07 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.15 * cool) * np.random.randn(mask.sum())
        elif mod == 3:
            blk = max(2, n // 10)
            b = np.random.randint(0, n // blk)
            pert = np.zeros(n)
            pert[b*blk:(b+1)*blk] = (0.03 + 0.06 * cool) * np.random.randn(blk)
        elif mod == 4:
            pert = np.zeros(n)
            for _ in range(3):
                a, b2 = np.random.permutation(n)[:2]
                d = (best_h[b2] - best_h[a]) * (0.3 + 0.7 * cool)
                pert[a] += d

        else:
            mask = np.random.random(n) < 0.08
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.12 * cool) * np.sign(np.random.randn(mask.sum()))
        h_pert = project_h(work_h + pert)
        n_cd = 8
        for _ in range(n_cd):
            h_pert = coord_sweep(h_pert, n, np.random.permutation(n))
        c5v = compute_c5(h_pert)
        temp = 0.0006 * cool + 0.00004
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = h_pert
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_pert.copy()

    if work_c5 < best_c5:
        best_h = work_h.copy()
        best_c5 = work_c5

    # Phase 5: SLSQP high-T polish
    for T in [3000, 10000, 30000]:
        if tl() < 30:
            break

            best_c5 = c5v

    for sw in range(80):

        best_h = coord_sweep(best_h, n, order)

            best_c5 = c5v

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
