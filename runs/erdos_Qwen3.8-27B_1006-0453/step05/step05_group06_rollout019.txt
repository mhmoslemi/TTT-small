def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T, lam):
    """Smoothed max of correlations + penalty for sum constraint."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    se = exps.sum()
    val = dx * (mc + np.log(se) / T) + lam * (h.sum() - n/2.0)**2
    return val

def smooth_max_grad(h, T, lam):
    """Gradient of smoothed max objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n/2.0)
    return grad

def lbfgs(h0, T, lam, maxiter, ftol, t_end):
    n = len(h0)
    x0 = project_h(h0).copy()
    def f(x):
        return smooth_max_obj(x, T, lam)

                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_h(res.x)
    except Exception:
        return x0

def structured_perturbation(h, std):
    """Perturb h to reduce peak correlations."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    peak_idx = np.argmax(corr)
    peak_lag = peak_idx - n + 1
    
    # Create perturbation that reduces correlation at peak_lag
    noise = np.zeros(n)
    if peak_lag > 0:
        for i in range(n - peak_lag):
            if h[i] > h[i + peak_lag]:
                noise[i] += 0.5  # decrease h[i]
                noise[i + peak_lag] -= 0.5  # increase h[i+peak_lag]
            else:
                noise[i] -= 0.5
                noise[i + peak_lag] += 0.5
    elif peak_lag < 0:
        p = -peak_lag
        for i in range(p, n):
            if h[i] > h[i - p]:
                noise[i] += 0.5
                noise[i - p] -= 0.5
            else:
                noise[i] -= 0.5
                noise[i - p] += 0.5
    
    return project_h(h + std * noise + std * np.random.randn(n))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smax_val(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc) - np.max(T * (corr - mc))
    se = np.exp(Tc).sum()
    return dx * (mc + np.log(se) / T) + lam * (h.sum() - n/2.0)**2

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    """Update correlation after changing h[i] from old_val to new_val."""
    corr = corr.copy()
    m = 2*n - 1
    for k in range(-n+1, n):
        idx = k + n - 1
        # Contribution of h[i] to corr[k]:
        # corr[k] = sum_j h[j]*(1-h[j+k])
        # h[i] appears as h[j] when j=i: h[i]*(1-h[i+k])
        # h[i] appears as h[j+k] when j=i-k: h[i-k]*(1-h[i])
        delta = 0
        if k == 0:
            # h[i]*(1-h[i]) → new_val*(1-new_val)
            delta = new_val*(1-new_val) - old_val*(1-old_val)
        else:
            if 0 <= i + k < n:
                delta += (new_val - old_val) * (1 - h[i+k])
            if 0 <= i - k < n:
                delta += h[i-k] * (old_val - new_val)
        corr[idx] += delta
    return corr

One more optimization: in the coordinate descent, I can skip the full rebalance. Instead of minimize_scalar, I can do a simpler search:

Since f(x) = max_k (cb[k] + x*lin[k] + x²*quad[k]) and quad is 0 or -1:
- The function is piecewise: at most 2n-1 pieces (one for each k)
- On each piece, it's either quadratic (concave) or linear
- The minimum of the max is where two (or more) pieces intersect at their minimum

I can approximate this by:
1. Evaluate f at x=0, x=0.5, x=1
2. Find the best of the three
3. Do 20 steps of golden section around the best

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h

    else:

        h = h * (1.0 + diff / s)
