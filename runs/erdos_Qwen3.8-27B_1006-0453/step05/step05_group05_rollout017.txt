import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def frank_wolfe(h, n, n_iter, top_k=5, t_schedule='linear'):
    """Frank-Wolfe for minimizing max_k corr_k * dx"""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iter):
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Find top-k lags (we want to minimize the max)
        top_idx = np.argsort(corr)[-top_k:]
        # Compute "gradient" = weighted sum of gradients for top lags
        # corr[k] = sum_i h[i]*(1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]
        # d(corr[k])/d(h[i]) = 1 - h[i+k] + h[i-k] (for valid indices, with boundary care)
        # Actually: corr[k] = sum_n h[n+k]*(1-h[n])
        # d/d(h[i]): from h[n+k]=i (n=i-k): (1-h[i-k])
        #            from 1-h[n]=1-h[i] (n=i): -h[i+k]
        # So d(corr[k])/d(h[i]) = (1-h[i-k])*1[i-k valid] - h[i+k]*1[i+k valid]
        
        # For top lags, compute combined gradient
        grad = np.zeros(n)
        for k in top_idx:
            k_shift = k - (n - 1)  # actual lag
            for i in range(n):
                # contribution from h[n+k]=i term: (1-h[n]) where n = i-k_shift
                n1 = i - k_shift
                if 0 <= n1 < n:
                    grad[i] += (1.0 - best_h[n1])
                # contribution from -h[i] in (1-h[n]) where n=i: -h[i+k_shift]
                n2 = i + k_shift
                if 0 <= n2 < n:
                    grad[i] -= best_h[n2]
        
        # FW step: minimize grad^T v subject to 0<=v<=1, sum(v)=n/2
        # Solution: v[i]=1 for top n/2 entries of (-grad), 0 otherwise
        # (minimize grad^T v = maximize -grad^T v)
        neg_grad = -grad
        threshold = np.partition(neg_grad, n//2 - 1)[n//2 - 1]
        v = (neg_grad >= threshold).astype(float)
        # Handle ties at threshold
        if v.sum() != n/2:
            # Adjust
            excess = v.sum() - n/2
            if excess > 0:
                at_thresh = np.where((neg_grad >= threshold) & (neg_grad < threshold + 1e-10))[0]
                if len(at_thresh) >= int(excess):
                    v[at_thresh[:int(excess)]] = 0
            else:
                at_thresh = np.where((neg_grad <= threshold) & (neg_grad > threshold - 1e-10))[0]
                if len(at_thresh) >= int(-excess):
                    v[at_thresh[:int(-excess)]] = 1
        
        # Step size
        if t_schedule == 'linear':
            t = 2.0 / (it + 2)
        else:
            t = 0.5  # fixed
        
        h_new = (1 - t) * best_h + t * v
        h_new = project_h(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fw_single_step(h, top_k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for m in top_idx:
        lag = m - (n - 1)
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
        else:
            a = -lag
            grad[a:] += (1.0 - h[:n-a])
            grad[:n-a] -= h[a:]
    # FW step: minimize grad^T v s.t. 0<=v<=1, sum(v)=n/2
    # v[i]=1 for n/2 smallest grad entries
    order = np.argsort(grad)
    v = np.zeros(n)
    v[order[:n//2]] = 1.0
    return v, corr

    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()

        if t_mode == 'lin':
            t = 2.0 / (it + 2)

        else:

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        cur_h = h_new
    return best_h, best_c5

def ternary_coord(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    m_vals = np.arange(-(n-1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2*n-1)
    coefs[valid1] += (1.0 - h[np.clip(vm1[valid1], 0, n-1)])
    coefs[valid2] += (-h[np.clip(vm2[valid2], 0, n-1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr_base + m1 * coefs)
        f2 = np.max(corr_base + m2 * coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:

            hi_new = ternary_coord(best_h, n, i)

            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

        v = fw_step(cur_h, top_k)
        if t_mode == 'lin':
            t = 2.0 / (it + 2.0)
        elif t_mode == 'half':
            t = 0.5
        else:
            t = 1.0
        h_new = (1.0 - t) * cur_h + t * v
        h_new = project_h(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        cur_h = h_new
    return best_h, best_c5

def ternary_coord(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    m_vals = np.arange(-(n-1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2*n-1)
    coefs[valid1] += (1.0 - h[np.clip(vm1[valid1], 0, n-1)])
    coefs[valid2] += (-h[np.clip(vm2[valid2], 0, n-1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = ternary_coord(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def generate_perturbation(n, cool, mode, rng):
    if mode == 0:
        return (0.02 + 0.08*cool) * rng.standard_normal(n)
    elif mode == 1:
        f = rng.randint(1, 12)
        ph = rng.uniform(0, 6.2832)
        return (0.02 + 0.06*cool) * np.sin(2*np.pi*f*np.arange(n)/n + ph)
    elif mode == 2:
        mask = rng.random(n) < 0.12
        p = np.zeros(n)
        p[mask] = (0.05 + 0.1*cool) * rng.standard_normal(mask.sum())
        return p
    elif mode == 3:
        blk = max(1, n // 10)
        b = rng.randint(0, n // blk)
        p = np.zeros(n)
        p[b*blk:(b+1)*blk] = (0.03 + 0.06*cool) * rng.standard_normal(blk)
        return p
    elif mode == 4:
        idx = rng.permutation(n)[:max(2, n//15)]
        p = np.zeros(n)
        for j in range(len(idx)-1):
            d = 0.03 * cool * (h_val[idx[j]] - h_val[idx[j+1]])
            p[idx[j]] += d
            p[idx[j+1]] -= d
        p[idx[0]] = 0
        return p
    else:
        mask = rng.random(n) < 0.08
        p = np.zeros(n)
        p[mask] = (0.04 + 0.12*cool) * np.sign(rng.standard_normal(mask.sum()))
        return p

h_val = None

def basin_hop(h, n, t0, budget_s, frac_start, frac_end, seed):
    rng = np.random.RandomState(seed)
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    best_h = h.copy()
    best_c5 = work_c5
    no_imp = 0
    max_no_imp = 25
    it = 0
    while time.time() < t0 + budget_s * frac_end:
        if time.time() > t0 + budget_s * frac_start + 120:
            break
        it += 1
        cool = max(0.05, 1.0 - 0.95 * it / 400.0)
        mode = it % 5
        global h_val
        h_val = work_h
        pert = generate_perturbation(n, cool, mode, rng)
        h_pert = project_h(work_h + pert)
        h_cand, c5_cand = fw_optimize(h_pert, 800, top_k=3)
        h_cand, c5_cand = fw_optimize(h_cand, 400, top_k=5)
        h_cand, c5_cand = coord_descent(h_cand, n, time.time() + 2.0, max_sweeps=8)
        if c5_cand < work_c5:
            work_h, work_c5 = h_cand, c5_cand
            no_imp = 0
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()
        else:
            temp = 0.0004 * cool + 0.00002
            if rng.rand() < np.exp(-(c5_cand - work_c5) / temp):
                work_h, work_c5 = h_cand, c5_cand
            no_imp += 1
        if no_imp > max_no_imp:
            work_h = best_h.copy()
            work_c5 = best_c5
            no_imp = 0
            max_no_imp += 5
    if work_c5 < best_c5:
        best_c5, best_h = work_c5, work_h
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: FW optimization (multiple rounds with different top_k)
    phase1_end = t0 + budget_s * 0.18
    for top_k in [3, 5, 7, 3]:
        if elapsed() > phase1_end:
            break
        best_h, best_c5 = fw_optimize(best_h, 3000, top_k=top_k)

    # Phase 2: CD polish
    phase2_end = t0 + budget_s * 0.28
    best_h, best_c5 = coord_descent(best_h, n, phase2_end, max_sweeps=50)

    # Phase 3: Basin hopping (main exploration)
    phase3_end = 0.62
    best_h, best_c5 = basin_hop(best_h, n, t0, budget_s, 0.28, phase3_end, seed)

    # Phase 4: Second basin hop with different seed and larger perturbations
    rng2 = np.random.RandomState(seed + 1000)
    h_start2 = project_h(best_h + 0.05 * rng2.standard_normal(n))
    h_cand2, c5_2 = fw_optimize(h_start2, 2000, top_k=3)
    h_cand2, c5_2 = coord_descent(h_cand2, n, t0 + budget_s * 0.70, max_sweeps=20)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h_cand2
    best_h, best_c5 = basin_hop(best_h, n, t0, budget_s, 0.70, 0.82, seed + 500)

    # Phase 5: Final FW + CD polish
    if elapsed() < budget_s * 0.88:
        best_h, best_c5 = fw_optimize(best_h, 4000, top_k=3)
        best_h, best_c5 = fw_optimize(best_h, 2000, top_k=5)
    if elapsed() < budget_s * 0.94:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.96, max_sweeps=60)

    # Phase 6: Try alternative n values
    if elapsed() < budget_s * 0.99:
        for n_try in [128, 200, 192]:
            if elapsed() > budget_s * 0.99 - 20:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_h(h_interp)
            h_try, c5_try = fw_optimize(h_interp, 2000, top_k=3)
            h_try, c5_try = coord_descent(h_try, n_try, t0 + budget_s * 0.99, max_sweeps=15)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                n = n_try

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
