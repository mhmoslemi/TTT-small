import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_J(h):
    n = len(h)
    lags = np.arange(-(n-1), n)
    j_idx = np.arange(n)
    idx1 = j_idx[None, :] + lags[:, None]
    idx2 = j_idx[None, :] - lags[:, None]
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    return (1.0 - h[np.clip(idx1, 0, n-1)]) * v1 - h[np.clip(idx2, 0, n-1)] * v2

def slp_optimize(h, n, maxiter=100, damping=0.5):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(maxiter):
        corr_cur = np.correlate(best_h, 1.0 - best_h, mode='full')
        J = compute_J(best_h)
        rhs = -corr_cur + J @ best_h
        A_ub = np.hstack([J, -np.ones((2*n-1, 1))])
        b_ub = rhs
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        bounds = [(0.0, 1.0)] * n + [(-1.0, 5.0)]
        try:
            res = linprog(np.zeros(n+1), A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        h_lp = res.x[:n]
        # Damped step
        for d in [damping, damping*0.5, damping*0.25]:
            h_try = (1-d)*best_h + d*np.clip(h_lp, 0, 1)
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                break
    return best_h, best_c5

def basin_hopping(h, n, t_end, n_iter=30):

    best_c5 = compute_c5(best_h)
    work_h = h.copy()
    work_c5 = best_c5

        amp = 0.02 + 0.05 * cool
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        hc = slsqp_smooth(h_pert, 5000*cool + 1000, n, maxiter=200)
        c5v = compute_c5(hc)
        temp = 0.0005 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v-work_c5)/temp):
            work_h, work_c5 = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc
    if work_c5 < best_c5:

import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_J(h):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    j_idx = np.arange(n)
    idx1 = j_idx[None, :] + lags[:, None]
    idx2 = j_idx[None, :] - lags[:, None]
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    return (1.0 - h[np.clip(idx1, 0, n - 1)]) * v1 - h[np.clip(idx2, 0, n - 1)] * v2

def slp_optimize(h, n, maxiter=80, damping=0.5, t_end=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(maxiter):
        if t_end is not None and time.time() > t_end:
            break
        corr_cur = np.correlate(best_h, 1.0 - best_h, mode='full')
        J = compute_J(best_h)
        rhs = -corr_cur + J @ best_h
        A_ub = np.hstack([J, -np.ones((2 * n - 1, 1))])
        b_ub = rhs
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n + [(-1.0, 5.0)]
        try:
            res = linprog(np.zeros(n + 1), A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        h_lp = np.clip(res.x[:n], 0, 1)
        improved_this_iter = False
        for d in [damping, damping * 0.6, damping * 0.3]:
            h_try = project_h((1 - d) * best_h + d * h_lp, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved_this_iter = True
                break
        if not improved_this_iter:
            break
    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_h(h, n)

def subgrad_opt(h, n, t_end, total_iters=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(total_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - it / total_iters
        alpha = 0.002 + 0.006 * cool
        k = 3 + int(5 * cool)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-k:]
        J = compute_J(best_h)
        grad = J[top_idx].sum(axis=0) / k
        h_try = project_h(best_h - alpha * grad, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def f1d_eval(a, effect, t):
    return float(np.max(a + t * effect))

def coord_descent(h, n, t_end, max_sweeps=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr_cache = np.correlate(best_h, 1.0 - best_h, mode='full')
    lags = np.arange(-(n - 1), np.arange(n - 1))
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            effect = np.zeros(2 * n - 1)
            idx1 = i + lags
            idx2 = i - lags
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            effect[v1] += (1.0 - best_h[idx1[v1]])
            effect[v2] -= best_h[idx2[v2]]
            a = corr_cache - old * effect
            res = minimize_scalar(f1d_eval, args=(a, effect),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9, 'maxiter': 40})
            t_new = res.x
            if abs(t_new - old) < 1e-10:
                continue
            trial = best_h.copy()
            trial[i] = t_new
            trial = project_h(trial, n)
            c5_trial = compute_c5(trial)
            if c5_trial < best_c5 - 1e-14:
                best_c5 = c5_trial
                best_h = trial
                corr_cache = np.correlate(best_h, 1.0 - best_h, mode='full')
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hopping(h, n, t_end, n_iter=25):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = h.copy()
    work_c5 = best_c5
    for it in range(n_iter):

            if c5v < best_c5:
                best_c5, best_h = c5v, hc
    if work_c5 < best_c5:

    return best_h, best_c5
