                    if 0 <= j - s < n:

                    imp = True
                else:

    n = len(h)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h *= (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-14:
            h *= (tgt / max(s, 1e-14))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        m = (h > 0.005) & (h < 0.995)
