        #   Net at shift L: (1-h[q+L])*vq - h[q-L]*vqm
        
        qq = q + L
        vqq = (qq >= 0) & (qq < N)
        qqc = np.clip(qq, 0, N-1)
        qm = q - L
        vqm = (qm >= 0) & (qm < N)
        qmc = np.clip(qm, 0, N-1)
        
        delta_q = np.where(vqq, 1.0 - h[qqc], 0.0) - np.where(vqm, h[qmc], 0.0)
        
        delta = delta_p + delta_q
        c_new = c + delta
        new_max = float(c_new.max())
        
        # Accept?
        if new_max <= current_max or rng.random() < np.exp((current_max - new_max) / T):
            # Accept
            h[p] = 0
            h[q] = 1
            one_pos[pi] = q
            zero_pos[qi] = p
            one_pos.sort()
            zero_pos.sort()
            c = c_new
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        
        # Cool
        if it % 100 == 0:
            T = max(T_min, T * cool_rate)
    
    return best_h, best_max * (2.0 / N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def fix_binary(h, N, rng):
    nh = N // 2
    h = h.copy().astype(float)
    n1 = int(h.sum())
    if n1 == nh:
        return h
    if n1 > nh:
        idx = np.where(h == 1)[0]
        to_rm = rng.choice(idx, n1 - nh, replace=False)
        h[to_rm] = 0
    else:
        idx = np.where(h == 0)[0]
        to_add = rng.choice(idx, nh - n1, replace=False)
        h[to_add] = 1
    return h

def compute_D(h, N):
    center = N - 1
    L_arr = np.arange(2 * N - 1) - center
    pos = np.arange(N)[:, None]
    i_pL = pos + L_arr[None, :]
    i_mL = pos - L_arr[None, :]
    vp = (i_pL >= 0) & (i_pL < N)
    vm = (i_mL >= 0) & (i_mL < N)
    ipc = np.clip(i_pL, 0, N - 1)
    imc = np.clip(i_mL, 0, N - 1)
    return np.where(vm, h[imc], 0.0) - np.where(vp, 1.0 - h[ipc], 0.0)
