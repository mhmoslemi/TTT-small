from scipy.optimize import minimize_scalar
import functools
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def _f1d(v, c_base, lin, old, nmid, dx):
    d = v - old
    val = c_base + d * lin
    val[nmid] -= d * d
    return float(np.max(val) * dx)

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42, xatol=1e-13):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    nmid = n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 5 == 4 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                best_h = project_h(best_h, n)
                return best_h, compute_c5(best_h)
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                if i + s < n:
                    lin[mn] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[mn] -= best_h[i - s]
            lin[nmid] = 1.0 - 2.0 * old
            c_base = corr.copy()
            res = minimize_scalar(
                functools.partial(_f1d, c_base=c_base, lin=lin, old=old, nmid=nmid, dx=dx),
                bounds=(0.0, 1.0), method='bounded', options={'xatol': xatol})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-14:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[nmid] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[nmid] += d * d
        if not improved and sweep > 5:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

def soft_gradient(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    exponents = np.clip(T * (corr - m), -50.0, 50.0)
    p = np.exp(exponents)
    p /= p.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = p[m_idx]
        if w < 1e-12:
            continue
        s = m_idx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        frac = hop / max(n_hops, 1)
        cool = 1.0 - frac
        # Smaller perturbations as we progress
        step = 0.005 + 0.03 * cool
        mode = hop % 7
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 8)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:

            pert = np.zeros(n)
            pert[mask] = step * 2 * rng.randn(max(1, mask.sum()))
        elif mode == 3:
            idx = rng.choice(n, size=max(3, n // 15), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * rng.randn(len(idx))
        elif mode == 4:
            # Block perturbation
            bl = rng.randint(10, 40)
            st = rng.randint(0, n - bl)
            pert = np.zeros(n)
            pert[st:st+bl] = step * rng.randn(bl)
        elif mode == 5:
            # Cosine windowed perturbation
            pert = step * rng.randn(n) * np.sin(np.pi * np.arange(n) / n)
        else:
            pert = step * 0.3 * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        # Short CD after perturbation
        hop_cd_time = max(1.0, 4.0 * cool)
        hop_end = min(t_end, time.time() + hop_cd_time)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=100, seed=seed + hop * 7)
        temp = 0.002 * cool + 0.0002
        accept = c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp))
        if accept:
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def jitter_polish(h, n, t_end, n_jitters=200, seed=42):

    best_h = h.copy()

import numpy as np
from scipy.optimize import minimize_scalar
import functools
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def _f1d(v, c_base, lin, old, nmid, dx):
    d = v - old
    val = c_base + d * lin
    val[nmid] -= d * d
    return float(np.max(val) * dx)

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42, xatol=1e-14):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    nmid = n - 1
    stuck = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                best_h = project_h(best_h, n)
                return best_h, compute_c5(best_h)
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + nmid
                mn = nmid - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                if i + s < n:
                    lin[mn] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[mn] -= best_h[i - s]
            lin[nmid] = 1.0 - 2.0 * old
            c_base = corr.copy()
            res = minimize_scalar(
                functools.partial(_f1d, c_base=c_base, lin=lin, old=old, nmid=nmid, dx=dx),
                bounds=(0.0, 1.0), method='bounded', options={'xatol': xatol})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-14:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[nmid] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[nmid] += d * d
        if not improved:
            stuck += 1
            if stuck >= 3:
                break
        else:
            stuck = 0
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=400, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        frac = hop / max(n_hops, 1)
        cool = 1.0 - frac
        step = 0.004 + 0.04 * cool * cool
        mode = hop % 8
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 10)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = step * 2.5 * rng.randn(max(1, int(mask.sum())))
        elif mode == 3:
            idx = rng.choice(n, size=max(4, n // 12), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * rng.randn(len(idx))
        elif mode == 4:
            bl = rng.randint(8, 35)
            st = rng.randint(0, max(1, n - bl))
            pert = np.zeros(n)
            pert[st:st + bl] = step * 1.5 * rng.randn(bl)
        elif mode == 5:
            pert = step * rng.randn(n) * np.sin(np.pi * np.arange(n) / n) ** 2
        elif mode == 6:
            idx1 = rng.randint(0, n // 2)
            idx2 = idx1 + rng.randint(n // 4, n - idx1 - 1)
            pert = step * 0.5 * rng.randn(n)
            pert[idx1:idx1 + 5] *= 3.0
            pert[idx2:idx2 + 5] *= 3.0
        else:
            pert = step * 0.4 * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        hop_cd_time = max(1.5, 6.0 * cool)
        hop_end = min(t_end, time.time() + hop_cd_time)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=200, seed=seed + hop * 7)
        temp = 0.0015 * cool + 0.0001
        accept = c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp))
        if accept:
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def jitter_polish(h, n, t_end, n_jitters=400, seed=42):

    best_h = h.copy()

    cur_h = h.copy()
    cur_c5 = best_c5
    for j in range(n_jitters):
        if time.time() > t_end:
            break
        frac = j / max(n_jitters, 1)
        step = 0.0008 + 0.003 * (1.0 - frac)
        pert = step * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        jit_end = min(t_end, time.time() + 2.5)
        h_opt, c5_opt = coord_descent(h_pert, n, jit_end, max_sweeps=200, seed=seed + j * 13)
        if c5_opt < cur_c5 + 0.00005:
            cur_h = h_opt
            cur_c5 = c5_opt
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5

def multi_seed_polish(h, n, t_end, seed=42):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for trial in range(6):
        if time.time() > t_end:
            break
        t_trial = min(t_end, time.time() + (t_end - time.time()) / (6 - trial))
        trial_seed = seed + trial * 1000 + 1
        h_try, c5_try = coord_descent(best_h, n, t_trial, max_sweeps=3000, seed=trial_seed, xatol=1e-15)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = compute_c5(best_h)
