```python
def swap_delta(h, i, j, n):
    """Compute change in corr_full when h[i]: 0->1, h[j]: 1->0"""
    delta = np.zeros(2*n-1)
    
    # h[i]: 0 -> 1
    # Part A: +1*(1-h[i-lag]) for i-lag in [0,n-1]
    m_lo = max(0, i)
    m_hi = min(2*n-2, i + n - 1)
    lags = np.arange(m_lo, m_hi+1) - (n-1)
    idx = i - lags  # = i - m + n - 1
    delta[m_lo:m_hi+1] += (1.0 - h[idx])
    
    # Part B: -h[i+lag] for i+lag in [0,n-1]
    m_lo2 = max(0, n-1-i)
    m_hi2 = min(2*n-2, n-1+i)
    lags2 = np.arange(m_lo2, m_hi2+1) - (n-1)
    idx2 = i + lags2  # = i + m - n + 1
    delta[m_lo2:m_hi2+1] -= h[idx2]
    
    # h[j]: 1 -> 0 (negate)
    m_lo3 = max(0, j)
    m_hi3 = min(2*n-2, j + n - 1)
    lags3 = np.arange(m_lo3, m_hi3+1) - (n-1)
    idx3 = j - lags3
    delta[m_lo3:m_hi3+1] -= (1.0 - h[idx3])
    
    m_lo4 = max(0, n-1-j)
    m_hi4 = min(2*n-2, n-1+j)
    lags4 = np.arange(m_lo4, m_hi4+1) - (n-1)
    idx4 = j + lags4
    delta[m_lo4:m_hi4+1] += h[idx4]
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        h = h + room * (diff / max(tot, 1e-14))
    else:
        h = h * (1.0 + diff / max(s, 1e-14))
    return np.clip(h, 0.0, 1.0)

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm

def binary_sa(h0, n, t_end, seed=42, n_iters=200000):
    rng = np.random.RandomState(seed)
    # Binarize
    h = (h0 >= 0.5).astype(float)
    # Ensure sum = n/2
    target = n // 2
    s = int(h.sum())
    if s > target:
        ones = np.where(h == 1)[0]
        extra = s - target
        to_zero = rng.choice(ones, extra, replace=False)
        h[to_zero] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        need = target - s
        to_one = rng.choice(zeros, need, replace=False)
        h[to_one] = 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    curr_max = float(corr.max())
    best_h = h.copy()
    best_max = curr_max
    
    T_start = 2.0
    T_end = 0.03
    n_steps = min(n_iters, max(10000, int((t_end - time.time()) * 80000)))
    
    for step in range(n_steps):
        if time.time() > t_end:
            break
        T = T_start * (T_end / T_start) ** (step / n_steps)
        # Pick random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        delta = np.zeros(2 * n - 1)
        # h[i]: 0->1
        a = max(0, i)
        b = min(2*n-2, i+n-1)
        idx = i - (np.arange(a, b+1) - (n-1))
        delta[a:b+1] += (1.0 - h[idx])
        a2 = max(0, n-1-i)
        b2 = min(2*n-2, n-1+i)
        idx2 = i + (np.arange(a2, b2+1) - (n-1))
        delta[a2:b2+1] -= h[idx2]
        # h[j]: 1->0
        a3 = max(0, j)
        b3 = min(2*n-2, j+n-1)
        idx3 = j - (np.arange(a3, b3+1) - (n-1))
        delta[a3:b3+1] -= (1.0 - h[idx3])
        a4 = max(0, n-1-j)
        b4 = min(2*n-2, n-1+j)
        idx4 = j + (np.arange(a4, b4+1) - (n-1))
        delta[a4:b4+1] += h[idx4]
        
        new_corr = corr + delta
        new_max = float(new_corr.max())
        
        accept = new_max <= curr_max or (new_max - curr_max) < T / max(1, rng.rand())
        # Actually standard SA: accept if new < old or with prob exp(-(new-old)/T)
        if new_max < curr_max or rng.rand() < np.exp(-(new_max - curr_max) / T):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            curr_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
    
    return best_h, best_max

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h.copy()

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            # Optimize h[i] given others
            h_save = best_h[i]
            best_hi = h_save
            best_val = best_c5
            # Try a few values
            for trial in [0.0, 0.25, 0.5, 0.75, 1.0, h_save]:
                best_h[i] = trial
                best_h = project_h(best_h, n)
                c5_try = compute_c5(best_h)
                if c5_try < best_val:
                    best_val = c5_try
                    best_hi = trial
            best_h[i] = best_hi
            best_h = project_h(best_h, n)
            if best_val < best_c5:
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return best_h, best_c5

        # Random

            h_sa, sa_max = binary_sa(cand, n, sa_time, seed=seed + ci * 100 + n)
            c5_sa = sa_max * 2.0 / n
            if c5_sa < best_c5:

        if elapsed() < budget_s * 0.85:
            # Use the best binary solution as starting point
            h_start = best_h.copy() if best_n == n else thue_morse(n)
            h_start = project_h(h_start, n)
            
            for T in [5000, 2000, 500, 100]:
                if elapsed() > budget_s * 0.85:

                hc = slsqp_smooth(h_start, T, n, maxiter=200)
                c5v = compute_c5(hc)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = hc.copy()
                    best_n = n
                h_start = hc

    if elapsed() < budget_s * 0.95 and best_h is not None:

    best_h = project_h(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, best_n
```

Let me reconsider the delta computation. For a swap where h[i]: 0→1 and h[j]: 1→0:

The correlation is: corr_full[m] = Σ_a h[a] * (1 - h[a - lag_m]) where lag_m = m - (n-1)

Change due to h[i]: 0→1:
- Terms where a = i: h[i] * (1-h[i-lag]) goes from 0 to (1-h[i-lag])
  Valid when 0 ≤ i-lag ≤ n-1, i.e., i-(n-1) ≤ lag ≤ i, i.e., i ≤ m ≤ i+n-1
  Delta += (1 - h[i - lag]) = (1 - h[i - (m-(n-1))]) = (1 - h[n-1 + i - m])
  
- Terms where a-lag = i, i.e., a = i+lag: (1-h[i]) * h[i+lag] goes from h[i+lag] to 0
  Valid when 0 ≤ i+lag ≤ n-1, i.e., -i ≤ lag ≤ n-1-i, i.e., n-1-i ≤ m ≤ n-1+i
  Delta -= h[i+lag] = h[i + (m-(n-1))] = h[m - (n-1-i)]

Change due to h[j]: 1→0 (negate the above with i→j):
- Delta -= (1 - h[n-1 + j - m]) for j ≤ m ≤ j+n-1
- Delta += h[m - (n-1-j)] for n-1-j ≤ m ≤ n-1+j

In numpy:
```python
delta = np.zeros(2*n-1)
# h[i] 0->1
m_lo = max(0, i); m_hi = min(2*n-2, i+n-1)
delta[m_lo:m_hi+1] += 1.0 - h[i - (np.arange(m_lo, m_hi+1) - (n-1))]
# = h[n-1 + i - np.arange(m_lo, m_hi+1)]

m_lo2 = max(0, n-1-i); m_hi2 = min(2*n-2, n-1+i)
delta[m_lo2:m_hi2+1] -= h[np.arange(m_lo2, m_hi2+1) - (n-1-i)]

# h[j] 1->0
m_lo3 = max(0, j); m_hi3 = min(2*n-2, j+n-1)
delta[m_lo3:m_hi3+1] -= 1.0 - h[n-1 + j - np.arange(m_lo3, m_hi3+1)]

m_lo4 = max(0, n-1-j); m_hi4 = min(2*n-2, n-1+j)
delta[m_lo4:m_hi4+1] += h[np.arange(m_lo4, m_hi4+1) - (n-1-j)]
```
