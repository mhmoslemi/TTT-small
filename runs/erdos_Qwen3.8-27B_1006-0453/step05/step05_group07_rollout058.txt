import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

_GT = [1.0]
_CS = [None, None]  # base, coef for coordinate descent

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-14:
            h = h * (tgt / max(s, 1e-14))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        cnt = int(m.sum())
        if cnt > 0:
            h[m] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    T = _GT[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return float((m + np.log(np.sum(np.exp(T * (c - m)))) / T) * (2.0 / n))

def _sg(x):
    T = _GT[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()

    for k in range(2 * n - 1):
        wk = float(w[k])
        if wk < 1e-14:
            continue
        lag = k - n + 1
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - x[lag:])
            g[lag:] -= wk * x[:n - lag]
        else:
            ll = -lag
            g[ll:] += wk * (1.0 - x[:n - ll])
            g[:n - ll] -= wk * x[ll:]
    return g * (2.0 / n)

                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-12, 'maxls': 50})
        return proj_h(r.x, n)
    except Exception:
        return h

    return float(np.max(_CS[0] + v * _CS[1]))

    mf = 2 * n - 1

            coef = np.zeros(mf)
            idx1 = i + lags
            v1 = (idx1 >= 0) & (idx1 < n)
            if v1.any():
                coef[v1] += (1.0 - best[idx1[v1]])
