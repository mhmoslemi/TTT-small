_CB = [None]
_CC = [None]

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            return np.full(n, 0.5)
        h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        m = (h > 0.02) & (h < 0.98)
        cnt = m.sum()
        if cnt > 0:
            h[m] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

_sT = [1.0]

def _soft_obj(x):
    T = _sT[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m + 1e-30)))) / T) * 2.0 / len(x)

def _soft_grad(x):
    T = _sT[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m + 1e-30))
    w = e / e.sum()
    g = np.zeros(n)
    s = np.arange(-(n - 1), n)
    for i in range(n):
        i1 = i + s
        i2 = i - s
        v1 = (i1 >= 0) & (i1 < n)
        v2 = (i2 >= 0) & (i2 < n)
        tmp = np.zeros(2 * n - 1)
        tmp[v1] = 1.0 - x[i1[v1]]
        tmp[v2] -= x[i2[v2]]
        g[i] = np.dot(w, tmp)
    return g * 2.0 / n

    best_h, best_c5 = _coord_desc(best_h, n, t0 + 0.87 * budget_s, 30, 400)
    # Phase 7: Try other resolutions
    for nt in [128, 200]:
        if time.time() > t0 + 0.92 * budget_s:
            break
        try:
            ht = _upsample(best_h, n, nt)
            ht = _lbfgs(ht, 2000.0, nt, 300)
            c5t = compute_c5(ht)
            ht, c5t = _coord_desc(ht, nt, t0 + 0.91 * budget_s, 10, 150)
            if c5t < best_c5:
                best_h, best_c5, n = ht, c5t, nt
        except Exception:
            pass
    # Phase 8: Final polish
    if time.time() < t0 + 0.95 * budget_s:
        best_h = _lbfgs(best_h, 80000.0, len(best_h), 600)
        best_c5 = compute_c5(best_h)
    if time.time() < t0 + 0.99 * budget_s:
        best_h, best_c5 = _coord_desc(best_h, len(best_h), t0 + 0.99 * budget_s, 10, 200)
    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
