import time
from scipy.optimize import minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')

        improved = False

        for i in order:

                return best_h, best_c5
            old = best_h[i]

        corr = np.correlate(h_c, 1.0 - h_c, mode='full')
        m = np.max(corr)
        exponents = np.clip(T * (corr - m), -80.0, 0.0)
        p = np.exp(exponents)
        ps = p.sum()
        if ps < 1e-30:
            p = np.zeros_like(corr)
            p[np.argmax(corr)] = 1.0
        else:
            p /= ps
        grad = np.zeros(n)
        for m_idx in range(2 * n - 1):
            w = p[m_idx]
            if w < 1e-15:
                continue
            s = m_idx - (n - 1)
            if s > 0:
                grad[s:] += w * (1.0 - h_c[:n - s])
                grad[:n - s] -= w * h_c[s:]
            elif s < 0:
                a = -s
                grad[:n - a] += w * (1.0 - h_c[a:])
                grad[a:] -= w * h_c[:n - a]
            else:
                grad += w * (1.0 - 2.0 * h_c)
        grad *= dx
        mom = beta * mom + grad
        gm = np.max(np.abs(mom))
        if gm < 1e-14:
            break
        alpha = 0.5 / gm
        h_new = project_h(h_c - alpha * mom, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c5

def do_basin_hop(h, n, t_end, n_hops=400, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.015 + 0.09 * cool
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        s_star = int(np.argmax(corr)) - (n - 1)
        mode = hop % 7
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = step * 2.5 * rng.randn(max(1, mask.sum()))
        elif mode == 3:
            pert = np.zeros(n)
            if s_star > 0:
                for _ in range(4):
                    i = rng.randint(0, n - s_star)
                    if cur_h[i] > 0.3 and cur_h[i + s_star] > 0.3:
                        pert[i] -= step
                        pert[i + s_star] -= step * 0.5
            elif s_star < 0:
                a = -s_star
                for _ in range(4):
                    i = rng.randint(0, n - a)
                    if cur_h[i + a] > 0.3 and cur_h[i] > 0.3:
                        pert[i + a] -= step
                        pert[i] -= step * 0.5
        elif mode == 4:
            sh = rng.randint(1, 5) * (1 if rng.rand() < 0.5 else -1)
            pert = 0.25 * (np.roll(cur_h, sh) - cur_h)
        elif mode == 5:
            idx = rng.choice(n, size=max(3, n // 15), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 1.5 * rng.randn(len(idx))
        else:
            f1 = rng.randint(1, 8)
            f2 = rng.randint(8, 20)
            ph1 = rng.uniform(0, 2 * np.pi)
            ph2 = rng.uniform(0, 2 * np.pi)
            t_arr = np.arange(n) / n
            pert = step * (0.6 * np.sin(2 * np.pi * f1 * t_arr + ph1) + 0.4 * np.sin(2 * np.pi * f2 * t_arr + ph2))
        h_pert = project_h(cur_h + pert, n)
        hop_time = max(2.0, 12.0 * cool)
        hop_end = min(t_end, time.time() + hop_time)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=120, seed=seed + hop * 7)
        temp = 0.006 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def quick_optimize_start(h0, n, t_end, seed):
    h0 = project_h(h0, n)
    h1, c1 = coord_descent(h0, n, min(t_end, time.time() + 8.0), max_sweeps=100, seed=seed)
    return h1, c1

def try_other_n(t_end, seed):
    best_extra_h = None
    best_extra_c5 = 999.0
    best_extra_n = 0
    for n_try in [80, 120, 200]:
        if time.time() > t_end:
            break
        rng = np.random.RandomState(seed + n_try)
        for trial in range(3):
            if time.time() > t_end:
                break
            h_try = rng.rand(n_try)
            h_try = project_h(h_try, n_try)
            h_opt, c5_opt = quick_optimize_start(h_try, n_try, t_end, seed + n_try + trial * 100)
            if c5_opt < best_extra_c5:
                best_extra_c5 = c5_opt
                best_extra_h = h_opt
                best_extra_n = n_try
    return best_extra_h, best_extra_c5, best_extra_n

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial coordinate descent
    t1 = t0 + budget_s * 0.08
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=400, seed=seed)

    # Phase 2: Gradient step
    t2 = t0 + budget_s * 0.14
    h_g, c5_g = gradient_step(best_h, n, t2, max_iters=800, seed=seed + 1)
    if c5_g < best_c5:
        best_h, best_c5 = h_g, c5_g
    t2b = t0 + budget_s * 0.17
    best_h, best_c5 = coord_descent(best_h, n, t2b, max_sweeps=150, seed=seed + 2)

    # Phase 3: Try other n values briefly
    t_try = t0 + budget_s * 0.22
    extra_h, extra_c5, extra_n = try_other_n(t_try, seed + 1000)
    if extra_c5 < best_c5:
        best_h = extra_h
        best_c5 = extra_c5
        n = extra_n
        t2b2 = min(t_try + 5.0, t0 + budget_s * 0.24)
        best_h, best_c5 = coord_descent(best_h, n, t2b2, max_sweeps=80, seed=seed + 2000)

    # Phase 4: Main basin hopping
    t4 = t0 + budget_s * 0.62
    best_h, best_c5 = do_basin_hop(best_h, n, t4, n_hops=500, seed=seed + 50)

    # Phase 5: Coordinate descent polish
    t5 = t0 + budget_s * 0.74
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=400, seed=seed + 1000)

    # Phase 6: Second basin hop with different seed
    t6 = t0 + budget_s * 0.85
    best_h, best_c5 = do_basin_hop(best_h, n, t6, n_hops=300, seed=seed + 2000)

    # Phase 7: Gradient + coord
    t7 = t0 + budget_s * 0.90
    h_g2, c5_g2 = gradient_step(best_h, n, t7, max_iters=600, seed=seed + 3000)
    if c5_g2 < best_c5:
        best_h, best_c5 = h_g2, c5_g2
    t7b = t0 + budget_s * 0.93
    best_h, best_c5 = coord_descent(best_h, n, t7b, max_sweeps=200, seed=seed + 4000)

    # Phase 8: Final intensive coordinate descent
    t8 = t0 + budget_s * 0.985
    best_h, best_c5 = coord_descent(best_h, n, t8, max_sweeps=300, seed=seed + 5000)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
