import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target + 1e-14:
            h = h * (target / s)
        elif s < target - 1e-14:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_v2(h, n, t_end, max_sweeps=2000, seed=42):
    """Faster coordinate descent with precomputed influence vectors."""
    rng = np.random.RandomState(seed)
    best_h = h.copy().astype(float)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            old = best_h[i]
            # Compute linear effect vector
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if s <= i:
                    lin[mp] += (1.0 - best_h[i - s])
                    lin[mn] -= best_h[i - s]
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                    lin[mn] += (1.0 - best_h[i + s])
            lin[n - 1] = 1.0 - 2.0 * old
            
            # 1D minimization: minimize max(corr + d*lin), with corr[n-1] -= d²
            c0 = corr.copy()
            # Use golden section or bounded minimization
            res = minimize_scalar(
                lambda d: float(np.max(np.where(
                    np.arange(2*n-1) == n-1,
                    c0 + d*lin - d*d,
                    c0 + d*lin
                )) * dx),
                bounds=(max(-old, -0.5), min(1.0-old, 0.5)),
                method='bounded', options={'xatol': 1e-10}
            )
            d = float(res.x)
            if abs(d) < 1e-12:
                continue
            new_val = old + d
            if new_val < 0 or new_val > 1:
                continue
            best_h[i] = new_val
            corr += d * lin
            corr[n-1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n-1] += d * d
        if not improved:
            break
    
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def soft_grad_step(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.clip(T * (corr - m), -50.0, 50.0)
    p = np.exp(exps)
    p /= p.sum()
    grad = np.zeros(n)
    center = n - 1
    for mi in range(2 * n - 1):
        w = p[mi]
        if w < 1e-12:
            continue
        s = mi - center
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

    beta = 0.6
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = min(1.0, it / max(max_iters, 1))
        T = 30.0 + 200.0 * frac
        g = soft_grad_step(hc, n, T)
        mom = beta * mom + g
        gm = np.max(np.abs(mom))
        if gm < 1e-14:
            break
        alpha = 0.3 / gm
        h_new = project_h(hc - alpha * mom, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        hc = h_new
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    t1 = t0 + budget_s * 0.12
    best_h, best_c5 = coord_descent(best_h, n, t1, 2000, seed)

    t2 = t0 + budget_s * 0.30
    hg, cg = grad_descent(best_h, n, t2, 1500, seed + 1)
    if cg < best_c5:
        best_h, best_c5 = hg, cg
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.34, 800, seed + 2)

    t3 = t0 + budget_s * 0.55
    best_h, best_c5 = sa_optimize(best_h, n, t3, seed + 10)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.60, 1500, seed + 11)

    t4 = t0 + budget_s * 0.75
    hg2, cg2 = grad_descent(best_h, n, t4, 800, seed + 20)
    if cg2 < best_c5:
        best_h, best_c5 = hg2, cg2
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.78, 600, seed + 21)

    t5 = t0 + budget_s * 0.90
    best_h, best_c5 = sa_optimize(best_h, n, t5, seed + 30)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.94, 2000, seed + 40)

    t6 = t0 + budget_s * 0.97
    hg3, cg3 = grad_descent(best_h, n, t6, 500, seed + 50)
    if cg3 < best_c5:
        best_h, best_c5 = hg3, cg3

    t7 = t0 + budget_s * 0.995
    best_h, best_c5 = coord_descent(best_h, n, t7, 1000, seed + 60)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
