```python
def binary_local_search(h_bin, n, t_end, max_iters=10000):
    h = h_bin.copy().astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        best_swap = None
        best_val = best_c5
        # Try random swaps
        for _ in range(20):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            # Simulate swap
            h[i] = 0; h[j] = 1
            corr_new = np.correlate(h, 1.0-h, mode='full')
            val = float(np.max(corr_new) * dx)
            h[i] = 1; h[j] = 0
            if val < best_val - 1e-12:
                best_val = val
                best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0; h[j] = 1
        corr = np.correlate(h, 1.0-h, mode='full')
        best_c5 = best_val
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return h, best_c5
```

This is O(n²) per iteration (correlate is O(n²) for n=160 which is 25600, fine). 20 swaps per iter * 10000 iters = 200K correlations = 5B operations. Might be too slow.

Let me optimize: instead of recomputing the full correlation, I can compute the change incrementally (like in the coordinate descent).

When I swap h[i] from 1 to 0 and h[j] from 0 to 1:
- This is equivalent to two coordinate changes
- I can use the `lin` vectors to update corr incrementally

Let me combine both approaches. Actually, for the binary case, the coordinate update is just 0→1 or 1→0, so d = ±1. The update is:
corr_new = corr ± lin[i] (for the position changing from 1 to 0: d=-1, so corr - lin[i])
plus the zero-lag quadratic correction.

Wait, for binary, the quadratic term at n-1: 
corr[n-1] = sum h[k]*(1-h[k]) = 0 always for binary.
When h[i] goes from 1 to 0: the term h[i]*(1-h[i]) goes from 0 to 0. No change!
When h[j] goes from 0 to 1: same, no change.

So for binary, the quadratic term doesn't matter! The update is purely linear:
corr_new = corr - lin[i] + lin[j] (swap i from 1→0, j from 0→1)

This makes the binary swap evaluation O(n) instead of O(n²).

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            h = np.clip(h * (target / s), 0.0, 1.0)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + room * ((target - s) / tot), 0.0, 1.0)
    return h

def get_lin(h, i, n):
    lin = np.zeros(2 * n - 1)
    s = np.arange(1, n)
    m1 = s <= i
    m2 = i + s <= n - 1
    if m1.any():
        lin[n - 1 + s[m1]] += 1.0 - h[i - s[m1]]
        lin[n - 1 - s[m1]] -= h[i - s[m1]]
    if m2.any():
        lin[n - 1 + s[m2]] -= h[i + s[m2]]
        lin[n - 1 - s[m2]] += 1.0 - h[i + s[m2]]
    lin[n - 1] = 1.0 - 2.0 * h[i]
    return lin

def coord_descent(h_in, n, t_end, max_sweeps=3000, seed=42):
    rng = np.random.RandomState(seed)
    h = h_in.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 7 == 6 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return h, best_c5
            old = h[i]
            lin = get_lin(h, i, n)
            c_base = corr.copy()

            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val) * dx)

            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13, 'maxiter': 150})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-14:
                continue
            corr = c_base + d * lin
            corr[n - 1] -= d * d
            h[i] = v_new
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                h[i] = old
                corr = c_base + d * lin
                corr[n - 1] -= d * d
        if not improved and sweep > 20:
            break
    h = project_h(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def binary_search(h_in, n, t_end, seed=42):
    h = (h_in >= 0.5).astype(float)
    if h.sum() != n / 2:
        idx = np.argsort(h)
        if h.sum() > n / 2:
            for k in idx[:int(h.sum() - n / 2)]:
                h[k] = 0.0
        else:
            for k in idx[::-1][:int(n / 2 - h.sum())]:
                h[k] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    rng = np.random.RandomState(seed)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(200000):
        if time.time() > t_end:
            break
        li = get_lin(h, ones[0], n)  # warm up
        best_val = best_c5
        best_pair = None
        for _ in range(30):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            li_i = get_lin(h, i, n)
            li_j = get_lin(h, j, n)
            corr_new = corr - li_i + li_j
            val = float(np.max(corr_new) * dx)
            if val < best_val - 1e-14:
                best_val = val
                best_pair = (i, j, li_i, li_j, corr_new.copy())
        if best_pair is None:
            break
        i, j, li_i, li_j, corr_new = best_pair
        h[i] = 0.0
        h[j] = 1.0
        corr = corr_new
        best_c5 = best_val
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h, best_c5

def basin_hop(h_in, n, t_end, n_hops=500, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h_in.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_in.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.02 + 0.08 * cool
        mode = hop % 7
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 20)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            nswap = rng.randint(5, 30)
            pert = np.zeros(n)
            ones_idx = np.where(cur_h > 0.5)[0]
            zeros_idx = np.where(cur_h < 0.5)[0]
            for _ in range(min(nswap, len(ones_idx), len(zeros_idx))):
                a = ones_idx[rng.randint(len(ones_idx))]
                b = zeros_idx[rng.randint(len(zeros_idx))]
                pert[a] += (cur_h[b] - cur_h[a])
                pert[b] += (cur_h[a] - cur_h[b])
        elif mode == 3:
            nswap = rng.randint(10, 50)
            pert = np.zeros(n)
            ones_idx = np.where(cur_h > 0.5)[0]
            zeros_idx = np.where(cur_h < 0.5)[0]
            for _ in range(min(nswap, len(ones_idx), len(zeros_idx))):
                a = ones_idx[rng.randint(len(ones_idx))]
                b = zeros_idx[rng.randint(len(zeros_idx))]
                pert[a] -= 1.0
                pert[b] += 1.0
        elif mode == 4:
            k = rng.randint(5, 40)
            start = rng.randint(0, n - k)
            pert = np.zeros(n)
            pert[start:start+k] = step * rng.randn(k)
        elif mode == 5:
            pert = np.zeros(n)
            nblock = rng.randint(1, 4)
            for _ in range(nblock):
                k = rng.randint(10, 60)
                start = rng.randint(0, n - k)
                pert[start:start+k] += 0.3 * cool * rng.choice([-1, 1])
        else:
            pert = step * 0.5 * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        hop_time = 3.0 + 6.0 * cool
        hop_end = min(t_end, time.time() + hop_time)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end,
                                       max_sweeps=int(50 + 80 * cool), seed=seed + hop)
        temp = 0.003 * cool + 0.0002
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[::2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Thorough coordinate descent
    t1 = t0 + budget_s * 0.18
    h1, c5_1 = coord_descent(best_h, n, t1, max_sweeps=5000, seed=seed)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Binary exploration
    t2 = t0 + budget_s * 0.26
    hb, c5_b = binary_search(best_h, n, t2, seed=seed + 1)
    if c5_b < best_c5:
        best_h, best_c5 = hb, c5_b
    if time.time() < t0 + budget_s * 0.30:
        h2, c5_2 = coord_descent(best_h, n, t0 + budget_s * 0.30, max_sweeps=2000, seed=seed + 2)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: Main basin hopping
    t3 = t0 + budget_s * 0.78
    best_h, best_c5 = basin_hop(best_h, n, t3, n_hops=600, seed=seed + 10)

    # Phase 4: Final coordinate descent polish
    t4 = t0 + budget_s * 0.93
    h4, c5_4 = coord_descent(best_h, n, t4, max_sweeps=5000, seed=seed + 100)
    if c5_4 < best_c5:
        best_h, best_c5 = h4, c5_4

    # Phase 5: Another short basin hop with finer steps
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.97, n_hops=100, seed=seed + 500)
        if time.time() < t0 + budget_s * 0.995:
            h5, c5_5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=1000, seed=seed + 600)
            if c5_5 < best_c5:
                best_h, best_c5 = h5, c5_5

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
