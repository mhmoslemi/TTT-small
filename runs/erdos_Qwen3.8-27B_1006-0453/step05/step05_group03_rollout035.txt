```python
    # where h0[j] = h[j] for j≠i, h0[i]=0
    # So (1-h0[j+k-(n-1)]) = (1-h[j+k-(n-1)]) unless j+k-(n-1)=i
    # If j+k-(n-1) = i: (1-h0[i]) = 1 (since h0[i]=0)
    
    # This is getting really messy. Let me just compute it differently.
    # 
    # Direct approach: corr[k] as function of x=h[i]:
    # corr[k] = sum_j h[j]*(1-h[j+k-(n-1)])
    # = (sum of terms not involving h[i]) + (terms involving h[i])
    # 
    # Terms involving h[i]:
    # (1) j=i: h[i]*(1-h[i+k-(n-1)]) = x*(1-h[i+m]) where m=k-(n-1), valid if 0≤i+m<n
    #     Special case m=0: x*(1-x) = x-x²
    # (2) j+k-(n-1)=i: h[j]*(1-h[i]) = h[i-m]*(1-x) where m=k-(n-1), valid if 0≤i-m<n
    #     Special case m=0: same as (1)
    #
    # So for m≠0:
    #   contribution = x*(1-h[i+m]) + h[i-m]*(1-x) [each term only if index valid]
    #   = x*(1-h[i+m]) + h[i-m] - x*h[i-m]
    #   = h[i-m] + x*((1-h[i+m]) - h[i-m])
    #
    # For m=0:
    #   contribution = x*(1-x) = x - x²
    #
    # Total: corr[k] = base[k] + contribution
    # where base[k] = sum of terms not involving h[i] at all
```

OK this is getting complicated. Let me just use a simpler but still effective approach:

For each coordinate i, compute how corr changes as a function of x=h[i]:
- corr[k] = base_k + linear_k * x + quadratic_k * x²
- Actually the only quadratic term is at m=0 (k=n-1): corr[n-1] gets x-x²
- For all other k, corr[k] is linear in x

So the objective max_k (corr[k]*dx) is:
- max of (linear functions + one quadratic) 
- The quadratic term x-x² at k=n-1 is bounded by 0.25, and for most solutions it's not the max
- So we can approximate as max of linear functions and use the convex envelope method

Actually, you know what, let me just use a practical approach:
1. For each coordinate, do a golden-section or ternary search over [0,1] to find the optimal x
2. This avoids the complexity of the exact solution
3. With n=160, each evaluation is O(n) for the correlation update (if we precompute)

Wait, np.correlate for n=160 takes maybe 50-100 microseconds. So 100 evaluations per coordinate × 160 coordinates = 16000 correlate calls ≈ 1-2 seconds per sweep. That's too slow for many sweeps.

Better: compute the correlation incrementally. When h[i] changes from old to new, the change in corr is:
delta_corr[k] = (new - old) * A[k] (approximately, ignoring the x² term at k=n-1)

where A[k] = (1-h[i+k-(n-1)]) - h[i-k+(n-1)] (with appropriate boundary handling)

Then we can evaluate max(corr + delta) very fast (O(1) per k, O(n) total).

Let me code this up properly:

```python
def coord_update(h, n, i):
    """Find best h[i] to minimize max(corr), keeping other h fixed. Ignores sum constraint."""
    old = h[i]
    corr = np.correlate(h, 1.0-h, mode='full')  # current correlation
    
    # A[k] = d(corr[k])/d(h[i]) (ignoring x² term)
    A = np.zeros(2*n - 1)
    m = np.arange(-(n-1), n)  # all possible lags
    # For each lag m: k = n-1+m
    # corr[n-1+m] gets contribution from:
    #   h[i]*(1-h[i+m]) if 0<=i+m<n: contributes (1-h[i+m]) to A
    #   h[i-m]*(1-h[i]) if 0<=i-m<n: contributes -h[i-m] to A
    idx1 = m + i  # = i+m, valid if 0<=idx1<n
    valid1 = (idx1 >= 0) & (idx1 < n)
    A[valid1] += (1.0 - h[idx1[valid1]])
    # But wait, when m=0: i+m = i, so 1-h[i] = 1-old. And we're computing d/dx, so it should be 1 (derivative of x*(1-x) wrt x is 1-2x, not 1-x)...
    # Actually for m=0: the term is x*(1-x), derivative is 1-2x
    # For m≠0, i+m≠i: the term is x*(1-h[i+m]), derivative is (1-h[i+m])
    # For m≠0, i-m≠i: the term is h[i-m]*(1-x), derivative is -h[i-m]
    # For m=0: both i+m and i-m equal i, term is x*(1-x), derivative is 1-2x
    
    # So: for m=0: A[n-1] = 1 - 2*old (from x*(1-x))
    # Actually wait, when m=0: i+m = i AND i-m = i
    # From first term: h[i]*(1-h[i+m]) = x*(1-h[i]) = x*(1-x), deriv = 1-2*old... 
    # But I already added (1-h[i]) = 1-old from valid1
    # And from second term: h[i-m]*(1-h[i]) = h[i]*(1-x) = old*(1-x), deriv = -old
    # Total from m=0: (1-old) + (-old) = 1-2*old. Correct!
    
    idx2 = i - m  # = i-m, valid if 0<=idx2<n  
    valid2 = (idx2 >= 0) & (idx2 < n)
    A[valid2] -= h[idx2[valid2]]
    
    # Now: corr_new[k] ≈ corr[k] + (x-old)*A[k] (ignoring -old*x + x² correction at k=n-1)
    # Actually the quadratic correction: at k=n-1, we have x*(1-x) instead of old*(1-old) + (x-old)*(1-2*old)
    # The difference is: x-x² - old+old² - (x-old)(1-2*old)
    # = x-x² - old+old² - x+old + x*2*old - x*old
    # = -x² - old + old² + x*old
    # Hmm this is getting messy. For small changes, the linear approximation is fine.
    # For the purposes of finding the optimal x, the linear approximation is good enough
    # since the correction is O((x-old)²).
    
    # Minimize max_k (corr[k] + (x-old)*A[k]) over x ∈ [0,1]
    y = x - old, range y ∈ [-old, 1-old]
    # This is: minimize max_k (corr[k] - old*A[k] + y*A[k])
    # Let b[k] = corr[k] - old*A[k], c[k] = A[k]
    # Minimize max_k (b[k] + y*c[k]) over y ∈ [-old, 1-old]
    
    b = corr - old * A
    c = A
    
    # Find y that minimizes max_k (b[k] + y*c[k])
    # This is a 1D convex piecewise-linear minimization
    # Optimal y is where the subgradient contains 0
    # i.e., max_{k: c[k]>0} (b[k]+y*c[k]) crosses max_{k: c[k]<0} (b[k]+y*c[k])
    
    # Simple approach: ternary search (works because function is convex)
    lo, hi = -old, 1.0 - old
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(b + m1 * c)
        f2 = np.max(b + m2 * c)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    y_opt = (lo + hi) / 2.0
    x_opt = old + y_opt
    return np.clip(x_opt, 0.0, 1.0)
```

This should work! Each iteration is O(n) for the max computation, 50 iterations, so O(50n) per coordinate. Total per sweep: O(50n²) = 50*160² = 1.28M operations. Very fast!

```python
def coord_descent_fast(h, n, t_end, max_sweeps=10000):

    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0-h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = h[i]
            # Compute A (derivative of corr wrt h[i])
            A = np.zeros(2*n - 1)
            m_arr = np.arange(-(n-1), n)
            idx1 = i + m_arr
            v1 = (idx1 >= 0) & (idx1 < n)
            A[v1] += (1.0 - h[idx1[v1]])
            idx2 = i - m_arr
            v2 = (idx2 >= 0) & (idx2 < n)
            A[v2] -= h[idx2[v2]]
            
            # Linearize: corr_new ≈ corr + (x-old)*A
            b = corr - old * A
            c = A
            
            # Ternary search for optimal y = x - old in [-old, 1-old]
            lo, hi = -old, 1.0 - old
            if lo >= hi:
                x_opt = old

                        hi = m2
                    else:
                        lo = m1

            delta = x_opt - old
            if abs(delta) < 1e-10:
                continue
            
            h[i] = x_opt
            corr += delta * A
            improved = True
        
        if not improved:
            break
        
        # Optional: project to maintain sum constraint
        # (do this less frequently to avoid disturbing the optimization)
```

```python
def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    for _ in range(100):
        if diff > 0:
            # Need to increase sum: increase values that are < 1
            mask = h < 1.0
            if not mask.any():
                return h
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                return h
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
    return h
```

SM = (1/T) * log(Σ exp(T*g_k))
∂SM/∂h_i = Σ_k [exp(T*g_k) / Σ_j exp(T*g_j)] * (∂g_k/∂h_i)
= Σ_k p_k * (dx * A_k) where p_k = exp(T*g_k)/Σ_j exp(T*g_j)

This is a weighted average of dx*A_k with weights p_k. For large T, p_k concentrates on the max, so this approaches dx * A_{k*} where k* = argmax g_k.

```python
def lbfgsb_softmax(h0, n, T, t_end, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0
    mu = 10.0  # penalty for sum constraint
    
    def obj_and_grad(h):
        corr = np.correlate(h, 1.0-h, mode='full')
        g = corr * dx
        # Soft-max
        g_max = np.max(g)
        exp_g = np.exp(T * (g - g_max))  # shift for numerical stability
        SM = (g_max + np.log(np.sum(exp_g)) / T)
        
        # Penalty
        s = h.sum() - target
        val = SM + mu * s * s
        
        # Gradient
        # d(SM)/d(h_i) = dx * Σ_k p_k * A[k,i] where p_k = exp_g[k]/sum(exp_g)
        p = exp_g / np.sum(exp_g)
        
        # Compute Σ_k p_k * A[k,i] for all i
        # A[k,i] = (1-h[i+k-(n-1)]) - h[i-k+(n-1)] (with boundary)
        # This is: for each k, contribute p[k] * [(1-h[...]) - h[...]]
        # Σ_k p[k] * (1-h[i+k-(n-1)]) for valid i+k-(n-1)
        # - Σ_k p[k] * h[i-k+(n-1)] for valid i-k+(n-1)
        # 
        # First sum: Σ_k p[k] * (1-h[i+k-(n-1)])
        # Let m = k-(n-1): Σ_m p[n-1+m] * (1-h[i+m]) for valid i+m
        # = (1 - Σ_m p[n-1+m]*h[i+m]) for valid m
        # This is like a weighted sum of h with weights p_win
        # 
        # Let's compute it as: for each i, sum over m of p[n-1+m]*(1-h[i+m])
        # = 1 - sum_i' h[i'] * (sum of p values where i+m=i', i.e., m=i'-i)
        # Hmm let me just do it directly with numpy
        m_arr = np.arange(-(n-1), n)
        # First term: for each i, sum_m p[n-1+m] * (1-h[i+m]) where 0<=i+m<n
        # = sum_m p[n-1+m] * (1-h[i+m]) * indicator(0<=i+m<n)
        # Let's compute: for each valid m, add p[n-1+m] to positions i where i+m is valid
        # i+m valid means 0<=i+m<n, i.e., -m<=i<n-m
        # So: first_term[i] = sum_{m: 0<=i+m<n} p[n-1+m] * (1-h[i+m])
        # = sum_{j: 0<=j<n} p[n-1+(j-i)] * (1-h[j]) where j=i+m
        # = 1 - sum_j p[n-1+(j-i)] * h[j]
        # = 1 - (h ⊛ p_flipped)[i] where p_flipped[m] = p[n-1+m]
        
        # Actually simpler: just compute the correlation-like sum
        # first_sum[i] = sum_m p[n-1+m] * (1-h[i+m]) for valid i+m in [0,n)
        # = sum_m p[n-1+m] - sum_m p[n-1+m]*h[i+m] for valid
        # The first part: sum of p over valid m (depends on i)
        # This is tedious. Let me use a different approach.
        
        # Alternative: use the fact that the gradient of corr wrt h is:
        # d(corr[k])/d(h[i]) = (1-h[i+k-(n-1)]) - h[i-k+(n-1)] (with boundary)
        # = A[i, k]
        # And grad_SM[i] = dx * sum_k p[k] * A[i, k]
        
        # Let's compute this as two weighted sums:
        # grad[i] = dx * (sum_k p[k]*(1-h[i+k-(n-1)]) - sum_k p[k]*h[i-k+(n-1)])
        # = dx * (sum_k p[k] - sum_k p[k]*h[i+k-(n-1)] - sum_k p[k]*h[i-k+(n-1)])
        #   (with appropriate boundary handling)
        
        # For the middle term: sum_k p[k]*h[i+k-(n-1)]
        # Let m=k-(n-1): sum_m p[n-1+m]*h[i+m] for valid i+m
        # This is a weighted "correlation" of h with p_win where p_win[m] = p[n-1+m]
        
        # I'll compute this using numpy convolve
        # Actually, let me just do it with a loop over the unique lags
        # For n=160, there are 319 lags, so this is fast enough
        
        grad_SM = np.zeros(n)
        for k in range(2*n - 1):
            if p[k] < 1e-15:
                continue
            m = k - (n-1)
            # A[k, i] = (1-h[i+m]) [if 0<=i+m<n] - h[i-m] [if 0<=i-m<n]
            idx1 = i_arr + m  # where i_arr = np.arange(n)
            v1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i_arr - m
            v2 = (idx2 >= 0) & (idx2 < n)
            contrib = np.zeros(n)
            contrib[v1] += (1.0 - h[idx1[v1]])
            contrib[v2] -= h[idx2[v2]]
            grad_SM += p[k] * contrib
        
        grad = dx * grad_SM + 2.0 * mu * (h.sum() - target)
        return val, grad
    
    try:
        res = minimize(obj_and_grad, h0, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h0
```

corr = np.correlate(h, 1-h, 'full')
∂corr[k]/∂h[i] = (1-h[i+k-(n-1)]) * 1[h[i+k-(n-1)] valid] + h[i-k+(n-1)] * (-1) * 1[h[i-k+(n-1)] valid]
Wait no: corr[k] = Σ_j h[j]*(1-h[j+k-(n-1)])
∂corr[k]/∂h[i]:
- j=i: (1-h[i+k-(n-1)]) if 0≤i+k-(n-1)<n
- j+k-(n-1)=i, i.e., j=i-k+(n-1): h[j],[-h[i]] differentiated = -h[j] if 0≤i-k+(n-1)<n

So ∂corr[k]/∂h[i] = (1-h[i+m_k]) * I[0≤i+m_k<n] - h[i-m_k] * I[0≤i-m_k<n]
where m_k = k-(n-1)

And grad_SM[i] = Σ_k p[k] * dx * ∂corr[k]/∂h[i]
= dx * Σ_k p[k] * [(1-h[i+m_k])*I[0≤i+m_k<n] - h[i-m_k]*I[0≤i-m_k<n]]

For fixed i:
= dx * [Σ_{k: 0≤i+m_k<n} p[k]*(1-h[i+m_k]) - Σ_{k: 0≤i-m_k<n} p[k]*h[i-m_k]]

Let j = i+m_k for the first sum: m_k = j-i, k = n-1+j-i
Valid when 0≤j<n and 0≤n-1+j-i<2n-1 (always true for valid j and i)
So first sum = Σ_j h... wait:
= Σ_{j=0}^{n-1} p[n-1+j-i] * (1-h[j])

Second sum: let j' = i-m_k: m_k = i-j', k = n-1+i-j'
Valid when 0≤j'<n
= Σ_{j'=0}^{n-1} p[n-1+i-j'] * h[j']

So: grad_SM[i] = dx * [Σ_j p[n-1+j-i]*(1-h[j]) - Σ_j p[n-1+i-j]*h[j]]
= dx * [Σ_j p[n-1+j-i] - Σ_j p[n-1+j-i]*h[j] - Σ_j p[n-1+i-j]*h[j]]

Note that p[n-1+j-i] and p[n-1+i-j] are different weightings (unless p is symmetric).

This is a matrix-vector product! If I define:
M1[i,j] = p[n-1+j-i] (if 0≤n-1+j-i<2n-1, else 0)
M2[i,j] = p[n-1+i-j] (if 0≤n-1+i-j<2n-1, else 0)

Then grad_SM = dx * (M1@1 - M1@h - M2@h)

For n=160, M is 160×160 = 25,600 entries. Computing M1 and M2 and doing the matrix multiplies is fast.

But building M takes O(n²) and we need to do it every gradient call. For L-BFGS-B with 200 iterations, that's 200 * O(n²) = 200 * 25600 = 5M operations. Fine.

Actually, I can avoid building the full matrix. The sums are:
S1[i] = Σ_j p[n-1+j-i]*(1-h[j]) = Σ_j p[n-1+j-i] - Σ_j p[n-1+j-i]*h[j]
S2[i] = Σ_j p[n-1+i-j]*h[j]

S1[i] - S2[i] = Σ_j [p[n-1+j-i]*(1-h[j]) - p[n-1+i-j]*h[j]]

For each i, this is a weighted sum over j. I can compute all i in O(n²) using a loop, or I can use FFT-based convolution if the weights have a special structure. But p is arbitrary (depends on T and the current h), so FFT won't help directly.

```python
i_arr = np.arange(n)

# This is a Toeplitz-like matrix

# Similarly M2[i,j] depends on (i-j)

# Using numpy: create the index matrix
idx1 = n - 1 + (i_arr[:, None] * 0 + np.arange(n)[None, :] - i_arr[:, None])  # j - i
# Wait: idx1[i,j] = n-1+j-i

valid1 = (idx1 >= 0) & (idx1 < 2*n-1)
W1 = np.where(valid1, p[idx1], 0.0)  # shape (n, n)

idx2 = (np.arange(n)[:, None] - np.arange(n)[None, :]) + (n-1)  # i - j
valid2 = (idx2 >= 0) & (idx2 < 2*n-1)
W2 = np.where(valid2, p[idx2], 0.0)

grad_SM = dx * (W1.sum(axis=1) - W1 @ h - W2 @ h)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_smooth(h0, n, T, t_end, maxiter=300):
    dx = 2.0 / n
    target = n / 2.0
    mu = 50.0
    p_len = 2 * n - 1
    i_arr = np.arange(n)
    idx1 = (i_arr[None, :] - i_arr[:, None]) + (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < p_len)
    idx2 = (i_arr[:, None] - i_arr[None, :]) + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < p_len)

    def obj_and_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        gmax = np.max(g)
        eg = np.exp(T * (g - gmax))
        s_eg = np.sum(eg)
        SM = gmax + np.log(s_eg) / T
        s = h.sum() - target
        val = SM + mu * s * s
        p = eg / s_eg
        W1 = np.where(valid1, p[idx1], 0.0)
        W2 = np.where(valid2, p[idx2], 0.0)
        grad_SM = dx * (W1.sum(axis=1) - W1 @ h - W2 @ h)
        grad = grad_SM + 2.0 * mu * s
        return val, grad

    try:
        res = minimize(obj_and_grad, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h0

def fast_coord_descent(h0, n, t_end, project_every=5):
    dx = 2.0 / n
    p_len = 2 * n - 1
    h = h0.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    m_arr = np.arange(-(n - 1), n)

    for sweep in range(5000):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            old = h[i]
            idx1 = i + m_arr
            v1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i - m_arr
            v2 = (idx2 >= 0) & (idx2 < n)
            A = np.zeros(p_len)
            A[v1] += (1.0 - h[idx1[v1]])
            A[v2] -= h[idx2[v2]]
            b = corr - old * A
            lo, hi = -old, 1.0 - old
            span = hi - lo
            if span < 1e-12:
                continue
            for _ in range(45):
                m1 = lo + span / 3.0
                m2 = hi - span / 3.0
                if np.max(b + m1 * A) < np.max(b + m2 * A):
                    hi = m2
                else:
                    lo = m1
                span = hi - lo
                if span < 1e-12:
                    break
            x_opt = np.clip(old + (lo + hi) / 2.0, 0.0, 1.0)
            delta = x_opt - old
            if abs(delta) < 1e-10:
                continue
            h[i] = x_opt
            corr += delta * A
            improved = True
        if sweep % project_every == project_every - 1:
            h = project_h(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            h = project_h(h, n)
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
            break
        h = project_h(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr) * dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def sa_coord(h0, n, t_end):

    p_len = 2 * n - 1
    m_arr = np.arange(-(n - 1), n)
    h = h0.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c5 = float(np.max(corr) * dx)
    best_c5 = cur_c5
    best_h = h.copy()
    T_start, T_end_sa = 0.005, 0.00005
    total_iters = 50000
    count = 0

    for it in range(total_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - count / total_iters
        T = T_start * (T_end_sa / T_start) ** (1.0 - cool) if cool > 0 else T_end_sa
        i = np.random.randint(n)
        old = h[i]
        idx1 = i + m_arr
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i - m_arr
        v2 = (idx2 >= 0) & (idx2 < n)
        A = np.zeros(p_len)
        A[v1] += (1.0 - h[idx1[v1]])
        A[v2] -= h[idx2[v2]]
        # Random move: perturb h[i]
        delta = np.random.randn() * 0.08 * (0.3 + cool)
        x_new = np.clip(old + delta, 0.0, 1.0)
        d = x_new - old
        if abs(d) < 1e-10:
            continue
        new_max = float(np.max(corr + d * A)) * dx
        if new_max < cur_c5 or np.random.rand() < np.exp(-(new_max - cur_c5) / T):
            h[i] = x_new
            corr += d * A
            cur_c5 = new_max
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        count += 1

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def block_optimize(h0, n, t_end):
    h = h0.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    block_sizes = [4, 8, 12, 16, 20]
    iters = 2000
    count = 0
    while count < iters and time.time() < t_end:
        bs = block_sizes[np.random.randint(len(block_sizes))]
        i = np.random.randint(0, n - bs)
        j = np.random.randint(0, n - bs)
        if abs(i - j) < bs:
            count += 1
            continue
        h_try = h.copy()
        swap_val = np.random.choice([-1, 1])
        h_try[i:i+bs] += 0.03 * swap_val
        h_try[j:j+bs] -= 0.03 * swap_val
        h_try = project_h(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            h = h_try
        count += 1
    return best_h, best_c5
