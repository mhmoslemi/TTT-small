import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h[:] = 0.5
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

_G = {'T': 100.0, 'n': 16, 'k': 0}
_CB = [np.zeros(1)]
_CC = [np.zeros(1)]

def _soft_obj(x):
    T, n, k = _G['T'], _G['n'], _G['k']
    c = np.correlate(x, 1.0 - x, mode='full')
    if 0 < k < len(c):
        idx = np.argsort(c)[-k:]
        vals = c[idx]
    else:
        vals = c
    m = vals.max()
    return (m + np.log(np.sum(np.exp(T * (vals - m)))) / T) * 2.0 / n

def _soft_grad(x):
    T, n, k = _G['T'], _G['n'], _G['k']
    c = np.correlate(x, 1.0 - x, mode='full')
    mf = 2 * n - 1
    if 0 < k < mf:
        idx = np.argsort(c)[-k:]
        w = np.zeros(mf)
        v = c[idx]
        m = v.max()
        e = np.exp(T * (v - m))
        w[idx] = e / e.sum()
    else:
        m = c.max()
        e = np.exp(T * (c - m))
        w = e / e.sum()
    g = np.zeros(n)
    for kk in range(mf):
        wk = w[kk]
        if wk < 1e-15:
            continue
        lo1, hi1 = max(0, n - 1 - kk), min(n, mf - kk)
        if lo1 < hi1:
            g[lo1:hi1] += wk * (1.0 - x[lo1 + kk - n + 1:hi1 + kk - n + 1])
        lo2, hi2 = max(0, kk - n + 1), min(n, kk + 1)
        if lo2 < hi2:
            g[lo2:hi2] -= wk * x[lo2 - kk + n - 1:hi2 - kk + n - 1]
    return g * (2.0 / n)

def lbfgs(h, T, n, iters=300, k=0):
    _G['T'], _G['n'], _G['k'] = T, n, k
    try:
        r = minimize(h, _soft_obj, _soft_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

def _f1d(hi):
    return float(np.max(_CB[0] + hi * _CC[0]))

def coord_desc(h, n, t_end, sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for kk in range(mf):
                j2 = i + kk - n + 1
                if 0 <= j2 < n:
                    coef[kk] += 1.0 - best[j2]
                j1 = i - kk + n - 1
                if 0 <= j1 < n:
                    coef[kk] -= best[j1]
            _CB[0] = corr - old * coef
            _CC[0] = coef
            r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-13})
            nv = float(r.x)

            trial = best.copy()
            trial[i] = nv

            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode='full')

    return best, best_c5
