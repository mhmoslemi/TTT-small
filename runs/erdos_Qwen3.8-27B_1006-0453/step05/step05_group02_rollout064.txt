```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            h = h + diff / n
        else:
            h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def solve_1d(c, b, lo=0.0, hi=1.0):
    """Solve min_x max_k (c[k] + b[k]*x) for x in [lo, hi]."""
    # Evaluate at boundaries
    f_lo = np.max(c)
    f_hi = np.max(c + b)
    best_x = lo if f_lo <= f_hi else hi
    best_f = min(f_lo, f_hi)
    
    # Find intersections
    n_lines = len(c)
    for i in range(n_lines):
        if b[i] == 0:
            continue
        for j in range(i+1, n_lines):
            if b[j] == b[i]:
                continue
            x_star = (c[j] - c[i]) / (b[i] - b[j])
            if x_star < lo or x_star > hi:
                continue
            # Check if this is on the upper envelope
            f_at = np.max(c + b * x_star)
            # Verify i and j are both (close to) maximal
            if f_at < best_f - 1e-15:
                best_f = f_at
                best_x = x_star
    return best_x, best_f

            # For each k, compute I1 = [0 <= idx-k < n], I2 = [0 <= idx+k < n]
            i1_idx = idx - k_range  # index in h for (1-h[idx-k])
            i2_idx = idx + k_range  # index in h for h[idx+k]

            I2 = (i2_idx >= 0) & (i2_idx < n)
```

```python
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(c + b * m1)
        f2 = np.max(c + b * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    f_star = np.max(c + b * x_star)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def solve_1d(c, b, lo=0.0, hi=1.0):
    """Solve min max_k(c[k]+b[k]*x) for x in [lo,hi] via ternary search."""
    for _ in range(80):
        m1 = lo + (hi - lo) * 0.333333
        m2 = hi - (hi - lo) * 0.333333
        if np.max(c + b * m1) < np.max(c + b * m2):
            hi = m2
        else:
            lo = m1
    x_star = 0.5 * (lo + hi)
    return x_star, float(np.max(c + b * x_star))

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')

            break

            if time.time() > t_end:
                return best_h, best_c5

            i2_idx = i + k_range

            I2 = (i2_idx >= 0) & (i2_idx < n)

            b[I2] -= best_h[i2_idx[I2]]
```

```python
def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()

            return h
```

```python
                continue
            h_try = best_h.copy()
            h_try[i], h_try[j] = best_h[j], best_h[i]
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp += 1
        if imp == 0:
            break
    return best_h, best_c5

def block_transfer_search(h, n, t_end, max_rounds):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        imp = 0
        for _ in range(200):
            if time.time() > t_end:
                return best_h, best_c5
            bs = rng_global.randint(3, max(4, n // 10))
            i = rng_global.randint(0, n - bs)
            j = rng_global.randint(0, n - bs)
            if abs(i - j) <= bs:
                continue
            amt = min(0.15, 0.5 * bs)
            h_try = best_h.copy()
            h_try[i:i+bs] = np.clip(h_try[i:i+bs] + amt / bs, 0, 1)
            h_try[j:j+bs] = np.clip(h_try[j:j+bs] - amt / bs, 0, 1)
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp += 1
        if imp == 0:
            break
    return best_h, best_c5

def make_pert(n, cool, rng):
    mod = rng.randint(0, 5)
    if mod == 0:
        bs = rng.randint(4, max(5, n // 8))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.05 + 0.15 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 15))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.04 + 0.1 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(4, max(5, n // 8))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.1 + 0.3 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.1 * cool * rng.standard_normal(half)
        p[half:] = -0.1 * cool * rng.standard_normal(n - half)
    return p

rng_global = np.random.RandomState(99)

def run(seed=42, budget_s=1100, **kwargs):
    global rng_global
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    rng_global = np.random.RandomState(seed + 7)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        best_h = np.zeros(n0)
        best_h[:n0 // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent on parent
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.12, 400)

    # Phase 2: Binary + swap local search
    best_h, best_c5 = binary_local_search(best_h, n, t0 + budget_s * 0.18, 40)
    best_h, best_c5 = swap_search(best_h, n, t0 + budget_s * 0.22, 25)

    # Phase 3: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 50
    for it in range(n_bh):
        if time.time() > t0 + budget_s * 0.52:
            break
        cool = 1.0 - 0.85 * (it / n_bh)
        pert = make_pert(n, cool, rng)
        h_pert = project_h(work_h + pert)
        h_c, c5_c = fast_coord_descent(h_pert, n, t0 + budget_s * 0.52, 40)
        temp = 0.0015 * cool + 0.0002
        if c5_c < work_c5 or rng.rand() < np.exp(-(c5_c - work_c5) / max(temp, 1e-9)):
            work_h, work_c5 = h_c, c5_c
            if c5_c < best_c5:
                best_h, best_c5 = h_c.copy(), c5_c
    if work_c5 < best_c5:
        best_h, best_c5 = work_h.copy(), work_c5

    # Phase 4: Block transfer + binary
    best_h, best_c5 = block_transfer_search(best_h, n, t0 + budget_s * 0.62, 25)
    best_h, best_c5 = binary_local_search(best_h, n, t0 + budget_s * 0.68, 30)

    # Phase 5: Try n=120
    if time.time() < t0 + budget_s * 0.80:
        n2 = 120
        h120 = np.interp(np.linspace(0, n - 1, n2), np.arange(n), best_h)
        h120 = project_h(h120)
        h120, c5_120 = fast_coord_descent(h120, n2, t0 + budget_s * 0.80, 200)
        h120, c5_120 = binary_local_search(h120, n2, t0 + budget_s * 0.80, 20)
        if c5_120 < best_c5:
            best_h, best_c5, n = h120, c5_120, n2

    # Phase 6: Try n=200
    n_orig = 160
    if time.time() < t0 + budget_s * 0.90 and n == 160:
        n3 = 200
        h200 = np.interp(np.linspace(0, 159, n3), np.arange(160), best_h)
        h200 = project_h(h200)
        h200, c5_200 = fast_coord_descent(h200, n3, t0 + budget_s * 0.90, 120)
        h200, c5_200 = binary_local_search(h200, n3, t0 + budget_s * 0.90, 15)
        if c5_200 < best_c5:
            best_h, best_c5, n = h200, c5_200, n3

    # Phase 7: Final coordinate descent
    best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.96, 200)

    # Phase 8: Final binary polish
    if time.time() < t0 + budget_s * 0.98:
        best_h, best_c5 = binary_local_search(best_h, n, t0 + budget_s * 0.99, 15)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
