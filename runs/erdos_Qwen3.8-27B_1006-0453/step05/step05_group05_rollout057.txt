from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(h, T, n):
    corr = get_corr(h)
    val = smooth_max_val(corr, T)
    return val * 2.0 / n

def slsqp_opt(h, T, n, maxiter=2000, ftol=1e-15):
    def obj(x):
        return slsqp_obj(x, T, n)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n-1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2*n-1)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n-1)])
    coefs[v2] += (-h[np.clip(vm2[v2], 0, n-1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=100):

    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:

        improved = False
        order = np.random.permutation(n)

For the "bulk" correlations (|k| << n), the first term is approximately n/2 (the total mass of h, minus boundary corrections). So corr[k] ≈ n/2 - Σ_i h[i+k]*h[i] for bulk k.

This means minimizing max_k corr[k] is approximately equivalent to minimizing (n/2 - min_k Σ_i h[i+k]*h[i]), i.e., MAXIMIZING the minimum autocorrelation.

The autocorrelation R[k] = Σ_i h[i+k]*h[i] is related to the power spectrum: R is the inverse DFT of |H|² (circular case). For the full (non-circular) correlation, it's more complex.

**Spectral approach**: If I work in Fourier space, the autocorrelation is the IFFT of the power spectrum. Maximizing the minimum autocorrelation becomes a problem about the power spectrum. But this only works cleanly for circular correlation.

Let me try yet another approach: **What if I use a "water-filling" style algorithm?**

The idea: at each step, identify the k* that maximizes corr[k*], then adjust h to reduce corr[k*] specifically. The subgradient of max_k corr[k] at h is the subgradient of corr[k*] (the active constraint).

The gradient of corr[k] with respect to h[i] can be computed analytically:
∂corr[k]/∂h[i] = (1-h[i+k]) - h[i+k]*(∂h[i+k]/∂h[i])... 

Actually for the full correlation:
corr[m] = Σ_i h[i] * (1 - h[i + m - (n-1)])

∂corr[m]/∂h[j] = (1 - h[j + m - (n-1)]) (if j is the first index)
              + Σ_{i: i + m - (n-1) = j} h[i] * (-1) (if j is the second index)
              = (1 - h[j + m - (n-1)]) - h[j - m + (n-1)]

(with appropriate bounds checking)

So the subgradient of max_m corr[m] at h is the gradient of corr[m*] where m* = argmax.

This gives a concrete direction to move h in the negative subgradient direction.

Let me implement a **subgradient descent** approach:
1. Find m* = argmax corr[m]
2. Compute grad = ∂corr[m*]/∂h
3. Update: h_new = project(h - alpha * grad)
4. Decay alpha

This is simpler than SLSQP and might escape different local minima.

    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

        return smooth_max_val(corr, T) * 2.0 / n

                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=3000, ftol=1e-16):
    def obj(x):
        c = get_corr(x)
        return smooth_max_val(c, T) * 2.0 / n
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def corr_grad_vec(h, n, k):
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] += -h[k:]
    else:
        grad[:n + k] += (1.0 - h[-k:])
        grad[-k:] += -h[:n + k]
    return grad

def subgrad_steps(h, n, n_steps, alpha0, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for t in range(n_steps):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        k_star = int(np.argmax(corr)) - (n - 1)
        # Check for near-ties
        top_idx = np.argsort(corr)[-3:]
        weights = np.exp(100.0 * (corr[top_idx] - corr[top_idx].max()))
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, mi in zip(weights, top_idx):
            ki = mi - (n - 1)
            grad += w * corr_grad_vec(best_h, n, ki)
        alpha = alpha0 / (1.0 + t / (n_steps * 0.3))
        h_new = project_h(best_h - alpha * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5

def coord_opt(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            nv = coord_opt(best_h, n, i)
            if abs(nv - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = nv
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

    c5v = compute_c5(hc)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc

    # Phase 8: Final subgradient + coordinate descent
    if time.time() < t0 + budget_s * 0.94:
        best_h, best_c5 = subgrad_steps(best_h, n, 150, 0.02, t0 + budget_s * 0.94)
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.97, max_sweeps=40)

    # Final polish SLSQP
    if time.time() < t0 + budget_s * 0.99:
        hc = slsqp_opt(best_h, 5000000, n, maxiter=3000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
