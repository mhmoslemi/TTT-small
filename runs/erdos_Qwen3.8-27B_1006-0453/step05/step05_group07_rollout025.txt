            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = proj_h(h_hi, n_try)
            h_hi = lbfgs_step(h_hi, 3000.0, n_try, iters=200)
            hb2 = (h_hi > 0.5).astype(float)
            s2 = hb2.sum()
            tgt2 = n_try * 0.5
            if s2 > tgt2:
                on2 = np.where(hb2 > 0.5)[0]
                hb2[np.random.choice(on2, int(s2-tgt2), replace=False)] = 0.0
            elif s2 < tgt2:
                zz2 = np.where(hb2 < 0.5)[0]
                hb2[np.random.choice(zz2, int(tgt2-s2), replace=False)] = 1.0
            hb2, c5_2 = bin_ls(hb2, n_try, t0 + budget_s * 0.86, max_rounds=3000)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = hb2
                n = n_try
        except Exception:
            pass

    # Phase 6: Final polish
    if elapsed() < budget_s * 0.94:
        best_h = lbfgs_step(best_h, 50000.0, n, iters=500)
        best_c5 = compute_c5(best_h)
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.99, sweeps=30)

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
