    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            h *= (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_f_grid(v_list, corr, lin, center, old, dx):
    d = v_list - old
    mat = corr[:, None] + d[None, :] * lin[:, None]
    mat[center, :] -= d ** 2
    return np.max(mat, axis=0) * dx

def find_opt_v(corr, lin, old, center, dx):
    v1 = np.linspace(0.0, 1.0, 256)
    f1 = eval_f_grid(v1, corr, lin, center, old, dx)
    idx1 = int(np.argmin(f1))
    if idx1 == 0:
        a, b = 0.0, v1[1]
    elif idx1 >= len(v1) - 1:
        a, b = v1[-2], 1.0
    else:
        a, b = v1[idx1 - 1], v1[idx1 + 1]
    v2 = np.linspace(a, b, 64)
    f2 = eval_f_grid(v2, corr, lin, center, old, dx)
    idx2 = int(np.argmin(f2))
    v3_lo, v3_hi = v2[max(0, idx2-1)], v2[min(len(v2)-1, idx2+1)]
    v3 = np.linspace(v3_lo, v3_hi, 64)
    f3 = eval_f_grid(v3, corr, lin, center, old, dx)
    idx3 = int(np.argmin(f3))
    return float(v3[idx3])

def coord_descent(h, n, t_end, max_sweeps=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    center = n - 1
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if sweep % 5 == 4:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = s + n - 1
                mn = n - 1 - s
                if i <= n - 1 - s:
                    lin[mp] += (1.0 - best_h[i + s])
                if i >= s:
                    lin[mp] -= best_h[i - s]
                if i >= s:
                    lin[mn] += (1.0 - best_h[i - s])
                if i <= n - 1 - s:
                    lin[mn] -= best_h[i + s]
            lin[center] = 1.0 - 2.0 * old
            v_new = find_opt_v(corr, lin, old, center, dx)
            d = v_new - old
            if abs(d) < 1e-10:
                continue
            corr += d * lin
            corr[center] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h[i] = v_new
                improved = True
            else:
                corr -= d * lin
                corr[center] += d * d
        if not improved and sweep > 30:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5
