def compute_delta(i, h, n):
    ncorr = 2*n - 1
    delta = np.zeros(ncorr)
    # First term: k in [i, n-1+i], delta[k] += 1-h[n-1-k+i]
    lo1 = max(0, i)
    hi1 = min(ncorr, n + i)  # exclusive upper bound: n-1+i+1 = n+i
    if lo1 < hi1:
        # k ranges lo1 to hi1-1, n-1-k+i ranges n-1-hi1+1+i to n-1-lo1+i
        # = i+n-hi1 to i+n-1-lo1
        h_idx_lo = i + n - hi1
        h_idx_hi = i + n - 1 - lo1
        delta[lo1:hi1] += 1.0 - h[h_idx_lo:h_idx_hi+1][::-1]
    
    # Second term: k in [n-1-i, 2n-2-i], delta[k] -= h[k-n+1+i]
    lo2 = max(0, n - 1 - i)
    hi2 = min(ncorr, 2*n - 1 - i)  # exclusive: 2n-2-i+1 = 2n-1-i
    if lo2 < hi2:
        # k ranges lo2 to hi2-1, k-n+1+i ranges lo2-n+1+i to hi2-1-n+1+i
        h_idx_lo = lo2 - n + 1 + i
        h_idx_hi = hi2 - 1 - n + 1 + i
        delta[lo2:hi2] -= h[h_idx_lo:h_idx_hi+1]
    
    return delta

            work_h, work_c5 = h_new, c5_new
        if work_c5 < best_c5:
            best_h, best_c5 = work_h.copy(), work_c5

    # Phase 5: Final pair swaps (5%)
    t5 = t0 + budget_s * 0.93
    best_h, best_c5 = pair_swap_improve(best_h, n, t5)

    # Phase 6: Final coordinate descent (7%)
    t6 = t0 + budget_s * 0.995
    best_h, best_c5 = fast_coord_descent(best_h, n, t6)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
