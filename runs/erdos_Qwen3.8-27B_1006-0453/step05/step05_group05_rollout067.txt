    # So grad[j] = (1-h[j-m])*1_{0≤j-m<n} - h[j+m]*1_{0≤j+m<n}
    # For m=0: grad[j] = (1-h[j]) - h[j] = 1-2*h[j]
    
    # Build gradient matrix
    A = np.zeros((len(active_lags), n))
    for idx, pos in enumerate(active_lags):
        m = pos - (n-1)
        for j in range(n):
            g = 0.0
            if 0 <= j-m < n:
                g += (1.0 - h[j-m])
            if 0 <= j+m < n:
                g -= h[j+m]
            A[idx, j] = g
    
    # QP: min t + lam*||dh||² s.t. A@dh <= t - corr[active] + c5_target, 0<=h+dh<=1, sum(h+dh)=n/2
    # Simplify: min lam*||dh||² + sum of soft constraints
    # Actually, let me use a simpler approach: 
    # min sum_k max(0, (corr_k + A_k·dh) - t) + lam*||dh||²
    # where t = c5_val (current max)
    # This is a convex QP (sum of hinge losses + quadratic)
    
    # For simplicity, let's just do projected gradient:
    grad = A.T @ np.ones(len(active_lags)) / len(active_lags)  # average gradient of active
    # Wait, that's not right. The subgradient of max(f_k) is the average of the gradients of all active k's.
    # Actually, for max, the subgradient at a point where k1 and k2 are both active is any convex combination of grad_k1 and grad_k2.
    # Using the "all active" subgradient (equal weight):
    subgrad = A.sum(axis=0) / len(active_lags)
    
    # Line search
    best_alpha = 0.1
    best_val = c5_val
    for alpha in np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.5]):
        h_try = project_h(h - alpha * subgrad)
        val = compute_c5(h_try)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    return project_h(h - best_alpha * subgrad), best_val

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_coef(h, n, i):
    coef = np.zeros(2 * n - 1)
    m_min = max(-(n - 1), i - n + 1)
    m_max = min(n - 1, i)
    if m_min <= m_max:
        idx = np.arange(m_min, m_max + 1)
        coef[n - 1 + idx] += (1.0 - h[i - idx])
    m_min2 = max(-(n - 1), -i)
    m_max2 = min(n - 1, n - 1 - i)
    if m_min2 <= m_max2:
        idx2 = np.arange(m_min2, m_max2 + 1)
        coef[n - 1 + idx2] -= h[i + idx2]
    return coef

def ternary_c5(corr_base, coef, n, lo=0.0, hi=1.0, iters=35):
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = corr_base + m1 * coef
        v1[n - 1] -= m1 * m1
        f1 = v1.max()
        v2 = corr_base + m2 * coef
        v2[n - 1] -= m2 * m2
        f2 = v2.max()
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=500, noise_every=50, noise_scale=0.04):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = get_corr(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            vf = best_h[i]
            best_h[i] = 0.0
            coef_i = compute_coef(best_h, n, i)
            corr_base = corr - vf * coef_i
            corr_base[n - 1] += vf * vf
            v_new = ternary_c5(corr_base, coef_i, n)
            if abs(v_new - vf) > 1e-11:
                corr += (v_new - vf) * coef_i
                corr[n - 1] -= (v_new * v_new - vf * vf)
                best_h[i] = v_new
                improved = True
            else:
                best_h[i] = vf
        best_h = project_h(best_h)
        corr = get_corr(best_h)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new
        if noise_every > 0 and (sweep + 1) % noise_every == 0 and not improved:
            noise = noise_scale * (1.0 - 0.5 * sweep / max(sweep, 1)) * np.random.randn(n)
            h_noisy = project_h(best_h + noise)
            c5_try = compute_c5(h_noisy)
            if c5_try < best_c5 + 0.003:
                best_h = h_noisy
                corr = get_corr(best_h)
                best_c5 = compute_c5(best_h)
        if not improved and sweep > 15:
            break
    return best_h, best_c5

def pair_descent(h, n, t_end, n_pairs=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = get_corr(best_h)
    for _ in range(n_pairs):
        if time.time() > t_end:
            break
        i, j = np.random.choice(n, size=2, replace=False)
        vi, vj = best_h[i], best_h[j]
        best_h[i] = 0.0
        best_h[j] = 0.0
        coef_i = compute_coef(best_h, n, i)
        coef_j = compute_coef(best_h, n, j)
        corr_base = corr - vi * coef_i - vj * coef_j
        corr_base[n - 1] += vi * vi + vj * vj
        best_vi, best_vj, best_f = vi, vj, np.inf
        for a in range(9):
            xa = a / 8.0
            tmp = corr_base + xa * coef_i
            tmp[n - 1] -= xa * xa
            for b in range(9):
                xb = b / 8.0
                tmp2 = tmp + xb * coef_j
                tmp2[n - 1] -= xb * xb
                f = tmp2.max()
                if f < best_f:
                    best_f, best_vi, best_vj = f, xa, xb
        lo_i, hi_i = max(0.0, best_vi - 0.25), min(1.0, best_vi + 0.25)
        lo_j, hi_j = max(0.0, best_vj - 0.25), min(1.0, best_vj + 0.25)
        for _ in range(15):
            mi1 = lo_i + (hi_i - lo_i) / 3
            mi2 = hi_i - (hi_i - lo_i) / 3
            mj1 = lo_j + (hi_j - lo_j) / 3
            mj2 = hi_j - (hi_j - lo_j) / 3
            vals = np.empty(4)
            for kk, (mi, mj) in enumerate([(mi1, mj1), (mi2, mj2), (mi1, mj2), (mi2, mj1)]):
                tmp = corr_base + mi * coef_i + mj * coef_j
                tmp[n - 1] -= mi * mi + mj * mj
                vals[kk] = tmp.max()
            if vals[0] <= vals[1] and vals[0] <= vals[2]:
                hi_i, hi_j = mi2, mj2
            elif vals[1] <= vals[0] and vals[1] <= vals[3]:
                lo_i, lo_j = mi1, mj1
            elif vals[2] <= vals[1] and vals[2] <= vals[3]:
                hi_i, lo_j = mi2, mj1
            else:
                lo_i, hi_j = mi1, mj2
        best_h[i] = (lo_i + hi_i) / 2.0
        best_h[j] = (lo_j + hi_j) / 2.0
        best_h = project_h(best_h)
        corr = get_corr(best_h)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new
    return best_h, best_c5

def subgrad_step(h, n, alpha_init=0.02):
    corr = get_corr(h)
    mx = corr.max()
    active = np.where(corr >= mx * 0.92)[0]
    A = np.zeros((len(active), n))
    for idx, pos in enumerate(active):
        m = pos - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j - m < n:
                g += (1.0 - h[j - m])
            if 0 <= j + m < n:
                g -= h[j + m]
            A[idx, j] = g
    subgrad = A.sum(axis=0) / len(active)
    best_h, best_val = h.copy(), compute_c5(h)
    for alpha in [alpha_init * 0.1, alpha_init * 0.3, alpha_init, alpha_init * 3]:
        h_try = project_h(h - alpha * subgrad)
        val = compute_c5(h_try)
        if val < best_val:
            best_val, best_h = val, h_try
    return best_h, best_val

def init_sinusoidal(n, freqs, amps, phases):
    i = np.arange(n)
    h = np.full(n, 0.5)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(2 * np.pi * f * i / n + p)
    return project_h(h)
