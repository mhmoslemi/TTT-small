def soft_obj(x, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def soft_grad(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        j2 = np.arange(max(0, n - 1 - k), min(n, 2 * n - 1 - k))
        idx2 = j2 + k - n + 1
        g[j2] += w[k] * (1.0 - x[idx2])
        j1 = np.arange(max(0, k - n + 1), min(n, k + 1))
        idx1 = j1 - k + n - 1
        g[j1] -= w[k] * x[idx1]
    return g * (2.0 / n)

        r = minimize(h, soft_obj, soft_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, args=(T,),
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

        best_h = lbfgs_opt(h_bin, 3000.0, n, iters=300)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.42, sweeps=40)

    # Phase 4: Basin hopping for global exploration
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.60, n_iters=70)

    # Phase 5: Extended coordinate descent polish
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.72, sweeps=100)

    # Phase 6: Second basin hopping from different perturbation
    if time.time() - t0 < budget_s * 0.78:
        h_alt = best_h + 0.05 * np.random.randn(n)
        h_alt = proj_h(h_alt, n)
        h_alt, c5_alt = basin_hop(h_alt, n, t0 + budget_s * 0.78, n_iters=40)
        if c5_alt < best_c5:
            best_h, best_c5 = h_alt, c5_alt
            best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.82, sweeps=40)

    # Phase 7: Try alternative resolutions
    for n_alt in [120, 200]:
        if time.time() - t0 > budget_s * 0.90:
            break
        try:
            x_old = np.linspace(0, 2, best_h.shape[0], endpoint=False) + 1.0 / best_h.shape[0]
            x_new = np.linspace(0, 2, n_alt, endpoint=False) + 1.0 / n_alt
            h_alt = np.interp(x_new, x_old, best_h)
            h_alt = proj_h(h_alt, n_alt)
            h_alt = lbfgs_opt(h_alt, 2000.0, n_alt, iters=250)
            h_alt, c5_alt = coord_desc(h_alt, n_alt, t0 + budget_s * 0.94, sweeps=40)
            if c5_alt < best_c5:
                best_h, best_c5 = h_alt, c5_alt
        except Exception:
            pass

    # Phase 8: Final high-precision polish
    if time.time() - t0 < budget_s * 0.95:
        best_h = lbfgs_opt(best_h, 50000.0, len(best_h), iters=600)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.995:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t0 + budget_s * 0.999, sweeps=60)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
