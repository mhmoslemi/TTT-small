def fourier_optimize(h, M, t_end, gamma=1000):
    n = len(h)
    # Compute current Fourier coefficients
    # h[i] ≈ 1/2 + sum_{m=1}^{M} (a_m cos(2πmi/n) + b_m sin(2πmi/n))
    # Use DFT
    H = np.fft.rfft(h)
    # DC component: H[0] = sum(h) = n/2
    # We fix h_mean = 1/2, so we optimize the AC components
    
    # Build the basis
    freqs = np.arange(1, M+1)
    i_idx = np.arange(n)
    cos_basis = np.column_stack([np.cos(2*np.pi*f*i_idx/n) for f in freqs])
    sin_basis = np.column_stack([np.sin(2*np.pi*f*i_idx/n) for f in freqs])
    
    # Current coefficients
    # h - 1/2 = cos_basis @ a + sin_basis @ b
    rhs = h - 0.5
    # Least squares (basis is orthonormal up to scaling)
    A = np.hstack([cos_basis, sin_basis])
    params0, _, _, _ = np.linalg.lstsq(A, rhs, rcond=None)
    
    def obj(params):
        h_trial = 0.5 + cos_basis @ params[:M] + sin_basis @ params[M:]
        h_trial = np.clip(h_trial, 0.0, 1.0)
        # Penalize constraint violation
        pen = gamma * (np.sum((np.maximum(0, -h_trial))**2) + np.sum((np.maximum(0, h_trial-1))**2))
        # Also penalize constraint violation after projection
        h_proj = project_h(h_trial)
        if not np.allclose(h_proj, h_trial, atol=1e-6):
            pen += gamma * np.sum((h_proj - h_trial)**2)
            h_trial = h_proj
        c5 = compute_c5(h_trial)
        return c5 + pen
    
    # Optimize
    res = minimize(obj, params0, method='Nelder-Mead', 
                   options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12})
    h_new = 0.5 + cos_basis @ res.x[:M] + sin_basis @ res.x[M:]
    h_new = project_h(h_new)
    return h_new

            c5_new = compute_c5(ph)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = ph
                improved = True
        if not improved:
            break
    return best_h, best_c5

Wait, golden section search with 40 iterations gives precision (hi-lo) * φ^40 ≈ 1 * 0.618^40 ≈ 10^(-10). That's good enough.

But each iteration of golden search evaluates f1d which is O(n) (max over 2n-1 elements). So 40 * O(n) = 40*320 = 12800 ops per coordinate.
Per sweep: 160 * (2*correlate + 40*max) = 160 * (2*160² + 40*320) = 160 * (51200 + 12800) = 160 * 64000 = 10.2M ops.
Actually, np.correlate for n=160 is implemented in C and takes maybe 1-2 μs. And the max over 319 elements is also fast. So per coordinate: ~10μs. Per sweep: 1.6ms. 2000 sweeps: 3.2 seconds. That's very fast!

But wait, I also need project_h and compute_c5 after each accepted update. compute_c5 is one np.correlate: ~2μs. project_h is more complex but still fast for n=160.

Actually, I realize the golden section search might not be the best choice here. The function f1d(x) = max_j (c0[j] + x*d[j] + (x-x²)δ_{j,n1}) is the max of linear/quadratic functions, so it's convex in x. Golden section works for convex functions. But I could also find the optimal x analytically:
- The max is achieved at some lag j*
- For the active lag j*, the function is c0[j*] + x*d[j*] + (if j*=n-1: x(1-x))
- The minimum of max over x occurs where the top few lags are equal

This is a 1D minimax problem that can be solved more efficiently, but golden section with 40 iterations is fine for our purposes.

Let me also add: after each coordinate descent sweep, I can check if any lags have become much smaller than the max, and if so, I can "free up" optimization capacity there.

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.clip(1.0 - h, 0, None)
            tot = room.sum()
            if tot < 1e-13:
                break
            h += room * (diff / tot)
        else:
            h[h > 0] *= (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_coord_descent(h, t_end, max_sweeps=5000):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n1 = n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break

        for i in range(n):
            saved = best_h[i]
            best_h[i] = 0.0
            c0 = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_h[i] = 1.0
            c1 = np.correlate(best_h, 1.0 - best_h, mode='full')
            d = c1 - c0
            # Golden section on [0,1]
            lo, hi = 0.0, 1.0
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            v1 = c0 + x1 * d
            v1[n1] += x1 * (1.0 - x1)
            f1 = v1.max()
            v2 = c0 + x2 * d
            v2[n1] += x2 * (1.0 - x2)
            f2 = v2.max()
            for _ in range(35):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    v1 = c0 + x1 * d
                    v1[n1] += x1 * (1.0 - x1)
                    f1 = v1.max()
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    v2 = c0 + x2 * d
                    v2[n1] += x2 * (1.0 - x2)
                    f2 = v2.max()

            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = ph
                improved = True
                iters_cd += 1
        if not improved:
            break
    return best_h, best_c5

def soft_max_obj(h, T, lam):
    """Log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    Tc = T * (corr - mc)
    Tc -= Tc.max()
    exps = np.exp(Tc)
    sm = mc + np.log(exps.sum()) / T
    return dx * sm + lam * (h.sum() - n / 2.0) ** 2

def soft_max_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    Tc = T * (corr - mc)
    Tc -= Tc.max()
    exps = np.exp(Tc)
    w = exps / exps.sum()
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    grad = np.zeros(n)
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return grad

def lbfgs_opt(h0, T, lam, maxiter, t_deadline):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x):
        return soft_max_obj(x, T, lam)
    def grad(x):
        return soft_max_grad(x, T, lam)
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': 10 * maxiter})
        if time.time() > t_deadline:
            return x0
        return project_h(res.x)
    except Exception:
        return x0
