def project_h(h, n):
    target = n / 2.0

        h = h * (1.0 + diff / s)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def fast_coord_opt(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    lags = np.arange(-(n - 1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    def f(t):
        return np.max(corr_base + t * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 50})
    return res.x

def targeted_subgrad(h, n, k_top=6, alpha=0.02):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[-k_top:]
    grad = np.zeros(n)
    for ki in top:
        lag = ki - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:n])
            grad[lag:] += (-h[:n - lag])
        else:
            l = -lag
            grad[l:] += (1.0 - h[:n - l])
            grad[:n - l] += (-h[l:])
    return project_h(h - alpha * grad, n)

def fourier_perturb(h, n, amplitude, rng):
    n_modes = min(4, n // 4)
    modes = rng.choice(np.arange(1, n // 2), size=n_modes, replace=False)
    phases = rng.uniform(0, 2 * np.pi, size=n_modes)
    amps = rng.uniform(0.4, 1.0, size=n_modes) * amplitude
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for m, ph, a in zip(modes, phases, amps):
        pert += a * np.cos(2 * np.pi * m * x / n + ph)
    return project_h(h + pert, n)

def coord_descent_smart(h, n, t_end, max_sweeps=400):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stale = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-10:]
        scores = np.zeros(n)
        for ki in top_idx:
            lag = ki - (n - 1)
            if lag >= 0:
                scores[:n - lag] += best_h[:n - lag] * (1 - best_h[lag:n])
            else:
                scores[-lag:] += best_h[-lag:] * (1 - best_h[:-lag])
        order = np.argsort(-scores)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = fast_coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            stale += 1
            if stale > 4:
                break
        else:
            stale = 0
    return best_h, best_c5

def pair_swap_opt(h, n, t_end, n_tries=2000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    top_idx = np.argsort(corr)[-6:]
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        ki = top_idx[np.random.randint(len(top_idx))]
        lag = ki - (n - 1)
        if lag >= 0:
            i = np.random.randint(0, n - lag)
            j = i + lag
        else:
            j = np.random.randint(0, n + lag)
            i = j - lag
        h_try = best_h.copy()
        sw = h_try[i]
        h_try[i] = h_try[j]
        h_try[j] = sw
        h_try = project_h(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    # Phase 1: Initial polish + diverse starts (10%)
    t1 = t0 + budget_s * 0.10

    for _ in range(4):

        h_alt = fourier_perturb(best_h, n, 0.2, rng)

        if c5v < best_c5:

    # Phase 2: SLSQP annealing (15%)
    t2 = t0 + budget_s * 0.25
    T_list = [30000, 12000, 5000, 2000, 800, 300, 120, 50, 20, 8]
    for T in T_list:
        if time.time() > t2:

        hc = slsqp_smooth(best_h, T, n, maxiter=350)
        c5v = compute_c5(hc)

    # Phase 3: Targeted subgradient + SLSQP (20%)
    t3 = t0 + budget_s * 0.45
    for it in range(35):
        if time.time() > t3:
            break
        cool = 1.0 - it / 35.0
        alpha = 0.004 + 0.02 * cool
        k_top = 3 + int(5 * cool)

    # Phase 4: Basin hopping with Fourier + subgrad perturbations (25%)

    work_c5 = best_c5

            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:

    if work_c5 < best_c5:

    # Phase 5: Coordinate descent (15%)
