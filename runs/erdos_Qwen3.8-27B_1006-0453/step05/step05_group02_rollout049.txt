from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > 1e-12:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -40.0, 40.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_step(h, T, n, maxiter=1000, ftol=1e-15):

        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def pair_transfer(h, n, rng, n_pairs=50):
    """Try pair transfers: h[i]+=d, h[j]-=d"""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    improved = 0
    for _ in range(n_pairs):
        i, j = rng.choice(n, 2, replace=False)
        if best_h[i] < 0.01 or best_h[j] < 0.01:
            continue
        d = 0.05 * rng.uniform(-1, 1)
        if best_h[i] + d < 0 or best_h[i] + d > 1:
            continue
        if best_h[j] - d < 0 or best_h[j] - d > 1:
            continue
        h_try = best_h.copy()
        h_try[i] += d
        h_try[j] -= d
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
            improved += 1
    return best_h, best_c5, improved

def block_move(h, n, rng, cool=1.0):

    p = np.zeros(n)
    block = p[st:st+bs].copy()
    shift = rng.randint(-3, 4)
    if shift != 0:

        if new_st < 0 or new_st + bs > n:
            return p
        p[new_st:new_st+bs] = h[st:st+bs] * (0.03 + 0.08*cool)
        p[st:st+bs] = -h[st:st+bs] * (0.03 + 0.08*cool)
    else:
        p[st:st+bs] = (0.03 + 0.08*cool) * rng.standard_normal(bs)
    return p

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > 1e-13:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -40.0, 40.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def slsqp_step(h, T, n, maxiter=1000, ftol=1e-15):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_vector(h, top_idx, n):
    """Compute average gradient over top_idx lags."""
    g = np.zeros(n)
    nm1 = n - 1
    for k in top_idx:
        ell = k - nm1
        # (1-h[p-ell]) term: p-ell in [0,n)
        a = max(0, ell)
        b = min(n - 1, n - 1 + ell)
        if a <= b:
            idx_src = np.arange(a, b + 1) - ell
            g[a:b + 1] += 1.0 - h[idx_src]
        # -h[p+ell] term: p+ell in [0,n)
        c = max(0, -ell)
        d = min(n - 1, n - 1 - ell)
        if c <= d:
            idx_src2 = np.arange(c, d + 1) + ell
            g[c:d + 1] -= h[idx_src2]
    g /= len(top_idx)
    return g

def coord_focused(h, n, i, top_k=12):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    kr = np.arange(-(n - 1), n)
    j1 = i + kr
    j2 = i - kr
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    sub_c = corr_base[top_idx]
    sub_co = coefs[top_idx]
    dx = 2.0 / n

        T_bh = 3000 + 12000 * cool
        hc = slsqp_step(h_pert, T_bh, n, maxiter=700, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: Final coordinate descent + binarize
    fin1 = t0 + budget_s * 0.92
    for sweep in range(150):
        if time.time() > fin1:
            break
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > fin1:
                break
            hi_new = coord_focused(best_h, n, i, top_k=20)
            if abs(hi_new - best_h[i]) > 1e-13:
                ht = best_h.copy()
                ht[i] = hi_new
                ht = project_h(ht)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-15:
                    best_c5 = c5t
                    best_h = ht
                    imp = True
        if not imp and sweep > 30:
            break
    best_h, c5b = binarize_push(best_h, n, rng, 400)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = best_h
    if time.time() < t0 + budget_s * 0.95:
        for sweep in range(60):
            if time.time() > t0 + budget_s * 0.95:
                break
            imp = False
            for i in rng.permutation(n):
                if time.time() > t0 + budget_s * 0.95:
                    break
                hi_new = coord_focused(best_h, n, i, top_k=25)
                if abs(hi_new - best_h[i]) > 1e-13:
                    ht = best_h.copy()
                    ht[i] = hi_new
                    ht = project_h(ht)
                    c5t = compute_c5(ht)
                    if c5t < best_c5 - 1e-15:
                        best_c5 = c5t
                        best_h = ht
                        imp = True
            if not imp:
                break

    # Final SLSQP + pair transfer
    for T in [500000, 3000000]:
        if elapsed() > budget_s * 0.97:
            break
        hc = slsqp_step(best_h, T, n, maxiter=2000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if elapsed() < budget_s * 0.99:
        best_h, c5t = pair_transfer(best_h, n, rng, 300)
        if c5t < best_c5:
            best_c5 = c5t

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
