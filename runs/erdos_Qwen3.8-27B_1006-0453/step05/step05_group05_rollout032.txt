import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, T):
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=2000, ftol=1e-15):
    def obj(x):
        return smooth_max(get_corr(x), T) * (2.0 / n)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_step(h, t_end, alpha_init=0.002, n_iter=200, top_k=5):
    """Subgradient descent on max of correlations"""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = alpha_init
    for it in range(n_iter):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        m = np.max(corr)
        # Active set: top top_k shifts
        idx = np.argsort(corr)[-top_k:]
        # Compute subgradient
        sg = np.zeros(n)
        for k in idx:
            shift = k - (n - 1)  # convert from correlate index to actual shift
            for i in range(n):
                # d corr[shift] / d h[i] = (1-h[i+shift]) - h[i-shift]
                ip = i + shift
                iq = i - shift
                if 0 <= ip < n:
                    sg[i] += (1.0 - best_h[ip])
                if 0 <= iq < n:
                    sg[i] -= best_h[iq]
        sg /= top_k
        # Projected step
        h_new = best_h - alpha * sg
        h_new = project_h(h_new)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new
            alpha *= 1.02  # increase step
        else:
            alpha *= 0.95  # decrease step
        if alpha < 1e-6:
            break
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = get_corr(h_base)
            m_vals = np.arange(-(n-1), n)
            vm1 = i - m_vals
            vm2 = i + m_vals
            v1 = (vm1 >= 0) & (vm1 < n)
            v2 = (vm2 >= 0) & (vm2 < n)
            coefs = np.zeros(2*n-1)
            coefs[v1] += (1.0 - best_h[np.clip(vm1[v1], 0, n-1)])
            coefs[v2] += (-best_h[np.clip(vm2[v2], 0, n-1)])
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi-lo)/3.0
                m2 = hi - (hi-lo)/3.0
                if np.max(corr_base + m1*coefs) < np.max(corr_base + m2*coefs):
                    hi = m2
                else:
                    lo = m1
            hi_new = (lo + hi) / 2.0
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

    best_c5 = compute_c5(best_h)
    alpha = alpha_init
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        m = np.max(corr)
        idx = np.argsort(corr)[-top_k:]
        sg = np.zeros(n)
        for k_out in idx:
            s = k_out - (n - 1)
            if s >= 0:
                sg[:n - s] += (1.0 - best_h[s:n])
                sg[s:n] -= best_h[0:n - s]
            else:
                ss = -s
                sg[ss:n] += (1.0 - best_h[0:n - ss])
                sg[0:n - ss] -= best_h[ss:n]
        sg /= top_k
        h_new = project_h(best_h - alpha * sg)
        c5v = compute_c5(h_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_new
            alpha *= 1.03
        else:
            alpha *= 0.92
        if alpha < 1e-7:
            break
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=50, use_pairs=True):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = get_corr(h_base)
            m_vals = np.arange(-(n - 1), n)
            vm1 = i - m_vals
            vm2 = i + m_vals
            v1 = (vm1 >= 0) & (vm1 < n)
            v2 = (vm2 >= 0) & (vm2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - best_h[np.clip(vm1[v1], 0, n - 1)])
            coefs[v2] += (-best_h[np.clip(vm2[v2], 0, n - 1)])
            lo, hi = 0.0, 1.0
            for _ in range(38):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
                    hi = m2
                else:
                    lo = m1
            hi_new = (lo + hi) / 2.0
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if use_pairs and sweep % 2 == 1 and time.time() < t_end:
            for _ in range(n // 6):
                if time.time() > t_end:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                h_base = best_h.copy()
                h_base[i] = 0.0
                h_base[j] = 0.0
                corr_base = get_corr(h_base)
                m_vals = np.arange(-(n - 1), n)
                ci = np.zeros(2 * n - 1)
                cj = np.zeros(2 * n - 1)
                for idx_c, arr in [(i, ci), (j, cj)]:
                    vm1 = idx_c - m_vals
                    vm2 = idx_c + m_vals
                    v1 = (vm1 >= 0) & (vm1 < n)
                    v2 = (vm2 >= 0) & (vm2 < n)
                    arr[v1] += (1.0 - best_h[np.clip(vm1[v1], 0, n - 1)])
                    arr[v2] += (-best_h[np.clip(vm2[v2], 0, n - 1)])
                best_val = np.max(corr_base)
                bi, bj = best_h[i], best_h[j]
                for gi in np.linspace(0, 1, 7):
                    row = corr_base + gi * ci
                    for gj in np.linspace(0, 1, 7):
                        vv = np.max(row + gj * cj)
                        if vv < best_val:
                            best_val = vv
                            bi, bj = gi, gj
                h_try = best_h.copy()
                h_try[i] = bi
                h_try[j] = bj
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h160 = np.array(initial_h_values, dtype=float)
    else:
        h160 = np.zeros(160)
        h160[:80] = 1.0

    best_h = h160.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    # Phase 1: SLSQP annealing on n=160
    for T in [40, 400, 4000, 40000, 150000]:
        if elapsed() > budget_s * 0.10:
            break
        hc = slsqp_opt(best_h, T, best_n, maxiter=2500, ftol=1e-16)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5 = c
            best_h = hc

    # Subgradient refinement
    if elapsed() < budget_s * 0.18:
        best_h, best_c5 = subgrad_opt(best_h, t0 + budget_s * 0.18, alpha_init=0.001, top_k=5, max_iter=500)

    # Coordinate descent n=160
    if elapsed() < budget_s * 0.33:
        best_h, best_c5 = coord_descent(best_h, best_n, t0 + budget_s * 0.33, max_sweeps=30, use_pairs=True)

    # Phase 2: Upsample to n=200 and optimize
    if elapsed() < budget_s * 0.35:
        n2 = 200
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n2)
        h200 = np.interp(x_new, x_old, best_h)
        h200 = project_h(h200)
        c200 = compute_c5(h200)
        for T in [80, 800, 8000, 80000]:
            if elapsed() > budget_s * 0.50:
                break
            hc = slsqp_opt(h200, T, n2, maxiter=2000, ftol=1e-15)
            c = compute_c5(hc)
            if c < c200:
                c200 = c
                h200 = hc
        if elapsed() < budget_s * 0.57:
            h200, c200 = subgrad_opt(h200, t0 + budget_s * 0.57, alpha_init=0.0008, top_k=6, max_iter=300)
        if elapsed() < budget_s * 0.68:
            h200, c200 = coord_descent(h200, n2, t0 + budget_s * 0.68, max_sweeps=20, use_pairs=True)
        if c200 < best_c5:
            best_c5 = c200
            best_h = h200
            best_n = n2

    # Phase 3: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    wn = best_n
    n_bh = 130
    for it in range(n_bh):
        if elapsed() > budget_s * 0.87:
            break
        cool = max(0.0, 1.0 - it / n_bh)
        mod = it % 7
        if mod == 0:
            pert = (0.015 + 0.06 * cool) * np.random.randn(wn)
        elif mod == 1:
            freq = np.random.randint(1, 18)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.015 + 0.05 * cool) * np.sin(2 * np.pi * freq * np.arange(wn) / wn + ph)
        elif mod == 2:
            mask = np.random.random(wn) < 0.12
            pert = np.zeros(wn)
            pert[mask] = (0.03 + 0.1 * cool) * np.random.randn(mask.sum())
        elif mod == 3:
            blk = max(1, wn // 12)
            b = np.random.randint(0, max(1, wn // blk))
            pert = np.zeros(wn)
            pert[b*blk:min(wn, (b+1)*blk)] = (0.02 + 0.04 * cool) * np.random.randn(blk)
        elif mod == 4:
            pert = np.zeros(wn)
            a, b2 = np.random.permutation(wn)[:2]
            d = (work_h[b2] - work_h[a]) * (0.3 + 0.7 * cool)
            pert[a] += d
            pert[b2] -= d
        elif mod == 5:
            f1 = np.random.randint(1, 10)
            f2 = np.random.randint(10, 25)
            pert = (0.01 + 0.03 * cool) * (np.sin(2*np.pi*f1*np.arange(wn)/wn) +
                    np.sin(2*np.pi*f2*np.arange(wn)/wn + 0.5))
        else:
            mask = np.random.random(wn) < 0.06
            pert = np.zeros(wn)
            pert[mask] = (0.05 + 0.12 * cool) * np.sign(np.random.randn(mask.sum()))
        h_p = project_h(work_h + pert)
        T_bh = max(50, 300 + 25000 * cool)
        hc = slsqp_opt(h_p, T_bh, wn, maxiter=400, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0003 * cool + 0.00002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Final polish
    if elapsed() < budget_s * 0.93:
        hc = slsqp_opt(best_h, 400000, best_n, maxiter=4000, ftol=1e-16)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5 = c
            best_h = hc
    if elapsed() < budget_s * 0.96:
        best_h, best_c5 = subgrad_opt(best_h, t0 + budget_s * 0.96, alpha_init=0.0005, top_k=4, max_iter=200)
    if elapsed() < budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, best_n, t0 + budget_s * 0.99, max_sweeps=15, use_pairs=False)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
