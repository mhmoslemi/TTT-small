    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        # Add/subtract proportionally to available room
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * ((target - s) / tot)
        else:
            if s < 1e-14:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * ((target - s) / tot)
        else:
            if s < 1e-14:
                break
            h *= (target / s)
            h = np.clip(h, 0.0, 1.0)
    return h

    v = base + lin * t

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s < target:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * ((target - s) / tot)
        else:
            if s < 1e-14:
                break
            h *= (target / s)
            h = np.clip(h, 0.0, 1.0)
    return h

def _f1d(xv, old, base, lin, n):
    t = xv - old
    v = base + lin * t
    v[n - 1] -= t * t
    return float(np.max(v))

def fast_coord_desc(h, t_end, max_sweeps=2000, seed=0):
    n = len(h)
    m = 2 * n - 1
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx = 2.0 / n
    base = np.correlate(h, 1.0 - h, mode='full').copy()
    lg_arr = np.arange(1, n)
    idx_p = n - 1 + lg_arr
    idx_m = n - 1 - lg_arr
    best_c5 = np.max(base) * dx
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng.permutation(n):
            old = h[i]
            lin = np.zeros(m)
            lin[n - 1] = 1.0 - 2.0 * old
            vp1 = (i + lg_arr < n)
            vp2 = (i - lg_arr >= 0)
            cp = np.zeros(n - 1)
            cm = np.zeros(n - 1)
            if vp1.any():
                vals = h[i + lg_arr[vp1]]
                cp[vp1] = 1.0 - vals
                cm[vp1] = -vals
            if vp2.any():
                vals2 = h[i - lg_arr[vp2]]
                cp[vp2] -= vals2
                cm[vp2] = 1.0 - vals2
            lin[idx_p] = cp
            lin[idx_m] = cm
            res = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                  args=(old, base, lin, n), options={'xatol': 1e-13})
            new_x = res.x
            if abs(new_x - old) < 1e-14:
                continue
            dt = new_x - old
            base += lin * dt
            base[n - 1] -= dt * dt
            h[i] = new_x
            new_c5 = np.max(base) * dx
            if new_c5 < best_c5 - 1e-16:
                best_c5 = new_c5
                improved = True
            else:
                base -= lin * dt
                base[n - 1] += dt * dt
                h[i] = old
        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from initial (45% budget)
    p1_end = t0 + budget_s * 0.45
    best_h, best_c5 = fast_coord_desc(best_h, p1_end, max_sweeps=5000, seed=seed)

    # Phase 2: Multi-restart (30% budget)
    p2_end = t0 + budget_s * 0.75
    candidates = []
    h_a = np.zeros(n); h_a[::2] = 1.0
    candidates.append(project_h(h_a))
    for bs in [3, 5, 7, 10, 13]:
        h_b = np.zeros(n)
        for blk in range(0, n, 2 * bs):
            h_b[blk:blk + bs] = 1.0
        candidates.append(project_h(h_b))
    for fr in [1, 2, 3, 5, 7, 12]:
        h_s = np.clip(0.5 + 0.49 * np.sin(2 * np.pi * fr * np.arange(n) / n), 0, 1)
        candidates.append(project_h(h_s))
    for _ in range(4):
        candidates.append(project_h(np.random.beta(2, 2, n).copy()))
    for sc in [0.04, 0.1, 0.2, 0.35]:
        candidates.append(project_h(best_h + sc * np.random.randn(n)))

    for ci, h_init in enumerate(candidates):
        if time.time() > p2_end:
            break
        hc, c5v = fast_coord_desc(h_init, p2_end, max_sweeps=300, seed=seed + 100 + ci)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc.copy()

    # Phase 3: Basin hopping (18% budget)
    p3_end = t0 + budget_s * 0.93
    work_h = best_h.copy()
    work_c5 = best_c5
    n_hops = 50
    for it in range(n_hops):
        remaining = p3_end - time.time()
        if remaining < 2.0:
            break
        t_hop_end = time.time() + max(1.0, remaining / (n_hops - it))
        cool = max(0.1, 1.0 - it / float(n_hops))
        sc = 0.02 + 0.09 * cool
        h_p = project_h(work_h + sc * np.random.randn(n))
        hc, c5v = fast_coord_desc(h_p, t_hop_end, max_sweeps=150, seed=seed + 500 + it)
        temp = 0.0004 * cool + 0.00008
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc.copy()
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = work_h.copy()

    # Phase 4: Final polish (7% budget)
    p4_end = t0 + budget_s * 0.99
    best_h, best_c5 = fast_coord_desc(best_h, p4_end, max_sweeps=2000, seed=seed + 999)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
