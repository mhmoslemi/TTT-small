            lin = make_lin(best_h, i, n)

            def f1d(v):
                d = v - old
                v2 = np.max(corr + d * lin)
                return v2

            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13, 'maxiter': 120})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-14:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved and sweep >= 3:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=500, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    arange = np.arange(n, dtype=float)

    for hop in range(n_hops):
        if time.time() > t_end:
            break
        progress = min(1.0, hop / max(n_hops, 1))
        cool = 1.0 - progress
        step = 0.004 + 0.09 * cool * cool
        mode = hop % 7

        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(2, 20)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * arange / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = step * 3.0 * rng.randn(mask.sum())
        elif mode == 3:
            idx = rng.choice(n, size=max(4, n // 6), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * rng.randn(len(idx))
        elif mode == 4:
            blk = max(1, n // 4)
            b = rng.randint(0, 4)
            pert = np.zeros(n)
            sl = slice(min(b * blk, n), min((b + 1) * blk, n))
            pert[sl] = step * 0.6 * rng.randn(blk)
        elif mode == 5:
            i1 = rng.randint(0, n)
            i2 = rng.randint(0, n)
            pert = np.zeros(n)
            pert[i1] = step * 4.0 * rng.randn()
            pert[i2] = step * 4.0 * rng.randn()
        else:
            idx = rng.choice(n, size=rng.randint(2, 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * 5.0 * rng.randn(len(idx))

        h_pert = project_h(cur_h + pert, n)
        remaining = t_end - time.time()
        per_hop = max(0.5, min(4.0, remaining * 0.08))
        hop_end = min(t_end, time.time() + per_hop)
        n_sw = max(8, int(per_hop * 8))
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=n_sw, seed=seed + hop * 7)

        temp = 0.002 * cool + 0.00015
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Initial CD
    t1 = t0 + budget_s * 0.12
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=400, seed=seed)

    # Phase 2: First basin hopping (broad exploration)
    t2 = t0 + budget_s * 0.50
    best_h, best_c5 = basin_hop(best_h, n, t2, n_hops=600, seed=seed + 10)

    # Phase 3: CD polish
    t3 = t0 + budget_s * 0.62
    best_h, best_c5 = coord_descent(best_h, n, t3, max_sweeps=400, seed=seed + 100)

    # Phase 4: Second basin hopping (finer, from best)
    t4 = t0 + budget_s * 0.85
    h2, c5_2 = basin_hop(best_h, n, t4, n_hops=400, seed=seed + 500)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 5: Final long CD
    t5 = t0 + budget_s * 0.97
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=800, seed=seed + 200)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
