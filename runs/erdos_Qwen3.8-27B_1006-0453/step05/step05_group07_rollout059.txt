    n = len(h)

    return float(corr.max() * 2.0 / n)

def proj_h(h, n):
    # Projection onto {0<=h<=1, sum(h)=n/2}
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    # Water-filling projection
    if s > tgt:
        # Need to reduce: scale down non-1 values
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        excess = s - tgt
        reducible = (h[mask] - 0.0).sum()  # can reduce to 0
        if reducible >= excess:
            h[mask] -= excess / mask.sum()
        else:
            h[mask] = 0.0
            remaining = excess - reducible
            mask2 = h > 0.0
            if mask2.sum() > 0:
                h[mask2] -= remaining / mask2.sum()
    else:
        # Need to increase: scale up non-0 values
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        deficit = tgt - s
        increasable = (1.0 - h[mask]).sum()
        if increasable >= deficit:
            h[mask] += deficit / mask.sum()
        else:
            h[mask] = 1.0
            remaining = deficit - increasable
            mask2 = h < 1.0
            if mask2.sum() > 0:
                h[mask2] += remaining / mask2.sum()
    h = np.clip(h, 0.0, 1.0)
    # Fix rounding
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * 2.0 / n

def _sm_grad(x, T, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    w = np.exp(T * (corr - m))
    w /= w.sum()
    mf = 2 * n - 1
    k_arr = np.arange(mf)
    grad = np.zeros(n)
    for i in range(n):
        j1 = i - k_arr + n - 1
        j2 = i + k_arr - n + 1
        m1 = (j1 >= 0) & (j1 < n)
        m2 = (j2 >= 0) & (j2 < n)
        t1 = (w[m1] * (1.0 - x[j1[m1]])).sum() if m1.any() else 0.0
        t2 = (w[m2] * x[j2[m2]]).sum() if m2.any() else 0.0
        grad[i] = (2.0 / n) * (t1 - t2)
    return grad

def lbfgs_run(h, T, n, iters=300):
    try:
        r = minimize(h, _sm_obj, _sm_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, args=(T, n),
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_descent(h, n, t_end, sweeps=300):
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    mf = 2 * n - 1
    k_arr = np.arange(mf)
    for sweep in range(sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            old = best[i]
            j1 = i - k_arr + n - 1
            j2 = i + k_arr + 1 - n
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coef = np.zeros(mf)
            if v1.any():
                coef[v1] += 1.0 - best[j1[v1]]
            if v2.any():
                coef[v2] -= best[j2[v2]]
            a = corr - old * coef
            lo, hi2 = 0.0, 1.0
            for _ in range(50):
                mid = 0.5 * (lo + hi2)
                vals = a + coef * mid
                mx = vals.max()
                act = vals >= mx - 1e-13
                ba = coef[act]
                if ba.min() <= 0.0 and ba.max() >= 0.0:
                    lo = hi2 = mid
                    break
                elif ba.min() > 0.0:
                    hi2 = mid
                else:
                    lo = mid
            nv = 0.5 * (lo + hi2)
            if abs(nv - old) < 1e-13:
                continue
            best[i] = nv
            corr += (nv - old) * coef
            improved = True
        best = proj_h(best, n)
        c5t = compute_c5(best)
        if c5t < best_c5:
            best_c5 = c5t
        else:
            if not improved:
                break
    return best, best_c5

def subgrad_mom(h, n, t_end, lr_init=0.002, beta=0.7, iters=50000):
    best = h.copy()
    best_c5 = compute_c5(best)
    v = np.zeros(n)
    lr = lr_init
    dx = 2.0 / n
    mf = 2 * n - 1
    k_arr = np.arange(mf)
    for it in range(iters):
        if time.time() > t_end:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        mx = corr.max()
        active = (corr >= mx - 1e-10)
        w = active.astype(float)
        w /= w.sum()
        grad = np.zeros(n)
        for i in range(n):
            j1 = i - k_arr + n - 1
            j2 = i + k_arr + 1 - n
            m1 = (j1 >= 0) & (j1 < n)
            m2 = (j2 >= 0) & (j2 < n)
            t1 = (w[m1] * (1.0 - best[j1[m1]])).sum() if m1.any() else 0.0
            t2 = (w[m2] * best[j2[m2]]).sum() if m2.any() else 0.0
            grad[i] = (2.0 / n) * (t1 - t2)
        v = beta * v - lr * grad
        trial = proj_h(best + v, n)
        c5t = compute_c5(trial)
        if c5t < best_c5:
            best = trial
            best_c5 = c5t
        else:
            lr *= 0.95
            v *= 0.5
        if it % 500 == 0:
            lr = max(lr_init * 0.3, lr * 1.05)
    return best, best_c5

def basin_hop(h, n, t_end, n_iters=150):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mode = it % 4
        if mode == 0:
            amp = 0.03 + 0.12 * cool
            pert = amp * np.random.randn(n)
        elif mode == 1:
            f = np.random.randint(1, 16)
            ph = np.random.uniform(0, 2 * np.pi)
            amp = 0.03 + 0.10 * cool
            pert = amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = np.random.random(n) < (0.25 + 0.35 * cool)
            pert = np.zeros(n)
            n_mask = max(mask.sum(), 1)
            pert[mask] = (0.04 + 0.18 * cool) * np.random.randn(n_mask)
        else:
            nb = (work > 0.5).astype(float)
            ones = np.where(nb == 1)[0]
            zeros = np.where(nb == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                swap_n = min(len(ones), len(zeros), max(2, n // 20))
                si = np.random.choice(ones, swap_n, replace=False)
                sj = np.random.choice(zeros, swap_n, replace=False)
                pert = np.zeros(n)
                pert[si] = -work[si]
                pert[sj] = 1.0 - work[sj]
            else:
                pert = 0.05 * np.random.randn(n)
        hp = proj_h(work + pert, n)
        T_val = 500.0 + 15000.0 * cool
        ht = lbfgs_run(hp, T_val, n, iters=200)
        ht, _ = coord_descent(ht, n, min(time.time() + 5, t_end), sweeps=20)
        c5t = compute_c5(ht)
        temp = 0.001 * cool + 0.00003
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B
    for T in [100, 1000, 5000]:
        if time.time() - t0 > budget_s * 0.05:
            break
        h_new = lbfgs_run(best_h, T, n, iters=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.15, sweeps=300)

    # Phase 3: Subgradient with momentum
    best_h, best_c5 = subgrad_mom(best_h, n, t0 + budget_s * 0.28, iters=30000)

    # Phase 4: More CD
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.38, sweeps=400)

    # Phase 5: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.65, n_iters=180)

    # Phase 6: Try other n values
    for n_try in [128, 200, 256]:
        if time.time() - t0 > budget_s * 0.85:
            break
        try:
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_try = np.interp(x_new, x_old, best_h)
            h_try = proj_h(h_try, n_try)
            h_try = lbfgs_run(h_try, 3000.0, n_try, iters=200)
            h_try, c5_try = coord_descent(h_try, n_try, t0 + budget_s * 0.88, sweeps=100)
            if c5_try < best_c5:
                best_h, best_c5, n = h_try, c5_try, n_try
        except Exception:
            pass

    # Phase 7: Final L-BFGS-B + CD
    if time.time() - t0 < budget_s * 0.92:
        best_h = lbfgs_run(best_h, 50000.0, n, iters=500)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.985, sweeps=200)

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
