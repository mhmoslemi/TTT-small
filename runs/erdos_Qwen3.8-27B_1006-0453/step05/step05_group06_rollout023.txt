import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        frac = diff / tot
        h = h + room * frac
    else:
        if s < 1e-14:
            return h
        floor = h
        tot_f = floor.sum()
        if tot_f < 1e-14:
            return h
        frac = 1.0 + diff / s
        h = h * frac
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, corr, eta, eps_frac=0.01):
    """Take a subgradient step on the max correlation."""
    n = len(h)
    m = len(corr)  # 2n-1
    max_c = np.max(corr)
    # Critical set: lags within eps_frac of max
    critical = np.where(corr >= max_c - eps_frac * max_c)[0]
    # Compute subgradient (average over critical set)
    grad = np.zeros(n)
    lag_arr = np.arange(m) - n + 1
    for idx in critical:
        lg = lag_arr[idx]
        if lg >= 0:
            if lg < n:
                grad[lg:] += (1.0 - h[:n-lg])
                grad[:n-lg] += -h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += (1.0 - h[p:])
                grad[p:] += -h[:n-p]
    grad /= len(critical)
    return grad

def subgradient_method(h, t_end, max_iters=50000):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    work_h = h.copy()
    work_c5 = best_c5
    eta = 0.05 / n  # initial step size
    no_improve = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        c5, corr = compute_c5(work_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = work_h.copy()
            no_improve = 0
        else:
            no_improve += 1
        grad = subgradient_step(work_h, corr, eta)
        work_h = work_h - eta * grad
        work_h = project_h(work_h)
        # Adaptive step size
        if no_improve > 200:
            eta *= 0.9
            no_improve = 0
        if eta < 1e-10:
            break
    return best_h, best_c5

def coord_descent_fast(h, t_end, max_sweeps=200):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            h0 = best_h.copy()
            # Compute baseline corr with h[i]=0
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            # Linear and quadratic terms for h[i]
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0 and lg < n:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                elif lg < 0 and -lg < n:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            
            def f1d(x):
                return np.max(cb + x * lin + x * x * quad)
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-8})
            if abs(res.x - best_h[i]) > 1e-10:
                ht = best_h.copy()
                ht[i] = res.x
                ht = project_h(ht)
                c5t, _ = compute_c5(ht)
                if c5t < best_c5 - 1e-15:
                    best_c5 = c5t
                    best_h = ht
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def softmax_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(Tc)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lag_arr[idx]
        wi = w[idx]
        if wi < 1e-15:
            continue
        if lg >= 0 and lg < n:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        elif lg < 0 and -lg < n:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter):
    x0 = project_h(h0)
    try:
        res = minimize(lambda x: softmax_obj_grad(x, T, lam)[0], x0,
                       jac=lambda x: softmax_obj_grad(x, T, lam)[1],
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * len(x0),
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_h(res.x)
    except:
        return x0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (diff / tot)
        else:
            if s < 1e-14:
                break
            h *= (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_at_lag(h, lag):
    """Gradient of corr[lag] w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0 and lag < n:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] += -h[lag:]
    elif lag < 0 and -lag < n:
        p = -lag
        g[:n-p] += (1.0 - h[p:])
        g[p:] += -h[:n-p]
    return g

def exact_subgrad(h, corr):
    """Compute average subgradient over near-max lags."""
    n = len(h)
    mc = np.max(corr)
    eps = 0.05 * mc + 1e-10
    active = np.where(corr >= mc - eps)[0]
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    g = np.zeros(n)
    for idx in active:
        g += grad_corr_at_lag(h, lag_arr[idx])
    g /= len(active)
    return g

def softmax_obj(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= Tc.max()
    ex = np.exp(Tc)
    sm = mc + np.log(ex.sum()) / T
    return dx * sm + lam * (h.sum() - n/2.0)**2

def softmax_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= Tc.max()
    ex = np.exp(Tc)
    w = ex / ex.sum()
    g = np.zeros(n)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-12:
            continue
        g += wi * grad_corr_at_lag(h, lag_arr[idx])
    return dx * g + 2.0 * lam * (h.sum() - n/2.0)

                     options={'maxiter':maxiter,'ftol':1e-16,'gtol':1e-10})

Let me also think about the **Frank-Wolfe (conditional gradient)** method:
1. At h, compute the gradient (or subgradient) g

4. Line search for α
