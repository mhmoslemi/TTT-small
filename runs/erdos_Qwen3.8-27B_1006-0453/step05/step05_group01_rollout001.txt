    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=500, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * dx

        return project_h(np.clip(res.x, 0, 1))
    except Exception:

            if no_improve >= 3:

    return best_h, best_c5

def targeted_subgrad(h, n, t_end, n_iters=600):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    v = np.zeros(n)
   -beta_unused = 0
    beta = 0.55
    for it in range(n_iters):
        if time.time() > t_end:
            break
        corr = full_corr(best_h)
        topk = min(8 + int(25 * (1 - it / n_iters)), 33)
        top_idx = np.argsort(corr)[-topk:]
        shifts = top_idx - (n - 1)
        i_idx = np.arange(n)[:, None]
        s_idx = shifts[None, :]
        jp = i_idx + s_idx
        jm = i_idx - s_idx
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n - 1)
        jmc = np.clip(jm, 0, n - 1)
        grad = ((1.0 - best_h[jpc]) * v1.astype(np.float64)
                - best_h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
        v = beta * v + (1.0 - beta) * grad
        cool = 1.0 - it / n_iters
        alpha = 0.002 + 0.008 * cool
        h_new = project_h(best_h - alpha * v)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter=35, scale=0.08):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.35:
            corr = full_corr(best_h)
            top_idx = np.argsort(corr)[-6:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n - k] += (best_h[k:n] - 0.5) * 0.4
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (best_h[:n + k] - 0.5) * 0.4
            pert *= scale * cool
        elif r < 0.55:
            f = np.random.randint(2, 18)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.5 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.75:
            blen = np.random.randint(max(n // 12, 2), max(n // 4, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * cool * np.random.randn(blen)
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(best_h + pert)
        T = 3000 * cool + 1000
        h_c = slsqp_smooth(h_pert, T, n, 250, 1e-14)
        c5_v = compute_c5(h_c)
        temp = 0.0006 * cool + 0.0002
        if c5_v < best_c5 or (c5_v > best_c5 and np.random.rand() < np.exp(-(c5_v - best_c5) / temp)):
            best_h = h_c
            best_c5 = c5_v
    return best_h, best_c5

def optimize_n(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # Phase 1: SLSQP warmup with increasing T
    if t_end - time.time() > 40:
        for T in [20, 200, 2000, 20000, 200000, 2000000]:
            if t_end - time.time() < 20:
                break
            hc = slsqp_smooth(best_h, T, n, 400, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Phase 2: Targeted subgradient
    if t_end - time.time() > 60:
        t_sub = t_end - (t_end - time.time()) * 0.25
        best_h, best_c5 = targeted_subgrad(best_h, n, t_sub, 500)
    # Phase 3: Basin hopping
    if t_end - time.time() > 80:
        t_bh = t_end - (t_end - time.time()) * 0.30
        best_h, best_c5 = basin_hop(best_h, n, t_bh, 30, 0.09)
    # Phase 4: Coordinate descent (main workhorse)
    if t_end - time.time() > 30:
        t_cd1 = t_end - 15
        best_h, best_c5 = coord_descent(best_h, n, t_cd1, 200, 45)
    # Phase 5: Final SLSQP + coord
    if t_end - time.time() > 12:
        hc = slsqp_smooth(best_h, 5000000, n, 500, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if t_end - time.time() > 6:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, 80, 50)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Main optimization at parent resolution (use ~80% of time)
    t_main_end = t0 + budget_s * 0.80
    best_h, best_c5 = optimize_n(best_h, n, t_main_end)

    # second attempt: slightly perturbed restart
    if t_end - time.time() > budget_s * 0.12:
        h2 = project_h(best_h + 0.04 * np.random.randn(n))
        h2_opt, c5_2 = optimize_n(h2, n, t_end - 3)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
