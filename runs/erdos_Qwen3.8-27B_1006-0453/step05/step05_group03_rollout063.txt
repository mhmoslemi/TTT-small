    best_c5 = compute_c5(best_h)

    weights = np.ones(n_corr)

        if t_end and time.time() > t_end:
            break

        T_w = 1.0 / (corr.max() * 2.0)  # adaptive temperature

            break
        cool = max(0.1, 1.0 - it / max(n_iter, 1))
        amp = base_amp * (0.4 + 0.6 * cool)
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        T = 40 + 2000 * cool
        hc = slsqp_smooth(h_pert, T, n, maxiter=250)
        c5_v = compute_c5(hc)
        temp = 0.00015 * cool + 0.00002
        if c5_v < work_c5 or (np.random.rand() < np.exp(-(c5_v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5_v
            if c5_v < best_c5:
                best_c5 = c5_v
                best_h = hc.copy()
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=150):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    nc = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        order = np.random.permutation(n)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(nc)
            for k in range(nc):
                if k <= n - 1:
                    lag = k
                    a = 0.0
                    if i <= n - 1 - lag:
                        a += (1.0 - best_h[np.minimum(i + lag, n-1)])
                    if lag <= i:
                        a += (-best_h[np.maximum(i - lag, 0)])
                    coefs[k] = a
                else:
                    lag = k - (n - 1)
                    a = 0.0
                    if lag <= i:
                        a += (1.0 - best_h[np.maximum(i - lag, 0)])
                    if i <= n - 1 - lag:
                        a += (-best_h[np.minimum(i + lag, n-1)])
                    coefs[k] = a
            hi_old = best_h[i]
            base = corr - hi_old * coefs
            def f(hi):
                return np.max(base + hi * coefs)
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                               options={'xatol': 1e-11, 'maxiter': 40})
            hi_new = float(res.x)
            if abs(hi_new - hi_old) > 1e-10:
                best_h[i] = hi_new
                best_h = project_h(best_h, n)
                c5_new = compute_c5(best_h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n0 = 160
        best_h = (0.5 + 0.45 * np.cos(np.linspace(0, 2 * np.pi, n0, endpoint=False))).copy()
        best_h = project_h(best_h, n0)
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP annealing with analytical gradient
    for T in [15000, 5000, 2000, 700, 250, 80, 30]:
        if elapsed() > budget_s * 0.12:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=350)
        c5_v = compute_c5(hc)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = hc

    # Phase 2: Iterative reweighting
    best_h, best_c5 = reweight_loop(best_h, n, t0 + budget_s * 0.30, n_iter=25, T_start=3.0, T_end=12.0)

    # Phase 3: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.52, n_iter=35, base_amp=0.035)

    # Phase 4: More reweighting (sharper)
    best_h, best_c5 = reweight_loop(best_h, n, t0 + budget_s * 0.68, n_iter=20, T_start=6.0, T_end=20.0)

    # Phase 5: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.84, max_sweeps=100)

    # Phase 6: Final smooth max polish
    for T in [60, 15, 5]:
        if elapsed() > budget_s * 0.93:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=450)
        c5_v = compute_c5(hc)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = hc

    # Phase 7: Final coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=60)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
