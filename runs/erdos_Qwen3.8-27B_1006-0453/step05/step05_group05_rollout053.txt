import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    dx = 2.0 / n
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def directed_perturb(h, n, scale=0.05):
    """Perturb h in the direction that reduces the current max correlation."""
    corr = get_corr(h)
    k_star = np.argmax(corr)  # index in full correlation
    # Convert to shift
    shift = k_star - (n - 1)  # actual shift value
    # Gradient of corr[shift] w.r.t. h
    m_vals = np.arange(-(n-1), n)
    # grad[i] = d corr[shift] / d h[i]
    # corr[shift] = sum_i h[i] * (1-h[i+shift])
    # d/dh[j] = (1-h[j+shift]) - h[j-shift] (with bounds)
    grad = np.zeros(n)
    for j in range(n):
        jp = j + shift
        jm = j - shift
        if jp >= 0 and jp < n:
            grad[j] += (1.0 - h[jp])

            grad[j] -= h[jm]
    # Move in direction -grad (to reduce corr[k*])
    # But we need to maintain sum constraint, so center the perturbation
    grad_centered = grad - grad.mean()
    pert = -scale * grad_centered
    return pert

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    return val * (2.0 / n)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_ternary(h, n, i, niter=55):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(vm1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(vm2[valid2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_opt_pair(h, n, i, j, niter=28):
    h_base = h.copy()
    h_base[i] = 0.0
    h_base[j] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    ci = np.zeros(2*n-1); cj = np.zeros(2*n-1)
    for ii, cc in [(i, ci), (j, cj)]:
        vm1 = ii - m_vals; vm2 = ii + m_vals
        v1 = (vm1 >= 0) & (vm1 < n); v2 = (vm2 >= 0) & (vm2 < n)
        cc[v1] += (1.0 - h[np.clip(vm1[v1], 0, n-1)])
        cc[v2] += (-h[np.clip(vm2[v2], 0, n-1)])
    lo_i, hi_i, lo_j, hi_j = 0.0, 1.0, 0.0, 1.0
    for _ in range(niter):
        mi1 = lo_i + (hi_i-lo_i)/3.0; mi2 = hi_i - (hi_i-lo_i)/3.0
        mj1 = lo_j + (hi_j-lo_j)/3.0; mj2 = hi_j - (hi_j-lo_j)/3.0
        vals = [np.max(corr_base + a*ci + b*cj) for a, b in
                [(mi1,mj1),(mi2,mj2),(mi1,mj2),(mi2,mj1)]]
        bx = int(np.argmin(vals))
        if bx == 0: hi_i, hi_j = mi2, mj2
        elif bx == 1: lo_i, lo_j = mi1, mj1
        elif bx == 2: hi_i, lo_j = mi2, mj1
        else: lo_i, hi_j = mi1, mj2
    return (lo_i+hi_i)/2.0, (lo_j+hi_j)/2.0

def directed_perturb(h, n, scale):
    corr = get_corr(h)
    top_k = np.argsort(corr)[-3:]
    grad = np.zeros(n)
    for kk in top_k:
        shift = kk - (n - 1)
        for j in range(n):
            if 0 <= j + shift < n:
                grad[j] += (1.0 - h[j + shift])
            if 0 <= j - shift < n:
                grad[j] -= h[j - shift]
    pert = -scale * (grad - grad.mean())
    return pert

        hc = slsqp_opt(best_h, T, n, maxiter=1000, ftol=1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent with pairs (14%)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.24,
                                     max_sweeps=40, pair_frac=0.45, triple_frac=0.12)

    # Phase 3: Basin hopping (44%)
    work_h = best_h.copy()
    work_c5 = best_c5
    t_bhop_end = t0 + budget_s * 0.68
    it = 0
    while time.time() < t_bhop_end:
        it += 1
        cool = max(0.02, 1.0 - 0.97 * it / 200.0)
        pert = make_perturbation(work_h, n, cool, it)
        h_pert = project_h(work_h + pert)
        T_bh = 300 + 20000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=500, ftol=1e-13)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.00003
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
        # Periodic coord polish during bhop
        if it % 12 == 0 and time.time() < t_bhop_end:
            work_h, work_c5 = coord_descent(work_h, n, time.time() + budget_s*0.01,
                                             max_sweeps=3, pair_frac=0.2, triple_frac=0)
            if work_c5 < best_c5:
                best_c5 = work_c5
                best_h = work_h.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: High-T SLSQP refinement (7%)
    for T in [200000, 800000, 2000000]:
        if time.time() > t0 + budget_s * 0.75:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Final intensive coordinate descent (17%)
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.93,
                                     max_sweeps=80, pair_frac=0.55, triple_frac=0.15)

    # Phase 6: Final polish (7%)
    if time.time() < t0 + budget_s * 0.96:
        hc = slsqp_opt(best_h, 3000000, n, maxiter=3000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99,
                                     max_sweeps=15, pair_frac=0.4, triple_frac=0)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
