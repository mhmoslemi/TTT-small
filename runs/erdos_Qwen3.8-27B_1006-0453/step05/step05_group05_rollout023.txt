import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_vals = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False

        for i in range(n):
            slope = np.zeros(2 * n - 1)
            ip = i + m_vals
            im = i - m_vals
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            slope[vp] += (1.0 - best_h[ip[vp]])
            slope[vm] -= best_h[im[vm]]

            corr0 = corr - best_h[i] * slope

            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = (lo + hi) * 0.5
                vals = corr0 + slope * mid
                mv = np.max(vals)
                active = vals >= mv - 1e-13
                smin = np.min(slope[active])
                smax = np.max(slope[active])
                if smin > 1e-13:
                    hi = mid
                elif smax < -1e-13:
                    lo = mid
                else:
                    lo = hi = mid
                    break

            new_v = (lo + hi) * 0.5
            if abs(new_v - best_h[i]) < 1e-14:
                continue
            delta = new_v - best_h[i]
            corr += delta * slope
            best_h[i] = new_v
            improved = True

            best_h = project_h(best_h)
            c5 = compute_c5(best_h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
        else:
            break
