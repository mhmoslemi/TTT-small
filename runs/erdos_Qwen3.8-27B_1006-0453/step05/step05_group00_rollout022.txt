import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    """Convert to binary with exactly N/2 ones."""
    h = h.copy()
    nh = N // 2
    n1 = int(np.sum(h > 0.5))
    if n1 == nh:
        return (h > 0.5).astype(float)
    # Convert to continuous first, then threshold with sorting
    vals = h.copy()
    vals[vals > 1] = 1
    vals[vals < 0] = 0
    # Sort and pick top nh
    sorted_idx = np.argsort(-vals)
    out = np.zeros(N)
    out[sorted_idx[:nh]] = 1.0
    return out

def fast_binary_opt(h, N, deadline, T0=3.0, T1=0.01, chunk=30):

    h = make_binary(h, N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    corr = fft_corr(h)
    best_c = float(corr.max())
    best_h = h.copy()
    T = T0
    total_est = int((deadline - time.time()) * 20)  # estimate ~20 iters/sec
    it = 0
    stagnation = 0
    min_T = 0.01
    
    while time.time() < deadline - 0.5:
        it += 1
        T = max(T1, T0 * (T1/T0) ** (it / max(total_est, 1)))
        
        # Compute D matrix: D[i,m] = (1-h)[i+N-1-m] if valid, else 1
        m_idx = np.arange(2*N - 1)
        ref = np.arange(N)[:, None] + N - 1 - m_idx[None, :]  # (N, 2N-1)
        valid = (ref >= 0) & (ref < N)
        ref_c = np.clip(ref, 0, N-1)
        D = np.where(valid, 1.0 - h[ref_c], 1.0)  # (N, 2N-1)
        
        # Evaluate swaps
        cur_max = float(corr.max())
        base = corr[None, :] - D[ones, :]  # (n1, 2N-1)
        
        best_new = cur_max + 1e6
        best_pi = -1
        best_qi = -1
        
        for start in range(0, len(ones), chunk):
            end = min(start + chunk, len(ones))
            # (chunk, n0, 2N-1)
            vals = base[start:end, :, None] + D[zeros][None, :, :]
            maxes = vals.max(axis=2)  # (chunk, n0)
            local_best = float(maxes.min())
            if local_best < best_new:
                li, lj = np.unravel_index(int(np.argmin(maxes)), maxes.shape)
                best_new = local_best
                best_pi = start + li
                best_qi = lj
        
        if best_pi < 0:
            break

        q = zeros[best_qi]

def nesterov_adam(h, N, deadline, k=50, lr=0.01):
    """Adam optimizer for continuous relaxation."""
    h = h.astype(float).copy()
    target = N / 2.0
    m = np.zeros(N)
    v = np.zeros(N)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    
    while time.time() < deadline:
        t += 1
        y = h.copy()
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        
        # Gradient of max_k corr[k] w.r.t. h
        # corr[m] = sum_i h[i] * (1-h)[i+N-1-m]
        # d(corr[m])/d(h[j]) = (1-h)[j+N-1-m] - h[j] * delta(j+N-1-m still needs care)
        # Actually: corr[m] = sum_i h[i]*(1-h[i+N-1-m]) for valid i+N-1-m
        # d/dh[j]: term where i=j: (1-h[j+N-1-m]) + terms where i+N-1-m=j: -h[i]
        
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                # corr[center+L] = sum_{i=0}^{N-1-L} h[i]*(1-h[i+L]) + sum_{i=N-L}^{N-1} h[i]

                grad[:N-L] -= y[L:]
            else:
                M = -L
                grad[:N-M] += (1.0 - y[M:])
                grad[M:] -= y[:N-M]
        grad /= k
        
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        h_new = project_sum(h - lr_t * m / (np.sqrt(v) + eps), target)
        h = h_new
    
    return best_h, best_pk * (2.0 / N)

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, N):
    nh = N // 2
    h = np.clip(h, 0.0, 1.0)
    sorted_idx = np.argsort(-h, kind='stable')
    out = np.zeros(N)
    out[sorted_idx[:nh]] = 1.0
    return out

def compute_D(h, N):

    ref = np.arange(N)[:, None] + N - 1 - m_idx[None, :]
    valid = (ref >= 0) & (ref < N)
    ref_c = np.clip(ref, 0, N - 1)
    return np.where(valid, 1.0 - h[ref_c], 1.0)

def adam_opt(h, N, deadline, k=60, lr=0.008):

    v_buf = np.zeros(N)

    t = 0
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    center = N - 1

    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += (1.0 - h[:N - L])
                grad[:N - L] -= h[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - h[M:])
                grad[M:] -= h[:N - M]

        v_buf = b2 * v_buf + (1 - b2) * grad**2

    return best_h, best_pk * (2.0 / N)
