```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar, fmin
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        excess = h
        tot = excess.sum()
        if tot < 1e-14:
            return h
        h = h * (1.0 - (s - target) / tot)
    else:
        deficit = 1.0 - h
        tot = deficit.sum()
        if tot < 1e-14:
            return h
        h = h + deficit * ((target - s) / tot)
    return np.clip(h, 0.0, 1.0)

def subgradient_max_corr(h):
    """Compute subgradient of max_k f_k(h) where f_k is correlation at lag k."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_max = np.max(corr)
    # Subgradient of the maximizing correlation
    # For lag index idx (0 to 2n-2), shift s = idx - (n-1)
    m_full = 2 * n - 1
    g = np.zeros(n)
    # Find all lags achieving the max (within tolerance)
    tol = m_max * 1e-10 + 1e-14
    active = np.where(corr >= m_max - tol)[0]
    # Average subgradient over active lags
    count = len(active)
    for idx in active:
        s = idx - (n - 1)  # actual shift
        for j in range(n):
            # Contribution from h[j] as first index
            if 0 <= (j + s) < n:
                g[j] += (1.0 - h[j + s])
            # Contribution from h[j] as second index  
            if 0 <= (j - s) < n:
                g[j] -= h[j - s]
    g /= count
    return m_max, g

def sma_optimize(h0, max_iter=5000, t_end=None):
    """SMA (Subgradient Method with Adaptive step) for min max correlation."""
    n = len(h0)
    dx = 2.0 / n
    h = project_h(h0.copy())
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = 0.1  # initial step size
    
    for it in range(max_iter):
        if t_end is not None and time.time() > t_end:
            break
        f_max, g = subgradient_max_corr(h)
        f_val = f_max * dx
        gnorm2 = np.dot(g, g)
        if gnorm2 < 1e-20:
            break
        eta = min(eta, f_val / gnorm2)
        h_new = project_h(h - eta * g)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
```

A binary signal with the right pattern could have A(s) > n/4 at most lags, giving C5 < 1/2.

The theoretical minimum of C5 is achieved when A(s) is maximized uniformly. For a binary sequence, A(s) can't exceed (n/2)²/n = n/4 on average, so the best we can do for binary is C5 ≈ 1/2.

But for non-binary h, we can do better! The key insight: if h takes values in [0,1] and is NOT binary, then:
- At lag 0: O(0) = n/2 - sum h[i]². To make this small, we want sum h[i]² large (binary-like).
- At other lags: O(s) = (sum h in overlap) - A(s). To make this small, we want A(s) large.

The tension: binary gives large sum h² (good for lag 0) but limited A(s) at other lags. Non-binary gives smaller sum h² (bad for lag 0) but potentially larger A(s) at other lags.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar, fmin
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    target = n / 2.0
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > 1e-14:
            h = h * (target / s)
        else:
            h = np.full(n, target / n)
            break
    return np.clip(h, 0.0, 1.0)

def subgrad_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_max = np.max(corr)
    g = np.zeros(n)
    tol = 1e-6
    active = np.where(corr >= m_max - tol)[0]
    for idx in active:
        s = idx - (n - 1)
        gs = np.zeros(n)
        if s >= 0:
            gs[:n - s] += (1.0 - h[s:])
            gs[s:] -= h[:n - s]
        else:
            p = -s
            gs[p:] += (1.0 - h[:n - p])
            gs[:n - p] -= h[p:]
        g += gs
    g /= len(active)
    return m_max, g

def sma_run(h0, t_end):
    n = len(h0)
    dx = 2.0 / n
    h = project_h(h0.copy())
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = 0.3
    for it in range(8000):
        if time.time() > t_end:
            break
        f_max, g = subgrad_max_corr(h)
        fv = f_max * dx
        g2 = np.dot(g, g)
        if g2 < 1e-18:
            break
        eta = min(eta, max(fv / g2, 1e-6))
        h_new = project_h(h - eta * g)
        c5n = compute_c5(h_new)
        if c5n < best_c5 - 1e-15:
            best_c5 = c5n
            best_h = h_new.copy()
        h = h_new
        if it > 0 and it % 200 == 0:
            eta = min(eta, 0.15)
    return best_h, best_c5

def fourier_h(params, n, M):
    i = np.arange(n)
    h = np.full(n, 0.5)
    for m in range(1, M + 1):
        a = params[2 * (m - 1)]
        b = params[2 * (m - 1) + 1]
        ang = 2.0 * np.pi * m * i / n
        h += a * np.cos(ang) + b * np.sin(ang)
    return project_h(np.clip(h, 0.0, 1.0))

def fourier_obj(params, n, M):
    h = fourier_h(params, n, M)
    return compute_c5(h)

def fourier_phase(h_init, n, M, t_end):
    i = np.arange(n)
    hp = h_init - 0.5
    init_p = np.zeros(2 * M)
    for m in range(1, M + 1):
        ang = 2.0 * np.pi * m * i / n
        init_p[2 * (m - 1)] = 2.0 * np.dot(hp, np.cos(ang)) / n
        init_p[2 * (m - 1) + 1] = 2.0 * np.dot(hp, np.sin(ang)) / n
    best_p = init_p.copy()
    best_c5 = fourier_obj(best_p, n, M)
    best_h = fourier_h(best_p, n, M)
    attempts = 0
    while time.time() < t_end and attempts < 12:
        attempts += 1
        if attempts > 1:
            rng = np.random.RandomState(attempts * 7 + 3)
            pert = 0.05 / attempts
            p_try = best_p + pert * rng.randn(2 * M)
        else:
            p_try = best_p.copy()
        res = fmin(fourier_obj, p_try, args=(n, M), xtol=1e-10, ftol=1e-12, maxiter=300, disp=False)
        c5n = fourier_obj(res, n, M)
        if c5n < best_c5 - 1e-12:
            best_c5 = c5n
            best_p = res.copy()
            best_h = fourier_h(best_p, n, M)
        else:
            break
    return best_h, best_c5

def lbfgs_smooth(h0, T, lam, maxiter, t_end):
    n = len(h0)
    dx = 2.0 / n
    x0 = project_h(h0.copy())

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        Tc = T * (corr - mc)
        Tc -= np.max(Tc)
        ev = np.exp(Tc)
        sm = mc + np.log(ev.sum()) / T
        return dx * sm + lam * (x.sum() - n / 2.0) ** 2

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        Tc = T * (corr - mc)
        Tc -= np.max(Tc)
        ev = np.exp(Tc)
        w = ev / ev.sum()
        grad = np.zeros(n)
        m_full = 2 * n - 1
        lag_arr = np.arange(m_full) - n + 1
        for idx in range(m_full):
            lg = lag_arr[idx]
            wi = w[idx]
            if wi < 1e-14:
                continue
            if lg >= 0:
                if lg < n:
                    grad[lg:] += wi * (1.0 - x[:n - lg])
                grad[:n - lg] += -wi * x[lg:]
            else:
                p = -lg
                grad[:n - p] += wi * (1.0 - x[p:])
                grad[p:] += -wi * x[:n - p]
        grad = dx * grad + 2.0 * lam * (x.sum() - n / 2.0)
        return grad

    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-10})
        if time.time() > t_end:
            return x0
        return project_h(res.x)
    except Exception:
        return x0

def coord_descend(h0, t_end, max_sweeps=200):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - n + 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            h0v = best_h.copy()
            h0v[i] = 0.0
            cb = np.correlate(h0v, 1.0 - h0v, mode='full')
            lin = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 2.0 * best_h[i] - 1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            def f1d(x):
                return np.max(cb + x * lin)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def block_optimize(h0, n, B, t_end):
    block_size = n // B
    if block_size < 2:
        return h0, compute_c5(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(30):
        if time.time() > t_end:
            break
        for b in range(B):
            start = b * block_size
            end = start + block_size if b < B - 1 else n
            if end - start < 2:
                continue
            h_save = best_h.copy()
            h_vals = best_h[start:end].copy()
            def block_obj(p):
                ht = h_save.copy()
                ht[start:end] = np.clip(p, 0.0, 1.0)
                ht = project_h(ht)
                return compute_c5(ht)
            res = fmin(block_obj, h_vals, xtol=1e-8, maxiter=100, disp=False)
            ht = h_save.copy()
            ht[start:end] = np.clip(res, 0.0, 1.0)
            ht = project_h(ht)
            c5n = compute_c5(ht)
            if c5n < best_c5 - 1e-14:
                best_c5 = c5n
                best_h = ht
    return best_h, best_c5
```
