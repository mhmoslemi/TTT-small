import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.real(np.fft.irfft(H * G, L))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_grad(h, N, top_k=50, temp=0.5):
    corr = fft_corr(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    top = corr[idx]
    w = np.exp((top - top.max()) / temp)
    w /= w.sum()
    grad = np.zeros(N)
    center = N - 1
    for j in range(top_k):
        k = int(idx[j] - center)
        wi = w[j]
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            c1 = np.zeros(N)
            c2 = np.zeros(N)
            c1[:N-k] = (1.0 - h[k:])
            c2[k:] = -h[:N-k]
            grad += wi * (c1 + c2)
        else:
            M = -k
            c1 = np.zeros(N)
            c2 = np.zeros(N)
            c1[M:] = (1.0 - h[:N-M])
            c2[:N-M] = -h[M:]
            grad += wi * (c1 + c2)
    return grad, float(corr.max()) * (2.0 / N)

def adam_cont(h, N, deadline, lr=0.08, top_k=50, temp=0.4):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    lr_cur = lr
    temp_cur = temp
    stagnant = 0
    while time.time() < deadline:
        t += 1
        g, c = smooth_max_grad(h, N, top_k, temp_cur)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        m_hat = m / (1.0 - 0.9**t)
        v_hat = v / (1.0 - 0.999**t)
        h = project_sum(h - lr_cur * m_hat / (np.sqrt(v_hat) + 1e-8), target)
        c = compute_c5(h, N)
        if c < best_c - 1e-7:
            best_c = c
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 300:
            h = best_h.copy() + np.random.randn(N) * 0.04
            h = project_sum(h, target)
            m[:] = 0.0
            v[:] = 0.0
            lr_cur = lr
            temp_cur = 0.5
            stagnant = 0
        if t % 300 == 0:
            lr_cur *= 0.94
            temp_cur = max(0.1, temp_cur * 0.88)
    return best_h, best_c

def swap_local_search(h, N, deadline, max_sweeps=300):
    h = (h > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 > nh:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[:n1-nh]] = 0
    elif n1 < nh:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        h[zeros[:nh-n1]] = 1
    best_c = compute_c5(h, N)
    center = N - 1
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = fft_corr(h)
        P = float(corr.max()) * (2.0 / N)
        if P < best_c:
            best_c = P
        # Find top contributing lags
        top_idx = np.argpartition(corr, -30)[-30:]
        top_set = set(top_idx.tolist())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Score each one-position: how much it contributes to top lags
        one_score = np.zeros(len(ones))
        for i, p in enumerate(ones):
            s = 0.0
            for ai in top_idx:
                k = ai - center
                if k > 0:
                    if p + k < N:
                        s += (1.0 - h[p+k]) * corr[ai]
                    if p - k >= 0:
                        s += h[p-k] * corr[ai]
                elif k < 0:
                    M = -k
                    if p >= M:
                        s += (1.0 - h[p-M]) * corr[ai]
                    if p + M < N:
                        s += h[p+M] * corr[ai]
            one_score[i] = s
        # Score each zero-position: how much removing it would help
        zero_score = np.zeros(len(zeros))
        for i, q in enumerate(zeros):
            s = 0.0
            for ai in top_idx:
                k = ai - center
                if k > 0:
                    if q + k < N:
                        s += (1.0 - h[q+k]) * corr[ai]
                    if q - k >= 0:
                        s += h[q-k] * corr[ai]
                elif k < 0:
                    M = -k
                    if q >= M:
                        s += (1.0 - h[q-M]) * corr[ai]
                    if q + M < N:
                        s += h[q+M] * corr[ai]
            zero_score[i] = s
        # Best one to remove (high score = most harmful)
        pi = int(np.argmax(one_score))
        # Best zero to fill (low score = least beneficial to keep as zero)
        qi = int(np.argmin(zero_score))
        p = int(ones[pi])
        q = int(zeros[qi])
        h[p], h[q] = 0.0, 1.0
        c_new = compute_c5(h, N)
        if c_new > best_c + 0.002:
            h[p], h[q] = 1.0, 0.0
    return h, best_c

def upsample_binary(h_old, N_old, N_new):
    x_old = np.linspace(0, 2.0, N_old, endpoint=False) + 1.0 / N_old
    x_new = np.linspace(0, 2.0, N_new, endpoint=False) + 1.0 / N_new
    h_interp = np.interp(x_new, x_old, np.concatenate([h_old, [h_old[0]]]), period=2.0)

        idx = np.where(h_bin == 0)[0]

def downsample_binary(h, N, N_target):
