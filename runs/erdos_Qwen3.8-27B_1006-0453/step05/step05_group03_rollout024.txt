import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -80, 80)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def fast_coord_search(corr, h, i, n, old):
    """Golden section search for best delta to h[i]. Returns new value."""
    lo = max(1e-9, -old + 1e-9)
    hi = min(1 - 1e-9, 1.0 - old - 1e-9)
    if hi - lo < 1e-10:
        return old
    gr = (5.0 ** 0.5 - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    nc_template = corr.copy()
    fc = np.max((nc_template + c * (1.0 - 2.0 * h)).__setitem__(slice(i, i+n), 
                  corr[i:i+n] + c * (1.0 - 2.0 * h)) and 
                nc_template.__setitem__(i, nc_template[i]) and
                (nc_template[2*i] if 2*i < len(nc_template) else 0) and 0)
    # Simpler approach: just evaluate directly
    def eval_delta(delta):
        nc = corr.copy()
        nc[i:i+n] += delta * (1.0 - 2.0 * h)
        if 2 * i < len(nc):
            nc[2 * i] -= delta * delta
        return np.max(nc)
    fc = eval_delta(c)
    fd = eval_delta(d)
    for _ in range(20):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a); fc = eval_delta(c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a); fd = eval_delta(d)
    best_delta = (a + b) / 2.0
    return old + best_delta

def incremental_cd(h, n, t_end, max_sweeps=200):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_max = np.max(corr)
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            old = h[i]
            # Find best new value via golden section
            new_val = fast_coord_search(corr, h, i, n, old)

            delta = new_val - old

            corr[i:i+n] += delta * (1.0 - 2.0 * h)
            if 2 * i < len(corr):
                corr[2 * i] -= delta * delta
            h[i] = new_val
            if np.max(corr) < current_max - 1e-12:
                current_max = np.max(corr)
                improved = True
        h = project_h(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            if sweep > 2:
                break
    h = project_h(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

    h = project_h(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def subgrad_step(h, n, k=6, alpha=0.005):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for ki in top_k:
        lag = ki - (n - 1)
        for i in range(n):
            if 0 <= i + lag < n:
                grad[i] += (1.0 - h[i + lag]) if lag >= 0 else h[i - lag]
            if 0 <= i - lag < n:
                grad[i] += (-h[i - lag]) if lag <= 0 else (1.0 - h[i + lag])
    return project_h(h - alpha * grad, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP to improve starting point
    if time.time() < t0 + budget_s * 0.08:
        for T in [5000, 2000]:
            hc = slsqp_smooth(best_h, T, n, maxiter=200)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 2: Incremental coordinate descent (main workhorse)
    t_phase2 = t0 + budget_s * 0.30
    best_h, best_c5 = incremental_cd(best_h, n, t_phase2, max_sweeps=300)

    # Phase 3: SLSQP refinement
    if time.time() < t0 + budget_s * 0.38:
        for T in [8000, 3000, 1000]:
            hc = slsqp_smooth(best_h, T, n, maxiter=300)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 4: More incremental CD
    t_phase4 = t0 + budget_s * 0.52
    best_h, best_c5 = incremental_cd(best_h, n, t_phase4, max_sweeps=300)

    # Phase 5: Binary push
    if time.time() < t0 + budget_s * 0.62:
        hb, c5b = binary_push_cd(best_h, n, t0 + budget_s * 0.62, max_sweeps=40)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb
        # CD after binary push
        best_h, best_c5 = incremental_cd(best_h, n, t0 + budget_s * 0.68, max_sweeps=100)

    # Phase 6: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    t_phase6 = t0 + budget_s * 0.80
    it = 0
    while time.time() < t_phase6 and it < 30:
        cool = max(0.1, 1.0 - it / 30.0)
        amp = 0.03 + 0.06 * cool
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        T_bh = 300 + 2000 * cool
        hc = slsqp_smooth(h_pert, T_bh, n, maxiter=250)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.00005
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        it += 1
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 7: Final incremental CD
    t_phase7 = t0 + budget_s * 0.92
    best_h, best_c5 = incremental_cd(best_h, n, t_phase7, max_sweeps=200)

    # Phase 8: Final SLSQP + CD
    if time.time() < t0 + budget_s * 0.96:
        hc = slsqp_smooth(best_h, 500, n, maxiter=200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = incremental_cd(best_h, n, t_end - 1.0, max_sweeps=50)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
