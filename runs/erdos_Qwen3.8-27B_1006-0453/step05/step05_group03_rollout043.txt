    n = len(h)

    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_step(h, n, i):
    # Precompute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    old = h[i]
    lags = np.arange(-(n-1), n)
    j1 = i + lags
    j2 = i - lags
    c = np.zeros(2*n-1)
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    if v1.any():
        c[v1] += (1.0 - h[j1[v1]])
    if v2.any():
        c[v2] -= h[j2[v2]]
    # Bisection on subgradient of max_k(corr[k] + t*c[k]), t in [-old, 1-old]
    lo_t, hi_t = -old, 1.0 - old
    if lo_t >= hi_t:
        return old

        bh, bc5 = coord_desc(bh, n, min(time.time() + 4, e2), sweeps=25, rnd=True)

    # Phase 3: basin hopping with perturbation + SLSQP + coord descent
    e3 = t0 + budget_s * 0.83
    wh, wc5 = bh.copy(), bc5
    for it in range(40):
        if time.time() > e3:
            break
        cool = max(0.08, 1.0 - it / 40.0)
        amp = 0.025 + 0.065 * cool
        hp = project_h(wh + amp * np.random.randn(n), n)
        T = 200 + 2500 * cool
        hc = slsqp_opt(hp, T, n, mi=130)
        hc, cc = coord_desc(hc, n, min(time.time() + 3, e3), sweeps=18, rnd=True)
        temp = 0.0012 * cool + 0.00008
        if cc < wc5:
            wh, wc5 = hc, cc
            if cc < bc5:
                bc5, bh = cc, hc
        elif (cc - wc5) < 0.004 and np.random.rand() < np.exp(-(cc - wc5) / temp):
            wh, wc5 = hc, cc
    if wc5 < bc5:
        bc5, bh = wc5, wh

    # Phase 4: final polish
    e4 = t0 + budget_s * 0.97
    bh, bc5 = coord_desc(bh, n, e4, sweeps=150, rnd=True)
    if time.time() < e4:
        for T in [80000, 300000]:
            if time.time() > t0 + budget_s * 0.99:
                break
            hc = slsqp_opt(bh, T, n, mi=350)
            cv = compute_c5(hc)
            if cv < bc5:
                bc5, bh = cv, hc
            bh, bc5 = coord_desc(bh, n, t0 + budget_s * 0.99, sweeps=50, rnd=True)

    bh = project_h(bh, n)
    bc5 = compute_c5(bh)
    return bh, bc5, n
