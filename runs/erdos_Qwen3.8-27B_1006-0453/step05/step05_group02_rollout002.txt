def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, T, n, maxiter=500, ftol=1e-14):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

    h_base[i] = 0.0

        corr_base = corr_cache

    best_c5 = compute_c5(best_h)

            best_h = h_try

One more thought: the `minimize_scalar` approach for the 1D subproblem might not be the most efficient. Since the objective is max of linear functions (convex, piecewise linear), I can find the minimum more directly:

The function f(x) = max_k (a_k + b_k * x) is minimized where the slope changes from negative to positive. I can find this by:
1. Compute which k gives the max at x=0 and x=1
2. If the argmax is the same at both endpoints, the function is linear and the minimum is at one of the endpoints
3. Otherwise, the minimum is at a kink point. I can find it by:
   - At x=0, let k0 = argmax
   - The function is a_k0 + b_k0 * x until it crosses another line
   - Find the crossing point, move there, repeat

But this is O(n²) in the worst case (n lines can cross n/2 times each). For n=319 lags, that's ~100K operations, which is fine.

Actually, a simpler approach for convex piecewise linear: use the fact that the subgradient contains 0 at the minimum. I can binary search:
- At x, the "slope" is the average of b_k for k in argmax (or any subgradient)
- If all active b_k > 0, move left; if all < 0, move right; if 0 is in the range, we're at the minimum

Let me implement this as a fast 1D solver:

```python
def solve_1d_convex_pl(c, d):
    """Minimize max_k(c[k]+d[k]*x) for x in [0,1]. c,d are arrays."""
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo+hi)/2
        vals = c + d*mid
        active = np.where(vals == vals.max())[0]
        if d[active].min() <= 0 <= d[active].max():
            break
        elif d[active].min() > 0:
            hi = mid
        else:
            lo = mid
    return (lo+hi)/2
```

This is O(50 * n) per coordinate instead of O(50 * n) for minimize_scalar, but with less overhead (no scipy function call overhead).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, T, n, maxiter=600, ftol=1e-14):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def solve_1d_convex_pl(c, d):
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = c + d * mid
        vmax = vals.max()
        active = np.where(vals >= vmax - 1e-10)[0]
        dmin = d[active].min()
        dmax = d[active].max()
        if dmin <= 0.0 and dmax >= 0.0:
            break
        elif dmin > 0.0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def coord_descent(h, n, t_end, max_sweeps=200, seed_off=0):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(1000 + seed_off)
    k_range = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            orig = best_h[i]
            best_h[i] = 0.0
            corr0 = np.correlate(best_h, 1.0 - best_h, mode='full')
            j1 = i + k_range
            j2 = i - k_range
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - best_h[np.clip(j1[v1], 0, n - 1)])
            coefs[v2] += (-best_h[np.clip(j2[v2], 0, n - 1)])
            best_h[i] = orig
            new_val = solve_1d_convex_pl(corr0, coefs)
            if abs(new_val - orig) > 1e-11:
                h_try = best_h.copy()
                h_try[i] = new_val
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try
                    imp = True
        if not imp and sweep > 3:
            break
    return best_h, best_c5

def upsample(h, factor=2):
    n = len(h)
    h_new = np.repeat(h, factor)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    # Phase 1: Intensive coordinate descent (multi-order)

    # Phase 2: SLSQP continuation
    for T in [3000, 15000, 80000, 400000]:
        if time.time() > t0 + budget_s * 0.38:

        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: More coordinate descent with different ordering
    t_p3 = t0 + budget_s * 0.52
    best_h, best_c5 = coord_descent(best_h, n160, t_p3, max_sweeps=300, seed_off=77)

    # Phase 4: Upsample to n=320 and optimize
    if time.time() < t0 + budget_s * 0.55:
        h320 = upsample(best_h, 2)
        n320 = len(h320)
        h320 = project_h(h320)
        c5_320 = compute_c5(h320)
        if c5_320 < best_c5 + 0.01:
            t_p4a = t0 + budget_s * 0.72
            h320, c5_320 = coord_descent(h320, n320, t_p4a, max_sweeps=120, seed_off=200)
            if time.time() < t0 + budget_s * 0.78:
                for T in [20000, 200000]:
                    hc320 = slsqp_step(h320, T, n320, maxiter=400, ftol=1e-14)
                    c5v = compute_c5(hc320)
                    if c5v < c5_320:
                        c5_320 = c5v
                        h320 = hc320
            h320, c5_320 = coord_descent(h320, n320, t0 + budget_s * 0.80, max_sweeps=80, seed_off=300)
            if c5_320 < best_c5:
                best_h = h320
                best_c5 = c5_320
                best_n = n320

    # Phase 5: Basin hopping with focused perturbations
    work_h = best_h.copy()
    work_c5 = best_c5
    work_n = best_n
    t_p5 = t0 + budget_s * 0.90
    for it in range(20):
        if time.time() > t_p5:
            break
        cool = max(0.1, 1.0 - 0.05 * it)
        ptype = it % 4
        if ptype == 0:
            pert = 0.05 * cool * rng.standard_normal(work_n)
        elif ptype == 1:
            freq = rng.randint(3, max(4, work_n // 20))

            pert = 0.04 * cool * np.sin(2 * np.pi * freq * np.arange(work_n) / work_n + ph)
        elif ptype == 2:

            idx = rng.choice(work_n, 20, replace=False)
            pert[idx] = 0.15 * cool * rng.standard_normal(20)
        else:
            half = work_n // 2
            pert = np.zeros(work_n)
            pert[:half] = 0.06 * cool * rng.standard_normal(half)
            pert[half:] = -0.06 * cool * rng.standard_normal(work_n - half)
        h_pert = project_h(work_h + pert)
        T_bh = 5000 + 80000 * cool
        hc = slsqp_step(h_pert, T_bh, work_n, maxiter=500, ftol=1e-14)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.00005
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
                best_n = work_n
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h
        best_n = work_n

    # Phase 6: Final coordinate descent
    if time.time() < t0 + budget_s * 0.96:
        best_h, best_c5 = coord_descent(best_h, best_n, t0 + budget_s * 0.97, max_sweeps=150, seed_off=555)

    # Phase 7: Final SLSQP polish
    if time.time() < t0 + budget_s * 0.985:
        for T in [500000, 2000000]:
            if time.time() > t0 + budget_s * 0.99:
                break
            hc = slsqp_step(best_h, T, best_n, maxiter=1200, ftol=1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
