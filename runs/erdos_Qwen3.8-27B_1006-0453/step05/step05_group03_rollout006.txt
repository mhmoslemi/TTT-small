def targeted_lag_descent(h, n, t_end, max_iters=5000):
    """Iteratively reduce the worst lag."""
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = compute_corr(best)
        worst_idx = np.argmax(corr)
        lag = worst_idx - (n - 1)
        # Compute gradient for this lag
        grad = np.zeros(n)
        # For h[j]*(1-h[j+lag]): grad[j] += (1-h[j+lag]) for valid j
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:n])
            grad[lag:] += (-h[:n-lag])
        else:
            grad[-lag:] += (1.0 - h[:n+lag])
            grad[:n+lag] += (-h[-lag:])
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        # Line search
        alpha_candidates = [0.01, 0.005, 0.002, 0.001, 0.0005]
        best_alpha = alpha_candidates[0]
        best_val = best_c5

def coord_opt_full(h, n, i, corr_base):
    """Optimize h[i] given base correlation (with h[i]=0)."""
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    # Build coefficient vector for h[i]
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n-1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n-1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n-1)])
    def f(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 30})
    return res.x

def genetic_pop(h0, n, t_end, pop_size=12, gen_max=200):
    """Simple genetic algorithm."""
    pop = [h0.copy() for _ in range(pop_size)]

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def targeted_lag_descent(h, n, t_end, max_iters=20000):
    best = h.copy()
    best_c5 = compute_c5(best)
    prev_c5 = best_c5
    stall = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = compute_corr(best)
        top_k = np.argsort(corr)[-3:]
        for ki in top_k:
            if time.time() > t_end:
                break
            lag = ki - (n - 1)
            grad = np.zeros(n)
            if lag >= 0:
                grad[:n - lag] += (1.0 - best[lag:n])
                grad[lag:] += (-best[:n - lag])
            else:
                s = -lag
                grad[s:] += (1.0 - best[:n - s])
                grad[:n - s] += (-best[s:])
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                continue
            grad_n = grad / gnorm
            improved = False
            for a in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                h_try = project_h(best - a * grad_n, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return best, best_c5

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            best_h[i] = 0.0
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            best_h[i] = old
            coefs = np.zeros(2 * n - 1)
            lags = np.arange(-(n - 1), n)
            j1 = i + lags
            j2 = i - lags
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coefs[v1] += (1.0 - best_h[np.clip(j1[v1], 0, n - 1)])
            coefs[v2] += (-best_h[np.clip(j2[v2], 0, n - 1)])
            def f(hi):
                return np.max(corr_base + hi * coefs)
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 30})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def genetic_pop(h0, n, t_end, pop_size=14, gen_max=150):

            break

        bi = np.argmin(pops_c5)
        if pops_c5[bi] < best_c5:
            best_c5 = pops_c5[bi]
            best = pop[bi].copy()
    return best, best_c5

def make_thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        tm[i] = 1.0 if c % 2 == 0 else 0.0
