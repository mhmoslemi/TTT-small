        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with decreasing temperature
    t_p1 = t0 + budget_s * 0.22
    for T, fp in [(2000, 0.05), (800, 0.04), (300, 0.02), (100, 0.01), (50, 0.005)]:
        if time.time() > t_p1:
            break
        hc = slsqp_opt(best_h, T, n, 600, 1e-15, fp, 5)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Gradient polish to help escape
    t_p2 = t0 + budget_s * 0.30
    best_h, best_c5 = gradient_polish(best_h, n, t_p2, 60, 0.004, 10)

    # Phase 3: Main coordinate descent (heavy)
    t_p3 = t0 + budget_s * 0.58
    best_h, best_c5 = coord_descent(best_h, n, t_p3, 1000, 40)

    # Phase 4: SLSQP refinement
    t_p4 = t0 + budget_s * 0.72
    for T, fp in [(200, 0.01), (100, 0.005), (50, 0.0)]:
        if time.time() > t_p4:
            break
        hc = slsqp_opt(best_h, T, n, 800, 1e-16, fp, 5)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Final coordinate descent with more top shifts
    t_p5 = t0 + budget_s * 0.92
    best_h, best_c5 = coord_descent(best_h, n, t_p5, 600, 50)

    # Phase 6: Final SLSQP with very tight tolerance
    if time.time() < t_end - 12:
        hc = slsqp_opt(best_h, 1e7, n, 1200, 1e-17, 0.0, 3)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 7: Last coordinate descent
    if time.time() < t_end - 2:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 0.5, 150, 50)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
