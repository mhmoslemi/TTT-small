from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def soft_max_val_grad(h, T, n, pen=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    exp_corr = np.exp(T * (corr - cmax))
    exp_sum = exp_corr.sum()
    weights = exp_corr / exp_sum
    sm_val = (cmax + np.log(exp_sum) / T) * dx
    
    grad = np.zeros(n)
    thresh = 1e-6
    sig = np.where(weights > thresh)[0]
    lags = np.arange(-(n-1), n)
    for ki in sig:
        w = weights[ki]
        lag = lags[ki]
        ps = max(0, lag)
        pe = min(n, n + lag)
        if ps < pe:
            grad[ps:pe] += w * (1.0 - h[ps-lag:pe-lag])
        qs = max(0, -lag)
        qe = min(n, n - lag)
        if qs < qe:
            grad[qs:qe] += w * (-h[qs+lag:qe+lag])
    grad *= dx
    
    # Penalty for constraint
    if pen > 0:
        s = h.sum() - n / 2.0
        sm_val += pen * s * s
        grad += 2.0 * pen * s * np.ones(n)
    
    return sm_val, grad

def lbfgs_optimize(h, T, n, maxiter=2000, pen=1e3):
    def obj(x):
        return soft_max_val_grad(x, T, n, pen)[0]
    def grad(x):
        return soft_max_val_grad(x, T, n, pen)[1]
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h

    """Optimize h[i] while keeping others fixed."""
    old = h[i]
    h2 = h.copy()
    h2[i] = 0.0
    corr0 = np.correlate(h2, 1.0 - h2, mode='full')
    # coeff[k] = d(corr[k])/d(h[i])

    # This is a 1D problem: min over x in [0,1] of max_k (corr0[k] + x*coeff[k])
    # The max of linear functions is convex, so we can use ternary search or just evaluate
    # Actually minimize_scalar with bounded method works
    
    def f(x):

    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',

    return res.x

def compute_coeff_i(h, n, i):
    lags = np.arange(-(n-1), n)
    coeff = np.zeros(2*n-1)
    # First part: valid when 0 <= i-lag < n, i.e., i-n < lag <= i
    # Intersection with [-(n-1), n-1]: max(-(n-1), i-n+1) <= lag <= min(n-1, i)
    lo1 = max(-(n-1), i-n+1)  # = max(1-n, i-n+1) = i-n+1 for i < n-1, else 1-n
    hi1 = min(n-1, i)
    # Actually: 0 <= i-lag < n means i-n < lag <= i, i.e., lag in [i-n+1, i]
    # Clamped to [-(n-1), n-1]
    lo1 = max(-(n-1), i-n+1)
    hi1 = min(n-1, i)
    if lo1 <= hi1:
        l1 = lags[lo1+n-1:hi1+n-1+1]
        coeff[lo1+n-1:hi1+n-1+1] += (1.0 - h[i - l1])
    
    # Second part: valid when 0 <= i+lag < n, i.e., -i < lag < n-i, i.e., lag in [-i+1, n-i-1]
    # Wait: 0 <= i+lag < n means -i <= lag < n-i, i.e., lag in [-i, n-i-1]
    # Clamped to [-(n-1), n-1]
    lo2 = max(-(n-1), -i)
    hi2 = min(n-1, n-i-1)
    if lo2 <= hi2:
        l2 = lags[lo2+n-1:hi2+n-1+1]
        coeff[lo2+n-1:hi2+n-1+1] += -h[i + l2]
    
    return coeff

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(h, T, n, pen=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    exp_c = np.exp(T * (corr - cmax))
    es = exp_c.sum()
    val = (cmax + np.log(es) / T) * dx
    if pen > 0:
        sv = h.sum() - n / 2.0
        val += pen * sv * sv
    return val

def soft_max_grad(h, T, n, pen=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    exp_c = np.exp(T * (corr - cmax))
    es = exp_c.sum()
    w = exp_c / es
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    idx = np.where(w > 1e-5)[0]
    for ki in idx:
        wi = w[ki]
        lag = lags[ki]
        ps = max(0, lag)
        pe = min(n, n + lag)
        if ps < pe:
            grad[ps:pe] += wi * (1.0 - h[ps-lag:pe-lag])
        qs = max(0, -lag)
        qe = min(n, n - lag)
        if qs < qe:
            grad[qs:qe] += wi * (-h[qs+lag:qe+lag])
    grad *= dx
    if pen > 0:
        sv = h.sum() - n / 2.0
        grad += 2.0 * pen * sv
    return grad

def lbfgs_run(h, T, n, maxiter=3000, pen=500.0):
    def obj(x):
        return soft_max_obj(x, T, n, pen)
    def jac(x):
        return soft_max_grad(x, T, n, pen)
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 50})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_desc_fast(h, n, t_end, max_sweeps=60):
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    lags = np.arange(-(n-1), n)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            coeff = np.zeros(2*n - 1)
            lo1 = max(-(n-1), i - n + 1)
            hi1 = min(n-1, i)
            if lo1 <= hi1:
                sl = slice(lo1 + n - 1, hi1 + n)
                l1 = lags[sl]
                coeff[sl] += (1.0 - best_h[i - l1])
            lo2 = max(-(n-1), -i)
            hi2 = min(n-1, n - i - 1)
            if lo2 <= hi2:
                sl2 = slice(lo2 + n - 1, hi2 + n)
                l2 = lags[sl2]
                coeff[sl2] += -best_h[i + l2]
            old = best_h[i]
            cur_max = np.max(corr)
            def f(x):
                return np.max(corr + (x - old) * coeff)
            r = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-9, 'maxiter': 40})
            xn = r.x
            if abs(xn - old) > 1e-10:
                new_max = np.max(corr + (xn - old) * coeff)
                if new_max < cur_max - 1e-13:
                    corr += (xn - old) * coeff
                    best_h[i] = xn
                    best_c5 = float(new_max * dx)
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def make_initial(n, rng):
    h = np.zeros(n)
    h[:n//2] = 1.0
    h = project_h(h, n)
    alts = []
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    alts.append(project_h(tm, n))
    for _ in range(3):
        rb = np.zeros(n)
        rb[rng.choice(n, n//2, replace=False)] = 1.0
        alts.append(rb)
    # Fractional construction: smooth step
    x = np.linspace(0, 2, n, endpoint=False)
    for width in [0.3, 0.5, 0.8, 1.0]:
        fs = np.clip(0.5 * (1 + np.tanh((1.0 - x) / width)), 0, 1)
        fs = project_h(fs, n)
        alts.append(fs)
    return h, alts

def structured_mutation(h, n, rng, amp):
    hm = h.copy()
    mode = rng.randint(4)
    if mode == 0:
        s = rng.randint(0, n - 8)
        L = rng.randint(4, min(20, n - s))
        hm[s:s+L] = -hm[s:s+L] + 1.0
    elif mode == 1:
        s = rng.randint(0, n - 16)
        L = rng.randint(4, 12)
        a, b = s, s + L
        hm[a:b] = h[b:min(b+L, n)] if b + L <= n else h[b:]
    elif mode == 2:
        mask = rng.random(n) < 0.3
        hm[mask] = 1.0 - h[mask]
    else:
        hm = h + amp * rng.randn(n)
    return project_h(hm, n)
