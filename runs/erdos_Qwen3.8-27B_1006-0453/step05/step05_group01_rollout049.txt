import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        h = h * (target / s) if s > 1e-14 else h
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, t_end, top_k=25, max_sweeps=800):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max() * dx)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        top_idx = np.argsort(corr)[-top_k:]
        top_shifts = (top_idx - (n - 1)).astype(int)
        any_improve = False
        for j in np.random.permutation(n):
            if time.time() > t_end:
                return h, best_c5
            hj = h[j]
            coefs = np.empty(top_k)
            bases = np.empty(top_k)
            for t in range(top_k):
                k = top_shifts[t]
                base = corr[top_idx[t]]
                c = 0.0
                jp, jm = j + k, j - k
                if 0 <= jp < n:
                    base -= hj * (1.0 - h[jp])
                    c += (1.0 - h[jp])
                if 0 <= jm < n:
                    base -= h[jm] * (1.0 - hj)
                    c -= h[jm]
                coefs[t] = c
                bases[t] = base
            cur_v = np.max(bases + hj * coefs)
            best_t, best_v = hj, cur_v
            for cand in (0.0, 1.0):
                v = np.max(bases + cand * coefs)
                if v < best_v - 1e-14:
                    best_v, best_t = v, cand
            idx5 = np.argsort(bases + hj * coefs)[::-1][:6]
            for a in range(len(idx5)):
                for b in range(a + 1, len(idx5)):
                    ia, ib = idx5[a], idx5[b]
                    d = coefs[ia] - coefs[ib]
                    if abs(d) < 1e-14:
                        continue
                    ti = (bases[ib] - bases[ia]) / d
                    if 0.0 <= ti <= 1.0:
                        v = np.max(bases + ti * coefs)
                        if v < best_v - 1e-14:
                            best_v, best_t = v, ti
            if abs(best_t - hj) < 1e-13:
                continue
            delta = best_t - hj
            h[j] = best_t
            for t in range(top_k):
                k = top_shifts[t]
                m = top_idx[t]
                dm = 0.0
                jp, jm = j + k, j - k
                if 0 <= jp < n:
                    dm += delta * (1.0 - h[jp])
                if 0 <= jm < n:
                    dm -= delta * h[jm]
                corr[m] += dm
            any_improve = True
        h = project_h(h)
        if sweep % 5 == 0 or any_improve:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_now = float(corr.max() * dx)
            if c5_now < best_c5:
                best_c5 = c5_now
    return h, best_c5

def nesterov_subgrad(h, t_end, n_iter=300, top_k=8):
    n = len(h)
    dx = 2.0 / n
    y = h.copy()
    v = np.zeros(n)
    beta = 0.85
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        alpha = (0.02 * cool + 0.003) / (1.0 + 0.05 * it)
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        shifts = (top_idx - (n - 1)).astype(int)
        grad = np.zeros(n)
        for k in shifts:
            for j in range(n):
                g = 0.0
                jp, jm = j + k, j - k
                if 0 <= jp < n:
                    g += (1.0 - y[jp])
                if 0 <= jm < n:
                    g -= y[jm]
                grad[j] += g
        grad *= dx / len(shifts)
        v = beta * v - alpha * grad
        y_new = project_h(y + v)
        c5v = compute_c5(y_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = y_new.copy()
        y = y_new
    return best_h, best_c5

def basin_hop(h, t_end, n_iter, scale, use_structure=True):
    n = len(h)
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if use_structure and r < 0.35:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-6:]
            shifts = (top_idx - (n - 1)).astype(int)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n - k] += (work_h[k:n] - 0.5) * 0.3
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n + k] - 0.5) * 0.3
            pert *= scale * cool
        elif r < 0.5:
            f = np.random.randint(1, 15)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.6 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.65:
            mask = np.random.random(n) < 0.25
            pert = np.zeros(n)
            pert[mask] = scale * 1.2 * cool * np.random.randn(max(int(mask.sum()), 1))
        elif r < 0.8:
            blen = np.random.randint(max(n // 12, 2), max(n // 4, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * cool * np.random.randn(blen)
        else:
            pert = scale * cool * np.random.randn(n) * 0.8
        h_pert = project_h(work_h + pert)
        c5v = compute_c5(h_pert)
        temp = 0.001 * cool + 0.0002
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = h_pert
            work_c5 = c5v
    return work_h, work_c5

def refine(h, n, t_end):
    best_h = project_h(h)
    best_c5 = compute_c5(best_h)
    if t_end - time.time() > 8:
        bh_h, bh_c5 = basin_hop(best_h, time.time() + 8, 30, 0.08, True)
        if bh_c5 < best_c5:
            best_h, best_c5 = bh_h, bh_c5

    if t_end - time.time() > 15:
        best_h, best_c5 = fast_coord_descent(best_h, min(time.time() + 15, t_end), 25, 600)
    if t_end - time.time() > 6:
        bh_h, bh_c5 = basin_hop(best_h, time.time() + 6, 25, 0.04, False)
        if bh_c5 < best_c5:
            best_h, best_c5 = bh_h, bh_c5
    if t_end - time.time() > 10:

    if t_end - time.time() > 4:
        h2, c5v = nesterov_subgrad(best_h, min(time.time() + 4, t_end), 100, 5)
        if c5v < best_c5:
            best_h, best_c5 = h2, c5v
    if t_end - time.time() > 8:

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
