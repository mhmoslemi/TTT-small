        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-14:
            h = h * (tgt / max(s, 1e-14))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h = np.where(m, h + d / m.sum(), h)
            h = np.clip(h, 0.0, 1.0)
    return h

_T = [1.0]
_KP = [3]
def _sobj(x):
    T = _T[0]
    kp = _KP[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    # Top-k power mean as smooth max
    idx = np.argpartition(c, -kp)[-kp:]
    top = c[idx]
    m = top.max()
    e = np.exp(T * (top - m))
    sm = np.log(e.sum() / kp) / T + m
    return sm * 2.0 / len(x)

def _sgrad(x):
    T = _T[0]
    kp = _KP[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(c, -kp)[-kp:]
    top = c[idx]
    m = top.max()
    e = np.exp(T * (top - m))
    w = e / e.sum()
    g = np.zeros(n)
    for j in range(kp):
        wk = w[j]
        if wk < 1e-14:
            continue
        k = idx[j]
        k1 = np.arange(0, min(k + 1, n))
        i1 = k - k1  # = 0..k, maps to h[i1]
        g[i1] += wk * (1.0 - x[k - i1 + n - 1 - (n-1)])  # wrong, let me redo
        # c[k] = sum_i h[i+k-(n-1)]*(1-h[i]) for valid i
        # dc/dh[j] = (1-h[j-k]) * I(0<=j-k<n) - h[j+k] * I(0<=j+k<n)
        # For j: term1 = (1-h[j-k]) if 0<=j-k<n → j>=k
        #         term2 = -h[j+k] if 0<=j+k<n → j<=n-1-k
        for ii in range(n):
            t1 = 0.0
            t2 = 0.0
            if ii >= k and ii - k < n:
                t1 = 1.0 - x[ii - k]
            if ii + k < n:
                t2 = -x[ii + k]
            g[ii] += wk * (t1 + t2)
    return g * (2.0 / n)

def lbfgs_opt(h, T, kp, n, iters=300):
    _T[0] = T
    _KP[0] = kp
    try:
        r = minimize(h, _sobj, _sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-9})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=80):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    dx = 2.0 / n
    order = np.random.permutation(n)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr = get_corr(best)
        for _ in range(n):
            i = order[(_ + sw) % n]
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            k1 = np.arange(0, min(i + 1, mf))
            coef[k1] += 1.0 - best[i - k1]
            k2 = np.arange(0, min(n - i, mf))
            coef[k2] -= best[i + k2]
            base = corr - old * coef
            def f1d(t):
                return float(np.max(base + t * coef))
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-10})
            nv = float(res.x)
            if abs(nv - old) > 1e-11:
                trial = best.copy()
                trial[i] = nv
                trial = proj_h(trial, n)
                c5t = compute_c5(trial)
                if c5t < best_c5 - 1e-14:
                    best = trial
                    best_c5 = c5t
                    corr = get_corr(best)
                    imp = True
        if not imp:
            break
    return best, best_c5

def bin_sa(h, n, t_end, max_iters=500000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    if len(ones) > n // 2:
        h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        h_b[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    dx = 2.0 / n
    corr = get_corr(h_b)
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T = 0.04
    for it in range(max_iters):
        if time.time() > t_end:
            break
        # Move: swap one 1 and one 0
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_b[i], h_b[j] = 0.0, 1.0
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        nc = get_corr(h_b)
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0, 0.0
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
        T = max(0.0003, T * (1.0 - 0.5 / max(max_iters, 1)))
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=150):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        r = np.random.rand()
        if r < 0.3:
            pert = (0.03 + 0.12 * cool) * np.random.randn(n)
        elif r < 0.6:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        else:
            mask = np.random.random(n) < (0.15 + 0.4 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.2 * cool) * np.random.randn(int(mask.sum()))
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 8000.0 * cool
        kp = 3 if cool > 0.5 else 5
        ht = lbfgs_opt(hp, T_val, kp, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.001 * cool + 0.00003
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 160
        best_h = np.zeros(n0)
        best_h[:n0 // 2] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with top-k power mean, increasing T
    for T, kp in [(50, 5), (200, 4), (800, 3), (3000, 3), (10000, 3), (50000, 5)]:
        if time.time() - t0 > budget_s * 0.07:
            break
        h_new = lbfgs_opt(best_h, T, kp, n, iters=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Coordinate descent
    t_cd1 = t0 + budget_s * 0.17
    best_h, best_c5 = coord_desc(best_h, n, t_cd1, sweeps=100)

    # Phase 3: Binary SA
    t_bin = t0 + budget_s * 0.38
    h_bin, c5_bin = bin_sa(best_h, n, t_bin, max_iters=600000)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin
        best_h = lbfgs_opt(best_h, 3000.0, 3, n, iters=300)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.44, sweeps=50)

    # Phase 4: Basin hopping
    t_bh = t0 + budget_s * 0.64
    best_h, best_c5 = basin_hop(best_h, n, t_bh, n_iters=180)

    # Phase 5: Coordinate descent after basin hopping
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.74, sweeps=80)

    # Phase 6: Try higher resolution
    for n_hi in [200, 240]:
        if time.time() - t0 > budget_s * 0.82:
            break
        try:
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_hi, endpoint=False) + 1.0 / n_hi
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = proj_h(h_hi, n_hi)
            h_hi = lbfgs_opt(h_hi, 2000.0, 3, n_hi, iters=300)
            c5_hi = compute_c5(h_hi)
            h_hi, c5_hi = coord_desc(h_hi, n_hi, t0 + budget_s * 0.88, sweeps=40)
            if c5_hi < best_c5:
                best_h = h_hi
                best_c5 = c5_hi
                n = n_hi
        except Exception:
            pass

    # Phase 7: Another binary round at current n
    if time.time() - t0 < budget_s * 0.88:
        h_bin2, c5_bin2 = bin_sa(best_h, n, t0 + budget_s * 0.91, max_iters=300000)
        if c5_bin2 < best_c5:
            best_h = h_bin2.copy()
            best_c5 = c5_bin2
            best_h = lbfgs_opt(best_h, 5000.0, 3, n, iters=200)
            best_c5 = compute_c5(best_h)

    # Phase 8: Final refinement
    if time.time() - t0 < budget_s * 0.95:
        best_h = lbfgs_opt(best_h, 100000.0, 3, len(best_h), iters=600)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.98:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t0 + budget_s * 0.99, sweeps=40)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
