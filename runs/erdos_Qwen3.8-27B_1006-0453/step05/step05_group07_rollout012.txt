    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    mf = 2 * n - 1
    for k in range(mf):
        wk = w[k]
        if wk < 1e-15:
            continue
        for i in range(n):
            j1 = i + k - n + 1
            j2 = i - k + n + 1 - n
            j2 = i - k + n - 1
            contrib = 0.0
            if 0 <= j1 < n:
                contrib += 1.0 - h[j1]
            if 0 <= j2 < n:
                contrib -= h[j2]
            g[i] += wk * contrib
    return g * 2.0 / n

def lbfgs_opt(h, T, n, iters=400):
    try:
        r = minimize(h, _sobj, _sgrad, args=(T,), method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-9, 'maxls': 30})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_fast(h, n, t_end, sweeps=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode='full')
    mf = 2 * n - 1
    order = np.random.permutation(n)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for si in range(n):
            i = order[si % n]
            old = best[i]
            a = np.zeros(mf)
            for k in range(mf):
                p = i + k - n + 1
                q = i - k + n - 1
                cv = 0.0
                if 0 <= p < n:
                    cv += 1.0 - best[p]
                if 0 <= q < n:
                    cv -= best[q]
                a[k] = cv
            base = corr - old * a
            r = minimize_scalar(lambda x: np.max(base + x * a),
                                bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-14})
            nv = float(r.x)
            if abs(nv - old) > 1e-14:
                best[i] = nv
                corr += (nv - old) * a
                nc5 = float(corr.max() * 2.0 / n)
                if nc5 < best_c5 - 1e-15:
                    best_c5 = nc5
                    imp = True
        if not imp:
            break
        order = np.random.permutation(n)
    best = proj_h(best, n)
    return best, compute_c5(best)

def binary_sa(h, n, t_end, max_iters=600000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    if len(ones) > n // 2:
        h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T_temp = 0.03
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h_b[i] == h_b[j]:
            continue
        h_b[i], h_b[j] = h_b[j], h_b[i]
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T_temp):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        T_temp = max(0.0003, T_temp * 0.99997)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=150):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 4
        if mod == 0:
            pert = (0.01 + 0.08 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 6.2832)
            pert = (0.01 + 0.06 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.15 + 0.25 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.03 + 0.12 * cool) * np.random.randn(int(mask.sum()))
        else:
            k = np.random.randint(3, 12)
            idx = np.random.choice(n, k, replace=False)
            pert = np.zeros(n)
            pert[idx] = (0.05 + 0.15 * cool) * np.random.randn(k)
        hp = proj_h(work + pert, n)
        T_val = 500.0 + 20000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=200)
        ht, c5t = coord_fast(ht, n, min(time.time() + 12, t_end), sweeps=40)
        temp = 0.0005 * cool + 0.00003
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Smooth L-BFGS with various T
    for T in [80, 300, 1000, 4000, 15000, 60000]:
        if time.time() - t0 > budget_s * 0.12:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Aggressive coordinate descent
    best_h, best_c5 = coord_fast(best_h, n, t0 + budget_s * 0.25, sweeps=300)

    # Phase 3: Binary SA
    h_bin = (best_h > 0.5).astype(float)
    ones = np.where(h_bin == 1)[0]
    if len(ones) > n // 2:
        h_bin[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_bin == 0)[0]
        h_bin[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    h_sa, c5_sa = binary_sa(h_bin, n, t0 + budget_s * 0.42)
    if c5_sa < best_c5:
        best_h = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n), h_sa)
        best_h = proj_h(best_h, n)
        best_h = lbfgs_opt(best_h, 3000.0, n, iters=400)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_fast(best_h, n, t0 + budget_s * 0.50, sweeps=100)

    # Phase 4: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.72, n_iters=130)

    # Phase 5: Coordinate descent refinement
    best_h, best_c5 = coord_fast(best_h, n, t0 + budget_s * 0.82, sweeps=200)

    # Phase 6: Second basin hop round with smaller perturbations
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.90, n_iters=60)

    # Phase 7: Final high-T smoothing + coordinate descent
    if time.time() - t0 < budget_s * 0.93:
        best_h = lbfgs_opt(best_h, 100000.0, n, iters=500)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_fast(best_h, n, t0 + budget_s * 0.985, sweeps=150)

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
