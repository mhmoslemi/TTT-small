    # corr_k(v) where h[j]=v, h[j2]=s-v

    # = corr_k(current) + (v - h[j])*grad_k[j] + (h[j] - v)*grad_k[j2]
    # = corr_k(current) + v*(grad_k[j] - grad_k[j2]) + h[j]*(grad_k[j2] - grad_k[j])
    # = corr_k(current) + h[j]*(grad_k[j2] - grad_k[j]) + v*(grad_k[j] - grad_k[j2])
    
    # So it's linear in v with slope (grad_k[j] - grad_k[j2]) and intercept corr_k + h[j]*(grad_k[j2]-grad_k[j])

    v_at_j2 = h[j2]
    h[j] = 0; h[j2] = 0

    h[j] = 0; h[j2] = 1
    corr_j2 = np.correlate(h, 1.0 - h, mode='full')
    h[j] = v_at_j; h[j2] = v_at_j2
    
    grad_j = corr_j - corr_base
    grad_j2 = corr_j2 - corr_base

    intercept = corr_base + s * grad_j2
    slope = grad_j - grad_j2
    
    # Minimize max_k (intercept[k] + v*slope[k]) for v in [max(0, s-1), min(1, s)]
    v_min = max(0.0, s - 1.0)
    v_max = min(1.0, s)

    # When h[j]=0: corr_{n-1} = ... + 0*(1-0) + ... = ... (h[j] doesn't contribute at lag 0 since 0*(1-0)=0)
    # When h[j]=1: corr_{n-1} = ... + 1*(1-1) + ... = ... (h[j] doesn't contribute at lag 0 since 1*(1-1)=0)

    """Update corr array when h[old_j] changes from old_v to new_v."""

    if abs(dv) < 1e-15:
