                    best_nv = nv
            # Refine with golden section
            if 0 < best_nv < 1:
                a, b = max(0, best_nv - 0.05), min(1, best_nv + 0.05)
                _ga = (np.sqrt(5) - 1) / 2
                c_pt = b - _ga * (b - a)
                d_pt = a + _ga * (b - a)
                fc = np.max(base + c_pt * coef)
                fd = np.max(base + d_pt * coef)
                for _ in range(20):
                    if fc < fd:
                        b = d_pt; d_pt = c_pt; fd = fc
                        c_pt = b - _ga * (b - a)
                        fc = np.max(base + c_pt * coef)
                    else:
                        a = c_pt; c_pt = d_pt; fc = fd
                        d_pt = a + _ga * (b - a)
                        fd = np.max(base + d_pt * coef)
                best_nv = (a + b) / 2
            if abs(best_nv - old) < 1e-12:
                continue
            trial = best.copy()
            trial[i] = best_nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                corr_cache = np.correlate(best, 1.0 - best, mode='full')
                imp += 1
        if imp == 0:
            break
    return best, best_c5

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = 0.5 * n
    s = h.sum()
    if abs(s - tgt) < 1e-12:
        return h
    if s > tgt:
        h *= (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-12:
            h *= (tgt / max(s, 1e-12))
        else:
            h += rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0, 1)
    return h

def _sm_grad(cc, T, n):
    m = cc.max()
    e = np.exp(T * (cc - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - [0] * 0)  # placeholder
    return g

def lbfgs_smooth(h, T, n, iters=400):
    def obj(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        mm = cc.max()
        return (mm + np.log(np.sum(np.exp(T * (cc - mm)))) / T) * 2.0 / n
    def grad(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        mm = cc.max()
        e = np.exp(T * (cc - mm))
        w = e / e.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            lo1 = max(0, n - 1 - k)
            hi1 = min(n, 2 * n - 1 - k)
            if lo1 < hi1:
                idx = np.arange(lo1, hi1) + k - n + 1
                g[lo1:hi1] += wk * (1.0 - x[idx])
            lo2 = max(0, k - n + 1)
            hi2 = min(n, k + 1)
            if lo2 < hi2:
                idx = np.arange(lo2, hi2) - k + n - 1
                g[lo2:hi2] -= wk * x[idx]
        return g * 2.0 / n
    try:
        r = minimize(h, obj, grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 50})
        return proj_h(r.x, n)
    except Exception:
        return proj_h(h, n)

def coord_desc(h, n, t_end, sweeps=300):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = 0
        corr_cache = np.correlate(best, 1.0 - best, mode='full').copy()
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            base = corr_cache - old * coef
            def f1d(nv):
                return float(np.max(base + nv * coef))
            r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-12, 'maxiter': 50})
            nv = float(r.x)
            if abs(nv - old) < 1e-12:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr_cache = np.correlate(best, 1.0 - best, mode='full')
                imp += 1
        if imp == 0:
            break
    return best, best_c5

def topical_opt(h, n, t_end, n_rounds=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    for rd in range(n_rounds):
        if time.time() > t_end:
            break
        cc = np.correlate(best, 1.0 - best, mode='full')
        top_idx = np.argsort(cc)[::-1][:5]
        T_w = np.exp(80.0 * (cc - cc.max()))
        T_w[top_idx] *= 3.0
        T_w /= T_w.sum()
        m = cc.max()
        w_eff = T_w
        T_eff = 5000.0
        def obj_w(x):
            cc2 = np.correlate(x, 1.0 - x, mode='full')
            mm = cc2.max()
            e = np.exp(T_eff * (cc2 - mm))
            sw = np.dot(w_eff, e)
            return (mm + np.log(sw) / T_eff) * 2.0 / n
        def grad_w(x):
            cc2 = np.correlate(x, 1.0 - x, mode='full')
            mm = cc2.max()
            e = np.exp(T_eff * (cc2 - mm))
            w = e * w_eff
            w /= w.sum()
            g = np.zeros(n)
            for k in range(2 * n - 1):
                wk = w[k]
                if wk < 1e-14:
                    continue
                lo1 = max(0, n - 1 - k)
                hi1 = min(n, 2 * n - 1 - k)
                if lo1 < hi1:
                    idx = np.arange(lo1, hi1) + k - n + 1
                    g[lo1:hi1] += wk * (1.0 - x[idx])
                lo2 = max(0, k - n + 1)
                hi2 = min(n, k + 1)
                if lo2 < hi2:
                    idx = np.arange(lo2, hi2) - k + n - 1
                    g[lo2:hi2] -= wk * x[idx]
            return g * 2.0 / n
        try:
            r = minimize(best, obj_w, grad_w, method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 80, 'ftol': 1e-16, 'gtol': 1e-9})
            hp = proj_h(r.x, n)
            c5p = compute_c5(hp)
            if c5p < best_c5:
                best = hp
                best_c5 = c5p
        except Exception:
            pass
    return best, best_c5

def binary_block_sa(h, n, t_end, max_iters=200000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    n_target = n // 2
    if len(ones) > n_target:
        h_b[np.random.choice(ones, len(ones) - n_target, replace=False)] = 0
    elif len(ones) < n_target:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, n_target - len(ones), replace=False)] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T = 0.02
    for it in range(max_iters):
        if time.time() > t_end:
            break
        r = np.random.rand()
        if r < 0.4:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if h_b[i] != h_b[j]:
                h_b[i], h_b[j] = h_b[j], h_b[i]
                nc = np.correlate(h_b, 1.0 - h_b, mode='full')
                nc5 = float(np.max(nc) * dx)
                if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
                    corr = nc
                    if nc5 < best_c5:
                        best_c5 = nc5
                        best_h = h_b.copy()
                else:
                    h_b[i], h_b[j] = h_b[j], h_b[i]
        elif r < 0.7:
            blen = np.random.randint(2, min(8, n // 4) + 1)
            start = np.random.randint(0, n - blen)
            seg = h_b[start:start+blen].copy()
            shift = np.random.randint(1, blen)
            h_b[start:start+blen] = np.roll(seg, shift)
            nc = np.correlate(h_b, 1.0 - h_b, mode='full')
            nc5 = float(np.max(nc) * dx)
            if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
                corr = nc
                if nc5 < best_c5:
                    best_c5 = nc5
                    best_h = h_b.copy()
            else:
                h_b[start:start+blen] = seg
        else:
            i = np.random.randint(n)
            h_b[i] = 1.0 - h_b[i]
            if h_b.sum() != n_target:
                if h_b.sum() > n_target:
                    oo = np.where(h_b == 1)[0]
                    h_b[np.random.choice(oo, 2, replace=False)][0] = 0
                else:
                    zz = np.where(h_b == 0)[0]
                    h_b[np.random.choice(zz, 2, replace=False)][0] = 1
            nc = np.correlate(h_b, 1.0 - h_b, mode='full')
            nc5 = float(np.max(nc) * dx)
            if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
                corr = nc
                if nc5 < best_c5:
                    best_c5 = nc5
                    best_h = h_b.copy()
            else:
                h_b[i] = 1.0 - h_b[i]
                if h_b.sum() != n_target:
                    if h_b.sum() > n_target:
                        oo = np.where(h_b == 1)[0]
                        h_b[np.random.choice(oo, 1, replace=False)[0]] = 1
                    else:
                        zz = np.where(h_b == 0)[0]
                        h_b[np.random.choice(zz, 1, replace=False)[0]] = 0
        T = max(0.0003, T * 0.99997)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=100):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 4
        if mod == 0:
            pert = (0.01 + 0.08 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.01 + 0.06 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.1 + 0.2 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.03 + 0.12 * cool) * np.random.randn(mask.sum())
        else:
            idx = np.random.choice(n, size=max(1, n // 10), replace=False)
            pert = np.zeros(n)
            pert[idx] = (0.05 + 0.1 * cool) * np.random.randn(len(idx))
        hp = proj_h(work + pert, n)
        T_val = 500.0 + 8000.0 * cool
        ht = lbfgs_smooth(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.0006 * cool + 0.00003
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B geometric T schedule (8% time)
    t1 = t0 + budget_s * 0.08
    for T in [100, 500, 2000, 10000, 50000, 200000]:
        if time.time() > t1:
            break
        h_new = lbfgs_smooth(best_h, T, n, iters=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Extended coordinate descent (22% time)
    t2 = t0 + budget_s * 0.30
    best_h, best_c5 = coord_desc(best_h, n, t2, sweeps=500)

    # Phase 3: Topical optimization (15% time)
    t3 = t0 + budget_s * 0.45
    best_h, best_c5 = topical_opt(best_h, n, t3, n_rounds=150)
    best_h, best_c5 = coord_desc(best_h, n, t3 + budget_s * 0.03, sweeps=100)

    # Phase 4: Binary block SA + refine (15% time)
    t4 = t0 + budget_s * 0.63
    h_bin, c5_bin = binary_block_sa(best_h, n, t4, max_iters=300000)
    if c5_bin < best_c5:
        best_h = lbfgs_smooth(h_bin, 5000.0, n, iters=300)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_desc(best_h, n, t4 + budget_s * 0.04, sweeps=100)

    # Phase 5: Basin hopping (20% time)
    t5 = t0 + budget_s * 0.83
    best_h, best_c5 = basin_hop(best_h, n, t5, n_iters=150)
    best_h, best_c5 = coord_desc(best_h, n, t5 + budget_s * 0.03, sweeps=80)

    # Phase 6: Second topical + coord (8% time)
    t6 = t0 + budget_s * 0.91
    best_h, best_c5 = topical_opt(best_h, n, t6, n_rounds=100)
    best_h, best_c5 = coord_desc(best_h, n, t6 + budget_s * 0.03, sweeps=100)

    # Phase 7: Final refinement with very low T
    t7 = t0 + budget_s * 0.97
    if time.time() < t7:
        best_h = lbfgs_smooth(best_h, 500000.0, n, iters=600)
        best_c5 = compute_c5(best_h)
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.985, sweeps=50)

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
