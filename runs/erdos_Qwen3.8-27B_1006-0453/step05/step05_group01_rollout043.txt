    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def solve_1d_maxlines(base, coefs):
    """Minimize max(base + t*coefs) for t in [0,1]"""
    nc = len(base)
    best_t = 0.0
    best_val = float(np.max(base))
    v1 = float(np.max(base + coefs))
    if v1 < best_val:
        best_val = v1
        best_t = 1.0
    for a in range(nc):
        for b in range(a + 1, nc):
            dc = coefs[b] - coefs[a]
            if abs(dc) < 1e-14:
                continue
            t = (base[b] - base[a]) / dc
            if t <= 0.0 or t >= 1.0:
                continue
            v = float(np.max(base + t * coefs))
            if v < best_val:
                best_val = v
                best_t = t
    return best_t

def cd_sweep(h, n, n_top, order):
    """One coordinate descent sweep (Gauss-Seidel style)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_top:]
    ts = top_shifts = top_idx - (n - 1)
    for i in order:
        jp = i + ts
        jm = i - ts
        coefs = np.zeros(n_top)
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        if np.any(vp):
            coefs[vp] += 1.0 - h[jp[vp]]
        if np.any(vm):
            coefs[vm] -= h[jm[vm]]
        base = corr[top_idx] - h[i] * coefs
        h[i] = solve_1d_maxlines(base, coefs)
    return h

def coord_descent(h, n, t_end, max_sweeps=300, n_top_start=15, n_top_end=45):
    """Adaptive coordinate descent"""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    order = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        frac = sweep / max(max_sweeps - 1, 1)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        np.random.shuffle(order)
        cd_sweep(best_h, n, n_top, order)
        best_h = project_h(best_h)
        c5_now = compute_c5(best_h)
        if c5_now < best_c5:
            best_c5 = c5_now
        else:
            if sweep > 10 and c5_now >= best_c5:
                # No improvement, maybe converged
                pass
    return best_h, best_c5

def slsqp_refine(h, n, maxiter=300):
    """Quick SLSQP polish using log-sum-exp smooth max"""
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        T = 2000.0
        return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        T = 2000.0
        w = np.exp(T * (corr - m))
        w /= w.sum()
        # d/dx_i of corr[k] = sum_j x_j(1-x_{j+k})
        # = (1 - x_{i+k}) * delta(i+k in range) - x_{i-k} * delta(i-k in range) ... 
        # This is complex, skip gradient and let SLSQP approximate
        return None
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hp = project_h(np.clip(res.x, 0, 1))
        return hp, compute_c5(hp)
    except Exception:
        return h, compute_c5(h)

    except Exception:
        return h.copy(), compute_c5(h)

def thue_morse_init(n, seed_frac=0.5):
    tm = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x > 0:
            c ^= (x & 1)
            x >>= 1
        tm[i] = c
    target = n // 2
    n_ones = int(tm.sum())
    if n_ones == target:
        return tm
    idx_ones = np.where(tm == 1)[0]
    idx_zeros = np.where(tm == 0)[0]
    if n_ones > target:
        flip = len(idx_ones) - target
        tm[idx_ones[np.random.choice(len(idx_ones), flip, replace=False)]] = 0
    else:
        flip = target - n_ones
        tm[idx_zeros[np.random.choice(len(idx_zeros), flip, replace=False)]] = 1
    return tm

def periodic_init(n, period, ones_in_period):
    h = np.zeros(n)
    pat = np.zeros(period)
    pat[:ones_in_period] = 1.0
    repeats = n // period
    h[:repeats * period] = np.tile(pat, repeats)
    if n % period != 0:
        h[repeats * period:] = pat[:n % period]
    target = n // 2
    n_ones = int(h.sum())
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h == 1)[0]
            np.random.shuffle(idx)
            h[idx[:n_ones - target]] = 0
        else:
            idx = np.where(h == 0)[0]
            np.random.shuffle(idx)
            h[idx[:target - n_ones]] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.zeros(200)
        parent[:100] = 1.0

    n = len(parent)
    best_h = project_h(parent.copy())
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from parent
    t_p1 = min(t0 + budget_s * 0.20, t_end - 300)
    if time.time() < t_p1:
        best_h, best_c5 = coord_descent(best_h, n, t_p1, 800, 15, 55)

    # Phase 2: Multi-start with diverse initializations
    inits = []
    # Binary rounding of parent
    inits.append(round_to_binary(best_h, n).astype(float))
    # Thue-Morse
    inits.append(thue_morse_init(n))
    # Periodic patterns
    inits.append(periodic_init(n, 16, 8))
    inits.append(periodic_init(n, 24, 12))
    inits.append(periodic_init(n, 40, 20))
    # Random binary
    h_rand = np.random.rand(n)
    thresh = np.sort(h_rand)[n // 2]
    inits.append((h_rand >= thresh).astype(float))
    # Smeared parent
    inits.append(0.5 * best_h + 0.3 * (thue_morse_init(n) - 0.5))

    n_inits = len(inits)
    per_init = min((t_end - time.time() - 200) / max(n_inits, 1), 120)
    for s in range(n_inits):
        if time.time() > t_end - 200:
            break
        h_start = project_h(inits[s].copy())
        t_sub = min(time.time() + max(per_init, 25), t_end - 150)
        h_try, c5_try = coord_descent(h_start, n, t_sub, 400, 15, 55)
        # Try rounding and re-optimizing
        h_bin = round_to_binary(h_try, n).astype(float)
        if time.time() < t_sub:
            h_try2, c5_try2 = coord_descent(h_bin, n, min(time.time() + 30, t_sub), 200, 20, 50)
            if c5_try2 < c5_try:
                h_try, c5_try = h_try2, c5_try2
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try

    # Phase 3: Try n=300
    remaining = t_end - time.time()
    if remaining > 250:
        n2 = 300
        h2_init = np.interp(np.linspace(0, 2, n2, endpoint=False),
                            np.linspace(0, 2, n, endpoint=False), best_h)
        h2_init = project_h(h2_init)
        t_p3 = min(time.time() + min(remaining * 0.5, 180), t_end - 100)
        h2_opt, c5_2 = coord_descent(h2_init, n2, t_p3, 600, 15, 60)
        # Also try binary at n=300
        h2_bin = round_to_binary(h2_opt, n2).astype(float)
        t_p3b = min(time.time() + 60, t_end - 80)
        h2_opt2, c5_2b = coord_descent(h2_bin, n2, t_p3b, 400, 20, 60)
        if c5_2b < c5_2:
            h2_opt, c5_2 = h2_opt2, c5_2b
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt
            n = n2

    # Phase 4: Fine coordinate descent with large n_top
    if time.time() < t_end - 60:
        t_p4 = t_end - 30
        best_h, best_c5 = coord_descent(best_h, n, t_p4, 1000, 30, 80)

    # Phase 5: SLSQP polish
    if time.time() < t_end - 30:
        h_pol, c5_pol = slsqp_refine(best_h, n, 500)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 6: Final binary round + CD
    if time.time() < t_end - 15:
        h_bin = round_to_binary(best_h, n).astype(float)
        h_bin_opt, c5_bin = coord_descent(h_bin, n, t_end - 3, 300, 25, 70)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin_opt

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
