from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        sat = 1.0 - h
        tot = sat.sum()
        if tot < 1e-15:
            h = h * (target / s) if s > 0 else h
        else:
            h = h - sat * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def topk_grad_step(h, n, k, alpha):
    """One step of gradient descent on top-k correlations."""
    corr = get_corr(h)
    idx = np.argpartition(corr, -k)[-k:]
    g = np.zeros(n)
    for ki in idx:
        lag = ki - (n - 1)
        if lag >= 0:
            L = lag
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
        else:
            L = -lag
            g[:n-L] += (1.0 - h[L:])
            g[L:] -= h[:n-L]
    h_new = project_h(h - alpha * g, n)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        room = h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h - room * (diff / tot)
    return np.clip(h, 0.0, 1.0)

def softmax_obj_wrapper(h, T, n):
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def jac(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -50, 50)
        p = np.exp(sh)
        p = p / p.sum()
        g = np.zeros(2 * n - 1)
        for ki in range(2 * n - 1):
            if p[ki] < 1e-12:
                continue
            lag = ki - (n - 1)
            if lag >= 0:
                g[lag:] += (1.0 - x[:n-lag]) * p[ki]
                g[:n-lag] -= x[lag:] * p[ki]
            else:
                L = -lag
                g[:n-L] += (1.0 - x[L:]) * p[ki]
                g[L:] -= x[:n-L] * p[ki]
        return dx * g
    return obj, jac

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    try:
        obj, jac = softmax_obj_wrapper(h, T, n)
        def cons(x):
            return x.sum() - n / 2.0
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def topk_grad_step(h, n, k, alpha):
    corr = get_corr(h)
    idx = np.argpartition(corr, -k)[-k:]
    g = np.zeros(n)
    for ki in idx:
        lag = ki - (n - 1)
        if lag >= 0:
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
        else:
            L = -lag
            g[:n-L] += (1.0 - h[L:])
            g[L:] -= h[:n-L]
    return project_h(h - alpha * g, n)

    c5v = compute_c5(hc)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = hc
    # Also try SLSQP from binarized best
    h_bin2 = (best_h > 0.5).astype(float)
    d2 = n / 2.0 - h_bin2.sum()
    if d2 > 0:
        idx0 = np.where(h_bin2 == 0)[0]
        np.random.shuffle(idx0)
        h_bin2[idx0[:int(d2)]] = 1.0
    elif d2 < 0:
        idx1 = np.where(h_bin2 == 1)[0]
        np.random.shuffle(idx1)
        h_bin2[idx1[:int(-d2)]] = 0.0
    h_bin2 = project_h(h_bin2, n)
    hb2, cb2 = binary_local_search(h_bin2, n, t0 + budget_s * 0.88, max_iter=50000)
    hc2 = slsqp_smooth(hb2, 100, n, maxiter=200)
    c5v2 = compute_c5(hc2)
    if c5v2 < best_c5:
        best_c5 = c5v2
        best_h = hc2

    # Phase 6: Final coordinate descent (most precise)
    best_h, best_c5 = coord_sweep(best_h, n, n_sweeps=20, t_end=t0 + budget_s * 0.94)

    # Phase 7: Ultra-fine SLSQP
    for T in [3, 1]:
        if elapsed() > budget_s * 0.97:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final polish
    best_h, best_c5 = coord_sweep(best_h, n, n_sweeps=10, t_end=t0 + budget_s * 0.995)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
