        p=(0.005+0.02*cool)*np.random.randn(n)
        hp=project_h(wh2+p)
        hc=slsqp(hp, 50000+200000*cool, n, 300, 1e-15)
        c=compute_c5(hc)
        temp=0.0003*cool+0.00002
        if c<wc52 or np.random.rand()<np.exp(-(c-wc52)/temp):
            wh2=hc; wc52=c
            if c<best_c5: best_c5=c; best_h=hc
    if wc52<best_c5: best_c5=wc52; best_h=wh2

    # Final polish
    if el() < budget_s*0.985:
        hc = slsqp(best_h, 5000000, n, 3000, 1e-16)
        c = compute_c5(hc)
        if c < best_c5: best_c5=c; best_h=hc
    if el() < budget_s*0.995:
        best_h, best_c5 = cd(best_h, n, t0+budget_s*0.995, 30)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
