import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    dx = 2.0 / n
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def block_lp_opt(h, n, start, end):
    """Solve LP for one block of variables h[start:end]."""
    m = end - start
    nf = 2 * n - 1  # number of correlation points
    
    curr_corr = get_corr(h)
    
    # Build coefficient matrix A[m_corrs, m] where
    # corr_new[k] = curr_corr[k] + A[k,j] * delta[j]
    # For variable j (absolute index), delta_j changes h[j]:
    #   coef[k, j] = (1 - h[j+k-n+1])*I(valid) - h[j-k+n-1]*I(valid)
    
    A = np.zeros((nf, m))
    for jj in range(m):
        j = start + jj
        # Term 1: h[j] as first arg, (1-h) at position j+k-n+1
        idx1 = j + np.arange(nf) - (n - 1)  # positions in (1-h)
        valid1 = (idx1 >= 0) & (idx1 < n)
        A[valid1, jj] += (1.0 - h[idx1[valid1]])
        # Term 2: (1-h[j]) as second arg, h at position j-k+n-1
        idx2 = j - (np.arange(nf) - (n - 1))  # positions in h
        valid2 = (idx2 >= 0) & (idx2 < n)
        A[valid2, jj] += (-h[idx2[valid2]])
    
    # LP: minimize t
    # s.t. curr_corr[k] + A[k,:] @ delta <= t  (for all k)
    #      0 <= h_block + delta <= 1
    #      sum(delta) = 0
    # Variables: [delta_0, ..., delta_{m-1}, t]  (m+1 variables)
    
    c = np.zeros(m + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: curr_corr + A@delta - t <= 0
    # i.e., A@delta - t <= -curr_corr
    # x-scheme: [A, -1] @ [delta; t] <= -curr_corr
    A_ineq = np.hstack([A, -np.ones((nf, 1))])
    b_ineq = -curr_corr
    
    # Box constraints: -h_block <= delta <= 1 - h_block
    # (lower bound): delta >= -h_block  =>  -delta <= h_block
    # (upper bound): delta <= 1 - h_block
    
    # Sum constraint: sum(delta) = 0
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(-h[start+jj], 1.0 - h[start+jj]) for jj in range(m)]
    bounds.append((None, None))  # t is free
    
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    new_block = h[start:end] + res.x[:m]
    return np.clip(new_block, 0, 1)

    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-h[start + jj], 1.0 - h[start + jj]) for jj in range(m)]
    bounds.append((None, None))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            return h[start:end] + res.x[:m]
    except Exception:
        pass
    return None

def block_cd(h, n, t_end, block_size=8, max_sweeps=300):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for start in range(0, n, block_size):
            if time.time() > t_end:
                break
            end = min(start + block_size, n)
            if end - start < 2:
                continue
            new_block = block_lp(best_h, n, start, end)
            if new_block is not None:
                cand = best_h.copy()
                cand[start:end] = np.clip(new_block, 0, 1)
                c5_try = compute_c5(cand)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = cand
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: Gradient descent warmup (fast)
    for T in [200, 2000, 20000]:
        if elapsed() > budget_s * 0.06:
            break
        hc, _ = proj_grad(best_h, n, T, t0 + budget_s * 0.06, lr=0.003, max_iter=1500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Fine block CD
    best_h, best_c5 = block_cd(best_h, n, t0 + budget_s * 0.18, block_size=4, max_sweeps=400)
    best_h, best_c5 = block_cd(best_h, n, t0 + budget_s * 0.30, block_size=8, max_sweeps=300)

    # Phase 3: Coarse block CD + gradient
    best_h, best_c5 = block_cd(best_h, n, t0 + budget_s * 0.40, block_size=16, max_sweeps=200)
    if elapsed() < budget_s * 0.42:
        hc, _ = proj_grad(best_h, n, 50000, t0 + budget_s * 0.42, lr=0.002, max_iter=1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Basin hopping with block CD local opt
    work_h = best_h.copy()
    work_c5 = best_c5
    n_bh = 60
    for it in range(n_bh):
        if elapsed() > budget_s * 0.72:
            break
        cool = 1.0 - 0.85 * (it / float(n_bh))
        mod = it % 6
        if mod == 0:
            step = (0.02 + 0.08 * cool)
            h_pert = project_h(work_h + step * np.random.randn(n))
        elif mod == 1:
            freq = np.random.randint(1, 10)
            ph = np.random.uniform(0, 2 * np.pi)
            amp = 0.02 + 0.06 * cool
            h_pert = project_h(work_h + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + ph))
        elif mod == 2:
            mask = np.random.random(n) < 0.2
            h_pert = work_h.copy()
            h_pert[mask] = np.clip(h_pert[mask] + (0.03 + 0.1 * cool) * np.random.randn(mask.sum()), 0, 1)
            h_pert = project_h(h_pert)
        elif mod == 3:
            blk = n // 6
            b = np.random.randint(0, n - blk)
            h_pert = work_h.copy()
            h_pert[b:b+blk] = np.clip(h_pert[b:b+blk] + 0.05*cool*np.random.randn(blk), 0, 1)
            h_pert = project_h(h_pert)
        elif mod == 4:
            a, b2 = np.random.permutation(n)[:2]
            h_pert = work_h.copy()
            d = (work_h[b2] - work_h[a]) * (0.4 + 0.6 * cool)
            h_pert[a] = np.clip(h_pert[a] + d, 0, 1)
            h_pert[b2] = np.clip(h_pert[b2] - d, 0, 1)
            h_pert = project_h(h_pert)
        else:
            step = 0.01 + 0.03 * cool
            h_pert = project_h(work_h + step * np.random.randn(n))

        hc, c5v = block_cd(h_pert, n, t0 + budget_s * 0.72, block_size=8, max_sweeps=40)
        if c5v > best_c5 + 0.002:
            continue
        temp = 0.0004 * cool + 0.00002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h.copy()

    # Phase 5: High-T gradient + SLSQP polish
    for T in [100000, 500000]:
        if elapsed() > budget_s * 0.82:
            break
        hc, _ = proj_grad(best_h, n, T, t0 + budget_s * 0.82, lr=0.001, max_iter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        if elapsed() < budget_s * 0.82:
            hc2 = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-16)
            c5v2 = compute_c5(hc2)
            if c5v2 < best_c5:
                best_c5 = c5v2
                best_h = hc2

    # Phase 6: Final block CD (multiple block sizes)
    for bs in [4, 8, 16, 8, 4]:
        if elapsed() > budget_s * 0.95:
            break
        best_h, best_c5 = block_cd(best_h, n, t0 + budget_s * 0.95, block_size=bs, max_sweeps=200)

    # Final gradient nudge
    if elapsed() < budget_s * 0.98:
        hc, _ = proj_grad(best_h, n, 1000000, t0 + budget_s * 0.98, lr=0.0005, max_iter=2000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
