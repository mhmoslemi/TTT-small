import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_fn(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / T

def eq_cnstr(x):
    return x.sum() - len(x) / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=600, ftol=1e-15):
    try:
        res = minimize(smooth_c5_fn, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_cnstr}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def fast_coord_single(h, n, i):
    old = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_idx = np.arange(2 * n - 1)
    idx1 = i + m_idx - n + 1
    idx2 = i - m_idx + n + 1 - (n - 1)
    idx2 = i - m_idx + n - 1
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(v1, h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where(v2, h[np.clip(idx2, 0, n - 1)], 0.0)
    coef = (1.0 - h1) - h2
    intercept = corr - old * coef
    slope = coef
    def obj_1d(x):
        return np.max(intercept + x * slope)
    try:
        res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-14})
        return float(res.x)
    except Exception:
        grid = np.linspace(0, 1, 4096)
        vals = np.max(intercept[None, :] + grid[:, None] * slope[None, :], axis=1)
        return float(grid[np.argmin(vals)])

def coord_descent(h, n, t_end, max_sweeps=2000, rng=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = fast_coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def flip_push(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_iter = 3000
    for it in range(n_iter):
        if time.time() > t_end:
            break
        idx = rng.randint(n)
        if best_h[idx] < 0.05 or best_h[idx] > 0.95:
            continue
        val = rng.random()
        if val < 0.4:
            flip_to = 0.0
        elif val < 0.8:
            flip_to = 1.0
        else:
            flip_to = rng.uniform(0, 1)
        h_try = best_h.copy()
        h_try[idx] = flip_to
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def make_small_pert(n, cool, rng):
    n_pert = rng.randint(1, 4)
    p = np.zeros(n)
    idx = rng.choice(n, n_pert, replace=False)
    amp = (0.02 + 0.15 * cool)
    p[idx] = amp * rng.standard_normal(n_pert)
    r = rng.random()
    if r < 0.3:
        freq = rng.randint(2, max(3, n // 20))
        ph = rng.uniform(0, 2 * np.pi)
        p += (0.01 + 0.04 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif r < 0.6:
        b = rng.randint(2, max(3, n // 10))
        s = rng.randint(0, n - b)
        p[s:s+b] += (0.01 + 0.05 * cool) * rng.standard_normal(b)
    return p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation with multiple T
    t_slsqp_end = t0 + budget_s * 0.08
    for T in [50, 300, 2000, 15000]:
        if elapsed() > t_slsqp_end:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=500, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Aggressive coordinate descent
    t_cd_end = t0 + budget_s * 0.35
    best_h, best_c5 = coord_descent(best_h, n, t_cd_end, max_sweeps=5000, rng=rng)

    # Phase 3: Basin hopping with small perturbations + fast local opt
    t_bh_end = t0 + budget_s * 0.75
    work_h = best_h.copy()
    work_c5 = best_c5
    max_iter = 1000
    while elapsed() < t_bh_end:
        cool = max(0.0, 1.0 - 0.8 * (min(elapsed() - t0 - budget_s*0.35, budget_s*0.4) / max(budget_s*0.4, 1)))
        pert = make_small_pert(n, cool, rng)
        h_pert = project_h(work_h + pert)
        # Fast local optimization: a few sweeps of coordinate descent
        h_loc, c5_loc = coord_descent(h_pert, n, t0 + budget_s * 0.75, max_sweeps=80, rng=rng)
        if c5_loc < work_c5:
            work_h = h_loc
            work_c5 = c5_loc
            if c5_loc < best_c5:
                best_c5 = c5_loc
                best_h = h_loc.copy()
        elif c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc.copy()
        temp = 0.0005 * cool + 0.00005
        if c5_loc <= work_c5 or rng.rand() < np.exp(-(c5_loc - work_c5) / max(temp, 1e-12)):
            if c5_loc <= work_c5:
                work_h = h_loc
                work_c5 = c5_loc
            elif rng.rand() < 0.3:
                work_h = h_loc
                work_c5 = c5_loc

    # Phase 4: Flip push + coordinate descent
    t_flip_end = t0 + budget_s * 0.82
    best_h, best_c5 = flip_push(best_h, n, t_flip_end, rng)
    t_cd2_end = t0 + budget_s * 0.90
    best_h, best_c5 = coord_descent(best_h, n, t_cd2_end, max_sweeps=3000, rng=rng)

    # Phase 5: Final SLSQP polish
    t_slsqp2_end = t0 + budget_s * 0.96
    for T in [50000, 300000]:
        if elapsed() > t_slsqp2_end:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=1000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Final coordinate descent
    t_final = t0 + budget_s * 0.99
    best_h, best_c5 = coord_descent(best_h, n, t_final, max_sweeps=500, rng=rng)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
