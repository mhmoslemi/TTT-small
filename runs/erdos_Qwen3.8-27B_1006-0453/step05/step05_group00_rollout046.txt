    # For finding the new max, we need max(corr + delta).
    # Since delta is sparse (nonzero only where p+L or q+L is in [0,N-1]),
    # we can compute it directly.
    
    # Actually for binary h, (1-h[p+L]) is either 0 or 1.
    # So delta values are in {-1, 0, 1}.
    
    # The new max is at most max(corr) + 1 and at least max(corr) - 1.
    # To check if new max < current max, we need:
    #   For all n where corr[n] == max: delta[n] < 0 (or corr[n]+delta[n] < max)
    #   For all n where corr[n] < max: corr[n] + delta[n] < max
    
    # More efficient: just check the relevant indices.
    new_max = float(np.max(corr))
    # Check indices where corr is near max
    near_max = np.where(corr >= new_max - 1)[0]
    for n in near_max:
        L = n - center
        d = 0.0
        if 0 <= p + L < N:
            d -= (1.0 - h[p + L])
        if 0 <= q + L < N:
            d += (1.0 - h[q + L])
        if corr[n] + d > new_max:
            new_max = corr[n] + d
    return new_max
