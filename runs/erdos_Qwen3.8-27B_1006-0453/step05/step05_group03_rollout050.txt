def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)

def compute_coefs(h, n, i):
    """Compute the coefficient of h[i] in each correlation entry."""
    m = 2 * n - 1
    coefs = np.zeros(m)
    lags = np.arange(-(n-1), n, dtype=int)

        # h[i] appears as (1-h[j+l]) with j+l=i, so j=i-l: contributes -h[i-l]
        j1 = i - l
        if 0 <= j1 < n:
            coefs[k] += (-h[j1])
    return coefs

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_coefs(h, n, i):
    m = 2 * n - 1
    coefs = np.zeros(m)
    for k in range(m):
        l = k - (n - 1)
        j2 = i + l
        if 0 <= j2 < n:
            coefs[k] += (1.0 - h[j2])
        j1 = i - l
        if 0 <= j1 < n:
            coefs[k] += (-h[j1])
    return coefs

def exact_max_min(base, coefs, lo=0.0, hi=1.0):
    tol = 1e-13
    vals0 = base + lo * coefs
    f0 = np.max(vals0)
    act0 = coefs[vals0 >= f0 - tol]
    if act0.min() > tol:
        return lo
    vals1 = base + hi * coefs
    f1 = np.max(vals1)
    act1 = coefs[vals1 >= f1 - tol]
    if act1.max() < -tol:
        return hi
    a, b = lo, hi
    for _ in range(80):
        mid = (a + b) * 0.5
        vals = base + mid * coefs
        fv = np.max(vals)
        act = coefs[vals >= fv - tol]
        if act.min() > tol:
            b = mid
        elif act.max() < -tol:
            a = mid
        else:
            break
    return (a + b) * 0.5

def coord_descent_exact(h, n, max_sweeps=200, t_end=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if t_end is not None and time.time() > t_end:
            break
        improved = 0
        for i in range(n):
            old = best_h[i]
            best_h[i] = 0.0
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            coefs = compute_coefs(best_h, n, i)
            t_new = exact_max_min(corr_base, coefs, 0.0, 1.0)
            best_h[i] = t_new
            if abs(t_new - old) > 1e-11:
                improved += 1
        best_h = project_h(best_h, n)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new
        if improved == 0:
            break
    return best_h, best_c5

def block_descent(h, n, block_size=4, max_iter=100, t_end=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iter):
        if t_end is not None and time.time() > t_end:
            break
        improved = False
        for start in range(0, n, block_size):
            idx = list(range(start, min(start + block_size, n)))
            b = len(idx)
            if b == 1:
                old = best_h[idx[0]]
                best_h[idx[0]] = 0.0
                corr_b = np.correlate(best_h, 1.0 - best_h, mode='full')
                coefs = compute_coefs(best_h, n, idx[0])
                t_new = exact_max_min(corr_b, coefs, 0.0, 1.0)
                best_h[idx[0]] = t_new
                if abs(t_new - old) > 1e-11:
                    improved = True
                continue
            vals_old = best_h[idx].copy()
            best_h[idx] = 0.0
            corr_base = np.correlate(best_h, 1.0 - best_h, mode='full')
            all_coefs = np.array([compute_coefs(best_h, n, i) for i in idx])
            best_h[idx] = vals_old

            def f_block(x):
                return np.max(corr_base + np.dot(x, all_coefs))
            res = minimize(f_block, vals_old, method='SLSQP',
                           bounds=[(0.0, 1.0)] * b,
                           options={'maxiter': 50, 'ftol': 1e-16})
            x_new = np.clip(res.x, 0.0, 1.0)

                best_h = project_h(best_h, n)

        if not improved:
            break
    return best_h, best_c5

def active_gradient(h, n, top_k=5, step=0.005):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for ki in top_idx:
        l = ki - (n - 1)
        for i in range(n):
            j2 = i + l
            if 0 <= j2 < n:
                grad[i] += (1.0 - h[j2])
            j1 = i - l
            if 0 <= j1 < n:
                grad[i] += (-h[j1])
    h_new = project_h(h - step * grad, n)
    return h_new

def slsqp_smooth(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
