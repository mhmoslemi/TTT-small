```python
def project_to_simplex_box(v, target_sum):
    """Project v onto {h: 0<=h_i<=1, sum(h)=target_sum}"""
    # Bisection on lambda: sum(clamp(v + lambda, 0, 1)) = target_sum
    lo = target_sum - v.sum() - n  # too small
    hi = target_sum - v.sum()  # too large
    # Actually: sum(clamp(v_i + lambda, 0, 1)) is monotone in lambda
    lo = -v.max() - 1  # ensures all clamped to 0
    hi = 1 - v.min()  # ensures all clamped to 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(v + mid, 0, 1).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v + lam, 0, 1)
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h = h + room * (diff / tot)
    else:
        if h.sum() < 1e-12:
            return h
        h = h * (1.0 + diff / h.sum())
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(sh))) / T) * (2.0 / len(h))

def slsqq_opt(h, T, n, maxiter=800, ftol=1e-15):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_single(h, n, i):
    """Optimize coordinate i, return optimal value"""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    k_range = np.arange(-(n-1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2*n-1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n-1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n-1)])
    dx = 2.0 / n
    
    def obj(val):
        return float(np.max(corr_base + val * coefs) * dx)
    
    try:
        res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-13})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_deadline, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_imp_count = 0
    for sweep in range(max_sweeps):
        if time.time() > t_deadline:
            break
        imp = False
        for i in range(n):
            hi_new = coord_opt_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            no_imp_count += 1
            if no_imp_count >= 3:
                break
        else:
            no_imp_count = 0
    return best_h, best_c5

def basin_hop(h, n, rng, t_start, t_end, n_iters, cool_start, cool_end):
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(v):
    n = len(v)
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(sh))) / T) * (2.0 / len(h))

def constraint_fun(x, n):
    return x.sum() - n * 0.5

def slsq_opt(h, T, n, maxiter=500, ftol=1e-15):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_fun, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def eval_1d(val, cb, co, dx):
    return float(np.max(cb + val * co) * dx)

def coord_opt(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    cb = get_corr(h0)
    kr = np.arange(-(n - 1), n)
    j1, j2 = i + kr, i - kr
    v1, v2 = (j1 >= 0) & (j1 < n), (j2 >= 0) & (j2 < n)
    co = np.zeros(2 * n - 1)
    co[v1] += 1.0 - h[np.clip(j1[v1], 0, n - 1)]
    co[v2] += -h[np.clip(j2[v2], 0, n - 1)]
    dx = 2.0 / n
    try:
        r = minimize_scalar(eval_1d, args=(cb, co, dx), bounds=(0.0, 1.0),
                           method='bounded', options={'xatol': 1e-13})
        return float(r.x)
    except Exception:
        return h[i]

def coord_desc(h, n, t_deadline, max_sweeps=200):
    bh = h.copy()
    bc5 = compute_c5(bh)
    no_imp = 0
    for _ in range(max_sweeps):
        if time.time() > t_deadline:
            break
        imp = False
        for i in range(n):
            nv = coord_opt(bh, n, i)
            if abs(nv - bh[i]) < 1e-13:
                continue
            h_try = bh.copy()
            h_try[i] = nv
            h_try = project_h(h_try)
            c = compute_c5(h_try)
            if c < bc5 - 1e-14:
                bc5 = c
                bh = h_try
                imp = True
        no_imp = 0 if imp else no_imp + 1
        if no_imp >= 3:
            break
    return bh, bc5

def binarize(h, n, rng):
    """Round to binary, preserving sum"""
    b = (h > 0.5).astype(float)
    # Fix sum: need exactly n/2 ones
    target = n // 2
    n_ones = int(b.sum())
    if n_ones > target:
        # Remove excess ones: pick the ones with smallest h values
        one_idx = np.where(b == 1)[0]
        order = np.argsort(h[one_idx])
        for idx in order[:n_ones - target]:
            b[one_idx[idx]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:target - n_ones]:
            b[zero_idx[idx]] = 1.0
    return b

def sa_swap(h, n, c5_cur, T_sa, rng):
    """One SA step: swap a random 1 and 0, accept/reject"""
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, c5_cur
    i, j = ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]
```
