    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -100, 100)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def _slsqp_cons(x):
    return x.sum() - _gp['n'] / 2.0

def slsqp_smooth(h, T, n, maxiter=500):
    _gp['dx'] = 2.0 / n
    _gp['T'] = T
    _gp['n'] = n
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _slsqp_cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _coord_1d(hi, corr_base, coefs):
    return np.max(corr_base + hi * coefs)

def coord_opt(h, n, i, lo=None, hi=None):
    old = h[i]
    h_tmp = h.copy()
    h_tmp[i] = 0.0
    corr_base = np.correlate(h_tmp, 1.0 - h_tmp, mode='full')
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n - 1), n)
    j1 = i + lags
    j2 = i - lags
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs[v1] += (1.0 - h_tmp[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h_tmp[np.clip(j2[v2], 0, n - 1)])
    if lo is None:
        lo = max(0.0, old - 0.12)
    if hi is None:
        hi = min(1.0, old + 0.12)
    lo = max(0.0, lo)
    hi = min(1.0, hi)
    if hi - lo < 1e-10:
        return old
    res = minimize_scalar(_coord_1d, bounds=(lo, hi), args=(corr_base, coefs),
                          method='bounded', options={'xatol': 1e-13, 'maxiter': 100})
    return res.x

def _de_eval(x, n):
    xp = project_h(x, n)
    return compute_c5(xp)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP annealing
    for T in [20000, 5000, 1200, 300, 80, 25]:
        if time.time() - t0 > budget_s * 0.12:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Differential evolution (global exploration)
    de_deadline = t0 + budget_s * 0.45
    bounds = [(0.0, 1.0)] * n
    for restart in range(4):
        if time.time() > de_deadline:
            break
        perturb = best_h + 0.04 * np.random.randn(n)
        try:
            res = differential_evolution(_de_eval, bounds, args=(n,),
                                         maxiter=60, popsize=12,
                                         tol=0.05, seed=seed + restart * 7,
                                         init='latinhypercube', polish=False,
                                         updating='delayed', workers=1,
                                         mutation=(0.5, 1.0))
            cand = project_h(res.x, n)
            c5c = compute_c5(cand)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = cand
        except Exception:
            pass
        # Quick SLSQP polish on DE result
        if time.time() < de_deadline:
            hc = slsqp_smooth(best_h, 200, n, maxiter=200)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 3: Pair-coordinate descent
    t_end3 = t0 + budget_s * 0.62
    for sweep in range(40):
        if time.time() > t_end3:
            break
        improved = False
        for i in range(0, n - 1, 2):
            if time.time() > t_end3:
                break
            j = i + 1
            h_try = best_h.copy()
            nv_i = coord_opt(h_try, n, i)
            h_try[i] = nv_i
            nv_j = coord_opt(h_try, n, j)
            h_try[j] = nv_j
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            # Try full-range coord pass
            if sweep < 5:
                for i in range(n):
                    if time.time() > t_end3:
                        break
                    nv = coord_opt(best_h, n, i, 0.0, 1.0)
                    if abs(nv - best_h[i]) > 1e-12:
                        h_try = best_h.copy()
                        h_try[i] = nv
                        h_try = project_h(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            best_c5 = c5_try
                            best_h = h_try
                            improved = True
        if not improved and sweep > 5:
            break

    # Phase 4: Basin hopping with SLSQP
    work_h = best_h.copy()
    work_c5 = best_c5
    t_end4 = t0 + budget_s * 0.78
    for it in range(35):
        if time.time() > t_end4:
            break
        cool = max(0.05, 1.0 - it / 35.0)
        amp = 0.006 + 0.012 * cool
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        T_bh = 50 + 600 * cool
        hc = slsqp_smooth(h_pert, T_bh, n, maxiter=200)
        c5v = compute_c5(hc)
        temp = 8e-5 * cool + 3e-5
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Fine single-coordinate descent
    t_end5 = t0 + budget_s * 0.92
    best_c5_prev = best_c5
    for sweep in range(60):
        if time.time() > t_end5:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end5:
                break
            nv = coord_opt(best_h, n, i)
            if abs(nv - best_h[i]) > 1e-13:
                h_try = best_h.copy()
                h_try[i] = nv
                h_try = project_h(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break

    # Phase 6: Final SLSQP polish at very low T
    for T in [30, 10, 5]:
        if time.time() - t0 > budget_s * 0.97:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final coord descent
    if time.time() - t0 < budget_s * 0.99:
        for sweep in range(30):
            if time.time() > t0 + budget_s * 0.995:
                break
            improved = False
            for i in range(n):
                if time.time() > t0 + budget_s * 0.995:
                    break
                nv = coord_opt(best_h, n, i, max(0, best_h[i]-0.05), min(1, best_h[i]+0.05))
                if abs(nv - best_h[i]) > 1e-13:
                    h_try = best_h.copy()
                    h_try[i] = nv
                    h_try = project_h(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-16:
                        best_c5 = c5_try
                        best_h = h_try
                        improved = True
            if not improved:
                break

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
