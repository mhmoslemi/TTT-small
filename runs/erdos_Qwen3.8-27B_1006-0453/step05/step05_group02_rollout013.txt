That works. Now let me think about the binary swap search more carefully.

For a binary vector h of length n with exactly n/2 ones:
- Correlation at shift k (0-indexed from -(n-1) to (n-1)): 
  - For k >= 0: corr[n-1+k] = ∑_{i=0}^{n-1-k} h[i+k] * (1-h[i])
  - For k < 0: corr[n-1+k] = ∑_{i=-k}^{n-1} h[i+k] * (1-h[i]) = ∑_{i=0}^{n-1+k} h[i] * (1-h[i-k])

When we swap h[a] and h[b] (where h[a]=1, h[b]=0):
- For each shift k, the correlation changes by at most 2 (one less overlap at some positions, one more at others)
- Specifically, positions i where i+k = a or i = a or i+k = b or i = b are affected

This means we can compute the effect of a swap in O(1) per shift, or O(n) total.

Actually, for the full swap search:
- There are C(n/2, 2) possible pairs of 1s and C(n/2, 2) pairs of 0s, but we only care about swapping a 1 with a 0
- That's (n/2) * (n/2) = n²/4 possible swaps
- For n=160, that's 6400 swaps
- Each swap evaluation is O(n) for computing the new max correlation
- Total: 6400 * 160 = ~1M operations per full sweep

That's very fast! Let me implement a full sweep swap search:

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

    if abs(diff) < 1e-14:

    if diff > 0:

        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = corr_full(h)
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

                       constraints=[{'type': 'eq', 'fun': constraint_fn, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def eval_coord(corr_base, coefs, dx, hi):
    return float(np.max(corr_base + hi * coefs) * dx)

def coord_opt(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = corr_full(h_base)
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)

    try:

                              bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-10})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    if rng is None:
        rng = np.random.RandomState(0)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def binarize(h):
    n = len(h)
    h_b = (h > 0.5).astype(float)
    ones = h_b.sum()
    target = n // 2
    if ones != target:
        if ones > target:
            diffs = np.abs(h - 0.5)
            idx = np.where(h_b == 1)[0]
            order = idx[np.argsort(diffs[idx])]
            for i in order[:int(ones - target)]:
                h_b[i] = 0
        else:
            diffs = np.abs(h - 0.5)
            idx = np.where(h_b == 0)[0]
            order = idx[np.argsort(diffs[idx])]
            for i in order[:int(target - ones)]:
                h_b[i] = 1
    return h_b

def binary_sa(h_b, n, t_end, rng):
    """Simulated annealing on binary swap space."""
    best_h = h_b.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_b.copy()
    cur_c5 = best_c5
    t_start = time.time()
    total_time = max(t_end - t_start, 1.0)
    iteration = 0
    while time.time() < t_end:
        iteration += 1
        progress = min((time.time() - t_start) / total_time, 1.0)
        T_sa = 0.01 * (1.0 - 0.95 * progress) + 0.0005
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        cur_h[a] = 0.0
        cur_h[b] = 1.0
        c5_try = compute_c5(cur_h)
        delta = c5_try - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T_sa, 1e-9)):
            cur_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = cur_h.copy()
        else:
            cur_h[a] = 1.0
            cur_h[b] = 0.0
    return best_h, best_c5

def greedy_swap(h_b, n, t_end):
    """Greedy: always apply the best improving swap."""
    best_h = h_b.copy()
    best_c5 = compute_c5(best_h)
    while time.time() < t_end:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_local = best_c5
        best_pair = None
        for a in ones:
            for b in zeros:
                best_h[a] = 0.0
                best_h[b] = 1.0
                c5_try = compute_c5(best_h)
                if c5_try < best_local:
                    best_local = c5_try
                    best_pair = (int(a), int(b))
        if best_pair is not None:
            a, b = best_pair
            best_h[a] = 0.0
            best_h[b] = 1.0
            best_c5 = best_local
        else:
            a_save, b_save = best_pair if best_pair else (int(ones[0]), int(zeros[0]))
            best_h[a_save] = 1.0
            best_h[b_save] = 0.0
            best_c5 = compute_c5(best_h)
            break
    return best_h, best_c5

def targeted_perturb(h, n, rng, amplitude):
    corr = corr_full(h)
    m = np.max(corr)
    kmax = int(np.argmax(corr))
    shift = kmax - (n - 1)
    if shift >= 0:
        i_lo, i_hi = 0, n - 1 - shift
    else:
        i_lo, i_hi = -shift, n - 1
    active_idx = np.arange(i_lo, i_hi + 1)
    pert = np.zeros(n)
    n_act = len(active_idx)
    if n_act > 0:
        pert[active_idx] = amplitude * rng.standard_normal(n_act)
    pert += 0.3 * amplitude * rng.standard_normal(n)
    return h + pert
