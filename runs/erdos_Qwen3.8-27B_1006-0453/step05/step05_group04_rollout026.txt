```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            lo, hi = 0.0, 1.0 - h.min() + 1.0
            for _ in range(50):
                mid = (lo + hi) / 2
                if np.minimum(h + mid, 1.0).sum() < target:
                    lo = mid
                else:
                    hi = mid
            h = np.minimum(h + (lo+hi)/2, 1.0)
        else:
            lo, hi = 0.0, h.max() + 1.0
            for _ in range(50):
                mid = (lo + hi) / 2
                if np.maximum(h - mid, 0.0).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h = np.maximum(h - (lo+hi)/2, 0.0)
    return np.clip(h, 0.0, 1.0)

def cd_fast(h, n, t_end, max_sweeps=5000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0/n))
    dx = 2.0 / n
    n2m1 = 2*n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        for i in range(n):
            if time.time() > t_end:
                best_h = project_h(best_h, n)
                best_c5 = compute_c5(best_h)
                return best_h, best_c5
            old = best_h[i]
            # Build lin vector
            s_pos = n - 1 - i  # max positive shift
            s_neg = i  # max negative shift (i - s >= 0)
            lin = np.zeros(n2m1)
            lin[n-1] = 1.0 - 2.0*old
            if s_pos > 0:
                lin[n:n+s_pos] += (1.0 - best_h[i+1:i+s_pos+1])
            if s_neg > 0:
                lin[n:n+s_neg] -= best_h[i-1::-1]
            if s_neg > 0:
                lin[n-2:n-2-s_neg:-1] += (1.0 - best_h[i-1::-1])
            if s_pos > 0:
                lin[n-2:n-2-s_pos:-1] -= best_h[i+1:i+s_pos+1]
            # 1D optimization
            c_base = corr.copy()
            def f1d(d):
                v = c_base + d * lin
                v[n-1] -= d*d
                return float(np.max(v) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 50})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-12:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n-1] -= d*d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved += 1
            else:

                corr -= d * lin
                corr[n-1] += d*d
        if improved == 0 and sweep > 5:
            break
    best_h = project_h(best_h, n)
```

```python
def lbfgs_smoothed(h, n, t_end, T_seq=None, penalty=1e4):
    if T_seq is None:
        T_seq = [50, 200, 1000, 5000, 20000]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    x0 = best_h.copy()
    bounds = [(0.0, 1.0)] * n
    for T in T_seq:
        if time.time() > t_end:
            break
        def obj(x):
            dx = 2.0/n
            corr = np.correlate(x, 1.0-x, mode='full') * dx
            a = T * corr
            a_max = a.max()
            s = np.exp(a - a_max).sum()
            if s < 1e-300:
                o = a_max / T
            else:
                o = a_max / T + np.log(s) / T
            o += penalty * (x.sum() - n/2.0)**2
            return float(o)
        
        def grad(x):
            dx = 2.0/n
            corr = np.correlate(x, 1.0-x, mode='full') * dx
            a = T * corr
            a_max = a.max()
            w = np.exp(a - a_max)
            w /= w.sum()
            g = np.zeros(n)
            for m in range(2*n-1):
                wi = w[m]
                if wi < 1e-12:
                    continue
                s = m - (n-1)
                if s > 0:
                    g[s:] += wi * (1.0 - x[:n-s])
                    g[:n-s] -= wi * x[s:]
                elif s < 0:
                    a2 = -s
                    g[:n-a2] += wi * (1.0 - x[a2:])
                    g[a2:] -= wi * x[:n-a2]
                else:
                    g += wi * (1.0 - 2.0*x)
            g *= dx
            g += 2.0 * penalty * (x.sum() - n/2.0)
            return g
        
        res = minimize(obj, x0, method='L-BFGS-B', jac=grad, bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10})
        x_new = project_h(res.x, n)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_h = x_new.copy()
            best_c5 = c5_new
        x0 = x_new
    return best_h, best_c5
```

For binary h: c_s = |{i : h[i]=1, h[i+s]=0}| (count of 1s followed by 0s at offset s). C5 = max_s c_s * dx.

For n=160, dx=1/80, and c_s ranges from 0 to 80 (at most 80 since only 80 are 1s). C5 = max_s c_s / 80.

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-11:
            break
        if diff > 0:
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2
                if np.minimum(h + mid, 1.0).sum() < target:
                    lo = mid
                else:
                    hi = mid
            h = np.minimum(h + (lo + hi) / 2, 1.0)
        else:
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2
                if np.maximum(h - mid, 0.0).sum() > target:
                    lo = mid
                else:
                    hi = mid
            h = np.maximum(h - (lo + hi) / 2, 0.0)
    return np.clip(h, 0.0, 1.0)

def cd_fast(h, n, t_end, max_sweeps=50000, seed=42):
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    n2m1 = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = 0
        for i in range(n):
            if time.time() > t_end:
                best_h = project_h(best_h, n)
                return best_h, compute_c5(best_h)
            old = best_h[i]
            s_pos = n - 1 - i
            s_neg = i
            lin = np.zeros(n2m1)
            lin[n - 1] = 1.0 - 2.0 * old
            if s_pos > 0:
                lin[n:n + s_pos] += (1.0 - best_h[i + 1:i + s_pos + 1])
            if s_neg > 0:
                lin[n:n + s_neg] -= best_h[i - 1::-1]
            if s_neg > 0:
                lin[n - 2:n - 2 - s_neg:-1] += (1.0 - best_h[i - 1::-1])
            if s_pos > 0:
                lin[n - 2:n - 2 - s_pos:-1] -= best_h[i + 1:i + s_pos + 1]
            c_b = corr + dx * 0
            def f1d(d):
                v = c_b + d * lin
                v[n - 1] -= d * d
                return float(np.max(v) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11, 'maxiter': 30})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-12:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved += 1
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if improved == 0 and sweep > 3:
            break
    best_h = project_h(best_h, n)
    return best_h, compute_c5(best_h)

def lbfgs_smooth(h, n, t_end, T_list=None):
    if T_list is None:
        T_list = [30, 100, 400, 1500, 6000, 20000]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    x = best_h.copy()
    bounds = [(0.0, 1.0)] * n
    pen = 5e3
    for T in T_list:
        if time.time() > t_end:
            break
        def obj(xv):
            dx = 2.0 / n
            corr = np.correlate(xv, 1.0 - xv, mode='full') * dx
            a = T * corr
            am = a.max()
            e = np.exp(a - am)
            s = e.sum()
            o = am / T + (np.log(s) if s > 0 else -700.0) / T
            o += pen * (xv.sum() - n / 2.0) ** 2
            return float(o)
        def jacc(xv):
            dx = 2.0 / n
            corr = np.correlate(xv, 1.0 - xv, mode='full') * dx
            a = T * corr
            am = a.max()
            w = np.exp(a - am)
            w /= w.sum()
            g = np.zeros(n)
            for m in range(2 * n - 1):
                wi = w[m]
                if wi < 1e-10:
                    continue
                s = m - (n - 1)
                if s > 0:
                    g[s:] += wi * (1.0 - xv[:n - s])
                    g[:n - s] -= wi * xv[s:]
                elif s < 0:
                    a2 = -s
                    g[:n - a2] += wi * (1.0 - xv[a2:])
                    g[a2:] -= wi * xv[:n - a2]
                else:
                    g += wi * (1.0 - 2.0 * xv)
            g *= dx
            g += 2.0 * pen * (xv.sum() - n / 2.0)
            return g
        try:
            res = minimize(obj, x, method='L-BFGS-B', jac=jacc, bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12})
            x = project_h(res.x, n)
        except Exception:
            pass
        c5_new = compute_c5(x)
        if c5_new < best_c5:
            best_h = x.copy()
            best_c5 = c5_new
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=400, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.008 + 0.05 * cool
        mode = hop % 6
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 10)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = step * 2 * rng.randn(mask.sum())
        elif mode == 3:
            k = rng.randint(2, 8)
            idx = rng.choice(n, size=min(n, k * 3), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * rng.randn(len(idx))
        elif mode == 4:
            blk = rng.randint(2, 12)
            pert = np.zeros(n)
            for b in range(0, n, blk):
                pert[b:b + blk] = step * rng.randn(blk)
        else:
            pert = step * 0.3 * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        cd_budget = min(1.5 + 3.0 * cool, 4.0)
        h_opt, c5_opt = cd_fast(h_pert, n, time.time() + cd_budget,
                                max_sweeps=300, seed=seed + hop * 7)
        temp = 0.002 * cool + 0.0002
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h, cur_c5 = h_opt, c5_opt
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt.copy(), c5_opt
    return best_h, best_c5

def try_upsampled(h160, t_end, budget):

    x1 = np.linspace(0, 2, 160, endpoint=False) + 1.0 / 160
    x2 = np.linspace(0, 2, 320, endpoint=False) + 1.0 / 320

    return h320, c5_320, n2
```
