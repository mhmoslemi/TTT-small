    dx = 2.0 / n

def compute_corr(h):

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            if s < 1e-14:
                break
            h = h * (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, T, n):
    """Smooth approximation of max correlation * dx."""
    dx = 2.0 / n
    corr = np.correlate(np.clip(x, 0, 1), 1.0 - np.clip(x, 0, 1), mode='full')
    m = np.max(corr)
    shifted = (corr - m) * T
    shifted = np.clip(shifted, -50, 50)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

        m = np.max(corr)

        return (m + np.log(np.sum(np.exp(sh))) / T) * dx

        L = np.random.randint(2, min(20, n//4))
        s1 = np.random.randint(0, n - 2*L)
        s2 = np.random.randint(s1 + L, n - L + 1)
        # Ensure blocks don't overlap
        if s1 + L > s2:

        h_new[s1:s1+L], h_new[s2:s2+L] = cur_h[s2:s2+L].copy(), cur_h[s1:s1+L].copy()

            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

def slsqp_opt(h, T, n, maxiter=300):
    def obj(x):
        return smooth_obj_grad(x, T, n)[0]
    def grad(x):
        return smooth_obj_grad(x, T, n)[1]
    def con_f(x):
        return x.sum() - n / 2.0
    def con_j(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_f, 'jac': con_j}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(res.x, n)
    except Exception:
        return h

def exact_subgrad(h, n, k_top=5):
    corr = compute_corr(h)
    idx = np.argsort(corr)[-k_top:]
    grad = np.zeros(n)
    for ki in idx:
        s = ki - (n - 1)
        if s >= 0:
            if s > 0:
                grad[0:s] += (1.0 - h[s:2*s])
            lo, hi = s, n - s
            if lo < hi:
                grad[lo:hi] += (1.0 - h[lo+s:hi+s]) - h[lo-s:hi-s]
            if n - s < n:
                grad[n-s:n] += -h[0:s]
        else:
            al = -s
            if al > 0:
                grad[0:al] += -h[al:2*al]
            lo, hi = al, n - al
            if lo < hi:
                grad[lo:hi] += (1.0 - h[lo-al:hi-al]) - h[lo+al:hi+al]
            if n - al < n:
                grad[n-al:n] += 1.0 - h[n-2*al:n-al]
    return grad / len(idx)

def subgrad_method(h, n, t_end, iters=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_c5 = best_c5
    for it in range(iters):
        if time.time() > t_end:
            break
        cool = max(0.05, 1.0 - it / float(iters))
        alpha = 0.002 + 0.006 * cool
        k_top = 3 + int(5 * cool)
        g = exact_subgrad(h, n, k_top)
        # Project gradient onto constraint subspace
        g = g - g.mean()
        # Armijo line search
        c1 = 1e-4
        step = alpha
        f0 = compute_c5(h)
        h_new = project_h(h - step * g, n)
        f_new = compute_c5(h_new)
        for _ in range(10):
            if f_new < f0 - c1 * step * np.dot(g, h - h_new):
                break
            step *= 0.5
            h_new = project_h(h - step * g, n)
            f_new = compute_c5(h_new)
        h = h_new
        if f_new < best_c5:
            best_h = h.copy()
            best_c5 = f_new
    return best_h, best_c5

def block_swap_move(h, n, rng):
    L = rng.randint(2, min(25, n // 4) + 1)
    max_s1 = n - 2 * L
    if max_s1 <= 0:
        return h.copy()
    s1 = rng.randint(0, max_s1 + 1)
    s2 = rng.randint(s1 + L, n - L + 1)
    h_new = h.copy()
    tmp = h_new[s1:s1+L].copy()
    h_new[s1:s1+L] = h_new[s2:s2+L]
    h_new[s2:s2+L] = tmp
    return h_new

def sa_block_swap(h, n, t_end, max_iter=3000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    T_init = 0.003
    T_end = 0.00005
    T = T_init
    n_i = max_iter
    for it in range(n_i):
        if time.time() > t_end:
            break
        T = T_init * (T_end / T_init) ** (it / max(n_i, 1))
        h_new = block_swap_move(cur_h, n, np.random)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 or np.random.rand() < np.exp(-(c5_new - cur_c5) / T):
            cur_h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    return best_h, best_c5

def coord_opt_1d(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = compute_corr(h)
    h[i] = old
    ncorr = 2 * n - 1
    coefs = np.zeros(ncorr)
    for ki in range(ncorr):
        s = ki - (n - 1)
        # h[i] contribution to corr[ki]
        if s >= 0:
            # h[i]*(1-h[i+s]) valid if i < n-s and i+s < n
            if 0 <= i < n - s:
                coefs[ki] += (1.0 - h[i + s])
            # h[i-s]*(1-h[i]) valid if 0 <= i-s < n-s, i.e., s <= i < n
            if 0 <= i - s < n - s:
                coefs[ki] += -h[i - s]
        else:
            al = -s
            # h[i+al]*(1-h[i]) valid if al <= i < n and i < n-al
            if al <= i < n - al:
                coefs[ki] += -h[i + al]
            # h[i]*(1-h[i-al]) valid if i >= al and i-al < n-al
            if al <= i < n:
                coefs[ki] += (1.0 - h[i - al])
    def f(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 40})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=50):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_1d(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-10:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def leveling(h, n, t_end, max_iter=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        corr = compute_corr(best_h)
        top_k = 6
        top_idx = np.argsort(corr)[-top_k:]
        top_avg = corr[top_k:-1] if top_k > 1 else corr[top_k]
        # Find position affecting top lags most
        best_improve = 0.0001
        best_move = None
        for ki in top_idx[:3]:
            s = ki - (n - 1)
            for _ in range(5):
                i = np.random.randint(0, n)
                delta = 0.02 * np.random.choice([-1, 1])
                h_try = best_h.copy()
                h_try[i] = best_h[i] + delta
                h_try = project_h(h_try, n)
                c5_try = compute_c5(h_try)
                if best_c5 - c5_try > best_improve:
                    best_improve = best_c5 - c5_try
                    best_move = h_try
        if best_move is not None:
            best_h = best_move
            best_c5 = compute_c5(best_h)
    return best_h, best_c5

def make_thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    # Initial solution
    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 160
        best_h = np.zeros(n)
        best_h[:n//2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP from current solution with decreasing T
    t_phase1 = t0 + budget_s * 0.12
    for T in [20000, 5000, 1500, 500, 150, 50]:
        if time.time() > t_phase1:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=250)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Multi-start with diverse inits
    t_phase2 = t0 + budget_s * 0.22
    inits = []
    inits.append(make_thue_morse(n))
    for _ in range(3):
        inits.append(make_random_binary(n, rng))
    # Two-block with transition
    hb = np.zeros(n)
    trans = int(n * 0.05)
    half = n // 2
    hb[:half - trans//2] = 1.0
    hb[half + trans//2:] = 0.0
    hb[half - trans//2:half + trans//2] = 0.5
    inits.append(hb)
    for alt in inits:
        if time.time() > t_phase2:
            break
        alt = project_h(alt, n)
        for T in [5000, 1000]:
            if time.time() > t_phase2:
                break
            hc = slsqp_opt(alt, T, n, maxiter=200)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 3: Exact subgradient descent
    t_phase3 = t0 + budget_s * 0.38
    best_h, best_c5 = subgrad_method(best_h, n, t_phase3, iters=400)

    # Phase 4: Block swap simulated annealing
    t_phase4 = t0 + budget_s * 0.62
    best_h, best_c5 = sa_block_swap(best_h, n, t_phase4, max_iter=8000)

    # Phase 5: SLSQP refinement after SA
    t_phase5 = t0 + budget_s * 0.72
    for T in [2000, 500, 150]:
        if time.time() > t_phase5:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Subgradient + SLSQP alternating
    t_phase6 = t0 + budget_s * 0.82
    for it in range(15):
        if time.time() > t_phase6:
            break
        cool = 1.0 - it / 15.0
        hc, _ = subgrad_method(best_h, n, min(time.time() + budget_s*0.01, t_phase6), iters=50)
        T = 500 + 2000 * cool
        hc2 = slsqp_opt(hc, T, n, maxiter=150)
        c5v = compute_c5(hc2)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc2

    # Phase 7: Another block swap SA round (finer)
    t_phase7 = t0 + budget_s * 0.88
    best_h, best_c5 = sa_block_swap(best_h, n, t_phase7, max_iter=4000)

    # Phase 8: Coordinate descent
    t_phase8 = t0 + budget_s * 0.95
    best_h, best_c5 = coord_descent(best_h, n, t_phase8, max_sweeps=60)

    # Phase 9: Leveling
    t_phase9 = t0 + budget_s * 0.98
    best_h, best_c5 = leveling(best_h, n, t_phase9, max_iter=100)

    # Final: short SLSQP + coord descent
    if time.time() < t0 + budget_s * 0.99:
        hc = slsqp_opt(best_h, 100, n, maxiter=100)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=15)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
