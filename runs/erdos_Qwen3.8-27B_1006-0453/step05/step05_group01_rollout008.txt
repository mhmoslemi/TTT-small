        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp(h_in, T, n, maxiter=800, ftol=1e-16, topk=0, flat_pen=0.0):
    h = h_in.copy()
    try:
        res = minimize(soft_max_obj, h, args=(T, topk, flat_pen), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def grad_subgrad(h, alpha, n_topk, n, mom_v, beta=0.45):
    dx = 2.0 / n
    corr = get_corr(h)
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if mom_v is not None:
        grad = beta * mom_v + (1.0 - beta) * grad
    return project_h(h - alpha * grad), grad

def binary_anneal(h, n, t_end, n_sweeps=300, T0=0.003, T1=0.0001):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = h.copy()
    work_c5 = best_c5
    step = 1 / n
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        T = T0 * (T1 / T0) ** (sweep / max(n_sweeps - 1, 1))
        for _ in range(n):
            if time.time() > t_end:
                break
            i = np.random.randint(n)
            if work_h[i] < 0.01:
                j = np.random.choice(np.where(work_h > 0.99)[0]) if np.sum(work_h > 0.99) > 0 else None
            elif work_h[i] > 0.99:
                j = np.random.choice(np.where(work_h < 0.01)[0]) if np.sum(work_h < 0.01) > 0 else None
            else:
                continue
            if j is None or j is i:
                continue
            work_h[i] = 1.0 if work_h[i] < 0.5 else 0.0
            work_h[j] = 0.0 if work_h[j] < 0.5 else 1.0
            work_h = project_h(work_h)
            c5_try = compute_c5(work_h)
            if c5_try < work_c5 or np.random.rand() < np.exp(-(c5_try - work_c5) / max(T, 1e-10)):
                work_c5 = c5_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = work_h.copy()
            else:
                work_h[i] = 1.0 if work_h[i] > 0.5 else 0.0
                work_h[j] = 1.0 if work_h[j] > 0.5 else 0.0
                work_h = project_h(work_h)
    return best_h, best_c5

def coord_desc(h, n, t_end, max_sweeps=300, n_top=20, xatol=1e-14):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = get_corr(best_h)
        top_idx = np.argsort(corr)[-n_top:]
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_idx[t] - (n - 1)
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += 1.0 - best_h[jp]
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': xatol})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            old_val = best_h[i]
            best_h[i] = t_new
            best_h = project_h(best_h)
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                improved = True
            else:
                best_h[i] = old_val
                best_h = project_h(best_h)
        if not improved and sweep > 80:
            break
        if improved:
            n_top = min(30, n_top + 1)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi, targeted=True):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(1.0 - it / max(n_iter, 1), 0.01)
        r = np.random.random()
        if targeted and r < 0.35:
            corr = get_corr(work_h)
            top_idx = np.argsort(corr)[-6:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if 0 <= k < n:
                    pert[:n - k] += work_h[k:n] - 0.5
                elif -n < k < 0:
                    pert[-k:] += work_h[:n + k] - 0.5
            pert *= scale * 0.5 * cool
        elif r < 0.55:
            f = np.random.randint(1, 15)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.8 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.75:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.2 * cool * np.random.randn(max(int(mask.sum()), 1))
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp(h_pert, T, n, 500, 1e-15, topk=8)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-10)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_n(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # SLSQP warmup with increasing T
    remaining = t_end - time.time()
    if remaining > 30:
        for T in [50, 500, 5000, 50000, 500000]:
            if time.time() > t_end - 10:
                break
            hc = slsqp(best_h, T, n, 600, 1e-15, topk=10)
            c5v = compute_c5(hc)
            if c5v < best_c5 - 1e-16:
                best_c5 = c5v
                best_h = hc
    # Momentum subgradient + SLSQP
    remaining = t_end - time.time()
    if remaining > 50:
        mom_v = np.zeros(n)
        n_iters = min(50, int(remaining / 2.5))
        for it in range(n_iters):
            if time.time() > t_end - 10:
                break
            cool = 1.0 - it / max(n_iters, 1)
            alpha = 0.001 + 0.004 * cool
            n_topk = 5 + int(15 * cool)
            hs, g = grad_subgrad(best_h, alpha, n_topk, n, mom_v, beta=0.5)
            mom_v = g
            T = 300 + 20000 * cool
            fl = 0.08 * cool
            hc = slsqp(hs, T, n, 400, 1e-15, topk=12, flat_pen=fl)
            c5v = compute_c5(hc)
            if c5v < best_c5 - 1e-16:
                best_c5 = c5v
                best_h = hc
    # Basin hopping
    remaining = t_end - time.time()
    if remaining > 60:
        t_bh = time.time() + remaining * 0.5
        bh_h, bh_c5 = basin_hop(best_h, n, t_bh, int(remaining * 0.12), 0.10, 500, 8000, True)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Coordinate descent (extended)
    remaining = t_end - time.time()
    if remaining > 40:
        t_cd = time.time() + remaining * 0.7
        best_h, best_c5 = coord_desc(best_h, n, t_cd, 250, 20, 1e-14)
    # Fine basin hop
    remaining = t_end - time.time()
    if remaining > 30:
        t_bh2 = time.time() + remaining * 0.5
        bh_h, bh_c5 = basin_hop(best_h, n, t_bh2, int(remaining * 0.08), 0.03, 2000, 10000, False)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Binary anneal (small perturbation)
    remaining = t_end - time.time()
    if remaining > 25 and best_c5 > 0.381:
        t_ba = time.time() + remaining * 0.4
        ba_h, ba_c5 = binary_anneal(best_h, n, t_ba, 150, 0.002, 0.0001)
        if ba_c5 < best_c5:
            best_c5 = ba_c5
            best_h = ba_h
    # Final coord descent
    remaining = t_end - time.time()
    if remaining > 15:
        best_h, best_c5 = coord_desc(best_h, n, t_end - 5, 150, 30, 1e-15)
    # Final SLSQP polish
    remaining = t_end - time.time()
    if remaining > 8:
        hc = slsqp(best_h, 1e7, n, 1000, 1e-16, topk=1)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Main optimization at parent resolution
    t_p1 = t0 + budget_s * 0.52
    best_h, best_c5 = optimize_n(best_h, n, t_p1)

    # Try n=240
    if t_end - time.time() > budget_s * 0.32 and n != 240:
        n2 = 240
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        t_p2 = t0 + budget_s * 0.80
        h2_opt, c5_2 = optimize_n(h2, n2, t_p2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt
            n = n2

    # Try n=180 if time remains
    if t_end - time.time() > budget_s * 0.20 and n != 180:
        n3 = 180
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = project_h(h3)
        t_p3 = t0 + budget_s * 0.94
        h3_opt, c5_3 = optimize_n(h3, n3, t_p3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3_opt
            n = n3

    # Final polish
    if t_end - time.time() > 10:
        hc = slsqp(best_h, 1e8, len(best_h), 1200, 1e-16, topk=1)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    if t_end - time.time() > 6:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t_end - 1, 80, 25, 1e-15)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
