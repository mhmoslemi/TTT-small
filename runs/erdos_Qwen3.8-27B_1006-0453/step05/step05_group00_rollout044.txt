import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    # D[i, idx] = change in corr[idx] when flipping h[i]
    # corr[idx] corresponds to lag L = idx - (N-1)
    # D[i, L] = (1-h[i+L]) - h[i-L] (with 0 for out of bounds)
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)  # -（N-1) to (N-1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pp = pos + sv  # i + L
    pm = pos - sv  # i - L
    vp = (pp >= 0) & (pp < N)
    vm = (pm >= 0) & (pm < N)
    pp_c = np.clip(pp, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    return (1.0 - np.where(vp, h[pp_c], 0.0)) - np.where(vm, h[pm_c], 0.0)

def greedy_swap(h, N, deadline, max_sweeps=1000, sample_n=150, top_k=30):

    if nh != N_half:
        # Fix to binary with correct count

        nh = int(h.sum())
        if nh > N_half:
            ones = np.where(h == 1)[0]
            remove = ones[N_half:]
            h[remove] = 0
        elif nh < N_half:
            zeros = np.where(h == 0)[0]
            add = zeros[:N_half - nh]
            h[add] = 1
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        # Sample
        if len(ones) > sample_n:
            p_list = np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]

def anneal_swap(h, N, deadline, max_sweeps=500, sample_n=100, T0=0.05, T1=0.001):

            q = int(zeros[np.argmin(cand_max)])

            h[p] = 0.0
            h[q] = 1.0

def nesterov_opt(h, N, deadline, k=60, lr0=0.02):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.99
            beta = min(0.97, 0.88 + it * 0.0003)
    return best_h, best_pk * dx

def make_random_binary(N, rng):
    h = np.zeros(N)
    ones_positions = rng.choice(N, N//2, replace=False)
    h[ones_positions] = 1

def make_block_pattern(N, block_len, rng):
    h = np.zeros(N)
    pos = 0
    while pos < N:
        # Random gap then block
        block = min(block_len, N - pos)
        h[pos:pos+block] = 1
        gap = rng.randint(1, block_len + 1)
        pos += block + gap
    # Fix sum
    h = (h > 0.5).astype(float)
    nh = int(h.sum())
    N_half = N // 2
    if nh > N_half:
        ones = np.where(h == 1)[0]
        h[ones[N_half:]] = 0
    elif nh < N_half:
        zeros = np.where(h == 0)[0]
        h[zeros[:N_half - nh]] = 1
    return h

    if time.time() < deadline - 100:
        d4 = min(deadline - 80, time.time() + 70)
        h_g, _ = greedy_swap(best_h, N0, d4, max_sweeps=500, sample_n=150)
        c5_g = compute_c5(h_g, N0)
        if c5_g < best_c5:
            best_c5, best_h = c5_g, h_g.copy()
    
    # Phase 5: Block moves
    if time.time() < deadline - 60:
        d5 = min(deadline - 50, time.time() + 40)
        h_b, _ = block_move(best_h, N0, d5)
        c5_b = compute_c5(h_b, N0)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b.copy()
    
    # Phase 6: Nesterov continuous polish
    if time.time() < deadline - 20:
        hF = best_h.astype(float) + np.random.randn(len(best_h)) * 0.05
        hF = project_sum(hF, len(best_h) / 2.0)
        d6 = min(deadline - 5, time.time() + 50)
        hF, c5F = nesterov_opt(hF, len(best_h), d6, k=80, lr0=0.015)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()
    
    # Phase 7: Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = (best_h > 0.5).astype(float)
        hF2, _ = greedy_swap(hF2, len(best_h), deadline - 0.5, max_sweeps=100, sample_n=150)
        c5F2 = compute_c5(hF2, len(best_h))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2
    
    # Phase 8: Try different N
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [400, 500, 600]:
        if time.time() > deadline - 5:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        nh = N_try // 2
        n1 = int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            perm = rng.permutation(len(idx))
            h_bt[idx[perm[nh:]]] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)
