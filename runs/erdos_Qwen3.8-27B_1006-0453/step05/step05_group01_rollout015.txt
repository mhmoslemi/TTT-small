```python
def softmax_smoothed_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    n = len(h)
    dx = 2.0 / n

def nesterov_opt(h, n, t_end, n_iter=200, T_start=100, T_end=50000):
    """Nesterov momentum optimization with softmax-smoothed objective."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    curr_h = h.copy()
    v = np.zeros(n)
    beta = 0.9  # momentum
    lr = 0.003
    no_improve = 0
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / n_iter
        T = T_start + (T_end - T_start) * (1 - cool)  # T increases
        # Nesterov lookahead
        y = curr_h + v * beta
        y = project_h(y)
        grad, _ = full_gradient(y, T)
        # Adaptive step
        step = lr * (0.5 + 0.5 * cool)
        new_h = project_h(curr_h - step * grad)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5 - 1e-15:
            v = beta * v - step * grad
            curr_h = new_h
            best_c5 = new_c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.7
                v *= 0.5
        # Occasionally reproject
        if it % 20 == 0:
            curr_h = project_h(curr_h)
    
    return best_h if compute_c5(best_h) < compute_c5(curr_h) else curr_h, min(best_c5, compute_c5(curr_h))

def coord_descent_v2(h, n, t_end, max_sweeps=50, n_top=30):
    """Coordinate descent focusing on top shifts."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            # Compute base and coefficients
            coefs = np.zeros(n_top)
            base_vals = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                jp = i + k
                jm = i - k
                c = 0.0
                val = 0.0
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                    val += best_h[i] * (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                    val += (1.0 - best_h[jm]) * best_h[i]
                # corr[top_idx[t]] = sum of h[j]*(1-h[j+shift]) over valid j
                # When h[i] changes to t:
                # contribution from h[i]*(1-h[i+shift]): t*(1-h[jp])
                # contribution from (1-h[i-shift])*h[i]: (1-h[jm])*t
                # So new_correlation = old - old_h[i]*coef + t*coef
                coefs[t] = c
                base_vals[t] = corr[top_idx[t]] - best_h[i] * c
            
            # Minimize max(base + t*coef) over t in [0,1]
            # This is a 1D convex optimization
            res = minimize_scalar(
                lambda t: np.max(base_vals + t * coefs),
                bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5
```

Actually, for the coordinate descent, the key is: when we change h[i] from its current value to a new value t, the correlation values change as:
new_corr[k] = old_corr[k] - h[i] * coef_k(i) + t * coef_k(i)

where coef_k(i) = (1-h[i+k]*I[valid]) - h[i-k]*I[valid]

And we want to minimize max_k new_corr[k].

This is a 1D minimax problem for each i. The optimal t is where the maximum of the linear functions (base_k + t*coef_k) is minimized. This can be solved exactly:
- For each k, the line is base_k + t*coef_k
- We want t ∈ [0,1] that minimizes max_k (base_k + t*coef_k)
- This is a linear programming problem in 1D, solvable by checking the intersection points of pairs of lines, or by bisection

The scipy minimize_scalar with bounded method should work, but for 30+ shifts, it might not be precise enough. Let me implement a more exact solver.

For 1D minimax of linear functions:
- The function f(t) = max_k (a_k + b_k * t) is convex piecewise-linear in t
- The minimum is at a point where two (or more) lines cross and are the maximum
- We can find it by: the minimum is at the smallest t where f(t) is minimized
- Algorithm: sort the breakpoints, check intervals between breakpoints

Actually, for the bounded case [0,1], the minimum of a convex piecewise-linear function is:
- At a vertex (intersection of two lines), or
- At the boundary (t=0 or t=1)

To find it exactly:
1. At each t, the active set (lines achieving the max) changes at intersection points
2. We can binary search on the value: for a given level L, check if there exists t ∈ [0,1] such that a_k + b_k*t ≤ L for all k
   - This is: for all k with b_k > 0: t ≤ (L - a_k)/b_k
   - For all k with b_k < 0: t ≥ (L - a_k)/b_k
   - For all k with b_k = 0: need a_k ≤ L
   - So we need max{0, max_{b_k<0} (L-a_k)/b_k} ≤ min{1, min_{b_k>0} (L-a_k)/b_k} and all a_k ≤ L for b_k=0
3. Binary search on L

This is O(n_top * log(precision)) which is very fast.

Let me implement this:

```python
def solve_1d_minimax(base, coefs, lo=0.0, hi=1.0):
    """Solve min_t max_k (base_k + coef_k * t) for t in [lo, hi]."""
    n_k = len(base)
    # Binary search on the optimal value L
    # Feasibility check: exists t in [lo, hi] s.t. base_k + coef_k * t <= L for all k
    
    lo_L = np.max(base)  # lower bound on L (t doesn't help below this at t=0... actually no)
    # Actually, the minimum of max is at least max_k min_t (base_k + coef_k*t)
    # For each k: min over t in [lo,hi] of base_k + coef_k*t
    # If coef_k > 0: min at t=lo, value = base_k + coef_k*lo
    # If coef_k < 0: min at t=hi, value = base_k + coef_k*hi
    # If coef_k = 0: value = base_k
    mins = np.where(coefs > 0, base + coefs * lo, np.where(coefs < 0, base + coefs * hi, base))
    lo_L = max(np.max(mins), np.max(base[coefs == 0]) if np.any(coefs == 0) else -np.inf)
    
    hi_L = max(np.max(base + coefs * lo), np.max(base + coefs * hi))
    
    for _ in range(60):  # binary search iterations
        mid_L = (lo_L + hi_L) / 2
        # Check feasibility: exists t in [lo, hi] s.t. base_k + coef_k * t <= mid_L for all k
        t_lo = lo
        t_hi = hi
        for k in range(n_k):
            if coefs[k] > 1e-15:
                t_hi = min(t_hi, (mid_L - base[k]) / coefs[k])
            elif coefs[k] < -1e-15:
                t_lo = max(t_lo, (mid_L - base[k]) / coefs[k])
            else:
                if base[k] > mid_L:
                    return t_lo  # infeasible, will converge
        if t_lo <= t_hi:
            hi_L = mid_L
            t_star = (t_lo + t_hi) / 2
        else:
            lo_L = mid_L
            t_star = (lo + hi) / 2
    
    # Retrieve t_star
    t_lo = lo
    t_hi = hi
    for k in range(n_k):
        if coefs[k] > 1e-15:
            t_hi = min(t_hi, (hi_L - base[k]) / coefs[k])
        elif coefs[k] < -1e-15:
            t_lo = max(t_lo, (hi_L - base[k]) / coefs[k])
    t_star = (t_lo + t_hi) / 2
    t_star = max(lo, min(hi, t_star))
    return t_star
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def full_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    w = np.exp(T * (corr - cm))
    w /= w.sum()
    shifts = np.arange(len(corr)) - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx - s_idx
    jm = i_idx + s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    g1 = (w[None, :] * v2).sum(axis=1)
    g2 = (w[None, :] * h[jmc] * v2).sum(axis=1)
    g3 = (w[None, :] * h[jpc] * v1).sum(axis=1)
    return (g1 - g2 - g3) * dx

def nesterov_opt(h, n, t_end, n_iter=300, T_start=50, T_end=100000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    curr_h = h.copy()
    v = np.zeros(n)
    beta = 0.85
    lr = 0.004
    no_improve = 0
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        T = T_start + (T_end - T_start) * (1.0 - cool)
        y = project_h(curr_h + v * beta)
        grad = full_gradient(y, T)
        step = lr * (0.3 + 0.7 * cool)
        new_h = project_h(curr_h - step * grad)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5 - 1e-15:
            v = beta * v - step * grad
            curr_h = new_h
            best_c5 = new_c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.65
                v *= 0.4
    final_c5 = compute_c5(curr_h)
    if final_c5 < best_c5:
        return curr_h, final_c5
    return best_h, best_c5

def solve_1d_minimax(base, coefs, lo=0.0, hi=1.0):
    n_k = len(base)
    lo_L = np.max(base)
    hi_L = np.max(np.maximum(base + coefs * lo, base + coefs * hi))
    t_star = 0.5
    for _ in range(50):
        mid_L = (lo_L + hi_L) / 2.0
        t_lo = lo
        t_hi = hi
        feasible = True
        for k in range(n_k):
            c = coefs[k]
            if c > 1e-14:
                t_hi_val = (mid_L - base[k]) / c
                if t_hi_val < t_hi:
                    t_hi = t_hi_val
            elif c < -1e-14:
                t_lo_val = (mid_L - base[k]) / c
                if t_lo_val > t_lo:
                    t_lo = t_lo_val
            else:
                if base[k] > mid_L:
                    feasible = False
                    break
        if feasible and t_lo <= t_hi:
            hi_L = mid_L
            t_star = (t_lo + t_hi) / 2.0
        else:
            lo_L = mid_L
    t_lo = lo
    t_hi = hi
    for k in range(n_k):
        c = coefs[k]
        if c > 1e-14:
            t_hi = min(t_hi, (hi_L - base[k]) / c)
        elif c < -1e-14:
            t_lo = max(t_lo, (hi_L - base[k]) / c)
    t_star = max(lo, min(hi, (t_lo + t_hi) / 2.0))
    return t_star

def coord_descent(h, n, t_end, max_sweeps=60, n_top=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            base = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
                base[t] = corr[top_idx[t]] - best_h[i] * c
            t_new = solve_1d_minimax(base, coefs, 0.0, 1.0)
            if abs(t_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_refine(h, n, T, maxiter=600):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        dx = 2.0 / n
        if T < 1e8:
            val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        else:
            val = m
        return val * dx
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def optimize_round(h, n, t_end, tag=""):
    best_h = project_h(h)
    best_c5 = compute_c5(best_h)
    # Nesterov phase
    if t_end - time.time() > 40:
        t_sub = min(t_end - 20, time.time() + (t_end - time.time()) * 0.5)
        h1, c5_1 = nesterov_opt(best_h, n, t_sub, 200, 50, 80000)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1
    # SLSQP
    if t_end - time.time() > 30:
        h2 = slsqp_refine(best_h, n, 5000, 600)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
    # Coordinate descent
    if t_end - time.time() > 20:
        best_h, best_c5 = coord_descent(best_h, n, t_end, 40, 30)
    return best_h, best_c5
```
