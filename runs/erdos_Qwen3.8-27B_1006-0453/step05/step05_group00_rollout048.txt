        # Change for h[p]: 1→0 at shift s: -(1-h[p+s]) + h[p-s] (from before)
        # Change for h[q]: 0→1 at shift s: (1-h[q+s]) - h[q-s]
        # Total swap change = [-(1-h[p+s]) + h[p-s]] + [(1-h[q+s]) - h[q-s]]
        
        # Using the D matrix: if D represents 0→1 change:
        # swap_change = -D[p,:] + D[q,:]
        # new_c = c - D[p,:] + D[q,:]
        
        # For speed, vectorize over zeros for each p:
        best_new_max = c.max() + 1.0
        best_p, best_q = -1, -1
        best_cand = None
        
        for p in ones:
            base = c - D[p, :]
            cands = base[None, :] + D[zeros, :]
            new_maxes = cands.max(axis=1)
            mi = int(np.argmin(new_maxes))
            if new_maxes[mi] < best_new_max:
                best_new_max = float(new_maxes[mi])
                best_p = p
                best_q = zeros[mi]
                best_cand = cands[mi]
        
        if best_p < 0:
            break
        
        cur_max = float(c.max())
        delta = best_new_max - cur_max
        accept = False
        if delta <= 0:
            accept = True
            stuck = 0
        else:
            if delta < T:
                accept = True
                stuck += 1
            else:
                stuck += 1
                if np.random.random() < np.exp(-delta / max(T, 1e-10)):
                    accept = True
        
        if not accept and stuck > 100:
            # Random restart-like move
            p = int(ones[np.random.randint(len(ones))])
            best_q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[best_q] = 0.0, 1.0
            c = c - D[p, :] + D[best_q, :]
            T = max(Tc, T * 0.95)
            continue
        
        h[best_p], h[best_q] = 0.0, 1.0
        c = best_cand if accept and delta <= 0 else c - D[best_p, :] + D[best_q, :]
        
        if c.max() < best_c:
            best_c = float(c.max())
            best_h = h.copy()
            stuck = 0
        T = max(Tc, T * 0.997)
    
    return best_h, best_c * (2.0/N)

The key bottleneck is: the greedy binary search converges to a local minimum around 0.380893. To go below 0.38085, we need to either:
1. Find a better local minimum (requires escaping the current one)
2. Use a different discretization (different N)
3. Use a continuous solution that happens to have a better binary rounding

Let me think about option 3. If we allow h ∈ [0,1] (not just {0,1}), the problem becomes:
minimize max_k c_k subject to 0 ≤ h[i] ≤ 1, Σh[i] = N/2

This is a nonlinear (bilinear) optimization. But we can use subgradient methods.

The subgradient of c_k with respect to h[i] is:
∂c_k/∂h[i] = (1-h[i+k]) + h[i-k] - 2*h[i]*(1-h[i+k]) - h[i-k]*h[i] ... 

Actually for bilinear: c_k = Σ_j h[j](1-h[j+k])
∂c_k/∂h[i] = (1-h[i+k]) + [if i-j-k=0 for some j, i.e., j=i-k] h[i-k]
Wait: c_k = Σ_j h[j] - Σ_j h[j]h[j+k]
∂c_k/∂h[i] = 1 - h[i+k] - h[i-k] (when the terms exist)

So the subgradient of max_k c_k is the subgradient of the active (maximizing) c_k.

For proximal gradient:
h_new[i] = clip(h[i] - lr * (1 - h[i+k*] - h[i-k*]), 0, 1)
where k* is the maximizing shift.

Then project back to Σh = N/2.

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.rfft(h, L)
    G = np.fft.rfft(np.flip(1.0 - h), L)
    return np.fft.irfft(H * G, L)[:L]

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-9:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None].astype(np.int64)
    sv = shifts[None, :].astype(np.int64)
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm).astype(float) * vm.astype(float) + hp.astype(float) * vp.astype(float)

def binary_anneal(h, N, deadline, T0=0.008, T_floor=0.0005, cool=0.9985, max_steps=5000):
    h = h.astype(float).copy()
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    T = T0
    no_improve = 0
    step = 0
    while time.time() < deadline and step < max_steps:
        step += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        cur_max = float(c.max())
        best_delta = 1e9
        best_p, best_q = -1, -1
        best_cand = None
        for p in ones:
            base = c - D[p, :]
            cands = base[None, :] + D[zeros, :]
            nm = cands.max(axis=1)
            mi = int(np.argmin(nm))
            if nm[mi] - cur_max < best_delta:
                best_delta = float(nm[mi] - cur_max)
                best_p, best_q = int(p), int(zeros[mi])
                best_cand = cands[mi]
        if best_p < 0:
            break
        if best_delta <= 0:
            accept = True
            no_improve = 0
        else:
            if T > 1e-6 and best_delta < 3.0 * T:
                accept = np.random.random() < np.exp(-best_delta / T)
            else:
                accept = False
            if not accept:
                no_improve += 1
                if no_improve > 200:
                    # Random shuffle escape
                    kp = int(np.random.choice(len(ones)))
                    kq = int(np.random.choice(len(zeros)))
                    h[ones[kp]], h[zeros[kq]] = 0.0, 1.0
                    c = c - D[ones[kp], :] + D[zeros[kq], :]
                    T = max(T_floor, T * 1.5)
                    no_improve = 0
                    continue
        if accept:
            h[best_p], h[best_q] = 0.0, 1.0
            c = best_cand.copy()
            if c.max() < best_c:
                best_c = float(c.max())
                best_h = h.copy()
        T = max(T_floor, T * cool)
    return best_h, best_c * (2.0 / N)

def subgrad_opt(h, N, deadline, lr0=0.03, k_top=40):
    h = h.astype(float).copy()
    target = N / 2.0
    h = project_sum(h, target)
    best_pk = float(np.max(fft_corr(h)))
    best_h = h.copy()
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        pk = float(c.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(c, -k_top)[-k_top:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad[:] += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k_top
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 300 == 0:
            lr *= 0.99
            beta = min(0.97, 0.85 + it * 0.0003)
    return best_h, best_pk * (2.0 / N)
