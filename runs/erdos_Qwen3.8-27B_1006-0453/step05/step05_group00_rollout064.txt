def fft_corr(h):

    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):

            lo = mid
        else:
            hi = mid

def precompute_deltas(h):
    """Precompute delta_p matrix: delta_p[p, s] = change in corr at lag s if h[p] goes from 1 to 0."""
    N = len(h)
    L = 2 * N - 1
    s = np.arange(-(N-1), N)  # (2N-1,)
    
    N_idx = np.arange(N)[:, None]  # (N,1)
    S = s[None, :]  # (1, 2N-1)

    return delta_p  # (N, 2N-1)

import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def precompute_deltas(h):
    N = len(h)
    L = 2 * N - 1
    N_idx = np.arange(N)[:, None].astype(np.int64)
    S = np.arange(-(N-1), N)[None, :].astype(np.int64)
    PS = N_idx + S
    PS_valid = ((PS >= 0) & (PS < N)).astype(np.float64)
    PSM = N_idx - S
    PSM_valid = ((PSM >= 0) & (PSM < N)).astype(np.float64)
    delta_p = h[np.clip(PS, 0, N-1)] * PS_valid - (1.0 - h[np.clip(PSM, 0, N-1)]) * PSM_valid
    return delta_p

def update_delta(h, p, q, delta_p, N):
    """Update delta_p after h[p]:1->0 and h[q]:0->1"""
    s_vals = np.arange(-(N-1), N)
    col_off = N - 1
    
    # h[p]: 1->0
    m1 = (p - s_vals >= 0) & (p - s_vals < N)
    if m1.any():
        delta_p[p - s_vals[m1], s_vals[m1] + col_off] -= 1
    m2 = (p + s_vals >= 0) & (p + s_vals < N)
    if m2.any():
        delta_p[p + s_vals[m2], s_vals[m2] + col_off] -= 1
    
    # h[q]: 0->1
    m3 = (q - s_vals >= 0) & (q - s_vals < N)
    if m3.any():
        delta_p[q - s_vals[m3], s_vals[m3] + col_off] += 1
    m4 = (q + s_vals >= 0) & (q + s_vals < N)
    if m4.any():
        delta_p[q + s_vals[m4], s_vals[m4] + col_off] += 1

def swap_local_search(h, deadline, K=25, sample=100, T0=0.3, cool=0.995):
    N = len(h)
    h = h.astype(float).copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    delta_p = precompute_deltas(h)
    corr = fft_corr(h)
    best_h = h.copy()
    best_c = float(corr.max())
    T = T0
    no_improve = 0
    
    while time.time() < deadline:
        # Update top K
        top_k = np.argpartition(corr, -K)[-K:]
        c_max = float(corr.max())
        
        if c_max < best_c - 1e-12:
            best_c = c_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Re-randomize some
            n_rand = min(10, len(ones) // 2)
            rp = ones[np.random.randint(len(ones), n_rand)]
            rq = zeros[np.random.randint(len(zeros), n_rand)]

                corr += delta_p[rp[i]] - delta_p[rq[i]]

        q_list = zeros[np.random.choice(len(zeros), min(sample, len(zeros)), replace=False)] if len(zeros) > sample else zeros

        best_pq = None
        
        for p in p_list:

            dq = delta_p[q_list, top_k]
            nv = (base[None, :] - dq).max(axis=1)
            mi = int(np.argmin(nv))
            if nv[mi] < c_max - best_gain:

            h[p], h[q] = 0.0, 1.0
            corr += delta_p[p] - delta_p[q]
            update_delta(h, p, q, delta_p, N)
            ones = np.delete(ones, np.where(ones == p)[0][0])
            zeros = np.insert(zeros, len(zeros), p)
            zeros = np.delete(zeros, np.where(zeros == q)[0][0])
            ones = np.insert(ones, len(ones), q)

            if np.random.random() < np.exp(-(new_max - c_max) / max(T, 1e-8)):
                h[p], h[q] = 0.0, 1.0
                corr += delta_p[p] - delta_p[q]

                ones = np.delete(ones, np.where(ones == p)[0][0])
                zeros = np.insert(zeros, len(zeros), p)
                zeros = np.delete(zeros, np.where(zeros == q)[0][0])

    if final_c * N / 2.0 < best_c:
        best_h = h.copy()
        best_c = final_c * N / 2.0
    return best_h, best_c * 2.0 / N

def nesterov_polish(h, N, deadline, K=60, lr=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_pk = float(fft_corr(h).max())
    m = np.zeros(N)
    beta = 0.9
    beta_inc = 0.0003
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -K)[-K:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= K
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 100 == 0:
            lr *= 0.995
            beta = min(0.98, 0.85 + it * beta_inc)
    return best_h, best_pk * 2.0 / N
