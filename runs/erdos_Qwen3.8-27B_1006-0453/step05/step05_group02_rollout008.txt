    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def proj_grad_descent(h, T, n, steps=5000, lr=None):
    """Projected gradient descent with backtracking line search."""
    best_h = h.copy()
    best_f = smooth_c5(best_h, T)
    h_curr = h.copy()
    
    for step in range(steps):
        if time.time() > t_end_global:
            break
        grad = compute_grad_smooth(h_curr, T)
        # Project gradient to feasible direction (remove mean component for sum constraint)
        grad_proj = grad - grad.mean()
        # Also handle bound constraints: if h[i]=0, grad should be >= 0 for descent
        # (we only want to increase h[i]), similarly for h[i]=1
        if h_curr >= 1.0 - 1e-10:
            grad_proj[h_curr >= 1.0 - 1e-10] = max(0, grad_proj[h_curr >= 1.0 - 1e-10])
        if h_curr <= 1e-10:
            grad_proj[h_curr <= 1e-10] = min(0, grad_proj[h_curr <= 1e-10])
        
        if np.max(np.abs(grad_proj)) < 1e-10:
            break
        
        # Line search
        g_norm = np.linalg.norm(grad_proj)
        alpha = lr if lr else 1.0 / (T * g_norm)
        for ls in range(20):
            h_try = project_h(h_curr - alpha * grad_proj)
            f_try = smooth_c5(h_try, T)
            if f_try < best_f - 1e-14:
                best_f = f_try
                best_h = h_try.copy()
            if f_try <= smooth_c5(h_curr, T) - 1e-4 * alpha * g_norm * g_norm:
                h_curr = h_try
                break
            alpha *= 0.5
        else:
            h_curr = h_try  # accept anyway
            
    # Also track actual C5
    actual_c5 = compute_c5(best_h)
    if actual_c5 < compute_c5(h):
        return best_h, actual_c5
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    W = np.exp(shifted)
    W /= W.sum()
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for j in range(n):
        i1 = j - lags
        v1 = (i1 >= 0) & (i1 < n)
        i2 = j + lags
        v2 = (i2 >= 0) & (i2 < n)
        grad[j] = np.dot(W[v1], 1.0 - h[i1[v1]]) - np.dot(W[v2], h[i2[v2]])
    return dx * grad
