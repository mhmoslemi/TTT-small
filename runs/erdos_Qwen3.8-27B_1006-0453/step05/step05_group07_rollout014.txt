import time

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _obj1d(u, a, b):
    return float(np.max(a + u * b))

def _sm_obj(x, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def _sm_grad(x, T):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    grad = np.zeros(n)
    mf = 2 * n - 1
    ka = np.arange(mf)
    for k in range(mf):
        wk = w[k]
        if wk < 1e-15:
            continue
        lo1 = max(0, k - n + 1)
        hi1 = min(n, k + 1)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) - k + n - 1
            grad[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, n - k)
        hi2 = min(n, 2 * n - 1 - k)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) + k - n + 1
            grad[lo2:hi2] -= wk * x[idx]
    return grad * (2.0 / n)

def lbfgs_step(h, T, n, iters=300):
    try:
        r = minimize(h, _sm_obj, _sm_grad, method='L-BFGS-B',
                     args=(T,),
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-8})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_descent(h, n, t_end, sweeps=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    ka = np.arange(mf)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = 0
        corr_c = compute_corr(best)
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            idx1 = i - ka + n - 1
            idx2 = i + ka - n + 1
            coeff = np.zeros(mf)
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            coeff[m1] += 1.0 - best[idx1[m1]]
            coeff[m2] -= best[idx2[m2]]
            a_arr = corr_c - old * coeff
            b_arr = coeff
            r = minimize_scalar(_obj1d, args=(a_arr, b_arr),
                               bounds=(0.0, 1.0), method='bounded',
                               options={'xatol': 1e-10})
            nv = float(r.x)
            if abs(nv - old) < 1e-11:

            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr_c = compute_corr(best)
                imp += 1
        if imp == 0:
            break
    return best, best_c5
