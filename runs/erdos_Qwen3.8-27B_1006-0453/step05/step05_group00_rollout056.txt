def project_binary(h, target_ones):
    """Project to binary with exact count of ones."""
    h = (h > 0.5).astype(float)
    n1 = int(h.sum())
    if n1 == target_ones:
        return h
    if n1 > target_ones:
        ones = np.where(h == 1)[0]
        # Remove (n1 - target_ones) ones with largest values
        # But they're all 1, so just pick random
        to_remove = np.random.choice(ones, n1 - target_ones, replace=False)
        h[to_remove] = 0
    else:
        zeros = np.where(h == 0)[0]
        to_add = np.random.choice(zeros, target_ones - n1, replace=False)
        h[to_add] = 1
    return h
