import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i, niter=25):
    h0 = h.copy(); h0[i] = 0.0
    h1 = h.copy(); h1[i] = 1.0
    c0 = get_corr(h0)
    c1 = get_corr(h1)
    d = c1 - c0
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(c0 + m1 * d) < np.max(c0 + m2 * d):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def full_cd(h, n, t_end, max_swep=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sw in range(max_swep):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                break
            ov = coord_opt(best_h, n, i)
            if abs(ov - best_h[i]) < 1e-13:

            ht = best_h.copy(); ht[i] = ov
            ht = project_h(ht, n)

            if c5t < best_c5 - 1e-15:

        if not imp:

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, n, i):
    h0 = h.copy(); h0[i] = 0.0
    c0 = get_corr(h0)
    h1 = h.copy(); h1[i] = 1.0
    c1 = get_corr(h1)
    d = c1 - c0
    lo, hi = 0.0, 1.0
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(c0 + m1 * d) < np.max(c0 + m2 * d):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def full_cd(h, n, t_end, max_swep=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sw in range(max_swep):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            ov = coord_opt(best_h, n, i)
            if abs(ov - best_h[i]) < 1e-14:
                continue
            ht = best_h.copy(); ht[i] = ov
            ht = project_h(ht, n)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_h = ht; best_c5 = c5t; imp = True
        if not imp:
            break
    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=2000):
    def obj(x):
        corr = get_corr(x)
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        return val * (2.0 / n)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def basin_hop(h, n, t_end, n_iter=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = h.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * it / max(n_iter, 1)
        mod = it % 8
        if mod == 0:

        elif mod == 1:

            ph = np.random.uniform(0, 2 * np.pi)
            x = np.arange(n) / n

        elif mod == 2:

        elif mod == 3:

            b = np.random.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            pert[b*blk:min(n,(b+1)*blk)] = (0.02+0.07*cool) * np.random.randn(blk)
        elif mod == 4:
            pert = np.zeros(n)
            for _ in range(4):
                a, b2 = np.random.permutation(n)[:2]
                d = (best_h[b2]-best_h[a])*(0.4+0.6*cool)
                pert[a] += d; pert[b2] -= d
        elif mod == 5:

        elif mod == 6:
            pt = np.zeros(n)

        hc, cc = full_cd(hp, n, min(time.time()+3, t_end-2), max_swep=8)

        if cc < work_c5 or np.random.rand() < np.exp(-(cc-work_c5)/temp):
            work_h = hc; work_c5 = cc
