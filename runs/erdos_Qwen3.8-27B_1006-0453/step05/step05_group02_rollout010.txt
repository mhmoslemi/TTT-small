    """Sum of correlations for active lags"""
    dx = 2.0/n
    total = 0.0
    for k in active_k:
        if k >= 0:
            corr = np.sum(h[:n-k] * (1 - h[k:]))
        else:
            corr = np.sum(h[-k:] * (1 - h[:n+k]))
        total += corr
    return total * dx

def active_set_grad(h, active_k, n):

    dx = 2.0/n
    grad = np.zeros(n)
    for k in active_k:
        if k >= 0:

                if j < n - k:

                if j >= k:
                    g -= h[j-k]

        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.04 + 0.12 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 15))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(4, max(5, n // 8))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.08 + 0.25 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.1 * cool * rng.standard_normal(half)
        p[half:] = -0.1 * cool * rng.standard_normal(n - half)
    return p

def binarize_push(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng_local = np.random.RandomState(123)
    for it in range(300):
        if time.time() > t_end:
            break
        idx = rng_local.randint(n)
        if best_h[idx] < 0.1 or best_h[idx] > 0.9:
            continue
        flip_to = 0.0 if best_h[idx] > 0.5 else 1.0
        h_try = best_h.copy()
        h_try[idx] = flip_to
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation (moderate time)
    for T in [50, 200, 800, 3000, 12000, 50000, 200000]:
        if elapsed() > budget_s * 0.08:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=800, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent (main workhorse)
    t_cd1 = t0 + budget_s * 0.25
    best_h, best_c5 = coord_descent(best_h, n, t_cd1, max_sweeps=300, rng=rng)

    # Phase 3: Basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 40
    for it in range(n_iter):
        if elapsed() > budget_s * 0.40:
            break
        cool = 1.0 - 0.8 * (it / n_iter)
        pert = make_pert(n, cool, rng)
        h_pert = project_h(work_h + pert)
        T_bh = 300 + 12000 * cool
        hc = slsqp_smooth(h_pert, T_bh, n, maxiter=500, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.0002
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: More coordinate descent
    t_cd2 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, n, t_cd2, max_sweeps=250, rng=rng)

    # Phase 5: Pair coordinate descent
    t_pc = t0 + budget_s * 0.62
    best_h, best_c5 = pair_coord(best_h, n, t_pc, max_sweeps=40, rng=rng)

    # Phase 6: Binarization push + coord
    t_bin = t0 + budget_s * 0.68
    best_h, best_c5 = binarize_push(best_h, n, t_bin)
    t_cd3 = t0 + budget_s * 0.75
    best_h, best_c5 = coord_descent(best_h, n, t_cd3, max_sweeps=200, rng=rng)

    # Phase 7: Finer basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(30):
        if elapsed() > budget_s * 0.82:
            break
        cool = 1.0 - 0.7 * (it / 30)
        pert = make_pert(n, cool * 0.4, rng)
        h_pert = project_h(work_h + pert)
        T_bh = 5000 + 15000 * cool
        hc = slsqp_smooth(h_pert, T_bh, n, maxiter=400, ftol=1e-15)
        c5v = compute_c5(hc)
        temp = 0.0005 * cool + 0.0001
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 8: Final coordinate descent + SLSQP polish
    t_cd4 = t0 + budget_s * 0.90
    best_h, best_c5 = coord_descent(best_h, n, t_cd4, max_sweeps=200, rng=rng)

    for T in [500000, 2000000, 8000000]:
        if elapsed() > budget_s * 0.95:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=1500, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Final micro-polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.985, max_sweeps=80, rng=rng)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
