import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def solve_minimax_1d(base, coefs):
    n = len(coefs)
    pts = [0.0, 1.0]
    for i in range(n):
        for j in range(i + 1, n):
            d = coefs[i] - coefs[j]
            if abs(d) > 1e-13:
                x = (base[j] - base[i]) / d
                if -1e-9 <= x <= 1.0 + 1e-9:
                    pts.append(max(0.0, min(1.0, x)))
    cands = np.array(pts)
    vals = np.max(base[None, :] + cands[:, None] * coefs[None, :], axis=1)
    return cands[np.argmin(vals)]

def get_coefs(h, n, i, shifts):
    coefs = np.zeros(len(shifts))
    for t in range(len(shifts)):
        k = shifts[t]
        c = 0.0
        jm = i - k
        jp = i + k
        if 0 <= jm < n:
            c += 1.0 - h[jm]
        if 0 <= jp < n:
            c -= h[jp]
        coefs[t] = c
    return coefs

def exact_cd(h, n, t_end, n_top=25, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    sq = n - 1

        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - sq
        improved = False
        idxs = np.random.permutation(n) if sweep > 0 else np.arange(n)
        for i in idxs:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = get_coefs(best_h, n, i, top_shifts)
            base = corr[top_idx] - best_h[i] * coefs
            x_new = solve_minimax_1d(base, coefs)
            if abs(x_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = x_new
            h_try = project_h(h_try)

                best_h = h_try
                improved = True
        if not improved:

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ec = np.exp(corr - m)
    sw = ec.sum()
    val = (m + np.log(sw)) / T
    w = ec / sw * dx
    sq = n - 1
    grad = np.zeros(n)
    ar = np.arange(n)
    for s_idx in range(2 * n - 1):
        if w[s_idx] < 1e-12:
            continue
        k = s_idx - sq
        wi = w[s_idx]
        jm = ar - k
        jp = ar + k
        vm = (jm >= 0) & (jm < n)
        vp = (jp >= 0) & (jp < n)
        g = np.zeros(n)
        g[vm] += 1.0 - h[jm[vm]]
        g[vp] -= h[jp[vp]]
        grad += wi * g
    return val, grad

def slsqp_solve(h, T, n, maxiter=600):
    def obj_fun(x):
        v, _ = smooth_obj_grad(x, T)
        return v
    def obj_grad(x):
        _, g = smooth_obj_grad(x, T)
        return g
    try:
        res = minimize(obj_fun, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

        best_c5 = bh2_c5
        best_h = bh2_h
    best_h, best_c5 = exact_cd(best_h, n, t0 + budget_s * 0.98, n_top=40, max_sweeps=150)

    # Final SLSQP polish
    if time.time() < t_end - 5:
        hc = slsqp_solve(best_h, 100000, n, 800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
