    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

        diff = target - s

            break

            h *= (target / s)
        else:

            if room < 1e-14:
                break
            h += (1.0 - h) * (diff / room)
        h = np.clip(h, 0.0, 1.0)
    return h

    old = h[i]
    for s in range(1, n):

        mn = n - 1 - s

        if s <= i:
            lin[mn] -= h[i - s]
    lin[n - 1] = 1.0 - 2.0 * old
    return lin

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15 * n:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = (1.0 - h).sum()
            if room < 1e-14:
                break
            h = h + (1.0 - h) * (diff / room)
        h = np.clip(h, 0.0, 1.0)
    return h


def build_lin(h, n, i):
    lin = np.zeros(2 * n - 1)
    s = np.arange(1, n)
    mp = (n - 1) + s
    mn = (n - 1) - s
    m1 = s <= i
    m2 = (i + s) < n
    if m1.any():
        lin[mp[m1]] += 1.0 - h[i - s[m1]]
        lin[mn[m1]] -= h[i - s[m1]]
    if m2.any():
        lin[mp[m2]] -= h[i + s[m2]]
        lin[mn[m2]] += 1.0 - h[i + s[m2]]
    lin[n - 1] = 1.0 - 2.0 * h[i]
    return lin


def single_cd(h, n, corr, dx, t_end, max_sweeps, seed):
    rng = np.random.RandomState(seed)
    best_c5 = float(np.max(corr) * dx)
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.arange(n) if sw % 4 != 3 else rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return h, corr, best_c5
            old = h[i]
            lin = build_lin(h, n, i)
            cb = corr
            def f1d(v):
                dv = v - old
                val = cb + dv * lin
                val[n - 1] -= dv * dv
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-14})
            vn = float(res.x)
            dv = vn - old
            if abs(dv) < 1e-15:
                continue
            nc = cb + dv * lin
            nc[n - 1] -= dv * dv
            c5n = float(np.max(nc) * dx)
            if c5n < best_c5:
                h[i] = vn
                corr = nc
                best_c5 = c5n
                improved = True
        if not improved and sw > 10:
            break
    return h, corr, best_c5


def pair_cd(h, n, corr, dx, t_end, max_sweeps, seed):
    rng = np.random.RandomState(seed)
    best_c5 = float(np.max(corr) * dx)
    pairs = [(i, i + 1) for i in range(n - 1)]
    pairs += [(i, i + 2) for i in range(n - 2)]
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for idx in rng.permutation(len(pairs)):
            if time.time() > t_end:
                return h, corr, best_c5
            i, j = pairs[idx]
            oi, oj = h[i], h[j]
            li = build_lin(h, n, i)
            lj = build_lin(h, n, j)
            d = j - i
            X = np.zeros(2 * n - 1)
            X[n - 1 + d] = 1.0
            X[n - 1 - d] = 1.0
            cb = corr
            def f2d(p):
                di = p[0] - oi
                dj = p[1] - oj
                val = cb + di * li + dj * lj - di * dj * X
                val[n - 1] -= di * di + dj * dj
                return float(np.max(val) * dx)
            res = minimize(f2d, x0=[oi, oj],
                           bounds=[(0.0, 1.0), (0.0, 1.0)],
                           method='Nelder-Mead',
                           options={'xatol': 1e-13, 'fatol': 1e-15, 'maxiter': 200})
            if res.fun < best_c5 - 1e-15:
                h[i] = float(res.x[0])
                h[j] = float(res.x[1])
                corr = np.correlate(h, 1.0 - h, mode='full')
                best_c5 = float(np.max(corr) * dx)
                improved = True
        if not improved and sw > 3:
            break
    return h, corr, best_c5


def anneal_search(h_in, n, c5_in, t_end, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    best_h = h_in.copy()
    best_c5 = c5_in
    cur_h = h_in.copy()
    cur_c5 = c5_in
    n_iter = 0
    while time.time() < t_end:
        n_iter += 1
        prog = min(1.0, n_iter / 3000.0)
        step = 3e-4 * (1.0 - prog) + 5e-6
        k = rng.randint(2, 10)
        idx = rng.choice(n, size=k, replace=False)
        cur_h[idx] += step * rng.randn(k)
        cur_h = project_h(cur_h, n)
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        tr = min(t_end, time.time() + max(0.5, 5.0 * (1.0 - prog)))
        cur_h, corr, c5_new = single_cd(cur_h, n, corr, dx, tr, max_sweeps=150,
                                        seed=seed + n_iter)
        temp = max(step * 5.0, 1e-7)
        accept = c5_new < cur_c5 or (rng.random() < np.exp(-(c5_new - cur_c5) / temp))
        if accept:
            cur_h, cur_c5 = cur_h.copy(), c5_new
            if c5_new < best_c5:
                best_h, best_c5 = cur_h.copy(), c5_new
    return best_h, best_c5
