```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        frac = diff / tot
        h = h + room * frac
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, t_end):
    m = 2 * n - 1
    improved = False
    timed_out = False
    for i in range(n):
        if time.time() > t_end:
            timed_out = True
            break
        old = h[i]
        h[i] = 0.0
        base = np.correlate(h, 1.0 - h, mode='full')
        
        # Build lin vector
        lin = np.zeros(m)
        lin[n-1] = 1.0  # lag 0 linear part
        # positive lags
        for l in range(1, n):
            if i + l < n:
                lin[n-1+l] += (1.0 - h[i+l])
            if i >= l:
                lin[n-1-l] += -h[i-l]
        # negative lags
        for p in range(1, n):
            if i >= p:
                lin[n-1-p] += (1.0 - h[i-p])
            if i + p < n:
                lin[n-1+p] += -h[i+p]
        
        # 1D minimization: min_x max_k (base[k] + x*lin[k] - delta_k0 * x^2)
        # where delta_k0 = 1 if k == n-1 else 0
        # Simplify: f(x) = max(max_{k≠n-1}(base[k]+x*lin[k]), base[n-1]+x-x^2)
        
        # Precompute for speed
        base_no0 = base.copy()
        base_no0[n-1] = -np.inf  # mask out
        lin_no0 = lin.copy()
        
        def f1d(x):
            vals = base_no0 + x * lin_no0
            v0 = base[n-1] + x - x*x
            return max(np.max(vals), v0)
        
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-10})
        new_val = res.x
        if abs(new_val - old) > 1e-11:
            h[i] = new_val
            improved = True
    return h, improved, timed_out

def smooth_max_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= shifted.max()
    exps = np.exp(shifted)
    w = exps / exps.sum()
    sm_val = mc + np.log(exps.sum()) / T
    
    m = 2 * n - 1
    lag_arr = np.arange(m) - n + 1
    grad = np.zeros(n)
    for idx in range(m):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lg = lag_arr[idx]
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n/2.0)
    val = dx * sm_val + lam * (h.sum() - n/2.0)**2
    return val, grad

def lbfgs(h0, T, lam, maxiter, t_end):
    n = len(h0)
    x0 = project_h(h0)
    try:
        res = minimize(lambda x: smooth_max_grad(x, T, lam)[0],
                       x0,
                       jac=lambda x: smooth_max_grad(x, T, lam)[1],
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 5000})
        # Time check
        if time.time() < t_end:
            return project_h(res.x)
        return x0
    except:
        return x0
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    m = 2 * n - 1
```

```python
    lag_arr = np.arange(m) - n + 1
    lg = lag_arr[idx]
    grad = np.zeros(n)
    if lg >= 0:
        if lg < n:

            grad[:n-lg] += -h[lg:]
    else:
        p = -lg
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] += -h[:n-p]

        c5_new = compute_c5(h_new)
        if c5_new < f0:

        alpha *= 0.5
    return h, f0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, t_end, order=None):
    m = 2 * n - 1
    k_arr = np.arange(m)
    improved = False
    timed_out = False
    coords = order if order is not None else range(n)
    for i in coords:
        if time.time() > t_end:
            timed_out = True
            break
        old = h[i]
        h[i] = 0.0
        base = np.correlate(h, 1.0 - h, mode='full')
        idx_b = i + k_arr - n + 1
        valid_b = (idx_b >= 0) & (idx_b < n)
        idx_a = i - k_arr + n - 1
        valid_a = (idx_a >= 0) & (idx_a < n)
        lin = np.zeros(m)
        if valid_b.any():
            lin[valid_b] += (1.0 - h[idx_b[valid_b]])
        if valid_a.any():
            lin[valid_a] += -h[idx_a[valid_a]]
        b0 = base[n - 1]
        base_no0 = base.copy()
        base_no0[n - 1] = -1e10
        lo, hi = 0.0, 1.0
        for _ in range(32):
            m1 = lo + (hi - lo) * 0.381966
            m2 = hi - (hi - lo) * 0.381966
            f1 = max(np.max(base_no0 + m1 * lin), b0 + m1 - m1 * m1)
            f2 = max(np.max(base_no0 + m2 * lin), b0 + m2 - m2 * m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        new_val = (lo + hi) * 0.5
        if abs(new_val - old) > 1e-11:
            h[i] = new_val
            improved = True
    return h, improved, timed_out

def smooth_max_val_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = T * (corr - mc)
    sh -= sh.max()
    exps = np.exp(sh)
    w = exps / exps.sum()
    m = 2 * n - 1
    lag_arr = np.arange(m) - n + 1
    grad = np.zeros(n)
    for idx in range(m):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lg = lag_arr[idx]
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n - lg])
                grad[:n - lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n - p]
    grad *= dx
    sdev = h.sum() - n * 0.5
    grad += 2.0 * lam * sdev
    val = dx * (mc + np.log(exps.sum()) / T) + lam * sdev * sdev
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, t_end):
    n = len(h0)
    x0 = project_h(h0)
    try:
        res = minimize(smooth_max_val_grad, x0, args=(T, lam),
                       jac=True, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-13, 'maxfun': 4000})
        if time.time() < t_end:
            return project_h(res.x)
    except Exception:
        pass
    return x0

def subgrad_step(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    active = np.where(corr >= mc - 1e-10)[0]
    idx = int(np.random.choice(active))
    m = 2 * n - 1
    lg = idx - n + 1
    grad = np.zeros(n)
    if lg >= 0:
        if lg < n:
            grad[lg:] += (1.0 - h[:n - lg])
            grad[:n - lg] += -h[lg:]
    else:
        p = -lg
        grad[:n - p] += (1.0 - h[p:])
        grad[p:] += -h[:n - p]
    grad *= dx
    f0 = mc * dx
    alpha = 0.5
    for _ in range(25):
        h_new = project_h(h - alpha * grad)
        c5n = compute_c5(h_new)
        if c5n < f0:
            return h_new, c5n
        alpha *= 0.5
    return h, f0

    if 'initial_h_values' in globals() and initial_h_values is not None:
```
