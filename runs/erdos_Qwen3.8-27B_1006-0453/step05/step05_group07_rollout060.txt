from scipy.optimize import minimize_scalar, minimize

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    if len(ones) > n // 2:
        # Remove extras (remove those closest to 0.5)
        overs = ones[h_b[ones] < 1.0]
        if len(overs) >= len(ones) - n // 2:
            h_b[np.sort(overs, key=lambda i: h_b[i])[:len(ones) - n // 2]] = 0
        else:
            h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    return h_b

def coord_desc(h, n, t_end, sweeps=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _ in range(sweeps):

            break
        imp = False
        corr_cache = np.correlate(best, 1.0 - best, mode='full')

                return best, best_c5

                corr = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def binary_sa(h, n, t_end, max_iters=250000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    if len(ones) > n // 2:
        h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(corr.max() * dx)
    best_h = h_b.copy()
    cur_c5 = best_c5
    T = 0.028
    for it in range(max_iters):
        if time.time() > t_end:
            break
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_b[i], h_b[j] = 0.0, 1.0
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(nc.max() * dx)
        if nc5 < cur_c5 or np.random.rand() < np.exp(-(nc5 - cur_c5) / T):
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0, 0.0
        T = max(0.0002, T * 0.99998)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=40):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 3
        if mod == 0:
            pert = (0.02 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        else:
            mask = np.random.random(n) < (0.15 + 0.35 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.18 * cool) * np.random.randn(int(mask.sum()))
        hp = proj_h(work + pert, n)
        ht = lbfgs_opt(hp, 4000.0 + 6000.0 * cool, n, iters=250)
        c5t = compute_c5(ht)
        temp = 0.0006 * cool + 0.00004
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS smooth optimization with T schedule
    for T, iters in [(30, 200), (300, 350), (3000, 500), (30000, 600)]:
        if time.time() - t0 > budget_s * 0.12:
            break
        h_new = lbfgs_opt(best_h, T, n, iters)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Coordinate descent (long)
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.28, sweeps=100)

    # Phase 3: Binary SA for diversification
    h_bin, c5_bin = binary_sa(best_h, n, t0 + budget_s * 0.40, max_iters=200000)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin.copy(), c5_bin
    # Continuous polish of best binary
    hp = lbfgs_opt(best_h, 2000.0, n, 400)
    cp = compute_c5(hp)
    if cp < best_c5:
        best_h, best_c5 = hp, cp
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.50, sweeps=60)

    # Phase 4: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.72, n_iters=35)

    # Phase 5: Coord descent after basin hopping
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.80, sweeps=60)

    # Phase 6: Try other resolutions
    for n_try in [120, 200, 144]:
        if time.time() - t0 > budget_s * 0.90:
            break
        try:
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            ht = np.interp(x_new, x_old, best_h)
            ht = proj_h(ht, n_try)
            ht = lbfgs_opt(ht, 5000.0, n_try, 350)
            ht, c5t = coord_desc(ht, n_try, t0 + budget_s * 0.93, sweeps=30)
            if c5t < best_c5:
                best_h, best_c5, n = ht, c5t, n_try
        except Exception:
            pass

    # Phase 7: Final refinement
    if time.time() - t0 < budget_s * 0.94:
        best_h = lbfgs_opt(best_h, 80000.0, len(best_h), 700)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.98:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t0 + budget_s * 0.99, sweeps=40)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
