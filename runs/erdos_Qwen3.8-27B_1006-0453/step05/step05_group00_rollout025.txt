def continuous_opt(h, N, deadline, top_m=10, lr=0.01, momentum=0.9):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()

    vel = np.zeros(N)
    center = N - 1
    
    it = 0
    while time.time() < deadline:
        it += 1
        c = get_corr(h)
        cm = float(c.max())
        if cm * (2.0/N) < best_c:
            best_c = cm * (2.0/N)
            best_h = h.copy()
        
        # Top-M lags
        idx_top = np.argpartition(c, -top_m)[-top_m:]
        c_top = c[idx_top]
        # Soft weights
        w = c_top - c_top.min() + 1.0
        w = w / w.sum()
        
        # Gradient
        grad = np.zeros(N)
        for j in range(top_m):
            sh = idx_top[j] - center
            if sh == 0:
                grad += w[j] * (1.0 - 2.0 * h)
            elif sh > 0:
                if sh < N:
                    grad[:N-sh] += w[j] * (1.0 - h[sh:])
                    grad[sh:] -= w[j] * h[:N-sh]

                    grad[m:] += w[j] * (1.0 - h[:N-m])
                    grad[:N-m] -= w[j] * h[m:]

            lr *= 0.8
            vel *= 0.5
            h = h + 0.5 * vel  # smaller step
            h = project_sum(h, target)
        
        h = h_new
        # Adapt lr
        if it % 100 == 0:
            lr *= 0.98
    
    return best_h, best_c

The objective is max_s c(s). A subgradient is (1/c_max) * ∂c(s*)/∂h where s* is the argmax. To minimize, we do h ← h - lr * subgrad.

With soft-max: ∂S/∂h[j] = Σ_s w_s * ∂c(s)/∂h[j]

And ∂c(s)/∂h[j]:
- c(s) = Σ_i h[i] * (1 - h[i+shift]) where shift = s - center
- If j = i: ∂/∂h[j] = (1-h[j+shift]) [when j+shift valid]
- If j = i+shift (i = j-shift): ∂/∂h[j] of (1-h[j]) = -1, so contribution is -h[j-shift]

So ∂c(s)/∂h[j] = (1-h[j+shift])*valid(j+shift) - h[j-shift]*valid(j-shift)

To minimize: h ← h - lr * Σ_s w_s * ∂c(s)/∂h[j]

        hF, _ = binary_local_search(hF, len(best_h), deadline - 1, top_m=8, sample=100, max_stall=100)
        cF = compute_c5(hF)
        if cF < best_c5:
            best_c5 = cF
            best_h = hF
    
    final_c5 = compute_c5(best_h)
    # Round to ensure binary if close
    if np.max(np.abs(best_h - np.round(best_h))) < 0.1:
        best_h = np.round(best_h)
        best_h = ((best_h > 0.5) * 1.0)
        # Fix sum
        target = len(best_h) // 2
        n1 = int(best_h.sum())
        if n1 != target:
            idx = np.argsort(best_h)
            if n1 > target:
                best_h[idx[target:]] = 0
            else:
                best_h[idx[:n1]] = 1
                extra = target - n1
                idx2 = np.where(best_h == 0)[0]
                best_h[idx2[:extra]] = 1
    
    return best_h, final_c5, len(best_h)

import numpy as np
import time

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    return float(get_corr(h).max()) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def make_binary(h, N):
    target = N // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(N)
    hb[idx[:target]] = 1.0
    return hb

def binary_local_search(h, N, deadline, top_m=5, sample=80, max_stall=250):
    best_h = h.copy()
    best_c = float(get_corr(h).max())
    h = h.copy()
    rng = np.random
    stall = 0
    center = N - 1
    while time.time() < deadline:
        c = get_corr(h)
        cm = float(c.max())
        if cm < best_c - 1e-12:
            best_c = cm
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        idx_top = np.argpartition(c, -top_m)[-top_m:]
        shifts = idx_top - center
        c_top = c[idx_top]
        if len(ones) > sample:
            p_samples = rng.choice(ones, sample, replace=False)
        else:
            p_samples = ones.astype(int)
        q_arr = zeros.astype(int)
        qeff = np.zeros((top_m, len(q_arr)))
        for j in range(top_m):
            sh = shifts[j]
            qp = q_arr + sh
            qm = q_arr - sh
            mp_ = (qp >= 0) & (qp < N)
            mm_ = (qm >= 0) & (qm < N)
            qp_c = np.clip(qp, 0, N - 1)
            qm_c = np.clip(qm, 0, N - 1)
            qeff[j] = (1.0 - h[qp_c]) * mp_.astype(float) - h[qm_c] * mm_.astype(float)
        best_gain = 0.0
        best_pq = None
        for p in p_samples:
            peff = np.zeros(top_m)
            for j in range(top_m):
                sh = shifts[j]
                pp = p + sh
                pm = p - sh
                if 0 <= pp < N:
                    peff[j] -= (1.0 - h[pp])
                if 0 <= pm < N:
                    peff[j] += h[pm]
            vals = (c_top + peff)[:, None] + qeff
            new_max = vals.max(axis=0)
            mi = int(np.argmin(new_max))
            if new_max[mi] < cm - 1e-12:
                gain = cm - new_max[mi]
                if gain > best_gain:
                    best_gain = gain
                    best_pq = (int(p), int(q_arr[mi]))
        if best_pq is not None:
            h[best_pq[0]], h[best_pq[1]] = 0.0, 1.0
        else:
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0

    return best_h, best_c

def continuous_opt(h, N, deadline, top_m=10, lr0=0.015):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    vel = np.zeros(N)
    center = N - 1
    lr = lr0
    mom = 0.88
    dx = 2.0 / N
    it = 0
    while time.time() < deadline:
        it += 1
        c = get_corr(h)
        cm = float(c.max()) * dx
        if cm < best_c:
            best_c = cm
            best_h = h.copy()
        idx_top = np.argpartition(c, -top_m)[-top_m:]
        c_top = c[idx_top]
        w = c_top - c_top.min() + 1.0
        w = w / w.sum()
        grad = np.zeros(N)
        for j in range(top_m):
            sh = idx_top[j] - center
            if sh == 0:
                grad += w[j] * (1.0 - 2.0 * h)
            elif sh > 0 and sh < N:
                grad[:N - sh] += w[j] * (1.0 - h[sh:])
                grad[sh:] -= w[j] * h[:N - sh]
            elif sh < 0 and -sh < N:
                m = -sh
                grad[m:] += w[j] * (1.0 - h[:N - m])
                grad[:N - m] -= w[j] * h[m:]
        vel = mom * vel - lr * grad
        h_new = project_sum(h + vel, target)
        cn = get_corr(h_new)
        if cn.max() > c.max() * 1.08 + 2:
            lr *= 0.6
            vel *= 0.3
            h = project_sum(h - 0.3 * lr * grad, target)
            continue
        h = h_new
        if it % 150 == 0:
            lr *= 0.97
            mom = min(0.95, mom + 0.002)
    return best_h, best_c

def multi_resolution_search(best_h, best_c5, deadline, rng):

        h_interp = np.interp(x_new, x_old, best_h)
        h_bt = make_binary(h_interp, N_try)
        sub_dl = min(deadline - 15, time.time() + 70)
        h_rt, _ = binary_local_search(h_bt, N_try, sub_dl, top_m=5, sample=60, max_stall=150)
        c5_rt = compute_c5(h_rt)
        if c5_rt < best_c5:
            best_c5 = c5_rt
            best_h = h_rt.copy()
    return best_h, best_c5
