        else:

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D(h, N):
    idx = np.arange(2 * N - 1)
    shifts = idx - (N - 1)
    pos = np.arange(N)[:, None]
    sv = shifts[None, :]
    pm = pos - sv
    pp = pos + sv
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    pm_c = np.clip(pm, 0, N - 1)
    pp_c = np.clip(pp, 0, N - 1)
    hm = np.where(vm, h[pm_c], 0.0)
    hp = np.where(vp, h[pp_c], 0.0)
    return -(1.0 - hm) * vm.astype(float) + hp * vp.astype(float)

def binary_search(h, N, deadline, max_sweeps=500, sample_n=100, anneal_T=0.0):
    h = h.astype(float).copy()
    best_c = float(np.max(fft_corr(h)))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = fft_corr(h)
        P = float(c.max())
        if P < best_c - 1e-12:
            best_c = P
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100 and anneal_T <= 0:
                break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        D = compute_D(h, N)
        if len(ones) > sample_n:
            p_list = np.random.choice(ones, sample_n, replace=False)
        else:
            p_list = ones
        best_new_P = P
        best_pq = None
        for p in p_list:
            base = c - D[p, :]
            cand = base[None, :] - D[zeros, :]
            npk = cand.max(axis=1)
            mi = int(np.argmin(npk))
            if npk[mi] < best_new_P:
                best_new_P = float(npk[mi])
                best_pq = (int(p), int(zeros[mi]))
        if best_pq is None:
            if anneal_T > 0 and np.random.random() < anneal_T:
                p = int(ones[np.random.randint(len(ones))])
                q = int(zeros[np.random.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
            elif anneal_T > 0:
                if no_improve > 120:
                    break
            else:
                break
            continue
        h[best_pq[0]] = 0.0
        h[best_pq[1]] = 1.0
    return h, best_c

def subgradient_opt(h, N, deadline, n_active=15, lr0=0.3):
    h = h.astype(float).copy()
    target = N / 2.0
    best_h = h.copy()
    best_val = float(np.max(fft_corr(h)))
    mvel = np.zeros(N)
    beta = 0.9
    lr = lr0
    it = 0
    while time.time() < deadline:
        y = h + mvel
        c = fft_corr(y)
        c_max = float(c.max())
        top = np.argpartition(c, -n_active)[-n_active:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= n_active
        h_new = project_sum(y - lr * grad, target)
        mvel = beta * (h_new - h)
        h = h_new
        val = float(np.max(fft_corr(h)))
        if val < best_val:
            best_val = val
            best_h = h.copy()
        it += 1
        if it % 40 == 0:
            lr *= 0.994
            beta = min(0.96, 0.86 + it * 0.0004)
    return best_h, best_val * (2.0 / N)

def enforce_binary(h, N):
    h = h.copy()
    nh = N // 2
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n1 = int(h.sum())
    if n1 > nh:
        idx = np.where(h == 1)[0]
        perm = np.random.permutation(len(idx))
        h[idx[perm[nh:]]] = 0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        perm = np.random.permutation(len(idx))
        h[idx[perm[:nh - n1]]] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary local search (larger sample)
    d1 = min(deadline - 550, t0 + 130)
    h_bin = (h0 > 0.5).astype(float)
    h_s, _ = binary_search(h_bin, N0, d1, max_sweeps=1200, sample_n=250)
    c5_s = compute_c5(h_s, N0)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Continuous subgradient (main fine-tuning)
    if time.time() < deadline - 350:
        d2 = min(deadline - 300, time.time() + 180)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.05
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = subgradient_opt(hc, N0, d2, n_active=20, lr0=0.25)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary refinement from continuous
    if time.time() < deadline - 220:
        d3 = min(deadline - 180, time.time() + 90)
        h_b3 = enforce_binary(best_h, N0)
        h_s3, _ = binary_search(h_b3, N0, d3, max_sweeps=900, sample_n=300)
        c5_3 = compute_c5(h_s3, N0)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Multi-resolution exploration
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    n_list = [320, 400, 480, 560, 650]
    for N_try in n_list:
        if time.time() > deadline - 100:
            break
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h, left=0.0, right=0.0)
        h_bt = enforce_binary(h_interp, N_try)
        sub_dl = min(deadline - 60, time.time() + 30)
        h_rt, _ = binary_search(h_bt, N_try, sub_dl, max_sweeps=350, sample_n=150)
        c5_rt = compute_c5(h_rt, N_try)
        improved = False
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = x_new
            improved = True
        if time.time() < deadline - 40:
            sub_dl2 = min(deadline - 25, time.time() + 18)
            hc2 = h_rt.astype(float) + np.random.randn(N_try) * 0.03
            hc2 = project_sum(hc2, N_try / 2.0)
            hn2, c5n2 = subgradient_opt(hc2, N_try, sub_dl2, n_active=12, lr0=0.2)
            if c5n2 < best_c5:
                best_c5, best_h = c5n2, hn2.copy()
                x_best = x_new
                improved = True

    # Phase 5: Extended continuous polish at best resolution
    if time.time() < deadline - 15:
        hF = best_h.astype(float) + np.random.randn(len(best_h)) * 0.025
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = subgradient_opt(hF, len(best_h), deadline - 4, n_active=25, lr0=0.12)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF

    # Phase 6: Final binary cleanup
    if time.time() < deadline - 1:
        hF2 = enforce_binary(best_h, len(best_h))
        hF2, _ = binary_search(hF2, len(best_h), deadline - 0.5, max_sweeps=40, sample_n=250)
        c5F2 = compute_c5(hF2, len(best_h))
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
