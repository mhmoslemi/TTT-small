    target = n / 2.0
    if binary:
        # Find the closest binary solution with exactly n/2 ones

        h_new = np.zeros(n)

            return h
        h = h + room * (diff / tot)

        h = h * (1.0 + diff / s)

def build_eff(h, n):
    # h has length n
    # eff[p, m] where m = 0..2n-2, off = m - (n-1)
    # eff[p, m] = (1-h[p+off])*I(0≤p+off<n) - h[p-off]*I(0≤p-off<n)

    # p ranges 0..n-1, off ranges -(n-1)..(n-1)
    p = np.arange(n)[:, None]  # n x 1
    off = np.arange(-(n-1), n)[None, :]  # 1 x (2n-1)
    
    # p + off: n x (2n-1)
    po = p + off  # broadcast
    # p - off: n x (2n-1)
    pm = p - off
    
    mask1 = (po >= 0) & (po < n)  # valid for first term
    mask2 = (pm >= 0) & (pm < n)  # valid for second term
    
    # h[po] where valid
    h_pos = np.where(mask1, h[po], 0)  # h[p+off] where valid
    h_neg = np.where(mask2, h[pm], 0)  # h[p-off] where valid
    
    eff = (1 - h_pos) * mask1 - h_neg * mask2
    
    return eff

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_binary(h):
    n = len(h)
    idx = np.argsort(-h)
    out = np.zeros(n)
    out[idx[:n//2]] = 1.0
    return out

def build_eff(h, n):
    p = np.arange(n)[:, None]
    off = np.arange(-(n-1), n)[None, :]
    po = p + off
    pm = p - off
    m1 = (po >= 0) & (po < n)
    m2 = (pm >= 0) & (pm < n)
    hp = np.zeros_like(po, dtype=float)
    hn = np.zeros_like(pm, dtype=float)
    hp[m1] = h[po[m1]]
    hn[m2] = h[pm[m2]]
    eff = (1.0 - hp) * m1 - hn * m2
    eff[:, n-1] = 0.0
    return eff

def anneal_binary(h0, n, t_end, max_iters=200000):
    h = project_binary(h0)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_val = float(np.max(corr))
    dx = 2.0 / n
    T0, Tmin = 1.0, 0.05
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        T = T0 * (Tmin / T0) ** (min(it, max_iters) / max_iters)
        eff = build_eff(h, n)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if it % 100 == 0 and len(ones) > 0:
            # Steepest descent step every 100 iterations
            best_d = best_val
            best_i, best_j = -1, -1
            for i in ones:
                for j in zeros:
                    nv = float(np.max(corr - eff[i] + eff[j]))
                    if nv < best_d:
                        best_d = nv
                        best_i, best_j = i, j
            if best_i >= 0:
                h[best_i], h[best_j] = 0.0, 1.0
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                if best_d < best_val:
                    best_val = best_d
                    best_h = h.copy()
                continue
        
        # Random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        nv = float(np.max(corr - eff[i] + eff[j]))
        if nv < best_val or np.random.rand() < np.exp(-(nv - best_val) / max(T, 1e-6)):
            h[i], h[j] = 0.0, 1.0
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            if nv < best_val:
                best_val = nv
                best_h = h.copy()
    
    return best_h, best_val * dx

def lins_polish(h, n, t_end):
    """Quick L-BFGS polish on continuous relaxation."""
    dx = 2.0 / n
    T = 5000.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return dx * (corr.max() + np.log(np.exp((corr - corr.max()) / T).sum()) * T)
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = corr.max()
        Tc = (corr - mc) / T
        Tc -= Tc.max()
        w = np.exp(Tc)
        w /= w.sum()
        g = np.zeros(n)
        m_full = 2 * n - 1
        lag = np.arange(m_full) - n + 1
        for idx in range(m_full):
            lg = lag[idx]
            wi = w[idx]
            if wi < 1e-15:
                continue
            if lg >= 0:
                if lg < n:
                    g[lg:] += wi * (1.0 - x[:n-lg])
                    g[:n-lg] += -wi * x[lg:]
            else:
                p = -lg
                g[:n-p] += wi * (1.0 - x[p:])
                g[p:] += -wi * x[:n-p]
        return dx * g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 500, 'ftol': 1e-18})
        return project_binary(res.x)  # Round back to binary
    except:
        return h
