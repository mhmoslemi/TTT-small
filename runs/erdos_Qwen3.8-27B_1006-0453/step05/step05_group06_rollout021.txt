```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def softmax_max(corr, T):
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    return mc + np.log(np.sum(np.exp(np.minimum(Tc, 500)))) / T

def lbfgs_step(h0, T, lam, maxiter, ftol):
    n = len(h0)
    dx = 2.0 / n
    m = 2 * n - 1
    lag_arr = np.arange(m) - (n - 1)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return dx * softmax_max(corr, T) + lam * (x.sum() - n / 2.0) ** 2
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        Tc = T * (corr - mc)
        Tc -= np.max(Tc)
        w = np.exp(np.minimum(Tc, 500))
        w /= w.sum()
        g = np.zeros(n)
        for idx in range(m):
            lg = lag_arr[idx]
            wi = dx * w[idx]
            if wi < 1e-15:
                continue
            if lg >= 0:
                if lg < n:
                    g[lg:] += wi * (1.0 - x[:n - lg])
                    g[:n - lg] += -wi * x[lg:]
            else:
                p = -lg
                g[:n - p] += wi * (1.0 - x[p:])
                g[p:] += -wi * x[:n - p]
        g += 2.0 * lam * (x.sum() - n / 2.0)
        return g
    
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return project_h(h0)

def effect_vector(h, i, lag_arr, m, n):
    """Compute how corr changes per unit change in h[i]."""
    eff = np.zeros(m)
    for j in range(m):
        lg = lag_arr[j]
        if lg >= 0:
            k = lg
            if k < n:
                if 0 <= i - k and i - k <= n - 1 - k:
                    eff[j] += (1.0 - h[i - k])
                if 0 <= i and i <= n - 1 - k:
                    eff[j] += -h[i + k]
        else:
            p = -lg
            if 0 <= i and i <= n - 1 - p:
                eff[j] += (1.0 - h[i + p])
            if p <= i and i <= n - 1:
                eff[j] += -h[i - p]
    return eff

def coord_desc_v2(h, t_end, max_sweeps=1000, seed=0):
    rng = np.random.default_rng(seed)
    n = len(h)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    m = 2 * n - 1
    lag_arr = np.arange(m) - (n - 1)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            cur = best_h[i]
            eff = effect_vector(best_h, i, lag_arr, m, n)
            # 1D problem: min_{x in [0,1]} max_j (corr[j] + (x-cur)*eff[j] - (x-cur)^2 * K[j])
            # K[j] = 1 if j == n-1 (lag 0), else 0
            # For simplicity, just do the linear approximation (neglect quadratic)
            # and verify with full computation
            
            def f1d(x):
                delta = x - cur
                vals = corr + delta * eff
                vals[n - 1] -= delta * delta  # quadratic correction for lag 0
                return np.max(vals)
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            x_new = res.x
            if abs(x_new - cur) < 1e-10:
                continue
            # Verify with full computation
            ht = best_h.copy()
            ht[i] = x_new
            ht = project_h(ht)
            corr_new = np.correlate(ht, 1.0 - ht, mode='full')
            c5t = float(np.max(corr_new) * dx)
            if c5t < best_c5 - 1e-13:
                best_c5 = c5t
                best_h = ht
                corr = corr_new
                improved = True
        if not improved:
            break
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def softmax_max(corr, T):
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    return mc + np.log(np.sum(np.exp(np.minimum(Tc, 500.0)))) / T

def lbfgs_step(h0, T, lam, maxiter, ftol):
    n = len(h0)
    dx = 2.0 / n
    m = 2 * n - 1
    lag_arr = np.arange(m) - (n - 1)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return dx * softmax_max(corr, T) + lam * (x.sum() - n / 2.0) ** 2

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        Tc = T * (corr - mc)
        Tc -= np.max(Tc)
        w = np.exp(np.minimum(Tc, 500.0))
        w /= w.sum()
        g = np.zeros(n)
        for idx in range(m):
            lg = lag_arr[idx]
            wi = dx * w[idx]
            if wi < 1e-15:
                continue
            if lg >= 0:
                if lg < n:
                    g[lg:] += wi * (1.0 - x[:n - lg])
                    g[:n - lg] += -wi * x[lg:]
            else:
                p = -lg
                g[:n - p] += wi * (1.0 - x[p:])
                g[p:] += -wi * x[:n - p]
        g += 2.0 * lam * (x.sum() - n / 2.0)
        return g

    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return project_h(h0)

def effect_vector(h, i, lag_arr):
    n = len(h)
    m = len(lag_arr)
    eff = np.zeros(m)
    for j in range(m):
        lg = lag_arr[j]
        if lg >= 0:
            k = lg
            if k < n:
                if 0 <= i - k and i - k <= n - 1 - k:
                    eff[j] += (1.0 - h[i - k])
                if 0 <= i and i <= n - 1 - k:
                    eff[j] += -h[i + k]
        else:
            p = -lg
            if 0 <= i and i <= n - 1 - p:
                eff[j] += (1.0 - h[i + p])
            if p <= i and i <= n - 1:
                eff[j] += -h[i - p]
    return eff

def coord_desc_fast(h, t_end, max_sweeps=5000, seed=0):
    rng = np.random.default_rng(seed)
    n = len(h)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    m = 2 * n - 1
    lag_arr = np.arange(m) - (n - 1)
    j_zero = n - 1  # index of lag-0 in full correlation

    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            cur = best_h[i]
            eff = effect_vector(best_h, i, lag_arr)
            corr_base = corr.copy()

            def f1d(x):
                d = x - cur
                vals = corr_base + d * eff
                vals[j_zero] -= d * d
                return np.max(vals)

            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            x_new = res.x
            if abs(x_new - cur) < 1e-10:
                continue
            ht = best_h.copy()
            ht[i] = x_new
            ht = project_h(ht)
            corr_new = np.correlate(ht, 1.0 - ht, mode='full')
            c5t = float(np.max(corr_new) * dx)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t
                best_h = ht
                corr = corr_new
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

            best_h = hc

    # Phase 2: Fast coordinate descent (30% budget)

    best_h, best_c5 = coord_desc_fast(best_h, p2_end, 5000, seed)

    # Phase 3: High-T L-BFGS refinement (15% budget)
    p3_end = t0 + budget_s * 0.60
    for T in [10000, 50000, 200000]:
        if time.time() > p3_end:

        c5v = compute_c5(hc)

            best_h = hc

    # Phase 4: Basin hopping with perturbation (20% budget)
    p4_end = t0 + budget_s * 0.80

    work_c5 = best_c5

    if time.time() < t0 + budget_s * 0.995:

    best_c5 = compute_c5(best_h)
```
