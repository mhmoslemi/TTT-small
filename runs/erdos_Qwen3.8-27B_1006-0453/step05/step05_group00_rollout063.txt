import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_deltas(h, N):
    """Compute Δ_down[p, s] and Δ_up[q, s] for all positions p, q and shifts s."""
    # corr[s] for s = -(N-1), ..., N-1
    # offset: index = s + (N-1) in the corr array
    offset = N - 1
    # Δ_down[p, s] = -(1-h[p+s]) * [0<=p+s<N] + h[p-s] * [0<=p-s<N]
    # Δ_up[q, s] = (1-h[q+s]) * [0<=q+s<N] - h[q-s] * [0<=q-s<N]
    
    # Build index arrays
    p_idx = np.arange(N)[:, None]  # (N, 1)
    s_idx = np.arange(-N+1, N)  # (2N-1,)
    
    # For Δ_down[p, s]:
    # p+s valid: 0 <= p+s < N
    ps = p_idx + s_idx[None, :]  # (N, 2N-1)
    ps_valid = (ps >= 0) & (ps < N)
    ps_clipped = np.clip(ps, 0, N-1)
    
    # p-s valid: 0 <= p-s < N
    pm = p_idx - s_idx[None, :]  # (N, 2N-1)
    pm_valid = (pm >= 0) & (pm < N)
    pm_clipped = np.clip(pm, 0, N-1)
    
    delta_down = -((1.0 - h[ps_clipped]) * ps_valid) + (h[pm_clipped] * pm_valid)
    
    # For Δ_up[q, s]:
    # (1-h[q+s]) * [valid] - h[q-s] * [valid]
    delta_up = (1.0 - h[ps_clipped]) * ps_valid - h[pm_clipped] * pm_valid
    
    return delta_down, delta_up  # each (N, 2N-1)

def full_swap_search(h, N, deadline, max_sweeps=10000):
    h = h.copy()
    nh = N // 2
    corr = fft_corr(h)
    best_c = float(corr.max()) * (2.0 / N)

            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # For each p in ones, compute base = corr + delta_down[p]

            # For all zeros: max_k(base + zeros_d[j])

            max_vals = combined.max(axis=1)  # (190,)
            j = int(np.argmin(max_vals))

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def compute_deltas(h, N):
    offset = N - 1
    p_idx = np.arange(N)[:, None].astype(int)
    s_vals = np.arange(-N + 1, N)
    s_idx = s_vals[None, :].astype(int)
    
    ps = p_idx + s_idx
    pm = p_idx - s_idx
    ps_v = (ps >= 0) & (ps < N)
    pm_v = (pm >= 0) & (pm < N)
    ps_c = np.clip(ps, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    
    dd = -((1.0 - h[ps_c]) * ps_v.astype(float)) + (h[pm_c] * pm_v.astype(float))
    du = ((1.0 - h[ps_c]) * ps_v.astype(float)) - (h[pm_c] * pm_v.astype(float))
    dd[:, offset] = 0.0
    du[:, offset] = 0.0
    return dd, du

def exhaustive_swap(h, N, deadline, max_sweeps=10000, kick_every=200, kick_size=3):
    h = h.astype(float).copy()
    nh = N // 2
    corr = fft_corr(h)
    best_c5 = float(corr.max()) * (2.0 / N)
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        dd, du = compute_deltas(h, N)
        ones_dd = dd[ones]
        base_all = corr[None, :] + ones_dd  # (n1, 2N-1)
        
        best_val = float('inf')
        best_pi = -1
        best_qi = -1
        for qi in range(len(zeros)):
            vals = (base_all + du[zeros[qi]]).max(axis=1)
            pi = int(np.argmin(vals))
            if vals[pi] < best_val:
                best_val = float(vals[pi])
                best_pi = pi
                best_qi = qi
        
        if best_pi < 0 or best_val >= corr.max() - 1e-15:
            stall += 1
            if stall > 3 and kick_size > 0:
                # Random kick
                for _ in range(kick_size):
                    p = ones[np.random.randint(len(ones))]
                    q = zeros[np.random.randint(len(zeros))]
                    h[p], h[q] = h[q], h[p]
                corr = fft_corr(h)
                stall = 0
                if float(corr.max()) * (2.0/N) < best_c5:
                    best_c5 = float(corr.max()) * (2.0/N)
                continue
            else:
                break
        else:
            stall = 0
        
        p = ones[best_pi]
        q = zeros[best_qi]
        h[p] = 0.0
        h[q] = 1.0
        corr = fft_corr(h)
        c5 = float(corr.max()) * (2.0 / N)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_deltas(h, N):
    offset = N - 1
    p_idx = np.arange(N)[:, None].astype(np.int64)
    s_vals = np.arange(-N + 1, N, dtype=np.int64)
    s_idx = s_vals[None, :]
    ps = p_idx + s_idx
    pm = p_idx - s_idx
    ps_v = ((ps >= 0) & (ps < N)).astype(np.float64)
    pm_v = ((pm >= 0) & (pm < N)).astype(np.float64)
    ps_c = np.clip(ps, 0, N - 1)
    pm_c = np.clip(pm, 0, N - 1)
    hps = h[ps_c]
    hpm = h[pm_c]
    dd = -((1.0 - hps) * ps_v) + (hpm * pm_v)
    du = ((1.0 - hps) * ps_v) - (hpm * pm_v)
    dd[:, offset] = 0.0
    du[:, offset] = 0.0
    return dd, du

def exhaustive_swap(h, N, deadline, max_sweeps=10000, kick_size=3):
    h = h.astype(np.float64).copy()
    nh = N // 2
    corr = fft_corr(h)
    best_c5 = float(corr.max()) * (2.0 / N)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        dd, du = compute_deltas(h, N)
        ones_dd = dd[ones]
        base_all = corr[None, :] + ones_dd
        n1 = len(ones)
        nz = len(zeros)
        best_val = np.inf
        best_pi = -1
        best_qi = -1
        chunk = 50
        for cs in range(0, nz, chunk):
            ce = min(cs + chunk, nz)
            zc = zeros[cs:ce]
            v = base_all[:, None, :] + du[zc][None, :, :]
            vm = v.max(axis=2)
            for pi in range(n1):
                li = int(np.argmin(vm[pi]))
                if vm[pi, li] < best_val:
                    best_val = float(vm[pi, li])
                    best_pi = pi
                    best_qi = cs + li
        if best_pi < 0 or best_val >= float(corr.max()) - 1e-14:
            stall += 1
            if stall <= 4 and kick_size > 0:
                for _ in range(kick_size):
                    p = ones[np.random.randint(len(ones))]
                    q = zeros[np.random.randint(len(zeros))]
                    h[p], h[q] = h[q], h[p]
                corr = fft_corr(h)
                ctmp = float(corr.max()) * (2.0 / N)
                if ctmp < best_c5:
                    best_c5 = ctmp
                stall = 0
                continue
            else:
                break
        else:
            stall = 0
        h[ones[best_pi]] = 0.0
        h[zeros[best_qi]] = 1.0
        corr = fft_corr(h)
        c5 = float(corr.max()) * (2.0 / N)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def two_swap_escape(h, N, deadline, n_trials=200):
    h = h.astype(np.float64).copy()
    corr = fft_corr(h)
    best_c5 = float(corr.max()) * (2.0 / N)
    dd, du = compute_deltas(h, N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for t in range(n_trials):
        if time.time() > deadline:
            break
        p1, p2 = np.random.choice(ones, 2, replace=False)
        q1, q2 = np.random.choice(zeros, 2, replace=False)
        nc = corr + dd[p1] + dd[p2] + du[q1] + du[q2]
        if nc.max() < corr.max() - 1e-12:
            h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
            corr = fft_corr(h)
            best_c5 = min(best_c5, float(corr.max()) * (2.0 / N))
            dd, du = compute_deltas(h, N)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    return h, best_c5

def nesterov_opt(h, N, deadline, k=50, lr0=0.018):
    h = h.astype(np.float64).copy()
    target = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                grad[:N - M] += (1.0 - y[M:])
                grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.992
            beta = min(0.97, 0.86 + it * 0.0004)
    return best_h, best_pk * dx

def ensure_sum(h, N):
    h = np.clip(h, 0.0, 1.0)
    nh = N // 2
    n1 = int(np.round(h.sum()))
    if n1 != nh:
        h = (h > 0.5).astype(float)
        n1 = int(h.sum())
        if n1 > nh:
            idx = np.where(h == 1)[0]
            perm = np.random.permutation(len(idx))
            h[idx[perm[nh:]]] = 0
        elif n1 < nh:
            idx = np.where(h == 0)[0]
            perm = np.random.permutation(len(idx))
            h[idx[perm[:nh - n1]]] = 1
    return h
