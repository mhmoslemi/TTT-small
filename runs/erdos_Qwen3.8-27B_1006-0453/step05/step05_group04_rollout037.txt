For the gradient method, I'll use a "hottest lag" approach:
- Identify the top K lags (where K is small, like 5-10)
- Compute a subgradient w.r.t. h that emphasizes these lags
- Take a step in the negative gradient direction
- Project

For basin hopping:
- Structured perturbations: 
  - Add a smooth bump (Gaussian) at a random location
  - Remove mass from a region and add to another (maintains integral better)
  - Shift a portion of h
  - Random smooth perturbation (low-pass filtered noise)

    """Optimize variable i. Returns (new_val, L, d_opt)."""
    old = h[i]
    L = np.zeros(2 * n - 1)
    # Term 1: (1-h[i+s]) for s where 0<=i+s<n
    s_lo = max(-i, -(n-1))
    s_hi = min(n-1-i, n-1)
    if s_lo <= s_hi:

        L[s_arr + n - 1] += 1.0 - h[i + s_arr]
    # Term 2: -h[i-s] for s where 0<=i-s<n
    s_lo2 = max(i-(n-1), -(n-1))
    s_hi2 = min(i, n-1)
    if s_lo2 <= s_hi2:
        s_arr2 = np.arange(s_lo2, s_hi2+1)
        L[s_arr2 + n - 1] -= h[i - s_arr2]
    
    d_lo = -old
    d_hi = 1.0 - old
    if d_hi - d_lo < 1e-14:
        return old, L, 0.0
    
    # Find best d using grid + refinement
    def f_of_d(d):
        v = corr + d * L
        v[n-1] -= d * d
        return np.max(v)
    
    # Coarse grid
    grid = np.linspace(d_lo, d_hi, 16)
    vals = np.array([f_of_d(d) for d in grid])
    best_idx = np.argmin(vals)
    
    # Refine around best
    lo = grid[max(0, best_idx-1)]
    hi = grid[min(len(grid)-1, best_idx+1)]
    fine_grid = np.linspace(lo, hi, 10)
    fine_vals = np.array([f_of_d(d) for d in fine_grid])
    best_fine = np.argmin(fine_vals)
    d_opt = fine_grid[best_fine]
    
    return old + d_opt, L, d_opt

            new_val, L, d_opt = cd_one_var(best_h, corr, i, n)
            d = d_opt
            if abs(d) < 1e-12:

            best_h[i] = new_val
            corr += d * L
            corr[n-1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:

                improved = True
            else:

                corr -= d * L
                corr[n-1] += d * d
        if not improved and sweep > 3:
            break
    return best_h, best_c5

import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(50):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (d / tot)
        else:
            if s < 1e-14:
                break
            h *= target / s
        h = np.clip(h, 0.0, 1.0)
    return h

def heat_gradient(h, corr, beta, n):
    """Compute softmax-weighted subgradient of max_k c_k w.r.t. h."""
    dx = 2.0 / n
    m = np.max(corr)
    exponents = np.clip(beta * (corr - m), -50.0, 50.0)
    p = np.exp(exponents)
    p /= p.sum()
    g = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = p[m_idx]
        if w < 1e-12:
            continue
        s = m_idx - (n - 1)
        if s > 0:
            g[:n-s] += w * (1.0 - h[s:])
            g[s:] -= w * h[:n-s]
        elif s < 0:
            a = -s
            g[a:] += w * (1.0 - h[:n-a])
            g[:n-a] -= w * h[a:]
        else:
            g += w * (1.0 - 2.0 * h)
    return dx * g

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    mom = np.zeros(n)
    beta_mom = 0.7
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = min(1.0, it / max(1, max_iters))
        T = 30.0 + 200.0 * frac

        if gm < 1e-14:

            corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=300, seed=42):
