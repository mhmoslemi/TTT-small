def project_h(h, n):

def grad_max_h(h, n, active_frac=0.01):
    """Compute subgradient of max correlation w.r.t. h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    threshold = m - m * active_frac
    active = corr >= threshold
    n_active = active.sum()
    
    grad = np.zeros(n)
    for m_idx in np.where(active)[0]:
        w = 1.0 / n_active
        s = m_idx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    w = 1.0 / len(active)
    grad = np.zeros(n)
    for m_idx in active:
        s = m_idx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return (2.0 / n) * grad


def make_perturb(h, n, seed, mode, intensity):
    rng = np.random.RandomState(seed)
    pert = np.zeros(n)
    if mode == 0:
        pert = intensity * rng.randn(n)
    elif mode == 1:
        f = rng.randint(1, max(2, n // 6))
        ph = rng.uniform(0, 2.0 * np.pi)
        pert = intensity * np.sin(2.0 * np.pi * f * np.arange(n) / n + ph)
    elif mode == 2:
        k = max(3, n // 12)
        idx = rng.choice(n, size=k, replace=False)
        pert[idx] = intensity * 2.0 * rng.choice([-1, 1], k)
    elif mode == 3:
        g = grad_subgrad(h, n)
        ng = np.max(np.abs(g))
        if ng > 1e-15:
            pert = -intensity * g / ng
    elif mode == 4:
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-3:]
        for m_idx in top_k:
            s = m_idx - (n - 1)
            if s > 0:
                contrib = h[:n-s] * (1.0 - h[s:])
                top = np.argsort(contrib)[-max(2, n // 15):]
                pert[top] -= intensity * 0.6
            elif s < 0:
                a = -s
                contrib = h[:n-a] * (1.0 - h[a:])
                top = np.argsort(contrib)[-max(2, n // 15):]
                pert[top] -= intensity * 0.6
    elif mode == 5:
        k = rng.randint(3, max(4, n // 8))
        start = rng.randint(0, n - k)
        pert[start:start+k] = intensity * rng.choice([-1, 1]) * rng.uniform(0.5, 1.0, k)
    else:
        pert = intensity * 0.4 * rng.randn(n)
    return pert


def basin_hopping(h, n, t_end, n_hops=400, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    stall = 0

    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - hop / n_hops)
        if stall > 40:
            step = 0.09 + 0.04 * cool
            stall = 0
        elif stall > 20:
            step = 0.06 + 0.04 * cool
        else:
            step = 0.025 + 0.05 * cool

        mode = hop % 7
        pert = make_perturb(cur_h, n, seed + hop, mode, step)
        h_pert = project_h(cur_h + pert, n)

        if cool > 0.6:
            cd_time = 2.5
            cd_sweeps = 120
        elif cool > 0.3:
            cd_time = 2.0
            cd_sweeps = 90
        else:
            cd_time = 1.5
            cd_sweeps = 60

        hop_end = min(t_end, time.time() + cd_time)
        h_opt, c5_opt = fast_coord_descent(h_pert, n, hop_end, max_sweeps=cd_sweeps, seed=seed + hop)

        temp = 0.0035 * cool + 0.00015
        if c5_opt < cur_c5:
            accept = True
            stall = 0
        else:
            delta = c5_opt - cur_c5
            accept = rng.rand() < np.exp(-delta / max(temp, 1e-10))
            if accept:
                stall += 1

        if accept:
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5


def interpolate_h(h, n_from, n_to):
    x_from = np.arange(n_from) + 0.5
    x_to = np.arange(n_to) + 0.5
    h_ext = np.concatenate([h[-n_from//4:], h, h[:n_from//4]])
    x_ext = np.concatenate([np.arange(n_from, 3*n_from) - n_from//4, x_from, x_to * 0 - 1])
    h_new = np.interp(x_to, x_from, h)
    return h_new


def multi_res_search(h0, n0, t_end, seed):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = n0
    budget_per = (t_end - time.time()) / 5.0

    for n_try in [140, 160, 180, 200, 220]:
        if time.time() > t_end:
            break
        if n_try == n0:
            continue
        h_nr = interpolate_h(h0, n0, n_try)
        h_nr = project_h(h_nr, n_try)
        c5_nr = compute_c5(h_nr)
        t_cd = min(t_end, time.time() + budget_per * 0.5)
        h_opt, c5_opt = fast_coord_descent(h_nr, n_try, t_cd, max_sweeps=300, seed=seed + n_try)
        if c5_opt < best_c5 - 1e-6:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try

    return best_h, best_c5, best_n


def pair_swap_refine(h, n, t_end, n_tries=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')

    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i, j = rng.choice(n, 2, replace=False)
        hi, hj = best_h[i], best_h[j]
        # Try swapping
        best_h[i], best_h[j] = hj, hi
        c5_s = compute_c5_fast(corr, best_h, i, j, n, dx)
        if c5_s < best_c5 - 1e-12:
            best_c5 = c5_s
            hi, hj = base_hi, base_hj
        else:
            best_h[i], best_h[j] = hi, hj

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent polish (20%)
    t1 = t0 + budget_s * 0.18
    best_h, best_c5 = fast_coord_descent(best_h, n, t1, max_sweeps=4000, seed=seed)

    # Phase 2: Multi-resolution search (12%)
    if time.time() < t0 + budget_s * 0.70:
        t2 = t0 + budget_s * 0.30
        h_mr, c5_mr, n_mr = multi_res_search(best_h, n, t2, seed)
        if c5_mr < best_c5:
            best_h, best_c5, n = h_mr, c5_mr, n_mr
        if time.time() < t0 + budget_s * 0.35:
            best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.35, max_sweeps=1500, seed=seed + 7)

    # Phase 3: Basin hopping - main exploration (42%)
    if time.time() < t0 + budget_s * 0.72:
        t3 = t0 + budget_s * 0.72
        best_h, best_c5 = basin_hopping(best_h, n, t3, n_hops=500, seed=seed + 10)

    # Phase 4: Final heavy coordinate descent (16%)
    if time.time() < t0 + budget_s * 0.90:
        best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=8000, seed=seed + 100)

    # Phase 5: Second basin hopping with different seed (6%)
    if time.time() < t0 + budget_s * 0.96:
        best_h, best_c5 = basin_hopping(best_h, n, t0 + budget_s * 0.96, n_hops=150, seed=seed + 500)

    # Phase 6: Final polish
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = fast_coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=3000, seed=seed + 999)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    assert abs(best_h.sum() - n / 2.0) < 0.05
    return best_h, best_c5, n
