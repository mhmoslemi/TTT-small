```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

_T = [1.0]

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h.copy()
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-14:
            h = h * (tgt / max(s, 1e-14))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        m = (h > 0.02) & (h < 0.98)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def coeff_at(p, h, n, lags):
    c = np.zeros(2 * n - 1)
    idx1 = p + lags
    m1 = (idx1 >= 0) & (idx1 < n)
    c[m1] += 1.0 - h[idx1[m1]]
    idx2 = p - lags
    m2 = (idx2 >= 0) & (idx2 < n)
    c[m2] -= h[idx2[m2]]
    return c

def pair_obj(delta, base, dc):
    return float(np.max(base + delta * dc))

def pairwise_cd(h, n, t_end, n_sweeps=300):
    best = h.copy()
    best_c5 = compute_c5(best)
    lags = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        n_try = n * 3
        for _ in range(n_try):
            if time.time() > t_end:
                return best, best_c5
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            oi, oj = best[i], best[j]
            s = oi + oj
            lo = -min(oi, 1.0 - oj)
            hi = min(1.0 - oi, oj)
            if hi - lo < 1e-14:
                continue
            ci = coeff_at(i, best, n, lags)
            cj = coeff_at(j, best, n, lags)
            dc = ci - cj
            r = minimize_scalar(pair_obj, args=(corr, dc),
                               bounds=(lo, hi), method='bounded',
                               options={'xatol': 1e-13})
            d = float(r.x)
            if abs(d) < 1e-13:
                continue
            ni = oi + d
            nj = s - ni
            if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                continue
            ni = np.clip(ni, 0.0, 1.0)
            nj = np.clip(nj, 0.0, 1.0)
            trial = best.copy()
            trial[i] = ni
            trial[j] = nj
            c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode='full')
                improved = True
        if sweep > 50 and not improved:
            break
    return best, best_c5

def _sobj(x):
    T = _T[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n

def _sgrad(x):
    T = _T[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    full_c = np.correlate(w, x, mode='full')
    A = full_c[n - 1::-1]
    conv_b = np.convolve(x, w, mode='full')
    B = conv_b[n - 1:2 * n - 1]
    g = 1.0 - A - B
    return g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=300):
    _T[0] = T
    try:
        r = minimize(h, _sobj, _sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-16, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.full(160, 0.5)
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B to get into a good basin
    for T in [200, 2000, 20000]:
        if time.time() - t0 > budget_s * 0.06:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Pairwise coordinate descent (main)
    best_h, best_c5 = pairwise_cd(best_h, n, t0 + budget_s * 0.40, n_sweeps=500)

    # Phase 3: L-BFGS-B refinement
    if time.time() - t0 < budget_s * 0.50:
        for T in [10000, 50000, 200000]:
            if time.time() - t0 > budget_s * 0.55:
                break
            h_new = lbfgs_opt(best_h, T, n, iters=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: More pairwise CD
    best_h, best_c5 = pairwise_cd(best_h, n, t0 + budget_s * 0.70, n_sweeps=400)

    # Phase 5: Diversification - try perturbed restarts
    if time.time() - t0 < budget_s * 0.80:
        for att in range(4):
            if time.time() - t0 > budget_s * 0.82:
                break
            amp = 0.03 + 0.04 * att
            pert = amp * np.random.randn(n)
            h_alt = proj_h(best_h + pert, n)
            h_alt, c5_alt = pairwise_cd(h_alt, n, t0 + budget_s * 0.82, n_sweeps=80)
            if c5_alt < best_c5:
                best_h = h_alt
                best_c5 = c5_alt

    # Phase 6: Final L-BFGS-B with very high T
    if time.time() - t0 < budget_s * 0.90:
        h_new = lbfgs_opt(best_h, 500000.0, n, iters=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 7: Final pairwise CD
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = pairwise_cd(best_h, n, t0 + budget_s * 0.98, n_sweeps=200)

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
