```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / exp_shifted.sum()
    val = (m + (max_shifted + np.log(exp_shifted.sum())) / T) * dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        j = np.arange(n)
        a = j - lag
        b = j + lag
        va = np.where((a >= 0) & (a < n), 1.0 - h[np.clip(a, 0, n-1)], 0.0)
        vb = np.where((b >= 0) & (b < n), h[np.clip(b, 0, n-1)], 0.0)
        grad += w * (va - vb)
    grad *= dx
    return val, grad

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def subgrad_descent(h, n, t_end, T_init=500, T_end=5000, max_iter=5000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_curr = h.copy()
    vel = np.zeros(n)
    lr = 0.02
    for it in range(max_iter):
        if time.time() > t_end:
            break
        T = T_init + (T_end - T_init) * it / max(max_iter, 1)
        val, grad = smooth_c5_and_grad(h_curr, T)

        # Grad descent step

            best_h = h_try.copy()
            h_curr = h_try
            lr = min(lr * 1.05, 0.1)
        else:
            # Rejected, reduce lr
            h_curr = project_h(best_h + 0.1 * (h_try - best_h))
            lr *= 0.95
    return best_h, best_c5
```

Now let me make the coordinate descent efficient. The key is to precompute the correlation once and update it as we change one variable.

When I change h[i] from h[i]_old to h[i]_new, the change in corr[k_idx] is:
corr[k_idx] involves h[m] and h[m+lag]. h[i] appears when m=i or m+lag=i.

For lag = k_idx - (n-1):
- m=i: term h[i]*(1-h[i+lag]) changes by (h_new - h_old)*(1-h[i+lag])
- m+lag=i → m=i-lag: term h[i-lag]*(1-h[i]) changes by h[i-lag]*(h_old - h_new)

So: Δcorr[k_idx] = (h_new - h_old) * [(1-h[i+lag]) - h[i-lag]] (with bounds checking)

This is O(n) to update all corr entries after changing one h[i].

Then I need to find the best h[i]_new that minimizes max(corr). Since corr[k_idx] is linear in h_new:
corr_new[k_idx] = corr_old[k_idx] + (h_new - h_old) * coeff[k_idx]

where coeff[k_idx] = (1-h[i+lag]) - h[i-lag] (the bracketed term above).

So I need to find h_new in [0,1] that minimizes max_k (corr_old[k] + delta * coeff[k]) where delta = h_new - h_old.

This is a 1D problem! I can solve it with a ternary search or by finding the intersection points.

Actually, the function f(delta) = max_k (a_k + delta * b_k) where a_k = corr_old[k] - h_old * coeff[k] and b_k = coeff[k]. This is a convex function of delta (max of linear functions). I can minimize it with a simple approach:
- The minimum is where the two "active" (maximum) lines cross
- Or at a boundary (delta ∈ [​-h_old, 1-h_old])

For n=160, I can just do a golden section search on [​-h_old, 1-h_old] for delta.

Let me implement this:

```python
def coord_single_fast(h, i, n, corr):

        c = 0.0

        if 0 <= b < n:
            c -= h[b]

    # = (corr[k] - h_old * coeff[k]) + h_new * coeff[k]

    # Minimize max_k (a_k + h_new * coeff[k]) over h_new in [0,1]
    # This is: max_k (a_k + h_new * coeff[k])
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def slsqp_opt(h, T, n, maxiter=500):
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = T * (corr - m)
        shm = np.max(sh)
        return (m + shm + np.log(np.sum(np.exp(sh - shm)))) / T * dx
    try:
        cons = {'type': 'eq', 'fun': _con_fun, 'args': (n,)}
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[cons],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def fast_coord_single(h, i, n, corr):
    h_old = h[i]
    coeff = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        c = 0.0
        a = i + lag
        if 0 <= a < n:
            c += 1.0 - h[a]
        b = i - lag
        if 0 <= b < n:
            c -= h[b]
        coeff[k] = c
    a_k = corr - h_old * coeff
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    for _ in range(40):
        m1 = hi - (hi - lo) / gr
        m2 = lo + (hi - lo) / gr
        if np.max(a_k + m1 * coeff) < np.max(a_k + m2 * coeff):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent_fast(h, n, t_end, max_sweeps=500, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            h_new = fast_coord_single(best_h, i, n, corr)
            if abs(h_new - best_h[i]) < 1e-10:
                continue
            best_h[i] = h_new
            best_h = project_h(best_h)
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
        if not improved:
            break
    return best_h, best_c5

def compute_effects(h, n):
    """effect[i][k] = change in corr[k] when h[i] increases by 1."""
    eff = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            c = 0.0
            a = i + lag
            if 0 <= a < n:
                c += 1.0 - h[a]
            b = i - lag
            if 0 <= b < n:
                c -= h[b]
            eff[i, k] = c
    return eff

def binary_steepest(h, n, t_end, rng, max_passes=500):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_curr = (best_h >= 0.5).astype(float)
    n_ones = int(h_curr.sum())
    target = n // 2
    if n_ones > target:
        oi = np.where(h_curr == 1)[0]
        rm = rng.choice(len(oi), n_ones - target, replace=False)
        h_curr[oi[rm]] = 0
    elif n_ones < target:
        zi = np.where(h_curr == 0)[0]
        ad = rng.choice(len(zi), target - n_ones, replace=False)
        h_curr[zi[ad]] = 1
    c5c = compute_c5(h_curr)
    if c5c < best_c5:
        best_c5 = c5c
        best_h = h_curr.copy()
    corr = np.correlate(h_curr, 1.0 - h_curr, mode='full').copy()
    eff = compute_effects(h_curr, n)
    for pas in range(max_passes):
        if time.time() > t_end:
            break
        oi = np.where(h_curr == 1)[0]
        zi = np.where(h_curr == 0)[0]
        best_delta = 0.0
        best_ij = None
        for i in oi:
            for j in zi:
                c_new = corr + eff[j] - eff[i]
                mx = np.max(c_new) * dx
                d = mx - best_c5
                if d < best_delta - 1e-15:
                    best_delta = d
                    best_ij = (i, j)
                    best_cnew = c_new.copy()
        if best_ij is None:
            # Try random pair as escape
            if int(time.time() * 10) % 3 == 0 and len(oi) > 0 and len(zi) > 0:
                i = oi[rng.randint(len(oi))]
                j = zi[rng.randint(len(zi))]
                c_new = corr + eff[j] - eff[i]
                if np.max(c_new) * dx < best_c5 + 0.002:
                    h_curr[i], h_curr[j] = 0, 1
                    corr = c_new
                    eff = compute_effects(h_curr, n)
                    continue
            break
        i, j = best_ij
        h_curr[i], h_curr[j] = 0, 1
        corr = best_cnew
        eff = compute_effects(h_curr, n)
        best_c5 += best_delta
        best_h = h_curr.copy()
    return best_h, best_c5

def random_swap_search(h, n, t_end, rng, n_tries=20000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_curr = (best_h >= 0.5).astype(float)
    for it in range(n_tries):
        if time.time() > t_end:
            break
        oi = np.where(h_curr == 1)[0]
        zi = np.where(h_curr == 0)[0]
        i = oi[rng.randint(len(oi))]
        j = zi[rng.randint(len(zi))]
        h_curr[i], h_curr[j] = 0, 1
        c5_try = compute_c5(h_curr)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_curr.copy()
        else:
            h_curr[i], h_curr[j] = 1, 0
    return best_h, best_c5
```
