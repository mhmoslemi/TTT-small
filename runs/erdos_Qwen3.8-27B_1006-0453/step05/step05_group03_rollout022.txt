import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_autocorr(h):
    n = len(h)
    return np.correlate(h, h, mode='full')  # length 2n-1

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_corr(h, T, n):
    """Smooth approximation of max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -100, 100)
    return (m + np.log(np.sum(np.exp(sh))) / T) * (2.0/n)

def smooth_min_autocorr(h, T, n):
    """Smooth approximation of -min autocorrelation (to minimize)."""
    ac = np.correlate(h, h, mode='full')
    m = np.min(ac)
    sh = np.clip(-T * (ac - m), -100, 100)
    return -(m + np.log(np.sum(np.exp(sh))) / T)

def slsqp_smooth(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        return smooth_max_corr(x, T, n)
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def subgrad_min_autocorr(h, n, k=8, alpha=0.01):
    """Gradient step to increase the minimum k autocorrelations."""
    ac = np.correlate(h, h, mode='full')
    idx = np.argsort(ac)[:k]  # bottom k autocorrelations
    grad = np.zeros(n)
    for ki in idx:
        lag = ki - (n - 1)
        # grad of autocorr[lag] w.r.t h[i] is h[i+lag] + h[i-lag]
        for i in range(n):
            v = 0.0
            if 0 <= i + lag < n:
                v += h[i + lag]
            if 0 <= i - lag < n:
                v += h[i - lag]
            grad[i] += v
    return project_h(h + alpha * grad, n)

Let me optimize: instead of running minimize_scalar, I can analytically find the optimal h[i]. Since f(δ) = max_k (c_k + δ * a_{k,i}) is a convex piecewise-linear function of δ, its minimum over [0,1] is at either δ=0, δ=1, or at a breakpoint where two linear pieces cross.

The breakpoints occur where c_k + δ*a_{k,i} = c_j + δ*a_{j,i}, i.e., δ = (c_j - c_k)/(a_{k,i} - a_{j,i}) for pairs where a_{k,i} ≠ a_{j,i}.

For n=160, there are 319 correlation values, so O(319²) = ~100K breakpoints to check. That's actually faster than 30 function evaluations of minimize_scalar (each O(n) = O(319)).

But implementing this correctly is tricky. Let me just use a simpler approach: ternary search or golden section with fewer iterations.

Actually, the piecewise-linear function max_k (c_k + δ*a_k) has at most O(n) breakpoints (the upper envelope of O(n) lines). I can find the minimum by:
1. Evaluate at δ=0 and δ=1
2. If neither is the minimum, find the breakpoint where the minimum occurs

But finding the exact breakpoint requires sorting, which is O(n log n). For n=319, that's fast.

Actually wait - I realize the coefficient computation also has an issue. Let me re-derive:

The correlation at lag L (where L = k - (n-1) for index k in the full correlation) is:
c(L) = ∑_i h[i] * (1-h[i+L]) for valid i (0≤i<n and 0≤i+L<n)

When we change h[i] from old to new (by δ = new - old):
c(L) changes by:
- Term where i is the first index: δ * (1-h[i+L]) if i+L is valid
- Term where i+L is the second index (i.e., the i in the sum is i-L): 
  h[i-L] * (-δ) if i-L is valid

So: Δc(L) = δ * [(1-h[i+L]) * 1_{valid(i+L)} - h[i-L] * 1_{valid(i-L)}]

In the full correlation array (indexed from 0 to 2n-2), lag L corresponds to index L + (n-1).

So coefs[L+(n-1)] = (1-h[i+L])*1_{valid} - h[i-L]*1_{valid}

This is exactly what the code computes. Good.

But the issue is: after changing h[i], the coefficients for OTHER coordinates change too. So the incremental update `corr += δ * coefs` is only valid for the FIRST step. After that, we need to recompute.

For the coordinate descent, the correct approach is:
1. At each step i:
   a. Compute coefs for position i (based on current h)
   b. Find optimal δ
   c. Apply: h[i] += δ
   d. Update corr: the change is δ * coefs (this is exact for this one step)
   e. But coefs for other positions have changed, so next iteration we recompute them

Actually step (d) IS exact! Because the correlation is bilinear, the change in corr when we change h[i] by δ is exactly δ * coefs (computed before the change). So we CAN do incremental updates!

Let me verify: c(L) = ∑_j h[j](1-h[j+L])
After h[i] → h[i]+δ:
c_new(L) = c(L) + δ(1-h[i+L]) - δ*h[i-L] (with validity checks)
= c(L) + δ * coefs[L]

Yes! This is exact. So I can maintain corr incrementally.

But wait, I also need to update coefs for the NEXT coordinate. The coefs for position j depend on h, and h has changed. So I need to recompute coefs each time. That's O(n) per coordinate, times n coordinates = O(n²) per sweep. Plus O(n) to evaluate max(corr). Total O(n²) per sweep.

For n=160: 160² = 25600 operations per sweep. Very fast!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt_fast(corr, coefs):
    a, b = 0.0, 1.0
    for _ in range(28):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(corr + m1 * coefs) < np.max(corr + m2 * coefs):
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def coord_descent_fast(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').copy()
    best_c5 = float(np.max(corr) * (2.0 / n))
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(2 * n - 1)
            lags = np.arange(-(n - 1), n)
            j1 = i + lags
            j2 = i - lags
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            coefs[v1] += (1.0 - best_h[np.clip(j1[v1], 0, n - 1)])
            coefs[v2] += (-best_h[np.clip(j2[v2], 0, n - 1)])
            old_val = best_h[i]
            new_val = coord_opt_fast(corr, coefs)
            delta = new_val - old_val
            if abs(delta) < 1e-12:
                continue
            best_h[i] = new_val
            corr += delta * coefs
            c5_new = float(np.max(corr) * (2.0 / n))
            if c5_new > best_c5 + 1e-14:
                best_h[i] = old_val
                corr -= delta * coefs
            elif c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_min_ac(h, n, k=6, alpha=0.005):
    ac = np.correlate(h, h, mode='full')
    idx = np.argsort(ac)[:k]
    grad = np.zeros(n)
    for ki in idx:
        lag = ki - (n - 1)
        if lag == 0:
            grad += 2.0 * h
        else:
            for i in range(n):
                v = 0.0
                if 0 <= i + lag < n:
                    v += h[i + lag]
                if 0 <= i - lag < n:
                    v += h[i - lag]
                grad[i] += v
    return project_h(h + alpha * grad, n)

def gen_thue_morse(n):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    return project_h(tm, n)

def gen_golden(n, frac=0.5):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    idx = np.arange(n)
    h = ((idx * phi) % 1.0 < frac).astype(float)
    return project_h(h, n)

def gen_sine(n, f, amp=0.3):
    idx = np.arange(n, dtype=float)
    h = 0.5 + amp * np.sin(2.0 * np.pi * f * idx / n)
    return project_h(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    inits = [best_h, gen_thue_morse(n), gen_golden(n), gen_golden(n, 0.42),
             gen_sine(n, 7, 0.35), gen_sine(n, 13, 0.28)]
    for _ in range(3):
        rb = np.zeros(n)
        rb[np.random.choice(n, n // 2, replace=False)] = 1.0
        inits.append(rb)

    # Phase 1: SLSQP from diverse inits (12%)
    t1 = t0 + budget_s * 0.12
    for h0 in inits:
        if time.time() > t1:
            break
        hc = slsqp_smooth(h0, 4000, n, maxiter=200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: SLSQP multi-T annealing (18%)
    t2 = t0 + budget_s * 0.30
    for T in [15000, 6000, 2500, 1000, 400, 180, 70]:
        if time.time() > t2:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Fast coordinate descent (12%)
    t3 = t0 + budget_s * 0.42
    best_h, best_c5 = coord_descent_fast(best_h, n, t3, max_sweeps=500)

    # Phase 4: Subgradient + SLSQP + coord (22%)
    t4 = t0 + budget_s * 0.64
    for it in range(35):
        if time.time() > t4:
            break
        cool = 1.0 - it / 35.0
        alpha = 0.002 + 0.008 * cool
        k_top = 3 + int(5 * cool)
        hs = subgrad_min_ac(best_h, n, k_top, alpha)
        T_s = 400 + 2500 * cool
        hc = slsqp_smooth(hs, T_s, n, maxiter=250)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent_fast(best_h, n, min(time.time() + 4, t4), max_sweeps=40)

    # Phase 5: Basin hopping with structured perturbations (18%)
    t5 = t0 + budget_s * 0.82
    work_h = best_h.copy()
    work_c5 = best_c5
    i_arr = np.arange(n)
    for it in range(50):
        if time.time() > t5:
            break
        cool = max(0.15, 1.0 - it / 50.0)
        amp = 0.015 + 0.055 * cool
        f_p = np.random.uniform(1.5, 12.0)
        ph = np.random.uniform(0, 2 * np.pi)
        pert = amp * (0.6 * np.random.randn(n) + 0.4 * np.sin(2 * np.pi * f_p * i_arr / n + ph))
        h_p = project_h(work_h + pert, n)
        T_bh = 250 + 2000 * cool
        hc = slsqp_smooth(h_p, T_bh, n, maxiter=300)
        c5v = compute_c5(hc)
        temp = 0.0004 * cool + 0.00004
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 6: Extended coordinate descent (8%)
    t6 = t0 + budget_s * 0.90
    best_h, best_c5 = coord_descent_fast(best_h, n, t6, max_sweeps=400)

    # Phase 7: Final SLSQP polish + coord (8%)
    if time.time() < t0 + budget_s * 0.97:
        hc = slsqp_smooth(best_h, 80, n, maxiter=400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent_fast(best_h, n, t0 + budget_s * 0.99, max_sweeps=100)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
