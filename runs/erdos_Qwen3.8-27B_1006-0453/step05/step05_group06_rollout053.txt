from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            s2 = h.sum()
            if s2 < 1e-14:
                break
            h = np.clip(h * (1.0 + diff / s2), 0.0, 1.0)
    return h

def soft_max_obj(h, T, lam, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(-corr, topk)[:topk]
        c = corr[idx]
        mc = c.max()
        Tc = T * (c - mc)
        Tc -= Tc.max()
        exps = np.exp(Tc)
        sm = mc + np.log(exps.sum()) / T
    else:
        mc = corr.max()
        Tc = T * (corr - mc)
        Tc -= Tc.max()
        exps = np.exp(Tc)
        sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n/2.0)**2
    return val

def soft_max_grad(h, T, lam, topk=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = 2*n - 1
    if topk is not None and topk < m:
        idx = np.argpartition(-corr, topk)[:topk]
        c = corr[idx]
        mc = c.max()
        Tc = T * (c - mc)
        Tc -= Tc.max()
        exps = np.exp(Tc)

    else:
        mc = corr.max()
        Tc = T * (corr - mc)
        Tc -= Tc.max()
        exps = np.exp(Tc)
        w_full = exps / exps.sum()
    grad = np.zeros(n)
    lag_arr = np.arange(m) - n + 1
    for idx in range(m):
        wi = w_full[idx]
        if wi < 1e-15:
            continue
        lg = lag_arr[idx]
        if lg == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lg > 0:
            if lg < n:
                grad[:n-lg] += wi * (1.0 - h[lg:])
                grad[lg:] += -wi * h[:n-lg]
        else:
            p = -lg
            grad[p:] += wi * (1.0 - h[:n-p])
            grad[:n-p] += -wi * h[p:]
    grad = dx * grad + 2.0 * lam * (h.sum() - n/2.0)
    return grad

def lbfgs_step(h0, T, lam, maxiter, ftol, topk=None):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x):
        return soft_max_obj(x, T, lam, topk)
    def grad(x):
        return soft_max_grad(x, T, lam, topk)
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})

    except:

def coord_update_1d(corr, h, i, n):

    old = h[i]
    # Compute A vector (linear coefficients for each lag)

    L = np.arange(1, n)

                                  options={'xatol': 1e-12})
            new_val = float(np.clip(res.x + old, 0.0, 1.0))
            if abs(new_val - old) > 1e-13:
                delta = new_val - old
                corr += delta * A
                corr[n - 1] -= delta * delta
                best[i] = new_val
                improved = True
        iter_count += 1
        if time.time() >= t_end:
            break
        best = project_h(best)
        nc = compute_c5(best)
        if nc < best_c5 - 1e-16:
            best_c5 = nc
        elif not improved:
            break
    return best, best_c5

def h_safe_pos(h, i, L, mask):
    n = len(h)
    out = np.zeros(len(L))
    if mask.any():
        idx = i + L[mask]
        out[mask] = h[idx]
    return out

def h_safe_neg(h, i, L, mask):
    n = len(h)
    out = np.zeros(len(L))
    if mask.any():
        idx = i - L[mask]
        out[mask] = h[idx]
    return out

def make_candidate(n, seed):
    rng = np.random.RandomState(seed)
    cands = []
    h1 = np.zeros(n)
    h1[:n // 2] = 1.0
    cands.append(h1)
    h2 = np.zeros(n)
    h2[::2] = 1.0
    cands.append(h2)
    for fr in [2, 3, 5, 7, 11]:
        h3 = 0.5 + 0.45 * np.sin(2 * np.pi * fr * np.arange(n) / n)
        cands.append(h3)
    for st in [0.05, 0.1, 0.2, 0.4]:
        h4 = 0.5 + st * np.sin(2 * np.pi * np.arange(n) / n) + st * np.sin(4 * np.pi * np.arange(n) / n)
        cands.append(h4)
    for _ in range(4):
        cands.append(rng.beta(2, 2, n))
    return [project_h(c) for c in cands]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B warm-up (12%)
    p1 = t0 + budget_s * 0.12
    for T in [50, 200, 800, 3000]:
        if time.time() > p1:
            break
        hc = lbfgs_step(best_h, T, 500.0, 2000, topk=None)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent main (33%)
    p2 = t0 + budget_s * 0.45
    best_h, best_c5 = coord_sweep(best_h, p2)

    # Phase 3: Active-set L-BFGS-B (12%)
    p3 = t0 + budget_s * 0.57
    for T, tk in [(2000, 10), (5000, 8), (10000, 6), (20000, 5)]:
        if time.time() > p3:
            break
        hc = lbfgs_step(best_h, T, 500.0, 3000, topk=tk)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 4: Multi-restart (25%)
    p4 = t0 + budget_s * 0.82
    cands = make_candidate(n, seed + 1)
    for sc in [0.05, 0.15, 0.3, 0.6]:
        cands.append(project_h(best_h + sc * np.random.randn(n)))
    for h_init in cands:
        if time.time() > p4:
            break
        hc = lbfgs_step(h_init, 1500, 500.0, 1500, topk=20)
        hc, c5v = coord_sweep(hc, min(time.time() + 8, p4))
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Final coordinate descent (10%)
    p5 = t0 + budget_s * 0.94
    best_h, best_c5 = coord_sweep(best_h, p5)

    # Phase 6: Final L-BFGS-B polish (6%)
    if time.time() < t0 + budget_s * 0.99:
        hc = lbfgs_step(best_h, 50000, 800.0, 5000, topk=5)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        # One more coord polish
        if time.time() < t0 + budget_s * 0.99:
            best_h, best_c5 = coord_sweep(best_h, t0 + budget_s * 0.99)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
