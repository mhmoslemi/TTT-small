import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    s = h.sum()
    d = t - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        if r.sum() < 1e-14:
            return h
        h += r * (d / r.sum())
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def _smooth_val(x, T, n):
    c = get_corr(x)
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n + 1e7 * (x.sum() - n / 2) ** 2

def slsqp(h, T, n, mi=1000, ft=1e-15):
    try:
        r = minimize(_smooth_val, h, args=(T, n), method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n, options={'maxiter': mi, 'ftol': ft})
        return project_h(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def coord1(h, n, i, ni=50):
    hb = h.copy()
    hb[i] = 0.0
    cb = get_corr(hb)
    kv = np.arange(-(n - 1), n)
    v1 = i - kv
    v2 = i + kv
    o1 = (v1 >= 0) & (v1 < n)
    o2 = (v2 >= 0) & (v2 < n)
    co = np.zeros(2 * n - 1)
    co[o1] += 1.0 - h[v1[o1]]
    co[o2] -= h[v2[o2]]
    lo, hi = 0.0, 1.0
    for _ in range(ni):
        a = lo + (hi - lo) / 3.0
        b = hi - (hi - lo) / 3.0
        if (cb + a * co).max() < (cb + b * co).max():
            hi = b
        else:
            lo = a
    return (lo + hi) / 2.0

def cd_fast(h, n, t_end, ms=30):
    bh = h.copy()
    bc = compute_c5(bh)
    for _ in range(ms):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return bh, bc
            v = coord1(bh, n, i)
            if abs(v - bh[i]) > 1e-11:
                bh[i] = v
                bh = project_h(bh, n)
                c = compute_c5(bh)
                if c < bc - 1e-15:
                    bc = c
                    imp = True
        if not imp:
            break
    return bh, bc

def _perturb(wh, n, cool, it, rng):
    m = it % 7
    if m == 0:
        return (0.015 + 0.05 * cool) * rng.randn(n)
    elif m == 1:
        f = rng.randint(1, 30)
        ph = rng.uniform(0, 6.2832)
        return (0.02 + 0.035 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif m == 2:
        mk = rng.random(n) < 0.2
        p = np.zeros(n)
        p[mk] = (0.03 + 0.1 * cool) * rng.randn(mk.sum())
        return p
    elif m == 3:
        p = np.zeros(n)
        for _ in range(6):
            a, b = rng.permutation(n)[:2]
            d = (wh[b] - wh[a]) * (0.2 + 0.6 * cool)
            p[a] += d
            p[b] -= d
        return p
    elif m == 4:
        blk = max(3, n // 5)
        b = rng.randint(0, max(1, n // blk))
        p = np.zeros(n)
        p[b * blk:min(n, (b + 1) * blk)] = (0.02 + 0.04 * cool) * rng.randn(blk)
        return p
    elif m == 5:
        f1 = rng.randint(1, 12)
        f2 = rng.randint(13, 35)
        return (0.015 + 0.03 * cool) * (np.sin(2 * np.pi * f1 * np.arange(n) / n) +
                                          np.sin(2 * np.pi * f2 * np.arange(n) / n + 1.1))
    else:
        mk = rng.random(n) < 0.12
        p = np.zeros(n)
        p[mk] = (0.04 + 0.1 * cool) * np.sign(rng.randn(mk.sum()))
        return p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float)
    else:
        bh = np.zeros(160)
        bh[:80] = 1.0
    n = len(bh)
    bh = project_h(bh, n)
    bc = compute_c5(bh)

    # Phase 1: SLSQP annealing (12% budget)
    for T in [30, 300, 3000, 30000, 150000, 600000]:
        if time.time() > t0 + budget_s * 0.12:
            break
        h = slsqp(bh, T, n, 1500, 1e-16)
        c = compute_c5(h)
        if c < bc:
            bc, bh = c, h

    # Phase 2: Coordinate descent (10% budget)
    bh, bc = cd_fast(bh, n, t0 + budget_s * 0.22, 25)

    # Phase 3: Basin hopping (50% budget)
    wh, wc = bh.copy(), bc
    stuck = 0
    t_bh_end = t0 + budget_s * 0.72
    for it in range(800):
        if time.time() > t_bh_end - 30:
            break
        cool = max(0.01, 1.0 - it / 400.0)
        p = _perturb(wh, n, cool, it, rng)
        hp = project_h(wh + p, n)
        Tbh = 300 + 50000 * cool
        hc = slsqp(hp, Tbh, n, 350, 1e-14)
        c = compute_c5(hc)
        temp = 0.0003 * cool + 0.00002
        if c < wc or rng.rand() < np.exp(-(c - wc) / temp):
            wh, wc = hc, c
            stuck = 0
            if c < bc:
                bc, bh = c, hc
                if time.time() < t_bh_end - 60:
                    bh, bc = cd_fast(bh, n, t_bh_end - 60, 4)
        else:
            stuck += 1
        if stuck > 70:
            wh = project_h(wh + 0.2 * rng.randn(n), n)
            stuck = 0

    if wc < bc:
        bc, bh = wc, wh

    # Phase 4: SLSQP polish with high T (10% budget)
    for T in [200000, 1000000, 5000000]:
        if time.time() > t0 + budget_s * 0.85:
            break
        h = slsqp(bh, T, n, 3000, 1e-16)
        c = compute_c5(h)
        if c < bc:
            bc, bh = c, h

    # Phase 5: Extended coordinate descent (12% budget)
    bh, bc = cd_fast(bh, n, t0 + budget_s * 0.97, 50)

    # Final micro-polish
    if time.time() < t0 + budget_s * 0.99:
        h = slsqp(bh, 10000000, n, 2000, 1e-16)
        c = compute_c5(h)
        if c < bc:
            bc, bh = c, h
        bh, bc = cd_fast(bh, n, t0 + budget_s * 0.995, 10)

    bh = project_h(bh, n)
    bc = compute_c5(bh)
    return bh, bc, n
