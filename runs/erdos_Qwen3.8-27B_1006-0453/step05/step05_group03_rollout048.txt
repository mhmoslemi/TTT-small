import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_1d_exact(h, n, i):
    """Exact minimization of max-k (a_k + b_k * x) for x in [0,1]."""
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    # coef for h[i]: contribution to corr[m] when h[i] changes
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n - 1), n)
    j_fwd = i + lags      # h[i] as the first factor: (1-h[i+k])
    j_bwd = i - lags      # h[i] as the second factor: -h[i-k]
    v1 = (j_fwd >= 0) & (j_fwd < n)
    v2 = (j_bwd >= 0) & (j_bwd < n)
    coefs[v1] += (1.0 - h[j_fwd[v1]])
    coefs[v2] += (-h[j_bwd[v2]])
    # Now minimize max_k(corr_base[k] + x * coefs[k]) for x in [0,1]
    # Binary search on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = corr_base + mid * coefs
        mx = np.max(vals)
        active = np.where(vals >= mx - 1e-12)[0]
        slope_lo = np.min(coefs[active])
        slope_hi = np.max(coefs[active])
        if slope_hi < -1e-12:
            hi = mid
        elif slope_lo > 1e-12:
            lo = mid
        else:
            break
    x_opt = (lo + hi) / 2
    # Also check endpoints
    best_x, best_val = x_opt, np.max(corr_base + x_opt * coefs)
    for xc in [0.0, 1.0]:
        vc = np.max(corr_base + xc * coefs)
        if vc < best_val:
            best_val, best_x = vc, xc
    return best_x

def coord_descent(h, n, t_end, max_sweeps=200):

                best_h = h_try

def slsqp_smooth(h, T, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        sh = np.clip(T * (corr - m), -50, 50)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def sa_block_moves(h, n, c5, t_end, max_iters=5000):

    best_h = h.copy()
    best_c5 = c5
    h_cur = h.copy()
    c5_cur = c5
    for it in range(max_iters):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max_iters
        temp = 0.002 * cool + 0.0001
        h_try = h_cur.copy()
        move_type = np.random.randint(4)
        if move_type == 0:
            # Swap two positions
            i, j = np.random.choice(n, 2, replace=False)
            di = h_cur[i] - h_cur[j]
            h_try[i] = h_cur[j]
            h_try[j] = h_cur[i]
        elif move_type == 1:
            # Flip a block
            blen = np.random.randint(3, min(9, n // 4))
            start = np.random.randint(0, n - blen)
            h_try[start:start+blen] = 1.0 - h_cur[start:start+blen]
            h_try = project_h(h_try, n)
        elif move_type == 2:
            # Perturb 3-5 random positions
            k = np.random.randint(3, 6)
            idx = np.random.choice(n, k, replace=False)
            h_try[idx] = np.clip(h_cur[idx] + np.random.randn(k) * 0.15, 0, 1)
            h_try = project_h(h_try, n)
        else:
            # Reflect a segment
            slen = np.random.randint(4, min(12, n // 3))
            start = np.random.randint(0, n - slen)
            h_try[start:start+slen] = h_cur[start:start+slen][::-1].copy()
            h_try = project_h(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur or np.random.rand() < np.exp(-(c5_try - c5_cur) / temp):
            h_cur = h_try
            c5_cur = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5

def gen_init(n, method='mixed'):
    if method == 'half':
        h = np.zeros(n)
        h[:n//2] = 1.0
    elif method == 'thue':
        h = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    elif method == 'random':
        h = np.zeros(n)
        h[np.random.choice(n, n//2, replace=False)] = 1.0
    elif method == 'ramp':
        x = np.linspace(0, 1, n)
        h = ((x * 4) % 1.0 < 0.5).astype(float)
    elif method == 'sine':
        x = np.linspace(0, 2*np.pi, n, endpoint=False)
        h = (np.sin(3*x) > 0).astype(float)
    else:
        h = np.random.rand(n)
    return project_h(h, n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _slsqp_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    sh = np.clip(T * (corr - m), -50, 50)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx
