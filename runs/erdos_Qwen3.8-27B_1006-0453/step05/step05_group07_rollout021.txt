def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5

        return h.copy()

    else:

        else:
