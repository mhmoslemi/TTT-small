        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # Swap: h[i] 1->0, h[j] 0->1
        h_b[i] = 0
        h_b[j] = 1
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        improved = nc5 < best_c5
        if improved or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            corr = nc
            if improved:
                best_c5 = nc5
                best_h = h_b.copy()
            # Update ones/zeros
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
        else:
            h_b[i] = 1
            h_b[j] = 0
        T = max(T_min, T * cool_rate)
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

_T = [1.0]

def _sobj(x):
    T = _T[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def _sgrad(x):
    T = _T[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=300):
    _T[0] = T
    try:
        r = minimize(h, _sobj, _sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

_CB = [np.zeros(1)]
_CC = [np.zeros(1)]

def _f1d(hi):
    return float(np.max(_CB[0] + hi * _CC[0]))

def coord_desc(h, n, t_end, sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for _sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr_cache = np.correlate(best, 1.0 - best, mode='full')
        cur_max = corr_cache.max()
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            _CB[0] = corr_cache - old * coef
            _CC[0] = coef
            r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-12})
            nv = float(r.x)
            if abs(nv - old) < 1e-11:
                continue
            # Quick check without projection
            trial_corr = corr_cache + (nv - old) * coef
            if trial_corr.max() >= cur_max - 1e-13:
                continue
            best[i] = nv
            best = proj_h(best, n)
            corr_cache = np.correlate(best, 1.0 - best, mode='full')
            cur_max = corr_cache.max()
            imp = True
        if not imp:
            break
    best_c5 = compute_c5(best)
    return best, best_c5

def bin_sa(h, n, t_end, max_iters=400000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    if len(ones) > n // 2:
        h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        h_b[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T_init = 0.02
    T_min = 0.00005
    cool_rate = (T_min / T_init) ** (1.0 / max(max_iters, 1))
    T = T_init
    for it in range(max_iters):
        if time.time() > t_end:
            break
        i = int(ones[np.random.randint(len(ones))])
        j = int(zeros[np.random.randint(len(zeros))])
        h_b[i] = 0.0
        h_b[j] = 1.0
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5:
            corr = nc
            best_c5 = nc5
            best_h = h_b.copy()
            ones = np.delete(ones, np.where(ones == i)[0])
            ones = np.append(ones, j)
            zeros = np.delete(zeros, np.where(zeros == j)[0])
            zeros = np.append(zeros, i)
        elif np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            corr = nc
            ones = np.delete(ones, np.where(ones == i)[0])
            ones = np.append(ones, j)
            zeros = np.delete(zeros, np.where(zeros == j)[0])
            zeros = np.append(zeros, i)
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
        T = max(T_min, T * cool_rate)
    return best_h, best_c5

def subgrad_step(h, n, steps=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = 0.005
    for s in range(steps):
        if time.time() > t0 + budget_s * 0.95:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        mx = corr.max()
        # Subgradient: average gradient over lags near max
        close = corr > mx - 0.001
        n_close = max(1, int(close.sum()))
        g = np.zeros(n)
        idx = np.where(close)[0]
        for k in idx:
            shift = k - n + 1
            if shift >= 0:
                for i in range(n - shift):
                    g[i] += (1.0 - best[i + shift])
                    g[i + shift] -= best[i]
            else:
                s2 = -shift
                for i in range(s2, n):
                    g[i] += (1.0 - best[i - s2])
                    g[i - s2] -= best[i]
        g = g * (2.0 / n) / n_close
        # Projected subgradient step
        trial = proj_h(best - lr * g, n)
        c5t = compute_c5(trial)
        if c5t < best_c5:
            best = trial
            best_c5 = c5t
        lr *= 0.95
    return best, best_c5

t0 = 0.0
budget_s = 1100.0

def basin_hop(h, n, t_end, n_iters=80):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        pert_scale = 0.03 + 0.12 * cool
        # Mix of noise types
        r = np.random.random()
        if r < 0.4:
            pert = pert_scale * np.random.randn(n)
        elif r < 0.7:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = pert_scale * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        else:
            mask = np.random.random(n) < 0.3
            pert = np.zeros(n)
            pert[mask] = pert_scale * 2 * np.random.randn(mask.sum())
        hp = proj_h(work + pert, n)
        T_val = 500.0 + 8000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.001 * cool + 0.00005
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global t0, budget_s
    np.random.seed(seed)
    t0 = time.time()
    budget_s = budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B warmup (8% = ~88s)
    t1 = t0 + budget_s * 0.06
    for T in [500, 3000, 15000]:
        if time.time() > t1:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Coordinate descent (12% = ~132s)
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.18, sweeps=120)

    # Phase 3: Binary SA (25% = ~275s)
    t3_end = t0 + budget_s * 0.43
    h_bin, c5_bin = bin_sa(best_h, n, t3_end, max_iters=300000)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin
        # Refine binary with L-BFGS-B
        best_h = lbfgs_opt(best_h, 3000.0, n, iters=500)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.50, sweeps=60)

    # Phase 4: Basin hopping (20% = ~220s)
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.70, n_iters=100)

    # Phase 5: Coordinate descent after basin hopping (15% = ~165s)
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.85, sweeps=100)

    # Phase 6: Subgradient polishing (5%)
    if time.time() < t0 + budget_s * 0.95:
        best_h, c5_sg = subgrad_step(best_h, n, steps=80)
        if c5_sg < best_c5:
            best_c5 = c5_sg

    # Phase 7: Final L-BFGS-B with very high T (5%)
    if time.time() < t0 + budget_s * 0.98:
        best_h = lbfgs_opt(best_h, 100000.0, n, iters=600)
        best_c5 = compute_c5(best_h)
        # Tiny coord desc polish
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.995, sweeps=15)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
