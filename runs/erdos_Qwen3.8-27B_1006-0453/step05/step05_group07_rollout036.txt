```python
def compute_coef(h, i, n):
    mf = 2*n - 1
    coef = np.zeros(mf)
    # (1-h) part: j1 = i + k - (n-1), valid when 0 <= j1 < n
    # k = j1 - i + (n-1), valid k: n-1-i <= k <= n-1-i+n-1
    lo = max(0, n-1-i)
    hi = min(mf, n-1-i+n)
    if lo < hi:
        j1 = np.arange(lo, hi) - (n-1) + i
        coef[lo:hi] += 1.0 - h[j1]
    # -h part: j2 = i - k + (n-1), valid when 0 <= j2 < n
    # k = i - j2 + (n-1), valid k: i-(n-1) <= k <= i
    lo2 = max(0, i-(n-1))
    hi2 = min(mf, i+1)
    if lo2 < hi2:
        j2 = i + (n-1) - np.arange(lo2, hi2)
        coef[lo2:hi2] -= h[j2]
    return coef
```

This is O(n) per coordinate with numpy operations, which is fast.

```python
def compute_c5(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = 0.5 * n
    for _ in range(100):
        s = h.sum()
        d = s - tgt
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h > 1e-12
            if mask.sum() == 0:
                break
            h[mask] -= d / mask.sum()
        else:
            mask = h < 1.0 - 1e-12
            if mask.sum() == 0:
                break
            h[mask] += (-d) / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgs_opt(h, T, n, iters=300):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n
    def grad(x):

        m = c.max()

        w = e / e.sum()
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-14:
                continue
            lo1 = max(0, n-1-k)
            hi1 = min(n, 2*n-1-k)
            if lo1 < hi1:

            lo2 = max(0, k-(n-1))
            hi2 = min(n, k+1)
            if lo2 < hi2:
                g[lo2:hi2] -= wk * x[np.arange(lo2,hi2) - k + (n-1)]
        return g * (2.0 / n)
    try:
        r = minimize(h, obj, grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})

    best_c5 = compute_c5(best)
    mf = 2 * n - 1
```
