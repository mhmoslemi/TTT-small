import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_1d(v, corr, lin, old, n):
    """Objective for 1D subproblem."""
    dv = v - old
    m = np.max(corr + dv * lin)
    vn1 = corr[n - 1] + dv * lin[n - 1] - dv * dv
    return max(m, vn1)

def vec_cd(h, n, t_end, max_sweeps=5000, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    N2 = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            if old < 1e-7 or old > 1.0 - 1e-7:
                continue
            k_lo = i
            k_hi = min(i + n - 1, N2 - 1)
            ks = np.arange(k_lo, k_hi + 1)
            j1 = i - ks + n - 1
            j2 = i + ks - n + 1
            m1 = (j1 >= 0) & (j1 < n)
            m2 = (j2 >= 0) & (j2 < n)
            lin = np.zeros(N2)
            seg = np.zeros(len(ks))
            seg[m1] += (1.0 - h[j1[m1]])
            seg[m2] -= h[j2[m2]]
