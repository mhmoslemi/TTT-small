            if abs(res.x - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = res.x
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved and sweep > 5:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter=35, scale=0.08):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(1.0 - it / max(n_iter, 1), 0.1)
        r = np.random.random()
        if r < 0.35:
            pert = scale * cool * np.random.randn(n)
        elif r < 0.55:
            blen = np.random.randint(4, max(n // 5, 5))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs+blen] = scale * 1.5 * cool * np.random.randn(blen)
        elif r < 0.75:
            f = np.random.randint(1, 10)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        else:
            # Targeted: perturb positions involved in worst shifts
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            top_idx = np.argsort(corr)[-5:]
            shifts = (top_idx - (n - 1)).astype(int)
            pert = np.zeros(n)
            for k in shifts:
                if 0 <= k < n:
                    pert[:n-k] += (cur_h[k:n] - 0.5)
                elif -n < k < 0:
                    pert[-k:n] += (cur_h[:n+k] - 0.5)
            pert *= scale * 0.3 * cool
        h_pert = project_h(cur_h + pert, n)
        T = 2000 + 20000 * cool
        h_ref = slsqp_refine(h_pert, n, T, 250, t_end)
        c5_ref = compute_c5(h_ref)
        temp = 0.0004 * cool + 0.00008
        if c5_ref < cur_c5 or (c5_ref > cur_c5 and np.random.rand() < np.exp(-(c5_ref - cur_c5) / temp)):
            cur_h, cur_c5 = h_ref, c5_ref
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, cur_h.copy()
    return best_h, best_c5

def optimize_n(h_init, n, t_end):
    h_init = project_h(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    # SLSQP warmup
    for T in [50, 200, 1000, 5000]:
        if time.time() > t_end - 30:
            break
        hc = slsqp_refine(best_h, n, T, 500, t_end - 30)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
    # Subgradient
    if time.time() < t_end - 40:
        hg, cg = subgrad_top(best_h, n, time.time() + (t_end - time.time()) * 0.25, 120, 12)
        if cg < best_c5:
            best_c5, best_h = cg, hg
    # SLSQP with flat penalty
    if time.time() < t_end - 40:
        hc = slsqp_refine(best_h, n, 20000, 700, t_end - 40, flat_pen=0.02)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
    # Basin hopping
    if time.time() < t_end - 50:
        bh_h, bh_c5 = basin_hop(best_h, n, time.time() + (t_end - time.time()) * 0.35, 30, 0.07)
        if bh_c5 < best_c5:
            best_c5, best_h = bh_c5, bh_h
    # Coordinate descent
    if time.time() < t_end - 15:
        best_h, best_c5 = coord_desc(best_h, n, t_end - 15, 50, 14)
    # Final SLSQP
    if time.time() < t_end:
        hc = slsqp_refine(best_h, n, 500000, 1200, t_end)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
    return project_h(best_h, n), best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = 0.5 * np.ones(200)
    n0 = len(init_h)
    # Main optimization
    best_h, best_c5 = optimize_n(init_h, n0, t0 + budget_s * 0.45)
    # Try other resolutions
    trys = []
    for nc in [160, 240, 120, 200]:
        if nc != n0 and nc != len(best_h):
            trys.append(nc)
    for nc in trys:
        if nc != len(best_h) and time.time() < t_end - budget_s * 0.2:
            xs = np.linspace(0, 2, nc, endpoint=False)
            xo = np.linspace(0, 2, len(best_h), endpoint=False)
            h_interp = np.interp(xs, xo, best_h)
            h_interp = project_h(h_interp, nc)
            t_alloc = min(t_end - budget_s * 0.08, time.time() + budget_s * 0.22)
            hc, cc = optimize_n(h_interp, nc, t_alloc)
            if cc < best_c5:
                best_c5, best_h = cc, hc
    # Final polish
    nb = len(best_h)
    if time.time() < t_end - 10:
        best_h, best_c5 = coord_desc(best_h, nb, t_end - 10, 45, 20)
    if time.time() < t_end - 3:
        hc = slsqp_refine(best_h, nb, 2000000, 800, t_end - 2)
        c5 = compute_c5(hc)
        if c5 < best_c5:
            best_c5, best_h = c5, hc
    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
