import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _smo(x, T, M):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    if M >= len(c):
        top = c
    else:
        idx = np.argpartition(-c, M)[:M]
        top = c[idx]
    tm = top.max()
    return (tm + np.log(np.sum(np.exp(T * (top - tm)))) / T) * 2.0 / n

def _smg(x, T, M):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    if M >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(-c, M)[:M]
    top = c[idx]
    tm = top.max()
    e = np.exp(T * (top - tm))
    w = e / e.sum()
    g = np.zeros(n)
    for ki, wk in zip(idx, w):
        if wk < 1e-14:
            continue
        d = ki - n + 1
        if d == 0:
            g += wk * (1.0 - 2.0 * x)
        elif d > 0:
            g[:n-d] += wk * (1.0 - x[d:])
            g[d:] -= wk * x[:n-d]
        else:
            d2 = -d
            g[d2:] += wk * (1.0 - x[:n-d2])
            g[:n-d2] -= wk * x[d2:]
    return g * (2.0 / n)

                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})
        return proj_h(r.x, n)

        return h
