from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T, flat_pen=0.0, n_pen=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e6:
        val = m
    else:
        # For numerical stability
        shifted = T * (corr - m)
        val = m + np.log(np.sum(np.exp(shifted))) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-n_pen:]
        val += flat_pen * np.std(top)
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_run(x, T, n, maxiter=500, ftol=1e-14, flat_pen=0.0):
    try:
        res = minimize(lambda v: smooth_obj(v, T, flat_pen), x, method='SLSQP',

        return project_h(np.clip(res.x, 0, 1))
    except:
        return x

                improved = True
        if not improved:

    return best_h, best_c5

def subgrad_step(h, alpha, n_topk, mom_v, beta=0.5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if mom_v is not None:
        new_grad = beta * mom_v + (1.0 - beta) * grad
    else:
        new_grad = grad
    return project_h(h - alpha * grad), new_grad

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, T, flat_pen=0.0, n_pen=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e6:
        val = m
    else:
        shifted = T * (corr - m)
        shifted = np.clip(shifted, -500, 500)
        val = m + np.log(np.sum(np.exp(shifted))) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-n_pen:]
        val += flat_pen * np.std(top)
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_run(x, T, n, maxiter=500, ftol=1e-14, flat_pen=0.0):
    try:
        res = minimize(lambda v: smooth_obj(v, T, flat_pen), x, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda v: v.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return x

def coord_descent(h, n, t_end, max_sweeps=80, n_top=25):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in range(n):
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            if abs(res.x - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = res.x
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_step(h, alpha, n_topk, mom_v, beta=0.5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if mom_v is not None:
        new_grad = beta * mom_v + (1.0 - beta) * grad
    else:
        new_grad = grad
    return project_h(h - alpha * grad), new_grad

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(1.0 - it / max(n_iter, 1), 0.0)
        r = np.random.random()
        if r < 0.3:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-6:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n-k] += (work_h[k:n] - 0.5)
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n+k] - 0.5)
            pert *= scale * 0.5 * cool
        elif r < 0.5:
            f = np.random.randint(1, 16)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.5 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.7:
            mask = np.random.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = scale * 1.3 * cool * np.random.randn(max(int(mask.sum()), 1))
        elif r < 0.85:
            blen = np.random.randint(max(n // 8, 2), max(n // 3, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs+blen] = scale * cool * np.random.randn(blen)
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp_run(h_pert, T, n, 250, 1e-14)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.00008
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_h(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # Phase 1: SLSQP warmup with increasing smoothness
    if t_end - time.time() > 25:
        for T in [50, 200, 1000, 5000, 20000, 80000]:
            if t_end - time.time() < 12:
                break
            hc = slsqp_run(best_h, T, n, 400, 1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Phase 2: Subgradient + SLSQP
    if t_end - time.time() > 35:
        mom_v = np.zeros(n)
        for it in range(35):
            if t_end - time.time() < 12:
                break
            cool = 1.0 - it / 35.0
            alpha = 0.004 + 0.01 * cool
            n_topk = 5 + int(15 * cool)
            hs, g = subgrad_step(best_h, alpha, n_topk, mom_v, beta=0.5)
            mom_v = g
            T = 500 + 12000 * cool
            hc = slsqp_run(hs, T, n, 300, 1e-14, flat_pen=0.04 * cool)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Phase 3: Aggressive coordinate descent
    if t_end - time.time() > 35:
        best_h, best_c5 = coord_descent(best_h, n, t_end - (t_end - time.time()) * 0.3,
                                        max_sweeps=60, n_top=25)
    # Phase 4: Basin hopping
    if t_end - time.time() > 45:
        bh_h, bh_c5 = basin_hop(best_h, n, t_end - (t_end - time.time()) * 0.35,
                                30, 0.1, 500, 6000)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
        if t_end - time.time() > 15:
            best_h, best_c5 = coord_descent(best_h, n, t_end - (t_end - time.time()) * 0.4,
                                            max_sweeps=40, n_top=20)
    # Phase 5: SLSQP polish
    if t_end - time.time() > 12:
        hc = slsqp_run(best_h, 200000, n, 600, 1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Phase 6: Final coord descent
    if t_end - time.time() > 8:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, max_sweeps=25, n_top=30)
    return project_h(best_h), best_c5
