import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if T > 1e8:
        return m * dx
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp_solve(h, T, n, maxiter=600, ftol=1e-15):
    try:
        res = minimize(lambda x: smooth_obj(x, T, n), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_nesterov(h, n, n_topk, lr, v=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if v is None:
        v = np.zeros(n)
    v_new = 0.6 * v - lr * grad
    return project_h(h + v_new), v_new

def coord_descent_full(h, n, t_end, max_sweeps=80, start_sweep=0):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    shifts_all = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            ip = i + shifts_all
            im = i - shifts_all
            B = np.where((ip >= 0) & (ip < n), 1.0 - best_h[np.clip(ip, 0, n-1)], 0.0) \
                - np.where((im >= 0) & (im < n), best_h[np.clip(im, 0, n-1)], 0.0)
            base = corr - best_h[i] * B
            res = minimize_scalar(lambda t: np.max(base + t * B),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop_improved(h, n, t_end, n_iter, scale):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.25:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-8:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n-k] += (work_h[k:n] - 0.5)
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n+k] - 0.5)
            pert *= scale * 0.5 * cool
        elif r < 0.5:
            f = np.random.randint(1, 15)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.8 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.7:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.8 * cool * np.random.randn(max(int(mask.sum()), 1))
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = 500 + 20000 * cool
        hc = slsqp_solve(h_pert, T, n, 300, 1e-14)
        c5v = compute_c5(hc)
        temp = 0.0015 * cool + 0.0002
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_at_n(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # SLSQP warmup with increasing T
    for T in [50, 200, 1000, 5000, 20000]:
        if t_end - time.time() < 15:
            break
        hc = slsqp_solve(best_h, T, n, 500, 1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Nesterov subgradient
    if t_end - time.time() > 40:
        v = np.zeros(n)
        for it in range(50):
            if t_end - time.time() < 10:
                break
            cool = 1.0 - it / 50.0
            lr = (0.003 + 0.008 * cool) / (1 + it * 0.02)
            n_topk = 8 + int(15 * cool)
            hs, v = subgrad_nesterov(best_h, n, n_topk, lr, v)
            T = 200 + 10000 * cool
            hc = slsqp_solve(hs, T, n, 300, 1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Basin hopping (coarse)
    if t_end - time.time() > 50:
        bh_end = t_end - (t_end - time.time()) * 0.25
        bh_h, bh_c5 = basin_hop_improved(best_h, n, bh_end, 35, 0.15)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Coordinate descent with ALL shifts
    if t_end - time.time() > 30:
        cd_end = t_end - (t_end - time.time()) * 0.2
        best_h, best_c5 = coord_descent_full(best_h, n, cd_end, 60)
    # Basin hopping (fine)
    if t_end - time.time() > 25:
        bh_end = t_end - (t_end - time.time()) * 0.3
        bh_h, bh_c5 = basin_hop_improved(best_h, n, bh_end, 20, 0.05)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # More coordinate descent
    if t_end - time.time() > 20:
        best_h, best_c5 = coord_descent_full(best_h, n, t_end - 5, 40)
    # Final SLSQP polish
    if t_end - time.time() > 10:
        hc = slsqp_solve(best_h, 500000, n, 1000, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    return project_h(best_h), compute_c5(best_h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:

        h3 = project_h(h3)
