    beta = 0.9
    temp = temp0
    stall = 0
    it = 0
    while time.time() < deadline:
        y = h + beta * (h - h_prev)
        h_prev = h
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk - 1e-15:
            best_pk = pk
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        top = np.argpartition(corr, -k)[-k:]
        w = np.exp((corr[top] - corr[top].max()) / max(temp, 1e-10))
        w /= w.sum()
        grad = np.zeros(N)
        center = N - 1
        for wi, ai in zip(w, top):
            L = int(ai - center)
            if L == 0:
                grad += wi * (1.0 - 2.0 * y)
            elif L > 0:
                seg = N - L
                grad[L:] += wi * (1.0 - y[:seg])
                grad[:seg] -= wi * y[L:]
            else:
                M = -L
                if M < N:
                    seg = N - M
                    grad[:seg] += wi * (1.0 - y[M:])
                    grad[M:] -= wi * y[:seg]
        h_new = project_sum(y - lr * grad, target)
        h = h_new
        it += 1
        if it % 200 == 0:
            lr = max(0.003, lr * 0.996)
            temp = max(0.0005, temp * 0.995)
            beta = min(0.97, 0.88 + it * 0.00015)
        if stall > 400:
            h = best_h + np.random.randn(N) * 0.025
            h = project_sum(h, target)
            h_prev = h.copy()
            lr = max(lr, lr0 * 0.9)
            stall = 0
    return best_h, best_pk * (2.0 / N)

def binary_opt(h0, N, deadline, n_tries=15, kick_freq=80):
    h = (h0 > 0.5).astype(float).copy()
    nh = N // 2
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            idx = idx[np.random.permutation(len(idx))][:n1 - nh]
            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh - n1]
            h[idx] = 1
    c = fft_corr(h)
    best_int = int(np.max(c))
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_sweep_int = int(c.max())
        best_sweep_h = None
        best_sweep_c = None
        for _ in range(n_tries):
            p = ones[np.random.randint(len(ones))]
            q = zeros[np.random.randint(len(zeros))]
            h[p] = 0
            h[q] = 1
            c_t = fft_corr(h)
            ct_int = int(c_t.max())
            if ct_int < best_sweep_int:
                best_sweep_int = ct_int
                best_sweep_h = h.copy()
                best_sweep_c = c_t.copy()
            h[p] = 1
            h[q] = 0
        if best_sweep_h is not None:
            h = best_sweep_h
            c = best_sweep_c
            stall = 0
            if best_sweep_int < best_int:
                best_int = best_sweep_int
                best_h = h.copy()
        else:
            stall += 1
            if stall > kick_freq:
                if len(ones) >= 2 and len(zeros) >= 2:
                    i1 = np.random.choice(len(ones), 2, replace=False)
                    i2 = np.random.choice(len(zeros), 2, replace=False)
                    ps = ones[i1]
                    qs = zeros[i2]
                    for pp in ps:
                        h[pp] = 0
                    for qq in qs:
                        h[qq] = 1
                    c = fft_corr(h)
                stall = 0
    return best_h, best_int * (2.0 / N)

def interp_to_N(h, N_new):
    N_old = len(h)
    x_old = np.linspace(0, 2, N_old, endpoint=False)
    x_new = np.linspace(0, 2, N_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary optimization from initial (120s)
    d1 = min(deadline - 750, t0 + 120)
    hb, cb = binary_opt(h0, N0, d1, n_tries=12, kick_freq=60)
    c5b = compute_c5(hb, N0)
    if c5b < best_c5:
        best_c5, best_h = c5b, hb.copy()

    # Phase 2: Continuous subgradient optimization (380s)
    d2 = min(deadline - 350, max(d1, t0) + 380)
    hc = best_h + np.random.randn(N0) * 0.05
    hc = project_sum(hc, N0 / 2.0)
    hn, c5n = continuous_opt(hc, N0, d2, k=50, lr0=0.022, temp0=0.0025)
    if c5n < best_c5:
        best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary polish from best (180s)
    d3 = max(d2, t0) + 180
    d3 = min(deadline - 150, d3)
    hb3, cb3 = binary_opt(best_h, N0, d3, n_tries=20, kick_freq=100)
    c5b3 = compute_c5(hb3, N0)
    if c5b3 < best_c5:
        best_c5, best_h = c5b3, hb3.copy()

    # Phase 4: Second continuous from perturbed best (180s)
    d4 = max(d3, t0) + 180
    d4 = min(deadline - 60, d4)
    hc4 = best_h + np.random.randn(N0) * 0.04
    hc4 = project_sum(hc4, N0 / 2.0)
    hn4, c5n4 = continuous_opt(hc4, N0, d4, k=60, lr0=0.018, temp0=0.002)
    if c5n4 < best_c5:
        best_c5, best_h = c5n4, hn4.copy()

    # Phase 5: Multi-resolution attempts (100s remaining)
    for N_try in [570, 760, 950]:
        if time.time() > deadline - 20:
            break
        N_try = N_try if N_try % 2 == 0 else N_try + 1
        h_interp = interp_to_N(best_h, N_try)
        h_bt = (h_interp > 0.5).astype(float)
        nh = N_try // 2
        n1 = int(h_bt.sum())
        if n1 > nh:
            idx = np.where(h_bt == 1)[0]
            idx = idx[np.random.permutation(len(idx))][:n1 - nh]
            h_bt[idx] = 0
        elif n1 < nh:
            idx = np.where(h_bt == 0)[0]
            idx = idx[np.random.permutation(len(idx))][:nh - n1]
            h_bt[idx] = 1
        sub_dl = min(deadline - 10, time.time() + 40)
        h_rt, c5_rt = binary_opt(h_bt, N_try, sub_dl, n_tries=10, kick_freq=50)
        c5_rt = compute_c5(h_rt, N_try)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()

    # Phase 6: Final Nesterov polish
    if time.time() < deadline - 3:
        hF = best_h + np.random.randn(len(best_h)) * 0.02
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = continuous_opt(hF, len(best_h), deadline - 2, k=70, lr0=0.01, temp0=0.001)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
