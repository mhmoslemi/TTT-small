def compute_grad(h, n, shifts, weights):
    """Compute weighted sum of gradients for given shifts"""
    n = len(h)
    grad = np.zeros(n)
    i_idx = np.arange(n)
    for k_idx, w in zip(shifts, weights):

            continue

        jp = i_idx + shift
        jm = i_idx - shift
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n-1)
        jmc = np.clip(jm, 0, n-1)
        grad += w * ((1 - h[jpc]) * v1 - h[jmc] * v2)
    return grad * (2.0 / n)

    return 0.5 * (lo + hi)

def coord_descent(h, n, top_k, t_end, max_sweeps=5000, recompute_every=1):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        top_shifts = (top_idx - (n - 1)).astype(np.int64)
        improved = 0
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            d = np.empty(top_k)
            for t in range(top_k):
                k = top_shifts[t]
                val = 0.0
                ip = i + k
                im = i - k
                if 0 <= ip < n:
                    val += 1.0 - best_h[ip]
                if 0 <= im < n:
                    val -= best_h[im]
                d[t] = val
            base = corr[top_idx] - best_h[i] * d
            t_opt = min_max_linear(base, d)
            if abs(t_opt - best_h[i]) > 1e-13:
                best_h[i] = t_opt
                improved += 1
        if improved == 0:
            break
        best_h = project_h(best_h)
        c5_new = compute_c5(best_h)
        if c5_new <= best_c5 + 1e-15:
            best_c5 = c5_new
    return best_h, best_c5

def smooth_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    if T > 1e5:
        return m * dx
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx

def slsqp_step(h, n, T, maxiter=800):
    try:
        res = minimize(lambda x: smooth_obj(x, n, T), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def grad_of_smooth(h, n, T, top_m=100):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    p = np.exp(T * (corr - m))
    p /= p.sum()
    order = np.argsort(p)[::-1][:top_m]
    i_idx = np.arange(n).astype(np.int64)
    grad = np.zeros(n)
    for ki in order:
        w = p[ki]
        if w < 1e-10:
            break
        shift = int(ki - (n - 1))
        jp = i_idx + shift
        jm = i_idx - shift
        v1 = (jp >= 0) & (jp < n)
        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n - 1)
        jmc = np.clip(jm, 0, n - 1)
        grad += w * ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64))
    return grad * dx

def subgrad_loop(h, n, t_end, n_iter=200, T0=500, T1=200000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    mom = np.zeros(n)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        T = T0 + (T1 - T0) * (1 - cool)
        g = grad_of_smooth(best_h, n, T)
        mom = 0.5 * mom + 0.5 * g
        alpha = 0.001 + 0.004 * cool
        h_try = project_h(best_h - alpha * mom)
        hc = slsqp_step(h_try, n, max(T, 100), 300)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    return best_h, best_c5

def basin_hopping(h, n, t_end, n_rounds, base_scale):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        cool = 1.0 - r / max(n_rounds, 1)
        s = base_scale * (0.3 + 0.7 * cool)
        r_choice = np.random.random()
        if r_choice < 0.3:
            pert = s * np.random.randn(n)
        elif r_choice < 0.5:
            f = np.random.randint(2, 15)
            p = np.random.uniform(0, 2 * np.pi)
            pert = s * 0.8 * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r_choice < 0.7:
            blen = np.random.randint(max(n // 8, 3), max(n // 3, 4))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = s * 1.2 * np.random.randn(blen)
        else:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = s * 2.0 * np.random.randn(max(int(mask.sum()), 1))
        h_p = project_h(best_h + pert)
        T = 500 + 15000 * cool
        hc = slsqp_step(h_p, n, T, 400)
        hc, c5_cd = coord_descent(hc, n, 40, min(t_end, time.time() + 8), 500)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.00008
        if c5v < best_c5 or (c5v > best_c5 and np.random.rand() < np.exp(-(c5v - best_c5) / temp)):
            best_h = hc
            best_c5 = c5v
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=np.float64)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive coordinate descent
    t1 = t0 + budget_s * 0.22
    best_h, best_c5 = coord_descent(best_h, n, 50, t1, 5000)

    # Phase 2: SLSQP with increasing T
    t2 = t0 + budget_s * 0.32
    for T in [200, 2000, 20000, 200000]:
        if time.time() > t2:
            break
        hc = slsqp_step(best_h, n, T, 700)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: More coordinate descent
    t3 = t0 + budget_s * 0.48
    best_h, best_c5 = coord_descent(best_h, n, 60, t3, 5000)

    # Phase 4: Subgradient + SLSQP
    t4 = t0 + budget_s * 0.62
    best_h, best_c5 = subgrad_loop(best_h, n, t4, 100, 500, 200000)

    # Phase 5: Basin hopping
    t5 = t0 + budget_s * 0.82
    best_h, best_c5 = basin_hopping(best_h, n, t5, 35, 0.06)

    # Phase 6: Try finer grid
    if time.time() < t_end - budget_s * 0.2 and n < 300:
        n2 = 300
        xs_old = np.linspace(0, 2, n, endpoint=False)
        xs_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(xs_new, xs_old, best_h)
        h2 = project_h(h2)
        t6 = t_end - budget_s * 0.1
        h2, c5_2 = coord_descent(h2, n2, 40, t6, 3000)
        hc2 = slsqp_step(h2, n2, 100000, 600)
        c5_2b = compute_c5(hc2)
        if c5_2b < c5_2:
            h2 = hc2
            c5_2 = c5_2b
        h2, c5_2 = coord_descent(h2, n2, 50, t6, 2000)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            n = n2

    # Phase 7: Final polish
    if time.time() < t_end - 8:
        tf = t_end - 3
        best_h, best_c5 = coord_descent(best_h, n, 80, tf, 3000)
    if time.time() < t_end - 4:
        hc = slsqp_step(best_h, n, 1000000, 1000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
