import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_grad(h, T, flat_pen=0.0, n_pen=5):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    m = np.max(corr)
    if T > 1e6:
        val = m
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        shifted = (corr - m) * T
        max_s = np.max(shifted)
        exps = np.exp(shifted - max_s)
        Z = np.sum(exps)
        weights = exps / Z
        val = m + np.log(Z + 1e-300) / T
    if flat_pen > 0.0:
        top = np.sort(corr)[-n_pen:]
        val += flat_pen * np.std(top)
    # Gradient of log-sum-exp w.r.t. h
    # d(corr_k)/dh[i] = (1-h[i+k]) for shift k, -h[i-k] for shift -k
    # Full correlation: corr[j] = sum_l h[l] * (1-h[l + j - (n-1)])
    # d(corr[j])/dh[i] = (1-h[i + j - (n-1)]) if in range, and -h[i - j + (n-1)] if in range
    grad = np.zeros(n)
    shifts = np.arange(len(corr)) - (n - 1)
    # For efficiency, only use top weights
    thresh = 1e-10
    active = weights > thresh
    if not np.any(active):
        return val * dx, grad
    wa = weights[active]
    sa = shifts[active]
    i_idx = np.arange(n)[:, None]
    s_idx = sa[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1 - h[jmc] * v2) @ wa
    return val * dx, grad * dx

def project_h(h):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n * 0.5
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_fast(h0, T, n, maxiter=600, ftol=1e-16, flat_pen=0.0):
    def fun(x):
        v, _ = smooth_obj_grad(x, T, flat_pen)
        return v
    def jac(x):
        _, g = smooth_obj_grad(x, T, flat_pen)
        return g
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h0

def coord_descent(h, t_end, max_sweeps=300, n_top=25):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = compute_corr(best_h)
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        perm = np.random.permutation(n)
        for i in perm:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved and sweep > 20:
            break
    return best_h, best_c5

def pair_coord_descent(h, t_end, n_pairs=5000, n_top=20):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = compute_corr(best_h)
    for p in range(n_pairs):
        if time.time() > t_end:
            break
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        old_i, old_j = best_h[i], best_h[j]
        # Compute coefficients for both variables
        coefs_i = np.zeros(n_top)
        coefs_j = np.zeros(n_top)
        for t in range(n_top):
            k = top_shifts[t]
            ci = 0.0
            cj = 0.0
            jp = i + k
            jm = i - k
            if 0 <= jp < n:
                ci += (1.0 - best_h[jp])
            if 0 <= jm < n:
                ci -= best_h[jm]
            coefs_i[t] = ci
            jp2 = j + k
            jm2 = j - k
            if 0 <= jp2 < n:
                cj += (1.0 - best_h[jp2])
            if 0 <= jm2 < n:
                cj -= best_h[jm2]
            coefs_j[t] = cj
        base = corr[top_idx] - old_i * coefs_i - old_j * coefs_j
        # Minimize max(base + ti*coefs_i + tj*coefs_j) over ti, tj in [0,1]
        def obj_pv(v):
            ti = np.clip(v[0], 0, 1)
            tj = np.clip(v[1], 0, 1)
            return np.max(base + ti * coefs_i + tj * coefs_j)
        best_val = obj_pv([old_i, old_j])
        res = minimize(obj_pv, [old_i, old_j], method='Nelder-Mead',
                       options={'xatol': 1e-12, ' fatol': 1e-14, 'maxiter': 200})
        new_i = np.clip(res.x[0], 0, 1)
        new_j = np.clip(res.x[1], 0, 1)
        if res.fun < best_val - 1e-14:
            h_try = best_h.copy()
            h_try[i] = new_i
            h_try[j] = new_j
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                corr = compute_corr(best_h)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.2:
            # Shift perturbation (structural)
            shift = np.random.randint(1, 4) * np.sign(np.random.random() - 0.5)
            pert = np.zeros(n)
            pert[shift:] = work_h[:-shift] - 0.5
            pert[:-shift] += work_h[shift:] - 0.5
            pert *= scale * 0.3 * cool
        elif r < 0.4:
            f = np.random.randint(1, 15)
            p = np.random.uniform(0, 2 * np.pi)
            pert = scale * 0.6 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.55:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.2 * cool * np.random.randn(max(int(mask.sum()), 1))
        elif r < 0.7:
            blen = np.random.randint(max(n // 12, 2), max(n // 4, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * cool * np.random.randn(blen)
        elif r < 0.85:
            # Top-shift targeting
            corr = compute_corr(work_h)
            top_idx = np.argsort(corr)[-6:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n - k] += (work_h[k:n] - 0.5) * 0.3
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n + k] - 0.5) * 0.3
            pert *= scale * 0.5 * cool
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp_fast(h_pert, T, n, 350, 1e-15)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5
