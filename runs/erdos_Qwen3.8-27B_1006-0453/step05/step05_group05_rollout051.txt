        return h
    if s > target:

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    if s > target:
        lo, hi = 0.0, 2.0
        for _ in range(50):
            mid = (lo + hi) * 0.5
            if np.maximum(h - mid, 0.0).sum() > target:
                lo = mid
            else:
                hi = mid
        return np.maximum(h - (lo + hi) * 0.5, 0.0)
    else:
        lo, hi = 0.0, 2.0
        for _ in range(50):
            mid = (lo + hi) * 0.5
            if np.minimum(h + mid, 1.0).sum() < target:
                lo = mid
            else:
                hi = mid
        return np.minimum(h + (lo + hi) * 0.5, 1.0)

def grad_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    grad = np.zeros(n)
    active = w > 1e-10
    offsets = np.arange(2 * n - 1) - (n - 1)
    idx = np.where(active)[0]
    for k in idx:
        wk = w[k]
        off = offsets[k]
        jl = max(0, off)
        jh = min(n, n + off)
        if jl < jh:
            grad[jl:jh] += wk * (1.0 - h[jl - off:jh - off])
        jl2 = max(0, -off)
        jh2 = min(n, n - off)
        if jl2 < jh2:
            grad[jl2:jh2] -= wk * h[jl2 + off:jh2 + off]
    return dx * grad

    return h

    h0[i] = 0.0
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    coefs = np.zeros(2 * n - 1)
    off_arr = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        off = off_arr[k]
        c = 0.0
        if 0 <= i + off < n:
            c += 1.0 - h[i + off]
