        hop_end = min(t_end, time.time() + hop_budget)
        h_opt, c5_opt = cd_h(h_pert, n, hop_end, max_sweeps=60, seed=seed + hop * 7)
        temp = 0.005 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Get starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[::2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive coordinate descent from parent
    t1 = t0 + budget_s * 0.15
    best_h, best_c5 = cd_h(best_h, n, t1, max_sweeps=3000, seed=seed)

    # Phase 2: Simulated annealing for exploration
    t2 = t0 + budget_s * 0.30
    h_sa, c5_sa = sa_phase(best_h, n, t2, n_steps=80000, seed=seed + 1)
    if c5_sa < best_c5:
        best_h, best_c5 = h_sa, c5_sa
    # Polish after SA
    best_h, best_c5 = cd_h(best_h, n, t0 + budget_s * 0.35, max_sweeps=1500, seed=seed + 2)

    # Phase 3: Basin hopping (main exploration)
    t3 = t0 + budget_s * 0.70
    best_h, best_c5 = basin_hopping(best_h, n, t3, n_hops=200, seed=seed + 10)

    # Phase 4: Second round - SA from current best
    t4 = t0 + budget_s * 0.78
    h_sa2, c5_sa2 = sa_phase(best_h, n, t4, n_steps=40000, seed=seed + 100)
    if c5_sa2 < best_c5:
        best_h, best_c5 = h_sa2, c5_sa2
    best_h, best_c5 = cd_h(best_h, n, t0 + budget_s * 0.82, max_sweeps=1000, seed=seed + 200)

    # Phase 5: Another basin hopping round
    t5 = t0 + budget_s * 0.90
    best_h, best_c5 = basin_hopping(best_h, n, t5, n_hops=100, seed=seed + 500)

    # Phase 6: Final intensive coordinate descent
    t6 = t0 + budget_s * 0.97
    best_h, best_c5 = cd_h(best_h, n, t6, max_sweeps=5000, seed=seed + 300)

    # Phase 7: Last polish
    if time.time() < t0 + budget_s * 0.995:
        best_h, best_c5 = cd_h(best_h, n, t0 + budget_s * 0.995, max_sweeps=500, seed=seed + 400)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
