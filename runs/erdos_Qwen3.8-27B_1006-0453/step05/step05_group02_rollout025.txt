import time
from scipy.optimize import minimize

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h = np.clip(h + room * (diff / tot), 0, 1)
        else:
            if s < 1e-13:
                break
            h = np.clip(h * (target / s), 0, 1)

def smooth_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50, 50)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

    def cons_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(smooth_obj, h, args=(T,), method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})

            return project_h(np.clip(res.x, 0, 1))
    except:
