```python
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_effect_vec(h, i, n):
    """O(n) vectorized computation of effect of changing h[i]."""
    effect = np.zeros(2 * n - 1)
    p_start = max(0, i - n + 1)
    p_end = min(n - 1, i + n - 1)
    if p_start <= p_end:
        p_range = np.arange(p_start, p_end + 1)
        k_range = i + n - 1 - p_range
        effect[k_range] += (1.0 - h[p_range])
    q_start = max(0, i + 1 - n)
    q_end = min(n - 1, i + 1)
    if q_start <= q_end:
        q_range = np.arange(q_start, q_end + 1)
        k_range2 = q_range - i + n - 1
        effect[k_range2] += (-h[q_range])
    return effect

def opt_coord_1d(corr, effect, h_val):
    """Find best delta for one coordinate, keeping h in [0,1]."""
    lo = -h_val
    hi = 1.0 - h_val
    if hi - lo < 1e-15:
        return 0.0
    def f(d):
        return np.max(corr + d * effect)
    res = minimize_scalar(f, bounds=(lo, hi), method='bounded', options={'xatol': 1e-12, 'maxiter': 50})
    return res.x

def fast_coord_descent(h, n, corr, t_end, max_sweeps=500):
    """Fast coordinate descent using precomputed effects."""
    best_c5 = np.max(corr) * (2.0 / n)
    improved = True
    sweep = 0
    while sweep < max_sweeps and time.time() < t_end:
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return h, corr, best_c5
            effect = compute_effect_vec(h, i, n)
            delta = opt_coord_1d(corr, effect, h[i])
            if abs(delta) > 1e-13:
                h[i] += delta
                corr += delta * effect
                new_max = np.max(corr)
                if new_max < best_c5 - 1e-14:
                    best_c5 = new_max * (2.0 / n)
                    improved = True
        sweep += 1
        if not improved and sweep > 5:
            break
    return h, corr, best_c5

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h
```

So there's a -δ² term at lag 0. For small δ this is negligible, but for δ up to 1 it could matter. Since we're doing a 1D minimization, the linear approximation might not be exact.

However, for the purposes of finding a good direction, the linear approximation is usually sufficient. The coordinate will be set to the min of the linear approximation, and the actual improvement will be close. If we care about exactness, we can recompute the correlation after the update.

Actually, since we're updating corr += delta * effect, we're using the linear approximation. The δ² error at one lag is at most δ²/4 (max of -δ² when h[i]=1/2, δ=1). In absolute terms this is tiny compared to the correlation values (~n/4 scale). So it's fine.

But wait - there's another issue. When we're doing coordinate descent and we've already updated some coordinates, the corr we're maintaining is approximately correct (linear updates). Over many updates, the errors could accumulate. Let me add periodic recomputation:

After every m sweeps, recompute corr = np.correlate(h, 1-h, mode='full').

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_effect_vec(h, i, n):
    effect = np.zeros(2 * n - 1)
    p_start = max(0, i - n + 1)
    p_range = np.arange(p_start, n)
    k_range = i + n - 1 - p_range
    effect[k_range] += (1.0 - h[p_range])
    q_start = max(0, i + 1 - n)
    q_range = np.arange(q_start, n)
    k_range2 = q_range - i + n - 1
    effect[k_range2] += (-h[q_range])
    return effect

def opt_delta(corr, effect, h_val):
    lo = -h_val
    hi = 1.0 - h_val
    if hi - lo < 1e-14:
        return 0.0
    def f(d):
        return np.max(corr + d * effect)
    res = minimize_scalar(f, bounds=(lo, hi), method='bounded',
                          options={'xatol': 1e-11, 'maxiter': 40})
    return res.x

def fast_cd(h, n, corr, t_end, max_sweeps=200, recompute_every=20):
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    sweep = 0
    while sweep < max_sweeps and time.time() < t_end:
        any_improve = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return h, corr, best_c5
            effect = compute_effect_vec(h, i, n)
            d = opt_delta(corr, effect, h[i])
            if abs(d) > 1e-12:
                h[i] += d
                corr += d * effect
                nm = np.max(corr)
                if nm < best_c5 - 1e-15:
                    best_c5 = float(nm * dx)
                    any_improve = True
        sweep += 1
        if sweep % recompute_every == 0:
            corr = get_corr(h)
            nm = np.max(corr)
            best_c5 = float(nm * dx)
        if not any_improve and sweep > 3:
            break
    return h, corr, best_c5

def slsqp_smooth(h, T, n, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(T * (c - m), -50, 50)
        return (m + np.log(np.sum(np.exp(s))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cons}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_corr = get_corr(best_h)
    best_c5 = float(np.max(best_corr) * (2.0 / n))
    
    # Phase 1: Intensive coordinate descent
    t1 = t0 + min(120, budget_s * 0.12)
    best_h, best_corr, best_c5 = fast_cd(best_h, n, best_corr, t1, max_sweeps=300)
    
    # Phase 2: SLSQP passes
    t2 = t0 + min(250, budget_s * 0.23)
    for T in [30000, 8000, 2000, 600, 200, 80]:
        if time.time() > t2:

        hc = slsqp_smooth(best_h, T, n, maxiter=150)
        hc, hc_c, hc_c5 = fast_cd(hc, n, get_corr(hc), t0 + 25, max_sweeps=40)
        if hc_c5 < best_c5:
            best_c5, best_h, best_corr = hc_c5, hc, hc_c
    best_corr = get_corr(best_h)
    best_c5 = float(np.max(best_corr) * (2.0 / n))
    
    # Phase 3: Basin hopping with CD
    t3 = t0 + min(550, budget_s * 0.5)
    work_h = best_h.copy()
    work_corr = best_corr.copy()
    work_c5 = best_c5
    for it in range(300):
        if time.time() > t3:
            break
        cool = max(0.05, 1.0 - it / 300.0)
        amp = 0.015 + 0.05 * cool
        h_p = project_h(work_h + amp * np.random.randn(n), n)
        hc, hc_c, hc_c5 = fast_cd(h_p, n, get_corr(h_p), min(t0+15, t3), max_sweeps=40)
        temp = 0.0004 * cool + 0.00003
        if hc_c5 < work_c5 or np.random.rand() < np.exp(-(hc_c5 - work_c5) / temp):
            work_h, work_corr, work_c5 = hc, hc_c, hc_c5
        if work_c5 < best_c5:
            best_c5, best_h = work_c5, work_h.copy()
            best_corr = work_corr.copy()

    t4 = t0 + min(750, budget_s * 0.68)
    for T in [15000, 4000, 1000, 300, 100]:

        hc = slsqp_smooth(best_h, T, n, maxiter=200)
        hc, hc_c, hc_c5 = fast_cd(hc, n, get_corr(hc), t0 + 25, max_sweeps=60)
        if hc_c5 < best_c5:
            best_c5, best_h, best_corr = hc_c5, hc, hc_c
    best_corr = get_corr(best_h)
    best_c5 = float(np.max(best_corr) * (2.0 / n))
    
    # Phase 5: Final intensive CD
    t5 = t0 + min(1020, budget_s * 0.93)
    for _ in range(10):
        if time.time() > t5:

        best_h, best_corr, best_c5 = fast_cd(best_h, n, best_corr, t5, max_sweeps=80)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_effect_vec(h, i, n):
    effect = np.zeros(2 * n - 1)
    p_start = max(0, i - n + 1)
    p_range = np.arange(p_start, n)
    k_range = i + n - 1 - p_range
    effect[k_range] += (1.0 - h[p_range])
    q_start = max(0, i + 1 - n)
    q_range = np.arange(q_start, n)
    k_range2 = q_range - i + n - 1
    effect[k_range2] += (-h[q_range])
    return effect

def opt_delta(corr, effect, h_val):
    lo = -h_val
    hi = 1.0 - h_val
    if hi - lo < 1e-14:
        return 0.0
    def f(d):
        return np.max(corr + d * effect)
    res = minimize_scalar(f, bounds=(lo, hi), method='bounded',
                          options={'xatol': 1e-11, 'maxiter': 40})
    return res.x

def fast_cd(h, n, corr, t_end, max_sweeps=200):
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        any_improve = False
        if sweep < 3:
            order = np.arange(n)
        else:
            order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return h, corr, best_c5
            effect = compute_effect_vec(h, i, n)
            d = opt_delta(corr, effect, h[i])
            if abs(d) > 1e-12:
                h[i] += d
                corr += d * effect
                nm = np.max(corr)
                if nm < best_c5:
                    best_c5 = float(nm * dx)
                    any_improve = True
        h = project_h(h, n)
        corr = get_corr(h)
        nm = np.max(corr)
        if nm * dx < best_c5:
            best_c5 = float(nm * dx)
        if not any_improve:
            break
    return h, corr, best_c5

def slsqp_smooth(h, T, n, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(T * (c - m), -50, 50)
        return (m + np.log(np.sum(np.exp(s))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cons}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(r.x, 0, 1), n)
    except Exception:
        return h

def segment_swap(h, n):
    a = np.random.randint(0, n)
    b = np.random.randint(0, n)
    seg = np.random.randint(2, max(3, n // 8))
    a = np.clip(a, 0, n - seg)
    b = np.clip(b, 0, n - seg)
    if a == b:
        return h
    h2 = h.copy()
    h2[a:a+seg], h2[b:b+seg] = h2[b:b+seg].copy(), h2[a:a+seg].copy()
    return project_h(h2, n)

def make_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[:n//2] = 1.0
    # Perturb with structure
    h += 0.1 * rng.randn(n)
    return project_h(h, n)

        if r < 0.5:

        hc, hc_c, hc_c5 = fast_cd(h_p, n, get_corr(h_p), min(t0 + 12, t3), max_sweeps=35)
```
