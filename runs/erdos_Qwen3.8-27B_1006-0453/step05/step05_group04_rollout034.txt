- Evaluating at critical points (where the max switches between different k's, or where the quadratic equals a linear, or at boundaries 0 and 1).

A faster approach:

2. The function g(v) = max(C_0 + v - v², max_k(A_k*v + B_k)) is the max of one concave quadratic and lines.
3. The minimum of g on [0,1] occurs at:
   - v=0 or v=1, or
   - A point where two of the functions are equal and the derivative of the max is zero.
   
For the concave quadratic and a line: f(v) = C + v - v² and g(v) = Av + B.
They're equal when v - v² = Av + B - C, i.e., -v² + (1-A)v + (C-B) = 0.
This has at most 2 solutions.

For two lines: they're equal at a single point.

So the critical points are:
- v = 0, v = 1
- Intersections of the quadratic with each line (at most 2 per line)
- Intersections of pairs of lines (at most 1 per pair)

The number of affected k's is at most max(i, n-1-i) + 1, which is at most n. So there are O(n) critical points to check. For each, evaluate g(v) and keep the minimum. This is O(n²) per coordinate in the worst case, but in practice much less.

            corr[0] -= d * d
            bh[i] = vn
            c5n = max(float(np.max(corr)) , float(mout)) * dx
            c5n = float(np.max(corr) * dx)
            if c5n < bc5 - 1e-15:
                bc5 = c5n
                improved = True
            else:
                corr[:kmax + 1] -= d * coeff
                corr[0] += d * d
                bh[i] = old
        if not improved and sweep > 0:
            break
    bh = project_h(bh, n)
    bc5 = compute_c5(bh)
    return bh, bc5

def gen_init(n, seed, style):
    rng = np.random.RandomState(seed)
    if style == 0:
        h = (rng.rand(n) < 0.5).astype(float)
    elif style == 1:
        h = np.zeros(n)
        for i in range(n):
            x, p = i, 0
            while x:
                p ^= 1
                x >>= 1
            h[i] = 1.0 - p
    elif style == 2:
        i = np.arange(n)
        h = 0.5 + 0.49 * np.sin(2 * np.pi * rng.randint(2, 15) * i / n)
    elif style == 3:
        bs = rng.randint(2, max(3, n // 10))
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i // bs) % 2 == 0 else 0.0
    else:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    return project_h(h, n)

def basin_hop(h, n, t_end, seed=42, nhops=300):
    rng = np.random.RandomState(seed)
    bh, bc5 = h.copy(), compute_c5(h)
    ch, cc5 = h.copy(), bc5
    for hop in range(nhops):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - hop / float(nhops))
        step = 0.02 + 0.08 * cool
        mode = hop % 4
        if mode == 0:
            p = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 20)
            p = step * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 6.28))
        elif mode == 2:
            p = np.zeros(n)
            mask = rng.random(n) < 0.3
            p[mask] = step * 2 * rng.randn(int(mask.sum()))
        else:
            idx = rng.choice(n, size=max(4, n // 8), replace=False)
            p = np.zeros(n)
            p[idx] = step * rng.randn(len(idx))
        hp = project_h(ch + p, n)
        he = min(t_end, time.time() + max(1.0, 5.0 * cool + 2.0))
        ho, co = coord_desc(hp, n, he, nsweeps=60, seed=seed + hop)
        temp = 0.003 * cool + 0.0003
        if co < cc5 or (co > cc5 and rng.rand() < np.exp(-(co - cc5) / temp)):
            ch, cc5 = ho, co
            if co < bc5:
                bc5, bh = co, ho.copy()
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h160_init = np.array(initial_h_values, dtype=float)
        n_init = len(h160_init)
    else:
        h160_init = np.zeros(160)
        h160_init[:80] = 1.0
        n_init = 160
    
    best_h = h160_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_init
    
    # Phase 1: Coord descent on initial (n=160)
    t1 = t0 + budget_s * 0.18
    best_h, best_c5 = coord_desc(best_h, best_n, t1, nsweeps=5000, seed=seed)
    
    # Phase 2: Try multiple n values with diverse inits
    n_candidates = [80, 120, 160, 200, 240, 320]
    if best_n not in n_candidates:
        n_candidates.insert(0, best_n)
    
    time_per_n = (t0 + budget_s * 0.62 - t0) / len(n_candidates)
    for ni in n_candidates:
        if time.time() > t0 + budget_s * 0.62:
            break
        t_n_end = min(t0 + budget_s * 0.62, time.time() + time_per_n * 1.3)
        n_styles = 3 if ni <= 160 else 2
        for st in range(n_styles):
            if time.time() > t_n_end:
                break
            if ni == best_n and st == 0:
                h0 = best_h.copy()
            else:
                h0 = gen_init(ni, seed + ni * 10 + st, st % 5)
            tcd = min(t_n_end, time.time() + time_per_n / n_styles)
            hc, c5c = coord_desc(h0, ni, tcd, nsweeps=3000, seed=seed + ni + st)
            if c5c < best_c5:
                best_c5, best_h, best_n = c5c, hc, ni
    
    # Phase 3: Basin hopping on best
    if time.time() < t0 + budget_s * 0.82:
        best_h, best_c5 = basin_hop(best_h, best_n, t0 + budget_s * 0.82,
                                     seed=seed + 100, nhops=250)
    
    # Phase 4: Basin hopping with different seed for diversity
    if time.time() < t0 + budget_s * 0.90:
        bh2, bc5_2 = basin_hop(best_h, best_n, t0 + budget_s * 0.90,
                               seed=seed + 500, nhops=100)
        if bc5_2 < best_c5:
            best_c5, best_h = bc5_2, bh2
    
    # Phase 5: Final coord descent polish
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = coord_desc(best_h, best_n, t0 + budget_s * 0.97,
                                     nsweeps=8000, seed=seed + 999)
    
    # Final projection and evaluation
    best_h = project_h(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Ensure valid
    assert best_c5 > 0
    assert all(best_h >= 0) and all(best_h <= 1)
    
    return best_h, best_c5, best_n
