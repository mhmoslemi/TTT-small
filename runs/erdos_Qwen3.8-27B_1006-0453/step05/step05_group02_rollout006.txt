import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = T * (corr - m)
    val = m + np.log(np.sum(np.exp(s))) / T
    return val * dx

def constraint_sum(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_full(h, T, n, maxiter=600, ftol=1e-14):
    try:
        res = minimize(smooth_obj, h, args=(T, n), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def multi_coord_opt(h, n, K, T, maxiter=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n_blocks = n // K
    for blk in range(n_blocks):
        lo = blk * K
        hi = lo + K
        fixed = best_h.copy()
        def sub_obj(y, T=T, n=n, fixed=fixed, lo=lo, hi=hi):
            x = fixed.copy()
            x[lo:hi] = y
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            s = T * (corr - m)
            return (m + np.log(np.sum(np.exp(s))) / T) * dx
        def sub_con(y, n=n, lo=lo, hi=hi):
            return y.sum() - (fixed[lo:hi].sum())
        y0 = best_h[lo:hi].copy()
        try:
            res = minimize(sub_obj, y0, method='SLSQP',
                           bounds=[(0.0, 1.0)] * K,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            y_new = np.clip(res.x, 0, 1)
            h_try = best_h.copy()
            h_try[lo:hi] = y_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
        except Exception:
            pass
    return best_h, best_c5

def coord_single_fast(h, n, i):
    h_base = h.copy()
    h_base[i] = 0.0
    cb = np.correlate(h_base, 1.0 - h_base, mode='full')
    k_range = np.arange(-(n - 1), n)
    j1 = i + k_range
    j2 = i - k_range
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(j2[v2], 0, n - 1)])
    dx = 2.0 / n
    def f1d(hi_val):
        return float(np.max(cb + hi_val * coefs) * dx)
    try:
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
        return float(res.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single_fast(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def make_pert(n, cool, rng, kind=None):
    if kind is None:
        kind = rng.randint(0, 6)
    p = np.zeros(n)
    if kind == 0:
        bs = rng.randint(4, max(5, n // 6))
        st = rng.randint(0, max(1, n - bs))
        p[st:st+bs] = (0.05 + 0.15*cool) * rng.standard_normal(bs)
    elif kind == 1:
        freq = rng.randint(2, max(3, n // 12))
        ph = rng.uniform(0, 2*np.pi)
        p = (0.04 + 0.1*cool) * np.sin(2*np.pi*freq*np.arange(n)/n + ph)
    elif kind == 2:
        np_ = rng.randint(3, max(4, n // 10))
        idx = rng.choice(n, np_, replace=False)
        p[idx] = (0.1 + 0.3*cool) * rng.standard_normal(np_)
    elif kind == 3:
        p = (0.02 + 0.05*cool) * rng.standard_normal(n)
    elif kind == 4:
        half = n // 2
        p[:half] = 0.08*cool * rng.standard_normal(half)
        p[half:] = -0.08*cool * rng.standard_normal(n-half)
    else:
        for _ in range(3):
            st = rng.randint(0, n-1)
            ln = rng.randint(2, max(3, n//10))
            p[st:min(n, st+ln)] += (0.2*cool) * rng.choice([-1, 1])
    return p

def upsample(h, factor=2):
    n = len(h)
    h_new = np.zeros(n * factor)
    for i in range(n):
        h_new[i*factor] = h[i]
        h_new[i*factor+1] = h[i]
    return h_new
