For k≠0: ∂corr_k/∂h[j] = (1-h[j+k])*[0≤j+k<n] - h[j-k]*[0≤j-k<n]

This is the same lin[j][k] I computed before. So the gradient of the soft-max is:
grad[j] = Σ_k p_k * lin[j][k] (with the -1 correction for k=0)

This is O(n²) to compute, which is fine for n=160.

Actually, let me think about this more carefully. The projection is critical. A simple approach:
1. Clip to [0,1]
2. If sum > target, reduce the largest values
3. If sum < target, increase the smallest values

Or better: use a water-filling approach:
- To reduce sum by δ: find a threshold τ such that sum(clip(h-τ, 0, ∞)) = δ. Then h_new = h - τ for h > τ, h_new = h for h ≤ τ.
- To increase sum by δ: find a threshold τ such that sum(clip(1-(h+τ), 0, ∞)) = δ. Then h_new = h + τ for h < 1-τ, h_new = h for h ≥ 1-τ.

This is O(n log n) per projection.

    return float(np.max(corr) * dx)

def project_h(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to reduce by diff
            # Water fill from top
            sorted_h = np.sort(h)[::-1]
            cumsum = np.cumsum(sorted_h)
            # Find how many top values to reduce
            # If we reduce top m values by τ, we need m*τ = diff (approximately)
            # More precisely: sum of (h_i - τ) for h_i > τ = diff
            # Use binary search on τ
            lo, hi = 0.0, h.max()
            for _ in range(50):
                mid = (lo + hi) / 2
                val = np.maximum(h - mid, 0).sum()
                if val > diff:
                    lo = mid
                else:
                    hi = mid
            tau = (lo + hi) / 2
            h = np.maximum(h - tau, 0)
        else:
            # Need to increase by -diff
            # Water fill from bottom
            lo, hi = 0.0, 1.0 - h.min()
            for _ in range(50):
                mid = (lo + hi) / 2
                val = np.minimum(1.0 - h, mid).sum()
                if val > -diff:
                    hi = mid
                else:
                    lo = mid
            tau = (lo + hi) / 2
            h = np.minimum(h + tau, 1.0)
    # Final exact fix
    diff = h.sum() - target
    if abs(diff) > 1e-14:
        # Adjust one element
        if diff > 0:
            idx = np.argmax(h)
            h[idx] = max(0, h[idx] - diff)
        else:
            idx = np.argmin(h)
            h[idx] = min(1, h[idx] - diff)
    return h

                pass

            # with corr[n-1] also having -v² (but we handle this by adjusting)
            # Actually: change d = v - old

                d = v - old
                val = corr + d * lin
                val[n-1] -= d * d
                return float(np.max(val) * dx)

            if abs(d) < 1e-12:

            corr[n-1] -= d * d

    bh = h.copy()

        cool = 1.0 - hop / n_hops
        step = 0.02 + 0.08 * cool

        mode = hop % 4
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2*np.pi)
            pert = step * np.sin(2*np.pi*f*np.arange(n)/n + ph)

grad[j] = (Σ_k p_k * lin_j_k) * dx where p_k = exp(β*corr_k) / Σ exp(β*corr_k)

And lin_j_k is what I computed before (the linear coefficient of corr_k w.r.t. h[j]).

I can compute the gradient in O(n²):
```python
def compute_grad_soft(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    exponents = beta * (corr - corr.max())  # for numerical stability
    p = np.exp(exponents)
    p /= p.sum()
    
    k_arr = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for j in range(n):
        lin = np.zeros(2*n-1)
        jk = j + k_arr
        v1 = (jk >= 0) & (jk < n)
        lin[v1] = (1.0 - h[jk[v1]])
        jkm = j - k_arr
        v2 = (jkm >= 0) & (jkm < n)
        lin[v2] -= h[jkm[v2]]
        # Special case k=0 (index n-1): additional -h[j] term? 
        # Actually lin[j][n-1] = (1-h[j]) - h[j] = 1-2h[j], which is correct for d(corr_0)/d(h[j]).
        grad[j] = p @ lin * dx
    
    return grad
```

This is O(n²) per gradient evaluation. For n=160, that's 25,600 multiplications. Very fast.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            break
        h[free] -= diff / nf
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) > 1e-10:
        if diff > 0:
            h[np.argmax(h)] -= diff
        else:
            h[np.argmin(h)] += -diff
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_1d(d, corr, lin, n):
    vals = corr + d * lin
    vals[n-1] -= d * d
    return vals.max()

def solve_1d(corr, lin, old, n):
    a, b = 0.0, 1.0
    gr = 0.618033988749895
    c = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = obj_1d(c - old, corr, lin, n)
    fd = obj_1d(d_pt - old, corr, lin, n)
    for _ in range(55):
        if fc < fd:
            b = d_pt
            d_pt = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj_1d(c - old, corr, lin, n)
        else:
            a = c
            c = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = obj_1d(d_pt - old, corr, lin, n)
        if b - a < 1e-12:
            break
    return (a + b) / 2.0

def fast_cd(h, n, t_end, max_sweeps=10000, seed=42):
    rng = np.random.RandomState(seed)
    bh = h.copy()
    bc5 = compute_c5(bh)
    dx = 2.0 / n
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    k_arr = np.arange(-(n-1), n)
    
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        if sw % 4 == 3:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for i in order:
            if time.time() > t_end:
                return bh, bc5
            old = bh[i]
            lin = np.zeros(2*n-1)
            jk = i + k_arr
            v1 = (jk >= 0) & (jk < n)
            lin[v1] = (1.0 - bh[jk[v1]])
            jkm = i - k_arr
            v2 = (jkm >= 0) & (jkm < n)
            lin[v2] -= bh[jkm[v2]]
            
            v_new = solve_1d(corr, lin, old, n)
            d = v_new - old
            if abs(d) < 1e-11:
                continue
            new_corr = corr + d * lin
            new_corr[n-1] -= d * d
            c5_new = float(new_corr.max() * dx)
            if c5_new < bc5:
                corr = new_corr
                bh[i] = v_new
                bc5 = c5_new
                improved = True
        if not improved and sw > 3:
            break
    return bh, bc5
