import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_D1_D0(h, N):
    shifts = np.arange(2 * N - 1) - (N - 1)
    pos = np.arange(N)
    pm = pos[:, None] - shifts[None, :]
    pp = pos[:, None] + shifts[None, :]
    vm = (pm >= 0) & (pm < N)
    vp = (pp >= 0) & (pp < N)
    hm = np.where(vm, h[pm.clip(0, N - 1)], 0.0)
    hp = np.where(vp, h[pp.clip(0, N - 1)], 0.0)
    D1 = -1.0 + hm + hp
    D0 = 1.0 - hm - hp
    return D1, D0

def fix_binary(h, N):
    nh = N // 2
    n1 = int(round(h.sum()))
    if n1 > nh:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[0] = 0.0
        h[idx[:n1 - nh]] = 0.0
    elif n1 < nh:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:nh - n1]] = 1.0
    h = np.where(h > 0.5, 1.0, 0.0)
    return h

def binary_search_full(h, N, deadline, max_sweeps=20000, T0=0.3):
    h = fix_binary(h, N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = fft_corr(h)
    D1, D0 = compute_D1_D0(h, N)
    dx = 2.0 / N
    best_c = float(corr.max()) * dx
    patience = 0
    T = T0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if len(ones) < 1 or len(zeros) < 1:
            break
        P = float(corr.max())
        best_new_P = P
        best_pq = None
        n_ones = len(ones)
        for i in range(n_ones):
            p = int(ones[i])
            A = corr + D1[p, :]
            C = A[None, :] + D0[zeros, :]
            M = C.max(axis=1)
            mi = int(np.argmin(M))
            if M[mi] < best_new_P:
                best_new_P = float(M[mi])
                best_pq = (p, int(zeros[mi]))
        if best_pq is not None:
            p, q = best_pq
            h[p], h[q] = 0.0, 1.0
            corr = corr + D1[p, :] + D0[q, :]
            D1, D0 = compute_D1_D0(h, N)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c_now = float(corr.max()) * dx
            if c_now < best_c:
                best_c = c_now
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            T = max(0.01, T0 * (0.995 ** patience))
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])

            if delta_P < 0 or np.random.random() < np.exp(-delta_P / (T * dx)):
                h[p], h[q] = 0.0, 1.0
                corr = corr + D1[p, :] + D0[q, :]

                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]

                    best_c = c_now
                    patience = 0
        if patience > 500:
            T *= 1.5
            patience = 250
    return h, best_c

def twoswap_search(h, N, deadline, n_tries=5000):
    h = fix_binary(h, N)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = fft_corr(h)
    D1, D0 = compute_D1_D0(h, N)
    dx = 2.0 / N
    best_c = float(corr.max()) * dx
    for t in range(n_tries):
        if time.time() > deadline:
            break
        p1 = int(ones[np.random.randint(len(ones))])
        p2 = int(ones[np.random.randint(len(ones))])
        if p1 == p2:
            continue
        q1 = int(zeros[np.random.randint(len(zeros))])
        q2 = int(zeros[np.random.randint(len(zeros))])
        if q1 == q2:
            continue
        delta = D1[p1, :] + D0[q1, :] + D1[p2, :] + D0[q2, :]
        new_max = float((corr + delta).max())
        old_max = float(corr.max())
        if new_max < old_max:
            h[p1], h[q1] = 0.0, 1.0
            h[p2], h[q2] = 0.0, 1.0
            corr = corr + delta
            D1, D0 = compute_D1_D0(h, N)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            c_now = new_max * dx
            if c_now < best_c:
                best_c = c_now
    return h, best_c

def nesterov_smooth(h, N, deadline, k=40, lr0=0.015, max_iter=20000):
    h = h.astype(float).copy()
    target = N / 2.0
    dx = 2.0 / N
    h = project_sum(h, target)
    best_h = h.copy()
    corr = fft_corr(h)
    best_pk = float(corr.max())
    m = np.zeros(N)
    lr = lr0
    beta = 0.85
    temp = 50.0
    it = 0
    while it < max_iter and time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(corr.max())
        if pk < best_pk:
            best_pk = pk
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad + np.random.randn(N) * temp * 0.01, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 500 == 0:
            lr *= 0.995
            temp *= 0.97
            beta = min(0.95, 0.85 + it * 0.00003)
    return best_h, best_pk * dx
