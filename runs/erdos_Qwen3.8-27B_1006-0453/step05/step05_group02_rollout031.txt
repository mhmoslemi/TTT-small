import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, T, n, maxiter=500):
    try:
        res = minimize(smooth_c5, h, args=(T,), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except:
        return h

    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')

    # result[k] = sum_m h[m] * (1-h[m-k+n-1])
    # h[i] appears as h[m] when m=i: valid when 0 <= i-k+n-1 < n → k in [i, i+n-1]
    # h[i] appears in (1-h[...]) when i-k+n-1 = j, i.e., k = i-j+n-1: 
    #   valid when 0 <= j < n and 0 <= k <= 2n-2
    #   j = i-k+n-1, so k = i-j+n-1 for j in [0,n-1]
    #   k ranges from i-(n-1)+n-1 = i to i-0+n-1 = i+n-1... wait
    #   k = i-j+n-1, j from 0 to n-1: k from i+n-1 down to i
    #   And the contribution is h[j] * (-h[i]) → coefficient is -h[j] = -h[i-k+n-1]

    #       i-k+n-1 <= n-1 → i-k <= 0 → k >= i

    k_hi = min(i + n - 1, 2 * n - 2)

        coefs[k_lo:k_hi + 1] += (1.0 - h[g_idx])
    
    # Second factor: i-k+n-1 = j (the index in the second array)

    try:

    except:

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_single(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def binarize(h, n):
    """Convert to binary, preserving sum = n/2"""
    target = n // 2
    # Sort by value, keep top 'target' as 1s
    sorted_idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[sorted_idx[:target]] = 1.0
    return hb

def swap_search_binary(hb, n, t_end, max_sweeps=200):
    """Exhaustive swap search on binary h"""
    hb = hb.copy()
    c5 = compute_c5(hb)
    corr = np.correlate(hb, 1.0 - hb, mode='full')
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        improved = False
        # Try all swaps
        improved_any = False
        for a in ones:
            if time.time() > t_end:
                break
            for b in zeros:
                # Swap: a goes 1→0, b goes 0→1
                hb[a], hb[b] = 0.0, 1.0
                new_corr = np.correlate(hb, 1.0 - hb, mode='full')
                new_c5 = float(np.max(new_corr) * dx)
                if new_c5 < c5 - 1e-15:
                    c5 = new_c5
                    corr = new_corr
                    improved = True
                    improved_any = True
                else:
                    hb[a], hb[b] = 1.0, 0.0
        if not improved:
            break
    return hb, c5

```python
def swap_delta(hb, n, a, b):
    """Compute the change in correlation array for swap a:1→0, b:0→1"""
    delta = np.zeros(2*n-1)
    # a as first factor: k in [a, a+n-1] ∩ [0, 2n-2], contribution -(1-hb[a-s]) where s=k-n+1
    # a-s = a-(k-n+1) = a-k+n-1
    k_lo = max(a, 0)
    k_hi = min(a+n-1, 2*n-2)
    if k_lo <= k_hi:
        idx = a - np.arange(k_lo, k_hi+1) + n - 1
        delta[k_lo:k_hi+1] += -(1.0 - hb[idx])
    # a as second factor: k in [n-1-a, n-1+a] ∩ [0, 2n-2], contribution +hb[a+s] where s=k-n+1
    # a+s = a+k-n+1
    k_lo = max(n-1-a, 0)
    k_hi = min(n-1+a, 2*n-2)
    if k_lo <= k_hi:
        idx = a + np.arange(k_lo, k_hi+1) - n + 1
        delta[k_lo:k_hi+1] += hb[idx]
    # b as first factor: k in [b, b+n-1] ∩ [0, 2n-2], contribution +(1-hb[b-s]) where s=k-n+1
    # b-s = b-k+n-1
    k_lo = max(b, 0)
    k_hi = min(b+n-1, 2*n-2)
    if k_lo <= k_hi:
        idx = b - np.arange(k_lo, k_hi+1) + n - 1
        delta[k_lo:k_hi+1] += (1.0 - hb[idx])
    # b as second factor: k in [n-1-b, n-1+b] ∩ [0, 2n-2], contribution -hb[b+s] where s=k-n+1
    # b+s = b+k-n+1
    k_lo = max(n-1-b, 0)
    k_hi = min(n-1+b, 2*n-2)
    if k_lo <= k_hi:
        idx = b + np.arange(k_lo, k_hi+1) - n + 1
        delta[k_lo:k_hi+1] += -hb[idx]
    return delta
```
