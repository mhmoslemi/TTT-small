            if tot < 1e-14:

            h_sm = smooth_h(cur_h, n, 2.5)
            step = 0.005 + 0.03 * cool
            h_p = project_h(h_sm + step * rng.randn(n), n)
        else:
            i0 = rng.randint(0, n - 2)
            i1 = rng.randint(0, n - 2)
            h_p = cur_h.copy()
            h_p[i0], h_p[i1] = h_p[i1], h_p[i0]
            h_p = project_h(h_p, n)
        hop_time = max(2.0, 10.0 * cool * (0.5 + 0.5 * cool))
        hop_end = min(t_end, time.time() + hop_time)
        h_opt, c5_opt = cd_h(h_p, n, hop_end, max_sweeps=150, seed=seed + hop)
        temp = 0.0025 * cool + 0.0002
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        n = len(best_h)
    else:
        n = 160
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    t1 = t0 + budget_s * 0.20
    best_h, best_c5 = cd_h(best_h, n, t1, max_sweeps=4000, seed=seed)

    t2 = t0 + budget_s * 0.60
    best_h, best_c5 = basin_hop(best_h, n, t2, n_hops=400, seed=seed + 50)

    best_h, best_c5 = cd_h(best_h, n, t0 + budget_s * 0.68, max_sweeps=2000, seed=seed + 200)

    for n_try in [200, 128, 192]:
        if time.time() > t0 + budget_s * 0.82:
            break
        h_try = interpolate_h(best_h, n, n_try)
        h_try, c5_try = cd_h(h_try, n_try, t0 + budget_s * 0.86, max_sweeps=1200, seed=seed + n_try)
        if c5_try < best_c5:
            best_h, best_c5, n = h_try, c5_try, n_try

    # Second basin hop round from new best
    if time.time() < t0 + budget_s * 0.93:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.93, n_hops=200, seed=seed + 500)

    t_end = max(t0 + 1.0, t0 + budget_s * 0.985)
    best_h, best_c5 = cd_h(best_h, n, t_end, max_sweeps=3000, seed=seed + 999)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
