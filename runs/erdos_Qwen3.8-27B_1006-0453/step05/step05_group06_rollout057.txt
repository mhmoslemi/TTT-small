def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)

        g[:n-lag] += (-h[lag:])  # -h[n+lag] for n = 0..n-1-lag

        g[p:] += (-h[:n-p])  # -h[m] for m = 0..n-1-p, j = m+p = p..n-1
        g *= dx
    return g

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 ≤ h[j] ≤ 1, t ≥ 0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

def lp_line_search(h, h_lp, max_c5):
    """Line search along h -> h_lp."""
    def f(alpha):
        h_t = h + alpha * (h_lp - h)
        h_t = project_h(h_t)
        return compute_c5(h_t)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    return h + res.x * (h_lp - h), f(res.x)

def soft_max_obj(h, T, lam):
    """Soft-max approximation of max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    Tc = T * (corr - corr.max())
    Tc -= Tc.max()
    exps = np.exp(Tc)
    sm = corr.max() + np.log(exps.sum()) / T
    val = sm + lam * (h.sum() - n/2.0)**2
    # Gradient
    w = exps / exps.sum()
    grad = np.zeros(n)

            grad[p:] += -wi * h[:n-p]
    grad += 2.0 * lam * (h.sum() - n/2.0)
    return val, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def get_lag_grad(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] += (-h[lag:])
        g *= dx
    else:
        p = -lag
        g[:n-p] += (1.0 - h[p:])
        g[p:] += (-h[:n-p])
        g *= dx
    return g
