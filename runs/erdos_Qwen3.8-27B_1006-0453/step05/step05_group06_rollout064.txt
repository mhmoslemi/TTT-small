import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    target = n / 2.0
    for _ in range(6):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            if s < 1e-14:
                break
            h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    Tc = T * (corr - mc)
    Tc -= np.max(Tc)
    exps = np.exp(np.clip(Tc, -500, 500))
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    for idx in range(m):
        lg = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-14:
            continue
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] -= wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] -= wi * h[:n-p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs(h0, T, lam, maxiter, ftol):
    x0 = project_h(h0)
    n = len(x0)
    def f(x):
        return smooth_max(x, T, lam)[0]
    def g(x):
        return smooth_max(x, T, lam)[1]
    try:
        res = minimize(f, x0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_h(res.x)
    except Exception:
        return x0

def coord_descent(h, t_end, max_sweeps=4000):
    n = len(h)
    m = 2 * n - 1
    n1 = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                break
            a = best_h[i]
            lin = np.zeros(m)
            if n - i > 0:
                lin[n1:n1 + n - i] += 1.0 - best_h[i:n]
            if i >= 0:
                lin[n1 - i:n1 + 1] -= best_h[i::-1]
            def f1d(x):
                d = x - a
                cv = corr + d * lin
                cv[n1] -= d * d
                return cv.max()
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-8})
            if abs(res.x - a) > 1e-9:
                ht = best_h.copy()
                ht[i] = res.x
                ht = project_h(ht)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-13:
                    best_h = ht
                    best_c5 = c5t
                    improved = True
                    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        if not improved:
            break
    return best_h, best_c5

def subgrad_polish(h, t_end, steps=30000):
    n = len(h)
    m = 2 * n - 1
    n1 = n - 1
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for step in range(steps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        mc = np.max(corr)
        thresh = mc - 0.05 * mc
        active = corr >= thresh
        if not np.any(active):
            break
        sg = np.zeros(n)
        na = float(np.sum(active))
        for idx in range(m):
            if not active[idx]:
                continue
            lg = idx - n1
            if lg >= 0:
                if lg < n:
                    sg[lg:] += (1.0 - best_h[:n-lg])
                    sg[:n-lg] -= best_h[lg:]
            else:
                p = -lg
                if p < n:
                    sg[:n-p] += (1.0 - best_h[p:])
                    sg[p:] -= best_h[:n-p]
        sg *= dx / na
        alpha = 0.03 / (1 + 0.005 * step)
        ht = project_h(best_h - alpha * sg)
        c5t = compute_c5(ht)
        if c5t < best_c5 - 1e-14:
            best_h = ht
            best_c5 = c5t
    return best_h, best_c5

    best_h, best_c5 = coord_descent(best_h, p4_end, max_sweeps=8000)

    # Phase 5: Subgradient polish + boundary perturbation (12% budget)
    p5_end = t0 + budget_s * 0.96
    best_h, best_c5 = subgrad_polish(best_h, t0 + budget_s * 0.88, steps=30000)
    # Boundary perturbation: nudge a few 0/1 coords and re-optimize
    for _ in range(5):
        if time.time() > p5_end:
            break
        work_h = best_h.copy()
        zeros = np.where(work_h < 0.01)[0]
        ones = np.where(work_h > 0.99)[0]
        if len(zeros) > 0:
            idx = np.random.choice(zeros, min(3, len(zeros)), replace=False)
            work_h[idx] = 0.05
        if len(ones) > 0:
            idx = np.random.choice(ones, min(3, len(ones)), replace=False)
            work_h[idx] = 0.95
        work_h = project_h(work_h)
        work_h, work_c5 = coord_descent(work_h, t0 + budget_s * 0.96, max_sweeps=1000)
        if work_c5 < best_c5:
            best_c5, best_h = work_c5, work_h

    # Phase 6: Final polish
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, t0 + budget_s * 0.99, max_sweeps=2000)
    if time.time() < t0 + budget_s * 0.999:
        hc = lbfgs(best_h, 200000, 500.0, 1500, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
