        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        frac = 1.0 + diff / s
        if frac > 0:
            h = h * frac
        else:
            mask = h > 0
            h[mask] = h[mask] * (target / h[mask].sum())
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, maxiter=500, ftol=1e-14):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if T > 1e10:
            return (m) * dx
        shifted = np.clip(T * (corr - m), -500, 0)
        return (m + np.log(np.sum(np.exp(shifted))) / T) * dx
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent(h, n, t_end, max_sweeps=300, n_top=20):
    best_h = h.copy()
    dx = 2.0 / n
    corr = get_corr(best_h).copy()
    best_max = np.max(corr)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = 0
        for i in range(n):
            if time.time() > t_end:
                break
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = int(top_shifts[t])
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            def obj_t(t):
                return np.max(base + t * coefs)
            res = minimize_scalar(obj_t, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10})
            t_new = res.x
            dh = t_new - best_h[i]
            if abs(dh) < 1e-12:
                continue
            # Incremental correlation update
            for k in range(-(n-1), n):
                idx = n - 1 + k
                jp = i + k
                jm = i - k
                dc = 0.0
                if 0 <= jp < n:
                    dc += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    dc -= best_h[jm]
                if dc != 0.0:
                    corr[idx] += dh * dc
            best_h[i] = t_new
        # End of sweep: project and verify
        best_h = project_h(best_h)
        corr_new = get_corr(best_h)
        new_max = np.max(corr_new)
        if new_max < best_max - 1e-14:
            best_max = new_max
            corr = corr_new
            improved = 1
        else:
            corr = corr_new  # Sync
        # Check if any improvement this sweep
        if np.max(corr_new) >= np.max(corr) - 1e-10:
            # No improvement, maybe break after a few failed sweeps
            pass
    return best_h, float(best_max * dx)

The key insight is: the incremental correlation update makes each coordinate step very fast. The main cost is the O(n) loop for updating all shifts when one coordinate changes. For n=200, that's 399 shifts to update per coordinate, and 200 coordinates per sweep, so ~80000 operations per sweep. With 300 sweeps, that's 24M operations total. Very fast.

The 1D minimization is the other cost: minimize_scalar with bounded method does ~50 function evaluations, each O(n_top=20). So per coordinate: ~1000 ops. Per sweep: 200 * 1000 = 200K. Per 300 sweeps: 60M. Also fast.

Total: maybe a few seconds for the entire coordinate descent phase. 

Now, the issue is that without maintaining the sum constraint during the sweep, the incremental updates aren't quite right (because the projection at the end changes all coordinates). But I can handle this by:
1. Doing the coordinate updates without projection
2. At end of sweep, projecting and recomputing corr
3. The projection is a small adjustment that gets absorbed in the next sweep

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        frac = 1.0 + diff / s
        if frac > 0:
            h = h * frac
        else:
            mask = h > 0
            h[mask] = h[mask] * (target / h[mask].sum())
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, maxiter=500, ftol=1e-14):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if T > 1e10:
            return m * dx
        shifted = np.clip(T * (corr - m), -500, 0)
        return (m + np.log(np.sum(np.exp(shifted))) / T) * dx
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': 0})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent_incremental(h, n, t_end, max_sweeps=500, n_top=25):
    best_h = h.copy()
    dx = 2.0 / n
    corr = get_corr(best_h).astype(np.float64).copy()
    best_max = float(np.max(corr))
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = (top_idx - (n - 1)).astype(int)
        sweep_best = best_max
        for i in range(n):
            if time.time() > t_end:
                break
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = int(top_shifts[t])
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            if np.max(np.abs(coefs)) < 1e-14:
                continue
            base = corr[top_idx] - best_h[i] * coefs
            def obj_t(t):
                return float(np.max(base + t * coefs))
            res = minimize_scalar(obj_t, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            t_new = float(res.x)
            dh = t_new - best_h[i]
            if abs(dh) < 1e-13:
                continue
            for k in range(-(n - 1), n):
                idx = n - 1 + k
                jp = i + k
                jm = i - k
                dc = 0.0
                if 0 <= jp < n:
                    dc += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    dc -= best_h[jm]
                if dc != 0.0:
                    corr[idx] += dh * dc
            best_h[i] = t_new
            cm = float(np.max(corr))
            if cm < sweep_best - 1e-14:
                sweep_best = cm
        best_h = project_h(best_h)
        corr_true = get_corr(best_h).astype(np.float64)
        new_max = float(np.max(corr_true))
        if new_max < best_max - 1e-13:
            best_max = new_max
            corr = corr_true
            no_improve = 0
        else:
            corr = corr_true
            no_improve += 1
        if no_improve >= 15:
            break
    return best_h, best_max * dx

def subgrad_step(h, alpha, n_topk, corr=None):
    n = len(h)
    dx = 2.0 / n
    if corr is None:
        corr = get_corr(h)
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = (topk_idx - (n - 1)).astype(int)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    return grad

def optimize_n(h_init, n, t_end):
    best_h = project_h(h_init.astype(np.float64).copy())
    best_c5 = compute_c5(best_h)
    # SLSQP warmup with increasing T
    for T in [200, 2000, 20000]:
        if time.time() > t_end - 20:
            break
        hc = slsqp_opt(best_h, T, 400, 1e-14)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Subgradient + SLSQP
    if time.time() < t_end - 30:
        mom_v = np.zeros(n)
        for it in range(30):
            if time.time() > t_end - 15:
                break
            cool = 1.0 - it / 30.0
            alpha = 0.003 + 0.006 * cool
            n_topk = 8 + int(14 * cool)
            corr_cur = get_corr(best_h)
            g = subgrad_step(best_h, alpha, n_topk, corr_cur)
            g = 0.45 * mom_v + 0.55 * g
            mom_v = g
            h_sub = project_h(best_h - alpha * g)
            T = 1000 + 8000 * cool
            hc = slsqp_opt(h_sub, T, 350, 1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Heavy coordinate descent
    if time.time() < t_end - 20:
        t_cd_end = min(t_end - 20, time.time() + (t_end - time.time()) * 0.55)
        best_h, best_c5 = coord_descent_incremental(best_h, n, t_cd_end, 400, 25)
    # More SLSQP polish
    if time.time() < t_end - 15:
        hc = slsqp_opt(best_h, 50000, 500, 1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Final coordinate descent
    if time.time() < t_end - 5:
        best_h, best_c5 = coord_descent_incremental(best_h, n, t_end, 200, 30)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5
