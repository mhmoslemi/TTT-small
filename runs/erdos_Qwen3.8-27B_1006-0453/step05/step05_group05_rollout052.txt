def coord_opt_fast(h, n, i, niter=40):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[valid1] += (1.0 - h[np.clip(vm1[valid1], 0, n - 1)])
    coefs[valid2] += (-h[np.clip(vm2[valid2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        mid = (lo + hi) * 0.5
        vals = corr_base + mid * coefs
        best = np.max(vals)
        active = vals >= best - 1e-10
        if hi - lo < 1e-12:
            break
        if np.all(coefs[active] >= 0.0):
            hi = mid
        elif np.all(coefs[active] <= 0.0):
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

    for i in range(n):
        if time.time() > t_end:
            return h
        hi_new = coord_opt_fast(h, n, i)
        if abs(hi_new - h[i]) > 1e-10:
            h[i] = hi_new
    return project_h(h, n)

        if c5_new < best_c5 - 1e-15:

            best_h = h_new
        else:
            break
    return best_h, best_c5

        p[mask] = (1 - 2*h[mask]) * scale  # push towards 0 or 1

        return project_h(h + p, n)
    else:
        return project_h(h + scale * np.random.randn(n), n)

def slsqp_smooth(h, T, n, maxiter=2000):
    def obj(x):
        corr = get_corr(x)
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
        dx = 2.0 / n
        return val * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h
