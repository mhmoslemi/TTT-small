import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-10:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-10:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def _smooth_grad(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -500, 500)
    exp_sh = np.exp(sh)
    w = exp_sh / exp_sh.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wt = w[idx]
        if wt < 1e-14:
            continue
        l = idx - (n - 1)
        i1 = max(0, l)
        i2 = min(n, n + l)
        if i1 < i2:
            grad[i1:i2] += wt * (1.0 - x[i1 - l:i2 - l])
        j1 = max(0, -l)
        j2 = min(n, n - l)
        if j1 < j2:
            grad[j1:j2] -= wt * x[j1 + l:j2 + l]
    return grad * dx

def _lbfgs(h, T, n, maxiter=3000, lam=5000):
    def obj(x):
        return _smooth_obj(x, T, n) + lam * (x.sum() - n / 2.0) ** 2
    def grad(x):
        g = _smooth_grad(x, T, n)
        g += 2.0 * lam * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _slsqp(h, T, n, maxiter=2000):
    def obj(x):
        return _smooth_obj(x, T, n)
    def grad(x):
        return _smooth_grad(x, T, n)
    def cf(x):
        return x.sum() - n / 2.0
    def cg(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cg}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def _coord_opt(h, n, i):
    h2 = h.copy()
    h2[i] = 0.0
    corr_base = np.correlate(h2, 1.0 - h2, mode='full')
    coefs = np.zeros(2 * n - 1)
    for l in range(-(n - 1), n):
        c = 0.0
        if 0 <= i - l < n:
            c += (1.0 - h[i - l])
        if 0 <= i + l < n:
            c -= h[i + l]
        coefs[l + n - 1] = c
    def f(hi):
        return np.max(corr_base + hi * coefs)
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-11, 'maxiter': 50})
    return res.x

def _coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            hi_new = _coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def _binary_sa(h, n, t_end, steps=20000):
    """Simulated annealing on binary-like solutions."""
    best_h = np.clip(h, 0, 1)
    best_h = project_h(best_h, n)
    cur_h = best_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_c5_best = cur_c5
    T_start, T_end = 0.02, 0.001
    ones = np.where(cur_h > 0.5)[0]
    zeros = np.where(cur_h <= 0.5)[0]
    for step in range(steps):
        if time.time() > t_end:
            break
        T = T_start * (T_end / T_start) ** (step / steps)
        if len(ones) < 2 or len(zeros) < 2:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c5 = compute_c5(cur_h)
        dE = new_c5 - cur_c5
        if dE < 0 or np.random.rand() < np.exp(-dE / max(T, 1e-10)):
            cur_c5 = new_c5
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h <= 0.5)[0]
            if new_c5 < best_c5_best:
                best_c5_best = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = 1.0, 0.0
    return best_h, best_c5_best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    alts = []

    alts.append(project_h(tm, n))
    for _ in range(3):

        rb[np.random.choice(n, n // 2, replace=False)] = 1.0
        alts.append(rb)

    for alt in alts:
        if time.time() > t0 + 12:
            break
        hc = _lbfgs(alt, 3000, n, maxiter=800, lam=20000)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 1: L-BFGS-B annealing (fast, analytical gradient)
    T_list = [30, 80, 200, 600, 2000, 6000, 20000, 60000, 200000]
    for T in T_list:
        if time.time() > t0 + budget_s * 0.18:
            break
        hc = _lbfgs(best_h, T, n, maxiter=2000, lam=30000)
        c5v = compute_c5(hc)

            best_h = hc

    # SLSQP polish after L-BFGS-B

        if time.time() > t0 + budget_s * 0.25:
            break
        hc = _slsqp(best_h, T, n, maxiter=3000)

        if c5v < best_c5:

            best_h = hc

    # Phase 2: Coordinate descent
    best_h, best_c5 = _coord_descent(best_h, n, t0 + budget_s * 0.42, max_sweeps=250)

    # Phase 3: Binary simulated annealing
    h_bin = np.where(best_h > 0.3, 1.0, 0.0)

            best_h = hc

    work_h = best_h.copy()

            amp = 0.015 + 0.05 * cool

    return best_h, best_c5, n
