    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    val = m + np.log(np.sum(np.exp(shifted))) / T
    return val * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

            m_curr = np.max(corr)

            j1 = i - ks + (n - 1)

            v2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - best_h[j1[v1]])

            # base = corr with h[i]=0

            best_val = m_curr

    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.exp(np.clip(T * (corr - m), -50, 50)).sum()
    return (m + np.log(s) / T) * dx

def constraint_eq(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_phase(h, n, t_end, Ts):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for T in Ts:
        if time.time() > t_end:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(T,), method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                         options={'maxiter': 1000, 'ftol': 1e-15})
            hc = project_h(np.clip(res.x, 0, 1))
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        except Exception:
            pass
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    ks = np.arange(2 * n - 1)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = best_h[i]
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            m_curr = np.max(corr)
            j1 = i - ks + (n - 1)
            v1 = (j1 >= 0) & (j1 < n)
            j2 = i + ks - (n - 1)
            v2 = (j2 >= 0) & (j2 < n)
            coefs = np.zeros(2 * n - 1)
            coefs[v1] += (1.0 - best_h[j1[v1]])
            coefs[v2] -= best_h[j2[v2]]
            base = corr - hi * coefs
            best_x = hi
            best_val = m_curr
            for x in np.array([0.0, 0.12, 0.25, 0.5, 0.75, 0.88, 1.0]):
                v = np.max(base + x * coefs)
                if v < best_val - 1e-13:
                    best_val = v
                    best_x = x
            lo = max(0.0, best_x - 0.07)
            hi_b = min(1.0, best_x + 0.07)
            for x in np.linspace(lo, hi_b, 15):
                v = np.max(base + x * coefs)
                if v < best_val - 1e-13:
                    best_val = v
                    best_x = x
            if abs(best_x - hi) > 1e-8:
                h_try = best_h.copy()
                h_try[i] = best_x
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try <= best_c5 + 1e-13:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def make_pert(n, cool, rng):
    mod = rng.randint(0, 6)
    if mod == 0:
        bs = rng.randint(3, max(4, n // 8))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.06 + 0.18 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 12))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.05 + 0.12 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(3, max(4, n // 6))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.12 + 0.3 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.03 + 0.06 * cool) * rng.standard_normal(n)
    elif mod == 4:
        n_pairs = rng.randint(3, max(4, n // 15))
        p = np.zeros(n)
        for _ in range(n_pairs):
            a, b = rng.choice(n, 2, replace=False)
            amt = (0.06 + 0.1 * cool) * rng.standard_normal()
            p[a] += amt
            p[b] -= amt
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = (0.05 + 0.1 * cool) * rng.standard_normal(half)
        p[half:n - (n - half)] = -p[:n - half]
    return p

def basin_hopping(h, n, t_end, rng, n_iter, T_base):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    best_h = work_h.copy()
    best_c5 = work_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(n_iter, 1))
        pert = make_pert(n, cool, rng)
        h_pert = project_h(work_h + pert)
        T = T_base + 8000.0 * cool
        try:
            res = minimize(smooth_obj, h_pert, args=(T,), method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                         options={'maxiter': 500, 'ftol': 1e-14})
            hc = project_h(np.clip(res.x, 0, 1))
        except Exception:
            hc = h_pert
        c5v = compute_c5(hc)
        temp = 0.002 * cool + 0.0003
        if c5v < work_c5 or (c5v - work_c5 < 0.01 and rng.rand() < np.exp(-max(c5v - work_c5, 0) / max(temp, 1e-9))):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    return best_h, best_c5

def binarize_refine(h, n, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(800):
        if time.time() > t_end:
            break
        mid = np.where((best_h > 0.2) & (best_h < 0.8))[0]
        if len(mid) == 0:
            break
        idx = mid[rng.randint(len(mid))]
        flip_to = 0.0 if best_h[idx] > 0.5 else 1.0
        h_try = best_h.copy()
        h_try[idx] = flip_to
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm-up
    best_h, best_c5 = slsqp_phase(best_h, n, t0 + budget_s * 0.10, [80, 300, 1200, 5000, 20000])

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.22, 400, rng)

    # Phase 3: Basin hopping (main)
    best_h, best_c5 = basin_hopping(best_h, n, t0 + budget_s * 0.45, rng, 35, 400.0)

    # Phase 4: Coord descent + binarize
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.52, 250, rng)
    best_h, best_c5 = binarize_refine(best_h, n, t0 + budget_s * 0.58, rng)

    # Phase 5: Try upsampled n=200
    n2 = 200
    h2 = interp_h(best_h, n2)
    h2, c5_2 = slsqp_phase(h2, n2, t0 + budget_s * 0.68, [200, 1000, 5000])
    h2, c5_2 = coord_descent(h2, n2, t0 + budget_s * 0.72, 150, rng)
    if c5_2 < best_c5:
        best_h = h2
        best_c5 = c5_2
        n = n2

    # Phase 6: More basin hopping
    best_h, best_c5 = basin_hopping(best_h, n, t0 + budget_s * 0.82, rng, 30, 2000.0)

    # Phase 7: Final coord descent
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, 200, rng)
    best_h, best_c5 = binarize_refine(best_h, n, t0 + budget_s * 0.93, rng)

    # Phase 8: SLSQP polish
    best_h, best_c5 = slsqp_phase(best_h, n, t0 + budget_s * 0.97, [50000, 200000])

    # Phase 9: Final coord
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, 50, rng)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
