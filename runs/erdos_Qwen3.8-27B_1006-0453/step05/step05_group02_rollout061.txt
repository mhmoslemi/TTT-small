def coord_step_fast(h, n, i, corr_base):
    """Find best h[i] given all else fixed. Uses the fact that max of linear is piecewise linear."""
    # c_k(t) = corr_base[k] + (t - h[i]) * a_k
    # where a_k = (1-h[i-k])*[valid] - h[i+k]*[valid]
    a = np.zeros(2*n-1)
    k_range = np.arange(-(n-1), n)
    for idx, k in enumerate(k_range):
        ik = i - k
        ik2 = i + k
        a[idx] = 0.0
        if 0 <= ik < n:
            a[idx] += 1.0 - h[ik]
        if 0 <= ik2 < n:
            a[idx] -= h[ik2]
    
    # c_k(t) = corr_base[k] + t * a_k - h[i] * a_k
    # = (corr_base[k] - h[i]*a_k) + t * a_k
    b = corr_base - h[i] * a
    # Find t in [0,1] minimizing max_k (b[k] + t*a[k])
    # This is a 1D convex piecewise linear optimization
    # Minimum is at t=0, t=1, or a breakpoint
    # Breakpoints: t where b[k1]+t*a[k1] = b[k2]+t*a[k2] → t = (b[k2]-b[k1])/(a[k1]-a[k2])
    # For simplicity, just use minimize_scalar with the exact max
    def obj(t):
        return np.max(b + t * a)
    
    # Try a few points
    candidates = [0.0, 1.0]
    # Add breakpoints from top active k's
    active = np.argsort(b)[-10:]  # top 10 at t=0
    for ai in active:
        for aj in active:
            if ai != aj and abs(a[ai] - a[aj]) > 1e-15:
                t_bp = (b[aj] - b[ai]) / (a[ai] - a[aj])
                if 0.0 < t_bp < 1.0:
                    candidates.append(t_bp)
    
    best_t = h[i]
    best_val = obj(best_t)
    for t_c in candidates + [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]:
        t_c = np.clip(t_c, 0.0, 1.0)
        v = obj(t_c)
        if v < best_val - 1e-15:
            best_val = v
            best_t = t_c
    
    return best_t, best_val
