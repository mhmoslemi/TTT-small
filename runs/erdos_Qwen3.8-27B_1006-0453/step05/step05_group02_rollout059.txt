def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def sobj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def sgrad(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    sw = np.sum(ex)
    w = ex / sw
    grad = np.zeros(n)
    nk1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lo = max(0, k - nk1)
        hi = min(nk1, k - 1)
        if lo <= hi:
            j1 = np.arange(lo, hi + 1) - k + nk1
            grad[lo:hi+1] += wk * (1.0 - x[j1])
        lo2 = max(0, nk1 - k)
        hi2 = min(nk1, 2 * nk1 - k)
        if lo2 <= hi2:
            j2 = np.arange(lo2, hi2 + 1) + k - nk1
            grad[lo2:hi2+1] -= wk * x[j2]
    return grad * dx

                       method='SLSQP', bounds=[(0.0, 1.0)] * n,

                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h

def coord_single(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    cb = np.correlate(h0, 1.0 - h0, mode='full')
    kr = np.arange(-(n-1), n)
    j1 = i + kr; j2 = i - kr
    v1 = (j1 >= 0) & (j1 < n); v2 = (j2 >= 0) & (j2 < n)
    co = np.zeros(2*n-1)
    co[v1] += (1.0 - h[np.clip(j1[v1], 0, n-1)])
    co[v2] -= h[np.clip(j2[v2], 0, n-1)]
    dx = 2.0/n
    def f(hi):
        return float(np.max(cb + hi * co) * dx)
    try:
        r = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
        return float(r.x)
    except:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = rng.permutation(n)

                return best_h, best_c5
            hi = coord_single(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:

            ht = best_h.copy()

            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-13:
                best_c5 = c5t
                best_h = ht
                imp = True
        if not imp:
            break
    return best_h, best_c5

def swap_search(h, n, t_end, rng, n_iter=5000):
    """Try random pairwise swaps that preserve the sum."""
    best_h = h.copy()

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def sobj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def sgrad(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    sw = np.sum(ex)
    w = ex / sw
    grad = np.zeros(n)
    nk1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lo = max(0, k - nk1)
        hi = min(nk1, k - 1)
        if lo <= hi:
            j1 = np.arange(lo, hi + 1) - k + nk1
            grad[lo:hi + 1] += wk * (1.0 - x[j1])
        lo2 = max(0, nk1 - k)
        hi2 = min(nk1, 2 * nk1 - k)
        if lo2 <= hi2:
            j2 = np.arange(lo2, hi2 + 1) + k - nk1
            grad[lo2:hi2 + 1] -= wk * x[j2]
    return grad * dx

def sfunc(x, T, n):
    return sobj(x, T, n)

def sfunc_jac(x, T, n):
    return sgrad(x, T, n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, T, n, maxiter=800):
    try:
        res = minimize(sfunc, h, args=(T, n), jac=sfunc_jac,
                       method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con,
                                     'jac': sum_con_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_single(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    cb = np.correlate(h0, 1.0 - h0, mode='full')
    kr = np.arange(-(n - 1), n)
    j1 = i + kr
    j2 = i - kr
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    co = np.zeros(2 * n - 1)
    co[v1] += (1.0 - h[np.clip(j1[v1], 0, n - 1)])
    co[v2] -= h[np.clip(j2[v2], 0, n - 1)]
    dx = 2.0 / n

    def eval_1d(hi):
        return float(np.max(cb + hi * co) * dx)

    try:
        r = minimize_scalar(eval_1d, bounds=(0.0, 1.0), method='bounded',
                            options={'xatol': 1e-12})
        return float(r.x)
    except Exception:
        return h[i]

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi = coord_single(best_h, n, i)
            if abs(hi - best_h[i]) < 1e-12:
                continue
            ht = best_h.copy()
            ht[i] = hi
            ht = project_h(ht, n)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-13:
                best_c5 = c5t
                best_h = ht
                imp = True
        if not imp:
            break
    return best_h, best_c5

def swap_search(h, n, t_end, rng, n_iter=10000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        i, j = rng.choice(n, 2, replace=False)
        frac = rng.uniform(0.2, 1.0)
        delta = (best_h[j] - best_h[i]) * frac
        if abs(delta) < 1e-10:
            continue
        hi_new = best_h[i] + delta
        hj_new = best_h[j] - delta
        if hi_new < -0.01 or hi_new > 1.01 or hj_new < -0.01 or hj_new > 1.01:
            continue
        ht = best_h.copy()
        ht[i] = hi_new
        ht[j] = hj_new
        ht = project_h(ht, n)
        c5t = compute_c5(ht)
        if c5t < best_c5 - 1e-14:
            best_c5 = c5t
            best_h = ht
    return best_h, best_c5

def make_pert(n, cool, rng):
    mod = rng.randint(0, 6)
    if mod == 0:
        bs = rng.randint(3, max(4, n // 8))
        st = rng.randint(0, max(1, n - bs))
        p = np.zeros(n)
        p[st:st + bs] = (0.05 + 0.15 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        f = rng.randint(1, max(2, n // 16))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.04 + 0.12 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(3, max(4, n // 8))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.1 + 0.3 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    elif mod == 4:
        p = np.zeros(n)
        for _ in range(rng.randint(2, 5)):
            a, b = rng.choice(n, 2, replace=False)
            d = (0.04 + 0.1 * cool) * rng.standard_normal()
            p[a] = d
            p[b] = -d
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.08 * cool * rng.standard_normal(half)
        p[half:] = -0.08 * cool * rng.standard_normal(n - half)
    return p

def basin_hop(work_h, work_c5, best_h, best_c5, n, t_start, t_end, rng, n_iter, T_base, T_amp, temp_base, temp_floor, scale=1.0):
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(1, n_iter)
        pert = make_pert(n, cool * scale, rng)
        h_pert = project_h(work_h + pert, n)
        T_bh = T_base + T_amp * cool
        hc = slsqp_step(h_pert, T_bh, n, maxiter=500)
        c5v = compute_c5(hc)
        temp = temp_base * cool + temp_floor
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    return best_h, best_c5, work_h, work_c5
