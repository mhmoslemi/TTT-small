        return h

        tot = room.sum()

            return h

    return np.clip(h, 0.0, 1.0)

    """Compute objective and gradient for top-k log-sum-exp smoothed max."""
    dx = 2.0 / n
    corr = get_corr(h)
    # Get top-k indices
    if k >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argpartition(corr, -k)[-k:]
    vals = corr[idx]
    m = vals.max()
    exp_vals = np.exp(T * (vals - m))
    Z = exp_vals.sum()
    obj = (m + np.log(Z) / T) * dx
    weights = exp_vals / Z
    
    # Gradient
    grad = np.zeros(n)
    for w, pi in zip(weights, idx):
        lag = int(pi) - (n - 1)
        if lag >= 0:
            # h[n] * (1-h[n+lag]) for n in [0, n-1-lag]

            i1 = np.arange(0, n - lag)
            gr[i1] += (1.0 - h[i1 + lag])
            i2 = np.arange(lag, n)
            gr[i2] -= h[i2 - lag]
        else:
            L = -lag
            gr = np.zeros(n)
            i1 = np.arange(L, n)
            gr[i1] += (1.0 - h[i1 - L])
            i2 = np.arange(0, n - L)

                      bounds=[(0.0, 1.0)] * n,

        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt(h, n, i, niter=50):
    """Ternary search to optimize h[i]."""
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m = np.arange(-(n-1), n)
    vm1 = i - m
    vm2 = i + m
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2*n - 1)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n-1)])

            hi = m2
        else:

    return (lo + hi) / 2.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def corr_lag_grad(h, n, lag):
    gr = np.zeros(n)
    if lag >= 0:
        i1 = np.arange(0, n - lag)
        gr[i1] += (1.0 - h[i1 + lag])
        i2 = np.arange(lag, n)
        gr[i2] -= h[i2 - lag]
    else:
        L = -lag
        i1 = np.arange(L, n)
        gr[i1] += (1.0 - h[i1 - L])
        i2 = np.arange(0, n - L)
        gr[i2] -= h[i2 + L]
    return gr

def obj_grad_topk(h, n, k, T):
    dx = 2.0 / n
    corr = get_corr(h)
    if k >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argpartition(corr, -k)[-k:]
    vals = corr[idx]
    m = vals.max()
    ev = np.exp(T * (vals - m))
    Z = ev.sum()
    obj = (m + np.log(Z) / T) * dx
    w = ev / Z
    grad = np.zeros(n)
    for wi, pi in zip(w, idx):
        lag = int(pi) - (n - 1)
        grad += wi * corr_lag_grad(h, n, lag)
    grad *= dx
    return obj, grad

def _wrap_obj(x, n, k, T):
    return obj_grad_topk(x, n, k, T)[0]

def _wrap_jac(x, n, k, T):
    return obj_grad_topk(x, n, k, T)[1]

def _wrap_con(x, n):
    return x.sum() - n / 2.0

def slsqp_topk(h, n, k, T, maxiter=400):
    try:
        res = minimize(_wrap_obj, h, jac=_wrap_jac, args=(n, k, T),
                      method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _wrap_con, 'args': (n,)}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt(h, n, i, niter=45):
    hb = h.copy(); hb[i] = 0.0
    cb = get_corr(hb)
    m = np.arange(-(n-1), n)
    v1 = i - m; v2 = i + m
    p1 = (v1 >= 0) & (v1 < n); p2 = (v2 >= 0) & (v2 < n)
    co = np.zeros(2*n - 1)
    co[p1] += (1.0 - h[np.clip(v1[p1], 0, n-1)])
    co[p2] += (-h[np.clip(v2[p2], 0, n-1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        t1 = lo + (hi-lo)/3.0; t2 = hi - (hi-lo)/3.0
        if np.max(cb + t1*co) < np.max(cb + t2*co):
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_deadline, max_sweeps=50, use_pairs=False):
    best = h.copy()
    best_c5 = compute_c5(best)
    for sw in range(max_sweeps):
        if time.time() > t_deadline:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_deadline:
                return best, best_c5
            hv = coord_opt(best, n, i)
            if abs(hv - best[i]) < 1e-13:
                continue
            ht = best.copy(); ht[i] = hv
            ht = project_h(ht, n)
            c = compute_c5(ht)
            if c < best_c5 - 1e-15:
                best_c5 = c; best = ht; imp = True
        if use_pairs and time.time() < t_deadline:
            for _ in range(n // 3):
                if time.time() > t_deadline:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                hb = best.copy(); hb[i] = 0; hb[j] = 0
                cb = get_corr(hb)
                mi = np.arange(-(n-1), n)
                co_i = np.zeros(2*n-1); co_j = np.zeros(2*n-1)
                vi1 = i-mi; vi2 = i+mi; vj1 = j-mi; vj2 = j+mi
                pi1 = (vi1>=0)&(vi1<n); pi2 = (vi2>=0)&(vi2<n)
                pj1 = (vj1>=0)&(vj1<n); pj2 = (vj2>=0)&(vj2<n)
                co_i[pi1] += (1-best[np.clip(vi1[pi1],0,n-1)])
                co_i[pi2] += (-best[np.clip(vi2[pi2],0,n-1)])
                co_j[pj1] += (1-best[np.clip(vj1[pj1],0,n-1)])
                co_j[pj2] += (-best[np.clip(vj2[pj2],0,n-1)])
                bv = np.inf; bx = best[i]; by = best[j]
                for xi in np.linspace(0, 1, 7):
                    for xj in np.linspace(0, 1, 7):
                        v = np.max(cb + xi*co_i + xj*co_j)
                        if v < bv:
                            bv = v; bx = xi; by = xj
                if abs(bx-best[i]) < 1e-12 and abs(by-best[j]) < 1e-12:
                    continue
                ht = best.copy(); ht[i] = bx; ht[j] = by
                ht = project_h(ht, n)
                c = compute_c5(ht)
                if c < best_c5 - 1e-15:
                    best_c5 = c; best = ht; imp = True
        if not imp:
            break
    return best, best_c5
