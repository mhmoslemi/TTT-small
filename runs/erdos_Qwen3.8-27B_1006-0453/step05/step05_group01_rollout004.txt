    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_objective(h, T, n_pen=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    # Add penalty for spread of top correlations
    top = np.sort(corr)[-n_pen:]
    val += 0.1 * (top[-1] - top[0])
    return val * dx

                       options={'maxiter': maxiter, 'ftol': ftol})

    except:
        pass

def subgradient_step(h, T, n_top, nu=0.7):
    """Nesterov-accelerated subgradient step"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = np.argsort(corr)[-n_top:]
    shifts = topk - (n - 1)
    
    grad = np.zeros(n)

        jp = i_idx + s
        jm = i_idx - s

        v2 = (jm >= 0) & (jm < n)
        jpc = np.clip(jp, 0, n-1)
        jmc = np.clip(jm, 0, n-1)
        g = (1.0 - h[jpc]) * v1 - h[jmc] * v2
        grad += g
    grad /= (len(shifts) * dx)
    
    return grad

        if c5_new < best_c5:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_h(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def analytic_opt_t(base, coefs):
    n = len(base)
    best_t = 0.0
    best_val = np.max(base)
    v1 = np.max(base + coefs)
    if v1 < best_val:
        best_t = 1.0
        best_val = v1
    if best_val <= 0:
        return best_t
    # Collect candidate breakpoints
    cands = [0.0, 1.0]
    for i in range(n):
        bi = base[i]
        ci = coefs[i]
        for j in range(i + 1, n):
            cj = coefs[j]
            d = ci - cj
            if abs(d) > 1e-14:
                tc = (base[j] - bi) / d
                if -1e-10 <= tc <= 1.0 + 1e-10:
                    cands.append(max(0.0, min(1.0, tc)))
    for tc in cands:
        val = np.max(base + tc * coefs)
        if val < best_val - 1e-15:
            best_val = val
            best_t = tc
    return best_t

def coord_descent(h, n, t_end, n_top, max_sweeps=60):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx].copy() - best_h[i] * coefs
            t_new = analytic_opt_t(base, coefs)
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def nesterov_subgrad(h, n, t_end, n_iter, n_top_lo, n_top_hi):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        n_top = max(3, n_top_lo + int((n_top_hi - n_top_lo) * cool))
        alpha = 0.0008 * (1.0 + 3.0 * cool)
        h_look = project_h(best_h - v)
        g = subgrad(h_look, n_top)
        v = 0.85 * v + 0.15 * g
        h_new = project_h(best_h - alpha * v)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def basin_toggle(h, n, t_end, n_iter):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        r = np.random.random()
        h_try = best_h.copy()
        if r < 0.4:
            blen = np.random.randint(3, max(4, n // 6))
            bs = np.random.randint(0, max(1, n - blen))
            h_try[bs:bs + blen] = 1.0 - h_try[bs:bs + blen]
        elif r < 0.7:
            blen = np.random.randint(2, max(3, n // 8))
            bs = np.random.randint(0, max(1, n - blen))
            h_try[bs:bs + blen] += np.random.randn(blen) * 0.15
        else:
            idx = np.random.choice(n, size=max(3, n // 4), replace=False)
            h_try[idx] = 1.0 - h_try[idx]
        h_try = project_h(h_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def optimize_pipeline(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # Phase 1: SLSQP warm
    if t_end - time.time() > 40:
        for T in [30, 300, 3000, 30000]:
            if t_end - time.time() < 15:
                break
            hc = slsqp_opt(best_h, n, T, 350, 1e-14)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Phase 2: Coordinate descent main
    if t_end - time.time() > 30:
        best_h, best_c5 = coord_descent(best_h, n, t_end - (t_end - time.time()) * 0.25, 40, 50)
    # Phase 3: Nesterov + CD cycles
    if t_end - time.time() > 40:
        for rd in range(3):
            if t_end - time.time() < 20:
                break
            t_sub = time.time() + min(15, (t_end - time.time()) * 0.25)
            best_h, best_c5 = nesterov_subgrad(best_h, n, t_sub, 25, 5, 30)
            t_cd = time.time() + min(20, (t_end - time.time()) * 0.35)
            best_h, best_c5 = coord_descent(best_h, n, t_cd, 80, 30)
    # Phase 4: Basin toggle
    if t_end - time.time() > 20:
        h_tog, c5_tog = basin_toggle(best_h, n, t_end - (t_end - time.time()) * 0.3, 60)
        if c5_tog < best_c5:
            best_c5 = c5_tog
            best_h = h_tog
            if t_end - time.time() > 15:
                best_h, best_c5 = coord_descent(best_h, n, t_end - 10, 100, 30)
    # Phase 5: Final high-precision CD
    if t_end - time.time() > 10:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, min(150, n * 2 - 1), 20)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0
    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)
    # Main optimization at initial n
    t_main_end = t0 + budget_s * 0.60
    best_h, best_c5 = optimize_pipeline(best_h, n, t_main_end)
    # Try coarser n=160
    if t_end - time.time() > budget_s * 0.25 and n > 160:
        n2 = 160
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        t2_end = t0 + budget_s * 0.80
        h2_opt, c5_2 = optimize_pipeline(h2, n2, t2_end)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_opt
            n = n2
    # Try finer n=240
    if t_end - time.time() > budget_s * 0.20 and n != 240:
        n3 = 240
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = project_h(h3)
        t3_end = t0 + budget_s * 0.94
        h3_opt, c5_3 = optimize_pipeline(h3, n3, t3_end)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3_opt
            n = n3
    # Final polish
    if t_end - time.time() > 12:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 2, min(150, n * 2 - 1), 15)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
