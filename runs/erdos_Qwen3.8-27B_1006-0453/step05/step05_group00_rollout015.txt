            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def fix_count(h_bin, N):
    target = N // 2
    n1 = int(h_bin.sum())
    if n1 > target:
        idx = np.where(h_bin==1)[0]
        h_bin[idx[np.random.choice(len(idx), n1-target, replace=False)]] = 0
    elif n1 < target:
        idx = np.where(h_bin==0)[0]
        h_bin[idx[np.random.choice(len(idx), target-n1, replace=False)]] = 1
    return h_bin

def swap_delta(h, N, p, q, p_is_one=True):
    s = np.arange(-(N-1), N)
    ps, pm = p+s, p-s
    qs, qm = q+s, q-s
    vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
    vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
    d = np.zeros(2*N-1)
    pc=np.clip(ps,0,N-1)[vps]; pmc=np.clip(pm,0,N-1)[vpm]
    qsc=np.clip(qs,0,N-1)[vqs]; qmc=np.clip(qm,0,N-1)[vqm]
    if p_is_one:
        d[vps] -= (1.0-h[pc])
        d[vpm] += h[pmc]
    else:
        d[vps] += (1.0-h[pc])
        d[vpm] -= h[pmc]
    d[vqs] += (1.0-h[qsc])
    d[vqs] -= (1.0-h[qsc])  # q goes 0->1: add (1-h[q+s])
    d[vqs][:] += 0  # placeholder
    # Redo q part correctly:
    d[qsc] += 0  # overwritten below
    # Clean approach:
    d2 = np.zeros(2*N-1)
    if p_is_one:
        d2[vps] -= (1.0-h[pc])
        d2[vpm] += h[pmc]
    else:
        d2[vps] += (1.0-h[pc])
        d2[vpm] -= h[pmc]
    d2[vqs] += (1.0-h[qsc])
    d2[vqm] -= h[qmc]
    return d2

def anneal(h, N, deadline, T0=0.06, cool=0.9992, max_no_improve=800, multi_restart=2):
    h = h.astype(float).copy()
    h = fix_count(h, N)
    corr = fft_corr(h)
    best_pk = float(corr.max())
    best_h = h.copy()
    current_pk = best_pk
    s_arr = np.arange(-(N-1), N)
    dx = 2.0/N
    for restart in range(multi_restart):
        T = T0 if restart == 0 else T0*0.4
        no_imp = 0
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        while time.time() < deadline and T > 0.001:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            ps = p+s_arr; pm = p-s_arr
            qs = q+s_arr; qm = q-s_arr
            vps=(ps>=0)&(ps<N); vpm=(pm>=0)&(pm<N)
            vqs=(qs>=0)&(qs<N); vqm=(qm>=0)&(qm<N)
            d = np.zeros(2*N-1)
            pc=np.clip(ps,0,N-1); pmc=np.clip(pm,0,N-1)
            qsc=np.clip(qs,0,N-1); qmc=np.clip(qm,0,N-1)
            d[vps] -= (1.0-h[pc[vps]])
            d[vpm] += h[pmc[vpm]]
            d[vqs] += (1.0-h[qsc[vqs]])
            d[vqm] -= h[qmc[vqm]]
            nc = corr + d
            npk = float(nc.max())
            delta = npk - current_pk
            if delta < 0 or np.random.random() < np.exp(-delta/max(T,1e-8)):
                h[p]=0; h[q]=1; corr=nc; current_pk=npk
                ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
                if npk < best_pk:
                    best_pk=npk; best_h=h.copy(); no_imp=0
                else:
                    no_imp+=1
            else:
                no_imp+=1
            T*=cool
            if no_imp > max_no_improve:
                no_imp=0
                if restart < multi_restart-1:
                    break
                else:
                    h=best_h.copy(); corr=fft_corr(h); current_pk=best_pk
                    ones=np.where(h==1)[0]; zeros=np.where(h==0)[0]
    return best_h, best_pk

def continuous_opt(h, N, deadline, top_k=30, lr0=0.004):
    h = h.astype(float).copy()
    m = np.zeros(N); v = np.zeros(N)
    b1,b2,eps = 0.85,0.999,1e-8
    t=0; best_pk=float(np.max(fft_corr(h))); best_h=h.copy()
    center=N-1; lr=lr0
    while time.time() < deadline:
        corr=fft_corr(h)
        pk=float(corr.max())
        if pk<best_pk: best_pk=pk; best_h=h.copy()
        idx=np.argpartition(corr,-top_k)[-top_k:]
        grad=np.zeros(N)
        for ai in idx:
            s=int(ai-center)
            if s==0: grad += (1.0-2.0*h)
            elif s>0:
                grad[:N-s]+=(1.0-h[s:]); grad[s:]-=h[:N-s]
            else:
                s2=-s; grad[s2:] +=(1.0-h[:N-s2]); grad[:N-s2] -= h[s2:]
        grad/=top_k
        t+=1
        m=b1*m+(1-b1)*grad; v=b2*v+(1-b2)*grad*grad
        mh=m/(1-b1**t); vh=v/(1-b2**t)
        h=proj_sum(h - lr*mh/np.sqrt(vh+eps), N/2.0)
        if t%500==0: lr*=0.97
    return best_h, best_pk

def run(seed=42, budget_s=1100, **kwargs):
    t0=time.time(); deadline=t0+budget_s-3
    np.random.seed(seed)
    h0=np.array(initial_h_values, dtype=float)
    N0=len(h0)
    best_h=h0.copy(); best_c5=float(np.max(fft_corr(h0)))*(2.0/N0)

    # Phase 1: Continuous optimization from start
    d1=min(deadline-300, t0+180)
    hc=h0+np.random.randn(N0)*0.04
    hc=proj_sum(hc, N0/2.0)
    hC,pkC=continuous_opt(hc,N0,d1,top_k=35,lr0=0.005)
    c5C=pkC*(2.0/N0)
    if c5C<best_c5: best_c5,best_h=c5C,hC.copy()

    # Phase 2: Binary annealing from best
    d2=min(deadline-200, time.time()+200)
    hb=fix_count((best_h>0.5).astype(float), N0)
    hB,pkB=anneal(hb,N0,d2,T0=0.07,cool=0.9993,max_no_improve=1000,multi_restart=2)
    c5B=pkB*(2.0/N0)
    if c5B<best_c5: best_c5,best_h=c5B,hB.copy()

    # Phase 3: Short continuous polish
    if time.time()<deadline-120:
        d3=min(deadline-100, time.time()+100)
        hc2=best_h.astype(float)+np.random.randn(len(best_h))*0.02
        hc2=proj_sum(hc2,len(best_h)/2.0)
        hC2,pkC2=continuous_opt(hc2,len(best_h),d3,top_k=50,lr0=0.003)
        c5C2=pkC2*(2.0/len(best_h))
        if c5C2<best_c5: best_c5,best_h=c5C2,hC2.copy()

    # Phase 4: Multi-N exploration
    for Nt in [320,400,460,540,680,820]:
        if time.time()>deadline-50: break
        if Nt%2!=0: Nt+=1
        x_old=np.linspace(0,2,len(best_h),endpoint=False)
        x_new=np.linspace(0,2,Nt,endpoint=False)
        hi=np.interp(x_new,x_old,best_h)
        hbt=fix_count((hi>0.5).astype(float),Nt)
        sub=min(deadline-5,time.time()+35)
        hrt,pkr=anneal(hbt,Nt,sub,T0=0.05,cool=0.9995,max_no_improve=500,multi_restart=1)
        c5r=pkr*(2.0/Nt)
        if c5r<best_c5: best_c5,best_h=c5r,hrt.copy()

    # Phase 5: Final binary polish on best
    if time.time()<deadline-3:
        hf=fix_count((best_h>0.5).astype(float),len(best_h))
        hfp,pkf=anneal(hf,len(best_h),deadline-1,T0=0.02,cool=0.9997,max_no_improve=600,multi_restart=1)
        c5f=pkf*(2.0/len(best_h))
        if c5f<best_c5: best_c5,best_h=c5f,hfp.copy()

    final_c5=float(np.max(fft_corr(best_h)))*(2.0/len(best_h))
    return best_h, final_c5, len(best_h)
