def project_population(x, n):
    """Handle bounds and integral"""
    a = np.clip(x * (0.5 / 0.999), -0.999, 0.999)
    adjusted = np.clip(a * 0.9 + 0.5, 0.49, 0.51)
    sum_ = np.sum(adjusted * 0.999)
    if np.abs(sum_ - 80) > 1.0:
        imbalance = 100 * (80 - np.round(sum_))
        adjusted += imbalance * (np.random.rand(n) < 0.01)
    return adjusted / 0.999 * 2

def multi_start_evolution(n_iterations, seed, initial_population):
    """Parallel evolution across multiple points"""
    np.random.seed(seed)
    population = initial_population.copy()
    for _ in range(n_iterations):
        np.random.seed(seed + np.random.randint(1000))
        new_population = []
        for idx in range(len(population) // 4):
            samples = population[np.random.choice(len(population), 5)]
            best = np.median(samples, axis=0)
            pert = np.random.uniform(-0.2, 0.2, len(best))
            try:
                candidate = best + pert
                if np.all((candidate <= 0.99) & (candidate >= 0.01)):
                    new_population.append(candidate)
            except:
                continue
        population = np.vstack([population, new_population])

        eval = find_benchmark(new_link, direction)
        
        if np.less(eval['c5'], best_c5):
            best_c5 = eval['c5']
            best_pop = eval['h_values']
            
        runs['objective'][i] = eval['objective']
        runs['c5'][i] = eval['c5']
        
        # Track summary points
        if i % summary_every == 0:
            summary = {}
            search_stats = evaluate_search_space(best_pop)
            summary['best_scalar'] = search_stats.get('best_val', best_c5)
            process_summary(summary, i)
    
    final_eval = compute_c5(best_pop)
    runs['best_c5'] = final_eval
    return best_pop, final_eval, n

def adjust_integral(x, n):
    """Heat flux correction"""
    excess = np.sum(x) - 80
    if excess > 0.5:
        excess_indices = x > 0.5
        reduc = excess * 0.4
        for idx in range(n):
            if excess_indices[idx] and reduc > 0.1:
                x[idx] -= 0.05
                reduc -= 0.05

        add_indices = x < 0.5
        add_points = -excess * 0.4
        for idx in range(n):
            if add_indices[idx] and add_points > 0.1:
                x[idx] += 0.05
                add_points -= 0.05
    return np.clip(x, 0, 1)
