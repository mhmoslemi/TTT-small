            tt[i] = hn
            tt = project_h(tt)
            cc = compute_c5(tt)
            if cc < bc - 1e-14:
                bc = cc
                bh = tt
                imp = True
        if not imp:
            break
    return bh, bc

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h)

def make_pert(n, cool, rng):
    m = rng.randint(5)
    p = np.zeros(n)
    if m == 0:
        bs = rng.randint(3, max(4, n // 7))
        st = rng.randint(0, n - bs)
        p[st:st + bs] = (0.04 + 0.14 * cool) * rng.standard_normal(bs)
    elif m == 1:
        fr = rng.randint(1, max(2, n // 12))
        ph = rng.uniform(0, 2 * np.pi)
        p = (0.03 + 0.10 * cool) * np.sin(2 * np.pi * fr * np.arange(n) / n + ph)
    elif m == 2:
        np_ = rng.randint(3, max(4, n // 6))
        ix = rng.choice(n, np_, replace=False)
        p[ix] = (0.08 + 0.25 * cool) * rng.standard_normal(np_)
    elif m == 3:
        p = (0.02 + 0.06 * cool) * rng.standard_normal(n)
    else:
        hf = n // 2
        p[:hf] = 0.08 * cool * rng.standard_normal(hf)
        p[hf:] = -0.08 * cool * rng.standard_normal(n - hf)
    return p

def basin_hop(h, n, t_end, nit, rng):
    wh = h.copy()
    wc = compute_c5(wh)
    bh = wh.copy()
    bc = wc
    for it in range(nit):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.85 * (it / max(nit, 1))
        hp = project_h(wh + make_pert(n, cool, rng))
        T = 300 + 15000 * cool
        hc = slsqp_step(hp, T, n, 900, 1e-15)
        cv = compute_c5(hc)
        temp = 0.0015 * cool + 0.0002
        if cv < wc or rng.rand() < np.exp(-(cv - wc) / max(temp, 1e-10)):
            wh = hc
            wc = cv
            if cv < bc:
                bc = cv
                bh = hc.copy()
    return bh, bc

def nm_reduced_obj(fa_r, h_base, active):
    h_r = h_base.copy()
    h_r[active] = fa_r
    h_r = project_h(h_r)
    return compute_c5(h_r)

def nm_active(h, n, t_end):
    active = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    if len(active) < 2 or len(active) > 35:
        return h, compute_c5(h)
    try:
        res = minimize(nm_reduced_obj, h[active].copy(),
                       args=(h, active), method='Nelder-Mead',
                       options={'maxiter': 8000, 'xatol': 1e-12, 'fatol': 1e-14})
        h_new = h.copy()
        h_new[active] = res.x
        h_new = project_h(h_new)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def optimize_at_n(h, n, t_end, rng):
    best = project_h(h.copy())
    bc = compute_c5(best)
    rem = t_end - time.time()
    if rem <= 5:
        return best, bc
    t1 = time.time() + rem * 0.18
    for T in [50, 200, 800, 3000, 10000, 40000, 150000, 500000]:
        if time.time() > t1:
            break
        hc = slsqp_step(best, T, n, 2000, 1e-16)
        cv = compute_c5(hc)
        if cv < bc:
            bc = cv
            best = hc
    t2 = time.time() + min(rem * 0.22, t_end - time.time())
    best, bc = basin_hop(best, n, t2, 45, rng)
    t3 = time.time() + min(rem * 0.15, t_end - time.time())
    best, bc = coord_descent(best, n, t3, 100)
    t4 = time.time() + min(rem * 0.08, t_end - time.time())
    best2, bc2 = nm_active(best, n, t4)
    if bc2 < bc:
        best, bc = best2, bc2
    t5 = time.time() + min(rem * 0.17, t_end - time.time())
    for T in [400000, 2000000, 8000000]:
        if time.time() > t5:
            break
        hc = slsqp_step(best, T, n, 3000, 1e-17)
        cv = compute_c5(hc)
        if cv < bc:
            bc = cv
            best = hc
    best, bc = coord_descent(best, n, t_end, 60)
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h160 = np.array(initial_h_values, dtype=float).copy()
    else:
        h160 = np.zeros(160)
        h160[:80] = 1.0
    n1 = len(h160)
    h160 = project_h(h160)
    t_mid = t0 + budget_s * 0.58
    best_h, best_c5 = optimize_at_n(h160, n1, t_mid, rng)
    if time.time() < t0 + budget_s * 0.88:
        t_end2 = t0 + budget_s * 0.94
        h200 = project_h(upsample(best_h, 200))
        h200b, c5_200 = optimize_at_n(h200, 200, t_end2, rng)
        if c5_200 < best_c5:
            best_h, best_c5 = h200b, c5_200
    if time.time() < t0 + budget_s * 0.94:
        t_end3 = t0 + budget_s * 0.985
        n_try = 240
        h_try = project_h(upsample(best_h, n_try))
        h_try = slsqp_step(h_try, 3000, n_try, 1500, 1e-15)
        h_try = slsqp_step(h_try, 50000, n_try, 1500, 1e-16)
        cv = compute_c5(h_try)
        if cv < best_c5:
            h_try = coord_descent(h_try, n_try, t_end3, 40)[0]
            cv = compute_c5(h_try)
            if cv < best_c5:
                best_h, best_c5 = h_try, cv
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
