def fast_coord_opt(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    corr_one = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    c = corr_one - corr_base  # coefficient of t
    
    # Minimize f(t) = max_k(corr_base[k] + t*c[k]) for t in [0,1]
    # Binary search on the optimal value M
    lo_M = np.max(np.minimum(corr_base, corr_base + c))  # lower bound on f
    hi_M = np.max(np.maximum(corr_base, corr_base + c))  # upper bound on f
    
    for _ in range(50):
        M = (lo_M + hi_M) / 2
        # Check feasibility: exists t in [0,1] s.t. corr_base[k] + t*c[k] <= M for all k
        # For c[k] > 0: t <= (M - corr_base[k]) / c[k]
        # For c[k] < 0: t >= (corr_base[k] - M) / (-c[k])
        # For c[k] = 0: need corr_base[k] <= M
        pos_mask = c > 0
        neg_mask = c < 0
        zero_mask = c == 0
        
        if np.any(zero_mask) and np.max(corr_base[zero_mask]) > M:
            lo_M = M
            continue
        
        t_lo = 0.0
        t_hi = 1.0
        if np.any(neg_mask):
            t_lo = np.max((corr_base[neg_mask] - M) / (-c[neg_mask]))
        if np.any(pos_mask):
            t_hi = np.min((M - corr_base[pos_mask]) / c[pos_mask])
        
        if t_lo <= t_hi and t_lo <= 1.0 and t_hi >= 0.0:
            hi_M = M
        else:
            lo_M = M
    
    # Find the optimal t
    M = (lo_M + hi_M) / 2
    pos_mask = c > 0
    neg_mask = c < 0
    t_lo = 0.0
    t_hi = 1.0
    if np.any(neg_mask):
        t_lo = float(np.max((corr_base[neg_mask] - M) / (-c[neg_mask])))
    if np.any(pos_mask):
        t_hi = float(np.min((M - corr_base[pos_mask]) / c[pos_mask]))
    t_lo = max(0.0, t_lo)
    t_hi = min(1.0, t_hi)
    if t_lo > t_hi:

    return (t_lo + t_hi) / 2

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_opt(h, n, i):
    """Find optimal h[i] to minimize max correlation, using binary search on value."""
    old = h[i]
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    c = corr1 - corr0  # coefficient of t
    
    # f(t) = max_k(corr0[k] + t * c[k]), minimize over t in [0,1]
    # f is convex piecewise-linear

    # Lower bound: max of min at t=0 and t=1
    lo_M = float(np.max(np.minimum(corr0, corr1)))
    hi_M = float(np.max(np.maximum(corr0, corr1)))
    
    if hi_M - lo_M < 1e-12:
        return old

        M = (lo_M + hi_M) * 0.5
        # Feasibility: exists t in [0,1] s.t. corr0[k] + t*c[k] <= M for all k
        pos_mask = c > 1e-15
        neg_mask = c < -1e-15
        zero_mask = ~(pos_mask | neg_mask)
        
        if np.any(zero_mask) and np.max(corr0[zero_mask]) > M + 1e-12:
            lo_M = M
            continue
        
        t_lo = 0.0
        t_hi = 1.0
        if np.any(neg_mask):
            t_lo = float(np.max((corr0[neg_mask] - M) / (-c[neg_mask])))
        if np.any(pos_mask):
            t_hi = float(np.min((M - corr0[pos_mask]) / c[pos_mask]))

            hi_M = M

    t_hi = min(1.0, t_hi)

f(t) is convex (max of linear functions). The minimum is at a point where at least two of the active constraints are tight, or at a boundary.

Binary search on M: the minimum value is the smallest M such that the feasible set {t : corr0[k] + t*c[k] ≤ M ∀k} intersects [0,1].

- For k with c[k] > 0: t ≤ (M - corr0[k]) / c[k]
- For k with c[k] < 0: t ≥ (corr0[k] - M) / (-c[k])
- For k with c[k] = 0: need corr0[k] ≤ M
- Also: 0 ≤ t ≤ 1

    t_hi = min(1.0, t_hi)
    t_opt = (t_lo + t_hi) * 0.5 if t_lo <= t_hi else old
    new_max = float(np.max(corr0 + t_opt * c))
    return t_opt, new_max

def coord_descent(h, n, t_end, max_sweeps=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = np.arange(n) if sweep < 3 else np.random.permutation(n)
        sweep_imp = 0
        for i in order:
            t_opt, new_max = fast_coord_opt(best_h, n, i)
            if abs(t_opt - best_h[i]) > 1e-12:
                best_h[i] = t_opt
                s = best_h.sum()
                target = n / 2.0
                if abs(s - target) > 1e-10:
                    interior = (best_h > 0.01) & (best_h < 0.99)
                    if interior.any():
                        best_h[interior] += (target - s) / interior.sum()
                        best_h = np.clip(best_h, 0, 1)
                    else:
                        best_h = project_h(best_h, n)
                c5_new = new_max * 2.0 / n
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    sweep_imp += 1
        if sweep_imp == 0:
            no_improve += 1
            if no_improve >= 8:
                break
        else:
            no_improve = 0
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def binary_swap(h, n, t_end, max_iters=50000):
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[np.random.choice(idx, num_ones - target, replace=False)] = 0
    elif num_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[np.random.choice(idx, target - num_ones, replace=False)] = 1
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        i1 = ones[np.random.randint(len(ones))]
        i0 = zeros[np.random.randint(len(zeros))]
        h_bin[i1] = 0
        h_bin[i0] = 1
        c5_new = compute_c5(h_bin)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_bin.copy()
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
        else:
            h_bin[i1] = 1
            h_bin[i0] = 0
    return best_h, best_c5

def slsqp_smooth(h, T, n, maxiter=150):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def pairing_swap(h, n, t_end, max_iters=30000):
    """Try pairs of coordinate swaps for binary sequences."""
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones != target:
        h_bin = project_h(h_bin.copy(), n)
        h_bin = (h_bin > 0.5).astype(float)
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    top_lags = np.argsort(corr)[-6:]
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        # Target high-correlation lags
        ki = top_lags[np.random.randint(len(top_lags))]
        lag = ki - (n - 1)
        # Pick i that contributes to this lag
        valid_i = []
        for i in range(n):
            j1 = i + lag
            j2 = i - lag
            if 0 <= j1 < n or 0 <= j2 < n:
                valid_i.append(i)
        if not valid_i:
            continue
        i = valid_i[np.random.randint(len(valid_i))]
        if h_bin[i] == 1:
            partners = np.where(h_bin == 0)[0]
        else:
            partners = np.where(h_bin == 1)[0]
        if len(partners) == 0:
            continue
        j = partners[np.random.randint(len(partners))]
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c5_new = compute_c5(h_bin)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_bin.copy()
            corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
            top_lags = np.argsort(corr)[-6:]
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 160
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP smooth init
    for T in [8000, 3000, 1000, 400]:
        if time.time() > t0 + budget_s * 0.12:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=120)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Fast coordinate descent
    t2 = t0 + budget_s * 0.50
    best_h, best_c5 = coord_descent(best_h, n, t2, max_sweeps=400)

    # Phase 3: Binary swap search
    t3 = t0 + budget_s * 0.70
    h_bin, c5_bin = binary_swap(best_h, n, t3, max_iters=40000)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin

    # Phase 4: Pairing swap on binary
    t4 = t0 + budget_s * 0.80
    h_pair, c5_pair = pairing_swap(best_h, n, t4, max_iters=20000)
    if c5_pair < best_c5:
        best_c5 = c5_pair
        best_h = h_pair

    # Phase 5: Coordinate descent from best
    t5 = t0 + budget_s * 0.93
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=300)

    # Phase 6: Final SLSQP polish
    if time.time() < t0 + budget_s * 0.98:
        hc = slsqp_smooth(best_h, 500, n, maxiter=100)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.995, max_sweeps=50)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
