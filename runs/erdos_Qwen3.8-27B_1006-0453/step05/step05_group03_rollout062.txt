```python
def coord_opt_fast(h, n, i):
    """Find optimal h[i] given all other h fixed. Returns new h[i]."""
    # Set h[i] = 0, compute base correlation
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    
    # Compute coefficients c[d] for d = -(n-1), ..., (n-1)
    # c[d] = 1 - h[i+d] - h[i-d] (out of range = 0)
    # In full correlation index: k = d + (n-1)
    c = np.full(2*n-1, 1.0)
    # Subtract h[i+d] for d where i+d is in [0,n-1]
    d_max = n-1
    d_min = -(n-1)
    for d in range(d_min, d_max+1):
        c[d + (n-1)] = 1.0
        if 0 <= i+d < n:
            c[d + (n-1)] -= h[i+d]
        if 0 <= i-d < n:
            c[d + (n-1)] -= h[i-d]
    
    # Now find v in [0,1] minimizing max(corr_base + v*c)
    # This is 1D convex minimax
    def f(v):
        return np.max(corr_base + v * c)
    
    # Use bounded minimization
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-10})
    return res.x
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        s = h.sum()
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def fast_coord_opt(h, n, i):
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    d = np.arange(-(n-1), n)
    c = np.ones(2*n-1)
    v1 = (i + d >= 0) & (i + d < n)
    c[v1] -= h[i + d[v1]]
    v2 = (i - d >= 0) & (i - d < n)
    c[v2] -= h[i - d[v2]]
    lo, hi = 0.0, 1.0
    for _ in range(35):
        mid = (lo + hi) * 0.5
        vals = corr_base + mid * c
        mv = vals.max()
        active = vals >= mv - 1e-13
        sl = c[active]
        if sl.min() > 1e-12:
            hi = mid
        elif sl.max() < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stagnant = 0
    while time.time() < t_end:
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            nv = fast_coord_opt(best_h, n, i)
            if abs(nv - best_h[i]) > 1e-10:
                best_h[i] = nv
                best_h = project_h(best_h, n)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3:
                break
    return best_h, best_c5

def slsqp_opt(h, T, n, maxiter=100):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        sh = np.clip(T * (corr - m), -50, 50)
        return float(m + np.log(np.sum(np.exp(sh))) / T)
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun * 1.0 < compute_c5(h) + 1e-6:  # crude comparison
            return project_h(np.clip(res.x, 0, 1), n)
    except:
        pass
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return project_h(h, n)

def make_random_bin(n):
    h = np.zeros(n)
    idx = np.random.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):

        temp = 0.0005 * cool + 0.00005
```

    dx = 2.0 / n
    target = n / 2.0
    def obj(x):

        m = corr.max()

                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 50})
