    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)
    mc = np.max(corr)
    # log-sum-exp
    shift = T * (corr - mc)
    shift -= np.max(shift)
    logsumexp = mc + np.log(np.exp(shift).sum()) / T
    val = dx * logsumexp + lam * (h.sum() - n/2.0)**2
    # Gradient
    w = np.exp(shift)
    w /= w.sum()
    grad = np.zeros(n)
    m_full = 2 * n - 1
    lags = np.arange(m_full) - n + 1
    for idx in range(m_full):
        lg = lags[idx]
        wi = w[idx]
        if wi < 1e-15:

        if lg >= 0:
            grad[lg:] += wi * (1.0 - h[:n-lg])
            grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            grad[:n-p] += wi * (1.0 - h[p:])
            grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n/2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    h = h.copy()
    n = len(h)
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            m = h < 1.0 - 1e-12
            if not m.any(): break
            h[m] += diff / m.sum()
        else:
            m = h > 1e-12
            if not m.any(): break
            h[m] += diff / m.sum()
    return h

def softmax_objgrad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    st = T * (corr - mc)
    st -= st.max()
    es = np.exp(st)
    esum = es.sum()
    sm = mc + np.log(esum) / T
    w = es / esum
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    g = np.zeros(n)
    m_full = 2 * n - 1
    lags = np.arange(m_full) - n + 1
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-14: continue
        lg = lags[idx]
        if lg >= 0:
            g[lg:] += wi * (1.0 - h[:n-lg])
            g[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            g[:n-p] += wi * (1.0 - h[p:])
            g[p:] += -wi * h[:n-p]
    g = dx * g + 2.0 * lam * (h.sum() - n / 2.0)
    return val, g

def topk_objgrad(h, K, lam):
    """Use top-K average instead of softmax for a different landscape"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-K:]
    sm = corr[idx].mean()
    w = np.zeros(len(corr))
    w[idx] = 1.0 / K
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    g = np.zeros(n)
    m_full = 2 * n - 1
    lags = np.arange(m_full) - n + 1
    for k in idx:
        wi = w[k]
        lg = lags[k]
        if lg >= 0:
            g[lg:] += wi * (1.0 - h[:n-lg])
            g[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            g[:n-p] += wi * (1.0 - h[p:])
            g[p:] += -wi * h[:n-p]
    g = dx * g + 2.0 * lam * (h.sum() - n / 2.0)
    return val, g

def lsg(h0, T, lam, maxiter, ftol):
    x0 = project_feasible(h0)
    n = len(x0)
    try:
        r = minimize(
            lambda x: softmax_objgrad(x, T, lam)[0], x0,
            jac=lambda x: softmax_objgrad(x, T, lam)[1],
            method='L-BFGS-B', bounds=[(0,1)]*n,
            options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-14})
        return project_feasible(r.x)
    except:
        return x0
