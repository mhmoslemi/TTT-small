from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    if s < 1e-12:
        h[:] = 0.5
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    es = np.exp(sh)
    se = es.sum()
    val = dx * (T * m + np.log(se))
    w = es / se
    g = np.zeros(n)
    for mi in range(2*n-1):
        wi = w[mi]
        if wi < 1e-10: continue
        k = mi - (n-1)
        if k == 0:
            g += wi * (1.0 - 2.0*h)
        elif k > 0:
            g[k:] += wi * (1.0 - h[:n-k])
            g[:n-k] -= wi * h[k:]
        else:
            kp = -k
            g[:n-kp] += wi * (1.0 - h[kp:])
            g[kp:] -= wi * h[:n-kp]
    g *= dx
    return val, g

def grad_step(h, T, step, n):
    val, g = smooth_val_grad(h, T)
    h_new = proj_h(h - step * g, n)
    return h_new, val

def grad_descent(h, T, n, max_iter, step, t_end):
    for _ in range(max_iter):
        if time.time() > t_end: break
        h_new, _ = grad_step(h, T, step, n)
        if np.allclose(h, h_new, atol=1e-12): break
        h = h_new
    return h

def basin_hop(h, n, t_end, rng, n_iter):
    best = h.copy()
    best_c5 = compute_c5(best)
    work = h.copy()
    work_c5 = best_c5

            if c5 < best_c5:

    dx = 2.0/n

                if c5 < best_c5 - 1e-13:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    if s < 1e-12:
        h[:] = 0.5
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    s2 = h.sum()
    if abs(s2 - target) > 1e-10:
        lo, hi = -1.0, 1.0
        for _ in range(40):
            mid = 0.5 * (lo + hi)
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    return h

def smooth_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -50.0, 50.0)
    es = np.exp(sh)
    se = es.sum()
    val = dx * (T * m + np.log(se))
    w = es / se
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = w[mi]
        if wi < 1e-10:
            continue
        k = mi - (n - 1)
        if k == 0:
            g += wi * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += wi * (1.0 - h[:n - k])
            g[:n - k] -= wi * h[k:]
        else:
            kp = -k
            g[:n - kp] += wi * (1.0 - h[kp:])
            g[kp:] -= wi * h[:n - kp]
    g *= dx
    return val, g

def gd_loop(h, T, n, max_iter, step, t_end):
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        _, g = smooth_val_grad(h, T)
        h_new = proj_h(h - step * g, n)
        if np.max(np.abs(h_new - h)) < 1e-13:
            break
        h = h_new
    return h

def basin_hop(h, n, t_end, rng, n_iter, scale=0.15):
    best = h.copy()
    best_c5 = compute_c5(best)
    work = h.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - float(it) / max(n_iter, 1)
        mod = rng.randint(3)
        p = np.zeros(n)
        if mod == 0:
            p = rng.standard_normal(n) * scale * cool
        elif mod == 1:
            blk = rng.randint(5, max(6, n // 6))
            st = rng.randint(0, n - blk)
            p[st:st + blk] = (0.3 + 0.5 * cool) * rng.standard_normal(blk)
        else:
            idx = rng.choice(n, size=rng.randint(3, n // 10 + 3), replace=False)
            p[idx] = (0.3 + 0.7 * cool) * rng.standard_normal(len(idx))
        hp = proj_h(work + p, n)
        Tv = 500.0 + 5000.0 * cool
        hp = gd_loop(hp, Tv, n, 200, 1.0 / Tv * 50.0, t_end)
        c5 = compute_c5(hp)
        temp = 0.001 * cool + 0.0001
        if c5 < work_c5 or rng.rand() < np.exp(-(c5 - work_c5) / max(temp, 1e-9)):
            work = hp
            work_c5 = c5
            if c5 < best_c5:
                best = hp.copy()
                best_c5 = c5
    return best, best_c5

def swap_search(h, n, t_end, max_rounds=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        rng_order = np.random.permutation(len(ones))
        for ii in rng_order:
            if time.time() > t_end:
                return best, best_c5
            i = ones[ii]
            for j in zeros:
                if i == j:
                    continue
                ht = best.copy()
                ht[i] = 0.0
                ht[j] = 1.0
                c5 = compute_c5(ht)
                if c5 < best_c5 - 1e-14:
                    best = ht
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def coord_descent(h, n, t_end, max_sweps=60):
    best = h.copy()
    best_c5 = compute_c5(best)
    for sw in range(max_sweps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            for val in [0.0, 1.0]:
                if abs(best[i] - val) < 1e-6:
                    continue
                ht = best.copy()
                ht[i] = val
                ht = proj_h(ht, n)
                c5 = compute_c5(ht)
                if c5 < best_c5 - 1e-13:
                    best = ht
                    best_c5 = c5
                    imp = True
        if not imp:
            break
    return best, best_c5

def multi_grad(h, n, t_end):
    best = h.copy()
    best_c5 = compute_c5(best)
    for T in [200, 800, 3000, 12000, 50000, 200000]:
        if time.time() > t_end:
            break
        step = 40.0 / T
        best = gd_loop(best, T, n, 800, step, t_end)
        c5 = compute_c5(best)
        if c5 < best_c5:
            best_c5 = c5
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    n = 160

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_h = proj_h(best_h, n)
    else:
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-T gradient (8%)
    t1 = t0 + budget_s * 0.08
    best_h, best_c5 = multi_grad(best_h, n, t1)

    # Phase 2: Basin hopping round 1 (22%)
    t2 = t0 + budget_s * 0.30
    best_h, best_c5 = basin_hop(best_h, n, t2, rng, 150, scale=0.18)

    # Phase 3: Gradient + coord descent (12%)
    t3 = t0 + budget_s * 0.42
    best_h, best_c5 = multi_grad(best_h, n, t3)
    best_h, best_c5 = coord_descent(best_h, n, t3 + 1, 40)

    # Phase 4: Swap search (12%)
    t4 = t0 + budget_s * 0.54
    best_h, best_c5 = swap_search(best_h, n, t4, 150)

    # Phase 5: Basin hopping round 2 (cooler) (20%)
    t5 = t0 + budget_s * 0.74
    best_h, best_c5 = basin_hop(best_h, n, t5, rng, 120, scale=0.10)

    # Phase 6: Swap + coord (12%)
    t6 = t0 + budget_s * 0.86
    best_h, best_c5 = swap_search(best_h, n, t6, 100)
    best_h, best_c5 = coord_descent(best_h, n, t6 + 1, 50)

    # Phase 7: Final gradient polish (8%)
    t7 = t0 + budget_s * 0.95
    best_h, best_c5 = multi_grad(best_h, n, t7)
    best_h, best_c5 = coord_descent(best_h, n, t_end - 2, 30)
    best_h, best_c5 = swap_search(best_h, n, t_end - 1, 50)

    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
