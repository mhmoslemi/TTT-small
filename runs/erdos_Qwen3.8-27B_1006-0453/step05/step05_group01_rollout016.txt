import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_c5(h, T, n_pen=5, pen_w=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp((corr - m) * T))) / T
    if pen_w > 0:
        top = np.sort(corr)[-n_pen:]
        val += pen_w * (top[-1] - top[0])
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        frac = 1.0 + diff / s
        h = h * frac
        # Clip and re-project
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        diff2 = target - s2
        if abs(diff2) > 1e-10:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h = h + room * (diff2 / tot)
            else:
                h = h * (target / max(s2, 1e-14))
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-16, pen_w=0.0):
    try:
        res = minimize(lambda x: soft_max_c5(x, T, pen_w=pen_w), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_descent_v2(h, n, t_end, max_sweeps=150, n_top=60):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in rng.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-14})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_step(h, alpha, n_topk, mom_v=None, beta=0.5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if mom_v is not None:
        grad = beta * mom_v + (1.0 - beta) * grad
    return project_h(h - alpha * grad), grad

    work_c5 = compute_c5(work_h)
    rng = np.random.RandomState(777)
    for it in range(n_iter):

            break
        cool = 1.0 - it / max(n_iter, 1)

        if mode == 'targeted' and r < 0.4:

            top_idx = np.argsort(corr)[-8:]

            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n-k] += (work_h[k:n] - 0.5)
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n+k] - 0.5)
            pert *= scale * 0.5 * cool
        elif r < 0.45:
            f = rng.randint(1, 20)
            p = rng.uniform(0, 2 * np.pi)
            pert = scale * 0.8 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.65:
            mask = rng.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 1.2 * cool * rng.randn(max(int(mask.sum()), 1))
        elif r < 0.85:
            blen = rng.randint(max(n // 15, 2), max(n // 4, 3))
            bs = rng.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs+blen] = scale * cool * rng.randn(blen)
        else:
            pert = scale * cool * rng.randn(n)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp_opt(h_pert, T, n, 300, 1e-15)
        c5v = compute_c5(hc)
        temp = 0.001 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and rng.random() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_c5(h, T, pen_w=0.0, n_pen=8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp((corr - m) * T))) / T
    if pen_w > 0:
        idx = np.argpartition(corr, -n_pen)[-n_pen:]
        val += pen_w * (corr[idx].max() - corr[idx].min())
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        d2 = target - s2
        if abs(d2) > 1e-10:
            room = 1.0 - h
            tot = room.sum()
            if tot > 1e-14:
                h += room * (d2 / tot)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=800, ftol=1e-16, pen_w=0.0):
    try:
        res = minimize(lambda x: soft_max_c5(x, T, pen_w=pen_w), h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def subgrad_step(h, alpha, n_topk, mom_v=None, beta=0.45):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_idx = np.argsort(corr)[-n_topk:]
    shifts = topk_idx - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    if mom_v is not None:
        grad = beta * mom_v + (1.0 - beta) * grad
    return project_h(h - alpha * grad), grad

def coord_desc(h, n, t_end, max_sweeps=200, n_top=60, recompute_every=10):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(42)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = top_idx - (n - 1)
        improved = False
        for j, i in enumerate(rng.permutation(n)):
            if time.time() > t_end:
                return best_h, best_c5
            if j > 0 and j % recompute_every == 0:
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                top_idx = np.argsort(corr)[-n_top:]
                top_shifts = top_idx - (n - 1)
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-14})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, scale, T_lo, T_hi, targeted_frac=0.35):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    rng = np.random.RandomState(999)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = rng.random()
        if r < targeted_frac:
            corr = np.correlate(work_h, 1.0 - work_h, mode='full')
            top_idx = np.argsort(corr)[-8:]
            shifts = top_idx - (n - 1)
            pert = np.zeros(n)
            for k in shifts:
                if k >= 0 and n - k > 0:
                    pert[:n-k] += (work_h[k:n] - 0.5)
                elif k < 0 and n + k > 0:
                    pert[-k:n] += (work_h[:n+k] - 0.5)
            pert *= scale * 0.5 * cool
        elif r < 0.5:
            f = rng.randint(1, 25)
            p = rng.uniform(0, 2 * np.pi)
            pert = scale * 0.8 * cool * np.sin(2 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.7:
            mask = rng.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = scale * 1.3 * cool * rng.randn(max(int(mask.sum()), 1))
        elif r < 0.85:
            blen = rng.randint(max(n // 12, 2), max(n // 3, 3))
            bs = rng.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs+blen] = scale * cool * rng.randn(blen)
        else:
            pert = scale * cool * rng.randn(n)
        h_pert = project_h(work_h + pert)
        T = T_lo + (T_hi - T_lo) * cool
        hc = slsqp_opt(h_pert, T, n, 350, 1e-15)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.0001
        if c5v < work_c5 or (c5v > work_c5 and rng.random() < np.exp(-(c5v - work_c5) / temp)):
            work_h = hc
            work_c5 = c5v
    return work_h, work_c5

def optimize_n(h_init, n, t_end):
    best_h = project_h(h_init)
    best_c5 = compute_c5(best_h)
    # SLSQP warmup
    if t_end - time.time() > 20:
        for T in [50, 300, 2000, 10000, 50000]:
            if t_end - time.time() < 8:
                break
            hc = slsqp_opt(best_h, T, n, 500, 1e-15)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Subgradient + SLSQP
    if t_end - time.time() > 35:
        mom_v = np.zeros(n)
        for it in range(35):
            if t_end - time.time() < 8:
                break
            cool = 1.0 - it / 35.0
            alpha = 0.004 * (0.3 + 0.7 * cool)
            n_topk = 8 + int(20 * cool)
            hs, g = subgrad_step(best_h, alpha, n_topk, mom_v, beta=0.45)
            mom_v = g
            T = 300 + 10000 * cool
            hc = slsqp_opt(hs, T, n, 400, 1e-15, pen_w=0.04 * cool)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    # Basin hopping targeted
    if t_end - time.time() > 45:
        bh_end = t_end - (t_end - time.time()) * 0.30
        bh_h, bh_c5 = basin_hop(best_h, n, bh_end, 40, 0.16, 500, 10000, 0.4)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Coordinate descent
    if t_end - time.time() > 20:
        cd_end = t_end - (t_end - time.time()) * 0.25
        best_h, best_c5 = coord_desc(best_h, n, cd_end, 150, 70, 8)
    # Fine basin hopping
    if t_end - time.time() > 20:
        bh_end = t_end - (t_end - time.time()) * 0.35
        bh_h, bh_c5 = basin_hop(best_h, n, bh_end, 25, 0.05, 3000, 20000, 0.2)
        if bh_c5 < best_c5:
            best_c5 = bh_c5
            best_h = bh_h
    # Final coord descent
    if t_end - time.time() > 12:
        best_h, best_c5 = coord_desc(best_h, n, t_end - 4, 100, 100, 6)
    # SLSQP polish
    if t_end - time.time() > 6:
        hc = slsqp_opt(best_h, 5000000, n, 1000, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0
    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)
    # Phase 1: main optimization
    t1 = t0 + budget_s * 0.48
    best_h, best_c5 = optimize_n(best_h, n, t1)
    # Phase 2: try smaller n
    if t_end - time.time() > budget_s * 0.28:
        n2 = 128
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = project_h(h2)
        t2 = t0 + budget_s * 0.72
        h2o, c52 = optimize_n(h2, n2, t2)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2o
            n = n2
    # Phase 3: try back at 200 or another size
    if t_end - time.time() > budget_s * 0.18 and n != 200:
        n3 = 200
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = project_h(h3)
        t3 = t0 + budget_s * 0.90
        h3o, c53 = optimize_n(h3, n3, t3)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3o
            n = n3
    # Final polish: multi-restart SLSQP
    if t_end - time.time() > 12:
        rng = np.random.RandomState(seed)
        for _ in range(3):
            if t_end - time.time() < 4:
                break
            h_p = project_h(best_h + 0.01 * rng.randn(n))
            hc = slsqp_opt(h_p, 8000000, n, 1200, 1e-16)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if t_end - time.time() > 8:
        best_h, best_c5 = coord_desc(best_h, n, t_end - 2, 80, 100, 5)
    if t_end - time.time() > 4:
        hc = slsqp_opt(best_h, 10000000, n, 1500, 1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
