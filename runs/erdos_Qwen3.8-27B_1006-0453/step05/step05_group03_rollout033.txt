    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h, 0.0, 1.0)

    if abs(diff) < 1e-14:

    if diff > 0:

        tot = room.sum()

            return h
        h = h + room * (diff / tot)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=300, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -500, 500)
        logits = corr - m
        # Use log-sum-exp trick
        max_logit = np.max(logits)
        lse = max_logit + np.log(np.sum(np.exp(logits - max_logit)))
        return (m + lse / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h

def pair_coord_opt(h, n, i, j):
    """Optimize h[i] and h[j] jointly, maintaining sum constraint."""
    # h[i] + h[j] = s fixed
    s = h[i] + h[j]
    h[i] = 0; h[j] = 0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    coef_i = np.zeros(2*n-1)
    coef_j = np.zeros(2*n-1)
    lags = np.arange(-(n-1), n)
    for idx, pos in [(i, 0), (j, 1)]:
        coefs = coef_i if pos == 0 else coef_j
        jfwd = idx + lags
        jback = idx - lags
        vf = (jfwd >= 0) & (jfwd < n)
        vb = (jback >= 0) & (jback < n)
        coefs[vf] += (1.0 - h[np.clip(jfwd[vf], 0, n-1)])
        coefs[vb] += (-h[np.clip(jback[vb], 0, n-1)])
    
    def f(t):
        # h[i] = t, h[j] = s - t
        # valid: 0 <= t <= 1, 0 <= s-t <= 1
        # corr = corr_base + t*coef_i + (s-t)*coef_j = corr_base + s*coef_j + t*(coef_i - coef_j)
        return np.max(corr_base + s * coef_j + t * (coef_i - coef_j))
    
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-15:
        return h
    res = minimize_scalar(f, bounds=(lo, hi), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 50})
    if res.fun < 1e300:
        h[i] = res.x
        h[j] = s - res.x
    return h

def coord_descent_pairs(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for ii in range(0, n-1, 2):
            i = order[ii]
            j = order[ii+1]
            if time.time() > t_end:
                return best_h, best_c5
            h_try = best_h.copy()
            h_try = pair_coord_opt(h_try, n, int(i), int(j))
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        ms = np.max(sh)
        lse = m + ms + np.log(np.sum(np.exp(sh - ms))) / T
        return lse * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_h(np.clip(res.x, 0, 1), n)
    except:
        return h

def pair_opt(h, n, i, j):
    s = h[i] + h[j]
    h[i] = 0.0
    h[j] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    lags = np.arange(-(n - 1), n)
    ci = np.zeros(2 * n - 1)
    cj = np.zeros(2 * n - 1)
    for pos, co in [(i, ci), (j, cj)]:
        jf = pos + lags
        jb = pos - lags
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        co[vf] += (1.0 - h[jf[vf]])
        co[vb] += (-h[jb[vb]])
    delta = ci - cj
    base = corr_base + s * cj
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-15:
        h[i] = s * 0.5
        h[j] = s * 0.5
        return h
    def f(t):
        return np.max(base + t * delta)
    res = minimize_scalar(f, bounds=(lo, hi), method='bounded',
                          options={'xatol': 1e-11, 'maxiter': 50})
    h[i] = res.x
    h[j] = s - res.x
    return h

def pair_cd(h, n, t_end, max_sweeps=60):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for ii in range(0, n - 1, 2):
            if time.time() > t_end:
                return best_h, best_c5
            i, j = int(order[ii]), int(order[ii + 1])
            h_try = best_h.copy()
            pair_opt(h_try, n, i, j)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def single_cd(h, n, t_end, max_sweeps=40):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lags = np.arange(-(n - 1), n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            i = int(i)
            h_try = best_h.copy()
            h_try[i] = 0.0
            corr_base = np.correlate(h_try, 1.0 - h_try, mode='full')
            co = np.zeros(2 * n - 1)
            jf = i + lags
            jb = i - lags
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            co[vf] += (1.0 - h_try[jf[vf]])
            co[vb] += (-h_try[jb[vb]])
            def f(t):
                return np.max(corr_base + t * co)
            res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 40})
            h_try[i] = res.x
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish from start
    for T in [5000, 1500, 500, 150, 50]:
        if elapsed() > budget_s * 0.06:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=200)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Basin hopping - main search
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter_done = 0
    for it in range(150):
        if elapsed() > budget_s * 0.50:
            break
        cool = max(0.05, 1.0 - it / 150.0)
        rt = np.random.rand()
        if rt < 0.20:
            amp = 0.008 + 0.05 * cool
            h_pert = project_h(work_h + amp * np.random.randn(n), n)
        elif rt < 0.35:
            blen = np.random.randint(3, 25)
            bstart = np.random.randint(0, max(1, n - blen))
            h_pert = work_h.copy()
            h_pert[bstart:bstart + blen] = 1.0 - h_pert[bstart:bstart + blen]
            h_pert = project_h(h_pert, n)
        elif rt < 0.50:
            freq = np.random.randint(1, 20)
            amp = 0.008 + 0.03 * cool
            phase = np.random.rand() * 2 * np.pi
            sig = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            h_pert = project_h(work_h + sig, n)
        elif rt < 0.65:
            nf = np.random.randint(2, 12)
            idx = np.random.choice(n, nf, replace=False)
            h_pert = work_h.copy()
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = project_h(h_pert, n)
        elif rt < 0.80:
            # Gaussian with heavier tail
            amp = 0.005 + 0.03 * cool
            noise = np.random.standard_t(4, size=n)
            h_pert = project_h(work_h + amp * noise, n)
        else:
            amp = 0.01 + 0.04 * cool
            h_pert = project_h(work_h + amp * np.random.randn(n), n)

        T = 200 + 6000 * cool
        hc = slsqp_smooth(h_pert, T, n, maxiter=250)
        c5v = compute_c5(hc)
        temp = 0.0005 * cool + 0.00002
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
        n_iter_done += 1

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 3: Pair coordinate descent
    best_h, best_c5 = pair_cd(best_h, n, t0 + budget_s * 0.67, max_sweeps=50)

    # Phase 4: Finer basin hopping
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(50):
        if elapsed() > budget_s * 0.82:
            break
        cool = max(0.1, 1.0 - it / 50.0)
        amp = 0.004 + 0.015 * cool
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        T = 400 + 3000 * cool
        hc = slsqp_smooth(h_pert, T, n, maxiter=250)
        c5v = compute_c5(hc)
        temp = 0.0003 * cool + 0.00001
        if c5v < work_c5 or np.random.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Single coordinate descent (finer)
    best_h, best_c5 = single_cd(best_h, n, t0 + budget_s * 0.90, max_sweeps=30)

    # Phase 6: Final pair CD
    best_h, best_c5 = pair_cd(best_h, n, t0 + budget_s * 0.95, max_sweeps=30)

    # Phase 7: Final SLSQP polish
    for T in [100, 30, 10]:
        if elapsed() > budget_s * 0.99:
            break
        hc = slsqp_smooth(best_h, T, n, maxiter=350)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
