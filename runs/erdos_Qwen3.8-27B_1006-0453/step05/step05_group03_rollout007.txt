import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_c5(h, T):
    """Smooth approximation of max using log-sum-exp."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    sh = np.clip(T * (corr - m), -100, 100)
    return (m + np.log(np.sum(np.exp(sh))) / T) * dx

def slsqp_smooth(h, T, n, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_opt_1d_fast(h, n, i):
    """Optimize coordinate i using piecewise-linear structure."""
    old = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    
    # Compute coefficient for each correlation element
    # corr[k] = sum_j h[j] * (1-h[j+k-n+1]) for valid j
    # dh[i]/dh[i] contribution:
    # For the term where h[i] is the first factor: (1 - h[i + k - n + 1])
    # For the term where h[i] is in (1-h[...]): -h[i - k + n - 1]
    coefs = np.zeros(2 * n - 1)
    lags = np.arange(-(n-1), n)  # actual lags
    # For element at index k (lag = k - n + 1):
    # h[i] contributes (1 - h[i + (k-n+1)]) if i + (k-n+1) in [0,n)
    # and -h[i - (k-n+1)] if i - (k-n+1) in [0,n)
    
    # Simpler: just compute numerically
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    coefs = c1 - c0  # change per unit of h[i]
    
    # Now minimize max(c0 + t*coefs) for t in [0,1]
    vals = c0 + np.outer([0.0, 1.0], coefs)
    # Actually, f(t) = max_k(c0[k] + t*coefs[k])
    # This is convex in t (max of linear functions)
    res = minimize_scalar(lambda t: np.max(c0 + t * coefs), 
                          bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    return res.x

def coord_descent(h, n, t_end, max_sweeps=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_1d_fast(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-12:

            h_try = best_h.copy()
            h_try[i] = hi_new

            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-15:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-15:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, T, n, maxiter=500):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(T * (corr - m), -100, 100)
        dx = 2.0 / n
        return (m + np.log(np.sum(np.exp(sh))) / T) * dx
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

    """Optimize single coordinate i using exact 1D minimization."""
