    corr = get_corr(h)
    S = corr.sum()
    m = corr.max()
    # Find active set
    thresh = m - (corr.max() - corr.min()) * active_frac
    active = corr >= thresh
    if not active.any():
        active = np.zeros(len(corr), bool)
        active[corr.argmax()] = True
    # Compute weighted gradient
    w = corr[active] / corr[active].sum()
    grad = np.zeros(n)
    for (idx, wi) in zip(np.where(active)[0], w):

    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals

def active_grad(h, n, n_active=5):
    corr = get_corr(h)
    m = corr.max()
    # Get top n_active
    idx = np.argsort(corr)[::-1][:n_active]
    weights = np.ones(n_active) / n_active
    grad = np.zeros(n)
    for s_idx, w in zip(idx, weights):
        s = s_idx - (n - 1)
        # grad_s[j] = (1-h[j-s])*[0<=j-s<n] - h[j+s]*[0<=j+s<n]
        lo1 = max(0, s)
        hi1 = min(n, s + n)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 + s:hi2 + s]
    return grad * (2.0 / n), corr

def projected_gradient(h, n, lr, n_active=5, n_steps=200, t_end=None):

    best_c5 = compute_c5(best_h)
    for step in range(n_steps):
        if t_end and time.time() > t_end:
            break
        grad, corr = active_grad(best_h, n, n_active)
        lr_eff = lr / (1 + step * 0.01)
        h_new = np.clip(best_h - lr_eff * grad, 0, 1)
        h_new = project_h(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

def coord_sweep(h, n, t_end, max_sweeps=200, niter_bins=50):

    best_c5 = compute_c5(best_h)
    improved_any = True
    sweep = 0
    while improved_any and sweep < max_sweeps:

        improved_any = False

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt_fast(h, n, i, niter=45):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        mid = (lo + hi) / 2.0
        vals = corr_base + mid * coefs
        m_val = vals.max()
        active = vals >= m_val - 1e-12
        slopes = coefs[active]
        if slopes.min() > 1e-12:
            hi = mid
        elif slopes.max() < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def coord_sweep(h, n, t_end, max_sweeps=300, niter_bins=45, shake_int=15):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_impr = 0
    sweep = 0
    while sweep < max_sweeps:
        if time.time() > t_end:
            break
        sweep += 1
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            t_opt = coord_opt_fast(best_h, n, i, niter_bins)
            if abs(t_opt - best_h[i]) > 1e-11:
                h_try = best_h.copy()
                h_try[i] = t_opt
                h_try = project_h(h_try)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
                    improved = True
        if improved:
            no_impr = 0
        else:
            no_impr += 1
            if no_impr >= shake_int:
                best_h = project_h(best_h + 0.008 * np.random.randn(n))
                best_c5 = compute_c5(best_h)
                no_impr = 0
    return best_h, best_c5

def smooth_c5(h, T):
    corr = get_corr(h)
    m = corr.max()
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    return val * (2.0 / n)

def smooth_c5_grad(h, T):
    corr = get_corr(h)
    m = corr.max()
    w = np.exp(T * (corr - m))
    w /= w.sum()
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        s = idx - (n - 1)
        lo1, hi1 = max(0, s), min(n, s + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 - s:hi1 - s])
        lo2, hi2 = max(0, -s), min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 + s:hi2 + s]
    return grad * (2.0 / n), smooth_c5(h, T)

def slsqp_pass(h, n, T, maxiter=2000, t_end=None):
    def obj(x):
        return smooth_c5(x, T)
    def jac(x):
        g, _ = smooth_c5_grad(x, T)
        return g
    def con(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if t_end and time.time() > t_end:
            return h
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def basin_hop(h, n, best_c5, t_end, n_iter=50):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(0.1, 1.0 - 0.85 * (it / n_iter))
        mod = it % 6
        if mod == 0:
            pert = (0.02 + 0.06 * cool) * np.random.randn(n)
        elif mod == 1:
            freq = np.random.randint(1, 12)
            ph = np.random.uniform(0, 6.28)
            pert = (0.015 + 0.05 * cool) * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = (0.03 + 0.10 * cool) * np.random.randn(max(1, mask.sum()))
        elif mod == 3:
            blk = max(1, n // 8)
            b = np.random.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            pert[b*blk:min(n, (b+1)*blk)] = (0.02 + 0.04 * cool) * np.random.randn(blk)
        elif mod == 4:
            pert = (0.008 + 0.025 * cool) * np.random.randn(n)
        else:
            mask = np.random.random(n) < 0.06
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.12 * cool) * np.sign(np.random.randn(max(1, mask.sum())))
        h_pert = project_h(work_h + pert)
        h_loc, c5_loc = coord_sweep(h_pert, n, t_end, max_sweeps=12, niter_bins=35, shake_int=10)
        temp = 0.0005 * cool + 0.00003
        if c5_loc < work_c5 or np.random.rand() < np.exp(-(c5_loc - work_c5) / temp):
            work_h = h_loc
            work_c5 = c5_loc
            if c5_loc < best_c5:
                best_c5 = c5_loc
                h = h_loc
        if time.time() > t_end * 0.98:
            break
    if work_c5 < best_c5:
        best_c5 = work_c5
        h = work_h
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 160
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with smoothing to reshape
    for T in [200, 2000, 20000]:
        if time.time() - t0 > budget_s * 0.12:
            break
        hc = slsqp_pass(best_h, n, T, maxiter=1200, t_end=t0 + budget_s * 0.12)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Heavy coordinate descent
    best_h, best_c5 = coord_sweep(best_h, n, t0 + budget_s * 0.38, max_sweeps=250, niter_bins=45, shake_int=12)

    # Phase 3: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, best_c5, t0 + budget_s * 0.65, n_iter=60)

    # Phase 4: SLSQP high-T smoothing
    for T in [50000, 200000, 800000]:
        if time.time() - t0 > budget_s * 0.80:
            break
        hc = slsqp_pass(best_h, n, T, maxiter=1500, t_end=t0 + budget_s * 0.80)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 5: Final coordinate descent (many sweeps, tighter)
    best_h, best_c5 = coord_sweep(best_h, n, t0 + budget_s * 0.96, max_sweeps=400, niter_bins=55, shake_int=25)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
