def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    """Project h to [0,1]^n with sum = n/2"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h, 0, 1)
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return np.clip(h, 0, 1)
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, T, top_k=-1):
    """Compute smooth max of correlations and gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

        mc = corr_sub.max()
        Tc = T * (corr_sub - mc)
        Tc = Tc - Tc.max()
        exps = np.exp(Tc)

        sm = mc + np.log(exps.sum()) / T

    return dx * grad

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_corr(h, corr, w):
    """Gradient of weighted sum of correlations w.r.t. h"""
    n = len(h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    grad = np.zeros(n)
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-14:

        if lg == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lg > 0:
            if lg < n:

                grad[lg:] += wi * (1.0 - h[:n-lg])
        else:
            p = -lg
            if p < n:

                grad[p:] += -wi * h[:n-p]

    for T in T_seq:
        for topk in ([top_k] if top_k > 0 else [0]):
            while time.time() < t_end:
                corr = get_corr(h)
                mc = corr.max()
                if topk > 0 and topk < 2*n-1:
                    idx = np.argpartition(corr, -topk)[-topk:]
                    cs = corr[idx]

        Tc -= Tc.max()

        w = np.zeros(len(corr))
        w[idx] = ws
    else:
        Tc = T * (corr - mc)
        Tc -= Tc.max()
        exps = np.exp(Tc)
        w = exps / exps.sum()
    return w
