import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, T, lam, top_k=0):
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k > 0 and top_k < m_full:
        top_idx = np.argsort(corr)[-top_k:]
        corr_use = corr[top_idx]
        idx_list = top_idx
    else:
        corr_use = corr
        idx_list = np.arange(m_full)
    mc = np.max(corr_use)
    shifted = T * (corr_use - mc)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    sm = mc + np.log(np.sum(exps)) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    lag_arr = np.arange(m_full) - (n - 1)
    for j, idx in enumerate(idx_list):
        wi = w[j]
        if wi < 1e-14:
            continue
        lg = lag_arr[idx]
        if lg >= 0:
            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])
                grad[:n-lg] += -wi * h[lg:]
        else:
            p = -lg
            if p < n:
                grad[:n-p] += wi * (1.0 - h[p:])
                grad[p:] += -wi * h[:n-p]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfgs_opt(h0, T, lam, maxiter, ftol, top_k=0):
    n = len(h0)
    x0 = project_h(h0)
    def obj(x):
        return smooth_obj_grad(x, T, lam, top_k)[0]
    def grad(x):
        return smooth_obj_grad(x, T, lam, top_k)[1]
    try:
        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(res.x)
    except Exception:
        return x0

def coord_descent(h, n, t_end, max_sweeps):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            h0 = best_h.copy()
            h0[i] = 0.0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                lg = lag_arr[idx]
                if lg == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[idx] += (1.0 - best_h[i - lg])
                    if i + lg < n:
                        lin[idx] += -best_h[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[idx] += (1.0 - best_h[i + p])
                    if i >= p:
                        lin[idx] += -best_h[i - p]
            def f1d(x):
                return np.max(cb + x * lin + x * x * quad)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-9})
            hi_new = res.x
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            ht = best_h.copy()
            ht[i] = hi_new
            ht = project_h(ht)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht
                improved = True
        if not improved:
            break
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def sg_obj(h, T, lam, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_full = 2 * n - 1
    if top_k > 0 and top_k < m_full:
        idx = np.argsort(corr)[-top_k:]
        cu = corr[idx]
    else:
        idx = np.arange(m_full)
        cu = corr
    mc = cu.max()
    sh = T * (cu - mc)
    sh -= sh.max()
    ex = np.exp(sh)
    w = ex / ex.sum()
    sm = mc + np.log(ex.sum()) / T
    return dx * sm + lam * (h.sum() - n/2.0)**2, w, idx

def sg_grad(h, T, lam, top_k):
    n = len(h)
    dx = 2.0 / n
    val, w, idx = sg_obj(h, T, lam, top_k)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    grad = np.zeros(n)
    for j, ii in enumerate(idx):
        wi = w[j]
        if wi < 1e-14:
            continue
        lg = lag_arr[ii]

            if lg < n:
                grad[lg:] += wi * (1.0 - h[:n-lg])

            grad[p:] += -wi * h[:n-p]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - n/2.0)
    return val, grad

                     bounds=[(0,1)]*n,
                     options={'maxiter':iters,'ftol':ftol,'gtol':1e-10})
        return project_h(r.x)
    except:
        return x0

    m_full = 2*n-1

        return sg_obj_grad(x, T, lam, top_k)[0]
    def g(x):
        return sg_obj_grad(x, T, lam, top_k)[1]
    try:
        r = minimize(f, x0, jac=g, method='L-BFGS-B',
                     bounds=[(0, 1)] * n,
                     options={'maxiter': iters, 'ftol': ftol, 'gtol': 1e-10})
        return project_h(r.x)
    except Exception:
        return x0

def coord_desc(h, n, t_end, sweeps):
    bh = h.copy()
    bc = compute_c5(bh)
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    for _ in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            h0 = bh.copy()
            h0[i] = 0
            cb = np.correlate(h0, 1.0 - h0, mode='full')
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for ii in range(m_full):
                lg = lag_arr[ii]
                if lg == 0:
                    lin[ii] = 1.0
                    quad[ii] = -1.0
                elif lg > 0:
                    if i >= lg:
                        lin[ii] += 1.0 - bh[i - lg]
                    if i + lg < n:
                        lin[ii] += -bh[i + lg]
                else:
                    p = -lg
                    if i + p < n:
                        lin[ii] += 1.0 - bh[i + p]
                    if i >= p:
                        lin[ii] += -bh[i - p]
            def f1(x):
                return np.max(cb + x * lin + x * x * quad)
            r = minimize_scalar(f1, bounds=(0, 1), method='bounded', options={'xatol': 1e-9})
            if abs(r.x - bh[i]) < 1e-13:
                continue
            ht = bh.copy()
            ht[i] = r.x
            ht = project_h(ht)
            ct = compute_c5(ht)
            if ct < bc - 1e-15:
                bc = ct
                bh = ht
                imp = True
        if not imp:
            break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Top-k focused L-BFGS (10%)
    pe = t0 + budget_s * 0.10
    for T in [80, 300, 1200]:
        if time.time() > pe:
            break
        hc = lbfgs(best_h, T, 200, 3000, 1e-16, top_k=8)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 2: Full smooth L-BFGS temperature sweep (20%)
    pe = t0 + budget_s * 0.30
    for T in [500, 2000, 8000, 30000]:
        if time.time() > pe:
            break
        hc = lbfgs(best_h, T, 300, 4000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 3: Multi-restart diversification (25%)
    pe = t0 + budget_s * 0.55
    cands = []
    hb = np.zeros(n); hb[:n // 2] = 1; cands.append(project_h(hb))
    ha = np.zeros(n); ha[::2] = 1; cands.append(project_h(ha))
    for fr in [2, 3, 5, 7, 11, 16, 25]:
        hp = 0.5 + 0.45 * np.sin(2 * np.pi * fr * np.arange(n) / n)
        cands.append(project_h(hp))
    for _ in range(6):
        cands.append(project_h(np.random.beta(4, 4, n)))
    for sc in [0.03, 0.08, 0.15, 0.3]:
        cands.append(project_h(best_h + sc * np.random.randn(n)))
    for ci, hini in enumerate(cands):
        if time.time() > pe:
            break
        Tr = [1500, 5000, 15000][ci % 3]
        tk = 8 if ci < 10 else 0
        hc = lbfgs(hini, Tr, 300, 2000, 1e-16, top_k=tk)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    # Phase 4: Coordinate descent (20%)
    pe = t0 + budget_s * 0.75
    best_h, best_c5 = coord_desc(best_h, n, pe, 600)

    # Phase 5: High-T refinement + basin hopping (15%)
    pe = t0 + budget_s * 0.90
    for T in [50000, 200000]:
        if time.time() > pe:
            break
        hc = lbfgs(best_h, T, 500, 4000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc
    wh, wc = best_h.copy(), best_c5
    for it in range(60):
        if time.time() > pe:
            break
        cool = max(0.05, 1.0 - 0.95 * (it / 60.0))
        sc = 0.01 + 0.1 * cool
        hp = project_h(wh + sc * np.random.randn(n))
        Tb = int(1000 * cool + 500)
        hc = lbfgs(hp, Tb, 300, 1000, 1e-16, top_k=8)
        c5v = compute_c5(hc)
        temp = 0.0008 * cool + 0.0001
        if c5v < wc or np.random.rand() < np.exp(-(c5v - wc) / temp):
            wh, wc = hc, c5v
            if c5v < best_c5:
                best_c5, best_h = c5v, hc

    # Phase 6: Final polish
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.97, 300)
    if time.time() < t0 + budget_s * 0.995:
        hc = lbfgs(best_h, 500000, 500, 3000, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5, best_h = c5v, hc

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
