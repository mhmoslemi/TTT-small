    """Project h onto {x : 0<=x_i<=1, sum(x)=target}."""
    h = np.clip(h, 0, 1)
    s = h.sum()

    lo, hi = -1.0, 1.0

        grad = np.zeros(N)

import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, rng):
    target = len(h) // 2
    h = (h >= 0.5).astype(float)
    n1 = int(h.sum())
    if n1 > target:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, n1 - target, replace=False)
        h[idx] = 0
    elif n1 < target:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, target - n1, replace=False)
        h[idx] = 1
    return h

def compute_sig(h, N):
    s_vals = np.arange(-(N - 1), N)
    i_vals = np.arange(N)
    idx_p = i_vals[:, None] + s_vals[None, :]
    idx_m = i_vals[:, None] - s_vals[None, :]
    vp = (idx_p >= 0) & (idx_p < N)
    vm = (idx_m >= 0) & (idx_m < N)
    hp = np.where(vp, h[np.clip(idx_p, 0, N - 1)], 0.0)
    hm = np.where(vm, h[np.clip(idx_m, 0, N - 1)], 0.0)
    loss = np.where(vp, 1.0 - hp, 0.0)
    return hm - loss

def targeted_search(h, N, deadline):
    h = make_binary(h, np.random)
    best_c = float(fft_corr(h).max())
    center = N - 1
    T = 0.5
    for it in range(50000):

        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
            T = max(0.05, T * 0.99)
        if P <= N * 0.189:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        sig = compute_sig(h, N)
        worst_k = int(np.argmax(c))
        worst_s = worst_k - center
        # Find positions contributing to worst shift
        if worst_s > 0:
            ci = np.arange(N - worst_s)
        elif worst_s < 0:
            ci = np.arange(-worst_s, N)
        else:
            ci = np.arange(N)
        contrib = ci[h[ci] == 1]
        found = False
        # Try targeted swaps from contributing positions
        if len(contrib) > 0:
            np.random.shuffle(contrib)
            for p in contrib[:25]:
                base = c + sig[p]
                delta = base[None, :] - sig[zeros]
                nm = delta.max(axis=1)
                mi = int(np.argmin(nm))
                if nm[mi] < P:
                    h[p] = 0; h[zeros[mi]] = 1

        if not found and len(ones) > 0 and len(zeros) > 0:

        c = fft_corr(h)
        P = float(c.max())
        if P < best_c:
            best_c = P
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        sig = compute_sig(h, N)
        sp = ones if len(ones) <= 200 else np.random.choice(ones, 200, replace=False)
        sq = zeros if len(zeros) <= 200 else np.random.choice(zeros, 200, replace=False)
        best_new_P = P
        best_pq = None
        for p in sp:
            base = c + sig[p]
            delta = base[None, :] - sig[sq]
            nm = delta.max(axis=1)
            mi = int(np.argmin(nm))
            if nm[mi] < best_new_P:
                best_new_P = float(nm[mi])
                best_pq = (int(p), int(sq[mi]))
        if best_pq is not None:
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
        else:
            # 2-swap attempt
            if len(ones) >= 2 and len(zeros) >= 2:
                improved = False
                for _ in range(15):
                    p1, p2 = np.random.choice(ones, 2, replace=False)
                    q1, q2 = np.random.choice(zeros, 2, replace=False)
                    nh = h.copy()
                    nh[p1] = 0; nh[p2] = 0; nh[q1] = 1; nh[q2] = 1
                    nP = float(fft_corr(nh).max())
                    if nP < P:
                        h = nh
                        improved = True
                        break
                if improved:
                    continue
            p = int(np.random.choice(ones))
            q = int(np.random.choice(zeros))
            h[p], h[q] = 0.0, 1.0
            if sweep > 0:
                break
    return h, best_c

def adam_cont(h, N, deadline, k=25, lr0=0.04):
    h = h.astype(float).copy()
    h = project_sum(h, N / 2.0)
    best_h = h.copy()
    best_val = float(fft_corr(h).max())
    m = np.zeros(N); v = np.zeros(N)
    b1, b2, eps = 0.85, 0.999, 1e-8
    center = N - 1
    t = 0; lr = lr0
    while time.time() < deadline:
        t += 1
        corr = fft_corr(h)
        val = float(corr.max())
        if val < best_val:
            best_val = val
            best_h = h.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        for ki in top:
            s = ki - center
            if s == 0:
                grad += (1.0 - 2.0 * h)
            elif s > 0:
                ve = N - s
                grad[:ve] += (1.0 - h[s:])
                grad[s:] -= h[:ve]
            else:
                sn = -s
                grad[:N - sn] -= h[sn:]
                grad[sn:] += (1.0 - h[:N - sn])
        grad /= k
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = project_sum(h - lr * mh / (np.sqrt(vh) + eps), N / 2.0)
        if t % 400 == 0:
            lr *= 0.96
        if t % 600 == 0:
            k = min(40, k + 5)
    return best_h, best_val * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)
    # Phase 1: Targeted binary search
    d1 = min(deadline - 350, t0 + 250)
    h1 = (h0 > 0.5).astype(float)
    h1, c1 = targeted_search(h1, N0, d1)
    c1s = c1 * (2.0 / N0)
    if c1s < best_c5:
        best_c5 = c1s
        best_h = h1.copy()
    # Phase 2: Full binary polish
    if time.time() < deadline - 250:
        d2 = min(deadline - 200, time.time() + 120)
        h2, c2 = full_binary_search(best_h, N0, d2, max_sweeps=2000)
        c2s = c2 * (2.0 / N0)
        if c2s < best_c5:
            best_c5 = c2s
            best_h = h2.copy()
    # Phase 3: Continuous Adam
    if time.time() < deadline - 150:
        d3 = min(deadline - 100, time.time() + 80)
        hc = best_h + rng.randn(len(best_h)) * 0.05
        hc = project_sum(hc, len(best_h) / 2.0)
        hc, chc = adam_cont(hc, len(best_h), d3, k=25, lr0=0.04)
        if chc < best_c5:
            best_c5 = chc
            best_h = hc.copy()
    # Phase 4: Binary after continuous
    if time.time() < deadline - 100:
        d4 = min(deadline - 60, time.time() + 80)
        h4, c4 = targeted_search(best_h, len(best_h), d4)
        c4s = c4 * (2.0 / len(best_h))
        if c4s < best_c5:
            best_c5 = c4s
            best_h = h4.copy()
    # Phase 5: Multi-resolution
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for Nt in [N0 - 20, N0 + 20, N0 + 40, 440, 520]:
        if time.time() > deadline - 40:
            break
        if Nt < 100:
            continue
        if Nt % 2 != 0:
            Nt += 1
        x_new = np.linspace(0, 2, Nt, endpoint=False)
        hi = np.interp(x_new, x_best, best_h)
        hbt = (hi > 0.5).astype(float)
        d_sub = min(deadline - 15, time.time() + 55)
        hbt, cbt = targeted_search(hbt, Nt, d_sub)
        cbts = cbt * (2.0 / Nt)
        if cbts < best_c5:
            best_c5 = cbts
            best_h = hbt.copy()
            x_best = np.linspace(0, 2, Nt, endpoint=False)
    # Phase 6: Final polish
    if time.time() < deadline - 3:
        hF = (best_h > 0.5).astype(float)
        hF, cF = full_binary_search(hF, len(best_h), deadline - 1, max_sweeps=300)
        cFs = cF * (2.0 / len(best_h))
        if cFs < best_c5:
            best_c5 = cFs
            best_h = hF
    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
