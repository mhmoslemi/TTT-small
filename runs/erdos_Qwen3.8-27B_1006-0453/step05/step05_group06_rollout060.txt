import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h_wf(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    eT = T * (corr - mc)
    eT -= np.max(eT)
    es = np.exp(eT)
    sm = mc + np.log(es.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    w = es / es.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        lg = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-14:
            continue
        if lg == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lg > 0:
            grad[:n - lg] += wi * (1.0 - h[lg:])
            grad[lg:] += -wi * h[:n - lg]
        else:
            p = -lg
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] += -wi * h[p:]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def _obj_fn(x, T, lam):
    return smooth_max_obj_grad(x, T, lam)[0]

def _grad_fn(x, T, lam):
    return smooth_max_obj_grad(x, T, lam)[1]

def lbfgs_smooth(h0, T, lam, maxiter=3000, ftol=1e-17):
    n = len(h0)
    x0 = project_h_wf(h0)
    try:
        res = minimize(_obj_fn, x0, args=(T, lam), jac=_grad_fn,
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_h_wf(res.x)
    except Exception:
        return x0

def _eval_coord(t, corr, delta, idx0, bq):
    val = corr + delta * t
    val[idx0] = corr[idx0] + bq * t - t * t
    return np.max(val)

def _coord_minimize(corr, delta, idx0, bq, t_min, t_max):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_min, t_max
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = _eval_coord(c, corr, delta, idx0, bq)
    fd = _eval_coord(d, corr, delta, idx0, bq)
    for _ in range(60):
        if b - a < 1e-11:
            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = _eval_coord(c, corr, delta, idx0, bq)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = _eval_coord(d, corr, delta, idx0, bq)
    return (a + b) / 2.0

def fast_coord_descent(h, n, t_end, max_sweeps=2000):
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
    m = 2 * n - 1
    idx0 = n - 1
    best_c5 = float(np.max(corr) * 2.0 / n)
    recalc_interval = 20
    sweeps = 0
    while time.time() < t_end:
        sweeps += 1
        improved = False
        order = np.random.permutation(n)
        for ci in range(len(order)):
            if time.time() > t_end:
                break
            i = order[ci]
            old_x = best_h[i]
            if old_x < 1e-14 or old_x > 1.0 - 1e-14:
                continue
            delta = np.zeros(m)
            for idx in range(m):
                if idx == idx0:
                    continue
                L = idx - (n - 1)
                c = 0.0
                ipL = i + L
                iL = i - L
                if 0 <= ipL < n:
                    c += 1.0 - best_h[ipL]
                if 0 <= iL < n:
                    c -= best_h[iL]
                delta[idx] = c
            bq = 1.0 - 2.0 * old_x
            t_min = max(-old_x, -1.0) + 1e-12
            t_max = min(1.0 - old_x, 1.0) - 1e-12
            if t_max - t_min < 1e-13:
                continue
            t_best = _coord_minimize(corr, delta, idx0, bq, t_min, t_max)
            new_x = old_x + t_best
            if abs(t_best) < 1e-12:
                continue
            corr_new0 = corr[idx0] + bq * t_best - t_best * t_best
            new_c5 = max(corr_new0 * 2.0 / n,
                         np.max(corr + delta * t_best) * 2.0 / n)
            if new_c5 < best_c5 - 1e-14:
                corr[idx0] = corr_new0
                for idx in range(m):
                    if idx != idx0:
                        corr[idx] += delta[idx] * t_best
                best_c5 = new_c5
                best_h[i] = new_x
                improved = True
        if sweeps % recalc_interval == 0:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
            best_c5 = float(np.max(corr) * 2.0 / n)
        if not improved and sweeps > 5:
            break
    best_h = project_h_wf(best_h)
    return best_h, compute_c5(best_h)
