from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:

    else:

            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

    return np.max(base + t * coefs)

def coord_descent(h, n, t_end, max_sweeps=80, n_top=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    total_shifts = 2 * n - 1
    if n_top is None or n_top >= total_shifts:
        n_top = total_shifts
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break

        if n_top < total_shifts:
            top_idx = np.argsort(corr)[-n_top:]
        else:
            top_idx = np.arange(total_shifts)
        top_shifts = top_idx - (n - 1)

                return best_h, best_c5
            jp = i + top_shifts
            jm = i - top_shifts

            base = corr[top_idx] - best_h[i] * coefs

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def f1(t, base, coefs):
    return np.max(base + t * coefs)

def coord_descent(h, n, t_end, max_sweeps=80, n_top=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    total_shifts = 2 * n - 1
    if n_top is None or n_top >= total_shifts:
        n_top = total_shifts
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        if n_top < total_shifts:
            top_idx = np.argsort(corr)[-n_top:]
        else:
            top_idx = np.arange(total_shifts)
        top_shifts = top_idx - (n - 1)
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best_h, best_c5
            jp = i + top_shifts
            jm = i - top_shifts
            v1 = (jp >= 0) & (jp < n)
            v2 = (jm >= 0) & (jm < n)
            jpc = np.clip(jp, 0, n - 1)
            jmc = np.clip(jm, 0, n - 1)
            coefs = (1.0 - best_h[jpc]) * v1.astype(np.float64) - best_h[jmc] * v2.astype(np.float64)
            base = corr[top_idx] - best_h[i] * coefs
            res = minimize_scalar(f1, args=(base, coefs), bounds=(0.0, 1.0),
                                  method='bounded', options={'xatol': 1e-10})
            t_new = res.x
            if abs(t_new - best_h[i]) < 1e-11:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter=30, scale=0.07):
    work_h = h.copy()
    work_c5 = compute_c5(work_h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        r = np.random.random()
        if r < 0.25:
            f = np.random.randint(1, 12)
            p = np.random.uniform(0, 2.0 * np.pi)
            pert = scale * cool * np.sin(2.0 * np.pi * f * np.arange(n) / n + p)
        elif r < 0.45:
            mask = np.random.random(n) < 0.15
            pert = np.zeros(n)
            pert[mask] = scale * 2.0 * cool * np.random.randn(max(int(mask.sum()), 1))
        elif r < 0.65:
            blen = np.random.randint(max(n // 8, 2), max(n // 3, 3))
            bs = np.random.randint(0, max(n - blen, 1))
            pert = np.zeros(n)
            pert[bs:bs + blen] = scale * 1.5 * cool * np.random.randn(blen)
        else:
            pert = scale * cool * np.random.randn(n)
        h_pert = project_h(work_h + pert)
        h_new, c5_new = coord_descent(h_pert, n, min(t_end, time.time() + budget_s * 0.04), 8, n_top=60)
        temp = 0.001 * cool + 0.0002
        if c5_new < work_c5 or (c5_new > work_c5 and np.random.rand() < np.exp(-(c5_new - work_c5) / temp)):
            work_h = h_new
            work_c5 = c5_new
    return work_h, work_c5

def slsqp_refine(h, T, n, maxiter=600):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if T > 1e6:
            return m
        return m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def multi_restart(h, n, t_end, n_restarts=3, scale=0.04):
    global budget_s
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for r in range(n_restarts):
        if time.time() > t_end:
            break
        pert = scale * np.random.randn(n)
        h_pert = project_h(best_h + pert)
        h_new, c5_new = coord_descent(h_pert, n, min(t_end, time.time() + budget_s * 0.06), 15, n_top=80)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive coordinate descent
    t1 = t0 + budget_s * 0.20
    best_h, best_c5 = coord_descent(best_h, n, t1, 50, n_top=50)
    t2 = t0 + budget_s * 0.38
    best_h, best_c5 = coord_descent(best_h, n, t2, 50, n_top=120)
    t3 = t0 + budget_s * 0.55
    best_h, best_c5 = coord_descent(best_h, n, t3, 80, n_top=None)

    # Phase 2: Basin hopping exploration
    t4 = t0 + budget_s * 0.72
    bh_h, bh_c5 = basin_hop(best_h, n, t4, 20, 0.05)
    if bh_c5 < best_c5:
        best_c5 = bh_c5
        best_h = bh_h
    best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.82, 50, n_top=None)

    # Phase 3: Multi-restart local search
    t5 = t0 + budget_s * 0.88
    best_h, best_c5 = multi_restart(best_h, n, t5, 4, 0.03)
    best_h, best_c5 = coord_descent(best_h, n, t_end - 10, 40, n_top=None)

    # Phase 4: SLSQP polish
    if t_end - time.time() > 30:
        hc = slsqp_refine(best_h, 30000, n, 700)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
    # Final coordinate descent cleanup
    best_h, best_c5 = coord_descent(best_h, n, t_end - 1, 25, n_top=None)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    return best_h, best_c5, n
