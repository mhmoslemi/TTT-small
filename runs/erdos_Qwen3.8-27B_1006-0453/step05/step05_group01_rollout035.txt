    target = n / 2.0
    s = h.sum()
    d = target - s
    if abs(d) < 1e-14:

        tot = room.sum()
        if tot < 1e-14:

        h = h + room * (d / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def slsqp(h0, T, k_top, n, maxiter=300):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        top = np.sort(corr)[-k_top:]
        m = top[-1]
        dx = 2.0 / n
        if T > 1e10:
            return m * dx

        corr = np.correlate(x, 1.0 - x, mode='full')

        top_vals = corr[top_idx]
        m = top_vals.max()

                if 0 <= jm < n:

    d = target - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (d / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + d / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h0, T, k, n, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(corr)[-k:]
        top = corr[idx]
        m = top[-1]
        if T > 1e10:
            return m * dx
        return (m + np.log(np.sum(np.exp(T * (top - m))) / k) / T) * dx
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_h(np.clip(res.x, 0, 1))

def coord_descent(h, t_end, n_top=15, max_sweeps=100):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode='full').astype(np.float64).copy()
    for sw in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            bi = best[i]
            delta = np.zeros(2 * n - 1)
            for s in range(2 * n - 1):
                sh = s - (n - 1)
                val = 0.0
                jp, jm = i + sh, i - sh
                if 0 <= jp < n:
                    val += 1.0 - best[jp]
                if 0 <= jm < n:
                    val -= best[jm]
                delta[s] = val
            res = minimize_scalar(lambda t: np.max(corr + (t - bi) * delta),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = res.x
            if abs(t_new - bi) < 1e-12:
                continue
            best[i] = t_new
            corr += (t_new - bi) * delta
            new_c5 = corr.max() * dx
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                improved = True
            else:
                corr -= (t_new - bi) * delta
                best[i] = bi
        if not improved:
            break
    best = project_h(best)
    best_c5 = compute_c5(best)
    return best, best_c5

def basin_hop_structured(h, n, t_end, n_iter=25, T_slsqp=300):
    work = h.copy()
    work_c5 = compute_c5(work)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - it / max(n_iter, 1)
        corr = np.correlate(work, 1.0 - work, mode='full')
        top_idx = np.argsort(corr)[-8:]
        shifts = top_idx - (n - 1)
        pert = np.zeros(n)
        for k in shifts[:5]:
            freq = max(abs(k), 1) / float(n)
            amp = 0.06 * cool
            phase = np.random.uniform(0, 2 * np.pi)
            pert += amp * np.sin(2 * np.pi * freq * np.arange(n) + phase)
        pert += 0.025 * cool * np.random.randn(n)
        h_pert = project_h(work + pert)
        hc = slsqp_smooth(h_pert, T_slsqp, 5, n, 200)
        c5v = compute_c5(hc)
        temp = 0.0012 * cool + 0.0002
        if c5v < work_c5 or (c5v > work_c5 and np.random.rand() < np.exp(-(c5v - work_c5) / temp)):
            work = hc
            work_c5 = c5v
    return work, work_c5

def subgrad_step(h, alpha, n_topk, mom_v, beta=0.4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_topk:]
    shifts = top_idx - (n - 1)
    grad = np.zeros(n)
    for t in range(n_topk):
        k = shifts[t]
        for i in range(n):
            val = 0.0
            jp, jm = i + k, i - k
            if 0 <= jp < n:
                val += (1.0 - h[jp])
            if 0 <= jm < n:
                val -= h[jm]
            grad[i] += val
    grad *= dx / n_topk
    if mom_v is not None:
        grad = beta * mom_v + (1.0 - beta) * grad
    new_h = project_h(h - alpha * grad)
    return new_h, grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(200)
        best_h[:100] = 1.0
    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with decreasing temperature
    t_p1 = t0 + budget_s * 0.20
    for T, k in [(50, 8), (200, 5), (1000, 3), (5000, 2)]:
        if time.time() > t_p1 - 5:
            break
        hc = slsqp_smooth(best_h, T, k, n, 400)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Subgradient + SLSQP refinement
    t_p2 = t0 + budget_s * 0.40
    mom_v = np.zeros(n)
    for it in range(60):
        if time.time() > t_p2 - 10:
            break
        cool = 1.0 - it / 60.0
        alpha = 0.001 + 0.004 * cool
        n_topk = 4 + int(8 * cool)
        hs, g = subgrad_step(best_h, alpha, n_topk, mom_v, beta=0.5)
        mom_v = g
        T_cur = 200 + 3000 * cool
        k_cur = 3 + int(3 * cool)
        hc = slsqp_smooth(hs, T_cur, k_cur, n, 250)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 3: Basin hopping
    t_p3 = t0 + budget_s * 0.60
    bh_h, bh_c5 = basin_hop_structured(best_h, n, t_p3, 20, T_slsqp=300)
    if bh_c5 < best_c5:
        best_h, best_c5 = bh_h, bh_c5

    # Phase 4: Coordinate descent
    t_p4 = t0 + budget_s * 0.80
    best_h, best_c5 = coord_descent(best_h, t_p4, n_top=15, max_sweeps=80)

    # Phase 5: Fine basin hopping
    t_p5 = t0 + budget_s * 0.90
    bh_h, bh_c5 = basin_hop_structured(best_h, n, t_p5, 12, T_slsqp=2000)
    if bh_c5 < best_c5:
        best_h, best_c5 = bh_h, bh_c5

    # Phase 6: Final coordinate descent
    t_p6 = t0 + budget_s * 0.96
    best_h, best_c5 = coord_descent(best_h, t_p6, n_top=20, max_sweeps=60)

    # Phase 7: Final SLSQP polish
    if time.time() < t_end - 10:
        hc = slsqp_smooth(best_h, 1e12, 1, n, 500)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Last coord descent
    if time.time() < t_end - 3:
        best_h, best_c5 = coord_descent(best_h, t_end - 1, n_top=25, max_sweeps=40)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
