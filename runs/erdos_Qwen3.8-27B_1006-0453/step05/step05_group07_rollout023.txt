import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(x, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def _sgrad(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    mf = 2 * n - 1
    for k in range(mf):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=300):
    try:
        r = minimize(h, lambda x: _sobj(x, T), lambda x: _sgrad(x, T),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=100, no_proj=False):
    """Coordinate descent with exact 1D line search.
    If no_proj=True, skip intermediate projection for speed/accuracy."""
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sweep in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr_cache = np.correlate(best, 1.0 - best, mode='full')
        order = np.random.permutation(n)
        for ii in range(n):
            if time.time() > t_end:
                return best, best_c5
            i = order[ii]
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            # Minimize max_k(corr_cache[k] + (nv - old) * coef[k]) over nv in [0,1]
            base = corr_cache - old * coef
            def f1d(nv):
                return float(np.max(base + nv * coef))
            r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-12})
            nv = float(r.x)
            if abs(nv - old) < 1e-13:
                continue
            trial = best.copy()
            trial[i] = nv
            if no_proj:
                # No projection - just check if it helps
                c5t = compute_c5(trial)
            else:
                trial = proj_h(trial, n)
                c5t = compute_c5(trial)
            if c5t < best_c5 - 1e-14:
                best = trial
                best_c5 = c5t
                corr_cache = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not no_proj and imp:
            best = proj_h(best, n)
        if not imp:
            break
    if not no_proj:
        best = proj_h(best, n)
    return best, compute_c5(best)

def binary_sa(h, n, t_end, max_iters=300000):

    ones = np.where(h_b == 1)[0]

    elif len(ones) < n // 2:

    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T = 0.02
    for _ in range(max_iters):
        if time.time() > t_end:
            break
        # Try swap: find one 1 and one 0
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_b[i], h_b[j] = 0.0, 1.0
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(min(0, -(nc5 - best_c5) / T)):
            corr = nc

                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = 1.0, 0.0
        T = max(0.0003, T * 0.99997)

def basin_hop(h, n, t_end, n_iters=200):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 4
        if mod == 0:
            pert = (0.03 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 15)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.3 + 0.4 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.05 + 0.2 * cool) * np.random.randn(mask.sum())
        else:
            # Swap perturbation: swap two nearby elements
            i = np.random.randint(n - 1)
            pert = np.zeros(n)
            pert[i] = 0.15 * cool * (h[(i+1)%n] - h[i])
            pert[(i+1)%n] = -pert[i]
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 8000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.001 * cool + 0.00005
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:

def make_periodic(n, period, fill):
    """Create a periodic pattern with given period and fill ratio."""
    h = np.zeros(n)
    pat = np.zeros(period)
    for i in range(fill):
        pat[i] = 1.0
    h = np.tile(pat, n // period + 1)[:n]
    # Adjust to get exactly n/2 ones
    ones = int(h.sum())
    if ones > n // 2:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:ones - n//2]] = 0
    elif ones < n // 2:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:n//2 - ones]] = 1
    return h

def make_quadratic_residue(n, p):
    """Based on quadratic residues mod p."""
    h = np.zeros(n)
    resid = set()
    for i in range(1, p):
        resid.add((i*i) % p)
    for i in range(n):
        if i % p in resid:
            h[i] = 1.0
    ones = int(h.sum())
    if ones > n // 2:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:ones - n//2]] = 0
    elif ones < n // 2:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:n//2 - ones]] = 1
    return h

For the 1D problem: minimize max_k (a_k + δ*b_k) over δ ∈ [0,1]
where a_k = c_k - old*coef_k and b_k = coef_k.

This is a linear program in 1D! The optimal δ is at a point where the maximum is minimized. We can find it by:
1. For each pair (k1, k2), find the δ where a_k1 + δ*b_k1 = a_k2 + δ*b_k2
2. Check all such δ values plus the boundaries 0 and 1
3. Pick the δ that minimizes the max

But this is O(mf²) which for mf=319 is about 100K operations - feasible but maybe slow in a tight loop.

Alternatively, we can use a more efficient approach: binary search on the target value. For a given target M, check if there exists δ ∈ [0,1] such that a_k + δ*b_k ≤ M for all k. This is a system of linear inequalities in 1D, which reduces to:
- For b_k > 0: δ ≤ (M - a_k) / b_k
- For b_k < 0: δ ≥ (M - a_k) / b_k
- For b_k = 0: need a_k ≤ M

So the feasible set is [max over b_k<0 of (M-a_k)/b_k, min over b_k>0 of (M-a_k)/b_k] ∩ [0,1].

Binary search on M: O(mf) per check, O(mf * log(precision)) total. This is O(mf * 50) ≈ 16000 operations, similar to the scipy approach.

        else:
            h = h + rm * ((tgt - s) / t)

    best_c5 = 999
    t_scr = t0 + budget_s * 0.05
    for hc in candidates:
        if time.time() > t_scr:
            break
        hc = proj_h(hc, n_d)
        ho = lbfgs_opt(hc, 300.0, n_d, iters=80)
        c5v = compute_c5(ho)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = ho.copy()

    t_p1 = t0 + budget_s * 0.14
    for T in [500, 2000, 8000, 30000, 120000]:
        if time.time() > t_p1:
            break
        hn = lbfgs_opt(best_h, T, n_d, iters=350)
        cv = compute_c5(hn)
        if cv < best_c5:
            best_h, best_c5 = hn, cv

    t_p2 = t0 + budget_s * 0.27
    best_h, best_c5 = coord_desc(best_h, n_d, t_p2, sweeps=70)

    t_p3 = t0 + budget_s * 0.42
    hb, c5b = binary_sa(best_h, n_d, t_p3, 250000)
    if c5b < best_c5:
        best_h = lbfgs_opt(hb, 3000.0, n_d, iters=250)
        best_c5 = compute_c5(best_h)
        t_p3b = t_p3 + (t_p3 - t0) * 0.3
        best_h, best_c5 = coord_desc(best_h, n_d, t_p3b, sweeps=35)

    t_p4 = t0 + budget_s * 0.62
    best_h, best_c5 = basin_hop(best_h, n_d, t_p4, 150)

    t_p5 = t0 + budget_s * 0.73
    best_h, best_c5 = coord_desc(best_h, n_d, t_p5, sweeps=50)

    for n_alt in [128, 200, 192]:
        if time.time() - t0 > budget_s * 0.86:
            break
        try:
            xo = np.linspace(0, 2, n_d, endpoint=False) + 1.0 / n_d
            xn = np.linspace(0, 2, n_alt, endpoint=False) + 1.0 / n_alt
            ha = np.interp(xn, xo, best_h)
            ha = proj_h(ha, n_alt)
            ha = lbfgs_opt(ha, 3000.0, n_alt, iters=250)
            c5a = compute_c5(ha)
            ha, c5a = coord_desc(ha, n_alt, min(t0 + budget_s * 0.91, time.time() + 50), 25)
            ha = proj_h(ha, n_alt)
            c5a = compute_c5(ha)
            if c5a < best_c5:
                best_h, best_c5 = ha, c5a
        except Exception:
            pass

    if time.time() - t0 < budget_s * 0.93:
        best_h = lbfgs_opt(best_h, 300000.0, len(best_h), iters=500)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.97:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t0 + budget_s * 0.985, 30)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
