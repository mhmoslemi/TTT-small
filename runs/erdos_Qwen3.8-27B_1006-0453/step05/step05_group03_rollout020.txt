    def obj(x):

        m = np.max(c)

        hb, cb = binary_sa(h_start, n, t_end_sa, max_iters=50000, T0=0.25 - 0.05 * rd)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
        # Soften
        if time.time() < t2:
            hs = slsqp_opt(hb, 2000, n, maxiter=350)
            cs = compute_c5(hs)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs

    # Phase 3: SLSQP annealing with analytical gradient
    t3 = t0 + budget_s * 0.48
    T_list = [30000, 12000, 5000, 2000, 800, 300, 100, 40, 15, 5, 2]
    for T in T_list:
        if time.time() > t3:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=600)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5 = c
            best_h = hc

    # Phase 4: Basin hopping
    t4 = t0 + budget_s * 0.76
    work_h = best_h.copy()
    work_c5 = best_c5
    niter = 0
    while time.time() < t4 and niter < 55:
        niter += 1
        cool = max(0.05, 1.0 - niter / 55.0)
        amp = 0.015 + 0.07 * cool
        h_pert = project_h(work_h + amp * np.random.randn(n), n)
        if np.random.rand() < 0.25:
            h_pert = (work_h + 0.4 * np.random.randn(n) > 0.5).astype(float)
            h_pert = project_h(h_pert, n)
        T_bh = 600 + 4000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=450)
        c = compute_c5(hc)
        temp = 0.0006 * cool + 0.00003
        if c < work_c5 or np.random.rand() < np.exp(-(c - work_c5) / max(temp, 1e-10)):
            work_h = hc
            work_c5 = c
            if c < best_c5:
                best_c5 = c
                best_h = hc

    # Phase 5: Coordinate descent
    t5 = t0 + budget_s * 0.90
    best_h, best_c5 = coord_descent(best_h, n, t5, max_sweeps=100)

    # Phase 6: Final fine SLSQP
    t6 = t0 + budget_s * 0.97
    for T in [3, 1, 0.5, 0.2]:
        if time.time() > t6:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=700)
        c = compute_c5(hc)
        if c < best_c5:
            best_c5 = c
            best_h = hc

    # Final coordinate descent
    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=40)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
