import time

def compute_c5(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

    tgt = n * 0.5

    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * 2.0 / len(x)

def _grad(x):
    T = _G['T']
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    e = np.exp(T * (corr - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):

        if wk < 1e-10: continue

            g[lo1:hi1] += wk * (1.0 - x[idx])
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0 / n)

    try:

                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-8})
        return proj_h(r.x, n)
    except Exception:

    best_c5 = compute_c5(best)

            best = h_new.copy()

    return best, best_c5

        h2, c5_2 = coord_cd(h2, n2, t0 + budget_s * 0.88, sweeps=25)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
            n = n2

    # Phase 8: Try n=240
    if time.time() < t0 + budget_s * 0.95:
        n3 = 240
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n3, endpoint=False) + 1.0 / n3
        h3 = np.interp(x_new, x_old, best_h)
        h3 = proj_h(h3, n3)
        for T in [500, 10000, 100000]:
            if time.time() > t0 + budget_s * 0.94:
                break
            h3 = lbfgs(h3, T, n3, iters=200)
        c5_3 = compute_c5(h3)
        h3, c5_3 = coord_cd(h3, n3, t0 + budget_s * 0.95, sweeps=20)
        if c5_3 < best_c5:
            best_h = h3
            best_c5 = c5_3

    # Phase 9: Final polish
    if time.time() < t0 + budget_s * 0.98:
        n = len(best_h)
        best_h = lbfgs(best_h, 200000.0, n, iters=400)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_cd(best_h, n, t0 + budget_s * 0.99, sweeps=25)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
