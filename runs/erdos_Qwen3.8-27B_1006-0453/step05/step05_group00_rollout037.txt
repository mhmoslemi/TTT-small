```python
import numpy as np
import time

def fft_corr(h):
    N = len(h)
    L = 2 * N - 1
    H = np.fft.fft(h, L)
    G = np.fft.fft(np.flip(1.0 - h), L)
    return np.real(np.fft.ifft(H * G))

def compute_c5(h, N):
    return float(np.max(fft_corr(h))) * (2.0 / N)

def init_LR(h, N):
    """Compute L and R matrices for the given binary h."""
    k_idx = np.arange(2 * N - 1)
    l = k_idx - (N - 1)  # shape (2N-1,)
    pos = np.arange(N)
    
    # L[q,k] = (1 - h[q + l[k]]) * [0 <= q+l[k] < N]
    idx_L = pos[:, None] + l[None, :]
    valid_L = (idx_L >= 0) & (idx_L < N)
    L = np.where(valid_L, 1.0 - h[idx_L.clip(0, N-1)], 0.0)
    
    # R[q,k] = h[q - l[k]] * [0 <= q-l[k] < N]
    idx_R = pos[:, None] - l[None, :]
    valid_R = (idx_R >= 0) & (idx_R < N)
    R = np.where(valid_R, h[idx_R.clip(0, N-1)], 0.0)
    
    return L, R, l

def greedy_swap(h, L, R, l, c, ones, zeros, N):
    """Find the best swap (p in ones -> 0, q in zeros -> 1) that minimizes max correlation."""
    best_P = float(c.max())
    best_pq = None
    
    L_zeros = L[zeros]  # (n_zeros, 2N-1)
    R_zeros = R[zeros]  # (n_zeros, 2N-1)
    C_zeros = L_zeros - R_zeros  # (n_zeros, 2N-1)
    
    for p in ones:
        B_p = c - L[p] + R[p]  # (2N-1,)
        vals = B_p[None, :] + C_zeros  # (n_zeros, 2N-1)
        maxes = vals.max(axis=1)  # (n_zeros,)
        mi = int(np.argmin(maxes))
        if maxes[mi] < best_P:
            best_P = float(maxes[mi])
            best_pq = (int(p), int(zeros[mi]))
    
    return best_pq, best_P

def update_LR_swap(L, R, p, q, N):
    """Update L and R after swapping: h[p] 1->0, h[q] 0->1."""
    # Flipping h[p] 1->0:
    # L[q2, p-q2+N-1] += 1 for all q2
    # R[q2, q2-p+N-1] -= 1 for all q2
    q2 = np.arange(N)
    k1 = p - q2 + N - 1  # for L update
    k2 = q2 - p + N - 1  # for R update
    L[np.arange(N), k1] += 1.0
    R[np.arange(N), k2] -= 1.0
    
    # Flipping h[q] 0->1:
    # L[q2, q-q2+N-1] -= 1 for all q2
    # R[q2, q2-q+N-1] += 1 for all q2
    k3 = q - q2 + N - 1
    k4 = q2 - q + N - 1
    L[np.arange(N), k3] -= 1.0
    R[np.arange(N), k4] += 1.0

def binary_local_search(h, N, deadline, max_sweeps=2000, perturb_every=50, n_perturb=3):
    h = h.astype(float).copy()
    nh = N // 2
    # Ensure exactly nh ones
    n1 = int(h.sum())
    if n1 != nh:
        if n1 > nh:
            idx = np.where(h == 1)[0]
            idx = idx[np.random.choice(len(idx), n1-nh, replace=False)]
            h[idx] = 0
        else:
            idx = np.where(h == 0)[0]
            idx = idx[np.random.choice(len(idx), nh-n1, replace=False)]
            h[idx] = 1
    
    L, R, l = init_LR(h, N)
    c = fft_corr(h)
    best_c = float(c.max())
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    sweep_no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        best_pq, new_P = greedy_swap(h, L, R, l, c, ones, zeros, N)
        if best_pq is None:
            sweep_no_improve += 1
            if sweep_no_improve >= 3:
                # Perturb
                if perturb_every > 0 and sweep > 0:
                    for _ in range(n_perturb):
                        p = ones[np.random.randint(len(ones))]
                        q = zeros[np.random.randint(len(zeros))]
                        h[p], h[q] = 0.0, 1.0
                        update_LR_swap(L, R, int(p), int(q), N)
                        c = fft_corr(h)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    sweep_no_improve = 0
                    continue
            break
        else:
            sweep_no_improve = 0
            p, q = best_pq
            h[p] = 0.0
            h[q] = 1.0
            update_LR_swap(L, R, p, q, N)
            c = fft_corr(h)
            if c.max() < best_c:
                best_c = float(c.max())
                best_h = h.copy()
            # Update ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    
    return best_h, best_c * (2.0/N)
```

```python
    return np.real(np.fft.ifft(H * G))
```

```python
def compute_grad_topk(h, k_list):
    """Compute gradient of sum of top-k correlations w.r.t. h."""
    N = len(h)
    grad = np.zeros(N)
    for k in k_list:
        lag = k - (N - 1)
        # ∂corr[k]/∂h[m] = (1-h[m-lag])*[0≤m-lag<N] - h[m+lag]*[0≤m+lag<N]
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            # m-lag valid: m >= lag, so m in [lag, N-1]: (1-h[m-lag])
            grad[lag:] += (1.0 - h[:N-lag])
            # m+lag valid: m+lag < N, so m in [0, N-1-lag]: -h[m+lag]
            grad[:N-lag] -= h[lag:]
        else:
            M = -lag  # M > 0
            # m-lag = m+M valid: m+M < N, so m in [0, N-1-M]: (1-h[m+M])
            grad[:N-M] += (1.0 - h[M:])
            # m+lag = m-M valid: m-M >= 0, so m in [M, N-1]: -h[m-M]
            grad[M:] -= h[:N-M]
    return grad / len(k_list)
```
