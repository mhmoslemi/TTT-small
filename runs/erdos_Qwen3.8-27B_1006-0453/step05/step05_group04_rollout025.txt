def basin_hop(h, n, t_end, n_hops=500, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    dx = 2.0/n
    
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        frac = hop / max(n_hops, 1)
        cool = 1.0 - frac
        
        # Generate perturbation
        mode = hop % 7
        if mode == 0:
            # Gaussian
            step = 0.02 + 0.05 * cool
            pert = step * rng.randn(n)
        elif mode == 1:
            # Sine wave
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2*np.pi)
            step = 0.03 + 0.07 * cool
            pert = step * np.sin(2*np.pi*f*np.arange(n)/n + ph)
        elif mode == 2:
            # Block
            a = rng.randint(0, n-20)
            b = a + rng.randint(5, 40)
            pert = np.zeros(n)
            pert[a:b] = 0.1 * rng.randn(b-a)
        elif mode == 3:
            # Random subset
            idx = rng.choice(n, size=n//8, replace=False)
            pert = np.zeros(n)
            pert[idx] = 0.15 * rng.randn(len(idx))
        elif mode == 4:
            # Complement flip on subset
            idx = rng.choice(n, size=n//5, replace=False)
            pert = np.zeros(n)
            pert[idx] = 1.0 - 2.0 * cur_h[idx] + 0.05*rng.randn(len(idx))
        elif mode == 5:
            # Large random restart (partial)
            # Keep half, randomize half
            mask = rng.random(n) < 0.5
            pert = np.zeros(n)
            pert[mask] = 0.5 - cur_h[mask] + 0.2*rng.randn(mask.sum())
        else:
            # Very small nudge
            step = 0.01
            pert = step * rng.randn(n)
        
        h_pert = project_h(cur_h + pert, n)
        
        # Run CD on perturbed solution
        remaining = max(1.0, (t_end - time.time()) / max(n_hops - hop - 1, 1))
        hop_cd_time = min(remaining, 3.0 + 10.0*cool)
        t_hop_end = time.time() + hop_cd_time
        n_sweeps = int(200 + 1000 * cool)
        h_opt, c5_opt = coord_descent(h_pert, n, t_hop_end, max_sweeps=n_sweeps, seed=seed+hop)
        
        # Accept/reject
        temp = 0.003 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5)/temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    return best_h, best_c5
