            return h.copy()

    h_save = h[i]
    h[i] = 0.0
    corr_base = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save

    base = corr_base + h_minus

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    h = np.clip(h, 0.0, 1.0).copy()
    target = len(h) / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h *= (1.0 + diff / s)

    h_save = h[i]
    h[i] = 0.0

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def project_h(h):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-12:
            return h
        h += room * (diff / tot)
    else:
        if s < 1e-12:
            return h
        h *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_opt(h, i, n):
    h_save = h[i]
    h[i] = 0.0
    cb = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save
    s = np.arange(-(n - 1), n)
    ip = i + s
    im = i - s
    vp = (ip >= 0) & (ip < n)
    vm = (im >= 0) & (im < n)
    hp = np.where(vp, h[np.clip(ip, 0, n - 1)], 0.0)
    hm = np.where(vm, h[np.clip(im, 0, n - 1)], 0.0)
    coef = (1.0 - hm) - hp
    k0 = n - 1  # index of tau=0 where delta is quadratic: x*(1-x)
    
    def eval_max(x):
        vals = cb + x * coef
        vals[k0] = cb[k0] + x * (1.0 - x)
        return np.max(vals)
    
    a, b = 0.0, 1.0
    gr = 0.6180339887
    for _ in range(40):

        x2 = a + gr * (b - a)
        if eval_max(x1) < eval_max(x2):
            b = x2
        else:
            a = x1
    return 0.5 * (a + b)

def coord_desc(h, n, t_end, max_sw, rng):

    bc = compute_c5(h)
    no_imp = 0
    for _ in range(max_sw):

            c = compute_c5(h)

                imp = True
        if imp:

def make_pert(h, rng, scale):

        return h + scale * rng.standard_normal(n)
    elif t == 1:

        hp[i], hp[j] = hp[j], hp[i]
        return hp
    elif t == 2:

def basin_hop(h, n, t_end, rng, iterations, base_scale):

            wh, wc = hc, cc
            scale = max(0.008, scale * 0.97)

        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(160)
        h0[:80] = 1.0
    n0 = len(h0)
    h0 = project_h(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Polish parent at n=160
    t1 = t0 + budget_s * 0.10
    best_h, best_c5 = coord_desc(best_h, n0, t1, 300, rng)

    # Phase 2: Upsample to n=320 and optimize
    n2 = 320
    h2 = np.zeros(n2)
    for i in range(n0):
        h2[2 * i] = best_h[i]
        h2[2 * i + 1] = best_h[i]
    h2 = project_h(h2)
    t2 = t0 + budget_s * 0.38
    h2, c5_2 = coord_desc(h2, n2, t2, 200, rng)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Basin hopping (main exploration)
    nw = len(best_h)
    t3 = t0 + budget_s * 0.72
    wh, wc = best_h.copy(), best_c5
    wh, wc = basin_hop(wh, nw, t3, rng, 200, 0.05)
    if wc < best_c5:
        best_c5, best_h = wc, wh

    # Phase 4: Second resolution attempt if we were at 160
    if len(best_h) < 320 and time.time() < t0 + budget_s * 0.82:
        n3 = 320
        h3 = np.array([best_h[i * n3 // len(best_h)] for i in range(n3)], dtype=float)
        h3 = project_h(h3)
        t3b = t0 + budget_s * 0.85
        h3, c5_3 = coord_desc(h3, n3, t3b, 100, rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3

    # Phase 5: Fine basin hopping
    if time.time() < t0 + budget_s * 0.92:
        nw = len(best_h)
        t5 = t0 + budget_s * 0.92
        wh, wc = best_h.copy(), best_c5
        wh, wc = basin_hop(wh, nw, t5, rng, 80, 0.02)
        if wc < best_c5:
            best_c5, best_h = wc, wh

    # Phase 6: Final coordinate descent polish
    t6 = t0 + budget_s * 0.98
    best_h, best_c5 = coord_desc(best_h, len(best_h), t6, 200, rng)

    # Final verification
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
