def smooth_c5(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    dx = 2.0 / n
    return val * dx

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

        return smooth_c5(x, T)

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h

def coord_opt(h, n, i, niter=35):

    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    if np.any(valid1):

    if np.any(valid2):

        f2 = np.max(corr_base + m2 * coefs)

def subgradient_step(h, n, step_size):
    """One subgradient step on the max objective."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Active set: lags within 1% of max
    active = corr > m - 0.01 * m
    # Gradient of f_k(h) w.r.t. h[i]:
    # f_k = sum_j h[j](1-h[j+k])  (for lag k)
    # df_k/dh[i] = (1-h[i+k]) + (-h[i-k]) for valid indices
    # Let's compute the weighted gradient
    grad = np.zeros(n)
    # For each active lag, add its gradient
    lags = np.arange(-(n-1), n)
    for idx in np.where(active)[0]:
        k = lags[idx]
        # f_k(h) = sum_i h[i] * (1-h[i+k]) for valid i
        # df_k/dh[i] = (1-h[i+k]) [from term i] + (-h[i-k]) [from term i-k, i.e., j=i-k: h[j](1-h[j+k]) = h[i-k](1-h[i])]
        # Wait: df_k/dh[i] appears in:
        #   term j=i: h[i](1-h[i+k]) → derivative: (1-h[i+k])
        #   term j=i-k: h[i-k](1-h[i]) → derivative: -h[i-k]
        for i in range(n):
            grad[i] += (1.0 - (h[i+k] if 0 <= i+k < n else 0)) - (h[i-k] if 0 <= i-k < n else 0)
    # Normalize
    gn = np.linalg.norm(grad)
    if gn > 0:
        h_new = h - step_size * grad / gn
    else:
        h_new = h
    return project_h(h_new)

    n = len(x)
    k = min(topk, len(corr))
    idx = np.argpartition(corr, -k)[-k:] if k < len(corr) else np.arange(len(corr))
    c = corr[idx]
    m = np.max(c)
    val = m + np.log(np.sum(np.exp(T * (c - m)))) / T
    return val * (2.0 / n)

def con_sum(x, n):
    return x.sum() - n / 2.0

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, T, n, maxiter=2000, ftol=1e-15, topk=0):
    if topk > 0:
        obj_fn = lambda x: smooth_c5_topk(x, T, topk)
        con_fn = lambda x: x.sum() - n / 2.0
    else:
        obj_fn = lambda x: smooth_c5(x, T)
        con_fn = lambda x: x.sum() - n / 2.0
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fn}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt(h, n, i, niter=35):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    valid1 = (vm1 >= 0) & (vm1 < n)
    valid2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    if np.any(valid1):
        coefs[valid1] += (1.0 - h[np.clip(vm1[valid1], 0, n - 1)])
    if np.any(valid2):
        coefs[valid2] += (-h[np.clip(vm2[valid2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(corr_base + m1 * coefs)
        f2 = np.max(corr_base + m2 * coefs)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=40, seed=42):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            h_try = best_h.copy()
            h_try[i] = hi_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_refine(h, n, steps=50, base_step=0.01):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for s in range(steps):
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        m = np.max(corr)
        active = corr > m - 0.05 * m
        lags = np.arange(-(n - 1), n)
        grad = np.zeros(n)
        act_lags = lags[active]
        for k in act_lags:
            k = int(k)
            if k >= 0:
                g1 = np.zeros(n)
                g1[:n - k] = 1.0 - best_h[k:n]
                g2 = np.zeros(n)
                g2[k:] = -best_h[:n - k]
            else:
                k2 = -k
                g1 = np.zeros(n)
                g1[k2:] = 1.0 - best_h[:n - k2]
                g2 = np.zeros(n)
                g2[:n - k2] = -best_h[k2:]
            grad += g1 + g2
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        step = base_step / (1.0 + 0.1 * s)
        h_try = project_h(best_h - step * grad / gn)
        c5_try = compute_c5(h_try)
        ls = step
        for _ in range(8):
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                break
            ls *= 0.5
            if ls < 1e-8:
                break
            h_try = project_h(best_h - ls * grad / gn)
            c5_try = compute_c5(h_try)
        else:
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
    return best_h, best_c5

def interpolate_h(h, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h[-1:], h, h[:1]])
    x_ext = np.concatenate([[-2.0/n_old], x_old, [2.0 + 2.0/n_old]])
    h_new = np.interp(x_new, x_ext, h_ext)
    return project_h(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0

    n = len(best_h)
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-temperature SLSQP (12% budget)
    p1_end = t0 + budget_s * 0.12
    for T in [100, 1000, 10000, 100000, 500000]:
        if time.time() > p1_end:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=2000, ftol=1e-15)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 1b: Active-set focused optimization
    if time.time() < p1_end + budget_s * 0.03:
        hc = slsqp_opt(best_h, 20000, n, maxiter=1500, ftol=1e-15, topk=30)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 2: Coordinate descent (13% budget)
    p2_end = t0 + budget_s * 0.25
    best_h, best_c5 = coord_descent(best_h, n, p2_end, max_sweeps=35, seed=seed)

    # Phase 2b: Subgradient refinement
    if time.time() < t0 + budget_s * 0.30:
        best_h, best_c5 = subgrad_refine(best_h, n, steps=60, base_step=0.015)
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.32, max_sweeps=10, seed=seed+1)

    # Phase 3: Basin hopping (33% budget)
    p3_end = t0 + budget_s * 0.65
    work_h = best_h.copy()
    work_c5 = best_c5
    n_iter = 120
    rng = np.random.RandomState(seed + 100)
    for it in range(n_iter):
        if time.time() > p3_end:
            break
        cool = max(0.05, 1.0 - 0.95 * (it / n_iter))
        mod = it % 7
        if mod == 0:
            pert = (0.02 + 0.06 * cool) * rng.randn(n)
        elif mod == 1:
            f = rng.randint(1, 25)
            ph = rng.uniform(0, 2 * np.pi)
            pert = (0.015 + 0.05 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = rng.random(n) < 0.12
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.10 * cool) * rng.randn(max(1, mask.sum()))
        elif mod == 3:
            blk = max(2, n // 12)
            b = rng.randint(0, max(1, n // blk))
            pert = np.zeros(n)
            pert[b*blk:min(n,(b+1)*blk)] = (0.02 + 0.06 * cool) * rng.randn(blk)
        elif mod == 4:
            pert = np.zeros(n)
            for _ in range(4):
                a, b2 = rng.choice(n, 2, replace=False)
                d = (work_h[b2] - work_h[a]) * rng.uniform(0.3, 1.0)
                pert[a] += d
                pert[b2] -= d
        elif mod == 5:
            # Binary perturbation
            mask = rng.random(n) < 0.06
            pert = np.zeros(n)
            pert[mask] = rng.choice([-1, 1], size=max(1, mask.sum())) * (0.1 + 0.1 * cool)
        else:
            # Large jump to escape
            pert = (0.04 + 0.08 * cool) * rng.randn(n)

        h_pert = project_h(work_h + pert)
        T_bh = 50 + 15000 * cool
        hc = slsqp_opt(h_pert, T_bh, n, maxiter=350, ftol=1e-13)
        c5v = compute_c5(hc)
        # Quick coord descent from perturbed point
        hc2, c5v2 = coord_descent(hc, n, time.time() + 1.5, max_sweeps=3, seed=seed+it)
        if c5v2 < c5v:
            hc, c5v = hc2, c5v2
        temp = 0.0008 * cool + 0.00005
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / temp):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 4: Try larger n (18% budget)
    p4_end = t0 + budget_s * 0.83
    for n_new in [200, 256]:
        if time.time() > p4_end:
            break
        h_interp = interpolate_h(best_h, n, n_new)
        hc = slsqp_opt(h_interp, 5000, n_new, maxiter=1500, ftol=1e-14)
        c5v = compute_c5(hc)
        hc, c5v = coord_descent(hc, n_new, time.time() + budget_s * 0.04, max_sweeps=8, seed=seed+n_new)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc
            n = n_new

    # Phase 5: High-T SLSQP polish (10% budget)
    p5_end = t0 + budget_s * 0.93
    for T in [200000, 1000000, 5000000]:
        if time.time() > p5_end:
            break
        hc = slsqp_opt(best_h, T, n, maxiter=3000, ftol=1e-16)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    # Phase 6: Final coordinate descent + subgradient
    if time.time() < t0 + budget_s * 0.97:
        best_h, best_c5 = subgrad_refine(best_h, n, steps=30, base_step=0.005)
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=25, seed=seed+999)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
