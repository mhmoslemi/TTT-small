import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h = h * (tgt / s)
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t < 1e-13:
            h = h * (tgt / max(s, 1e-13))
        else:
            h = h + rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_descent_fast(h, n, t_end, sweeps=200, xatol=1e-14):
    """Fast coordinate descent with pre-computed gradients."""
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = corr_full(best)
    dx = 2.0 / n
    mf = 2 * n - 1
    
    order = np.random.permutation(n)
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            # Compute gradient of corr wrt h[i]
            # g_k = [1-h[i-k+n-1]] * valid + [-h[i+k-(n-1)]] * valid
            # In correlate indexing: corr[k] = sum_j h[j] * (1-h)[j+k-(n-1)]
            # dc_k/dh_i: i appears as h[j] when j=i, and as (1-h)[j+k-n+1] when j=i-k+n-1
            g = np.zeros(mf)
            # h[i] as first factor: corr[k] includes h[i]*(1-h[i+k-n+1]) for valid i+k-n+1
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    g[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    g[k] -= best[j1]
            
            old = best[i]
            # f(delta) = max_k (corr[k] + delta * g[k])
            # We want to minimize this over delta such that old + delta in [0,1]
            def f1d(delta):
                return float(np.max(corr + delta * g))
            
            r = minimize_scalar(f1d, bounds=(-old, 1.0 - old), method='bounded',
                                options={'xatol': xatol, 'maxiter': 100})
            nv = old + float(r.x)
            if abs(nv - old) < 1e-14:
                continue
            best[i] = nv
            # Update corr incrementally
            corr += (nv - old) * g
            new_c5 = float(np.max(corr) * dx)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                imp = True
            else:
                # Revert
                best[i] = old
                corr -= (nv - old) * g
        
        if not imp:
            break
        order = np.random.permutation(n)
    return best, best_c5

Actually, let me reconsider the coordinate descent. The `minimize_scalar` call at each coordinate is expensive. For a piecewise linear convex function, I can find the minimum more directly.

f(δ) = max_k (c_k + δ * g_k), δ ∈ [δ_lo, δ_hi]

This is the upper envelope of (2n-1) lines. The minimum of this upper envelope occurs at a vertex (intersection of two lines) or at a boundary.

To find it:
1. Compute the value at boundaries: f(δ_lo), f(δ_hi)
2. Compute the derivative (subgradient) at each point. If the left subgradient > 0 and right subgradient < 0 at some point, that's the minimum.
3. The subgradient at δ is the set {g_k : c_k + δ*g_k = max_j(c_j + δ*g_j)}

A simple approach: binary search on the subgradient.
- Compute f'(δ) = max{g_k : k ∈ argmax(c + δg)} - min{g_k : k ∈ argmax(c + δg)}
- If f'(δ) > 0, increase δ; if < 0, decrease; if 0, we're at the minimum.

Actually, for a convex piecewise linear function, the "derivative" is well-defined except at breakpoints. At a breakpoint, the subgradient is an interval [min_sl, max_sl]. If 0 is in this interval, we're at the minimum.

Algorithm:
1. Start with δ_lo and δ_hi
2. Evaluate f at δ_lo and δ_hi
3. Find the active set (indices k where c_k + δ*g_k = max)
4. If all active g_k > 0, move δ_lo up
   If all active g_k < 0, move δ_hi down
   If there are both positive and negative g_k in active set, we're at a minimum (or need to find the exact breakpoint)
5. Bisect

This is essentially a variant of the golden section method but exploiting the piecewise-linear structure. For simplicity, I'll just use minimize_scalar.
