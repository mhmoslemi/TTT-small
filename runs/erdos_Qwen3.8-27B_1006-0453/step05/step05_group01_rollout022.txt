def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h.copy()
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h.copy()
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, t_end, n_active=20):
    """One full sweep of exact 1D coordinate descent."""
    dx = 2.0 / n
    proj_step = 20  # project every 20 coordinates
    
    for i in range(n):
        if time.time() > t_end:
            break
        old_val = h[i]
        k = np.arange(2*n-1) - (n-1)
        valid1 = (i + k >= 0) & (i + k < n)
        valid2 = (i - k >= 0) & (i - k < n)
        
        slope = np.zeros(2*n-1)
        if np.any(valid1):
            slope[valid1] = 1.0 - h[i + k[valid1]]
        if np.any(valid2):
            slope[valid2] += -h[i - k[valid2]]
        
        # Objective at t: max_j (corr[j] + (t - old_val)*slope[j])
        # = max_j (corr[j] - old_val*slope[j] + t*slope[j])
        corr = np.correlate(h, 1.0-h, mode='full')
        base = corr - old_val * slope
        
        # Find min of max(base + t*slope) for t in [0,1]
        # Evaluate at 0, 1, and intersections of top lines
        vals_at_old = base + old_val * slope  # = corr
        top = np.argsort(vals_at_old)[-n_active:]
        
        cands = [0.0, 1.0]
        for a in range(len(top)):

                sa, sb = slope[top[a]], slope[top[b]]

                    continue

                if -0.01 <= tc <= 1.01:
                    cands.append(np.clip(tc, 0, 1))
        
        best_t = old_val
        best_v = np.max(base + old_val * slope)
        for tc in cands:
            v = np.max(base + tc * slope)
            if v < best_v - 1e-15:
                best_v = v
                best_t = tc
        
        h[i] = best_t

    return h

    best_h = h.copy()

def targeted_shake(h, n, scale):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-5:]
    shifts = top_idx - (n - 1)
    pert = np.zeros(n)
    for s_val in shifts:
        if s_val >= 0:
            for i in range(n - s_val):
                w = h[i] * (1.0 - h[i + s_val])
                pert[i] -= w
                pert[i + s_val] += w
        elif s_val < 0:
            for i in range(-s_val, n):
                w = h[i] * (1.0 - h[i + s_val])
                pert[i] -= w
                pert[i + s_val] += w
    norm = np.linalg.norm(pert)
    if norm > 1e-12:
        pert = pert / norm * scale
    return project_h(h + pert)
