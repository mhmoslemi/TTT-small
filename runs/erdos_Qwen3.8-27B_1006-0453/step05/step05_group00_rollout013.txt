                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                h[p], h[q] = 0.0, 1.0
            c = fft_corr(h)
        else:
            stuck = 0
            h[best_pq[0]] = 0.0
            h[best_pq[1]] = 1.0
            c = fft_corr(h)
            if float(c.max()) < best_c5:
                best_c5 = float(c.max())
    return h, best_c5

def nesterov_soft(h_in, N, deadline, top_k=30, lr0=0.015, T0=40.0):
    h = h_in.astype(float).copy()
    target = N / 2.0
    center = N - 1
    best_h = h.copy()
    best_c = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.9
    T = T0
    it = 0
    while time.time() < deadline:
        y = h + m
        c = fft_corr(y)
        idx = np.argsort(c)[-top_k:]
        vals = c[idx]
        w = np.exp(T * (vals - vals.max()))
        w /= w.sum()
        grad = np.zeros(N)
        for t, ai in enumerate(idx):
            k = int(ai - center)
            if k == 0:
                g = 1.0 - 2.0 * y
            elif k > 0:
                g = np.zeros(N)
                g[:N - k] += (1.0 - y[k:])
                g[k:] -= y[:N - k]
            else:
                k2 = -k
                g[k2:] += (1.0 - y[:N - k2])
                g[:N - k2] -= y[k2:]
            grad += w[t] * g
        grad /= top_k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 30 == 0:
            c_now = float(np.max(fft_corr(h)))
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
            lr = max(0.003, lr * 0.996)
            T = min(300.0, T * 1.03)
            beta = min(0.97, 0.88 + it * 0.0003)
    c_final = float(np.max(fft_corr(h)))
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, N0)

    # Phase 1: Binary local search on initial (N=N0)
    h_bin = make_binary(h0, N0, rng)
    d1 = min(deadline - 300, t0 + 200)
    h_s, c5_s = binary_search(h_bin, N0, d1, sample_n=140, max_sweeps=500, anneal_p=0.0, rng=rng)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()

    # Phase 2: Nesterov continuous from perturbed binary
    if time.time() < deadline - 200:
        d2 = min(deadline - 150, time.time() + 100)
        hc = best_h.astype(float) + rng.randn(N0) * 0.05
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_soft(hc, N0, d2, top_k=40, lr0=0.012, T0=50.0)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()

    # Phase 3: Binary polish after continuous
    if time.time() < deadline - 150:
        d3 = min(deadline - 100, time.time() + 120)
        h_b3 = make_binary(best_h, N0, rng)
        h_s3, c5_3 = binary_search(h_b3, N0, d3, sample_n=160, max_sweeps=500, rng=rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()

    # Phase 4: Annealed binary (escape local minima)
    if time.time() < deadline - 120:
        d4 = min(deadline - 80, time.time() + 80)
        h_b4 = make_binary(best_h, N0, rng)
        # Perturb a few bits
        ones4 = np.where(h_b4 == 1)[0]
        zeros4 = np.where(h_b4 == 0)[0]
        for _ in range(8):
            p = int(ones4[rng.randint(len(ones4))])
            q = int(zeros4[rng.randint(len(zeros4))])
            h_b4[p], h_b4[q] = 0.0, 1.0
        h_s4, c5_4 = binary_search(h_b4, N0, d4, sample_n=100, max_sweeps=350, anneal_p=0.15, rng=rng)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h_s4.copy()

    # Phase 5: Multi-resolution - try larger N
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [460, 540, 720]:
        if time.time() > deadline - 40:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = make_binary(h_interp, N_try, rng)
        sub_dl = min(deadline - 20, time.time() + 55)
        h_rt, c5_rt = binary_search(h_bt, N_try, sub_dl, sample_n=100, max_sweeps=200, rng=rng)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)

    # Phase 6: Final Nesterov + binary polish at best resolution
    Nf = len(best_h)
    if time.time() < deadline - 10:
        hF = best_h.astype(float) + rng.randn(Nf) * 0.03
        hF = project_sum(hF, Nf / 2.0)
        hF, c5F = nesterov_soft(hF, Nf, deadline - 5, top_k=50, lr0=0.008, T0=60.0)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF.copy()

    if time.time() < deadline - 1:
        hF2 = make_binary(best_h, Nf, rng)
        hF2, _ = binary_search(hF2, Nf, deadline - 0.5, sample_n=100, max_sweeps=60, rng=rng)
        c5F2 = compute_c5(hF2, Nf)
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2.copy()

    final_c5 = compute_c5(best_h, len(best_h))
    return best_h, final_c5, len(best_h)
