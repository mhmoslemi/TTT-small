import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def active_grad(h, corr, top_k):
    """Gradient of max using only top-k lags."""
    n = len(h)
    dx = 2.0 / n
    m_full = 2 * n - 1
    mc = np.max(corr)
    # Get top k lags
    idxs = np.argsort(corr)[-top_k:]
    w = corr[idxs].copy()
    w -= w.min()  # shift so min is 0
    if w.max() > 0:
        w = w / w.sum()  # normalize
    else:
        w = np.ones(len(w)) / len(w)
    
    lag_arr = np.arange(m_full) - (n - 1)
    grad = np.zeros(n)
    for ii, idx in enumerate(idxs):
        wi = w[ii]
        L = lag_arr[idx]
        if L >= 0:
            if L < n:
                grad[L:] += wi * (1.0 - h[:n-L])
                grad[:n-L] += -wi * h[L:]
        else:
            P = -L
            if P < n:
                grad[:n-P] += wi * (1.0 - h[P:])
                grad[P:] += -wi * h[:n-P]
    grad *= dx
    return grad, mc

def projected_grad_descent(h, t_end, max_iters=5000, top_k=5, lr_init=0.01, momentum=0.9):
    """Projected gradient descent with momentum."""
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = get_corr(h)
        grad, mc = active_grad(h, corr, top_k)
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        h_new = project_h(h + v)
        
        # Check if improved
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Occasionally increase top_k for diversity
        elif it % 100 == 0:
            # Backtrack
            lr *= 0.7
            v *= 0.5
            h = best_h.copy()
        else:
            h = best_h.copy()
            v *= 0.9
    
    return best_h, best_c5

    """Coordinate descent focusing on most critical coordinates."""
    n = len(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def active_set_grad(h, corr, top_k=5):
    n = len(h)
    dx = 2.0 / n
    m_full = 2*n - 1
    lag_arr = np.arange(m_full) - (n-1)
    idxs = np.argsort(corr)[-top_k:]
    w = np.ones(len(idxs)) / len(idxs)
    grad = np.zeros(n)
    for ii, idx in enumerate(idxs):
        wi = w[ii]
        L = lag_arr[idx]
        if L >= 0:
            if L < n:
                grad[L:] += wi * (1.0 - h[:n-L])
                grad[:n-L] += -wi * h[L:]
        else:
            P = -L
            if P < n:
                grad[:n-P] += wi * (1.0 - h[P:])
                grad[P:] += -wi * h[:n-P]
    return grad * dx

    n = len(h)

    top_k = 5

            break

        cmax = np.max(corr)
        # Adaptive top_k
        diff = cmax - corr
        active = np.where(diff < cmax * 0.05)[0]
        if len(active) > 10:
            active = np.argsort(corr)[-10:]

                break

                break

    m_full = len(cb)

        return np.max(cb + lin * x + quad * x * x)
    
    # Candidate points: 0, 1, and vertices of active quadratics
    candidates = [0.0, 1.0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mc = np.max(corr)
    d = T * (corr - mc)
    d -= d.max()
    exps = np.exp(d)
    w = exps / exps.sum()
    sm = mc + np.log(exps.sum()) / T
    val = dx * sm + lam * (h.sum() - n / 2.0) ** 2
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    grad = np.zeros(n)
    for idx in range(m_full):
        wi = w[idx]
        if wi < 1e-15:
            continue
        L = lag_arr[idx]
        if L >= 0:
            if L < n:
                grad[L:] += wi * (1.0 - h[:n - L])
                grad[:n - L] += -wi * h[L:]
        else:
            P = -L
            if P < n:
                grad[:n - P] += wi * (1.0 - h[P:])
                grad[P:] += -wi * h[:n - P]
    grad = dx * grad + 2.0 * lam * (h.sum() - n / 2.0)
    return val, grad

def lbfqs_run(h0, T, lam, maxiter, ftol):
    n = len(h0)
    x0 = project_h(h0)
    res = minimize(soft_obj_grad, x0, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
    return project_h(res.x)

def fast_coord_descent(h, t_end, max_sweeps):

    dx = 2.0 / n
    m_full = 2 * n - 1
    lag_arr = np.arange(m_full) - (n - 1)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_h = h.copy()
    corr = get_corr(best_h)
    best_c5 = np.max(corr) * dx
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            x_old = best_h[i]
            lin = np.zeros(m_full)
            quad = np.zeros(m_full)
            for idx in range(m_full):
                L = lag_arr[idx]
                if L == 0:
                    lin[idx] = 1.0
                    quad[idx] = -1.0
                elif L > 0:
                    if i >= L:
                        lin[idx] += (1.0 - best_h[i - L])
                    if i + L < n:
                        lin[idx] += -best_h[i + L]
                else:
                    P = -L
                    if i + P < n:
                        lin[idx] += (1.0 - best_h[i + P])
                    if i >= P:
                        lin[idx] += -best_h[i - P]
            cb = corr - x_old * lin - x_old * x_old * quad
            # Candidate x values
            cands = np.array([0.0, 1.0, x_old, 0.25, 0.5, 0.75])
            qmask = np.abs(quad) > 1e-12
            if qmask.any():
                verts = -lin[qmask] / (2.0 * quad[qmask])
                verts = verts[(verts > 0.0) & (verts < 1.0)]
                if len(verts) > 0:
                    cands = np.concatenate([cands, verts[:10]])
            cands = np.clip(cands, 0.0, 1.0)
            vals = np.max(cb[None, :] + lin[None, :] * cands[:, None] + quad[None, :] * cands[:, None] ** 2, axis=1)
            bi = np.argmin(vals)
            bx = cands[bi]
            # Golden section
            lo = max(0.0, bx - 0.05)
            hi = min(1.0, bx + 0.05)
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)
            f1 = np.max(cb + lin * m1 + quad * m1 * m1)
            f2 = np.max(cb + lin * m2 + quad * m2 * m2)
            for _ in range(45):
                if f1 < f2:
                    hi = m2; m2 = m1; f2 = f1
                    m1 = hi - gr * (hi - lo)
                    f1 = np.max(cb + lin * m1 + quad * m1 * m1)
                else:
                    lo = m1; m1 = m2; f1 = f2
                    m2 = lo + gr * (hi - lo)
                    f2 = np.max(cb + lin * m2 + quad * m2 * m2)
            bx = (lo + hi) / 2.0
            if abs(bx - x_old) < 1e-12:
                continue
            new_corr = cb + lin * bx + quad * bx * bx
            c5_new = np.max(new_corr) * dx
            if c5_new < best_c5 - 1e-15:
                corr = new_corr
                best_h[i] = bx
                best_c5 = c5_new
                improved = True
        if not improved:
            break
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

    if 'initial_h_values' in globals() and initial_h_values is not None:
