    tgt = n * 0.5
    if abs(h.sum() - tgt) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    def fsum(lam):
        return np.clip(h + lam, 0.0, 1.0).sum() - tgt
    if h.sum() > tgt:
        lo, hi = -1.0, 0.0
    else:
        lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if fsum(mid) > 0:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / n

def smooth_grad(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n, 2 * n - 1 - k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1) + k - n + 1
            g[lo1:hi1] += wk * (1.0 - x[idx])
        lo2 = max(0, k - n + 1)
        hi2 = min(n, k + 1)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2) - k + n - 1
            g[lo2:hi2] -= wk * x[idx]
    return g * (2.0 / n)

def lbfgs(h, T, n, iters=300):
    try:
        r = minimize(h, lambda x: smooth_obj(x, T), lambda x: smooth_grad(x, T),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-10, 'factr': 1e10})
        return proj_h(r.x, n)
    except Exception:
        return h

_CB = [np.zeros(1)]
_CC = [np.zeros(1)]

def _f1d(a):
    return float(np.max(_CB[0] + a * _CC[0]))

def coord_desc(h, n, t_end, sweeps=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = np.random.permutation(n)
        corr = np.correlate(best, 1.0 - best, mode='full')
        for i in order:
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                j2 = i + k - n + 1
                if 0 <= j2 < n:
                    coef[k] += 1.0 - best[j2]
                j1 = i - k + n - 1
                if 0 <= j1 < n:
                    coef[k] -= best[j1]
            _CB[0] = corr - old * coef
            _CC[0] = coef
            try:
                r = minimize_scalar(_f1d, bounds=(0.0, 1.0), method='bounded',
                                    options={'xatol': 1e-13})
                nv = float(r.x)
            except Exception:
                continue
            if abs(nv - old) < 1e-14:
                continue
            trial = best.copy()
            trial[i] = nv
            trial = proj_h(trial, n)
            c5t = float(np.max(np.correlate(trial, 1.0 - trial, mode='full')) * 2.0 / n)
            if c5t < best_c5 - 1e-15:
                best = trial
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode='full')
                imp = True
        if not imp:
            break
    return best, best_c5

def binary_sa(h, n, t_end, max_iters=800000):
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    if len(ones) > n // 2:
        h_b[np.random.choice(ones, len(ones) - n // 2, replace=False)] = 0
    elif len(ones) < n // 2:
        zeros = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros, n // 2 - len(ones), replace=False)] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    Ti = 0.03
    for it in range(max_iters):
        if time.time() > t_end:
            break
        T = max(0.0002, Ti * (0.99997 ** it))
        r = np.random.rand()
        if r < 0.7:
            for _a in range(8):
                i, j = np.random.randint(n), np.random.randint(n)
                if h_b[i] != h_b[j]:
                    break
            else:
                continue
            h_b[i], h_b[j] = h_b[j], h_b[i]
        else:
            i = np.random.randint(n)
            if r < 0.85:
                j = np.random.randint(n)
                if i == j or h_b[i] == h_b[j]:
                    continue
                h_b[i], h_b[j] = h_b[j], h_b[i]
            else:
                for _a in range(4):
                    i1 = np.random.randint(n)
                    i2 = np.random.randint(n)
                    i3 = np.random.randint(n)
                    i4 = np.random.randint(n)
                    if h_b[i1] != h_b[i2] and h_b[i3] != h_b[i4]:
                        break
                else:
                    continue
                h_b[i1], h_b[i2] = h_b[i2], h_b[i1]
                h_b[i3], h_b[i4] = h_b[i4], h_b[i3]
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            if r < 0.7:
                h_b[i], h_b[j] = h_b[j], h_b[i]
            elif r < 0.85:
                h_b[i], h_b[j] = h_b[j], h_b[i]
            else:
                h_b[i1], h_b[i2] = h_b[i2], h_b[i1]
                h_b[i3], h_b[i4] = h_b[i4], h_b[i3]
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=140):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 4
        if mod == 0:
            pert = (0.012 + 0.06 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 16)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.012 + 0.05 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.12 + 0.2 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.03 + 0.1 * cool) * np.random.randn(mask.sum())
        else:
            blk = np.random.randint(3, 10)
            st = np.random.randint(0, n - blk)
            pert = np.zeros(n)
            pert[st:st + blk] = (0.02 + 0.08 * cool) * np.random.randn(blk)
        hp = proj_h(work + pert, n)
        T_val = 150.0 + 3000.0 * cool
        ht = lbfgs(hp, T_val, n, iters=180)
        ht, c5t = coord_desc(ht, n, min(t_end, time.time() + 6), sweeps=20)
        temp = 0.0005 * cool + 0.00002
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS warmup
    for T in [80, 500, 5000]:
        if time.time() - t0 > budget_s * 0.05:
            break
        h_new = lbfgs(best_h, T, n, iters=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Intensive coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.16, sweeps=150)

    # Phase 3: Binary SA
    end3 = t0 + budget_s * 0.34
    h_bin, c5_bin = binary_sa(best_h, n, end3, max_iters=900000)
    if c5_bin < best_c5:
        best_h = lbfgs(h_bin, 3000.0, n, iters=400)
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.42, sweeps=50)

    # Phase 4: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.62, n_iters=130)

    # Phase 5: Coordinate descent refinement
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.74, sweeps=90)

    # Phase 6: Try higher resolution
    for n_hi in [200, 240]:
        if time.time() - t0 > budget_s * 0.83:
            break
        try:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
            x_new = np.linspace(0, 2, n_hi, endpoint=False) + 1.0 / n_hi
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = proj_h(h_hi, n_hi)
            h_hi = lbfgs(h_hi, 1500.0, n_hi, iters=300)
            h_hi, c5_hi = coord_desc(h_hi, n_hi, min(t0 + budget_s * 0.88, time.time() + 35), sweeps=25)
            if c5_hi < best_c5:
                best_h, best_c5 = h_hi, c5_hi
        except Exception:
            pass

    # Phase 7: Final polish
    if time.time() - t0 < budget_s * 0.91:
        best_h = lbfgs(best_h, 40000.0, len(best_h), iters=600)
        best_c5 = compute_c5(best_h)
    if time.time() - t0 < budget_s * 0.975:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t0 + budget_s * 0.99, sweeps=50)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
