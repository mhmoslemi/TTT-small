from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n=None):
    if n is None:
        n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s) < 1e-14:
        return np.full(n, 0.5)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return np.clip(h + 0.5 * (1.0 - h), 0, 1)
        h = h + np.maximum(room, 0) * (diff / max(tot, 1e-14))
    else:
        h = h * max(1.0 + diff / s, 0)
    return np.clip(h, 0.0, 1.0)

def dc_lp_solve(h0, n, n_constraints=319):
    """One DC iteration: linearize f_k at h0, solve LP."""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    n_corr = len(corr)
    
    # Take top constraints only to keep LP tractable
    top_idx = np.argsort(-corr)[:n_constraints]
    
    js = np.arange(n)
    for ki in top_idx:
        lag = ki - (n - 1)
        grad = np.zeros(n)
        v1 = (js + lag >= 0) & (js + lag < n)
        grad[v1] += (1.0 - h0[js[v1] + lag])
        v2 = (js - lag >= 0) & (js - lag < n)
        grad[v2] += (-h0[js[v2] - lag])
        const = corr[ki] - np.dot(grad, h0)
        constraints.append(cp.sum(grad * h_var) + const <= t_var / dx)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solved = False
    for s in ['CLARABEL', 'ECOS', 'SLS', 'SCS', 'GLPK']:
        try:
            prob.solve(solver=s)
            solved = True
            break

def subgradient_step(h, n, G):
    """One projected subgradient step."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    lag = k_max - (n - 1)
    
    grad = np.zeros(n)
    js = np.arange(n)
    v1 = (js + lag >= 0) & (js + lag < n)
    grad[v1] += (1.0 - h[js[v1] + lag])
    v2 = (js - lag >= 0) & (js - lag < n)
    grad[v2] += (-h[js[v2] - lag])
    
    # BB step size
    g_norm = np.linalg.norm(grad)
    if g_norm < 1e-14:
        return h
    
    # Use diminishing step
    alpha = G['alpha_init'] / (1.0 + G['decay'] * G['iter'])
    h_new = project_h(h - alpha * grad * dx, n)
    G['iter'] += 1
    return h_new

def smooth_c5_multi(h, T, top_k=10):
    """Smooth max using top-k log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use top-k for better conditioning
    if top_k < len(corr):
        idx = np.argsort(-corr)[:top_k]
        top_vals = corr[idx]
        m = np.max(top_vals)
        val = m + np.log(np.sum(np.exp(top_vals - m))) / T
    else:
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(corrs - m))) / T
    return val * dx

def slsqp_optimize(h, n, T, maxiter=2000, ftol=1e-16):
    def obj(x):
        return smooth_c5_multi(x, T, top_k=20)
    
    def eq_con(x):
        return x.sum() - n / 2.0
    
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def coord_single(h, n, i):
    """Optimize single coordinate i."""

    if n is None:
        n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    diff = target - s
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        room = np.maximum(1.0 - h, 0.0)
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        scale = 1.0 + diff / s
        h = h * max(scale, 0.0)
    return np.clip(h, 0.0, 1.0)

def smooth_max(corr, T, top_k=15):
    kk = min(top_k, len(corr))
    idx = np.argpartition(-corr, kk - 1)[:kk]
    vals = corr[idx]
    m = vals.max()
    return m + np.log(np.sum(np.exp(vals - m))) / T

def slsqp_step(h, n, T, maxiter=2000, ftol=1e-16, top_k=15):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, T, top_k) * dx
    def eqcon(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eqcon}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        return project_h(h, n)

def coord_opt(h, n, i):
    h_save = h[i]
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_save
    k_range = np.arange(-(n - 1), n)
    jp = i + k_range
    jm = i - k_range
    vp = (jp >= 0) & (jp < n)
    vm = (jm >= 0) & (jm < n)
    coef = np.zeros(2 * n - 1)
    coef[vp] += (1.0 - h[jp[vp]])
    coef[vm] += (-h[jm[vm]])
    dx = 2.0 / n
    def f_t(t):
        return float(np.max(corr0 + t * coef) * dx)
    try:
        res = minimize_scalar(f_t, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-14})
        return float(res.x)
    except Exception:
        return h_save

def coord_descent(h, n, t_end, max_sweeps=100, rng=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt(best_h, n, i)
            if abs(hi_new - best_h[i]) < 1e-13:
                continue
            best_h[i] = hi_new
            best_h = project_h(best_h, n)
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                imp = True
        if not imp:
            break
    return best_h, best_c5

def dc_lp_solve(h0, n, n_top=150):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    top_idx = np.argsort(-corr)[:n_top]
    js = np.arange(n)
    for ki in top_idx:
        lag = ki - (n - 1)
        grad = np.zeros(n)
        v1 = (js + lag >= 0) & (js + lag < n)
        grad[v1] += (1.0 - h0[js[v1] + lag])
        v2 = (js - lag >= 0) & (js - lag < n)
        grad[v2] += (-h0[js[v2] - lag])
        const = corr[ki] - np.dot(grad, h0)
        constraints.append(cp.sum(grad * h_var) + const <= t_var / dx)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    for s in ['CLARABEL', 'ECOS', 'SLS', 'SCS']:
        try:
            prob.solve(solver=s)
            break
        except Exception:
            continue
    if h_var.value is None:
        return h0
    return project_h(np.asarray(h_var.value, dtype=float).flatten(), n)

def interp_h(h_src, n_src, n_dst):
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src, period=2.0)
    return project_h(h_dst, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h160 = np.array(initial_h_values, dtype=float)
    else:
        h160 = np.zeros(160)
        h160[:80] = 1.0

    best_h = None
    best_c5 = float('inf')

    # === Main optimization at n=160 ===
    n = 160
    h = project_h(h160.copy(), n)
    c5 = compute_c5(h)

    # Phase A: SLSQP continuation
    for T in [80, 400, 1500, 6000, 25000, 100000, 400000]:
        if elapsed() > budget_s * 0.18:
            break
        h = slsqp_step(h, n, T, maxiter=2000, ftol=1e-16, top_k=15)
        c5 = compute_c5(h)

    # Phase B: DC/LP iterations (exact max)
    dc_end = min(t0 + budget_s * 0.28, t0 + 120)
    for it in range(12):
        if time.time() > dc_end:
            break
        h_dc = dc_lp_solve(h, n, n_top=120)
        c5_dc = compute_c5(h_dc)
        if c5_dc < c5 - 1e-12:
            c5, h = c5_dc, h_dc

    # Phase C: Coordinate descent
    h, c5 = coord_descent(h, n, t0 + budget_s * 0.42, max_sweeps=120, rng=rng)

    # Phase D: Basin hopping (diverse)
    work_h, work_c5 = h.copy(), c5
    for it in range(45):
        if elapsed() > budget_s * 0.58:
            break
        cool = max(1.0 - it / 45.0, 0.05)
        mod = it % 4
        if mod == 0:
            h_pert = project_h(work_h + (0.02 + 0.1 * cool) * rng.standard_normal(n), n)
        elif mod == 1:
            fs = rng.randint(2, 12)
            ph = rng.uniform(0, 2 * np.pi)
            pert = (0.04 + 0.1 * cool) * np.sin(2 * np.pi * fs * np.arange(n) / n + ph)
            h_pert = project_h(work_h + pert, n)
        elif mod == 2:
            blk = rng.randint(5, 20)
            st = rng.randint(0, n - blk)
            pert = np.zeros(n)
            pert[st:st + blk] = (0.1 + 0.15 * cool) * rng.standard_normal(blk)
            h_pert = project_h(work_h + pert, n)
        else:
            h_pert = project_h(work_h + 0.08 * cool * rng.standard_normal(n), n)
        T_bh = int(3000 * cool + 300)
        h_opt = slsqp_step(h_pert, n, T_bh, maxiter=800, ftol=1e-15, top_k=10)
        c5_opt = compute_c5(h_opt)
        temp = 0.0006 * cool + 0.00005
        if c5_opt < work_c5 or rng.rand() < np.exp(-(c5_opt - work_c5) / max(temp, 1e-11)):
            work_h, work_c5 = h_opt, c5_opt
            if c5_opt < c5:
                c5, h = c5_opt, h_opt
    if work_c5 < c5:
        c5, h = work_c5, work_h

    # Phase E: More coordinate descent
    h, c5 = coord_descent(h, n, t0 + budget_s * 0.70, max_sweeps=120, rng=rng)

    # Phase F: High-T SLSQP polish
    for T in [800000, 3000000]:
        if elapsed() > budget_s * 0.78:
            break
        h2 = slsqp_step(h, n, T, maxiter=3000, ftol=1e-17, top_k=8)
        c52 = compute_c5(h2)
        if c52 < c5:
            c5, h = c52, h2

    h, c5 = coord_descent(h, n, t0 + budget_s * 0.82, max_sweeps=60, rng=rng)

    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # === Try n=200 (interpolated) ===
    if elapsed() < budget_s * 0.83:
        n2 = 200
        h2 = interp_h(best_h, len(best_h), n2)
        c52 = compute_c5(h2)
        for T in [500, 3000, 15000]:
            if elapsed() > budget_s * 0.90:
                break
            h2 = slsqp_step(h2, n2, T, maxiter=1500, ftol=1e-16, top_k=12)
            c52 = compute_c5(h2)
        h2, c52 = coord_descent(h2, n2, t0 + budget_s * 0.94, max_sweeps=40, rng=rng)
        if c52 < best_c5:
            best_c5, best_h = c52, h2

    # === Try n=120 ===
    if elapsed() < budget_s * 0.92:
        n3 = 120
        h3 = interp_h(best_h, len(best_h), n3)
        c53 = compute_c5(h3)
        for T in [400, 2000]:
            if elapsed() > budget_s * 0.95:
                break
            h3 = slsqp_step(h3, n3, T, maxiter=1000, ftol=1e-16, top_k=10)
            c53 = compute_c5(h3)
        h3, c53 = coord_descent(h3, n3, t0 + budget_s * 0.97, max_sweeps=30, rng=rng)
        if c53 < best_c5:
            best_c5, best_h = c53, h3

    # Final
    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
