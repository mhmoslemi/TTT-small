def lp_step(h, n, max_delta=0.5):
    """Solve LP to find best descent direction for linearized c[k]."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr * dx
    num_k = 2 * n - 1
    
    # Build gradient matrix A: A[k, j] = d c[k]/d h[j]
    # c[k] = sum_i h[i+k]*(1-h[i])
    # d c[k]/d h[j] = (1-h[j-k]) - h[j+k] (out-of-range treated as 0)
    A = np.zeros((num_k, n))
    for j in range(n):
        for k_idx in range(num_k):
            k = k_idx - (n - 1)
            val = 0.0
            jm = j - k
            jp = j + k
            if 0 <= jm < n:
                val += 1.0 - h[jm]
            if 0 <= jp < n:
                val -= h[jp]
            A[k_idx, j] = val
    
    # LP: min C
    # s.t. A @ delta + c_vals <= C  =>  A @ delta - C <= -c_vals
    #      sum(delta) = 0
    #      -h <= delta <= 1 - h
    #      |delta[j]| <= max_delta (trust region)
    
    # Variables: [delta(0), ..., delta(n-1), C]
    # c (objective): [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # Row for each k: A[k,:] @ delta - C <= -c_vals[k]
    A_ub = np.zeros((num_k, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -c_vals
    
    # Bounds: -min(h, max_delta) <= delta[j] <= 1-h[j] but also <= max_delta
    lb = np.maximum(-h, -max_delta)
    ub = np.minimum(1.0 - h, max_delta)
    bounds = list(zip(lb, ub))
    bounds.append((0, None))  # C >= 0
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='hihops')
    
    if res.success:
        delta = res.x[:n]
        C_new = res.x[n]
        return delta, C_new
    return None, None

import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot > 1e-14:
            h = h + diff * room / tot
    else:
        if s > 1e-14:
            h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def build_grad(h, n):
    num_k = 2 * n - 1
    A = np.zeros((num_k, n))
    for k_idx in range(num_k):
        k = k_idx - (n - 1)
        if k >= 0:
            A[k_idx, k:] = 1.0 - h[:n - k]
            if n > k:
                A[k_idx, :n - k] -= h[k:]
        else:
            A[k_idx, :n + k] = 1.0 - h[-k:]
            if n > -k:
                A[k_idx, -k:] -= h[:n + k]
    return A

def lp_direction(h, n, max_delta=0.3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr * dx
    num_k = 2 * n - 1
    A = build_grad(h, n)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((num_k, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = -c_vals
    lb = np.maximum(-h, -max_delta)
    ub = np.minimum(1.0 - h, max_delta)
    bounds = tuple(zip(lb, ub)) + ((0.0, None),)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return np.zeros(n)

def line_search(h, n, direction, start_alpha=1.0):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = start_alpha
    for _ in range(14):
        h_try = project_h(h + alpha * direction, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_h = h_try
            best_c5 = c5_try
            alpha *= 0.75
        else:
            alpha *= 0.5
            if alpha < 1e-5:
                break
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=40):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng_cd = np.random.RandomState(999)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in rng_cd.permutation(n):
            if time.time() > t_end:
                return best_h, best_c5
            h_base = best_h.copy()
            h_base[i] = 0.0
            corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
            b = np.zeros(2 * n - 1)
            for k_idx in range(2 * n - 1):
                k = k_idx - (n - 1)
                bk = 1.0
                j1 = i - k
                j2 = i + k
                if 0 <= j1 < n:
                    bk -= h_base[j1]
                if 0 <= j2 < n:
                    bk -= h_base[j2]
                b[k_idx] = bk
            a = corr_base
            try:
                def f1d(v, a=a, b=b):
                    return np.max(a + v * b)
                res_cs = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                         options={'xatol': 1e-12})
                v_opt = float(res_cs.x)
            except Exception:
                v_opt = best_h[i]
            if abs(v_opt - best_h[i]) > 1e-10:
                best_h[i] = v_opt
                best_h = project_h(best_h, n)
                c5_try = compute_c5(best_h)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    improved = True
                else:
                    best_h[i] = 0.0
                    best_h = project_h(best_h, n)
                    best_h[i] = v_opt
                    best_h = project_h(best_h, n)
        if not improved:
            break
    return best_h, best_c5

def slsqp_smooth(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(T * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def slsqp_constraint(x, n):
    return x.sum() - n / 2.0

def slsqp_step(h, n, T=5000, maxiter=400):
    try:
        res = minimize(slsqp_smooth, h, args=(T, n), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': slsqp_constraint, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success:
            return project_h(np.clip(res.x, 0, 1), n)
    except Exception:
        pass
    return h

def make_perturbation(n, cool, rng):
    mod = rng.randint(0, 5)
    if mod == 0:
        bs = rng.randint(4, max(5, n // 8))
        st = rng.randint(0, n - bs)
        p = np.zeros(n)
        p[st:st + bs] = (0.05 + 0.15 * cool) * rng.standard_normal(bs)
    elif mod == 1:
        freq = rng.randint(2, max(3, n // 10))
        ph = rng.uniform(0, 2.0 * np.pi)
        p = (0.04 + 0.12 * cool) * np.sin(2.0 * np.pi * freq * np.arange(n) / n + ph)
    elif mod == 2:
        np_ = rng.randint(4, max(5, n // 6))
        idx = rng.choice(n, np_, replace=False)
        p = np.zeros(n)
        p[idx] = (0.1 + 0.3 * cool) * rng.standard_normal(np_)
    elif mod == 3:
        p = (0.03 + 0.08 * cool) * rng.standard_normal(n)
    else:
        half = n // 2
        p = np.zeros(n)
        p[:half] = 0.1 * cool * rng.standard_normal(half)
        p[half:] = -0.1 * cool * rng.standard_normal(n - half)
    return p

def lp_descent_loop(h, n, t_end, max_delta=0.3, max_iters=500):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(max_iters):
        if time.time() > t_end:
            break
        delta = lp_direction(best_h, n, max_delta=max_delta)
        if delta is None or np.max(np.abs(delta)) < 1e-10:
            break
        h_new, c5_new = line_search(best_h, n, delta, start_alpha=1.0)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
        else:
            h_new, c5_new = line_search(best_h, n, delta, start_alpha=0.1)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
            else:
                delta2 = lp_direction(best_h, n, max_delta=0.15)
                if delta2 is not None and np.max(np.abs(delta2)) > 1e-10:
                    h_new, c5_new = line_search(best_h, n, delta2, start_alpha=0.5)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h_new
                    else:
                        break
                else:
                    break
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iter, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.7 * (it / max(n_iter, 1))
        pert = make_perturbation(n, cool, rng)
        h_pert = project_h(work_h + pert, n)
        h_pert, c5_pert = lp_descent_loop(h_pert, n, t_end, max_delta=0.25, max_iters=30)
        h_pert, c5_pert = coord_descent(h_pert, n, min(t_end, time.time() + 8), max_sweeps=4)
        temp = 0.002 * cool + 0.0003
        if c5_pert < work_c5 or rng.rand() < np.exp(-(c5_pert - work_c5) / max(temp, 1e-9)):
            work_h = h_pert
            work_c5 = c5_pert
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h
    return best_h, best_c5
