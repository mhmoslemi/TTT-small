def compute_c5(h):
    N = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode="full"))) * (2.0 / N)

def compute_contrib(h, N):
    L = 2 * N - 1
    s = np.arange(-(N - 1), N)
    p = np.arange(N)[:, None]
    ps = p + s[None, :]
    valid_a = (ps >= 0) & (ps < N)
    h_ps = np.where(valid_a, h[np.clip(ps, 0, N - 1)], 1.0)
    pm = p - s[None, :]
    valid_b = (pm >= 0) & (pm < N)
    h_pm = np.where(valid_b, h[np.clip(pm, 0, N - 1)], 0.0)
    return (1.0 - h_ps) * valid_a + h_pm * valid_b

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if float(np.clip(h - mid, 0.0, 1.0).sum()) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def ensure_count(h, N):
    nh = N // 2
    h = h.copy()
    n1 = int(np.sum(h > 0.5))
    if n1 > nh:
        idx = np.where(h > 0.5)[0]
        idx = idx[np.random.choice(len(idx), n1 - nh, replace=False)]
        h[idx] = 0
    elif n1 < nh:
        idx = np.where(h <= 0.5)[0]
        idx = idx[np.random.choice(len(idx), nh - n1, replace=False)]
        h[idx] = 1
    return (h > 0.5).astype(float)

def swap_search(h, N, deadline, top_k=40, max_sweeps=500, escape_prob=0.2):
    h = ensure_count(h, N)
    c = fft_corr(h)
    contrib = compute_contrib(h, N)
    best_val = float(c.max())
    best_h = h.copy()
    stall = 0
    
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        P = float(c.max())
        if P < best_val - 1e-12:
            best_val = P
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 80:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        top_idx = np.argsort(c)[-top_k:]
        c_top = c[top_idx]
        ct = contrib[:, top_idx]
        
        best_new = P
        best_p = -1
        best_q = -1
        
        for p in ones:
            base = c_top - ct[p]
            cands = (base[None, :] + ct[zeros, :]).max(axis=1)
            mi = int(np.argmin(cands))
            if cands[mi] < best_new:
                best_new = float(cands[mi])
                best_p = p
                best_q = int(zeros[mi])
        
        if best_p >= 0 and best_new < P - 1e-12:
            h[best_p], h[best_q] = 0.0, 1.0
        elif np.random.random() < escape_prob:
            p = int(ones[np.random.randint(len(ones))])
            q = int(zeros[np.random.randint(len(zeros))])
            h[p], h[q] = 0.0, 1.0
            c_new = fft_corr(h)
            if float(c_new.max()) > best_val + 0.005:
                h[p], h[q] = 1.0, 0.0
                continue
        else:
            break
        
        c = fft_corr(h)
        contrib = compute_contrib(h, N)
    
    return best_h, best_val * (2.0 / N)

def nesterov_opt(h, N, deadline, k=50, lr0=0.018):
    h = h.astype(float).copy()
    target = N / 2.0
    dl = dx = 2.0 / N
    best_h = h.copy()
    best_pk = float(np.max(fft_corr(h)))
    m = np.zeros(N)
    lr = lr0
    beta = 0.92
    it = 0
    while time.time() < deadline:
        y = h + m
        corr = fft_corr(y)
        pk = float(np.max(corr))
        if pk < best_pk:
            best_pk = pk
            best_h = y.copy()
        top = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(N)
        center = N - 1
        for ai in top:
            L = int(ai - center)
            if L == 0:
                grad += (1.0 - 2.0 * y)
            elif L > 0:
                grad[L:] += (1.0 - y[:N - L])
                grad[:N - L] -= y[L:]
            else:
                M = -L
                if M < N:
                    grad[:N - M] += (1.0 - y[M:])
                    grad[M:] -= y[:N - M]
        grad /= k
        h_new = project_sum(y - lr * grad, target)
        m = beta * (h_new - h)
        h = h_new
        it += 1
        if it % 150 == 0:
            lr *= 0.994
    return best_h, best_pk * dl

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    np.random.seed(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    N0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Core swap-based local search
    d1 = min(deadline - 250, t0 + 280)
    h_bin = (h0 > 0.5).astype(float)
    h_s, c5_s = swap_search(h_bin, N0, d1, top_k=50, max_sweeps=2000, escape_prob=0.15)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s.copy()
    
    # Phase 2: Nesterov continuous polish
    if time.time() < deadline - 180:
        d2 = min(deadline - 130, time.time() + 100)
        hc = best_h.astype(float) + np.random.randn(N0) * 0.04
        hc = project_sum(hc, N0 / 2.0)
        hn, c5n = nesterov_opt(hc, N0, d2, k=60, lr0=0.012)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
    
    # Phase 3: Extended swap search with larger top_k
    if time.time() < deadline - 120:
        d3 = min(deadline - 80, time.time() + 130)
        h_bin3 = (best_h > 0.5).astype(float)
        h_s3, c5_3 = swap_search(h_bin3, N0, d3, top_k=60, max_sweeps=1500, escape_prob=0.1)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h_s3.copy()
    
    # Phase 4: Multi-resolution exploration
    x_best = np.linspace(0, 2, len(best_h), endpoint=False)
    for N_try in [360, 400, 440, 500, 600, 700]:
        if time.time() > deadline - 50:
            break
        if N_try % 2 != 0:
            N_try += 1
        x_new = np.linspace(0, 2, N_try, endpoint=False)
        h_interp = np.interp(x_new, x_best, best_h)
        h_bt = (h_interp > 0.5).astype(float)
        h_bt = ensure_count(h_bt, N_try)
        sub_dl = min(deadline - 20, time.time() + 35)
        h_rt, c5_rt = swap_search(h_bt, N_try, sub_dl, top_k=35, max_sweeps=400, escape_prob=0.2)
        if c5_rt < best_c5:
            best_c5, best_h = c5_rt, h_rt.copy()
            x_best = np.linspace(0, 2, N_try, endpoint=False)
    
    # Phase 5: Nesterov refinement at best resolution
    if time.time() < deadline - 15:
        hF = best_h.astype(float) + np.random.randn(len(best_h)) * 0.025
        hF = project_sum(hF, len(best_h) / 2.0)
        hF, c5F = nesterov_opt(hF, len(best_h), deadline - 8, k=80, lr0=0.008)
        if c5F < best_c5:
            best_c5, best_h = c5F, hF
    
    # Phase 6: Final binary swap polish
    if time.time() < deadline - 2:
        hF2 = (best_h > 0.5).astype(float)
        hF2, c5F2 = swap_search(hF2, len(best_h), deadline - 0.5, top_k=50, max_sweeps=200, escape_prob=0.0)
        if c5F2 < best_c5:
            best_c5, best_h = c5F2, hF2
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
