def subgradient_direction(h, n, top_k=5):
    """Return a direction that decreases the worst lags."""
    corr = get_corr(h)
    n_lags = len(corr)
    top_lags = np.argsort(corr)[-top_k:]
    # For each top lag, compute the subgradient (column of Jacobian)
    # d corr[m]/d h[j] = (1-h[j+k])[valid] - h[j-k][valid] where k=m-(n-1)
    # We want to decrease corr[m] for the top lags
    # Direction: -sum over top lags of subgradient
    # But we need to be careful: the subgradient for the max is the subgradient
    # of the function achieving the max.
    # For a weighted combination: direction = -sum w_m * sgrad_m
    weights = np.zeros(n_lags)
    weights[top_lags] = 1.0 / top_k
    # Compute -sum_m weights[m] * d corr[m]/d h[j] for all j
    k_vals = np.arange(-(n-1), n)  # k = m - (n-1) for m in [0, 2n-1]
    # For efficiency, compute the sum directly
    # d corr[m]/d h[j] = (1-h[j+k_m]) * [0<=j+k_m<n] - h[j-k_m] * [0<=j-k_m<n]
    # where k_m = m - (n-1)
    # sum_m w_m * d corr[m}/d h[j] = sum_m w_m * [(1-h[j+k_m])[valid] - h[j-k_m][valid]]
    # This is like a weighted sum of shifted versions of (1-h) and -h
    # = (W * (1-h))[j] - (W * h)[j-(shifted)]
    # where W is the weighted convolution
    # More precisely:
    # S[j] = sum_m w_m * (1-h[j+k_m]) * [0<=j+k_m<n]
    #      - sum_m w_m * h[j-k_m] * [0<=j-k_m<n]
    # Let's compute this via convolution
    # First term: for each m, (1-h) shifted by -k_m, weighted by w_m, summed
    # This is: sum_m w_m * (1-h)[::-1 shifted]... it's getting complicated
    # Let me just do it directly for n=160 (O(n * top_k * n) = O(n^2 * top_k) ~ 160^2*5 ~ 128k)
    grad = np.zeros(n)
    for m in top_lags:
        k = m - (n - 1)
        for j in range(n):
            if 0 <= j + k < n:
                grad[j] += -w * (1.0 - h[j + k])  # negative for descent
            if 0 <= j - k < n:
                grad[j] += -w * (-h[j - k])  # = +w*h[j-k]
    # Hmm, this is O(n * top_k) which is fine
    # Actually wait, let me redo.
    # We want to MINIMIZE corr[m] for the top lags.
    # d corr[m]/d h[j] = (1-h[j+k])[0<=j+k<n] + (-h[j-k])[0<=j-k<n]
    # Descent direction: -d corr[m]/d h[j]
    # = -(1-h[j+k])[0<=j+k<n] + h[j-k][0<=j-k<n]
    
    grad = np.zeros(n)
    w_k = 1.0 / top_k
    for m in top_lags:
        k = m - (n - 1)
        for j in range(n):
            if 0 <= j + k < n:
                grad[j] += -w_k * (1.0 - h[j + k])
            if 0 <= j - k < n:
                grad[j] += w_k * h[j - k]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = np.clip(h + room * (diff / tot), 0, 1)
    else:
        if s < 1e-14:
            return h
        h = np.clip(h * (1.0 + diff / s), 0, 1)
    return h

def smooth_obj(x, T):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return (m + np.log(np.sum(np.exp(T * (corr - m))) + 1e-30) / T) * (2.0 / len(h))

def slsqp_step(h, T, n, maxiter=2000):
    def obj(x):
        return smooth_obj(x, T)
    def con_eq(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_single(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    k_vals = np.arange(-(n - 1), n)
    coefs = np.zeros(2 * n - 1)
    valid1 = (i + k_vals >= 0) & (i + k_vals < n)
    coefs[k_vals + (n - 1)][valid1] += (1.0 - h[i + k_vals[valid1]])
    valid2 = (i - k_vals >= 0) & (i - k_vals < n)
    coefs[k_vals + (n - 1)][valid2] += (-h[i - k_vals[valid2]])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:

                return best_h, best_c5

            if abs(hi_new - best_h[i]) < 1e-12:
                continue
            h_try = best_h.copy()

            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_c5

def subgrad_perturb(h, n, scale, top_k=3):
    corr = get_corr(h)
    top_lags = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    wk = 1.0 / top_k
    for m in top_lags:
        k = m - (n - 1)
        for j in range(n):
            if 0 <= j + k < n:
                grad[j] += -wk * (1.0 - h[j + k])
            if 0 <= j - k < n:
                grad[j] += wk * h[j - k]
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h
    direction = -grad / gn
    return project_h(h + scale * direction)
