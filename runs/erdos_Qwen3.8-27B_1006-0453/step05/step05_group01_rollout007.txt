def project_h(h):

    diff = target - s
    if abs(diff) < 1e-14:

        tot = room.sum()

            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_at_shifts(h, shifts):
    """Compute gradient of corr[k] with respect to h at given shifts."""
    n = len(h)
    dx = 2.0 / n
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    jp = i_idx + s_idx
    jm = i_idx - s_idx

    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)

    grad = ((1.0 - h[jpc]) * v1.astype(np.float64) - h[jmc] * v2.astype(np.float64)).mean(axis=1) * dx
    return grad

import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def grad_top_shifts(h, shifts):
    n = len(h)
    dx = 2.0 / n
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :].astype(int)
    jp = i_idx + s_idx
    jm = i_idx - s_idx
    v1 = (jp >= 0) & (jp < n)
    v2 = (jm >= 0) & (jm < n)
    jpc = np.clip(jp, 0, n - 1)
    jmc = np.clip(jm, 0, n - 1)
    g = ((1.0 - h[jpc]) * v1 - h[jmc] * v2).mean(axis=1) * dx
    return g

def gd_refine(h, n, t_end, n_iter=400):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    mom = np.zeros(n)
    beta = 0.4
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = max(1.0 - it / n_iter, 0.05)
        K = int(3 + 12 * cool)
        corr = compute_corr(best_h)
        top_idx = np.argsort(corr)[-K:]
        shifts = (top_idx - (n - 1)).astype(int)
        g = grad_top_shifts(best_h, shifts)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-10:
            break
        g = g / (norm_g + 1e-12)
        alpha = 0.004 * cool
        mom = beta * mom + (1.0 - beta) * g
        h_new = project_h(best_h - alpha * mom * (norm_g + 1e-12))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        else:
            alpha *= 0.85
            if alpha < 1e-6:
                break
    return best_h, best_c5

def coord_descent(h, n, t_end, max_sweeps=300, n_top=18):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    improved_total = 0
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        corr = compute_corr(best_h)
        top_idx = np.argsort(corr)[-n_top:]
        top_shifts = (top_idx - (n - 1)).astype(int)
        corr_top = corr[top_idx].copy()
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            coefs = np.zeros(n_top)
            for t in range(n_top):
                k = top_shifts[t]
                c = 0.0
                jp = i + k
                jm = i - k
                if 0 <= jp < n:
                    c += (1.0 - best_h[jp])
                if 0 <= jm < n:
                    c -= best_h[jm]
                coefs[t] = c
            base = corr_top - best_h[i] * coefs
            res = minimize_scalar(lambda t: np.max(base + t * coefs),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            t_new = float(res.x)
            if abs(t_new - best_h[i]) < 1e-14:
                continue
            h_try = best_h.copy()
            h_try[i] = t_new
            h_try = project_h(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try
                improved = True
                improved_total += 1
        if not improved:
            stall += 1
            if stall >= 4:
                break
        else:
            stall = 0
    return best_h, best_c5

def slsqp_step(h, n, T, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        if T > 500:
            return (cmax + np.log(np.sum(np.exp(T * (corr - cmax)))) / T) * dx
        return cmax * dx
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def perturb_h(h, n, scale, r):
    if r < 0.25:
        return h + scale * np.random.randn(n)
    elif r < 0.45:
        blen = np.random.randint(max(n // 15, 2), max(n // 5, 3))
        bs = np.random.randint(0, max(n - blen, 1))
        p = np.zeros(n)
        p[bs:bs + blen] = scale * 1.5 * np.random.randn(blen)
        return h + p
    elif r < 0.65:
        f = np.random.randint(1, 10)
        ph = np.random.uniform(0, 2 * np.pi)
        return h + scale * 0.8 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif r < 0.8:
        mask = np.random.random(n) < 0.15
        p = np.zeros(n)
        p[mask] = scale * 2.0 * np.random.randn(int(mask.sum()) + 1)
        return h + p
    else:
        corr = compute_corr(h)
        top_k = np.argsort(corr)[-4:]
        shifts = (top_k - (n - 1)).astype(int)
        p = np.zeros(n)
        for k in shifts:
            if k > 0 and n - k > 0:
                p[:n - k] += (h[k:n] - 0.5) * 0.3
            elif k < 0 and n + k > 0:
                p[-k:] += (h[:n + k] - 0.5) * 0.3
        return h + scale * 0.6 * p

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    else:
        init_h = np.zeros(200)
        init_h[:100] = 1.0

    n = len(init_h)
    best_h = project_h(init_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup
    t1 = t0 + budget_s * 0.15
    if t_end - time.time() > 20:
        for T in [100, 1000, 10000]:
            if time.time() > t1 - 3:
                break
            hc = slsqp_step(best_h, n, T, 300)
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc

    # Phase 2: Gradient descent refinement
    t2 = t0 + budget_s * 0.35
    if t_end - time.time() > 15:
        best_h, best_c5 = gd_refine(best_h, n, t2, 300)

    # Phase 3: Coordinate descent (main workhorse)
    t3 = t0 + budget_s * 0.60
    if t_end - time.time() > 10:
        best_h, best_c5 = coord_descent(best_h, n, t3, 250, 16)

    # Phase 4: Basin hopping
    t4 = t0 + budget_s * 0.82
    n_basins = 10
    for b in range(n_basins):
        if time.time() > t4:
            break
        scale = 0.04 + 0.04 * (b / n_basins)
        h_pert = project_h(perturb_h(best_h, n, scale, np.random.rand()))
        t_sub = min(time.time() + 40, t4)
        h_sub, c5_sub = gd_refine(h_pert, n, t_sub, 60)
        h_sub, c5_sub = coord_descent(h_sub, n, min(time.time() + 30, t4), 80, 14)
        if c5_sub < best_c5:
            best_h, best_c5 = h_sub, c5_sub

    # Phase 5: Final intensive coordinate descent
    if t_end - time.time() > 15:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 15, 300, 20)

    # Phase 6: Final SLSQP polish
    if t_end - time.time() > 20:
        hc = slsqp_step(best_h, n, 50000, 800)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_h, best_c5 = hc, c5v

    # Phase 7: Absolute final coordinate descent
    if t_end - time.time() > 8:
        best_h, best_c5 = coord_descent(best_h, n, t_end - 1, 200, 25)

    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
