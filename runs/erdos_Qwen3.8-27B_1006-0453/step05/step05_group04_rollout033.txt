def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()

            mask = h > 0
            h[mask] *= target / s
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_lin(h, i0, n):

    lin = np.zeros(2 * n - 1)
    rev_1mh = (1.0 - h)[::-1]

    lin[n - 1 - i0:2 * n - 1 - i0] -= h

def fast_cd(h, n, t_end, max_sweeps, seed=42):

    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 5 == 4 else np.arange(n)

            if time.time() > t_end:
                return best_h, best_c5
            old = h[i0]
            if old == 0.0 or old == 1.0:
                # Still can move, just note
                pass
            
            lin = build_lin(h, i0, n)

            # over δ such that 0 ≤ old+δ ≤ 1
            lo, hi = max(0 - old, -1e-15), min(1 - old, 1 + 1e-15)
            if lo >= hi:
                continue
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # ~0.618
            a, b = lo, hi
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            
            def eval_f(delta):
                nc = corr + delta * lin
                nc[n-1] -= delta * delta
                return np.max(nc) * dx
            
            fc = eval_f(c_)
            fd = eval_f(d_)
            
            for _ in range(40):  # ~40 iterations for good precision
                if fc < fd:
                    b = d_
                    d_ = c_
                    fd = fc
                    c_ = b - gr * (b - a)
                    fc = eval_f(c_)
                else:
                    a = c_
                    c_ = d_
                    fc = fd
                    d_ = a + gr * (b - a)
                    fd = eval_f(d_)
            
            delta = (a + b) / 2
            new_val = old + delta
            new_val = np.clip(new_val, 0.0, 1.0)
            delta = new_val - old
            if abs(delta) < 1e-14:
                continue
            
            corr += delta * lin
            corr[n-1] -= delta * delta
            h[i0] = new_val
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
            else:
                corr -= delta * lin
                corr[n-1] += delta * delta
                h[i0] = old
        if not improved:
            break
    
    return best_h, best_c5

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
        if s > target:
            h *= target / s
            h = np.minimum(h, 1.0)
        else:
            deficit = target - s
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-12:
                break
            h = h + room * (deficit / tot)
            h = np.minimum(h, 1.0)
            # Re-clip zeros if needed
            h = np.maximum(h, 0.0)
    # Final exact projection
    s = h.sum()
    if abs(s - target) > 1e-10:
        # Distribute remaining evenly among non-saturated points
        free = (h > 0.001) & (h < 0.999)
        if free.any():
            adj = (target - s) / free.sum()
            h[free] += adj
            h = np.clip(h, 0, 1)

    rev_1mh = (1.0 - h)[::-1]
    lin[i0:i0 + n] += rev_1mh
    lin[n - 1 - i0:2 * n - 1 - i0] -= h
    return lin

def minimize_1d(corr, lin, n, dx, lo, hi):

    # Evaluate at midpoint and endpoints to bracket
    # Then golden section
    best_d = 0.0
    best_f = float('inf')
    
    def f(d):
        nc = corr + d * lin
        nc[n-1] -= d * d
        return np.max(nc) * dx
    
    # Coarse grid
    pts = np.linspace(lo, hi, 11)
    for p in pts:
        val = f(p)
        if val < best_f:
            best_f = val
            best_d = p
    
    # Refine with golden section around best
    span = (hi - lo) / 10.0
    a = max(lo, best_d - span)
    b = min(hi, best_d + span)

        c_ = b - gr * (b - a)

                c_ = d_
                fc = fd
                d_ = a + gr * (b - a)
                fd = f(d_)

                corr[n-1] -= delta_actual * delta_actual

                corr[n-1] += delta * delta
                h[i0] = old
        
        if not improved:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s < target:
            deficit = target - s
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += room * (deficit / tot)
        else:
            excess = s - target
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= (target / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def build_lin(h, i0, n):
    lin = np.zeros(2 * n - 1)
    rev_1mh = (1.0 - h)[::-1]
    lin[i0:i0 + n] += rev_1mh
    lin[n - 1 - i0:2 * n - 1 - i0] -= h
    return lin

def minimize_1d(corr, lin, n, dx, lo, hi):
    def f(d):
        nc = corr + d * lin
        nc[n - 1] -= d * d
        return np.max(nc) * dx
    best_d = 0.0
    best_f = float('inf')
    pts = np.linspace(lo, hi, 9)
    for p in pts:
        v = f(p)
        if v < best_f:
            best_f = v
            best_d = p
    span = (hi - lo) / 8.0
    a = max(lo, best_d - span)
    b = min(hi, best_d + span)
    if b - a > 1e-12:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c_ = b - gr * (b - a)
        d_ = a + gr * (b - a)
        fc = f(c_)
        fd = f(d_)
        for _ in range(35):
            if fc < fd:
                b = d_
                d_ = c_
                fd = fc
                c_ = b - gr * (b - a)
                fc = f(c_)
            else:
                a = c_
                c_ = d_
                fc = fd
                d_ = a + gr * (b - a)
                fd = f(d_)
        best_d = (a + b) / 2.0
        best_f = f(best_d)
    return best_d, best_f

def fast_cd(h, n, t_end, max_sweeps, seed=42, pair_prob=0.15):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    nh2 = 2 * n - 1
    h = h.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i0 in order:
            if time.time() > t_end:
                break
            if rng.rand() < pair_prob and n > 1:
                j0 = rng.randint(0, n)
                if j0 == i0:
                    j0 = (j0 + 1) % n
                lo_p = max(-h[i0], h[j0] - 1.0)
                hi_p = min(1.0 - h[i0], h[j0])
                if hi_p - lo_p < 1e-13:
                    continue
                lin_i = build_lin(h, i0, n)
                lin_j = build_lin(h, j0, n)
                dcorr = lin_i - lin_j
                def fp(d):
                    nc = corr + d * dcorr
                    nc[n - 1] -= 2.0 * d * d
                    return np.max(nc) * dx
                best_dp = 0.0
                best_fp = float('inf')
                for p in np.linspace(lo_p, hi_p, 7):
                    v = fp(p)
                    if v < best_fp:
                        best_fp = v
                        best_dp = p
                if abs(best_dp) < 1e-13:
                    continue
                corr += best_dp * dcorr
                corr[n - 1] -= 2.0 * best_dp * best_dp
                h[i0] = np.clip(h[i0] + best_dp, 0, 1)
                h[j0] = np.clip(h[j0] - best_dp, 0, 1)
                c5_n = np.max(corr) * dx
                if c5_n < best_c5 - 1e-15:
                    best_c5 = float(c5_n)
                    best_h = h.copy()
                    improved = True
                    stall = 0
                else:
                    corr -= best_dp * dcorr
                    corr[n - 1] += 2.0 * best_dp * best_dp
                    h[i0] = np.clip(h[i0] - best_dp, 0, 1)
                    h[j0] = np.clip(h[j0] + best_dp, 0, 1)
            else:
                old = h[i0]
                lo_s = -old
                hi_s = 1.0 - old
                if hi_s - lo_s < 1e-13:

                    stall = 0
                else:

                    corr[n - 1] += da * da
                    h[i0] = old
        if not improved:
            stall += 1
            if stall >= 4:
                break

def soft_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.clip(T * (corr - m), -50.0, 50.0)
    wts = np.exp(exps)
    wts /= wts.sum()
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = wts[m_idx]
        if w < 1e-12:
            continue
        s = m_idx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

def basin_hop(h0, n, t_end, n_hops, seed=42):
    rng = np.random.RandomState(seed)
    bh = h0.copy()
    bc5 = compute_c5(bh)
    ch = h0.copy()
    cc5 = bc5
    best_c5_hist = [bc5]
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        st = 0.02 + 0.08 * cool
        mode = hop % 6
        if mode == 0:
            pert = st * rng.randn(n)
        elif mode == 1:
            L = rng.randint(4, 25)
            s0 = rng.randint(0, n - L)
            pert = np.zeros(n)
            pert[s0:s0 + L] = st * 2.0 * (rng.rand(L) > 0.5).astype(float) * 2 - st * 2
        elif mode == 2:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = st * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 3:
            idx = rng.choice(n, size=max(4, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = st * 1.5 * rng.randn(len(idx))
        elif mode == 4:
            m = int(0.05 * n * rng.rand() + 2)
            idx = rng.choice(n, size=m, replace=False)
            pert = np.zeros(n)
            pert[idx] = st * 3 * (2 * rng.rand(m) - 1)
        else:
            fj = rng.randint(1, 8)
            ph0 = rng.uniform(0, 2 * np.pi)
            ph1 = rng.uniform(0, 2 * np.pi)
            pert = st * (np.cos(2 * np.pi * fj * np.arange(n) / n + ph0)
                         + 0.5 * np.sin(2 * np.pi * (fj + 1) * np.arange(n) / n + ph1))
        hp = project_h(ch + pert, n)
        hop_t = min(t_end, time.time() + max(1.5, 6.0 * cool + 1.0))
        ho, co = fast_cd(hp, n, hop_t, max_sweeps=60, seed=seed + hop * 7)
        temp = 0.003 * cool + 0.0003
        if co < cc5 or (co > cc5 and rng.rand() < np.exp(-(co - cc5) / temp)):
            ch = ho
            cc5 = co
            if co < bc5:
                bc5 = co
                bh = ho.copy()
                best_c5_hist.append(bc5)
    return bh, bc5
