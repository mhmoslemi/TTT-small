def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    if T < 1e-10:
        return m
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    dx = 2.0 / n

def project_h(h):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:

            return h
        h = h * (1.0 + diff / s)

def upsample(h, factor):
    """Upsample h by factor using linear interpolation then constrain."""
    n_old = len(h)
    n_new = n_old * factor
    # Linear interpolation
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new)

def downsample(h, factor):

    n_old = len(h)

    h_blocks = h[:n_new * factor].reshape(n_new, factor)
    h_new = h_blocks.mean(axis=1)
    return project_h(h_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, T):
    corr = get_corr(h)
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    n = len(h)
    return val * (2.0 / n)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def upsample(h, factor):
    n_old = len(h)
    n_new = n_old * factor
    xo = np.linspace(0, 1, n_old)
    xn = np.linspace(0, 1, n_new)
    return project_h(np.interp(xn, xo, h))

def slsqp_opt(h, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        return smooth_c5(x, T)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h

def coord_opt_1d(h, n, i):
    hb = h.copy()
    hb[i] = 0.0
    cb = get_corr(hb)
    mv = np.arange(-(n - 1), n)
    v1 = i - mv; v2 = i + mv
    ok1 = (v1 >= 0) & (v1 < n); ok2 = (v2 >= 0) & (v2 < n)
    coef = np.zeros(2 * n - 1)
    coef[ok1] += (1.0 - h[np.clip(v1[ok1], 0, n - 1)])
    coef[ok2] += (-h[np.clip(v2[ok2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(cb + m1 * coef) < np.max(cb + m2 * coef):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=100):
    bh = h.copy()
    bc5 = compute_c5(bh)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in np.random.permutation(n):
            if time.time() > t_end:
                return bh, bc5
            hv = coord_opt_1d(bh, n, i)
            if abs(hv - bh[i]) < 1e-11:
                continue
            ht = bh.copy(); ht[i] = hv
            ht = project_h(ht)
            cv = compute_c5(ht)
            if cv < bc5 - 1e-14:
                bc5 = cv; bh = ht; imp = True
        if not imp:
            break
    return bh, bc5

def subgrad_step(h, n, step):
    corr = get_corr(h)
    c5v = np.max(corr)
    active = np.where(corr >= c5v - 1e-5)[0]
    na = len(active)
    if na == 0:
        return project_h(h), c5v
    grad = np.zeros(n)
    for k in active:
        off = k - (n - 1)
        j = np.arange(n)
        a1 = (j + off >= 0) & (j + off < n)
        a2 = (j - off >= 0) & (j - off < n)
        grad[a1] += 1.0 - h[np.clip(j[a1] + off, 0, n - 1)]
        grad[a2] += 1.0 - h[np.clip(j[a2] - off, 0, n - 1)]
    grad /= na
    return project_h(h - step * grad), c5v * (2.0 / n)
