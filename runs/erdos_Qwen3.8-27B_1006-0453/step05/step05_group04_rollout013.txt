from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} via bisection on lambda."""
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.max() - target, h.min()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def soft_max_gradient(h, n, tau):
    """Compute gradient of soft-max approximation of C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft-max weights
    shifted = corr - corr.max()
    w = np.exp(shifted / tau)
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        k = k_idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            a = -k
            g[:n-a] += (1.0 - h[a:])
            g[a:] -= h[:n-a]
        grad += wk * g
    return dx * grad, float(corr.max() * dx)

def soft_max_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp((corr - m) / tau)
