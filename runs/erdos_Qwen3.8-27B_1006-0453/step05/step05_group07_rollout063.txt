    h = h.copy()

    tgt = n * 0.5

    else:

from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = 0.5 * n
    d = tgt - h.sum()
    if abs(d) < 1e-14:
        return h
    if d > 0:
        w = 1.0 - h
        sw = w.sum()
        h += w * (d / sw) if sw > 1e-14 else d / n
    else:
        h += h * (d / h.sum()) if h.sum() > 1e-14 else d / n
    h = np.clip(h, 0.0, 1.0)
    d2 = tgt - h.sum()
    if abs(d2) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10) if d2 < 0 else (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            mask = h < 1.0 if d2 > 0 else h > 0.0
        if mask.sum() > 0:
            h[mask] += d2 / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    d3 = tgt - h.sum()
    if abs(d3) > 1e-15:
        idx = np.argmin(h) if d3 > 0 else np.argmax(h)
        h[idx] += d3
    return h

# Globals for L-BFGS objective
_GT = [1.0]

def _obj(x):
    T = _GT[0]
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def _grad(x):
    T = _GT[0]
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    e = np.exp(T * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        off = k - n + 1
        if off >= 0:
            g[:n-off] += wk * (1.0 - x[off:])
            g[off:] -= wk * x[:n-off]
        else:
            op = -off
            g[op:] += wk * (1.0 - x[:n-op])
            g[:n-op] -= wk * x[op:]
    return g * 2.0 / n

def lbfgs(h, T, n, iters=500):
    _GT[0] = T
    try:
        r = minimize(h, _obj, _grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'maxls': 50, 'ftol': 1e-20, 'gtol': 1e-12, 'factr': 1e10})
        return proj_h(r.x, n)
    except:
        return h

def coord_desc(h, n, t_end, max_sweeps=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode='full')
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            # Compute coefficient of h[i] in each c[k]
            coef = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                off = k - n + 1
                if off >= 0:
                    if 0 <= i < n - off:
                        coef[k] += (1.0 - best[i + off])
                    if off <= i < n:
                        coef[k] -= best[i - off]
                else:
                    op = -off
                    if op <= i < n:
                        coef[k] += (1.0 - best[i - op])
                    if 0 <= i < n - op:
                        coef[k] -= best[i + op]
            # 1D minimization: min_b max_k (corr[k] - old*coef[k] + b*coef[k]) * 2/n
            c_base = corr - old * coef
            # Use minimize_scalar
            def f1d(b):
                return float(np.max(c_base + b * coef) * 2.0 / n)
            r = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-14})
            nv = float(r.x)

def binary_local_search(h, n, t_end, max_iters=200000):
    """Local search on binary sequences via pair swaps"""
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    # Ensure exactly n/2 ones
    if len(ones) > n // 2:
        rm = np.random.choice(ones, len(ones) - n//2, replace=False)
        h_b[rm] = 0
        ones = np.where(h_b == 1)[0]
    elif len(ones) < n // 2:
        add = np.random.choice(zeros, n//2 - len(ones), replace=False)
        h_b[add] = 1
        ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(corr.max() * dx)
    best_h = h_b.copy()
    T_sa = 0.01
    for it in range(max_iters):
        if time.time() > t_end:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # Swap: h[i] 1→0, h[j] 0→1
        h_b[i], h_b[j] = 0, 1
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(nc.max() * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T_sa):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i], h_b[j] = 1, 0
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
        T_sa = max(0.0003, T_sa * 0.99998)
    return best_h, best_c5
