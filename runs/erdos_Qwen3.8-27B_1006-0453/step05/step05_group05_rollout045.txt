```python
    weights = np.exp(T * (corr - m)) / np.sum(np.exp(T * (corr - m)))
    # grad of corr[k] w.r.t. h
    # Need to compute weighted sum of gradients
    # For efficiency, compute the gradient of each corr[k] as a function of h
    # Then take weighted sum
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        # corr[k] = sum_i h[(i+k-n+1)] * (1-h[i]) for valid i
        # Let's compute d(corr[k])/dh[j]
        shift = k - (n - 1)
        # corr[k] = sum_i h[i] * (1-h[i-shift]) for valid (i, i-shift)
        # d/dh[j] = (1-h[j-shift]) [from h[j] term] + sum_i: i-shift=j => -h[i] = -h[j+shift]
        for j in range(n):
            idx = j - shift
            if 0 <= idx < n:
                grad[j] += w * (1.0 - h[idx])
            idx2 = j + shift
            if 0 <= idx2 < n:
                grad[j] += w * (-h[idx2])
    return grad
```

**Key Insight**: The problem can be viewed as: minimize the maximum over lags of a quadratic form. This is a min-max problem. We can use a bilevel approach:

- Inner: check if there exists h with all correlation lags ≤ C₅/dx

But the inner problem is still non-convex (quadratic constraints).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, T):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(T * (corr - m)))) / T)

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h0, T, n, maxiter=1000, ftol=1e-15):
    def obj(x):
        corr = get_corr(x)
        return smooth_max_val(corr, T) * (2.0 / n)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(res.x, 0, 1))
    except Exception:
        return h0.copy()

def coord_opt_single(h, n, i, niter=50):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = get_corr(h_base)
    m_vals = np.arange(-(n - 1), n)
    vm1 = i - m_vals
    vm2 = i + m_vals
    v1 = (vm1 >= 0) & (vm1 < n)
    v2 = (vm2 >= 0) & (vm2 < n)
    coefs = np.zeros(2 * n - 1)
    coefs[v1] += (1.0 - h[np.clip(vm1[v1], 0, n - 1)])
    coefs[v2] += (-h[np.clip(vm2[v2], 0, n - 1)])
    lo, hi = 0.0, 1.0
    for _ in range(niter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(corr_base + m1 * coefs) < np.max(corr_base + m2 * coefs):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descent(h, n, t_end, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            hi_new = coord_opt_single(best_h, n, i)
            if abs(hi_new - best_h[i]) > 1e-10:
                ht = best_h.copy()
                ht[i] = hi_new
                ht = project_h(ht)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t
                    best_h = ht
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def subgradient_refine(h, n, n_iter, t_limit):
    work = h.copy()
    best_c = compute_c5(work)
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        corr = get_corr(work)
        k_star = np.argmax(corr)
        shift = k_star - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 1.0
            idx1 = j - shift
            idx2 = j + shift
            if 0 <= idx1 < n:
                g -= work[idx1]
            if 0 <= idx2 < n:
                g -= work[idx2]
            grad[j] = g
        cool = 1.0 - 0.6 * (it / max(n_iter, 1))
        step = (0.003 + 0.015 * cool) / (np.max(np.abs(grad)) + 1e-10)
        h_new = project_h(work - step * grad)
        T_ref = int(2000 * cool + 200)
        h_new = slsqp_smooth(h_new, T_ref, n, maxiter=150, ftol=1e-13)
        c5v = compute_c5(h_new)
        temp = 0.0003 * cool + 0.00005
        if c5v < best_c or np.random.rand() < np.exp(-(c5v - best_c) / temp):
            work = h_new
            best_c = c5v
    return work, best_c

def basin_hop(h, n, t_start, t_end, n_iter):
    work = h.copy()
    wc = compute_c5(work)
    bc, bh = compute_c5(h), h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        cool = 1.0 - 0.8 * (it / max(n_iter, 1))
        mod = it % 7
        if mod == 0:
            p = (0.01 + 0.05 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 14)
            ph = np.random.uniform(0, 6.28)
            p = (0.015 + 0.05 * cool) * np.sin(2*np.pi*f*np.arange(n)/n + ph)
        elif mod == 2:
            msk = np.random.random(n) < 0.2
            p = np.zeros(n)
            p[msk] = (0.04 + 0.1 * cool) * np.random.randn(msk.sum())
        elif mod == 3:
            blk = max(1, n // 5)
            b = np.random.randint(0, n // blk)
            p = np.zeros(n)
            p[b*blk:(b+1)*blk] = (0.02 + 0.05*cool) * np.random.randn(blk)
        elif mod == 4:
            p = np.zeros(n)
            for _ in range(4):
                a, b2 = np.random.permutation(n)[:2]
                d = (work[b2]-work[a])*(0.3+0.5*cool)
                p[a] += d; p[b2] -= d
        elif mod == 5:
            f1, f2 = np.random.randint(1,6), np.random.randint(7,18)
            p = (0.01+0.03*cool)*(np.sin(2*np.pi*f1*np.arange(n)/n)+np.sin(2*np.pi*f2*np.arange(n)/n+0.5))
        else:
            msk = np.random.random(n) < 0.08
            p = np.zeros(n)
            p[msk] = (0.05+0.12*cool)*np.sign(np.random.randn(msk.sum()))
        hp = project_h(work + p)
        Tb = int(3000*cool + 500)
        hc = slsqp_smooth(hp, Tb, n, maxiter=300, ftol=1e-13)
        cv = compute_c5(hc)
        temp = 0.0004*cool + 0.00003
        if cv < wc or np.random.rand() < np.exp(-(cv-wc)/temp):
            work = hc
            wc = cv
            if cv < bc:
                bc, bh = cv, hc.copy()
    return bh, bc
```
