import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    if abs(s - tgt) < 1e-13:
        return h
    if s > tgt:
        h *= tgt / s
    else:
        rm = 1.0 - h
        t = rm.sum()
        if t > 1e-13:
            h += rm * ((tgt - s) / t)
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.01) & (h < 0.99)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _soft_obj(x, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * 2.0 / len(x)

def _soft_grad(x, T):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    w = np.exp(T * (c - m))
    w /= w.sum()
    corr1 = np.correlate(w, x, mode='full')
    corr2 = np.correlate(w[::-1], x, mode='full')
    g = (1.0 - corr1[:n]) - corr2[:n]
    return g * (2.0 / n)

def lbfgs_opt(h, T, n, iters=300):
    try:
        r = minimize(h, _soft_obj, _soft_grad, args=(T,),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-11})
        return proj_h(r.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, sweeps=60):
    best = h.copy()
    best_c5 = compute_c5(best)
    mf = 2 * n - 1
    for sw in range(sweeps):
        if time.time() > t_end:
            break
        imp = False
        corr_cache = np.correlate(best, 1.0 - best, mode='full')
        for i in range(n):
            if time.time() > t_end:
                return best, best_c5
            old = best[i]
            coef = np.zeros(mf)
            for k in range(mf):
                sh = k - n + 1
                j1 = i + sh
                if 0 <= j1 < n:
                    coef[k] += 1.0 - best[j1]
                j2 = i - sh
                if 0 <= j2 < n:
                    coef[k] -= best[j2]
            base = corr_cache - old * coef
            lo_b = 0.0 if old > 0 else 0.0
            hi_b = 1.0 if old < 1 else 1.0
            def f1d(v):
                return float(np.max(base + v * coef))
            r = minimize_scalar(f1d, bounds=(lo_b, hi_b), method='bounded',
                                options={'xatol': 1e-12})
            nv = float(r.x)
            if abs(nv - old) < 1e-13:
                continue
            best[i] = nv
            imp = True
        if imp:
            best = proj_h(best, n)
            best_c5 = compute_c5(best)
            corr_cache = np.correlate(best, 1.0 - best, mode='full')
        else:
            break
    return best, best_c5

def block_sa(h, n, t_end, max_iters=300000):
    h_b = (h > 0.5).astype(float)
    ones_idx = np.where(h_b == 1)[0]
    if len(ones_idx) > n // 2:
        h_b[np.random.choice(ones_idx, len(ones_idx) - n // 2, replace=False)] = 0
    elif len(ones_idx) < n // 2:
        zeros_idx = np.where(h_b == 0)[0]
        h_b[np.random.choice(zeros_idx, n // 2 - len(ones_idx), replace=False)] = 1
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h_b.copy()
    T = 0.02
    for it in range(max_iters):
        if time.time() > t_end:
            break
        done = False
        for _ in range(8):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if h_b[i] != h_b[j]:
                done = True
                break
        if not done:
            continue
        bs = np.random.choice([1, 2, 3])
        oi, oj = h_b[i], h_b[j]
        h_b[i:i+bs] = np.roll(h_b[i:i+bs], 1)
        h_b[j:j+bs] = np.roll(h_b[j:j+bs], -1)
        if h_b.sum() != n // 2:
            h_b[i:i+bs] = np.roll(h_b[i:i+bs], -1)
            h_b[j:j+bs] = np.roll(h_b[j:j+bs], 1)
            continue
        nc = np.correlate(h_b, 1.0 - h_b, mode='full')
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            corr = nc
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        else:
            h_b[i:i+bs] = np.roll(h_b[i:i+bs], -1)
            h_b[j:j+bs] = np.roll(h_b[j:j+bs], 1)
        T = max(0.0003, T * 0.99997)
    return best_h, best_c5

def basin_hop(h, n, t_end, n_iters=150):
    work = h.copy()
    work_c5 = compute_c5(work)
    best = h.copy()
    best_c5 = work_c5
    for it in range(n_iters):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - it / n_iters)
        mod = it % 4
        if mod == 0:
            pert = (0.02 + 0.12 * cool) * np.random.randn(n)
        elif mod == 1:
            f = np.random.randint(1, 16)
            ph = np.random.uniform(0, 2 * np.pi)
            pert = (0.03 + 0.1 * cool) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mod == 2:
            mask = np.random.random(n) < (0.15 + 0.35 * cool)
            pert = np.zeros(n)
            pert[mask] = (0.04 + 0.18 * cool) * np.random.randn(mask.sum())
        else:
            idx = np.random.choice(n, size=min(n//4, 40), replace=False)
            pert = np.zeros(n)
            pert[idx] = (0.05 + 0.15 * cool) * np.random.randn(len(idx))
        hp = proj_h(work + pert, n)
        T_val = 200.0 + 8000.0 * cool
        ht = lbfgs_opt(hp, T_val, n, iters=200)
        c5t = compute_c5(ht)
        temp = 0.001 * cool + 0.00005
        if c5t < work_c5 or np.random.rand() < np.exp(-(c5t - work_c5) / temp):
            work = ht
            work_c5 = c5t
            if c5t < best_c5:
                best_c5 = c5t
                best = ht
    return best, best_c5

def multi_start(n, t_end):
    starts = []
    h1 = np.zeros(n); h1[:n//2] = 1.0
    starts.append(h1)
    h2 = np.zeros(n)
    h2[n//4:3*n//4] = 1.0
    starts.append(h2)
    h3 = (np.sin(np.pi * np.arange(n) / n) ** 0.3)
    starts.append(h3)
    per = t_end - time.time()
    if per < 30:
        return starts
    best_h, best_c5 = starts[0], compute_c5(starts[0])
    for h0 in starts[1:]:
        if time.time() > t_end:
            break
        h0 = proj_h(h0, n)
        h0 = lbfgs_opt(h0, 200.0, n, iters=300)
        c5t = compute_c5(h0)
        if c5t < best_c5:
            best_h, best_c5 = h0, c5t
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = proj_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS with soft-max (fast vectorized gradient)
    for T in [80, 400, 2000, 10000, 60000]:
        if elapsed() > budget_s * 0.10:
            break
        h_new = lbfgs_opt(best_h, T, n, iters=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Coordinate descent
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.22, sweeps=80)

    # Phase 3: Block SA
    h_bin, c5_bin = block_sa(best_h, n, t0 + budget_s * 0.38)
    if c5_bin < best_c5:
        best_h = lbfgs_opt(h_bin, 3000.0, n, iters=300)
        best_c5 = compute_c5(best_h)
        best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.46, sweeps=40)

    # Phase 4: Basin hopping
    best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.65, n_iters=130)

    # Phase 5: Coord descent after basin hopping
    best_h, best_c5 = coord_desc(best_h, n, t0 + budget_s * 0.76, sweeps=60)

    # Phase 6: Try n=200 refinement
    if elapsed() < budget_s * 0.82:
        n_hi = 200
        try:
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0/n
            x_new = np.linspace(0, 2, n_hi, endpoint=False) + 1.0/n_hi
            h_hi = np.interp(x_new, x_old, best_h)
            h_hi = proj_h(h_hi, n_hi)
            h_hi = lbfgs_opt(h_hi, 2000.0, n_hi, iters=300)
            h_hi, c5_hi = coord_desc(h_hi, n_hi, t0 + budget_s * 0.90, sweeps=35)
            if c5_hi < best_c5:
                best_h = h_hi
                best_c5 = c5_hi
                n = n_hi
        except Exception:
            pass

    # Phase 7: Final polish
    if elapsed() < budget_s * 0.94:
        best_h = lbfgs_opt(best_h, 100000.0, len(best_h), iters=600)
        best_c5 = compute_c5(best_h)
    if elapsed() < budget_s * 0.98:
        best_h, best_c5 = coord_desc(best_h, len(best_h), t0 + budget_s * 0.99, sweeps=25)

    best_h = proj_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
