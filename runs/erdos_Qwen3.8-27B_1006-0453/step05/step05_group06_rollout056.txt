    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tot = room.sum()
        if tot < 1e-14:
            return h
        h = h + room * (diff / tot)
    else:
        if s < 1e-14:
            return h
        h = h * (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def active_lags(corr, top_k=3):
    """Return indices of top-k correlation values"""
    return np.argsort(corr)[::-1][:top_k]

def grad_active_lag(h, lag, n):
    """Gradient of correlation at a specific lag w.r.t. h"""
    m_full = 2 * n - 1
    # lag in correlate corresponds to index lag + n - 1
    idx = lag + n - 1
    # corr[idx] = sum_i h[i] * (1-h[i+lag]) for valid i
    # d/dx_j of corr = (1-h[lag+j]) if j < n-lag (when lag >= 0)
    #                 = -h[j-lag] if j >= lag (when lag >= 0)
    grad = np.zeros(n)
    if lag >= 0:
        # For j in [0, n-lag): term h[j]*(1-h[j+lag]) -> d/dh[j] = (1-h[j+lag])
        #                          d/dh[j+lag] = -h[j]
        grad[:n-lag] += (1.0 - h[lag:])
        grad[lag:] += -h[:n-lag]
    else:
        p = -lag
        grad[:n-p] += (1.0 - h[p:])
        grad[p:] += -h[:n-p]
    return grad

    """Subgradient of max(corr) w.r.t. h (uniform over active set)"""
    max_c = np.max(corr)
    active = np.where(corr >= max_c - 1e-10)[0]
    grad = np.zeros(n)
    for idx in active:
        lag = idx - n + 1
        grad += grad_active_lag(h, lag, n)

    dx = 2.0 / n
    x = h.copy()

    best_c5, _ = compute_c5(x)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    return float(np.max(compute_corr(h)) * 2.0 / len(h))

def project_h(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    s = h.sum()
    d = tgt - s
    if abs(d) < 1e-14:
        return h
    if d > 0:
        r = 1.0 - h
        t = r.sum()
        if t < 1e-14:
            return h
        h = np.clip(h + r * (d / t), 0.0, 1.0)
    else:
        if s < 1e-14:
            return h
        h = np.clip(h * (1.0 + d / s), 0.0, 1.0)
    return h

def lg_grad(h, idx, n):
    g = np.zeros(n)
    l = idx - n + 1
    if 0 <= l < n:
        g[:n-l] += 1.0 - h[l:]
        g[l:] += -h[:n-l]
    elif -n < l < 0:
        p = -l
        g[:n-p] += 1.0 - h[p:]
        g[p:] += -h[:n-p]
    return g

def opt_one_lag(h, idx, n, lam, it, te):
    def F(x):
        return compute_corr(x)[idx] + lam * (x.sum() - n * 0.5) ** 2
    def G(x):
        return lg_grad(x, idx, n) + 2.0 * lam * (x.sum() - n * 0.5)
    try:
        r = minimize(F, project_h(h), jac=G, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': it, 'ftol': 1e-18, 'gtol': 1e-12})
        if time.time() < te:
            return project_h(r.x)
    except Exception:
        pass
    return project_h(h.copy())

def active_loop(h, n, te, rounds=50):
    b = h.copy()
    c5 = compute_c5(b)
    for _ in range(rounds):
        if time.time() > te:
            break
        corr = compute_corr(b)
        k = int(np.argmax(corr))
        nh = opt_one_lag(b, k, n, 800.0, 2500, te)
        nc = compute_c5(nh)
        if nc < c5 - 1e-14:
            c5, b = nc, nh
        else:
            break
    return b, c5

def cdesc(h, n, te, sweeps=500):
    b = h.copy()
    c5 = compute_c5(b)
    mf = 2 * n - 1
    lags = np.arange(mf) - n + 1
    for _ in range(sweeps):
        if time.time() > te:
            break
        imp = False
        for i in range(n):
            h0 = b.copy()
            h0[i] = 0.0
            cb = compute_corr(h0)
            ln = np.zeros(mf)
            qa = np.zeros(mf)
            for jdx in range(mf):
                lg = lags[jdx]
                if lg == 0:
                    ln[jdx] = 1.0
                    qa[jdx] = -1.0
                elif 0 < lg < n:
                    if i >= lg:
                        ln[jdx] += 1.0 - b[i - lg]
                    if i + lg < n:
                        ln[jdx] -= b[i + lg]
                elif -n < lg < 0:
                    p = -lg
                    if i + p < n:
                        ln[jdx] += 1.0 - b[i + p]
                    if i >= p:
                        ln[jdx] -= b[i - p]
            def f1(x):
                return np.max(cb + x * ln + x * x * qa)
            r = minimize_scalar(f1, bounds=(0.0, 1.0), method='bounded',
                                options={'xatol': 1e-12})
            if abs(r.x - b[i]) < 1e-14:
                continue
            ht = b.copy()
            ht[i] = r.x
            ht = project_h(ht)
            nc = compute_c5(ht)
            if nc < c5 - 1e-15:
                c5, b, imp = nc, ht, True
        if not imp:
            break
    return b, c5

def bin_local_search(h, n, te, passes=10):
    """Greedy local search on binary rounding"""
    xn = np.round(h)
    xn = (xn >= 0.5).astype(float)
    diff = n // 2 - int(xn.sum())
    if diff > 0:
        order = np.argsort(h - xn)[::-1]
        for j in range(min(diff, n)):
            xn[order[j]] = 1.0
    elif diff < 0:
        order = np.argsort(h - xn)
        for j in range(min(-diff, n)):
            xn[order[j]] = 0.0
    xn = project_h(xn)
    bc = compute_c5(xn)
    for _ in range(passes):
        if time.time() > te:
            break
        imp = False
        for i in range(n):
            xn2 = xn.copy()
            xn2[i] = 1.0 - xn2[i]
            xn2 = project_h(xn2)
            c = compute_c5(xn2)
            if c < bc - 1e-14:
                xn, bc, imp = xn2, c, True
        if not imp:
            break
    return xn, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        bh = np.array(initial_h_values, dtype=float).copy()
    else:
        bh = np.zeros(160)
        bh[:80] = 1.0
    n = len(bh)
    bh = project_h(bh)
    bc5 = compute_c5(bh)

    # Phase 1: Active-lag L-BFGS-B (18%)
    bh, bc5 = active_loop(bh, n, t0 + budget_s * 0.18, 40)

    # Phase 2: Thorough coordinate descent (30%)
    bh, bc5 = cdesc(bh, n, t0 + budget_s * 0.48, 600)

    # Phase 3: Binary local search + re-continuous (12%)
    p3e = t0 + budget_s * 0.60
    hb, cb = bin_local_search(bh, n, p3e, 8)
    if cb < bc5:
        bc5, bh = cb, hb
    bh, bc5 = active_loop(bh, n, p3e, 20)
    bh, bc5 = cdesc(bh, n, p3e, 200)

    # Phase 4: Multi-scale try n=200, n=120 (22%)
    p4e = t0 + budget_s * 0.82
    for n2 in [200, 120]:
        if time.time() > p4e:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn2 = np.linspace(0, 2, n2, endpoint=False)
        h2 = project_h(np.interp(xn2, xo, bh))
        h2, c2 = active_loop(h2, n2, min(time.time() + 80, p4e), 15)
        h2, c2 = cdesc(h2, n2, min(time.time() + 120, p4e), 150)
        if c2 < bc5:
            bc5, bh, n = c2, h2, n2

    # Phase 5: Basin-hop perturbations (10%)
    p5e = t0 + budget_s * 0.92
    wh = bh.copy()
    wc5 = bc5
    for it in range(20):
        if time.time() > p5e:
            break
        cool = 1.0 - 0.9 * (it / 20.0)
        sc = 0.005 + 0.04 * cool
        hp = project_h(wh + sc * np.random.randn(n))
        hc = opt_one_lag(hp, int(np.argmax(compute_corr(hp))), n, 800, 1500, p5e)
        hc, _ = cdesc(hc, n, min(time.time() + 15, p5e), 50)
        c5v = compute_c5(hc)
        temp = 1e-4 * cool
        if c5v < wc5 or np.random.rand() < np.exp(-(c5v - wc5) / temp):
            wh, wc5 = hc, c5v
            if c5v < bc5:
                bc5, bh = c5v, hc

    # Phase 6: Final polish (8%)
    p6e = t0 + budget_s * 0.99
    bh, bc5 = active_loop(bh, n, p6e, 30)
    bh, bc5 = cdesc(bh, n, p6e, 400)
    bh = project_h(bh)
    bc5 = compute_c5(bh)
    return bh, bc5, n
