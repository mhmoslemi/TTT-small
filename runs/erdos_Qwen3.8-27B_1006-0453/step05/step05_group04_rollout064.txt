def targeted_swap(h, n, t_end, max_iters=50000):
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * (2.0/n))
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    rng = np.random.RandomState(123)
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        # Identify worst shift
        max_idx = np.argmax(corr)
        s_star = max_idx - (n-1)
        
        # Try to find a good swap
        improved = False
        # Focus on positions that contribute to the worst shift
        if s_star > 0:
            # Positions i where h[i]=1 and h[i+s_star]=0 contribute to corr at shift s_star
            bad_positions = [i for i in ones if i + s_star < n and h[i+s_star] == 0]

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, t_end, max_sweeps=5000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    corr = compute_corr(best_h)
    best_c5 = float(np.max(corr) * (2.0 / n))
    dx = 2.0 / n
    converged_sweeps = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 5 == 4 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                kp = n - 1 + s
                km = n - 1 - s
                if s <= n - 1 - i:
                    lin[kp] += (1.0 - best_h[i + s])
                if s <= i:
                    lin[kp] -= best_h[i - s]
                if i >= s:
                    lin[km] += (1.0 - best_h[i - s])
                if i + s <= n - 1:
                    lin[km] -= best_h[i + s]
            lin[n - 1] = 1.0 - 2.0 * old
            c_base = corr.copy()
            def f1d(v):
                d = v - old
                val = c_base + d * lin
                val[n - 1] -= d * d
                return float(np.max(val) * dx)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-11})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-13:
                continue
            best_h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
            c5_new = float(np.max(corr) * dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = old
                corr -= d * lin
                corr[n - 1] += d * d
        if not improved:
            converged_sweeps += 1
            if converged_sweeps >= 3:
                break
        else:
            converged_sweeps = 0
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def binary_swap_opt(h, n, t_end, max_rounds=500, seed=42):
    """Optimize via binary swaps (lift one 1 to 0, drop one 0 to 1)."""
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * (2.0 / n))
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    stagnant = 0
    for rd in range(max_rounds):
        if time.time() > t_end:
            break
        max_idx = int(np.argmax(corr))
        s_star = max_idx - (n - 1)
        if s_star > 0:
            bad = [i for i in ones if i + s_star < n and h[i + s_star] == 0]
        elif s_star < 0:
            a = -s_star
            bad = [i for i in ones if i >= a and h[i - a] == 0]
        else:
            bad = list(ones)
        try_list = bad if len(bad) > 0 else list(ones)
        try_n = min(len(try_list), 15)
        improved = False
        for _ in range(try_n * 4):
            if time.time() > t_end:
                return h, best_c5
            i = rng.choice(try_list)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_c5 = float(np.max(new_corr) * dx)
            if new_c5 < best_c5 - 1e-14:
                corr = new_corr
                best_c5 = new_c5
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            stagnant += 1
            if stagnant >= 5:
                break
        else:
            stagnant = 0
    h = project_h(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = 1.0 - hop / max(n_hops, 1)
        step = 0.01 + 0.05 * cool
        mode = hop % 6
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 16)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = step * 2 * rng.randn(int(mask.sum()))
        elif mode == 3:
            idx = rng.choice(n, size=max(4, n // 8), replace=False)
            pert = np.zeros(n)
            pert[idx] = step * rng.randn(len(idx))
        elif mode == 4:
            k = rng.randint(4, n // 4)
            pert = np.roll(rng.randn(n) * step, k) - np.roll(rng.randn(n) * step, -k)
        else:
            pert = step * 0.3 * rng.randn(n)
        h_pert = project_h(cur_h + pert, n)
        hop_end = min(t_end, time.time() + max(1.5, 6.0 * cool))
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=80, seed=seed + hop)
        temp = 0.003 * cool + 0.0003
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
