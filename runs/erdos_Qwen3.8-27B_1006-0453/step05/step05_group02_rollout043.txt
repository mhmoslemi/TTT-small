**Key idea: Subgradient/active-set method**

At the optimum, the active set (shifts k achieving the max) is small. The subgradient of the max function at h is a uniform average of the gradients of the active correlations. We can:

1. Compute all correlations
2. Find active set: k where corr[k] > max_corr - ε (for small ε like 1e-3)
3. Compute the subgradient: average of dcorr_k/dh over active k
4. Take a projected step: h_new = project(h - α * subgrad)
5. If the max decreased, accept; otherwise reduce α

This is like projected subgradient descent on the nonsmooth max. It's O(n) per iteration (for the active set, which is small) plus O(n) for the correlation computation.

def subgrad_step(h, n, alpha=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    active = np.where(corr > max_c - 0.01)[0]
    # Subgradient: average gradient over active set
    grad = np.zeros(n)
    kr = np.arange(-(n-1), n)
    for ki in active:
        shift = kr[ki]
        # dcorr[ki]/dh[j] for each j
        # corr[ki] = sum_valid h[j] * (1-h[j+shift])
        # d/dh[j] = (1-h[j+shift]) if j+shift valid, contributes +1 to h[j]
        #          - h[j-shift] if j-shift valid, contributes -1 from chain rule on 1-h[j-shift]
        j_range = np.arange(n)
        jp = j_range + shift
        jm = j_range - shift
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        g = np.zeros(n)
        g[vp] += 1.0 - h[jp[vp]]
        # For the -h[j-shift] term: when h[j] appears as 1-h[j-shift],
        # j-shift = i means j = i+shift, so for the term h[i]*(1-h[i+shift]),
        # h[j] appears in the second factor when j = i+shift, i.e., i = j-shift
        # d/dh[j] of h[i]*(1-h[i+shift]) where i+shift = j: 
        #   = h[i] * (-1) = -h[j-shift] where j-shift = i
        if np.any(vm):
            g -= h[jm[vm]]
        grad += g
    grad /= len(active)
    # Projected step
    h_new = project_h(h - alpha * grad, n)
    return h_new, max_c

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip((corr - m) / T, -50.0, 50.0)
    return (m + T * np.log(np.sum(np.exp(shifted)))) * dx

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        sp = np.clip(h - mid, 0.0, 1.0).sum()
        if sp > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    d = target - hp.sum()
    if abs(d) > 1e-10:

def slsqp_step(h, T, n, maxiter=800):

        return x.sum() - n / 2.0

                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(res.x, n)
    except Exception:

def coord_single(h, n, i):
    h0 = h.copy()
    h0[i] = 0.0
    cb = np.correlate(h0, 1.0 - h0, mode='full')
    kr = np.arange(-(n - 1), n)
    j1, j2 = i + kr, i - kr
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    cf = np.zeros(2 * n - 1)
    cf[v1] += 1.0 - h[np.clip(j1[v1], 0, n - 1)]
    cf[v2] -= h[np.clip(j2[v2], 0, n - 1)]
    dx = 2.0 / n
    def f1(x):
        return np.max(cb + x * cf) * dx
    try:
        return float(minimize_scalar(f1, bounds=(0.0, 1.0),
                                     method='bounded', options={'xatol': 1e-13}).x)
    except Exception:
        return h[i]

            break

            ct = compute_c5(tt)
            if ct < bc - 1e-14:

        if not imp:
            break
    return bh, bc

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smooth_c5(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip((corr - m) / T, -50.0, 50.0)
    return (m + T * np.log(np.sum(np.exp(sh)))) * (2.0 / n)

def project_h(h, n):
    t = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - t) < 1e-12:
        return h
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0, 1)
    d = t - hp.sum()
    if abs(d) > 1e-10:
        fr = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        if fr.sum() > 0:
            hp[fr] += d / fr.sum()
            hp = np.clip(hp, 0, 1)
    return hp

def _con_eq(x, n):
    return x.sum() - n / 2.0

    try:

                     bounds=[(0.0, 1.0)] * n,

                     options={'maxiter': mi, 'ftol': 1e-16})
        return project_h(r.x, n)
    except Exception:
        return h
