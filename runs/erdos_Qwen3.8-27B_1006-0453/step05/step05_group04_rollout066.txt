import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            room = h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h - h / tot * (s - target)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Add/subtract to the element closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def fast_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_descent(h, n, t_end, max_sweeps=5000, seed=42, pair_prob=0.1):

    best_h = h.copy()

    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            # Compute linear coefficients for each lag
            # corr[lag] changes linearly with h[i], except the h[i]^2 term at lag 0
            lin = np.zeros(2 * n - 1)
            # Contribution to lag s (where s = idx - (n-1)):
            # h[i] appears in (1-h[i]) for lags where i is the "second" index
            # and h[i] appears in h[i+s] for lags where i is the "first" index
            for s in range(1, n):
                mp = s + n - 1  # positive lag s
                mn = n - 1 - s   # negative lag s (= -(s))
                # h[i] in h[...+s] position (first factor): h[i] = h[j+s] where j = i-s
                if i - s >= 0:
                    lin[mp] += (1.0 - best_h[i - s])
                # h[i] in (1-h[...]) position (second factor): h[j]*(1-h[i]) where j = i, need j+s valid
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                # Negative lag: h[i] in h[...] position where i = j-s, so j = i+s
                if i + s < n:

                # Negative lag: h[i] in (1-h[...]) where i = j, need j-s valid, s' = s
                if i - s >= 0:

            # Lag 0: h[i]^2 - h[i], so derivative is 2*h[i] - 1, linear part is -1
            lin[n - 1] = 1.0 - 2.0 * old
            
            c_base = corr.copy()

                dval = v - old
                val = c_base + dval * lin
                val[n - 1] -= dval * dval
                return float(np.max(val) * dx)

        if not improved and sweep > 20:
            break
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def fast_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def cd_sweep(h, corr, dx, n, t_end, seed=0):
    """One sweep of coordinate descent. Returns (h, corr, c5, improved)."""
    rng = np.random.RandomState(seed)
    order = rng.permutation(n)
    c0 = float(np.max(corr) * dx)
    improved = False
    for i in order:
        if time.time() > t_end:
            break
        old = h[i]
        lin = np.zeros(2 * n - 1)
        for s in range(1, n):
            mp = n - 1 + s
            mn = n - 1 - s
            if i >= s:
                lin[mp] += (1.0 - h[i - s])
                lin[mn] -= h[i - s]
            if i + s < n:
                lin[mp] -= h[i + s]
                lin[mn] += (1.0 - h[i + s])
        lin[n - 1] = 1.0 - 2.0 * old
        cbase = corr
        best_v = old
        best_f = float(np.max(cbase) * dx)
        # Bounded scalar minimization
        def f1d(v):
            d = v - old
            val = cbase + d * lin
            val[n - 1] -= d * d
            return np.max(val) * dx
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12})
        v_new = float(res.x)
        d = v_new - old
        if abs(d) < 1e-14:
            continue
        h[i] = v_new
        corr += d * lin
        corr[n - 1] -= d * d
        f_new = float(np.max(corr) * dx)
        if f_new < best_f - 1e-15:
            best_f = f_new
            improved = True
        else:
            h[i] = old
            corr -= d * lin
            corr[n - 1] += d * d
    return h, corr, best_f, improved

def coord_descent(h, n, t_end, max_sweeps=5000, seed=42):
    best_h = h.copy()
    corr = fast_corr(best_h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    sw = 0
    while time.time() < t_end and sw < max_sweeps:
        best_h, corr, c5, imp = cd_sweep(best_h, corr, dx, n, t_end, seed + sw)
        sw += 1
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
        elif not imp and sw > 30:
            break
    return best_h, best_c5

def softmax_grad(h, n, T):
    """Compute softmax-smoothed gradient of C5."""
    corr = fast_corr(h)
    dx = 2.0 / n
    m = np.max(corr)
    exponents = np.clip(T * (corr - m), -50.0, 50.0)
    p = np.exp(exponents)
    p /= p.sum()
    grad = np.zeros(n)
    for mIdx in range(2 * n - 1):
        w = p[mIdx]
        if w < 1e-14:
            continue
        s = mIdx - (n - 1)
        if s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        elif s < 0:
            a = -s
            grad[:n - a] += w * (1.0 - h[a:])
            grad[a:] -= w * h[:n - a]
        else:
            grad += w * (1.0 - 2.0 * h)
    return dx * grad

def adam_opt(h, n, t_end, max_iters=3000, seed=42):
    """Adam-based optimization of softmax-smoothed C5."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    h_c = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    for it in range(max_iters):
        if time.time() > t_end:
            break
        frac = min(1.0, it / max(1, max_iters))
        T = 30.0 + 200.0 * frac
        g = softmax_grad(h_c, n, T)
        lr = 0.08 / (1.0 + 0.5 * it / max(1, max_iters))
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * g * g
        mc = m_v / (1 - beta1 ** (it + 1))
        vc = v_v / (1 - beta2 ** (it + 1))
        step = lr * mc / (np.sqrt(vc) + eps)
        h_new = project_h(h_c - step, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_c = h_new
    return best_h, best_c5

    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    best_c5 = float(np.max(corr) * dx)
    sw = 0
    stall = 0
    while time.time() < t_end and sw < max_sweeps:
        order = rng.permutation(n) if sw % 4 == 3 else np.arange(n)
        improved = False
        for i in order:
            if time.time() > t_end:
                return best_h, best_c5
            old = best_h[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = n - 1 + s
                mn = n - 1 - s
                if i >= s:
                    lin[mp] += 1.0 - best_h[i - s]
                    lin[mn] -= best_h[i - s]
                if i + s < n:
                    lin[mp] -= best_h[i + s]
                    lin[mn] += 1.0 - best_h[i + s]
            lin[n - 1] = 1.0 - 2.0 * old
            def f1d(v):
                d = v - old
                val = corr + d * lin
                val[n - 1] -= d * d
                return np.max(val) * dx
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-12})
            v_new = float(res.x)
            d = v_new - old

            corr[n - 1] -= d * d

                corr -= d * lin
                corr[n - 1] += d * d

        else:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use Lp norm
    cp = np.abs(corr) ** (p - 1) * np.sign(corr)
    cp = cp / (np.sum(np.abs(corr) ** p)) ** (1.0 / p)

    T = p
    exponents = np.clip(T * (corr - m), -50.0, 0.0)
    w = np.exp(exponents)
    w /= w.sum()
    grad = np.zeros(n)

            grad[a:] -= wi * h[:n - a]

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        if s > target:
            h = h * (target / s)
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * ((target - s) / tot)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def cd_optimize(h, n, t_end, max_sweeps=5000, seed=42):
    rng = np.random.RandomState(seed)
    bh = h.copy()
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    dx = 2.0 / n
    bc = float(np.max(corr) * dx)
    sw = 0
    stall = 0
    while time.time() < t_end and sw < max_sweeps:
        order = rng.permutation(n) if sw % 4 == 3 else np.arange(n)
        imp = False
        for i in order:
            if time.time() > t_end:
                return bh, bc
            old = bh[i]
            lin = np.zeros(2 * n - 1)
            for s in range(1, n):
                mp = n - 1 + s
                mn = n - 1 - s
                if i >= s:
                    lin[mp] += 1.0 - bh[i - s]
                    lin[mn] -= bh[i - s]
                if i + s < n:
                    lin[mp] -= bh[i + s]
                    lin[mn] += 1.0 - bh[i + s]
            lin[n - 1] = 1.0 - 2.0 * old
            cbase = corr
            def f1d(v, _c=cbase, _l=lin, _o=old, _d=dx, _n=n):
                dv = v - _o
                val = _c + dv * _l
                val[_n - 1] -= dv * dv
                return np.max(val) * _d
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-13})
            vn = float(res.x)
            dv = vn - old
            if abs(dv) < 1e-14:
                continue
            bh[i] = vn
            corr += dv * lin
            corr[n - 1] -= dv * dv
            cn = float(np.max(corr) * dx)
            if cn < bc - 1e-15:
                bc = cn
                imp = True
            else:
                bh[i] = old
                corr -= dv * lin
                corr[n - 1] += dv * dv
        sw += 1
        if imp:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    return bh, bc

def grad_soft(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ex = np.clip(T * (corr - m), -50.0, 0.0)
    w = np.exp(ex)
    w /= w.sum()
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = w[mi]
        if wi < 1e-14:
            continue
        s = mi - (n - 1)
        if s > 0:
            g[s:] += wi * (1.0 - h[:n - s])
            g[:n - s] -= wi * h[s:]
        elif s < 0:
            a = -s
            g[:n - a] += wi * (1.0 - h[a:])
            g[a:] -= wi * h[:n - a]
        else:
            g += wi * (1.0 - 2.0 * h)
    return dx * g

def adam_opt(h, n, t_end, max_it=2000, seed=42):
    bh = h.copy()
    bc = compute_c5(bh)
    hc = h.copy()
    mv = np.zeros(n)
    vv = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    for it in range(max_it):
        if time.time() > t_end:
            break
        fr = min(1.0, it / max(1, max_it))
        T = 25.0 + 120.0 * fr
        g = grad_soft(hc, n, T)
        lr = 0.1 * (1.0 - 0.6 * fr)
        mv = b1 * mv + (1 - b1) * g
        vv = b2 * vv + (1 - b2) * g * g
        mc = mv / (1 - b1 ** (it + 1))
        vc = vv / (1 - b2 ** (it + 1))
        st = lr * mc / (np.sqrt(vc) + eps)
        hn = project_h(hc - st, n)
        cn = compute_c5(hn)
        if cn < bc:
            bc = cn
            bh = hn.copy()
        hc = hn
    return bh, bc

            p[mk] = stp * 3 * rng.randn(mk.sum())
        elif md == 3:
            ix = rng.choice(n, size=max(5, n // 6), replace=False)
            p = np.zeros(n)
            p[ix] = stp * 2 * rng.randn(len(ix))
        elif md == 4:
            sh = rng.randint(1, 6)
            p = 0.3 * (np.roll(ch, -sh) - ch)
        elif md == 5:
            p = np.zeros(n)
            for _ in range(3):
                p += stp * rng.uniform(0.5, 1.5) * np.sin(
                    2 * np.pi * rng.randint(1, 12) * np.arange(n) / n + rng.uniform(0, 6.28))
        elif md == 6:
            c = rng.randint(0, n)
            w = rng.randint(3, 15)
            a, b = max(0, c - w), min(n, c + w)
            p = np.zeros(n)
            p[a:b] = stp * 2 * rng.randn(b - a)
        else:
            p = stp * 0.5 * rng.randn(n)
        hp2 = project_h(ch + p, n)
