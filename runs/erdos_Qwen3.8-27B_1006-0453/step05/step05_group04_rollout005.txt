```python
def balance_lags(h, n, t_end, max_iters=500):
    """Try to balance the top correlation lags."""
    dx = 2.0 / n
    h_c = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_c)
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h_c, 1.0 - h_c, mode='full') * dx
        m = corr.max()
        # Find active lags (within 2% of max)
        active = corr > m * 0.98
        # Compute gradient for each active lag
        grad = np.zeros(n)
        for idx in np.where(active)[0]:
            s = idx - (n - 1)
            if s == 0:
                grad += (1.0 - 2.0 * h_c)
            elif s > 0:
                grad[s:] += (1.0 - h_c[:n-s])
                grad[:n-s] -= h_c[s:]
            else:
                a = -s
                grad[:n-a] += (1.0 - h_c[a:])
                grad[a:] -= h_c[:n-a]
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            h *= target / s
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (target - s) / tot * room
        h = np.clip(h, 0.0, 1.0)
    return h

def cd_sweep(h, corr, n, dx, order):
    """One coordinate descent sweep. Returns updated h and corr."""
    best_local = None
    for i in order:
        old = h[i]
        lin = np.zeros(2 * n - 1)
        for s in range(1, n):
            mp = s + n - 1
            mn = n - 1 - s
            if s <= i:
                lin[mp] += (1.0 - h[i - s])
                lin[mn] -= h[i - s]
            if i + s < n:
                lin[mp] -= h[i + s]
                lin[mn] += (1.0 - h[i + s])
        lin[n - 1] = 1.0 - 2.0 * old
        # Minimize max(corr + d*lin) * dx - d^2 * dx (for lag 0)
        # f(d) = max_s (corr[s] + d*lin[s]) * dx, but lag 0 has extra -d^2
        c0 = corr[n-1]
        l0 = lin[n-1]
        # For non-zero lags: linear in d
        # For lag 0: c0 + d*l0 - d^2
        # Overall: max(max_{s!=0} (corr[s]+d*lin[s]), c0+d*l0-d^2) * dx
        # This is a concave function in d (max of concave functions... wait no)
        # max of linear functions is convex, but -d^2 is concave
        # So the objective is max(convex, concave) which is not necessarily convex
        # But it IS unimodal in practice for small d
        # Use bounded scalar minimization
        base = corr.copy()
        def f1d(v):
            d = v - old
            val = base + d * lin
            val[n - 1] -= d * d
            return float(np.max(val) * dx)
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12, 'maxiter': 100})
        v_new = float(res.x)
        d = v_new - old
        if abs(d) > 1e-13:
            h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
    return h, corr

def coord_descent(h, n, t_end, max_sweeps=3000, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    prev_c5 = best_c5
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        if sweep % 5 == 2:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        h, corr = cd_sweep(h, corr, n, dx, order)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if abs(prev_c5 - c5) < 1e-12:
            break
        prev_c5 = c5
    return best_h, best_c5

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - hop / n_hops)
        step = 0.01 + 0.05 * cool ** 1.5
        mode = hop % 6
        if mode == 0:
            pert = step * rng.randn(n)
        elif mode == 1:
            f = rng.randint(1, 15)
            ph = rng.uniform(0, 2 * np.pi)
            pert = step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif mode == 2:
            mask = rng.random(n) < 0.2
            pert = np.zeros(n)
            pert[mask] = step * 2.0 * rng.randn(int(mask.sum()))
        elif mode == 3:
            # Mass transport: move mass from high to low regions
            perm = rng.permutation(n)
            pert[:, 1] = 0  # no-op
            pert = np.zeros(n)
            for _ in range(n // 20):
                a, b = perm[rng.randint(n)], perm[rng.randint(n)]
                amt = step * 0.5 * rng.rand()
                pert[a] += amt
                pert[b] -= amt
        elif mode == 4:
            # Block perturbation
            blen = rng.randint(4, n // 4)
            start = rng.randint(0, n - blen)
            pert = np.zeros(n)
            pert[start:start+blen] = step * rng.uniform(-1, 1)
        else:
            # Two-scale perturbation
            f1, f2 = rng.randint(1, 8), rng.randint(9, 20)
            ph1, ph2 = rng.uniform(0, 2*np.pi), rng.uniform(0, 2*np.pi)
            pert = step * (np.sin(2*np.pi*f1*np.arange(n)/n + ph1) + 
                          0.5*np.sin(2*np.pi*f2*np.arange(n)/n + ph2))
        h_pert = project_h(cur_h + pert, n)
```

```python
def compute_lin(k, h, n):
    lin = np.zeros(2*n-1)
    # Positive lags s=1..n-1
    s = np.arange(1, n)
    # For index n-1+s: contribution (1-h[k+s]) if k+s<n, contribution -h[k-s] if s<=k
    idx_pos = n - 1 + s  # indices n..2n-2
    mask_pos1 = (k + s) < n  # (1-h[k+s]) term
    mask_pos2 = s <= k  # -h[k-s] term
    lin[idx_pos[mask_pos1]] += (1.0 - h[k + s[mask_pos1]])
    lin[idx_pos[mask_pos2]] -= h[k - s[mask_pos2]]
    # For index n-1-s: contribution (1-h[k-s]) if s<=k, contribution -h[k+s] if k+s<n
    idx_neg = n - 1 - s  # indices n-2..0
    lin[idx_neg[mask_pos2]] += (1.0 - h[k - s[mask_pos2]])
    lin[idx_neg[mask_pos1]] -= h[k + s[mask_pos1]]
    # Lag 0
    lin[n-1] = 1.0 - 2.0 * h[k]
    return lin
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        if s > target:
            h *= target / s
        else:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (target - s) / tot * room
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_lin_vec(k, h, n):
    s = np.arange(1, n)
    lin = np.zeros(2 * n - 1)
    m1 = (k + s) < n
    m2 = s <= k
    if m1.any():
        lin[n - 1 + s[m1]] += (1.0 - h[k + s[m1]])
        lin[n - 1 - s[m1]] -= h[k + s[m1]]
    if m2.any():
        lin[n - 1 + s[m2]] -= h[k - s[m2]]
        lin[n - 1 - s[m2]] += (1.0 - h[k - s[m2]])
    lin[n - 1] = 1.0 - 2.0 * h[k]
    return lin

def cd_sweep(h, corr, n, dx, order):
    for i in order:
        old = h[i]
        lin = compute_lin_vec(i, h, n)
        base = corr.copy()
        def f1d(v):
            d = v - old
            val = base + d * lin
            val[n - 1] -= d * d
            return float(np.max(val) * dx)
        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-12, 'maxiter': 80})
        v_new = float(res.x)
        d = v_new - old
        if abs(d) > 1e-13:
            h[i] = v_new
            corr += d * lin
            corr[n - 1] -= d * d
    return h, corr

def coord_descent(h, n, t_end, max_sweeps=3000, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    prev_c5 = best_c5
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        order = rng.permutation(n) if sweep % 4 == 3 else np.arange(n)
        h, corr = cd_sweep(h, corr, n, dx, order)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if abs(prev_c5 - c5) < 1e-13:
            break
        prev_c5 = c5
    return best_h, best_c5

def make_perturbation(mode, n, step, rng):
    if mode == 0:
        return step * rng.randn(n)
    elif mode == 1:
        f = rng.randint(1, 16)
        ph = rng.uniform(0, 2 * np.pi)
        return step * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif mode == 2:
        mask = rng.random(n) < 0.2
        pert = np.zeros(n)
        pert[mask] = step * 2.0 * rng.randn(int(mask.sum()))
        return pert
    elif mode == 3:
        pert = np.zeros(n)
        nc = max(5, n // 15)
        for _ in range(nc):
            a, b = rng.randint(n), rng.randint(n)
            amt = step * 0.5 * rng.rand()
            pert[a] += amt
            pert[b] -= amt
        return pert
    elif mode == 4:
        blen = rng.randint(4, max(5, n // 4))
        start = rng.randint(0, n - blen)
        pert = np.zeros(n)
        pert[start:start+blen] = step * rng.uniform(-1, 1)
        return pert
    else:
        f1, f2 = rng.randint(1, 8), rng.randint(9, 22)
        p1, p2 = rng.uniform(0, 2*np.pi), rng.uniform(0, 2*np.pi)
        return step * (np.sin(2*np.pi*f1*np.arange(n)/n+p1)+0.5*np.sin(2*np.pi*f2*np.arange(n)/n+p2))

def basin_hop(h, n, t_end, n_hops=300, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    for hop in range(n_hops):
        if time.time() > t_end:
            break
        cool = max(0.0, 1.0 - hop / n_hops)
        step = 0.008 + 0.05 * cool ** 1.3
        mode = hop % 6
        pert = make_perturbation(mode, n, step, rng)
        h_pert = project_h(cur_h + pert, n)
        hop_budget = max(1.0, 5.0 * cool + 0.5)
        hop_end = min(t_end, time.time() + hop_budget)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=50, seed=seed+hop)
        temp = 0.0025 * cool + 0.0002
        accept = c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt-cur_c5)/temp))
        if accept:
            cur_h, cur_c5 = h_opt, c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def try_resolution(h_src, target_n, t_end, seed):
    h_src = np.array(h_src, dtype=float)
    n_src = len(h_src)
    if target_n == n_src:
        h = project_h(h_src, target_n)
    else:
        x_old = np.linspace(0, 2, n_src)
        x_new = np.linspace(0, 2, target_n)
        h_interp = np.interp(x_new, x_old, h_src)
        h = project_h(h_interp, target_n)
    h, c5 = coord_descent(h, target_n, t_end, max_sweeps=400, seed=seed)
    return h, c5, target_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

            if c5_try < best_c5:

                best_h = h_try
                n = n_try
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.30, max_sweeps=500, seed=seed+7)

    # Phase 3: Intensive basin hopping

        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.70, n_hops=500, seed=seed+10)
```
