def solve_coord_exact(c, coeff_i, h_i, n_corr):
    # c: current correlation (length 2n-1)
    # coeff_i: coefficient vector for coordinate i (length 2n-1)
    # h_i: current value of h[i]
    # Find t in [0,1] minimizing max_k (c[k] + (t - h_i) * coeff_i[k])
    
    # Vectorized bisection on V
    # Feasibility: for all k, c[k] + (t-h_i)*coeff_i[k] <= V
    # => t <= (V - c[k])/coeff_i[k] + h_i for coeff_i[k] > 0
    # => t >= (V - c[k])/coeff_i[k] + h_i for coeff_i[k] < 0
    # Need: max(lower_bounds) <= min(upper_bounds) and result in [0,1]
    
    pos = coeff_i > 1e-15
    neg = coeff_i < -1e-15
    zero = ~pos & ~neg
    
    V_lo = float(np.max(c))  # at t=h_i
    V_hi = 0.0
    # Better initial bounds
    # At t=0: max(c - h_i * coeff_i)
    # At t=1: max(c + (1-h_i) * coeff_i)
    V_hi = min(float(np.max(c - h_i * coeff_i)), float(np.max(c + (1-h_i) * coeff_i)), V_lo)
    
    # Bisection
    lo = 0.0  # absolute lower bound on V
    hi = V_hi
    
    for _ in range(50):
        mid = (lo + hi) / 2
        # Upper bounds on t
        ub = 1.0
        if np.any(pos):
            ub = min(ub, float(np.min((mid - c[pos]) / coeff_i[pos] + h_i)))
        # Lower bounds on t
        lb = 0.0
        if np.any(neg):
            lb = max(lb, float(np.max((mid - c[neg]) / coeff_i[neg] + h_i)))
        # Zero-coeff constraint
        if np.any(zero) and np.any(c[zero] > mid):
            lo = mid
            continue
        if lb <= ub:
            hi = mid
        else:
            lo = mid
    
    # Now find t that achieves V = hi
    ub = 1.0
    if np.any(pos):
        ub = min(ub, float(np.min((hi - c[pos]) / coeff_i[pos] + h_i)))
    lb = 0.0
    if np.any(neg):
        lb = max(lb, float(np.max((hi - c[neg]) / coeff_i[neg] + h_i)))
    t_opt = max(0.0, min(1.0, (lb + ub) / 2))
    return t_opt

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def proj_h(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = 0.5 * n
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h *= tgt / s
    else:
        room = 1.0 - h
        tr = room.sum()
        if tr > 1e-14:
            h += room * (tgt - s) / tr
    h = np.clip(h, 0, 1)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.001) & (h < 0.999)
        cnt = m.sum()
        if cnt > 0:
            h[m] += d / cnt
            h = np.clip(h, 0, 1)
    return h

def solve_coord(c, coeff, old_h):
    pos = coeff > 1e-14
    neg = coeff < -1e-14
    V_hi = min(float(np.max(c - old_h * coeff)), float(np.max(c + (1.0 - old_h) * coeff)))
    V_lo = 0.0
    for _ in range(35):
        mid = (V_lo + V_hi) / 2.0
        ub = 1.0
        if np.any(pos):
            ub = min(ub, float(np.min((mid - c[pos]) / coeff[pos] + old_h)))
        lb = 0.0
        if np.any(neg):
            lb = max(lb, float(np.max((mid - c[neg]) / coeff[neg] + old_h)))
        if lb <= ub + 1e-15:
            V_hi = mid
        else:
            V_lo = mid
    ub = 1.0
    if np.any(pos):
        ub = min(ub, float(np.min((V_hi - c[pos]) / coeff[pos] + old_h)))
    lb = 0.0
    if np.any(neg):
        lb = max(lb, float(np.max((V_hi - c[neg]) / coeff[neg] + old_h)))
    return max(0.0, min(1.0, (lb + ub) / 2.0))

def coord_desc(h, n, t_end, max_sweeps=5000):
    h = h.copy()
    n_c = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    order = np.random.permutation(n)
    sweeps = 0
    stall = 0
    while sweeps < max_sweeps and time.time() < t_end:
        improved = False
        max_change = 0.0
        for idx in range(n):
            if time.time() > t_end:
                break
            i = order[idx]
            coeff = np.zeros(n_c)
            coeff[i:i+n] += (1.0 - h[::-1])
            coeff[n-1-i:2*n-1-i] -= h
            old = h[i]
            t_opt = solve_coord(c, coeff, old)
            delta = t_opt - old
            if abs(delta) > 1e-12:
                h[i] = t_opt
                c += delta * coeff
                max_change = max(max_change, abs(delta))
                improved = True
        if improved:
            h = proj_h(h, n)
            c2 = np.correlate(h, 1.0 - h, mode='full')
            if np.max(np.abs(c2 - c)) > 1e-5:
                c = c2
            stall = 0
        else:
            stall += 1
            if stall >= 5:
                break
            # Shuffle order to escape cycling
            order = np.random.permutation(n)
        sweeps += 1
    return h, compute_c5(h)

def bin_swaps(h, n, t_end, max_iters=200000):
    """Binary swap simulated annealing."""
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    if len(ones) != n // 2:
        # Fix
        if len(ones) > n // 2:
            rm = np.random.choice(ones, len(ones) - n//2, replace=False)
            h_b[rm] = 0
        else:
            add = np.random.choice(zeros, n//2 - len(ones), replace=False)
            h_b[add] = 1
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
    dx = 2.0 / n
    c = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    best_c5 = float(np.max(c) * dx)
    best_h = h_b.copy()
    T = 0.01
    for it in range(max_iters):
        if time.time() > t_end:
            break
        # Pick random 1 and random 0
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        # Compute change
        n_c = 2 * n - 1
        ci = np.zeros(n_c)
        ci[i:i+n] += (1.0 - h_b[::-1])
        ci[n-1-i:2*n-1-i] -= h_b
        cj = np.zeros(n_c)
        cj[j:j+n] += (1.0 - h_b[::-1])
        cj[n-1-j:2*n-1-j] -= h_b
        # Swap: h[i] 1->0 (delta=-1 for i), h[j] 0->1 (delta=+1 for j)
        nc = c - ci + cj
        nc5 = float(np.max(nc) * dx)
        if nc5 < best_c5 or np.random.rand() < np.exp(-(nc5 - best_c5) / T):
            h_b[i], h_b[j] = 0, 1
            c = nc
            # Update ones/zeros
            # i moves from ones to zeros, j from zeros to ones
            ones = np.where(h_b == 1)[0]  # expensive but correct
            zeros = np.where(h_b == 0)[0]
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h_b.copy()
        T = max(0.0

001, T * 0.9999)
    return best_h, best_c5

def compute_c5(h):

def proj_h(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = 0.5 * n
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    if s > tgt:
        h *= tgt / s
    else:
        room = 1.0 - h
        tr = room.sum()
        if tr > 1e-14:
            h += room * (tgt - s) / tr
    h = np.clip(h, 0, 1)
    d = tgt - h.sum()
    if abs(d) > 1e-9:
        m = (h > 0.001) & (h < 0.999)
        cnt = m.sum()
        if cnt > 0:
            h[m] += d / cnt
            h = np.clip(h, 0, 1)
    return h

    neg = coeff < -1e-14

        mid = (V_lo + V_hi) / 2.0
        ub = 1.0
        if np.any(pos):
            ub = min(ub, float(np.min((mid - c[pos]) / coeff[pos] + old_h)))
        lb = 0.0
        if np.any(neg):
            lb = max(lb, float(np.max((mid - c[neg]) / coeff[neg] + old_h)))
        if lb <= ub + 1e-15:
            V_hi = mid
        else:
            V_lo = mid
    ub = 1.0
    if np.any(pos):
        ub = min(ub, float(np.min((V_hi - c[pos]) / coeff[pos] + old_h)))
    lb = 0.0
    if np.any(neg):
        lb = max(lb, float(np.max((V_hi - c[neg]) / coeff[neg] + old_h)))

def coord_desc(h, n, t_end, max_sweeps=5000):
    h = h.copy()
    n_c = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    order = np.random.permutation(n)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        for idx in range(n):
            i = order[idx]
            coeff = np.zeros(n_c)
            coeff[i:i+n] += (1.0 - h[::-1])
            coeff[n-1-i:2*n-1-i] -= h
            old = h[i]
            t_opt = solve_coord(c, coeff, old)
            delta = t_opt - old
            if abs(delta) > 1e-11:
                h[i] = t_opt
                c += delta * coeff
                improved = True
        if improved:
            h = proj_h(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
            order = np.random.permutation(n)
    return h, compute_c5(h)
