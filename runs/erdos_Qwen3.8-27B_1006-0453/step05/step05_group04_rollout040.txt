def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + room * (diff / tot)
        else:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

    best_h = h.copy()

    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5 = float(np.max(corr) * dx)

                lin[i : n-1] -= best_h[i+1:i+1+right][::-1]

            def f1d(v):
                d = v - old
                vals = c_base + d * lin
                vals[n-1] += v*(1.0-v) - old_quad
                return float(np.max(vals) * dx)
            
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 30})
            v_new = float(res.x)
            d = v_new - old
            if abs(d) < 1e-12:
                continue
            new_quad = v_new * (1.0 - v_new)
            corr += d * lin
            corr[n-1] += new_quad - old_quad
            c5_new = float(np.max(corr) * dx)
            if c5_new <= best_c5 + 1e-15:
                best_c5 = min(best_c5, c5_new)
                best_h[i] = v_new
                improved = True
            else:
                corr -= d * lin
                corr[n-1] -= new_quad - old_quad
        if not improved:
            break
    return best_h, best_c5

            pert = np.zeros(n)
            pert[start:start+blen] += step * 0.5
            ps = (start + n // 2) % n
            pert[ps:ps+blen] -= step * 0.5
        else:
            f1 = rng.randint(2, 8)
            f2 = rng.randint(8, 20)
            ph1 = rng.uniform(0, 2 * np.pi)
            ph2 = rng.uniform(0, 2 * np.pi)
            x = np.arange(n) / n
            pert = step * 0.7 * (np.sin(2*np.pi*f1*x + ph1) + np.cos(2*np.pi*f2*x + ph2))
        h_pert = project_h(cur_h + pert, n)
        hop_time = max(1.5, 6.0 * cool)
        hop_end = min(t_end, time.time() + hop_time)
        h_opt, c5_opt = coord_descent(h_pert, n, hop_end, max_sweeps=80, seed=seed + hop)
        temp = 0.004 * cool + 0.0004
        if c5_opt < cur_c5 or (c5_opt > cur_c5 and rng.rand() < np.exp(-(c5_opt - cur_c5) / temp)):
            cur_h = h_opt
            cur_c5 = c5_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    return best_h, best_c5

def micro_refine(h, n, t_end, n_trials=2000, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_trials):
        if time.time() > t_end:
            break
        idx = rng.randint(n)
        delta = rng.randn() * 0.002
        h_t = best_h.copy()
        h_t[idx] += delta
        h_t = project_h(h_t, n)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.zeros(160)
        best_h[:80] = 1.0
    n = len(best_h)
    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Heavy coordinate descent from initial
    t1 = t0 + budget_s * 0.20
    best_h, best_c5 = coord_descent(best_h, n, t1, max_sweeps=20000, seed=seed)

    # Phase 2: Subgradient + coord polish
    if time.time() < t0 + budget_s * 0.35:
        h_g, c5_g = subgradient_phase(best_h, n, t0 + budget_s * 0.33, max_iters=2000, seed=seed)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.38, max_sweeps=3000, seed=seed + 1)

    # Phase 3: Basin hopping (main exploration)
    if time.time() < t0 + budget_s * 0.72:
        best_h, best_c5 = basin_hop(best_h, n, t0 + budget_s * 0.72, n_hops=400, seed=seed + 10)

    # Phase 4: Final heavy coordinate descent
    if time.time() < t0 + budget_s * 0.90:
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.90, max_sweeps=20000, seed=seed + 100)

    # Phase 5: Another subgradient pass
    if time.time() < t0 + budget_s * 0.94:
        h_g2, c5_g2 = subgradient_phase(best_h, n, t0 + budget_s * 0.935, max_iters=800, seed=seed + 60)
        if c5_g2 < best_c5:
            best_h, best_c5 = h_g2, c5_g2
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.955, max_sweeps=2000, seed=seed + 200)

    # Phase 6: Micro-refinement
    if time.time() < t0 + budget_s * 0.98:
        best_h, best_c5 = micro_refine(best_h, n, t0 + budget_s * 0.98, n_trials=3000, seed=seed + 300)
        best_h, best_c5 = coord_descent(best_h, n, t0 + budget_s * 0.99, max_sweeps=2000, seed=seed + 400)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), n
