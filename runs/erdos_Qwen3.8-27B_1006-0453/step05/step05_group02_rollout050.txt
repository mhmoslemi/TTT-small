        p = make_pert(n, cool, rng)
        h_p = project_h(src + p, n)
        T_bh = 400 + 12000 * cool
        hc = slsqp_opt(h_p, T_bh, n, 700, 1e-15)
        c5v = compute_c5(hc)
        # Quick local refinement (3 sweeps)
        hc, c5r = coord_sweep(hc, n, 3, rng, time.time() + 5)
        if c5r < c5v:
            hc, c5v = hc, c5r
        temp = 0.0025 * cool + 0.0003
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 3: Thorough coordinate descent (15%)
    t3 = t0 + budget_s * 0.65
    best_h, best_c5 = coord_sweep(best_h, n, 300, rng, t3)

    # Phase 4: Finer basin hopping (15%)
    t4 = t0 + budget_s * 0.80
    work_h = best_h.copy()
    work_c5 = best_c5
    for it in range(80):
        if time.time() > t4:
            break
        cool = max(0.1, 1.0 - 0.9 * (it / 80.0))
        p = 0.06 * cool * rng.standard_normal(n)
        h_p = project_h(work_h + p, n)
        hc = slsqp_opt(h_p, 2000 + 6000 * cool, n, 500, 1e-15)
        c5v = compute_c5(hc)
        hc, c5r = coord_sweep(hc, n, 3, rng, time.time() + 4)
        if c5r < c5v:
            hc, c5v = hc, c5r
        temp = 0.001 * cool + 0.0002
        if c5v < work_c5 or rng.rand() < np.exp(-(c5v - work_c5) / max(temp, 1e-9)):
            work_h = hc
            work_c5 = c5v
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
    if work_c5 < best_c5:
        best_c5 = work_c5
        best_h = work_h

    # Phase 5: Extended coordinate descent (10%)
    t5 = t0 + budget_s * 0.90
    best_h, best_c5 = coord_sweep(best_h, n, 200, rng, t5)

    # Phase 6: SLSQP polish + final coord (10%)
    t6 = t0 + budget_s * 0.97
    for T in [400000, 1500000, 5000000]:
        if time.time() > t6:
            break
        hc = slsqp_opt(best_h, T, n, 2500, 1e-17)
        c5v = compute_c5(hc)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hc

    if time.time() < t0 + budget_s * 0.99:
        best_h, best_c5 = coord_sweep(best_h, n, 100, rng, t0 + budget_s * 0.995)

    best_h = project_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
