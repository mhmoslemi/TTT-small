import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = n * 0.5
    for _ in range(5):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        mask = (h < 1.0 - 1e-12) if d > 0 else (h > 1e-12)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def softmax_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * tau * dx
    m = s.max()
    vals = np.exp(s - m)
    result = (m + np.log(vals.sum())) / tau
    result += 1e6 * (h.sum() - n * 0.5) ** 2
    return result

def softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * tau * dx
    m = s.max()
    vals = np.exp(s - m)
    p = vals / vals.sum()
    r = np.zeros(n)
    offset = n - 1
    for k in range(2 * n - 1):
        m_lag = k - offset
        pk = p[k]
        if pk < 1e-14:
            continue
        if m_lag == 0:
            r[:] += pk * (1.0 - 2.0 * h)
        elif m_lag > 0:
            r[m_lag:] += pk * (1.0 - h[:n - m_lag])
            r[:n - m_lag] += (-pk) * h[m_lag:]
        else:
            mm = -m_lag
            r[:n - mm] += pk * (1.0 - h[mm:])
            r[mm:] += (-pk) * h[:n - mm]
    grad = r * dx
    grad += 2e6 * (h.sum() - n * 0.5)
    return grad

    best_c5 = c5
    cur_h = h.copy()
    cur_c5 = c5
    T = c5 * 0.03
    T_min = c5 * 2e-8

    t_start = time.time()

        i, j = rng.choice(n, 2, replace=False)
        delta = rng.uniform(-step, step)
        
        new_h = cur_h.copy()
        new_h[i] = min(1.0, max(0.0, cur_h[i] + delta))
        new_h[j] = min(1.0, max(0.0, cur_h[j] - delta))

        new_c5 = compute_c5(new_h)
        d_obj = new_c5 - cur_c5

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        
        T = max(T_min, T * 0.99996)
        step = max(0.005, step * 0.99999)

def local_refine(h, n, t_budget):
