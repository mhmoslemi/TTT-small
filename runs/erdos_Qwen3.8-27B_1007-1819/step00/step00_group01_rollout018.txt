    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

    m = np.max(vals)
    return float(m + eps * np.log(np.sum(np.exp((vals - m) / eps))))

def smooth_obj(h, eps):
    return smooth_max(get_corr(h), eps)

def analytic_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    w = np.exp((corr - m) / eps)
    w /= np.sum(w)
    
    # corr has length 2n-1, index s corresponds to lag k = s - (n-1)
    # C_k = sum_m h[m]*(1-h[m-k]) for valid m, m-k
    # dC_k/dh[j] = (1-h[j-k])*I[j-k valid] - h[j+k]*I[j+k valid]

        if abs(w[s]) < 1e-15:
            continue
        # For lag k, dC_k/dh[j] for each j:
        # (1-h[j-k]) if j-k in [0,n-1] minus h[j+k] if j+k in [0,n-1]
        for j in range(n):

            jm = j - k
            if 0 <= jm < n:

            jp = j + k
            if 0 <= jp < n:
                term -= h[jp]
            grad[j] += w[s] * term

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    w = np.exp((corr - m) / eps)

        k = s - (n - 1)
        ws = w[s]
        if ws < 1e-15:
            continue

        # Vectorized:
        # (1-h[j-k]) for j where j-k in [0,n-1]
        # h[j+k] for j where j+k in [0,n-1]

        j_end = min(n, n + k)

        j_end2 = min(n, n - k)

            grad[j_start2:j_end2] -= ws * h[j_start2 + k:j_end2 + k]
    
    return grad
