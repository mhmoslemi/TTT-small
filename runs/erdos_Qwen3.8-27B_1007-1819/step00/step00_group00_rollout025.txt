    # Try a few other n values
    for n2 in [44, 48, 50, 60, 40]:
        if time.time() > deadline:
            break
        target2 = n2 / 2.0
        bounds2 = [(0.0, 1.0)] * n2
        def eq_con2(p):
            return np.sum(p) - target2
        constraints2 = [{'type': 'eq', 'fun': eq_con2}]

        # Initialize: interpolate from best or use uniform
        h2 = np.interp(
            np.linspace(0, 1, n2, endpoint=False),
            np.linspace(0, 1, len(best_h), endpoint=False),
            best_h
        )
        # Shift interpolation
        x_old = np.linspace(0, 1, len(best_h))
        x_new = np.linspace(0, 1, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = fix_sum(h2, target2)

        for T in [0.1, 0.05, 0.02, 0.01, 0.005]:
            if time.time() > deadline:
                break
            h_new = try_optimize(h2, T, n2, target2, bounds2, constraints2, 400)
            if h_new is not None:
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n2
                    # Re-optimize with tighter T
                    h_new = try_optimize(h_new, 0.002, n2, target2, bounds2, constraints2, 300)
                    if h_new is not None:
                        c5_new2 = compute_c5(h_new)
                        if c5_new2 < best_c5:
                            best_c5 = c5_new2
                            best_h = h_new.copy()

    # Final polish with very low temperature
    if time.time() < deadline - 5:
        nfin = len(best_h)
        targetfin = nfin / 2.0
        boundsfin = [(0.0, 1.0)] * nfin
        def eq_fin(p):
            return np.sum(p) - targetfin
        con_fin = [{'type': 'eq', 'fun': eq_fin}]
        for T in [0.001, 0.0005]:
            if time.time() > deadline:
                break
            h_new = try_optimize(best_h, T, nfin, targetfin, boundsfin, con_fin, 500)
            if h_new is not None:
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Verify sum constraint
    nfinal = len(best_h)
    best_h = fix_sum(best_h, nfinal / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, nfinal
