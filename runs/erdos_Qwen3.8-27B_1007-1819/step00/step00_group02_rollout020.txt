def smooth_max_corr(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return (mx + np.log(s) / beta) * dx

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def obj_b40(x):
    return smooth_max_corr(fix_feasible(x), 40.0)

def obj_b80(x):
    return smooth_max_corr(fix_feasible(x), 80.0)

def obj_b200(x):
    return smooth_max_corr(fix_feasible(x), 200.0)

def obj_b400(x):
    return smooth_max_corr(fix_feasible(x), 400.0)

def gen_start(n, r, best_h):
    if r == 0:
        return tm_seq(n)
    if r == 1:
        x = np.arange(n)
        return 0.5 + 0.35 * np.sin(4.0 * np.pi * x / n)
    if r == 2:
        return (np.random.random(n) > 0.5).astype(float)
    if r == 3:
        h = np.random.uniform(0, 1, n)
        k = max(1, n // 10)
        return np.convolve(h, np.ones(k) / k, mode='same')
    if r == 4:
        blk = max(2, n // 8)
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i // blk) % 2 == 0 else 0.0
        return h
    if r == 5:
        x = np.arange(n)
        return 0.5 + 0.3 * np.cos(2.0 * np.pi * 3.0 * x / n)
    if best_h is not None and len(best_h) == n:
        return best_h + 0.2 * np.random.randn(n)
    return 0.5 + 0.3 * np.random.randn(n)

def coord_refine(h, max_rounds=20):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    deltas = [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.25, -0.25]
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            h_trial = h.copy()
            for d in deltas:
                v = h[i] + d
                if v < 0 or v > 1:
                    continue
                h_trial[i] = v
                h_trial = fix_feasible(h_trial)
                obj = compute_c5(h_trial)
                if obj < cur - 1e-10:
                    cur = obj
                    h = h_trial.copy()
                    improved = True
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    best_h = None
    best_c5 = 1e10

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is not None:
        h_init = fix_feasible(h_init)
        c5v = compute_c5(h_init)
        best_c5 = c5v
        best_h = h_init.copy()

    sizes = []
    if h_init is not None:
        sizes.append(len(h_init))
    for nn in [30, 36, 42, 48, 54, 60, 72, 84, 100]:
        if nn not in sizes:
            sizes.append(nn)

    for n in sizes:
        rem = budget_s * 0.9 - (time.time() - t0)
        if rem < 5.0:
            break
        n_trials = max(2, min(6, int(rem / max(n, 40))))

        for r in range(n_trials):
            if time.time() - t0 > budget_s * 0.95:
                break
            if r == 0 and h_init is not None and n == len(h_init):
                h = h_init.copy()
            else:
                h = gen_start(n, r, best_h)
            h = fix_feasible(h)

            c5v = compute_c5(h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h.copy()

            obj_fns = [obj_b40, obj_b80]
            for objf in obj_fns:
                if time.time() - t0 > budget_s * 0.96:
                    break
                try:
                    res = minimize(objf, h, method='SLSQP',
                                  constraints=[{'type': 'eq', 'fun': sum_eq}],
                                  bounds=[(0, 1)] * n,
                                  options={'maxiter': 400, 'ftol': 1e-9})
                    h_new = fix_feasible(res.x)
                    c5v = compute_c5(h_new)
                    if c5v < best_c5:
                        best_c5 = c5v
                        best_h = h_new.copy()
                    h = h_new
                except Exception:
                    pass

            if n <= 60 and time.time() - t0 < budget_s * 0.95:
                h_cd = coord_refine(h, max_rounds=15)
                c5v = compute_c5(h_cd)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_cd

    if best_h is not None and time.time() - t0 < budget_s * 0.97:
        nb = len(best_h)
        for objf in [obj_b200, obj_b400]:
            if time.time() - t0 > budget_s * 0.985:
                break
            try:
                res = minimize(objf, best_h, method='SLSQP',
                              constraints=[{'type': 'eq', 'fun': sum_eq}],
                              bounds=[(0, 1)] * nb,
                              options={'maxiter': 500, 'ftol': 1e-11})
                h_f = fix_feasible(res.x)
                c5f = compute_c5(h_f)
                if c5f < best_c5:
                    best_c5 = c5f
                    best_h = h_f
            except Exception:
                pass

    if best_h is not None and len(best_h) <= 60 and time.time() - t0 < budget_s * 0.985:
        h_cd = coord_refine(best_h, max_rounds=10)
        c5v = compute_c5(h_cd)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_cd

    if best_h is None:
        best_h = fix_feasible(np.full(45, 0.5))
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
