    new_h[j] -= amt

        new_h[k] = 0

        new_h[k] = 1

        new_h[j] = 0
        mask = new_h < 1.0

            new_h[mask] += excess / np.sum(mask)
            new_h = np.clip(new_h, 0, 1)

        new_h[j] = 1
        mask = new_h > 0.0
        if np.sum(mask) > 0:
            new_h[mask] -= excess / np.sum(mask)
            new_h = np.clip(new_h, 0, 1)

    n = len(h)

    return float(np.max(corr) * dx)

def soft_c5(h, alpha=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    shifted = corr - m

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        total = np.sum(room)
        if total < 1e-14:
            return h
        h += (diff / total) * room
    else:
        total = np.sum(h)
        if total < 1e-14:
            return h
        h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

            if elapsed() > budget_s * 0.92:
                break
            scale = 0.015 * (1.0 + 0.3 * rng.random())
            h_init = best_h + rng.normal(0, scale, n)
            h_init = project_feasible(h_init, n)
            h_new = slsqp_opt(h_init, n, alpha=300.0, max_iter=200)
            v = c5_value(h_new)
            if v < best_c5:
                best_h, best_c5 = h_new, v
                # Coordinate descent after improvement
                if elapsed() < budget_s * 0.92:
                    best_h, best_c5 = coord_descent(best_h, n, n_passes=1)

    # Phase 6: Final SLSQP polish
    if elapsed() < budget_s * 0.97:
        h_new = slsqp_opt(best_h, n, alpha=500.0, max_iter=400)
        v = c5_value(h_new)
        if v < best_c5:
            best_h, best_c5 = h_new, v

    best_h = project_feasible(best_h, n)
    best_c5 = c5_value(best_h)
    return best_h, best_c5, n
