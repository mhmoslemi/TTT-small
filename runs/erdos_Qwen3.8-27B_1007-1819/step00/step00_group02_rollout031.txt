from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_objective(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c = np.max(corr)
    if temp < 1e-10:
        return c
    return c + (1.0 / temp) * np.log(np.sum(np.exp(temp * (corr - c))))

def slsqp_refine(h, target_sum, temp, maxiter=200):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(
            lambda x: smooth_objective(x, temp),
            h, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        x = np.clip(res.x, 0, 1)
        x = x + (target_sum - np.sum(x)) / n
        x = np.clip(x, 0, 1)
        return x, compute_c5(x)
    except Exception:
        return h, compute_c5(h)

    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:

    # Distribute uniformly
    step = diff / len(h)

    # If still not feasible, iterate
    for _ in range(10):
