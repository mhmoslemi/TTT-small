import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max(x, alpha):
    x = np.asarray(x, dtype=float)
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(p, dx, alpha, target, pen):
    h = np.clip(p, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    base = soft_max(corr, alpha)
    s = np.sum(h) - target
    return base + pen * s * s

def optimize_h(h_init, n, max_alphas=4):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h = fix_sum(h_init.astype(float), target)
    alphas = [15, 50, 200, 600, 1500][:max_alphas]
    for alpha in alphas:
        pen = max(5.0, 0.01 * alpha)
        try:
            res = opt.minimize(obj_fn, h, args=(dx, alpha, target, pen),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 800, 'gtol': 1e-13})
            h = fix_sum(res.x, target)
        except Exception:
            pass
    # Polish: round near-binary values
    h = np.where(h < 0.05, 0.0, h)
    h = np.where(h > 0.95, 1.0, h)
    h = fix_sum(h, target)
    # One more optimization pass after rounding
    try:
        res = opt.minimize(obj_fn, h, args=(dx, 600, target, 50.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'gtol': 1e-14})
        h = fix_sum(res.x, target)
    except Exception:
        pass
    return h
