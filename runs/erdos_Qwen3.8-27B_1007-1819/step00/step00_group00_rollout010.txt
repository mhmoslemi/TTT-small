    weights = np.exp(shifted)
    weights /= np.sum(weights)
    obj = max_c + np.log(np.sum(weights)) / temp  # = max_c + logsumexp/temp - max_c = logsumexp/temp
    # Actually: (1/temp)*log(sum(exp(temp*corr))) = (1/temp)*(temp*max_c + log(sum(exp(temp*(corr-max_c)))))
    # = max_c + (1/temp)*log(sum(weights*sum(weights))) = max_c + (1/temp)*log(sum(exp(shifted)))
    # Since weights = exp(shifted)/sum(exp(shifted)):
    # sum(exp(shifted)) = sum(weights) * sum(exp(shifted)) / sum(weights) ... 
    # Let me just compute directly:
    obj = max_c + np.log(np.sum(np.exp(shifted))) / temp
    # Gradient: g[j] = dx * sum_m w[m] * (1 - 2*h[m-j]) for valid m,j
    # = dx * correlate(w, 1-2*h, "valid")[j]
    grad = np.correlate(weights, 1.0 - 2.0*h, mode="valid") * dx
    return obj, grad
