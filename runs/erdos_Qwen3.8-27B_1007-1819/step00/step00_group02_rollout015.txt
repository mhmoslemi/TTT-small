def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(40):
        h = np.clip(h, 0.0, 1.0)
        d = target_sum - h.sum()
