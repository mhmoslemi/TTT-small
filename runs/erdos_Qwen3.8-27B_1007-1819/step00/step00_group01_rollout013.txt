import numpy as np
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_c5_and_grad(h):
    N = len(h)
    dx = 2.0 / N
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode="full")
    idx = int(np.argmax(corr))
    c5 = float(corr[idx] * dx)
    lag = idx - (N - 1)
    j = np.arange(N)
    g = np.zeros(N)
    m1 = (j - lag >= 0) & (j - lag < N)
    m2 = (j + lag >= 0) & (j + lag < N)
    g[m1] += (1.0 - h[j[m1] - lag])
    g[m2] -= h[j[m2] + lag]
    return c5, g * dx


def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - float(np.sum(h))
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def round_to_binary(h, target_sum):
    N = len(h)
    n_ones = N // 2
    n_half = N - 2 * n_ones
    order = np.argsort(h)[::-1]
    h_new = np.zeros(N)
    h_new[order[:n_ones]] = 1.0
    if n_half > 0:
        h_new[order[n_ones:n_ones + n_half]] = 0.5
    return project(h_new, target_sum)


def optimize(h0, t_end, lr0, momentum, target_sum, rng):
    N = len(h0)
    h = h0.copy()
    buf = np.zeros(N)
    lr = lr0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    stall = 0
    while time.time() < t_end:
        _, g = compute_c5_and_grad(h)
        buf = momentum * buf - lr * g
        h_new = h + buf
        h_new = project(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5:
            h = h_new
            cur_c5 = c5_new
            lr = min(lr * 1.01, 0.5)
            stall = 0
        else:
            lr *= 0.85
            stall += 1
            if stall > 300:
                h = best_h + rng.randn(N) * 0.05
                h = project(h, target_sum)
                buf = np.zeros(N)
                lr = lr0 * 0.3
                cur_c5 = compute_c5(h)
                stall = 0
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5
