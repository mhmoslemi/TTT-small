            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n, alpha, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (target,)}]
    res = minimize(
        smooth_obj, h0, args=(alpha,),
        method='SLSQP', bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return project_feasible(res.x, n)

def lfgs_optimize(h0, n, alpha, penalty_w, maxiter=500):
    target = n / 2.0
    def obj_pen(x):
        base = smooth_obj(x, alpha)
        pen = penalty_w * (np.sum(x) - target) ** 2
        return base + pen
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_pen, h0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def binary_search(h, n, max_rounds=200):
    dx = 2.0 / n
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    target = n // 2
    while np.sum(h) > target:
        h[np.argmax(h)] = 0.0
    while np.sum(h) < target:
        h[np.argmin(h)] = 1.0
    best_c5 = compute_c5(h, n)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_pair = None
        best_val = best_c5
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h, n)
                if c5 < best_val - 1e-14:
                    best_val = c5
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_pair is not None:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            best_c5 = best_val
            improved = True
        if not improved:
            break
    return h, best_c5

def sa_optimize(h0, n, t_budget, rng):
    h = h0.copy()
    best_c5 = compute_c5(h, n)
    T = 0.03
    T_min = 0.0005
    cooling = 0.9995
    while T > T_min:
        if time.time() > t_budget:
            break
        n_pert = max(1, n // 15)
        idx = rng.choice(n, size=n_pert, replace=False)
        pert = rng.uniform(-0.08, 0.08, size=n_pert)
        h_new = h.copy()
        h_new[idx] += pert
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        c5_old = compute_c5(h, n)
        delta = c5_new - c5_old
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
        T *= cooling
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init)

    best_c5 = compute_c5(h_init, n_init)
    best_h = h_init.copy()
    best_n = n_init

    # Phase 1: SLSQP on initial
    for alpha in [10.0, 30.0, 80.0, 200.0]:
        if remaining() < 20:
            break
        h_res = slsqp_optimize(h_init, n_init, alpha, maxiter=200)
        c5 = compute_c5(h_res, n_init)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_res.copy()
            best_n = n_init
        h_init = h_res

    # Phase 2: L-BFGS-B with penalty
    for alpha in [50.0, 150.0]:
        if remaining() < 20:
            break
        h_res = lfgs_optimize(best_h, best_n, alpha, penalty_w=50.0, maxiter=300)
        c5 = compute_c5(h_res, best_n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_res.copy()

    # Phase 3: Try other n values
    n_candidates = [44, 46, 50, 60, 80, 100, 120, 160]
    for n_try in n_candidates:
        if remaining() < 50:
            break
        if n_try % 2 != 0:
            continue
        h_try = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, best_n), best_h)
        h_try = project_feasible(h_try, n_try)
        for alpha in [30.0, 100.0]:
            if remaining() < 15:
                break
            h_res = slsqp_optimize(h_try, n_try, alpha, maxiter=200)
            c5 = compute_c5(h_res, n_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_res.copy()
                best_n = n_try
            h_try = h_res

    # Phase 4: Binary local search (even n)
    if remaining() > 80:
        for n_bin in [44, 46, 50, 60, 80, 100]:
            if remaining() < 20:
                break
            h_bin = np.interp(np.linspace(0, 1, n_bin), np.linspace(0, 1, best_n), best_h)
            h_bin, c5_bin = binary_search(h_bin, n_bin, max_rounds=100)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n_bin

    # Phase 5: Simulated annealing
    if remaining() > 60:
        t_end = time.time() + remaining() * 0.7
        h_sa, c5_sa = sa_optimize(best_h, best_n, t_end, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

        # SA on binary
        if remaining() > 30:
            h_bin = best_h.copy()
            h_bin = np.where(h_bin > 0.5, 1.0, 0.0)
            target = best_n // 2
            while np.sum(h_bin) > target:
                h_bin[np.argmax(h_bin)] = 0.0
            while np.sum(h_bin) < target:
                h_bin[np.argmin(h_bin)] = 1.0
            t_end2 = time.time() + remaining() * 0.5
            h_sa2, c5_sa2 = sa_optimize(h_bin, best_n, t_end2, rng)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()

    # Phase 6: Final polish
    if remaining() > 15:
        for alpha in [200.0, 500.0]:
            if remaining() < 5:
                break
            h_pol = slsqp_optimize(best_h, best_n, alpha, maxiter=400)
            c5_pol = compute_c5(h_pol, best_n)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
