import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(x, n, alpha):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    return m + np.log(np.sum(np.exp((vals - m) * alpha))) / alpha

def fix_sum(h, n):
    d = n / 2.0 - h.sum()
    if abs(d) < 1e-12:
        return h
    tol = (h > 1e-6) & (h < 1.0 - 1e-6)
    if tol.sum() == 0:
        return np.clip(h, 0, 1)
    h = h.copy()
    h[tol] += d / tol.sum()
    return np.clip(h, 0, 1)

def snap_binary(h, n):
    k = n // 2
    if k == 0:
        return np.zeros(n)
    out = np.zeros(n)
    out[np.argsort(h)[::-1][:k]] = 1.0
    return out

def try_slsqp(h0, n, alpha, t_limit, t0):
    if time.time() - t0 > t_limit:
        return h0, compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n / 2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h0, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        h_out = fix_sum(np.clip(res.x, 0, 1), n)
        return h_out, compute_c5(h_out)
    except Exception:
        return h0, compute_c5(h0)

    h_ls = h0.copy()

            h_new[idx] += np.random.uniform(-0.3, 0.3)
        elif np.random.random() < 0.7:
            i1, i2 = np.random.choice(n, 2, replace=False)
            h_new = h_ls.copy()
            s = np.random.uniform(-0.15, 0.15)
            h_new[i1] += s
            h_new[i2] -= s
        else:
            i1, i2 = np.random.choice(n, 2, replace=False)
            h_new = h_ls.copy()
            h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
        h_new = np.clip(h_new, 0, 1)
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_ls or (c5_new - c5_ls) < 0.02 and np.random.random() < np.exp(-(c5_new - c5_ls) / max(T, 1e-8)):
            h_ls = h_new
            c5_ls = c5_new
        T *= 0.997
        step += 1
    return h_ls, c5_ls
