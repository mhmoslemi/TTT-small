def compute_subgrad(h, n, dx, active_shifts):
    grad = np.zeros(n)
    for s in active_shifts:
        # (1-h[t+s]) for valid t: i.e., t from max(0,-s) to min(n-1, n-1-s)
        # -h[t-s] for valid t: i.e., t from max(0,s) to min(n-1, n-1+s)
        if s >= 0:
            # t+s valid: t from 0 to n-1-s
            grad[:n-s] += (1.0 - h[s:])
            # t-s valid: t from s to n-1
            grad[s:] -= h[:n-s]
        else:
            s_abs = -s
            # t+s valid: t from s_abs to n-1
            grad[s_abs:] += (1.0 - h[:n-s_abs])
            # t-s valid: t from 0 to n-1-s_abs
            grad[:n-s_abs] -= h[s_abs:]
    grad *= dx / len(active_shifts)
    return grad

            best_h = pair_search(best_h, n, dx, rng, t_rem * 0.3)
            best_c5 = eval_c5(best_h, n)

    if best_h is None:
        n = 44
        best_h = project(np.full(n, 0.5), n)
        best_n = n
        best_c5 = eval_c5(best_h, n)

    return best_h, best_c5, best_n
