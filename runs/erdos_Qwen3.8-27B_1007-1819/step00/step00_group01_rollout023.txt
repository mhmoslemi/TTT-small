So df_k/dh[i] = dx * [(1-h[i-s]) * 1_{0<=i-s<n} - h[i+s] * 1_{0<=i+s<n}]

And d(smooth_max)/dh[i] = sum_k weights[k] * df_k/dh[i]

This is still O(n^2) per component if done naively, but we can vectorize.

import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
            if diff < 1e-14:
                break
    else:
        order = np.argsort(h)[::-1]
        for i in order:
            rem = min(h[i], -diff)
            h[i] -= rem
            diff += rem
            if diff > -1e-14:
                break
    return h


def smooth_max_val(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    m = f.max()
    fc = np.minimum(f - m, 0.0)
    e = np.exp(T * fc)
    return m + np.log(e.sum()) / T


def smooth_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    m = f.max()
    fc = np.minimum(f - m, 0.0)
    e = np.exp(T * fc)
    w = e / e.sum()
    # For shift s = k-(n-1): df_k/dh[i] = dx*[(1-h[i-s])*1_{valid} - h[i+s]*1_{valid}]
    grad = np.zeros(n)
    shifts = np.arange(-(n - 1), n)
    for si, s in enumerate(shifts):
        wk = w[s + n - 1]
        if wk < 1e-16:
            continue
        # (1-h[i-s]) for valid i-s
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += wk * dx * (1.0 - h[idx - s])
        # -h[i+s] for valid i+s
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= wk * dx * h[idx + s]
    return grad


def optimize_continuous(h_init, n, t_max, T0=20.0, T1=80.0):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h = h_init.copy()
    best_val = evaluate_c5(h_init)
    
    for T in [T0, 30.0, 50.0, T1]:
        def obj(x):
            return smooth_max_val(x, n, T)
        
        def jac(x):
            return smooth_max_grad(x, n, T)
        
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h_init, method='SLSQP', jac=jac,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-12})
            h_try = make_feasible(res.x, n)
            c5 = evaluate_c5(h_try)
            if c5 < best_val:
                best_val = c5
                best_h = h_try.copy()
                h_init = h_try.copy()
        except Exception:
            pass
    
    return best_h, best_val


def local_binary_search(h, n, n_iter=2000):
    """Perturbation search: flip/adjust individual elements."""
    target_sum = n / 2.0
    best = h.copy()
    best_c5 = evaluate_c5(best)
    rng = np.random.default_rng(12345)
    
    for it in range(n_iter):
        i = rng.integers(n)
        delta = 0.1 * rng.standard_normal()
        h_new = best.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        # Adjust another element to maintain sum
        diff = best_c5 and (target_sum - h_new.sum())
        diff = target_sum - h_new.sum()
        if abs(diff) > 1e-14:
            j = rng.integers(n)
            if j == i:
                j = (j + 1) % n
            adj = np.clip(diff / 2.0, -1.0, 1.0)
            h_new[i] = np.clip(h_new[i] - adj, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] + (diff - (h_new[i] - best[i])), 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best = h_new
            best_c5 = c5
    
    return best, best_c5
