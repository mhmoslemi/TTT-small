        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')

        corr = np.correlate(h, one_minus_h, mode='full') * dx

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def _fix_sum(v, ts, n):
    v = np.array(v, dtype=float)
    v += (ts - v.sum()) / n
    v = np.clip(v, 0.0, 1.0)
    for _ in range(12):
        d = ts - v.sum()
        if abs(d) < 1e-14:
            break
        free = (v > 1e-12) & (v < 1.0 - 1e-12)
        if not np.any(free):
            v += d / n
        else:
            v[free] += d / np.sum(free)
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5(v, dx):
    return float(np.max(np.correlate(v, 1.0 - v, mode='full')) * dx)

def _smax(v, t, dx):
    c = np.correlate(v, 1.0 - v, mode='full') * dx
    return float(t * logsumexp(c / t))

def _slsqp_opt(h, ts, n, dx, bounds, cons, t0, budget_s, frac):
    best = h.copy()
    best_c = _c5(best, dx)
    for t in [0.04, 0.015, 0.005, 0.0015, 0.0005]:

            r = minimize(lambda x: _smax(x, t, dx), best, method='SLSQP',
                        bounds=bounds, constraints=cons,

            v = _fix_sum(r.x, ts, n)
            cv = _c5(v, dx)
            if cv < best_c:
                best_c = cv
                best = v
        except Exception:
            pass
    return best, best_c
