import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def logsumexp_vals(vals, t):
    m = float(np.max(vals))
    return m + float(np.log(np.sum(np.exp((vals - m) * t)) / len(vals))) / t

def project_sum(h, S):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, dx, t):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return logsumexp_vals(corr, t)

def optimize_slqp(h_start, dx, S, n, t, maxiter=300):
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        return smooth_obj(hc, dx, t)
    def cons_eq(x):
        return float(np.sum(x) - S)
    try:
        res = minimize(obj, h_start, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_res = project_sum(res.x, S)
        return h_res
    except Exception:
        return h_start

def random_perturb(h, rng, amp):
    h_try = h + rng.uniform(-amp, amp, len(h))
    return h_try

def swap_perturb(h, rng):
    h_try = h.copy()
    i, j = rng.choice(len(h), size=2, replace=False)
    h_try[i], h_try[j] = h_try[j], h_try[i]
    return h_try

def block_shift_perturb(h, rng):
