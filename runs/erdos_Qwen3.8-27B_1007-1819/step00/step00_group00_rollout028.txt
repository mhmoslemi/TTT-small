    if h_init is None or len(h_init) < 4:
        n = 45
        h_init = np.full(n, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h_init = project_feasible(h_init, target)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def obj(x):
        return compute_c5(x)

    def con_eq(x):
        return x.sum() - target

    cons = [{'type': 'eq', 'fun': con_eq}]

    # Phase 1: SLSQP from parent
    while time.time() < deadline:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False})
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
        else:
            break

    # Phase 2: Nelder-Mead with penalty (escape local minima)
    if time.time() < deadline:
        def obj_pen(x):
            h = np.clip(x, 0.0, 1.0)
            pen = 50.0 * (h.sum() - target) ** 2
            return compute_c5(h) + pen

        res = minimize(obj_pen, best_h, method='Nelder-Mead',
                       options={'maxiter': 30000, 'xatol': 1e-10,
                                'fatol': 1e-13, 'adaptive': True})
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: COBYLA polish
    if time.time() < deadline:
        res = minimize(obj, best_h, method='COBYLA',
                       constraints=[con_eq],
                       bounds=bounds,
                       options={'maxiter': 8000, 'tol': 1e-12, 'rhobeg': 0.05})
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Random perturbations + SLSQP
    scales = np.array([0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5])
    while time.time() < deadline:
        s = rng.choice(scales)
        h_r = best_h + rng.normal(0.0, s, n)
        h_r = project_feasible(h_r, target)
        res = minimize(obj, h_r, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False})
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
            # Re-run SLSQP from improved point
            if time.time() < deadline:
                res2 = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                                constraints=cons,
                                options={'maxiter': 3000, 'ftol': 1e-14, 'disp': False})
                h_new2 = project_feasible(res2.x, target)
                c5_new2 = compute_c5(h_new2)
                if c5_new2 < best_c5:
                    best_h = h_new2
                    best_c5 = c5_new2

    # Phase 5: Targeted local search on the max-correlation lag
    if time.time() < deadline:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        k_max = int(np.argmax(corr))
        # Try small coordinated moves to reduce the dominant lag
        for trial in range(20):
            if time.time() > deadline:
                break
            ind = rng.choice(n, size=min(n, 6), replace=False)
            h_try = best_h.copy()
            for i in ind:
                h_try[i] += rng.uniform(-0.05, 0.05)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_h = h_try
                best_c5 = c5_try
                if time.time() < deadline:
                    res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                                   constraints=cons,
                                   options={'maxiter': 2000, 'ftol': 1e-14, 'disp': False})
                    h_new = project_feasible(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new

    # Final feasibility check
    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
