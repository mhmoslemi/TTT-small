            current_h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
        
        iterations += 1
        if iterations % 10000 == 0:
            T *= cooling
            if T < 0.001:

    # Phase 2: Local search (pair adjustments)
    end2 = start_time + budget_s * 0.99
    h = best_h.copy()
    c5 = best_c5
    while time.time() < end2:
        improved = False
        for i in range(n):
            if time.time() > end2:
                break
            for j in range(n):
                if i == j:
                    continue
                for d in [0.02, 0.005, 0.001]:
                    h_try = h.copy()
                    h_try[i] += d
                    h_try[j] -= d
                    h_try = repair(h_try, target_sum)
                    c5_try = evaluate_c5(h_try)
                    if c5_try < c5 - 1e-12:
                        h = h_try
                        c5 = c5_try
                        improved = True
                        if c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = c5
                    h_try = h.copy()
                    h_try[i] -= d
                    h_try[j] += d
                    h_try = repair(h_try, target_sum)
                    c5_try = evaluate_c5(h_try)
                    if c5_try < c5 - 1e-12:
                        h = h_try
                        c5 = c5_try
                        improved = True
                        if c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = c5
        if not improved:
            break
    
    return best_h, best_c5, n

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if s < 1e-15:
        h[:] = target_sum / len(h)
        return h
    h *= target_sum / s
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s2 = float(np.sum(h))
        diff = target_sum - s2
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def gen_move(h, n, rng, mt, T):
    hn = h.copy()
    if mt == 0:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        while j == i:
            j = int(rng.integers(0, n))
        d = rng.uniform(-0.06, 0.06)
        hn[i] += d
        hn[j] -= d
    elif mt == 1:
        L = int(rng.integers(2, min(8, n // 2 + 1)))
        i = int(rng.integers(0, n - L))
        j = int(rng.integers(0, n - L))
        ok = False
        for _ in range(30):
            if not (j + L > i and j < i + L):
                ok = True
                break
            j = int(rng.integers(0, n - L))
        if ok:
            tmp = hn[i:i+L].copy()
            hn[i:i+L] = hn[j:j+L]
            hn[j:j+L] = tmp
        else:
            mt = 0
            i2 = int(rng.integers(0, n))
            j2 = int(rng.integers(0, n))
            while j2 == i2:
                j2 = int(rng.integers(0, n))
            d = rng.uniform(-0.06, 0.06)
            hn[i2] += d
            hn[j2] -= d
    elif mt == 2:
        i = int(rng.integers(0, n))
        d = 1.0 if hn[i] < 0.5 else -1.0
        hn[i] += d
        j = int(rng.integers(0, n))
        while j == i:
            j = int(rng.integers(0, n))
        hn[j] -= d
    elif mt == 3:
        mask = rng.random(n) < 0.12
        cnt = int(mask.sum())
        if cnt > 0:
            hn[mask] = rng.random(cnt)
    else:
        sigma = 0.015 + 0.01 * (T / 0.02)
        hn += rng.normal(0, sigma, n)
    return hn

def local_refine(h, n, target_sum, budget_end, rng):
    c5 = evaluate_c5(h)
    deltas = [0.03, 0.01, 0.003, 0.001, 0.0003]
    while time.time() < budget_end:
        improved = False
        order = rng.permutation(n)
        for ii in range(n):
            if time.time() > budget_end:
                break
            i = int(order[ii])
            for d in deltas:
                for sgn in [1, -1]:
                    ht = h.copy()
                    ht[i] += sgn * d
                    # spread compensation
                    rem = n - 1
                    if rem > 0:
                        others = np.concatenate([np.arange(i), np.arange(i+1, n)])
                        ht[others] -= sgn * d / rem
                    ht = repair(ht, target_sum)
                    c5t = evaluate_c5(ht)
                    if c5t < c5 - 1e-13:
                        h = ht
                        c5 = c5t
                        improved = True
        if not improved:
            break
    return h, c5
