        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except (NameError, AttributeError):
        n0 = 50
        h0 = np.zeros(n0)
        h0[0::2] = 1.0
    best_h = h0.copy()
    best_c5, _ = compute_c5_and_corr(best_h, n0)

    # Phase 1: scipy smooth optimization
    if time.time() - t0 < budget_s * 0.15:
        n = n0
        def smooth_obj(x):
            hx = project_feasible(x, n)
            dx = 2.0 / n
            corr = np.correlate(hx, 1.0 - hx, mode="full") * dx
            mx = np.max(corr)
            t_s = 2.0
            return mx + t_s * np.log(np.mean(np.exp((corr - mx) / t_s)))
        res = minimize(smooth_obj, h0, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': 2000, 'ftol': 1e-14})
        hc = project_feasible(res.x, n)
        c5c, _ = compute_c5_and_corr(hc, n)
        if c5c < best_c5:
            best_h, best_c5 = hc, c5c

    # Phase 2: intensive local search on n0
    if time.time() - t0 < budget_s * 0.55:
        elapsed = time.time() - t0
        rem = budget_s * 0.55 - elapsed
        best_h, best_c5 = multi_local_search(best_h, n0, max(rem, 5.0))

    # Phase 3: try even n values
    for n_try in [46, 48, 50, 54, 58, 62]:
        if time.time() - t0 > budget_s * 0.82:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_b = binarize(h_interp, n_try)
        hc, c5c = binary_local_search(h_b, n_try, max_iter=150000, use_anneal=True, t0=10.0)
        if c5c < best_c5:
            best_h, best_c5 = hc, c5c
            # further polish
            if time.time() - t0 < budget_s * 0.82:
                best_h, best_c5 = multi_local_search(best_h, n_try, 30.0)

    # Phase 4: random restarts on best n
    n_best = len(best_h)
    if time.time() - t0 < budget_s * 0.93:
        end_t = time.time() + (budget_s * 0.93 - (time.time() - t0))
        for restart in range(20):
            if time.time() > end_t:
                break
            h_rand = np.zeros(n_best)
            n_ones = int(n_best / 2)
            positions = np.random.choice(n_best, n_ones, replace=False)
            h_rand[positions] = 1.0
            if n_best % 2 == 1:
                h_rand = project_feasible(h_rand, n_best)
            hc, c5c = multi_local_search(h_rand, n_best, min(15.0, end_t - time.time()))
            if c5c < best_c5:
                best_h, best_c5 = hc, c5c

    # Phase 5: final polish
    if time.time() - t0 < budget_s * 0.98:
        rem = budget_s * 0.98 - (time.time() - t0)
        best_h, best_c5 = multi_local_search(best_h, len(best_h), max(rem, 2.0))

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5_and_corr(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
