def og(h, T):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1-h, "full") * dx
    m = c.max()
    e = np.exp(T*(c-m))
    S = e.sum()

def c5(h):

    return float(np.max(np.correlate(h, 1-h, "full")) * 2.0/n)

    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        d = s - h.sum()

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full") * dx
    m = c.max()

import numpy as np
import time

def c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * 2.0 / n)

def proj(h, s):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        d = s - h.sum()
        if abs(d) < 1e-15:
            break
        h = np.clip(h + d / n, 0.0, 1.0)
    return h

def og(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full") * dx
    m = c.max()
    e = np.exp(T * (c - m))
    S = e.sum()
    w = e / S
    obj = m + np.log(S) / T
    g = np.zeros(n)
    for lag in range(-(n - 1), n):
        wi = w[n - 1 + lag] * dx
        if wi < 1e-15:
            continue
        a, b = max(0, -lag), min(n - 1, n - 1 - lag)
        if a <= b:
            g[a:b + 1] += wi * (1.0 - h[a + lag:b + 1 + lag])
        a2, b2 = max(0, lag), min(n - 1, n - 1 + lag)
        if a2 <= b2:
            g[a2:b2 + 1] -= wi * h[a2 - lag:b2 + 1 - lag]
    return obj, g

def pdesc(h, Tlo, Thi, nit, end):
    n = len(h)
    s = n / 2.0
    h = proj(h, s)
    bh, bc = h.copy(), c5(h)
    for it in range(nit):
        if time.time() > end:
            break
        T = Tlo * (Thi / Tlo) ** (it / max(nit, 1))
        _, g = og(h, T)
        gp = g - g.mean()
        gn = np.linalg.norm(gp)
        if gn < 1e-12:
            break
        lr = 0.1 / max(gn, 0.5)
        h = proj(h - lr * gp, s)
        cur = c5(h)
        if cur < bc:
            bh, bc = h.copy(), cur
    return bh, bc

def bswap(h, ntry, end):
    n, s = len(h), len(h) / 2.0
    h = proj(h, s)
    bc = c5(h)
    for _ in range(ntry):
        if time.time() > end:
            break
        imp = False
        i, j = int(np.argmax(h)), int(np.argmin(h))

            hn[i], hn[j] = hn[j], hn[i]
            hn = proj(hn, s)

        i2, j2 = int(np.random.randint(n)), int(np.random.randint(n))

            hn = proj(hn, s)
            if c5(hn) < bc - 1e-13:
                h, bc, imp = hn, c5(hn), True
        if not imp:

    return h, bc

def thue_morse(n):
    bits = np.zeros(n, dtype=float)
    for i in range(1, n):
        c, p = i, 0
        while c:
            p ^= (c & 1)
            c >>= 1
        bits[i] = p
    return bits
