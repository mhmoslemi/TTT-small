from scipy.optimize import minimize
import time

def c5_exact(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def c5_smooth(h, beta=200.0):
    n = len(h)
    dx = 2.0/n
    vals = np.correlate(h, 1.0-h, mode='full') * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(beta*(vals-m))))/beta

def c5_smooth_grad(h, beta=200.0):
    n = len(h)
    dx = 2.0/n
    vals = np.correlate(h, 1.0-h, mode='full') * dx
    m = np.max(vals)
    exps = np.exp(beta*(vals-m))
    s = np.sum(exps)
    val = m + np.log(s)/beta
    grad = np.zeros(n)
    eps = 1e-7
    for j in range(n):
        hp = h.copy(); hp[j] += eps
        hm = h.copy(); hm[j] -= eps
        vp = np.correlate(hp, 1.0-hp, mode='full')*dx
        vm = np.correlate(hm, 1.0-hm, mode='full')*dx
        mp = np.max(vp)

        mm = np.max(vm)
        fm = mm + np.log(np.sum(np.exp(beta*(vm-mm))))/beta
        grad[j] = (fp-fm)/(2*eps)
    return val, grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

    h = h0.copy()
    c5 = c5_exact(h)

    best_h = h.copy()

def slsqp_optimize(h0, maxiter=500):

import numpy as np
from scipy.optimize import minimize
import time


def c5_exact(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def c5_smooth_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta


def c5_smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(vals)
    exps = np.exp(beta * (vals - m))
    s = np.sum(exps)
    val = m + np.log(s) / beta
    grad = np.zeros(n)
    eps = 1e-7
    for j in range(n):
        hp = h.copy(); hp[j] += eps
        hm = h.copy(); hm[j] -= eps
        vp = np.correlate(hp, 1.0 - hp, mode='full') * dx
        vm = np.correlate(hm, 1.0 - hm, mode='full') * dx
        mp = np.max(vp)
        fp = mp + np.log(np.sum(np.exp(beta * (vp - mp)))) / beta
        mm = np.max(vm)
        fm = mm + np.log(np.sum(np.exp(beta * (vm - mm)))) / beta
        grad[j] = (fp - fm) / (2.0 * eps)
    return val, grad


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h0, maxiter=400, beta=200.0):
    n = len(h0)
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)

    def obj(h):
        return c5_smooth_val(h, beta)

    def grad(h):
        _, g = c5_smooth_grad(h, beta)
        return g

    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj, h0, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_feasible(result.x, target_sum)
        c_opt = c5_exact(h_opt)
        if c_opt < c5_exact(h0):
            return h_opt, c_opt
    except Exception:
        pass
    return h0, c5_exact(h0)


def sa_optimize(h0, deadline, rng, n_perturb=3, T0=5e-4, T_floor=1e-7):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    c5_cur = c5_exact(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = T0
    stalling = 0
    while time.time() < deadline:
        h_new = h.copy()
        idx = rng.choice(n, size=n_perturb, replace=False)
        h_new[idx] += rng.normal(0, 0.15, size=n_perturb)
        h_new = project_feasible(h_new, target_sum)
        c5_new = c5_exact(h_new)
        delta = c5_new - c5_cur
        if delta < -1e-15 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
                stalling = 0
            else:
                stalling += 1
        else:
            stalling += 1
        T *= 0.9997
        if T < T_floor:
            T = T_floor
        if stalling > 5000:
            T = max(T, T0 * 0.5)
            stalling = 0
    return best_h, best_c5

    x_src = np.arange(n_src) * (2.0 / n_src)
    x_dest = np.arange(n_dest) * (2.0 / n_dest)
    return np.interp(x_dest, x_src, h_src)
