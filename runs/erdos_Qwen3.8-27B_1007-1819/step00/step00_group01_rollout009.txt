import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum, n_pts):
    h = np.clip(h, 0.0, 1.0)
    h += (target_sum - np.sum(h)) / n_pts
    h = np.clip(h, 0.0, 1.0)
    rem = target_sum - np.sum(h)
    if abs(rem) > 1e-13:
        if rem > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if len(free) > 0:
            h[free] += rem / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_obj_grad(alpha, lam, dx, target_sum, n_pts):
    n_l = n_pts
    off = n_l - 1
    k = np.arange(2 * n_l - 1) - off
    p_idx = np.arange(n_l)[:, None]
    k_arr = k[None, :]
    idx1 = p_idx + k_arr
    v1 = (idx1 >= 0) & (idx1 < n_l)
    idx2 = p_idx - k_arr
    v2 = (idx2 >= 0) & (idx2 < n_l)
    idx1_c = np.clip(idx1, 0, n_l - 1)
    idx2_c = np.clip(idx2, 0, n_l - 1)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        sm = m + (1.0 / alpha) * np.log(np.sum(np.exp(np.clip(alpha * (corr - m), -800, 800))))
        pen = lam * (np.sum(x) - target_sum) ** 2
        return sm + pen

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(corr)
        ew = np.exp(np.clip(alpha * (corr - m), -800, 800))
        w = ew / np.sum(ew)
        x1 = np.where(v1, x[idx1_c], 0.0)
        x2 = np.where(v2, x[idx2_c], 0.0)
        wc = w[None, :]
        autocorr = np.sum(wc * x1, axis=1) + np.sum(wc * x2, axis=1)
        gcorr = dx * (1.0 - autocorr)
        gpen = 2.0 * lam * (np.sum(x) - target_sum)
        return gcorr + gpen

    return obj, grad

def optimize_phase(h0, n_pts, dx, target_sum, alpha, lam, maxiter=8000):
    obj, grad = make_obj_grad(alpha, lam, dx, target_sum, n_pts)
    try:
        res = minimize(obj, h0, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n_pts,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                'maxfun': maxiter * 3, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, target_sum, n_pts)
    except Exception:
        pass
    return h0

                valid_s = (i_minus >= 0) & (i_minus < n_pts)
                B = np.where(valid_f, 1.0 - h_cd[np.clip(i_plus, 0, n_pts-1)], 0.0) \
                    - np.where(valid_s, h_cd[np.clip(i_minus, 0, n_pts-1)], 0.0)
                B *= dx
                A = corr0  # corr with h[i]=0
                # Minimize max_m (A[m] + B[m]*t) over t in [0,1]
                # Convex piecewise linear -> bisection on subgradient
                lo, hi = 0.0, 1.0

                    active = np.where(vals >= mx - 1e-14)[0]

                    if sg > 0:
                        hi = mid
                    else:
                        lo = mid
                t_opt = (lo + hi) / 2.0
                t_opt = np.clip(t_opt, 0.0, 1.0)
                h_cd[i] = t_opt
                h_cd = project_feasible(h_cd, target_sum, n_pts)
                h_cd[i] = np.clip(t_opt, 0.0, 1.0)
                if abs(t_opt - x_old) > 1e-10:
                    improved = True
            c5_cd = compute_c5(h_cd, dx)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                h_cur = h_cd.copy()
            if not improved and sweep > 10:
                break

    # Final validation
    best_h = project_feasible(best_h, target_sum, n_pts)
    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_pts
