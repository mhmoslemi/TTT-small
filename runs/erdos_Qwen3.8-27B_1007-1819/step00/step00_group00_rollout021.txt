import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        tr = room.sum()
        if tr > 1e-14:
            h = h + diff * room / tr
    else:
        tm = float(h.sum())
        if tm > 1e-14:
            h = h + diff * h / tm
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def sa_optimize(h, target, n, T_init, t_limit, rng):
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    T = T_init

        else:

            i2 = rng.integers(n)
