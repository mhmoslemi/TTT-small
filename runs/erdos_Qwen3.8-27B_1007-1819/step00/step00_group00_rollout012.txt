import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fast(h, dx):
    """Compute C5 = max_k correlate(h, 1-h) * dx"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_set(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum} using water-filling"""
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(10):  # iterate to convergence
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count == 0:
            break
        adj = diff / count
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

def smooth_obj(h, dx, T):
    """Smooth max via softmax"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = T * (corr - m)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

import numpy as np
from scipy.optimize import minimize
import time

# Module-level parameters for constraint/objective (avoid closures)
_TARGET_SUM = 0.0
_DX = 0.0
_T_PARAM = 1.0

def _smooth_obj(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = T * (corr - m)
    return (m + np.log(np.sum(np.exp(shifted))) / T) * dx

def _obj_fn(x):
    return _smooth_obj(x, _DX, _T_PARAM)

def _sum_con(x):
    return np.sum(x) - _TARGET_SUM

def _sum_con_jac(x):
    return np.ones(len(x))

def compute_c5_fast(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_set(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count == 0:
            break
        h[mask] += diff / count
        np.clip(h, 0.0, 1.0, out=h)
    return h

def do_slsqp(h0, dx, target, T, maxiter=300):

    _TARGET_SUM = target
    _DX = dx
    _T_PARAM = T
    n = len(h0)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    try:
        res = minimize(_obj_fn, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_onto_set(res.x, target)
        return h_new, compute_c5_fast(h_new, dx)
    except Exception:
        return h0.copy(), compute_c5_fast(h0, dx)

def exhaustive_swap_search(h0, dx, target, timeout):

        best_delta = -1e-15
        best_i, best_j = -1, -1

            for j in range(i + 1, n):
                h[i], h[j] = h[j], h[i]
                c5_try = compute_c5_fast(h, dx)
                h[i], h[j] = h[j], h[i]
                delta = c5_try - c5
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        if best_i >= 0:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            c5 = compute_c5_fast(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            improved = True
        if not improved:
            break
    return best_h, best_c5

def sa_loop(h0, dx, target, timeout, rng):
