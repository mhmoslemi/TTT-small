import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        z_test = np.clip(x - mid, 0.0, 1.0)
        if z_test.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def objective_smooth(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx


def constraint_sum(x, n_pts):
    return x.sum() - n_pts / 2.0


def make_bounds(n_pts):
    return [(0.0, 1.0)] * n_pts


def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.arange(n_from) * (2.0 / n_from)
    x_new = np.arange(n_to) * (2.0 / n_to)
    return np.interp(x_new, x_old, h)
