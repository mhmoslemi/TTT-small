import numpy as np
from scipy.optimize import minimize
import time


def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)))


def _cvec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def _sm(c, a):
    m = c.max()
    return m + np.log(np.sum(np.exp(a * (c - m)))) / a


def _fix(h, T):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = T - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h < 1.0 - 1e-13) if d > 0 else (h > 1e-13)
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _exh_swap(h, T):
    n = len(h)
    best_c = _c5(h)
    for _ in range(80):
        improved = False
        best_gain = 1e-14
        best_h2 = None
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in ones:
            for j in zeros:
                h2 = h.copy()
                h2[i], h2[j] = h2[j], h2[i]
                h2 = _fix(h2, T)
                c = _c5(h2)
                gain = best_c - c
                if gain > best_gain:
                    best_gain = gain
                    best_h2 = h2.copy()
        if best_h2 is not None:
            best_c = _c5(best_h2)
            h = best_h2
            improved = True
        if not improved:
            break
    return h, best_c


def _slsqp_refine(h, T, alpha, maxiter=500):
    n = len(h)
    def f(x):
        return _sm(_cvec(x), alpha) + 1e5 * (x.sum() - T) ** 2

        r = minimize(f, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix(r.x, T)
    except:
        return h
