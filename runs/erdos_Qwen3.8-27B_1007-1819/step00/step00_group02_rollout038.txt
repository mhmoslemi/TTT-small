import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def best_pair_swap(h, n):
    """Try all pair swaps, return best one"""
    target = n / 2.0
    current_c5 = compute_c5(h)
    ones = np.where(h > 0.999)[0]
    zeros = np.where(h < 0.001)[0]
    
    best_c5 = current_c5
    best_pair = None
    best_h = h
    
    for i in ones:
        for j in zeros:
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            c5 = compute_c5(h)
            if c5 < best_c5:

            h[j] = 0.0

    t0 = time.time()
    h = h.copy()
    best_h = h.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    T = 0.02
    T_min = 0.001
    
    n_improve = 0
    max_no_improve = 2000
    
    while time.time() - t0 < time_limit:
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        
        if new_c5 < current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                n_improve = 0
            else:
                n_improve += 1
        else:
            h[i], h[j] = 1.0, 0.0
            n_improve += 1
        
        T = max(T * 0.999, T_min)
        
        if n_improve > max_no_improve:

            # Random perturbation
            for _ in range(rng.integers(2, 10)):
                ones = np.where(h > 0.999)[0]

                if len(ones) > 0 and len(zeros) > 0:
                    h[rng.choice(ones)] = 0.0
                    h[rng.choice(zeros)] = 1.0
            current_c5 = compute_c5(h)
            T = 0.02
            n_improve = 0
    
    return best_h, best_c5

            sum(h) = n/2

        z_idx = np.where(h_bin == 0)[0]
        if len(z_idx) > 0:
            h_bin[z_idx[rng.integers(len(z_idx))]] = 0.5
        else:
            o_idx = np.where(h_bin == 1)[0]
            h_bin[o_idx[rng.integers(len(o_idx))]] = 0.5
    h_bin = repair(h_bin, n)
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    h0 = repair(h0, n0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Optimize at n0
    if n0 % 2 == 0:
        h_b = (h0 > 0.5).astype(float)
    else:
        h_b = np.round(h0).clip(0, 1)
        h_b = repair(h_b, n0)
    t_hc = min(30.0, budget_s * 0.06)
    h_hc, c5_hc = hill_climb_binary(h_b, n0, t_hc, rng)
    if c5_hc < best_c5:
        best_h, best_c5 = h_hc, c5_hc
    if elapsed() < budget_s * 0.22:
        t_sa = min(80.0, budget_s * 0.22 - elapsed())
        h_sa, c5_sa = sa_binary(best_h, n0, t_sa, rng)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa, c5_sa
    if elapsed() < budget_s * 0.35:
        h_sl, c5_sl = slsqp_optimize(best_h, n0, budget_s * 0.35 - elapsed())
        if c5_sl < best_c5:
            best_h, best_c5 = h_sl, c5_sl

    # Phase 2: Try multiple n values with restarts
    n_list = []
    for nn in [44, 46, 48, 50, 52, 54, 56, 60, 64, 72, 80, 96, 100, 120]:
        if nn != n0:
            n_list.append(nn)

    for n_try in n_list:
        if elapsed() > budget_s * 0.90:
            break
        remaining = budget_s * 0.90 - elapsed()
        n_restarts = 3 if remaining > 120 else 2

        for r in range(n_restarts):
            if elapsed() > budget_s * 0.90:
                break
            if r == 0:
                h_init = make_binary(n_try, rng, best_h)
            else:
                h_init = make_binary(n_try, rng, None)

            t_hc = min(25.0, remaining * 0.25)
            if t_hc < 1.0:
                break
            h_hc, c5_hc = hill_climb_binary(h_init, n_try, t_hc, rng)

            if elapsed() < budget_s * 0.85:
                t_sa = min(50.0, (budget_s * 0.85 - elapsed()) * 0.4)
                if t_sa > 1.0:
                    h_sa, c5_sa = sa_binary(h_hc, n_try, t_sa, rng)
                    if c5_sa < c5_hc:
                        h_hc, c5_hc = h_sa, c5_sa

            if c5_hc < best_c5:
                best_h, best_c5 = h_hc, c5_hc

            if elapsed() < budget_s * 0.88:
                h_sl, c5_sl = slsqp_optimize(h_hc, n_try, budget_s * 0.88 - elapsed())
                if c5_sl < best_c5:
                    best_h, best_c5 = h_sl, c5_sl

    # Phase 3: Final polish
    n_best = len(best_h)
    if elapsed() < budget_s * 0.96:
        h_sl, c5_sl = slsqp_optimize(best_h, n_best, budget_s * 0.96 - elapsed())
        if c5_sl < best_c5:
            best_h, best_c5 = h_sl, c5_sl

    n_final = len(best_h)
    best_h = repair(best_h, n_final)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
