            lr = min(lr * 1.05, 0.1)  # Slightly increase lr

            lr *= 0.95

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, n_sweeps, grid, t0, deadline):
    h = h.copy()
    val = evaluate_c5(h)
    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            h_save = h.copy()
            old = h[j]
            best_v = val
            best_gj = None
            for gj in grid:
                if abs(gj - old) < 1e-9:
                    continue
                h[j] = gj
                h = project_feasible(h, n)
                v = evaluate_c5(h)
                if v < best_v - 1e-13:
                    best_v = v
                    best_gj = gj
                    improved = True
            if best_gj is not None:
                h[j] = best_gj
                h = project_feasible(h, n)
            else:
                h = h_save
            val = best_v
        if not improved:
            break
    return h, val

def grad_step(h, n, lr, t0, deadline, max_steps=200):
    h = h.copy()
    val = evaluate_c5(h)
    for _ in range(max_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst = int(np.argmax(corr))
        L = worst - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j + L < n:
                grad[j] += (1.0 - h[j + L])
            if 0 <= j - L < n:
                grad[j] -= h[j - L]
        grad -= np.mean(grad)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        v_new = evaluate_c5(h_new)
        if v_new < val - 1e-14:
            h = h_new
            val = v_new
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.92
            if lr < 1e-5:
                break
    return h, val
