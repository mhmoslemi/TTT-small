import numpy as np
from scipy.optimize import minimize
import time


def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if diff != 0:
        h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    # Second pass for exact sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argsort(h)
        if diff > 0:
            for i in reversed(idx):
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            for i in idx:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff >= -1e-12:
                    break
    return h


def _smooth_c5(h, T=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if m < 1e-15:
        return m
    shifted = (corr - m) * T
    return m + np.log(np.sum(np.exp(shifted)) / len(corr)) / T
