from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute adjustment
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def graduated_sum(h, target_sum):
    """Try harder to meet the sum constraint exactly."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += d / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def lsq_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    e = np.exp(alpha * dx * (corr - mc))
    s = float(e.sum())
    if s < 1e-300:
        return np.zeros(n)
    p = e / s
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        m = idx - (n - 1)
        pk = p[idx] * dx
        if pk < 1e-16:
            continue
        lo1, hi1 = max(0, -m), min(n, n - m)
        if hi1 > lo1:
            g[lo1:hi1] += pk * (1.0 - h[lo1 + m:hi1 + m])
        lo2, hi2 = max(0, m), min(n, n + m)
        if hi2 > lo2:
            g[lo2:hi2] -= pk * h[lo2 - m:hi2 - m]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)

    n = len(h0)
    target = n / 2.0
    h0 = repair_sum(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]

    def slsqp_poly(h, maxiter=3000, ftol=1e-14):
        try:
            r = minimize(compute_c5, h, method='SLSQP', bounds=bounds,
                         constraints=cons,
                         options={'maxiter': maxiter, 'ftol': ftol})
            if r.x is not None and np.all(np.isfinite(r.x)):
                return repair_sum(r.x, target)
        except Exception:
            pass
        return None

    # Phase 1: SLSQP from initial
    h1 = slsqp_poly(best_h, 5000)
    if h1 is not None:
        cv = compute_c5(h1)
        if cv < best_c5:
            best_h, best_c5 = h1, cv

    # Phase 2: Log-sum-exp gradient descent (continuation)
    gd_deadline = t0 + budget_s * 0.30
    if time.time() < gd_deadline:
        h_gd = best_h.copy()
        for alpha in [20, 50, 100, 200, 400, 800, 1600]:
            if time.time() > gd_deadline:
                break
            lr = 0.08
            for step in range(400):
                if time.time() > gd_deadline:
                    break
                g = lsq_grad(h_gd, alpha)
                h_new = repair_sum(h_gd - lr * g, target)
                cv_new = compute_c5(h_new)
                cv_cur = compute_c5(h_gd)
                if cv_new <= cv_cur + 1e-12:
                    h_gd = h_new
                    lr = min(lr * 1.02, 0.3)
                else:
                    lr *= 0.6
        cv = compute_c5(h_gd)
        if cv < best_c5:
            best_h, best_c5 = h_gd, cv

    # Quick SLSQP polish after GD
    if time.time() < deadline - 200:
        hp = slsqp_poly(best_h, 5000)
        if hp is not None:
            cv = compute_c5(hp)
            if cv < best_c5:
                best_h, best_c5 = hp, cv

    # Phase 3: Iterated SLSQP from random perturbations
    p3_deadline = t0 + budget_s * 0.75
    n_iter = 0
    while time.time() < p3_deadline:
        n_iter += 1
        # Vary perturbation scale
        base_scale = 0.15 * (1.0 + 0.5 * np.sin(n_iter * 0.7))
        scale = np.random.uniform(0.05, max(0.1, base_scale))
        hp = repair_sum(best_h + np.random.randn(n) * scale, target)
        hr = slsqp_poly(hp, 1500, 1e-12)
        if hr is not None:
            cv = compute_c5(hr)
            if cv < best_c5:
                best_h, best_c5 = hr, cv

    # Phase 4: Binary push + SLSQP
    if time.time() < deadline - 80:
        hb = np.where(best_h > 0.5, 1.0, 0.0)
        hb = repair_sum(hb, target)
        hbr = slsqp_poly(hb, 5000)
        if hbr is not None:
            cv = compute_c5(hbr)
            if cv < best_c5:
                best_h, best_c5 = hbr, cv
        else:
            cv = compute_c5(hb)
            if cv < best_c5:
                best_h, best_c5 = hb, cv

        # Also try thresholding at 0.4
        hb2 = np.where(best_h > 0.4, 1.0, 0.0)
        hb2 = repair_sum(hb2, target)
        hbr2 = slsqp_poly(hb2, 3000)
        if hbr2 is not None:
            cv = compute_c5(hbr2)
            if cv < best_c5:
                best_h, best_c5 = hbr2, cv

    # Phase 5: Short SA to escape local optima, SLSQP after improvement
    sa_end = t0 + budget_s * 0.92
    T = 0.02
    ss = 0.06
    while time.time() < sa_end:
        nc = max(1, int(n * np.random.exponential(0.2)))
        nc = min(nc, n)
        idx = np.random.choice(n, size=nc, replace=False)
        new_h = best_h.copy()
        new_h[idx] += np.random.randn(nc) * ss
        new_h = repair_sum(new_h, target)
        cv_new = compute_c5(new_h)
        d = cv_new - best_c5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-6)):
            best_h = new_h
            best_c5 = cv_new
            if time.time() < sa_end - 30:
                hr = slsqp_poly(best_h, 800)
                if hr is not None:
                    cv_r = compute_c5(hr)
                    if cv_r < best_c5:
                        best_h, best_c5 = hr, cv_r
        T = max(0.003, T * 0.999)
        ss = max(0.02, ss * 0.9997)

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 3:
        hf = slsqp_poly(best_h, 10000, 1e-15)
        if hf is not None:
            cv = compute_c5(hf)
            if cv < best_c5:
                best_h, best_c5 = hf, cv

    return best_h, best_c5, n
