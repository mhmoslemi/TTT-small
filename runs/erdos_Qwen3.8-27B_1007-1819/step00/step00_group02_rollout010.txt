        )

            best_h, best_c5 = h_cand, c5_cand

    # Phase 4: Multiple random restarts (small perturbations)
    if remaining() > 30:
        for trial in range(20):
            if remaining() < 15:
                break
            h_pert = best_h + rng.normal(0, 0.03, n)
            h_pert = project_feasible(h_pert, target_sum)
            h_local, c5_local = gradient_descent(
                h_pert, target_sum, max_iters=200, lr=0.003, beta=20.0, rng=rng, time_check=remaining
            )
            if c5_local < best_c5:
                best_h, best_c5 = h_local, c5_local

    # Phase 5: Try structured initializations
    if remaining() > 40:
        structures = []
        # Sign pattern
        sign_pat = np.array([1 if (i % 4) < 2 else -1 for i in range(n)])
        h_struct = 0.5 + 0.35 * sign_pat
        structures.append(h_struct)
        # Sine
        structures.append(0.5 + 0.35 * np.sin(np.pi * np.arange(n) / (n / 4.0)))
        # Two-block with transition
        h_two = np.ones(n) * 0.5
        h_two[:n // 3] = 0.8
        h_two[n // 3:2 * n // 3] = 0.2
        h_two[2 * n // 3:] = 0.8
        structures.append(h_two)

        for hs in structures:
            if remaining() < 15:
                break
            hs = project_feasible(np.clip(hs, 0, 1), target_sum)
            h_local, c5_local = gradient_descent(
                hs, target_sum, max_iters=300, lr=0.004, beta=15.0, rng=rng, time_check=remaining
            )
            if c5_local < best_c5:
                best_h, best_c5 = h_local, c5_local

    # Phase 6: Final polishing
    if remaining() > 10:
        h_cand, c5_cand = gradient_descent(
            best_h, target_sum, max_iters=1000, lr=0.0005, beta=60.0, rng=rng, time_check=remaining
        )
        if c5_cand < best_c5:
            best_h, best_c5 = h_cand, c5_cand

    if remaining() > 5:
        try:
            h_sl = slsqp_refine(best_h, target_sum, n)
            if evaluate_c5(h_sl) < best_c5:
                best_h = h_sl
                best_c5 = evaluate_c5(best_h)
        except Exception:
            pass

    # Final SLSQP pass
    if remaining() > 5:
        try:
            def obj_final(x):
                return np.max(np.correlate(x, 1.0 - x, mode='full'))
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                     'jac': lambda x: np.ones(n)}]
            bnds = [(0.0, 1.0)] * n
            res = minimize(obj_final, best_h, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
            if res.success:
                h_fin = project_feasible(np.clip(res.x, 0, 1), target_sum)
                if evaluate_c5(h_fin) < best_c5:
                    best_h = h_fin
                    best_c5 = evaluate_c5(best_h)
        except Exception:
            pass

    # Ensure exact feasibility
    best_h = project_feasible(best_h, target_sum)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n
