import time

def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx"""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def get_max_lag(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.argmax(corr)

def project_to_sum(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while keeping 0<=h<=1"""
    h = h.copy()
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase; add to indices with most room
        order = np.argsort(1 - h)
        for idx in order:
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        # Need to decrease
        order = np.argsort(h)
        for idx in order:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if diff >= -1e-12:
                break
    return h

    sv = (float(np.sum(xc)) - target) ** 2

def interp_to_n(h_src, n_dst):

    x_new = np.linspace(0.0, 2.0, n_dst)
    return np.interp(x_new, x_old, h_src)


def directed_refine(h, dx, target, n_iter, rng):
    """Refine by targeting the worst lag"""
    n = len(h)
    ha = h.copy()
    c5a = compute_c5_val(ha, dx)

    for _ in range(n_iter):
        corr = np.correlate(ha, 1.0 - ha, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)

        # Find worst contributing pair
        if lag >= 0:
            # pairs (n, n+lag) for valid n
            n_max = n - lag
            contribs = ha[:n_max] * (1.0 - ha[lag:])
            if n_max > 0 and contribs.max() > 0:
                idx = int(np.argmax(contribs))
                i, j = idx, idx + lag
                # Try to reduce: decrease ha[i] or decrease ha[j]
                if rng.random() < 0.5:
                    i, j = idx, idx
                else:
                    i, j = j, j
                # Compensate
                k = rng.randint(n)
                if k == i:
                    k = (k + 1) % n
                d = 0.03 + 0.07 * rng.random()
                ha[i] -= d
                ha[k] += d
                if ha[i] < 0 or ha[k] > 1:
                    ha[i] += d
                    ha[k] -= d
                    continue
            else:
                continue
        else:
            lag2 = -lag
            n_max = n - lag2
            contribs = ha[lag2:] * (1.0 - ha[:n_max])
            if n_max > 0 and contribs.max() > 0:
                idx = int(np.argmax(contribs))
                i, j = idx + lag2, idx
                i = i if rng.random() < 0.5 else j
                k = rng.randint(n)
                if k == i:

                d = 0.03 + 0.07 * rng.random()
                ha[i] -= d
                ha[k] += d
                if ha[i] < 0 or ha[k] > 1:
                    ha[i] += d
                    ha[k] -= d
                    continue
            else:
                continue

        c5n = compute_c5_val(ha, dx)
        if c5n < c5a:
            c5a = c5n

    return ha, c5a
