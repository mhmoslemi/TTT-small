def compute_c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")

def project(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} via bisection on shift."""
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5

def grad_smooth(h, beta):
