def _proj(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            if diff2 > 0:
                idx = np.argsort(h)[-min(int(np.ceil(diff2)) + 1, n):]
            else:
                idx = np.argsort(h)[:min(int(np.ceil(-diff2)) + 1, n)]
            h[idx] += diff2 / max(len(idx), 1)
            h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp(h0, n, maxiter):
    global _g_n, _g_target
    _g_n = n
    _g_target = n / 2.0
    h0 = _proj(h0.copy(), n)
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_compute_c5, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15,
                                                  'maxls': 200})
        h_out = np.clip(res.x, 0.0, 1.0)
        c5 = _compute_c5(h_out)
    except Exception:
        h_out = h0
        c5 = _compute_c5(h0)
    return h_out, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    def elapsed():
        return time.time() - t0

    def update(h, c5):
        nonlocal best_c5, best_h, best_n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = len(h)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        iv = np.array(initial_h_values, dtype=float)
    else:
        iv = np.zeros(45)
        iv[11:34] = 1.0
    init_n = len(iv)
    init_n = init_n  # keep as int

    # Phase 1: Polish the initial solution
    h1, c5_1 = _slsqp(iv, init_n, 5000)
    update(h1, c5_1)
    if elapsed() > budget_s - 5:
        return _proj(best_h, best_n), _compute_c5(_proj(best_h, best_n)), best_n

    # Phase 2: Multi-start with geometrically shrinking perturbations
    attempt = 0
    while elapsed() < budget_s - 120:
        attempt += 1
        n_cur = best_n if best_h is not None else init_n
        scale = 0.4 * (0.92 ** (attempt // 2))
        if scale < 0.02:
            scale = 0.02
        if best_h is not None:
            h_pert = best_h + rng.randn(n_cur) * scale
        else:
            h_pert = iv + rng.randn(init_n) * 0.2
        h_pert = _proj(h_pert, n_cur)
        h_res, c5_res = _slsqp(h_pert, n_cur, 800)
        update(h_res, c5_res)

    # Phase 3: Try other resolutions by interpolation
    for n_try in [30, 36, 50, 54, 60, 72, 80, 96, 120]:
        if elapsed() > budget_s - 15:
            break
        if best_h is not None and best_n != n_try:
            x_old = (np.arange(best_n) + 0.5) * 2.0 / best_n
            x_new = (np.arange(n_try) + 0.5) * 2.0 / n_try
            h_try = np.interp(x_new, x_old, np.concatenate([best_h, best_h[:1]]))
        elif best_h is None:
            x_old = (np.arange(init_n) + 0.5) * 2.0 / init_n
            x_new = (np.arange(n_try) + 0.5) * 2.0 / n_try
            h_try = np.interp(x_new, x_old, np.concatenate([iv, iv[:1]]))
        else:
            continue
        h_try = _proj(h_try, n_try)
        h_res, c5_res = _slsqp(h_try, n_try, 2000)
        update(h_res, c5_res)

    # Phase 4: Final fine-tuning if time permits
    if best_h is not None and elapsed() < budget_s - 5:
        h_ref, c5_ref = _slsqp(best_h, best_n, 3000)
        update(h_ref, c5_ref)

    # Fallback
    if best_h is None:
        best_h = iv
        best_n = init_n
        best_c5 = _compute_c5(best_h)

    # Final projection to ensure exact feasibility
    best_h = _proj(best_h, best_n)
    best_c5 = _compute_c5(best_h)
    return (best_h, best_c5, best_n)
