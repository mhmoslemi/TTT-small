            if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-15)):
                h_cur = h_t
                c5_cur = c5_t
                if c5_cur < c5_best:
                    c5_best = c5_cur
                    h_saved = h_cur.copy()
        T = max(T_min, T * 0.97)
        if T <= T_min * 1.1:
            reheat_T = max(T_min, reheat_T * 0.6)
            T = reheat_T
            velocity = max(0.01, velocity * 0.7)
            h_cur = project_feasible(h_saved + rng.normal(0, 0.04, n), n)
            c5_cur = compute_c5(h_cur)
            if c5_best < compute_c5(h_best):
                h_best = h_saved.copy()
    return h_saved, c5_best


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12.0

    # Initialize from parent
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    if h0 is None or len(h0) == 0:
        h0 = np.full(45, 0.5)

    n = len(h0)
    h_best = project_feasible(h0.copy(), n)
    c5_best = compute_c5(h_best)

    # Phase 1: Smooth-max gradient optimization with increasing alpha
    alphas = [20, 50, 100, 250, 600, 1500, 4000]
    for ai, alpha in enumerate(alphas):
        if time.time() > deadline:
            break
        max_it = 400 if alpha < 100 else 250
        n_restarts = 2 if alpha < 200 else 1
        for r in range(n_restarts):
            if time.time() > deadline:
                break
            if r == 0:
                h_init = h_best.copy()
            else:
                h_init = np.clip(h_best + np.random.normal(0, 0.06, n), 0, 1)
                h_init = project_feasible(h_init, n)
            h_t = gradient_step(h_init, n, alpha, max_it, seed + ai * 10 + r)
            c5_t = compute_c5(h_t)
            if c5_t < c5_best:
                c5_best = c5_t
                h_best = h_t

    # Phase 2: Simulated Annealing
    if time.time() < deadline - 30:
        sa_deadline = min(deadline - 20, t0 + budget_s * 0.75)
        h_best, c5_best = sa_search(h_best, n, sa_deadline, seed)

    # Phase 3: Final gradient polish
    if time.time() < deadline - 5:
        for alpha in [5000, 10000]:
            if time.time() > deadline:
                break
            h_t = gradient_step(h_best, n, alpha, 150, seed + 100 + alpha)
            c5_t = compute_c5(h_t)
            if c5_t < c5_best:
                c5_best = c5_t
                h_best = h_t

    # Phase 4: One more SA burst if time remains
    if time.time() < deadline - 5:
        h_best, c5_best = sa_search(h_best, n, deadline, seed + 999)

    # Final projection and evaluation
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    return h_best, c5_best, n
