def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_overlaps(h)) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        err = h.sum() - target
        if abs(err) < 1e-14:
            break
        if err > 0:
            m = h > 1e-14
            c = int(m.sum())
            if c == 0:
                break
            h[m] -= min(err / c, h[m].max())
        else:
            m = h < 1.0 - 1e-14
            c = int(m.sum())
            if c == 0:
                break
            h[m] += min(-err / c, 1.0 - h[m].min())
    return np.clip(h, 0.0, 1.0)

def grad_overlap(h, s, n):
    g = np.zeros(n)
    if s >= 0 and s < n:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    elif s < 0 and -s < n:
        ss = -s
        g[ss:] += (1.0 - h[:n - ss])
        g[:n - ss] -= h[ss:]
    return g
