        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)

def coord_descent(h, n, rounds=10):

    _gn = n
    for _ in range(rounds):

        for i in range(n):
            old = h[i]
            lo = max(0.0, n/2.0 - (h.sum() - h[i]) )
            hi = 1.0
            if lo > hi:
                continue
            # Ternary search
            a, b = lo, hi
            for _ in range(30):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h[i] = m1
                f1 = e_c5(p_sum(h, n), n)
                h[i] = m2
                f2 = e_c5(p_sum(h, n), n)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            h[i] = (a + b) / 2.0
            h = p_sum(h, n)
            if h[i] != old:
                improved = True
        if not improved:
            break
    return h, e_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

_gn = 45

def e_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def p_sum(h, n):
    t = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        nm = m.sum()
        if nm == 0:
            break
        h[m] += d / nm
        h = np.clip(h, 0.0, 1.0)
    return h

def f(x):
    return e_c5(p_sum(x, _gn), _gn)

def jf(x):
    n = _gn
    h = p_sum(x, n)
    eps = 1e-6
    grad = np.empty(n)
    for i in range(n):
        hp = h.copy()
        hp[i] = min(1.0, h[i] + eps)
        hm = h.copy()
        hm[i] = max(0.0, h[i] - eps)
        grad[i] = (e_c5(p_sum(hp, n), n) - e_c5(p_sum(hm, n), n)) / (2.0 * eps)
    return grad

def slsqp_opt(h0, n, iters=200):
    global _gn
    _gn = n
    h0 = p_sum(h0, n)
    try:
        r = minimize(f, h0, method='SLSQP', jac=jf,
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-12})
        return p_sum(r.x, n), e_c5(p_sum(r.x, n), n)
    except Exception:

    _gn = n

                     bounds=[(0.0, 1.0)] * n,

        return p_sum(r.x, n), e_c5(p_sum(r.x, n), n)

    _gn = n

            lo = max(0.0, n / 2.0 - (h.sum() - h[i]))

                    best_v = h[i]
