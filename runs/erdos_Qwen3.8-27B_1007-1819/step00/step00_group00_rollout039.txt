def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_binary(h, n):
    """Round h to binary with exactly n/2 ones."""
    hb = (h >= 0.5).astype(float)
    s = int(np.sum(hb))
    target = n // 2
    if s > target:
        ones = np.where(hb == 1)[0]
        # Remove (s - target) ones, prefer those with smallest h value
        order = np.argsort(h[ones])
        for idx in order[:s-target]:
            hb[ones[idx]] = 0
    elif s < target:
        zeros = np.where(hb == 0)[0]
        order = np.argsort(-h[zeros])  # prefer those with largest h value
        for idx in order[:target-s]:
            hb[zeros[idx]] = 1
    return hb

import numpy as np
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_binary(h, n, rng):
    hb = (h >= 0.5).astype(float)
    target = n // 2
    s = int(hb.sum())
    if s > target:
        idx = np.where(hb == 1)[0]
        perm = rng.permutation(idx)
        for i in range(s - target):
            hb[perm[i]] = 0.0
    elif s < target:
        idx = np.where(hb == 0)[0]
        perm = rng.permutation(idx)
        for i in range(target - s):
            hb[perm[i]] = 1.0
    return hb

def init_random(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def init_thue_morse(n, rng):
    h = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)
    return fix_binary(h, n, rng)

def init_spaced(n, rng):
    h = np.zeros(n)
    positions = np.setdiff1d(np.arange(n), np.arange(0, n, 2))
    positions = positions[:n // 2]
    h[positions] = 1.0
    for _ in range(200):
        i = rng.randint(n // 2)
        j = rng.randint(n // 2)
        if positions[i] != positions[j]:
            p, q = positions[i], positions[j]
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                new_q = zeros[rng.randint(len(zeros))]
                if new_q != p:
                    h[p] = 0.0
                    h[new_q] = 1.0
                    positions[i] = new_q
                    np.sort(positions)
    return fix_binary(h, n, rng)

def sa_optimize(h, n, rng, time_limit, T_init=0.03, T_min=0.0005):
    deadline = time.time() + time_limit
    cur = h.copy()
    cur_c5 = compute_c5(cur, n)
    best = cur.copy()
    best_c5 = cur_c5
    T = T_init
    stuck_count = 0
    last_improve_time = time.time()

    while time.time() < deadline:
        p = int(rng.random() * n)
        q = int(rng.random() * n)
        if p == q or cur[p] == cur[q]:
            continue
        cur[p], cur[q] = cur[q], cur[p]
        new_c5 = compute_c5(cur, n)
        delta = new_c5 - cur_c5
        if delta <= 0 or (T > T_min * 0.5 and rng.random() < math.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best[:] = cur
                last_improve_time = time.time()
                stuck_count = 0
            else:
                stuck_count += 1
        else:
            cur[p], cur[q] = cur[q], cur[p]
            stuck_count += 1

        T = max(T * 0.9995, T_min)
        if stuck_count > 20000:
            T = max(T, min(0.04, T_init * 0.5))
            stuck_count = 0
            cur[:] = best

    return best, best_c5

def hill_climb(h, n, rng, time_limit):
    deadline = time.time() + time_limit
    cur = h.copy()
    cur_c5 = compute_c5(cur, n)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        rng.shuffle(ones)
        zeros_list = list(zeros)
        for p in ones:
            if time.time() > deadline:
                break
            best_q = -1
            best_val = cur_c5
            for q in zeros_list:
                cur[p], cur[q] = cur[q], cur[p]
                v = compute_c5(cur, n)
                cur[p], cur[q] = cur[q], cur[p]
                if v < best_val - 1e-14:
                    best_val = v
                    best_q = q
            if best_q >= 0:
                cur[p], cur[best_q] = cur[best_q], cur[p]
                cur_c5 = best_val
                improved = True
                ones = np.where(cur == 1)[0]
                zeros_list = list(np.where(cur == 0)[0])
    return cur, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_c5 = float('inf')
    best_h = None
    best_n = 0

            c5_b = compute_c5(h_b, n0)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    n_list = [44, 46, 48, 50, 52, 54, 56, 60, 64, 72, 80, 96, 128, 160]
    total_n = len(n_list)

    for ni, n in enumerate(n_list):
        elapsed = time.time() - t0
        remaining = budget_s * 0.82 - elapsed
        if remaining < 20:
            break
        time_per_n = max(15.0, remaining / max(1, total_n - ni))

        inits = [init_random(n, rng), init_thue_morse(n, rng)]
        if gh is not None and n == len(gh) and n % 2 == 0:
            inits.insert(0, fix_binary(gh, n, rng))
        inits.append(init_spaced(n, rng))

        for h_init in inits:
            elapsed = time.time() - t0
            if budget_s * 0.85 - elapsed < 5:
                break
            c5 = compute_c5(h_init, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_init.copy()
                best_n = n

            t_sa = max(5.0, min(40.0, (budget_s * 0.6 - elapsed) / 5.0))
            h_sa, c5_sa = sa_optimize(h_init, n, rng, t_sa)

            t_hc = max(2.0, min(8.0, (budget_s * 0.6 - (time.time() - t0)) / 5.0))
            h_hc, c5_hc = hill_climb(h_sa, n, rng, t_hc)

            if c5_hc < best_c5:
                best_c5 = c5_hc
                best_h = h_hc.copy()
                best_n = n

    # Final refinement on best solution
    elapsed = time.time() - t0
    if best_h is not None and budget_s - elapsed > 30:
        t_fin = budget_s * 0.7 - elapsed
        h_f, c5_f = sa_optimize(best_h, best_n, rng, max(5.0, t_fin * 0.5))
        h_f, c5_f = hill_climb(h_f, best_n, rng, max(2.0, t_fin * 0.3))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    elapsed = time.time() - t0
    if best_h is not None and budget_s - elapsed > 15:
        h_f2, c5_f2 = hill_climb(best_h, best_n, rng, budget_s - elapsed - 3)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    if best_h is None:
        n = 44
        best_h = init_random(n, rng)
        best_n = n
        best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, best_n
