def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
    return h

def smooth_c5_and_grad(h, m=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = np.max(corr)
    shifted = m * (corr - mx)
    shifted = np.clip(shifted, -60, 0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
        wsum = 1.0
    weights /= wsum
    val = mx + np.log(wsum) / m
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-12:
            continue
        offset = k - (n - 1)
        for p in range(n):
            q = p - offset
            if 0 <= q < n:
                grad[p] += weights[k] * (1.0 - h[q])
            r = p + offset
            if 0 <= r < n:
                grad[p] -= weights[k] * h[r]
    grad *= dx
    return val, grad

def slsqp_polish(h, n):
    x0 = np.append(h, evaluate_c5(h) + 0.01)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    def obj(x):
        return x[-1]

    def grad_obj(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g

    def cons_func(x):
        h = x[:-1]
        t = x[-1]
        out = []
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        for k in range(ncorr):
            out.append(t - corr[k])
        out.append(np.sum(h) - n / 2.0)
        return out

    def cons_jac(x):
        h = x[:-1]
        nvar = len(x)
        J = np.zeros((ncorr + 1, nvar))
        corr = np.correlate(h, 1.0 - h, mode="full")

            J[k, -1] = 1.0

                r = p + offset
                if 0 <= r < n:

        J[-1, :n] = 1.0
