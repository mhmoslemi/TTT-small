import numpy as np
from scipy.optimize import minimize
import time


def repair(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] -= diff / cnt
            h[mask] = np.maximum(h[mask], 0.0)
        else:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += (-diff) / cnt
            h[mask] = np.minimum(h[mask], 1.0)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    rem = target - s
    if abs(rem) > 1e-15:
        idx = np.argsort(np.abs(h - 0.5))
        for i in idx:
            if rem > 0:
                add = min(rem, 1.0 - h[i])
                h[i] += add
                rem -= add
            elif rem < 0:
                sub = min(-rem, h[i])
                h[i] -= sub
                rem += sub
            if abs(rem) < 1e-15:
                break
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def nm_optimize(h_init, n, maxiter, simplex_size):
    def obj(h_flat):
        return compute_c5(repair(h_flat, n))
    res = minimize(obj, h_init, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-14,
                            '_initial_simplex_size': simplex_size})
    return repair(res.x, n)


def sa_phase(h_init, n, t_end, rng, T_start=0.025, T_min=1e-9):
    cur = h_init.copy()
    cur_c5 = compute_c5(cur)
    best_h = cur.copy()
    best_c5 = cur_c5
    T = T_start
    alpha = 0.99998

    while time.time() < t_end:
        mt = rng.integers(0, 6)
        h_new = cur.copy()

        if mt == 0:
            i = rng.integers(n)
            h_new[i] += rng.choice([-1, 1]) * rng.uniform(0.005, 0.06)
        elif mt == 1:
            i, j = rng.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        elif mt == 2:
            i = rng.integers(0, max(1, n - 2))
            L = min(rng.integers(2, 6), n - i)
            h_new[i:i + L] += rng.normal(0, 0.03, L)
        elif mt == 3:
            i = rng.integers(n)
            h_new[i] = 1.0 if h_new[i] > 0.45 else 0.0
        elif mt == 4:
            L = min(rng.integers(2, 8), n // 2)
            i = rng.integers(0, n - L)
            j = rng.integers(0, n - L)
            if abs(j - i) >= L:
                a = i
                b = j
                if a > b:
                    a, b = b, a
                tmp = h_new[a:a + L].copy()
                h_new[a:a + L] = h_new[b:b + L]
                h_new[b:b + L] = tmp
        else:
            L = min(rng.integers(1, 4), n)
            idx = rng.choice(n, L, replace=False)
            h_new[idx] = 1.0 - h_new[idx]

        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new)
        delta = c5_new - cur_c5

        if delta < 0 or rng.random() < np.exp(-min(delta / max(T, 1e-12), 60.0)):
            cur = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

        T = max(T * alpha, T_min)

    return best_h, best_c5

            best_c5 = c5_try
            best_h = h_try

    # Verify and return
    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h.tolist(), best_c5, n)
