def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    
    start_time = time.time()
    
    n = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float).copy()
    dx = 2.0 / n
    
    def make_feasible(h):
        h = np.clip(h, 0, 1)
        target = n / 2.0
        diff = target - np.sum(h)
        h = h + diff / n
        h = np.clip(h, 0, 1)
        # Second pass to fix sum
        diff = target - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                order = np.argsort(-h)
                for i in order:
                    if h[i] < 1:
                        add = min(diff, 1 - h[i])
                        h[i] += add
                        diff -= add
                        if abs(diff) < 1e-10:
                            break
            else:
                order = np.argsort(h)
                for i in order:
                    if h[i] > 0:
                        rem = min(-diff, h[i])
                        h[i] -= rem
                        diff += rem
                        if abs(diff) < 1e-10:
                            break
        return h
    
    def evaluate(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    h = make_feasible(h0)
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    
    rng = np.random.default_rng(seed)
    
    # Subgradient method with projection
    lr = 0.01
    for iteration in range(500000):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute objective and gradient
        corr = np.correlate(h, 1-h, mode="full")
        obj_vals = corr * dx
        max_val = np.max(obj_vals)
        
        # Active set: lags where objective is close to max
        active_mask = obj_vals > max_val - 0.005
        active_lags = np.where(active_mask)[0]
        
        # Subgradient (average over active set)
        grad = np.zeros(n)
        for aidx in active_lags:
            lag = aidx - (n - 1)
            # For this lag, contribution to gradient at j:
            # (1-h[j-lag]) if 0 <= j-lag < n-lag_abs AND valid pair
            # -h[j+lag] if 0 <= j+lag < n AND valid pair
            # More precisely:
            # c_lag = sum_{n=max(0,-lag)}^{n-1-max(0,lag)} h[n+lag]*(1-h[n])
            # where n+lag must be in [0,n-1] and n must be in [0,n-1]
            
            n_start = max(0, -lag)
            n_end = min(n, n - lag)  # exclusive
            
            for nn in range(n_start, n_end):
                j1 = nn + lag  # index of h in first factor
                j2 = nn        # index of h in second factor (1-h)
                if 0 <= j1 < n:
                    grad[j1] += (1 - h[j2])
                if 0 <= j2 < n:
                    grad[j2] += (-h[j1])
        
        grad /= len(active_lags)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project onto feasible set
        h_new = make_feasible(h_new)
        
        # Check improvement
        c5_new = evaluate(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            lr = min(lr * 1.05, 0.5)
        else:

            lr *= 0.95
            h = best_h.copy()  # revert
        
        # Occasionally add noise to escape local optima
        if iteration % 5000 == 1:

            h = make_feasible(h + noise)
            c5 = evaluate(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    return best_h, best_c5, n

def _grad_smooth(h, n, obj, tau):
    mx = obj.max()
    w = np.exp((obj - mx) / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(obj))
        w[np.argmax(obj)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    n_lags = len(obj)
    for k in range(n_lags):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-8:
            continue
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] += wk * (-h[lag:])
        else:
            lp = -lag
            grad[:n - lp] += wk * (1.0 - h[lp:])
            grad[lp:] += wk * (-h[:n - lp])
    return grad

def _grad_topk(h, n, obj, topk=3):
    idx = np.argsort(obj)[::-1][:topk]
    grad = np.zeros(n)
    for k in idx:
        lag = k - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] += (-h[lag:])
        else:
            lp = -lag
            grad[:n - lp] += (1.0 - h[lp:])
            grad[lp:] += (-h[:n - lp])
    return grad / float(len(idx))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    n = len(initial_h_values)
    dx = 2.0 / n
    h = np.array(initial_h_values, dtype=float)
    h = _make_feasible(h, n)

    best_h = h.copy()
    best_obj = _compute_obj(best_h, dx).max()

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient optimization with soft-max (55% budget)
    tau = 0.04
    lr = 0.04
    phase1_end = t0 + 0.55 * budget_s
    stall = 0

    while elapsed() < phase1_end:
        obj = _compute_obj(h, dx)
        cur = obj.max()
        if cur < best_obj - 1e-10:
            best_h, best_obj = h.copy(), cur
            stall = 0
        else:
            stall += 1

        if stall % 2000 == 1999:
            h = _make_feasible(h + rng.normal(0, 0.03, n), n)
            lr = 0.05
            tau = max(tau * 1.5, 0.04)

        grad = _grad_smooth(h, n, obj, tau)
        h_new = _make_feasible(h - lr * grad, n)
        obj_new = _compute_obj(h_new, dx)

        if obj_new.max() < cur - 1e-12:
            h = h_new
            lr = min(lr * 1.015, 0.2)
        else:
            lr *= 0.94
            if lr < 0.002:
                h = _make_feasible(h + rng.normal(0, 0.03, n), n)
                lr, tau = 0.05, 0.04

        # Slowly decrease tau
        if int(elapsed()) > int(phase1_end * 0.3):
            tau = max(tau * 0.9995, 0.005)

        # Occasional perturbation
        if rng.random() < 0.003:
            h = _make_feasible(h + rng.normal(0, 0.015, n), n)

    # Phase 2: Local random search (25% budget)
    phase2_end = t0 + 0.80 * budget_s
    step = 0.008
    while elapsed() < phase2_end:
        h_try = best_h + rng.normal(0, step, n)
        h_try = _make_feasible(h_try, n)
        c5 = _compute_obj(h_try, dx).max()
        if c5 < best_obj - 1e-12:
            best_h, best_obj = h_try, c5
            step = min(step * 1.005, 0.06)
        else:
            step *= 0.999
            if step < 0.002:
                step = 0.015

    # Phase 3: Coordinate descent polishing (20% budget)
    order = np.arange(n)
    phase3_end = t0 + budget_s - 5
    while elapsed() < phase3_end:
        any_improve = False
        rng.shuffle(order)
        for i in order:
            if elapsed() >= phase3_end:
                break
            old_val = best_h[i]
            for delta in (0.01, -0.01, 0.005, -0.005, 0.002, -0.002):
                best_h[i] = np.clip(old_val + delta, 0.0, 1.0)
                best_h = _make_feasible(best_h, n)
                c5 = _compute_obj(best_h, dx).max()
                if c5 < best_obj - 1e-13:
                    best_obj = c5
                    old_val = best_h[i]
                    any_improve = True
                else:
                    best_h[i] = old_val
                    best_h = _make_feasible(best_h, n)
                    break
        if not any_improve:
            break

    best_h = _make_feasible(best_h, n)
    best_obj = _compute_obj(best_h, dx).max()
    return best_h, best_obj, n
