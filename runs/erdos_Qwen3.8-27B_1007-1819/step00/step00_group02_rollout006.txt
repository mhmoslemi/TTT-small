    def elapsed():
        return time.time() - t0

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def try_optimize(h_init, maxiter=600):
        nonlocal best_h, best_c5
        if elapsed() > budget_s - 90:
            return
        n = len(h_init)
        target = n / 2.0
        h_init = fix_sum(h_init, target)

        def obj(x):
            return compute_c5(x)

        def obj_grad(x):
            v, g = c5_and_grad(x)
            return v, g

        cons = [{'type': 'eq',
                 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-12,
                                    'disp': False})
            h_opt = fix_sum(res.x, target)
            c5 = compute_c5(h_opt)
            # Polish
            h_opt = polish(h_opt, n)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 1: Optimize from provided initial
    try_optimize(h0, maxiter=800)
    best_h = polish(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    # Phase 2: Try different resolutions
    n_list = [45, 50, 60, 70, 80, 90, 100, 120, 150, 180]
    for n_pts in n_list:
        if elapsed() > budget_s - 150:
            break
        if n_pts == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_pts, endpoint=False)
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = fix_sum(h_interp, n_pts / 2.0)
        try_optimize(h_interp, maxiter=600)
        if c5 < best_c5:
            best_h = polish(best_h, len(best_h))
            best_c5 = compute_c5(best_h)

    # Phase 3: Multi-start random perturbations
    n_best = len(best_h)
    attempts = 0
    while elapsed() < budget_s - 60:
        attempts += 1
        strategy = attempts % 4
        if strategy == 0:
            h_pert = best_h + np.random.randn(n_best) * 0.04
        elif strategy == 1:
            h_pert = best_h * np.exp(np.random.randn(n_best) * 0.08)
        elif strategy == 2:
            mask = np.random.rand(n_best) < 0.3
            h_pert = best_h.copy()
            h_pert[mask] = np.random.rand(np.sum(mask))
        else:
            # Try a different n
            n_alt = [40, 50, 60, 75, 90, 110][attempts % 6]
            x_new = np.linspace(0, 2, n_alt, endpoint=False)
            x_old = np.linspace(0, 2, n_best, endpoint=False)
            h_pert = np.interp(x_new, x_old, best_h)
            h_pert = fix_sum(h_pert, n_alt / 2.0)
            try_optimize(h_pert, maxiter=400)
            continue

        h_pert = fix_sum(h_pert, n_best / 2.0)
        try_optimize(h_pert, maxiter=300)
        # Repolish best
        if elapsed() < budget_s - 30:
            best_h = polish(best_h, len(best_h))
            best_c5 = compute_c5(best_h)

    # Final ensure feasibility
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
