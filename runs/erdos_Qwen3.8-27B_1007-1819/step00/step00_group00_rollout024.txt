import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_objective(x):
    x = np.clip(x, 0.0, 1.0)
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * dx)

def repair_sum(h):
    n = len(h)
    target_sum = n / 2.0
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(20):
        h += (target_sum - np.sum(h)) / n
        np.clip(h, 0.0, 1.0, out=h)
        if abs(np.sum(h) - target_sum) < 1e-12:
            break
    return h

def make_random_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    h += (rng.random(n) - 0.5) * 0.3
    return repair_sum(h)

def make_sinusoidal_start(n, rng):
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    freq = rng.randint(2, max(3, n // 3))
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.sin(freq * x + phase)
    return repair_sum(h)

def project_gradient_step(h, grad, lr, n):
    target_sum = n / 2.0
    h = h - lr * grad
    for _ in range(30):
        h += (target_sum - np.sum(h)) / n
        np.clip(h, 0.0, 1.0, out=h)
        if abs(np.sum(h) - target_sum) < 1e-14:
            break
    return h

def compute_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    shift = k_max - (n - 1)
    s = shift
    grad = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
    else:
        s2 = -s
        grad[s2:] += (1.0 - h[:n - s2])
        grad[:n - s2] -= h[s2:]
    return grad * dx

                           options={'maxiter': 2000, 'ftol': 1e-12})
            h_res = repair_sum(res.x)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
        except Exception:
            pass

    # Final multi-start SLSQP on best with small perturbation
    if time.time() - t0 < budget_s * 0.95:
        n = len(best_h)
        for _ in range(5):
            if time.time() - t0 > budget_s * 0.95:
                break
            h_pert = best_h + (rng.random(n) - 0.5) * 0.02
            h_pert = repair_sum(h_pert)
            bounds = [(0.0, 1.0)] * n
            target_sum = n / 2.0
            A_eq = np.ones((1, n))
            con_lin = LinearConstraint(A_eq, target_sum, target_sum)
            try:
                res = minimize(c5_objective, h_pert, method='SLSQP',
                               bounds=bounds, constraints=[con_lin],
                               options={'maxiter': 1000, 'ftol': 1e-12})
                h_res = repair_sum(res.x)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res.copy()
            except Exception:
                pass

    best_h = repair_sum(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
