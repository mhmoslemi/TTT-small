import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, 1.0 - h[mask].min())
            h[mask] += step
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, h[mask].min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def _soft_max_obj(h, n, t):
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(vals)
    shifted = t * (vals - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / t

def _soft_max_grad(h, n, t):
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(vals)
    shifted = t * (vals - m)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # grad of vals[k] wrt h[i]... this is complex
    # Use finite difference instead

        hp[i] += eps
        grad[i] = (_soft_max_obj(hp, n, t) - f0) / eps
    return grad

                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-9})
    
    if res.x is not None:
        h_out = _repair_sum(np.clip(res.x, 0.0, 1.0), target)
        c5_out = _compute_c5(h_out)
        c5_in = _compute_c5(h)

def _binary_snap(h, n):
    target = n / 2.0
    b = (h > 0.5).astype(float)
    excess = np.sum(b) - target
    if excess > 0:
        order = np.argsort(h)
        for i in order:
            if excess <= 0:
                break
            if b[i] == 1.0:
                b[i] = 0.0
                excess -= 1
    elif excess < 0:
        order = np.argsort(-h)
        for i in order:
            if excess >= 0:
                break
            if b[i] == 0.0:
                b[i] = 1.0
                excess += 1
    return b
