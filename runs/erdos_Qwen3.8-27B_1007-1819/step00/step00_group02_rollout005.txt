import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

_g_n = 45
_g_T = 50.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _soft_obj(x):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return float(dx * logsumexp(_g_T * corr) / _g_T)

def _con_fun(x):
    return float(np.sum(x) - _g_n / 2.0)

def _con_jac(x):
    return np.ones(_g_n)

def reproject(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, T, maxiter, ftol):
    global _g_n, _g_T
    _g_n = n
    _g_T = T
    h0 = reproject(h0, n)
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_soft_obj, h0, method='SLSQP', bounds=bnds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return reproject(res.x, n)
    except Exception:
        return h0

            h_new[j] += a
        else:

                return None
            j = int(rng.choice(cnd))

            if not (0.0 <= h_new[j] <= 1.0):
