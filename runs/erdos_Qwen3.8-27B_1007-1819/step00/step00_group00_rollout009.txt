    def feasible(V):

        t_lo = lo
        t_hi = hi
        for k in range(K):
            if b[k] > 1e-15:
                # a[k] + b[k]*t <= V => t <= (V - a[k])/b[k]
                t_hi = min(t_hi, (V - a[k]) / b[k])
            elif b[k] < -1e-15:
                # a[k] + b[k]*t <= V => t >= (V - a[k])/b[k]
                t_lo = max(t_lo, (V - a[k]) / b[k])
            else:
                # b[k] ≈ 0: need a[k] <= V
                if a[k] > V + 1e-12:
                    return None
        if t_lo > t_hi + 1e-12:
            return None
        # Return a feasible t
        t = (t_lo + t_hi) / 2
        if t < lo: t = lo
        if t > hi: t = hi
        return t
    
    # Binary search on V
    # Lower bound: min of a[k] (when t=0) or endpoint values
    f_lo = np.max(a + b * lo)
    f_hi = np.max(a + b * hi)
    # The minimum V is at most min(f_lo, f_hi)
    V_hi = min(f_lo, f_hi)
    V_lo = np.min(a)  # very rough lower bound
    
    # Actually, better lower bound:
    # V must be >= max_k a[k] - |b[k]| * max(|lo|,|hi|)
    V_lo = np.max(a - np.abs(b) * max(abs(lo), abs(hi)))
    
    for _ in range(60):
        V_mid = (V_lo + V_hi) / 2
        t = feasible(V_mid)
        if t is not None:
            V_hi = V_mid
        else:
            V_lo = V_mid
    
    # Find optimal t at V_hi
    t = feasible(V_hi)
    if t is None:
        t = (lo + hi) / 2
    return t, V_hi
