def local_search_swaps(h, n, dx, max_time):

    start = time.time()
    best_c5 = compute_c5_array(h, dx)
    improved = True
    while improved and (time.time() - start) < max_time:

        best_delta = 0
        best_pair = None
        for i in range(n):

            h[i], h[j] = h[j], h[i]

            improved = True
