from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def swap_improve(h, max_rounds=1000):
    """Greedily improve by trying all single swaps."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h > 0.4)[0]
        zeros = np.where(h < 0.6)[0]
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                new_c5 = compute_c5(h)
                if new_c5 < c5 - 1e-12:
                    c5 = new_c5
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
            if improved:
                break
        if not improved:
            break
    return h, c5

def simulated_annealing(n, n_iters, seed, t_start=0.5, t_end=0.0001):
    np.random.seed(seed)
    h = np.zeros(n)
    n_ones = n // 2
    h[:n_ones] = 1.0
    np.random.shuffle(h)
    if n % 2 == 1:
        h[n_ones] = 0.5
    
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    
    for it in range(n_iters):
        t = t_start * (t_end / t_start) ** (it / max(n_iters, 1))
        ones = np.where(h > 0.4)[0]
        zeros = np.where(h < 0.6)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or np.random.random() < np.exp(-max(delta, -10) / max(t, 1e-10)):
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

    """Continuous refinement using Nelder-Mead with penalty for sum constraint."""
    total_target = n / 2.0
    
    def objective(x):

        pen = abs(np.sum(x) - total_target) * 100
