import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(corr)
    sh = (corr - cmax) / temp
    sh -= sh.max()
    ew = np.exp(sh)
    sw = ew.sum()
    w = ew / sw
    obj = cmax + temp * np.log(sw)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - n + 1
        wk = w[k] * dx
        js_p = np.arange(n) + s
        m_p = (js_p >= 0) & (js_p < n)
        js_m = np.arange(n) - s
        m_m = (js_m >= 0) & (js_m < n)
        g = np.zeros(n)
        if np.any(m_p):
            g[m_p] += (1.0 - h[js_p[m_p]])
        if np.any(m_m):
            g[m_m] -= h[js_m[m_m]]
        grad += wk * g
    return obj, grad

    n = len(h_init)

        return smooth_obj_grad(x, temp)[0]

        return smooth_obj_grad(x, temp)[1]
