from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
            h = np.clip(h, 0.0, 1.0)
            # Second pass if needed
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-12:
                if diff2 > 0:
                    mask = h < 1.0
                else:
                    mask = h > 0.0
                count = np.sum(mask)
                if count > 0:
                    h[mask] += diff2 / count
    return h

def smooth_objective_and_grad(h, t=50.0):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    max_c = np.max(corr)
    s = (corr - max_c) * t
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    obj = dx * max_c + (dx / t) * np.log(np.sum(exp_s))

    # corr[k] = sum_{i: 0<=i<n, 0<=i+k_{idx}<n} h[i+k_{idx}] * (1-h[i])
    # where k_idx = k - (n-1) maps the full correlation index to actual lag

    # Let lag = m - (n-1), then result[m] = sum_i h[i+lag] * omh[i]
    # dc/dh[j] = sum_m w_m * d(result[m])/dh[j]
    # d(result[m])/dh[j] = omh[j-lag] * [valid] + h[j+lag] * (-1) * [valid]
    #                    = omh[j-lag] * [valid] - h[j+lag] * [valid]

    # Actually, let's just use the weighted sum approach
    grad = np.zeros(n)
    # for each lag m (0 to 2n-2), lag_k = m - (n-1)
    # result[m] = sum_{i=max(0,-lag_k)}^{min(n-1, n-1-lag_k)} h[i+lag_k] * omh[i]

    #                        = (1-h[j-lag_k])*[0<=j-lag_k<n] - h[j+lag_k]*[0<=j+lag_k<n]
    
    # This is O(n^2) per j, O(n^3) total. For n=45, fine.
    # But let's vectorize:
    for m in range(2*n-1):
        if w[m] < 1e-10:
            continue
        lag_k = m - (n-1)
        # d(result[m])/dh[j] for all j
        # Term 1: (1-h[j-lag_k]) where 0<=j-lag_k<n, i.e., lag_k<=j<n+lag_k

        hi1 = min(n, n + lag_k)
        if lo1 < hi1:

        lo2 = max(0, -lag_k)
        hi2 = min(n, n - lag_k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w[m] * h[lo2+lag_k:hi2+lag_k]

    return obj, grad

    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()

    start = time.time()

            break
        options = {'maxiter': max_iter, 'ftol': 1e-10, 'disp': False}
        res = minimize(
            lambda h: smooth_objective_and_grad(h, t)[0],
            best_h.copy(),
            jac=lambda h: smooth_objective_and_grad(h, t)[1],
            method='SLSQP',

            options=options

            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        elapsed = time.time() - start
    return best_h, best_c5

        hi2 = min(n, n - lag_k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w[m] * h[lo2 + lag_k:hi2 + lag_k]
    grad *= dx
    return obj, grad


def _s_obj(h, t):
    return smooth_obj_grad(h, t)[0]


def _s_jac(h, t):
    return smooth_obj_grad(h, t)[1]


def _sum_fun(h, n):
    return np.sum(h) - n / 2.0


def _sum_jac(h, n):
    return np.ones(n)


def run_slsqp(h0, t_list, max_iter, time_limit, start_elapsed):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for t in t_list:
        elapsed = (time.time() - start_elapsed)
        if elapsed > time_limit * 0.85:
            break
        options = {'maxiter': max_iter, 'ftol': 1e-12, 'disp': False}
        cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
        try:
            res = minimize(_s_obj, best_h.copy(), jac=_s_jac, args=(t,),
                           method='SLSQP', bounds=[(0, 1)] * n,
                           constraints=cons, options=options)
        except Exception:
            continue
        if res.x is not None:
            h_new = repair_h(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    return best_h, best_c5


def run_sa(h0, time_limit, start_temp, rng):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    cur_h = h0.copy()
    cur_c5 = best_c5
    t0 = time.time()
    while time.time() - t0 < time_limit:
        frac = (time.time() - t0) / max(time_limit, 1e-9)
        temp = start_temp * np.exp(-6.0 * frac)
        if temp < 1e-8:
            break
        n_pert = max(1, n // 12)
        idx = rng.choice(n, size=n_pert, replace=False)
        delta = rng.uniform(-0.15, 0.15, size=n_pert)
        h_new = cur_h.copy()
        h_new[idx] += delta
        h_new = repair_h(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5:
            cur_h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        else:
            if rng.random() < np.exp(-(c5_new - cur_c5) / temp):
                cur_h = h_new
                cur_c5 = c5_new
    return best_h, best_c5


def try_round(h, n):
    """Try rounding to nearest 0.1 increments and accept if improves."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(200):
        h_try = np.round(h, decimals=1)
        h_try = repair_h(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            h = h_try
        else:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t_start = time.time()

    h0 = initial_h_values.copy().astype(np.float64)
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP on parent
    t_left = budget_s - (time.time() - t_start)
    best_h, best_c5 = run_slsqp(best_h, [15, 40, 80, 160, 320], 300, t_left, t_start)

    # Phase 2: SA
    t_left = budget_s - (time.time() - t_start)
    if t_left > 20:
        best_h, best_c5 = run_sa(best_h, t_left * 0.45, 0.08, rng)

    # Phase 3: SLSQP again on SA result
    t_left = budget_s - (time.time() - t_start)
    if t_left > 20:
        best_h, best_c5 = run_slsqp(best_h, [100, 250, 500], 200, t_left, t_start)

    # Phase 4: Fine SA
    t_left = budget_s - (time.time() - t_start)
    if t_left > 10:
        best_h, best_c5 = run_sa(best_h, t_left * 0.6, 0.02, rng)

    # Phase 5: Final SLSQP polish
    t_left = budget_s - (time.time() - t_start)
    if t_left > 15:
        best_h, best_c5 = run_slsqp(best_h, [500, 1000], 150, t_left, t_start)

    # Phase 6: Try other n values with interpolated starts
    t_left = budget_s - (time.time() - t_start)
    for n_alt in [30, 60, 90]:
        t_left = budget_s - (time.time() - t_start)
        if t_left < 30:
            break
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_alt)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = repair_h(h_init, n_alt)
        h_alt, c5_alt = run_slsqp(h_init, [50, 150, 400], 150, t_left * 0.5, t_start)
        t_left_tmp = budget_s - (time.time() - t_start)
        if t_left_tmp > 10:
            h_alt, c5_alt = run_sa(h_alt, min(t_left_tmp * 0.5, 20), 0.03, rng)
        if c5_alt < best_c5:
            best_h = h_alt
            best_c5 = c5_alt
            n = n_alt

    # Final repair and evaluation
    best_h = repair_h(best_h, n)
    best_c5 = compute_c5(best_h)

    elapsed = time.time() - t_start
    if elapsed > budget_s * 0.95:
        pass  # We're close to budget, return immediately

    return (best_h, best_c5, n)
