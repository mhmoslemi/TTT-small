import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def make_alternating(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if i % 2 == 0 else 0.0
    return project_h(h, n / 2.0)

def make_block_pattern(n, block_len=4):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % block_len) < block_len // 2 else 0.0
    return project_h(h, n / 2.0)

def make_smooth_step(n):
    x = np.linspace(0, 2, n)
    h = 1.0 / (1.0 + np.exp(-10 * (x - 1.0)))
    return project_h(h, n / 2.0)

def nm_optimize(h_init, target_sum, max_iter=20000, step=0.05):
    def obj(x):
        return evaluate_c5(project_h(x, target_sum))
    res = minimize(obj, h_init, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-12, 'fatol': 1e-14,
                            'initial_simplex_size': step})
    return project_h(res.x, target_sum)

def sa_optimize(h_init, target_sum, n, rng, t_limit, T0=0.015, T_min=0.0003):
    cur = h_init.copy()
    cur_c5 = evaluate_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    T = T0
    cool_rate = 0.9998
    step = 0.08
    n_calls = 0
    while t_limit() > 3:
        n_pert = rng.randint(1, 4)
        idx = rng.choice(n, n_pert, replace=False)
        trial = cur.copy()
        for i in idx:
            trial[i] += rng.uniform(-step, step)
        trial = project_h(trial, target_sum)
        trial_c5 = evaluate_c5(trial)
        if trial_c5 < cur_c5 or rng.random() < np.exp(-(trial_c5 - cur_c5) / T):
            cur = trial
            cur_c5 = trial_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
        T = max(T_min, T * cool_rate)
        step = max(0.005, step * 0.9999)
        n_calls += 1
        if n_calls % 5000 == 0 and T <= T_min * 2:
            T = T0 * 0.5
            step = 0.05
    return best, best_c5

def slsqp_optimize(h_init, target_sum, n):

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        pen = 500.0 * (np.sum(h) - target_sum) ** 2

                   options={'maxiter': 5000, 'xatol': 1e-11, 'fatol': 1e-13,
                            'initial_simplex_size': 0.03})
    return project_h(res.x, target_sum)
