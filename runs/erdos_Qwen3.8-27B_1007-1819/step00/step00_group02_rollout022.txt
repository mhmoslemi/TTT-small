import numpy as np
from scipy.optimize import minimize

_N = 45

def c5_val(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def sobj(h, a):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    s = np.exp(a * (c - m))
    return m + np.log(s.sum()) / a

def sgrad(h, a):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    e = np.exp(a * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        l = k - (n - 1)
        for j in range(n):
            v = 0.0
            j2 = j + l
            if 0 <= j2 < n:
                v += 1.0 - h[j2]
            j3 = j - l
            if 0 <= j3 < n:
                v -= h[j3]
            if v != 0.0:
                g[j] += wk * v
    return g

def proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    for _ in range(500):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not m.any():
            m = h < 1.0 if d > 0 else h > 0.0
            if not m.any():
                break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
