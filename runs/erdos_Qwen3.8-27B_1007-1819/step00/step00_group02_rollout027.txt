import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = np.sum(h)
        diff = n / 2.0 - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 0.005) & (h < 0.995)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, T, pw):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

    val = (m + np.log(np.sum(np.exp(T * (corr - m)))) / T) * dx
    val += pw * (np.sum(h) - n / 2.0) ** 2
