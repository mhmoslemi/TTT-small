import time

def compute_c5(h):

    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * (2.0 / n)

    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n)
    m = np.max(corr)

    for _ in range(100):

            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0

        h = np.clip(h, 0.0, 1.0)
