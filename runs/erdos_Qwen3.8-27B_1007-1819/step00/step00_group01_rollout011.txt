import numpy as np
import time

def compute_c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

def quad_residue_h(n):
    h = np.zeros(n)
    residues = [(i * i) % n for i in range(1, n)]
    seen = sorted(set(residues))
    for r in seen:
        h[r] = 1.0
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    if s < target:
        diff = target - s
        for i in range(n):
            if h[i] == 0.0:
                h[i] = min(diff, 1.0)
                diff -= h[i]
            if diff < 1e-9:
                break
    else:
        diff = s - target
        for i in range(n):
            if h[i] == 1.0:
                cut = min(diff, 1.0)
                h[i] -= cut
                diff -= cut
            if diff < 1e-9:
                break
    return h

def random_binary_h(n, rng):
    h = np.zeros(n)
    target = n / 2.0
    perm = rng.permutation(n)
    if n % 2 == 0:
        h[perm[:n // 2]] = 1.0
    else:
        h[perm[:n // 2]] = 1.0
        h[perm[n // 2]] = 0.5
    return h

def local_search(h, t0, deadline, rng):
    n = len(h)
    c5 = compute_c5_value(h)
    temp = 0.005
    stagnation = 0
    best = h.copy()
    best_c5 = c5

    while time.time() < deadline:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        delta = float(rng.uniform(-0.5, 0.5))
        ni = h[i] - delta
        nj = h[j] + delta
        if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
            continue
        h[i] = ni
        h[j] = nj
        nc5 = compute_c5_value(h)
        if nc5 < c5 or (nc5 - c5) < temp * float(rng.standard_normal()):
            c5 = nc5
            stagnation = 0
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        else:
            h[i] += delta
            h[j] -= delta
            stagnation += 1
        if stagnation > 400:
            temp *= 0.55
            stagnation = 0
            if temp < 1e-8:
                for _ in range(int(rng.integers(3, 12))):
                    a = int(rng.integers(n))
                    b = int(rng.integers(n))
                    if a != b:
                        d = float(rng.uniform(-0.5, 0.5))
                        if 0.0 <= h[a] - d <= 1.0 and 0.0 <= h[b] + d <= 1.0:
                            h[a] -= d
                            h[b] += d
                c5 = compute_c5_value(h)
                temp = 0.005
                stagnation = 0
    return best, best_c5

def coordinate_polish(h, t0, deadline, rng):
    n = len(h)
    c5 = compute_c5_value(h)
    deltas = [-0.25, -0.15, -0.1, -0.05, -0.02, 0.02, 0.05, 0.1, 0.15, 0.25]
    for i in range(n):
        if time.time() > deadline:
            break
        for j in range(i + 1, n):
            if time.time() > deadline:
                break
            best_d = 0.0
            best_c = c5
            for d in deltas:
                ni = h[i] - d
                nj = h[j] + d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i] = ni
                    h[j] = nj
                    nc = compute_c5_value(h)
                    if nc < best_c:
                        best_c = nc
                        best_d = d
            h[i] += best_d
            h[j] -= best_d
            c5 = best_c
    return h, c5

def fix_sum(h):
    n = len(h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 0.005) & (h < 0.995)
        if mask.any():
            h[mask] += diff / int(mask.sum())
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def try_solution(h, c5):
        nonlocal best_h, best_c5, best_n
        if h is not None and c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = len(h)

    # Phase 1: Optimize from initial_h_values (if available)
    phase1_end = t0 + budget_s * 0.30
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        h0 = fix_sum(h0)
        c5_init = compute_c5_value(h0)
        try_solution(h0, c5_init)
        h_opt, c5_opt = local_search(h0.copy(), t0, phase1_end, rng)
        try_solution(h_opt, c5_opt)

    # Phase 2: Quadratic residue starts (primes mod 4 = 3 preferred)
    phase2_end = t0 + budget_s * 0.55
    prime_candidates = [43, 47, 59, 67, 71, 79, 83, 103, 107, 127]
    for n in prime_candidates:
        if time.time() > phase2_end:
            break
        if not is_prime(n):
            continue
        h_qr = quad_residue_h(n)
        chunk_end = min(time.time() + max(2.0, budget_s * 0.03), phase2_end)
        h_opt, c5_opt = local_search(h_qr.copy(), t0, chunk_end, rng)
        try_solution(h_opt, c5_opt)

    # Phase 3: Random binary multi-start
    phase3_end = t0 + budget_s * 0.85
    for n in [47, 53, 59, 61, 63, 71, 73, 79, 83, 89, 95, 101, 107, 113, 127]:
        if time.time() > phase3_end:
            break
        for trial in range(2):
            if time.time() > phase3_end:
                break
            h_rand = random_binary_h(n, rng)
            chunk_end = min(time.time() + max(2.0, budget_s * 0.025), phase3_end)
            h_opt, c5_opt = local_search(h_rand, t0, chunk_end, rng)
            try_solution(h_opt, c5_opt)

    # Phase 4: Re-optimize the best found with more time
    phase4_end = t0 + budget_s * 0.93
    if best_h is not None and time.time() < phase4_end:
        h_pol, c5_pol = local_search(best_h.copy(), t0, phase4_end, rng)

    # Phase 5: Coordinate descent polish
    polish_end = t0 + budget_s * 0.97
    if best_h is not None and time.time() < polish_end:
        best_h, best_c5 = coordinate_polish(best_h.copy(), t0, polish_end, rng)
        best_c5 = compute_c5_value(best_h)

    # Final: ensure exact sum and bounds

        best_h = fix_sum(best_h)
        best_h = np.clip(best_h, 0.0, 1.0)

        best_h = random_binary_h(n_fb, rng)
        best_h = fix_sum(best_h)
        best_c5 = compute_c5_value(best_h)
        best_n = n_fb

    return best_h, best_c5, best_n
