```python
    h = h.copy()
    for _ in range(100):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to decrease; reduce entries that are above target/n
            mask = h > target_sum / len(h)
        else:
            mask = h < target_sum / len(h)

        h[mask] -= np.sign(diff) * min(abs(adj),
```

```python
def _proj(h, tgt):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    n = len(h)
    d = h.sum() - tgt
    if abs(d) < 1e-15:
        return h
    if d > 0:
        # Need to reduce by d
        idx = np.argsort(h)[::-1]  # largest first
        remaining = d
        for i in idx:
            can_reduce = h[i]
            take = min(remaining, can_reduce)
            h[i] -= take
            remaining -= take
            if remaining < 1e-15:
                break
    else:
        # Need to increase by -d
        idx = np.argsort(h)  # smallest first
        remaining = -d
        for i in idx:
            can_increase = 1.0 - h[i]
            take = min(remaining, can_increase)
            h[i] += take
            remaining -= take
            if remaining < 1e-15:
                break
    return np.clip(h, 0.0, 1.0)
```

```python
                break
            h[fr] += (-d) / fr.sum()
    return np.clip(h, 0.0, 1.0)

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _smax(v, t):
    m = v.max()
    return m + np.log(np.exp(np.clip(t * (v - m), -400.0, 400.0)).sum()) / t

def _obj(h):
    dx, tau, pen, tgt = _P['dx'], _P['tau'], _P['pen'], _P['tgt']
    lags = np.correlate(h, 1.0 - h, mode="full") * dx
    return _smax(lags, tau) + pen * (h.sum() - tgt) ** 2

def _opt(h, dx, tau, pen, iters):
    _P['dx'] = dx
    _P['tau'] = tau
    _P['pen'] = pen
    _P['tgt'] = len(h) / 2.0
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj, h, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': iters, 'ftol': 1e-13, 'gtol': 1e-9, 'maxls': 50})
    return _proj(np.clip(res.x, 0.0, 1.0), n / 2.0)

def _coord(h, dx, tgt, passes):
    n = len(h)
    h = h.copy()
    best = _c5(h, dx)
    for _p in range(passes):
        imp = False
        order = np.argsort(np.abs(h - 0.5))
        for i in order:
            for v in (0.0, 1.0):
                if abs(h[i] - v) < 1e-10:
                    continue
                old = h[i]
                h[i] = v
                h = _proj(h, tgt)
                c = _c5(h, dx)
                if c < best - 1e-12:
                    best = c
                    imp = True
                else:
                    h[i] = old
                    h = _proj(h, tgt)
        if not imp:
            break
    return h, best

def _snap(h, tgt):
    h = h.copy()
    near = (h > 0.9998) | (h < 2e-4)
    if near.any():
        h[near] = np.where(h[near] > 0.5, 1.0, 0.0)
        h = _proj(h, tgt)
        return h
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        h0 = np.zeros(47)
        h0[:23] = 1.0
        h0[23] = 0.5

    n = len(h0)
    dx = 2.0 / n
    tgt = n / 2.0
    h0 = _proj(h0, tgt)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: progressive L-BFGS-B with increasing tau
    for tau in (3.0, 8.0, 20.0, 60.0, 150.0, 400.0, 1000.0, 3000.0):
        if time.time() - t0 > budget_s * 0.30:
            break
        hn = _opt(best_h, dx, tau, 200.0, 250)
        cn = _c5(hn, dx)
        if cn < best_c5:
            best_h, best_c5 = hn, cn
        best_h = _snap(best_h, tgt)

    # Phase 2: coordinate descent (binary flips)
    if time.time() - t0 < budget_s * 0.50:
        best_h, best_c5 = _coord(best_h, dx, tgt, 50)
        best_h = _snap(best_h, tgt)
        best_c5 = _c5(best_h, dx)

    # Phase 3: re-optimize from binary-rounded start
    if time.time() - t0 < budget_s * 0.60:
        hb = _proj(np.where(best_h >= 0.5, 1.0, 0.0), tgt)
        cb = _c5(hb, dx)
        if cb <= best_c5:
            for tau in (50.0, 300.0, 1000.0):
                if time.time() - t0 > budget_s * 0.68:
                    break
                hb = _opt(hb, dx, tau, 200.0, 200)
                cb = _c5(hb, dx)
            if cb < best_c5:
                best_h, best_c5 = hb, cb
            best_h, best_c5 = _coord(best_h, dx, tgt, 30)

    # Phase 4: upscale to finer grid
    for f in (2, 3):
        n2 = n * f
        if n2 > 400 or time.time() - t0 > budget_s * 0.72:
            break
        x1 = np.linspace(0.0, 2.0, n, endpoint=False)
        x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = _proj(np.interp(x2, x1, best_h), n2 / 2.0)
        dx2 = 2.0 / n2
        for tau in (5.0, 30.0, 150.0, 600.0):
            if time.time() - t0 > budget_s * 0.84:
                break
            h2 = _opt(h2, dx2, tau, 200.0, 150)
        c2 = _c5(h2, dx2)
        if c2 < best_c5:
            h2, c2 = _coord(h2, dx2, n2 / 2.0, 20)
            if c2 < best_c5:
                best_h, best_c5, n, dx, tgt = h2, c2, n2, dx2, n2 / 2.0
                best_h = _snap(best_h, tgt)
                best_c5 = _c5(best_h, dx)

    # Phase 5: random restarts
    if time.time() - t0 < budget_s * 0.88:
        for r in range(15):
            if time.time() - t0 > budget_s * 0.96:
                break
            scale = 0.15 + 0.1 * r
            hr = best_h + scale * rng.standard_normal(n)
            hr = _proj(hr, n / 2.0)
            hr = _opt(hr, dx, 30.0, 200.0, 100)
            hr = _opt(hr, dx, 200.0, 200.0, 100)
            cr = _c5(hr, dx)
            if cr < best_c5:
                hr, cr = _coord(hr, dx, n / 2.0, 15)
                if cr < best_c5:
                    best_h, best_c5 = hr, cr
                    best_h = _snap(best_h, tgt)
                    best_c5 = _c5(best_h, dx)

    # Final: ensure exact constraint and clean up
    best_h = _proj(best_h, tgt)
    best_h = _snap(best_h, tgt)
    best_c5 = _c5(best_h, dx)

    return (best_h, best_c5, n)
```
