    dx = 2.0 / n
    one_mh = 1 - h
    c = np.correlate(h, one_mh, mode="full") * dx
    # c has 2n-1 elements, indexed by k=0..2n-2
    # shift s_k = k - (n-1)
    shifts = np.arange(2*n-1) - (n-1)  # -(n-1) to (n-1)
    
    # weights

    lw = beta * (c - c_max)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(c)

def project_feasible(h, S):
    """Project onto {x: 0<=x<=1, sum(x)=S} using bisection on water level"""
    n = len(h)
    def sum_clipped(t):
        return np.sum(np.clip(h - t, 0.0, 1.0))
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if sum_clipped(mid) > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.clip(h - t, 0.0, 1.0)
    # Correct tiny numerical errors
    diff = S - np.sum(result)
    if abs(diff) > 1e-12:
        # Adjust the most "interior" element
        interior = (result > 1e-10) & (result < 1 - 1e-10)
        if np.sum(interior) > 0:
            result[interior] += diff / np.sum(interior)
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_max(c, beta):
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta

def grad_smooth_max(h, n, beta):
    dx = 2.0 / n
    one_mh = 1.0 - h
    c = np.correlate(h, one_mh, mode="full") * dx
    c_max = np.max(c)
    lw = beta * (c - c_max)
    w = np.exp(lw)
    w_sum = np.sum(w)
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.ones_like(c) / len(c)
    
    shifts = np.arange(len(c)) - (n - 1)
    g = np.zeros(n)
    for k in range(len(c)):
        s = shifts[k]
        wk = w[k]
        if wk < 1e-14:
            continue
        # Term 1: +wk*(1-h[j-s]) for j in [max(0,s), min(n-1, s+n-1)]
        js = max(0, s)
        je = min(n - 1, s + n - 1)
        if js <= je:
            idx = np.arange(js, je + 1) - s
            g[js:je+1] += wk * (1.0 - h[idx])
        # Term 2: -wk*h[j+s] for j in [max(0,-s), min(n-1, n-1-s)]
        js2 = max(0, -s)
        je2 = min(n - 1, n - 1 - s)
        if js2 <= je2:
            idx2 = np.arange(js2, je2 + 1) + s
            g[js2:je2+1] -= wk * h[idx2]
    
    return dx * g

def projected_gradient_step(h, grad, S, lr):
    h_new = h - lr * grad
    return project_feasible(h_new, S)

    t_end = t0 + budget_frac * budget_s

                h[i] = rng.uniform(0, 1)
        h = project_feasible(h, S)
        c5, c = compute_c5(h, n)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                h += rng.normal(0, 0.05, n)
                h = project_feasible(h, S)
                stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 999.0
    best_n = 1
    cands = []

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.clip(np.array(initial_h_values, dtype=float), 0, 1)
        n0 = len(h0)
        h0 = project_feasible(h0, n0 / 2.0)
        c5, _ = compute_c5(h0, n0)
        cands.append((c5, h0, n0))

    for n in [46, 48, 64, 96, 128, 192]:
        for tr in range(3):
            h = make_binary_start(n, seed * 31 + n * 7 + tr * 13)
            c5, _ = compute_c5(h, n)
            cands.append((c5, h, n))

    cands.sort(key=lambda x: x[0])
    T1 = t0 + budget_s * 0.55
    T2 = t0 + budget_s * 0.80
    T3 = t0 + budget_s * 0.97

    n_tries = min(7, len(cands))
    for i in range(n_tries):
        if time.time() > T1:
            break
        c5i, hi, ni = cands[i]
        remaining = T1 - time.time()
        tl = time.time() + remaining / (n_tries - i)
        bts = [3, 8, 20, 50, 100, 200, 400]
        ho, c5o = optimize_pg(hi, ni, tl, bts)
        if c5o < best_c5:
            best_c5 = c5o
            best_h = ho
            best_n = ni

    if best_h is not None and time.time() < T2:
        S = best_n / 2.0
        h2, c52 = local_swap_search(best_h, best_n, T2, S)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    if best_h is not None and time.time() < T3:
        bts2 = [30, 80, 200, 500]
        h3, c53 = optimize_pg(best_h, best_n, T3, bts2, steps_per_beta=6000)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3

    if best_h is None:
        n_fallback = 48
        best_h = make_binary_start(n_fallback, seed)
        best_n = n_fallback
        best_c5, _ = compute_c5(best_h, best_n)

    best_h = np.clip(best_h, 0, 1)
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5, _ = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
