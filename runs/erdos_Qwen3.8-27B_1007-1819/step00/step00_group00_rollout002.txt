def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def soft_max_c5(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(vals - m))) / beta


def soft_max_grad(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    weights = np.exp(vals * beta - m * beta) / (np.sum(np.exp(vals * beta - m * beta)))
    # d(corr_k)/d(h_i) for np.correlate(h, 1-h, full)
    # r[k] = sum_j h[j] * (1 - h[j - (k - n + 1)])
    # Let's compute gradient numerically for safety with n=45
    g = np.zeros_like(h)
    eps = 1e-6
    f0 = soft_max_c5(h, dx, beta)
    for i in range(len(h)):
        h[i] += eps
        f1 = soft_max_c5(h, dx, beta)
        h[i] -= eps
        g[i] = (f1 - f0) / eps
    return g


def fix_sum(h, sum_target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:

        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h
