    return fix_sum(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    # Determine n_points from initial if available
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h0)
    except (NameError, Exception):
        n_points = 45
        h0 = gen_random_h(n_points, n_points / 2.0, rng)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = fix_sum(h0, target_sum)
    best_c5 = compute_c5(best_h, dx)

    bounds = [(0.0, 1.0)] * n_points
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n_points)}]

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        return compute_c5(h, dx)

    def try_opt(h_init, maxiter=1500, ftol=1e-13):
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < best_c5 - 1e-12:
            new_h = fix_sum(np.clip(res.x, 0, 1), target_sum)
            new_c5 = compute_c5(new_h, dx)
            return new_h, new_c5
        return None, None

    def elapsed():
        return time.time() - t_start

    # Phase 1: Optimize from initial
    r, c = try_opt(best_h, maxiter=3000, ftol=1e-14)
    if r is not None:
        best_h, best_c5 = r, c

    # Phase 2: Multi-start random restarts
    n_strats = [gen_random_h, gen_block_h, gen_smooth_h]
    attempt = 0
    while elapsed() < budget_s - 30:
        attempt += 1
        # Choose strategy
        strat = n_strats[attempt % len(n_strats)]
        h_init = strat(n_points, target_sum, rng)

        # Occasionally perturb the best solution
        if attempt % 5 == 0:
            h_init = best_h + rng.normal(0, 0.15, n_points)
            h_init = fix_sum(np.clip(h_init, 0, 1), target_sum)
        elif attempt % 5 == 1:
            h_init = best_h + rng.normal(0, 0.3, n_points)
            h_init = fix_sum(np.clip(h_init, 0, 1), target_sum)

        r, c = try_opt(h_init, maxiter=1000, ftol=1e-12)
        if r is not None:
            best_h, best_c5 = r, c
            # Gradient step from new best
            r2, c2 = try_opt(best_h, maxiter=2000, ftol=1e-14)
            if r2 is not None:
                best_h, best_c5 = r2, c2

    # Phase 3: Local perturbation refinement
    while elapsed() < budget_s - 10:
        improved = False
        for _ in range(20):
            if elapsed() > budget_s - 10:
                break
            h_pert = best_h + rng.normal(0, 0.05, n_points)
            h_pert = fix_sum(np.clip(h_pert, 0, 1), target_sum)
            r, c = try_opt(h_pert, maxiter=500, ftol=1e-13)
            if r is not None:
                best_h, best_c5 = r, c
                improved = True
        if not improved:
            break

    # Final polish
    if elapsed() < budget_s - 5:
        r, c = try_opt(best_h, maxiter=5000, ftol=1e-15)
        if r is not None:
            best_h, best_c5 = r, c

    best_h = fix_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, n_points
