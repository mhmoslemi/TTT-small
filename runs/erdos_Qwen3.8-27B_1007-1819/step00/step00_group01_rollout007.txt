                    h[j] += delta
                    h[k2] -= delta
                    if 0.0 <= h[j] <= 1.0 and 0.0 <= h[k2] <= 1.0:
                        c = _c5(h)
                        if c < best - 1e-11:
                            best = c
                            improved_sweep = True
                            continue
                    h[j] -= delta
                    h[k2] += delta
        if not improved_sweep:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s - 25
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h_init = np.zeros(47)
        h_init[12:24] = 1.0
    n = len(h_init)
    h_init = _project(h_init, n)

    best_h = h_init.copy()
    best_c5 = _c5(h_init)

    # Phase 1: PSGD with smooth gradient
    for beta in [30, 80, 200, 500]:
        for lr in [0.01, 0.03, 0.08, 0.15]:
            if time.time() > t_end - 120:
                break
            h_res, c5_res = _psgd(h_init, n, beta, lr, t_end)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
            # Also try from current best
            if c5_res < best_c5 + 0.01:
                h_res2, c5_res2 = _psgd(h_res, n, beta * 2, lr * 0.5, t_end)
                if c5_res2 < best_c5:
                    best_c5 = c5_res2
                    best_h = h_res2.copy()

    # Phase 2: Hard subgradient
    for lr in [0.005, 0.02, 0.05]:
        if time.time() > t_end - 80:
            break
        h_res, c5_res = _psgd(best_h, n, 0, lr, t_end, use_hard=True)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 3: Random restarts
    if time.time() < t_end - 100:
        for rr in range(5):
            if time.time() > t_end - 80:
                break
            h_pert = best_h + 0.05 * (rng.random(n) - 0.5)
            h_pert = _project(h_pert, n)
            h_res, c5_res = _psgd(h_pert, n, 100, 0.05, t_end)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()

    # Phase 4: Coordinate descent polish
    remain = max(5, (budget_s - 20) - (time.time() - t0))
    h_cd, c5_cd = _coord_desc(best_h.copy(), remain)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, int(n)
