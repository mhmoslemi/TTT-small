So for k=0, the best we can do is make ∫h² as large as possible, which means h should be as "concentrated" as possible (close to 0 or 1).

But for other k, we want h(x) ≈ h(x+k), which favors h being "spread out" (constant-like).

So there's a trade-off. The optimal h is somewhere in between.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def obj_smooth(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, alpha) * dx

def obj_exact(params, n):
    h = project_feasible(params, n)
    return compute_c5(h)

def refine_slsqp(h0, n, alpha, maxiter=500):
    try:
        res = minimize(
            obj_smooth, h0.copy(), args=(n, alpha),
            method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def refine_exact(h0, n, maxiter=300):
    try:
        res = minimize(
            obj_exact, h0.copy(), args=(n,),
            method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def simulated_annealing(h0, n, budget_remain):
    h = h0.copy()
    curr = compute_c5(h)
    best_local = h.copy()
    best_val = curr
    T0 = 0.008
    n_steps = int(budget_remain / 0.004)
    if n_steps < 100:
        return best_local, best_val
    for step in range(n_steps):
        if time.time() < 0:
            break
        n_pert = np.random.choice([1, 2, 3], p=[0.6, 0.3, 0.1])
        idxs = np.random.choice(n, size=n_pert, replace=False)
        h_new = h.copy()
        for i in idxs:
            h_new[i] = np.clip(h_new[i] + np.random.randn() * 0.08, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        T = T0 * (1.0 - step / n_steps) ** 2
        if c_new < curr or np.random.rand() < np.exp(-(c_new - curr) / max(T, 1e-15)):
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best_local = h.copy()
    return best_local, best_val

def coordinate_polish(h, n, n_rounds=200):
    h = h.copy()
    curr = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            best_val = curr
            best_v = h[i]
            for rv in np.linspace(0, 1, 11):
                if rv == h[i]:
                    continue
                h[i] = rv
                h = project_feasible(h, n)
                v = compute_c5(h)
                if v < best_val:
                    best_val = v
                    best_v = h[i]
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_val
        if not improved:
            break
    return h, curr
