    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    m = np.max(v)
    if m > 1e6:
        return float(m)
    return float(m + np.log(np.sum(np.exp(b * (v - m)))) / b)

    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(v, b):
    m = np.max(v)
    if m > 1e6:
        return float(m)
    return float(m + np.log(np.sum(np.exp(b * (v - m)))) / b)

def slsqp_obj(x, n, beta):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(c * dx, beta)

def slsqp_con(x, tgt):
    return float(np.sum(x) - tgt)

def nm_obj(x, n, beta, pw):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full')
    sd = h.sum() - n / 2.0
    return smooth_max(c * dx, beta) + pw * sd * sd

def do_slsqp(h0, n, mi, b):

    cons = [{'type': 'eq', 'fun': slsqp_con, 'args': (tgt,)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(slsqp_obj, h0, args=(n, b), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': mi, 'ftol': 1e-14})
    h = np.clip(res.x, 0.0, 1.0)
    h = fix_sum(h, n)
