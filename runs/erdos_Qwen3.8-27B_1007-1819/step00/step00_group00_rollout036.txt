import numpy as np
import scipy.optimize as opt
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target, tol=1e-13):
    h = h.astype(float).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < tol:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr_vals, beta):
    mx = np.max(corr_vals)
    shifted = beta * (corr_vals - mx)
    shifted = np.clip(shifted, -500, 500)
    return mx + np.log(np.sum(np.exp(shifted))) / beta

def opt_lbf(h0, n, beta=200, pen=1e5, maxiter=10000):
    dx = 2.0 / n
    target = n / 2.0
    h = fix_sum(np.clip(h0, 0, 1), target)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        sm = smooth_max(corr, beta)
        return sm + pen * (x.sum() - target) ** 2

    res = opt.minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    h_opt = fix_sum(np.clip(res.x, 0, 1), target)
    return h_opt, compute_c5(h_opt, dx)

def opt_slsqp(h0, n, beta=100, maxiter=500):

        return x[n]
