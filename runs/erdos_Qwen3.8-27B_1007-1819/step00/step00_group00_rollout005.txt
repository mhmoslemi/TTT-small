import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, n, rng=None):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if rng is not None:
            perm = rng.permutation(n)
        else:
            perm = np.arange(n)
        for j in perm:
            if diff > 0:
                adj = min(diff, 1.0 - h[j])
            else:
                adj = max(diff, -h[j])
            h[j] += adj
            diff -= adj
            if abs(diff) < 1e-13:
                break
    return h

def sa_perturb(h, n, rng):
