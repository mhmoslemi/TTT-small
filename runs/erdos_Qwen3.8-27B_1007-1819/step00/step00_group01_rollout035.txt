def compute_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    vals = corr * dx
    m = vals.max()
    s = np.clip(tau * (vals - m), -50, 0)
    w = np.exp(s)
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(2*n-1):
        if w[j] < 1e-12:
            continue
        k = j - (n-1)  # shift
        coeff = w[j] * tau * dx
        # corr[j] = sum_{i: 0<=i<n, 0<=i+k<n} h[i]*(1-h[i+k])
        # d/dh[i] = (1-h[i+k]) - h[i-k] (boundary: 0<=i+k<n, 0<=i-k<n)
        i_lo = max(0, -k)
        i_hi = min(n-1, n-1-k)
        if i_lo > i_hi:
            continue
        # Build gradient contribution
        ii = np.arange(i_lo, i_hi+1)
        g = np.zeros(n)
        g[ii] = coeff * (1 - h[ii+k])
        # Subtract h[i-k] where valid
        valid = (ii - k >= 0) & (ii - k < n)
        g[ii[valid]] -= coeff * h[ii[valid]-k]
        grad += g
    return grad
