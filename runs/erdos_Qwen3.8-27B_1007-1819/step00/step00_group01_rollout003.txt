def compute_c5(h, n_points=None):

def objective_and_grad(h_flat, N):
    h = h_flat.reshape(-1)
    # Objective: max of corr * dx
    dx = 2.0 / N
    gh = 1.0 - h
    corr = np.correlate(h, gh, mode="full")
    vals = corr * dx
    max_val = vals.max()
    active = np.where(vals >= max_val - 1e-12)[0]
    
    # Subgradient
    grad = np.zeros_like(h_flat)
    for k in active:
        lag = k - (N - 1)
        i_start = max(0, -lag)
        i_end = min(N - 1, N - 1 - lag)
        for i in range(i_start, i_end + 1):
            j = i + lag
            grad[j] += (1.0 - h[i]) * dx
            grad[i] += -h[j] * dx
    grad /= len(active)
    return max_val, grad

def project_constraint(h, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0

        proj = np.clip(h - mid, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points=None):
    if n_points is None:
        n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_subgradient(h, N):
    gh = 1.0 - h
    corr = np.correlate(h, gh, mode="full")
    dx = 2.0 / N
    vals = corr * dx
    max_val = np.max(vals)
    active = np.where(vals >= max_val - 1e-14)[0]
    grad = np.zeros(N)
    for k in active:
        lag = k - (N - 1)
        n_start = max(0, -lag)
        n_end = min(N - 1, N - 1 - lag)
        for n in range(n_start, n_end + 1):
            m = n + lag
            grad[m] += (1.0 - h[n])
            grad[n] += -h[m]
    grad /= len(active)
    grad *= dx
    return max_val, grad

def project_constraint(h, N):
    target = N / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_optimize(h0, N, n_iters, start_lr, rng, shake_interval=200, shake_mag=0.1):
    h = h0.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = compute_c5(h, N)
    for t in range(n_iters):
        lr = start_lr / (1.0 + t * start_lr / 5.0)
        c5, grad = compute_subgradient(h, N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h = project_constraint(h - lr * grad, N)
        if (t + 1) % shake_interval == 0 and best_c5 > 0.37:
            noise = rng.randn(N) * shake_mag * max(0.01, 1.0 - t / n_iters)
            h = project_constraint(h + noise, N)
    return best_h, best_c5

        h_trial[i] += move
        h_trial[j] -= move
        h_trial = np.clip(h_trial, 0.0, 1.0)

            h = h_trial
        elif rng.rand() < np.exp(-(c5_trial - best_c5) / (temperature * 0.1)):
            h = h_trial
    return best_h, best_c5
