np.correlate(h, 1-h, mode="full") for h of length n:
- Output has length 2n-1
- Output[k] = sum_{valid i} h[i] * (1-h[i + k - (n-1)])

For k = 0 (shift = -(n-1)): only i=n-1 is valid, output[0] = h[n-1] * (1-h[0])
For k = n-1 (shift = 0): all i valid, output[n-1] = sum(h[i]*(1-h[i]))
For k = 2n-2 (shift = n-1): only i=0 valid, output[2n-2] = h[0]*(1-h[n-1])

- Compute C₅

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-12:
        h[:] = 0.5
        return h
    h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            sr = np.sum(room)
            if sr < 1e-14:
                break
            h += diff * room / sr
        else:
            avail = h.copy()
            sa = np.sum(avail)
            if sa < 1e-14:
                break
            h += diff * avail / sa
        h = np.clip(h, 0.0, 1.0)
    return h

def make_neighbor(h, n, rng):
    h_new = h.copy()
    strat = rng.random()
    if strat < 0.25:
        blen = int(rng.integers(2, min(9, n // 3)))
        start = int(rng.integers(0, n - blen))
        h_new[start:start + blen] += rng.random(blen) * 0.6 - 0.3
    elif strat < 0.45:
        blen = int(rng.integers(1, min(4, n // 5)))
        start = int(rng.integers(0, n - blen))
        h_new[start:start + blen] = 1.0 - h_new[start:start + blen]
    elif strat < 0.65:
        idx = int(rng.integers(0, n))
        h_new[idx] += rng.random() * 0.5 - 0.25
    elif strat < 0.85:
        bl = int(rng.integers(2, min(6, n // 4)))
        i1 = int(rng.integers(0, n - bl))
        i2 = int(rng.integers(0, n - bl))
        if abs(i1 - i2) >= bl:
            tmp = h_new[i1:i1+bl].copy()
            h_new[i1:i1+bl] = h_new[i2:i2+bl]
            h_new[i2:i2+bl] = tmp
    else:
        h_new += rng.normal(0, 0.08, size=n)
    return repair(h_new, n)

def sa_optimize(h, n, time_limits, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_c = h.copy()
    c5_c = best_c5
    T = 0.04
    T_min = 0.002
    cool = 0.9995
    t0 = time.time()
    while time.time() - t0 < time_limits:
        h_n = make_neighbor(h_c, n, rng)
        c5_n = compute_c5(h_n)
        d = c5_n - c5_c
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
            h_c = h_n
            c5_c = c5_n
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_c.copy()
        T = max(T * cool, T_min)
    return best_h, best_c5

def grad_refine(h, n, time_limits, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_c = h.copy()
    lr = 0.03
    eps = 0.005
    t0 = time.time()
    stall = 0
    while time.time() - t0 < time_limits:
        c5_c = compute_c5(h_c)
        grad = np.zeros(n)
        for i in range(n):
            hp = h_c.copy()
            hp[i] = min(1.0, hp[i] + eps)
            hp = repair(hp, n)
            cm = h_c.copy()
            cm[i] = max(0.0, cm[i] - eps)
            cm = repair(cm, n)
            grad[i] = (compute_c5(hp) - compute_c5(cm)) / (2.0 * eps)
        h_n = h_c - lr * grad
        h_n = repair(h_n, n)
        c5_n = compute_c5(h_n)
        if c5_n < c5_c - 1e-8:
            h_c = h_n
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_c.copy()
                stall = 0
            lr = min(lr * 1.03, 0.15)
        else:
            stall += 1
            lr *= 0.92
            if stall > 30:
                break
    return best_h, best_c5

def make_initial(n, rng, variant):
    h = np.zeros(n)
    if variant == 0:
        w = max(2, n // 4)
        s = n // 4
        h[s:s+w] = 1.0
    elif variant == 1:
        w = max(1, n // 8)
        h[n//8 : n//8+w] = 1.0
        h[3*n//8 : 3*n//8+w] = 1.0
    elif variant == 2:
        h[:] = rng.random(n) * 0.5 + 0.25
    elif variant == 3:
        h[int(n//5):int(4*n//5)] = 1.0
    elif variant == 4:
        freq = rng.uniform(1, 3)
        x = np.linspace(0, 2, n)
        h[:] = 0.5 + 0.4 * np.cos(2 * np.pi * freq * x / 2.0)
    else:
        h[:] = (rng.random(n) < 0.5).astype(float)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        init = np.array(initial_h_values, dtype=float)
    except NameError:
        init = None

    if init is not None and len(init) > 0:
        best_n = len(init)
        best_h = repair(init.copy(), best_n)
        best_c5 = compute_c5(best_h)
    else:
        best_n = 45
        best_h = make_initial(best_n, rng, 0)
        best_c5 = compute_c5(best_h)

    # Phase 1: SA on initial
    elapsed = time.time() - t_start
    sa_time = min(250, budget_s * 0.30) - elapsed
    if sa_time > 10:
        h1, c1 = sa_optimize(best_h, best_n, sa_time, rng)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1

    # Phase 2: Gradient refine
    elapsed = time.time() - t_start
    gr_time = min(200, budget_s * 0.25) - (elapsed - sa_time) if sa_time > 10 else min(200, budget_s * 0.25)
    gr_time = budget_s * 0.25 - (time.time() - t_start) + min(200, budget_s * 0.25)
    gr_time = min(200, budget_s - (time.time() - t_start) - 100)
    if gr_time > 10:
        h2, c2 = grad_refine(best_h, best_n, gr_time, rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 3: SA again from best
    remain = budget_s - (time.time() - t_start)
    if remain > 60:
        sa2_time = min(150, remain * 0.5)
        h3, c3 = sa_optimize(best_h, best_n, sa2_time, rng)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3
        remain = budget_s - (time.time() - t_start)
        if remain > 30:
            h4, c4 = grad_refine(best_h, best_n, min(60, remain - 10), rng)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4

    # Phase 4: Try other n values
    remain = budget_s - (time.time() - t_start)
    for n_try in [30, 60, 90, 120]:
        if remain < 40:
            break
        for var in range(4):
            if time.time() - t_start > budget_s - 30:
                break
            h_t = make_initial(n_try, rng, var)
            c_t = compute_c5(h_t)
            sa_t = min(50, (budget_s - 30) - (time.time() - t_start))
            if sa_t < 10:
                break
            h_t, c_t2 = sa_optimize(h_t, n_try, sa_t, rng)
            if c_t2 < best_c5:
                best_c5 = c_t2
                best_h = h_t.copy()
                best_n = n_try
            remain = budget_s - (time.time() - t_start)
            if remain > 20:
                h_t, c_t3 = grad_refine(h_t, n_try, min(25, remain - 15), rng)
                if c_t3 < best_c5:
                    best_c5 = c_t3
                    best_h = h_t.copy()
                    best_n = n_try

    # Phase 5: Final gradient polish
    remain = budget_s - (time.time() - t_start)
    if remain > 15:
        h_f, c_f = grad_refine(best_h, best_n, remain - 5, rng)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    best_h = repair(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
