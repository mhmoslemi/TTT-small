```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    ones = np.ones(n)
    corr = np.correlate(h, ones - h, mode='full')
    return np.max(corr) * dx

def soft_max_obj_gradient(h, temp=30.0):
    n = len(h)
    dx = 2.0 / n
    ones = np.ones(n)
    corr = np.correlate(h, ones - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_vals = np.exp(temp * shifted)
    weights = exp_vals / np.sum(exp_vals)
    obj = (max_c + np.log(np.sum(exp_vals)) / temp) * dx
    v = 1.0 - 2.0 * h
    grad = np.correlate(weights, v, 'valid') * dx
    return obj, grad

def repair(h, target_sum):
    """Clip to [0,1] and adjust sum."""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if diff > 0:
        # Need to add diff; add to elements with room
        mask = h < 1
        if np.sum(mask) == 0:
            return h
        add_per = diff / np.sum(mask)
        h[mask] = np.minimum(h[mask] + add_per, 1.0)
    elif diff < 0:
        mask = h > 0
        if np.sum(mask) == 0:
            return h
        sub_per = -diff / np.sum(mask)
        h[mask] = np.maximum(h[mask] - sub_per, 0.0)
    return h

def resample_h(h_old, n_new):
    """Resample from n_old points to n_new points."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def simulate_annealing(h, n_evals, rng, target_sum, start_temp=0.05, end_temp=0.001):
    """SA optimization."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    current_c5 = best_c5
    n = len(h)
    
    for i in range(n_evals):
        progress = i / max(n_evals - 1, 1)
        temp = start_temp * (end_temp / start_temp) ** progress
        n_perturb = rng.integers(1, max(2, n // 5))
        idx = rng.integers(0, n, size=n_perturb)
        new_h = current_h.copy()
        new_h[idx] += rng.normal(0, temp, size=n_perturb)
        new_h = repair(new_h, target_sum)
        new_c5 = compute_c5(new_h)
        
        if new_c5 < current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / max(temp * 0.01, 1e-10)):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_TEMP = 20.0
_TS = 22.5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _soft_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    ev = np.exp(_TEMP * (corr - mc))
    return (mc + np.log(np.sum(ev)) / _TEMP) * dx

def _soft_grad(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    ev = np.exp(_TEMP * (corr - mc))
    w = ev / np.sum(ev)
    return np.correlate(w, 1.0 - 2.0 * x, 'valid') * dx

def _soft_fandg(x):
    return _soft_obj(x), _soft_grad(x)

def _sum_con(x):
    return np.sum(x) - _TS

def _sum_jac(x):
    return np.ones(len(x))

def repair(h, ts):
    h = np.clip(h, 0.0, 1.0)
    d = ts - np.sum(h)
    if abs(d) > 1e-14:
        m = (h > 0.0) | (h < 1.0)
        s = int(np.sum(m))
        if s > 0:
            h[m] = np.clip(h[m] + d / s, 0.0, 1.0)
    return h

    bc = compute_c5(best)
    cur = h.copy()
    cc = bc
    n = len(h)
    for i in range(nevals):
        p = i / max(nevals - 1, 1)
        ps = p0 + (p1 - p0) * p
        at = at0 * (at1 / at0) ** p
        a = rng.integers(0, n)
        b = rng.integers(0, n)
        if a == b:
            b = (a + 1) % n
        amt = rng.normal(0.0, ps)
        nh = cur.copy()
        nh[a] += amt
        nh[b] -= amt
        nh = np.clip(nh, 0.0, 1.0)
        diff = np.sum(cur) - np.sum(nh)
        if abs(diff) > 1e-14:
            m = (nh > 0.0) & (nh < 1.0)
            sm = int(np.sum(m))
            if sm > 0:
                nh[m] = np.clip(nh[m] + diff / sm, 0.0, 1.0)
        nc = compute_c5(nh)
        delta = nc - cc
        if delta < 0 or rng.random() < np.exp(-delta / max(at, 1e-12)):
            cur = nh
            cc = nc
            if nc < bc:
                best = nh.copy()
                bc = nc

    res = minimize(_soft_fandg, x, jac=True, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 400, 'ftol': 1e-13})

    ts = n / 2.0
    h = np.zeros(n)
    k = int(frac * n)
    h[:k] = 1.0
    rem = ts - k
    if 0 < rem < k and k < n:

    elif k < n and rem > 0:

    return h
```
