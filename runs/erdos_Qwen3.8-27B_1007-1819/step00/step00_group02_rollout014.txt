    T = T_init
    for _ in range(max_iters):
        if T < T_min:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[int(i)] = 0.0
        h[int(j)] = 1.0
        c5_new = compute_c5(h, dx)
        if c5_new > c5_cur and rng.random() >= np.exp(-(c5_new - c5_cur) / T):
            h[int(j)] = 0.0
            h[int(i)] = 1.0
        else:
            c5_cur = c5_new
        T *= 0.99997
    return h, c5_cur

def continuous_local_search(h, n, dx, target, rng, max_iters, T_init, T_min):
    h = h.copy()
    c5_cur = compute_c5(h, dx)
    T = T_init
    for _ in range(max_iters):
        if T < T_min:
            break
        j = rng.randint(n)
        delta = rng.uniform(-0.08, 0.08)
        h_new = h.copy()
        h_new[j] = np.clip(h_new[j] + delta, 0.0, 1.0)
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-10:
            k = (j + 1 + rng.randint(n - 1)) % n
            h_new[k] = np.clip(h_new[k] + diff, 0.0, 1.0)
        c5_new = compute_c5(h_new, dx)
        if c5_new <= c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T):
            h = h_new
            c5_cur = c5_new
        T *= 0.99998
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    global _dx, _target
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    best_c5 = np.inf
    best_h = None
    best_n = 0

    n_list = [44, 46, 48, 50, 52, 56, 60, 64]
    if h_init is not None:
        n_list.insert(0, len(h_init))
    n_list = sorted(set(n for n in n_list if 20 <= n <= 200))

    for n in n_list:
        if time.time() - t_start > budget_s * 0.72:
            break
        dx = 2.0 / n
        target = n / 2.0
        _dx = dx
        _target = target
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _constraint_eq, 'jac': _constraint_jac}]

        starts = []
        if h_init is not None and n == len(h_init):
            starts.append(np.clip(h_init, 0.0, 1.0).copy())
        for _ in range(8):
            h0 = np.zeros(n)
            idx = rng.choice(n, int(target), replace=False)
            h0[idx] = 1.0
            starts.append(h0)
        for _ in range(2):
            h0 = rng.random(n)
            h0 = h0 * target / max(h0.sum(), 1e-9)
            starts.append(np.clip(h0, 0.0, 1.0))

        for h0 in starts:
            if time.time() - t_start > budget_s * 0.82:
                break
            h0 = np.clip(h0, 0.0, 1.0)

            try:
                res = minimize(_objective, h0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-13})
                h_sol = project_feasible(res.x, target)
                c5 = compute_c5(h_sol, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sol.copy()
                    best_n = n
            except Exception:
                pass

            if time.time() - t_start < budget_s * 0.7:
                try:
                    res = minimize(_objective, h0, method='Nelder-Mead',
                                   options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-12})
                    h_sol = project_feasible(res.x, target)
                    c5 = compute_c5(h_sol, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_sol.copy()
                        best_n = n
                except Exception:
                    pass

    if best_h is None:
        n = 44
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_c5 = compute_c5(best_h, 2.0 / n)
        best_n = n

    # Binary local search
    if best_n % 2 == 0 and time.time() - t_start < budget_s * 0.5:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        h_bin = (best_h > 0.5).astype(float)
        h_ls, c5_ls = binary_local_search(h_bin, n, dx, target, rng, 60000, 0.008, 1e-5)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n

    # Continuous local search
    if time.time() - t_start < budget_s * 0.25:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        h_cs, c5_cs = continuous_local_search(best_h, n, dx, target, rng, 80000, 0.006, 1e-5)
        if c5_cs < best_c5:
            best_c5 = c5_cs
            best_h = h_cs.copy()
            best_n = n

    # Final binary snap + search
    if best_n % 2 == 0 and time.time() - t_start < budget_s * 0.1:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        h_bin2 = (best_h > 0.5).astype(float)
        h_ls2, c5_ls2 = binary_local_search(h_bin2, n, dx, target, rng, 40000, 0.005, 1e-6)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    return best_h, best_c5, best_n
