import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_flat(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        h = h.clip(0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = h.clip(0.0, 1.0)
    return h

def smooth_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    scaled = corr / temp
    m = np.max(scaled)
    return temp * (m + np.log(np.sum(np.exp(scaled - m))))

def num_grad(h, n, temp, eps):
    grad = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        hm = h.copy()
        hp[i] += eps
        hm[i] -= eps
        grad[i] = (smooth_obj(hp, n, temp) - smooth_obj(hm, n, temp)) / (2.0 * eps)
    return grad

def smooth_optimize(h, n, temp, max_iter, lr, time_limit):
    h = h.copy()
    eps = 1e-6
    start = time.time()
    momentum = np.zeros(n)
    beta = 0.9
    current_lr = lr
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = num_grad(h, n, temp, eps)
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = h - current_lr * momentum
        h_new = project_feasible(h_new, n)
        c5_old = compute_c5_flat(h, n)
        c5_new = compute_c5_flat(h_new, n)
        if c5_new < c5_old:
            h = h_new
        else:
            current_lr *= 0.5
            if current_lr < 1e-10:
                break
    return h

        stalled = False

            if stalled:

def make_block_initial(n, rng):
    h = np.zeros(n)
    nblocks = rng.randint(2, 5)
    for b in range(nblocks):
        start = int(b * n / nblocks)
        end = int((b + 1) * n / nblocks)
        h[start:end] = rng.uniform(0.2, 0.8)
    h = project_feasible(h, n)
    return h

def make_cos_initial(n, rng):
    freq = rng.uniform(1, 4)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.cos(2 * np.pi * freq * (x / 2.0) + phase)
    h = project_feasible(h, n)
    return h
