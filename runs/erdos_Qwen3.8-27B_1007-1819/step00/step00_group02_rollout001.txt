import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time

_G_n = 45
_G_T = 100

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def objective_lse(h):
    n = _G_n
    T = _G_T
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * (2.0 / n)
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def slsqp_refine(h, n, target_sum, T, maxiter, budget_s, start_time):

    elapsed = time.time() - start_time

                       bounds=bnds, constraints=lc,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = repair_sum(res.x, target_sum)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h, compute_c5(h)
