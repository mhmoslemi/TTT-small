    h[:target] = 1.0
    if n % 2 == 1:
        h[target] = 0.5
    rng.shuffle(h)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get initial
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    h_init = project_feasible(h_init, n_init)
    
    best_h = h_init.copy()
    best_c5 = get_c5(best_h)
    
    # Try local search from initial
    if time_left() > 30:
        h_ls, c5_ls = local_search(h_init.copy(), n_init, max_rounds=300)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls
    
    # Try PGS from initial
    if time_left() > 60:
        h_pgs, c5_pgs = pgsd_step(h_init.copy(), n_init, lr=0.03, tau=3.0, max_iter=300)
        if c5_pgs < best_c5:
            best_h, best_c5 = h_pgs, c5_pgs
        # Then local search on the result
        if time_left() > 30:
            h_ls2, c5_ls2 = local_search(h_pgs.copy(), n_init, max_rounds=200)
            if c5_ls2 < best_c5:
                best_h, best_c5 = h_ls2, c5_ls2
    
    # Try other n values
    for n_try in [46, 48, 50, 52, 54, 60, 80, 100]:
        if time_left() < 20:
            break
        
        # Random starts + local search
        n_starts = max(3, int(60 / max(1, n_try // 10)))
        for s in range(n_starts):
            if time_left() < 10:
                break
            if s == 0 and n_try == n_init:
                h_cur = h_init.copy()
            else:
                h_cur = random_feasible(n_try, rng)
            h_ls, c5_ls = local_search(h_cur, n_try, max_rounds=100)
            if c5_ls < best_c5:
                best_h, best_c5 = h_ls, c5_ls
        
        # PGS on best found for this n
        if time_left() > 40:
            h_pgs, c5_pgs = pgsd_step(best_h.copy() if len(best_h) == n_try else random_feasible(n_try, rng),
                                       n_try, lr=0.03, tau=3.0, max_iter=200)
            if c5_pgs < best_c5:
                best_h, best_c5 = h_pgs, c5_pgs
            if time_left() > 20:
                h_ls3, c5_ls3 = local_search(h_pgs.copy(), n_try, max_rounds=150)
                if c5_ls3 < best_c5:
                    best_h, best_c5 = h_ls3, c5_ls3
    
    # Final refinement on best solution
    if time_left() > 30:
        n_best = len(best_h)
        h_fin, c5_fin = local_search(best_h.copy(), n_best, max_rounds=500)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin
    
    # Final PGS polish
    if time_left() > 40:
        h_pol, c5_pol = pgsd_step(best_h.copy(), len(best_h), lr=0.01, tau=10.0, max_iter=200)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
        if time_left() > 20:
            h_fin2, c5_fin2 = local_search(h_pol.copy(), len(best_h), max_rounds=300)
            if c5_fin2 < best_c5:
                best_h, best_c5 = h_fin2, c5_fin2
    
    # Ensure final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    
    # Verify
    final_c5 = get_c5(best_h)
    if np.sum(best_h) != n_final / 2.0:
        best_h = project_feasible(best_h, n_final)
        final_c5 = get_c5(best_h)
    
    if time_left() < 0:
        pass  # We're over time, just return what we have
    
    return best_h, float(final_c5), n_final
