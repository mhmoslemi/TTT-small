def project_onto_constraint(v, n):
    """Project v onto {x: 0<=x<=1, sum(x)=n/2}"""
    x = np.clip(v, 0, 1)
    # Adjust to get sum = n/2
    target = n / 2.0
    diff = target - np.sum(x)
    if abs(diff) < 1e-12:
        return x
    if diff > 0:
        # Need to add: add to elements with most room (x < 1)
        room = 1.0 - x
        # Distribute diff according to room
        if room.sum() > 0:
            # Water filling: add to elements with most room
            order = np.argsort(-room)  # most room first
            x[order] += diff / room.sum() * room
            x = np.clip(x, 0, 1)

            if abs(diff2) > 1e-10:
                # Uniform adjustment among non-bound elements
                mask = (x > 0) & (x < 1)
                if mask.sum() > 0:

                    x = np.clip(x, 0, 1)

        space = x
        if space.sum() > 0:
            order = np.argsort(-space)  # most space first

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_constrained(v, n):
    target = n / 2.0
    lo = float(v.min()) - 2.0
    hi = float(v.max()) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            x[mask] += diff / cnt
        else:
            x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

def spsa_optimize(h0, n, t_start, t_budget, target_c5):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    rng = np.random.default_rng(int(t_start * 1e6) % 2**31)
    steps = 0
    gamma = 0.01
    delta = 0.1
    stuck = 0
    while time.time() - t_start < t_budget:
        jk = (rng.integers(0, 2, n) * 2 - 1).astype(np.float64)
        delta_t = max(delta * (1.0 / (1.0 + 0.001 * steps)) ** 0.5, 0.01)
        h_plus = project_constrained(h + delta_t * jk, n)
        h_minus = project_constrained(h - delta_t * jk, n)
        f_plus = compute_c5(h_plus)
        f_minus = compute_c5(h_minus)
        g_t = (f_plus - f_minus) / (2.0 * delta_t)
        h = project_constrained(h - gamma * g_t * jk, n)
        f = compute_c5(h)
        if f < best_c5 - 1e-8:
            best_c5 = f
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 500:
            gamma = max(gamma * 0.5, 0.001)
            delta_t = max(delta_t * 0.7, 0.01)
            stuck = 0
        steps += 1
        if best_c5 <= target_c5:
            break
    return best_h, best_c5, steps

def coordinate_refine(h, n, t_start, t_budget, target_c5):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    order = np.random.default_rng(42).permutation(n)
    it = 0
    while time.time() - t_start < t_budget:
        improved = False
        for j in order:
            if abs(compute_c5(best_h) - best_c5) > 1e-9:
                best_c5 = compute_c5(best_h)
            for trial in [best_h[j] - 0.05, best_h[j] + 0.05, best_h[j] - 0.01, best_h[j] + 0.01, 0.0, 1.0, 0.5]:
                h_trial = best_h.copy()
                h_trial[j] = trial
                h_trial = project_constrained(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-10:
                    best_h = h_trial
                    best_c5 = c5_t
                    improved = True
                    break
        it += 1
        if not improved and it > 3:
            break
        if best_c5 <= target_c5:
            break
    return best_h, best_c5

def random_start(n, rng):
    h = rng.random(n)
    return project_constrained(h, n)

def structured_start(n):
    h = np.zeros(n)
    m = n // 2
    for i in range(m):
        pos = int(round(i * (n - 1.0) / m))
        h[pos] = 1.0
    return project_constrained(h, n)

def two_interval_start(n):
    m = n // 2
    h = np.zeros(n)
    h[:m // 2] = 1.0
    h[m + m // 4 : m + m // 4 + m // 2] = 1.0
    h = np.clip(h, 0, 1)
    need = m - int(h.sum())
    for i in range(n):
        if need <= 0:
            break
        if h[i] < 1.0:
            h[i] = min(1.0, h[i] + 1.0)
            need -= 1
    return project_constrained(h, n)

    return best_h, best_c5, best_n
