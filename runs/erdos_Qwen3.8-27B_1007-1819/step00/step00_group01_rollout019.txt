    for step in range(500000):
        if elapsed() > budget_s * 0.9:

        T = max(0.0001, 0.01 * (0.999 ** (step / 1000)))

        n_pert = np.random.randint(1, 4)
        idx = np.random.choice(n, n_pert, replace=False)

        for i in idx:
            delta = np.random.uniform(-0.15, 0.15)
            trial[i] += delta

        trial = np.clip(trial, 0.0, 1.0)
        # Adjust sum
        diff = tsum - np.sum(trial)
        if abs(diff) > 1e-8:
            # Distribute uniformly to non-boundary entries
            mask = (trial > 1e-6) & (trial < 1.0 - 1e-6)
            if np.sum(mask) == 0:
                mask = (trial < 1.0 - 1e-6)
            cnt = np.sum(mask)
            if cnt > 0:
                trial[mask] += diff / cnt
                trial = np.clip(trial, 0.0, 1.0)
        
        # Verify constraint
        if abs(np.sum(trial) - tsum) > 1e-4:
            continue
        
        nc = c5(trial)
        delta_c = nc - best_c5
        if delta_c < 0 or (np.random.rand() < np.exp(-delta_c / T)):
            best_h = trial
            best_c5 = nc
    
    # Local polish: coordinate-wise
    if elapsed() < budget_s * 0.95:
        for _ in range(20):
            improved = False
            for i in range(n):
                for dv in np.linspace(-0.3, 0.3, 13):
                    nv = best_h[i] + dv
                    if nv < 0 or nv > 1:
                        continue
                    trial = best_h.copy()
                    trial[i] = nv
                    diff = tsum - np.sum(trial)
                    mask = np.arange(n) != i
                    if diff > 0:
