            r = minimize(opt_obj, best_h, args=(dx, 0.0), method='Nelder-Mead',
                         options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-11,
                                  'adaptive': True})
            h = project_h(r.x, ts)
            c = compute_c5(h, dx)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
        except Exception:
            pass

    # Phase 5: Final SLSQP
    if budget_s - (time.time() - t0) > 20:
        try:
            r = minimize(opt_obj, best_h, args=(dx, 0.0), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 5000, 'ftol': 1e-14})
            h = project_h(r.x, ts)
            c = compute_c5(h, dx)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
        except Exception:
            pass

    # Phase 6: Try other resolutions
    cur_n = len(best_h)
    for n_try in [36, 54, 72, 90, 120]:
        if time.time() - t0 > budget_s - 30:
            break
        if n_try == cur_n:
            continue
        x_old = np.linspace(0, 2, cur_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_h(h_try, n_try / 2.0)
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        cons_t = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (ts_t,)}]
        bnds_t = [(0.0, 1.0)] * n_try
        try:
            r = minimize(opt_obj, h_try, args=(dx_t, 0.0), method='SLSQP',
                         bounds=bnds_t, constraints=cons_t,
                         options={'maxiter': 3000, 'ftol': 1e-12})
            h = project_h(r.x, ts_t)
            c = compute_c5(h, dx_t)
            if c < best_c5:
                best_c5, best_h = c, h.copy()
        except Exception:
            pass

    # Final safety check
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))

    return (best_h.tolist(), best_c5, len(best_h))
