import numpy as np
from scipy.optimize import minimize
import time

_TARGET_SUM = 0.0
_DX = 1.0
_BETA = 50.0


def constraint_sum(x):
    return x.sum() - _TARGET_SUM


def obj_smooth(x):
    corr = np.correlate(x, 1.0 - x, mode='full') * _DX
    m = corr.max()
    e = np.exp(_BETA * (corr - m))
    s = e.sum()
    return m + np.log(s) / _BETA


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def fix_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(3000):
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return h


def make_structured_inits(n, target_sum):
    inits = []
    for freq in [1, 2, 3, 5, 7]:
        x = np.arange(n)
        s = 0.5 + 0.25 * np.sin(2.0 * np.pi * freq * x / n)
        s = fix_sum(s, target_sum)
        inits.append(s)
        c = 0.5 + 0.25 * np.cos(2.0 * np.pi * freq * x / n)
        c = fix_sum(c, target_sum)
        inits.append(c)
    return inits

    np.random.seed(seed)
    t0 = time.time()

    best_n = len(h_best)

    n_list = sorted(set([best_n, 50, 60, 80, 100, 120, 150, 200]))

    for target_n in n_list:
        if time.time() - t0 > budget_s - 100:
            break

        _TARGET_SUM = target_n / 2.0
        _DX = 2.0 / target_n
        bounds = [(0.0, 1.0)] * target_n
        cons = [{'type': 'eq', 'fun': constraint_sum}]

        if target_n == best_n:
            h0 = h_best.copy()
        else:

            h0 = np.interp(x_new, x_old, h_best)
        h0 = fix_sum(h0, _TARGET_SUM)

        structured = make_structured_inits(target_n, _TARGET_SUM)

        # Build list of starting points
        starts = [h0]
        starts.append(fix_sum(h0 + 0.03 * np.random.randn(target_n), _TARGET_SUM))
