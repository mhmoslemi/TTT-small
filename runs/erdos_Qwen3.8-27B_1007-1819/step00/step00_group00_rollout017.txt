import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def compute_all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s > 1e-10:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        # Fine adjustment to exact constraint
        for _ in range(10):
            diff = target - np.sum(h)
            if abs(diff) < 1e-12:
                break
            # Add/subtract from points not at boundary
            indices = np.where((h > 1e-8) & (h < 1 - 1e-8))[0]
            if len(indices) == 0:
                break
            h[indices] += diff / len(indices)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_initial(n):
    # Start with a structured pattern
    h = np.zeros(n)
    # Two blocks
    block_len = n // 4
    h[:block_len] = 1.0
    h[n//2:n//2+block_len] = 1.0
    return repair(h, n)

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old  # centers
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return repair(h_new, n_new)

def sa_optimize(h, n, budget_s, rng):
    h = h.copy()
    c5 = compute_c5(h)
    T = 0.02
    T_min = 0.0001
    cooling = 0.997
    best_local = c5
    best_local_h = h.copy()
    
    t0 = time.time()
    iters = 0
    while time.time() - t0 < budget_s and T > T_min:
        # Generate mutation
        mut_type = rng.random()
        h_new = h.copy()
        
        if mut_type < 0.3:
            # Point perturbation
            idx = rng.integers(n)
            delta = rng.uniform(-0.3, 0.3)
            h_new[idx] += delta
        elif mut_type < 0.5:
            # Block shift
            block_size = rng.integers(1, max(2, n//8))
            start = rng.integers(0, n - block_size)
            # Shift block values
            shift = rng.integers(1, max(2, n//10))
            h_new = np.roll(h_new, shift)
            h_new = np.roll(h_new, -shift, axis=0)  # undo, use circular
            # Actually, let's do a different block mutation
            h_new[start:start+block_size] += rng.uniform(-0.2, 0.2, block_size)
        elif mut_type < 0.7:
            # Swap two regions
            s1 = rng.integers(0, n//3)
            s2 = rng.integers(n//3, 2*n//3)
            length = min(5, n//6)
            h_new[s1:s1+length], h_new[s2:s2+length] = h_new[s2:s2+length].copy(), h_new[s1:s1+length].copy()
        elif mut_type < 0.85:
            # Gaussian noise on random section
            start = rng.integers(0, n - max(3, n//5))
            length = rng.integers(3, max(4, n//4))
            h_new[start:start+length] += rng.normal(0, 0.1, length)
        else:
            # Larger perturbation
            idx = rng.integers(n//3)
            h_new[idx] = rng.uniform(0, 1)
            idx2 = rng.integers(2*n//3)
            h_new[idx2] = rng.uniform(0, 1)
        
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new)
        
        # Accept
        if c5_new < c5 or rng.random() < np.exp(-(c5_new - c5) / T):
            h = h_new
            c5 = c5_new
            if c5 < best_local:
                best_local = c5
                best_local_h = h.copy()
        
        T *= cooling
        iters += 1
    
    return best_local_h

def local_refine(h, n, budget_s, rng):
    h = h.copy()
    c5 = compute_c5(h)
    
    t0 = time.time()
    while time.time() - t0 < budget_s:
        improved = False
        corrs = compute_all_corrs(h)
        max_val = np.max(corrs)
        # Find worst shifts
        worst_indices = np.where(corrs > max_val - 1e-6)[0]
        
        # Try adjusting each point
        for i in range(n):
            if time.time() - t0 > budget_s:
                return h
            for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                h_new = h.copy()
                h_new[i] += delta

                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-10:
                    h = h_new
                    c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(np.sum(h))
    if s > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        for _ in range(50):
            diff = target - float(np.sum(h))
            if abs(diff) < 1e-14:
                break
            mask = (h > 1e-10) & (h < 1.0 - 1e-10)
            cnt = int(np.sum(mask))
            if cnt == 0:
                if diff > 0:
                    mask = h < 1.0
                else:
                    mask = h > 0.0
                cnt = int(np.sum(mask))
                if cnt == 0:
                    break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def make_fallback(n):
