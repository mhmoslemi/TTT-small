import numpy as np
from scipy.optimize import minimize
import time

def project_to_constraint(h, target):
    """Project h onto {x in [0,1]^n : sum(x) = target}"""
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        t = (lo + hi) / 2.0
        s = np.sum(np.clip(h - t, 0.0, 1.0))
        if s > target:
            lo = t
        else:
            hi = t
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_penalty(h, p=4.0, lam=1000.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Smooth max
    obj = float(np.sum(np.maximum(corr, 0)**p) ** (1.0/p))
    # Constraint penalty
    constraint_viol = (np.sum(h) - n/2.0)
    obj += lam * constraint_viol**2
    return obj

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 48
        h0 = np.zeros(n)
        h0[::2] = 1.0

    h0 = project_to_constraint(h0, target_sum)

    def __call__(self, h):
        c = np.correlate(h, 1.0 - h, "full") * self.dx
        cp = np.maximum(c, 0.0) + 1e-15
        v = float(np.sum(cp ** self.p)) ** (1.0 / self.p)
        s = np.sum(h) - self.tgt
        return v + self.lam * s * s


class ObjFunc2:

    def __init__(self, n, beta, lam):

        self.beta = beta
        self.lam = lam

        self.tgt = 0.5 * n

        # log-sum-exp for numerical stability
        cm = c - np.max(c)
        v = float(np.max(c) + np.log(np.sum(np.exp(self.beta * cm))) / self.beta)
        s = np.sum(h) - self.tgt
        return v + self.lam * s * s

            lo = mid
        else:
            hi = mid

    return float(np.max(np.correlate(h, 1.0 - h, "full")) * (2.0 / n))

class ObjLp:
