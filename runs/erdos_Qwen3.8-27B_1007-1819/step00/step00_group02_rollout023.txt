```python
def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Binary search for c
    lo, hi = min(0, target - s) - 1, max(0, target - s) + 1
    for _ in range(60):
        c = (lo + hi) / 2
        h_new = np.clip(h + c, 0, 1)
        s_new = np.sum(h_new)
        if s_new < target:
            lo = c
        else:
            hi = c
    c = (lo + hi) / 2
    return np.clip(h + c, 0, 1)
```

def optimize_h(h0, max_iters, lr=None):
    n = len(h0)
    h = h0.copy()
    h = project(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    momentum = np.zeros(n)
    lr0 = 0.01
    mu = 0.9
    
    for it in range(max_iters):
        g = compute_gradient(h)
        momentum = mu * momentum + (1-mu) * g
        lr = lr0 / (1 + it * 0.01)  # simple decay
        h_new = h - lr * momentum
        h_new = project(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5
