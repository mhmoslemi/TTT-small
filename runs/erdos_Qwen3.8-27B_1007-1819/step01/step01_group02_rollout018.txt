from scipy.optimize import linprog

def solve_with_cuts(h0, n, max_iter_lp=200):
    h = h0.copy()
    cuts = []  # (grad, offset) pairs
    dx = 2.0 / n
    
    for it in range(max_iter_lp):
        # Evaluate at current point
        corr = corr_vec(h)
        c_max = np.max(corr)
        
        # Add cut at current point
        # Linearization: c_k(h_new) ≈ c_k(h) + grad_k(h)·(h_new - h)
        # We need: c_k(h_new) ≤ t for all k
        # Linearized: grad_k(h)·h_new ≤ t - c_k(h) + grad_k(h)·h
        # But we only need this for k near the max (or all k for safety)
        
        # For efficiency, add cuts for all k but weight by proximity to max
        active = np.where(corr > c_max - 0.01 * c_max)[0]
        for ki in active:
            # Compute gradient of c_ki at h
            s = ki - (n - 1)  # shift value
            g = np.zeros(n)
            js = max(0, s)
            je = min(n, n + s)
            if js < je:
                g[js:je] += (1.0 - h[js-s:je-s])
            js2 = max(0, -s)
            je2 = min(n, n - s)
            if js2 < je2:
                g[js2:je2] -= h[js2+s:je2+s]
            offset = corr[ki] - g @ h  # c_k(h) - grad·h
            cuts.append((g * dx, offset * dx))
        
        # Solve LP
        # Variables: h[0], h[1], ..., h[n-1], t
        # Minimize t
        # Subject to: g_k·h - t ≤ -offset_k for each cut
        #             sum(h) = n/2
        #             0 ≤ h ≤ 1
        #             t ≥ 0
        
        m_cuts = len(cuts)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        A_ub = np.zeros((m_cuts + n, n + 1))
        b_ub = np.zeros(m_cuts + n)
        for i, (g, off) in enumerate(cuts):
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0
            b_ub[i] = -off
        
        # Upper bounds on h: h[i] ≤ 1 → handled by bounds
        # Lower bounds: 0 ≤ h[i] → handled by bounds
        # t ≥ 0 → t ≥ -M (large M as lower bound)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h = res.x[:n]
            t_lp = res.x[n]
        else:
            break
        
        new_corr = corr_vec(h)
        new_max = np.max(new_corr) * dx
        if new_max <= t_lp + 1e-6:
            break  # Optimal!
    
    return h, compute_c5(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    g = np.zeros(n)
    for s in range(-(n - 1), n):
        wi = w[s + n - 1]
        if wi < 1e-14:
            continue
        a1, b1 = max(0, s), min(n, n + s)
        if a1 < b1:
            g[a1:b1] += wi * (1.0 - h[a1 - s:b1 - s])
        a2, b2 = max(0, -s), min(n, n - s)
        if a2 < b2:
            g[a2:b2] -= wi * h[a2 + s:b2 + s]
    return g * dx

def obj_pen(h, n, beta, pen):

    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * (2.0 / n)
    s = h.sum() - n / 2.0
    return val + pen * s * s

    s = h.sum() - len(h) / 2.0

    g = grad_smooth_max(h, beta)
