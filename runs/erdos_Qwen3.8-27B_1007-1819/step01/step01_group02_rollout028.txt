```python
def frank_wolfe_step(h, n, dx, gamma):
    # Compute gradient of f(h) = max_l C_l(h) * dx
    corr = np.correlate(h, 1.0 - h, mode="full")
    l_star = np.argmax(corr)
    
    # Subgradient: for the correlation at lag l_star (which corresponds to shift s = l_star - (n-1))
    s = l_star - (n - 1)  # actual shift
    # ∂C_s/∂h[i] = (1 - h[i+s]) for valid i (where i+s is in [0,n-1])
    grad = np.zeros(n)
    if s >= 0:
        # i ranges from 0 to n-1-s
        grad[0:n-s] = (1.0 - h[s:n])
    else:
        # i ranges from -s to n-1
        grad[-s:] = (1.0 - h[0:n+s])
    
    grad *= dx
    
    # Frank-Wolfe: find y in P that minimizes g^T y
    # P = {y: 0≤y≤1, sum(y)=n/2}
    # Optimal: y[i] = 1 for n/2 smallest g[i], 0 otherwise
    y = np.zeros(n)
    indices = np.argsort(grad)
    y[indices[:n//2]] = 1.0
    
    h_new = h + gamma * (y - h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new
```

```python
    scaled = beta * corr * dx
    scaled -= np.max(scaled)  # for numerical stability
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Sum of weighted gradients
    grad = np.zeros(n)
    for l in range(len(corr)):
        if weights[l] < 1e-10:
            continue
        s = l - (n - 1)
        if s >= 0:
            grad[0:n-s] += weights[l] * (1.0 - h[s:n])
        else:
            grad[-s:] += weights[l] * (1.0 - h[0:n+s])
    
    return grad * dx, np.max(corr) * dx
```

```python
        h_alt = project_feasible(interp_to_size(best_h, best_n, n_alt), n_alt)
        gh, gc = grad_descent(h_alt, n_alt, 0.02, t0, budget_s, betas=(50, 100))
        if gc < best_c5:
            best_c5, best_h, best_n = gc, gh, n_alt

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
```
