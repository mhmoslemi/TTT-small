from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj_grad(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exps = np.exp(shifted - shifted.max())  # numerical stability
    w = exps / exps.sum()
    val = (m + np.log(exps.sum()) / alpha) * dx
    
    # Gradient
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w_s = w[s + n - 1]
        if w_s < 1e-15:
            continue
        if s == 0:
            grad += w_s * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += w_s * (1.0 - h[:n-s])
            grad[:n-s] -= w_s * h[s:]
        else:
            t = -s
            grad[:n-t] += w_s * (1.0 - h[t:])
            grad[t:] -= w_s * h[:n-t]
    return val, grad * dx

def gradient_descent(h0, n, alpha, lr=0.01, max_iter=2000, tol=1e-14):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):
        val, grad = smooth_obj_grad(h, n, alpha)
        # Project gradient onto feasible space (orthogonal to all-ones)
        grad_proj = grad - grad.mean()
        h_new = h - lr * grad_proj
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < best_val:
            best_val = c
            best_h = h_new.copy()
        if abs(val - compute_c5(h)) < tol or np.max(np.abs(grad)) < 1e-10:
            break
        # Adaptive learning rate
        if c > val * 1.01:
            lr *= 0.5
        else:
            lr = min(lr * 1.05, 0.05)
        h = h_new
    return best_h, best_val

                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return project_feasible(res.x, n)
    except:
        return h

def local_search(h0, n, max_rounds=300, n_trials=5):

    curr = compute_c5(h)

        h_t, c = pair_swap_search(best_h, best_n, n_rounds=150, n_pairs=12)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 5: Simulated annealing
    if tleft() > 30:
        sa_budget = min(tleft() - 10, 200)
        h_t, c = gradient_informed_sa(best_h, best_n, sa_budget)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 6: Final polish - SLSQP + pair search
    if tleft() > 25:
        h_t = slsqp_solve(best_h, best_n, 800, maxiter=200)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
        if tleft() > 15:
            h_t, c = pair_swap_search(best_h, best_n, n_rounds=80, n_pairs=10)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 7: One more gradient descent + SLSQP cycle
    if tleft() > 20:
        h_t, c = grad_descent_solve(best_h, best_n, 200, max_iter=1000)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
            h_t = slsqp_solve(best_h, best_n, 500, maxiter=150)
            c = compute_c5(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
