        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    # Tiny fix
    diff = target - h.sum()

            i = int(np.argmin(h))
        else:
            i = int(np.argmax(h))
        h[i] = np.clip(h[i] + diff, 0, 1)
    return h

def proj_feasible(x):
    n = len(x)
    target = n / 2.0
    lo, hi = -x.max() - 0.5, 1.0 - x.min() + 0.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(x + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    sh = np.clip(beta * (corr - mx), -30, 30)
    return (mx + np.log(np.sum(np.exp(sh))) / beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    sh = np.clip(beta * (corr - mx), -30, 30)
    mask = sh > -25
    if not np.any(mask):
        idx = np.argmax(corr)
        mask = np.zeros_like(corr)
        mask[idx] = 1
    w = np.where(mask, np.exp(sh), 0.0)
    ws = w.sum()
    if ws < 1e-300:
        idx = np.argmax(corr)
        w = np.zeros_like(corr)
        w[idx] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    active = np.where(w > 1e-15)[0]
    for k in active:
        wk = w[k]
        off = k - (n - 1)
        lo = max(0, -off)
        hi = min(n - 1, n - 1 - off)
        idx = np.arange(lo, hi + 1)
        gidx = idx + off
        grad[idx] += wk * (1.0 - h[gidx])
        jlo = max(0, lo + off)
        jhi = min(n - 1, hi + off)
        if jlo <= jhi:
            jidx = np.arange(jlo, jhi + 1)
            grad[jidx] -= wk * h[jidx - off]
    return dx * grad

def pgd(h, beta, max_iter, step0):
    h = h.copy()
    step = step0
    best = compute_c5(h)
    best_h = h.copy()
    stall = 0
    for _ in range(max_iter):
        g = smooth_max_grad(h, beta)
        h_new = proj_feasible(h - step * g)
        c = compute_c5(h_new)
        if c < best - 1e-15:
            best = c
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.03, 0.5)
            stall = 0
        else:
            step *= 0.6
            if step < 1e-13 or stall > 80:
                break
            stall += 1
    return best_h, best

def coord_refine(h, passes=5, dsmall=0.002, dbig=0.01):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    deltas = [dsmall, -dsmall, dbig, -dbig, 0.5*dsmall, -0.5*dsmall]
    for _ in range(passes):
        imp = False
        for i in range(n):
            for d in deltas:
                v = h[i] + d
                if v < -1e-10 or v > 1 + 1e-10:
                    continue
                h[i] = max(0.0, min(1.0, v))
                h = fix_feasible(h)
                o = compute_c5(h)
                if o < cur - 1e-15:
                    cur = o
                    imp = True
                else:
                    h[i] = max(0.0, min(1.0, h[i] - d))
                    h = fix_feasible(h)
        if not imp:
            break
    return h, cur

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c ^= 1
            x >>= 1
        t[i] = c
    return t
