        r = minimize(smooth_obj_and_grad, h0, args=(n, alpha), method='SLSQP',
                     jac=True, bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(r.x, n)
    except Exception:
        return h0


def block_sa(h0, n, budget_s):
    h = h0.copy()
    curr = compute_c5(h)
    best = h.copy()
    best_val = curr
    t0 = time.time()
    while time.time() - t0 < budget_s:
        blen = int(np.random.choice([2, 3, 5, 8, 12], p=[0.25, 0.25, 0.2, 0.2, 0.1]))
        start = np.random.randint(0, max(1, n - blen + 1))
        h_new = h.copy()
        h_new[start:start + blen] += np.random.randn(blen) * 0.05
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        frac = (time.time() - t0) / budget_s
        T = 0.004 * max(0.0, 1.0 - frac) ** 2
        if c_new < curr or (T > 1e-15 and np.random.rand() < np.exp(-(c_new - curr) / T)):
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best = h.copy()
    return best, best_val


def coord_polish(h, n, rounds):
    h = h.copy()
    curr = compute_c5(h)
    grid = np.linspace(0.0, 1.0, 25)
    for _ in range(rounds):
        improved = False
        for i in range(n):
            orig = h[i]
            bv = orig
            bc = curr
            for v in grid:
                if abs(v - orig) < 3e-3:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < bc:
                    bc = c
                    bv = h[i]
                    improved = True
            h[i] = bv
            h = project_feasible(h, n)
            curr = bc
        if not improved:
            break
    return h, compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    try:
        init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        init = np.full(80, 0.5)
    n = len(init)
    init = project_feasible(init, n)

    best_h = init.copy()
    best_c5 = compute_c5(init)
    best_n = n

    # Phase 1: SLSQP with analytic gradient, increasing alpha
    for alpha in [80, 250, 800, 2000]:
        if time_left_fn(t0, budget_s) < 20:
            break
        ht = slsqp_opt(best_h, best_n, alpha, 500)
        c = compute_c5(ht)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 2: Random restart + SLSQP
    if time_left_fn(t0, budget_s) > 60:
        h_rand = project_feasible(np.clip(0.5 + 0.3 * np.random.randn(best_n), 0, 1), best_n)
        ht = slsqp_opt(h_rand, best_n, 300, 400)
        c = compute_c5(ht)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 3: Block simulated annealing
    if time_left_fn(t0, budget_s) > 80:
        sb = min(100, time_left_fn(t0, budget_s) - 20)
        ht, c = block_sa(best_h, best_n, sb)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 4: SLSQP refinement post-SA
    if time_left_fn(t0, budget_s) > 40:
        ht = slsqp_opt(best_h, best_n, 1500, 500)
        c = compute_c5(ht)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 5: Try nearby resolutions
    for dn in [-3, -2, -1, 1, 2, 3, 5, -5]:
        if time_left_fn(t0, budget_s) < 35:
            break
        ns = best_n + dn
        if ns < 20:
            continue
        h_int = np.interp(np.linspace(0, 1, ns), np.linspace(0, 1, best_n), best_h)
        h_int = project_feasible(h_int, ns)
        ht = slsqp_opt(h_int, ns, 400, 350)
        c = compute_c5(ht)
        if c < best_c5:
            best_c5, best_h, best_n = c, ht.copy(), ns

    # Phase 6: Another SA round
    if time_left_fn(t0, budget_s) > 50:
        ht, c = block_sa(best_h, best_n, min(40, time_left_fn(t0, budget_s) - 10))
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 7: High-alpha SLSQP
    if time_left_fn(t0, budget_s) > 30:
        ht = slsqp_opt(best_h, best_n, 3000, 400)
        c = compute_c5(ht)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 8: Coordinate polish
    if time_left_fn(t0, budget_s) > 18:
        ht, c = coord_polish(best_h, best_n, 40)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    # Phase 9: Final SLSQP squeeze
    if time_left_fn(t0, budget_s) > 12:
        ht = slsqp_opt(best_h, best_n, 5000, 200)
        c = compute_c5(ht)
        if c < best_c5:
            best_c5, best_h = c, ht.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
