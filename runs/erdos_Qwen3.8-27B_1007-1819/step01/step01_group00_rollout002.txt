    """Gradient of corr[k_star] with respect to h"""
    n = len(h)
    g = np.zeros(n)
    # corr[k*] = sum_{i=0}^{n-1-k*} h[i] * (1-h[i+k*])

    for i in range(k_star, n):
        g[i] -= h[i - k_star]
    return g

    target = n / 2.0
    for _ in range(4*n):
        r = target - np.sum(h)
        if abs(r) < 1e-12:
            break
        if r > 0:
            # Distribute r among elements not at 1
            mask = h < 1.0
            if not np.any(mask):
                break
            add = r / np.sum(mask)
            h[mask] = np.minimum(1.0, h[mask] + add)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            sub = (-r) / np.sum(mask)

        k_star = int(np.argmax(corr)) + n - 1  # offset in 'full' mode
        # corr array has indices 0 to 2n-2
        # corr[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} h[i] * (1-h[i+k-(n-1)])
        # Actually let me just compute the gradient directly

        # result[m] = sum_{i} h[i] * (1-h)[i + n - 1 - m]

        # result[n-1-k] = sum_{i: 0<=i<n, 0<=i+k<n} h[i]*(1-h[i+k])

        for i in valid_range:
            j = i + k_shift
            g[i] += (1.0 - h[j])
            g[j] -= h[i]

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(6 * n):
        r = target - np.sum(h)
        if abs(r) < 1e-13:
            break
        if r > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] = np.minimum(1.0, h[mask] + r / np.sum(mask))
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] = np.maximum(0.0, h[mask] - (-r) / np.sum(mask))
    return h

def smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return (mx + np.log(s) / beta) * dx

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

    return np.ones(len(x))

    t = np.zeros(n)

    """Gradient of corr[m_star] w.r.t. h, where corr = np.correlate(h,1-h,'full')"""
    n = len(h)
    k_shift = n - 1 - m_star
    g = np.zeros(n)
    lo = max(0, -k_shift)
    hi = min(n, n - k_shift)
    for i in range(lo, hi):
        j = i + k_shift
        g[i] += (1.0 - h[j])
        g[j] -= h[i]
    return g

    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(max_steps):
        corr = get_corr(h)
        m_star = int(np.argmax(corr))
        g = grad_corr_at(h, m_star)
        h_new = fix_feasible(np.clip(h - lr * g, 0, 1))
        new_val = compute_c5(h_new)
        if new_val < cur - tol:
            h = h_new
            cur = new_val
        else:
            lr *= 0.6
            if lr < 1e-8:
                break
            h_new = fix_feasible(np.clip(h - lr * g, 0, 1))
            new_val = compute_c5(h_new)
            if new_val < cur - tol:
                h = h_new
                cur = new_val
    return h, cur

def coord_fine(h, steps=None, max_rounds=25):
    if steps is None:
        steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    for ss in steps:
        for _ in range(max_rounds):
            improved = False
            for i in range(n):
                old = h[i]
                for d in (ss, -ss):
                    v = old + d
                    if v < 0 or v > 1:
                        continue
                    h[i] = v
                    hf = fix_feasible(h)
                    obj = compute_c5(hf)
                    if obj < cur - 1e-12:
                        cur = obj
                        h = hf
                        improved = True
                        break
                    else:
                        h[i] = old
            if not improved:
                break
    return h, cur

    n = len(h)

        return smooth_max(fix_feasible(x), beta)

                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return fix_feasible(res.x)
    except Exception:
        return h
