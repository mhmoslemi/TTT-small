def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        cnt = np.count_nonzero(m)
        if cnt == 0:
            break
        h[m] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        cnt = int(np.count_nonzero(m))
        if cnt == 0:
            break
        h[m] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400, 0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def obj_smooth(p, n, alpha):
    h = project_feasible(p, n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(c, alpha) * dx

def obj_exact(p, n):
    return compute_c5(project_feasible(p, n))

def slsqp_opt(h0, n, alpha, mit):
    try:
        r = minimize(obj_smooth, h0.copy(), args=(n, alpha),
                     method='SLSQP', options={'maxiter': mit, 'ftol': 1e-15})
        return project_feasible(r.x, n)
    except Exception:
        return h0.copy()

def nm_opt(h0, n, mit):
    try:
        r = minimize(obj_exact, h0.copy(), args=(n,),
                     method='Nelder-Mead',
                     options={'maxiter': mit, 'xatol': 1e-10, 'fatol': 1e-14,
                              'adaptive': True})
        return project_feasible(r.x, n)
    except Exception:
        return h0.copy()

def coord_hd(h, n, rounds, gn):
    h = h.copy()
    best = compute_c5(h)
    grid = np.linspace(0.0, 1.0, gn)
    for _ in range(rounds):
        imp = False
        for i in range(n):
            orig = h[i]
            bc = best
            bv = orig
            for gv in grid:
                if abs(gv - orig) < 1e-6:
                    continue
                h[i] = float(gv)
                h2 = project_feasible(h, n)
                c = compute_c5(h2)
                if c < bc:
                    bc = c
                    bv = float(h2[i])
            if bc < best - 1e-15:
                imp = True
                h[i] = bv
                h = project_feasible(h, n)
                best = bc
            else:
                h[i] = orig
        if not imp:
            break
    return h, best

def sa_impl(h0, n, dur):
    h = h0.copy()
    cur = compute_c5(h)
    bh = h.copy()
    bc = cur
    te = time.time() + dur
    T0 = 0.012
    stp = 0
    est = max(int(dur / 0.003), 100)
    while time.time() < te:
        stp += 1
        np_ = np.random.choice([1, 2, 3, 4, 5, 6],
                               p=[0.3, 0.25, 0.2, 0.12, 0.08, 0.05])
        ix = np.random.choice(n, size=np_, replace=False)
        hn = h.copy()
        hn[ix] = np.clip(hn[ix] + np.random.randn(np_) * 0.18, 0.0, 1.0)
        hn = project_feasible(hn, n)
        cn = compute_c5(hn)
        fr = stp / est
        T = T0 * max((1.0 - fr) ** 2, 0.001)
        if cn < cur or np.random.rand() < np.exp(-(cn - cur) / T):
            h = hn
            cur = cn
            if cur < bc:
                bc = cur
                bh = h.copy()
    return bh, bc

def interp_h(hs, ns, nd):
    xs = np.linspace(0.0, 2.0, ns)
    xd = np.linspace(0.0, 2.0, nd)
    return project_feasible(np.interp(xd, xs, hs), nd)

def random_h(n, s):
    rng = np.random.RandomState(s)
    return project_feasible(np.clip(0.5 + 0.3 * rng.randn(n), 0.0, 1.0), n)
