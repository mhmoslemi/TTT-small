import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    r = target - np.sum(h)
    if abs(r) < 1e-12:
        return h
    # Distribute the correction
    h += r / n
    h = np.clip(h, 0.0, 1.0)
    # Iterate to fix
    for _ in range(200):
        r = target - np.sum(h)
        if abs(r) < 1e-12:
            break
        if r > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += r / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += r / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def project_constraint(z, n):
    """Project z onto {h: 0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Find tau such that sum(clamp(z_i - tau, 0, 1)) = target
    # clamp(z_i - tau, 0, 1) = 0 if z_i <= tau, z_i - tau if tau < z_i < tau+1, 1 if z_i >= tau+1
    # By sorting, we can find tau by bisection
    lo, hi = z.min() - 1, z.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(z - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(z - tau, 0, 1)

def grad_corr(h):
    """Compute gradients of all corr[k] values with respect to h. Returns (2n-1, n) array."""
    n = len(h)
    grads = np.zeros((2*n-1, n))
    one_h = 1.0 - h
    for lag in range(n):  # lag 0 to n-1
        if lag == 0:
            # corr[0] = sum h[i]*(1-h[i])
            # grad[i] = 1 - 2*h[i]
            grads[n-1] = 1.0 - 2.0*h
        else:
            # Positive lag: corr = sum_{m=0}^{n-1-lag} h[m+lag]*(1-h[m])
            # grad[i] = (1-h[i-lag])*[lag<=i<=n-1] - h[i+lag]*[0<=i<=n-1-lag]
            idx = n - 1 + lag  # index in full correlation
            # positive lag
            i_pos = np.arange(lag, n)
            grads[idx, i_pos] += one_h[i_pos - lag]
            i_neg = np.arange(0, n - lag)
            grads[idx, i_neg] -= h[i_neg + lag]
            # negative lag (by symmetry of the problem structure but NOT the same value)
            # corr at lag -lag: sum_{m=lag}^{n-1} h[m-lag]*(1-h[m]) = sum_{m=0}^{n-1-lag} h[m]*(1-h[m+lag])
            # grad[i] = (1-h[i+lag])*[0<=i<=n-1-lag] - h[i-lag]*[lag<=i<=n-1]
            idx_neg = n - 1 - lag
            i_pos2 = np.arange(0, n - lag)
            grads[idx_neg, i_pos2] += one_h[i_pos2 + lag]
            i_neg2 = np.arange(lag, n)
            grads[idx_neg, i_neg2] -= h[i_neg2 - lag]
    return grads

def smooth_max_and_grad(h, beta):
    """Compute smooth max of corr*dx and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Log-sum-exp for numerical stability
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    s = np.sum(exps)
    p = exps / s  # softmax weights
    val = (m + np.log(s) / beta)
    # Gradient: p @ grads * dx... wait
    # val = (1/beta)*log(sum exp(beta*corr_k * dx))
    # dval/dh = sum_k p_k * dx * d(corr_k)/dh
    grads = grad_corr(h)  # (2n-1, n)
    grad = (p * dx) @ grads  # weighted sum of gradients, weighted by p_k * dx
    return val, grad

For the coordinate descent, instead of trying 8 different deltas per coordinate (which is O(8*n) evaluations per sweep, each O(n) for correlation = O(8*n²) per sweep), I can:

- This involves finding the value of h[i] that equalizes the top correlations

Actually, for coordinate descent on the exact objective:
- Fix all h[j] for j≠i
- The objective as a function of h[i] is: max_k (a_k + b_k * h[i]) * dx
  where a_k, b_k depend on the fixed coordinates.
- This is a max of linear functions of h[i], so the minimum is where the top two lines cross, or at a boundary.

This is actually very efficient! For each coordinate:
1. Compute the current correlations (all 2n-1 values)
2. For coordinate i, compute how each correlation changes when h[i] → t:
   corr_k(t) = corr_k_current + delta_k * (t - h[i])
   where delta_k is the partial derivative of corr_k w.r.t. h[i] (which we already have from the gradient)
3. The optimal t minimizes max_k (corr_k_current + delta_k * (t - h[i]))
   This is the smallest t (or largest, depending) where the max is minimized.
   It's at the intersection of the two currently-largest lines, or at a boundary [0,1].

def coord_descent_exact(h, max_rounds=50):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for rnd in range(max_rounds):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode="full")  # (2n-1,)
        for i in range(n):
            # Compute delta_k for each lag
            deltas = np.zeros(2*n-1)
            one_h = 1.0 - h
            for lag in range(n):
                if lag == 0:
                    # grad of corr[0] w.r.t. h[i] = 1 - 2*h[i]
                    deltas[n-1] += (1.0 - 2.0*h[i])
                else:

                    if lag <= i <= n-1:
                        deltas[n-1+lag] += one_h[i-lag]
                    if 0 <= i <= n-1-lag:
                        deltas[n-1+lag] -= h[i+lag]
                    # Negative lag
                    if 0 <= i <= n-1-lag:
                        deltas[n-1-lag] += one_h[i+lag]
                    if lag <= i <= n-1:
                        deltas[n-1-lag] -= h[i-lag]

            # = (corr[k] - deltas[k]*h[i]) + deltas[k]*t

            a = corr - deltas * h[i]
            b = deltas

            # Or at a boundary t=0 or t=1

            # Better: binary search on t

            lo, hi = 0.0, 1.0
            for _ in range(40):

                f_mid = np.max(a + b * mid)
                f_hi = np.max(a + b * hi)
                # Check subgradient at mid

                # If all active b's are positive, f is increasing at mid, go left
                # If all active b's are negative, f is decreasing, go right
                active = np.where(a + b * mid == np.max(a + b * mid))[0]
                # Actually use tolerance
                fmax = np.max(a + b * mid)
                active = np.where(a + b * mid >= fmax - 1e-12)[0]

                    active = np.array([np.argmax(a + b * mid)])
                if np.all(b[active] > 0):
                    hi = mid
                elif np.all(b[active] < 0):
                    lo = mid
                else:
                    # Minimum is between lo and hi (or exactly at mid if 0 in subgradient)
                    if 0 in b[active] or (b[active].min() < 0 < b[active].max()):
                        break
                    # Otherwise, the active set changes; just narrow
                    if f_mid < f_lo:
                        hi = mid
                    else:
                        lo = mid
            t_opt = (lo + hi) / 2
            # But we also need to maintain the sum constraint!
            # If we change h[i] from h[i] to t, we need to adjust another coordinate
            # This complicates things. Let's just try the simple version first (no constraint)
            # and then fix feasibility
            h[i] = t_opt
        # Fix feasibility after each round
        h = fix_feasible(h)
        # Check if improved
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(100):
        r = target - np.sum(h)
        if abs(r) < 1e-14:
            break
        eps = 1e-12
        if r > 0:
            mask = h < 1.0 - eps
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.minimum(1.0, h[mask] + r / cnt)
        else:
            mask = h > eps
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.maximum(0.0, h[mask] + r / cnt)
    return h

def project_box_sum(z, n):
    """Project z onto {h: 0<=h_i<=1, sum(h)=n/2} via bisection on tau."""
    target = n / 2.0
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(z - tau, 0.0, 1.0)

def smooth_max_grad(h, beta):
    """Return (S_beta, gradient) of smooth max of corr*dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    g = corr * dx  # (2n-1,)
    m = g.max()
    exps = np.exp(beta * (g - m))
    s = exps.sum()
    p = exps / s  # (2n-1,) softmax weights
    val = m + np.log(s) / beta
    # Gradient: sum_k p_k * dx * grad(corr_k)
    # grad(corr_k) for each k can be computed vectorially
    # Let's compute it as: for each i, grad_i = sum_k p_k * dx * (partial corr_k / partial h_i)
    # partial corr_k / partial h_i:
    #   For lag k (where k = idx - (n-1), idx in 0..2n-2):
    #   If k > 0: (1-h[i-k])*[k<=i<n] - h[i+k]*[0<=i<n-k]
    #   If k < 0: (1-h[i+k])*[0<=i<n-|k|] - h[i-|k|]*[|k|<=i<n]  (where |k|=-k)
    #   If k = 0: 1 - 2*h[i]
    
    # Vectorized computation:
    grad = np.zeros(n)
    # k = 0 (idx = n-1)
    grad += p[n-1] * dx * (1.0 - 2.0 * h)
    for lag in range(1, n):
        # Positive lag k=lag (idx = n-1+lag)
        w_pos = p[n-1+lag] * dx
        # Negative lag k=-lag (idx = n-1-lag)
        w_neg = p[n-1-lag] * dx
        # Positive lag grad: (1-h[i-lag])*[lag<=i<n] - h[i+lag]*[0<=i<n-lag]
        if lag < n:
            grad[lag:] += w_pos * (1.0 - h[:n-lag])
            grad[:n-lag] -= w_pos * h[lag:]
        # Negative lag grad: (1-h[i+lag])*[0<=i<n-lag] - h[i-lag]*[lag<=i<n]
        grad[:n-lag] += w_neg * (1.0 - h[lag:])
        grad[lag:] -= w_neg * h[:n-lag]
    
    return val, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(200):
        r = target - h.sum()
        if abs(r) < 1e-13:
            break
        if r > 0:
            m = h < 1.0
            if not m.any(): break
            h[m] = np.minimum(1.0, h[m] + r / m.sum())
        else:
            m = h > 0.0
            if not m.any(): break
            h[m] = np.maximum(0.0, h[m] + r / m.sum())
    return h

def project_box_sum(z, n):
    target = n * 0.5
    if z.sum() == target and z.min() >= 0 and z.max() <= 1:
        return z.copy()
    lo, hi = z.min() - 1.0, z.max() + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - 0.5*(lo+hi), 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n

    g = corr * dx
    m = g.max()
    e = np.exp(beta * (g - m))
    s = e.sum()
    p = e / s
    val = m + np.log(s) / beta
    # Gradient computation
    grad = np.zeros(n)
    grad += p[n-1] * dx * (1.0 - 2.0*h)
    for lag in range(1, n):
        wp = p[n-1+lag] * dx
        wn = p[n-1-lag] * dx

        grad[:n-lag] -= wp * h[lag:]
        grad[:n-lag] += wn * (1.0 - h[lag:])
