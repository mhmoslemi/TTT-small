        h, c5 = coord_descent(best_h, max_rounds=80, fine=True)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Binarize + refine
    if elapsed() < budget_s * 0.8:
        hb, cb = binarize_refine(best_h, max_flips=800, rng=rng)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
            h_cd, c5c = coord_descent(best_h, max_rounds=40, fine=True)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = h_cd.copy()

    # Phase 6: Multiple perturbation-reoptimize cycles
    cycle = 0
    while elapsed() < budget_s * 0.93 and cycle < 8:
        start_h = best_h + 0.02 * rng.randn(n)
        start_h = fix_feasible(start_h)
        _beta_val = 200
        try:
            res = minimize(_soft_obj, start_h, method='SLSQP',
                          constraints=[{'type': 'eq', 'fun': sum_eq}],
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-12})
            h_new = fix_feasible(res.x)
            h_new, c5n = coord_descent(h_new, max_rounds=40, fine=True)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()
        except Exception:
            pass
        cycle += 1

    # Phase 7: Final fine-tuning
    if elapsed() < budget_s * 0.97:
        h, c5 = coord_descent(best_h, max_rounds=80, fine=True)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if elapsed() < budget_s * 0.98:
            h, c5 = pair_swap(best_h, max_rounds=20, rng=rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            h, c5 = coord_descent(best_h, max_rounds=40, fine=True)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    best_h = fix_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
