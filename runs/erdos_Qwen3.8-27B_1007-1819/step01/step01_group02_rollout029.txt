from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_lags(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_lag(h, m, dx):
    n = len(h)
    g = np.zeros(n)
    lo1, hi1 = max(0, m), min(n-1, n-1+m)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1+1)
        g[idx] += 1.0 - h[idx - m]
    lo2, hi2 = max(0, -m), min(n-1, n-1-m)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2+1)
        g[idx] -= h[idx + m]
    return g * dx

    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)

    n = 180
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    except NameError:
        h_init = np.full(90, 0.5)
        n = len(h_init)

    h_start = project_feasible(h_init.copy(), n)
    dx = 2.0 / n
    best_c5 = compute_c5(h_start)
    best_h = h_start.copy()

    # Phase 1: Subgradient exploration (45% budget)
    p1_end = t0 + budget_s * 0.45
    h_sub, c5_sub = subgrad_phase(h_start, n, dx, rng, p1_end, t0)
    if c5_sub < best_c5:
        best_c5 = c5_sub
        best_h = h_sub.copy()

    # Phase 2: SLSQP refinement (40% budget)
    p2_end = t0 + budget_s * 0.85
    h_slq, c5_slq = slsqp_phase(best_h, n, p2_end, t0)
    if c5_slq < best_c5:
        best_c5 = c5_slq
        best_h = h_slq.copy()

    # Phase 3: Coordinate descent polish (10% budget)
    p3_end = t0 + budget_s * 0.97
    h_cd, c5_cd = coord_descent(best_h, n, p3_end, t0)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 4: One more SLSQP round if time allows and improvement was made
    if time.time() < t0 + budget_s * 0.98 and c5_cd < c5_slq - 1e-6:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
        res = minimize(sm_obj, best_h, args=(n, 1000),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-16})
        if res.x is not None:
            h_opt = project_feasible(res.x, n)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
