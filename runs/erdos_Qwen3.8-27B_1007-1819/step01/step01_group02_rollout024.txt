import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj_lbfgs(x, n, beta, mu):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    penalty = mu * (h.sum() - n / 2.0) ** 2
    return sm * dx + penalty


def smooth_obj_slsqp(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx


def topk_obj(x, n, k, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argpartition(corr, -k)[-k:]
    vals = corr[idx]
    m = vals.max()
    return (m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta) * dx


def sum_con(x, n):
    return x.sum() - n / 2.0


def make_bounds(n):
    return [(0.0, 1.0)] * n


def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.arange(n_from) * 2.0 / n_from
    x_new = np.arange(n_to) * 2.0 / n_to
    return np.interp(x_new, x_old, h)


def binary_threshold(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n_ones]] = 1.0
    return hb


def binary_swap_search(h_bin, n, t0, budget_s, frac, max_rounds=200):
    c5_curr = compute_c5(h_bin)
    best_c5 = c5_curr
    h_curr = h_bin.copy()
    n_ones = n // 2
    for rnd in range(max_rounds):
        if time.time() - t0 > budget_s * frac:
            break
        one_idx = np.where(h_curr == 1)[0]
        zero_idx = np.where(h_curr == 0)[0]
        improved = False
        for a in one_idx:
            if time.time() - t0 > budget_s * frac:
                break
            for b in zero_idx:
                if abs(a - b) < 2:
                    continue
                h_try = h_curr.copy()
                h_try[a] = 0.0
                h_try[b] = 1.0
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    h_curr = h_try
                    one_idx = np.where(h_curr == 1)[0]
                    zero_idx = np.where(h_curr == 0)[0]
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h_curr, best_c5


def coord_descend(h, n, t0, budget_s, frac, max_epochs=6):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    for ep in range(max_epochs):
        if time.time() - t0 > budget_s * frac:
            break
        improved = False
        steps = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]
        for i in range(n):
            if time.time() - t0 > budget_s * frac:
                break
            for d in steps:
                h_try = h_curr.copy()
                h_try[i] = h_try[i] + d
                if h_try[i] < 0 or h_try[i] > 1:
                    continue
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    h_curr = h_try.copy()
                    improved = True
                    break
                else:
                    h_try[i] = h_curr[i]
        if not improved:
            break
    return h_curr, best_c5
