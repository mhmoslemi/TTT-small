import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def interp_h(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)


def smooth_obj(x, n, beta, pen_coef):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    smax = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    pen = pen_coef * (h.sum() - n / 2.0) ** 2
    return (smax + pen) * (2.0 / n)


def binarize_with_sum(h, n, rng):
    h_b = (h > 0.5).astype(float)
    nt = n // 2
    while int(h_b.sum()) > nt:
        o = np.where(h_b == 1)[0]
        h_b[rng.choice(o)] = 0
    while int(h_b.sum()) < nt:
        z = np.where(h_b == 0)[0]
        h_b[rng.choice(z)] = 1
    return h_b


def binary_ls(h_init, n, t_limit, t0, rng, max_rounds=3000):
    h = h_init.astype(float).copy()
    nt = n // 2
    correct_sum(h, nt, rng)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr)) * (2.0 / n)

    for _ in range(max_rounds):
        if time.time() - t0 > t_limit:
            break
        z = np.where(h == 0)[0]
        o = np.where(h == 1)[0]
        cur_max = np.max(corr)
        best_nm, best_i, best_j = cur_max, -1, -1
        for i in z:
            for j in o:
                h[i], h[j] = 1.0, 0.0
                nc = np.correlate(h, 1.0 - h, mode="full")
                nm = np.max(nc)
                h[i], h[j] = 0.0, 1.0
                if nm < best_nm:
                    best_nm, best_i, best_j = nm, i, j
        if best_i >= 0 and best_nm < cur_max:
            h[best_i], h[best_j] = 1.0, 0.0
            corr = np.correlate(h, 1.0 - h, mode="full")
            best_c5 = best_nm * (2.0 / n)
        else:
            break
    return h, best_c5


def correct_sum(h, nt, rng):
    while int(h.sum()) > nt:
        o = np.where(h == 1)[0]
        h[rng.choice(o)] = 0
    while int(h.sum()) < nt:
        z = np.where(h == 0)[0]
        h[rng.choice(z)] = 1


def greedy_binary(n, rng):
    h = np.zeros(n)
    positions = list(range(n))
    rng.shuffle(positions)
    nt = n // 2
    for k in range(nt):
        best_p, best_mx = positions[0], 1e9
        for p in positions:
            h[p] = 1.0
            mx = np.max(np.correlate(h, 1.0 - h, mode="full"))
            h[p] = 0.0
            if mx < best_mx:
                best_mx, best_p = mx, p
        h[best_p] = 1.0
        positions.remove(best_p)
    return h

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
