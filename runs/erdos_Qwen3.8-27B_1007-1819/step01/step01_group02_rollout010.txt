def compute_c5(h):
    n = len(h)

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def objective_smooth(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    logsum = np.log(np.sum(np.exp(beta * (corr - m))))
    return (m + logsum / beta) * dx

        grad[:n - k] -= h[k:]

        grad[:n - p] += (1.0 - h[p:])
        grad[p:] -= h[:n - p]
    return grad
