```python
def corr_and_grad(h):
    """Return (max_corr, gradient_of_max_corr, argmax_index)"""
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_star = int(np.argmax(corr))
    k = m_star - (N - 1)  # actual lag
    # grad[i] = (1-h[i-k]) if 0<=i-k<N else 0  -  h[i+k] if 0<=i+k<N else 0
    grad = np.zeros(N)
    # First term: (1-h[i-k]) for i where 0<=i-k<N, i.e., k<=i<N+k
    lo1 = max(0, k)
    hi1 = min(N, N + k)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[idx] += (1.0 - h[idx - k])
    # Second term: -h[i+k] for i where 0<=i+k<N, i.e., -k<=i<N-k
    lo2 = max(0, -k)
    hi2 = min(N, N - k)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] -= h[idx + k]
    return corr, m_star, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) * beta, -700, 0)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / beta) * dx

def top_k_gradient(h, k_top=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-k_top:]
    grad = np.zeros(n)
    for mi in top_idx:
        lag = int(mi) - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += (1.0 - h[idx - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= h[idx + lag]
    grad /= k_top
    return corr, grad

def proj_gradient_steps(h, lr0, n_steps, k_top=3):
    n = len(h)
    cur = compute_c5(h)
    lr = lr0
    for _ in range(n_steps):
        _, grad = top_k_gradient(h, k_top)
        grad_feas = grad - np.mean(grad)
        h_new = h - lr * grad_feas
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_feasible(h_new)
        c = compute_c5(h_new)
        if c <= cur + 1e-13:
            h = h_new
            cur = c
        else:
            lr *= 0.95
        if lr < 1e-6:
            break
    return h, cur

def coord_descent(h, steps, max_rounds=20):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for d in steps:
                v = h[i] + d
                if v < -1e-12 or v > 1 + 1e-12:
                    continue
                ht = h.copy()
                ht[i] = v
                ht = fix_feasible(ht)
                c = compute_c5(ht)
                if c < cur - 1e-13:
                    cur = c
                    h = ht
                    improved = True
        if not improved:
            break
    return h

def slsqp_smooth(h, beta, maxiter=300):
    n = len(h)
    def obj(x):
        return smooth_obj(fix_feasible(x), beta)
    def constr(x):
        return np.sum(x) - n / 2.0
    def jac_c(x):
        return np.ones(n)
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': constr, 'jac': jac_c}],
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        return fix_feasible(res.x)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    best_h = None
    best_c5 = 1e10

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is not None:
        h_init = fix_feasible(h_init)
        best_c5 = compute_c5(h_init)
        best_h = h_init.copy()
    else:
        n0 = 100
        best_h = fix_feasible(np.full(n0, 0.5))
        best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    n = len(best_h)

    # Phase 1: Projected gradient descent with decreasing lr
    for phase_lr in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]:
        if elapsed() > budget_s * 0.45:
            break
        n_steps = 400
        h_pg, c = proj_gradient_steps(best_h, phase_lr, n_steps, k_top=3)
        if c < best_c5:
            best_c5 = c
            best_h = h_pg

    # Phase 2: SLSQP with smooth max
    for beta in [200, 600, 1500]:
        if elapsed() > budget_s * 0.6:
            break
        h_s = slsqp_smooth(best_h, beta, maxiter=400)
        c = compute_c5(h_s)
        if c < best_c5:
            best_c5 = c
            best_h = h_s

    # Phase 3: More projected gradient with 1-lag focus
    for phase_lr in [0.02, 0.01, 0.005]:
        if elapsed() > budget_s * 0.72:
            break
        h_pg, c = proj_gradient_steps(best_h, phase_lr, 600, k_top=1)
        if c < best_c5:
            best_c5 = c
            best_h = h_pg

    # Phase 4: Coordinate descent with adaptive steps
    if elapsed() < budget_s * 0.88:
        for steps in [
            [0.05, -0.05, 0.1, -0.1, 0.02, -0.02],
            [0.02, -0.02, 0.01, -0.01, 0.03, -0.03],
            [0.005, -0.005, 0.002, -0.002, 0.008, -0.008],
        ]:
            if elapsed() > budget_s * 0.88:
                break
            h_cd = coord_descent(best_h, steps, max_rounds=12)
            c = compute_c5(h_cd)
            if c < best_c5:
                best_c5 = c
                best_h = h_cd

    # Phase 5: Basin hopping - perturb and re-optimize
    if elapsed() < budget_s * 0.92:
        n_max_hops = max(3, int((budget_s * 0.92 - elapsed()) / 12))
        for hop in range(n_max_hops):
            if elapsed() > budget_s * 0.92:
                break
            scale = 0.03 + 0.08 * np.random.random()
            pert = best_h + np.random.randn(n) * scale
            pert = np.clip(pert, 0, 1)
            pert = fix_feasible(pert)
            # Quick gradient descent
            h_h, c_h = proj_gradient_steps(pert, 0.03, 200, k_top=3)
            # Quick SLSQP
            if elapsed() < budget_s * 0.92:
                h_h = slsqp_smooth(h_h, 400, maxiter=150)
                c_h = compute_c5(h_h)
            if c_h < best_c5:
                best_c5 = c_h
                best_h = h_h

    # Phase 6: Final fine polishing
    if best_h is not None and elapsed() < budget_s * 0.97:
        h_fin = coord_descent(best_h, [0.003, -0.003, 0.001, -0.001, 0.005, -0.005], max_rounds=10)
        c = compute_c5(h_fin)
        if c < best_c5:
            best_c5 = c
            best_h = h_fin
        # One more SLSQP pass

            h_fin2 = slsqp_smooth(best_h, 2000, maxiter=200)
            c2 = compute_c5(h_fin2)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h_fin2

    if best_h is None:
        best_h = fix_feasible(np.full(50, 0.5))
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
