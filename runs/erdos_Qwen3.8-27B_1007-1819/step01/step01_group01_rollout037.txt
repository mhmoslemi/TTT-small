        i_range = np.arange(n)

        mask = (idx >= 0) & (idx < n)
        grad[np.where(mask)[0]] += weights[s] * (1 - h[idx[mask]])

        mask2 = (idx2 >= 0) & (idx2 < n)

    grad *= dx
    return grad

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -100, 100)
    return (m + np.log(np.sum(np.exp(s))) / alpha) * dx

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -100, 100)
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    for si in range(2*n-1):
        wi = w[si]
        if wi < 1e-14:
            continue
        k = shifts[si]
        # d corr[si]/d h[i] = (1-h[i-k])*1_{0<=i-k<n} - h[i+k]*1_{0<=i+k<n}
        idx1 = np.arange(n) - k
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += wi * (1.0 - h[idx1[mask1]])
        idx2 = np.arange(n) + k
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] -= wi * h[idx2[mask2]]
    return grad * dx

def slsqp_optimize(h0, n, alpha, maxiter=200):
    def f(x):
        return smooth_max_obj(x, n, alpha)
    def g(x):
        return smooth_max_grad(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def nesterov_gradient(h0, n, alpha, lr0, n_iter, total_step_factor=1.0):
    h = h0.copy()

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_val_and_weights(corr, alpha):
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -100, 100)
    e = np.exp(s)
    w = e / e.sum()
    return (m + np.log(e.sum()) / alpha), w

def compute_smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    _, w = smooth_max_val_and_weights(corr, alpha)
    grad = np.zeros(n)
    idx = np.arange(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for si in range(2 * n - 1):
        wi = w[si]
        if wi < 1e-14:
            continue
        k = shifts[si]
        i1 = idx - k
        m1 = (i1 >= 0) & (i1 < n)
        grad[m1] += wi * (1.0 - h[i1[m1]])
        i2 = idx + k
        m2 = (i2 >= 0) & (i2 < n)
        grad[m2] -= wi * h[i2[m2]]
    return grad * dx

def slsqp_smooth(h0, n, alpha, maxiter=300):
    def f(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        v, _ = smooth_max_val_and_weights(c, alpha)
        return v * (2.0 / n)

    def g(x):
        return compute_smooth_grad(x, n, alpha)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def slsqp_exact(h0, n, maxiter=200):
    def f(x):
        return compute_c5(x)

    def g(x):
        h = np.clip(x, 0, 1)
        return compute_smooth_grad(h, n, 800)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), jac=g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def nesterov_step(h0, n, alpha, lr, n_iter):
    h = h0.astype(float).copy()
    v = np.zeros(n)
    mu = 0.9
    for t in range(n_iter):
        lr_t = lr / (1.0 + 0.01 * t)
        g = compute_smooth_grad(h, n, alpha)
        v = mu * v - lr_t * g
        h_new = project_feasible(h + v, n)
        h = h_new
    return h

def coord_desc(h, n, rounds=40, n_trials=31):
    h = h.copy()
    curr = compute_c5(h)
    vals = np.linspace(0, 1, n_trials)
    for _ in range(rounds):
        improved = False
        for i in range(n):
            best_c = curr
            best_v = h[i]
            old = h[i]
            for rv in vals:
                if abs(rv - old) < 1e-7:
                    continue
                h[i] = rv
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = h[i]
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_c
        if not improved:
            break
    return h, curr

def sim_anneal(h0, n, steps):
    h = h0.astype(float).copy()
    curr = compute_c5(h)
    best = h.copy()
    best_c = curr
    t = time.time()
    for step in range(steps):
        if time.time() - t > steps * 0.01:
            break
        n_pert = np.random.choice([1, 2, 3, 5], p=[0.4, 0.3, 0.2, 0.1])
        idxs = np.random.choice(n, size=n_pert, replace=False)
        h_new = h.copy()
        scale = 0.1 * (1.0 - step / steps) + 0.02
        h_new[idxs] = np.clip(h_new[idxs] + np.random.randn(n_pert) * scale, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        T = 0.005 * (1.0 - step / steps) + 1e-6
        if c_new < curr or np.random.rand() < np.exp(-(c_new - curr) / T):
            h = h_new
            curr = c_new
            if curr < best_c:
                best_c = curr
                best = h.copy()
    return best, best_c
