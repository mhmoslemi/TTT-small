def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(3 * n + 10):
            r = target - np.sum(h)
            if abs(r) < 1e-14:
                break
            if r > 0:
                i = int(np.argmin(1.0 - h))
                amt = min(r, 1.0 - h[i])
                h[i] += amt
            else:
                i = int(np.argmax(h))
                amt = min(-r, h[i])
                h[i] -= amt
    return h

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

    c = corr_full(h)
    mx = np.max(c)
    return float((mx + np.log(np.sum(np.exp(beta*(c-mx))))/beta) * dx)

    try:

                       bounds=[(0,1)]*n,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
        return fix_feasible(res.x)
    except:
        return h

def optimize_slsqp(h, beta, max_iter=400):

    def obj(x):
        return smooth_c5(x, beta)
    def constraint(x):
        return np.sum(x) - n/2.0

                       bounds=[(0,1)]*n,
                       options={'maxiter': max_iter, 'ftol': 1e-12})

    except:

def coord_refine(h, rounds=20):

    cur = compute_c5(h)
    deltas = [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.2, -0.2, 0.03, -0.03]

    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(3 * n + 10):

            if abs(r) < 1e-14:
                break
            if r > 0:
                i = int(np.argmin(1.0 - h))
                h[i] = min(1.0, h[i] + r)
            else:

def swap_hill_climb(h, max_rounds=500, time_limit=60.0):
    """Hill-climbing via swaps (preserves sum constraint)."""
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    t_start = time.time()
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        best_imp = -1.0
        best_i = best_j = -1
        best_val = cur
        for i in range(n):
            for j in range(i + 1, n):
                # Skip if values are equal
                if h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h)
                h[i], h[j] = h[j], h[i]
                imp = cur - c5
                if imp > best_imp:
                    best_imp = imp
                    best_i, best_j = i, j
                    best_val = c5
        if best_i >= 0:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            cur = best_val
        else:
            break
    return h, cur
