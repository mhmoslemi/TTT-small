def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_from)
    x_new = np.linspace(0.0, 2.0, n_to)
    return np.interp(x_new, x_old, h)

def smooth_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_grad(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    n = n_pts
    dx = 2.0 / n

                    if hasattr(res, 'x') and res.x is not None:
                        h_opt = project_feasible(res.x, n_pts)
                        c5_v = compute_c5(h_opt)
                        if c5_v < best_c5:
                            best_c5 = c5_v
                            best_h = h_opt.copy()
                            best_n = n_pts
                            best_h_curr = h_opt.copy()
                except Exception:
                    pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
