    c = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        c[i] = c[i >> 1] + (i & 1)
    return (c % 2).astype(float)

    """Simulated annealing on binary sequences."""
    h = make_binary(h, n, rng).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for it in range(iters):
        T = 0.01 * (0.5 ** (it / max(1, iters//3)))
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        else:

    h = h.astype(float).copy()
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(iters):

            break
        grad = np.zeros(n)

            elif s < 0:
