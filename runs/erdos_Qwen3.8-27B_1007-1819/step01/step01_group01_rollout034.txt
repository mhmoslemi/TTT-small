            best_c5 = c5
            best_h = h.copy()
        if time.time() - t_start > time_budget:
            break
    return best_h, best_c5

def local_flip(h, n, n_trials=200):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        i = np.random.randint(n)
        best_local = best_h[i]
        best_local_c5 = best_c5
        for trial in [0.0, 1.0, 0.5, max(0.0, h[i] - 0.15), min(1.0, h[i] + 0.15)]:
            if abs(trial - h[i]) < 1e-10:
                continue
            h_t = best_h.copy()
            h_t[i] = trial
            h_t = project_feasible(h_t, n)
            c_t = compute_c5(h_t)
            if c_t < best_local_c5:
                best_local_c5 = c_t
                best_local = h_t[i]
        best_h[i] = best_local
        best_h = project_feasible(best_h, n)
        best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def tleft():
        return budget_s - 10 - (time.time() - t0)

    try:
        init = np.array(initial_h_values, dtype=float)
        n_init = len(init)
    except NameError:
        init = np.full(50, 0.5)
        n_init = 50
    init = project_feasible(init, n_init)
    best_h = init.copy()
    best_c5 = compute_c5(init)
    best_n = n_init

    # Phase 1: Binary exhaustive n=20
    if tleft() > 15:
        h_b, c_b = binary_exhaustive(20, min(tleft() - 5, 20))
        if c_b < best_c5:
            best_c5, best_h, best_n = c_b, h_b.copy(), 20

    # Phase 2: Binary exhaustive n=24
    if tleft() > 40:
        h_b, c_b = binary_exhaustive(24, min(tleft() - 10, 50))
        if c_b < best_c5:
            best_c5, best_h, best_n = c_b, h_b.copy(), 24

    # Phase 3: Projected gradient from initial at various alpha
    if tleft() > 80:
        for alpha in [30, 100, 400, 2000, 10000]:
            if tleft() < 20:
                break
            h_o, c_o = proj_grad_opt(init, n_init, alpha, 3000, lr0=0.06)
            if c_o < best_c5:
                best_c5, best_h = c_o, h_o.copy()

    # Phase 4: Try multiple n values
    n_tries = sorted(set([best_n, n_init, max(20, n_init - 20), n_init + 20, n_init + 40]))
    for n_t in n_tries:
        if tleft() < 30 or n_t < 10:
            continue
        if n_t == best_n:
            h_s = best_h.copy()
        else:
            h_s = np.interp(np.linspace(0, 1, n_t), np.linspace(0, 1, best_n), best_h)
            h_s = project_feasible(h_s, n_t)
        for alpha in [100, 800, 5000]:
            if tleft() < 15:
                break
            h_o, c_o = proj_grad_opt(h_s, n_t, alpha, 2000, lr0=0.05)
            if c_o < best_c5:
                best_c5, best_h, best_n = c_o, h_o.copy(), n_t

    # Phase 5: Random restarts at best_n
    if tleft() > 60:
        for _ in range(8):
            if tleft() < 15:
                break
            h_r = project_feasible(np.clip(0.5 + 0.35 * np.random.randn(best_n), 0, 1), best_n)
            h_o, c_o = proj_grad_opt(h_r, best_n, 500, 2500, lr0=0.06)
            if c_o < best_c5:
                best_c5, best_h = c_o, h_o.copy()

    # Phase 6: High-alpha refinement (near-exact max)
    if tleft() > 30:
        for alpha in [5000, 20000, 80000]:
            if tleft() < 10:
                break
            h_o, c_o = proj_grad_opt(best_h, best_n, alpha, 4000, lr0=0.03)
            if c_o < best_c5:
                best_c5, best_h = c_o, h_o.copy()

    # Phase 7: Local flip refinement
    if tleft() > 20:
        h_o, c_o = local_flip(best_h, best_n, n_trials=300)
        if c_o < best_c5:
            best_c5, best_h = c_o, h_o.copy()

    # Phase 8: One more projected gradient pass
    if tleft() > 15:
        h_o, c_o = proj_grad_opt(best_h, best_n, 50000, 3000, lr0=0.02)
        if c_o < best_c5:
            best_c5, best_h = c_o, h_o.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
