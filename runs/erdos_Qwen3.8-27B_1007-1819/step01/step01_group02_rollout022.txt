def subgradient_of_max_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    s = k_star - (n - 1)
    
    grad = np.zeros(n)
    # Term 1: (1-h[j-s]) for j-s in [0, n-1]
    # j ranges from max(0, s) to min(n-1, n-1+s)
    if s >= 0:
        # j-s in [0,n-1] => j in [s, s+n-1] => j in [s, n-1]
        grad[s:] += (1.0 - h[:n-s])
        # j+s in [0,n-1] => j in [-s, n-1-s] => j in [0, n-1-s]
        grad[:n-s] -= h[s:]
    else:
        # j-s in [0,n-1] => j in [s, s+n-1] => j in [0, n-1+s] (since s<0)
        grad[:n+s] += (1.0 - h[-s:])
        # j+s in [0,n-1] => j in [-s, n-1-s] => j in [0, n-1-s] (since -s>0)
        grad[-s:] -= h[:n+s]
    
    return grad * dx, corr[k_star] * dx, k_star

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_max_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx


def sum_con(x, n_pts):
    return x.sum() - n_pts / 2.0


def interp(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xo = np.arange(n_from) * 2.0 / n_from
    xn = np.arange(n_to) * 2.0 / n_to
    return np.interp(xn, xo, h)


def slsqp_opt(h, n, beta, maxiter=1000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]
    res = minimize(smooth_max_obj, h, args=(n, beta), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if hasattr(res, 'x') and res.x is not None:
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    return h, compute_c5(h)


def subgrad_step(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        lo1, hi1 = s, n
        grad[lo1:hi1] += (1.0 - h[0:n - s])
        grad[0:n - s] -= h[s:]
    else:
        lo1, hi1 = 0, n + s
        grad[lo1:hi1] += (1.0 - h[-s:])
        grad[-s:] -= h[0:n + s]
    return grad * dx, k_star


def subgrad_method(h, n, max_iter=5000, lr_init=0.003):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    stall = 0
    for it in range(max_iter):
        grad, _ = subgrad_step(best_h, n)
        h_new = best_h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.02, 0.05)
        else:
            stall += 1
            if stall % 20 == 0:
                lr *= 0.8
            if stall > 200:
                break
    return best_h, best_c5


def coord_descent(h, n, steps_list=None, max_passes=50):
    if steps_list is None:
        steps_list = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for p in range(max_passes):
        ss = steps_list[min(p, len(steps_list) - 1)]
        improved = False
        for i in range(n):
            for sign in (1, -1):
                ht = best_h.copy()
                ht[i] += sign * ss
                ht = project_feasible(ht, n)
                c = compute_c5(ht)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = ht
                    improved = True
                    break
        if not improved and p >= 3:
            break
    return best_h, best_c5


def pairwise_cd(h, n, step=0.005, max_passes=80):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(7)
    for p in range(max_passes):
        improved = False
        for _ in range(20):
            i, j = rng.randint(n - 1), rng.randint(n - 1)
            if i == j:
                continue
            for di in (1, -1):
                for dj in (1, -1):
                    ht = best_h.copy()
                    ht[i] += di * step
                    ht[j] += dj * step
                    ht = np.clip(ht, 0.0, 1.0)
                    ht = project_feasible(ht, n)
                    c = compute_c5(ht)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        best_h = ht
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved and p >= 5:
            break
    return best_h, best_c5

    # Final validation
