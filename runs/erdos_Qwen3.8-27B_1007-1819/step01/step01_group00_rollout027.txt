import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n + 2):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = int(np.argmin(1.0 - h))
                h[idx] = min(1.0, h[idx] + min(r, 1.0 - h[idx]))
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] - min(-r, h[idx]))
    return h

def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = float(np.max(corr))
    rng = mx - float(np.min(corr))
    if rng < 1e-15:
        return mx
    if beta * rng > 400:
        k = min(20, len(corr))
        top = np.sort(corr)[-k:]
        return float(mx + np.log(np.sum(np.exp(beta * (top - mx)))) / beta)
    s = np.sum(np.exp(beta * (corr - mx)))
    return float(mx + np.log(s) / beta)

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c, x = 0, i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c & 1
    return t

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def fine_cd(h, rounds, budget_check):
    n = len(h)
    h = fix_feasible(h.copy())
    cur = compute_c5(h)
    steps = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002,
             0.001, -0.001, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]
    for r in range(rounds):
        if budget_check():
            break
        improved = False
        for i in range(n):
            if budget_check():
                break
            h_save = h.copy()
            cv = h[i]
            best_local = cur
            best_h = h_save
            for d in steps:
                v = cv + d
                if v < 0 or v > 1:
                    continue
                ht = h_save.copy()
                ht[i] = v
                ht = fix_feasible(ht)
                obj = compute_c5(ht)
                if obj < best_local - 1e-13:
                    best_local = obj
                    best_h = ht.copy()
            if best_local < cur - 1e-13:
                h = best_h
                cur = best_local
                improved = True
        if not improved:
            break
    return h, cur

def pair_cd(h, rounds, budget_check):
    n = len(h)
    h = fix_feasible(h.copy())
    cur = compute_c5(h)
    for r in range(rounds):
        if budget_check():
            break
        improved = False
        rng = np.random.default_rng(42 + r)
        for _ in range(n * 2):
            if budget_check():
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            d = float(rng.choice([0.02, -0.02, 0.05, -0.05, 0.01, -0.01]))
            ht = h.copy()
            vi, vj = ht[i] + d, ht[j] - d
            if vi < 0 or vi > 1 or vj < 0 or vj > 1:
                continue
            ht[i], ht[j] = vi, vj
            ht = fix_feasible(ht)
            obj = compute_c5(ht)
            if obj < cur - 1e-13:
                h = ht
                cur = obj
                improved = True
        if not improved:
            break
    return h, cur

        return time.time() - t0

    def tk(frac):

    best_h = None
