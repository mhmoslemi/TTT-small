                if cv < best_c5:
                    best_c5 = cv
                    best_h = ho.copy()
                    best_n = n_try

    # Phase 3: Intensive coordinate descent at best resolution
    n_pts = best_n
    h_c = best_h.copy()
    cv = compute_c5(h_c)
    if cv < best_c5:
        best_c5, best_h = cv, h_c.copy()

    steps_cd = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    max_passes_cd = [3, 3, 4, 4, 5, 5, 6, 8]
    for s, mp in zip(steps_cd, max_passes_cd):
        if elapsed() > budget_s * 0.85:
            break
        h_new, cv_new = coord_descent(h_c, n_pts, [s], mp, budget_s * 0.85, t0, best_c5)
        if cv_new < best_c5:
            best_c5 = cv_new
            best_h = h_new.copy()
            h_c = h_new.copy()

    # Phase 4: Pairwise mass transfer
    if elapsed() < budget_s * 0.55:
        h_c = best_h.copy()
        amts = [0.02, 0.01, 0.005]
        n_trials = min(20000, int((budget_s * 0.85 - elapsed()) * 200))
        h_new, cv_new = pairwise_search(h_c, n_pts, rng, n_trials, amts, budget_s * 0.85, t0)
        if cv_new < best_c5:
            best_c5 = cv_new
            best_h = h_new.copy()

    # Phase 5: Final ultra-fine coordinate descent
    if elapsed() < budget_s * 0.55:
        h_c = best_h.copy()
        for s in [0.0005, 0.0002, 0.0001, 0.00005]:
            if elapsed() > budget_s * 0.92:
                break
            h_new, cv_new = coord_descent(h_c, n_pts, [s], 6, budget_s * 0.92, t0, best_c5)
            if cv_new < best_c5:
                best_c5 = cv_new
                best_h = h_new.copy()
                h_c = h_new.copy()

    # Phase 6: Random multi-coordinate perturbations (simulated annealing lite)
    if elapsed() < budget_s * 0.6:
        h_c = best_h.copy()
        cv_c = compute_c5(h_c)
        n_perturb = min(5000, int((budget_s * 0.92 - elapsed()) * 300))
        for _ in range(n_perturb):
            if elapsed() > budget_s * 0.92:
                break
            n_mod = rng.randint(1, 4)
            idxs = rng.choice(n_pts, n_mod, replace=False)
            ht = h_c.copy()
            for idx in idxs:
                ht[idx] = np.clip(ht[idx] + rng.uniform(-0.03, 0.03), 0.0, 1.0)
            ht = project_feasible(ht, n_pts)
            cv = compute_c5(ht)
            if cv < cv_c - 1e-14:
                h_c = ht
                cv_c = cv
                if cv_c < best_c5:
                    best_c5 = cv_c
                    best_h = ht.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
