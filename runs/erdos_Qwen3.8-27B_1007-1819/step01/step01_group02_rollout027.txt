from scipy.optimize import minimize
import time


def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def interp_h(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xo = np.arange(n_from) * 2.0 / n_from
    xn = np.arange(n_to) * 2.0 / n_to
    return np.interp(xn, xo, h)


def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta


def obj_lbfgs(x, n_pts, beta, mu):
    dx = 2.0 / n_pts
    corr = np.correlate(x, 1.0 - x, mode="full")
    sm = smooth_max_val(corr, beta)
    sv = x.sum() - n_pts / 2.0
    return sm * dx + mu * sv * sv


def obj_slsqp(x, n_pts, beta):
    dx = 2.0 / n_pts
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max_val(corr, beta) * dx


def cons_sum(x, n_pts):
    return x.sum() - n_pts / 2.0


def make_anticperiodic(n_pts, seed):
    rng = np.random.RandomState(seed)
    half = n_pts // 2
    h_first = rng.uniform(0.0, 1.0, half)
    h = np.zeros(n_pts)
    h[:half] = h_first
    h[half:] = 1.0 - h_first
    return project_feasible(h, n_pts)


def make_cosine(n_pts, freq, amp, seed):
    rng = np.random.RandomState(seed)
    xg = np.arange(n_pts) * 2.0 / n_pts
    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * xg + rng.uniform(0, 2 * np.pi))
    return project_feasible(h, n_pts)

    h = np.zeros(n_pts)
    ones = rng.choice(n_pts, n_pts // 2, replace=False)
    h[ones] = 1.0
    return h
