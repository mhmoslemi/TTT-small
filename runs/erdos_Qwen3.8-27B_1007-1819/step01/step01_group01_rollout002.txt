from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_probs(corr, alpha):
    shifted = alpha * (corr - corr.max())
    exp_t = np.exp(shifted)
    s = exp_t.sum()
    probs = exp_t / s
    val = corr.max() + np.log(s) / alpha
    return val, probs

def obj_and_grad(params, n, alpha, penalty):
    h = np.clip(params, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val_sm, probs = smooth_max_and_probs(corr, alpha)
    obj = val_sm * dx + penalty * (h.sum() - n/2.0)**2
    
    # Gradient
    grad_corr = np.zeros(n)
    for m in range(-(n-1), n):
        p = probs[m + n - 1]
        if p < 1e-30:
            continue
        # First term: (1-h[i+m]) for valid i
        istart = max(0, -m)
        iend = min(n, n - m)
        if istart < iend:
            grad_corr[istart:iend] += p * (1.0 - h[istart+m:iend+m])
        # Second term: -h[i-m] for valid i
        istart2 = max(0, m)
        iend2 = min(n, n + m)
        if istart2 < iend2:
            grad_corr[istart2:iend2] += p * (-h[istart2-m:iend2-m])
    
    grad = dx * grad_corr + 2 * penalty * (h.sum() - n/2.0)
    return obj, grad

def lbfqsb_optimize(h0, n, alphas, penalty=100.0, time_limit=60):
    h = h0.copy()
    t0 = time.time()
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        res = minimize(
            obj_and_grad, h.copy(),
            args=(n, alpha, penalty),
            method='L-BFGS-B',
            jac=True,
            bounds=[(0,1)]*n,
            options={'maxiter': 200, 'gtol': 1e-10, 'maxls': 50}
        )
        h = np.clip(res.x, 0, 1)
    h = project_feasible(h, n)
    return h

def coordinate_descent(h, n, max_rounds=50):
    h = h.copy()
    curr = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            # Try a few values
            candidates = [0.0, 1.0, h[i]*0.5, h[i]*1.5, h[i]+0.1, h[i]-0.1]
            candidates = [np.clip(c, 0, 1) for c in candidates]
            best_v = h[i]
            best_c = curr
            for cv in set(np.round(candidates, 4)):
                h[i] = cv
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = h[i]
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_c
        if not improved:
            break
    return h, curr

def sa_optimize(h0, n, budget, seed=42):

    curr = compute_c5(h)
    best = h.copy()

        h_new = project_feasible(h_new, n)

        elapsed = time.time() - t0

            curr = c_new
            if curr < best_val:

    return best, best_val

corr[k] = Σ_j h[j] * (1-h)[n-1-k+j] = Σ_j h[j] * (1-h[j + n-1-k])

Let m = n-1-k. Then corr[k] = Σ_j h[j] * (1-h[j+m]) where m = n-1-k.
As k goes from 0 to 2n-2, m goes from n-1 to -(n-1).

So: m = n-1-k, or k = n-1-m.

∂corr[k]/∂h[i] where m = n-1-k:
- First factor: j=i, needs 0≤i<n and 0≤i+m<n → (1-h[i+m])
- Second factor: j+m=i → j=i-m, needs 0≤i-m<n and 0≤i<n → -h[i-m]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_probs(corr, alpha):
    shifted = alpha * (corr - corr.max())
    exp_t = np.exp(shifted)
    s = exp_t.sum()
    probs = exp_t / s
    val = corr.max() + np.log(s) / alpha
    return val, probs

def obj_and_grad(params, n, alpha, penalty):
    h = np.clip(params, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val_sm, probs = smooth_max_probs(corr, alpha)
    obj = val_sm * dx + penalty * (h.sum() - n / 2.0) ** 2
    grad_corr = np.zeros(n)
    for m in range(-(n - 1), n):
        p = probs[m + n - 1]
        if p < 1e-30:
            continue
        istart = max(0, -m)
        iend = min(n, n - m)
        if istart < iend:
            grad_corr[istart:iend] += p * (1.0 - h[istart + m:iend + m])
        istart2 = max(0, m)
        iend2 = min(n, n + m)
        if istart2 < iend2:
            grad_corr[istart2:iend2] += p * (-h[istart2 - m:iend2 - m])
    grad = dx * grad_corr + 2.0 * penalty * (h.sum() - n / 2.0)
    return obj, grad

def lbfqsb_opt(h0, n, alphas, penalty=500.0, time_limit=40):
    h = h0.copy()
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(obj_and_grad, h.copy(), args=(n, alpha, penalty),
                           method='L-BFGS-B', jac=True, bounds=bounds,
                           options={'maxiter': 300, 'gtol': 1e-12, 'maxls': 30})
            h = np.clip(res.x, 0.0, 1.0)
        except Exception:
            pass
    h = project_feasible(h, n)
    return h

def coord_descent(h, n, max_rounds=40):
    h = h.copy()
    curr = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            candidates = np.unique(np.clip(
                [0.0, 1.0, old * 0.5, old * 1.5, old + 0.1, old - 0.1,
                 0.25, 0.5, 0.75], 0, 1))
            best_v, best_c = old, curr
            for cv in candidates:
                if abs(cv - old) < 1e-6:
                    continue
                h[i] = cv
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c, best_v = c, h[i]
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_c
        if not improved:
            break
    return h, curr

def sa_opt(h0, n, budget, seed=42):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    curr = compute_c5(h)
    best, best_val = h.copy(), curr
    t0 = time.time()
    while time.time() - t0 < budget:
        mt = rng.randint(3)
        h_new = h.copy()
        if mt == 0:
            npert = rng.randint(1, 4)
            idxs = rng.choice(n, size=npert, replace=False)
            for i in idxs:
                h_new[i] = np.clip(h_new[i] + rng.randn() * 0.12, 0, 1)
            h_new = project_feasible(h_new, n)
        elif mt == 1:
            i, j = rng.choice(n, size=2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            blen = rng.randint(2, 7)
            st = rng.randint(0, n - blen)
            h_new[st:st + blen] = np.clip(
                h[st:st + blen] + rng.randn(blen) * 0.08, 0, 1)
            h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        frac = (time.time() - t0) / budget
        T = 0.004 * (1.0 - frac) ** 2 + 0.0001
        if c_new < curr or rng.rand() < np.exp(-(c_new - curr) / T):
            h, curr = h_new, c_new
            if curr < best_val:
                best_val, best = curr, h.copy()
    return best, best_val
