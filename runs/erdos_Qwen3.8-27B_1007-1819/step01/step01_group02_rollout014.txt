        g[idx + k] -= h[idx]

    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -500, 500)))) / beta) * dx

def constraint_sum(x, n_pts):
    return x.sum() - n_pts / 2.0

def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)

def subgradient_polish(h, n, n_iters, top_k=3, base_alpha=0.02, seed=0):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    for it in range(n_iters):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            g += grad_lag_vec(h_curr, k, n)
        g /= max(1, top_k)
        alpha = base_alpha / (1.0 + 0.005 * it)
        alpha *= (0.5 + rng.rand())
        h_new = project_feasible(h_curr - alpha * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new
        elif rng.rand() < 0.05 * np.exp(-800.0 * (c5_new - best_c5)):
            h_curr = h_new
    return best_h, best_c5

def coord_descent(h, n, max_sweeps=4, step=0.004):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            for d in [step, -step, 2*step, -2*step]:
                h_test = best_h.copy()
                h_test[i] = np.clip(h_test[i] + d, 0.0, 1.0)
                h_test = project_feasible(h_test, n)
                c5_t = compute_c5(h_test)
                if c5_t < best_c5 - 1e-12:
                    best_h = h_test
                    best_c5 = c5_t
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_init = len(h0)
    except NameError:
        h0 = np.full(180, 0.5)
        n_init = 180

    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n_init

    # Phase 1: SLSQP at initial resolution with increasing sharpness
    if elapsed() < budget_s * 0.30:
        n_pts = best_n
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_pts,)}]
        for beta in [40.0, 100.0, 200.0]:
            if elapsed() > budget_s * 0.30:
                break
            res = minimize(objective_smooth, best_h, args=(n_pts, beta),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-14})
            if hasattr(res, 'x') and res.x is not None:
                h_opt = project_feasible(res.x, n_pts)
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt
                    best_n = n_pts

    # Phase 2: Subgradient polish (different landscape navigation)
    if elapsed() < budget_s * 0.50:
        n_iters = 1500
        best_h, best_c5 = subgradient_polish(best_h, best_n, n_iters, top_k=3, base_alpha=0.012, seed=seed)

    # Phase 3: Higher beta SLSQP + diverse restarts
    if elapsed() < budget_s * 0.70:
        n_pts = best_n
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_pts,)}]
        for beta in [300.0, 600.0]:
            if elapsed() > budget_s * 0.70:
                break
            res = minimize(objective_smooth, best_h, args=(n_pts, beta),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 800, 'ftol': 1e-14})
            if hasattr(res, 'x') and res.x is not None:
                h_opt = project_feasible(res.x, n_pts)
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt
                    best_n = n_pts
        # Diverse restart: small perturbation + re-optimize
        if elapsed() < budget_s * 0.70:
            h_pert = project_feasible(best_h + rng.randn(n_pts) * 0.03, n_pts)
            res = minimize(objective_smooth, h_pert, args=(n_pts, 200.0),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 600, 'ftol': 1e-14})
            if hasattr(res, 'x') and res.x is not None:
                h_opt = project_feasible(res.x, n_pts)
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt
                    best_n = n_pts

    # Phase 4: Upsample to 2x resolution and refine
    if elapsed() < budget_s * 0.88:
        n_hi = best_n * 2
        if n_hi <= 720:
            h_hi = project_feasible(interp_to_size(best_h, best_n, n_hi), n_hi)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi
                best_n = n_hi
            else:
                best_c5 = min(best_c5, c5_hi)
                if c5_hi < best_c5 + 0.001:
                    best_h = h_hi
                    best_n = n_hi

            if elapsed() < budget_s * 0.90:
                bounds_hi = [(0.0, 1.0)] * n_hi
                cons_hi = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_hi,)}]
                for beta in [100.0, 300.0]:
                    if elapsed() > budget_s * 0.92:
                        break
                    res = minimize(objective_smooth, best_h, args=(n_hi, beta),
                                  method='SLSQP', bounds=bounds_hi, constraints=cons_hi,
                                  options={'maxiter': 500, 'ftol': 1e-13})
                    if hasattr(res, 'x') and res.x is not None:
                        h_opt = project_feasible(res.x, n_hi)
                        c5_val = compute_c5(h_opt)
                        if c5_val < best_c5:
                            best_c5 = c5_val
                            best_h = h_opt
                            best_n = n_hi

    # Phase 5: Final coordinate descent polish
    if elapsed() < budget_s * 0.97:
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=3, step=0.003)

    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
