    ht = _G_H0.copy()
    ht[_G_GROUP] = np.clip(x, 0.0, 1.0)
    ht = fix_fast(ht)
    return compute_c5(ht)

def group_nm(h, group, cur_c5, maxiter=100):
    global _G_H0, _G_GROUP
    _G_H0 = h.copy()
    _G_GROUP = group
    try:
        res = minimize(_group_obj, h[group].copy(), method='Nelder-Mead',
                      options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-11, 'adaptive': True})
        ht = h.copy()
        ht[group] = np.clip(res.x, 0.0, 1.0)
        ht = fix_feasible(ht)
        return ht, compute_c5(ht)
    except Exception:
        return h, cur_c5

def coord_descent(h, cur_c5, max_rounds=100, t_limit=None, fine=False):
    n = len(h)
    if fine:
        deltas = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.03, -0.03, 0.05, -0.05]
    else:
        deltas = [0.01, -0.01, 0.02, -0.02, 0.03, -0.03, 0.05, -0.05, 0.08, -0.08, 0.12, -0.12]
    for rnd in range(max_rounds):
        if t_limit and time.time() > t_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if t_limit and time.time() > t_limit:
                break
            hi = h[i]
            for d in deltas:
                v = hi + d
                if v < -1e-9 or v > 1 + 1e-9:
                    continue
                h[i] = v
                h = fix_fast(h)
                c5v = compute_c5(h)
                if c5v < cur_c5 - 1e-13:
                    cur_c5 = c5v
                    improved = True
                    hi = v
                    break
                else:
                    h[i] = hi
                    # Re-fix after undo (fix_fast may have shifted others)
                    h = fix_fast(h)
        if not improved:
            break
    return h, cur_c5

def downsample(h, n_new):
    n_old = len(h)
    edges = np.linspace(0, n_old, n_new + 1)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        lo = int(edges[i])
        hi = min(int(edges[i + 1]), n_old)
        if hi > lo:
            h_new[i] = np.mean(h[lo:hi])
        else:
            h_new[i] = h[min(lo, n_old - 1)]
    return fix_feasible(h_new)

def upsample(h, n_new):
    n_old = len(h)
    edges = np.linspace(0, n_old, n_new + 1)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        lo = int(edges[i])
        hi = min(int(edges[i + 1]) + 1, n_old)
        hi = max(hi, lo + 1)
        h_new[i] = np.mean(h[lo:hi])
    return fix_feasible(h_new)

def local_opt(h, t_limit, max_groups=25, max_cd=40):
    c5v = compute_c5(h)
    n = len(h)
    for _ in range(max_groups):
        if time.time() > t_limit:
            break
        gs = int(np.random.choice([4, 6, 8, 10]))
        gs = min(gs, n - 1)
        grp = np.random.choice(n, gs, replace=False)
        h, c5v = group_nm(h, grp, c5v, maxiter=120)
    h, c5v = coord_descent(h, c5v, max_rounds=max_cd, t_limit=t_limit)
    return h, c5v

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError, Exception):
        h_best = tm_seq(100).astype(float)
    h_best = fix_feasible(h_best)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: Intensive local search at initial n (65% budget)
    t_p1 = t0 + budget_s * 0.65
    for cycle in range(200):
        if time.time() > t_p1:
            break
        # Group Nelder-Mead
        n_gs = max(8, int((t_p1 - time.time()) / 0.4))
        n_gs = min(n_gs, 50)
        for _ in range(n_gs):
            if time.time() > t_p1:
                break
            gs = int(np.random.choice([4, 6, 8, 10]))
            gs = min(gs, n - 1)
            grp = np.random.choice(n, gs, replace=False)
            h_new, c5_new = group_nm(h_best, grp, best_c5, maxiter=100)
            if c5_new < best_c5 - 1e-12:
                h_best = h_new
                best_c5 = c5_new
        # Coord descent
        h_best, best_c5 = coord_descent(h_best, best_c5, max_rounds=25, t_limit=t_p1)
        # Perturbation + re-optimize
        if time.time() < t_p1 - 3.0:
            k = int(np.random.randint(3, 12))
            idx = np.random.choice(n, k, replace=False)
            h_pert = h_best.copy()
            h_pert[idx] = np.clip(h_pert[idx] + np.random.uniform(-0.2, 0.2, k), 0, 1)
            h_pert = fix_feasible(h_pert)
            h_opt, c5_opt = local_opt(h_pert, t_p1, max_groups=12, max_cd=20)
            if c5_opt < best_c5:
                h_best = h_opt
                best_c5 = c5_opt
            # Also try perturbing in the opposite direction
            if time.time() < t_p1 - 3.0:
                h_pert2 = h_best.copy()
                idx2 = np.random.choice(n, k, replace=False)
                h_pert2[idx2] = np.clip(h_pert2[idx2] + np.random.uniform(-0.15, 0.15, k), 0, 1)
                h_pert2 = fix_feasible(h_pert2)
                h_opt2, c5_opt2 = local_opt(h_pert2, t_p1, max_groups=8, max_cd=15)
                if c5_opt2 < best_c5:
                    h_best = h_opt2
                    best_c5 = c5_opt2

    # Phase 2: Try other sizes (20% budget)
    t_p2 = t0 + budget_s * 0.85
    candidates = [45, 60, 80, 120, 48, 72]
    for n_try in candidates:
        if time.time() > t_p2:
            break
        if n_try == n:
            continue
        if n_try < n:
            h_try = downsample(h_best, n_try)
        else:
            h_try = upsample(h_best, n_try)
        h_try, c5_try = local_opt(h_try, t_p2, max_groups=15, max_cd=30)
        if c5_try < best_c5:
            best_c5 = c5_try
            h_best = h_try
            n = len(h_best)

    # Phase 3: Final fine polish (15% budget)
    t_p3 = t0 + budget_s * 0.97
    h_best, best_c5 = coord_descent(h_best, best_c5, max_rounds=80, t_limit=t_p3, fine=True)

    # L-BFGS-B on smoothed objective
    if time.time() < t_p3 - 3.0:
        global _G_H0, _G_GROUP
        try:
            def _smoothed(x):
                return smooth_obj(fix_fast(x), 300.0)
            res = minimize(_smoothed, h_best, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * len(h_best),
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10})
            h_l = fix_feasible(res.x)
            c5_l = compute_c5(h_l)
            if c5_l < best_c5:
                h_best = h_l
                best_c5 = c5_l
        except Exception:
            pass

    h_best, best_c5 = coord_descent(h_best, best_c5, max_rounds=40,
                                     t_limit=t0 + budget_s * 0.99, fine=True)
    h_best = fix_feasible(h_best)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
