def compute_grad(corr, h, beta):
    n = len(h)
    dx = 2.0 / n
    c = corr
    max_c = np.max(c)
    if beta == np.inf:
        # true max
        k_star = np.argmax(c)
        m = k_star - (n - 1)
        grad = np.zeros(n)
        # A part: (1-h[j-m]) for valid j
        for j in range(n):
            jm = j - m
            if 0 <= jm < n:
                grad[j] += (1 - h[jm])
            jp = j + m
            if 0 <= jp < n:
                grad[j] -= h[jp]
        return grad * dx
    else:
        weights = np.exp(beta * (c - max_c))
        weights /= np.sum(weights)
        grad = np.zeros(n)

                if 0 <= jm < n:
                    grad[j] += weights[k] * (1 - h[jm])
                jp = j + m
                if 0 <= jp < n:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(50):
        r = target - np.sum(h)
        if abs(r) < 1e-12:
            break
        if r > 0:
            # Need to add r total. Add to coordinates with most room.
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                add = min(r, room[i])
                h[i] += add
                r -= add
                if abs(r) < 1e-12:
                    break
        else:
            # Need to remove -r total. Remove from coordinates with most value.
            order = np.argsort(-h)
            for i in order:
                rem = min(-r, h[i])
                h[i] -= rem
                r += rem
                if abs(r) < 1e-12:
                    break
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return (mx + np.log(s) / beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    w = np.exp(beta * (corr - mx))
    w /= np.sum(w)
    # Weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        m = k - (n - 1)
        # grad_k[j] = (1-h[j-m])*1{0<=j-m<n} - h[j+m]*1{0<=j+m<n}
        if m >= 0:
            A = np.zeros(n)
            A[m:] = 1.0 - h[:n-m]
            B = np.zeros(n)
            B[:n-m] = h[m:]
        else:
            A = np.zeros(n)
            A[:n+m] = 1.0 - h[-m:]
            B = np.zeros(n)
            B[-m:] = h[:n+m]
        grad += w[k] * (A - B)
    return grad * dx

def grad_line_search(h, g, c5_current, max_iter=30):
    n = len(h)
    alpha = 0.1
    best_h = h
    best_c5 = c5_current
    for _ in range(max_iter):
        h_trial = fix_feasible(h - alpha * g)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-12:
            best_h = h_trial
            best_c5 = c5_trial
            return h_trial, c5_trial, alpha
        alpha *= 0.5
        if alpha < 1e-10:
            break
    return best_h, best_c5, alpha

def momentum_grad_descent(h, betas, iters_per_beta, t0, budget_s):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)  # momentum
    
    for beta in betas:
        for it in range(iters_per_beta):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.65:
                return best_h, best_c5
            h = best_h.copy()
            g = smooth_max_grad(h, beta)
            # Project gradient to remove component along all-ones (for sum constraint)
            g = g - np.mean(g)  # this helps with the sum constraint
            v = 0.9 * v - 0.05 * g  # momentum step (learning rate 0.05)
            h_new = fix_feasible(h + v)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    w = np.exp(beta * (corr - mx))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        m = k - (n - 1)  # actual lag
        # Compute ∂c[m]/∂h
        if m >= 0:
            # (1-h[j+m]) for j=0..n-m-1
            term1 = np.zeros(n)
            term1[:n-m] = 1.0 - h[m:]
            # -h[j-m] for j=m..n-1
            term2 = np.zeros(n)
            term2[m:] = -h[:n-m]
        else:
            km = -m  # positive
            # (1-h[j-m]) = (1-h[j+km]) for j=km..n-1
            term1 = np.zeros(n)
            term1[km:] = 1.0 - h[:n-km]
            # -h[j+m] = -h[j-km] for j=0..n-km-1
            term2 = np.zeros(n)
            term2[:n-km] = -h[km:]
        grad += wk * (term1 + term2)
    
    return grad * dx

def true_max_grad(h, m, dx):
    n = len(h)
    grad = np.zeros(n)
    if m >= 0:
        # First term: (1-h[j-m]) for j=m..n-1
        if n - m > 0:
            grad[m:] += (1.0 - h[:n-m])
        # Second term: -h[j+m] for j=0..n-1-m
        if n - m > 0:
            grad[:n-m] += -h[m:]
    else:
        k = -m
        # First term: (1-h[j+k]) for j=0..n-1-k
        if n - k > 0:
            grad[:n-k] += (1.0 - h[k:])
        # Second term: -h[j-k] for j=k..n-1
        if n - k > 0:
            grad[k:] += -h[:n-k]
    return grad * dx
