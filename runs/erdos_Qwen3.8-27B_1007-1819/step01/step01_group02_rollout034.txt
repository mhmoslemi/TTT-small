def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
