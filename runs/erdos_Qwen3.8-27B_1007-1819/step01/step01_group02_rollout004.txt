    cuts_g = []
    cuts_b = []
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        c5_curr = np.max(corr) * dx
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

        top_n = min(8, len(corr))
        top_idx = np.argsort(corr)[-top_n:]
        for m_idx in top_idx:
            lag = m_idx - (n_pts - 1)
            g = corr_gradient_single(h_curr, n_pts, lag) * dx
            b = corr[m_idx] * dx - np.dot(g, h_curr)
            cuts_g.append(g)
            cuts_b.append(b)

        if len(cuts_g) > 40:
            cuts_g = cuts_g[-40:]
            cuts_b = cuts_b[-40:]

        m_cuts = len(cuts_g)
        if m_cuts < 2:
            continue
        A_cuts = np.array(cuts_g)
        b_cuts = np.array(cuts_b)

        c_obj = np.zeros(n_pts + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((m_cuts, n_pts + 1))
        A_ub[:, :n_pts] = A_cuts
        A_ub[:, n_pts] = -1.0
        b_ub = -b_cuts
        A_eq = np.zeros((1, n_pts + 1))
        A_eq[0, :n_pts] = 1.0
        b_eq = np.array([n_pts / 2.0])
        bnds = [(0.0, 1.0)] * n_pts + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs')
            if res.success:
                h_new = project_feasible(res.x[:n_pts], n_pts)
                c5_new = compute_c5(h_new)
                h_curr = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_curr.copy()
        except Exception:
            pass

        if it > 5 and it % 10 == 0:
            corr_now = np.correlate(best_h, 1.0 - best_h, mode="full")
            max_lag = np.argmax(corr_now)
            gap = corr_now[max_lag] * dx - best_c5
            if gap < 1e-12:
                break

    return best_h, best_c5


def coord_descent(h_init, n_pts, time_limit=60.0):
    t0 = time.time()
    h_curr = h_init.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    step_sizes = [0.01, 0.005, 0.002, 0.001]

    for epoch in range(200):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = np.random.RandomState(epoch).permutation(n_pts)
        for i in order:
            if time.time() - t0 > time_limit:
                break
            for d in [step_sizes[0], -step_sizes[0], step_sizes[1], -step_sizes[1],
                      step_sizes[2], -step_sizes[2]]:
                h_test = h_curr.copy()
                h_test[i] = np.clip(h_test[i] + d, 0.0, 1.0)
                h_test = project_feasible(h_test, n_pts)
                c5_t = compute_c5(h_test)
                if c5_t < best_c5 - 1e-13:
                    h_curr = h_test
                    best_c5 = c5_t
                    best_h = h_curr.copy()
                    improved = True
                    break
        if not improved:
            if step_sizes[0] > 0.0005:
                step_sizes = step_sizes[1:] + [step_sizes[-1] / 2.0]
            else:
                break

    return best_h, best_c5


def multi_start_lbfgs(h_start, n_pts, bounds, beta_schedule, max_it_each, time_limit):
    t0 = time.time()
    h_curr = h_start.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()

    for beta in beta_schedule:
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(
                obj_smooth, h_curr, args=(n_pts, beta),
                jac=lambda x: grad_smooth(x, n_pts, beta),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': max_it_each, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxls': 50}
            )
            h_new = project_feasible(res.x, n_pts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_curr = h_new
        except Exception:
            pass

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_init = len(h0)
    except NameError:
        h0 = np.full(90, 0.5)
        n_init = 90

    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n_init

    resolutions = sorted(set([min(n_init, 180), 90, 180]))

    for n_pts in resolutions:
        if elapsed() > budget_s * 0.70:
            break

        if n_pts == best_n:
            h_curr = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n_pts, endpoint=False)
            h_curr = np.interp(x_new, x_old, best_h)
            h_curr = project_feasible(h_curr, n_pts)

        bounds = [(0.0, 1.0)] * n_pts
        time_alloc = budget_s * 0.65 / len(resolutions)
        h_curr, c5_curr = multi_start_lbfgs(
            h_curr, n_pts, bounds,
            [30, 80, 200, 500], 800, time_alloc
        )
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n_pts

        # Second pass with random perturbations
        if elapsed() < budget_s * 0.70:
            for trial in range(2):
                h_pert = h_curr + rng.randn(n_pts) * 0.05
                h_pert = project_feasible(h_pert, n_pts)
                h_try, c5_try = multi_start_lbfgs(
                    h_pert, n_pts, bounds, [200, 500], 400, 80.0
                )
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_pts
                if elapsed() > budget_s * 0.75:
                    break

    # Cutting-plane refinement
    if elapsed() < budget_s * 0.45:
        cp_limit = min(budget_s * 0.25, 300.0)
        h_cp, c5_cp = cutting_plane(best_h, best_n, max_iter=300, time_limit=cp_limit)
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()

    # Coordinate descent
    if elapsed() < budget_s * 0.30:
        cd_limit = min(budget_s * 0.20, 180.0)
        h_cd, c5_cd = coord_descent(best_h, best_n, time_limit=cd_limit)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final micro-refinement: try small linear combinations
    if elapsed() < budget_s * 0.10:
        n_pts = best_n
        for _ in range(30):
            if elapsed() > budget_s * 0.95:
                break
            direction = rng.randn(n_pts)
            direction -= direction.sum() / n_pts
            direction /= np.max(np.abs(direction)) + 1e-15
            for scale in [0.005, 0.002, 0.001]:
                h_try = project_feasible(best_h + scale * direction, n_pts)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_h = h_try
                    best_c5 = c5_try
                    break

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
