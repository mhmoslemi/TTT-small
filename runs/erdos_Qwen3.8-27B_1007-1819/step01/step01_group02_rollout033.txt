                    a_opt = (lo_a + hi_a) / 2.0

                        h_new = h_curr.copy()
                        h_new[i] += sign * a_opt
                        h_new[j] -= sign * a_opt
                        h_new = np.clip(h_new, 0.0, 1.0)
                        h_new = project_feasible(h_new, n)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5 - 1e-13:
                            h_curr = h_new
                            best_c5 = c5_new
                            improved = True
        if not improved:
            break
    return h_curr, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_init = len(h0)
    except NameError:
        h0 = np.full(45, 0.5)
        n_init = 45

    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n_init

    # Phase 1: SLSQP warm-up
    n_pts = n_init
    bounds = make_bounds(n_pts)
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_pts,)}]

    starts = [h0.copy()]
    for _ in range(2):
        h_r = h0 + rng.randn(n_pts) * 0.04
        starts.append(project_feasible(h_r, n_pts))

    for beta in [40.0, 100.0, 300.0]:
        for h_start in starts[:2]:
            if elapsed() > budget_s * 0.12:
                break
            res = minimize(objective_smooth, h_start, args=(n_pts, beta),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-16})
            if hasattr(res, 'x') and res.x is not None:
                h_opt = project_feasible(res.x, n_pts)
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
                    best_n = n_pts

    # Phase 2: Active-set LP refinement
    if elapsed() < budget_s * 0.12:
        h_lp = best_h.copy()
        c5_lp = best_c5
        n_lp = best_n
        step_frac = 1.0
        stall_count = 0

        for iteration in range(500):
            if elapsed() > budget_s * 0.60:
                break
            M = min(40 + iteration * 4, 180)
            h_new, c5_new, improved = active_set_lp_step(h_lp, n_lp, M=M, step_frac=step_frac)
            if improved:
                h_lp = h_new
                c5_lp = c5_new
                if c5_lp < best_c5 - 1e-14:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
                    best_n = n_lp
                stall_count = 0
                if iteration % 30 == 29:
                    step_frac = max(0.3, step_frac * 0.85)
            else:
                stall_count += 1
                step_frac *= 0.7
                if step_frac < 0.15:
                    step_frac = 1.0
                if stall_count > 15:
                    break

    # Phase 3: LP again with different step sizes
    if elapsed() < budget_s * 0.45:
        h_lp2 = best_h.copy()
        c5_lp2 = best_c5
        for sf in [0.3, 0.5, 0.7, 1.0, 0.4]:
            if elapsed() > budget_s * 0.70:
                break
            for it in range(100):
                if elapsed() > budget_s * 0.70:
                    break
                h_new, c5_new, imp = active_set_lp_step(h_lp2, best_n, M=100, step_frac=sf)
                if imp:
                    h_lp2 = h_new
                    c5_lp2 = c5_new
                    if c5_lp2 < best_c5 - 1e-14:
                        best_c5 = c5_lp2
                        best_h = h_lp2.copy()
                        best_n = best_n

    # Phase 4: Pairwise coordinate descent
    if elapsed() < budget_s * 0.55:
        h_pd = best_h.copy()
        n_pd = best_n
        n_rounds = 2
        h_pd, c5_pd = pair_descent(h_pd, n_pd, n_rounds=n_rounds, rng=rng)
        if c5_pd < best_c5:
            best_c5 = c5_pd
            best_h = h_pd.copy()
            best_n = n_pd

    # Phase 5: Final SLSQP polish
    if elapsed() < budget_s * 0.75:
        for beta in [200.0, 500.0]:
            if elapsed() > budget_s * 0.90:
                break
            bounds_f = make_bounds(best_n)
            cons_f = [{'type': 'eq', 'fun': constraint_sum, 'args': (best_n,)}]
            res = minimize(objective_smooth, best_h, args=(best_n, beta),
                           method='SLSQP', bounds=bounds_f, constraints=cons_f,
                           options={'maxiter': 800, 'ftol': 1e-16})
            if hasattr(res, 'x') and res.x is not None:
                h_opt = project_feasible(res.x, best_n)
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()

    # Phase 6: One more LP pass
    if elapsed() < budget_s * 0.85:
        for it in range(80):
            if elapsed() > budget_s * 0.95:
                break
            h_new, c5_new, imp = active_set_lp_step(best_h, best_n, M=150, step_frac=0.5)
            if imp:
                best_h = h_new
                best_c5 = c5_new
            else:
                break

    # Final verification
    best_h = np.clip(project_feasible(best_h, best_n), 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
