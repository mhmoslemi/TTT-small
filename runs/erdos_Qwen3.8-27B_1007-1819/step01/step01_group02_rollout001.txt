import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def golden_sequence(n, frac):
    h = np.zeros(n)
    n_ones = int(round(n * frac))
    golden = (1.0 + np.sqrt(5.0)) / 2.0
    used = set()
    for k in range(n_ones * 4):
        pos = int((k * golden) % n)
        if pos not in used:
            used.add(pos)
            h[pos] = 1.0
            if len(used) >= n_ones:
                break
    return h

def thue_morse_seq(n):
    h = np.zeros(n)
    for i in range(n):
        v = 0
        x = i
        while x:
            v ^= 1
            x &= (x - 1)
        h[i] = v
    return h

def smooth_max_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = corr.max()
    return (mx + np.log(np.sum(np.exp(beta * (corr - mx)))) / beta) * dx
