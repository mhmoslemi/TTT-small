def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n + 1):
            r = target - np.sum(h)
            if abs(r) < 1e-10:
                break
            if r > 0:
                i = int(np.argmin(1.0 - h))
                h[i] = min(1.0, h[i] + min(r, 1.0 - h[i]))
            else:
                i = int(np.argmax(h))
                h[i] = max(0.0, h[i] - min(-r, h[i]))
    return h

def smooth_max_val(corr, beta):
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return mx + np.log(s) / beta

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(3 * n):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = np.argsort(1.0 - h)[::-1]
                for ii in idx:
                    amt = min(r, 1.0 - h[ii])
                    h[ii] += amt
                    r -= amt
                    if r < 1e-12:
                        break
            else:
                idx = np.argsort(h)[::-1]
                for ii in idx:
                    amt = min(-r, h[ii])
                    h[ii] -= amt
                    r += amt
                    if r > -1e-12:

    return h

def fourier_optimize(n, n_harmonics, time_limit):
    # h(x) = 0.5 + sum_{k=1}^{K} (a_k*sin(k*pi*x/1) + b_k*cos(k*pi*x/1))
    # where x ∈ [0,2], so fundamental frequency is π/1 = π
    # Actually: x ∈ [0,2], period 2, so fundamental is 2π/2 = π
    # h(x) = 0.5 + Σ (a_k sin(kπx) + b_k cos(kπx))
    # Constraint: ∫₀² h dx = 1 → 0.5*2 + Σ b_k * [sin(kπx)/(kπ)]₀² = 1
    # sin(kπx)/(kπ) evaluated at 2 and 0: sin(2kπ)/(kπ) - sin(0)/(kπ) = 0
    # So the constraint is automatically satisfied! (as long as h is periodic with period 2)
    
    x = np.linspace(0, 2, n, endpoint=False)
    K = n_harmonics
    n_params = 2 * K
    
    def evaluate(params):
        h = np.full(n, 0.5)
        for k in range(1, K+1):
            a_k = params[2*(k-1)]
            b_k = params[2*(k-1)+1]
            h += a_k * np.sin(k * np.pi * x) + b_k * np.cos(k * np.pi * x)
        return np.clip(h, 0, 1)
    
    def obj(params):
        h = evaluate(params)
        # Check sum constraint
        s = np.sum(h) * (2.0/n)
        if abs(s - 1.0) > 0.01:  # penalty for violating constraint
            return 1.0 + 10*(s-1.0)**2
        return compute_c5(h)
    
    x0 = np.zeros(n_params)
    res = minimize(obj, x0, method='Nelder-Mead',
                   options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12})
    h = evaluate(res.x)
    h = fix_feasible(h)
    return h, compute_c5(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(4 * n):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = int(np.argmin(1.0 - h))
                amt = min(r, 1.0 - h[idx])
                h[idx] += amt
            else:
                idx = int(np.argmax(h))
                amt = min(-r, h[idx])
                h[idx] -= amt
    return h

        return np.clip(res.x, 0, 1)

        return h0.copy()

                h = fix_feasible(h)
                cur = bo
                improved = True
        if not improved:
