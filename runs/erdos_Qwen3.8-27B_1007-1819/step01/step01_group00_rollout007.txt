    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n + 1):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = np.argsort(1.0 - h)[-min(5, n):]
                for i in idx:
                    add = min(r, 1.0 - h[i])
                    h[i] += add
                    r -= add
                    if r < 1e-12:
                        break
            else:
                idx = np.argsort(h)[-min(5, n):][::-1]
                for i in idx:
                    sub = min(-r, h[i])
                    h[i] -= sub
                    r += sub
                    if r > -1e-12:
                        break
    return h

    i2 = j_idx[:, None] + k_idx[None, :] - n + 1

    dx = 2.0 / n
    target = n / 2.0
    
    def objective(x):

        # fix sum
        diff = target - np.sum(x)
        if abs(diff) > 1e-10:
            x = x + diff / n
            x = np.clip(x, 0.0, 1.0)
        val, _ = smooth_corr_and_grad(x, beta)
        return val

        x = np.clip(x, 0.0, 1.0)
        diff = target - np.sum(x)
        if abs(diff) > 1e-10:
            x = x + diff / n
            x = np.clip(x, 0.0, 1.0)
        _, g = smooth_corr_and_grad(x, beta)
        return g
    
    res = minimize(objective, h, jac=gradient, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol,
                            'maxfun': max_iter * 3})
    h_new = fix_feasible(res.x)
    return h_new

def coord_refine(h, max_rounds=30, time_limit=None, t0=None):

    cur = compute_c5(h)

                break

                h = fix_feasible(h)
                cur = best_val
                improved = True
