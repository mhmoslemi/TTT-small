        x_old = np.arange(len(best_h)) * 2.0 / len(best_h)
        x_new = np.arange(n_try) * 2.0 / n_try
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        c5_i = compute_c5(h_interp)
        if c5_i < best_c5:
            best_c5 = c5_i
            best_h = h_interp.copy()
            best_n = n_try

        bounds_t = [(0.0, 1.0)] * n_try
        cons_t = [{'type': 'eq', 'fun': con_sum, 'args': (n_try,)}]
        for beta in [50, 120, 250]:
            if elapsed() > budget_s * 0.68:
                break
            res = minimize(
                lse_objective, h_interp, args=(n_try, beta),
                method='SLSQP', bounds=bounds_t,
                constraints=cons_t,
                options={'maxiter': 350, 'ftol': 1e-15}
            )
            if res.x is not None:
                h_opt = project_feasible(res.x, n_try)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n = n_try
                    h_interp = h_opt.copy()

    # Phase 4: Per-coordinate local search with multiple scales
    if elapsed() < budget_s * 0.25:
        h_curr = best_h.copy()
        n_c = len(h_curr)
        deltas = [0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.015, -0.015]
        for epoch in range(6):
            if elapsed() > budget_s * 0.78:
                break
            any_improved = False
            order = list(range(n_c))
            rng.shuffle(order)
            for i in order:
                if elapsed() > budget_s * 0.78:
                    break
                for d in deltas:
                    h_test = h_curr.copy()
                    h_test[i] += d
                    h_test = project_feasible(h_test, n_c)
                    c5_t = compute_c5(h_test)
                    if c5_t < best_c5 - 1e-13:
                        h_curr = h_test
                        best_c5 = c5_t
                        best_h = h_curr.copy()
                        any_improved = True
                        break
            if not any_improved:
                break

    # Phase 5: Pairwise swap search (limited)
    if elapsed() < budget_s * 0.15:
        h_curr = best_h.copy()
        n_c = len(h_curr)
        for epoch in range(3):
            if elapsed() > budget_s * 0.82:
                break
            any_improved = False
            for i in range(0, n_c, 2):
                if elapsed() > budget_s * 0.82:
                    break
                if h_curr[i] <= 0.05 or h_curr[i] >= 0.95:
                    continue
                j = (i + n_c // 2 + rng.randint(0, n_c // 4)) % n_c
                if j == i or h_curr[j] <= 0.05 or h_curr[j] >= 0.95:
                    continue
                for d in [0.03, 0.06]:
                    for sgn in [1.0]:
                        h_test = h_curr.copy()
                        h_test[i] += sgn * d
                        h_test[j] -= sgn * d
                        h_test = np.clip(h_test, 0.0, 1.0)
                        h_test = project_feasible(h_test, n_c)
                        c5_t = compute_c5(h_test)
                        if c5_t < best_c5 - 1e-13:
                            h_curr = h_test
                            best_c5 = c5_t
                            best_h = h_curr.copy()
                            any_improved = True
                            break
                    if any_improved:
                        break
            if not any_improved:
                break

    # Phase 6: Final SLSQP polish with very high beta
    if elapsed() < budget_s * 0.6:
        n_p = best_n
        bounds_f = [(0.0, 1.0)] * n_p
        cons_f = [{'type': 'eq', 'fun': con_sum, 'args': (n_p,)}]
        for beta in [500, 800]:
            if elapsed() > budget_s * 0.85:
                break
            res = minimize(
                lse_objective, best_h, args=(n_p, beta),
                method='SLSQP', bounds=bounds_f,
                constraints=cons_f,
                options={'maxiter': 200, 'ftol': 1e-16}
            )
            if res.x is not None:
                h_opt = project_feasible(res.x, n_p)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
