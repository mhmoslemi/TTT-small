def grad_corr_max(h, epsilon=1e-6):
    n = len(h)
    g = 1.0 - h
    # Compute all correlation values
    corr = np.correlate(h, g, mode='full')
    max_val = np.max(corr)
    # Active set
    active = np.where(corr >= max_val - epsilon)[0]
    # Gradient
    grad = np.zeros(n)
    for m in active:
        L = m - (n-1)
        # grad[j] += (1-h[j+L]) - h[j-L]
        # Vectorized:
        # For j=0..n-1, j+L valid when 0<=j+L<n, j-L valid when 0<=j-L<n
        j = np.arange(n)
        jp = j + L
        jm = j - L
        contrib = np.zeros(n)
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        contrib[mask_p] += 1.0 - h[jp[mask_p]]
        contrib[mask_m] -= h[jm[mask_m]]
        grad += contrib
    grad /= len(active)
    return max_val, grad, active

        for i in range(n):
            orig = h[i]
            for d in [0.08, -0.08, 0.15, -0.15, 0.04, -0.04, 0.25, -0.25]:
                v = orig + d
                if v < 0.0 or v > 1.0:
                    continue
                h[i] = v
                h = project_polytope(h, target)
                c = compute_c5(h)
                if c < cur - 1e-10:
                    cur = c
                    improved = True
                    break
            else:
                h[i] = orig
                h = project_polytope(h, target)
        if not improved:
            break
    return h, cur

def binary_local_search(h, time_limit=30.0):
    n = len(h)
    target = n / 2.0
    h = (h > 0.5).astype(float)
    h = project_polytope(h, target)
    cur = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        improved = False
        for _ in range(200):
            i, j = np.random.choice(n, 2, replace=False)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h)
            if c < cur - 1e-12:
                cur = c
                improved = True
        if not improved:
            break
    return h, cur

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t

def smooth_max_obj(h, beta, target):
    h = project_polytope(h, target)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return (mx + np.log(s) / beta) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    best_h = None
    best_c5 = 1e10

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is not None:
        n0 = len(h_init)
        h_init = project_polytope(h_init, n0 / 2.0)
        c5v = compute_c5(h_init)
        best_c5 = c5v
        best_h = h_init.copy()

    sizes = [100, 80, 60, 48]
    if h_init is not None:
        sizes.remove(100)
        sizes.insert(0, 100)

    elapsed_frac = 0.88
    for n in sizes:
        rem = budget_s * elapsed_frac - (time.time() - t0)
        if rem < 10.0:
            break

        t_starts = []
        if h_init is not None:
            if n == len(h_init):
                t_starts.append(h_init.copy())
            else:
                t_starts.append(upsample(h_init, n))
        t_starts.append(tm_seq(n))
        t_starts.append(project_polytope(0.5 + 0.3 * np.sin(4 * np.pi * np.arange(n) / n) + 0.1 * np.random.randn(n), n / 2.0))
        t_starts.append(project_polytope(np.random.uniform(0, 1, n), n / 2.0))
        if best_h is not None and len(best_h) != n:
            t_starts.append(project_polytope(upsample(best_h, n) + 0.1 * np.random.randn(n), n / 2.0))

        time_per_trial = max(15.0, rem / max(1, len(t_starts)))
        for start_h in t_starts:
            if time.time() - t0 > budget_s * elapsed_frac:
                break
            start_h = project_polytope(start_h, n / 2.0)
            c5v = compute_c5(start_h)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = start_h.copy()

            # Phase 1: Active-set gradient
            rem2 = budget_s * elapsed_frac - (time.time() - t0)
            tl1 = min(time_per_trial * 0.5, max(10.0, rem2 * 0.4))
            h1, c1 = active_set_opt(start_h, time_limit=tl1)
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()

            # Phase 2: SLSQP with smoothing
            if n <= 100 and time.time() - t0 < budget_s * elapsed_frac:
                rem3 = budget_s * elapsed_frac - (time.time() - t0)
                tl2 = min(time_per_trial * 0.3, max(8.0, rem3 * 0.3))
                try:
                    def obj_sqp(x):
                        return smooth_max_obj(x, 150.0, n / 2.0)
                    def sum_eq(x):
                        return np.sum(x) - n / 2.0
                    res = minimize(obj_sqp, h1, method='SLSQP',
                                  constraints=[{'type': 'eq', 'fun': sum_eq}],
                                  bounds=[(0, 1)] * n,
                                  options={'maxiter': 200, 'ftol': 1e-10})
                    h2 = project_polytope(res.x, n / 2.0)
                    c2 = compute_c5(h2)
                    if c2 < best_c5:
                        best_c5 = c2
                        best_h = h2.copy()
                        h1 = h2
                except Exception:
                    pass

            # Phase 3: Coordinate polish
            if time.time() - t0 < budget_s * elapsed_frac:
                rem4 = budget_s * elapsed_frac - (time.time() - t0)
                tl3 = min(25.0, max(5.0, rem4 * 0.2))
                h3, c3 = coord_polish(h1, time_limit=tl3)
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = h3

        # Binary local search at this scale
        if n <= 80 and time.time() - t0 < budget_s * elapsed_frac:
            hb, cb = binary_local_search(best_h if len(best_h) == n else upsample(best_h, n), time_limit=15.0)
            if cb < best_c5:
                best_c5 = cb
                best_h = hb

    # Final refinement on best solution
    if best_h is not None and time.time() - t0 < budget_s * 0.97:
        nb = len(best_h)
        rem_final = budget_s * 0.97 - (time.time() - t0)
        tl_a = max(10.0, rem_final * 0.4)
        h_fin, c_fin = active_set_opt(best_h, time_limit=tl_a, max_iter=8000)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin

    if best_h is not None and len(best_h) <= 100 and time.time() - t0 < budget_s * 0.985:
        rem_cd = budget_s * 0.985 - (time.time() - t0)
        h_cd, c_cd = coord_polish(best_h, time_limit=max(5.0, rem_cd * 0.5), max_rounds=80)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    if best_h is None:
        best_h = project_polytope(tm_seq(60), 30.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
