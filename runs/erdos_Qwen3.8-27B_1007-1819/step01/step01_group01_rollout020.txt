from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3000):

        if diff > 0:

            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)

    corr = get_corr(h)
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    return val

        return smooth_max_obj(params, n, alpha)
    try:
        res = minimize(f, h0.copy(), method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

    curr = compute_c5(h)
    grid = np.linspace(0, 1, grid_size)
    for rd in range(n_rounds):

        idx_order = np.random.permutation(n)
        for i in idx_order:
            old = h[i]
            for rv in grid:
                if abs(rv - old) < 1e-12:
                    continue

                h = project_feasible(h, n)
                v = compute_c5(h)

                    improved = True
            # After trying all grid values, h[i] is at the best one
            curr = compute_c5(h)
            if not improved:

                pass

    curr = compute_c5(h)
    best = h.copy()
    best_val = curr
    T_max = 0.02

    h = h0.copy()
    curr = compute_c5(h)
    best = h.copy()
    best_val = curr
    t_end = t_start + max_time
    not_improved = 0
    while time.time() < t_end:
        corr = get_corr(h)
        # Find active constraint
        k_active = np.argmax(corr)
        # Compute gradient at active constraint
        k_shift = k_active - (n - 1)
        g = np.zeros(n)
        for i in range(n):
            j = i + k_shift
            if 0 <= j < n:
                g[i] += (1.0 - h[j])
            j2 = i - k_shift
            if 0 <= j2 < n:
                g[i] -= h[j2]
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn > 1e-12:
            g /= gn
        
        # Step in negative gradient direction + noise
        progress = (time.time() - t_start) / max(max_time, 1e-9)
        alpha = 0.01 * max(0, 1.0 - progress) + 0.001
        noise_scale = 0.02 * max(0, 1.0 - progress) + 0.002
        h_new = h - alpha * g + np.random.randn(n) * noise_scale
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        
        if c_new < curr:
            h = h_new
            curr = c_new
            not_improved = 0
            if curr < best_val:
                best_val = curr
                best = h.copy()
        else:
            not_improved += 1
            if not_improved > 200:
                # Random restart perturbation
                h_new = h + np.random.randn(n) * 0.05
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3000):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        if not np.any(m):
            break
        h[m] += d / np.count_nonzero(m)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(params, n, alpha):
    h = project_feasible(params, n)
    c = get_corr(h)
    mx = c.max()
    return (mx + np.log(np.sum(np.exp(alpha * (c - mx)))) / alpha) * (2.0 / n)

def refine_slsqp(h0, n, alpha, maxiter=600):
    try:
        r = minimize(smooth_obj, h0.copy(), args=(n, alpha),
                     method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(r.x, n)
    except Exception:
        return h0.copy()

def grad_at_shift(h, s):
    """Gradient of corr at shift s (s = m - (n-1)) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    for i in range(n):
        if 0 <= i + s < n:
            g[i] += 1.0 - h[i + s]
        if 0 <= i - s < n:
            g[i] -= h[i - s]
    return g

def directed_sa(h0, n, t_start, dur):
    h = h0.copy()
    cv = compute_c5(h)
    best = h.copy()
    bv = cv
    t_end = t_start + dur
    stall = 0
    while time.time() < t_end:
        c = get_corr(h)
        # Top-2 active
        top2 = np.argsort(c)[-2:]
        g = sum(grad_at_shift(h, k - (n - 1)) for k in top2) / 2.0
        gn = np.linalg.norm(g)
        prog = min(1.0, (time.time() - t_start) / max(dur, 1e-9))
        step = 0.005 * (1 - prog) + 0.001
        noise = 0.015 * (1 - prog) + 0.003
        if gn > 1e-10:
            h_new = h - step * g / gn + np.random.randn(n) * noise
        else:
            h_new = h + np.random.randn(n) * noise
        h_new = project_feasible(h_new, n)
        c2 = compute_c5(h_new)
        if c2 < cv:
            h, cv = h_new, c2
            stall = 0
            if cv < bv:
                bv, best = cv, h.copy()
        else:
            stall += 1
            if stall > 500:
                idx = np.random.choice(n, 5, replace=False)
                h_new = h.copy()
                h_new[idx] = np.clip(h_new[idx] + np.random.randn(5) * 0.08, 0, 1)
                h_new = project_feasible(h_new, n)
                if compute_c5(h_new) < bv + 0.003:
                    h, cv = h_new, compute_c5(h_new)
                stall = 0
    return best, bv

def fine_cd(h0, n, t_start, dur):
    h = h0.copy()
    cv = compute_c5(h)
    grid = np.linspace(0, 1, 41)
    t_end = t_start + dur
    while time.time() < t_end:
        imp = False
        for i in np.random.permutation(n):
            old = h[i]
            for rv in grid:
                if abs(rv - old) < 1e-10:
                    continue
                h[i] = rv
                h = project_feasible(h, n)
                v = compute_c5(h)
                if v < cv:
                    cv = v
                    imp = True
            cv = compute_c5(h)
        if not imp:
            break
    return h, cv

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(3000):
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0 - 1e-15
        else:
            m = h > 1e-15
        if not np.any(m):
            break
        h[m] += d / np.count_nonzero(m)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(params, n, alpha):
    h = project_feasible(params, n)
    c = get_corr(h)
    mx = np.max(c)
    return (mx + np.log(np.sum(np.exp(alpha * (c - mx)))) / alpha) * (2.0 / n)

def refine_slsqp(h0, n, alpha, maxiter=600):
    try:
        r = minimize(smooth_obj, h0.copy(), args=(n, alpha),
                     method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(r.x, n)
    except Exception:
        return h0.copy()

def grad_at_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    hi_p = h[s:] if s >= 0 else h[:s]
    hi_m = h[-s:] if s > 0 else h[:-s] if s < 0 else h
    if s >= 0:
        g[:len(hi_p)] += 1.0 - hi_p
        g[:len(hi_m)] -= hi_m
    else:
        g[len(hi_p):] += 1.0 - hi_p
        g[len(hi_m):] -= hi_m
    return g

def directed_sa(h0, n, t_start, dur):
    h = h0.copy()
    cv = compute_c5(h)
    best = h.copy()
    bv = cv
    t_end = t_start + dur
    stall = 0
    vel = np.zeros(n)
    beta = 0.85
    while time.time() < t_end:
        c = get_corr(h)
        k_act = int(np.argmax(c))
        s = k_act - (n - 1)
        g = grad_at_shift(h, s)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            g = np.random.randn(n)
            gn = 1.0
        prog = min(1.0, (time.time() - t_start) / max(dur, 1e-9))
        step = 0.005 * (1.0 - 0.7 * prog)
        noise = 0.008 * (1.0 - 0.7 * prog) + 0.002
        vel = beta * vel - step * g / gn
        vel = np.clip(vel, -0.04, 0.04)
        h_new = h + vel + np.random.randn(n) * noise
        h_new = project_feasible(h_new, n)
        c2 = compute_c5(h_new)
        if c2 < cv:
            h, cv = h_new, c2
            stall = 0
            if cv < bv:
                bv = cv
                best = h.copy()
        else:
            stall += 1
            if stall > 350:
                n_esc = int(np.random.choice([5, 10, 15]))
                idx = np.random.choice(n, n_esc, replace=False)
                h_new = h.copy()
                h_new[idx] = np.clip(h_new[idx] + np.random.randn(n_esc) * 0.09, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                v = compute_c5(h_new)
                if v < bv + 0.004:
                    h, cv = h_new, v
                vel = np.zeros(n)
                stall = 0
    return best, bv

def fine_cd(h0, n, t_start, dur, grid_sz):
    h = h0.copy()
    cv = compute_c5(h)
    grid = np.linspace(0.0, 1.0, grid_sz)
    t_end = t_start + dur
    while time.time() < t_end:
        imp = False
        for i in np.random.permutation(n):
            old = h[i]
            for rv in grid:
                if abs(rv - old) < 1e-10:
                    continue
                h[i] = rv
                h = project_feasible(h, n)
                v = compute_c5(h)
                if v < cv - 1e-14:
                    cv = v
                    imp = True
            cv = compute_c5(h)
        if not imp:
            break
    return h, cv
