        for i in order:
            best_v = h[i]
            best_c = curr
            for v in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(v - best_v) < 1e-6:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = v
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_c
        if not improved:
            break
    return h, curr

def bfgs_refine(h, n, alpha, maxiter=800):
    try:
        res = minimize(smooth_obj, h.copy(), args=(n, alpha),
                      method='BFGS', options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_refine(h, n, maxiter=300):
    try:
        res = minimize(exact_obj, h.copy(), args=(n,),
                      method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def perturb(h, n, scale, n_pert):
    h2 = h.copy()
    idxs = np.random.choice(n, size=min(n_pert, n), replace=False)
    for i in idxs:
        h2[i] = np.clip(h2[i] + np.random.randn() * scale, 0.0, 1.0)
    return project_feasible(h2, n)

def threshold_h(h, n, thresh_lo, thresh_hi):
    h2 = h.copy()
    h2[h < thresh_lo] = 0.0
    h2[h > thresh_hi] = 1.0
    return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        best_h = np.full(80, 0.5)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    def tleft():
        return budget_s - 8.0 - (time.time() - t0)

    # Phase 1: BFGS with increasing alpha on initial
    for alpha in [50, 150, 400, 1000, 3000]:
        if tleft() < 80:
            break
        h_try = bfgs_refine(best_h, best_n, alpha, maxiter=600)
        c = compute_c5(h_try)
        if c < best_c5:
            best_c5 = c
            best_h = h_try

    # Phase 2: Coordinate descent on best
    if tleft() > 120:
        best_h, best_c5 = coord_descent(best_h, best_n, n_rounds=25, window=0.2)

    # Phase 3: Try different n values
    n_list = [40, 50, 60, 100, 120, 160, 200]
    for nn in n_list:
        if tleft() < 100:
            break
        if nn == best_n:
            continue
        h_interp = np.interp(np.linspace(0, 1, nn), np.linspace(0, 1, best_n), best_h)
        h_interp = project_feasible(h_interp, nn)
        h_rand = project_feasible(np.clip(0.5 + 0.35 * np.random.randn(nn), 0, 1), nn)
        for h0 in [h_interp, h_rand]:
            if tleft() < 60:
                break
            h_opt = bfgs_refine(h0, nn, 300, maxiter=500)
            if tleft() > 40:
                h_opt, c = coord_descent(h_opt, nn, n_rounds=12, window=0.2)
            else:
                c = compute_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt
                best_n = nn

    # Phase 4: SLSQP polish
    if tleft() > 50:
        h_try = slsqp_refine(best_h, best_n, maxiter=200)
        c = compute_c5(h_try)
        if c < best_c5:
            best_c5 = c
            best_h = h_try

    # Phase 5: Fine coordinate descent (narrow window)
    if tleft() > 40:
        best_h, best_c5 = coord_descent(best_h, best_n, n_rounds=20, window=0.08)

    # Phase 6: Threshold + refine
    if tleft() > 60:
        for tl, th in [(0.15, 0.85), (0.1, 0.9), (0.05, 0.95)]:
            if tleft() < 30:
                break
            h_th = threshold_h(best_h, best_n, tl, th)
            h_th = bfgs_refine(h_th, best_n, 500, maxiter=300)
            h_th, c = coord_descent(h_th, best_n, n_rounds=10, window=0.15)
            if c < best_c5:
                best_c5 = c
                best_h = h_th

    # Phase 7: Random restarts around best
    if tleft() > 80:
        for attempt in range(6):
            if tleft() < 30:
                break
            for scale in [0.05, 0.1, 0.15, 0.2]:
                if tleft() < 20:
                    break
                h_pert = perturb(best_h, best_n, scale, best_n // 4)
                h_pert = bfgs_refine(h_pert, best_n, 500, maxiter=300)
                if tleft() > 25:
                    h_pert, c = coord_descent(h_pert, best_n, n_rounds=8, window=0.12)
                else:
                    c = compute_c5(h_pert)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_pert

    # Phase 8: Final fine polish
    if tleft() > 30:
        best_h, best_c5 = coord_descent(best_h, best_n, n_rounds=15, window=0.05)
    if tleft() > 20:
        h_try = slsqp_refine(best_h, best_n, maxiter=150)
        c = compute_c5(h_try)
        if c < best_c5:
            best_c5 = c
            best_h = h_try
        best_h, best_c5 = coord_descent(best_h, best_n, n_rounds=10, window=0.03)

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
