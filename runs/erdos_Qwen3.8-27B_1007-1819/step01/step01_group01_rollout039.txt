        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_corr_single(h, shift):
    """O(n) gradient of correlation at a specific shift."""
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        jk = j + shift
        jmk = j - shift
        if 0 <= jk < n:
            g += (1.0 - h[jk])
        if 0 <= jmk < n:
            g -= h[jmk]
        grad[j] = g
    return grad

def active_set_optimize(h, n, n_iter=500, topm=5):
    """Optimize using active-set gradient method with line search."""
    h = h.copy()
    curr = compute_c5(h)
    best_h = h.copy()
    best_val = curr
    step_base = 0.3
    for it in range(n_iter):
        corr, dx = compute_all_corrs(h)
        # Identify active shifts
        max_corr = np.max(corr)
        threshold = max_corr - 0.001
        active_idx = np.where(corr > threshold)[0]
        if len(active_idx) == 0:
            active_idx = [np.argmax(corr)]
        active_idx = active_idx[:topm]
        # Weighted gradient (weights proportional to corr deviation)
        weights = np.exp((corr[active_idx] - max_corr) / 0.005)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, idx in zip(weights, active_idx):
            shift = idx - (n - 1)
            grad += w * grad_corr_single(h, shift)
        # Line search with backtracking
        step = step_base
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - step * grad, n)
            c_new = compute_c5(h_new)
            if c_new < curr - 1e-14:
                h = h_new
                curr = c_new
                improved = True
                if curr < best_val:
                    best_val = curr
                    best_h = h.copy()
                break
            step *= 0.5
        if not improved:
            step_base = max(0.01, step_base * 0.8)
        else:
            step_base = min(0.8, step_base * 1.1)
    return best_h, best_val

def coordinate_polish(h, n, n_rounds=40):
    """Coordinate descent with fine grid + parabolic refinement."""
    h = h.copy()
    curr = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for i in range(n):
            best_v = h[i]
            best_c = curr
            # Fine grid search
            base = max(0, min(1, h[i]))
            candidates = np.unique(np.concatenate([
                [base],
                np.linspace(0, 1, 7),
                np.clip(np.array([base-0.08, base-0.04, base+0.04, base+0.08,
                                  base-0.02, base+0.02]), 0, 1)
            ]))
            for v in candidates:
                if abs(v - h[i]) < 1e-10:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = h[i]
                    improved = True
                h[i] = h[i]  # keep projected value for next try
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_c
            if not improved:
                break
        if not improved:
            break
    return h, curr

def simulated_annealing(h, n, budget_s):
    """Adaptive simulated annealing."""
    h = h.copy()
    curr = compute_c5(h)
    best_h = h.copy()
    best_val = curr
    t_end = time.time() + budget_s
    T = 0.004
    step = 0
    n_total = int(budget_s / 0.006)
    for step in range(max(n_total, 50)):
        if time.time() > t_end:
            break
        n_pert = np.random.choice([1, 2, 3, 4], p=[0.5, 0.25, 0.15, 0.1])
        idxs = np.random.choice(n, size=n_pert, replace=False)
        h_new = h.copy()
        for i in idxs:
            h_new[i] = np.clip(h_new[i] + np.random.randn() * 0.06, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        T_cur = T * (1.0 - step / n_total) ** 1.5
        if c_new < curr or np.random.rand() < np.exp(-(c_new - curr) / max(T_cur, 1e-15)):
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best_h = h.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    try:
        init = np.array(initial_h_values, dtype=float)
        n_init = len(init)
    except NameError:
        init = np.full(80, 0.5)
        n_init = 80
    init = project_feasible(init, n_init)
    best_h = init.copy()
    best_c5 = compute_c5(init)
    best_n = n_init
    def time_left():
        return budget_s - 10 - (time.time() - t0)

    # Phase 1: Active-set optimization (main driver)
    if time_left() > 60:
        h_t, c = active_set_optimize(best_h, best_n, n_iter=600, topm=5)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 2: Coordinate polish
    if time_left() > 40:
        h_t, c = coordinate_polish(best_h, best_n, n_rounds=30)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 3: Active-set with more iterations + different topm
    if time_left() > 60:
        for tm in [3, 7, 10]:
            if time_left() < 40:
                break
            h_t, c = active_set_optimize(best_h, best_n, n_iter=300, topm=tm)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()
        h_t, c = coordinate_polish(best_h, best_n, n_rounds=30)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 4: Try different n values
    n_candidates = [best_n, best_n-5, best_n+5, best_n-10, best_n+10,
                    best_n-20, best_n+20, 60, 100, 120, 64, 96, 48]
    n_candidates = sorted(set(x for x in n_candidates if 16 <= x <= 250))
    for n_try in n_candidates:
        if time_left() < 45:
            break
        if n_try == best_n:
            continue
        h_start = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, best_n), best_h)
        h_start = project_feasible(h_start, n_try)
        # Active-set on new n
        h_t, c = active_set_optimize(h_start, n_try, n_iter=200, topm=5)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
            best_n = n_try
        # Quick coordinate polish
        if time_left() > 30:
            h_t, c = coordinate_polish(h_t, n_try, n_rounds=15)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()
                best_n = n_try

    # Phase 5: SLSQP refinement on best
    if time_left() > 40:
        def obj_slsqp(params, nn):
            hh = project_feasible(params, nn)
            return compute_c5(hh)
        try:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - best_n/2}]
            bounds = [(0.0, 1.0)] * best_n
            res = minimize(obj_slsqp, best_h.copy(), args=(best_n,),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 300, 'ftol': 1e-15})
            h_t = project_feasible(res.x, best_n)
            c = compute_c5(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()
        except Exception:
            pass

    # Phase 6: Simulated annealing
    if time_left() > 25:
        sa_budget = min(time_left() - 5, 120)
        h_t, c = simulated_annealing(best_h, best_n, sa_budget)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 7: Final active-set + coordinate polish
    if time_left() > 20:
        h_t, c = active_set_optimize(best_h, best_n, n_iter=400, topm=5)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
        h_t, c = coordinate_polish(best_h, best_n, n_rounds=40)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
        # One more active-set round
        if time_left() > 10:
            h_t, c = active_set_optimize(best_h, best_n, n_iter=200, topm=7)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
