import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def softmax_obj(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    a = np.clip(alpha * (corr - m), -40, 40)
    e = np.exp(a)
    return (m + np.log(e.sum() + 1e-300) / alpha) * dx

def run_lbfgsb(h0, n, alpha, maxiter=400):
    def obj(p):
        h = project_feasible(p, n)
        return softmax_obj(h, n, alpha)
    try:
        res = minimize(obj, h0.copy(), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 20, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def coord_polish(h, n, n_rounds=40):
    h = h.copy()
    curr = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            best_v = old
            best_val = curr
            trials = np.unique(np.concatenate([
                [0.0, 1.0],
                [old + d for d in np.linspace(-0.3, 0.3, 7)]
            ]))
            trials = np.clip(trials, 0, 1)
            for rv in trials:
                if abs(rv - old) < 1e-8:

                h[i] = rv

                    best_val = v
                    best_v = h[i]
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_val
        if not improved:
            break
    return h, curr

    h = h.copy()

            j = (i + 7) % n

                continue

                h[i] = np.clip(ai + eps, 0, 1)
                h[j] = np.clip(aj - eps, 0, 1)
                h = project_feasible(h, n)
                v = compute_c5(h)
                if v < curr:
                    curr = v
                    improved = True
            h[i], h[j] = ai, aj
            h = project_feasible(h, n)
        if not improved:
            break
    return h, curr

    h = h0.copy()

    best_val = curr
