    """Gradient of corr[k_idx] w.r.t. h"""
    n = len(h)
    m = k_idx - (n - 1)  # lag
    grad = np.zeros(n)
    if m >= 0:

        # As h[i]: grad[i] += (1 - h[i-m])
        grad[i_start:i_end] += (1 - h[0 : n-m])
        # As h[i-m]: grad[i-m] -= h[i], for i in [m, n-1], so j=i-m in [0, n-1-m]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def corr_grad(h, k_idx):
    n = len(h)
    m = k_idx - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[m:n] += 1.0 - h[:n-m]
        g[:n-m] -= h[m:]
    else:
        m2 = -m
        g[:n-m2] += 1.0 - h[m2:]
        g[m2:] -= h[:n-m2]
    return g

def smooth_obj(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

def subgrad_opt(h0, n, n_iter, eta0, decay, t_dl):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    stale = 0
    for it in range(n_iter):
        if time.time() > t_dl:
            break
        corr = get_corr(h)
        c = float(np.max(corr)) * (2.0 / n)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 5000 and it < n_iter - 100:
                h = project_feasible(h + 0.1 * np.random.randn(n), n)
                stale = 0
        top3 = np.argsort(corr)[-3:]
        g = np.zeros(n)
        for k in top3:
            g += corr_grad(h, k)
        g /= 3.0
        eta = eta0 / (1.0 + decay * it)
        h = project_feasible(h - eta * g, n)
    return best_h, best_c

def slsqp_run(h0, n, alpha, maxiter, t_dl):
    best_h = h0.copy()
    best_c = compute_c5(h0)
    try:
        res = minimize(smooth_obj, h0.copy(), args=(n, alpha),
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    except Exception:
        pass
    return best_h, best_c

def interp_resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def coord_polish(h0, n, rounds, t_dl):
    h = h0.copy()
    c = compute_c5(h)
    for r in range(rounds):
        if time.time() > t_dl:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() > t_dl:
                break
            old = h[i]
            for rv in np.linspace(0, 1, 9):
                if abs(rv - old) < 1e-6:
                    continue
                h[i] = rv
                h = project_feasible(h, n)
                v = compute_c5(h)
                if v < c - 1e-12:
                    c = v
                    improved = True
                    old = h[i]
            h[i] = old
            h = project_feasible(h, n)
        if not improved:
            break
    return h, c

def binarize_push(h0, n, frac, t_dl):
    h = h0.copy()
    target = n / 2.0
    n_ones = int(round(target))
    order = np.argsort(np.abs(h - 0.5))[::-1]
    idxs = order[:int(frac * n)]
    for i in idxs:
        h[i] = 1.0 if h[i] > 0.5 else 0.0
    h = project_feasible(h, n)
    return h

    for alpha in [50, 150, 400, 1000]:

        h2, c2 = slsqp_run(best_h, best_n, alpha, 300, t_end - 40)

    n_trials = [best_n]
    for dn in [-30, -20, -10, -5, 5, 10, 20, 30, 50]:
        nt = best_n + dn
        if 15 <= nt <= 300:
            n_trials.append(nt)
    n_trials = sorted(set(n_trials))
    for nt in n_trials:
        if tl() < 50:
            break
        h_start = interp_resample(best_h, nt)
        for pert in [0.0, 0.1, 0.2]:
            if tl() < 40:
                break
            h_s = h_start.copy()
            if pert > 0:

            h3, c3 = subgrad_opt(h_s, nt, 20000, 0.03, 0.002, t_end - 60)

    best_c = compute_c5(best_h)
    return best_h, best_c, best_n
