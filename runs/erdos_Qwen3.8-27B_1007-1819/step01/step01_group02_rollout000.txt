import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj_psum(x, n, p):
    h = project_feasible(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    shifted = corr - cmax + 1e-10
    smax = cmax + np.log(np.sum(np.exp(p * shifted)) / len(corr)) / p
    return smax * dx


def smooth_obj_lse(x, n, beta):
    h = project_feasible(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx


def parametric_obj(params, n):
    A, f1, phi1, B, f2, phi2 = params
    xg = np.arange(n) * 2.0 / n
    h = 0.5 + A * np.cos(2.0 * np.pi * f1 * xg + phi1) + B * np.cos(2.0 * np.pi * f2 * xg + phi2)
    h = project_feasible(h, n)
    return compute_c5(h)


def make_starts(n, rng, h_ref):
    starts = []
    if len(h_ref) != n:
        x_old = np.arange(len(h_ref)) * 2.0 / len(h_ref)
        x_new = np.arange(n) * 2.0 / n
        hi = np.interp(x_new, x_old, h_ref)
    else:
        hi = h_ref.copy()
    starts.append(project_feasible(hi, n))
    for _ in range(4):
        hr = hi + rng.randn(n) * 0.12
        starts.append(project_feasible(hr, n))
    xg = np.arange(n) * 2.0 / n
    for period in [4, 6, 8, 10, 12]:
        hblk = np.zeros(n)
        half = period // 2
        for i in range(n):
            hblk[i] = 1.0 if (i % period) < half else 0.0
        starts.append(project_feasible(hblk + rng.randn(n) * 0.04, n))
    for freq in [2, 3, 4, 5, 6]:
        for amp in [0.2, 0.3, 0.35]:
            hcos = 0.5 + amp * np.cos(2.0 * np.pi * freq * xg)
            starts.append(project_feasible(hcos, n))
    return starts


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_init = len(h0)
    except NameError:
        h0 = np.full(90, 0.5)
        n_init = 90

    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n_init

    # Multi-resolution schedule
    n_schedule = sorted(set([45, 90, n_init, 180]))
    n_schedule = [nn for nn in n_schedule if nn <= 360]

    current_h = h0.copy()

    for n_pts in n_schedule:
        if elapsed() > budget_s * 0.65:
            break

        if len(current_h) != n_pts:
            x_old = np.arange(len(current_h)) * 2.0 / len(current_h)
            x_new = np.arange(n_pts) * 2.0 / n_pts
            interp_h = np.interp(x_new, x_old, current_h)
        else:
            interp_h = current_h.copy()

        starts = make_starts(n_pts, rng, interp_h)
        bounds = [(0.0, 1.0)] * n_pts

        for h_start in starts:
            if elapsed() > budget_s * 0.65:
                break

            work_h = h_start.copy()
            for p in [8, 20, 50]:
                if elapsed() > budget_s * 0.65:
                    break

                def obj_p(x, _n=n_pts, _p=p):
                    return smooth_obj_psum(x, _n, _p)

                res = minimize(obj_p, work_h, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-13, 'maxfun': 8000})
                if res.x is not None:
                    h_opt = project_feasible(res.x, n_pts)
                    c5_val = compute_c5(h_opt)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_opt.copy()
                        best_n = n_pts
                        current_h = h_opt.copy()
                    work_h = h_opt.copy()

    # SLSQP refinement with LSE at best resolution
    if elapsed() < budget_s * 0.5:
        n_pts = best_n
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n_pts / 2.0}]

        for beta in [30, 80, 150, 300]:
            if elapsed() > budget_s * 0.7:
                break

            def obj_lse(x, _n=n_pts, _b=beta):
                return smooth_obj_lse(x, _n, _b)

            res = minimize(obj_lse, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            if res.x is not None:
                h_opt = project_feasible(res.x, n_pts)
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()

    # Parametric search for better structure
    if elapsed() < budget_s * 0.45:
        n_pts = max(best_n, 180)
        p_bounds = [(0.0, 0.5), (1.0, 8.0), (0.0, 2.0 * np.pi),
                    (0.0, 0.4), (1.0, 10.0), (0.0, 2.0 * np.pi)]
        p_start = np.array([0.3, 3.0, 0.0, 0.15, 5.0, 1.0])
        res = minimize(lambda p: parametric_obj(p, n_pts), p_start,
                       method='Nelder-Mead', bounds=p_bounds,
                       options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
        if res.fun < best_c5:
            xg = np.arange(n_pts) * 2.0 / n_pts
            A, f1, phi1, B, f2, phi2 = res.x
            hp = 0.5 + A * np.cos(2.0 * np.pi * f1 * xg + phi1) + B * np.cos(2.0 * np.pi * f2 * xg + phi2)
            hp = project_feasible(hp, n_pts)
            c5_p = compute_c5(hp)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = hp.copy()
                best_n = n_pts

    # Coordinate descent with adaptive steps
    if elapsed() < budget_s * 0.3:
        n_pts = best_n
        h_curr = best_h.copy()
        step_sizes = [0.01, 0.005, 0.002, 0.001, 0.0005]
        for step in step_sizes:
            if elapsed() > budget_s * 0.72:
                break
            for _ in range(3):
                if elapsed() > budget_s * 0.72:
                    break
                improved = False
                order = np.random.permutation(n_pts)
                for i in order:
                    for d in [step, -step]:
                        h_test = h_curr.copy()
                        h_test[i] = np.clip(h_test[i] + d, 0.0, 1.0)
                        h_test = project_feasible(h_test, n_pts)
                        c5_t = compute_c5(h_test)
                        if c5_t < best_c5 - 1e-12:
                            h_curr = h_test.copy()
                            best_c5 = c5_t
                            best_h = h_curr.copy()
                            improved = True
                            break
                if not improved:
                    break

    # Binary snap heuristics
    if elapsed() < budget_s * 0.55:
        n_pts = best_n
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n_pts)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        # Try sorting-based partial binary
        idx = np.argsort(best_h)
        h_pb = np.zeros(n_pts)
        h_pb[idx[n_pts // 2:]] = 1.0
        h_pb = project_feasible(h_pb, n_pts)
        c5_pb = compute_c5(h_pb)
        if c5_pb < best_c5:
            best_c5 = c5_pb
            best_h = h_pb.copy()

    # Final large-beta SLSQP polish
    if elapsed() < budget_s * 0.25:
        n_pts = best_n
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n_pts / 2.0}]

        for beta in [200, 500]:
            if elapsed() > budget_s * 0.8:
                break
            def obj_final(x, _n=n_pts, _b=beta):
                return smooth_obj_lse(x, _n, _b)
            res = minimize(obj_final, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 400, 'ftol': 1e-15})
            if res.x is not None:
                h_opt = project_feasible(res.x, n_pts)
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()

    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
