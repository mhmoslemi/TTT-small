        return fix_feasible(res.x)
    except Exception:

                h[j] = h[j] - best_dv
                h = fix_feasible(h)
                cur = best_c5
                improved = True
        if not improved:
            break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n + 2):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for i in order:
                if r > 0:
                    amt = min(1.0 - h[i], r)
                    h[i] += amt
                else:
                    amt = min(h[i], -r)
                    h[i] -= amt
                r = target - np.sum(h)
                if abs(r) < 1e-12:
                    break
    return h

def smooth_max_val(h, beta):
    corr = get_corr(h)
    n = len(h)
    dx = 2.0 / n
    mx = np.max(corr)
    shifted = corr - mx
    ms = np.max(shifted)
    exp_arg = beta * (shifted - ms)
    exp_arg = np.clip(exp_arg, -500, 0)
    s = np.sum(np.exp(exp_arg))
    return (mx + ms + np.log(s) / beta) * dx

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_run(h, beta, maxiter):
    n = len(h)
    def obj(x):
        xf = fix_feasible(x)
        return smooth_max_val(xf, beta)
    try:
        res = minimize(obj, h, method='SLSQP',
                      constraints=[{'type': 'eq', 'fun': sum_eq}],
                      bounds=[(0, 1)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return fix_feasible(res.x)
    except Exception:
        return h

def coord_descent(h, max_rounds=25):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > 0:
            pass
        improved = False
        for i in range(n):
            hi = h[i]
            for d in [0.04, 0.08, 0.12, 0.2]:
                for sign in [1, -1]:
                    vi = hi + sign * d
                    if vi < 0 or vi > 1:
                        continue
                    if sign > 0:
                        j = int(np.argmax(h))
                    else:
                        j = int(np.argmin(h))
                    if j == i:
                        j = (i + 1) % n
                    vj = h[j] - sign * d
                    if vj < 0 or vj > 1:
                        continue
                    h[i] = vi
                    h[j] = vj
                    c5 = compute_c5(h)
                    if c5 < cur - 1e-12:
                        cur = c5
                        improved = True
                        break
                    h[i] = hi
                    h[j] = h[j] + sign * d
                else:
                    continue
                break
        if not improved:
            break
    return h

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t

def gen_start(n, rng, parent):
    r = rng.random()
    if r < 0.2:
        return fix_feasible(tm_seq(n))
    elif r < 0.4:
        x = np.arange(n)
        freq = int(rng.randint(2, 7))
        return fix_feasible(0.5 + 0.35 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 6.28)))
    elif r < 0.6:
        blk = max(2, n // int(rng.randint(4, 10)))
        h = np.zeros(n)
        ph = int(rng.randint(0, blk * 2))
        for i in range(n):
            h[i] = 1.0 if (i // blk + ph) % 2 == 0 else 0.0
        return fix_feasible(h)
    elif r < 0.8 and parent is not None:
        return fix_feasible(parent + 0.15 * rng.randn(n))
    else:
        return fix_feasible(rng.uniform(0.2, 0.8, n))
