import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        z_test = np.clip(x - mid, 0.0, 1.0)
        if z_test.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def obj_smooth(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx


def cons_sum(x, n_pts):
    return x.sum() - n_pts / 2.0


def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)


def subgrad_step(h, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_idx = np.argmax(corr)
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    jm = j - k
    jp = j + k
    for j in range(n):
        g = 0.0
        if 0 <= j - k < n:
            g += 1.0 - h[j - k]
        if 0 <= j + k < n:
            g -= h[j + k]
        grad[j] = g
    return grad


def subgrad_descent(h, n, max_iter, t0, deadline):
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr)
    running = h_curr.copy()
    momentum = np.zeros(n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = subgrad_step(h_curr, n)
        alpha = 0.3 / np.sqrt(it + 1)
        momentum = 0.9 * momentum + 0.1 * grad
        h_new = project_feasible(h_curr - alpha * momentum, n)
        h_curr = h_new
        running += h_new
        if (it + 1) % 20 == 0:
            h_avg = project_feasible(running / (it + 1), n)
            c5_avg = compute_c5(h_avg)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = h_avg.copy()
    return best_h, best_c5


def coord_descent(h, n, t0, deadline):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    steps_large = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    steps_small = [0.001, -0.001, 0.0005, -0.0005, 0.0002, -0.0002]
    
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h_curr, best_c5
            for step in steps_large:
                h_test = h_curr.copy()
                h_test[i] = np.clip(h_test[i] + step, 0.0, 1.0)
                h_test = project_feasible(h_test, n)
                c5_t = compute_c5(h_test)
                if c5_t < best_c5 - 1e-12:
                    h_curr = h_test
                    best_c5 = c5_t
                    improved = True
                    break
        if not improved:
            fine_improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h_curr, best_c5
                for step in steps_small:
                    h_test = h_curr.copy()
                    h_test[i] = np.clip(h_test[i] + step, 0.0, 1.0)
                    h_test = project_feasible(h_test, n)
                    c5_t = compute_c5(h_test)
                    if c5_t < best_c5 - 1e-13:
                        h_curr = h_test
                        best_c5 = c5_t
                        fine_improved = True
                        break
            if not fine_improved:
                break
    return h_curr, best_c5


def pair_coord_descent(h, n, t0, deadline, rng):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    while time.time() < deadline:
        improved = False
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1
        for di in [0.01, -0.01, 0.005, -0.005]:
            for dj in [0.01, -0.01, 0.005, -0.005]:
                h_test = h_curr.copy()
                h_test[i] = np.clip(h_test[i] + di, 0.0, 1.0)
                h_test[j] = np.clip(h_test[j] + dj, 0.0, 1.0)
                h_test = project_feasible(h_test, n)
                c5_t = compute_c5(h_test)
                if c5_t < best_c5 - 1e-12:
                    h_curr = h_test
                    best_c5 = c5_t
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h_curr, best_c5
