    # Simplified: for valid i, j=i+LAG gives +, j=i gives -
    
    grad_h = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)
    for mi in range(2*n-1):
        w = weights[mi] * dx
        if abs(w) < 1e-15:
            continue
        lag = lags[mi]
        # Valid i range: max(0,-lag) <= i <= min(n-1, n-1-lag)
        i_lo = max(0, -int(lag))
        i_hi = min(n-1, n-1-int(lag))
        for i in range(i_lo, i_hi+1):
            j_plus = i + int(lag)
            j_minus = i
            grad_h[j_plus] += w * (1.0 - h[j_minus])
            grad_h[j_minus] -= w * h[j_plus]
    
    return smax, grad_h

    grad_h = np.zeros(n)
    for mi in max_k:
        lag = int(lags[mi])
        i_lo = max(0, -lag)

        # Wait, need to be more careful

def sa_search(h, n, budget_s):

    h = h.copy()

    best = h.copy()

        T = 0.01 * (1 - progress) ** 2

        block_size = np.random.choice([1, 2, 3, 4, 6, 8], p=[0.3, 0.25, 0.2, 0.1, 0.1, 0.05])
        start = np.random.randint(0, max(1, n - block_size + 1))

        h_new[start:start+block_size] = np.clip(h_new[start:start+block_size] + rng, 0, 1)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new)
        if c_new < curr or np.random.rand() < np.exp(-(c_new - curr) / max(T, 1e-15)):
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best = h.copy()
    
    return best, best_val

def get_corr(h):

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

        else:

            break
        h[mask] += diff / np.count_nonzero(mask)
        np.clip(h, 0.0, 1.0, out=h)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -400, 400)
    exps = np.exp(shifted)
    S = np.sum(exps)
    smax = (m + np.log(S) / alpha) * dx
    weights = exps / S
    grad_h = np.zeros(n)
    lags = np.arange(2 * n - 1) - (n - 1)
    for mi in range(2 * n - 1):
        w = weights[mi] * dx
        if abs(w) < 1e-14:
            continue
        lag = int(lags[mi])
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        for i in range(i_lo, i_hi + 1):
            jp = i + lag
            jn = i
            grad_h[jp] += w * (1.0 - h[jn])
            grad_h[jn] -= w * h[jp]
    return smax, grad_h

def slsqp_opt(h0, n, alpha, maxiter=200):
    def fun(x):
        h = project_feasible(x, n)
        return smooth_max_and_grad(h, alpha)[0]
    def jac(x):
        h = project_feasible(x, n)
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    x0 = project_feasible(h0, n)
    try:
        res = minimize(fun, x0, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return x0

def sharpen(h, n):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in [1.2, 1.5, 2.0, 3.0, 5.0]:
        h_s = np.clip(0.5 + t * (h - 0.5), 0.0, 1.0)

        if c < best_c5:

            best_h = h_s.copy()
    return best_h, best_c5

        bs = min(bs, n - 1)
        start = np.random.randint(0, n - bs + 1)
        h_new = h.copy()
        h_new[start:start + bs] = np.clip(h_new[start:start + bs] + np.random.randn(bs) * 0.08, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < curr or np.random.rand() < np.exp(-(c_new - curr) / max(T, 1e-15)):
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best = h.copy()
    return best, best_val

def subgrad_descent(h, n, budget_s):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    step = 0.04
    dx = 2.0 / n
    lags = np.arange(2 * n - 1) - (n - 1)
    while time.time() - t_start < budget_s:
        corr = get_corr(h)
        max_v = np.max(corr)
        max_idx = np.where(corr >= max_v - 1e-10)[0]
        grad_h = np.zeros(n)
        for mi in max_idx:
            lag = int(lags[mi])
            i_lo = max(0, -lag)
            i_hi = min(n - 1, n - 1 - lag)
            for i in range(i_lo, i_hi + 1):
                jp = i + lag
                jn = i
                grad_h[jp] += (1.0 - h[jn])
                grad_h[jn] -= h[jp]
        grad_h *= dx / len(max_idx)
        h = project_feasible(h - step * grad_h, n)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        step *= 0.997
    return best_h, best_c5
