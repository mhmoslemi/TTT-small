        rem = budget_s * 0.68 - (time.time() - t0)
        if rem < 15:
            break
        n_starts = max(1, min(3, int(rem / max(80, n * 0.8))))
        for v in range(n_starts + 1):
            if time.time() - t0 > budget_s * 0.72:
                break
            if h_parent is not None and n == len(h_parent) and v == 0:
                h = h_parent.copy()
            else:
                h = gen_start(n, v, best_h, rng)
            h = fix_feasible(h)
            c5v = compute_c5(h)
            if c5v < best_c5:
                best_c5 = c5v; best_h = h.copy()
            for beta in [25, 80, 250]:
                if time.time() - t0 > budget_s * 0.75:
                    break
                h = run_slsqp(h, beta, 350, 1e-9)
                c5v = compute_c5(h)
                if c5v < best_c5:
                    best_c5 = c5v; best_h = h.copy()
            if n <= 56 and time.time() - t0 < budget_s * 0.80:
                h_bin, c5_bin = structured_bin_search(h, 40, rng)
                if c5_bin < best_c5:
                    best_c5 = c5_bin; best_h = h_bin.copy()
                h_rf = run_slsqp(h_bin, 150, 250, 1e-10)
                c5v = compute_c5(h_rf)
                if c5v < best_c5:
                    best_c5 = c5v; best_h = h_rf.copy()
            if n <= 44 and time.time() - t0 < budget_s * 0.84:
                steps = [0.02, -0.02, 0.04, -0.04, 0.07, -0.07, 0.01, -0.01]
                h_cd, c5_cd = coord_descent(h, 10, steps)
                if c5_cd < best_c5:
                    best_c5 = c5_cd; best_h = h_cd.copy()
    # Phase 2: Intensive refinement of best
    if best_h is not None and time.time() - t0 < budget_s * 0.96:
        nb = len(best_h)
        for beta in [300, 700, 1500]:
            if time.time() - t0 > budget_s * 0.96:
                break
            h_f = run_slsqp(best_h, beta, 500, 1e-11)
            c5f = compute_c5(h_f)
            if c5f < best_c5:
                best_c5 = c5f; best_h = h_f
        if nb <= 64 and time.time() - t0 < budget_s * 0.975:
            steps = [0.01, -0.01, 0.02, -0.02, 0.03, -0.03, 0.005, -0.005]
            h_cd, c5_cd = coord_descent(best_h, 20, steps)
            if c5_cd < best_c5:
                best_c5 = c5_cd; best_h = h_cd
        if nb <= 72 and time.time() - t0 < budget_s * 0.99:
            h_bin, c5_bin = structured_bin_search(best_h, 60, rng)
            if c5_bin < best_c5:
                best_c5 = c5_bin; best_h = h_bin
            h_rf = run_slsqp(h_bin, 500, 300, 1e-11)
            c5f = compute_c5(h_rf)
            if c5f < best_c5:
                best_c5 = c5f; best_h = h_rf
    if best_h is None:
        best_h = fix_feasible(np.full(48, 0.5))
        best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
