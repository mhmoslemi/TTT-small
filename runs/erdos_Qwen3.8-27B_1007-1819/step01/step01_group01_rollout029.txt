    curr = compute_c5(h)
    best_h = h.copy()
    best_val = curr

        # Generate batch of perturbations
        batch = np.zeros((batch_size, n))
        for i in range(batch_size):
            n_pert = rng.integers(1, 6)
            idxs = rng.integers(0, n, size=n_pert)
            h_pert = h.copy()
            h_pert[idxs] += rng.normal(0, 0.1, size=n_pert)
            batch[i] = project_feasible(h_pert, n)
        
        # Evaluate all
        best_in_batch = curr
        best_idx = -1
        for i in range(batch_size):
            c = compute_c5(batch[i])
            if c < best_in_batch:
                best_in_batch = c
                best_idx = i
        
        if best_idx >= 0:
            h = batch[best_idx].copy()
            curr = best_in_batch
            if curr < best_val:
                best_val = curr
                best_h = h.copy()
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_lse(corr, alpha):
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -400, 400)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def obj_smooth(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_lse(corr, alpha) * dx

def obj_exact(params, n):
    h = project_feasible(params, n)
    return compute_c5(h)

def run_slsqp(h0, n, alpha, maxiter=400, ftol=1e-13):
    try:
        res = minimize(obj_smooth, h0.copy(), args=(n, alpha),
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

            h_pert = project_feasible(h_pert, n)

                h = h_pert
                curr = c
                improvements += 1
                if curr < best_val:
                    best_val = curr
                    best_h = h.copy()
        if improvements == 0 and it > 20:
            scale *= 0.9
    return best_h, best_val
