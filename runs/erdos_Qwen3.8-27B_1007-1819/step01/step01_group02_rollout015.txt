import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj(h, beta, lam=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    val = (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx
    if lam > 0:
        val += lam * (h.sum() - n / 2.0) ** 2
    return val

def subgrad_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    lo1, hi1 = max(0, lag), min(n, n + lag)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[np.arange(lo1, hi1) - lag])
    lo2, hi2 = max(0, -lag), min(n, n - lag)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[np.arange(lo2, hi2) + lag]
    return grad * dx

def projected_subgrad_descent(h, n_steps=500, lr_init=0.01):
    n = len(h)
    h_curr = h.copy()
    best = compute_c5(h_curr)
    lr = lr_init
    for step in range(n_steps):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        k_max = np.argmax(corr)
        lag = k_max - (n - 1)
        sg = subgrad_lag(h_curr, lag)
        h_new = h_curr - lr * sg
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best - 1e-10:

            best = c5_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def project_template(t, p):
    target = p / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(t - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(t - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = beta * (corr - m)
    val = (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    if lam > 0:
        val += lam * (h.sum() - n / 2.0) ** 2
    return val

    pen = 10.0 * (np.sum(np.maximum(0.0, -t)**2) + np.sum(np.maximum(0.0, t - 1.0)**2))
    pen += 5.0 * (t.sum() - p / 2.0) ** 2
    return c5 + pen

def subgrad_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    lo1, hi1 = max(0, lag), min(n, n + lag)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[idx] += (1.0 - h[idx - lag])
    lo2, hi2 = max(0, -lag), min(n, n - lag)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] -= h[idx + lag]
    return grad * dx
