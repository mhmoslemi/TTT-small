def compute_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    weights = exps / np.sum(exps)
    
    # For each lag k (index in corr array), m_k = k - (n-1)
    lags = np.arange(2*n - 1) - (n - 1)  # shape (2n-1,)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-15:
            continue
        mk = lags[k]
        w = weights[k]
        # d c_k / d h[j] = (1-h[j-mk]) * [0<=j-mk<n] - h[j+mk] * [0<=j+mk<n]
        if mk >= 0:
            # j-mk >= 0 means j >= mk, j-mk < n means j < n+mk (always true for j<n, mk>=0)
            # j+mk < n means j < n-mk
            # Term 1: (1-h[j-mk]) for j in [mk, n-1]
            # Term 2: -h[j+mk] for j in [0, n-mk-1]
            for j in range(mk, n):
                grad[j] += w * (1.0 - h[j - mk])
            for j in range(0, n - mk):
                grad[j] += w * (-h[j + mk])
        elif mk < 0:
            # Let p = -mk > 0
            # j-mk = j+p >= 0 always, j+p < n means j < n-p
            # j+mk = j-p >= 0 means j >= p, j-p < n always
            # Term 1: (1-h[j+p]) for j in [0, n-p-1]
            # Term 2: -h[j-p] for j in [p, n-1]

            for j in range(0, n - p):
                grad[j] += w * (1.0 - h[j + p])
            for j in range(p, n):
                grad[j] += w * (-h[j - p])
        else:
            # mk = 0: d c_k / d h[j] = (1-h[j]) - h[j] = 1 - 2h[j]
            grad += w * (1.0 - 2.0 * h)
    
    return grad * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / np.sum(exps)
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        mk = lags[k]
        if mk == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif mk > 0:
            # d ck/dh[j]: (1-h[j-mk]) for j in [mk, n-1], -h[j+mk] for j in [0, n-mk-1]
            grad[mk:] += wk * (1.0 - h[:n - mk])
            grad[:n - mk] += wk * (-h[mk:])
        else:
            p = -mk
            # (1-h[j+p]) for j in [0, n-p-1], -h[j-p] for j in [p, n-1]
            grad[:n - p] += wk * (1.0 - h[p:])
            grad[p:] += wk * (-h[:n - p])
    return grad * dx

def obj_smooth_p(params, n, alpha):
    h = np.clip(params, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return smooth_max_val(corr, alpha)

def grad_smooth_p(params, n, alpha):
    h = np.clip(params, 0, 1)
    return grad_smooth_vec(h, alpha)

def lbfgsb_round(h, n, alphas, maxiter_each):
    for a in alphas:
        h = project_feasible(h, n)
        try:
            res = minimize(
                obj_smooth_p, h.copy(),
                args=(n, a), jac=lambda p, nn, aa: grad_smooth_p(p, nn, aa),
                method='L-BFGS-B', bounds=[(0, 1)] * n,
                options={'maxiter': maxiter_each, 'ftol': 1e-14, 'gtol': 1e-9}
            )
            h = project_feasible(res.x, n)
        except Exception:
            pass
    return h

def coordinate_polish(h, n, max_rounds=30):
    h = h.copy()
    curr, _ = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            best_v, best_c = old, curr
            for rv in np.linspace(max(0, old - 0.15), min(1, old + 0.15), 9):
                if abs(rv - old) < 1e-10:
                    continue
                h[i] = rv
                h = project_feasible(h, n)
                c, _ = compute_c5(h)
                if c < best_c:
                    best_c, best_v = c, h[i]
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_c
        if not improved:
            break
    return h, curr

def sa_refine(h, n, budget_s, rng):
    h = h.copy()
    curr, _ = compute_c5(h)
    best_h, best_val = h.copy(), curr
    T0 = 0.006
    t_start = time.time()
    step = 0
    while time.time() - t_start < budget_s:
        step += 1
        T = T0 * max(0.01, 1.0 - step * 0.002)
        n_pert = rng.choice([1, 2, 3, 4], p=[0.4, 0.3, 0.2, 0.1])
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_new = h.copy()
        for i in idxs:
            h_new[i] = np.clip(h_new[i] + rng.randn() * 0.06, 0, 1)
        h_new = project_feasible(h_new, n)
        c_new, _ = compute_c5(h_new)
        if c_new < curr or rng.rand() < np.exp(-(c_new - curr) / max(T, 1e-15)):
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best_h = h.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 8 - (time.time() - t0)

    try:
        init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(init)
    except NameError:
        n_init = 60
        init = np.interp(np.linspace(0, 1, n_init), np.linspace(0, 1, 40),
                         np.clip(0.5 + 0.3 * np.sin(2 * np.pi * np.linspace(0, 2, 40)), 0, 1))

    init = project_feasible(init, n_init)
    best_h = init.copy()
    best_c5, _ = compute_c5(init)
    best_n = n_init

    # Phase 1: L-BFGS-B from initial with alpha schedule
    if time_left() > 120:
        h_opt = lbfgsb_round(best_h, best_n, [40, 80, 150, 300, 600, 1200], 150)
        c, _ = compute_c5(h_opt)
        if c < best_c5:
            best_c5, best_h = c, h_opt.copy()

    # Phase 2: Try different n values (interpolate best to new n, optimize)
    n_cands = sorted(set([best_n] + list(range(max(30, best_n - 40), min(140, best_n + 41), 10))))
    for n in n_cands:
        if time_left() < 60:
            break
        if n == best_n:
            continue
        h_start = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, best_n), best_h)
        h_start = project_feasible(h_start, n)
        h_opt = lbfgsb_round(h_start, n, [100, 300, 800], 120)
        c, _ = compute_c5(h_opt)
        if c < best_c5:
            best_c5, best_h, best_n = c, h_opt.copy(), n

    # Phase 3: Second L-BFGS-B round with higher alpha
    if time_left() > 80:
        h_opt = lbfgsb_round(best_h, best_n, [500, 1500, 3000], 100)
        c, _ = compute_c5(h_opt)
        if c < best_c5:
            best_c5, best_h = c, h_opt.copy()

    # Phase 4: Simulated annealing
    if time_left() > 40:
        sa_time = min(40, time_left() - 5)
        h_t, c_t = sa_refine(best_h, best_n, sa_time, rng)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t.copy()

    # Phase 5: Coordinate polish
    if time_left() > 20:
        h_t, c_t = coordinate_polish(best_h, best_n, max_rounds=40)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t.copy()

    # Phase 6: Final L-BFGS-B + coordinate
    if time_left() > 15:
        h_t = lbfgsb_round(best_h, best_n, [5000], 80)
        h_t, c_t = coordinate_polish(h_t, best_n, max_rounds=20)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, best_n
