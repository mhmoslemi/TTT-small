def weighted_subgrad(h, n, top_k=8, beta=30):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode="full") * dx
    # Get top lags
    idx = np.argsort(corr)[-top_k:]
    w = np.exp(beta * (corr[idx] - corr[idx].max()))
    w /= w.sum()
    
    grad = np.zeros(n)
    for ii, i in enumerate(idx):
        k = i - (n - 1)
        wi = w[ii]
        if k >= 0 and k < n:
            grad[k:] += wi * (1.0 - h[:n-k])
            grad[:n-k] -= wi * h[k:]
        elif k < 0:
            M = -k
            if M < n:
                grad[:n-M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:n-M]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_obj_with_sum(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta + 10.0 * (h.sum() - n_pts / 2.0) ** 2

def topk_subgrad(h, n, top_k=8, beta=40):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = np.argsort(corr)[-top_k:]
    w = np.exp(beta * (corr[idx] - corr[idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for ii, i in enumerate(idx):
        k = i - (n - 1)
        wi = w[ii]
        if k >= 0:
            if k < n:
                grad[k:] += wi * (1.0 - h[:n - k])
                grad[:n - k] -= wi * h[k:]
        else:
            M = -k
            if M < n:
                grad[:n - M] += wi * (1.0 - h[M:])
                grad[M:] -= wi * h[:n - M]
    return grad * dx - grad.mean() * 0  # centering handled by projection

def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)

def lbfgs_refine(h_init, n, beta, max_iter=300, tol=1e-14):
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n / 2.0, 'args': ()}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        smooth_obj, h_init.copy(), args=(n, beta),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8, 'maxfun': max_iter * 3}
    )
    if hasattr(res, 'x') and res.x is not None:
        return project_feasible(res.x, n), res
    return h_init, None

def subgrad_descent(h_init, n, t_budget=60.0):
    t0 = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.02
    momentum = np.zeros(n)
    mom_alpha = 0.7
    patience = 0
    for it in range(50000):
        if time.time() - t0 > t_budget:
            break
        c5_cur = compute_c5(h)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 500:
                lr *= 0.9
                if lr < 1e-5:
                    break
        grad = topk_subgrad(h, n, top_k=8, beta=40)
        momentum = mom_alpha * momentum + (1 - mom_alpha) * grad
        h_new = h - lr * momentum
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_cur + lr * np.max(np.abs(grad)) * 0.5:
            h = h_new
        else:
            lr *= 0.8
            momentum = 0.5 * momentum
            h = project_feasible(h - 0.5 * lr * momentum, n)
    return best_h, best_c5

def coord_descent(h_init, n, t_budget=30.0):
    t0 = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    step_sizes = [0.01, 0.005, 0.002, 0.001]
    for epoch in range(50):
        if time.time() - t0 > t_budget:
            break
        improved = False
        for s in step_sizes:
            if time.time() - t0 > t_budget:
                break
            for i in range(n - 1):
                for j in range(i + 1, min(i + 4, n)):
                    for di in [s, -s]:
                        for dj in [s, -s]:
                            h_test = h.copy()
                            h_test[i] += di
                            h_test[j] += dj
                            h_test = project_feasible(h_test, n)
                            c5_t = compute_c5(h_test)
                            if c5_t < best_c5 - 1e-13:
                                h = h_test
                                best_c5 = c5_t
                                improved = True
        if not improved:
            break
    return h, best_c5
