import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n * 0.5
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_val(corr, alpha):
    m = corr.max()
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def slsqp_smooth(h0, n, alpha, maxiter=300):
    def obj(p):
        h = project_feasible(p, n)
        return smooth_max_val(compute_corr(h), alpha) * (2.0 / n)
    try:
        res = minimize(obj, h0.copy(), method='SLSQP',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def local_search(h0, n, n_iters, rng):
    h = h0.copy()
    curr = compute_c5(h)
    best = h.copy()
    best_val = curr
    T0 = 0.006
    for it in range(n_iters):
        r = rng.rand()
        h_new = h.copy()
        if r < 0.35:
            L = rng.randint(2, max(3, n // 5))
            s = rng.randint(0, n - L)
            h_new[s:s+L] += rng.randn(L) * 0.10
        elif r < 0.65:
            i, j = rng.choice(n, 2, replace=False)
            a = rng.uniform(-0.5, 0.5)
            h_new[i] += a
            h_new[j] -= a
        elif r < 0.85:
            i = rng.randint(n)
            h_new[i] += rng.randn() * 0.15
        else:
            # Targeted: perturb coords most affecting worst lags
            corr = compute_corr(h)
            worst = np.argsort(corr)[-3:]
            for k in worst:
                shift = k - (n - 1)
                if shift >= 0:
                    idxs = np.arange(min(n - shift))
                else:
                    idxs = np.arange(-shift, n)
                if len(idxs) > 0:
                    pick = rng.choice(idxs, size=min(3, len(idxs)), replace=False)
                    h_new[pick] += rng.randn(len(pick)) * 0.08
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        T = T0 * (1.0 - it / max(n_iters, 1)) ** 2
        if c_new < curr or (c_new > curr and rng.rand() < np.exp(-(c_new - curr) / max(T, 1e-15))):
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best = h.copy()
    return best, best_val

def coord_pair_polish(h0, n, n_rounds=40):
    h = h0.copy()
    curr = compute_c5(h)
    for r in range(n_rounds):
        improved = False
        order = np.random.permutation(n)
        for idx in range(n):
            i = order[idx]
            j = order[(idx + 7) % n]
            if i == j:
                continue
            best_v = curr
            best_h = h.copy()
            deltas = [-0.15, -0.08, -0.03, 0.03, 0.08, 0.15]
            for di in deltas:
                for dj in deltas:
                    if di == dj == 0:
                        continue
                    ht = h.copy()
                    ht[i] = np.clip(ht[i] + di, 0.0, 1.0)
                    ht[j] = np.clip(ht[j] + dj, 0.0, 1.0)
                    ht = project_feasible(ht, n)
                    v = compute_c5(ht)
                    if v < best_v:
                        best_v = v
                        best_h = ht.copy()
            if best_v < curr - 1e-14:
                h = best_h
                curr = best_v
                improved = True
        if not improved:
            break
    return h, curr

def interp_h(h_src, n_src, n_dst):
    if n_dst == n_src:
        return h_src.copy()
    xp = np.linspace(0, 1, n_src)
    fp = h_src
    new_x = np.linspace(0, 1, n_dst)
    h_new = np.interp(new_x, xp, fp)
    return project_feasible(h_new, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng_main = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15 - (time.time() - t0)

    try:
        init = np.array(initial_h_values, dtype=float)
        n0 = len(init)
    except NameError:
        init = np.full(80, 0.5)
        n0 = 80
    init = project_feasible(init, n0)

    best_h = init.copy()
    best_c5 = compute_c5(init)
    best_n = n0

    # Phase 1: SLSQP smooth max from initial
    for alpha in [80, 250, 800]:
        if tleft() < 50:
            break
        h_t = slsqp_smooth(best_h, best_n, alpha, maxiter=300)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 2: Aggressive local search
    if tleft() > 100:
        n_iter = min(int(tleft() * 0.4 / 0.00008), 500000)
        h_t, c = local_search(best_h, best_n, n_iter, rng_main)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 3: Resolution sweep
    n_candidates = sorted(set([best_n - 10, best_n - 5, best_n, best_n + 5, best_n + 10,
                                50, 64, 96, 120]))
    n_candidates = [x for x in n_candidates if 30 <= x <= 160 and x % 2 == 0]
    for n_try in n_candidates:
        if tleft() < 60:
            break
        h_start = interp_h(best_h, best_n, n_try)
        h_t = slsqp_smooth(h_start, n_try, 300, maxiter=200)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
            best_n = n_try
            if tleft() > 80:
                h_t2, c2 = local_search(h_t, n_try, min(int(tleft()*0.25/0.00008), 200000), rng_main)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h_t2.copy()

    # Phase 4: SLSQP refinement at best n
    if tleft() > 50:
        h_t = slsqp_smooth(best_h, best_n, 600, maxiter=300)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 5: Pair coordinate polish
    if tleft() > 60:
        h_t, c = coord_pair_polish(best_h, best_n, n_rounds=30)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 6: Another local search round with different seed
    if tleft() > 40:
        rng2 = np.random.RandomState(seed + 999)
        n_iter = min(int(tleft() * 0.5 / 0.00008), 300000)
        h_t, c = local_search(best_h, best_n, n_iter, rng2)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 7: Final SLSQP + polish
    if tleft() > 25:
        h_t = slsqp_smooth(best_h, best_n, 1000, maxiter=150)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
    if tleft() > 20:
        h_t, c = coord_pair_polish(best_h, best_n, n_rounds=15)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
