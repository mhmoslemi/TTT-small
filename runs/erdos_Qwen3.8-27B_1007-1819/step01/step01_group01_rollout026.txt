        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(corr, alpha):
    m = np.max(corr)
    x = alpha * (corr - m)
    x = np.clip(x, -300, 300)
    return m + np.log(np.sum(np.exp(x))) / alpha

def slsqp_smooth(h0, n, alpha, maxiter=400):
    dx = 2.0 / n
    def f(p):
        h = np.clip(p, 0.0, 1.0)
        c = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(c, alpha) * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def slsqp_exact(h0, n, maxiter=400):
    dx = 2.0 / n
    def f(p):
        h = np.clip(p, 0.0, 1.0)
        c = np.correlate(h, 1.0 - h, mode='full')
        return np.max(c) * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def coord_descent(h, n, n_rounds=40, grid_sz=31):
    h = h.copy()
    curr = compute_c5(h)
    grid = np.linspace(0.0, 1.0, grid_sz)
    for r in range(n_rounds):
        improved = False
        for i in range(n):
            save = h[i]
            best_v = save
            best_c = curr
            for cv in grid:
                h[i] = cv
                project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = h[i]
                    improved = True
            h[i] = best_v
            project_feasible(h, n)
            curr = min(curr, best_c)
        if not improved:
            break
    return h, curr

def gen_sinusoidal(n, freqs, amps, phases):
    h = np.zeros(n)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(2.0 * np.pi * f * np.linspace(0, 1, n) + p)
    h = 0.5 + h
    return project_feasible(h, n)

def gen_block(n, pattern_vals, pattern_lens):
    total = sum(pattern_lens)
    h_list = []
    idx = 0
    reps = n // (total * 2) + 1
    for _ in range(reps * 2):
        for v, L in zip(pattern_vals, pattern_lens):
            h_list.extend([v] * L)
            idx += L
            if idx >= n:
                break
        if idx >= n:
            break
    h = np.array(h_list[:n], dtype=float)
    return project_feasible(h, n)

def iterated_local_search(h0, n, slsqp_alpha, n_iters, budget_s):
    h = h0.copy()
    curr = compute_c5(h)
    t_start = time.time()
    for it in range(n_iters):
        if time.time() - t_start > budget_s or (time.time() - t_start) / max(1, it) > budget_s / max(1, n_iters):
            break
        k_pert = np.random.choice([3, 5, 8, 12], p=[0.3, 0.3, 0.25, 0.15])
        idxs = np.random.choice(n, size=min(k_pert, n), replace=False)
        h_p = h.copy()
        h_p[idxs] = np.clip(h_p[idxs] + np.random.randn(len(idxs)) * 0.12, 0.0, 1.0)
        h_p = project_feasible(h_p, n)
        h_p = slsqp_smooth(h_p, n, slsqp_alpha, maxiter=200)
        c_p = compute_c5(h_p)
        if c_p < curr:
            h = h_p
            curr = c_p
    return h, curr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def time_left():
        return budget_s - 20 - (time.time() - t0)

    try:
        init = np.array(initial_h_values, dtype=float)
        n_init = len(init)
    except NameError:
        init = np.full(60, 0.5)
        n_init = 60

    init = project_feasible(init, n_init)
    best_h = init.copy()
    best_c5 = compute_c5(init)
    best_n = n_init

    # Phase 1: Diverse initializations + SLSQP at initial n
    starts = [init.copy()]
    starts.append(gen_sinusoidal(n_init, [1, 2, 3], [0.2, 0.1, 0.05], [0, 1, 2]))
    starts.append(gen_sinusoidal(n_init, [2, 5], [0.25, 0.1], [0.5, 1.5]))
    starts.append(gen_block(n_init, [0.3, 0.7, 0.5], [3, 3, 4]))
    starts.append(project_feasible(np.clip(0.5 + 0.2 * np.random.randn(n_init), 0, 1), n_init))
    starts.append(project_feasible(np.clip(0.5 + 0.15 * np.random.randn(n_init), 0, 1), n_init))

    for alpha in [30, 80, 200]:
        if time_left() < 60:
            break
        for s in starts[:]:
            if time_left() < 30:
                break
            h_t = slsqp_smooth(s, best_n, alpha, maxiter=300)
            c = compute_c5(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()
                starts = [best_h.copy()] + [x for x in starts if x is not best_h][:2]
                break

    # Phase 2: Progressive alpha refinement on best
    for alpha in [200, 500, 1000]:
        if time_left() < 40:
            break
        h_t = slsqp_smooth(best_h, best_n, alpha, maxiter=500)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 3: Exact SLSQP
    if time_left() > 60:
        h_t = slsqp_exact(best_h, best_n, maxiter=500)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 4: Dense coordinate descent
    if time_left() > 60:
        h_t, c = coord_descent(best_h, best_n, n_rounds=30, grid_sz=31)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 5: Iterated local search
    if time_left() > 80:
        ilsa_budget = min(time_left() * 0.4, 250)
        h_t, c = iterated_local_search(best_h, best_n, 200, 50, ilsa_budget)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 6: Coordinate descent again after ILS
    if time_left() > 50:
        h_t, c = coord_descent(best_h, best_n, n_rounds=25, grid_sz=41)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 7: Resolution sweep
    n_sweeps = sorted(set([best_n, best_n - 10, best_n + 10, best_n + 20, 2 * best_n]))
    n_sweeps = [x for x in n_sweeps if 12 <= x <= 400 and x != best_n]
    for n_new in n_sweeps:
        if time_left() < 80:
            break
        x_old = np.linspace(0, 1, best_n)
        x_new = np.linspace(0, 1, n_new)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)
        for alpha in [100, 500]:
            if time_left() < 40:
                break
            h_t = slsqp_smooth(h_interp, n_new, alpha, maxiter=300)
            c = compute_c5(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()
                best_n = n_new
        if time_left() > 40:
            h_t = slsqp_exact(best_h, best_n, maxiter=300)
            c = compute_c5(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 8: Final polish
    if time_left() > 30:
        h_t = slsqp_smooth(best_h, best_n, 1000, maxiter=300)
        h_t = slsqp_exact(h_t, best_n, maxiter=300)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
    if time_left() > 20:
        h_t, c = coord_descent(best_h, best_n, n_rounds=20, grid_sz=41)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
