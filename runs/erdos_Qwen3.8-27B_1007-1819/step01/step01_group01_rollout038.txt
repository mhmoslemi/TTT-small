def compute_c5(h):

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-13:

        else:

        np.clip(h, 0.0, 1.0, out=h)

import numpy as np
from scipy.optimize import minimize
import time

_n_g = 80

def _con_fun(x):
    return np.sum(x) - _n_g / 2.0

def _con_jac(x):
    return np.ones(len(x))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = np.clip(alpha * (corr - m), -700, 0)
    s = np.sum(np.exp(scaled))
    return (m + np.log(s) / alpha) * dx

def exact_obj(params, n):
    return compute_c5(project_feasible(params, n))

def run_slsqp(h0, n, alpha, maxiter=500):
    global _n_g
    _n_g = n
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(n, alpha),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def run_exact(h0, n, maxiter=300):
    global _n_g
    _n_g = n
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

        res = minimize(exact_obj, h0, args=(n,),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def coord_descent(h, n, n_rounds=30):
    h = h.copy()
    curr = compute_c5(h)
    rng_cd = np.random.RandomState(42)

        improved = False
        for i in range(n):
            old = h[i]
            best_v = old
            best_c = curr
            cands = np.unique(np.clip(np.concatenate([
                np.linspace(0.0, 1.0, 11),
                old + rng_cd.randn(4) * 0.05
            ]), 0.0, 1.0))
            for v in cands:
                if abs(v - old) < 1e-7:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = h[i]
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_c
        if not improved:
            break
    return h, compute_c5(h)

def sa_refine(h0, n, time_budget, seed):

    h = h0.copy()
    curr = compute_c5(h)
    best = h.copy()
    best_c = curr

    step = 0
    est_steps = max(int(time_budget / 0.005), 100)

        step += 1

        amp = 0.25 * (1.0 - frac) + 0.015

        move = rng.rand()
        h_new = h.copy()
        if move < 0.35:

            curr = c_new

                best_c = curr
                best = h.copy()
    return best, best_c
