                h = project_feasible(h, n)
                v = compute_c5(h)
                if v < best_c:
                    best_c = v
                    best_v = h[i]
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_c
        if not improved:
            break
    return h, curr

def simulated_annealing(h0, n, t_start, t_end, max_steps):
    h = h0.copy()
    curr = compute_c5(h)
    best_h = h.copy()
    best_val = curr
    T = t_start
    cool = (max(t_end / max(t_start, 1e-15), 0.5)) ** (1.0 / max(max_steps, 1))
    for _ in range(max_steps):
        if time.time() >= t_end:
            break
        n_pert = np.random.choice([1, 2, 3], p=[0.5, 0.3, 0.2])
        idxs = np.random.choice(n, size=n_pert, replace=False)
        h_new = h.copy()
        noise = np.random.randn(n_pert) * 0.09
        h_new[idxs] = np.clip(h_new[idxs] + noise, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < curr:
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best_h = h.copy()
        elif T > 1e-14:
            if np.random.rand() < min(1.0, np.exp(-(c_new - curr) / T)):
                h = h_new
                curr = c_new
        T *= cool
    return best_h, best_val

def bang_bang_push(h, n, threshold):
    h2 = h.copy()
    h2[h2 < threshold] = 0.0
    h2[h2 > 1.0 - threshold] = 1.0
    return project_feasible(h2, n)

def de_objective(params, n):
    h = project_feasible(params, n)
    return compute_c5(h)

def multi_start_slsqp(h_best, n_best, deadline, alphas):
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    for a in alphas:
        if time.time() >= deadline:
            break
        h_t = slsqp_smooth(best_h, n_best, a, maxiter=500)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    deadline = t0 + budget_s - 5

    try:
        init = np.array(initial_h_values, dtype=float)
    except NameError:
        init = np.full(50, 0.5)
    n_init = len(init)
    init = project_feasible(init, n_init)

    best_h = init.copy()
    best_c5 = compute_c5(init)
    best_n = n_init

    # Phase 1: Progressive SLSQP from initial
    best_h, best_c5 = multi_start_slsqp(best_h, best_n, deadline - 200, [20, 50, 100, 200, 500, 1000, 2000])
    best_n = len(best_h)

    # Phase 2: Resolution sweep with interpolation
    n_candidates = sorted(set([best_n, best_n - 10, best_n + 10,
                                best_n - 5, best_n + 5,
                                40, 50, 60, 70, 80, 90, 100]))
    for n_try in n_candidates:
        if n_try < 20 or n_try > 150:
            continue
        if time.time() >= deadline - 80:
            break
        h_start = interp_to_n(best_h, n_try)
        h_t = slsqp_smooth(h_start, n_try, 200, maxiter=400)
        h_t = slsqp_exact(h_t, n_try, maxiter=300)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
            best_n = n_try

    # Phase 3: DE global search at low n
    if time.time() < deadline - 100:
        for n_de in [25, 35]:
            if time.time() >= deadline - 50:
                break
            try:
                bounds = [(0.0, 1.0)] * n_de
                de_res = differential_evolution(
                    de_objective, bounds, args=(n_de,),
                    maxiter=80, popsize=12, tol=1e-8,
                    seed=seed, polish=False, init='sobol',
                    updating='immediate', workers=1
                )
                h_de = project_feasible(de_res.x, n_de)
                h_de = slsqp_smooth(h_de, n_de, 500, maxiter=300)
                c_de = compute_c5(h_de)
                if c_de < best_c5:
                    h_interp = interp_to_n(h_de, max(n_de + 40, 70))
                    h_t = slsqp_smooth(h_interp, max(n_de + 40, 70), 300, maxiter=400)
                    h_t = slsqp_exact(h_t, max(n_de + 40, 70), maxiter=300)
                    c_t = compute_c5(h_t)
                    if c_t < best_c5:
                        best_c5 = c_t
                        best_h = h_t.copy()
                        best_n = max(n_de + 40, 70)
                    else:
                        best_c5 = c_de
                        best_h = h_de.copy()
                        best_n = n_de
            except Exception:
                pass

    # Phase 4: Multiple random restarts + SLSQP
    if time.time() < deadline - 60:
        for restart in range(5):
            if time.time() >= deadline - 30:
                break
            h_rand = np.clip(0.5 + 0.3 * np.random.randn(best_n), 0, 1)
            h_rand = project_feasible(h_rand, best_n)
            h_t = slsqp_smooth(h_rand, best_n, 100, maxiter=200)
            c = compute_c5(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 5: Coordinate descent
    if time.time() < deadline - 30:
        h_t, c_t = coordinate_descent(best_h, best_n, n_rounds=60, n_trial=9)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()

    # Phase 6: Bang-bang push + reoptimize
    if time.time() < deadline - 20:
        for thresh in [0.3, 0.2, 0.15]:
            if time.time() >= deadline - 10:
                break
            h_bb = bang_bang_push(best_h, best_n, thresh)
            h_t = slsqp_smooth(h_bb, best_n, 500, maxiter=300)
            h_t = slsqp_exact(h_t, best_n, maxiter=200)
            c_t = compute_c5(h_t)
            if c_t < best_c5:
                best_c5 = c_t
                best_h = h_t.copy()

    # Phase 7: Simulated annealing
    if time.time() < deadline - 10:
        remaining = deadline - time.time() - 5
        n_steps = int(remaining / 0.0025)
        if n_steps > 500:
            h_t, c_t = simulated_annealing(best_h, best_n, 0.012, 1e-7, n_steps)
            if c_t < best_c5:
                best_c5 = c_t
                best_h = h_t.copy()
            h_t = slsqp_smooth(best_h, best_n, 1000, maxiter=200)
            c_t = compute_c5(h_t)
            if c_t < best_c5:
                best_c5 = c_t
                best_h = h_t.copy()

    # Phase 8: Final fine-tuning
    if time.time() < deadline - 5:
        h_t = slsqp_smooth(best_h, best_n, 3000, maxiter=300)
        c_t = compute_c5(h_t)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()
        h_t, c_t = coordinate_descent(best_h, best_n, n_rounds=40, n_trial=11)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()
        h_t = slsqp_exact(best_h, best_n, maxiter=200)
        c_t = compute_c5(h_t)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = h_t.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
