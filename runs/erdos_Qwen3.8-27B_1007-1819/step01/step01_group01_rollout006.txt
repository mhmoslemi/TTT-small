import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def coord_descent(h, n, n_rounds=100, tol=1e-12):
    h = h.copy()
    curr = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for i in range(n):
            corr = np.correlate(h, 1-h, mode='full')
            k = np.arange(2*n-1)
            cond1 = (k >= n-1-i) & (k <= n-1+i)
            cond2 = (k >= i+1) & (k <= i+n-1)
            idx1 = np.where(cond1, i + k - (n-1), 0)
            idx2 = np.where(cond2, i - k + (n-1), 0)
            h_val1 = np.where(cond1, h[np.clip(idx1, 0, n-1)], 0.0)
            h_val2 = np.where(cond2, h[np.clip(idx2, 0, n-1)], 0.0)
            slopes = cond1.astype(float) - h_val1 - h_val2
            intercepts = corr - h[i] * slopes
            
            # Minimize max_k (intercepts + t*slopes) for t in [0,1]
            grid = np.linspace(0, 1, 51)
            vals = np.max(intercepts[None,:] + grid[:,None]*slopes[None,:], axis=1)
            gi = np.argmin(vals)
            best_t = grid[gi]
            # Refine
            lo = grid[max(0, gi-1)]
            hi = grid[min(len(grid)-1, gi+1)]
            for _ in range(30):
                m1 = lo + (hi-lo)*0.33
                m2 = hi - (hi-lo)*0.33
                if np.max(intercepts + m1*slopes) < np.max(intercepts + m2*slopes):
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-14:
                    break
            best_t = (lo+hi)/2
            new_val = np.max(intercepts + best_t*slopes)
            old_val = np.max(intercepts + h[i]*slopes)
            
            if new_val < old_val - tol:
                h[i] = best_t
                improved = True
        
        # Re-project
        h = project_feasible(h, n)
        new_curr = compute_c5(h)
        if new_curr < curr - 1e-14:
            curr = new_curr
        if not improved:
            break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def coord_descent(h, n, n_rounds=200, tol=1e-13):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    dx = 2.0 / n
    curr = np.max(corr) * dx
    for rnd in range(n_rounds):
        improved = False
        for i in range(n):
            k = np.arange(2 * n - 1)
            cond1 = (k >= n - 1 - i) & (k <= n - 1 + i)
            cond2 = (k >= i + 1) & (k <= i + n - 1)
            idx1 = np.where(cond1, i + k - (n - 1), 0)
            idx2 = np.where(cond2, i - k + (n - 1), 0)
            h_val1 = np.where(cond1, h[np.clip(idx1, 0, n - 1)], 0.0)
            h_val2 = np.where(cond2, h[np.clip(idx2, 0, n - 1)], 0.0)
            slopes = cond1.astype(float) - h_val1 - h_val2
            intercepts = corr - h[i] * slopes
            grid = np.linspace(0, 1, 51)
            vals = np.max(intercepts[None, :] + grid[:, None] * slopes[None, :], axis=1)
            gi = np.argmin(vals)
            lo = grid[max(0, gi - 2)]
            hi = grid[min(len(grid) - 1, gi + 2)]
            for _ in range(30):
                m1 = lo + (hi - lo) * 0.37
                m2 = hi - (hi - lo) * 0.37
                if np.max(intercepts + m1 * slopes) < np.max(intercepts + m2 * slopes):
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-15:
                    break
            best_t = (lo + hi) / 2.0
            new_val = np.max(intercepts + best_t * slopes)
            if new_val < np.max(corr) - tol:
                delta = best_t - h[i]
                corr += delta * slopes
                h[i] = best_t
                improved = True
        h = project_feasible(h, n)

        new_curr = np.max(corr) * dx
        if new_curr < curr - 1e-15:
            curr = new_curr
        if not improved:
            break
    return h, compute_c5(h)

def smooth_max_obj(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return sm * dx

def slsqp_refine(h0, n, alpha, maxiter=300):
    try:
        res = minimize(smooth_max_obj, h0.copy(), args=(n, alpha),
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def pair_swap_polish(h, n, n_rounds=20):
    h = h.copy()
    curr = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(n_rounds):
        improved = False

            for j in range(i + 1, n):

                    if abs(delta) < 1e-15:

                    if h[i] + delta < 0 or h[i] + delta > 1:

                    if v < best_val - 1e-14:

                    h[i] -= delta

                    h[j] -= best_delta
                    curr = best_val
                    improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h)
