    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, alpha) * dx

def slsqp_refine(h0, n, alpha, maxiter=500):
    try:
        res = minimize(obj_smooth, h0.copy(), args=(n, alpha),
                       method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def adaptive_local_search(h, n, budget_s, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    curr = compute_c5(h)
    best = h.copy()
    best_val = curr
    step = 0.08
    t0 = time.time()
    it = 0
    while time.time() - t0 < budget_s:
        it += 1
        r = rng.rand()
        h_old = h.copy()
        if r < 0.35:
            i = rng.randint(n)
            h[i] = np.clip(h[i] + rng.randn() * step, 0, 1)
        elif r < 0.65:
            i, j = rng.choice(n, 2, replace=False)
            d = rng.randn() * step
            h[i] = np.clip(h[i] + d, 0, 1)
            h[j] = np.clip(h[j] - d, 0, 1)
        else:
            i = rng.randint(max(1, n - 3))
            bl = min(3, n - i)
            h[i:i + bl] = np.clip(h[i:i + bl] + rng.randn(bl) * step * 0.6, 0, 1)
        h = project_feasible(h, n)
        new_val = compute_c5(h)
        if new_val <= curr:
            curr = new_val
            if curr < best_val:
                best_val = curr
                best = h.copy()
            step = max(0.003, step * 0.997)
        else:
            if rng.rand() < np.exp(-(new_val - curr) / 0.0015):
                curr = new_val
            else:
                h = h_old
            step = min(0.12, step * 1.012)
    return best, best_val

def coordinate_refine(h, n, n_rounds=30, n_cand=15):
    h = h.copy()
    curr = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            h_base = h.copy()
            bv, bc = h[i], curr
            cands = np.unique(np.clip(np.concatenate([
                [bv], np.linspace(0, 1, n_cand),
                bv + np.array([-0.1, -0.05, -0.02, 0.02, 0.05, 0.1])
            ]), 0, 1))
            for rv in cands:
                if abs(rv - bv) < 1e-7:
                    continue
                h_try = h_base.copy()
                h_try[i] = rv
                h_try = project_feasible(h_try, n)
                v = compute_c5(h_try)
                if v < bc:
                    bc, bv, improved = v, rv, True
            h[i] = bv
            h = project_feasible(h, n)
            curr = bc
        if not improved:
            break
    return h, curr

def binary_exhaustive(n, budget_s):
    from itertools import combinations
    nh = n // 2
    best_h = None
    best_val = float('inf')
    t0 = time.time()
    for pos in combinations(range(n), nh):
        if time.time() - t0 > budget_s:
            break
        h = np.zeros(n)
        h[list(pos)] = 1.0
        c = compute_c5(h)
        if c < best_val:
            best_val = c
            best_h = h.copy()
    return best_h, best_val

def multi_start_slsqp(ref_h, ref_n, n, n_starts, budget_s, seed):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = project_feasible(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, ref_n), ref_h), n)
    best_c = compute_c5(best_h)
    for s in range(n_starts):
        if time.time() - t0 > budget_s:
            break
        if s == 0:
            h0 = best_h.copy()
        else:
            h0 = project_feasible(np.clip(0.5 + 0.3 * rng.randn(n), 0, 1), n)
        for alpha in [100, 500]:
            hot = slsqp_refine(h0, n, alpha, 200)
            c = compute_c5(hot)
            if c < best_c:
                best_c, best_h = c, hot.copy()
                h0 = hot.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    try:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
    except NameError:
        best_h = np.full(80, 0.5)
        best_n = 80
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - 10 - (time.time() - t0)

    # Phase 1: SLSQP at multiple smoothness levels
    if tl() > 60:
        for alpha in [30, 80, 200, 600, 1500]:
            if tl() < 20:
                break
            h_t = slsqp_refine(best_h, best_n, alpha, 350)
            c = compute_c5(h_t)
            if c < best_c5:
                best_c5, best_h = c, h_t.copy()

    # Phase 2: Main adaptive local search (largest budget)
    if tl() > 80:
        b = min(tl() * 0.4, 400)
        h_t, c = adaptive_local_search(best_h, best_n, b, seed)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()

    # Phase 3: SLSQP + coordinate polish
    if tl() > 40:
        h_t = slsqp_refine(best_h, best_n, 400, 300)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()
    if tl() > 30:
        h_t, c = coordinate_refine(best_h, best_n, 25, 15)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()

    # Phase 4: Try other n_points
    for n in [40, 50, 60, 70, 90, 100, 120]:
        if n == best_n or tl() < 55:
            continue
        b = min(tl() * 0.2, 70)
        h_t, c = multi_start_slsqp(best_h, best_n, n, 3, b, seed + n)
        if c < best_c5:
            best_c5, best_h, best_n = c, h_t.copy(), n
        if tl() > 25 and best_n != n:
            h_t2, c2 = adaptive_local_search(h_t, n, min(tl() * 0.1, 30), seed + n + 1)
            if c2 < best_c5:
                best_c5, best_h, best_n = c2, h_t2.copy(), n

    # Phase 5: Binary exhaustive for n=20 (quick)
    if tl() > 40:
        bh, bc = binary_exhaustive(20, min(tl() * 0.1, 20))
        if bh is not None and bc < best_c5:
            best_c5, best_h, best_n = bc, bh, 20

    # Phase 6: Final SLSQP + coordinate + adaptive LS
    if tl() > 20:
        h_t = slsqp_refine(best_h, best_n, 1000, 250)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()
    if tl() > 15:
        h_t, c = coordinate_refine(best_h, best_n, 20, 21)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()
    if tl() > 15:
        h_t, c = adaptive_local_search(best_h, best_n, max(5, tl() - 5), seed + 9999)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
