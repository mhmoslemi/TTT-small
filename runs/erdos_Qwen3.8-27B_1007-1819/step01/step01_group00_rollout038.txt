            best_h = h_opt

    # Phase 3: Coordinate descent (medium step)
    if best_h is not None and rem(0.2) > 10:
        h_cd, c5_cd = coord_descent(best_h, rounds=30, base_step=0.03)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 4: Basin hopping - perturb and re-optimize
    if best_h is not None:
        n = len(best_h)
        for trial in range(4):
            if rem(0.12) < 20:
                break
            noise_scale = 0.08 + 0.04 * trial
            h_pert = best_h + noise_scale * rng.randn(n)
            h_pert = fix_feasible(h_pert)
            h_opt, c5_opt = optimize_slsqp(h_pert, [50, 120], 8, 150, t0, budget_s, 0.12)
            h_cd, c5_cd = coord_descent(h_opt, rounds=20, base_step=0.02)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Phase 5: Explore 1-h symmetry and different structured starts
    if best_h is not None and rem(0.08) > 15:
        h_flip = fix_feasible(1.0 - best_h)
        c5_flip = compute_c5(h_flip)
        if c5_flip < best_c5:
            best_c5 = c5_flip
            best_h = h_flip

    # Phase 6: Try lower resolutions
    for n_alt in [50, 72, 36]:
        if rem(0.06) < 25:
            break
        if best_h is not None:
            frac_old = np.linspace(0.0, 1.0, len(best_h))
            frac_new = np.linspace(0.0, 1.0, n_alt)
            h_alt = np.interp(frac_new, frac_old, best_h)
        else:
            h_alt = 0.5 * np.ones(n_alt)
        h_alt = fix_feasible(h_alt)
        c5_alt = compute_c5(h_alt)
        h_opt, c5_opt = optimize_slsqp(h_alt, [40, 80, 150], 6, 150, t0, budget_s, 0.06)
        h_cd, c5_cd = coord_descent(h_opt, rounds=25, base_step=0.03)
        if c5_cd < c5_alt:
            c5_alt = c5_cd
            h_alt = h_cd
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt

    # Phase 7: Final SLSQP polish at high beta
    if best_h is not None and rem(0.02) > 8:
        n = len(best_h)
        h_opt, c5_opt = optimize_slsqp(best_h, [150, 300, 500], 6, 300, t0, budget_s, 0.03)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 8: Fine coordinate descent
    if best_h is not None and rem(0.005) > 5:
        h_fine, c5_fine = fine_coord_descent(best_h, rounds=40, base_step=0.01)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine

    if best_h is None:
        best_h = fix_feasible(0.5 * np.ones(100))
        best_c5 = compute_c5(best_h)

    best_h = fix_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
