```python
def subgradient_at_topk(h, K=10, beta_sm=50):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top K indices
    top_k = np.argsort(corr)[-K:]
    # Softmax weights
    weights = np.exp((corr[top_k] - corr[top_k].max()) / beta_sm)
    weights /= weights.sum()
    
    # Compute subgradient
    g = np.zeros(n)
    for w, k in zip(weights, top_k):
        # For lag k, compute gradient wrt h
        # Valid t range
        t_lo = max(0, n - 1 - k)
        t_hi = min(n - 1, 2 * n - 2 - k)
        for t in range(t_lo, t_hi + 1):
            # Contribution from h[t] * (1-h[t+k-n+1])
            idx2 = t + k - n + 1
            g[t] += w * (1.0 - h[idx2])
            # Contribution from h[t] * (1-h[t+k-n+1]) wrt h[idx2]
            g[idx2] += w * (-h[t])
    return g, corr.max() * (2.0 / n)
```

```python
    t_lo = max(0, n - 1 - k)
    t_hi = min(n - 1, 2 * n - 2 - k)
    ts = np.arange(t_lo, t_hi + 1)

    g = np.zeros(n)
    g[ts] += (1.0 - h[idx2])
    g[idx2] -= h[ts]
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    """Project x onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    target = n / 2.0
    # Bisection on threshold
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def sum_constraint(x, n_pts):
    return x.sum() - n_pts / 2.0
```

```python
def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

    corr = np.correlate(h, 1.0 - h, mode="full")
    top = np.argsort(corr)[-K:]
    w = np.exp((corr[top] - corr[top].max()) / beta_sm)
    w /= w.sum()
    g = np.zeros(n)
    for wi, k in zip(w, top):
        k = int(k)
        t_lo = max(0, n - 1 - k)
```
