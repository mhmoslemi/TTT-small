    else:

            b1 = lo + gr * (hi - lo)

        starts.append(interp_to_size(best_h, best_n, n_pts))
        starts.append(make_antiperiodic(n_pts, 1.0, 0.0))
        starts.append(make_antiperiodic(n_pts, 2.0, np.pi / 4))
        xg = np.arange(n_pts) * 2.0 / n_pts
        for fr in [3, 5, 7]:
            starts.append(0.5 + 0.25 * np.cos(2.0 * np.pi * fr * xg))
        starts.append(rng.rand(n_pts) * 0.6 + 0.2)
        for hs in starts:
            if elapsed() > budget_s * 0.55:
                break
            hs = project_feasible(hs, n_pts)
            h_try = exact_coord_descent(hs, n_pts, passes=4, seed=seed + n_pts)
            h_try = project_feasible(h_try, n_pts)
            c = compute_c5(h_try)
            if c < best_c5:
                best_c5, best_h, best_n = c, h_try.copy(), n_pts

    # Phase 4: SLSQP refinement
    if elapsed() < budget_s * 0.35:
        n_pts = best_n
        h_start = best_h.copy()
        for beta in [80.0, 150.0, 300.0, 600.0]:
            if elapsed() > budget_s * 0.55:
                break
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n_pts / 2.0}]
            res = minimize(smooth_obj, h_start, args=(n_pts, beta),
                           method='SLSQP', bounds=[(0.0, 1.0)] * n_pts,
                           constraints=cons, options={'maxiter': 400, 'ftol': 1e-14})
            if hasattr(res, 'x') and res.x is not None:
                h_opt = project_feasible(res.x, n_pts)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h, best_n = c, h_opt.copy(), n_pts
                    h_start = h_opt.copy()

    # Phase 5: Final coordinate descent polish
    if elapsed() < budget_s * 0.25:
        for p in range(3):
            if elapsed() > budget_s * 0.7:
                break
            h_final = exact_coord_descent(best_h.copy(), best_n, passes=3, seed=999 + p)
            h_final = project_feasible(h_final, best_n)
            c = compute_c5(h_final)
            if c < best_c5:
                best_c5, best_h, best_n = c, h_final.copy(), best_n

    # Phase 6: Try higher resolution if time remains
    if elapsed() < budget_s * 0.2:
        n_hi = best_n * 2
        if n_hi <= 720:
            h_hi = project_feasible(interp_to_size(best_h, best_n, n_hi), n_hi)
            h_hi = exact_coord_descent(h_hi, n_hi, passes=2, seed=seed)
            h_hi = project_feasible(h_hi, n_hi)
            c = compute_c5(h_hi)
            if c < best_c5:
                best_c5, best_h, best_n = c, h_hi.copy(), n_hi

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
