import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n * 3):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = int(np.argmax(1.0 - h))
                h[idx] += min(r, 1.0 - h[idx])
            else:
                idx = int(np.argmax(h))
                h[idx] -= min(-r, h[idx])
    return h

def smooth_val_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    shifted = np.clip(beta * (corr - mx), -500, 0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = (mx + np.log(s) / beta) * dx
    w = exps / s
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        shift = k - n + 1
        a = max(0, -shift)
        b = min(n, n - shift)
        if b > a:
            grad[a:b] -= w[k] * h[a + shift:b + shift]
        a2 = max(0, shift)
        b2 = min(n, n + shift)
        if b2 > a2:
            grad[a2:b2] -= w[k] * h[a2 - shift:b2 - shift]
    grad += 1.0  # sum(w[k]) = 1
    return val, dx * grad

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_smooth(h, beta, maxiter=300):
    n = len(h)
    def obj(x):
        x = np.clip(x, 0, 1)
        v, _ = smooth_val_and_grad(x, beta)
        return v
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        _, g = smooth_val_and_grad(x, beta)
        return g
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   constraints=[{'type': 'eq', 'fun': sum_eq}],
                   bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    return fix_feasible(res.x)

def coord_descent(h, max_rounds=10, step=0.02):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for d in [step, -step, 2*step, -2*step]:
                v = h[i] + d
                if 0 <= v <= 1:
                    h[i] = v
                    h = fix_feasible(h)
                    c = compute_c5(h)
                    if c < cur - 1e-12:
                        cur = c
                        improved = True
                    else:
                        h[i] = v - d  # revert
                        h = fix_feasible(h)
        if not improved:
            break
    return h

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old + 1)[:-1] + 1.0/n_old
    x_new = np.linspace(0, 2, n_new + 1)[:-1] + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old)
    return fix_feasible(h_new)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n * 3):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = int(np.argmax(1.0 - h))
                h[idx] += min(r, 1.0 - h[idx])
            else:
                idx = int(np.argmax(h))
                h[idx] -= min(-r, h[idx])
    return h

def smooth_val_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    shifted = np.clip(beta * (corr - mx), -500, 0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = (mx + np.log(s) / beta) * dx
    w = exps / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        shift = k - n + 1
        a = max(0, -shift)
        b = min(n, n - shift)
        if b > a:
            grad[a:b] -= w[k] * h[a + shift:b + shift]
        a2 = max(0, shift)
        b2 = min(n, n + shift)
        if b2 > a2:
            grad[a2:b2] -= w[k] * h[a2 - shift:b2 - shift]
    grad += 1.0
    return val, dx * grad

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_smooth(h, beta, maxiter=250):
    n = len(h)
    def obj(x):
        xc = np.clip(x, 0, 1)
        v, _ = smooth_val_and_grad(xc, beta)
        return v
    def obj_g(x):
        xc = np.clip(x, 0, 1)
        _, g = smooth_val_and_grad(xc, beta)
        return g
    try:
        res = minimize(obj, h, jac=obj_g, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': sum_eq}],
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        return fix_feasible(res.x)
    except Exception:
        return h

def coord_descent(h, max_rounds=12, steps=None):
    n = len(h)
    if steps is None:
        steps = [0.02, -0.02, 0.04, -0.04, 0.08, -0.08]
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            orig = h[i]
            for d in steps:
                v = orig + d
                if 0 <= v <= 1:
                    h[i] = v
                    h = fix_feasible(h)
                    c = compute_c5(h)
                    if c < cur - 1e-13:
                        cur = c
                        improved = True
                        break
            h[i] = h[i]  # keep current
        if not improved:
            break
    return h

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t
