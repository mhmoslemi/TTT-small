import numpy as np
from scipy.optimize import minimize
import time

_obj_beta = 40.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(3 * n + 5):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = np.argsort(1.0 - h)
                for i in idx:
                    a = min(r, 1.0 - h[i])
                    h[i] += a
                    r -= a
                    if abs(r) < 1e-12:
                        break
            else:
                idx = np.argsort(h)
                for i in idx:
                    a = min(-r, h[i])
                    h[i] -= a
                    r += a
                    if abs(r) < 1e-12:
                        break
    return h

def smooth_max_corr(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = corr.max()
    return (mx + np.log(np.sum(np.exp(beta * (corr - mx)))) / beta) * dx

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = corr - corr.max()
    w = np.exp(beta * shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        p = m - n + 1
        lo1, hi1 = max(0, -p), min(n - 1, n - 1 - p)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w[m] * (1.0 - h[lo1 + p:hi1 + p + 1])
        lo2, hi2 = max(0, p), min(n - 1, n - 1 + p)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w[m] * h[lo2 - p:hi2 - p + 1]
    return grad * dx

def projected_grad_descent(h, beta, steps, lr_init):
    n = len(h)
    h = h.copy()
    lr = lr_init
    for s in range(steps):
        g = grad_smooth_max(h, beta)
        f = smooth_max_corr(h, beta)
        step = lr
        success = False
        for _ in range(25):
            h_new = fix_feasible(h - step * g)
            f_new = smooth_max_corr(h_new, beta)
            if f_new < f - 1e-14:
                h = h_new
                success = True
                break
            step *= 0.5
        if not success:
            break
        lr = min(lr * 1.05, 0.3) if step >= lr * 0.9 else max(step, lr * 0.6)
    return h

def _obj_fun(x):
    return smooth_max_corr(fix_feasible(x), _obj_beta)

def _sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def coord_descent(h, deltas, rounds):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(rounds):
        improved = False
        for i in range(n):
            for d in deltas:
                v = h[i] + d
                if v < -1e-9 or v > 1 + 1e-9:
                    continue
                old = h[i]
                h[i] = v
                h = fix_feasible(h)
                f = compute_c5(h)
                if f < cur - 1e-13:
                    cur = f
                    improved = True
                else:
                    h[i] = old
        if not improved:
            break
    return h

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c, x = 0, i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t

    t0 = time.time()
    np.random.seed(seed)

    best_c5 = 1e10

        best_c5 = compute_c5(h_init)
        best_h = h_init.copy()

    def tleft():
        return budget_s * 0.96 - (time.time() - t0)

    # Phase 1: Projected gradient descent from parent (key innovation)
    if h_init is not None and tleft() > 30:
        h = h_init.copy()
        for beta in [15, 30, 60, 120, 250, 500]:
            if tleft() < 10:
                break
            h = projected_grad_descent(h, beta, steps=300, lr_init=0.03)
            f = compute_c5(h)
            if f < best_c5:
                best_c5 = f
                best_h = h.copy()

    # Phase 2: Multi-start with various sizes
    n_list = []
    if h_init is not None:
        n_list.append(len(h_init))
    for nn in [36, 48, 54, 60, 72, 84, 96, 108, 120, 144, 180]:
        if nn not in n_list:
            n_list.append(nn)

    for n in n_list:
        if tleft() < 25:
            break
        n_starts = max(3, min(7, int(tleft() / max(n, 50) / 8)))
        for s in range(n_starts):
            if tleft() < 12:
                break
            if s == 0 and best_h is not None and len(best_h) == n:
                h = fix_feasible(best_h + 0.15 * np.random.randn(n))
            elif s == 1:
                h = fix_feasible(tm_seq(n) + 0.1 * np.random.randn(n))
            elif s == 2:
                x = np.arange(n)
                h = fix_feasible(0.5 + 0.3*np.sin(4*np.pi*x/n) + 0.15*np.sin(8*np.pi*x/n) + 0.1*np.sin(12*np.pi*x/n))
            elif s == 3:
                blk = max(2, n // 6)
                h = np.zeros(n)
                for j in range(0, n, 2 * blk):
                    h[j:j+blk] = 1.0
                h = fix_feasible(h + 0.25 * np.random.randn(n))
            elif s == 4:
                h_raw = np.random.uniform(0, 1, n)
                k = max(2, n // 10)
                h = fix_feasible(np.convolve(h_raw, np.ones(k) / k, mode='same'))
            else:
                h = fix_feasible(pm_seq(n) if False else 0.5 + 0.35 * np.random.randn(n))

            f = compute_c5(h)
            if f < best_c5:
                best_c5 = f
                best_h = h.copy()

            # SLSQP optimization
            for beta in [50, 120]:
                if tleft() < 10:
                    break
                _obj_beta = beta
                try:
                    res = minimize(_obj_fun, h, method='SLSQP',
                                   constraints=[{'type': 'eq', 'fun': _sum_eq}],
                                   bounds=[(0, 1)] * n,
                                   options={'maxiter': 350, 'ftol': 1e-11})
                    h_new = fix_feasible(res.x)
                    f = compute_c5(h_new)
                    if f < best_c5:
                        best_c5 = f
                        best_h = h_new.copy()
                    h = h_new
                except Exception:
                    pass

            # Gradient descent polish
            if tleft() < 8:
                continue
            for beta in [200, 400]:
                if tleft() < 6:
                    break
                h = projected_grad_descent(h, beta, steps=150, lr_init=0.015)
                f = compute_c5(h)
                if f < best_c5:
                    best_c5 = f
                    best_h = h.copy()

    # Phase 3: Intensive refinement of best solution
    if best_h is not None and tleft() > 20:
        h = best_h.copy()
        n = len(h)
        for beta in [300, 600, 1200, 2000]:
            if tleft() < 6:
                break
            # Gradient descent
            h = projected_grad_descent(h, beta, steps=200, lr_init=0.01)
            f = compute_c5(h)
            if f < best_c5:
                best_c5 = f
                best_h = h.copy()
            # SLSQP
            if tleft() < 5:
                break
            _obj_beta = beta
            try:
                res = minimize(_obj_fun, h, method='SLSQP',
                               constraints=[{'type': 'eq', 'fun': _sum_eq}],
                               bounds=[(0, 1)] * n,
                               options={'maxiter': 400, 'ftol': 1e-13})
                h_new = fix_feasible(res.x)
                f = compute_c5(h_new)
                if f < best_c5:
                    best_c5 = f
                    best_h = h_new.copy()
                h = h_new
            except Exception:
                pass

    # Phase 4: Fine coordinate descent
    if best_h is not None and len(best_h) <= 180 and tleft() > 10:
        for deltas, rnds in [([0.02, -0.02, 0.01, -0.01, 0.05, -0.05], 40),
                             ([0.005, -0.005, 0.002, -0.002], 30)]:
            if tleft() < 5:
                break
            best_h = coord_descent(best_h, deltas, rounds=rnds)
            f = compute_c5(best_h)
            if f < best_c5:
                best_c5 = f

    if best_h is None:
        best_h = fix_feasible(np.full(50, 0.5))
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
