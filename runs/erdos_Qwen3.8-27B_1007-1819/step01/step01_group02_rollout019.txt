from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    s = x.sum()
    if s == target:
        return np.clip(x, 0.0, 1.0)
    n_clipped = np.sum((x < 0) | (x > 1))
    if n_clipped >= n:
        return np.full(n, 0.5)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def obj_lse(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    c = corr_raw(h)
    m = c.max()
    return (m + np.log(np.sum(np.exp(beta * (c - m)))) / beta) * dx

def g_sum(x, n):
    return x.sum() - n / 2.0

def interp_resize(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xo = np.linspace(0, 2, n_from, endpoint=False)
    xn = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xn, xo, h)

def sgs_step(h, n, alpha, rng, temp=0.0):
    """One subgradient step on the max correlation."""
    c = corr_raw(h)
    # Add tiny noise to break ties
    if temp > 0:
        c = c + rng.randn(len(c)) * temp
    k_star = int(np.argmax(c))
    m = k_star - (n - 1)  # lag
    
    g = np.zeros(n)
    if m >= 0:
        idx = np.arange(n - m)
        g[idx] += (1.0 - h[idx + m])
        g[idx + m] -= h[idx]
    else:
        idx = np.arange(-m, n)
        g[idx + m] += (1.0 - h[idx])
        g[idx] -= h[idx + m]
    
    h_new = h - alpha * g
    h_new = project_feasible(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def obj_lse(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    c = corr_raw(h)
    m = c.max()
    return (m + np.log(np.sum(np.exp(beta * (c - m)))) / beta) * dx

def g_sum(x, n):
    return x.sum() - n / 2.0

def interp_resize(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xo = np.linspace(0, 2, n_from, endpoint=False)
    xn = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xn, xo, h)

def sgs_subgrad(h, n, m):
    g = np.zeros(n)
    if m >= 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        q = -m
        g[:n - q] += (1.0 - h[q:])
        g[q:] -= h[:n - q]
    return g

def sgs_step(h, n, alpha, rng):
    c = corr_raw(h)
    c += rng.randn(len(c)) * 1e-6
    k_star = int(np.argmax(c))
    m = k_star - (n - 1)
    g = sgs_subgrad(h, n, m)
    h_new = h - alpha * g
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h0)
    except NameError:
        h0 = np.full(180, 0.5)
        n_init = 180

    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n_init

    # Phase 1: SLSQP warmup on initial grid
    n_pts = best_n
    bounds = [(0.0, 1.0)] * n_pts
    cons = [{'type': 'eq', 'fun': g_sum, 'args': (n_pts,)}]
    for beta in [30.0, 100.0, 300.0]:
        if elapsed() > budget_s * 0.18:
            break
        res = minimize(obj_lse, best_h, args=(n_pts, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-14})
        if res.x is not None:
            h_opt = project_feasible(res.x, n_pts)
            c = compute_c5(h_opt)
            if c < best_c5:
                best_c5, best_h = c, h_opt.copy()

    # Phase 2: Subgradient descent with SLSQP restarts
    if elapsed() < budget_s * 0.22:
        n_pts = best_n
        h_sgs = best_h.copy()
        alpha = 0.04
        iter_count = 0
        while elapsed() < budget_s * 0.55:
            h_sgs = sgs_step(h_sgs, n_pts, alpha, rng)
            iter_count += 1
            if iter_count % 200 == 0:
                c_sgs = compute_c5(h_sgs)
                if c_sgs < best_c5:
                    best_c5, best_h = c_sgs, h_sgs.copy()
                alpha = max(alpha * 0.9985, 0.003)
            if iter_count % 3000 == 0 and elapsed() < budget_s * 0.5:
                res = minimize(obj_lse, h_sgs, args=(n_pts, 200.0),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-12})
                if res.x is not None:
                    h_opt = project_feasible(res.x, n_pts)
                    c = compute_c5(h_opt)
                    if c < best_c5:
                        best_c5, best_h = c, h_opt.copy()
                        h_sgs = best_h.copy()

    # Phase 3: Try different grid sizes via interpolation
    if elapsed() < budget_s * 0.4:
        for n_try in [n_init * 2, n_init // 2, n_init + 20]:
            if elapsed() > budget_s * 0.55:
                break
            if n_try <= 20 or n_try > 720 or n_try == best_n:
                continue
            hi = interp_resize(best_h, best_n, n_try)
            hi = project_feasible(hi, n_try)
            bnd_t = [(0.0, 1.0)] * n_try
            cns_t = [{'type': 'eq', 'fun': g_sum, 'args': (n_try,)}]
            for bt in [80.0, 250.0]:
                if elapsed() > budget_s * 0.55:
                    break
                res = minimize(obj_lse, hi, args=(n_try, bt),
                               method='SLSQP', bounds=bnd_t, constraints=cns_t,
                               options={'maxiter': 300, 'ftol': 1e-14})
                if res.x is not None:
                    h_opt = project_feasible(res.x, n_try)
                    c = compute_c5(h_opt)
                    if c < best_c5:
                        best_c5, best_h, best_n = c, h_opt.copy(), n_try

    # Phase 4: Pairwise swap local search
    if elapsed() < budget_s * 0.5:
        n_pts = best_n
        h_curr = best_h.copy()
        c_curr = best_c5
        t_end = min(budget_s * 0.82, elapsed() + (budget_s * 0.32))
        while elapsed() < t_end:
            i, j = rng.randint(0, n_pts, 2)
            if i == j:
                continue
            for d in (0.015, 0.04, 0.09, 0.005):
                ht = h_curr.copy()
                ht[i] += d
                ht[j] -= d
                if 0.0 <= ht[i] <= 1.0 and 0.0 <= ht[j] <= 1.0:
                    c = compute_c5(ht)
                    if c < c_curr - 1e-12:
                        h_curr, c_curr = ht, c
                        break
                ht = h_curr.copy()
                ht[i] -= d
                ht[j] += d
                if 0.0 <= ht[i] <= 1.0 and 0.0 <= ht[j] <= 1.0:
                    c = compute_c5(ht)
                    if c < c_curr - 1e-12:
                        h_curr, c_curr = ht, c
                        break
        if c_curr < best_c5:
            best_c5, best_h = c_curr, h_curr

    # Phase 5: Stochastic perturb + SLSQP restart
    if elapsed() < budget_s * 0.65:
        n_pts = best_n
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': g_sum, 'args': (n_pts,)}]
        n_tries = 6
        t_avail = budget_s * 0.82 - elapsed()
        if t_avail > n_tries * 5:
            for _ in range(n_tries):
                if elapsed() > budget_s * 0.8:
                    break
                hp = best_h.copy()
                np_idx = rng.choice(n_pts, rng.randint(3, 8), replace=False)
                hp[np_idx] += rng.uniform(-0.08, 0.08, len(np_idx))
                hp = project_feasible(hp, n_pts)
                res = minimize(obj_lse, hp, args=(n_pts, 400.0),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-12})
                if res.x is not None:
                    h_opt = project_feasible(res.x, n_pts)
                    c = compute_c5(h_opt)
                    if c < best_c5:
                        best_c5, best_h = c, h_opt.copy()

    # Phase 6: Final SLSQP polish with high beta
    if elapsed() < budget_s * 0.75:
        n_pts = best_n
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': g_sum, 'args': (n_pts,)}]
        for beta in [500.0, 900.0, 1500.0]:
            if elapsed() > budget_s * 0.9:
                break
            res = minimize(obj_lse, best_h, args=(n_pts, beta),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-15})
            if res.x is not None:
                h_opt = project_feasible(res.x, n_pts)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()

    # Phase 7: Fine coordinate descent
    if elapsed() < budget_s * 0.85:
        h_curr = best_h.copy()
        for _ in range(3):
            if elapsed() > budget_s * 0.96:
                break
            improved = False
            for i in range(best_n):
                for d in (0.002, 0.007, 0.02, 0.05, -0.002, -0.007, -0.02, -0.05):
                    ht = h_curr.copy()
                    ht[i] = float(np.clip(ht[i] + d, 0.0, 1.0))
                    ht = project_feasible(ht, best_n)
                    c = compute_c5(ht)
                    if c < best_c5 - 1e-13:
                        h_curr = ht
                        best_c5, best_h = c, h_curr.copy()
                        improved = True
                        break
            if not improved:
                break

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
