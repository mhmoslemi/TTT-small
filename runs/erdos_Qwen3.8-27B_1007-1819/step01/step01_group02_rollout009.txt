        else:
            hi = mid

    x_new = np.arange(n_to) * (2.0 / n_to)
    return np.interp(x_new, x_old, h)

def lp_iteration(h_star, n):
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    
    cons = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    
    for lag in range(-(n-1), n):
        i_min = max(0, -lag)
        i_max = min(n-1, n-1-lag)
        if i_min > i_max:
            continue
        
        # Linear part: sum_{i=i_min}^{i_max} h[i]
        # Bilinear part: f = sum_{i=i_min}^{i_max} h[i]*h[i+lag]
        
        # Gradient of bilinear part at h_star
        grad = np.zeros(n)
        for i in range(i_min, i_max + 1):
            grad[i] += h_star[i + lag]      # ∂(h_i * h_{i+lag})/∂h_i
            grad[i + lag] += h_star[i]      # ∂(h_i * h_{i+lag})/∂h_{i+lag}
        
        f_star = np.dot(h_star[i_min:i_max+1], h_star[i_min+lag:i_max+1+lag])
        
        # corr[lag] ≈ sum_{i=a}^{b} h_i - (grad @ h + f_star - grad @ h_star)
        # = (ones[a:b+1] - grad) @ h + (grad @ h_star - f_star)
        
        lin_coeffs = np.zeros(n)
        lin_coeffs[i_min:i_max+1] = 1.0
        lin_coeffs -= grad
        
        const = np.dot(grad, h_star) - f_star
        
        # Constraint: dx * (lin_coeffs @ h + const) ≤ t
        # → t - dx * (lin_coeffs @ h) ≥ dx * const
        cons.append(t - dx * (lin_coeffs @ h) >= dx * const)
    
    prob = cp.Problem(cp.Minimize(t), cons)
    prob.solve(solver=cp.SCS, verbose=False)
    
    if h.value is None:
        return None
    return h.value

import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.arange(n_from) * (2.0 / n_from)
    x_new = np.arange(n_to) * (2.0 / n_to)
    return np.interp(x_new, x_old, h)


def lp_iteration(h_star, n, max_lags=None):
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    cons = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]

    if max_lags is None:
        max_lags = n - 1
    else:
        max_lags = min(max_lags, n - 1)

    for lag in range(-max_lags, max_lags + 1):
        i_min = max(0, -lag)
        i_max = min(n - 1, n - 1 - lag)
        if i_min > i_max:
            continue

        grad = np.zeros(n)
        idx1 = np.arange(i_min, i_max + 1)
        idx2 = idx1 + lag
        grad[idx1] += h_star[idx2]
        grad[idx2] += h_star[idx1]

        f_star = float(np.dot(h_star[idx1], h_star[idx2]))

        lin_coeffs = np.zeros(n)
        lin_coeffs[i_min:i_max + 1] = 1.0
        lin_coeffs -= grad

        const = float(np.dot(grad, h_star)) - f_star

        cons.append(t - dx * (lin_coeffs @ h) >= dx * const)

    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=200000)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if h.value is None:
        return None
    return np.clip(h.value, 0.0, 1.0)


def find_active_lags(h, n, top_k=40):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    lags = np.arange(-(n - 1), n)
    order = np.argsort(corr)[::-1]
    active = lags[order[:top_k]]
    return sorted(active.tolist())


def smooth_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx


def sum_con(x, n_pts):
    return x.sum() - n_pts / 2.0
