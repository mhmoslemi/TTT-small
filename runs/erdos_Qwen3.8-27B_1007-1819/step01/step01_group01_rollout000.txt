                             constraints=[{'type': 'eq', 'fun': lambda p: p.sum() - best_n / 2}],
                             options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_feasible(res.x, best_n)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_c5, best_h = c, h_new.copy()
            except Exception:
                pass
        if tl() > 20:
            h_new, c = subgrad_descent(best_h, best_n, 0.003, 1000, 0.999, topk=1)
            if c < best_c5:
                best_c5, best_h = c, h_new.copy()

    # Phase 6: Tiny perturbation search
    if tl() > 15:
        for _ in range(200):
            if tl() < 5:
                break
            h_pert = best_h.copy()
            n_pert = np.random.randint(1, 4)
            idxs = np.random.choice(best_n, size=n_pert, replace=False)
            h_pert[idxs] = np.clip(h_pert[idxs] + np.random.randn(n_pert) * 0.02, 0, 1)
            h_pert = project_feasible(h_pert, best_n)
            c = compute_c5(h_pert)
            if c < best_c5:
                best_c5, best_h = c, h_pert.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
