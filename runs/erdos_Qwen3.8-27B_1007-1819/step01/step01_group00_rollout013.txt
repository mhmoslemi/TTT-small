def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5(x, beta):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    s = np.exp(beta * (corr - mx))
    return (mx + np.log(np.sum(s)) / beta) * dx

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n + 2):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = np.argsort(1.0 - h)
            else:
                idx = np.argsort(h)[::-1]
            for i in idx:
                if r > 0:
                    amt = min(r, 1.0 - h[i])
                    h[i] += amt
                else:
                    amt = min(-r, h[i])
                    h[i] -= amt
                r = target - np.sum(h)
    return h

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

        c, x = 0, i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t

def corr_gradient_lag(h, lag, eps=1e-7):
    """Compute gradient of corr[lag] with respect to h, via central differences."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        grad[i] = (np.correlate(h_plus, 1-h_plus, 'full')[lag] - 
                   np.correlate(h_minus, 1-h_minus, 'full')[lag]) / (2*eps)
    return grad
