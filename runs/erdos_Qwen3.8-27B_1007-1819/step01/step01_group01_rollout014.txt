        corr = compute_corr(h)
        m = np.max(corr)

        return (m + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

            obj, h0.copy(),
            method='SLSQP',
            bounds=[(0, 1)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2}],
            options={'maxiter': maxiter, 'ftol': 1e-14}

        new_val = compute_c5_val(h)

            curr = new_val
            if curr < best_val:
                best_val = curr
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
        T = max(T * 0.9995, 0.001)
    return best_h, best_val

def binary_greedy(h0, n, time_limit=15):
    h = h0.astype(float).copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    curr = compute_c5_val(h)
    best_h = h.copy()
    best_val = curr
    t_start = time.time()
    stall = 0
    while time.time() - t_start < time_limit and stall < 150:
        corr = compute_corr(h)
        k_star = np.argmax(corr)
        improved = False
        for _ in range(40):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            v = compute_c5_val(h)
            h[i], h[j] = h[j], h[i]
            if v < curr:
                h[i], h[j] = h[j], h[i]
                curr = v
                improved = True
                if curr < best_val:
                    best_val = curr
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                break
        if not improved:
            stall += 1
        else:
            stall = 0
    return best_h, best_val

def slsqp_refine(h0, n, alpha, maxiter=250):
    def obj(params):
        h = project_feasible(params, n)
        corr = compute_corr(h)
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -500, 500)
        return (m + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)
    try:
        res = minimize(
            obj, h0.copy(), method='SLSQP',
            bounds=[(0, 1)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def coord_descend(h0, n, n_rounds=25):
    h = h0.copy()
    curr = compute_c5_val(h)
    cands = np.array([0.0, 0.15, 0.3, 0.5, 0.7, 0.85, 1.0])
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            best_v = curr
            best_c = curr
            for rv in cands:
                if abs(rv - old) < 0.01:
                    continue
                h[i] = rv
                h = project_feasible(h, n)
                v = compute_c5_val(h)
                if v < best_c:
                    best_c = v
                    best_v = h[i]
            h[i] = best_v
            h = project_feasible(h, n)
            if best_c < curr:
                curr = best_c
                improved = True
        if not improved:
            break
    return h, curr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def time_left():
        return budget_s - 10 - (time.time() - t0)

    try:
        init = np.array(initial_h_values, dtype=float)
        n_init = len(init)
    except NameError:
        init = make_random_bin(80)
        n_init = 80
    init = project_feasible(init, n_init)
    best_h = init.copy()
    best_c5 = compute_c5_val(init)
    best_n = n_init

    # Phase 1: Diverse binary starts + SA
    n_list = sorted(set([n_init, 50, 60, 70, 80, 90, 100, 120]))
    n_list = [x for x in n_list if 20 <= x <= 200]
    for n_s in n_list:
        if time_left() < 20:
            break
        starts = []
        if n_s == n_init:
            starts.append(init.copy())
        starts.append(make_random_bin(n_s))
        starts.append(make_beatty(n_s))
        for p in [3, 4, 5, 7]:
            d = max(1, round(n_s / (2.0 * p)))
            if d < p:
                starts.append(make_periodic(n_s, p, d))
        starts.append(make_reflected(n_s))
        for h_s in starts:
            if time_left() < 10:
                break
            h_sa, v_sa = binary_sa(h_s, n_s, time_limit=min(8, max(3, time_left() - 8)))
            if v_sa < best_c5:
                best_c5 = v_sa
                best_h = h_sa.copy()
                best_n = n_s

    # Phase 2: Greedy refinement on best
    if time_left() > 15:
        h_g, v_g = binary_greedy(best_h, best_n, time_limit=min(20, time_left() - 10))
        if v_g < best_c5:
            best_c5 = v_g
            best_h = h_g.copy()

    # Phase 3: SLSQP continuous refinement
    if time_left() > 30:
        for alpha in [80, 250, 800]:
            if time_left() < 15:
                break
            h_t = slsqp_refine(best_h, best_n, alpha, maxiter=200)
            c = compute_c5_val(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 4: Coordinate descent
    if time_left() > 15:
        h_t, c = coord_descend(best_h, best_n, n_rounds=25)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 5: Second SA round + refine
    if time_left() > 20:
        h_sa, v_sa = binary_sa(best_h, best_n, time_limit=min(15, time_left() - 10))
        if v_sa < best_c5:
            best_c5 = v_sa
            best_h = h_sa.copy()
        if time_left() > 10:
            h_t = slsqp_refine(best_h, best_n, 500, maxiter=150)
            c = compute_c5_val(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 6: Try interpolation to nearby n values
    if time_left() > 30:
        for n_try in [best_n - 5, best_n + 5]:
            if n_try < 20 or time_left() < 15:
                continue
            h_interp = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, best_n), best_h)
            h_bin = (h_interp > 0.5).astype(float)
            fix_balance(h_bin, n_try)
            h_sa, v_sa = binary_sa(h_bin, n_try, time_limit=min(10, time_left() - 10))
            h_t = slsqp_refine(h_sa, n_try, 300, maxiter=150)
            c = compute_c5_val(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()
                best_n = n_try

    # Phase 7: Final polish
    if time_left() > 10:
        h_t, c = coord_descend(best_h, best_n, n_rounds=20)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
    if time_left() > 8:
        h_t = slsqp_refine(best_h, best_n, 2000, maxiter=100)
        c = compute_c5_val(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5_val(best_h)
    return best_h, best_c5, best_n
