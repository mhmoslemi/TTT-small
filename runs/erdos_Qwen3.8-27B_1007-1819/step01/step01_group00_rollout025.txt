import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def enforce_exact(y, S):
    y = np.clip(np.asarray(y, dtype=float), 0.0, 1.0)
    n = len(y)
    for _ in range(20 * n + 10):
        r = S - float(np.sum(y))
        if abs(r) < 1e-12:
            break
        if r > 0.0:
            room = 1.0 - y
            i = int(np.argmax(room))
            y[i] += min(r, room[i])
        else:
            i = int(np.argmax(y))
            y[i] -= min(-r, y[i])
    return y


def project_feasible(x, S):
    y = np.asarray(x, dtype=float)
    if y.size == 0:
        return y
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        v = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if v > S:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(y - tau, 0.0, 1.0)
    return enforce_exact(y, S)


def tm_seq(n):
    t = np.zeros(n, dtype=float)
    for i in range(n):
        x = i
        c = 0
        while x:
            c += x & 1
            x >>= 1
        t[i] = float(c & 1)
    return t


def make_pattern(n, kind):
    if n <= 0:
        return np.empty(0, dtype=float)
    if kind == 0:
        return tm_seq(n)
    if kind == 1:
        a = 0.5 * (np.sqrt(5.0) - 1.0)
        return (np.mod(np.arange(n) * a, 1.0) < 0.5).astype(float)
    if kind == 2:
        return (np.random.random(n) < 0.5).astype(float)
    if kind == 3:
        x = np.linspace(0.0, 4.0 * np.pi, n)
        return 0.5 + 0.35 * np.sin(x)
    if kind == 4:
        p = max(3, n // 8)
        return ((np.arange(n) // p) % 2 == 0).astype(float)
    y = np.random.random(n)
    k = max(1, n // 8)
    return np.convolve(y, np.ones(k) / k, mode="same")


def resample_h(src, n):
    if src is None or len(src) == 0:
        return (np.random.random(n) < 0.5).astype(float)
    m = len(src)
    if m == n:
        return src.astype(float, copy=True)
    if n % m == 0:
        return np.repeat(src, n // m).astype(float)
    if m % n == 0:
        return src.reshape(n, m // n).mean(axis=1).astype(float)
    old = np.linspace(0.0, 1.0, m)
    new = np.linspace(0.0, 1.0, n)
    return np.interp(new, old, src)


def lag_values(h):
    n = len(h)
    if n == 0:
        return np.empty(0)
    q = 1.0 - h
    out = np.empty(2 * n - 1)
    for idx in range(n - 1):
        m = n - 1 - idx
        out[idx] = np.dot(h[m:], q[:n - m])
    out[n - 1] = np.dot(h, q)
    for idx in range(n, 2 * n - 1):
        m = idx - n + 1
        out[idx] = np.dot(h[:n - m], q[m:])
    return out


def grad_at_idx(h, idx, n):
    g = np.zeros(n, dtype=float)
    if idx < n - 1:
        m = n - 1 - idx
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    elif idx == n - 1:
        return 1.0 - 2.0 * h
    else:
        m = idx - n + 1
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g


def smooth_weighted_grad(h, c, beta):
    n = len(h)
    if c.size == 0:
        return np.zeros(n, dtype=float)
    mx = float(np.max(c))
    if beta <= 0.0:
        w = np.full(c.size, 1.0 / c.size)
    else:
        z = beta * (c - mx)
        z = z - float(np.max(z))
        e = np.exp(z)
        s = float(np.sum(e))
        if s <= 0.0:
            w = np.full(c.size, 1.0 / c.size)
        else:
            w = e / s
    g = np.zeros(n, dtype=float)
    for idx in range(c.size):
        wi = float(w[idx])
        if wi <= 1e-14:
            continue
        if idx < n - 1:
            m = n - 1 - idx
            g[m:] += wi * (1.0 - h[:n - m])
            g[:n - m] -= wi * h[m:]
        elif idx == n - 1:
            g += wi * (1.0 - 2.0 * h)
        else:
            m = idx - n + 1
            g[:n - m] += wi * (1.0 - h[m:])
            g[m:] -= wi * h[:n - m]
    return g


def smooth_value_grad(h, beta):
    c = lag_values(h)
    if c.size == 0:
        return 0.0, np.zeros(len(h))
    mx = float(np.max(c))
    if beta <= 0.0:
        val = mx
    else:
        z = beta * (c - mx)
        z = z - float(np.max(z))
        val = mx + float(np.log(np.sum(np.exp(z)))) / beta
    grad = smooth_weighted_grad(h, c, beta)
    return val, grad


def pen_obj(x, beta, rho):
    n = len(x)
    val, _ = smooth_value_grad(x, beta)
    s = float(np.sum(x)) - 0.5 * n
    return val + 0.5 * rho * s * s


def pen_jac(x, beta, rho):
    n = len(x)
    _, g = smooth_value_grad(x, beta)
    s = float(np.sum(x)) - 0.5 * n
    return g + rho * s * np.ones(n, dtype=float)


def lbfgs_polish(h, beta, rho, maxiter=80):
    n = len(h)
    S = 0.5 * n
    x0 = project_feasible(h, S)
    if n < 2:
        return x0, compute_c5(x0)
    if n > 80:
        maxiter = min(maxiter, 50)
    elif n > 50:
        maxiter = min(maxiter, 80)
    try:
        res = minimize(
            pen_obj,
            x0,
            jac=pen_jac,
            args=(beta, rho),
            method="L-BFGS-B",
            bounds=[(0.0, 1.0)] * n,
            options={
                "maxiter": maxiter,
                "gtol": 1e-8,
                "maxfun": max(120, maxiter * 40),
            },
        )
        xn = project_feasible(res.x, S)
        return xn, compute_c5(xn)
    except Exception:
        return x0, compute_c5(x0)


def adjust_one(h, i, d, S):
    v = h[i] + d
    if v <= 0.0 or v >= 1.0:
        return None
    t = h.copy()
    t[i] = v
    r = S - float(np.sum(t))
    if r > 1e-12:
        room = 1.0 - t
        room[i] = 0.0
        order = np.argsort(-room)
        rem = r
        for j in order:
            if rem <= 1e-12:
                break
            take = float(room[j])
            if take > rem:
                take = rem
            t[j] += take
            rem -= take
        if rem > 1e-9:
            return project_feasible(t, S)
    elif r < -1e-12:
        val = t.copy()
        val[i] = 0.0
        order = np.argsort(-val)
        rem = -r
        for j in order:
            if rem <= 1e-12:
                break
            take = float(val[j])
            if take > rem:
                take = rem
            t[j] -= take
            rem -= take
        if rem > 1e-9:
            return project_feasible(t, S)
    t = np.clip(t, 0.0, 1.0)
    if abs(S - float(np.sum(t))) < 1e-9:
        return t
    return project_feasible(t, S)


def coordinate_refine(h, rounds=3, deadline=None):
    n = len(h)
    S = 0.5 * n
    best = h.copy()
    bestc = compute_c5(best)
    deltas = (0.2, -0.2, 0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01)
    for _ in range(rounds):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if deadline is not None and time.time() > deadline:
                return best, bestc
            lb = best
            lbc = bestc
            for d in deltas:
                tr = adjust_one(best, i, d, S)
                if tr is None:
                    continue
                c = compute_c5(tr)
                if c < lbc - 1e-12:
                    lb = tr
                    lbc = c
            if lbc < bestc - 1e-12:
                best = lb
                bestc = lbc
                improved = True
        if not improved:
            break
    return best, bestc
