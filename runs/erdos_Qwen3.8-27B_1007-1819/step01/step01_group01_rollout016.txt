def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_subgrad(h, n, n_active=5, eps=0.01):
    """Compute subgradient of C5 at h using active lags."""
    corr = compute_corr(h)
    cmax = np.max(corr)
    active = np.where(corr >= cmax - eps * cmax)[0]
    if len(active) < n_active:
        active = np.argsort(corr)[-n_active:]
    
    grad = np.zeros(n)
    for k in active:
        tau = k - (n - 1)
        # grad[j] += (1-h[j-tau])*I[j-tau in [0,n)) - h[j+tau]*I[j+tau in [0,n))
        for j in range(n):
            g = 0.0
            jt = j - tau
            if 0 <= jt < n:
                g += (1.0 - h[jt])
            jp = j + tau
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += g
    return grad / len(active), cmax, corr

def subgrad_descent(h0, n, lr0=0.02, max_iter=5000, decay=0.999):
    h = h0.copy()
    lr = lr0
    best_h = h.copy()
    best_c = compute_c5(h)
    stall = 0
    
    for it in range(max_iter):
        grad, cmax, corr = compute_subgrad(h, n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.1, lr0)  # increase slightly
        else:
            stall += 1
            lr *= decay
        
        h = h_new
        if stall > 200:
            break
        if abs(c_new - best_c) < 1e-10:
            stall += 1
            if stall > 500:
                break
    
    return best_h, best_c
