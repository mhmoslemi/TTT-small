import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(3 * n + 10):
            r = target - np.sum(h)
            if abs(r) < 1e-14:
                break
            if r > 0:
                i = int(np.argmin(1.0 - h))
                h[i] = min(1.0, h[i] + r)
            else:
                i = int(np.argmax(h))
                h[i] = max(0.0, h[i] + r)
    return h

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mx = np.max(corr)
    shifted = np.clip(beta * (corr - mx), -400, 400)
    s = np.sum(np.exp(shifted))
    return (mx + np.log(s) / beta) * dx

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def soft_coord_descent(h, max_rounds=30):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    deltas = [0.04, -0.04, 0.08, -0.08, 0.12, -0.12]
    for _ in range(max_rounds):
        improved = False
        corr = get_corr(h)
        active = set(np.argpartition(corr, -8)[-8:].tolist())
        for i in range(n):
            for d in deltas:
                v = h[i] + d
                if v < 0 or v > 1:
                    continue
                h_t = h.copy()
                h_t[i] = v
                h_t = fix_feasible(h_t)
                obj = compute_c5(h_t)
                if obj < cur - 1e-14:
                    cur = obj
                    h = h_t
                    improved = True
                    break
            if improved:
                corr = get_corr(h)
                active = set(np.argpartition(corr, -8)[-8:].tolist())
        if not improved:
            break
    return h

            cur = obj
            h = h_t
