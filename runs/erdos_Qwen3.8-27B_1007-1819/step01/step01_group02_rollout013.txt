                    best_h = ht.copy()
                    best_c5 = c5t
                    cur = nv
                    improved = True
        if not improved:
            break
    return best_h, best_c5


def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_init = len(h0)
    except NameError:
        h0 = np.full(180, 0.5)
        n_init = 180

    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n_init

    n_pts = n_init
    bounds = [(0.0, 1.0)] * n_pts

    # Phase 1: SLSQP with increasing beta
    for beta in [30.0, 60.0, 120.0]:
        if elapsed() > budget_s * 0.40:
            break
        cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_pts,)}]
        res = minimize(
            obj_wrapper, best_h, args=(n_pts, beta),
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 600, 'ftol': 1e-15, 'disp': False}
        )
        if hasattr(res, 'x') and res.x is not None:
            h_opt = project_feasible(res.x, n_pts)
            c5_val = compute_c5(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()

    # Phase 2: Projected gradient with analytic gradient
    if elapsed() < budget_s * 0.65:
        h_pg, c5_pg = projected_gradient(best_h, n_pts, max_iters=400, beta=100)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: Multiple starts with perturbation
    for pstart in range(3):
        if elapsed() > budget_s * 0.72:
            break
        h_pert = best_h + rng.randn(n_pts) * 0.03
        h_pert = project_feasible(h_pert, n_pts)
        cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_pts,)}]
        res_p = minimize(
            obj_wrapper, h_pert, args=(n_pts, 80.0),
            method='SLSQP', bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-15, 'disp': False}
        )
        if hasattr(res_p, 'x') and res_p.x is not None:
            h_opt_p = project_feasible(res_p.x, n_pts)
            c5_p = compute_c5(h_opt_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_opt_p.copy()

    # Phase 4: Coordinate descent refinement
    if elapsed() < budget_s * 0.85:
        h_cd, c5_cd = coord_descent(best_h, n_pts, epochs=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Try finer resolution n=360
    if elapsed() < budget_s * 0.50:
        n2 = 360
        h2 = project_feasible(interp_to_size(best_h, n_pts, n2), n2)
        bounds2 = [(0.0, 1.0)] * n2
        for beta2 in [60.0, 120.0]:
            if elapsed() > budget_s * 0.72:
                break
            cons2 = [{'type': 'eq', 'fun': constraint_sum, 'args': (n2,)}]
            res2 = minimize(
                obj_wrapper, h2, args=(n2, beta2),
                method='SLSQP', bounds=bounds2,
                constraints=cons2,
                options={'maxiter': 400, 'ftol': 1e-15, 'disp': False}
            )
            if hasattr(res2, 'x') and res2.x is not None:
                h_opt2 = project_feasible(res2.x, n2)
                c5_2 = compute_c5(h_opt2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h_opt2.copy()
                    best_n = n2
                    h2 = h_opt2.copy()
        if elapsed() < budget_s * 0.78:
            h_pg2, c5_pg2 = projected_gradient(h2, n2, max_iters=250, beta=100)
            if c5_pg2 < best_c5:
                best_c5 = c5_pg2
                best_h = h_pg2.copy()
                best_n = n2
        if elapsed() < budget_s * 0.88 and best_n == n2:
            h_cd2, c5_cd2 = coord_descent(best_h, n2, epochs=2)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    # Phase 6: Final coordinate descent on best solution
    if elapsed() < budget_s * 0.92:
        h_fin, c5_fin = coord_descent(best_h, best_n, epochs=2)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
