    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def fix_feasible(h):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(n):
        r = target - h.sum()
        if abs(r) < 1e-13:
            break
        if r > 0:
            i = int(np.argmin(1.0 - h))
            h[i] = min(1.0, h[i] + r)
        else:
            i = int(np.argmax(h))
            h[i] = max(0.0, h[i] + r)
    return h

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = np.max(corr)
    return mx + np.log(np.sum(np.exp(beta * (corr - mx)))) / beta

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    return t

def coord_desc(h, max_rounds=30, t0=None, deadline=None):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    vals = [0.0, 1.0, 0.25, 0.5, 0.75]
    deltas = [0.05, -0.05, 0.1, -0.1, 0.15, -0.15]
    for _ in range(max_rounds):
        if t0 is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old = h[i]
            for v in vals:
                if abs(v - old) < 1e-6:
                    continue
                h[i] = v
                h = fix_feasible(h)
                c = compute_c5(h)
                if c < cur - 1e-12:
                    cur = c
                    old = v
                    improved = True
                else:
                    h[i] = old
                    h = fix_feasible(h)
            for dv in deltas:
                v = old + dv
                if v < 0 or v > 1:
                    continue
                h[i] = v
                h = fix_feasible(h)
                c = compute_c5(h)
                if c < cur - 1e-12:
                    cur = c
                    old = v
                    improved = True
                else:
                    h[i] = old
                    h = fix_feasible(h)
        if not improved:
            break
    return h, cur

def binary_sa(h_bin, t0, deadline, iters=5000):
    n = len(h_bin)
    h = h_bin.copy()
    dx = 2.0 / n
    cur = compute_c5(h)
    T = 0.02
    for it in range(iters):
        if time.time() > deadline:
            break
        T = max(0.001, T * 0.9995)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(np.random.choice(zeros))
        j = int(np.random.choice(ones))
        h[i] = 1.0
        h[j] = 0.0
        new_c = compute_c5(h)
        delta = new_c - cur
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            cur = new_c
        else:
            h[i] = 0.0
            h[j] = 1.0
    return h, cur

def lbfgs_opt(h, beta, t0, deadline, maxiter=300):
    n = len(h)
    def obj(x):
        hf = fix_feasible(x)
        return smooth_max_obj(hf, beta)
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                      bounds=[(0, 1)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return fix_feasible(res.x)
    except Exception:
        return h
