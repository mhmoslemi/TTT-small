        k = idx - offset
        # h[i] coefficient in corr at lag k:
        # term1: (1-h[i+k]) for i+k in [0,n)
        # term2: -h[i-k] for i-k in [0,n)
        if k >= 0:
            coeff1 = 1.0 - h[k:nn]
            coeff2 = np.zeros(nn)
            coeff2[:nn-k] = -h[:nn-k]
        else:
            coeff1 = np.zeros(nn)
            coeff1[-k:] = 1.0 - h[:nn+k]
            coeff2 = -h[-k:]
            tmp = np.zeros(nn)
            tmp[:nn+k] = -h[-k:]
            coeff2 = tmp
        grad += coeff1 + coeff2
    # Projected gradient step
    h_new = np.clip(h - lr * grad, 0.0, 1.0)
    h_new = project_feasible(h_new, nn)
    return h_new


def coord_descent_true(h, n, epochs=20, n_sweeps=3):
    """Coordinate descent on the true max-correlation objective."""
    best_c5 = compute_c5(h)
    h = h.copy()
    dx = 2.0 / n
    for ep in range(epochs):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            # Try moving h[i] to several candidate values
            old = h[i]
            candidates = [0.0, 1.0, old - 0.1, old - 0.03, old + 0.03, old + 0.1]
            if i % n_sweeps == 0:
                candidates += [old - 0.3, old + 0.3, 0.25, 0.75]
            best_v = old
            best_f = compute_c5(h)
            for v in candidates:
                v = np.clip(v, 0.0, 1.0)
                h[i] = v
                h = project_feasible(h, n)
                f = compute_c5(h)
                if f < best_f - 1e-12:
                    best_f = f
                    best_v = v
            h[i] = best_v
            h = project_feasible(h, n)
            if best_f < best_c5 - 1e-12:
                best_c5 = best_f
                improved = True
        if not improved:
            break
    return h, best_c5


def swap_local_search(h, n, max_swaps=3000):
    """Try swapping values between pairs of coordinates."""
    best_c5 = compute_c5(h)
    h = h.copy()
    dx = 2.0 / n
    rng = np.random.RandomState(12345)
    for _ in range(max_swaps):
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        ai, aj = h[i], h[j]
        # Try swapping
        h[i], h[j] = aj, ai
        h = project_feasible(h, n)
        f = compute_c5(h)
        if f >= best_c5:
            h[i], h[j] = ai, aj
            h = project_feasible(h, n)
        else:
            best_c5 = f
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    rng = np.random.RandomState(seed)

    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_init = len(h0)
    except NameError:
        h0 = 0.5 + 0.3 * np.sin(2.0 * np.pi * np.arange(180) / 90.0)
        n_init = 180

    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n_init

    # Phase 1: L-BFGS-B with annealed beta on the original size
    n_pts = n_init
    if elapsed() > budget_s * 0.85:
        return best_h, best_c5, best_n

    bounds = [(0.0, 1.0)] * n_pts
    sum_con = n_pts / 2.0

    def obj_lbfgs(x, beta):
        h = np.clip(x, 0.0, 1.0)
        dx = 2.0 / n_pts
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

    for beta in [30.0, 80.0, 200.0, 500.0]:
        if elapsed() > budget_s * 0.5:
            break
        def obj_wrapped(x, b=beta):
            return obj_lbfgs(x, b)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - sum_con, 'args': ()}]
        res = minimize(
            obj_wrapped, best_h, method='L-BFGS-B',
            bounds=bounds, options={'maxiter': 800, 'ftol': 1e-14, 'gtol': 1e-12}
        )
        if res.x is not None:
            h_opt = project_feasible(res.x, n_pts)
            c5_v = compute_c5(h_opt)
            if c5_v < best_c5:
                best_c5 = c5_v
                best_h = h_opt.copy()

    # Phase 2: Active-set subgradient refinement
    if elapsed() < budget_s * 0.35:
        h_curr = best_h.copy()
        lr = 0.005
        for it in range(200):
            if elapsed() > budget_s * 0.55:
                break
            h_new = active_subgrad_step(h_curr, n_pts, lr, 0.02)
            f_new = compute_c5(h_new)
            if f_new < best_c5 + 1e-12:
                h_curr = h_new
                if f_new < best_c5:
                    best_c5 = f_new
                    best_h = h_new.copy()
            else:
                lr *= 0.95
            if lr < 1e-5:
                break

    # Phase 3: Coordinate descent on true max
    if elapsed() < budget_s * 0.4:
        h_cd, c5_cd = coord_descent_true(best_h, best_n, epochs=10, n_sweeps=5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Try different resolutions
    for n_alt in [90, 120, 240]:
        if elapsed() > budget_s * 0.75:
            break
        # Interpolate best to new size
        x_old = np.arange(best_n) * (2.0 / best_n)
        x_new = np.arange(n_alt) * (2.0 / n_alt)
        h_alt = project_feasible(np.interp(x_new, x_old, best_h), n_alt)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()
            best_n = n_alt
        # Quick L-BFGS-B at this size
        if elapsed() > budget_s * 0.8:
            continue
        bnds_alt = [(0.0, 1.0)] * n_alt
        sc = n_alt / 2.0

        def obj_alt(x, b=100.0, nn=n_alt):
            h = np.clip(x, 0.0, 1.0)
            dxv = 2.0 / nn
            corr = np.correlate(h, 1.0 - h, mode="full")
            m = np.max(corr)
            return (m + np.log(np.sum(np.exp(b * (corr - m)))) / b) * dxv

        res_alt = minimize(
            obj_alt, h_alt, method='L-BFGS-B',
            bounds=bnds_alt, options={'maxiter': 400, 'ftol': 1e-14}
        )
        if res_alt.x is not None:
            h_try = project_feasible(res_alt.x, n_alt)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_alt

    # Phase 5: Active-set refinement on best
    if elapsed() < budget_s * 0.3:
        h_curr = best_h.copy()
        lr = 0.003
        for it in range(150):
            if elapsed() > budget_s * 0.7:
                break
            h_new = active_subgrad_step(h_curr, best_n, lr, 0.01)
            f_new = compute_c5(h_new)
            if f_new < best_c5 + 1e-13:
                h_curr = h_new
                if f_new < best_c5:
                    best_c5 = f_new
                    best_h = h_new.copy()
            else:
                lr *= 0.97
            if lr < 1e-5:
                break

    # Phase 6: Final coordinate descent
    if elapsed() < budget_s * 0.25:
        h_cd2, c5_cd2 = coord_descent_true(best_h, best_n, epochs=8, n_sweeps=3)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 7: Random perturbation search (small kicks)
    if elapsed() < budget_s * 0.15:
        for trial in range(200):
            if elapsed() > budget_s * 0.85:
                break
            h_kick = best_h + rng.randn(best_n) * 0.02
            h_kick = project_feasible(h_kick, best_n)
            c5_k = compute_c5(h_kick)
            if c5_k < best_c5:
                best_c5 = c5_k
                best_h = h_kick.copy()
            # If the kick was close, do a mini coordinate descent
            elif c5_k < best_c5 + 0.001:
                h_m, c5_m = coord_descent_true(h_kick, best_n, epochs=3, n_sweeps=5)
                if c5_m < best_c5:
                    best_c5 = c5_m
                    best_h = h_m.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
