from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(1000):
        r = target - np.sum(h)
        if abs(r) < 1e-13:
            break
        if r > 0:
            idx = int(np.argmin(1.0 - h))
            h[idx] = min(1.0, h[idx] + r)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + r)
    return h

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    vals = beta * (corr - mx)
    vals = np.where(vals > -60, vals, -60)
    return (mx + np.log(np.sum(np.exp(vals))) / beta) * dx

    return np.sum(x) - len(x) / 2.0

    t = np.zeros(n, dtype=float)

        t[i] = c % 2
    return t

    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    if steps is None:
        steps = [0.08, -0.08, 0.04, -0.04, 0.12, -0.12]
    for _ in range(rounds):
        moved = False
        for i in range(n):
            bv, bo = h[i], cur
            for d in steps:
                v = h[i] + d
                if v < 0 or v > 1:
                    continue
                tr = h.copy()
                tr[i] = v
                tr = fix_feasible(tr)
                o = compute_c5(tr)
                if o < bo - 1e-13:
                    bo, bv = o, v
            if bv != h[i]:
                h[i] = bv
                h = fix_feasible(h)
                cur = bo
                moved = True
        if not moved:
            break
    return h
