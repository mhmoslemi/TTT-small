from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(4 * n):
        r = target - np.sum(h)
        if abs(r) < 1e-13:
            break
        if r > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                a = min(r, 1.0 - h[i])
                h[i] += a
                r -= a
                if r < 1e-13:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                a = min(-r, h[i])
                h[i] -= a
                r += a
                if -r < 1e-13:
                    break
    return h

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return (mx + np.log(s) / beta) * dx

def sum_res(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_opt(h0, beta, maxit=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    jac = np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_res, 'jac': lambda x: jac}]
    res = minimize(lambda x: smooth_obj(x, beta), h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxit, 'ftol': 1e-12})
    return fix_feasible(res.x)

def basin_hop(h0, n_iterations, sigma, beta, rng, time_limit):
    h = h0.copy()
    cur = compute_c5(h)
    for it in range(n_iterations):
        if time.time() > time_limit:
            break
        # Perturb
        h_pert = h + rng.normal(0, sigma, len(h))
        h_pert = fix_feasible(h_pert)
        # Optimize
        h_opt = slsqp_opt(h_pert, beta, maxit=200)
        c_opt = compute_c5(h_opt)
        if c_opt < cur:
            h = h_opt
            cur = c_opt
    return h, cur

def coord_refine(h, n_rounds=8):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    deltas = [0.03, -0.03, 0.07, -0.07, 0.12, -0.12, 0.2, -0.2, 0.4, -0.4]
    changed = True
    for _ in range(n_rounds):

        changed = False
        for i in range(n):
            for d in deltas:

                if v < 0 or v > 1:

                h[i] = v
                h = fix_feasible(h)
                c = compute_c5(h)
                if c < cur - 1e-13:
                    cur = c
                    changed = True
                    break
    return h, cur

    h_bin = fix_feasible(h_bin)
    c_bin = compute_c5(h_bin)
    c_orig = compute_c5(h)
    if c_bin < c_orig:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(4 * n):
        r = target - np.sum(h)
        if abs(r) < 1e-14:
            break
        if r > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                a = min(r, 1.0 - h[i])
                h[i] += a
                r -= a
                if r < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                a = min(-r, h[i])
                h[i] -= a
                r += a
                if -r < 1e-14:
                    break
    return h

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return (mx + np.log(s) / beta) * dx

    return np.sum(x) - len(x) / 2.0

    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_res, 'jac': lambda x: np.ones(len(x))}]
    res = minimize(lambda x: smooth_obj(x, beta), h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxit, 'ftol': 1e-13})
    return fix_feasible(res.x)

        h_res, c_res = basin_hop(best_h, 10, sigma, 600, rng, t_limit)
        if c_res < best_c5:
            best_c5, best_h = c_res, h_res

    # Phase 4: Lag-targeted optimization
    if time.time() < t_limit:
        h_lt, c_lt = lag_target_opt(best_h, n_iter=25)
        if c_lt < best_c5:
            best_c5, best_h = c_lt, h_lt

    # Phase 5: More SLSQP after lag targeting
    if time.time() < t_limit:
        h_opt = slsqp_opt(best_h, 800, maxit=500)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt

    # Phase 6: Pair-flip refinement
    if time.time() < t_limit:
        h_pf, c_pf = pair_flip_refine(best_h, n_rounds=4, rng=rng)
        if c_pf < best_c5:
            best_c5, best_h = c_pf, h_pf

    # Phase 7: Coordinate descent
    if time.time() < t_limit:
        h_ref, c_ref = coord_refine(best_h, n_rounds=12)
        if c_ref < best_c5:
            best_c5, best_h = c_ref, h_ref

    # Phase 8: Binarize check
    h_bin = (best_h > 0.5).astype(float)
    h_bin = fix_feasible(h_bin)
    c_bin = compute_c5(h_bin)
    if c_bin < best_c5:
        best_c5, best_h = c_bin, h_bin

    # Phase 9: Try other sizes with Thue-Morse start
    for nn in [50, 60, 80, 120]:
        if time.time() > t_limit:
            break
        h_try = fix_feasible(tm_seq(nn))
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try.copy()
        h_try = slsqp_opt(h_try, 400, maxit=200)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try.copy()
        h_try = slsqp_opt(h_try, 1000, maxit=300)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5, best_h = c_try, h_try.copy()

    # Phase 10: Final polish on best
    if time.time() < t_limit:
        h_opt = slsqp_opt(best_h, 1200, maxit=400)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt

    if time.time() < t_limit:
        h_ref, c_ref = coord_refine(best_h, n_rounds=6)
        if c_ref < best_c5:
            best_c5, best_h = c_ref, h_ref

    if time.time() < t_limit:
        h_pf, c_pf = pair_flip_refine(best_h, n_rounds=3, rng=rng)
        if c_pf < best_c5:
            best_c5, best_h = c_pf, h_pf

    if best_h is None:
        best_h = fix_feasible(np.full(50, 0.5))
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
