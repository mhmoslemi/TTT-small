from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def sum_constraint(x, n_pts):
    return x.sum() - n_pts / 2.0

def make_bounds(n_pts):
    return [(0.0, 1.0)] * n_pts

def slsqp_optimize(h, n, beta, maxiter=300):
    bounds = make_bounds(n)
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    res = minimize(smooth_obj, h, args=(n, beta), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if hasattr(res, 'x') and res.x is not None:
        return project_feasible(res.x, n)
    return h

    best_c5 = compute_c5(h)
    h_curr = h.copy()
    target = n / 2.0
    
    # Start with moderate deltas, shrink over time
    base_deltas = [0.1, 0.05, 0.02]
    
    while time.time() - t_start < time_limit:
        improved = False
        # Determine which pairs to try based on h values
        # Prioritize pairs with large difference in h values
        idx = np.argsort(h_curr)
        
        # Try pairs: low-h with high-h
        tried = 0
        max_tries = 20000
        
        for a in range(0, n, 1):
            if time.time() - t_start > time_limit:

            for b in range(n-1, -1, -1):
                if a == b:
                    continue
                tried += 1
                if tried > max_tries:
                    break
                
                if h_curr[a] <= 0.005 or h_curr[b] >= 0.995:

                if abs(h_curr[a] - h_curr[b]) < 0.01:
                    continue
                
                # Try moving from b (high) to a (low)
                hi_val, lo_val = h_curr[b], h_curr[a]
                for d in base_deltas:
                    if d > hi_val - 0.001 or d > 0.999 - lo_val:

                    h_curr[b] -= d
                    h_curr[a] += d

    return h_curr, best_c5

These are actually the same by reindexing! Because sum_i h[i+k](1-h[i]) = sum_j h[j](1-h[j-k]) where j = i+k.

So the full correlation is symmetric in the relevant sense.

The key shifts are those where the correlation value is near the maximum. For those shifts, I want to reduce the correlation.

For a specific shift k (say positive), corr_k = sum_i h[i+k] * (1-h[i])
To reduce this, I want to:
- Decrease h[i+k] for terms where h[i] is small (since (1-h[i]) is large)
- Increase h[i] for terms where h[i+k] is large (since h[i+k] is large)

This gives a natural set of (i, j) pairs to try: for each i where h[i] is small and h[i+k] is large, try transferring mass from h[i+k] to h[i].

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def sum_constraint(x, n_pts):
    return x.sum() - n_pts / 2.0

def make_bounds(n_pts):
    return [(0.0, 1.0)] * n_pts

def do_slsqp(h, n, beta, maxiter=300):
    bounds = make_bounds(n)
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    res = minimize(smooth_obj, h, args=(n, beta), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if hasattr(res, 'x') and res.x is not None:
        return project_feasible(res.x, n)
    return h

def pairwise_search(h, n, time_limit, seed=0):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    deltas = [0.1, 0.05, 0.02]

    while time.time() - t_start < time_limit:
        improved = False
        order = rng.permutation(n)
        for a in range(n):
            if time.time() - t_start > time_limit:
                return h_curr, best_c5
            i = order[a]
            for b in range(a + 1, n):
                j = order[b]
                vi, vj = h_curr[i], h_curr[j]
                if vi < 0.005 or vj < 0.005:
                    continue
                if vi > 0.995 and vj > 0.995:
                    continue
                # move i -> j
                for d in deltas:
                    if d > vi or d > 1.0 - vj:
                        continue
                    h_curr[i] -= d
                    h_curr[j] += d
                    c5 = compute_c5(h_curr)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        improved = True
                    else:
                        h_curr[i] += d
                        h_curr[j] -= d
                    break
                # move j -> i
                vi, vj = h_curr[i], h_curr[j]
                for d in deltas:
                    if d > vj or d > 1.0 - vi:
                        continue
                    h_curr[j] -= d
                    h_curr[i] += d
                    c5 = compute_c5(h_curr)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        improved = True
                    else:
                        h_curr[j] += d
                        h_curr[i] -= d
                    break
        if not improved:
            deltas = [d * 0.5 for d in deltas]
            if min(deltas) < 0.003:
                break
    return h_curr, best_c5

def fine_search(h, n, time_limit, seed=100):
    """Smaller steps focused on pairs with intermediate h values."""
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    deltas = [0.02, 0.01, 0.005]
    while time.time() - t_start < time_limit:
        improved = False
        order = rng.permutation(n)
        for a in range(n):
            if time.time() - t_start > time_limit:
                return h_curr, best_c5
            i = order[a]
            for b in range(a + 1, n):
                j = order[b]
                vi, vj = h_curr[i], h_curr[j]
                if vi < 0.02 or vj < 0.02:
                    continue
                if vi > 0.98 and vj > 0.98:
                    continue
                if abs(vi - vj) < 0.005:
                    continue
                for d in deltas:
                    if d > vi or d > 1.0 - vj:
                        continue
                    h_curr[i] -= d
                    h_curr[j] += d
                    c5 = compute_c5(h_curr)
                    if c5 < best_c5 - 1e-15:
                        best_c5 = c5
                        improved = True
                    else:
                        h_curr[i] += d
                        h_curr[j] -= d
                    break
                vi, vj = h_curr[i], h_curr[j]
                for d in deltas:
                    if d > vj or d > 1.0 - vi:
                        continue
                    h_curr[j] -= d
                    h_curr[i] += d
                    c5 = compute_c5(h_curr)
                    if c5 < best_c5 - 1e-15:
                        best_c5 = c5
                        improved = True
                    else:
                        h_curr[j] += d
                        h_curr[i] -= d
                    break
        if not improved:
            deltas = [d * 0.5 for d in deltas]
            if min(deltas) < 0.001:
                break
    return h_curr, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    except NameError:
        n = 180
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: SLSQP with escalating smoothness
    for beta in [80, 200, 500]:
        if time.time() - t0 > budget_s * 0.28:
            break
        h_new = do_slsqp(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Broad pairwise local search
    t_now = time.time()
    if t_now < budget_s * 0.25:
        alloc = min(budget_s * 0.45, budget_s - (t_now - t0) - budget_s * 0.3)
        if alloc > 30:
            h2, c5_2 = pairwise_search(best_h, n, alloc, seed=seed)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 3: SLSQP refinement at high beta
    t_now = time.time()
    if t_now < budget_s * 0.65:
        h_new = do_slsqp(best_h, n, 800, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Fine-grained local search
    t_now = time.time()
    if t_now < budget_s * 0.82:
        alloc2 = min(budget_s * 0.12, budget_s * 0.95 - (t_now - t0))
        if alloc2 > 15:
            h3, c5_3 = fine_search(best_h, n, alloc2, seed=seed + 7)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()

    # Phase 5: Final SLSQP polish
    t_now = time.time()
    if t_now < budget_s * 0.90:
        h_new = do_slsqp(best_h, n, 1000, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final verification and constraint enforcement
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(best_h.sum() - n / 2.0) > 1e-9:
        best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
