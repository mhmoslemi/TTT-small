```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        z_test = np.clip(x - mid, 0.0, 1.0)
        if z_test.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def objective(h, n_pts, beta):
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    logsum = np.log(np.sum(np.exp(beta * (corr - m) - 500))) + m * beta + 500
    return (np.exp(logsum) / beta) * dx

def constraint_sum(x, n_pts):
    return x.sum() - n_pts / 2.0

def make_bounds(n_pts):
    return [(0.0, 1.0)] * n_pts

def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def gen_structured_inits(n, seed):
    rng = np.random.RandomState(seed)
    inits = []
    
    # Thue-Morse
    tm = thue_morse(n)
    inits.append(tm.copy())
    
    # Anti-Thue-Morse
    inits.append(1.0 - tm)
    
    # Periodic patterns
    for period in [4, 6, 8, 12, 16]:
        if n % period == 0:
            pat = np.zeros(period)
            half = period // 2
            pat[:half] = 1.0
            # Vary: not always first half
            if period >= 6:
                pat[0] = 1; pat[1] = 0; pat[2] = 1; pat[3:] = 0
                if period >= 8:
                    pat[4] = 1; pat[5] = 0; pat[6] = 1; pat[7:] = 0
            for rep in range(n // period):
                inits.append(np.tile(pat, n // period))
    
    # Cosine patterns
    xg = np.linspace(0, 2, n, endpoint=False)
    for freq in [2, 3, 4, 5]:
        for amp in [0.3, 0.4, 0.5]:
            inits.append(0.5 + amp * np.cos(2 * np.pi * freq * xg / 2))
    
    # Random
    inits.append(rng.rand(n))
    inits.append(rng.rand(n))
    
    return inits

    def elapsed():
        return time.time() - t0
    
    # Load initial

    best_h = h0.copy()
    best_n = n_init

        # Build initial guesses

        bounds = make_bounds(n_pts)

                            best_h = h_cd.copy()
                            best_n = n_pts
                            improved = True
                            break
                if not improved:
                    break
    
    # Final SLSQP polish
    if elapsed() < budget_s * 0.3:
        n_pts = best_n
        bounds = make_bounds(n_pts)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        z_test = np.clip(x - mid, 0.0, 1.0)
        if z_test.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def objective_fn(h, n_pts, beta):
    dx = 2.0 / n_pts
    one_minus = 1.0 - h
    corr = np.correlate(h, one_minus, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    logsum = np.log(np.sum(np.exp(shifted)))
    return (m + logsum / beta) * dx


def constraint_sum(x, n_pts):
    return x.sum() - n_pts / 2.0


def make_bounds(n_pts):
    return [(0.0, 1.0)] * n_pts


def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(x_new, x_old, h, left=0.0, right=0.0)


def thue_morse_arr(n):
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm


def gen_starts(n, seed, best_h, best_n):
    rng = np.random.RandomState(seed)
    starts = []
    h_int = project_feasible(interp_to_size(best_h, best_n, n), n)
    starts.append(h_int)
    # Thue-Morse and complement
    tm = thue_morse_arr(n)
    starts.append(project_feasible(tm, n))
    starts.append(project_feasible(1.0 - tm, n))
    # Periodic patterns
    for p in [4, 6, 8, 10, 12]:
        if n % p == 0:
            base = np.zeros(p)
            n_ones = p // 2
            base[:n_ones] = 1.0
            pat = np.tile(base, n // p)
            starts.append(project_feasible(pat, n))
            # Shifted pattern
            if p >= 6:
                base2 = np.zeros(p)
                idx = list(range(0, p, 2))[:n_ones]
                base2[idx] = 1.0
                pat2 = np.tile(base2, n // p)
                starts.append(project_feasible(pat2, n))
    # Cosine
    xg = np.linspace(0, 2, n, endpoint=False)
    for freq in [2, 3, 4]:
        for amp in [0.35, 0.5]:
            starts.append(project_feasible(0.5 + amp * np.cos(np.pi * freq * xg), n))
    # Perturbed best
    for _ in range(3):
        starts.append(project_feasible(h_int + rng.randn(n) * 0.1, n))
    # Random binary-ish
    for _ in range(2):
        rb = (rng.rand(n) > 0.5).astype(float)
        starts.append(project_feasible(rb, n))
    return starts


def slsqp_run(h_start, n_pts, beta, maxiter, ftol):
    bounds = make_bounds(n_pts)
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n_pts,)}]
    res = minimize(objective_fn, h_start, args=(n_pts, beta),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if hasattr(res, 'x') and res.x is not None:
        return project_feasible(res.x, n_pts)
    return h_start.copy()


def coord_descent(h, n_pts, best_c5, budget_fn, time_limit_frac):
    h_curr = h.copy()
    curr_c5 = compute_c5(h_curr)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for step in steps:
        if budget_fn() > time_limit_frac:
            break
        for epoch in range(4):
            if budget_fn() > time_limit_frac:
                break
            improved = False
            for i in range(n_pts):
                for d in [step, -step, step * 0.5, -step * 0.5, step * 2.0, -step * 2.0]:
                    h_t = h_curr.copy()
                    h_t[i] = np.clip(h_t[i] + d, 0.0, 1.0)
                    h_t = project_feasible(h_t, n_pts)
                    c5_t = compute_c5(h_t)
                    if c5_t < curr_c5 - 1e-13:
                        h_curr = h_t
                        curr_c5 = c5_t
                        improved = True
                        break
            if not improved:
                break
    return h_curr, curr_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_init = len(h0)
    except NameError:
        h0 = np.full(180, 0.5)
        n_init = 180

    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n_init

    # Sort resolutions by priority
    resolutions = sorted(set([n_init, 90, 180, 270, 360]))

    # Phase 1: Quick multi-resolution exploration
    phase1_end = budget_s * 0.18
    for n_pts in resolutions:
        if elapsed() > phase1_end:
            break
        starts = gen_starts(n_pts, seed + n_pts * 7, best_h, best_n)
        for h_start in starts[:4]:
            if elapsed() > phase1_end:
                break
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n_pts
            # Quick SLSQP
            h_opt = slsqp_run(h_start, n_pts, 20.0, 300, 1e-13)
            c5_o = compute_c5(h_opt)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_opt.copy()
                best_n = n_pts

    # Phase 2: Main optimization at best resolution + neighbors
    phase2_end = budget_s * 0.72
    main_reses = []
    for nr in resolutions:
        if abs(nr - best_n) / best_n < 0.5:
            main_reses.append(nr)
    main_reses = sorted(set(main_reses))
    if best_n not in main_reses:
        main_reses.append(best_n)

    beta_schedule = [15.0, 30.0, 60.0, 120.0, 250.0, 500.0, 1000.0]

    for n_pts in main_reses:
        if elapsed() > phase2_end:
            break
        h_seed_pts = project_feasible(interp_to_size(best_h, best_n, n_pts), n_pts)
        inits = [h_seed_pts]
        # A few diverse starts
        for _ in range(2):
            inits.append(project_feasible(h_seed_pts + rng.randn(n_pts) * 0.06, n_pts))
        # TM-based
        tm = thue_morse_arr(n_pts)
        inits.append(project_feasible(0.3 * tm + 0.7 * h_seed_pts, n_pts))

        for h_start in inits:
            if elapsed() > phase2_end:
                break
            current = h_start.copy()
            for j, beta in enumerate(beta_schedule):
                if elapsed() > phase2_end:
                    break
                maxiter = max(200, int(600 * 180.0 / n_pts))
                h_opt = slsqp_run(current, n_pts, beta, maxiter, 1e-14)
                c5_o = compute_c5(h_opt)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_opt.copy()
                    best_n = n_pts
                    current = h_opt.copy()
                else:
                    # Nudge slightly
                    current = project_feasible(h_opt + rng.randn(n_pts) * 0.01, n_pts)

    # Phase 3: Coordinate descent refinement
    phase3_end = budget_s * 0.90
    if elapsed() < phase3_end:
        n_pts = best_n
        h_cd, c5_cd = coord_descent(best_h, n_pts, best_c5, elapsed, phase3_end / budget_s)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

        # Try one more resolution
        for n_alt in [best_n + 90, best_n + 180]:
            if n_alt <= 500 and elapsed() < phase3_end:
                h_alt = project_feasible(interp_to_size(best_h, best_n, n_alt), n_alt)
                h_cd2, c5_cd2 = coord_descent(h_alt, n_alt, best_c5, elapsed, phase3_end / budget_s)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2.copy()
                    best_n = n_alt

    # Phase 4: Final SLSQP polish
    if elapsed() < budget_s * 0.97:
        n_pts = best_n
        for beta in [500.0, 1500.0]:
            if elapsed() > budget_s * 0.97:
                break
            h_opt = slsqp_run(best_h, n_pts, beta, 800, 1e-15)
            c5_o = compute_c5(h_opt)
            if c5_o < best_c5:
                best_c5 = c5_o
                best_h = h_opt.copy()

        # Tiny coordinate descent
        if elapsed() < budget_s * 0.99:
            h_cd, c5_cd = coord_descent(best_h, best_n, best_c5, elapsed, 0.99)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact sum
    target_sum = best_n / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        # Distribute correction
        mask = (best_h > 0.01) & (best_h < 0.99)
        if mask.sum() > 0:
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
```
