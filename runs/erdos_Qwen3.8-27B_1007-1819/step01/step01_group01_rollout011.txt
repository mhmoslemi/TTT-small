import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def obj_smooth(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, alpha) * dx

def obj_smooth_grad(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Smooth max value
    exp_a = np.exp(alpha * (corr - corr.max()))
    w = exp_a / exp_a.sum()
    val = (corr.max() + np.log(exp_a.sum()) / alpha) * dx
    # Gradient: d/dh_j of corr[k] for each k
    # corr[k] = sum_i h[i+k](1-h[i]) (roughly)
    # This is complex; let's use finite differences or skip gradient
    # For L-BFGS-B, we can use the objective only
    return val

def optimize_lbfgsb(h0, n, alpha, maxiter=2000):
    try:
        res = minimize(
            obj_smooth, h0.copy(), args=(n, alpha),
            method='L-BFGS-B',

            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10}

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(mask):
            mask = h < 1.0 if diff > 0 else h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    return (m + max_shifted / alpha + np.log(np.sum(np.exp(shifted - max_shifted))) / alpha)

def obj_smooth(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, alpha) * dx

def optimize_lbfgsb(h0, n, alpha, maxiter=3000):
    try:
        res = minimize(
            obj_smooth, h0.copy(), args=(n, alpha),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def binary_sa(h0, n, budget_s, T0=0.004):
    t_start = time.time()
    h = (h0 > 0.5).astype(float)
    target = n // 2
    ones = np.sum(h)
    if ones > target:
        idx = np.where(h == 1)[0]
        for i in np.random.choice(idx, int(ones - target), replace=False):
            h[i] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        for i in np.random.choice(idx, int(target - ones), replace=False):
            h[i] = 1
    curr = compute_c5(h)
    best = h.copy()
    best_val = curr
    while time.time() - t_start < budget_s:
        T = T0 * (1.0 - (time.time() - t_start) / budget_s) ** 2
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        h[i], h[j] = 0, 1
        c_new = compute_c5(h)
        if c_new < curr or np.random.rand() < np.exp(-(c_new - curr) / max(T, 1e-15)):
            curr = c_new
            if curr < best_val:
                best_val = curr
                best = h.copy()
        else:
            h[i], h[j] = 1, 0
    return best, best_val

                if curr < best_val:

            else:

            s2 = h[j:j+blen].copy()
            h[i:i+blen] = s2

def periodic_init(p, n):
    m = n // p
    a = np.full(p, (n / 2.0) / n)  # uniform
    h = np.tile(a, m)
    if n > m * p:
        h = np.append(h, a[:n - m * p])
    return project_feasible(h, n)

def optimize_periodic(p, n, alpha=200, maxiter=2000):
    """Optimize period-p solution."""
    m = n // p
    target_sum = m / 2.0  # sum of one period
    
    def pobj(a):
        a = np.clip(a, 0, 1)
        diff = target_sum - a.sum()
        mask = (a > 1e-10) & (a < 1.0 - 1e-10)
        if np.any(mask):
            a[mask] += diff / np.sum(mask)
        a = np.clip(a, 0, 1)
        h = np.tile(a, m)
        if n > m * p:
            h = np.append(h, a[:n - m * p])
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr, alpha) * dx
    
    x0 = np.full(p, target_sum / p)
    try:
        res = minimize(pobj, x0, method='L-BFGS-B',
                       bounds=[(0, 1)] * p,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        a = np.clip(res.x, 0, 1)
        h = np.tile(a, m)
        if n > m * p:
            h = np.append(h, a[:n - m * p])
        return project_feasible(h, n)
    except Exception:
        return periodic_init(p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    
    try:
        init = np.array(initial_h_values, dtype=float)
        n_init = len(init)
    except NameError:
        n_init = 80
        init = np.full(n_init, 0.5)
    
    init = project_feasible(init, n_init)
    best_h = init.copy()
    best_c5 = compute_c5(init)
    best_n = n_init
    
    def tleft():
        return budget_s - 25 - (time.time() - t0)
    
    # Phase 1: Periodic initialization for various n
    n_list = sorted(set([n_init] + list(range(30, 151, 10))))
    for n in n_list:
        if tleft() < 30:
            break
        for p in [2, 3, 4, 5, 6, 7, 8, 10, 12, 16, 20]:
            if n % p != 0:
                continue
            h_p = optimize_periodic(p, n, alpha=300)
            c = compute_c5(h_p)
            if c < best_c5:
                best_c5 = c
                best_h = h_p.copy()
                best_n = n
    
    # Phase 2: L-BFGS-B from best
    if tleft() > 40:
        for alpha in [100, 500, 2000]:
            if tleft() < 20:
                break
            h_t = optimize_lbfgsb(best_h, best_n, alpha)
            c = compute_c5(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()
    
    # Phase 3: Binary SA
    if tleft() > 40:
        bd = min(tleft() * 0.25, 100)
        h_b, c_b = binary_sa(best_h, best_n, bd)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = h_b.copy()
    
    # Phase 4: L-BFGS-B from binary
    if tleft() > 30:
        h_t = optimize_lbfgsb(best_h, best_n, 1000)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
    
    # Phase 5: Block SA
    if tleft() > 40:
        bd = min(tleft() * 0.25, 100)
        h_b, c_b = block_sa(best_h, best_n, bd)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = h_b.copy()
    
    # Phase 6: Try other n with interpolation + optimize
    for n in n_list:
        if tleft() < 40:
            break
        if abs(n - best_n) < 5:
            continue
        h_s = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, best_n), best_h)
        h_s = project_feasible(h_s, n)
        h_opt = optimize_lbfgsb(h_s, n, 500, maxiter=1500)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()
            best_n = n
            if tleft() > 30:
                h_b, c_b = binary_sa(best_h, best_n, min(tleft() * 0.15, 40))
                if c_b < best_c5:
                    best_c5 = c_b
                    best_h = h_b.copy()
                    best_n = n
    
    # Phase 7: Iterate round-optimize
    for _ in range(3):
        if tleft() < 30:
            break
        h_bin = (best_h > 0.5).astype(float)
        # Fix count
        target = best_n // 2
        ones = np.sum(h_bin)
        if ones > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[np.random.choice(idx, int(ones-target), replace=False)] = 0
        elif ones < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[np.random.choice(idx, int(target-ones), replace=False)] = 1
        h_t = optimize_lbfgsb(h_bin, best_n, 2000)
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
    
    # Phase 8: Final L-BFGS-B + coordinate polish
    if tleft() > 15:
        h_t = optimize_lbfgsb(best_h, best_n, 5000, maxiter=5
