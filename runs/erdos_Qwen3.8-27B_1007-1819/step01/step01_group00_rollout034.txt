from scipy.optimize import minimize
import time

_BETA = 200.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(6 * n + 10):
        r = target - np.sum(h)
        if abs(r) < 1e-11:
            break
        h += r / n
        np.clip(h, 0.0, 1.0, out=h)
    r = target - np.sum(h)
    if abs(r) > 1e-9:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + r, 0.0, 1.0)
    return h

def smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    shifted = corr - mx
    m = np.max(shifted)
    val = mx + m + np.log(np.sum(np.exp(beta * (shifted - m)))) / beta
    return val * dx

def obj_smooth(x):
    return smooth_max(fix_feasible(x), _BETA)

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    return t

    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return fix_feasible(h_new)

def coord_descend(h, max_rounds=20):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    steps = [0.03, -0.03, 0.06, -0.06, 0.1, -0.1, 0.015, -0.015]
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for d in steps:
                v = h[i] + d
                if v < 0 or v > 1:
                    continue
                h_trial = h.copy()
                h_trial[i] = v
                h_trial = fix_feasible(h_trial)
                obj = compute_c5(h_trial)
                if obj < cur - 1e-13:
                    cur = obj
                    h = h_trial
                    improved = True
        if not improved:
            break
    return h, cur

    h = h.copy()

    steps = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.03, -0.03]
