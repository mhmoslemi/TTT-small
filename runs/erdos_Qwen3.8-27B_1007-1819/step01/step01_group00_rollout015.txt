import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(vals, beta):
    mx = np.max(vals)
    shifted = vals - mx
    return mx + np.log(np.sum(np.exp(beta * shifted))) / beta

def obj_full(h, beta, target_sum):
    h_c = np.clip(h, 0.0, 1.0)
    corr = compute_corr(h_c)
    dx = 2.0 / len(h_c)
    return smooth_max(corr, beta) * dx + 100.0 * (np.sum(h_c) - target_sum) ** 2

def obj_critical(h, cidx, beta, target_sum):
    h_c = np.clip(h, 0.0, 1.0)
    corr = compute_corr(h_c)
    dx = 2.0 / len(h_c)
    vals = corr[cidx]
    return smooth_max(vals, beta) * dx + 100.0 * (np.sum(h_c) - target_sum) ** 2

def get_critical(h, K=16):
    corr = compute_corr(h)
    idx = np.argsort(corr)[-K:]
    return idx, float(np.max(corr))

def lbfgsb_opt(h0, beta, cidx, time_limit, maxiter=4000):
    n = len(h0)
    ts = n / 2.0
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n

    def _obj(h):
        if cidx is not None:
            return obj_critical(h, cidx, beta, ts)
        return obj_full(h, beta, ts)

    try:
        res = minimize(_obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxfun': maxiter * max(n, 10)})
        return fix_feasible(res.x)
    except Exception:
        return fix_feasible(h0)

def discrete_hill climb(h, n_iter=500, time_limit=10):
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    cur = compute_c5(h)
    t_start = time.time()
    deltas = [0.1, -0.1, 0.05, -0.05, 0.2, -0.2, 0.4, -0.4]
    for _ in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        i = np.random.randint(n)
        best_d = None
        best_v = cur
        for d in deltas:
            v = h[i] + d
            if v < 0 or v > 1:
                continue
            h[i] = v
            h = fix_feasible(h)
            val = compute_c5(h)
            if val < best_v - 1e-12:
                best_v = val
                best_d = v
        if best_d is not None:
            h[i] = best_d
            h = fix_feasible(h)
            cur = best_v
        else:
            h[i] = h[i]  # no change
    return h, cur

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return fix_feasible(h_new)
