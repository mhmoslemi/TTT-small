import numpy as np
from scipy.optimize import minimize, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def lse_obj(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -400, 400)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def slsqp_opt(h0, n, alpha, maxiter=400):
    try:
        res = minimize(lse_obj, h0.copy(), args=(n, alpha),
                      method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def nelder_mead_polish(h0, n, maxiter=200):
    try:
        res = minimize(lambda p: compute_c5(project_feasible(p, n)), h0.copy(),
                      method='Nelder-Mead', options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

        if c_new <= curr or np.random.rand() < np.exp(-(c_new - curr) / max(T, 1e-15)):
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best = h.copy()
    return best, best_val

def targeted_coord_descent(h, n, max_rounds=40):
    h = h.copy()
    c = compute_c5(h)
    for _ in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_lags = np.argsort(corr)[-4:]
        active_idx = set()
        for lag in top_lags:
            offset = lag - (n - 1)
            lo = max(0, offset)
            hi = min(n, n - offset)
            for i in range(lo, hi):
                active_idx.add(i)
                active_idx.add(i - offset)
        active_idx = sorted(active_idx)
        if not active_idx:
