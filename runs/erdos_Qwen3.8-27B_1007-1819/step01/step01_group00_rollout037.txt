    return float(np.max(corr) * dx)

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0

        if abs(diff) < 1e-12:

        h += diff / n
        h = np.clip(h, 0.0, 1.0)

    r = target - np.sum(h)
    if abs(r) > 1e-12:

            idx = np.argsort(1.0 - h)
            for i in idx:
                add = min(r, 1.0 - h[i])

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    r = target - np.sum(h)
    if abs(r) > 1e-13:
        if r > 0:
            idx = np.argsort(1.0 - h)
            for i in idx:
                add = min(r, 1.0 - h[i])
                h[i] += add
                r -= add
                if abs(r) < 1e-13:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                sub = min(-r, h[i])
                h[i] -= sub
                r += sub
                if abs(r) < 1e-13:
                    break
    return h

def smooth_max_corr(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    c = np.clip(corr - mx, -80.0, 0.0)
    s = np.sum(np.exp(beta * c))
    return (mx + np.log(s) / beta) * dx

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def tm_seq(n):
    t = np.zeros(n, dtype=float)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t

        for i in range(n // 2 - n_ones):

        best_c5 = cur_c5

        for i in ones:

                h_bin[i] = 0

            h_bin[best_j] = 1
            cur_c5 = best_c5

        if time.time() < deadline - 15:
            h_cd, c5_cd = coord_refine(h_frac, n_rounds=4, deadline=deadline - 5)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Final high-beta polish on best solution
    if time.time() < deadline - 40:
        for beta in [200, 500, 1000]:
            if time.time() > deadline - 10:
                break
            def obj_final(x):
                return smooth_max_corr(fix_feasible(x), beta)
            try:
                res = minimize(obj_final, best_h, method='SLSQP',
                               constraints=[{'type': 'eq', 'fun': sum_eq}],
                               bounds=[(0, 1)] * n,
                               options={'maxiter': 400, 'ftol': 1e-12})
                h_f = fix_feasible(res.x)
                c5_f = compute_c5(h_f)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f
            except Exception:
                pass

    # Final coordinate refinement
    if time.time() < deadline - 5 and n <= 100:
        h_final, c5_final = coord_refine(best_h, n_rounds=6, deadline=deadline - 1)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Verify feasibility
    best_h = fix_feasible(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
