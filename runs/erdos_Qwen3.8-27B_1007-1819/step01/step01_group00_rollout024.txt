```python
def fix_feasible(h):
```

```python
def subgradient_step(h, top_k=10, alpha=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode="full")
    # Get top-k shifts
    idx = np.argsort(corr)[::-1][:top_k]
    # Average gradient over top-k
    g = np.zeros(n)
    for k in idx:
        wk = 1.0 / top_k
        if k < n:
            g[:n-k] += wk * (1.0 - h[k:])
            g[k:] += -wk * h[:n-k]
        else:
            s = k - n + 1
            g[s:] += wk * (1.0 - h[:n-s])
            g[:s] += -wk * h[s:]
    g *= dx
    # Projected gradient step
    h_new = h - alpha * g
    h_new = fix_feasible(h_new)
    return h_new
```

```python
def slsqp_smooth(h, beta, maxiter=300):
    n = len(h)
    def obj(x):
        return smooth_obj(np.clip(x, 0, 1), beta)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', constraints=cons,
                       bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-11})
        return fix_feasible(res.x)
    except Exception:
        return h

def subgrad_descent(h, top_k, n_iter, alpha0, time_fn):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for it in range(n_iter):
        if time_fn() > 0.85:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for k in idx:
            wk = 1.0 / top_k
            if k < n:
                g[:n-k] += wk * (1.0 - h[k:])
                g[k:] += -wk * h[:n-k]
            else:
                s = k - n + 1
                g[s:] += wk * (1.0 - h[:n-s])
                g[:s] += -wk * h[s:]
        g *= dx
        alpha = alpha0 / (1.0 + it * 0.002)
        h_new = fix_feasible(h - alpha * g)
        h = h_new
    return h

def coord_minimax(h, n_rounds=40):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for i in range(n):
            old_val = h[i]
            h[i] = 0.0
            h = fix_feasible(h)
            base = np.correlate(h, 1.0 - h, mode="full")
            h[i] = 1.0
            h = fix_feasible(h)
            one = np.correlate(h, 1.0 - h, mode="full")
            A = (one - base)
            B = base
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(A * m1 + B) < np.max(A * m2 + B):
                    hi = m2
                else:
                    lo = m1
            x_opt = (lo + hi) / 2.0
            h[i] = x_opt
            h = fix_feasible(h)
            c_new = compute_c5(h)
            if c_new < cur - 1e-13:
                cur = c_new
                improved = True
            else:
                h[i] = old_val
                h = fix_feasible(h)
        if not improved:
            break
    return h

def fine_coord(h, n_rounds=20):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    steps = [0.001, -0.001, 0.002, -0.002, 0.005, -0.005, 0.01, -0.01]
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            best_v = h[i]
            best_c = cur
            for d in steps:
                v = h[i] + d
                if v < 0 or v > 1:
                    continue
                h[i] = v
                h = fix_feasible(h)
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v
            h[i] = best_v
            h = fix_feasible(h)
            if best_c < cur - 1e-13:
                cur = best_c
                improved = True
        if not improved:
            break
    return h

def gen_start(n, r, best_h):
    if r == 0:
        t = np.zeros(n)
        for i in range(n):
            c, x = 0, i
            while x:
                c += x & 1
                x >>= 1
            t[i] = c % 2
        return t
    if r == 1:
        x = np.arange(n)
        return 0.5 + 0.3 * np.sin(6.0 * np.pi * x / n)
    if r == 2:
        blk = max(2, n // 7)
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i // blk) % 2 == 0 else 0.0
        return h
    if r == 3 and best_h is not None and len(best_h) == n:
        return best_h + 0.15 * np.random.randn(n)
    x = np.arange(n)
    return 0.5 + 0.25 * np.random.randn(n) + 0.1 * np.sin(3.0*np.pi*x/n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def tf():
        return (time.time() - t0) / budget_s

    best_h = None
    best_c5 = 1e10

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is not None:
        h_init = fix_feasible(h_init)
        best_c5 = compute_c5(h_init)
        best_h = h_init.copy()

    n = len(h_init) if h_init is not None else 100
    h = best_h.copy() if best_h is not None else (0.5 + 0.3*np.sin(6*np.pi*np.arange(n)/n))
    h = fix_feasible(h)

    # Phase 1: Subgradient descent (fast, many iterations)
    for top_k in [5, 10, 20, 30]:
        if tf() > 0.25:
            break
        h = subgrad_descent(h, top_k, 2000, alpha0=0.5, time_fn=tf)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 2: SLSQP with smooth max
    for beta in [200, 500, 1000]:
        if tf() > 0.5:
            break
        h = slsqp_smooth(h, beta, maxiter=400)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 3: Coordinate minimax descent
    if tf() < 0.7:
        h = coord_minimax(best_h, n_rounds=50)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 4: Basin hopping with small perturbations
    if tf() < 0.88:
        for trial in range(6):
            if tf() > 0.88:
                break
            scale = 0.02 + 0.01 * trial
            h_p = fix_feasible(best_h + scale * np.random.randn(n))
            h_p = subgrad_descent(h_p, 15, 500, alpha0=0.3, time_fn=tf)
            h_p = slsqp_smooth(h_p, 500, maxiter=200)
            c = compute_c5(h_p)
            if c < best_c5:
                best_c5 = c
                best_h = h_p.copy()

    # Phase 5: Final coordinate refinement
    if tf() < 0.95:
        h = coord_minimax(best_h, n_rounds=30)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        h = fine_coord(best_h, n_rounds=15)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 6: Final SLSQP polish
    if tf() < 0.985:
        h = slsqp_smooth(best_h, 2000, maxiter=300)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    if best_h is None:
        best_h = fix_feasible(np.full(100, 0.5))
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
