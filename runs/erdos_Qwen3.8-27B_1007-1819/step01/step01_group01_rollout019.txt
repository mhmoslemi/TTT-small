import time


def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)


def proj(h, n):
    h = np.clip(np.asarray(h, float), 0.0, 1.0)
    for _ in range(500):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        if not m.any():
            break
        h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smo(h, a):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -50.0, 50.0)
    return (m + np.log(np.exp(s).sum()) / a) * (2.0 / len(h))

                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': mi, 'ftol': 1e-15, 'gtol': 1e-10})
        return proj(r.x, n)
    except Exception:
        return h0.copy()
