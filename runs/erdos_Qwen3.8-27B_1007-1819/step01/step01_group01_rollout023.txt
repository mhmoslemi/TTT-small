```python
import numpy as np
import time
from scipy.optimize import minimize, linprog


def corr_values(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def c5_values(h):
    h = np.asarray(h, dtype=float)
    return corr_values(h) * (2.0 / len(h))


def compute_c5(h):
    return float(np.max(c5_values(h)))


def project_feasible(x):
    h = np.clip(np.asarray(x, dtype=float).copy(), 0.0, 1.0)
    n = len(h)
    target = 0.5 * n
    for _ in range(80):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_value(c, alpha):
    c = np.asarray(c, dtype=float)
    m = float(np.max(c))
    if alpha <= 1e-12:
        return m
    z = alpha * (c - m)
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    return m + float(np.log(e.sum())) / alpha


def softmax_weights(c, alpha):
    c = np.asarray(c, dtype=float)
    m = float(np.max(c))
    if alpha <= 1e-12:
        return np.full(c.shape, 1.0 / max(1, len(c)))
    z = alpha * (c - m)
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        return np.full(c.shape, 1.0 / max(1, len(c)))
    return e / s


def grad_corrs(h, w):
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    n = len(h)
    g = np.zeros(n, dtype=float)
    off = n - 1
    g += w[off] * (1.0 - 2.0 * h)
    for q in range(1, n):
        wp = w[off + q]
        if wp > 1e-18:
            g[q:] += wp * (1.0 - h[:-q])
            g[:-q] -= wp * h[q:]
        wm = w[off - q]
        if wm > 1e-18:
            g[:-q] += wm * (1.0 - h[q:])
            g[q:] -= wm * h[:-q]
    return g


def obj_smooth(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = corr_values(x)
    return smooth_max_value(c, alpha) * dx


def grad_smooth(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = corr_values(x)
    w = softmax_weights(c, alpha)
    return dx * grad_corrs(x, w)


def eq_sum(x):
    return float(np.sum(x) - 0.5 * len(x))


def eq_sum_jac(x):
    return np.ones(len(x), dtype=float)


def slsq_once(h0, alpha, maxiter=150):
    h = project_feasible(h0)
    n = len(h)
    if n < 4:
        return h
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": eq_sum, "jac": eq_sum_jac}]
    try:
        res = minimize(
            obj_smooth,
            h,
            args=(alpha,),
            method="SLSQP",
            jac=grad_smooth,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-10},
        )
    except Exception:
        return h
    if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
        return project_feasible(np.asarray(res.x, dtype=float))
    return h


def phase_slsqp(h0, deadline, alphas=(500.0, 200.0, 80.0, 120.0, 300.0)):
    h = project_feasible(h0)
    cur_c = compute_c5(h)
    for alpha in alphas:

        mt = 180 if len(h) <= 110 else 90

        c2 = compute_c5(h2)
        if c2 < cur_c:
            h = h2

    return h, cur_c


def grad_lag_scaled(h, q, sign):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n, dtype=float)
    if q == 0:
        return dx * (1.0 - 2.0 * h)
    if sign >= 0:
        g[q:] += 1.0 - h[:-q]
        g[:-q] -= h[q:]
    else:
        g[:-q] += 1.0 - h[q:]
        g[q:] -= h[:-q]
    return dx * g


def lp_step(h, active_idx):
    h = np.asarray(h, dtype=float)
    n = len(h)
    off = n - 1
    C = c5_values(h)
    active_idx = np.asarray(active_idx, dtype=int).ravel()
    if active_idx.size == 0:
        return None
    rows = []
    rhs = []
    for idx in active_idx:
        k = int(idx) - off
        d = grad_lag_scaled(h, abs(k), 1 if k >= 0 else -1)
        rows.append(np.concatenate((d, np.array([-1.0]))))
        rhs.append(float(np.dot(d, h)) - float(C[idx]))
    if not rows:
        return None
    A = np.vstack(rows)
    b = np.asarray(rhs, dtype=float)
    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[n] = 1.0
    A_eq = np.ones((1, n + 1), dtype=float)
    b_eq = np.array([0.5 * n], dtype=float)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = None
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method="simplex")
        except Exception:
            res = None
    if res is not None and int(getattr(res, "status", -1)) == 0:
        return res
    return None


def phase_lp(h0, deadline, max_active=None):
    h = project_feasible(h0)
    n = len(h)
    best = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    if max_active is None:
        max_active = min(2 * n - 1, 100)
    max_active = max(8, min(int(max_active), 2 * n - 1))

    stall = 0
    while time.time() < deadline:
        C = c5_values(h)
        m = float(np.max(C))
        if m < cur_c:
            cur_c = m
            best = h.copy()
            best_c = m
        elif m < best_c:
            best_c = m
            best = h.copy()

        tol = max(1e-6, 1e-5 * m)
        near = np.where(C >= m - tol)[0]
        if len(near) < max_active:
            top = np.argsort(C)[-max_active:][::-1]
            near = np.unique(np.concatenate((near, top)))
        active = near[np.argsort(C[near])[::-1]][:max_active]

        res = lp_step(h, active)
        if res is None and len(active) > 8:
            res = lp_step(h, active[:8])
        if res is None:
            stall += 1
            if stall > 8:
                break
            continue

        z = res.x[:n]
        if not np.all(np.isfinite(z)):
            break
        dnorm = float(np.linalg.norm(z - h))
        if dnorm < 1e-10:
            stall += 1
            if stall > 8:
                break
            continue

        best_cand = None
        best_val = cur_c
        for a in (1.0, 0.7, 0.5, 0.33, 0.22, 0.14, 0.08, 0.05, 0.03, 0.015):
            cand = project_feasible(h + a * (z - h))
            val = compute_c5(cand)
            if val < best_val - 1e-14:
                best_val = val
                best_cand = cand
        if best_cand is not None:
            h = best_cand
            cur_c = best_val
            best = h.copy()
            best_c = cur_c
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break
    return best, best_c


def interpolate_to_n(h, n):
    if len(h) == n:
        return project_feasible(h)
    xs = np.linspace(0.0, 1.0, len(h))
    xn = np.linspace(0.0, 1.0, n)
    return project_feasible(np.interp(xn, xs, h))


def jitter_start(h, scale):
    h = project_feasible(h)
    return project_feasible(h + scale * np.random.normal(size=len(h)))


def random_start(n, mode):
    if mode == 0:
        x = np.linspace(0.0, 1.0, n)
        h = 0.5 * np.ones(n)
        for m in range(1, 5):
            a = np.random.uniform(-0.25, 0.25) / m
            b = np.random.uniform(-0.25, 0.25) / m
            h += a * np.sin(np.pi * m * x) + b * np.cos(np.pi * m * x)
    else:
        h = np.empty(n, dtype=float)
        i = 0
        val = bool(np.random.rand() < 0.5)
        while i < n:
            L = int(np.random.randint(2, 9)) + 1
            h[i:min(n, i + L)] = 1.0 if val else 0.2
            val = not val
            i += L
    h += np.random.normal(0.0, 0.04, n)
    return project_feasible(np.clip(h, 0.0, 1.0))


def coordinate_polish(h0, deadline, max_rounds=6):
    h = project_feasible(h0)
    best = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    deltas = (0.015, -0.015, 0.03, -0.03, 0.06, -0.06)
    n = len(h)
    for _ in range(max_rounds):
        if time.time() >= deadline:
            break
        improved = False
        for i in range(n):
            if time.time() >= deadline:
                break
            base = h.copy()
            base_val = cur_c
            best_cand = base.copy()
            for d in deltas:
                if time.time() >= deadline:
                    break
                v = base[i] + d
                if v <= 0.0 or v >= 1.0:
                    continue
                cand = base.copy()
                cand[i] = v
                cand = project_feasible(cand)
                val = compute_c5(cand)
                if val < base_val - 1e-14:
                    base_val = val
                    best_cand = cand
            if base_val < cur_c - 1e-14:
                h = best_cand
                cur_c = base_val
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best = h.copy()
        if not improved:
            break
    return best, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    try:
        init = initial_h_values
        if init is None or len(init) == 0:
            raise ValueError
        h0 = np.asarray(init, dtype=float).copy()
    except Exception:
        n = 80
        x = np.linspace(0.0, 1.0, n)
        h0 = 0.5 + 0.30 * np.cos(np.pi * x) + 0.10 * np.cos(3.0 * np.pi * x) + 0.05 * np.sin(2.0 * np.pi * x)

    h0 = project_feasible(h0)
    best_h = h0.copy()
    best_n = len(h0)
    best_c = compute_c5(best_h)

    if not np.isfinite(budget_s):
        budget_s = 1100.0
    budget_s = max(60.0, float(budget_s))
    reserve = min(30.0, 0.03 * budget_s)
    use = max(30.0, budget_s - reserve)

    d1 = t0 + 0.40 * use
    d2 = t0 + 0.55 * use
    d3 = t0 + 0.78 * use
    d4 = t0 + 0.90 * use
    d5 = t0 + budget_s - 10.0

    if time.time() < d1:
        best_h, best_c = phase_lp(best_h, min(d1, d5), max_active=None)

    if time.time() < d2:
        best_h, best_c = phase_slsqp(best_h, min(d2, d5))

    if time.time() < d3:
        base_n = best_n
        cand_ns = set()
        for mm in (base_n - 10, base_n - 5, base_n + 5, base_n + 10):
            if mm < 20:
                continue
            if mm % 2:
                mm += 1
            if mm <= 320:
                cand_ns.add(mm)
        mm = 2 * base_n
        if mm <= 320:
            if mm % 2:
                mm += 1
            cand_ns.add(mm)
        for mm in sorted(cand_ns):
            if time.time() >= d3:
                break
            limit = min(d3, time.time() + 45.0)
            start = interpolate_to_n(best_h, mm)
            h2, c2 = phase_lp(start, limit, max_active=80)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = mm

    if time.time() < d4:
        n_try = best_n if best_n <= 160 else 120
        for r in range(6):
            if time.time() >= d4:
                break
            if r % 3 == 0:
                start = random_start(n_try, r % 2)
            else:
                if best_n == n_try:
                    start = jitter_start(best_h, 0.02 + 0.025 * r)
                else:
                    start = interpolate_to_n(best_h, n_try)
            limit = min(d4, time.time() + 25.0)
            h2, c2 = phase_lp(start, limit, max_active=70)
            if time.time() < d4:
                h2 = slsq_once(h2, 180.0, 80)
                c2 = compute_c5(h2)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = n_try

    if time.time() < d5:
        limit = min(d5, time.time() + 120.0)
        ma = 2 * best_n - 1 if best_n <= 120 else 120
        best_h, best_c = phase_lp(best_h, limit, max_active=ma)

    if time.time() < d5 - 1.0:
        h2 = slsq_once(best_h, 300.0, 120)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < d5 - 1.0:
        mr = 8 if best_n <= 120 else 3
        best_h, best_c = coordinate_polish(best_h, d5 - 1.0, mr)

    best_h = project_feasible(best_h)
    best_c = compute_c5(best_h)
    return best_h, best_c, best_n
```
