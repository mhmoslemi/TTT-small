def smooth_max_val(corr, alpha):
    m = np.max(corr)
    s = np.clip(corr - m, -2.0, 0.0)
    return m + float(np.log(np.sum(np.exp(alpha * s)))) / alpha

def make_smooth_obj(n, alpha):
    def obj(params):
        h = project_feasible(params, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max_val(corr * (2.0/n), alpha)
    return obj

def make_exact_obj(n):
    def obj(params):
        return compute_c5(project_feasible(params, n))
    return obj

    t0 = time.time()
    np.random.seed(seed)
    
    def tl():
        return budget_s - 15 - (time.time() - t0)
    
    # Get initial
    try:
        init = np.array(initial_h_values, dtype=float)
    except NameError:
        init = np.full(80, 0.5)
    n = len(init)
    init = project_feasible(init, n)
    
    best_h = init.copy()
    best_c = compute_c5(init)

    cons = [{'type': 'eq', 'fun': lambda p: np.sum(p) - n/2.0,
             'jac': lambda p: np.ones(n)}]
    
    # Phase 1: SLSQP with smooth max, increasing alpha
    for alpha in [100, 300, 800, 2000, 5000]:
        if tl() < 30:
            break

        iters = 400
        try:
            res = minimize(obj_fn, best_h.copy(), method='SLSQP',

            h_t = project_feasible(res.x, n)
            c = compute_c5(h_t)
            if c < best_c:
                best_c = c
                best_h = h_t.copy()
        except Exception:
            pass
