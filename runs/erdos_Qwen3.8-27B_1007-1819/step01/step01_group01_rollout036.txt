    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-12:

def smooth_max(corr, alpha):
    m = np.max(corr)
    # Numerically stable
    shifted = alpha * (corr - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return m + log_sum / alpha

    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -50, 50)
    exp_val = np.exp(shifted)
    weights = exp_val / np.sum(exp_val)
    return weights

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

    dx = 2.0 / n

        return compute_c5(h)
    try:
        res = minimize(fun, h0.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h0.copy()

    h = h0.copy()
    curr = compute_c5(h)
    best = h.copy()
    best_val = curr
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-k:]
        # Subgradient of sum of top-k correlations
        grad = np.zeros(n)
        for m in top_idx:
            for p in range(n):
                i1 = p - m
                if 0 <= i1 <= n - 1:
                    grad[p] += (1.0 - h[i1])
                i2 = p + m
                if 0 <= i2 <= n - 1:
                    grad[p] -= h[i2]
        norm = np.sqrt(np.sum(grad**2))
        if norm < 1e-15:
            break
        lr = lr_init / (1 + it * 0.1)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)

            c_new = compute_c5(h_new)

            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best = h.copy()
    return best, best_val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0 - 1e-13) if diff > 0 else (h > 1e-13)
        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_grad_smooth(params, n, alpha):
    h = project_feasible(params, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_max = np.max(corr)
    shifted = np.clip(alpha * (corr - m_max), -50, 50)
    exp_v = np.exp(shifted)
    s = exp_v.sum()
    obj = (m_max + np.log(s) / alpha) * dx
    weights = exp_v / s
    n2 = 2 * n - 1
    grad = np.zeros(n)
    for p in range(n):
        m_lo = max(0, p - n + 1)
        m_hi = min(p, n2 - 1)
        if m_lo <= m_hi:
            idx = p - np.arange(m_hi - m_lo + 1)
            grad[p] += weights[m_lo:m_hi + 1] @ (1.0 - h[idx])
        m_hi2 = min(n - 1 - p, n2 - 1)
        if m_hi2 >= 0:
            grad[p] -= weights[0:m_hi2 + 1] @ h[p:p + m_hi2 + 1]
    grad *= dx
    return obj, grad

def nelder_polish(h0, n, maxiter=200):
    def fun(x):
        return compute_c5(project_feasible(x, n))
    try:
        res = minimize(fun, h0.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def topk_sg(h0, n, k=5, n_it=300, lr0=0.008):
    h = h0.copy()
    curr = compute_c5(h)
    best = h.copy()
    bv = curr
    for it in range(n_it):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(corr)[-k:]
        grad = np.zeros(n)
        for m in top:
            for p in range(n):
                i1 = p - m
                if 0 <= i1 <= n - 1:
                    grad[p] += (1.0 - h[i1])
                i2 = p + m
                if 0 <= i2 <= n - 1:
                    grad[p] -= h[i2]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        lr = lr0 / (1.0 + 0.05 * it)
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new)
        if c_new > curr * 1.02:
            h_new = project_feasible(h - lr * 0.1 * grad, n)
            c_new = compute_c5(h_new)
        if c_new < curr:
            h = h_new
            curr = c_new
            if curr < bv:
                bv = curr
                best = h.copy()
    return best, bv

def interpolate_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    h_new = np.interp(np.linspace(0, 1, n_dst), np.linspace(0, 1, n_src), h_src)
    return project_feasible(h_new, n_dst)

            best_c5 = c

    for n in n_list:
        if tl() < 30:
            break
        h_interp = interpolate_h(best_h, best_n, n)
        h_rand = project_feasible(np.clip(0.5 + 0.35 * np.random.randn(n), 0, 1), n)
        h_block = np.zeros(n)
        bs = n // 4
        h_block[0:bs] = 1.0
        h_block[2 * bs:3 * bs] = 1.0
        h_block = project_feasible(h_block, n)
        for h0 in [h_interp, h_rand, h_block]:
            if tl() < 15:
                break
            h_o = slsqp_optimize(h0, n, 100, maxiter=200)
            h_o = nelder_polish(h_o, n, maxiter=100)
            c = compute_c5(h_o)
            if c < best_c5:
                best_c5 = c
                best_h = h_o.copy()
                best_n = n

    # Phase 4: Intensive SLSQP on best with various alphas
    for alpha in [50, 100, 300, 800]:
        if tl() < 15:
            break
        h_o = slsqp_optimize(best_h, best_n, alpha, maxiter=300)
        c = compute_c5(h_o)
        if c < best_c5:
            best_c5 = c
            best_h = h_o.copy()

    # Phase 5: Nelder-Mead + top-k subgradient
    if tl() > 20:
        h_nm = nelder_polish(best_h, best_n, maxiter=300)
        c = compute_c5(h_nm)
        if c < best_c5:
            best_c5 = c
            best_h = h_nm.copy()
    if tl() > 20:
        h_sg, c_sg = topk_sg(best_h, best_n, k=5, n_it=300, lr0=0.006)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()
    if tl() > 15:
        h_sg, c_sg = topk_sg(best_h, best_n, k=3, n_it=200, lr0=0.003)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()

    # Phase 6: Final SLSQP + Nelder + coordinate
    if tl() > 15:
        h_o = slsqp_optimize(best_h, best_n, 500, maxiter=200)
        c = compute_c5(h_o)
        if c < best_c5:
            best_c5 = c
            best_h = h_o.copy()
    if tl() > 10:
        h_nm = nelder_polish(best_h, best_n, maxiter=200)
        c = compute_c5(h_nm)
        if c < best_c5:
            best_c5 = c
            best_h = h_nm.copy()

    # Phase 7: Coordinate descent (targeted)
    if tl() > 10:
        h_cd = best_h.copy()
        c_cd = compute_c5(h_cd)
        for _ in range(80):
            improved = False
            for i in range(best_n):
                old = h_cd[i]
                for frac in [0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0]:
                    h_tri = h_cd.copy()
                    h_tri[i] = frac
                    h_tri = project_feasible(h_tri, best_n)
                    v = compute_c5(h_tri)
                    if v < c_cd:
                        c_cd = v
                        h_cd = h_tri
                        improved = True
                h_cd[i] = c_cd and h_cd[i]  # no-op, keep h_cd
            if not improved:
                break
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Phase 8: One final SLSQP round
    if tl() > 10:
        h_o = slsqp_optimize(best_h, best_n, 300, maxiter=150)
        h_o = nelder_polish(h_o, best_n, maxiter=100)
        c = compute_c5(h_o)
        if c < best_c5:
            best_c5 = c
            best_h = h_o.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
