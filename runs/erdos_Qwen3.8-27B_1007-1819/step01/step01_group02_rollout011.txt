import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        z = np.clip(x - mid, 0.0, 1.0)
        if z.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj(x, n_pts, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Shift for numerical stability
    shifted = corr - m
    logsumexp = m + np.log(np.sum(np.exp(beta * shifted)) + 1e-300) / beta
    return logsumexp * dx

def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.arange(n_from) * (2.0 / n_from)
    x_new = np.arange(n_to) * (2.0 / n_to)
    return np.interp(x_new, x_old, h, left=0, right=0)

def coordinate_descent(h, n, max_iters, tol, rng):
    """Coordinate descent with line search-like step selection"""
    dx = 2.0 / n
    best = compute_c5(h)
    h = h.copy()
    steps = [0.01, 0.005, 0.02, 0.002, 0.05, 0.001]
    for it in range(max_iters):
        improved = False
        idx = rng.permutation(n)
        for i in idx:
            for d in steps:
                for sign in [1, -1]:
                    h_test = h.copy()
                    delta = sign * d
                    h_test[i] = np.clip(h_test[i] + delta, 0.0, 1.0)
                    h_test = project_feasible(h_test, n)
                    c = compute_c5(h_test)
                    if c < best - tol:
                        h = h_test
                        best = c
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            # Try smaller steps
            steps = [s * 0.5 for s in steps]
            if steps[0] < 1e-5:
                break
    return h, best

def subgradient_descent(h, n, max_iters, lr_start, lr_end, rng):
    """Projected subgradient descent on the max of correlations"""
    dx = 2.0 / n
    h = project_feasible(h, n).copy()
    best = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        lr = lr_start * (lr_end / lr_start) ** (it / max_iters)
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Subgradient: find max lag
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)  # actual lag
        # Gradient of corr[k] w.r.t. h
        # corr[k] = sum_i h[i] * (1 - h[i+k]) for appropriate range
        # d/dh[j] corr[k] = (1-h[j+k]) for j where j+k valid (as first arg)
        #                  - h[j-k] for j where j-k valid (as second arg, 1-h[j])
        grad = np.zeros(n)
        if k >= 0:
            # h[i] * (1-h[i+k]), i from 0 to n-1-k
            # d/dh[j] = (1-h[j+k]) for j in [0, n-1-k]
            #           - h[j-k] for j in [k, n-1]
            for j in range(n - k):
                grad[j] += (1.0 - h[j + k])
            for j in range(k, n):
                grad[j] += (-h[j - k])

                grad[j] += (1.0 - h[j + k_abs])
            for j in range(k_abs, n):
                grad[j] += (-h[j - k_abs])
        
        grad *= dx
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new)
        if c < best:
            best = c
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best
