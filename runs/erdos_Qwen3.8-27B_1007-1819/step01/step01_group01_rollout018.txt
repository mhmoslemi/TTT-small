from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max(corr, alpha):
    m = np.max(corr)
    if alpha * (m - m) > 700:  # overflow guard (won't trigger but safety)
        return m
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def targeted_descent(h, n, n_iterations=1000, lr=0.01):
    """At each step, find worst lag and do gradient step."""
    h = h.copy()
    curr = compute_c5(h)
    dx = 2.0 / n
    for it in range(n_iterations):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        k = k_star - (n - 1)
        # Compute gradient of c_k_star w.r.t. h
        grad = np.zeros(n)
        if k >= 0:
            for i in range(n - k):
                grad[i + k] += (1 - h[i])
                grad[i] -= h[i + k]
        else:
            kp = -k
            for i in range(n - kp):
                grad[i] += (1 - h[i + kp])
                grad[i + kp] -= h[i]
        # Gradient step to reduce c_k
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < curr:
            h = h_new
            curr = c_new
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return h, curr

def lag_gradient(h, n, k):
    """Gradient of c_k (the k-th lag correlation) w.r.t. h."""
    grad = np.zeros(n)
    if k == 0:

    elif k > 0:

        grad[kp:] -= h[:n-kp]
    return grad

    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def lag_gradient(h, n, k):
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        kp = -k
        grad[:n - kp] += (1.0 - h[kp:])
        grad[kp:] -= h[:n - kp]
    return grad

def targeted_descent(h, n, max_iter=5000, lr0=0.03):
    h = h.copy()
    curr = compute_c5(h)
    lr = lr0
    best_h = h.copy()
    best_val = curr
    stall = 0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        g = lag_gradient(h, n, k)
        for lr_try in [lr, lr * 0.5, lr * 0.25]:
            h_new = project_feasible(h - lr_try * g, n)
            c_new = compute_c5(h_new)
            if c_new < curr - 1e-15:
                h = h_new
                curr = c_new
                lr = min(lr * 1.05, 0.08)
                stall = 0
                break
        else:
            lr *= 0.6
            stall += 1
            if stall > 80 or lr < 1e-10:
                break
        if curr < best_val:
            best_val = curr
            best_h = h.copy()
    return best_h, best_val

def multi_lag_descent(h, n, max_iter=3000, lr0=0.02, top_k=3):
    h = h.copy()
    curr = compute_c5(h)
    lr = lr0
    best_h = h.copy()
    best_val = curr
    stall = 0
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idxs = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        weights = np.zeros(len(top_idxs))
        for i, k_idx in enumerate(top_idxs):
            k = k_idx - (n - 1)
            g += lag_gradient(h, n, k)
            weights[i] = corr[k_idx] - corr[top_idxs[0]] + 1.0
        g *= np.sum(weights) / (len(top_idxs) * np.max(weights))
        g /= (np.max(np.abs(g)) + 1e-15)
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new)
        if c_new < curr - 1e-15:
            h = h_new
            curr = c_new
            lr = min(lr * 1.03, 0.06)
            stall = 0
        else:
            lr *= 0.7
            stall += 1
            if stall > 60 or lr < 1e-10:
                break
        if curr < best_val:
            best_val = curr
            best_h = h.copy()
    return best_h, best_val

def smooth_slsqp(h0, n, alpha, maxiter=300):
    def obj(x):
        h = project_feasible(x, n)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -600, 600)
        return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    try:
        res = minimize(obj, h0.copy(), method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def pairwise_swaps(h, n, n_rounds=300):
    h = h.copy()
    curr = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        n_pairs = min(400, n * n // 3)
        idxs = np.random.randint(0, n, size=(n_pairs, 2))
        for i, j in idxs:
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            v = compute_c5(h)
            if v < curr - 1e-13:
                curr = v
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, curr

def coordinate_refine(h, n, n_rounds=80):
    h = h.copy()
    curr = compute_c5(h)
    deltas = [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.25, -0.25]
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            best_v = h[i]
            best_c = curr
            for d in deltas:
                v_new = h[i] + d
                if v_new < 0 or v_new > 1:
                    continue
                h[i] = v_new
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = v_new
            h[i] = best_v
            h = project_feasible(h, n)
            if best_c < curr:
                improved = True
                curr = best_c
        if not improved:
            break
    return h, curr

def simulated_annealing(h0, n, t_deadline):
    h = h0.copy()
    curr = compute_c5(h)
    best_h = h.copy()
    best_val = curr
    total_steps = 4000
    for step in range(total_steps):
        if time.time() > t_deadline:
            break
        T = 0.004 * (1 - step / total_steps) + 0.0002
        mt = np.random.random()
        h_new = h.copy()
        if mt < 0.25:
            i = np.random.randint(n)
            h_new[i] = np.clip(h_new[i] + np.random.randn() * 0.12, 0, 1)
        elif mt < 0.50:
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        elif mt < 0.70:
            ln = np.random.choice([2, 3, 4])
            st = np.random.randint(0, max(1, n - ln))
            h_new[st:st+ln] = np.roll(h_new[st:st+ln], np.random.choice([-1, 1]))
        elif mt < 0.85:
            np_ = np.random.choice([2, 3, 4])
            idxs = np.random.choice(n, np_, replace=False)
            h_new[idxs] = np.clip(h_new[idxs] + np.random.randn(np_) * 0.08, 0, 1)
        else:
            k = np.random.randint(1, min(6, n // 2))
            st = np.random.randint(0, max(1, n - 2 * k))
            h_new[st+k:st+2*k], h_new[st:st+k] = h_new[st:st+k].copy(), h_new[st+k:st+2*k].copy()
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < curr or np.random.random() < np.exp(-(c_new - curr) / max(T, 1e-15)):
            h = h_new
            curr = c_new
            if curr < best_val:
                best_val = curr
                best_h = h.copy()
    return best_h, best_val

def block_exchange(h, n, n_trials=800):
    h = h.copy()
    curr = compute_c5(h)
    best_h = h.copy()
    best_val = curr
    for _ in range(n_trials):
        k1 = np.random.randint(1, max(2, n // 3))
        k2 = np.random.randint(1, max(2, n // 3))
        s1 = np.random.randint(0, n - k1)
        s2 = np.random.randint(0, n - k2)
        h_new = h.copy()
        h_new[s1:s1+k1], h_new[s2:s2+k2] = h_new[s2:s2+k2].copy(), h_new[s1:s1+k1].copy()
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < best_val:
            best_val = c
            best_h = h_new.copy()
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def tleft():
        return budget_s - 25 - (time.time() - t0)

    try:
        init = np.array(initial_h_values, dtype=float)
        n = len(init)
    except NameError:
        n = 50
        init = np.full(n, 0.5)
    init = project_feasible(init, n)

    best_h = init.copy()
    best_c5 = compute_c5(init)
    best_n = n

    # Phase 1: Targeted descent
    if tleft() > 120:
        h_t, c_t = targeted_descent(best_h, best_n, 4000, 0.03)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t

    # Phase 2: Multi-lag descent
    if tleft() > 100:
        h_t, c_t = multi_lag_descent(best_h, best_n, 3000, 0.015, top_k=3)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t

    # Phase 3: Smooth SLSQP
    if tleft() > 80:
        for alpha in [200, 600, 1500]:
            if tleft() < 15:
                break
            h_t = smooth_slsqp(best_h, best_n, alpha, 200)
            c_t = compute_c5(h_t)
            if c_t < best_c5:
                best_c5, best_h = c_t, h_t

    # Phase 4: Targeted + pairwise
    if tleft() > 60:
        h_t, c_t = targeted_descent(best_h, best_n, 5000, 0.01)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t
        h_t, c_t = pairwise_swaps(best_h, best_n, 400)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t

    # Phase 5: Block exchange + coordinate
    if tleft() > 40:
        h_t, c_t = block_exchange(best_h, best_n, 800)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t
        h_t, c_t = coordinate_refine(best_h, best_n, 60)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t

    # Phase 6: Simulated annealing
    if tleft() > 50:
        dl = min(time.time() + tleft() * 0.5, t0 + budget_s - 30)
        h_t, c_t = simulated_annealing(best_h, best_n, dl)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t

    # Phase 7: Try smaller n
    if tleft() > 30:
        for ns in [40, 50, 60]:
            if tleft() < 15 or ns >= best_n:
                continue
            hi = np.interp(np.linspace(0, 1, ns), np.linspace(0, 1, best_n), best_h)
            hi = project_feasible(hi, ns)
            h_t, c_t = targeted_descent(hi, ns, 3000, 0.02)
            h_t, c_t = coordinate_refine(h_t, ns, 40)
            if c_t < best_c5:
                best_c5, best_h, best_n = c_t, h_t, ns

    # Phase 8: Final polish
    if tleft() > 15:
        h_t, c_t = targeted_descent(best_h, best_n, 3000, 0.005)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t
        h_t, c_t = pairwise_swaps(best_h, best_n, 300)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t
        h_t, c_t = coordinate_refine(best_h, best_n, 40)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t

    # Phase 9: One more SLSQP with very high alpha
    if tleft() > 10:
        h_t = smooth_slsqp(best_h, best_n, 3000, 200)
        c_t = compute_c5(h_t)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t
        h_t, c_t = targeted_descent(best_h, best_n, 2000, 0.003)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
