def coordinate_refine(h, n, n_rounds=50, n_candidates=9):

            best_v = old
            best_c = curr

            curr = best_c

def compute_c5(h):

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break

        np.clip(h, 0.0, 1.0, out=h)

        if time_left() < 50:

        h_init = make_structured_init(best_n, pat)
        h_init = project_feasible(np.clip(h_init + 0.08 * np.random.randn(best_n), 0, 1), best_n)
        h_opt = anneal_alpha_optimize(h_init, best_n, [40, 200, 800], maxiter_each=150)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    # Phase 3: Resolution sweep
    n_candidates = sorted(set([best_n]))
    for dn in [-30, -20, -10, 10, 20, 30, 40]:
        nc = best_n + dn
        if 24 <= nc <= 200 and nc != best_n:
            n_candidates.append(nc)
    for n_try in n_candidates:
        if time_left() < 50:
            break
        if n_try == best_n:
            continue
        h_interp = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, best_n), best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt = anneal_alpha_optimize(h_interp, n_try, [50, 300], maxiter_each=150)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()
            best_n = n_try

    # Phase 4: Coordinate descent on exact objective
    if time_left() > 50:
        h_opt, c = coordinate_refine(best_h, best_n, n_rounds=25, n_cand=7)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    # Phase 5: High-alpha SLSQP for sharpening
    if time_left() > 40:
        h_opt = slsqp_smooth(best_h, best_n, 2000, maxiter=200)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    # Phase 6: Simulated annealing with block moves
    if time_left() > 30:
        sa_time = min(time_left() - 10, 120)
        n_sa = max(500, int(sa_time / 0.003))
        h_opt, c = simulated_annealing(best_h, best_n, n_sa, T0=0.004)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    # Phase 7: Final coordinate polish with finer grid
    if time_left() > 15:
        h_opt, c = coordinate_refine(best_h, best_n, n_rounds=15, n_cand=11)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    # Phase 8: One more SLSQP pass
    if time_left() > 10:
        h_opt = slsqp_smooth(best_h, best_n, 1500, maxiter=100)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
