    return project_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 20.0 - (time.time() - t0)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        best_h = np.full(80, 0.5)
    n = len(best_h)
    best_h = project_feasible(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [80, 200, 500, 1000]:
        if time_left() < 50:
            break
        h = slsqp_refine(best_h, alpha, maxiter=400)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 2: Coordinate descent
    if time_left() > 60:
        h, c = coord_descent(best_h, max_rounds=100, g_iters=28)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 3: Multi-start
    n_starts = 16
    intensities = [0.04, 0.08, 0.12, 0.18, 0.25]
    for s in range(n_starts):
        if time_left() < 35:
            break
        h0 = diverse_h(n, rng, best_h, intensities[s % len(intensities)])
        h = slsqp_refine(h0, 250, maxiter=250)
        h, c = coord_descent(h, max_rounds=40, g_iters=22)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 4: Simulated annealing (long)
    if time_left() > 40:
        sa_time = min(time_left() - 15, 300)
        h, c = sa_search(best_h, n, sa_time, rng)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        h, c = coord_descent(best_h, max_rounds=50, g_iters=25)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 5: Second round of multi-start (finer)
    for s in range(8):
        if time_left() < 35:
            break
        h0 = diverse_h(n, rng, best_h, intensities[s % len(intensities)] * 0.6)
        h = slsqp_refine(h0, 400, maxiter=300)
        h, c = coord_descent(h, max_rounds=30, g_iters=22)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 6: Final SA + polish
    if time_left() > 40:
        sa_time = min(time_left() - 15, 200)
        h, c = sa_search(best_h, n, sa_time, rng)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    if time_left() > 25:
        h = slsqp_refine(best_h, 800, maxiter=300)
        h, c = coord_descent(h, max_rounds=60, g_iters=28)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    if time_left() > 15:
        h, c = coord_descent(best_h, max_rounds=40, g_iters=30)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    best_h = project_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
