def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(2*n):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = np.argmin(1.0 - h)
                h[idx] = min(1.0, h[idx] + min(r, 1.0 - h[idx]))
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] - min(-r, h[idx]))
    return h

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    kmax = int(np.argmax(corr))
    shifted = beta * (corr - corr[kmax])
    shifted = np.clip(shifted, -700, 0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    softmax_w = exp_v / sum_exp
    S = (corr[kmax] + np.log(sum_exp) / beta) * dx
    
    weights = softmax_w.copy()
    weights[kmax] += 1.0
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        # Term 1: (1-h[m+(n-1)-k]) * I(0<=m+(n-1)-k<n)
        # m+(n-1)-k in [0,n-1] → m in [k-(n-1), n-1+k-(n-1)] = [k-n+1, k]
        # m in [0,n-1] → m in [max(0,k-n+1), min(n-1,k)]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            j = idx + (n - 1) - k
            grad[idx] += w * (1.0 - h[j])
        
        # Term 2: -h[m-(n-1)+k] * I(0<=m-(n-1)+k<n)
        # m-(n-1)+k in [0,n-1] → m in [(n-1)-k, n-1+n-1-k] = [n-1-k, 2n-2-k]
        # m in [0,n-1] → m in [max(0,n-1-k), min(n-1, 2n-2-k)]
        lo2 = max(0, n - 1 - k)
        hi2 = min(n - 1, 2 * n - 2 - k)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            j = idx - (n - 1) + k
            grad[idx] -= w * h[j]
    
    grad *= dx
    return S, grad

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(4 * n):
            r = target - np.sum(h)
            if abs(r) < 1e-12:
                break
            if r > 0:
                idx = int(np.argmin(1.0 - h))
                h[idx] = min(1.0, h[idx] + min(r, 1.0 - h[idx]))
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] - min(-r, h[idx]))
    return h

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    kmax = int(np.argmax(corr))
    shifted = np.clip(beta * (corr - corr[kmax]), -700, 0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    smw = exp_v / sum_exp
    S = (corr[kmax] + np.log(sum_exp) / beta) * dx
    weights = smw.copy()
    weights[kmax] += 1.0
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        lo1 = max(0, k - nm1)
        hi1 = min(nm1, k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += w * (1.0 - h[idx + nm1 - k])
        lo2 = max(0, nm1 - k)
        hi2 = min(nm1, 2 * nm1 - k)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] -= w * h[idx - nm1 + k]
    grad *= dx
    return S, grad

    n = len(h)
    h = h.astype(float).copy()
    ma = np.zeros(n)
    va = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stale = 0
    for t in range(1, steps + 1):
        if time.time() - t_ref > t_end:
            break
        frac = (t - 1) / max(steps - 1, 1)
        beta = b0 + (b1v - b0) * frac
        _, grad = smooth_max_and_grad(h, beta)
        for i in range(n):
            ma[i] = b1 * ma[i] + (1.0 - b1) * grad[i]
            va[i] = b2 * va[i] + (1.0 - b2) * grad[i] * grad[i]
        mh = ma / (1.0 - b1 ** t)
        vh = va / (1.0 - b2 ** t)
        h -= lr * mh / (np.sqrt(vh) + eps)
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h)
        if abs(diff) > 1e-8:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)

        if n <= 120 and budget_s * 0.86 - (time.time() - t0) > 8:
            h3, c3 = coord_desc(h1, rounds=18)
            if c3 < c1:
                h1, c1 = h3, c1 if c3 >= c1 else (h3, c3)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    if best_h is not None and budget_s * 0.95 - (time.time() - t0) > 15:
        nf = len(best_h)
        sf = max(200, min(800, int((budget_s * 0.95 - (time.time() - t0)) * 0.5)))
        hf, cf = adam_opt(best_h, sf, 0.006, 150, 500, t0, t0 + budget_s * 0.95)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        if nf <= 120 and budget_s * 0.98 - (time.time() - t0) > 8:
            hc, cc = coord_desc(best_h, rounds=12, dets=[0.04, -0.04, 0.08, -0.08, 0.02, -0.02])
            if cc < best_c5:
                best_c5, best_h = cc, hc

    if best_h is None:
        best_h = fix_feasible(tm_seq(48))
        best_c5 = compute_c5(best_h)
    best_h = fix_feasible(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5), len(best_h)
