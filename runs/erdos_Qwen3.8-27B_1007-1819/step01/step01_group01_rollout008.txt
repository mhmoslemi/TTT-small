def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -40, 40)
    exps = np.exp(shifted)
    total = np.sum(exps)

    grad = np.zeros(n)
    for k in range(nc):
        wk = w[k] * dx
        if wk < 1e-15:
            continue

            i2 = nn
            grad[i1] += wk * (1.0 - h[i2])
            grad[i2] -= wk * h[i1]
            if i1 == i2:
                grad[i1] -= wk * h[i1]
    return val, grad

def obj_and_jac(params, n, alpha):
    val, grad = smooth_max_and_grad(params, alpha)
    return val, grad

def slsqp_optimize(h0, n, alpha, maxiter=300):
    cons = [{'type': 'eq', 'fun': lambda p: np.sum(p) - n / 2.0,
             'jac': lambda p: np.ones(n)}]
    try:
        res = minimize(obj_and_jac, h0.copy(), args=(n, alpha),
                       method='SLSQP', constraints=cons,
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def pair_swap_sa(h0, n, budget_s, seed_val=0):
    rng = np.random.RandomState(seed_val)
    h = h0.copy()
    curr = compute_c5(h)
    best = h.copy()
    best_c = curr
    T0 = 0.003
    n_max = int(budget_s / 0.0005)
    for step in range(n_max):
        T = T0 * (1.0 - step / max(n_max, 1)) ** 2
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.15, 0.15)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        c_new = compute_c5(h_new)
        if c_new < curr or rng.rand() < np.exp(-(c_new - curr) / max(T, 1e-15)):
            h = h_new
            curr = c_new
            if curr < best_c:
                best_c = curr
                best = h.copy()
        if step % 1000 == 999 and time.time() - 0 > budget_s:
            break
    return best, best_c

def coord_descent(h0, n, n_rounds=30):
    h = h0.copy()
    curr = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            best_v = h[i]
            best_c = curr
            # Try binary moves and gradient-informed moves
            candidates = [0.0, 1.0, h[i] * 0.5, 1.0 - h[i] * 0.5]
            for rv in candidates:
                if abs(rv - h[i]) < 1e-6:
                    continue
                h[i] = rv
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_v = h[i]
                    improved = True
            h[i] = best_v
            h = project_feasible(h, n)
            curr = best_c
        if not improved:
            break
    return h, curr

def make_binary_seq(n, pattern='mseq'):

        n -= 1
    h = np.zeros(n)

        taps = [0, m - 1]
        h[:m] = reg
        for i in range(m, n):
            val = reg[0] * reg[taps[1]] % 2
            reg = np.roll(reg, 1)
            reg[0] = val
            h[i] = reg[0]
        # Balance: need n/2 ones

    elif pattern == 'block':
