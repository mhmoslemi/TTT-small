    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(2 * n + 2):
            r = target - np.sum(h)
            if abs(r) < 1e-10:
                break
            if r > 0:
                idx = np.argsort(1.0 - h)
                for i in idx:
                    a = min(r, 1.0 - h[i])
                    h[i] += a
                    r -= a
                    if r < 1e-12:

            else:
                idx = np.argsort(h)
                for i in idx:
                    a = min(-r, h[i])
                    h[i] -= a

    return h

    dx = 2.0 / n

    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
