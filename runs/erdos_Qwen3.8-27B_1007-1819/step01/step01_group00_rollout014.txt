    h = np.asarray(h, dtype=float).copy()
    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

_N = 100

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return (mx + np.log(s) / beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    w = np.exp(beta * (corr - mx))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        lo1, hi1 = max(0, s), min(n, n + s)
        if lo1 < hi1:
            g[lo1:hi1] += wk * (1.0 - h[lo1 - s:hi1 - s])
        lo2, hi2 = max(0, -s), min(n, n - s)
        if lo2 < hi2:
            g[lo2:hi2] -= wk * h[lo2 + s:hi2 + s]
    return g * dx

def _obj_smooth(x):
    return smooth_max_val(project_feasible(x), 300.0)

def _con_sum(x):
    return np.sum(x) - _N / 2.0

def tm_seq(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def pgd_run(h0, n_steps, beta, step_init, decay):
    h = project_feasible(h0)
    best = h.copy()
    best_val = compute_c5(h)
    step = step_init
    for i in range(n_steps):
        _, g = smooth_max_grad(h, beta)
        h = project_feasible(h - step * g)
        step *= decay
        v = compute_c5(h)
        if v < best_val:
            best_val = v
            best = h.copy()
    return best, best_val

def coord_descent(h, max_rounds=20):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            h_trial = h[i]
            for d in deltas:
                v = h_trial + d
                if v < 0 or v > 1:
                    continue
                h[i] = v
                h = project_feasible(h)
                val = compute_c5(h)
                if val < cur - 1e-12:
                    cur = val
                    h_trial = v
                    improved = True
                    break
                else:
                    h[i] = h_trial
            if h[i] != h_trial:
                h[i] = h_trial
        if not improved:
            break
    return h, cur

def coord_descent_wide(h, max_rounds=10):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    deltas = [0.05, -0.05, 0.1, -0.1, 0.2, -0.2, 0.3, -0.3]
    for _ in range(max_rounds):
        improved = False

                    break
                else:
                    h[i] = h_trial
            if h[i] != h_trial:
                h[i] = h_trial
        if not improved:
            break
    return h, cur

    np.random.seed(seed)

    global _N

    best_h = None

        best_c5 = compute_c5(best_h)

    def time_frac_mark(frac):

    # Phase 1: PGD from diverse starts
    starts = [
        best_h.copy(),
        tm_seq(n),
        project_feasible(np.random.uniform(0.25, 0.75, n)),

    ]

    for hs in starts:
        if not time_frac_mark(0.45):
            break
        for beta in [50, 100, 200]:
            if not time_frac_mark(0.5):

            h_r, c5_r = pgd_run(hs, 3000, beta, 0.01, 0.999)
            if c5_r < best_c5:

    if best_h is not None and time_frac_mark(0.65):

            h_new = project_feasible(res.x)

        best_c5 = compute_c5(best_h)
