                h[i] = max(0.0, h[i] - min(-r, h[i]))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n + 1):
            r = target - np.sum(h)
            if abs(r) < 1e-10:
                break
            if r > 0:
                i = int(np.argmin(1.0 - h))
                h[i] = min(1.0, h[i] + min(r, 1.0 - h[i]))
            else:
                i = int(np.argmax(h))
                h[i] = max(0.0, h[i] - min(-r, h[i]))
    return h

def smooth_max_corr(hf, beta):
    corr = np.correlate(hf, 1.0 - hf, mode="full")
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return (mx + np.log(s) / beta) * (2.0 / len(hf))

def sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t

    return smooth_max_corr(fix_feasible(x), 1500.0)

def coord_refine(h, max_rounds=20):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    deltas = [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.25, -0.25, 0.02, -0.02, 0.35, -0.35]
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            best_v = h[i]
            best_obj = cur
            for d in deltas:
                v = h[i] + d
                if v < -1e-9 or v > 1.0 + 1e-9:
                    continue
                h_trial = h.copy()
                h_trial[i] = np.clip(v, 0.0, 1.0)
                h_trial = fix_feasible(h_trial)
                obj = compute_c5(h_trial)
                if obj < best_obj:
                    best_obj = obj
                    best_v = np.clip(v, 0.0, 1.0)
            if best_v != h[i]:
                h[i] = best_v
                h = fix_feasible(h)
                cur = best_obj
                improved = True
        if not improved:
            break
    return h

def binary_flip_refine(h, max_rounds=10):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            targets = [0.0, 1.0, 0.25, 0.75, 0.5]
            if abs(h[i] - 0.0) < 0.01:
                targets.remove(0.0)
            if abs(h[i] - 1.0) < 0.01:
                targets.remove(1.0)
            for tv in targets:
                if abs(tv - h[i]) < 0.01:
                    continue
                h_trial = h.copy()
                h_trial[i] = tv
                h_trial = fix_feasible(h_trial)
                obj = compute_c5(h_trial)
                if obj < cur - 1e-12:
                    h = h_trial.copy()
                    cur = obj
                    improved = True
                    break
        if not improved:
            break
    return h

def slsqp_pass(h, objf, nb, nmax=400, ftol=1e-11):
    try:
        res = minimize(objf, h, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': sum_eq}],
                       bounds=[(0, 1)] * nb,
                       options={'maxiter': nmax, 'ftol': ftol})
        return fix_feasible(res.x)
    except Exception:
        return h.copy()
