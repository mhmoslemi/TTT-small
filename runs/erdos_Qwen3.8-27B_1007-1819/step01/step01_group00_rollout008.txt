                                       min(20, rem * 0.2))
                c5v = compute_c5(h_cd)
                if c5v < best_c5:
                    best_c5 = c5v
                    best_h = h_cd.copy()

    # Phase 4: Re-interpolate best to n_main if needed, then final polish
    if best_h is not None:
        nb = len(best_h)
        if nb != n_main:
            h_interp = np.interp(np.linspace(0, nb - 1, n_main),
                                 np.arange(nb), best_h)
            h_interp = fix_feasible(h_interp)
            c5v = compute_c5(h_interp)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_interp
        # High-beta L-BFGS-B polish
        rem = budget_s * 0.92 - (time.time() - t0)
        if rem > 10:
            h_pol = run_lbfgsb(best_h, [200, 500], 400, rem * 0.5)
            h_pol = fix_feasible(h_pol)
            c5v = compute_c5(h_pol)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_pol.copy()

    # Phase 5: Fine coordinate descent
    if best_h is not None:
        rem = budget_s * 0.96 - (time.time() - t0)
        if rem > 5:
            steps_fine = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.04, -0.04, 0.08, -0.08]
            h_fin, _ = coord_refine(best_h, 50, steps_fine, rem)
            c5v = compute_c5(h_fin)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_fin

    if best_h is None:
        best_h = fix_feasible(tm_seq(60))
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
