            h_hi = to_res(best_h, best_n, n_hi)
            h_hi = project_feasible(h_hi, n_hi)
            bounds_hi = [(0.0, 1.0)] * n_hi
            cons_hi = [{'type': 'eq', 'fun': sum_con, 'args': (n_hi,)}]
            for beta in [100.0, 300.0, 800.0]:
                if elapsed() > budget_s * 0.72:
                    break
                res = minimize(
                    smooth_obj, h_hi, args=(n_hi, beta),
                    method='SLSQP', bounds=bounds_hi, constraints=cons_hi,
                    options={'maxiter': 300, 'ftol': 1e-15}
                )
                if res.x is not None:
                    h_opt = project_feasible(res.x, n_hi)
                    c5_v, _ = compute_c5(h_opt)
                    if c5_v < best_c5:
                        best_c5 = c5_v
                        best_h = h_opt.copy()
                        best_n = n_hi
                    h_hi = h_opt
            if elapsed() < budget_s * 0.70:
                h_cd2, c5_cd2 = coord_descent(h_hi, n_hi, max_sweeps=100)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2
                    best_n = n_hi

    # Final careful coordinate descent
    if elapsed() < budget_s * 0.92:
        h_final, c5_final = coord_descent(best_h, best_n, max_sweeps=200, tol=1e-13)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Final projection and verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n)
    best_c5, _ = compute_c5(best_h)

    return best_h, best_c5, best_n
