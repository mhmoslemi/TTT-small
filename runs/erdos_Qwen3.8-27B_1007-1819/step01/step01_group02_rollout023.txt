```python
        if np.clip(x - mid, 0, 1).sum() > target:

        else:

        return h.copy()
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(2, 0, n_from, endpoint=False)  # wrap-around

    x_new = np.arange(n_to) * 2.0 / n_to

    m = corr.max()
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

    dx = 2.0 / n
    shift = k_star - (n - 1)
    j = np.arange(n)
    g_idx = j + shift
    h_idx = j - shift
    grad = np.zeros(n)

    m2 = (h_idx >= 0) & (h_idx < n)
    grad[m2] -= h[h_idx[m2]]

        for k in idx:
            sg += grad_sub(h, k, n)

            best_local = c5_new
            # Adaptive lr: increase slightly

        else:
            lr *= 0.7
```

```python
        m1 = (g_idx >= 0) & (g_idx < n)
        m2 = (h_idx >= 0) & (h_idx < n)
        gk = np.zeros(n)
        gk[m1] += (1.0 - h[g_idx[m1]])
        gk[m2] -= h[h_idx[m2]]
        grad += weights[k] * gk
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -0.5, 0.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.arange(n_from) * 2.0 / n_from
    x_new = np.arange(n_to) * 2.0 / n_to
    return np.interp(x_new, x_old, h)


def grad_smooth_max(h, n, beta):
    corr = compute_corrs(h)
    m = corr.max()
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        g_idx = j + shift
        h_idx = j - shift
        m1 = (g_idx >= 0) & (g_idx < n)
        m2 = (h_idx >= 0) & (h_idx < n)
        grad[m1] += w[k] * (1.0 - h[g_idx[m1]])
        grad[m2] -= w[k] * h[h_idx[m2]]
    return grad * dx


def projected_grad_descent(h, n, t_end, beta, lr_init=0.01, max_iter=5000):
    h = h.copy()
    best_local = compute_c5(h)
    lr = lr_init
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = grad_smooth_max(h, n, beta)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_local - 1e-16:
            h = h_new
            best_local = c5_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.8
            if lr < 1e-6:
                break
    return h, best_local


def constraint_sum(x, n_pts):
    return x.sum() - n_pts / 2.0


def obj_lse(h, beta):
    corr = compute_corrs(h)
    m = corr.max()
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta


def slsqp_refine(h, n, beta, max_iter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}]
    res = minimize(obj_lse, h, args=(beta,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    if res.x is not None:
        h_opt = project_feasible(res.x, n)
        return h_opt, compute_c5(h_opt)
    return h, compute_c5(h)


def coord_descent(h, n, t_end, n_passes=3):
    h = h.copy()
    best_local = compute_c5(h)
    steps = [0.01, 0.005, 0.02]
    for _ in range(n_passes):
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for d in steps:
                for sign in [1.0, -1.0]:
                    h_test = h.copy()
                    h_test[i] = np.clip(h_test[i] + sign * d, 0.0, 1.0)
                    h_test = project_feasible(h_test, n)
                    c5_t = compute_c5(h_test)
                    if c5_t < best_local - 1e-13:
                        h = h_test
                        best_local = c5_t
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_local
```
