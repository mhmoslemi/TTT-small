        alpha = alpha0 * (1.0 - 0.9 * progress)

        m = np.max(corr)

            g += 1.0

            else:

            else:

Wait, the coordinate descent with project_feasible is problematic. When I change one coordinate and project, the projection shifts ALL coordinates. This makes the "coordinate descent" not really per-coordinate.

Better approach: Fix the sum constraint by adjusting only the changed coordinate and one "compensating" coordinate. Or use a different projection that only adjusts the minimum amount.

Actually, for the constraint sum(h) = n/2, if I change h[i] by δ, I need to compensate by changing other coordinates. The simplest is to distribute the compensation evenly:
h_new[i] = h[i] + δ
h_new[j≠i] = h[j] - δ/(n-1) (for all j≠i)

But this might push some values out of [0,1]. A better approach: just use the full projection but be aware that it changes all values.

Alternatively, I can use a two-coordinate swap: change h[i] by +δ and h[j] by -δ, keeping the sum exactly. This is cleaner.

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)


def subgrad_compute(h, n, topk):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    k = min(topk, len(corr))
    idx = np.argpartition(-corr, k)[:k]
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            if lag == 0:
                g += 1.0 - 2.0 * h
            else:
                g[lag:] += 1.0
                g[lag:] -= h[:n - lag]
                g[:n - lag] -= h[lag:]
        else:
            p = -lag
            g[:n - p] += 1.0
            g[:n - p] -= h[p:]
            g[p:] -= h[:n - p]
    return g / float(k), m


def sg_momentum(h_init, n, n_iter, t_deadline, alpha0, mu, topk):
    h_curr = h_init.copy()
    v = np.zeros(n)
    best = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    stall = 0
    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        progress = float(it) / max(n_iter, 1)
        alpha = alpha0 * (1.0 - 0.8 * progress)
        mu_a = mu * (1.0 - 0.15 * progress)
        g, _ = subgrad_compute(h_curr, n, topk)
        v = mu_a * v - alpha * g
        h_next = project_feasible(h_curr + v, n)
        c5 = compute_c5(h_next)
        if c5 < best_c5:
            best_c5 = c5
            best = h_next.copy()
            stall = 0
        else:
            stall += 1
        h_curr = h_next
        if stall > 500 and progress > 0.1:
            h_curr = project_feasible(h_curr + 0.02 * np.random.randn(n), n)
            v = np.zeros(n)
            stall = 0
    return best, best_c5


def swap_search(h_init, n, t_deadline, n_rounds, rng, deltas):
    h_curr = h_init.copy()
    best_c5 = compute_c5(h_curr)
    for rnd in range(n_rounds):
        if time.time() > t_deadline:
            break
        improved = False
        n_pairs = max(60, n)
        for _ in range(n_pairs):
            if time.time() > t_deadline:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                for sign in (-1, 1):
                    h_test = h_curr.copy()
                    h_test[i] += sign * d
                    h_test[j] -= sign * d
                    if h_test.min() < -1e-12 or h_test.max() > 1.0 + 1e-12:
                        continue
                    h_test = np.clip(h_test, 0.0, 1.0)
                    h_test = project_feasible(h_test, n)
                    c5_t = compute_c5(h_test)
                    if c5_t < best_c5 - 1e-13:
                        h_curr = h_test
                        best_c5 = c5_t
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h_curr, best_c5


def interp_to_size(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return np.interp(x_new, x_old, h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_init = len(h0)
    except NameError:
        n_init = 45
        h0 = np.full(n_init, 0.5)

    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n_init

    # Phase 1: Subgradient with momentum at native resolution
    t_dl = t0 + budget_s * 0.35
    h_sg, c5_sg = sg_momentum(h0, n_init, 6000, t_dl, 0.04, 0.85, 5)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg
        best_n = n_init

    # Phase 2: Try alternate resolutions
    if time.time() - t0 < budget_s * 0.42:
        for n_try in (90, 135, 225):
            if time.time() - t0 > budget_s * 0.52:
                break
            h_interp = interp_to_size(best_h, best_n, n_try)
            h_interp = project_feasible(h_interp, n_try)
            t_dl2 = t0 + budget_s * 0.52
            h_sg2, c5_sg2 = sg_momentum(h_interp, n_try, 3000, t_dl2, 0.05, 0.80, 3)
            if c5_sg2 < best_c5:
                best_c5 = c5_sg2
                best_h = h_sg2
                best_n = n_try

    # Phase 3: Coarse swap-based local search
    if time.time() - t0 < budget_s * 0.62:
        t_dl3 = t0 + budget_s * 0.76
        deltas = (0.03, 0.08, 0.15, 0.25, 0.5)
        h_sw, c5_sw = swap_search(best_h, best_n, t_dl3, 25, rng, deltas)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw

    # Phase 4: Fine-grained coordinate swaps
    if time.time() - t0 < budget_s * 0.80:
        t_dl4 = t0 + budget_s * 0.92
        h_curr = best_h.copy()
        n_pts = best_n
        best_local = best_c5
        small_deltas = (0.005, 0.01, 0.02, 0.04)
        for _ in range(2000):
            if time.time() - t0 > budget_s * 0.92:
                break
            i = rng.randint(n_pts)
            j = rng.randint(n_pts)
            if i == j:
                continue
            improved = False
            for d in small_deltas:
                for sign in (-1, 1):
                    h_test = h_curr.copy()
                    h_test[i] += sign * d
                    h_test[j] -= sign * d
                    if h_test.min() < -1e-12 or h_test.max() > 1.0 + 1e-12:
                        continue
                    h_test = np.clip(h_test, 0.0, 1.0)
                    h_test = project_feasible(h_test, n_pts)
                    c5_t = compute_c5(h_test)
                    if c5_t < best_local - 1e-13:
                        h_curr = h_test
                        best_local = c5_t
                        improved = True
                        break
                if improved:
                    break
        if best_local < best_c5:
            best_c5 = best_local
            best_h = h_curr

    # Phase 5: Final subgradient polish with wider topk
    if time.time() - t0 < budget_s * 0.94:
        t_dl5 = t0 + budget_s * 0.97
        h_fin, c5_fin = sg_momentum(best_h, best_n, 4000, t_dl5, 0.008, 0.65, 12)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
