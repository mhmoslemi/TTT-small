import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_smooth(x, n, alpha):
    h = project_feasible(x, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    h = project_feasible(x, n)

def cons_fn(x):
    return np.sum(x) - len(x) / 2.0

    n_old = len(h)

    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)
