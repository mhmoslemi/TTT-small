        h += diff / n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n + 1):
            r = target - np.sum(h)
            if abs(r) < 1e-10:
                break
            if r > 0:
                i = int(np.argmin(1.0 - h))
                h[i] = min(1.0, h[i] + min(r, 1.0 - h[i]))
            else:
                i = int(np.argmax(h))
                h[i] = max(0.0, h[i] - min(-r, h[i]))
    return h

def corr_and_softmax(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = np.max(corr)
    e = np.exp(beta * (corr - M))
    s = np.sum(e)
    w = e / s
    soft = M + np.log(s) / beta
    return corr, soft, w

def soft_grad(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        term = np.zeros(n)
        if s == 0:
            term[:] = 1.0 - 2.0 * h
        elif s > 0:
            term[:n - s] = 1.0 - h[s:]
            term[s:] -= h[:n - s]
        else:
            a = -s
            term[a:] = 1.0 - h[:n - a]
            term[:n - a] -= h[a:]
        grad += w[k] * term
    return grad * dx

def obj_with_pen(h, beta, lam):
    hc = np.clip(h, 0, 1)
    _, soft, _ = corr_and_softmax(hc, beta)
    return soft * (2.0 / len(hc)) + lam * (np.sum(hc) - len(hc) / 2.0) ** 2

def grad_with_pen(h, beta, lam):
    hc = np.clip(h, 0, 1)
    _, _, w = corr_and_softmax(hc, beta)
    g = soft_grad(hc, w)
    g += 2.0 * lam * (np.sum(hc) - len(hc) / 2.0)
    return g

def lbfgs_step(h0, beta, lam, maxiter=400):
    n = len(h0)
    res = minimize(obj_with_pen, h0, jac=lambda x: grad_with_pen(x, beta, lam),
                   method='L-BFGS-B', bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-14, 'maxls': 40})
    return fix_feasible(res.x), res.fun

def tm_seq(n):
    t = np.zeros(n)
    for i in range(n):
        c, x = 0, i
        while x:
            c += x & 1
            x >>= 1
        t[i] = c % 2
    return t

def gen_starts(n, rng):
    starts = []
    starts.append(tm_seq(n))
    x = np.arange(n)
    for freq, amp in [(2, 0.35), (3, 0.3), (4, 0.25)]:
        starts.append(np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x / n), 0, 1))
    starts.append((rng.random(n) > 0.5).astype(float))
    starts.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0, 1))
    blk = max(2, n // 8)
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i // blk) % 2 == 0 else 0.0
    starts.append(h)
    starts.append(np.clip(0.5 + 0.25 * np.sin(6 * np.pi * x / n) + 0.15 * np.cos(2 * np.pi * x / n), 0, 1))
    return [fix_feasible(s) for s in starts]

def coord_refine(h, rounds=40, step=0.04):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    for _ in range(rounds):
        improved = False
        for i in range(n):
            for d in [step, -step, step * 0.5, -step * 0.5]:
                trial = h.copy()
                v = trial[i] + d
                if 0.0 <= v <= 1.0:
                    trial[i] = v
                    trial = fix_feasible(trial)
                    c = compute_c5(trial)
                    if c < cur - 1e-12:
                        cur = c
                        h = trial
                        improved = True
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1e10

    def elapsed():
        return time.time() - t0

    def time_left():
        return max(0.0, budget_s - elapsed())

    parent_h = None

        best_h = parent_h.copy()

    sizes = [len(parent_h)] if parent_h is not None else []
    for nn in [30, 40, 48, 60, 64, 80, 100]:
        if nn not in sizes:
            sizes.append(nn)
    sizes = sorted(set(s for s in sizes if 20 <= s <= 120))

    for n in sizes:

        n_starts = 5 if n <= 64 else 3
        starts = gen_starts(n, rng)

        for si in range(min(n_starts, len(starts))):
            if time_left() < 20.0:
                break
            if si == 0 and parent_h is not None and n == len(parent_h):

            else:
                h = starts[si % len(starts)]

            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c

            # Multi-beta L-BFGS-B optimization
            lam = 500.0
            for beta in [30, 100, 500, 2000]:
                if time_left() < 15.0:

                try:
                    h_new, _ = lbfgs_step(h, beta, lam, maxiter=300)
                    c = compute_c5(h_new)
                    if c < best_c5:

                    h = h_new
                except Exception:

                    best_h = h_cd

            # Perturbation + re-optimize
