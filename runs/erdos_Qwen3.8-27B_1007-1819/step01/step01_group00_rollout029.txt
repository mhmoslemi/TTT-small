import numpy as np
from scipy.optimize import minimize
import time

_g_beta = 500.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_feasible(h):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(n + 10):
            r = target - np.sum(h)
            if abs(r) < 1e-10:
                break
            if r > 0:
                i = int(np.argmin(1.0 - h))
                h[i] = min(1.0, h[i] + min(r, 1.0 - h[i]))
            else:
                i = int(np.argmax(h))
                h[i] = max(0.0, h[i] - min(-r, h[i]))
    return h

def smooth_max_corr(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    s = np.sum(np.exp(beta * (corr - mx)))
    return (mx + np.log(s) / beta) * dx

def objective_smooth(x):
    return smooth_max_corr(fix_feasible(x), _g_beta)

def constraint_sum(x):
    return np.sum(x) - len(x) / 2.0

            break

            if deadline is not None and time.time() > deadline:
                break
            for s in steps:

                    v = h[i] + d * s
                    if v < -1e-9 or v > 1 + 1e-9:

                    h_trial = h.copy()

                    h_trial = fix_feasible(h_trial)
                    obj = compute_c5(h_trial)
                    if obj < cur - 1e-13:
                        cur = obj
                        h = h_trial.copy()
                        round_improved = True
                        break
                if round_improved:
                    break
        if not round_improved:
            break
    return h, cur

            h_trial[i], h_trial[j] = h[j], h[i]
            h_trial = fix_feasible(h_trial)
            obj = compute_c5(h_trial)
            if obj < cur - 1e-13:
                cur = obj
                h = h_trial.copy()
                improved = True
        if not improved:
            break
    return h, cur
