    n = a.size

    # normalize to sum=1 (or not)

    if sum_a == 0:

    else:
        a /= sum_a  # simple normalization to sum=1

    best = a.copy()

    len_fft = 1 << ((2*n - 1).bit_length())

    # iteration
    init_alpha = 0.3
    decay_steps = max(1, num_steps // 10)

    for step in range(num_steps):

        fa = np.fft.rfft(a, n=len_fft)
        conv = np.fft.irfft(fa * fa, n=len_fft)[:2*n-1]

        max_val = conv.max()
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # range indices contributing to max conv
        idxs = np.arange(lo, hi + 1, dtype=np.int64)
        # compute weights = a[k - idxs]
        w = a[k - idxs]
        # find source index with largest w
        src_local = int(np.argmax(w))
        src = int(idxs[src_local])

        # choose destination index with smallest w (prefer outside range)
        # first try outside range; if none, pick global min
        if lo > 0:
            dst_candidates = np.arange(0, lo, dtype=np.int64)
        elif hi < n-1:
            dst_candidates = np.arange(hi+1, n, dtype=np.int64)
        else:
            dst_candidates = np.arange(0, n, dtype=np.int64)
        # among candidates, pick one with minimal weight
        dst_weights = a[dst_candidates]  # weight = a[k - dst] but k-dst out of range =>0
        # all out-of-range have zero weight, good to use first candidate
        if dst_candidates.size > 0:
            dst = int(dst_candidates[np.argmin(dst_weights)])
        else:
            dst = src  # shouldn't happen

        if src == dst:

        # amount to move
        alpha = init_alpha / (1.0 + step / decay_steps)
        delta = alpha * a[src]  # move fraction

        score = float(evaluate_sequence(cand.tolist()))

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _load_parent_sequence():
    """Return the provided parent sequence if it exists, else None."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    return None


def _arcsine_sequence(n):
    """
    Generate the truncated arcsine coefficients:
        a0 = 1,
        a_i = a_{i-1} * (1 - 0.5 / i)  for i >= 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _objective_from_conv(max_val, sum_a, n):
    """Compute the AC1 upper bound given max convolution entry."""
    return 2.0 * n * max_val / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=4000, **kwargs):
    """
    Search for a non‑negative step function that reduces the AC1 bound.
    A simple projected gradient descent is performed on the maximum
    convolution coefficient.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from parent or arcsine
    # ------------------------------------------------------------------
    parent = _load_parent_sequence()
    if parent is not None:
        a = parent.copy()
    else:
        # fallback to a modest arcsine of length 256
        a = _arcsine_sequence(256)

    n = a.size
    # Normalise to unit L1 norm – scaling does not affect the objective.
    total = a.sum()
    if total < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / total

    # Evaluate parent/arcsine candidates using the supplied evaluator.
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Try the pure arcsine of the same length (also normalised).
    arcsine = _arcsine_sequence(n)
    arcsine = arcsine / arcsine.sum()
    arcsine_score = float(evaluate_sequence(arcsine.tolist()))
    if arcsine_score < best_score:
        best = arcsine.copy()
        best_score = arcsine_score
        a = arcsine.copy()

    # ------------------------------------------------------------------
    # Pre‑compute FFT size
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    # sum of coefficients stays constant (unit L1 norm)
    sum_a = a.sum()
    current = a.copy()

    # schedule for the amount of mass moved each iteration
    init_alpha = 0.25
    decay_rate = max(1, num_steps // 10)

    for step in range(num_steps):
        # --------------------------------------------------------------
        # Compute full self‑convolution via FFT
        # --------------------------------------------------------------
        fa = np.fft.rfft(current, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

        max_val = conv.max()
        k = int(conv.argmax())
        current_score = _objective_from_conv(max_val, sum_a, n)

        # --------------------------------------------------------------
        # Identify indices influencing the maximal entry
        # --------------------------------------------------------------
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idxs = np.arange(lo, hi + 1, dtype=np.int64)
        # weight = a_{k-i}
        weights = current[k - idxs]
        src_local = int(np.argmax(weights))
        src = int(idxs[src_local])

        # Choose a destination index outside the influencing window
        if lo > 0:
            left = np.arange(0, lo, dtype=np.int64)
        else:
            left = np.empty(0, dtype=np.int64)
        if hi < n - 1:
            right = np.arange(hi + 1, n, dtype=np.int64)
        else:
            right = np.empty(0, dtype=np.int64)
        dst_pool = np.concatenate((left, right)) if (left.size + right.size) else np.arange(0, n, dtype=np.int64)
        dst = int(rng.choice(dst_pool))

        if src == dst:
            continue

        # --------------------------------------------------------------
        # Propose a mass transfer from src to dst
        # --------------------------------------------------------------
        alpha = init_alpha / (1.0 + step / decay_rate)
        delta = alpha * current[src]
        if delta <= 0.0:
            continue

        candidate = current.copy()
        candidate[src] -= delta
        candidate[dst] += delta
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # --------------------------------------------------------------
        # Evaluate the candidate (FFT based for speed)
        # --------------------------------------------------------------
        fa_c = np.fft.rfft(candidate, n=fft_len)
        conv_c = np.fft.irfft(fa_c * fa_c, n=fft_len)[: 2 * n - 1]
        cand_max = conv_c.max()
        cand_score = _objective_from_conv(cand_max, sum_a, n)

        # Keep the best ever seen
        if cand_score < best_score:
            best = candidate.copy()
            best_score = float(cand_score)

        # Simple acceptance rule with a tiny annealing component
        if cand_score < current_score:
            current = candidate
        else:
            # occasional uphill jumps
            temperature = 0.01 + 0.99 * (1.0 - step / num_steps)
            if np.exp((current_score - cand_score) / temperature) > rng.random():
                current = candidate

        # --------------------------------------------------------------
        # Occasional random block perturbation to shake out of plateaus
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            width = max(1, n // 150)
            start = rng.integers(0, n - width + 1)
            stop = start + width
            pert = current.copy()
            scale = np.exp(rng.normal(0.0, 0.09))
            pert[start:stop] *= scale

            np.clip(pert, 0.0, 1000.0, out=pert)

            fa_p = np.fft.rfft(pert, n=fft_len)
            conv_p = np.fft.irfft(fa_p * fa_p, n=fft_len)[: 2 * n - 1]
            p_score = _objective_from_conv(conv_p.max(), sum_a, n)

            if p_score < best_score:
                best = pert.copy()
                best_score = float(p_score)

            if p_score < current_score:
                current = pert

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
