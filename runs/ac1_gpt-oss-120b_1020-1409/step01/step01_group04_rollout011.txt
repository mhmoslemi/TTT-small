def subgradient_descent(a, max_iters=600, eta=0.5):
    n = a.shape[0]
    size = 2 * n
    for t in range(max_iters):
        fa = np.fft.rfft(a, n=size)
        b = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]
        k = int(np.argmax(b))
        # gradient for max convolution
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        grad = np.zeros_like(a)
        # idx range
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] = a[k - idx]
        # Step size
        step = eta / np.sqrt(t + 1.0)
        a = a - step * grad
        # enforce nonnegative
        a = np.maximum(a, 0.0)
        # renormalize to sum=1
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            break
        # Occasionally add small random perturbation to escape uniform
        if (t + 1) % 100 == 0:
            # add small noise
            noise = (rng.random(a.shape) - 0.5) * 0.01
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    return a

def _power_seq(alpha, n):
    """1/sqrt(i+alpha) sequence, clipped to allowed range."""

    best_score = np.inf

    for t in range(max_iters):
        # convolution via FFT (real‑valued)
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]
        k = int(np.argmax(conv))               # index of the current maximum

        # gradient of the active linear functional b_k
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        grad = np.zeros_like(a)
        if i0 <= i1:
            idx = np.arange(i0, i1 + 1, dtype=np.int64)
            grad[idx] = a[k - idx]

        # diminishing stepsize
        step = eta / np.sqrt(t + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)

        # project onto the simplex (sum == 1, non‑negative)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            break

        # occasional jitter to avoid stagnation at uniform points
        if (t + 1) % 100 == 0:
            jitter = (rng.random(a.shape) - 0.5) * 0.02
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # keep the best iterate seen so far (lightweight check)
        if (t + 1) % 50 == 0:
            try:
                sc = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = a.copy()
            except Exception:
                pass

    return best

def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and a fallback parent.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Assemble a modest portfolio of seeds (all L1‑normalised).
    # ------------------------------------------------------------------
    seeds = []

    seeds.append(_normalize(parent))

    # power‑law family
    alphas = np.logspace(-5, 1, num=12, dtype=np.float64)
    for a in alphas:
        seeds.append(_normalize(_power_seq(a, n)))

    # linear decay
    seeds.append(_normalize(_linear_seq(n)))

    # exponential decays with several time‑constants
    for tau in (n / 4, n / 2, n, n * 2):
        seeds.append(_normalize(_exp_seq(tau, n)))

    # random Dirichlet‑like start
    rand_vec = rng.random(n, dtype=np.float64)
    seeds.append(_normalize(rand_vec))

    # ------------------------------------------------------------------
    # Pick the best among the raw seeds (cheap O(n²) evaluation).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for seq in seeds:
        cand = np.clip(seq, 0.0, 1000.0)
        try:
            sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    if best_seq is None:
        best_seq = np.clip(parent, 0.0, 1000.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Refine the champion with projected sub‑gradient descent.
    # ------------------------------------------------------------------
    refined = _sgd_descent(best_seq.copy(), rng, max_iters=800, eta=0.5)
    try:
        refined_score = float(evaluate_sequence(refined.tolist()))
        if np.isfinite(refined_score) and refined_score < best_score:
            best_seq = refined
            best_score = refined_score
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Final safety net – guarantee non‑negativity, bounds and minimal mass.
    # ------------------------------------------------------------------
    final = np.clip(best_seq, 0.0, 1000.0)
    if final.sum() < 0.01 or not np.isfinite(best_score):
        final = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in final.tolist()]
