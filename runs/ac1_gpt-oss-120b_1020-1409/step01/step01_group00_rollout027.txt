import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the classic binomial (1/√(1‑x)) sequence.
# ----------------------------------------------------------------------
def _generate_binomial_sequence(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


# ----------------------------------------------------------------------
# Helper: compute full convolution via FFT and return max value + argmax.
# ----------------------------------------------------------------------
def _max_convolution(arr: np.ndarray, fft_len: int):
    # Real FFT of the zero‑padded signal
    f = np.fft.rfft(arr, fft_len)
    conv_full = np.fft.irfft(f * f, fft_len)[: 2 * arr.size - 1]
    max_val = float(np.max(conv_full))
    max_idx = int(np.argmax(conv_full))
    return max_val, max_idx, conv_full
