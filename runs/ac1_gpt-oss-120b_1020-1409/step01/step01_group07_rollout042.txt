import numpy as np

# ---------- helper functions (top‑level) ----------
def _fft_len(n):
    """Smallest power of two >= 2*n-1 (for a full convolution)."""
    target = 2 * n - 1
    return 1 << (target - 1).bit_length()


def _full_convolution_fft(a, L):
    """Return a*a (full convolution) using a real FFT of length L."""
    # rfft returns N/2+1 complex numbers for a real input of length N
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    # keep only the first 2*len(a)-1 entries (the valid part)
    return conv[: 2 * a.size - 1]


def _score(a, L, n):
    """Quadratic AC1 upper bound used by the evaluator."""
    s = a.sum()
    if s == 0.0:
        return np.inf
    conv = _full_convolution_fft(a, L)
    # numerical noise may produce tiny negatives – ignore them
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def _initial_arcsine_like(n):
    """
    Continuous arcsine‑type density (Beta(½,½)) discretised on a grid.
    a_i ∝ 1/√[(i+½)*(n-i-½)]  (high at both ends, low in the middle).
    """
    i = np.arange(n, dtype=np.float64)
    # Shift by 0.5 to avoid division by zero at the exact ends
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


def _gradient_at_lag(a, lag):
    """
    Exact gradient of the convolution entry C[lag] = Σ a_i a_{lag-i}
    with respect to a (non‑negative entries).  For interior points
    the derivative is 2*a_{lag-i} whenever the index exists.
    """
    n = a.size
    g = np.zeros_like(a)
    idx = lag - np.arange(n)            # index of the partner term
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * a[idx[mask]]
    return g

    current = _initial_arcsine_like(n)
    best = current.copy()
    best_score = _score(best, L, n)

    # keep a simple uniform fallback in case everything fails

    uniform_score = _score(uniform, L, n)
    if uniform_score < best_score:
