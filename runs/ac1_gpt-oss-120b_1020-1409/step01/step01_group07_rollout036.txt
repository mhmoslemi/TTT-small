import numpy as np
from numpy.fft import fft, ifft

# ---------- helper functions -------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_conv(seq: np.ndarray) -> np.ndarray:
    """Convolution of seq with itself using a power‑of‑two FFT."""
    n = seq.size
    nfft = _next_pow2(2 * n - 1)
    f = fft(seq, nfft)
    conv = ifft(f * f).real
    return conv[: 2 * n - 1]


def _max_conv(seq: np.ndarray):
    """Return (max value, index) of the self‑convolution."""
    conv = _full_conv(seq)
    # tiny negative parts from rounding are harmless for max
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _gradient_step(seq: np.ndarray, k: int, lr: float) -> np.ndarray:
    """One projected gradient step aimed at reducing the peak at index k."""
    n = seq.size
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(seq)
    grad[mask] = 2.0 * seq[idx[mask]]
    cand = seq - lr * grad
    np.clip(cand, 0.0, 1000.0, out=cand)
    total = cand.sum()
    if total > 0.0:
        cand /= total
    return cand


def _perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative block perturbation (keeps L1 mass ≈ 1)."""
    n = seq.size
    max_w = max(1, n // 10)                 # up to 10 % of length
    w = int(rng.integers(1, max_w + 1))
    s = int(rng.integers(0, n - w + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))   # ~7 % log‑normal change
    cand = seq.copy()
    cand[s : s + w] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    t = cand.sum()
    if t > 0.0:
        cand /= t
    return cand


def _highpass(seq: np.ndarray, damping: float = 0.9, cut_ratio: float = 0.05) -> np.ndarray:
    """Apply a simple low‑frequency damping in the DCT domain."""
    try:
        from scipy.fft import dct, idct
    except Exception:
        return seq.copy()      # scipy unavailable – skip this operator
    n = seq.shape[0]
    coeff = dct(seq, type=2, norm="ortho")
    cut = max(1, int(cut_ratio * n))
    coeff[:cut] *= damping
    out = idct(coeff, type=2, norm="ortho")
    np.clip(out, 0.0, 1000.0, out=out)
    s = out.sum()
    if s > 0.0:
        out /= s
    return out


def _score(seq: np.ndarray, n: int) -> float:
    """Internal proxy for the evaluator: 2·n·max(conv).  seq is L1‑normalised."""
    m, _ = _max_conv(seq)
    return 2.0 * n * m

    if total > 0.0:

    else:

    best = seq.copy()
    best_score = _score(best, n)

    max_iter = 800
