    import numpy as np
    rng=np.random.default_rng(seed)
    parent = globals().get('height_sequence_1')
    if isinstance(parent, np.ndarray) and parent.ndim==1 and parent.size>0:
        n=int(parent.shape[0])
        seq = parent.astype(np.float64).copy()
    else:
        n=256
        seq=generate_r_sequence(n).astype(np.float64)
    # Clip
    np.clip(seq, 0.0, 1000.0, out=seq)
    best = seq
    best_score = float(evaluate_sequence(best.tolist()))
    # Define tail indices: using np.arange(n)// maybe half
    tail_start = n//2
    tail_idx = np.arange(tail_start, n)
    tail_len = tail_idx.size
    # Deltas to try
    deltas = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for delta in deltas:
        improved = True
        while improved:
            # ensure we have enough a0 to give delta
            if best[0] < delta:
                break
            candidate = best.copy()
            candidate[0] -= delta
            inc = delta / tail_len
            candidate[tail_idx] += inc
            np.clip(candidate, 0.0, 1000.0, out=candidate)
            # recompute sum for sanity (should stay same)
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score
                # continue trying same delta again
            else:
                improved = False
    # fallback if not improved
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
    return [float(x) for x in best]

import numpy as np

def _generate_r_sequence(n: int) -> np.ndarray:
    """Return the extremal r‑sequence: r[i] = C(2i,i)/4**i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function that improves the AC1 bound.

    The routine starts from the provided parent sequence (if any) or from the
    classical r‑sequence, then repeatedly removes whole prefixes and redistributes
    their mass uniformly over the remaining positions.  This operation keeps the
    total sum unchanged while lowering the maximal autocorrelation value,
    consequently reducing the bound 2 n max/∑a².
    """
    # ----------------------------------------------------------------------
    # 1.  Initialise the sequence.
    # ----------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
        seq = parent.astype(np.float64).copy()
    else:
        n = 256
        seq = _generate_r_sequence(n)

    # enforce the problem limits
    np.clip(seq, 0.0, 1000.0, out=seq)

    # ----------------------------------------------------------------------
    # 2.  Record the current best candidate.
    # ----------------------------------------------------------------------
    best_seq = seq
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ----------------------------------------------------------------------
    # 3.  Greedy prefix‑removal moves.
    #
    #    For a set of prefix lengths p we zero out the first p entries and
    #    redistribute the removed mass uniformly over the remaining positions.
    #    The total sum stays constant, while the convolution peak is
    #    reduced because many large early contributions disappear.
    # ----------------------------------------------------------------------
    # Prefix lengths: powers of two up to roughly half the array.
    max_prefix = n // 2
    p = 1
    while p <= max_prefix:
        # copy the current best sequence and apply the transformation
        cand = best_seq.copy()
        prefix_mass = np.sum(cand[:p])
        if prefix_mass <= 0.0:
            # nothing to move – go to the next size
            p <<= 1
            continue

        # zero the prefix
        cand[:p] = 0.0

        # redistribute the removed mass uniformly over the suffix
        tail_len = n - p
        inc = prefix_mass / tail_len
        cand[p:] += inc

        # respect the absolute bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            # keep the improvement and try larger prefixes on the new sequence
        p <<= 1

    # ----------------------------------------------------------------------
    # 4.  A tiny local adjustment: move a small amount from the first
    #     non‑zero entry to the tail to see if a further improvement is
    #     possible.  The step size is deliberately tiny to avoid breaking
    #     the already‑good bound.
    # ----------------------------------------------------------------------
    epsilons = [1e-3, 5e-4, 2e-4, 1e-4]
    tail_start = n // 2
    tail_idx = np.arange(tail_start, n)
    tail_len = tail_idx.size

    for eps in epsilons:
        # find the first index with a positive value (normally index 0)
        src = np.argmax(best_seq > 0)
        if src == 0 and best_seq[0] < eps:
            continue
        if best_seq[src] < eps:
            continue

        cand = best_seq.copy()
        cand[src] -= eps
        cand[tail_idx] += eps / tail_len
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # ----------------------------------------------------------------------
    # 5.  Safety fallback (should never be needed).
    # ----------------------------------------------------------------------
    if not np.isfinite(best_score) or best_score >= 1e12:
        best_seq = np.ones(n, dtype=np.float64)

    # Return a plain Python list of floats, as required by the harness.
    return [float(x) for x in best_seq]
