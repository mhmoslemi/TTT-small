import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, exact for real non‑negative data)
def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full convolution arr*arr."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact evaluator (without the external wrapper)."""
    total = arr.sum()
    if total == 0.0:
        return float("inf")
    mx = _conv_max(arr)
    return float(2.0 * n * mx / (total * total))


# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p); optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


# ----------------------------------------------------------------------
def _build_array_from_masses(masses: np.ndarray, bucket_sizes: np.ndarray) -> np.ndarray:
    """Create the full length array from bucket masses."""
    heights = masses / bucket_sizes
    return np.repeat(heights, bucket_sizes)


def _masses_from_array(arr: np.ndarray, bucket_sizes: np.ndarray) -> np.ndarray:
    """Sum the array into bucket masses (preserves order)."""
    masses = np.empty(len(bucket_sizes), dtype=np.float64)
    start = 0
    for i, sz in enumerate(bucket_sizes):
        masses[i] = arr[start:start + sz].sum()
        start += sz
    return masses

            and np.isfinite(parent_seq).all():
        n = int(parent_seq.size)

    else:

    target_sum = np.sqrt(2.0 * n)                     # normalisation used by evaluator

    best_vec = np.full(n, target_sum / n, dtype=np.float64)   # uniform fallback

        if s > 0.0:

            if sc < best_score:

    p_vals = np.linspace(0.10, 1.00, 19)               # finer grid than before
