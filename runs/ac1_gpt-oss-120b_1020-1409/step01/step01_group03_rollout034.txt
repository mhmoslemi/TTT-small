import numpy as np

# ------------------------------------------------------------
# Helpers
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Truncate of a_i = C(2i,i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (avoids O(n^2) cost)."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound (mirrors evaluate_sequence)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    return 2.0 * seq.size * c.max() / (s * s)

# ------------------------------------------------------------

    """
    Generate a non‑negative height sequence that beats the target bound.
    The construction starts from the classic Catalan series,
    then applies a symmetric exponential rescaling that enlarges the
    tail while keeping the maximal self‑convolution unchanged (≈1).
    This dramatically raises the total sum and therefore lowers the AC1
    bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size from the provided parent if possible.
    # ------------------------------------------------------------------
    try:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    base = _make_series(n)

    # ------------------------------------------------------------------
    # Evaluate the three natural candidates: base, parent, and scaled.
    # ------------------------------------------------------------------
    best_seq = base.copy()
    best_score = _score(best_seq)

    if parent is not None:
        p = np.clip(parent, 0.0, 1000.0)
        p_score = _score(p)
        if p_score < best_score:

            best_score = p_score

    # ------------------------------------------------------------------
    # Pre‑compute the convolution of the base series (exactly 1 for
    # k=0..n‑1, decays for k>n‑1).
    # ------------------------------------------------------------------
    conv_base = _conv_self(base)                     # length 2*n-1
    k = np.arange(conv_base.size, dtype=np.float64)
    # exponent for scaling each convolution entry:
    # factor(k) = exp(gamma * 2 * (k/(n-1) - 1))
    exp_factor = 2.0 * (k / (n - 1) - 1.0)

    # ------------------------------------------------------------------
    # Determine the maximal admissible gamma so that all scaled
    # convolution entries stay ≤ 1 (the interior peak at k=n‑1 stays 1).
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # Ensure a non‑empty, finite result.
    # ------------------------------------------------------------------
