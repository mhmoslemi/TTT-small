    rng = np.random.default_rng(seed)
    parent = _parent_sequence()
    n = int(parent.size)

    best_score = float(evaluate_sequence(best.tolist()))
    arcsine = _arcsine_sequence(n)
    arcsine_score = float(evaluate_sequence(arcsine.tolist()))
    if arcsine_score < best_score:
        best = arcsine
        best_score = arcsine_score
    current = best.copy()
    # variables for stagnation detection
    last_improve_step = 0
    stagnation_limit = int(num_steps * 0.1)  # 10% of steps
    # parameters
    max_temp = 0.5
    min_temp = 0.001
    for step in range(num_steps):
        # compute convolution and max index
        conv = np.convolve(current, current)
        idx_max = int(np.argmax(conv))
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)

            i_high = idx_range[int(np.argmax(grad_weights))]

        # compute transfer capacity
        capacity = 1000.0 - current[i_low]

            # cannot transfer, maybe random small move

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return the parent height sequence if available, otherwise a fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    # fallback – small uniform vector that always satisfies the evaluator
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """Generate the truncated arcsine coefficients of length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _make_block_perturbation(base, rng):
    """Cheap O(n) random block perturbation for occasional jumps."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # occasional symmetry bias
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, num_steps=8000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 upper bound.
    Implements a deterministic high‑to‑low mass transfer guided by the maximal
    autocorrelation entry, together with occasional stochastic jumps.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = int(parent.size)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    arcsine = _arcsine_sequence(n)
    arcsine_score = float(evaluate_sequence(arcsine.tolist()))
    if arcsine_score < best_score:
        best = arcsine
        best_score = arcsine_score

    current = best.copy()

    # Stagnation handling
    last_improve_step = 0
    stagnation_limit = max(10, int(num_steps * 0.10))

    # Temperature schedule for simulated‑annealing acceptance
    max_temp = 0.5
    min_temp = 0.001

    # ------------------------------------------------------------------
    # Search loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # Compute full linear convolution and locate its maximal entry
        conv = np.convolve(current, current)
        idx_max = int(np.argmax(conv))

        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)

        # Gradient weights: a_{idx_max - i} for i in the contributing range
        grad_weights = current[idx_max - idx_range]

        # Choose the index with the largest contribution (i_high)
        i_high = idx_range[int(np.argmax(grad_weights))]

        # Build a full‑length weight vector where undefined entries are zero.
        # w[i] = a_{idx_max - i} if the pair contributes, else 0.
        w_all = np.zeros(n, dtype=np.float64)
        w_all[lo:hi + 1] = current[idx_max - np.arange(lo, hi + 1)]

        # Choose the index with the smallest contribution (i_low) – often
        # outside the contributing region, which is ideal.
        i_low = int(np.argmin(w_all))

        # Transfer capacity respecting the upper bound 1000
        capacity = 1000.0 - current[i_low]
        max_transfer = min(current[i_high], capacity)

        # If no transfer possible, fall back to a random small move
        if max_transfer <= 0.0:
            if rng.random() < 0.02:
                cand = _make_block_perturbation(current, rng)
                cand_score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(cand_score) and cand_score < best_score:
                    best, best_score, current = cand, cand_score, cand
                    last_improve_step = step
                continue
            else:
                continue

        # Decreasing step‑size factor (between 0.01 and ~0.21)
        factor = 0.2 * (1.0 - step / num_steps) + 0.01
        delta = factor * max_transfer

        # Apply the transfer
        cand = current.copy()
        cand[i_high] -= delta
        cand[i_low] += delta
        # (Clipping not strictly needed because of capacity, but keep for safety)
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score):
            if cand_score < best_score:
                best, best_score, current = cand, cand_score, cand
                last_improve_step = step
                continue
            # Simulated‑annealing acceptance for uphill moves
            temperature = max_temp * (1.0 - step / num_steps) + min_temp
            if np.exp((best_score - cand_score) / temperature) > rng.random():
                current = cand

        # Occasional global perturbation to escape plateaus
        if rng.random() < 0.01:
            cand2 = _make_block_perturbation(current, rng)
            cand2_score = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(cand2_score) and cand2_score < best_score:
                best, best_score, current = cand2, cand2_score, cand2
                last_improve_step = step

        # Restart if stagnation detected
        if step - last_improve_step > stagnation_limit:
            jitter = rng.normal(scale=0.05, size=n)
            cand_restart = best + jitter * best
            np.clip(cand_restart, 0.0, 1000.0, out=cand_restart)
            if np.sum(cand_restart) < 0.01:
                cand_restart = np.ones(n, dtype=np.float64)
            restart_score = float(evaluate_sequence(cand_restart.tolist()))
            if np.isfinite(restart_score):
                current = cand_restart
            last_improve_step = step

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
