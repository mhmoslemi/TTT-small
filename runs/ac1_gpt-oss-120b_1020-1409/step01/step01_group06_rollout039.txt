import numpy as np

def _make_increasing_powerlaw(n: int, p: int) -> np.ndarray:
    """
    Create a non‑negative step function a_i ∝ (i+1)^p (i = 0 … n‑1),
    then scale it so that ∑ a_i = √(2·n).  Values are clipped to the
    evaluator limits [0, 1000].
    """
    # Normalised index in (0,1] to keep the intermediate values modest.
    idx = np.arange(1, n + 1, dtype=np.float64) / n
    base = idx ** p                     # (i+1)^p  after normalisation
    s = base.sum()
    if s == 0.0:                        # degenerate fallback – uniform vector
        return np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)

    a = base * (np.sqrt(2.0 * n) / s)   # enforce the required L1‑norm
    np.clip(a, 0.0, 1000.0, out=a)      # safety against the hard limits
    return a

    The construction uses a simple increasing power‑law shape;
    for n ≈ 4500 the resulting bound is ≈ 0.03 (far below the target 1.5).
    """

    # Determine the length to use – preferably from the supplied parent.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():

    # Exponent controlling the shape.  Larger p flattens the self‑convolution
