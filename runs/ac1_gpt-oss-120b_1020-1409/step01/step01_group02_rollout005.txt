def _weight_vector(a, k):
    n = a.size

    mask = (idx >= 0) & (idx < n)
    w[mask] = a[idx[mask]]
    return w

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities

def _parent_sequence():
    """Return the supplied parent sequence if it is a valid 1‑D numeric array."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:
        pass
    # fallback – short arcsine seed (will be resized later if needed)
    return _arcsine_sequence(256)


def _arcsine_sequence(n):
    """Generate the classic arcsine coefficients of length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return np.clip(a, 0.0, 1000.0)


def _make_block_perturbation(base, rng):
    """Apply a cheap O(n) random scaling to a contiguous block of entries."""
    cand = base.copy()
    n = cand.size
    # width between 0.5 % and 25 % of the vector length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:               # occasional symmetrisation bias
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _weight_vector(a, k):
    """
    For a given lag k, return w_i = a_{k-i} when the index k-i is inside the
    support of a, otherwise 0.  This vector is proportional to the gradient of
    the convolution entry (a*a)_k with respect to a.
    """
    n = a.size
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    w = np.zeros_like(a)
    w[mask] = a[idx[mask]]
    return w
