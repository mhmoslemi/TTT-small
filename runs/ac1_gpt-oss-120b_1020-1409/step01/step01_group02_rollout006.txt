    """
    Generate the arcsine‑type decreasing sequence:
        a[0] = 1,

    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a

    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)
    return conv[: 2 * n - 1]
