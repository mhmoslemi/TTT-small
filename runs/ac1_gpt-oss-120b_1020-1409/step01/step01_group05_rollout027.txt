import numpy as np

def linear_conv(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)[:n]
    return conv

def correlation(w, a):
    # correlation w⋆a
    w_rev = w[::-1]
    return linear_conv(w_rev, a)

def gradient_smooth_max(a, p, eps=1e-12):
    conv = linear_conv(a, a)
    # we use exponent (p-1); conv may have zeros; ensure positivity
    w = np.maximum(conv, eps) ** (p-1)
    corr = correlation(w, a)
    n = len(a)
    # slice of length n starting at offset 2n-2
    start = 2 * n - 2
    grad = 2.0 * corr[start:start + n]
    return grad, conv

def propose_candidate(seed=42, num_iters=3000, **kwargs):

        n = 2000
        parent_ok = False

    # Helper: normalize vector to sum = 1
    def normalize(v):
        s = v.sum()
        if s <= 0 or not np.isfinite(s):
            return np.ones_like(v) / v.size
        return v / s

    # Generate list of initial candidates
    init_candidates = []

    if parent_ok:

    init_candidates.append(uniform)

    # Power-law (1/sqrt(i+1))
    power = 1.0 / np.sqrt(np.arange(1, n+1, dtype=np.float64))
    init_candidates.append(normalize(power))

    rand_exp = rng.exponential(scale=1.0, size=n)

def solve_lp(a, max_val, ks):
    n = len(a)
    rows = []
    cols = []
    data = []
    for r, k in enumerate(ks):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # j indexes row elements
        for j in range(low, high + 1):
            rows.append(r)
            cols.append(j)
            data.append(a[k - j])
    A = sp.csr_matrix((data, (rows, cols)), shape=(len(ks), n))
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= max_val]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=400, verbose=False)
    except Exception:
        return None
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    return b.value
```

def linear_conv(x, y):
    """
    Linear convolution using a real FFT.
    Returns a float ndarray of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


def _normalize(v):
    """Scale to sum == 1, safely handling degenerate inputs."""
    s = v.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _build_lp_matrix(a, ks):
    """
    Build a sparse matrix A such that (A @ b)[i] = (a * b)[ks[i]]
    for a fixed vector a and a list of convolution indices ks.
    """
    n = a.size
    rows, cols, data = [], [], []
    for row, k in enumerate(ks):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # index j runs over a, a[k-j] is the coefficient multiplying b_j
        for j in range(lo, hi + 1):
            rows.append(row)
            cols.append(j)
            data.append(a[k - j])
    return sp.csr_matrix((data, (rows, cols)), shape=(len(ks), n))
