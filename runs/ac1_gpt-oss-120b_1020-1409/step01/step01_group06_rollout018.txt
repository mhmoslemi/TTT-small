import numpy as np

# ----------------------------------------------------------------------
# Fixed‑size FFT convolution (full mode) returning the maximum entry.
def _conv_max(arr: np.ndarray, fft_size: int, n: int) -> float:
    """Maximum of the full convolution arr * arr using an FFT of given size."""
    fa = np.fft.rfft(arr, n=fft_size)
    conv = np.fft.irfft(fa * fa, n=fft_size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
# Score = 2 * n * max(conv) / (sum(arr) ** 2)  (identical to evaluate_sequence)
def _score(arr: np.ndarray, n: int, fft_size: int) -> float:
    a = np.clip(arr, 0.0, 1000.0)
    s = a.sum()
    if s == 0.0:
        return float("inf")
    m = _conv_max(a, fft_size, n)
    return float(2.0 * n * m / (s * s))


# ----------------------------------------------------------------------
def _block_perturb(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a small multiplicative perturbation to a random block."""
    cand = vec.copy()
    n = cand.size
    # block width: between 0.5% and 10% of the array
    lo = max(1, n // 200)
    hi = max(2, n // 10)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.04)))
    cand[start:start + width] *= factor
    # occasional global jitter (helps escape plateaus)
    if rng.random() < 0.25:
        cand *= np.exp(rng.normal(0.0, 0.02, size=n))
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
