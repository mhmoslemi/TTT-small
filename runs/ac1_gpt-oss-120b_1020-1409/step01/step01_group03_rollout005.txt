import numpy as np

# ------------------------------------------------------------
# Generate the classic Catalan‑type series a_i = C(2i,i)/4^i.
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Convolution of a vector with itself using a real FFT.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, n=fft_len)
    conv_full = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv_full


# ------------------------------------------------------------
# Exact AC1 bound (mirrors evaluate_sequence).
# ------------------------------------------------------------
def _score_seq(x: np.ndarray) -> float:
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float('inf')
    m = _conv_self(x).max()
    n = x.size
    return 2.0 * n * m / (s * s)

    averaging while keeping track of the best candidate seen.
    """
