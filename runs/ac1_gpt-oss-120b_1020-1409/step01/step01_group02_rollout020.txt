import numpy as np

def compute_score(arr, n=None):
    # assumes arr is 1D numpy array, sum>0
    if n is None: n = arr.size
    conv = np.convolve(arr, arr)
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=42, num_steps=12000, **kwargs):
    rng = np.random.default_rng(seed)

    # get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None

    # arcsine
    def arcsine_seq(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * (1.0 - 0.5 / i)
        return a

    if parent is not None:
        n = parent.size
    else:
        n = 1024  # fallback length (maybe moderate)
    # choose best initial
    best = arcsine_seq(n)
    best_score = compute_score(best, n)

    if parent is not None:
        parent_score = compute_score(parent, n)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # enforce symmetry on best
    best = 0.5 * (best + best[::-1])
    best = np.clip(best, 0.0, 1000.0)

    current = best.copy()
    current_score = best_score

    # Precompute step size and temperature schedule
    step_size = 0.2   # fraction of source magnitude to transfer
    # Temperature schedule: exponential decay
    # We'll compute temperature at each iteration: temp = max(0.01, 0.5 * (0.9995**step))

    for step in range(num_steps):
        # Recompute conv of current for gradient
        conv = np.convolve(current, current)
        idx_max = int(np.argmax(conv))
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        # compute gradient weights: contribution from each i = current[idx_max - i]
        partner_idx = idx_max - idx_range
        grad = current[partner_idx]  # shape same
        # weighting; choose i with probability proportional to grad
        if grad.sum() > 0 and rng.random() < 0.9:
            probs = grad / grad.sum()
            i = int(rng.choice(idx_range, p=probs))
        else:
            i = int(rng.integers(0, n))

        # partner j random
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # transfer delta
        max_transfer = current[i]  # can transfer up to whole mass
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer * step_size
        if delta <= 0:
            continue

        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta

        # symmetrize
        cand = 0.5 * (cand + cand[::-1])
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = compute_score(cand, n)

        # acceptance
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            current = cand
            current_score = cand_score
            continue
        # simulated annealing
        temperature = max(0.01, 0.5 * (0.9997 ** step))
        if np.exp((best_score - cand_score) / temperature) > rng.random():
            current = cand
            current_score = cand_score
            continue

        # occasional block perturbation
        if step % 2000 == 0 and step > 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 8)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.2)))
            block = current.copy()
            block[start:start+width] *= factor
            # symmetrize again
            block = 0.5 * (block + block[::-1])
            np.clip(block, 0.0, 1000.0, out=block)
            block_score = compute_score(block, n)
            if block_score < best_score:
                best = block
                best_score = block_score
                current = block
                current_score = block_score

    # final fallback: ensure non-empty, non-zero sum
    if best.size == 0 or best.sum() < 0.01:
        best = np.ones(256, dtype=np.float64)

    # return as list of floats
    return [float(x) for x in best]

    rng = np.random.default_rng(seed)

    parent = _parent_sequence()
    if parent is not None:
        n = parent.size
    else:
        n = 1024  # fallback length
    # initial best
    arcsine = _arcsine_sequence(n)
    best = arcsine
    best_score = _score(best)
    if parent is not None:
        parent_score = _score(parent)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    # Ensure best feasible: clip and nonneg
    best = np.clip(best, 0.0, 1000.0)
    # Current pointer
    current = best.copy()
    current_score = best_score

    # Main loop
    for step in range(num_steps):
        # compute convolution and extreme indices
        conv = np.convolve(current, current)
        idx_max = int(np.argmax(conv))
        idx_min = int(np.argmin(conv))

        # source range (max)
        lo_max = max(0, idx_max - (n - 1))
        hi_max = min(n - 1, idx_max)
        src_idx = np.arange(lo_max, hi_max + 1, dtype=int)
        partner_max = idx_max - src_idx
        src_weights = current[src_idx] * current[partner_max]
        if src_weights.sum() > 0 and rng.random() < 0.9:
            probs = src_weights / src_weights.sum()
            i = int(rng.choice(src_idx, p=probs))
        else:
            i = int(rng.integers(0, n))

        lo_min = max(0, idx_min - (n - 1))

def _score(seq):
    """
    Exact AC1 upper bound for a non‑negative sequence.
    Mirrors the logic of the provided evaluate_sequence.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=12000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the AC1 bound.
    The algorithm repeatedly flattens the autocorrelation by moving mass
    from indices that heavily contribute to the current maximal entry
    toward indices that contribute little to the minimal entry.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial candidate (parent or arcsine)
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    if parent is not None:
        n = parent.size
    else:
        n = 1024                     # fallback length if no parent is supplied

    # start from the better of arcsine and the parent (if any)
    best = _arcsine_sequence(n)
    best_score = _score(best)

    if parent is not None:
        parent_score = _score(parent)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    best = np.clip(best, 0.0, 1000.0)
    current = best.copy()
    current_score = best_score

    # ------------------------------------------------------------------
    # Stochastic optimisation loop
    # ------------------------------------------------------------------
    eps = 1e-12
    for step in range(num_steps):
        # full autocorrelation of the current state
        conv = np.convolve(current, current)
        idx_max = int(np.argmax(conv))
        idx_min = int(np.argmin(conv))

        # ---- source indices (those feeding the maximal entry) ----
        lo_max = max(0, idx_max - (n - 1))
        hi_max = min(n - 1, idx_max)
        src_idx = np.arange(lo_max, hi_max + 1, dtype=int)
        partner_max = idx_max - src_idx
        src_weights = current[src_idx] * current[partner_max]

        if src_weights.sum() > 0 and rng.random() < 0.9:
            probs = src_weights / src_weights.sum()
            i = int(rng.choice(src_idx, p=probs))
        else:
            i = int(rng.integers(0, n))

        # ---- destination indices (those feeding the minimal entry) ----
        lo_min = max(0, idx_min - (n - 1))
        hi_min = min(n - 1, idx_min)
        dst_idx = np.arange(lo_min, hi_min + 1, dtype=int)
        partner_min = idx_min - dst_idx
        dst_products = current[dst_idx] * current[partner_min]
        dst_weights = 1.0 / (dst_products + eps)

        if dst_weights.sum() > 0 and rng.random() < 0.9:
            probs = dst_weights / dst_weights.sum()
            j = int(rng.choice(dst_idx, p=probs))
        else:
            j = int(rng.integers(0, n))

        if i == j:
            continue

        # ---- transfer a fraction of mass from i to j ----
        max_transfer = current[i]
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * 0.3 * max_transfer   # up‑to 30 % of source mass
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _score(cand)

        # ---- keep the best ever found ----
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            current = cand
            current_score = cand_score
            continue

        # ---- simulated‑annealing acceptance for uphill moves ----
        temperature = max(0.01, 1.0 - step / num_steps)   # linear cooldown
        if np.exp((best_score - cand_score) / temperature) > rng.random():
            current = cand
            current_score = cand_score
            continue

        # ---- occasional block perturbation to jump out of plateaus ----
        if rng.random() < 0.02:
            width = int(rng.integers(max(1, n // 200), max(2, n // 10)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.2)))
            pert = current.copy()
            pert[start:start + width] *= factor
            np.clip(pert, 0.0, 1000.0, out=pert)
            pert_score = _score(pert)
            if pert_score < best_score:
                best = pert
                best_score = pert_score
                current = pert
                current_score = pert_score

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best.size == 0 or best.sum() < 0.01:
        best = np.ones(256, dtype=np.float64)

    # Return a plain Python list of floats
    return [float(x) for x in best]
