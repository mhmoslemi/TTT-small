    f = np.fft.rfft(vec, n=L)

    return conv[:conv_len]

import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _parent_sequence():
    """
    Return the provided parent sequence `height_sequence_1` if it exists and is
    valid, otherwise return None.
    """
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    return None


def _arcsine_sequence(n):
    """
    Generate the arcsine coefficients of length n:
        a_0 = 1,
        a_i = a_{i-1} * (1 - 0.5 / i)  for i >= 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _fft_full_convolution(vec, L, out_len):
    """
    Compute the linear self‑convolution of `vec` using a real FFT.
    The result length is `out_len` (= 2*len(vec)-1).
    """
    f = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(f * f, n=L)
    return conv[:out_len]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=5000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 upper bound.
    The algorithm starts from a good analytic seed (parent or arcsine) and
    performs a gradient‑guided stochastic local search with FFT‑accelerated
    convolutions.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    if parent is not None:
        init = parent
    else:
        init = _arcsine_sequence(256)

    n = init.size
    # Normalise to unit sum (the objective is scale‑invariant)
    total = init.sum()
    if total < 1e-12:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = np.clip(init / total, 0.0, 1000.0)

    # Best solution found so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Convolution parameters (FFT length is next power of two)
    conv_len = 2 * n - 1
    L = 1 << (conv_len - 1).bit_length()  # smallest power of two ≥ conv_len

    # Current convolution (FFT based) – used for gradient direction
    conv_cur = _fft_full_convolution(a, L, conv_len)

    eps = 1e-12

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # ---------- occasional global perturbation ----------
        if rng.random() < 0.004:
            cand = a + rng.normal(0.0, 0.1, size=n)
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = cand.sum()
            if s > eps:
                cand = cand / s
                conv_cand = _fft_full_convolution(cand, L, conv_len)
                approx = 2.0 * n * np.max(conv_cand)
                if approx < best_score:
                    real = float(evaluate_sequence(cand.tolist()))
                    if real < best_score:
                        best = cand.copy()
                        best_score = real
                    a = cand.copy()
                    conv_cur = conv_cand
            continue

        # ---------- locate maximal convolution entry ----------
        idx_max = int(np.argmax(conv_cur))

        # Indices that actually influence this maximal entry
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        idxs = np.arange(lo, hi + 1, dtype=np.int64)

        # ----- choose i (donor) ----- weighted by large contribution
        contrib_i = a[idx_max - idxs]                # a_{k0-i}
        if contrib_i.sum() > 0:
            probs_i = contrib_i / contrib_i.sum()
            i = int(rng.choice(idxs, p=probs_i))
        else:
            i = int(rng.integers(0, n))

        # ----- choose j (receiver) ----- weighted by small contribution
        # start with uniform weights, then downgrade indices with large a_{k0-j}
        weights_j = np.ones(n, dtype=np.float64)
        weights_j[idxs] = 1.0 / (a[idx_max - idxs] + eps)
        probs_j = weights_j / weights_j.sum()
        j = int(rng.choice(n, p=probs_j))
        # avoid trivial self‑transfer
        if j == i:
            j = (i + 1) % n

        # ----- perform a paired mass transfer -----
        if a[i] <= eps:
            continue
        delta_i = a[i] * rng.random() * 0.5               # up to 50 % of a[i]
        i2 = idx_max - i                                   # counterpart index
        delta_i2 = 0.0
        if i2 != i and a[i2] > eps:
            delta_i2 = a[i2] * rng.random() * 0.5

        cand = a.copy()
        cand[i] -= delta_i
        cand[i2] -= delta_i2
        cand[j] += delta_i + delta_i2
        np.clip(cand, 0.0, 1000.0, out=cand)   # safety projection

        # ---------- evaluate with cheap FFT bound ----------
        conv_cand = _fft_full_convolution(cand, L, conv_len)
        approx_score = 2.0 * n * np.max(conv_cand)

        # ---------- accept if it strictly improves the true score ----------
        if approx_score < best_score:
            real_score = float(evaluate_sequence(cand.tolist()))
            if real_score < best_score:
                best = cand.copy()
                best_score = real_score
                a = cand.copy()
                conv_cur = conv_cand
                continue

        # ---------- simulated‑annealing style acceptance ----------
        temp = 1.0 / (step + 1.0)
        if np.exp((best_score - approx_score) / temp) > rng.random():
            a = cand.copy()
            conv_cur = conv_cand

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64) / 256.0

    return [float(x) for x in best]
