def propose_candidate(seed=42, num_steps=20000, **kwargs):
    rng = np.random.default_rng(seed)

    # try to load parent
    parent = _parent_sequence()
    n = parent.size

    # arcsine init if better
    arcsine = _arcsine_sequence(n)
    # compute scores
    score_parent = float(evaluate_sequence(parent.tolist()))
    score_arcsine = float(evaluate_sequence(arcsine.tolist()))
    best = parent.copy()
    best_score = score_parent
    if score_arcsine < best_score:
        best = arcsine.copy()
        best_score = score_arcsine

    current = best.copy()

    # maybe keep best_freq scoring.
    for step in range(num_steps):
        # compute conv and parameters
        conv = np.convolve(current, current)
        idx_max = int(np.argmax(conv))
        max_val = conv[idx_max]
        sum_a = float(np.sum(current))

        lo = max(0, idx_max - (n - 1))
        hi = min(idx_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        # contributions array: for each i in idx_range, a[i] * a[idx_max - i]
        # we just need a[idx_max - idx_range] (the other factor)
        contribs = current[idx_max - idx_range]   # shape (len(idx_range),)

        # target = max_val / sum_a
        target = max_val / sum_a

        diff = contribs - target

        # Choose i_pos with largest positive diff, i_neg with most negative diff
        # if diff <= 0 for all, maybe break early (flat)
        max_diff = diff.max()
        min_diff = diff.min()
        if max_diff <= 0:
            # convergence
            break

        i_pos = int(idx_range[np.argmax(diff)])
        i_neg = int(idx_range[np.argmin(diff)])

        # compute delta: move some mass from i_pos to i_neg.
        # Determine max possible delta without making i_pos negative.
        max_transfer = current[i_pos]
        if max_transfer <= 0:
            continue
        # Choose delta = max_transfer * rng.random() * 0.5
        delta = max_transfer * rng.random() * 0.5
        if delta <= 0:
            continue

        cand = current.copy()
        cand[i_pos] -= delta
        cand[i_neg] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            current = cand
            continue

        # simulated annealing acceptance
        temperature = 1.0 / (step + 1.0)
        if np.isfinite(cand_score) and np.exp((best_score - cand_score) / temperature) > rng.random():
            current = cand

        # occasional random block perturb
        if step % 1000 == 0 and step > 0:
            cand2 = _make_block_perturbation(current, rng)
            cand2_score = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(cand2_score) and cand2_score < best_score:
                best = cand2
                best_score = cand2_score
                current = cand2
            else:
                # maybe accept occasionally
                temperature2 = 1.0 / (step + 1.0)
                if np.isfinite(cand2_score) and np.exp((best_score - cand2_score) / temperature2) > rng.random():
                    current = cand2

    # fallback
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]

def propose_candidate(seed=42, max_steps=8000, **kwargs):
    rng = np.random.default_rng(seed)
    # get parent
    parent = _parent_sequence()
    n = parent.size
    # arcsine baseline
    arcsine = _arcsine_sequence(n)
    # evaluate
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    arcsine_score = float(evaluate_sequence(arcsine.tolist()))
    if arcsine_score < best_score:
        best = arcsine.copy()
        best_score = arcsine_score
    current = best.copy()
    # main loop
    for step in range(max_steps):
        # compute convolution
        conv = np.convolve(current, current)
        idx_max = int(np.argmax(conv))
        max_val = conv[idx_max]
        sum_a = float(np.sum(current))
        # early exit if low bound reached ~1 (infeasible to beat)
        if best_score <= 1.001:  # optional break
            break
        # compute contributions at max index
        lo = max(0, idx_max - (n - 1))
        hi = min(idx_max, n - 1)
        idx_range = np.arange(lo, hi+1, dtype=int)
        # contributions of each i to b_{idx_max} (via partner index)
        contrib = current[idx_max - idx_range]   # length hi-lo+1
        target = max_val / sum_a
        diff = contrib - target
        # choose donor and recipient indices
        if diff.min() < 0:
            i_pos = int(idx_range[np.argmax(diff)])
            i_neg = int(idx_range[np.argmin(diff)])
        else:
            # all contributions >= target, pick donor as max contribution,
            # recipient as a random other index (prefer a small element)
            i_pos = int(idx_range[np.argmax(diff)])
            # pick index with smallest value overall (excluding i_pos)
            # compute small random candidate among low-value indices
            # To avoid O(n) each step, we can sample a few random indices and pick min.
            sample = rng.integers(0, n, size=5)
            # ensure not selecting i_pos
            sample = sample[sample != i_pos]
            if sample.size == 0:
                i_neg = (i_pos + 1) % n
            else:
                i_neg = int(sample[np.argmin(current[sample])])
        # compute delta
        max_transfer = current[i_pos]
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.5
        if delta <= 0:
            continue
        cand = current.copy()
        cand[i_pos] -= delta
        cand[i_neg] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = float(evaluate_sequence(cand.tolist()))
        # update best if improvement
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            current = cand
            continue
        # possibly accept worse move
        temperature = 1.0 / (step + 1.0)
        if np.isfinite(cand_score) and np.exp((best_score - cand_score) / temperature) > rng.random():
            current = cand

                current = block

    if best.size == 0:
        best = np.ones(256, dtype=np.float64)
    return [float(x) for x in best]

    """Return the provided parent sequence if it exists, else a fallback."""

    except Exception:  # pragma: no cover
        pass
    # fallback – short uniform vector that always satisfies the evaluator
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """
    Generate the truncated arcsine coefficients:
        a_0 = 1,
        a_i = a_{i-1} * (1 - 0.5 / i)  for i >= 1.
    These satisfy a*a = 1 on the first n convolution entries.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _make_block_perturbation(base, rng):
    """A cheap O(n) random block‑wise perturbation."""
    cand = base.copy()
    n = cand.size
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # occasional symmetry bias
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

                i_neg = (i_pos + 1) % n
            else:
                i_neg = int(samples[np.argmin(current[samples])])

        max_transfer = current[i_pos]
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.5
        if delta <= 0:
            continue

        cand = current.copy()
        cand[i_pos] -= delta
        cand[i_neg] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            current = cand
            continue

        # simulated‑annealing style acceptance for occasional uphill moves
        temperature = 1.0 / (step + 1.0)
        if np.isfinite(cand_score) and np.exp((best_score - cand_score) / temperature) > rng.random():
            current = cand

        # periodic block‑wise perturbation to escape flat plateaus
        if step % 2000 == 0 and step > 0:
            block = _make_block_perturbation(current, rng)
            block_score = float(evaluate_sequence(block.tolist()))
            if np.isfinite(block_score) and block_score < best_score:
                best = block
                best_score = block_score
                current = block
            elif np.isfinite(block_score) and np.exp((best_score - block_score) / temperature) > rng.random():
                current = block

    # ------------------------------------------------------------------
    # safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
