def propose_candidate(seed=42, **kwargs):

    Produce a non‑negative step sequence that minimises the AC1 upper bound.
    The construction uses an (almost) optimal exponentially decreasing shape.
    """
    rng = np.random.default_rng(seed)  # kept for possible extensions

    # ------------------------------------------------------------------
    # Determine the length n, using the provided parent if present.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.size)
        parent_ok = (
            parent_arr.ndim == 1
            and n > 0
            and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # ------------------------------------------------------------------
    # Analytic construction: a_i = a0 * q^i with q = n^{-1/(n-1)}.
    # This balances the largest convolution entry (a0^2) with the centre
    # entry, giving a provably tiny score.
    # ------------------------------------------------------------------
    if n > 1:
        # q = exp(-log(n)/(n-1)), p = 1 - q
        logn = np.log(float(n))
        q = np.exp(-logn / (n - 1))
    else:
        q = 0.5
    p = 1.0 - q

    # denominator = 1 - q**n   (use log for stability)
    # q**n = exp(log(q) * n) = exp(-logn * n/(n-1)) = n^{-n/(n-1)}
    q_n = np.exp(np.log(q) * n)
    denom = 1.0 - q_n
    if denom <= 0.0:
        denom = 1.0  # safety fallback

    a0 = p / denom
    # Build the exponential vector: a_i = a0 * q**i
    idx = np.arange(n, dtype=np.float64)
    a = a0 * np.exp(np.log(q) * idx)  # q ** idx
    np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    # Evaluate the analytic candidate.
    # ------------------------------------------------------------------
    best = a
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if parent_ok:
