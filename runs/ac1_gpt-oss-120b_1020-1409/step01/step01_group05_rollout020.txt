import numpy as np

def full_convolve(a):
    n = a.shape[0]
    nfft = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(fa * fa, nfft)[:2*n-1]
    return conv

def max_conv_gradient(a, conv):
    n = a.shape[0]
    k = int(np.argmax(conv))
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    idx = np.arange(low, high+1, dtype=np.int64)
    partner = k - idx
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[partner]
    return g

def compute_score(a):
    n = a.shape[0]
    conv = full_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

def _full_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution a*a using FFT (real signal)."""
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())          # next power of two ≥2n‑1
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[: 2 * n - 1]
    return conv


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the maximal entry of conv = a*a w.r.t. a.
    For the index k where conv attains its maximum, the gradient is
    g_j = 2·a_{k‑j} for those j with a valid partner, otherwise 0.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    idx = np.arange(low, high + 1, dtype=np.int64)
    partner = k - idx
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[partner]
    return g
