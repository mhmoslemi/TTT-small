        a[i] = a[i-1] * (1.0 - 0.5 / i)
    return a

def _powerlaw_sequence(n, beta, gamma):
    idx = np.arange(n, dtype=np.float64)
    seq = (idx + beta) ** (-gamma)
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq

def _make_block_perturbation(base, rng):
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= np.exp(rng.normal(0.0, 0.12))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _conv_fft(vec):
    n = vec.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fft = np.fft.rfft(vec, size)
    conv = np.fft.irfft(fft * fft, size)
    return conv[:2*n-1]

def _score_fft(vec):
    # compute objective 2 n max_b / sum_a^2
    n = vec.shape[0]
    conv = _conv_fft(vec)
    max_b = conv.max()
    sum_a = vec.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed=42, num_steps=2500, **kwargs):
    rng = np.random.default_rng(seed)
    parent = _parent_sequence()
    n = parent.shape[0]

    # start from best of several simple seeds
    best = parent.copy()
    best_score = _score_fft(best)
    # arcsine
    arcs = _arcsine_sequence(n)
    arcs_score = _score_fft(arcs)
    if arcs_score < best_score:
        best = arcs.copy()
        best_score = arcs_score
    # random powerlaw samples
    for _ in range(80):
        beta = rng.uniform(0.4, 3.0)
        gamma = rng.uniform(0.2, 0.8)
        cand = _powerlaw_sequence(n, beta, gamma)
        cand_score = _score_fft(cand)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

    current = best.copy()

    for step in range(num_steps):
        # occasional block perturb
        if rng.random() < 0.02:
            cand = _make_block_perturbation(current, rng)
            cand_score = _score_fft(cand)
            if np.isfinite(cand_score) and cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                current = cand.copy()
                continue

        conv = _conv_fft(current)
        max_idx = int(np.argmax(conv))

        # compute influence weights
        b = np.zeros_like(current)
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx_i = np.arange(lo, hi + 1)
        b[idx_i] = current[max_idx - idx_i]

        # choose donor (high influence) and receiver (low influence)
        max_b_val = b.max()
        min_b_val = b.min()
        donor_candidates = np.flatnonzero((b == max_b_val) & (current > 1e-12))
        receiver_candidates = np.flatnonzero((b == min_b_val) & (current < 1000.0))
        if donor_candidates.size and receiver_candidates.size:
            i = int(rng.choice(donor_candidates))
            j = int(rng.choice(receiver_candidates))
            if i == j:
                # fallback to random
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                while j == i:
                    j = int(rng.integers(0, n))
        else:
            # random move
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            while j == i:
                j = int(rng.integers(0, n))

        # amount to move
        max_transfer = current[i]
        delta = rng.random() * 0.5 * max_transfer
        if delta <= 0:
            continue
        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _score_fft(cand)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            current = cand.copy()
            continue

        # simulated annealing acceptance
        temperature = 0.5 / (step + 1.0)
        if np.isfinite(cand_score) and np.exp((best_score - cand_score) / temperature) > rng.random():
            current = cand.copy()

    # final verification using official evaluator (optional)
    # Ensure best not empty
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
