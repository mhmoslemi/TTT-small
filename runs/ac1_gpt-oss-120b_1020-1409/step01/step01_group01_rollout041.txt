import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the classic r‑sequence  r_i = C(2i,i)/4^i  for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step function that improves the AC1 bound.
    The method starts from the parent (if supplied) or the r‑sequence,
    removes the most restrictive first entry, and then greedily
    increases later entries while never letting the full convolution
    exceed 1.0.  The resulting sequence has a larger total sum and a
    smaller AC1 bound.
    """
    # -----------------------------------------------------------------
    # 1.  Initialise the sequence.

    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_r_sequence(n)

    np.clip(a, 0.0, 1000.0, out=a)

    # The first element of the r‑sequence forces every convolution entry to
    # be exactly 1, leaving no room for improvement.  Zero it to create slack.
    a[0] = 0.0

    # -----------------------------------------------------------------
    # 2.  Greedy filling under the fixed bound M_target = 1.
    # -----------------------------------------------------------------
    c = np.convolve(a, a)                  # full convolution, length 2n‑1
    M_target = 1.0                         # maximal allowed convolution value

    for j in range(n - 1, -1, -1):         # walk from the tail towards the front
        aj_old = a[j]
        if aj_old >= 1000.0:
            continue

        # Slice of the convolution that will change when a[j] grows.
        slice_c = c[j: j + n]              # view, length n
        mask = a > 0.0                      # indices that actually constrain the step

        # Linear constraints coming from already‑positive entries.
        diff = M_target - slice_c
        if np.any(mask) and np.any(diff[mask] <= 0.0):
            # some constraint already tight → nothing can be added here.
            max_inc = 0.0
        else:
            # candidate epsilon from linear constraints (if any positive entry exists)
            eps_linear = np.min(diff[mask] / a[mask]) if np.any(mask) else np.inf

            # Constraint from the self‑product when the current entry is zero.
            if aj_old == 0.0:

                eps_self = np.sqrt(diff_self) if diff_self > 0.0 else 0.0
                max_inc = min(eps_linear, eps_self)
            else:

        if max_inc <= 0.0 or not np.isfinite(max_inc):
