import numpy as np
import math
import random

def _bound_with_delta(delta, a, c, total_sum):
    """Return the AC1 bound after moving `delta` from a[0] to a[-1]
    and the updated convolution array."""
    n = a.shape[0]
    # copy the original convolution
    c_new = c.copy()
    # effect of decreasing a[0] by delta
    c_new[:n] -= 2.0 * delta * a
    # effect of increasing a[-1] by delta   (index n-1)
    c_new[n - 1:] += 2.0 * delta * a
    # quadratic (Δ*Δ) terms
    c_new[0] += delta * delta               # 2*0 index
    c_new[2 * n - 2] += delta * delta       # last index
    c_new[n - 1] -= 2.0 * delta * delta     # middle index
    # maximal convolution value
    m_val = float(np.max(c_new))
    # AC1 bound = 2 * n * max / (sum^2)
    bound = 2.0 * n * m_val / (total_sum * total_sum)
    return bound, c_new
