def compute_ac1(seq):
    # seq is np.ndarray
    # Clip to [0,1000] (avoid any rounding)
    np.clip(seq, 0.0, 1000.0, out=seq)
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = seq.size
    # compute convolution via FFT
    nfft = 1 << ((2*n-1).bit_length())
    fft = np.fft.rfft(seq, nfft)
    conv = np.fft.irfft(fft*fft, nfft)[:2*n-1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (total * total))
```

    arr = np.clip(seq, 0.0, 1000.0, out=np.empty_like(seq))
    total = arr.sum()

    conv = np.fft.irfft(fft*fft, nfft)[:2*n-1]

def _ac1_score(seq: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0, out=seq.copy())
    total = seq.sum()
    if total < 0.01:
        return float('inf')
    n = seq.size
    # next power‑of‑two for an efficient FFT
    nfft = 1 << ((2 * n - 1).bit_length())

    conv = np.fft.irfft(f * f, n=nfft)[: 2 * n - 1]
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (total * total))
