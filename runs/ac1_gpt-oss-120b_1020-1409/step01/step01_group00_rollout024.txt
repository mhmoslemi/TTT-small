def propose_candidate(seed=42, num_steps=3000, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        a0 = np.array(height_sequence_1, copy=True, dtype=np.float64)
        n = a0.shape[0]
    except Exception:
        n = 256
        a0 = _generate_binomial_sequence(n)  # define earlier
    candidate = a0.copy()
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    uniform = np.ones(n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
    # Simulated annealing
    early_end = n // 3
    late_start = 2 * n // 3
    temp0 = 0.1
    for step in range(num_steps):
        temp = temp0 * (0.999 ** step)
        if rng.random() < 0.7:
            # pair transfer early -> late
            i = rng.integers(0, early_end)
            j = rng.integers(late_start, n)
            if candidate[i] <= 0.0:
                continue
            delta = candidate[i] * rng.uniform(0.001, 0.2)
            new = candidate.copy()
            new[i] = max(0.0, new[i] - delta)
            new[j] = min(1000.0, new[j] + delta)
        else:
            # block scaling tail and early
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            # early block
            e_start = int(rng.integers(0, max(1, n - width - 1)))
            e_stop = e_start + width
            # late block
            l_start = int(rng.integers(late_start, max(1, n - width + 1)))
            l_stop = l_start + width
            new = candidate.copy()
            factor_early = float(np.exp(rng.normal(-0.05, 0.05)))
            factor_late = float(np.exp(rng.normal(0.05, 0.05)))
            new[e_start:e_stop] *= factor_early
            new[l_start:l_stop] *= factor_late
            # clip
            np.clip(new, 0.0, 1000.0, out=new)
        # evaluate
        sc = float(evaluate_sequence(new.tolist()))
        if np.isfinite(sc):
            # acceptance
            if sc < best_score or rng.random() < np.exp((best_score - sc) / max(temp, 1e-12)):
                candidate = new
                if sc < best_score:
                    best = new.copy()
                    best_score = sc
        # occasional random reset to best to avoid drift
        if step % 500 == 0 and rng.random() < 0.1:
            candidate = best.copy()
    return [float(v) for v in best]

import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i) / 4**i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a

def _score(seq: np.ndarray) -> float:
    """Compute the AC1 upper bound for a non‑negative sequence."""
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    n = seq.shape[0]
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _pair_transfer(arr: np.ndarray, rng: np.random.Generator,
                   early_len: int, late_start: int,
                   max_frac: float) -> np.ndarray:
    """Move a random fraction of mass from an early index to a late one."""
    i = rng.integers(0, early_len)
    j = rng.integers(late_start, arr.size)
    if arr[i] <= 0.0:
        return arr  # nothing to move
    frac = rng.uniform(0.001, max_frac if max_frac > 0.001 else 0.001)
    delta = arr[i] * frac
    new = arr.copy()
    new[i] = max(0.0, new[i] - delta)
    new[j] = min(1000.0, new[j] + delta)
    return new

def _block_scale(arr: np.ndarray, rng: np.random.Generator,
                 early_len: int, late_start: int,
                 sigma: float) -> np.ndarray:
    """Scale a random early block down and a random late block up."""
    n = arr.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 30)))
    # early block (within the early region)
    e_start = int(rng.integers(0, max(1, early_len - width + 1)))
    # late block (within the late region)
    l_start = int(rng.integers(late_start, max(1, n - width + 1)))
    factor_early = float(np.exp(rng.normal(-sigma, sigma)))
    factor_late = float(np.exp(rng.normal(sigma, sigma)))
    new = arr.copy()
    new[e_start:e_start + width] *= factor_early
    new[l_start:l_start + width] *= factor_late
    np.clip(new, 0.0, 1000.0, out=new)
    return new

def propose_candidate(seed: int = 42, num_steps: int = 4000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose
    AC1 upper bound is hoped to be lower than the previous 1.57085.
    The routine starts from the parent construction `height_sequence_1`
    when available, then performs a lightweight simulated‑annealing
    search using biased pair transfers (early → late) and block scaling.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the provided parent or fall back to the binomial.
    # ------------------------------------------------------------------
    try:
        candidate = np.array(height_sequence_1, copy=True, dtype=np.float64)
        n = candidate.shape[0]
    except Exception:
        n = 256
        candidate = _generate_binomial_sequence(n)
    np.clip(candidate, 0.0, 1000.0, out=candidate)

    best = candidate.copy()
    best_score = _score(best)

    # one simple fallback – uniform vector – in case the parent is bad
    uniform = np.ones(n, dtype=np.float64)
    uniform_score = _score(uniform)
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score
        candidate = best.copy()

    # ------------------------------------------------------------------
    # Parameters defining “early” and “late” parts of the sequence.
    # ------------------------------------------------------------------
    early_len = max(1, n // 12)          # size of the leading region
    late_start = n - early_len           # start of the trailing region

    # annealing schedule parameters
    init_temp = 0.05
    decay = 0.999

    for step in range(num_steps):
        temp = init_temp * (decay ** step)

        # ------------------------------------------------------------------
        # Choose a move type.
        # ------------------------------------------------------------------
        if rng.random() < 0.70:   # biased pair transfer dominates
            # decreasing maximal transfer fraction as the search proceeds
            max_frac = 0.5 * (1.0 - step / num_steps)
            new = _pair_transfer(candidate, rng,

        else:                     # occasional block scaling

            new = _block_scale(candidate, rng,
                               early_len, late_start, sigma)

        sc = _score(new)
        if not np.isfinite(sc):
            continue

        # ------------------------------------------------------------------
        # Simulated‑annealing acceptance.
        # ------------------------------------------------------------------
        improve = sc < best_score
        if improve or rng.random() < np.exp((best_score - sc) / max(temp, 1e-12)):
            candidate = new
            if improve:

                best_score = sc

        if (step % 400 == 0) and (rng.random() < 0.10):

    return [float(v) for v in best]
