def _inverse_sqrt_seq(n):

    """a_i = (i+1)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    return 1.0 / np.power(i + 1.0, p)

    return (1.0 / np.power(i + 1.0, p)) * np.exp(-alpha * i)
