def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    parent = _parent_sequence()
    n = int(parent.shape[0])
    arcsine = _arcsine_sequence(n)
    # Normalize to sum 1
    parent = parent / parent.sum()
    arcsine = arcsine / arcsine.sum()

    # evaluate
    def objective(seq):
        conv = np.convolve(seq, seq)
        return 2.0 * n * conv.max()
    best = parent if objective(parent) <= objective(arcsine) else arcsine
    best_score = objective(best)
    current = best.copy()
    current_score = best_score

    # loop
    for step in range(num_steps):
        # compute conv and max index
        conv = np.convolve(current, current)
        max_idx = int(conv.argmax())
        max_val = float(conv[max_idx])
        # compute partner values
        idx = np.arange(n)
        partner_idx = max_idx - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        neighbor = np.zeros(n, dtype=np.float64)
        neighbor[mask] = current[partner_idx[mask]]
        # weight for i selection proportional to contribution (neighbor * current)
        contrib = neighbor * current
        if contrib.sum() > 0:

        else:
            i = rng.integers(n)

        if i == j:

import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return the provided parent sequence if it exists, else a safe fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.sum() > 0.0:
                return arr
    except Exception:  # pragma: no cover
        pass
    # fallback – uniform vector of moderate length
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """Generate the classic arcsine coefficients of length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _objective(seq, n):
    """Exact objective 2*n*max(conv(seq,seq)) for a unit‑sum sequence."""
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 upper bound.
    Returns a list of coefficients (floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise best candidate
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = int(parent.shape[0])

    arcsine = _arcsine_sequence(n)

    # normalise to unit L1 norm (objective is scale‑invariant)
    parent = parent / parent.sum()
    arcsine = arcsine / arcsine.sum()

    # pick the better of the two
    parent_score = _objective(parent, n)
    arcsine_score = _objective(arcsine, n)

    if arcsine_score < parent_score:
        best = arcsine.copy()
        best_score = arcsine_score
    else:
        best = parent.copy()
        best_score = parent_score

    current = best.copy()
    current_score = best_score

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # full autocorrelation and its maximal index
        conv = np.convolve(current, current)
        max_idx = int(conv.argmax())

        # partner values a_{max_idx - i} (zero outside range)
        idx = np.arange(n)
        partner_idx = max_idx - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        partner = np.zeros(n, dtype=np.float64)
        partner[mask] = current[partner_idx[mask]]

        # contribution of each coordinate to the maximal entry
        contrib = current * partner

        # pick i with largest contribution (most responsible for the peak)
        i = int(np.argmax(contrib))

        # pick j with smallest partner value (least responsible for the peak)
        # mask out i to avoid picking the same index
        partner_masked = np.where(idx == i, np.inf, partner)
        j = int(np.argmin(partner_masked))

        # move a modest fraction of mass from i to j
        max_transfer = 0.5 * current[i]                # never empty i completely
        if max_transfer <= 0.0:
            continue
        # decreasing step factor to fine‑tune later in the run
        step_factor = 0.2 * (1.0 - step / num_steps) + 0.01
        delta = step_factor * max_transfer * rng.random()
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta

        # occasional symmetrisation (helps for many optimal shapes)
        if rng.random() < 0.02:
            cand = 0.5 * (cand + cand[::-1])

        # random global perturbation every few hundred iterations
        if (step % 250 == 0) and (rng.random() < 0.15):
            rnd = rng.random(n)
            rnd = rnd / rnd.sum()
            cand = 0.7 * cand + 0.3 * rnd

        # re‑normalise to unit L1 norm
        cand = cand / cand.sum()

        # evaluate candidate (single convolution)
        cand_score = _objective(cand, n)

        # update best ever found
        if cand_score < best_score:
            best = cand.copy()
            best_score = float(cand_score)

        # simple greedy + tiny probabilistic acceptance for current state
        if (cand_score < current_score) or (rng.random() < 0.001):
            current = cand
            current_score = float(cand_score)

    # ------------------------------------------------------------------
    # post‑processing and safe fallback
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64) / 256.0
    else:
        # enforce (optional) symmetry before return
        best = 0.5 * (best + best[::-1])
        best = best / best.sum()

    return [float(x) for x in best]
