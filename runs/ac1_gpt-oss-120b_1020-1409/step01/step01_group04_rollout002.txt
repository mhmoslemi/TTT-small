def conv_max(seq):
    n = len(seq)
    # choose size = 2*n-1
    m = 1 << ((2*n-1).bit_length())
    # compute FFT
    fft_a = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)
    # conv now length m, but actual needed first 2*n-1 entries.
    conv = conv[:2*n-1]
    max_val = np.max(conv)
    return max_val

import numpy as np

def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a non‑negative sequence using FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)
    return conv[: 2 * n - 1]


def _fast_score(seq: np.ndarray) -> float:
    """Evaluate the AC1 bound without calling the external scorer."""
    seq = np.asarray(seq, dtype=np.float64)
    s = np.sum(seq)
    if s == 0.0:
        return np.inf
    n = seq.shape[0]
    m = np.max(_conv_full(seq))
    return float(2.0 * n * m / (s * s))

            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor

        elif move_type == 1:              # pairwise mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            delta = cur_seq[i] * rng.random() * transfer_frac
            cand[i] -= delta
            cand[j] += delta

        elif move_type == 2:              # peak shrink & redistribution
            idx = int(np.argmax(cand))
            shrink = float(rng.uniform(*shrink_range))
            removed = cand[idx] * (1.0 - shrink)
            cand[idx] *= shrink
            k = int(rng.integers(1, 5))
            rec = rng.choice(n, size=k, replace=False)
            cand[rec] += removed / k

        else:                             # targeted reduction of the most
            # compute the current full convolution only for this move
            conv_full = _conv_full(cur_seq)
            k_max = int(np.argmax(conv_full))
            idx_arr = k_max - np.arange(n)
            mask = (idx_arr >= 0) & (idx_arr < n)
            if np.any(mask):
                prod = np.zeros(n, dtype=np.float64)
                prod[mask] = cur_seq[mask] * cur_seq[idx_arr[mask]]
                i_max = int(np.argmax(prod))
                delta = cur_seq[i_max] * rng.random() * 0.2
                cand[i_max] -= delta
                j = int(rng.integers(0, n - 1))
                if j >= i_max:
                    j += 1
                cand[j] += delta

        # occasional weak symmetrisation
        if rng.random() < 0.04:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _fast_score(cand)

        # Acceptance (Metropolis)
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur_seq = cand
            cur_score = cand_score

        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score

        temp *= decay

    # ------------------------------------------------------------------
    # Fallback sanity check.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        try:
            best_seq = np.clip(height_sequence_1, 0.0, 1000.0)
        except Exception:
            best_seq = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_seq]
