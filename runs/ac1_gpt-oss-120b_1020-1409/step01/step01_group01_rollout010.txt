import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Generate the r‑sequence  a_i = C(2i,i)/4^i  for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def _score(seq: np.ndarray) -> float:
    """Convenient wrapper around the supplied evaluator."""
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function that (heuristically) minimizes the AC1 bound."""
    # ----------------------------------------------------------------------
    # Determine problem size n
    # ----------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256                     # safe fallback length

    # ----------------------------------------------------------------------
    # Base candidate – the r‑sequence (clipped to the allowed range)
    # ----------------------------------------------------------------------
    base = generate_r_sequence(n)
    np.clip(base, 0.0, 1000.0, out=base)

    best = base.copy()
    best_score = _score(best)

    # ----------------------------------------------------------------------
    # If a parent construction exists and is better, start from it
    # ----------------------------------------------------------------------
    if isinstance(parent, np.ndarray):
        try:
            parent_score = _score(parent)
            if parent_score < best_score:
                best = parent.astype(np.float64).copy()
                best_score = parent_score
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # Coarse search: scale an early block (indices 1..K) by a factor f
    # ----------------------------------------------------------------------
    # Choose a handful of block sizes (log‑spaced)
    ratios = [0.001, 0.0025, 0.005, 0.01, 0.025, 0.05, 0.1, 0.2]
    K_vals = []
    for r in ratios:
        K = int(n * r)
        K = max(1, min(K, n - 1))
        K_vals.append(K)
    K_vals = sorted(set(K_vals))

    # Coarse factor grid (1 → 2)
    factor_grid = np.linspace(1.0, 2.0, num=9)   # 1.0, 1.125, …, 2.0

    best_K = 0
    best_f = 1.0

    for K in K_vals:
        start, end = 1, min(K + 1, n)            # slice 1…K inclusive
        for f in factor_grid:
            cand = best.copy()
            cand[start:end] *= f
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                best_K = K
                best_f = f

    # ----------------------------------------------------------------------
    # Fine search around the best (K, f) found above
    # ----------------------------------------------------------------------
    if best_K > 0:
        # Refine K (±10 % around the best value)
        delta_K = max(1, int(best_K * 0.1))
        K_fine = list({max(1, best_K - delta_K),
                       best_K,
                       min(n - 1, best_K + delta_K)})
        # Refine factor (±10 % around the best factor)
        f_low = max(1.0, best_f * 0.9)
        f_high = best_f * 1.1
        factor_fine = np.linspace(f_low, f_high, num=11)

        for K in K_fine:
            start, end = 1, min(K + 1, n)
            for f in factor_fine:
                cand = best.copy()
                cand[start:end] *= f
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _score(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc
                    best_K = K
                    best_f = f

    # ----------------------------------------------------------------------
    # Small additive tweaks on the (now) leading block
    # ----------------------------------------------------------------------
    if best_K > 0:
        epsilons = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        for eps in epsilons:
            cand = best.copy()
            cand[1:best_K + 1] += eps
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if sc < best_score:
                best = cand
                best_score = sc

    # ----------------------------------------------------------------------
    # Fallback sanity check
    # ----------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    # Return a plain Python list as required by the harness
    return [float(x) for x in best]
