import numpy as np

def propose_candidate(seed=42, num_iters=2500, top_k=7, lam=0.2, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the AC1 upper bound.
    Uses a multi‑lag sub‑gradient descent (top‑k maximal convolution entries)
    combined with occasional random block perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Detect parent sequence if provided (height_sequence_1)
    # --------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.size)
        parent_ok = (
            parent_arr.ndim == 1
            and n > 0
            and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_ok = False
        n = 2000

    # --------------------------------------------------------------
    # Initial candidate
    # --------------------------------------------------------------
    if parent_ok:
        a = np.clip(parent_arr, 0.0, 1000.0)
        s = a.sum()
        if s > 0.0 and np.isfinite(s):
            a = a / s
        else:
            a = np.ones(n, dtype=np.float64) / n
    else:
        # decreasing 1/sqrt(i) shape, normalised to sum = 1
        a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.ones(n, dtype=np.float64) / n
        else:
            a = a / s

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------
    # Parameters for the optimisation
    # --------------------------------------------------------------
    step0 = 0.38          # initial step size for multiplicative update
    decay = 0.998         # exponential decay per iteration
    block_prob = 0.25     # probability to apply a random block perturbation
    block_min = max(1, n // 300)
    block_max = max(2, n // 20)

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    for it in range(num_iters):
        # Full linear convolution (length 2*n-1)
        conv = np.convolve(a, a)

        # ----- Gradient from the top‑k maximal convolution entries -----
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(-conv, top_k - 1)[:top_k]

        g = np.zeros_like(a)
        for lag in top_idx:
            low = max(0, lag - (n - 1))
            high = min(lag, n - 1)
            idx = np.arange(low, high + 1, dtype=np.int64)
            partner = lag - idx
            g[idx] += 2.0 * a[partner]

        # ----- Optional contribution from the minimal entry (to raise it) -----
        if lam > 0.0:
            min_idx = int(np.argmin(conv))
            low = max(0, min_idx - (n - 1))
            high = min(min_idx, n - 1)
            idx = np.arange(low, high + 1, dtype=np.int64)
            partner = min_idx - idx
            g_min = np.zeros_like(a)
            g_min[idx] = 2.0 * a[partner]

        # ----- Multiplicative update -----
        step = step0 * (decay ** it)
        a = a * np.exp(-step * g)

        # ----- Renormalise to keep sum fixed (scale‑invariant objective) -----
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.ones(n, dtype=np.float64) / n
        else:
            a = a / s

        # ----- Random block perturbation (escape local minima) -----
        if (it % 250 == 0) and rng.random() < block_prob:
            width = int(rng.integers(block_min, block_max))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width

            best_score = cur_score

    # --------------------------------------------------------------
    if best_score == float('inf') or not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64) / n

    # Return plain Python floats
    return [float(x) for x in best]
