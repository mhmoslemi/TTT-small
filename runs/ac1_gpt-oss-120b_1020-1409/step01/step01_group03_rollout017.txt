import numpy as np

# ------------------------------------------------------------
# Helper: power‑law series a_i = (i+1)^{-p}
# ------------------------------------------------------------
def _power_series(length: int, p: float) -> np.ndarray:
    """Return a length‑`length` array with entries (i+1)^{-p}."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    return idx ** (-p)


# ------------------------------------------------------------
# Fast self‑convolution via FFT (real input)
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# One projected gradient step (same sub‑gradient as in the
# reference implementation, but with a modest line‑search).
# ------------------------------------------------------------
def _grad_step(x: np.ndarray) -> tuple[np.ndarray, float]:
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    k = int(conv.argmax())

    # sub‑gradient of the max‑convolution at the active index k
    g = np.zeros_like(x)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    g[idx] = x[k - idx]

    S = x.sum()
    if S == 0.0:
        return x, float("inf")

    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

    # simple back‑tracking line search
    lr = 1.0
    best_local = x
    best_score = evaluate_sequence(x.tolist())
    for _ in range(5):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_local = cand
            best_score = sc
            break
        lr *= 0.5
    return best_local, best_score

    """

    except Exception:

    # --------------------------------------------------------
