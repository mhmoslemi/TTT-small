    for p in p_grid:

            idx = np.arange(n, dtype=np.float64) + shift

            cand_score = score(seq.tolist())
            if np.isfinite(cand_score) and cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()

    # Local refinement: random multiplicative perturbations around best sequence
    # Use a modest number of iterations to improve
    steps = 200
    for _ in range(steps):
        # create perturbation scale factor per block
        cand = best_seq.copy()
        # choose block width as fraction of n
        max_width = max(1, n // 200)  # 0.5%?
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))

        # clip
        np.clip(cand, 0.0, 1000.0, out=cand)
        # ensure sum>0
        total = cand.sum()
        if total < 0.01:
            cand = cand * (0.01 / total)
        cand_score = score(cand.tolist())
        if np.isfinite(cand_score) and cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
    # final fallback if best_seq is None
    if best_seq is None or not best_seq.size:

    # Ensure final list valid
    result = best_seq.tolist()
    # Defensive verify
    if not result or any([not (isinstance(x, (int, float)) and (not np.isnan(x)) and (not np.isinf(x))) for x in result]):

    if total < 0.01:                           # avoid division‑by‑zero in the scorer
        seq = seq * (0.01 / total)
    return seq


def _score(seq: np.ndarray) -> float:
    """
    Wrapper around the supplied evaluator – returns a Python float.
    """
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:                         # any unexpected error → infinite score
        return float("inf")


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Search for a coefficient list whose AC1 upper bound is as low as possible.
    The routine explores a family a_i = (i + shift)^{-p} (p ≈ 0.5–0.7) with a
    compact local refinement.  It always returns a non‑empty, finite list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the sequence (fallback to 256 if unavailable)
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1.  Parent construction (if present) – simple baseline
    # --------------------------------------------------------------
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        np.clip(parent, 0.0, 1000.0, out=parent)
        sc = _score(parent)
        if np.isfinite(sc) and sc < best_score:
            best_score, best_seq = sc, parent.copy()
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2. Uniform sequence – known score = 2 (kept for reference)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    sc = _score(uniform)
    if np.isfinite(sc) and sc < best_score:
        best_score, best_seq = sc, uniform.copy()

    # --------------------------------------------------------------
    # 3.  Power‑law family search
    # --------------------------------------------------------------
    # coarse sweep
    p_vals = np.arange(0.45, 0.71, 0.01)      # 0.45 … 0.70
    shift_vals = [1.0]                       # shift = 1 reproduces the classic inverse‑sqrt
    for p in p_vals:
        for sh in shift_vals:
            cand = _make_power_sequence(p, sh, n)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score, best_seq, best_p, best_shift = sc, cand.copy(), p, sh

    # fine sweep around the best exponent discovered
    if best_score < float("inf"):
        p_center = best_p
        p_fine = np.arange(max(0.1, p_center - 0.02), p_center + 0.021, 0.002)
        for p in p_fine:
            cand = _make_power_sequence(p, best_shift, n)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score, best_seq, best_p = sc, cand.copy(), p

    # --------------------------------------------------------------
    # 4.  Small shift variations (optional, cheap)
    # --------------------------------------------------------------
    if best_score < float("inf"):
        for sh in [0.8, 0.9, 1.1, 1.2, 1.5]:
            cand = _make_power_sequence(best_p, sh, n)
            sc = _score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score, best_seq, best_shift = sc, cand.copy(), sh

    # --------------------------------------------------------------
    # 5.  Local stochastic refinement (tiny multiplicative blocks)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = _score(best_seq)

    steps = 200
    max_width = max(1, n // 200)                     # at most 0.5 % of the length
    for _ in range(steps):
        cand = best_seq.copy()
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.015)))   # ~1.5 % log‑normal change
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score, best_seq = sc, cand.copy()

    # --------------------------------------------------------------
    # 6. Defensive final check – guarantee a valid non‑empty list
    # --------------------------------------------------------------
    result = best_seq.tolist()
    if (not result or
        any((not isinstance(x, (int, float)) or np.isnan(x) or np.isinf(x)) for x in result)):
        result = [1.0 / n] * n

    return result
