import numpy as np

def _scale_and_clip(arr: np.ndarray, target_sum: float) -> np.ndarray | None:
    """Scale ``arr`` so that its L1‐norm equals ``target_sum`` and clip to [0,1000]."""
    s = float(np.sum(arr))
    if s == 0.0:
        return None
    scale = target_sum / s
    arr = arr * scale
    np.clip(arr, 0.0, 1000.0, out=arr)
    return arr


def _bound(arr: np.ndarray) -> float:
    """Return the AC1 upper bound for a NumPy array."""
    return float(evaluate_sequence(arr.tolist()))


def _make_candidate(raw: np.ndarray, target_sum: float):
    """
    Build a feasible candidate from a raw (unscaled) array.
    Returns a tuple (candidate_array, bound) or (None, inf) on failure.
    """
    arr = _scale_and_clip(raw.astype(np.float64, copy=False), target_sum)
    if arr is None:
        return None, np.inf
    return arr, _bound(arr)
