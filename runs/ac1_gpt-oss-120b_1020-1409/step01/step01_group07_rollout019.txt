import numpy as np

def _sidon_sequence(n, rng, min_size=122, max_trials=30):
    """
    Construct a Sidon (sum‑distinct) set of indices within [0, n).
    Returns a NumPy array of length n with ones at the selected indices
    and zeros elsewhere.  The set size is maximised by trying several
    random orders; the first sequence with at least `min_size` elements
    is returned, otherwise the best found.
    """
    best_seq = None
    best_m = 0

    for _ in range(max_trials):
        order = rng.permutation(n)                # random order of candidates
        selected = []                             # chosen indices
        used = set()                              # occupied sums i+j

        for idx in order:
            # check whether any sum would clash
            clash = False
            for s in selected:
                if idx + s in used:
                    clash = True
                    break
            if clash or (idx + idx) in used:
                continue

            # accept idx
            selected.append(idx)
            # register all new sums (including idx+idx)
            for s in selected:
                used.add(idx + s)

        m = len(selected)
        if m > best_m:
            best_m = m
            seq = np.zeros(n, dtype=np.float64)
            seq[selected] = 1.0
            best_seq = seq
            if m >= min_size:
                break                     # good enough – stop early

    return best_seq, best_m

        score = float('inf')

    # Defensive fallback – guarantee a valid non‑empty list.
    if not np.isfinite(score) or len(cand_seq) == 0:
        cand_seq = np.full(n, 1.0 / n, dtype=np.float64)

    return cand_seq.tolist()
