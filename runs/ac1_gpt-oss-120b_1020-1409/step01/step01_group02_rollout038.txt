def _block_perturbation(a, rng):
    n = a.shape[0]
    cand = a.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    conv = np.convolve(cand, cand)
    return cand, conv
```

import numpy as np

# ----------------------------------------------------------------------

def _arcsine_sequence(n):
    """Generate the arcsine sequence a_0=1, a_i = a_{i-1}*(1-0.5/i)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _parent_sequence():
    """Return the provided parent if available, otherwise a short arcsine fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    # fallback – short arcsine vector
    return _arcsine_sequence(256)


def _block_perturbation(seq, rng):
    """Create a cheap random block perturbation of `seq` and its full convolution."""
    n = seq.shape[0]
    cand = seq.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetry bias
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    conv = np.convolve(cand, cand)
    return cand, conv

    # ------------------------------------------------------------------
