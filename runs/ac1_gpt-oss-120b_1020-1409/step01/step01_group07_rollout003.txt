def compute_objective(arr):
    n = arr.shape[0]
    # compute convolution via FFT for speed
    m = 1 << ((2*n - 1).bit_length())
    # use rfft to handle real
    fft = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a == 0:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

import numpy as np

def _max_conv_obj(seq):
    n = seq.shape[0]
    # compute linear convolution via FFT
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[:2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a == 0.0:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def _shape_decreasing(n, eps):
    r = 1.0 - eps
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx, dtype=np.float64)
    total = seq.sum()
    if total == 0.0:
        return None
    return seq / total

def _shape_symmetric(n, eps):
    r = 1.0 - eps
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    seq = np.power(r, dist, dtype=np.float64)
    total = seq.sum()
    if total == 0.0:
        return None
    return seq / total

def _shape_dual(n, eps):
    r = 1.0 - eps
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx, dtype=np.float64) + np.power(r, n - 1 - idx, dtype=np.float64)
    total = seq.sum()
    if total == 0.0:
        return None
    return seq / total

import numpy as np

def _max_conv_obj(seq: np.ndarray) -> float:
    """Fast objective 2·n·max(conv(seq,seq))/sum(seq)^2 using FFT."""
    n = seq.shape[0]
    # length for linear convolution (next power of two)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_b = conv.max()
    s = seq.sum()
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)
