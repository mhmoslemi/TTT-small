    rng = np.random.default_rng(seed)

    # determine n and parent

        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # grid search
    p_vals = np.array([0.55, 0.6, 0.65, 0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 0.98])
    alpha_vals = np.array([0.5,1.0,2.0,3.0,5.0,7.0,10.0])
    for p in p_vals:
        for alpha in alpha_vals:
            seq = 1.0 / np.power(np.arange(n, dtype=np.float64) + alpha, p)
            seq = np.clip(seq, 0.0, 1000.0)
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score

    # random search
    for _ in range(2000):
        p = 0.5 + rng.random() * 0.49  # (0.5,0.99)
        # draw alpha log-uniform between 0.5 and 10
        log_alpha = np.log(0.5) + rng.random() * (np.log(10.0) - np.log(0.5))
        alpha = np.exp(log_alpha)
        seq = 1.0 / np.power(np.arange(n, dtype=np.float64) + alpha, p)
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best = seq
            best_score = score

    # local refinement: block multiplicative
    min_width = max(1, n // 200)
    max_width = max(2, n // 20)
    sigma = 0.03
    for _ in range(1000):
        cand = best.copy()
        width = int(rng.integers(min_width, max_width+1))
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, sigma)))
        cand[start:start+width] *= factor
        # option random symmetry enforcement
        if rng.random() < 0.05:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # final sanity check
    if not np.isfinite(best_score) or best.size == 0:
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]

import numpy as np
from scipy.optimize import linprog

def _power_seq(alpha, n):
    """a_i = 1/sqrt(i+alpha)  (clipped to allowed range)."""
    seq = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + alpha)
    return np.clip(seq, 0.0, 1000.0)


def _conv_max(arr):
    """Maximum of the full convolution of arr with itself."""
    return np.max(np.convolve(arr, arr))


def _build_A(f, idxs):
    """
    Build a dense constraint matrix A such that (f * g)[k] = A[k]·g.
    Only rows for indices in idxs are constructed.
    """
    n = len(f)
    m = len(idxs)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over entries of f that overlap with g at shift k
        for i in range(i_start, i_end + 1):
            j = k - i               # column index for g
            A[row, j] = f[i]
    return A

    """
    rng = np.random.default_rng(seed)

    # start from a reasonably good power law
    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    alphas = np.logspace(-5, 1, num=30, dtype=np.float64)
    for a in alphas:

        sc = float(evaluate_sequence(cand.tolist()))

    f = best.copy()

        A = _build_A(f, active_idx)

        c = -np.ones(n, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    min_width = max(1, n // 250)      # ~0.4 % of the length
