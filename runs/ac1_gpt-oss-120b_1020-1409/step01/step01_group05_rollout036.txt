import numpy as np

def _top_gradient(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """Sum of gradients of conv[k] for all k in ks (ordered‑pair derivative)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in ks:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=np.int64)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    return g
