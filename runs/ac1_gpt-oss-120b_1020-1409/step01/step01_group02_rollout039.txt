    n = a.size

            elif sc2 < cur_sc:
                current = cand2.copy()

    # ------------------------------------------------------------------
    # final safety check – evaluate the best candidate via the official
    # evaluator (in case of tiny numerical differences)
    # ------------------------------------------------------------------
    final_score = _evaluate_seq(best)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    # fallback: never return an empty list
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
