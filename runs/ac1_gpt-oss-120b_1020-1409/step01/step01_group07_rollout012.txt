def simplex_projection(v):
    # v: 1D numpy array
    # returns projection onto simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive after shift, return uniform?
        # But should not happen.
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np

def _simplex_projection(v):
    # Project points onto the probability simplex {x >= 0, sum(x)=1}
    # v: 1d numpy array
    # returns projection
    # reference: efficient algorithm (by Duchi et al.)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are negative or zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w

def _conv_full(seq, fft_len):
    # Compute full linear convolution seq * seq using FFT of length fft_len
    s_fft = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(s_fft * s_fft, n=fft_len)
    m = 2 * seq.size - 1
    return conv[:m]

    rng = np.random.default_rng(seed)

    # Determine length and initial vector

    except Exception:

    # Clip to evaluator bounds and normalise to a probability simplex
    cur = np.clip(cur, 0.0, 1000.0)
    s = cur.sum()
    if s == 0.0:
        cur = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur = cur / s
    cur = _simplex_projection(cur)   # ensures sum == 1

    # Pre‑compute FFT length (next power of two >= 2n‑1)
    m = 2 * n - 1
    fft_len = 1 << ((m - 1).bit_length())

    # Helper to evaluate objective using the same formula as evaluate_sequence
    def _score(seq):
        conv = _conv_full(seq, fft_len)
        max_conv = conv.max()
        # sum is 1 because of projection, but compute for safety
        sm = seq.sum()
        if sm == 0.0:

        return float(2.0 * n * max_conv / (sm * sm))

    best_seq = cur.copy()

        conv_cur = _conv_full(cur, fft_len)
        k_max = int(np.argmax(conv_cur))

        i = np.arange(n, dtype=np.int64)

        if sm == 0.0:
            return np.inf
        return float(2.0 * n * max_conv / (sm * sm))

    # --------------------------------------------------------------------
    # 4.  Track the best solution seen so far.
    # --------------------------------------------------------------------
    best_seq = cur.copy()
    best_score = _score(best_seq)

    # --------------------------------------------------------------------
    # 5.  Sub‑gradient descent on the convex objective.
    # --------------------------------------------------------------------
    max_iters = 1500
    patience = 300
    no_improve = 0

    for t in range(1, max_iters + 1):
        # Convolution of the current iterate.
        conv_cur = _conv_full(cur, fft_len)
        k_max = int(np.argmax(conv_cur))

        # Sub‑gradient of the selected maximal entry.
        i = np.arange(n, dtype=np.int64)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(cur)
        grad[mask] = cur[j[mask]]
        grad = 2.0 * grad
        if (k_max & 1) == 0:          # centre term counted only once
            cen = k_max // 2
            if cen < n:
                grad[cen] -= cur[cen]

        # Diminishing step size.
        step = 0.5 / np.sqrt(t)

        # Gradient step followed by projection onto the simplex.
        nxt = cur - step * grad
        nxt = _simplex_projection(nxt)

        # Occasionally inject tiny random noise to avoid stagnation.
        if t % 100 == 0:
            nxt += rng.normal(0.0, 0.008, size=n)
            nxt = _simplex_projection(nxt)

        score = _score(nxt)

        if np.isfinite(score) and score < best_score:
            best_seq = nxt.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        cur = nxt

        if no_improve >= patience:
            break

    # --------------------------------------------------------------------
    # 6.  Final sanity check and return.
    # --------------------------------------------------------------------
    result = best_seq.tolist()
    if not result or any(not np.isfinite(x) for x in result):
        result = [1.0 / n] * n
    return result
