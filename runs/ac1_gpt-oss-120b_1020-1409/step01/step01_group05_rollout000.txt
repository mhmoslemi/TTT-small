def evaluate_sequence(sequence: list) -> float:
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]

    else:

        a = 1.0 / np.sqrt(i)
        a = a / a.sum()

    # maintain best
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    # precompute FFT size

    while n_fft < 2*n-1:

    # iterate

    decay = 0.995
    for it in range(num_iters):
        # compute convolution using FFT
        Fa = np.fft.rfft(a, n=n_fft)
        conv = np.fft.irfft(Fa*Fa, n=n_fft)[:2*n-1]
        k = int(np.argmax(conv))

        # gradient g

        if low < 0:
            low = 0
        high = k if k < n-1 else n-1
        idx = np.arange(low, high+1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        # update

        # limit exponent to avoid overflow
        exp_factor = np.exp(-np.clip(step * g, -20, 20))
        a = a * exp_factor

    if s <= 0:

    else:
