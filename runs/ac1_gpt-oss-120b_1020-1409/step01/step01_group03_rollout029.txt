    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        f = np.clip(parent, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        # fallback: analytic decreasing series
        n = 256
        f = _make_series(n)
    n = f.shape[0]

    # ------------------------------------------------------------------
    # Normalise sum to sqrt(2n) – the objective is scale‑invariant.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    s = f.sum()
    if s > 0.0:
        f = f * (target_sum / s)
    else:
        f = np.full(n, target_sum / n)

    # ------------------------------------------------------------------
    # Pre‑compute static row/col indices for the LP constraint matrix.
    # For a given vector a, the matrix entry A[k,j] = a_{k-j}
    # (zero when k‑j is out of range).  Row indices are i+j,
    # column indices are j.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + j[None, :]).ravel()          # shape (n*n,)
    cols = np.tile(j, n)                               # shape (n*n,)
    m = 2 * n - 1                                       # number of convolution positions

    # ------------------------------------------------------------------
    # Keep the best candidate seen so far.
    # ------------------------------------------------------------------
    best = f.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # Phase 1 – simple projected gradient descent with occasional block
    # perturbations.  This quickly drives the bound down.
    # ------------------------------------------------------------------
    grad_steps = 1500
    lr0 = 0.25
    for step in range(grad_steps):
        conv = _conv_self(best)
        M = conv.max()
        k = int(conv.argmax())

        # gradient of max(conv)/sum^2 (ignoring the constant 2n)
        g = np.zeros_like(best)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = best[k - idx]

        S = best.sum()
        if S == 0.0:
            break
        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

        # adaptive learning rate (decreases like 1/√step)
        lr = lr0 / np.sqrt(step + 1.0)
        cand = best - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand = cand * (target_sum / cand_sum)
            np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional random block scaling
        if step % 37 == 0:
            width = max(1, n // rng.integers(20, 50))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.13))
            blk = best.copy()
            blk[start:start + width] *= factor
            np.clip(blk, 0.0, 1000.0, out=blk)
            blk = blk * (target_sum / blk.sum())
            np.clip(blk, 0.0, 1000.0, out=blk)
            blk_score = _score(blk)
            if blk_score < best_score:
                best = blk
                best_score = blk_score

    # ------------------------------------------------------------------
    # Phase 2 – linear‑program refinement (only if SciPy is available).
    # Solve: maximise Σ b_j  subject to b_j≥0 and (f * b)[k] ≤ M.
    # ------------------------------------------------------------------
    if _HAS_SCIPY:
        lp_iters = 8
        mix = 0.15
        for _ in range(lp_iters):
            conv = _conv_self(best)
            M = conv.max()

            # Build the constraint matrix A_ub (sparse)
            data = np.repeat(best, n)                     # f_i repeated for each column
            A_ub = _sp.csr_matrix((data, (rows, cols)), shape=(m, n))

            c = -np.ones(n)                               # maximise sum(b) => minimise -sum(b)
            ub_vec = np.full(m, M)

            # Solve the LP
            try:
                res = _linprog(
                    c,
                    A_ub=A_ub,
                    b_ub=ub_vec,
                    bounds=[(0, None)] * n,
                    method='highs',
                    options={'presolve': True, 'time_limit': 8},
                )
            except Exception:  # pragma: no cover
                break

            if not res.success or res.x is None:
                break

            b = res.x
            sb = b.sum()
            if sb == 0.0:
                break
            g = b * (target_sum / sb)

            cand = (1.0 - mix) * best + mix * g
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand = cand * (target_sum / cand.sum())
            np.clip(cand, 0.0, 1000.0, out=cand)

            cand_score = _score(cand)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
            else:
                mix = max(mix * 0.5, 0.01)   # shrink step if no progress

    # ------------------------------------------------------------------
    # Final random refinement (cheap, cheap local search).
    # ------------------------------------------------------------------
    for _ in range(200):
        rnd = rng.random(n)
        rnd = rnd / rnd.sum() * target_sum
        rnd_score = _score(rnd)
        if rnd_score < best_score:
            best = rnd
            best_score = rnd_score

    # ------------------------------------------------------------------
    # Defensive fallback – guarantee a non‑empty finite list.
    # ------------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.full(n, 1.0)

    return [float(v) for v in best]
