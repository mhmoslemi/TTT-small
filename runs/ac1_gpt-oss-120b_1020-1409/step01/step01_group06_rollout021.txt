import numpy as np
from math import exp, log
# define functions

def _conv_max(arr):
    n = arr.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[: 2*n - 1]
    return conv.max()

def _score(arr):
    s = arr.sum()
    if s == 0: return float('inf')
    maxb = _conv_max(arr)
    return 2.0 * arr.shape[0] * maxb / (s * s)

def _perturb_block(arr, rng):
    n = arr.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12))) # keep same as previous
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.06))) # log-normal factor
    new = arr.copy()
    new[start:start+width] *= factor
    # optionally mirror
    if rng.random() < 0.3:
        new = 0.5 * (new + new[::-1])
    np.clip(new, 0.0, 1000.0, out=new)
    # renormalize to original sum (target)
    s = new.sum()
    if s>0:
        new *= target_sum / s
    return new

def _perturb_transfer(arr, rng):
    n = arr.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return arr
    mass = arr[i]
    if mass == 0:
        return arr
    # move a fraction epsilon (lognormal delta)
    epsilon = max(0., np.tanh(rng.normal(0, 0.3))) # not good
    # Simpler: pick delta fraction uniformly 0..0.5 of arr[i]
    delta = float(rng.uniform(0.0, 0.5) * mass)
    new = arr.copy()
    new[i] -= delta
    new[j] += delta
    # clip
    np.clip(new, 0.0, 1000.0, out=new)
    # renorm to target_sum
    s = new.sum()
    if s > 0:
        new *= target_sum / s
    return new

def transfer_perturb(vec, rng, target_sum):
    n = vec.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return vec
    # amount to move: random fraction of vec[i]
    amount = float(rng.random() * 0.5) * vec[i]   # up to 50% of i
    new = vec.copy()
    new[i] -= amount
    new[j] += amount
    # clip & renorm
    np.clip(new, 0.0, 1000.0, out=new)
    s = new.sum()
    if s != 0:
        new *= target_sum / s
    return new

import numpy as np

# ----------------------------------------------------------------------
# Core convolution utilities (size depends only on n, passed explicitly)
def conv_max(arr: np.ndarray, size: int) -> float:
    """Maximum of full convolution arr * arr using FFT."""
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * arr.shape[0] - 1]
    return float(conv.max())


def score(arr: np.ndarray, size: int, n: int) -> float:
    """Exact evaluator formula (scale‑invariant)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    mb = conv_max(arr, size)
    return float(2.0 * n * mb / (s * s))


# ----------------------------------------------------------------------
# Normalisation + clipping helper (keeps sum = target_sum)
def scale_to_target(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale to target L1‑norm, clip to [0,1000], and re‑scale if needed."""
    s = vec.sum()
    if s != 0.0:
        vec = vec * (target_sum / s)
    else:
        vec = np.full_like(vec, target_sum / vec.size)
    np.clip(vec, 0.0, 1000.0, out=vec)
    s2 = vec.sum()
    if s2 != 0.0:
        vec = vec * (target_sum / s2)
    return vec


# ----------------------------------------------------------------------
# Perturbation operators (keep overall L1‑norm fixed)
def block_perturb(v: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    n = v.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.06)))  # log‑normal scaling
    cand = v.copy()
    cand[start:start + width] *= factor
    if rng.random() < 0.3:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return scale_to_target(cand, target_sum)


def transfer_perturb(v: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    n = v.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return v
    amount = float(rng.random() * 0.5) * v[i]          # move up to 50 % of source mass
    cand = v.copy()
    cand[i] -= amount
    cand[j] += amount
    np.clip(cand, 0.0, 1000.0, out=cand)
    return scale_to_target(cand, target_sum)

# Structured grid search over simple families
def grid_search(n: int, target_sum: float, size: int) -> np.ndarray:
    """Explore a few parametric families and return the best vector found."""
    best_vec = None
    best_sc = float("inf")
    idx = np.arange(n, dtype=np.float64)

    # ----- power‑law (asymmetric & symmetric) -----
    p_vals = np.linspace(0.2, 5.0, 15)                     # exponents
    beta_vals = np.logspace(-3, 2, 10)                     # shifts
    for p in p_vals:
        for beta in beta_vals:
            base = (idx + beta) ** (-p)                    # decreasing shape
            for sym in (False, True):
                a = base.copy()
                if sym:
                    a = a + (idx[::-1] + beta) ** (-p)   # symmetrised version
                a = scale_to_target(a, target_sum)
                sc = score(a, size, n)
                if sc < best_sc:
                    best_sc = sc
                    best_vec = a.copy()

    # ----- pure exponential decay -----
    alpha_vals = np.logspace(-4, -1, 10)                    # decay rates
    for alpha in alpha_vals:
        a = np.exp(-alpha * idx)
        a = scale_to_target(a, target_sum)
        sc = score(a, size, n)
        if sc < best_sc:
            best_sc = sc
            best_vec = a.copy()

    # ----- combined power‑law * exponential -----
    p_c = np.linspace(0.2, 5.0, 5)
    beta_c = np.logspace(-3, 2, 5)
    alpha_c = np.logspace(-4, -1, 5)
    for p in p_c:
        for beta in beta_c:
            for alpha in alpha_c:
                a = (idx + beta) ** (-p) * np.exp(-alpha * idx)
                a = scale_to_target(a, target_sum)
                sc = score(a, size, n)
                if sc < best_sc:
                    best_sc = sc
                    best_vec = a.copy()

    return best_vec
