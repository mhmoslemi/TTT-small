    n = vec.size

import numpy as np

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full convolution arr * arr (exact, using FFT)."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact value of the evaluator (without the extra clamping step)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p)   (symmetrised if requested)."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block, then renormalise."""
    cand = vec.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.06)))          # ≈±6 % on average
    cand[start:start + width] *= factor
    # occasional mirroring (helps symmetric candidates)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand


def _mass_transfer(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Move a random amount of mass from one index to another, keep L1‑norm."""
    cand = vec.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    if cand[i] <= 0.0:
        return cand
    delta = cand[i] * rng.random() * 0.5          # at most half of source mass
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)^2).

    Strategy:
      * initialise from the provided parent (if any) or from a uniform vector;
      * run a modest grid‑search over power‑law families;
      * perform simulated‑annealing with two lightweight perturbations:
        – block multiplicative perturbation;
        – transfer of mass between two positions.
      * return the best vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n (use the provided parent if it looks valid)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    target_sum = np.sqrt(2.0 * n)               # normalisation used by the evaluator

    # ------------------------------------------------------------------
    # Start from the parent (clipped & renormalised) and from a uniform vector
    def _prepare(vec):
        np.clip(vec, 0.0, 1000.0, out=vec)
        s = vec.sum()
        if s > 0.0:
            vec *= target_sum / s
        return vec

    best_vec = _prepare(parent.copy())
    best_score = _score(best_vec, n)

    uniform = np.full(n, target_sum / n, dtype=np.float64)
    sc_uni = _score(uniform, n)
    if sc_uni < best_score:
        best_vec = uniform.copy()
        best_score = sc_uni

    # ------------------------------------------------------------------
    # Small grid‑search over power‑law shapes (both symmetric and asymmetric)
    p_vals = np.linspace(0.35, 0.65, 13)      # finer interval
    beta_vals = np.logspace(-2, 2, 16)        # 0.01 … 100
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                base = _make_powerlaw(n, p, beta, sym)
                s = base.sum()
                if s == 0.0:
                    continue
                a = base * (target_sum / s)
                np.clip(a, 0.0, 1000.0, out=a)
                sc = _score(a, n)
                if sc < best_score:
                    best_score = sc
                    best_vec = a.copy()

    # ------------------------------------------------------------------
    # Simulated annealing
    total_steps = 4000
    T_initial = 1.0
    T_final = 1e-3
    cooling = (T_final / T_initial) ** (1.0 / (total_steps - 1))

    temperature = T_initial
    for step in range(total_steps):
        # choose perturbation type (≈50 % each)
        if rng.random() < 0.5:
            cand = _block_perturb(best_vec, rng, target_sum)
        else:
            cand = _mass_transfer(best_vec, rng, target_sum)

        sc = _score(cand, n)

        if sc < best_score:
            best_score = sc
            best_vec = cand
        else:
            # Metropolis acceptance
            if temperature > 0.0:
                prob = np.exp(-(sc - best_score) / temperature)
                if rng.random() < prob:
                    best_vec = cand
        temperature *= cooling

        # early stop if we already beat the published target
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Final safety: clamp and renormalise exactly as the official evaluator would
    final = best_vec.copy()
    np.clip(final, 0.0, 1000.0, out=final)
    s = final.sum()
    if s > 0.0:
        final *= target_sum / s
    return [float(x) for x in final]
