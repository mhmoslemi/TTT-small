import numpy as np

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # get n
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    sum_target = 1.0   # We'll normalize to sum=1
    # Define a function to normalize vector
    def normalize(v):
        s = v.sum()
        if s <= 0 or not np.isfinite(s):
            v = np.ones_like(v) / v.size
            s = v.sum()
        v = v / s * sum_target
        np.clip(v, 0.0, 1000.0, out=v)
        return v

    # Initialize best candidate
    if parent_ok:
        best = np.clip(parent_arr, 0.0, 1000.0)
        best = normalize(best)
    else:
        # uniform start
        best = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    # Ensure we track best.

    # Optimization parameters
    num_runs = 3
    iters_per_run = 2500
    step0 = 0.25
    decay = 0.995
    beta = 0.9

    for run in range(num_runs):
        # start from best with some random perturbation
        a = best.copy()
        # random perturbation to escape local minima
        if run > 0:
            # multiply random subset of entries by random factor
            mask = rng.random(n) < 0.2
            factor = np.exp(rng.normal(0.0, 0.2, size=n))
            a[mask] *= factor[mask]
            a = normalize(a)
        # momentum vector
        m = np.zeros_like(a)
        step = step0
        for it in range(iters_per_run):
            # Compute convolution and max indices
            conv = np.convolve(a, a)
            max_val = conv.max()
            max_idx = np.flatnonzero(conv == max_val)
            # gradient
            g = np.zeros_like(a)
            for k in max_idx:
                low = max(0, k - (n - 1))
                high = min(k, n - 1)
                idx = np.arange(low, high + 1, dtype=np.int64)
                partner = k - idx
                g[idx] += 2.0 * a[partner]
            # momentum update
            m = beta * m + (1 - beta) * g
            # multiplicative update
            a = a * np.exp(-step * m)
            a = normalize(a)
            # Decay step
            step *= decay

            # Occasionally apply random block perturbation
            if it % 200 == 0 and rng.random() < 0.3:
                width = int(rng.integers(max(1, n // 300), max(2, n // 20)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                factor = float(np.exp(rng.normal(0.0, 0.15)))
                a[start:start+width] *= factor
                a = normalize(a)

            # Evaluate and update best
            cur_score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(cur_score) and cur_score < best_score:
                best = a.copy()
                best_score = cur_score

        # End of run

    # After runs, maybe perform final local refinement with simple hill climbing:
    for extra in range(500):
        # propose small random local change
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # transfer a small amount from i to j
        delta = a[i] * 0.05 * rng.random()
        if delta > 0:
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = normalize(a_new)
            score_new = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(score_new) and score_new < best_score:
                best = a_new.copy()
                best_score = score_new
                a = a_new
                continue

    # Return best as python list
    return [float(x) for x in best]

import numpy as np

def _normalize(v: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """Scale to have the given sum and enforce the hard bounds."""
    s = v.sum()
    if s <= 0.0 or not np.isfinite(s):
        v = np.ones_like(v, dtype=np.float64) / v.size
        s = v.sum()
    v = v / s * target_sum
    np.clip(v, 0.0, 1000.0, out=v)
    return v


def _score(v: np.ndarray, n: int) -> float:
    """Exact AC1 upper‑bound score for a non‑negative vector."""
    s = v.sum()
    if s == 0.0:
        return np.inf
    conv = np.convolve(v, v)
    return float(2.0 * n * conv.max() / (s * s))


def _grad_max(v: np.ndarray, n: int, idxs: np.ndarray) -> np.ndarray:
    """Gradient of the convolution entries whose value equals the current maximum."""
    g = np.zeros_like(v)
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = k - i
        g[i] += 2.0 * v[j]
    return g


def propose_candidate(seed: int = 42, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and optional parent sequence.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = (
            parent_arr.ndim == 1
            and n > 0
            and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # ------------------------------------------------------------------
    # Initial candidate (normalised to sum = 1).
    # ------------------------------------------------------------------
    if parent_ok:
        init = np.clip(parent_arr, 0.0, 1000.0)
        best = _normalize(init, 1.0)
    else:
        best = np.full(n, 1.0 / n, dtype=np.float64)

    best_score = _score(best, n)

    # ------------------------------------------------------------------
    # Core optimisation – exponentiated‑gradient on the active convolution peak.
    # ------------------------------------------------------------------
    a = best.copy()
    m = np.zeros_like(a)               # momentum term
    step = 0.30
    decay = 0.9995
    beta = 0.9                         # momentum weight
    total_iters = 12000
    restart_every = 3000               # occasional fresh random restarts

    for it in range(total_iters):
        # Current convolution and score.
        conv = np.convolve(a, a)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val / (a.sum() * a.sum()))
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score

        # Gradient of the active max entry (or entries, if several tie).
        max_idx = np.flatnonzero(conv == max_val)
        g = _grad_max(a, n, max_idx)

        # Momentum update (exponential moving average of sub‑gradients).
        m = beta * m + (1.0 - beta) * g

        # Exponentiated‑gradient descent step (keeps non‑negativity).
        a = a * np.exp(-step * m)

        # Renormalise and enforce bounds.
        a = _normalize(a, 1.0)

        # Step‑size schedule.
        step *= decay

        # Random block perturbation – breaks symmetry and explores new regions.
        if (it % 150 == 0) and rng.random() < 0.35:

            a[start:start + width] *= factor

    # ------------------------------------------------------------------
    for _ in range(800):

            best_score = s2

    # Return the best sequence found (as plain Python floats).
    # ------------------------------------------------------------------
