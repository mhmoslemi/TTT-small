def _conv_max_fft(arr):
    n = arr.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fr = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fr * fr, n=size)[:2*n-1]
    return float(conv.max())

def _score(arr, n):
    s = arr.sum()

        return float('inf')
    max_c = _conv_max_fft(arr)
    return 2.0 * n * max_c / (s * s)

    base = (idx + beta) ** (-p)

def _mix_shapes(n, rng):

import numpy as np
import math

# ----------------------------------------------------------------------
def _conv_max_fft(arr: np.ndarray) -> float:
    """Maximum value of the full convolution arr * arr using FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # power of two >= 2n-1
    f = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact objective used by the official evaluator (scale‑invariant)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_c = _conv_max_fft(arr)
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p), optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + base[::-1]
    return base


def _block_perturb(vec: np.ndarray, rng: np.random.Generator,
                   sigma: float = 0.07) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = vec.size
    w_min = max(1, int(0.001 * n))
    w_max = max(2, int(0.1 * n))
    width = int(rng.integers(w_min, w_max))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = math.exp(rng.normal(0.0, sigma))
    out = vec.copy()
    out[start:start + width] *= factor
    if rng.random() < 0.3:                     # optional mirroring
        out = 0.5 * (out + out[::-1])
    np.clip(out, 0.0, 1000.0, out=out)
    # renormalise to keep the original L1‑norm (scale‑invariance makes it optional)
    s = out.sum()
    if s > 0.0:
        out *= vec.sum() / s
    return out


def _mass_transfer(vec: np.ndarray, rng: np.random.Generator,
                   eps: float = 0.2) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = vec.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    while j == i:
        j = rng.integers(0, n)
    amount = vec[i] * rng.uniform(0.0, eps)
    out = vec.copy()
    out[i] -= amount
    out[j] += amount
    np.clip(out, 0.0, 1000.0, out=out)
    s = out.sum()
    if s > 0.0:
        out *= vec.sum() / s
    return out
