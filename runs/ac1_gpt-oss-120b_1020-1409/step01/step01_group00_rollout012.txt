def propose_candidate(seed=42, num_steps=3000, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        n = int(height_sequence_1.shape[0])
        a = height_sequence_1.astype(np.float64).copy()
    except NameError:
        n = 256
        a = np.ones(n, dtype=np.float64)  # fallback uniform
    np.clip(a, 0.0, 1000.0, out=a)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    current = a.copy()
    current_score = best_score

    # Optionally include an early baseline uniform for fallback
    if best_score > 2.0:  # improbable
        uniform = np.ones(n, dtype=np.float64)
        uniform_score = float(evaluate_sequence(uniform.tolist()))
        if uniform_score < best_score:
            best = uniform.copy()
            best_score = uniform_score
            current = uniform.copy()
            current_score = uniform_score

    eps = 1e-12
    for step in range(num_steps):
        # Compute convolution and max
        conv = np.convolve(current, current)
        M = float(conv.max())
        # Choose a shift where conv is near M
        max_idxs = np.where(conv >= M - eps)[0]
        k = int(rng.choice(max_idxs))

        # Determine donor index within this shift
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # Compute products for these i
        # Use vectorized compute
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        products = current[i_range] * current[j_range]
        # Choose donor as index with largest product
        donor_idx = int(i_range[np.argmax(products)])
        # Avoid donor with negligible mass
        if current[donor_idx] <= 0:
            # fall back to random perturbation
            donor_idx = rng.integers(0, n)
        # Choose receiver index with low value
        # Sample a few candidates and pick the smallest value
        cand_receivers = rng.integers(0, n, size=5)
        # ensure not donor
        cand_receivers = cand_receivers[cand_receivers != donor_idx]
        if cand_receivers.size == 0:
            receiver_idx = rng.integers(0, n)
            while receiver_idx == donor_idx:
                receiver_idx = rng.integers(0, n)
        else:
            receiver_idx = int(cand_receivers[np.argmin(current[cand_receivers])])
        # Transfer fraction of donor's mass
        frac = float(rng.uniform(0.001, 0.02))
        delta = current[donor_idx] * frac
        if delta < 1e-12:
            continue
        cand = current.copy()
        cand[donor_idx] = max(0.0, cand[donor_idx] - delta)
        cand[receiver_idx] = min(1000.0, cand[receiver_idx] + delta)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        # occasional block scaling
        if step % 250 == 0 and step > 0:
            width = int(rng.integers(max(1, n // 250), max(2, n // 30)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.09)))
            block_cand = current.copy()
            block_cand[start:start+width] *= factor
            np.clip(block_cand, 0.0, 1000.0, out=block_cand)
            block_score = float(evaluate_sequence(block_cand.tolist()))
            if np.isfinite(block_score) and block_score < current_score:
                current = block_cand
                current_score = block_score
                if block_score < best_score:
                    best = block_cand.copy()
                    best_score = block_score

    # Ensure final result meets constraints (clip, non-negative)
    np.clip(best, 0.0, 1000.0, out=best)
    # return Python list
    return [float(v) for v in best]

import numpy as np

# ----------------------------------------------------------------------
# Helper to generate the classic binomial (Catalan) sequence if no parent.
# ----------------------------------------------------------------------
def _generate_binomial_sequence(length: int) -> np.ndarray:
    """Return a_i = C(2i,i) / 4^i for i=0..length‑1 as float64."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a

# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound used by `evaluate_sequence`.
    The algorithm starts from the supplied parent sequence `height_sequence_1`
    if present, otherwise from the binomial sequence, and performs a
    greedy mass‑transfer local search guided by the locations where the
    convolution attains its maximum.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Obtain initial vector `a`.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(height_sequence_1.shape[0])
            a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        except Exception:
            n = 256
            a = _generate_binomial_sequence(n)
    else:
        n = 256
        a = _generate_binomial_sequence(n)

    # enforce evaluator’s clipping rules up‑front
    np.clip(a, 0.0, 1000.0, out=a)

    # --------------------------------------------------------------
    # Set up bookkeeping.
    # --------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    current = a.copy()
    current_score = best_score

    # tiny tolerance for detecting the max convolution plateau
    eps = 1e-12
    target_bound = 1.50274349  # stop early if reached

    # --------------------------------------------------------------
    # Main local‑search loop.
    # --------------------------------------------------------------
    for step in range(num_steps):
        # 1. Full self‑convolution and the current maximum value.
        conv = np.convolve(current, current)
        M = float(conv.max())

        # 2. Pick a shift where the maximum is attained (randomised).
        max_positions = np.where(conv >= M - eps)[0]
        k = int(rng.choice(max_positions))

        # 3. Within this shift find the pair (i, j) with the largest product.
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        products = current[i_range] * current[j_range]
        donor_idx = int(i_range[np.argmax(products)])

        # If the chosen donor has essentially no mass, fall back to a random index.
        if current[donor_idx] <= 0.0:
            donor_idx = int(rng.integers(0, n))

        # 4. Choose a receiver index among a few low‑mass candidates.
        #    This encourages moving mass away from the “hot” region.
        cand_recv = rng.integers(0, n, size=7)
        cand_recv = cand_recv[cand_recv != donor_idx]
        if cand_recv.size == 0:
            recv_idx = int(rng.integers(0, n))
            while recv_idx == donor_idx:
                recv_idx = int(rng.integers(0, n))
        else:
            recv_idx = int(cand_recv[np.argmin(current[cand_recv])])

        # 5. Transfer a modest fraction of the donor’s mass.
        frac = float(rng.uniform(0.001, 0.02))
        delta = current[donor_idx] * frac
        if delta <= 0.0:
            continue
        cand = current.copy()
        cand[donor_idx] = max(0.0, cand[donor_idx] - delta)
        cand[recv_idx] = min(1000.0, cand[recv_idx] + delta)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                if best_score <= target_bound:
                    break

        # ----------------------------------------------------------
        # Occasionally try a block‑scale perturbation to escape flat regions.
        # ----------------------------------------------------------
        if step % 250 == 0 and step > 0:
            block_width = int(rng.integers(max(1, n // 250), max(2, n // 30)))
            block_start = int(rng.integers(0, max(1, n - block_width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.09)))
            block_cand = current.copy()
            block_cand[block_start:block_start + block_width] *= factor
            np.clip(block_cand, 0.0, 1000.0, out=block_cand)

            block_score = float(evaluate_sequence(block_cand.tolist()))
            if np.isfinite(block_score) and block_score < current_score:
                current = block_cand
                current_score = block_score
                if block_score < best_score:
                    best = block_cand.copy()
                    best_score = block_score
                    if best_score <= target_bound:
                        break

    # Final safety clipping before returning.
    np.clip(best, 0.0, 1000.0, out=best)

    # Convert to plain Python list of floats as required.
    return [float(v) for v in best]
