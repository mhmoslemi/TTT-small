def _conv_max_fft(arr):
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv.max()
```

import numpy as np

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """
    Maximum value of the full self‑convolution of a non‑negative vector.
    Implemented via FFT for O(n log n) speed.
    """
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """
    Exact value of the evaluator (without the security wrapper).
    The objective is 2·n·max_b / (sum_a)², which is scale‑invariant.
    """
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))
