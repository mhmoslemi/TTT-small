import numpy as np, math, random

def _conv_max(arr):
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2*n-1]
    return float(conv.max())

def _score(arr, n):
    s = arr.sum()
    if s == 0.0:
        return float('inf')
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))

def _normalize(vec, target):
    vec = np.clip(vec, 0.0, 1000.0, out=vec)
    total = vec.sum()
    if total == 0:
        # avoid division by zero; set uniform distribution
        vec[:] = target / len(vec)
    else:
        vec *= target / total
    return vec

def _make_powerlaw(n,p,beta,reverse=False):
    idx = np.arange(n, dtype=np.float64)
    if reverse:
        # decreasing: use (n - idx - 1 + beta)^{-p}
        base = (n - idx - 1 + beta) ** (-p)
    else:
        base = (idx + beta) ** (-p)
    return base

def _make_exponential(n,c,reverse=False):
    idx = np.arange(n, dtype=np.float64)
    if reverse:
        base = np.exp(-c * (n - idx - 1))
    else:
        base = np.exp(-c * idx)
    return base

def _make_gaussian(n, sigma):
    # centered at (n-1)/2
    center = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    base = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    return base

def _make_tent(n):
    idx = np.arange(n, dtype=np.float64)
    # linearly rising to middle, then descending
    half = n // 2
    if n % 2 == 0:
        # even length: two equal middle values
        left = idx[:half] / half
        right = (n - idx[half:]) / half

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full convolution arr * arr using FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
def _score(arr: np.ndarray, n: int) -> float:
    """Exact objective value 2*n*max(conv)/sum(arr)^2."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    return float(2.0 * n * _conv_max(arr) / (s * s))

# ----------------------------------------------------------------------
def _normalize(vec: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1000] and scale so that sum(vec) == target."""
    np.clip(vec, 0.0, 1000.0, out=vec)
    total = vec.sum()
    if total == 0.0:
        vec[:] = target / vec.size
    else:
        vec *= target / total
    return vec

def _make_powerlaw(n: int, p: float, beta: float, reverse: bool = False) -> np.ndarray:
    """a_i ∝ (i+β)^(-p) (or reversed if reverse=True)."""
    idx = np.arange(n, dtype=np.float64)
    if reverse:
        base = (n - idx - 1 + beta) ** (-p)
    else:
        base = (idx + beta) ** (-p)
    return base

    """a_i ∝ exp(-c*i) (or reversed)."""

    p_vals = np.linspace(0.20, 0.90, 15)          # 15 values
    beta_vals = np.logspace(-2, 2, 12)           # 12 values
    for rev in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                cand = _make_powerlaw(n, p, beta, reverse=rev)
                cand = _normalize(cand, target_sum)
                sc = _score(cand, n)
                if sc < best_score:
                    best_score = sc
                    best_vec = cand.copy()

    # ------------------------------------------------------------------
    # 2) Exponential decay families
    c_vals = np.linspace(5e-4, 1e-2, 12)        # decay rates
    for rev in (False, True):
        for c in c_vals:
            cand = _make_exponential(n, c, reverse=rev)
            cand = _normalize(cand, target_sum)
            sc = _score(cand, n)
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()

    # ------------------------------------------------------------------
    # 3) Broad Gaussian shapes
    sigma_vals = np.linspace(n / 8.0, n / 2.0, 6)
    for sigma in sigma_vals:
        cand = _make_gaussian(n, sigma)
        cand = _normalize(cand, target_sum)
        sc = _score(cand, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # 4) Simple tent shape
    cand = _make_tent(n)
    cand = _normalize(cand, target_sum)
    sc = _score(cand, n)
    if sc < best_score:
        best_score = sc
        best_vec = cand.copy()

    # ------------------------------------------------------------------
    # 5) Parent solution (if it happens to be good)
    cand = _normalize(parent.copy(), target_sum)
    sc = _score(cand, n)
    if sc < best_score:
        best_score = sc
        best_vec = cand.copy()

    # ------------------------------------------------------------------
    # Stochastic refinement (simulated‑annealing style)
    current_vec = best_vec.copy()
    current_score = best_score

    for it in range(1500):
        # choose a perturbation type
        ptype = rng.integers(0, 5)
        cand = current_vec.copy()

        if ptype == 0:                     # block scaling
            width = int(rng.integers(max(1, n // 500), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.08)))
            cand[start:start + width] *= factor

        elif ptype == 1:                   # additive Gaussian noise
            sigma_noise = 0.02 * target_sum / n
            noise = rng.normal(0.0, sigma_noise, size=n)
            cand += noise
            cand = np.maximum(cand, 0.0)

        elif ptype == 2:                   # three‑point smoothing
            cand = (cand + np.roll(cand, 1) + np.roll(cand, -1)) / 3.0

        elif ptype == 3:                   # symmetrisation
            cand = 0.5 * (cand + cand[::-1])

        else:                              # random cyclic shift averaging
            shift = int(rng.integers(1, n))
            cand = 0.5 * (cand + np.roll(cand, shift))

        cand = _normalize(cand, target_sum)
        sc = _score(cand, n)

        # acceptance test for the current state
        temp = max(0.1, 1.0 - it / 2000.0)          # linear cooling
        if sc <= current_score or rng.random() < math.exp(-(sc - current_score) / temp):
            current_vec = cand
            current_score = sc

        # record best ever seen
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # Final safety normalisation (ensures values lie inside the evaluator's limits)
    final_vec = _normalize(best_vec.copy(), target_sum)

    return [float(x) for x in final_vec]
