def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n and parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)
    # Initialize best candidate
    best = np.clip(parent, 0.0, 1000.0)
    best = best / best.sum() if best.sum() > 0 else best
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Scan power-law family
    alphas = np.logspace(-5, 1, num=30)
    p_vals = [0.3,0.4,0.5,0.6,0.7,0.8,1.0,1.2,1.5,2.0]
    for p in p_vals:
        for alpha in alphas:
            seq = (np.arange(n, dtype=np.float64) + alpha) ** (-p)
            # avoid overflow in power for large n: np.power works fine.
            seq = np.clip(seq, 0.0, 1000.0)
            seq = seq / seq.sum()
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = seq, score
    
    # Scan exponential family
    betas = np.logspace(-4, 1, num=25)
    for beta in betas:
        seq = np.exp(-beta * np.arange(n, dtype=np.float64))
        seq = np.clip(seq, 0.0, 1000.0)
        seq = seq / seq.sum()
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = seq, score
    
    # Hill climbing loop
    min_width = max(1, n // 200)
    max_width = max(2, n // 20)
    sigma = 0.07
    transfer_rate = 0.2
    for _ in range(2000):
        cand = best.copy()
        if rng.random() < 0.7:  # block multiplicative
            width = int(rng.integers(min_width, max_width+1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start+width] *= factor
        else:  # pairwise transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n-1))
            if j >= i:
                j += 1
            delta = cand[i] * rng.random() * transfer_rate
            cand[i] -= delta
            cand[j] += delta
        # Renormalize
        total = cand.sum()
        if total > 0:
            cand /= total
        # Clip
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score
    # Return as list
    return [float(x) for x in best]
