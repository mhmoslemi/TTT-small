import numpy as np

def _clip_and_normalize(v, n):
    """Clip to [0,1000] and scale so that sum(v)=1."""
    np.clip(v, 0.0, 1000.0, out=v)
    s = v.sum()
    if s <= 0.0 or not np.isfinite(s):
        v[:] = 1.0 / n
        s = v.sum()
    v /= s
    return v

def _gradient_for_indices(a, idxs):
    """Gradient of the sum of convolution entries at positions idxs."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in idxs:
        low = k - (n - 1)
        if low < 0:
            low = 0
        high = k
        if high > n - 1:
            high = n - 1
        idx = np.arange(low, high + 1, dtype=np.int64)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    return g
