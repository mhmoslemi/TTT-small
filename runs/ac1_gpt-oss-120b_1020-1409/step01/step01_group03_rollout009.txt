import numpy as np

# ------------------------------------------------------------
# Helper: classic Catalan-like series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the truncation of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast self‑convolution via real FFT.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Score exactly as the evaluator (but using FFT for speed).
# ------------------------------------------------------------
def _score(x: np.ndarray) -> float:
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(x)
    return 2.0 * x.size * c.max() / (s * s)

        # --------------------------------------------------------------
        if best_score <= target:
            break
        if stagnant > 600:
            break

    # --------------------------------------------------------------
    # Fallback safety net.
    # --------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)
        best = best / best.sum()

    return [float(v) for v in best]
