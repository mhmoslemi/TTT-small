import numpy as np

# ------------------------------------------------------------
# Core convolution (full) using FFT – linear memory, O(n log n).
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 bound (mirrors the harness evaluator).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    return 2.0 * seq.size * c.max() / (s * s)

# One projected‑gradient step on the max‑convolution value.
# ------------------------------------------------------------
def _gradient_step(x: np.ndarray, rng: np.random.Generator):
    n = x.size
    conv = _conv_self(x)
    k = int(conv.argmax())               # index attaining the max
    # gradient of max‑convolution: g_i = 2 * x_{k-i} (zero outside support)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1)
    g = np.zeros_like(x)
    g[idx] = 2.0 * x[k - idx]

    sum_g = g.sum()
    d = g - sum_g / n

    lr = 1.0
