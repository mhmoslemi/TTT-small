import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return the provided parent sequence if it exists; otherwise a safe fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    # fallback – a short uniform vector that always satisfies the evaluator
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """
    Generate the truncated arcsine coefficients:
        a_0 = 1,
        a_i = a_{i-1} * (1 - 0.5 / i)  for i >= 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


# ----------------------------------------------------------------------
# Core optimisation
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=8000, **kwargs):
    """
    Stochastic local search for a non‑negative step‑function that minimises
    the AC1 upper bound. Returns a list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial candidate selection
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = parent.shape[0]

    arcsine = _arcsine_sequence(n)

    # evaluate both seeds
    best = None
    best_score = np.inf
    for seq in (parent, arcsine):
        sc = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = seq.copy()
            best_score = sc

    if best is None:  # safety net
        best = np.ones(256, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    current = best.copy()
    rev_idx = np.arange(n - 1, -1, -1)

    target = 1.50274349  # early‑stop target from the problem statement

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # early exit if we already hit the target bound
        if best_score <= target:
            break

        # --------------------------------------------------------------
        # Periodic block perturbation (global exploration)
        # --------------------------------------------------------------
        if step % 500 == 0 and step > 0:
            width = int(rng.integers(max(2, n // 200), max(3, n // 10)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand = current.copy()
            cand[start:start + width] *= factor
            if rng.random() < 0.5:
                cand = 0.5 * (cand + cand[rev_idx])
            np.clip(cand, 0.0, 1000.0, out=cand)

            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best, best_score, current = cand, sc, cand
                continue
            # allow occasional uphill move
            temperature = 1.0 / (step + 1.0)
            if np.isfinite(sc) and np.exp((best_score - sc) / temperature) > rng.random():
                current = cand
            continue

        # --------------------------------------------------------------
        # Gradient‑guided mass transfer (local refinement)
        # --------------------------------------------------------------
        conv = np.convolve(current, current)
        k_max = int(np.argmax(conv))

        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)               # indices contributing to max
        weights = current[k_max - idx]                       # f_{k_max-i}
        contributions = current[idx] * weights               # f_i * f_{k_max-i}
        if contributions.size == 0:
            continue

        i = int(idx[int(np.argmax(contributions))])          # heavy contributor

        # pick j outside the contributing window (zero weight for the current max)
        while True:
            j = int(rng.integers(0, n))
            if j < lo or j > hi:
                break
        if i == j:
            continue

        if current[i] <= 0.0:
            continue
        frac = rng.random() * 0.9            # up to 90 % of the mass at i
        delta = frac * current[i]

        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta

        # occasional symmetrisation (helps flatten autocorrelation)
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[rev_idx])

        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc):
            if sc < best_score:
                best, best_score, current = cand, sc, cand
                continue
            # simulated‑annealing style acceptance
            temperature = 1.0 / (step + 1.0)
            if np.exp((best_score - sc) / temperature) > rng.random():
                current = cand

    # ------------------------------------------------------------------
    # Final safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best is None or best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
