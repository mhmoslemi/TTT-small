# ------------------------------------------------------------
# Helpers (top‑level, no closures)
# ------------------------------------------------------------

def _make_series(length: int) -> np.ndarray:
    """Catalan‑type series a_i = C(2i,i)/4^i, length `length`."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself using a power‑of‑two FFT."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv

    if s < 0.01:
        return float('inf')
    conv = _conv_self(seq)
    m = conv.max()
    return 2.0 * seq.size * m / (s * s)
