def conv_max(arr):
    n = arr.size
    sz = 1 << ((2*n-1).bit_length())
    fft = np.fft.rfft(arr, sz)

            np.clip(cur, 0.0, 1000.0, out=cur)
            cur_score = _compute_bound(cur)
            no_improve = 0

    # Fallback – should never happen, but guard against invalid results
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best_score = _compute_bound(best)

    return [float(v) for v in best]
