   1. rng = np.random.default_rng(seed)
   2. Try fetch parent: parent_arr = np.asarray(height_sequence_1, dtype=np.float64) else None.
   3. if parent_ok: set a = clipped and normalized parent (sum to 1). else generate a using 1/sqrt(i).
   4. set best = a.copy(); best_score = evaluate_sequence(best.tolist()).
   5. Loop for it in range(num_iters):
       - conv = np.convolve(a,a)
       - max_conv = conv.max()
       - # get topK indices (including all ties)
         sorted_indices = np.argpartition(-conv, min(top_k, len(conv)-1))[:top_k]   # topK small values
         # compute weights
         conv_top = conv[sorted_indices]
         # use exponent weight to emphasize max; using alpha, compute w = exp(alpha * (conv_top - max_conv))
         w_raw = np.exp(alpha * (conv_top - max_conv))
         w = w_raw / w_raw.sum()
         # gradient
         g = np.zeros_like(a)
         for idx_k, weight in zip(sorted_indices, w):
             low = max(0, idx_k - (n - 1))
             high = min(idx_k, n - 1)
             idx = np.arange(low, high + 1, dtype=np.int64)
             partner = idx_k - idx
             g[idx] += 2.0 * a[partner] * weight
       - step = step0 * (0.998 ** it)  # slower decay
       - a = a * np.exp(-step * g)
       - # normalization
         s = a.sum()
         if s <= 0 or not np.isfinite(s):
             a = np.ones(n, dtype=np.float64) / n
             s = a.sum()
         a = a / s  # sum = 1
         np.clip(a, 0.0, 1000.0, out=a)
       - # random block perturbation
         if (it % block_perturb_interval == 0) and rng.random() < block_perturb_prob:
             width = int(rng.integers(max(1, n // 250), max(2, n // 15)))
             start = int(rng.integers(0, max(1, n - width + 1)))
             factor = float(np.exp(rng.normal(0.0, block_perturb_factor_sigma)))
             a_pert = a.copy()

             a_pert = np.clip(a_pert, 0.0, 1000.0)
             a_pert /= a_pert.sum()
             score = float(evaluate_sequence(a_pert.tolist()))
             if np.isfinite(score) and score < best_score:
                 best = a_pert.copy()

    # Return the best sequence found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
