def propose_candidate(seed=42, **kwargs):
    import numpy as np, random, time, math
    random.seed(seed)
    np.random.seed(seed)
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size>0:
        n = int(parent.shape[0])
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_r_sequence(n).astype(np.float64)  # fallback function defined later
    # clip to bounds
    np.clip(a, 0.0, 1000.0, out=a)
    # ensure sum > 0.01 (r-sequence ensures)
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    cur_seq = a.copy()
    cur_score = best_score
    start = time.time()
    max_time = 4.5  # seconds
    T = 0.01
    decay = 0.9999
    # compute step relative to average magnitude
    avg = np.mean(cur_seq)
    # loops
    while time.time() - start < max_time:
        # choose indices
        i = random.randrange(n-1)  # early 0..n-2
        j = random.randrange(i+1, n)  # later
        # compute step: fraction of current at i
        delta = cur_seq[i] * (random.random() * 0.05)  # up to 5%
        if delta <= 0:
            continue
        # new candidate
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # clip after change to keep bounds
        if new_seq[i] < 0:
            continue
        if new_seq[j] > 1000:
            continue
        # evaluate
        new_score = float(evaluate_sequence(new_seq.tolist()))
        if not math.isfinite(new_score):
            continue
        # acceptance
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score)/T):
            cur_seq = new_seq
            cur_score = new_score
            if cur_score < best_score:
                best_seq = cur_seq.copy()
                best_score = cur_score
        # temperature decay
        T *= decay
        if T < 1e-8:
            T = 1e-8
    return best_seq.tolist()
```

import numpy as np
import random
import time
import math

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the first n terms of r_i = C(2i,i)/4^i."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function heights list that (heuristically)
    reduces the AC1 bound.  The routine uses a simple simulated‑annealing
    hill‑climber starting from the provided parent (height_sequence_1) or,
    if absent, from the r‑sequence.
    """
    random.seed(seed)
    np.random.seed(seed)

    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
        cur_seq = parent.astype(np.float64).copy()
    else:
        n = 256
        cur_seq = generate_r_sequence(n)

    # enforce bounds immediately
    np.clip(cur_seq, 0.0, 1000.0, out=cur_seq)

    # initial evaluation
    best_seq = cur_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    cur_score = best_score

    # parameters for the stochastic search
    start_time = time.time()
    max_seconds = 4.5               # generous wall‑time budget
    T = 1.0e-2                      # initial temperature
    decay = 0.9999                  # exponential cooling
    early_width = max(1, n // 20)   # region considered “early”
    late_start = max(n - early_width, early_width + 1)

    # main loop
    while time.time() - start_time < max_seconds:
        # pick a move that moves mass from an early index toward a later one
        if random.random() < 0.7 and early_width < n // 2:
            i = random.randrange(early_width)                     # early
            j = random.randrange(late_start, n)                  # late
        else:
            i = random.randrange(n - 1)
            j = random.randrange(i + 1, n)

        # size of the transfer – a small fraction of the source entry
        src_val = cur_seq[i]
        if src_val <= 0.0:
            continue
        delta = src_val * (random.random() * 0.03)   # up to 3 % move
        if delta <= 0.0:
            continue

        # propose new sequence (mass conserved)
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # enforce hard bounds
        if new_seq[i] < 0.0 or new_seq[j] > 1000.0:
            continue
        # evaluate candidate
        try:
            new_score = float(evaluate_sequence(new_seq.tolist()))
        except Exception:
            continue
        if not math.isfinite(new_score):
            continue

        # Metropolis acceptance
        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / T):
            cur_seq = new_seq
            cur_score = new_score
            if cur_score < best_score:
                best_seq = cur_seq.copy()
                best_score = cur_score

        # cool down
