import numpy as np

# ----------------------------------------------------------------------
def _conv_max_and_idx(arr: np.ndarray):
    """Return max value of full convolution arr*arr and its index (FFT, exact)."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv.max()), idx


def _conv_max(arr: np.ndarray) -> float:
    """Return max value of full convolution arr*arr (FFT, exact)."""
    mx, _ = _conv_max_and_idx(arr)
    return mx


def _eval(arr: np.ndarray):
    """Exact score = 2*n*max(conv)/(sum^2).  Returns (score, max_val, idx)."""
    n = arr.shape[0]
    s = arr.sum()
    if s == 0.0:
        return float("inf"), None, None
    max_val, idx = _conv_max_and_idx(arr)
    score = 2.0 * n * max_val / (s * s)
    return float(score), float(max_val), int(idx)

def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p), optionally symmetrised."""
    i = np.arange(n, dtype=np.float64)
    base = (i + beta) ** (-p)
    if sym:
        base = base + base[::-1]
    return base

    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    cand[start: start + width] *= factor
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    total = cand.sum()
    if total > 0.0:
        cand *= target_sum / total
    return cand


def _random_candidate(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random non‑negative candidate (power‑law / exp / smoothed)."""
    choice = rng.integers(0, 3)
    if choice == 0:                       # power‑law
        p = rng.uniform(0.2, 1.5)
        beta = 10 ** rng.uniform(-3, 2)   # 0.001 … 100
        sym = rng.random() < 0.5
        base = _make_powerlaw(n, p, beta, sym)
    elif choice == 1:                     # exponential decay
        r = rng.uniform(0.5, 0.99)
        base = r ** np.arange(n, dtype=np.float64)
        if rng.random() < 0.3:
            base = base[::-1]            # occasional reversal
    else:                                 # smoothed positive noise
        noise = rng.exponential(scale=1.0, size=n)
        w = int(rng.integers(max(2, n // 200), max(10, n // 20) + 1))
        kernel = np.ones(w, dtype=np.float64) / w
        smooth = np.convolve(noise, kernel, mode='same')
        base = smooth
        if rng.random() < 0.35:
            base = np.exp(base)           # make distribution heavier‑tailed
        base = np.maximum(base, 1e-12)   # avoid exact zeros
    # normalise to the required L1‑norm
    s = base.sum()
    if s == 0.0:
        base = np.full(n, target_sum / n, dtype=np.float64)
    else:
        base = base * (target_sum / s)
    np.clip(base, 0.0, 1000.0, out=base)
    # re‑scale after clipping
    s2 = base.sum()
    if s2 > 0.0:
        base *= target_sum / s2
    return base


def _gradient(arr: np.ndarray, max_val: float, idx: int) -> np.ndarray:
    """Gradient of the score w.r.t. arr assuming the max occurs at idx."""
    n = arr.shape[0]
    s = arr.sum()
    if s == 0.0:
        return np.zeros_like(arr)
    i = np.arange(n, dtype=np.int64)
    j = idx - i
    mask = (j >= 0) & (j < n)
    dM = np.zeros_like(arr)
    dM[mask] = 2.0 * arr[j[mask]]
    factor = (2.0 * n) / (s * s)
    const = 2.0 * max_val / s
    return factor * (dM - const)

def propose_candidate(seed=42, **kwargs):
    """
    Produce a non‑negative step function (list of floats) with a low AC1 bound.
    The routine mixes analytic families, random synthesis, gradient descent,
    and block‑wise stochastic refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n and a useful starting point (parent if available)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    target_sum = np.sqrt(2.0 * n)          # normalisation used by the evaluator

    # ------------------------------------------------------------------
    # Normalise parent to the required sum and clamp
    cand = parent.copy()
    np.clip(cand, 0.0, 1000.0, out=cand)
    s0 = cand.sum()
    if s0 > 0.0:
        cand *= target_sum / s0
    else:
        cand = np.full(n, target_sum / n, dtype=np.float64)

    best_vec = cand
    best_score, _, _ = _eval(best_vec)

    # Uniform fallback (score exactly 2)
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    u_score, _, _ = _eval(uniform)
    if u_score < best_score:
        best_vec = uniform
        best_score = u_score

    # ------------------------------------------------------------------
    # Grid search over power‑law families (as in the previous implementation)
    p_vals = np.linspace(0.20, 1.20, 11)                # broader than before
    beta_vals = np.logspace(-3, 2, 12)                 # 0.001 … 100
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                base = _make_powerlaw(n, p, beta, sym)
                total = base.sum()
                if total == 0.0:
                    continue
                a = base * (target_sum / total)
                np.clip(a, 0.0, 1000.0, out=a)
                a_sum = a.sum()
                if a_sum > 0.0:
                    a *= target_sum / a_sum
                score, _, _ = _eval(a)
                if score < best_score:
                    best_score, best_vec = score, a.copy()
                    if best_score < 1.51:
                        break

    # ------------------------------------------------------------------
    # Simple exponential‑decay grid search
    r_vals = np.linspace(0.5, 0.99, 12)
    idx_arr = np.arange(n, dtype=np.float64)
    for r in r_vals:
        base = r ** idx_arr
        total = base.sum()
        if total == 0.0:
            continue
        a = base * (target_sum / total)
        np.clip(a, 0.0, 1000.0, out=a)
        a_sum = a.sum()
        if a_sum > 0.0:
            a *= target_sum / a_sum
        score, _, _ = _eval(a)
        if score < best_score:
            best_score, best_vec = score, a.copy()
            if best_score < 1.51:
                break

    # ------------------------------------------------------------------
    # Randomly generated candidates (different families)
    for _ in range(200):
        a = _random_candidate(n, target_sum, rng)
        score, _, _ = _eval(a)
        if score < best_score:
            best_score, best_vec = score, a.copy()
            if best_score < 1.51:
                break

    # ------------------------------------------------------------------
    # Gradient‑descent refinement using the active max‑lag index
    lr = 0.12                         # initial learning rate
    for it in range(300):
        max_val, idx = _conv_max_and_idx(best_vec)
        grad = _gradient(best_vec, max_val, idx)
        new_vec = best_vec - lr * grad
        np.clip(new_vec, 0.0, 1000.0, out=new_vec)
        s_new = new_vec.sum()
        if s_new > 0.0:
            new_vec *= target_sum / s_new
        score, _, _ = _eval(new_vec)
        if score < best_score:
            best_score, best_vec = score, new_vec.copy()
            lr = min(lr * 1.10, 0.5)          # slightly enlarge step after improvement
            if best_score < 1.51:
                break
        else:
            lr *= 0.85                       # otherwise shrink the step
        if lr < 1e-4:
            break

    # ------------------------------------------------------------------
    # Final stochastic block‑wise refinement (cheap cheap proposals)
    for _ in range(500):
        cand = _block_perturb(best_vec, rng, target_sum)
        score, _, _ = _eval(cand)
        if score < best_score:
            best_score, best_vec = score, cand.copy()
            if best_score < 1.51:
                break

    # ------------------------------------------------------------------
    # Return a plain Python list of floats (the evaluator will clamp again)
    return [float(x) for x in best_vec]
