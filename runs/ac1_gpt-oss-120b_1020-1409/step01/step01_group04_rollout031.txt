    best = np.clip(parent, 0, 1000)
    best_score = float(evaluate_sequence(best.tolist()))
    # If best invalid bail to fallback later
    if not np.isfinite(best_score):
        best = np.clip(parent, 0, 1000)
        best_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(best_score):
            # fallback to uniform
            best = np.full(n, 1.0, dtype=np.float64)
            best_score = float(evaluate_sequence(best.tolist()))
    # Family search
    # For exponents p in linspace
    for p in np.linspace(0.2, 1.5, 14):
        for alpha in np.logspace(-4, 2, 12):
            seq = 1.0 / np.power(np.arange(n, dtype=np.float64) + alpha, p)
            cand = np.clip(seq, 0, 1000)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
    # Exponential family
    for beta in np.linspace(0.0005, 0.02, 20):
        seq = np.exp(-beta * np.arange(n, dtype=np.float64))
        cand = np.clip(seq, 0, 1000)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    # Combined family: power * exponential
    for p in np.linspace(0.3, 1.0, 8):
        for beta in np.linspace(0.001, 0.01, 8):
            seq = (1.0 / (np.arange(n, dtype=np.float64) + 1.0) ** p) * np.exp(-beta * np.arange(n, dtype=np.float64))
            cand = np.clip(seq, 0, 1000)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score

    # Hill climbing: random block scaling
    num_steps = 2000
    min_width = max(1, n // 200)
    max_width = max(2, n // 20)
    sigma = 0.07

    for _ in range(num_steps):
        cand = best.copy()
        # Choose operation
        if rng.random() < 0.7:
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start+width] *= factor
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            delta = best[i] * rng.random() * 0.2
            cand[i] -= delta
            cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Ensure final best is valid
    if not np.isfinite(best_score):
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]

import numpy as np

def _ac1_score(arr: np.ndarray) -> float:
    """Compute the AC1 bound 2*n*max(b)/(sum(a)^2) for a non‑negative array."""
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(arr, arr)
    return float(2.0 * arr.size * b.max() / (s * s))
