import numpy as np

def propose_candidate(seed=42, num_iters=3000, top_m=3, step0=0.35, **kwargs):
    """
    Search for a non‑negative step sequence that minimizes the AC1 bound.
    Uses a gradient‑descent on the (soft) max of the autocorrelation,
    enforces symmetry, and adds occasional random block perturbations.
    Falls back to a compact analytic initialization if the parent sequence
    ``height_sequence_1`` is unavailable or invalid.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine length and initial candidate.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_ok = (
            parent_arr.ndim == 1
            and parent_arr.size > 0
            and np.isfinite(parent_arr).all()
        )
        n = parent_arr.size
    except Exception:
        parent_ok = False
        n = 2000                     # fallback length

    if parent_ok:
        a = np.clip(parent_arr, 0.0, 1000.0)
        s = a.sum()
        a = a / s if s > 0 else np.ones_like(a) / a.size
    else:
        # Symmetric analytic start: blend uniform with edge‑heavy 1/√(i) shape
        i = np.arange(n, dtype=np.float64)
        p = 0.5
        edge = (i + 1) ** (-p) + (n - i) ** (-p)
        a = 0.5 * np.ones(n, dtype=np.float64) + 0.5 * edge
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        a = a / s if s > 0 else np.ones_like(a) / n

    a = a.astype(np.float64)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 2. Main optimisation loop.
    # ------------------------------------------------------------------
    sum_target = 1.0
    for it in range(num_iters):
        conv = np.convolve(a, a)                     # full autocorrelation
        max_val = conv.max()

        # Indices of the (up to) top_m largest entries of conv
        if top_m < conv.size:
            idx_top = np.argpartition(-conv, top_m - 1)[:top_m]
            idx_top = idx_top[np.argsort(-conv[idx_top])]
        else:
            idx_top = np.arange(conv.size, dtype=np.int64)

        # Gradient of the selected entries (averaged)
        g = np.zeros_like(a)
        for k in idx_top:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx = np.arange(low, high + 1, dtype=np.int64)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g = g / max(1, len(idx_top))

        # Decaying multiplicative step
        step = step0 * (0.998 ** it)
        a = a * np.exp(-step * g)

        # Enforce symmetry (mirroring)
        a = (a + a[::-1]) / 2.0

        # Renormalise and clip
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.ones_like(a) / a.size
            s = a.sum()
        a = a / s * sum_target
        np.clip(a, 0.0, 1000.0, out=a)

        # ------------------------------------------------------------------
        # Random block perturbation (occasionally)
        # ------------------------------------------------------------------
        if (it % 100 == 0) and rng.random() < 0.30:
            width = int(rng.integers(max(1, n // 300), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = float(np.exp(rng.normal(0.0, 0.09)))
            a_pert = a.copy()
            a_pert[start:stop] *= factor
            # mirror the perturbation to keep symmetry
            a_pert[n - stop:n - start] = a_pert[start:stop]
            np.clip(a_pert, 0.0, 1000.0, out=a_pert)
            a_pert = a_pert / a_pert.sum() * sum_target
            pert_score = float(evaluate_sequence(a_pert.tolist()))
            if np.isfinite(pert_score) and pert_score < best_score:
                best = a_pert.copy()
                best_score = pert_score
                a = a_pert.copy()

        # ------------------------------------------------------------------
        # Evaluate current candidate
        # ------------------------------------------------------------------
        cur_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best = a.copy()
            best_score = cur_score

    # ------------------------------------------------------------------
    # 3. Final local random refinement around the best found vector.
    # ------------------------------------------------------------------
    refined = best.copy()
    for _ in range(200):
        width = int(rng.integers(max(1, n // 400), max(2, n // 50)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        factor = float(np.exp(rng.normal(0.0, 0.07)))
        cand = refined.copy()
        cand[start:stop] *= factor
        cand[n - stop:n - start] = cand[start:stop]          # symmetry
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand = cand / cand.sum() * sum_target
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            refined = cand.copy()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if not np.isfinite(best_score) or best.size == 0:
        # ultimate fallback: uniform distribution
        fallback = np.ones(n, dtype=np.float64) / n
        return [float(x) for x in fallback]
    return [float(x) for x in best]
