import numpy as np

# ------------------------------------------------------------
# Helper: fast full convolution using FFT (real‐valued FFT)
# ------------------------------------------------------------
def _full_convolve(x: np.ndarray) -> np.ndarray:
    """Return np.convolve(x, x) using a single FFT (size = next power of two)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # smallest power of two >= 2n‑1
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)
    return conv[: 2 * n - 1]

def _inverse_sqrt_shape(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1), normalized to sum = 1."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / np.sqrt(idx)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    result = best_seq.tolist()
    try:
        # The harness imported evaluate_sequence; we just call it.
        final_score = float(evaluate_sequence(result))
        if not np.isfinite(final_score):
            raise ValueError
    except Exception:
        # Fallback: a simple uniform distribution (guaranteed to be valid)
        result = [1.0 / n] * n
    return result
