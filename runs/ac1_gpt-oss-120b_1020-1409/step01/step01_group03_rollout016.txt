def optimize(x_initial, rng, max_iters=3000, block_iter_mod=17):
    x = x_initial.copy()
    best = x.copy()
    best_score = _score(x)
    n = len(x)
    for it in range(max_iters):
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        # compute g vector
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # building g:
        # For i in i_low..i_high, g[i] = x[k - i]
        g = np.zeros_like(x)
        idx = np.arange(i_low, i_high + 1)
        g[idx] = x[k - idx]
        S = x.sum()
        if S == 0:
            break
        inv_S2 = 1.0/(S*S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3
        # line search
        step_sizes = [1.0, 0.5, 0.25, 0.125, 0.0625]
        improved = False
        for lr in step_sizes:
            cand = x - lr * grad
            np.clip(cand,0.0,1000.0,inplace=True) (but there is no inplace arg; we can assign directly)
            # may optionally symmetrize rarely
            if rng.random() < 0.05:
                cand = 0.5 * (cand + cand[::-1])
            sc = _score(cand)
            if sc < best_score - 1e-12:
                best = cand.copy()
                best_score = sc
                improved = True
                x = cand
                break
        if not improved:
            # maybe random perturb:
            width = max(1, n // rng.integers(12, 33))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.13))
            cand = x.copy()
            cand[start:start+width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
                x = cand
            else:
                # maybe accept occasional worse move with small probability
                if rng.random() < 0.01:
                    # accept
                    x = cand
        # optional periodic symmetrization:
        if it % 100 == 0 and rng.random() < 0.3:
            x = 0.5 * (x + x[::-1])
    return best, best_score

def _conv_self(x):
    n = x.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv
```

    # fallback

def _optimize(x0, rng, max_iters=2000):
    x = x0.copy()
    n = x.shape[0]
    best = x.copy()
    best_score = _score(x)
    # optionally enforce clipping
    np.clip(best, 0.0, 1000.0, out=best)
    # run iterations
    for it in range(max_iters):
        # compute conv and gradient
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        S = x.sum()
        if S == 0:
            break
        # compute g
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g = np.zeros_like(x)
        g[idx] = x[k - idx]
        inv_S2 = 1.0/(S*S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3
        # line search
        improved = False
        for lr in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = x - lr * grad
            np.clip(cand, 0.0, 1000.0, out=cand)
            # occasional symmetrization
            if rng.random() < 0.03:
                cand = 0.5 * (cand + cand[::-1])
            sc = _score(cand)
            if sc < best_score - 1e-12:
                best = cand.copy()
                best_score = sc
                x = cand
                improved = True
                break
        if not improved:
            # random block perturbation
            width = max(1, n // rng.integers(10, 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            cand = x.copy()
            cand[start:start+width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            # Symmetrize sometimes
            if rng.random() < 0.05:
                cand = 0.5 * (cand + cand[::-1])
            sc = _score(cand)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
                x = cand
            else:
                # Occasionally accept worse move to jump out
                if rng.random() < 0.01:
                    x = cand
        # occasional global symmetrization for stability
        if it % 200 == 0 and rng.random() < 0.2:
            x = 0.5 * (x + x[::-1])
    return best, best_score

import numpy as np

# ------------------------------------------------------------
# Helper: Catalan (central binomial) coefficients a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the length‑`length` truncation of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# FFT‑based full self‑convolution (real input)
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact score matching evaluate_sequence (the lower the better)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)


# ------------------------------------------------------------

# ------------------------------------------------------------
def _optimize(x0: np.ndarray, rng: np.random.Generator,
              max_iters: int = 6000) -> tuple[np.ndarray, float]:
    x = x0.copy()
    n = x.shape[0]
    best = x.copy()
    best_score = _score(best)
    # pre‑allocate buffer for gradient contributions
    g = np.empty_like(x)

    for it in range(max_iters):
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        S = x.sum()
        if S == 0.0:
            break

        # ----- build g_i = a_{k-i} on the overlapping region -----
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g.fill(0.0)
        g[idx] = x[k - idx]

        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

        # ----- line‑search on a short schedule -----
        improved = False
        for lr in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = x - lr * grad
            np.clip(cand, 0.0, 1000.0, out=cand)
            if rng.random() < 0.03:          # occasional symmetrisation
                cand = 0.5 * (cand + cand[::-1])
            sc = _score(cand)
            if sc < best_score - 1e-13:
                best = cand.copy()
                best_score = sc
                x = cand
                improved = True
                break
        if improved:
            continue

        # ----- random block perturbation (escape local minima) -----
        width = max(1, n // rng.integers(10, 30))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.12))
        cand = x.copy()
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        if rng.random() < 0.05:
            cand = 0.5 * (cand + cand[::-1])
        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            x = cand
        else:

        if (it + 1) % 250 == 0 and rng.random() < 0.2:
            x = 0.5 * (x + x[::-1])

    return best, best_score
