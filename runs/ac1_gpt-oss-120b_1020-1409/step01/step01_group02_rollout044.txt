import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _initial_sequence():
    """
    Return a non‑negative start vector.
    Uses the provided `height_sequence_1` if available,
    otherwise falls back to a compact arcsine construction of length 256.
    """
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass

    # arcsine‐type seed
    n = 256
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return np.clip(a, 0.0, 1000.0)


def _gradient_weights(seq, k_max):
    """
    For the convolution index `k_max` compute the gradient weights:
        w_i = seq[k_max - i]  if 0 ≤ k_max-i < len(seq) else 0
    """
    n = seq.size
    w = np.zeros_like(seq)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    w[idx] = seq[k_max - idx]
    return w


def _block_perturbation(base, rng):
    """Create a cheap O(n) random block perturbation."""
    n = base.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, num_steps=4000, **kwargs):
    """
    Search for a non‑negative step function that reduces the AC1 bound.
    Returns a list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    current = _initial_sequence()
    n = current.size
    if current.sum() < 0.01:                     # protect against degenerate sum
        current[0] = max(0.01, current[0])

    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # compute convolution of the current candidate
        conv = np.convolve(current, current)
        k_max = int(np.argmax(conv))

        # gradient weights for the current maximal entry
        w = _gradient_weights(current, k_max)

        # pick a donor index i (high weight) – weighted random draw
        total_w = w.sum()
        if total_w > 0 and rng.random() < 0.9:
            probs = w / total_w
            i = int(rng.choice(np.arange(n), p=probs))
        else:
            i = int(rng.integers(0, n))

        # pick a receiver index j (prefer low weight)
        attempts = 0
        while True:
            j = int(rng.integers(0, n))
            if j != i:
                break
            attempts += 1
            if attempts > 5:
                break

        # step magnitude – decaying fraction of the donor mass
        step_frac = 0.05 * (0.5 ** (step // 1000))
        delta = min(current[i], step_frac * current[i] + 1e-12)
        if delta <= 0.0:
            continue

        # projected move
        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            current = cand
            continue

        # simulated‑annealing acceptance (relative to best)
        temperature = 1.0 / (step + 1.0)
        if np.isfinite(score) and np.exp((best_score - score) / temperature) > rng.random():
            current = cand

        # occasional larger block‑wise perturbation to escape plateaus
        if rng.random() < 0.02:

    # ------------------------------------------------------------------
