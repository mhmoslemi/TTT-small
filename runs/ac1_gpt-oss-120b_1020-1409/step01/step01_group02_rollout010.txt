    except Exception:

    n = a.size

    sum_a = a.sum()
    if sum_a == 0:

    conv = full_convolution_fft(a)
    max_conv = conv.max()
    best_score = 2 * n * max_conv / (a.sum()**2)
    best = a.copy()

    for step in range(max_steps):

        conv = full_convolution_fft(a)

        eta = step0 / (1 + decay * step)   # decays slowly

def full_convolution_fft(v):
    n = v.shape[0]
    m = 1 << ((2*n - 1).bit_length())
    fv = np.fft.rfft(v, n=m)
    conv = np.fft.irfft(fv * fv, n=m)
    return conv[:2*n-1]

import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine‑type analytic seed
# ----------------------------------------------------------------------
def arcsine_seq(length):
    """Generate the decreasing arcsine coefficients."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


# ----------------------------------------------------------------------
# Helper: full convolution using a power‑of‑two FFT (O(n log n))
# ----------------------------------------------------------------------
def full_convolution_fft(vec):
    """Return np.convolve(vec, vec) without allocating an n×n matrix."""
    n = vec.shape[0]
    # next power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(vec, n=size)
    conv = np.fft.irfft(f * f, n=size)
    return conv[: 2 * n - 1]
