import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helpers
# ------------------------------------------------------------
def generate_r_sequence(n: int) -> np.ndarray:
    """Extremal r‑sequence: r_i = C(2i,i)/4^i."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def conv_max(seq: np.ndarray) -> float:
    """Maximum coefficient of the full convolution of seq with itself."""
    return float(np.max(np.convolve(seq, seq)))


def build_constraint_matrix(f: np.ndarray, rows: int, eps: float) -> sp.csc_matrix:
    """
    Build the sparse matrix A (rows × n) that represents the linear constraints
    (f * a)[k] ≤ M for k = 0..rows‑1, i.e. each row k contains the coefficients
    f[k‑j] for all j ≤ k.  Entries with f[d] < eps are omitted to keep the
    matrix sparse.
    """
    n = f.shape[0]
    data = []
    offsets = []
    # Only diagonal offsets –d (column index j = row – d) where d < rows
    max_d = min(n, rows)
    for d in range(max_d):
        coeff = f[d]
        if coeff < eps:
            continue
        length = rows - d                     # number of entries on this diagonal
        data.append(np.full(length, coeff, dtype=np.float64))
        offsets.append(-d)                    # offset = column - row = -d
    return sp.diags(data, offsets, shape=(rows, n), format='csc')


def solve_lp_max_sum(f: np.ndarray, M: float, rows: int, eps: float) -> np.ndarray | None:
    """
    Linear program:
        maximize   sum a_j
        subject to (f * a)[k] ≤ M   for k = 0..rows‑1,
                   a_j ≥ 0.
    Returns the optimal a (numpy array) or None if the solver fails.
    """
    A = build_constraint_matrix(f, rows, eps)
    n = f.shape[0]
    c = -np.ones(n, dtype=np.float64)        # minimise –sum a_j  →  maximise sum a_j
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n
    # HiGHS handles sparse matrices efficiently
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluator."""
    return float(evaluate_sequence(seq.tolist()))


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a step‑function (list of non‑negative heights) whose AC1 bound
    improves on the previous π/2 ≈ 1.5708 value.
    """
    # --------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256

    # --------------------------------------------------------
    # Initial candidate – the extremal r‑sequence
    # --------------------------------------------------------
    f = generate_r_sequence(n)
    f = np.clip(f, 0.0, 1000.0)
    best_seq = f.copy()
    best_score = evaluate(best_seq)

    # --------------------------------------------------------
    # Gather data for the LP steps
    # --------------------------------------------------------
    M = conv_max(f)                # current maximal convolution (≈1)
    rows = n                       # we enforce the first n convolution constraints
    epsilons = [0.01, 0.005, 0.002]       # pruning thresholds for the constraint matrix
    mix_factors = [0.2, 0.5, 0.8]          # how much of the LP solution to blend in

    # --------------------------------------------------------
    # Try a few LP‑based refinements
    # --------------------------------------------------------
    for eps in epsilons:
        g = solve_lp_max_sum(f, M, rows, eps)
        if g is None:
            continue

        # direct use of the LP solution (may already improve the bound)
        cand = np.clip(g, 0.0, 1000.0)
        score = evaluate(cand)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()

        # blended candidates  f←(1‑t)·f + t·g
        for t in mix_factors:
            cand = (1.0 - t) * f + t * g
            cand = np.clip(cand, 0.0, 1000.0)
            score = evaluate(cand)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()

    # --------------------------------------------------------
    # Final sanity – ensure we return a valid non‑negative list
    # --------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
