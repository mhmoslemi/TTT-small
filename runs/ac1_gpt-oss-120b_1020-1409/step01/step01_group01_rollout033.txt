import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Produce the extremal r‑sequence r_i = C(2i,i)/4^i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def _evaluate(arr: np.ndarray) -> float:
    """Wrap the provided evaluate_sequence."""
    return float(evaluate_sequence(arr.tolist()))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound used by evaluate_sequence.
    The routine starts from the supplied parent `height_sequence_1`
    (or from the r‑sequence) and performs a small greedy search that
    transfers mass from large coefficients to tiny ones.
    """
    # -----------------------------------------------------------------
    # 1) Initialise the base sequence.
    # -----------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        seq = parent.astype(np.float64).copy()
    else:
        n = 256
        seq = generate_r_sequence(n)

    # Keep all entries in the allowed range.
    np.clip(seq, 0.0, 1000.0, out=seq)

    best_seq = seq
    best_score = _evaluate(best_seq)

    # If the initial candidate is invalid fall back to a uniform sequence.
    if not np.isfinite(best_score):
        best_seq = np.ones_like(seq, dtype=np.float64)
        best_score = _evaluate(best_seq)

    # -----------------------------------------------------------------
    # 2) Greedy mass‑transfer search.
    # -----------------------------------------------------------------
    max_passes = 12                # how many outer iterations
    src_cnt = 5                    # number of largest coefficients examined
    dst_cnt = 5                    # number of smallest coefficients examined
    # Fractions of a source value that will be moved.
    fractions = [0.05, 0.10, 0.20, 0.40, 0.60, 0.80, 1.00]

    for _ in range(max_passes):
        improved = False

        # indices of the biggest and smallest entries (ties broken arbitrarily)
        src_idxs = np.argsort(-best_seq)[:src_cnt]
        dst_idxs = np.argsort(best_seq)[:dst_cnt]

        for src in src_idxs:
            src_val = best_seq[src]
            if src_val <= 0.0:
                continue
            for dst in dst_idxs:
                if src == dst:
                    continue
                for f in fractions:
                    delta = src_val * f
                    if delta <= 0.0 or delta > src_val:
                        continue

                    cand = best_seq.copy()
                    cand[src] -= delta
                    cand[dst] += delta
                    # Clip only the modified entries – the rest are already valid.
                    cand[src] = max(0.0, min(1000.0, cand[src]))
                    cand[dst] = max(0.0, min(1000.0, cand[dst]))

                    score = _evaluate(cand)
                    if np.isfinite(score) and score < best_score:
                        best_seq = cand
                        best_score = score
                        improved = True
                        break   # restart the search from the new best point
                if improved:
                    break
            if improved:
                break

        if not improved:
            break   # no single transfer improves the score – stop

    # -----------------------------------------------------------------
    # 3) Safety fallback (should never be triggered).
    # -----------------------------------------------------------------
    if not np.isfinite(best_score):
        best_seq = np.ones_like(best_seq, dtype=np.float64)

    # Return a plain Python list of floats, as required by the harness.
    return [float(v) for v in best_seq]
