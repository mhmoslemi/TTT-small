def _parent_sequence():
    """Return the supplied parent sequence or a generic fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover

    return np.ones(256, dtype=np.float64)


# ----------------------------------------------------------------------
# Helper: discrete arcsine sequence (used as a strong analytic seed)
# ----------------------------------------------------------------------
def _arcsine_sequence(n):
    """Generate the truncated arcsine coefficients."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a

# Helper: convolution via real FFT (O(n log n), modest memory)
# ----------------------------------------------------------------------
def _conv_fft(seq):
    """Full linear convolution of seq with itself using FFT."""
    n = seq.shape[0]
    size = 2 * n - 1
    f = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f * f, size)
    return conv

def _lp_refine(seq, rng, max_refine=5):
    """Iteratively improve seq by solving a small LP that keeps f*g <= M."""
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    cur = best_seq.copy()

    for _ in range(max_refine):
        conv = _conv_fft(cur)
        idx_max = int(np.argmax(conv))

        total = 2 * n - 1
        # pick a modest set of constraints (always include the worst index)
        inds = [idx_max]
        extra = min(80, total - 1)
        if extra > 0:
            cand = rng.choice(total, size=extra + 1, replace=False)
            cand = cand[cand != idx_max][:extra]
            inds.extend(cand.tolist())

        m = len(inds)
        A = np.zeros((m, n + 1), dtype=np.float64)

            hi = min(n - 1, k)
            j_idx = np.arange(lo, hi + 1, dtype=np.int64)

            A[row, j_idx] = cur[f_idx]
