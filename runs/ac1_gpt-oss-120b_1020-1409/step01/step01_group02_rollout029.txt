import numpy as np

def _arcsine_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1,n):
        a[i] = a[i-1] * (1.0 - 0.5 / i)
    return a

def _objective(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = len(seq)
    L = 1 << ((2*n-1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (total*total)

def _gradient(seq, conv, max_idx):
    n = len(seq)
    lo = max(0, max_idx - (n-1))
    hi = min(n-1, max_idx)
    i_range = np.arange(lo, hi+1, dtype=np.int64)
    j = max_idx - i_range
    # compute gradient contributions
    grad = np.zeros(n, dtype=np.float64)
    # case i==j
    eq = (i_range == j)
    # for i==j, grad[i] = seq[i]
    grad[i_range[eq]] = seq[i_range[eq]]
    # for i!=j, grad[i] = 2 * seq[j]
    not_eq = ~eq
    grad[i_range[not_eq]] = 2.0 * seq[j[not_eq]]
    return grad
```

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _arcsine_sequence(n):
    """Generate the arcsine‑type decreasing sequence of length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _objective(seq):
    """
    Fast evaluation of the AC1 upper bound:
        2·n·max(conv(seq,seq)) / (sum(seq) ** 2)
    Uses an FFT based convolution (O(n log n)).
    """
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_val = conv.max()
    return 2.0 * n * max_val / (total * total)


def _gradient_of_max(seq, max_idx):
    """
    Sub‑gradient of the convolution entry at index `max_idx`
    with respect to `seq`.  For a term (seq * seq)[k] we have
        ∂/∂seq_i = 2·seq_{k‑i}  (i≠k‑i)  and  = seq_i  (i = k‑i).
    """
    n = seq.shape[0]
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = max_idx - i
    grad = np.zeros(n, dtype=np.float64)

    eq_mask = (i == j)
    if np.any(eq_mask):
        grad[i[eq_mask]] = seq[i[eq_mask]]

    neq_mask = ~eq_mask
    if np.any(neq_mask):
        grad[i[neq_mask]] = 2.0 * seq[j[neq_mask]]
    return grad
