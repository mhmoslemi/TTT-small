import numpy as np

def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step sequence with a low AC1 bound.

    The search proceeds in two phases:
      1. Scan a one‑parameter exponential family a_i = exp(-beta*i)
         (including the uniform case beta=0) and its reversal.
      2. Perform cheap stochastic block perturbations starting from the
         best candidate found in phase 1.
    """
    # ----------------------------------------------------------------------
    # Determine problem size from the supplied parent construction.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:               # parent unavailable
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # 1) Scan the exponential family a_i = exp(-beta*i).
    # ----------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # beta range: 0 (uniform) up to a value that makes the tail very small.
    betas = np.linspace(0.0, 0.005, 101)   # step ≈5e‑5
    for beta in betas:
        if beta == 0.0:
            cand = np.ones(n, dtype=np.float64)    # uniform height 1.0
        else:
            cand = np.exp(-beta * idx)

        # Clamp to the allowed interval.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Evaluate forward direction.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

        # Evaluate the reversed sequence (large values at the end).
        rev = cand[::-1]
        score_rev = float(evaluate_sequence(rev.tolist()))
        if np.isfinite(score_rev) and score_rev < best_score:
            best, best_score = rev, score_rev

    # ----------------------------------------------------------------------
    # 2) Light stochastic block optimisation from the best candidate.
    # ----------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    # Parameters for block updates.
    min_width = max(1, n // 200)      # at least 1 element, ≈0.5 % of length
    max_width = max(2, n // 20)       # up to ≈5 % of length
    log_sigma = 0.02                  # log‑normal multiplicative noise

    for _ in range(500):
        cand = best.copy()

        # Choose a random contiguous block and scale it.
        width = int(rng.integers(min_width, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, log_sigma)))
        cand[start:start + width] *= factor

        # Ensure feasibility.
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

    # ----------------------------------------------------------------------
    # Final sanity check.
    # ----------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]
