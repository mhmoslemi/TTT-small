# ----------------------------------------------------------------------
def _arcsine_sequence(n: int) -> np.ndarray:
    """Return the length‑n truncated arcsine sequence a_i = C(2i,i)/4^i."""
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    if n > 1:

        factors = 1.0 - 0.5 / i                        # (1‑0.5/i)
        seq[1:] = np.cumprod(factors)                  # product_{k=1}^i (1‑0.5/k)
    return seq
