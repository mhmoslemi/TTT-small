def _conv_max(arr):
    n = arr.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv.max()

import numpy as np

def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full convolution arr * arr using FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact value of the evaluator (without calling evaluate_sequence)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


def _normalize(arr: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000], then rescale to the required L1‑norm."""
    np.clip(arr, 0.0, 1000.0, out=arr)
    s = arr.sum()
    if s > 0.0:
        arr *= target_sum / s
    # one more clipping in case scaling broke the bound
    np.clip(arr, 0.0, 1000.0, out=arr)
    s2 = arr.sum()
    if s2 > 0.0:
        arr *= target_sum / s2
    return arr


def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """Generate a_i ∝ (i+β)^(-p); optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + base[::-1]
    return base


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """Generate a_i = exp(-lam * i)."""
    return np.exp(-lam * np.arange(n, dtype=np.float64))


def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Multiply a random block by a random factor, keep L1‑norm fixed."""
    cand = vec.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 300), max(2, n // 15)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:start + width] *= factor
    # occasional global Gaussian noise
    if rng.random() < 0.15:
        cand += rng.normal(0.0, 0.005, size=n)
    np.clip(cand, 0.0, 1000.0, out=cand)
    return _normalize(cand, target_sum)


def _swap_two(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two random positions."""
    cand = vec.copy()
    i, j = rng.integers(0, vec.size, size=2)
    cand[i], cand[j] = cand[j], cand[i]
    return cand
