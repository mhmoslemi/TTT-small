def proj_simplex(v, z=1):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _project_to_simplex(v, z=1.0):
    # v: 1D np.ndarray
    # returns w, projection onto {w >= 0, sum(w) = z}
    if np.sum(v) == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        return np.full_like(v, z / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

This is typical algorithm.

def _full_convolution(seq):
    n = seq.size
    L = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    return conv
```

def _objective(seq):
    n = len(seq)
    conv = _full_convolution(seq)
    M = conv.max()
    S = seq.sum()
    return 2.0 * n * M / (S * S)

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _arcsine_sequence(n):
    """Generate the truncated arcsine coefficients."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (1.0 - 0.5 / i)
    return seq


def _project_to_simplex(v, z=1.0):
    """
    Euclidean projection of vector v onto the simplex {w >= 0, sum(w) = z}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.isclose(v.sum(), z) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero, distribute uniformly
        return np.full_like(v, z / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(seq):
    """
    Compute the linear convolution of seq with itself using an FFT of
    length next power of two. Returns the full (2n-1)-length result as real.
    """
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    return conv


def _objective_from_conv(conv, total):
    """Given a full convolution array and total sum, compute the AC1 bound."""
    return 2.0 * len(conv) // 2 * conv.max() / (total * total)

    The routine uses a projected gradient descent on the objective
    2·n·max(a*a)/sum(a)^2 with optional simulated‑annealing steps.
    """

    min_alpha = 1e-4
    decay = 0.5          # backtrack factor
    patience = 0
    patience_limit = 30  # after this many non‑improving steps trigger jitter

    for it in range(max_iter):
        # Compute convolution and locate the maximal entry
        conv_cur = _full_convolution(current)
        M_cur = conv_cur.max()
        k0 = int(np.argmax(conv_cur))
        S_cur = current.sum()

        # Gradient of the maximal convolution entry w.r.t. each coefficient
        grad = np.zeros_like(current)
        lo = max(0, k0 - (n - 1))
        hi = min(n - 1, k0)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k0 - idx
        grad[idx] = current[partner]
        # Diagonal term where i == partner contributes twice
        diag = idx == partner
        if np.any(diag):
            grad[idx[diag]] += current[idx[diag]]

        # Gradient of the full objective O = 2 n M / S^2
        factorA = 2.0 * n / (S_cur * S_cur)
        factorB = 4.0 * n * M_cur / (S_cur * S_cur * S_cur)
        obj_grad = factorA * grad - factorB

        # ----------------------------------------------------------------
        # 4. Line search with projection onto simplex
        # ----------------------------------------------------------------
        step_alpha = alpha
        improved = False
        while step_alpha >= min_alpha:
            cand = current - step_alpha * obj_grad
            cand = _project_to_simplex(cand, 1.0)

            conv_cand = _full_convolution(cand)
            M_cand = conv_cand.max()

                best_seq = cand.copy()
                best_score = float(cand_score)
                current = cand.copy()
                improved = True
                alpha = min(alpha * 1.2, 1.0)
                break

            temp = 1.0 / (it + 1.0)
            if rng.random() < np.exp((best_score - cand_score) / temp):
                current = cand.copy()
                improved = True
                break

            # Reset step size after disturbance
            alpha = 0.7

    # ------------------------------------------------------------------
    # 6. Final safety fallback
    # ------------------------------------------------------------------
    if best_seq.size == 0:

    # Return as a plain Python list of floats
    return [float(x) for x in best_seq]
