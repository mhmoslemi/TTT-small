import numpy as np

def generate_r_sequence(n):
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2*i - 1) / (2*i)
        seq[i] = r
    return seq

def evaluate(seq):
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine length
    parent = globals().get('height_sequence_1')
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256

    # initial sequence: r-sequence
    seq = generate_r_sequence(n)
    np.clip(seq, 0.0, 1000.0, out=seq)

    best_seq = seq.copy()
    best_score = evaluate(best_seq.tolist())
    # if evaluation fails, fallback to uniform
    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)
        best_score = evaluate(best_seq.tolist())

    # parameters
    max_iter = 500
    early_frac = 0.1   # fraction of indices considered early
    late_frac = 0.5    # fraction for late region for receiving mass
    early_end = max(1, int(n * early_frac))
    late_start = int(n * (1 - late_frac))
    if late_start <= early_end:
        late_start = early_end + 1

    # step size function: relative amount of source to transfer
    # we can schedule decreasing step size
    step = 0.05  # start with 5% of source value

    target_score = 1.50274349

    for it in range(max_iter):
        # adapt step size: decays slowly
        step_factor = 0.99 ** it
        curr_step = step * step_factor

        # pick source early index
        i = rng.integers(0, early_end)  # inclusive 0..early_end-1
        # pick target later index
        j = rng.integers(late_start, n)  # inclusive
        # ensure i != j
        if i == j:
            continue
        # propose transfer
        src_val = best_seq[i]
        if src_val <= 0:
            continue
        delta = rng.random() * src_val * curr_step
        # apply transfer
        candidate = best_seq.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # clip
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # evaluate
        score = evaluate(candidate.tolist())
        if np.isfinite(score) and score < best_score:
            best_seq = candidate
            best_score = score
            # optional early stop
            if best_score <= target_score:
                break

    # final fallback if still non-finite
    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_seq]

import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Generate r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def _score(arr: np.ndarray) -> float:
    """Wrap evaluate_sequence to accept a numpy array."""
    return float(evaluate_sequence(arr.tolist()))

def propose_candidate(seed: int = 42, **kwargs):
    """Return a step‑function whose AC1 bound improves on the baseline."""
    rng = np.random.default_rng(seed)

    # --- determine problem size -------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256

    # --- start from the extremal r‑sequence ------------------------------------
    base = generate_r_sequence(n)
    np.clip(base, 0.0, 1000.0, out=base)

    best_seq = base.copy()
    best_score = _score(best_seq)
    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)
        best_score = _score(best_seq)

    # -------------------------------------------------------------------------
    # 1. Deterministic reduction of the first entry (the dominant term)
    # -------------------------------------------------------------------------
    tail_start = n // 2               # distribute mass to the latter half
    tail_len = n - tail_start
    factors = np.linspace(0.70, 1.00, 16)  # scale of a_0, 1.0 = no change

            best_score = sc
