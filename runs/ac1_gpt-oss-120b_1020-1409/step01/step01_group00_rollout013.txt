def _gradient_step(a, step_size=0.01):
    n = len(a)
    s = a.sum()
    b = np.convolve(a, a)
    M = b.max()
    m = b.argmax()
    rev = np.zeros_like(a)
    lo = max(0, m - (n - 1))
    hi = min(m, n - 1)
    # indices i from lo..hi
    rev[lo:hi+1] = a[m - np.arange(lo, hi+1)]
    # update
    a_new = a - step_size * (rev - (M / s))
    a_new = np.clip(a_new, 0.0, 1000.0)
    # renorm to keep sum
    s_new = a_new.sum()
    if s_new > 0:
        a_new *= (s / s_new)
    return a_new

import numpy as np

# Helper: one gradient step that tries to reduce the maximal convolution entry.
def _gradient_step(a: np.ndarray, step_size: float = 0.02) -> np.ndarray:
    n = a.size
    s = a.sum()
    if s == 0.0:
        return a.copy()
    # convolution a*a (full length 2n-1)
    b = np.convolve(a, a)
    M = float(b.max())
    m = int(b.argmax())                     # index where the maximum occurs
    # rev[i] = a[m-i] when the index is in range, else 0
    rev = np.zeros_like(a)
    lo = max(0, m - (n - 1))
    hi = min(m, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        rev[idx] = a[m - idx]
    # gradient of the objective (ignoring the constant factor 4*n/s**2)
    # we move opposite to grad, i.e. a <- a - step * (rev - M/s)
    a_new = a - step_size * (rev - (M / s))
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    # re‑normalise sum (scale does not affect the bound)
    s_new = a_new.sum()
    if s_new > 0.0:
        a_new *= (s / s_new)
    return a_new
