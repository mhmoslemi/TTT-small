def run_opt(a, num_iters):
    best_a = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    step0 = 0.35
    decay = 0.998
    sum_target = 1.0
    for it in range(num_iters):
        conv = np.convolve(a, a)
        cur_score = 2 * n * conv.max() / (a.sum()**2)  # but we use evaluate_sequence later.
        # compute gradient at max conv index
        k = int(np.argmax(conv))
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high+1)
        partner = k - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[partner]

        # step size
        step = step0 * (decay ** it)
        a = a * np.exp(-step * g)
        # clip and re-normalize
        np.clip(a, 0.0, 1000.0, out=a)
        s = a.sum()
        if s == 0 or not np.isfinite(s):
            a.fill(1.0/n)
            s = a.sum()
        a = a / s * sum_target

        # evaluate if better
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional random block perturbation
        if it % 150 == 0 and rng.random() < 0.15:
            width = int(rng.integers(max(1, n // 300), max(2, n // 12)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.15)))
            a_tmp = a.copy()
            a_tmp[start:start+width] *= factor
            np.clip(a_tmp,0.0,1000.0,out=a_tmp)
            a_tmp = a_tmp / a_tmp.sum() * sum_target
            tmp_score = float(evaluate_sequence(a_tmp.tolist()))
            if np.isfinite(tmp_score) and tmp_score < best_score:
                best_score = tmp_score
                best_a = a_tmp.copy()
                a = a_tmp.copy()
    return best_a, best_score
```

import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _clip_and_normalise(a, target_sum=1.0):
    """Clip to [0,1000] and scale to have the given L1 sum."""
    np.clip(a, 0.0, 1000.0, out=a)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        a[:] = 1.0 / a.size
        s = a.sum()
    a /= s
    a *= target_sum
    return a

def _gradient_max_convolution(a, conv, k):
    """Gradient of the convolution entry k (the max entry)."""
    n = a.size
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    idx = np.arange(low, high + 1, dtype=np.int64)
    partner = k - idx
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[partner]
    return g

def _run_opt(initial_a, rng, max_iters=2000):
    """Perform the multiplicative gradient descent with occasional
    random block perturbations. Returns (best_array, best_score)."""
    a = initial_a.astype(np.float64, copy=True)
    a = _clip_and_normalise(a)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = a.size

    # hyper‑parameters (tuned modestly)
    step0 = 0.55          # initial step size
    decay = 0.996         # exponential decay factor
    sum_target = 1.0
    max_exp = 50.0        # safety for exp overflow
    no_imp_early = 0      # counter for early stopping

    for it in range(max_iters):
        # Full self‑convolution
        conv = np.convolve(a, a)
        cur_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_imp_early = 0
        else:
            no_imp_early += 1

        # Identifies the currently maximal convolution entry
        k = int(np.argmax(conv))
        g = _gradient_max_convolution(a, conv, k)

        # Adaptive step
        step = step0 * (decay ** it)

        # Multiplicative update (matrix‑free)
        factor = np.exp(np.clip(-step * g, -max_exp, max_exp))
        a = a * factor
        a = _clip_and_normalise(a, target_sum=sum_target)

        # Random block perturbation every few hundred iterations
        if (it % 250 == 0) and rng.random() < 0.22:
            width = int(rng.integers(max(1, n // 300), max(2, n // 12) + 1))
            start = int(rng.integers(0, n - width + 1))
            mult = float(np.exp(rng.normal(0.0, 0.18)))
            a_pert = a.copy()
            a_pert[start:start + width] *= mult
            a_pert = _clip_and_normalise(a_pert, target_sum=sum_target)
            pert_score = float(evaluate_sequence(a_pert.tolist()))
            if np.isfinite(pert_score) and pert_score < best_score:
                best = a_pert.copy()
                best_score = pert_score
                a = a_pert
                no_imp_early = 0

        # Early stop if no improvement for a long stretch
        if no_imp_early > 800:
            break

    return best, best_score

    """
    rng = np.random.default_rng(seed)
