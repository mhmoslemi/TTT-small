import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _arcsine_sequence(n: int) -> np.ndarray:
    """Generate the classic arcsine coefficients (length n)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative block‑wise perturbation – a cheap global jump."""
    cand = base.copy()
    n = cand.size
    w_min = max(1, n // 250)
    w_max = max(2, n // 40)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    scale = float(np.exp(rng.normal(0.0, 0.15)))
    cand[start:start + width] *= scale
    if rng.random() < 0.3:                      # occasional symmetry
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _objective(seq: np.ndarray, n: int) -> float:
    """Exact AC1 upper‑bound (the evaluator’s formula) for a NumPy array."""
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    return float(2.0 * n * np.max(conv) / (s * s))
