    if top_k > conv.size:
        top_k = conv.size
    idxs = np.argpartition(-conv, top_k)[:top_k]
    sum_g = np.zeros_like(x)
    for k in idxs:
        i_low = max(0, k - (n - 1))

def _make_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2.0 * i)

def _conv_self(x):
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv

def _score(seq):

import numpy as np

# ------------------------------------------------------------
# Helper: central binomial coefficient series (fallback init)
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return a_i = C(2i,i)/4^i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast convolution via real FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 score (mirrors evaluate_sequence)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)


# ------------------------------------------------------------
# Subgradient of the homogeneous objective (averaged over top‑k conv)
# ------------------------------------------------------------
def _subgradient(x: np.ndarray, top_k: int):
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    S = x.sum()
    if S == 0.0:
        return np.zeros_like(x), M, S
    ksize = top_k if top_k < conv.size else conv.size
    ks = np.argpartition(-conv, ksize)[:ksize]
    sum_g = np.zeros_like(x)
    for k in ks:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        sum_g[idx] += x[k - idx]
    g_avg = sum_g / ksize
    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = (2.0 * inv_S2) * g_avg - (2.0 * M) * inv_S3
    return grad, M, S
