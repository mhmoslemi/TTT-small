import numpy as np

def propose_candidate(seed=42, num_iters=1200, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound as evaluated by ``evaluate_sequence``.
    The algorithm:
      1. Build several simple initial candidates (parent, uniform, power‑laws,
         half‑sine, quadratic) and keep the best.
      2. Refine the best candidate with a momentum‑based multiplicative
         gradient descent that uses the top‑k convolution entries as a
         surrogate for the sub‑gradient of the max‑convolution.
      3. Occasionally apply random local perturbations to escape plateaus.
      4. Return the best sequence found.
    The sequence length equals the length of ``height_sequence_1`` when that
    variable exists; otherwise a default length of 2000 is used.
    The returned list is plain Python floats and satisfies the evaluator’s
    constraints (non‑negative, bounded by 1000, sum ≥ 0.01).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and optional parent sequence.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.size)
        parent_ok = (
            parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # ------------------------------------------------------------------
    # Helper to normalise a vector to unit L1 sum.
    # ------------------------------------------------------------------
    def _normalise(v):
        v = np.clip(v, 0.0, 1000.0)
        s = v.sum()
        if s <= 0 or not np.isfinite(s):
            return np.ones_like(v) / v.size
        return v / s

    # ------------------------------------------------------------------
    # Generate a pool of initial candidates.
    # ------------------------------------------------------------------
    candidates = []

    # Parent (if available)
    if parent_ok:
        p = _normalise(parent_arr.astype(np.float64, copy=False))
        candidates.append(p)

    # Uniform
    candidates.append(np.ones(n, dtype=np.float64) / n)

    # Power‑law families a_i ∝ (i+1)^(-p)
    for p in (0.2, 0.5, 0.8, 1.0):
        x = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        candidates.append(_normalise(x))

    # Half‑sine shape
    s = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1.0)))
    candidates.append(_normalise(s))

    # Quadratic (symmetrically peaked)
    idx = np.arange(1, n + 1, dtype=np.float64)
    q = idx * (n - idx + 1.0)
    candidates.append(_normalise(q))

    # ------------------------------------------------------------------
    # Pick the best among the initial pool.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for cand in candidates:
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq = cand.copy()
            best_score = sc

    # Fallback in the unlikely event that no candidate was valid
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Gradient‑descent refinement parameters.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    sum_target = 1.0                     # keep L1 norm fixed (scale‑invariant)
    step0 = 0.5                           # initial step magnitude
    decay = 0.999                         # exponential decay factor per iteration
    mu = 0.9                              # momentum coefficient
    g_mom = np.zeros_like(a)              # momentum accumulator
    top_k = max(5, n // 500)              # number of convolution entries to use

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    for it in range(num_iters):
        # Full convolution (fast FFT internally for this size)
        conv = np.convolve(a, a)
        max_val = conv.max()

        # Determine the (approximately) top‑k indices of conv
        if top_k < conv.size:
            idx_top = np.argpartition(conv, -top_k)[-top_k:]
        else:
            idx_top = np.arange(conv.size, dtype=np.int64)

        # Gradient contribution from the selected indices
        g = np.zeros_like(a)
        for kk in idx_top:
            low = max(0, kk - (n - 1))
            high = min(kk, n - 1)
            idx = np.arange(low, high + 1, dtype=np.int64)
            partner = kk - idx

            g[idx] += 2.0 * a[partner]

        g /= float(len(idx_top))

        a = _normalise(a) * sum_target

            if np.isfinite(pert_score) and pert_score < best_score:

                best_score = pert_score
                a = pert  # continue optimisation from the improvement

    # ------------------------------------------------------------------
    # Return the best sequence discovered as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
