    for step in range(num_steps):
        # compute reverse
        rev = best[::-1]

        top_k = max(1, int(n * 0.02))  # top 2%

        donor_idx = int(rng.choice(donor_candidates))
        if best[donor_idx] <= 0:
            continue
        # choose recipient low rev
        bot_k = max(1, int(n * 0.02))
        recipient_candidates = np.argpartition(rev, bot_k-1)[:bot_k]
        recipient_idx = int(rng.choice(recipient_candidates))
        if donor_idx == recipient_idx:
            continue

        delta = best[donor_idx] * frac
        if delta < 1e-9:
            continue
        cand = best.copy()
        cand[donor_idx] -= delta
        cand[recipient_idx] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score
            # optionally reset frac to larger for exploration

import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i-1] * ((2.0*i_f-1.0)*(2.0*i_f)) / (4.0*i_f*i_f)
    return a

def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    rng = np.random.default_rng(seed)

    try:
        n = int(height_sequence_1.shape[0])
        a = height_sequence_1.astype(np.float64).copy()
    except Exception:
        n = 256
        a = _generate_binomial_sequence(n)

    np.clip(a, 0.0, 1000.0, out=a)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # parameters for directed mass transfers
    frac_start = 0.2          # initial fraction of donor mass to move
    decay = 0.998             # decay factor per iteration
    min_frac = 0.001

    frac = frac_start

    for step in range(num_steps):
        rev = best[::-1]
        # indices with largest and smallest reverse value
        donor_idx = int(np.argmax(rev))
        recipient_idx = int(np.argmin(rev))

        # avoid no‑move cases
        if donor_idx == recipient_idx or best[donor_idx] <= 0.0:
            # fallback to a tiny random pair perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and best[i] > 0.0:
                delta = best[i] * 0.01
                cand = best.copy()
                cand[i] -= delta
                cand[j] += delta
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc
            continue

        delta = best[donor_idx] * frac
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[donor_idx] -= delta
        cand[recipient_idx] += delta

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

        # decay the step size
        frac = max(min_frac, frac * decay)

        # occasional random exploratory step
        if step % 100 == 0:
            i = rng.integers(0, n)

import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """
    Directed mass‑transfer search.
    Starts from the parent vector (or a binomial fallback) and repeatedly
    moves a fraction of the coefficient with the largest reverse‑partner to the
    coefficient with the smallest reverse‑partner, while also stealing a little
    from the first element.  This reduces the largest entry of the autocorrelation
    (which dominates the AC1 bound) without violating the constraints.
    """

    # ------------------------------------------------------------------

    frac_start = 0.20          # fraction of donor mass moved (tail → head)
    decay = 0.998              # exponential decay of that fraction
    min_frac = 0.001           # lower bound for the fraction
    frac = frac_start

    target = 1.50274349        # desired bound (stop early if reached)

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        rev = best[::-1]                         # reverse view (partner values)

        # index with smallest partner but NOT the very first element (to avoid
        # inflating a[0]², which would become the new maximum).
        rev_for_rec = rev.copy()
        rev_for_rec[0] = np.inf
        rec_idx = int(np.argmin(rev_for_rec))

        # alternate between two donor strategies:
        #   even steps: pull from the tail (largest partner)
        #   odd  steps: pull from the head element a[0] itself
        if step % 2 == 0:
            don_idx = int(np.argmax(rev))        # tail donor
            delta_frac = frac
        else:
            don_idx = 0                          # head donor
            # a[0] is the only element whose square can dominate the max;
            # use a slightly larger fraction to shrink it faster.
            delta_frac = max(frac, 0.05)

        # avoid degenerate selections
        if don_idx == rec_idx or best[don_idx] <= 0.0:
            # fall back to a tiny random pair move
            i = rng.integers(0, n)
