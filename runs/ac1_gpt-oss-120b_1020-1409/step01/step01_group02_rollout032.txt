from scipy.optimize import linprog

def _parent_sequence():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:
        pass
    return np.full(256, 1.0)

def _arcsine_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 0.5 / i)
    return a

def _normalize_sum(a, target=1.0):
    s = a.sum()
    if s <= 0:
        return np.full_like(a, target / a.size)
    return a * (target / s)

def _solve_lp(f, M, shifts):
    n = f.size
    K = len(shifts)
    rows = []
    cols = []
    data = []
    for r, k in enumerate(shifts):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        # For i in lo..hi: coefficient for g_{k-i} is f[i]
        # So iterate i, compute j = k - i
        for i in range(lo, hi+1):
            rows.append(r)
            cols.append(k - i)
            data.append(f[i])
    A = coo_matrix((data, (rows, cols)), shape=(K, n)).tocsc()
    b = np.full(K, M)
    c = -np.ones(n)  # maximize sum(g) -> minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        g = res.x
        # Clip to 0-1000 (should be already)
        np.clip(g, 0, 1000, out=g)
        return g
    else:
        return None

def propose_candidate(seed=42, max_iters=1500, **kwargs):
    rng = np.random.default_rng(seed)
    # initial
    parent = _parent_sequence()
    n = parent.size
    arcs = _arcsine_sequence(n)
    # combine: use better score
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    arcs_score = float(evaluate_sequence(arcs.tolist()))
    if arcs_score < best_score:
        best = arcs
        best_score = arcs_score
    # Normalize to unit sum to keep scale invariance
    best = _normalize_sum(best, target=1.0)
    cur = best.copy()
    cur_score = best_score

    # main loop
    for it in range(max_iters):
        # compute convolution and stats
        conv = np.convolve(cur, cur)
        M = conv.max()
        # quick bound check
        cur_score = float(2 * n * M / (cur.sum()**2))
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score

        # pick shifts for LP: top K values
        K = min(8, conv.size)

        top_idx = np.argpartition(conv, -K)[-K:]
        shifts = top_idx.tolist()

            # mix

            # evaluate

            if new_score < best_score:

                best_score = new_score
            # accept new as current if improves or with some prob

            if new_score <= cur_score or rng.random() < 0.1:

                cand[i] -= delta

                cand = _normalize_sum(cand, target=1.0)
                cand_score = float(evaluate_sequence(cand.tolist()))
                if cand_score < best_score:
                    best, best_score = cand, cand_score
                if cand_score < cur_score:
                    cur = cand

    # convert best to list of floats
    if best.size == 0:
        best = np.ones(256, dtype=np.float64) / 256.0
    return [float(x) for x in best]

import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return the supplied parent sequence if it looks valid, else None."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    return None


def _arcsine_sequence(n):
    """
    Truncated arcsine coefficients a_i = C(2i,i)/4^i .
    They satisfy (a*a)_k ≈ 1 for interior k.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _normalize_sum(vec, target=1.0):
    """Scale so that sum(vec) == target (while keeping non‑negative)."""
    s = vec.sum()
    if s <= 0.0:
        return np.full_like(vec, target / vec.size)
    return vec * (target / s)

def propose_candidate(seed=42, max_iters=1200, **kwargs):
    """
    Produce a non‑negative step function (list of floats) whose
    AC1 upper bound is as low as possible.
    The algorithm starts from an arcsine seed (or the provided parent)
    and performs a smooth‑max projected gradient descent with occasional
    random jitter.  All operations are O(n) or O(n log n) and use only
    linear memory.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    if parent is not None:
        n = parent.size
    else:
        n = 256
    # two candidates: parent (if any) and arcsine
    arcs = _arcsine_sequence(n)

    # pick the better of the two as the start point
    cand_parent = parent if parent is not None else arcs
    # both are already non‑negative and bounded
    cand_parent = _normalize_sum(cand_parent, target=1.0)
    arcs = _normalize_sum(arcs, target=1.0)

    # evaluate (using the same formula that evaluate_sequence uses)
    def _score(vec):
        conv = np.convolve(vec, vec)
        return float(2.0 * n * np.max(conv) / (vec.sum() ** 2))

    if _score(arcs) < _score(cand_parent):
        cur = arcs.copy()
    else:
        cur = cand_parent.copy()

    best = cur.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    for it in range(max_iters):
        # full autocorrelation and current score
        conv = np.convolve(cur, cur)
        cur_max = np.max(conv)
        cur_score = float(2.0 * n * cur_max / (cur.sum() ** 2))

        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score

        # ---------- smooth‑max weighted gradient ----------
        #   we treat max as a smooth log‑sum‑exp with temperature β
        beta = 1.0 + 9.0 * (it / max_iters)            # 1 … 10
        diff = conv - cur_max                         # ≤ 0
        # weights proportional to exp(β·diff); clip under‑flow
        w = np.exp(beta * diff, dtype=np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:  # degenerate – fall back to uniform
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0
        w /= w_sum

        # smooth estimate of the maximal autocorrelation value
        smooth_max = np.dot(w, conv)

        # weighted sum of shifted copies of cur:
        #   weighted_a[i] = Σ_k w[k]·cur[k‑i]  (out‑of‑range terms are zero)
        tmp = np.convolve(w, cur[::-1])               # length 3n‑2
        weighted_a = tmp[n - 1 : n - 1 + n]           # extract n entries

        # gradient of 2 n·max / (∑a)² (with ∑a = 1)  → 4 n·(weighted_a – smooth_max·a)
        grad = 4.0 * n * (weighted_a - smooth_max * cur)

        # line‑search (projected back onto the simplex)
        step = 0.2 * (1.0 - it / max_iters) + 0.005   # 0.205 → 0.005
        accepted = False
        while step > 1e-5:
            cand = cur - step * grad
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_sum = cand.sum()
            if cand_sum <= 0.0:
                cand = np.full_like(cand, 1.0 / cand.size)
                cand_sum = 1.0
            cand /= cand_sum

            cand_score = _score(cand)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                accepted = True
                break
            step *= 0.5

        # occasional random jitter to escape tiny plateaus
        if not accepted and rng.random() < 0.25:
            i = rng.integers(0, n)
            j = rng.integers(0, n)

    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.full(256, 1.0 / 256.0, dtype=np.float64)

    # return a plain Python list of finite floats
    return [float(x) for x in best]
