import numpy as np

def _parent_sequence():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:
        pass
    return np.ones(256, dtype=np.float64)

def _arcsine_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 0.5 / i)
    return a

def _full_convolution_fft(a, L):
    # return linear convolution a*a length 2*len(a)-1
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)
    n = a.size
    return conv[:2*n-1]

def propose_candidate(seed=42, num_steps=1500, **kwargs):
    rng = np.random.default_rng(seed)

    parent = _parent_sequence()
    n = parent.shape[0]

    # initial candidates
    # arcsine of same length
    arcsine = _arcsine_sequence(n)
    # evaluate both
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    arcsine_score = float(evaluate_sequence(arcsine.tolist()))
    if arcsine_score < best_score:
        best = arcsine.copy()
        best_score = arcsine_score

    current = best.copy()
    sum_a = float(current.sum())
    # precompute L
    L = 1 << ((2 * n - 1).bit_length())

    # parameter schedule
    transfer_frac = 0.5  # fraction of source mass moved each step
    # main loop
    for step in range(num_steps):
        # compute convolution via FFT
        conv = _full_convolution_fft(current, L)
        max_val = float(conv.max())
        idx_max = int(conv.argmax())

        # compute current score
        cur_score = 2.0 * n * max_val / (sum_a * sum_a)

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = current.copy()

        # adapt transfer fraction: gradually shrink
        if step % 300 == 0 and step > 0:
            transfer_frac = max(0.05, transfer_frac * 0.8)

        # Determine contributing index range
        lo = max(0, idx_max - (n - 1))
        hi = min(idx_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=int)

        # weights = partner values
        partner_vals = current[idx_max - idx_range]
        # choose source with largest partner weight and positive amount
        src_candidates = idx_range[partner_vals > 0]
        if src_candidates.size == 0:
            # fallback random source
            i_src = int(rng.integers(0, n))
        else:
            # find index among src_candidates with max partner weight
            src_weights = partner_vals[partner_vals > 0]
            i_src = int(src_candidates[np.argmax(src_weights)])

        # choose destination from indices that do NOT affect current max (weight zero)
        out_mask = np.ones(n, dtype=bool)
        out_mask[lo:hi+1] = False
        dest_candidates = np.nonzero(out_mask)[0]
        if dest_candidates.size == 0:
            # all indices affect, pick low weight among idx_range
            j_dest = int(idx_range[np.argmin(partner_vals)])
        else:
            # random element among those zero-weight indices
            j_dest = int(rng.choice(dest_candidates))

        if i_src == j_dest:
            continue

        # transfer amount
        max_transfer = current[i_src]
        if max_transfer <= 0:
            continue
        delta = transfer_frac * max_transfer * rng.random()
        if delta <= 0:
            continue

        cand = current.copy()
        cand[i_src] -= delta
        cand[j_dest] += delta
        # clip to bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        # recompute conv and score for candidate
        conv2 = _full_convolution_fft(cand, L)
        max_val2 = float(conv2.max())
        cand_score = 2.0 * n * max_val2 / (sum_a * sum_a)

        # accept if improvement, else SA
        if cand_score < cur_score:
            current = cand
            continue
        # simulated annealing temperature schedule
        temp = 0.01 + 0.99 * (1.0 - step / num_steps)
        if np.exp((cur_score - cand_score) / temp) > rng.random():
            current = cand
            continue

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.02:
            # block perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            block = np.exp(rng.normal(0.0, 0.1))
            cand2 = current.copy()
            cand2[start:start+width] *= block
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            # ensure sum unchanged via scaling
            cand2_sum = cand2.sum()
            if cand2_sum > 0:
                cand2 *= (sum_a / cand2_sum)
            # evaluate
            conv3 = _full_convolution_fft(cand2, L)
            cand_score2 = 2.0 * n * float(conv3.max()) / (sum_a * sum_a)
            if cand_score2 < best_score:
                best = cand2.copy()
                best_score = cand_score2
                current = cand2

    # final fallback if best invalid
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]

def _arcsine_sequence(n):
    """Generate the classical arcsine coefficients of length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _full_convolution_fft(x, L):
    """Linear convolution of x with itself using an FFT of length L."""
    X = np.fft.rfft(x, L)
    conv = np.fft.irfft(X * X, L)
    n = x.shape[0]
    return conv[: 2 * n - 1]

    The algorithm starts from the arcsine sequence (or a parent if better) and

    L = 1 << ((2 * n - 1).bit_length())

    # schedule for the fraction of mass that is moved each step
    transfer_frac = 0.7

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # full autocorrelation (via FFT) and current worst entry
        conv = _full_convolution_fft(current, L)
        max_val = float(conv.max())
        idx_max = int(conv.argmax())

        # current AC1 score (exactly the same expression used by the evaluator)
        cur_score = 2.0 * n * max_val / (sum_a * sum_a)

        # keep best seen so far
        if cur_score < best_score:
            best_score = cur_score
            best = current.copy()

        # slowly shrink the transfer fraction – coarse moves early, fine later
        if step % 300 == 0 and step > 0:
            transfer_frac = max(0.05, transfer_frac * 0.8)

        # ------------------------------------------------------------------
        # Determine indices that actually influence the maximal entry
        # ------------------------------------------------------------------
        lo = max(0, idx_max - (n - 1))
        hi = min(idx_max, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=int)

        # weight = partner value a[idx_max - i]; larger weight ⇒ larger
        # contribution to the maximal convolution entry
        partner_vals = current[idx_max - idx_range]

        # pick a source index with the largest partner weight (and non‑zero mass)
        src_mask = partner_vals > 0.0
        if src_mask.any():
            src_idx = idx_range[np.argmax(partner_vals * src_mask)]
        else:
            src_idx = int(rng.integers(0, n))

        # choose a destination index that does *not* affect this entry
        out_mask = np.ones(n, dtype=bool)
        out_mask[lo : hi + 1] = False
        zero_weight_candidates = np.nonzero(out_mask)[0]

        if zero_weight_candidates.size == 0:
            # all indices influence the max – fall back to the smallest weight
            dst_idx = int(idx_range[np.argmin(partner_vals)])
        else:
            dst_idx = int(rng.choice(zero_weight_candidates))

        if src_idx == dst_idx:
            continue

        # ------------------------------------------------------------------
        # Propose a mass transfer
        # ------------------------------------------------------------------
        max_transfer = float(current[src_idx])
        if max_transfer <= 0.0:
            continue

        delta = transfer_frac * max_transfer * rng.random()
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[src_idx] -= delta
        cand[dst_idx] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        # recompute sum (clipping can change it) and normalise back to the
        # original total mass – scaling does not affect the bound but keeps
        # numbers well behaved
        cand_sum = float(cand.sum())
        if cand_sum > 0.0:
            cand *= (sum_a / cand_sum)

        conv2 = _full_convolution_fft(cand, L)
        max_val2 = float(conv2.max())
        cand_score = 2.0 * n * max_val2 / (sum_a * sum_a)

        # ------------------------------------------------------------------
        # Acceptance: strictly better or Metropolis‐type uphill move
        # ------------------------------------------------------------------
        if cand_score < cur_score:
            current = cand
            sum_a = float(current.sum())
            continue

        temp = 0.01 + 0.99 * (1.0 - step / float(num_steps))
        if np.exp((cur_score - cand_score) / temp) > rng.random():
            current = cand
            sum_a = float(current.sum())
            continue

        if rng.random() < 0.02:

                best = block.copy()
                best_score = block_score

    # final safety fallback
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
