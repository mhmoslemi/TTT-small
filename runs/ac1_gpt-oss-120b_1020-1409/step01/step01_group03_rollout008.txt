import numpy as np

# ------------------------------------------------------------
# Convolution via FFT (real input, linear convolution)
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 bound (mirrors the evaluator)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    m = _conv_self(seq).max()
    return 2.0 * seq.size * m / (s * s)


# ------------------------------------------------------------
# Classic decreasing series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a

# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 2500, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    previous bound.  The routine builds a few simple analytic candidates,
    picks the best, then runs a long projected‑gradient / random‑perturbation
    loop (matrix‑free, O(n log n) per iteration).  The best sequence seen
    is finally returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size (use the provided parent if present)
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 1024  # safe fallback length

    # --------------------------------------------------------
    # Build a small portfolio of analytic starts
    # --------------------------------------------------------
    candidates = []

    # (a) the classic binomial series (decreasing)
    s1 = _make_series(n)
    candidates.append(s1)

    # (b) reversed series (increasing) – sometimes helpful
    candidates.append(s1[::-1].copy())

    # (c) a smooth half‑sine shape (positive, roughly unimodal)
    idx = np.arange(n, dtype=np.float64)
    s3 = np.sin(np.pi * (idx + 0.5) / (2.0 * n))
    candidates.append(s3)

    # (d) use the parent if it exists
    if parent is not None:
        candidates.append(parent)

    # Evaluate and keep the best analytic start
    best = None
    best_score = float('inf')
    for cand in candidates:
        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # --------------------------------------------------------
    # Helper: one projected gradient step (focuses on the maximal
    # convolution entry). Returns a candidate and its score.
    # --------------------------------------------------------
    def _grad_step(x: np.ndarray):
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        S = x.sum()
        if S == 0.0:
            return x, float('inf')

        # Build the gradient of M (ordered pairs) for the max index
        g = np.zeros_like(x)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] = x[k - idx]

        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

        # Simple back‑tracking line search (up to 6 trials)
        lr = 1.0
        best_local = x
        best_local_score = _score(x)
        for _ in range(6):
            cand = x - lr * grad
            np.clip(cand, 0.0, 1000.0, out=cand)
            if rng.random() < 0.12:                     # occasional symmetrisation
                cand = 0.5 * (cand + cand[::-1])
            sc = _score(cand)
            if sc < best_local_score:
                best_local = cand
                best_local_score = sc
                break
            lr *= 0.5
        return best_local, best_local_score

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    x = best.copy()
    stagn = 0                     # number of iterations without improvement

    for step in range(num_steps):
        # 1) Gradient step
        cand, cand_score = _grad_step(x)

        # 2) Random block perturbation (every few iterations)
        if step % 13 == 0:
            # block width is chosen roughly logarithmically
            width = max(1, int(n / rng.integers(12, 28) + 0.5))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.09))
            block = cand.copy()
            block[start:start + width] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            scb = _score(block)
            if scb < cand_score:
                cand, cand_score = block, scb

        # 3) Light symmetrisation on a longer schedule
        if step % 31 == 0 and rng.random() < 0.22:
            cand = 0.5 * (cand + cand[::-1])
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand_score = _score(cand)

        # 4) Global best update
        if cand_score + 1e-14 < best_score:
            best = cand.copy()
            best_score = cand_score
            stagn = 0
        else:
            stagn += 1

        x = cand

        # early stop if we appear stuck for a long time
        if stagn > 2000:
            break

    # --------------------------------------------------------
    # Fallback safety: guarantee a non‑empty, finite list
    # --------------------------------------------------------
    if best is None or best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(v) for v in best]
