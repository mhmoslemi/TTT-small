def _score(seq, L):
    # seq is np.ndarray float64, non-negative.
    n = seq.shape[0]
    # compute convolution via FFT
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)

    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

import numpy as np
import math

# ------------------------------------------------------------
# Helper to compute the AC1 upper bound (exactly as evaluate_sequence)
# ------------------------------------------------------------
def _score(seq: np.ndarray, L: int) -> float:
    """Return 2*n*max(conv(seq,seq))/(sum(seq)**2).  seq must be non‑negative."""
    n = seq.shape[0]
    # Linear convolution via FFT (zero‑padded to avoid wrap‑around)
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    max_b = float(conv[: 2 * n - 1].max())
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a list of non‑negative coefficients whose AC1 bound is low."""
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Determine length and initial sequence (use the provided parent when present)
    # -----------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])          # parent variable supplied at run‑time
        init_seq = height_sequence_1.astype(np.float64).copy()
    except Exception:
        # fallback: modest inverse‑square‑root shape (sum normalised)
        n = 256
        init_seq = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 1.0)
        init_seq /= init_seq.sum()

    np.clip(init_seq, 0.0, 1000.0, out=init_seq)

    # ------------------------------------------------------------
    # Pre‑compute the padded length for FFT‑based convolution
    # ------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # ------------------------------------------------------------
    # Initialise current and best solutions
    # ------------------------------------------------------------
    best = init_seq.copy()
    best_score = _score(best, L)

    if not np.isfinite(best_score):
        best = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score(best, L)

    cur = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------
    total_iters = 5000
    max_width = max(1, n // 100)          # for block perturbations
    target = 1.50274349                  # stop early if we beat this

    for it in range(total_iters):
        cand = cur.copy()

        # ----- Choose a stochastic move -------------------------------------------------
        if rng.random() < 0.7:   # mass‑transfer move (most common)
            i = rng.integers(n)
            # ensure j != i
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if cand[i] > 0.0:
                # move up to 40 % of the source entry
                delta = cand[i] * rng.random() * 0.4
                cand[i] -= delta

        else:    # modest block scaling move (exploratory)
            width = int(rng.integers(1, max_width + 1))
