import numpy as np

# ----------------------------------------------------------------------
# Helper: convolution maximum via FFT (exact for non‑negative real data)
_fft_cache = {}  # cache of FFT sizes for each length we encounter

def _max_conv(arr: np.ndarray, n: int) -> float:
    """Return max value of full convolution arr*arr using FFT (no padding beyond needed)."""
    if n not in _fft_cache:
        # size = next power of two >= 2n-1
        _fft_cache[n] = 1 << ((2 * n - 1).bit_length())
    size = _fft_cache[n]
    # Real FFT, multiply, inverse FFT (real)
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale non‑negative vector to have L1‑norm = target_sum."""
    s = vec.sum()
    if s == 0.0:
        # avoid division by zero; return a tiny uniform vector
        return np.full_like(vec, target_sum / vec.size)
    vec = vec * (target_sum / s)
    # values stay far below the 1000 bound for the lengths we handle
    return vec

# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p) ; optionally symmetrised by adding the reversed copy."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base

# ----------------------------------------------------------------------
def _make_exponential(n: int, lam: float, sym: bool) -> np.ndarray:
    """a_i ∝ exp(-λ i) ; optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = np.exp(-lam * idx)
    if sym:
        base = base + np.exp(-lam * (n - 1 - idx))
    return base

# ----------------------------------------------------------------------
def _make_window(n: int, kind: str, **kwargs) -> np.ndarray:
    """Generate a standard window and optionally raise it to a power."""
    if kind == "hann":
        w = np.hanning(n)
    elif kind == "hamming":
        w = np.hamming(n)
    elif kind == "blackman":
        w = np.blackman(n)
    elif kind == "kaiser":
        beta = kwargs.get("beta", 0.0)
        w = np.kaiser(n, beta)
    else:
        raise ValueError(f"unknown window {kind}")
    power = kwargs.get("power", 1.0)
    if power != 1.0:
        w = np.power(w, power)
    return w

# ----------------------------------------------------------------------
def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Random local multiplicative perturbation of a contiguous block, then renormalise."""
    n = vec.size
    cand = vec.copy()
    # choose a block width between ~0.5% and ~5% of n
    min_w = max(2, n // 200)
    max_w = max(min_w + 1, n // 20)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    # log‑normal factor
    factor = float(np.exp(rng.normal(0.0, 0.08)))
    cand[start : start + width] *= factor
    # optional mirroring to encourage symmetry
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])
    # renormalise to keep the L1‑norm fixed
    cand = _normalize(cand, target_sum)
    return cand

# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that beats the
    previous AC1 upper bound.  The routine
    * determines the problem size from `height_sequence_1` if it exists,
    * examines a diverse family of analytic shapes (power‑law, exponential,
      classic windows),
    * refines the best candidate with thousands of cheap random block
      perturbations,
    * returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n (use provided parent if valid)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
        use_parent = True
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)
        use_parent = False

    target_sum = np.sqrt(2.0 * n)   # normalisation used by the official evaluator

    # ------------------------------------------------------------------
    # Baseline uniform candidate (score = 2.0)
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    best_vec = uniform.copy()
    best_score = _max_conv(best_vec, n)   # with correct normalisation this equals the evaluator's score

    # ------------------------------------------------------------------
    # Incorporate the provided parent (if any) after normalisation
    if use_parent:
        cand = _normalize(np.clip(parent, 0.0, 1000.0), target_sum)
        sc = _max_conv(cand, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # 1) Analytic families ------------------------------------------------
    # Power‑law family (both asymmetric and symmetric)
    p_vals = np.linspace(0.30, 1.00, 15)          # exponent
    beta_vals = np.logspace(-2.0, 2.0, 10)       # shift
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                base = _make_powerlaw(n, p, beta, sym)
                if base.sum() == 0.0:
                    continue
                cand = _normalize(base, target_sum)
                sc = _max_conv(cand, n)
                if sc < best_score:
                    best_score = sc
                    best_vec = cand.copy()

    # Exponential decay family
    lam_vals = np.logspace(-4.0, -1.0, 8)        # decay rate
    for sym in (False, True):
        for lam in lam_vals:
            base = _make_exponential(n, lam, sym)
            if base.sum() == 0.0:
                continue
            cand = _normalize(base, target_sum)
            sc = _max_conv(cand, n)
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()

    # Classical windows (Hann, Hamming, Blackman, Kaiser) possibly raised to powers
    window_kinds = ["hann", "hamming", "blackman"]
    for kind in window_kinds:
        for power in (0.5, 1.0, 1.5, 2.0):
            w = _make_window(n, kind, power=power)
            cand = _normalize(w, target_sum)
            sc = _max_conv(cand, n)
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()

    # Kaiser window with a sweep of betas
    for beta in np.linspace(0.0, 15.0, 12):
        w = _make_window(n, "kaiser", beta=beta)
        cand = _normalize(w, target_sum)
        sc = _max_conv(cand, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # 2) Stochastic local refinement ---------------------------------------
    # Perform many cheap block perturbations starting from the best vector.
    # The number of iterations is chosen to stay comfortably within the
    # typical time budget for a few thousand FFTs.
    for _ in range(4000):
        cand = _block_perturb(best_vec, rng, target_sum)
        sc = _max_conv(cand, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # Final safety: the evaluator will clip and renormalise anyway, but
    # we ensure the output respects the simple deterministic contract.
    final = [float(x) for x in best_vec]
    return final
