import numpy as np

def _linear_convolve(seq, fft_len, conv_len):
    """Linear convolution of seq with itself using FFT (matrix‑free)."""
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[:conv_len]

def _clip_and_normalize(arr):
    """Enforce evaluator bounds and renormalise to sum = 1."""
    np.clip(arr, 0.0, 1000.0, out=arr)
    s = arr.sum()
    if s > 0.0:
        arr /= s
    return arr
