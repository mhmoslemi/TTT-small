def subgradient_max_convolution(a):
    n = a.shape[0]
    # compute convolution via FFT
    L = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[:2*n-1]
    M = conv.max()
    # indices of max
    idxs = np.flatnonzero(np.isclose(conv, M))
    g = np.zeros_like(a)
    for k in idxs:
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        for i in range(i_min, i_max+1):
            j = k - i
            if i > j:
                continue
            if i == j:
                g[i] += 2*a[i]
            else:
                g[i] += a[j]
                g[j] += a[i]
    return M, g
```

def descent(a, step_factor=0.5, max_iter=1000):
    n = a.shape[0]
    S0 = max(a.sum(), 0.01)
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    eta0 = step_factor * (S0 / n)
    for t in range(1, max_iter+1):
        M, g = subgradient_max_convolution(a)
        eta = eta0 / (1 + t * 0.001)  # simple decay
        a = a - eta * g
        a = np.clip(a, 0.0, 1000.0)
        # Rescale sum to S0 to avoid drift
        sum_a = a.sum()
        if sum_a > 0:
            a *= S0 / sum_a
        # occasional random perturbation
        if t % 200 == 0:
            # multiply a random block
            width = max(1, n // 100)
            start = np.random.randint(0, n-width+1)
            factor = np.exp(np.random.normal(0.0, 0.1))
            a[start:start+width] *= factor
            a = np.clip(a, 0.0, 1000.0)
            # rescale again
            sum_a = a.sum()
            if sum_a > 0:
                a *= S0 / sum_a
        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def descent(a, max_iter=1500, step_factor=0.3):
    n = a.shape[0]
    best_a = a.copy()
    sum_a = best_a.sum()
    best_score = 2 * n * (full_convolution(best_a).max()) / (sum_a**2)  # approximate
    eta0 = step_factor * (sum_a / n)  # average element magnitude times step_factor
    for t in range(1, max_iter+1):

        a = a - eta * g
        a = np.clip(a, 0.0, 1000.0)
        # occasional random block scaling
        if t % 250 == 0:

            start = np.random.randint(0, n - width + 1)
            factor = np.exp(np.random.normal(scale=0.1))
            a[start:start+width] *= factor

            best_a = a.copy()
    return best_a, best_score

import numpy as np

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Convolution a*a using FFT (linear convolution, length 2n‑1)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_max_convolution(a: np.ndarray):
    """
    Return the current maximum value of a*a and a sub‑gradient of the
    max‑operator with respect to a (i.e. derivative of max_k (a*a)_k).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    M = conv.max()
    # tolerance for floating‑point ties
    atol = max(1e-12, 1e-12 * abs(M))
    idxs = np.flatnonzero(np.abs(conv - M) <= atol)

    g = np.zeros_like(a)
    for k in idxs:
        i0 = max(0, k - (n - 1))
        i1 = min(k, n - 1)
        js = np.arange(i0, i1 + 1, dtype=np.int64)
        # derivative contribution 2*a_{k-j}
        g[js] += 2.0 * a[k - js]
    return M, g


def _descent(a: np.ndarray, rng: np.random.Generator,
             max_iter: int = 1500, step_factor: float = 0.4):

    Simple projected sub‑gradient descent on the quantity
    score = 2·n·max(a*a) / (sum(a)^2).  The descent keeps a
    feasible (non‑negative, bounded) and rescales occasionally.
    """
    n = a.shape[0]
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        a = np.full(n, 0.01, dtype=np.float64)
        sum_a = a.sum()

    M, _ = _subgradient_max_convolution(a)
    best_a = a.copy()
