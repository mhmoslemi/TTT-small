            seq = np.empty(n, dtype=np.float64)
            r = 1.0
            seq[0] = r
            for i in range(1,n):
                r = r * (2*i-1)/(2*i)
                seq[i] = r
            return seq

import numpy as np

def generate_r_sequence(n):
    seq = np.empty(n,float)
    r = 1.0
    seq[0] = r
    for i in range(1,n):
        r = r * (2*i -1) / (2*i)
        seq[i] = r
    return seq

def propose_candidate(seed=42, **kwargs):
    import random, math
    random.seed(seed)
    np.random.seed(seed)

    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size>0:
        seq = parent.astype(np.float64).copy()
        n = len(seq)
    else:
        n = 256
        seq = generate_r_sequence(n)

    np.clip(seq,0.0,1000.0,out=seq)

    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # exploration parameters
    fractions = [0.15, 0.2, 0.25, 0.3]
    epsilons = [0.001, 0.002, 0.005, 0.01, 0.02]

    for frac in fractions:
        tail_start = int(n * (1 - frac))
        if tail_start >= n:
            continue
        for eps in epsilons:
            cand = best_seq.copy()
            cand[tail_start:] += eps
            # clip after addition
            np.clip(cand,0.0,1000.0,out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand

    # Simple local refinement: try moving a tiny epsilon from a tail index to a near tail index to reduce self-convolution (optional)
    # but we can skip.

    # fallback if something went wrong
    if not np.isfinite(best_score):

    # return as plain list
    return best_seq.tolist()

import numpy as np

def _r_sequence(n: int) -> np.ndarray:
    """Catalan‑type extremal sequence: r_i = C(2i,i)/4^i."""
    out = np.empty(n, dtype=np.float64)
    r = 1.0
    out[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        out[i] = r
    return out


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 upper bound
    improves on the reference solution. The routine works as follows:

    1. Use the supplied `height_sequence_1` if present, otherwise fall back to
       the extremal r‑sequence (length 256).
    2. Clip to the allowed interval [0,1000].
    3. Try a small family of modifications that add a tiny constant `eps`
       to a suffix (the “tail”) of the sequence.  Adding mass far in the
       tail raises the total sum while increasing the maximal convolution
       only modestly (self‑convolution of a sparse tail is O(eps²·|tail|)).
       All candidates are evaluated with the provided `evaluate_sequence`.
    4. Keep the best candidate found; fall back to a uniform vector of ones
       if something goes wrong.
    """
    import random

    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise from the parent (if any) or from the r‑sequence.
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        seq = parent.astype(np.float64).copy()
        n = int(seq.shape[0])
    else:
        n = 256
        seq = _r_sequence(n)

    # enforce the hard bounds required by the evaluator
    np.clip(seq, 0.0, 1000.0, out=seq)

    # ------------------------------------------------------------------
    # 2.  Baseline score.
    # ------------------------------------------------------------------
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 3.  Search over a tiny grid of tail‑fractions and epsilons.
    #    Adding to the tail raises the sum but only adds at most
    #    eps to the max‑convolution (plus a small eps²·|tail| term).
    # ------------------------------------------------------------------
    # fractions of the whole length that will be turned into a tail
    tail_fractions = [0.15, 0.20, 0.25, 0.30]
    # tiny constants to add – all well below the 1000 ceiling
    epsilons = [0.001, 0.002, 0.005, 0.010, 0.020]

    for frac in tail_fractions:
        # start index of the tail (rounded down, always at least 1)
        tail_start = max(1, int(n * (1.0 - frac)))
        if tail_start >= n:
            continue
        for eps in epsilons:
            cand = best_seq.copy()
            cand[tail_start:] += eps
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_seq = cand
                best_score = score

    # ------------------------------------------------------------------
    # 4.  Safety fallback – should never be needed.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best_seq = np.ones_like(seq, dtype=np.float64)

    # Return a plain Python list as required by the harness.
    return best_seq.tolist()
