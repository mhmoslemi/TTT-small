        partners = seq[idx_max - idxs]                 # gradient components

        # source – the coefficient with the largest partner (most impact)
        i_source = lo + int(np.argmax(partners))

        # target – an index outside the influence range if possible
        if lo > 0:
            j_target = lo - 1
        elif hi < n - 1:
            j_target = hi + 1
        else:                                          # full range – pick smallest partner
            j_target = int(np.argmin(partners))

        if i_source == j_target:                       # fallback to a random swap
            j_target = int(rng.integers(0, n))

        # ----------------------------------------------------------
        # Compute a safe transfer amount that would (approximately)
        # bring the current maximal entry down to the allowed level.
        # ----------------------------------------------------------
        partner_max = partners.max()
        needed = max_val - max_allowed
        delta = needed / (partner_max + 1e-12)          # linear approximation
        delta = min(delta * 0.9, seq[i_source] * 0.5)   # stay conservative
        if delta <= 0.0:
            delta = seq[i_source] * 0.1

        # ----------------------------------------------------------
        # Apply the transfer, project onto [0,1000] and evaluate.
        # ----------------------------------------------------------
        cand = seq.copy()
        cand[i_source] -= delta
        cand[j_target] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        conv_cand = _conv_fft(cand)
        max_cand = conv_cand.max()
        score_cand = _score(cand, max_cand)

        if score_cand < best_score:
            # accept improvement
            seq = cand
            best_seq = cand.copy()
            best_score = score_cand
            max_allowed = target_score * (seq.sum() ** 2) / (2.0 * n)
            continue

        # occasional uphill move to escape shallow plateaus
        if rng.random() < 0.01:
            seq = cand

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64)

    # Return as a plain Python list of floats
    return [float(x) for x in best_seq]
