def compute_ac1_bound(seq, fft_len=None, fft_seq=None):
   # seq: 1d np.ndarray of float64.
   n = seq.shape[0]
   if fft_len is None:
       m = 1

   bound = 2.0 * n * max_b / (sum_a * sum_a) if sum_a > 0 else np.inf
   return bound

        return bnd, conv

import numpy as np
import math

def generate_r_sequence(n: int) -> np.ndarray:
    """Catalan‑type r‑sequence: r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def compute_bound_and_conv(seq: np.ndarray, fft_len: int):
    """Return the AC1 bound and the full convolution of seq with itself."""
    n = seq.shape[0]
    # convolution via real FFT (linear convolution with zero padding)
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a <= 0.0:
        bound = np.inf
    else:
        bound = 2.0 * n * max_b / (sum_a * sum_a)
    return float(bound), conv


def find_max_move(seq: np.ndarray, conv: np.ndarray):
    """
    Pick an index i that most contributes to the maximal convolution entry
    and a distinct index j having the smallest coefficient.
    Returns (i, j, idx_max).
    """
    n = seq.shape[0]
    idx_max = int(np.argmax(conv))

    # indices that actually appear in the max entry
    start = max(0, idx_max - (n - 1))
    end = min(idx_max, n - 1)
    i_vals = np.arange(start, end + 1, dtype=np.int64)
    j_vals = idx_max - i_vals
    contrib = seq[i_vals] * seq[j_vals]

    i_max = int(i_vals[int(np.argmax(contrib))])

    # choose a destination with the smallest value (avoid i_max itself)
    j_min = int(np.argmin(seq))
    if j_min == i_max:
        # second smallest if the absolute minimum coincides with i_max
        sorted_idx = np.argsort(seq)
        for idx in sorted_idx:
            if idx != i_max:
                j_min = int(idx)
                break
    return i_max, j_min, idx_max

    """Return a step‑function (list of non‑negative heights) whose AC1 bound
    is improved over the simple r‑sequence.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        seq = parent.astype(np.float64).copy()
    else:
        # fallback length – any positive length works; 256 keeps runtime tiny.
        n_fallback = 256
        seq = generate_r_sequence(n_fallback).astype(np.float64)

    np.clip(seq, 0.0, 1000.0, out=seq)
    n = seq.shape[0]

    # ------------------------------------------------------------------
    # 2. Pre‑compute a power‑of‑two FFT length for the convolution.
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = seq
    best_score, best_conv = compute_bound_and_conv(best_seq, fft_len)

    # ------------------------------------------------------------------
    # 4. Simple greedy / simulated‑annealing style improvement.
    # ------------------------------------------------------------------
    target_bound = 1.50274349          # wanted improvement threshold
    max_iters = 5000
    no_improve = 0                     # stop early if stagnating

    for it in range(max_iters):
        if best_score <= target_bound:

        i_src, j_dst, _ = find_max_move(best_seq, best_conv)
