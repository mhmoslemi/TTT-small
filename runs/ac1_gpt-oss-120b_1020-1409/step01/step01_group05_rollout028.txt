        if sc < best_score:
            best = a.copy()
            best_score = sc

    # --------------------------------------------------------------
    # Refine the current best using a multi‑index multiplicative update.
    # --------------------------------------------------------------
    a = best.copy()
    step0 = 0.35                # initial step size
    decay = 0.995               # geometric decay of the step
    no_improve = 0
    tol_rel = 1e-3              # tolerance for “active” convolution entries

    for it in range(num_iters):
        conv = np.convolve(a, a)                 # full convolution
        max_val = conv.max()

        # indices whose convolution value is within tol_rel of the maximum
        active = np.where(conv >= max_val * (1.0 - tol_rel))[0]
        if active.size == 0:
            active = np.array([int(np.argmax(conv))], dtype=np.int64)

        # gradient of the (smoothed) maximum w.r.t. a
        g = np.zeros_like(a)
        w = 1.0 / active.size                     # equal weight for each active k
        for k in active:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx_range = np.arange(low, high + 1, dtype=np.int64)
            partner = k - idx_range
            g[idx_range] += w * 2.0 * a[partner]

        # multiplicative update, stays non‑negative
        step = step0 * (decay ** it)
        a = a * np.exp(-step * g)

        # re‑normalise to keep sum = 1 and respect bounds
        a = _normalise(a)

        cur_score = float(evaluate_sequence(a.tolist()))
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
        else:
            no_improve += 1

        # occasional random block perturbation to escape plateaux
        if (it % 150 == 0) and rng.random() < 0.15:
            width = int(rng.integers(max(1, n // 300), max(2, n // 20)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.10)))
            pert = a.copy()
            pert[start:start + width] *= factor
            pert = _normalise(pert)
            pert_score = float(evaluate_sequence(pert.tolist()))
            if pert_score < best_score:
                a = pert
                best = pert.copy()
                best_score = pert_score
                no_improve = 0

        # long stagnation → re‑initialise from a random power‑law
        if no_improve >= 500:
            p = rng.uniform(0.1, 1.2)
            a = _normalise(idx ** (-p))
            no_improve = 0

    # --------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # --------------------------------------------------------------
    return [float(x) for x in best]
