def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Return full convolution (seq * seq) using FFT (real, size = next power of two)."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft = np.fft.rfft(seq, n=size)
    prod = fft * fft
    conv = np.fft.irfft(prod, n=size)[: 2 * n - 1]
    return conv

    # power‑law with exponent 0.6 (a little steeper than 0.5)
    alpha = 0.6
    pw = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** alpha)
    pw /= pw.sum()
    cand.append(pw)

    # random slight perturb of uniform (to break symmetry)
    u = np.full(n, 1.0 / n, dtype=np.float64)
    width = max(1, n // 200)
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.04)))
    u[start:start + width] *= factor
    u = np.clip(u, 0.0, 1000.0)
    u /= u.sum()
    cand.append(u)

    return cand


def _hill_climb(start: np.ndarray, rng: np.random.Generator, max_iter: int = 1800) -> tuple:
    """Simple local search that repeatedly reduces the dominant convolution term."""
    seq = start.copy()
    best = seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = seq.size
    no_improve = 0

    for it in range(max_iter):
        # full convolution and index of its maximum
        conv = _conv_full(seq)
        max_idx = int(conv.argmax())

        # contributions of each position to that maximal entry
        i = np.arange(n, dtype=np.int64)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        contrib = np.empty_like(seq)
        contrib.fill(0.0)
        contrib[mask] = seq[i[mask]] * seq[j[mask]]

        # pick a position that contributes the most and tweak it
        idx = int(np.argmax(contrib))

        # propose a log‑normal multiplicative change
        factor = float(np.exp(rng.normal(0.0, 0.02)))
        cand = seq.copy()
        cand[idx] *= factor
        cand = np.clip(cand, 0.0, 1000.0)

        # renormalise (objective is scale‑invariant)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            seq = cand
            no_improve = 0
            continue

        # occasional stochastic escape
        no_improve += 1
        if no_improve > 250:
            # a random block perturbation (global move)
            width = max(1, n // 100)
            start_pos = int(rng.integers(0, n - width + 1))
            blk_factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand = seq.copy()
            cand[start_pos:start_pos + width] *= blk_factor
            cand = np.clip(cand, 0.0, 1000.0)
            cand /= cand.sum()
            esc_score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(esc_score) and esc_score < best_score:
                best = cand
                best_score = esc_score
                seq = cand
                no_improve = 0
            else:
                no_improve = 0  # reset counter even if not accepted

    return best, best_score


# ---------- main entry point ----------
def propose_candidate(seed: int = 42, **kwargs) -> list:
    rng = np.random.default_rng(seed)

    # Determine length and (optionally) a parent sequence
    try:
        n = int(height_sequence_1.shape[0])
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() < 0.01:
            parent = None
    except Exception:
        n = 256
        parent = None

    # Build a pool of starters
    starters = []
    if parent is not None:
        p = parent.copy()
        p /= p.sum() if p.sum() > 0 else 1.0
        starters.append(p)

    starters.extend(_initial_candidates(n, rng))

    # Choose the best initial candidate
    best_seq = None
    best_score = np.inf
    for cand in starters:
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq = cand
            best_score = sc

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Run a focused hill‑climb from the best start
    max_iters = 2000 if n > 2000 else 1200
    best_seq, best_score = _hill_climb(best_seq, rng, max_iter=max_iters)

    result = best_seq.tolist()
    # Defensive guarantee – never return an empty or non‑finite list
    if (not result) or any((not np.isfinite(x) for x in result)):
        result = [1.0 / n] * n
    return result
