def _conv_max(arr):
    n = arr.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv.max()
```

import numpy as np

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full linear convolution arr*arr via FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())            # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray) -> float:
    """Exact value of the official evaluator (without calling evaluate_sequence)."""
    n = arr.shape[0]
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))

    p_vals = np.linspace(0.2, 1.2, 21)                     # exponent
    beta_vals = np.logspace(-3, 2, 15)                    # offset
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                base = (idx + beta) ** (-p)
                if sym:
                    base = base + (idx[::-1] + beta) ** (-p)
                s = base.sum()
                if s == 0.0:
                    continue
                cand = base * (target_sum / s)
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _score(cand)
                if sc < best_score:
                    best_score = sc
                    best_vec = cand.copy()

    # --------------------------------------------------------------
    # 3) Simple linear‑decreasing shape.
    lin = np.linspace(1.0, 0.1, n, dtype=np.float64)
    cand = lin * (target_sum / lin.sum())
    np.clip(cand, 0.0, 1000.0, out=cand)
    sc = _score(cand)
    if sc < best_score:
        best_score = sc
        best_vec = cand.copy()

    # --------------------------------------------------------------
    # 4) Stochastic local refinement around the current best.
    for _ in range(1500):
        # pick a random block
        width = int(rng.integers(max(2, n // 200), max(5, n // 20)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.05)))    # multiplicative noise
        cand = best_vec.copy()
        cand[start:start + width] *= factor
        # occasional mirroring (helps symmetry)
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        np.clip(cand, 0.0, 1000.0, out=cand)
        # renormalise to the required L1‑norm
        s = cand.sum()
        if s > 0.0:
            cand *= target_sum / s
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # --------------------------------------------------------------
    # 5) Try the original parent (if present) after normalisation.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            s = parent.sum()
            if s > 0.0:
                cand = parent * (target_sum / s)
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _score(cand)
                if sc < best_score:
                    best_score = sc
                    best_vec = cand.copy()
    except Exception:
        pass

    # --------------------------------------------------------------
    # Return the best candidate as a plain Python list.
    return [float(v) for v in best_vec]
