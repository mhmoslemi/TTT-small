    k = np.argmax(c)

    high = min(k, len(a)-1)

def smooth_step(a, step_factor):
    n = a.size
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    idx = np.arange(low, high+1)
    partner = k - idx
    # product
    prod = a[idx] * a[partner]
    # identify max product index (should be same as k but may be multiple)
    max_pos = np.argmax(prod)
    i = idx[max_pos]
    j = partner[max_pos]
    # compute delta proportional to product
    delta = step_factor * prod[max_pos]
    # ensure delta <= a[i], a[j] (to keep nonnegative)
    delta = min(delta, a[i], a[j])
    if delta <= 0:
        return a, False
    # subtract delta from i and j; redistribute
    a[i] -= delta
    a[j] -= delta
    # distribute equally to neighbors (four possible)
    share = delta / 4.0
    if i > 0:
        a[i-1] += share
    if i + 1 < n:
        a[i+1] += share
    if j > 0:
        a[j-1] += share
    if j + 1 < n:
        a[j+1] += share
    # renormalize
    total = a.sum()
    if total <= 0 or not np.isfinite(total):
        a[:] = 1.0 / n
    else:
        a /= total
    return a, True
```

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _clip_and_normalize(arr):
    """Clip to [0,1000] and scale so that the sum equals 1."""
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s <= 0.0 or not np.isfinite(s):
        arr = np.full_like(arr, 1.0 / arr.size)
    else:
        arr = arr / s
    return arr


def _evaluate(arr):
    """Return the AC1 upper bound for a numpy 1‑D array."""
    return float(evaluate_sequence(arr.tolist()))


def _smooth_optimize(a, rng, target_score=1.50274349):
    """
    Local improvement by moving a small amount of mass from the pair that
    contributes most to the maximal convolution entry to their neighbours.
    """
    n = a.size
    best_seq = a.copy()
    best_score = _evaluate(best_seq)
    no_improve = 0
    max_no_improve = 500
    base_step = 0.2
    # limit overall work – 2000 iterations are ample for n≈4000
    for it in range(2000):
        # step size shrinks slowly
        step_factor = base_step / np.sqrt(it + 1.0)

        # current convolution and its peak index
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))

        # indices that actually appear in conv[k]
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=np.int64)
        partner = k - idx

        # product a_i * a_j for each contributing pair
        prod = a[idx] * a[partner]
        max_pos = int(np.argmax(prod))
        i = int(idx[max_pos])
        j = int(partner[max_pos])

        # amount to shift – fraction of the smaller of the two entries
        delta = step_factor * min(a[i], a[j])
        if delta <= 0.0:
            # nothing to move; break early
            break

        # neighbours that will receive the removed mass
        neigh = set()
        if i > 0:
            neigh.add(i - 1)
        if i + 1 < n:
            neigh.add(i + 1)
        if j > 0:
            neigh.add(j - 1)
        if j + 1 < n:
            neigh.add(j + 1)
        neigh_cnt = len(neigh)
        if neigh_cnt == 0:
            # no place to put the mass – skip this iteration
            continue

        share = (2.0 * delta) / neigh_cnt  # total removed = 2*delta, redistribute equally

        a[i] -= delta
        a[j] -= delta
        for nb in neigh:
            a[nb] += share

        # keep within hard bounds, then renormalise (sum should stay 1, but clipping may disturb)
        np.clip(a, 0.0, 1000.0, out=a)
        total = a.sum()
        if total <= 0.0 or not np.isfinite(total):
            a[:] = 1.0 / n
        else:
            a /= total

        # occasional random block perturbation to avoid stagnation
        if (it % 400 == 0) and (rng.random() < 0.3):
            block = max(1, n // 200)
            start = int(rng.integers(0, n - block + 1))
            factor = float(np.exp(rng.normal(0.0, 0.2)))
            a[start:start + block] *= factor
            np.clip(a, 0.0, 1000.0, out=a)
            a /= a.sum()

        cur_score = _evaluate(a)
        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            if best_score < target_score:
                break
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            break

    return best_seq, best_score

def propose_candidate(seed=42, **kwargs):
    """
    Produce a non‑negative step sequence that attempts to minimise the AC1
    upper bound.  The routine tests a number of analytically‑motivated shapes,
    keeps the best, and then refines it with a lightweight local optimisation.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine length and optional parent sequence
    # --------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all():
            n = parent_arr.size
            parent_ok = True
        else:
            raise ValueError
    except Exception:
        parent_ok = False
        n = 2000  # fallback length when no parent is available

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Helper to test a candidate and keep the best one
    # --------------------------------------------------------------
    def _try_candidate(arr):
        nonlocal best_seq, best_score
        seq = _clip_and_normalize(arr)
        sc = _evaluate(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # --------------------------------------------------------------
    # 1. Parent (if any)
    # --------------------------------------------------------------
    if parent_ok:
        _try_candidate(parent_arr.copy())

    # --------------------------------------------------------------
    # 2. Simple deterministic families
    # --------------------------------------------------------------
    # Uniform
    _try_candidate(np.ones(n, dtype=np.float64))

    # Power‑law a_i = i^{-p}, p in [0.1, 2.0]
    exponents = np.linspace(-2.0, -0.1, 20)
    for exp in exponents:
        arr = np.arange(1, n + 1, dtype=np.float64) ** exp
        _try_candidate(arr)

    # Geometric decay a_i = r^{i}
    for r in (0.3, 0.5, 0.7, 0.85, 0.9, 0.95, 0.98):
        arr = r ** np.arange(n, dtype=np.float64)
        _try_candidate(arr)

    # Half‑sine shape (positive)
    arr = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64)) / (n + 1))
    _try_candidate(arr)

    # Triangular (rise‑then‑fall)
    half = n // 2
    tri = np.concatenate([np.linspace(1, half, half, endpoint=False),
                          np.linspace(half, 1, n - half)])
    _try_candidate(tri)

    # Random Dirichlet‑like (Gamma) with various concentration parameters
    for alpha in (0.1, 0.3, 0.5, 1.0, 2.0):
        arr = rng.gamma(shape=alpha, scale=1.0, size=n)
        _try_candidate(arr)

    # Random exponential (memory‑less) distribution
    arr = rng.exponential(scale=1.0, size=n)
    _try_candidate(arr)

    # --------------------------------------------------------------
    # Ensure we have at least one viable fallback
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _evaluate(best_seq)

    # --------------------------------------------------------------
    # 3. Local refinement on the best candidate found so far
    # --------------------------------------------------------------
    refined_seq, refined_score = _smooth_optimize(best_seq.copy(), rng,
                                                  target_score=1.50274349)
    if refined_score < best_score:
        best_seq, best_score = refined_seq, refined_score

    # --------------------------------------------------------------
    # Return the best sequence as plain Python floats
    # --------------------------------------------------------------
    return [float(x) for x in best_seq.tolist()]
