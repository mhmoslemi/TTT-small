import numpy as np

def compute_conv_max_and_shift(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    # map to effective shift
    if max_idx > n-1:
        shift = 2*n-2 - max_idx
    else:
        shift = max_idx
    return max_val, shift

def compute_contrib_weights(a, shift):
    n = a.shape[0]
    w = np.zeros_like(a)
    if shift == 0:
        # own squares
        w = a * a
        return w
    # overlapping region indices i: 0 to n-1-shift
    contrib = a[:n-shift] * a[shift:]
    w[:n-shift] += contrib
    w[shift:] += contrib
    return w

def conv_max_and_shift(a):
    n = a.shape[0]
    m = 2*n - 1
    size = 1 << ((m-1).bit_length())  # next power of two >= m
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:m]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    if max_idx > n-1:
        shift = 2*n-2 - max_idx
    else:
        shift = max_idx
    return max_val, shift

import numpy as np

# ----------------------------------------------------------------------
# Helper functions (matrix‑free, linear memory)
# ----------------------------------------------------------------------
def _conv_max_and_shift(a: np.ndarray):
    """Return (max value of a*a convolution, effective shift where it occurs)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:m]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    # convolution is symmetric; map to a shift in [0, n-1]
    if max_idx > n - 1:
        shift = 2 * n - 2 - max_idx
    else:
        shift = max_idx
    return max_val, shift


def _neighbor_sum(a: np.ndarray, shift: int):
    """For each index i, return a_i+shift + a_i-shift (missing terms are omitted)."""
    n = a.shape[0]
    neigh = np.zeros_like(a)
    if shift == 0:
        # derivative of a_i^2 is 2 a_i
        neigh = 2.0 * a
    else:
        # contribution from pairs (i, i+shift)
        contrib = a[:n - shift]
        neigh[:n - shift] += contrib          # a_{i+shift} part
        neigh[shift:] += contrib              # a_{i-shift} part
    return neigh


def _biased_pair_transfer(arr: np.ndarray, rng: np.random.Generator):
    """Move a random fraction of mass from an early index to a later one."""
    n = arr.size
    half = n // 2
    # pick donor from the first half (must have positive mass)
    early_pos = np.where(arr[:half] > 0.0)[0]
    if early_pos.size == 0:
        donor = rng.integers(0, half)
    else:
        donor = int(rng.choice(early_pos))
    # pick recipient from the second half
    recipient = int(rng.integers(half, n))
    if arr[donor] <= 0.0:
        return arr.copy()
    delta = arr[donor] * rng.uniform(0.01, 0.08)   # move 1‑8 % of donor value
    cand = arr.copy()
    cand[donor] = max(0.0, cand[donor] - delta)
    cand[recipient] = min(1000.0, cand[recipient] + delta)
    return cand


def _early_late_scaling(arr: np.ndarray, rng: np.random.Generator):
    """
    Reduce a small early block and increase a symmetric late block.
    The increase is chosen to roughly keep the total sum unchanged
    (or to increase it a little), while hopefully lowering the max convolution.
    """
    n = arr.size
    early_len = max(1, n // 100)          # ~1 % of the length
    late_len = early_len

    early_sum = arr[:early_len].sum()
    late_sum = arr[-late_len:].sum()
    if early_sum == 0.0:
        return arr.copy()

    # reduction fraction for early block, 1‑20 %
    red = rng.uniform(0.01, 0.20)
    # increase fraction for late block chosen so that total sum does not shrink
    inc = red * early_sum / max(late_sum, 1e-12)
    inc = inc * rng.uniform(0.9, 1.1)      # small random perturbation

    cand = arr.copy()
    cand[:early_len] *= (1.0 - red)
    cand[-late_len:] *= (1.0 + inc)
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _random_block_scale(arr: np.ndarray, rng: np.random.Generator):
    """Scale a random contiguous block (used for occasional exploratory jumps)."""
    n = arr.size
    w = int(rng.integers(max(1, n // 200), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - w + 1)))
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.07)))  # roughly 1 ± 7 %
    cand[start:start + w] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):

    Return a list of non‑negative floats (step‑function heights) that
    strives to minimise the AC1 upper bound defined by `evaluate_sequence`.
    The routine works without any external data except the optional

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        a = np.array(height_sequence_1, dtype=np.float64).copy()
        if a.ndim != 1:
            raise ValueError
        n = a.shape[0]
    except Exception:

        n = 256
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            f = float(i)
            a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
        np.clip(a, 0.0, 1000.0, out=a)

    np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    # Baseline score and safety fallback
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Keep a completely uniform fallback (sometimes better for tiny n)
    uniform = np.ones_like(best)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # ------------------------------------------------------------------
    # Iterative, matrix‑free search
    # ------------------------------------------------------------------
    # Adaptive step‑size for the simple pair transfer (starts modest)
    pair_step_frac = 0.07

    for step in range(num_steps):
        # --- gradient‑inspired biased transfer ---------------------------------
        cand = _biased_pair_transfer(best, rng)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            # a little bit larger transfer after success
            pair_step_frac = min(pair_step_frac * 1.05, 0.12)

        # --- occasional structured early/late scaling -------------------------
        if step % 55 == 0:
            cand2 = _early_late_scaling(best, rng)
            sc2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best = cand2
                best_score = sc2

        # --- random block perturbations for escaping plateaus -----------------
        if step % 200 == 0:
            cand3 = _random_block_scale(best, rng)
            sc3 = float(evaluate_sequence(cand3.tolist()))
            if np.isfinite(sc3) and sc3 < best_score:
                best = cand3
                best_score = sc3

        # --- very lightweight gradient step (optional, cheap) -----------------
        # compute current max shift and neighbor sums
        max_val, shift = _conv_max_and_shift(best)
        S = best.sum()
        if S > 0.0:
            target = 2.0 * max_val / S

            # scale direction to keep updates modest

                best = cand4
                best_score = sc4

    # ------------------------------------------------------------------
    # Return a plain Python list of floats (the best candidate found)
    # ------------------------------------------------------------------
    return [float(v) for v in best]
