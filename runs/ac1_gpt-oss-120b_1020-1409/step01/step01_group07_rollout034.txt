def _project_to_simplex(v):
    # ensures non-negative and sum = 1
    # Based on algorithm: sort v descending, find rho, compute theta.
    n = v.shape[0]
    u = np.sort(v)[::-1]   # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)

def _project_simplex(v):
    """Project non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total <= 0.0:
        # uniform fallback
        n = v.shape[0]
        return np.full_like(v, 1.0 / n)
    return v / total

    """Score 2·n·max(conv(seq,seq))/(sum(seq)^2) – assumes seq is a 1‑D ndarray."""
    total = seq.sum()
    if total <= 0.0:
        return np.inf
    conv = np.convolve(seq, seq)
    return float(2 * len(seq) * conv.max() / (total * total))

    result = best_seq.tolist() if best_seq is not None else [1.0 / n] * n
    try:
        final_score = float(evaluate_sequence(result))  # evaluate_sequence is pre‑imported
        if not np.isfinite(final_score):
            result = [1.0 / n] * n
    except Exception:
        result = [1.0 / n] * n

    # Defensive guard – guarantee a non‑empty, finite list.
    if not result or any(not np.isfinite(x) for x in result):
        result = [1.0 / n] * n
    return result
