    M = float(conv.max())

def _directed_pair_perturbation(arr, rng, head_frac=0.25, tail_frac=0.75):
    n = arr.size
    i = rng.integers(0, int(n * head_frac))
    j = rng.integers(int(n * tail_frac), n)
    # transfer fraction
    if arr[i] <= 0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.05, 0.2) # larger fraction
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand

    rng = np.random.default_rng(seed)

        f[0] = 1.0
        for i in range(1, n):
            i_f = float(i)
            f[i] = f[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    n = f.size
    np.clip(f, 0.0, 1000.0, out=f)

    # compute M
    conv_f = np.convolve(f, f)
    M = float(conv_f.max())
    # slack initial
    slack = np.full(2 * n - 1, M, dtype=np.float64)

    g = np.zeros_like(f)

    # Greedy fill
    for j in range(n-1, -1, -1):
        SL = slack[j:j+n]
        # compute per-lower bound
        # avoid division by zero (f > 0)
        ratios = SL / f
        delta = float(np.min(ratios))
        # maximum per entry limited by remaining capacity (1000 - g[j])
        maxcap = 1000.0 - g[j]
        delta = min(delta, maxcap, M)  # also not exceed M to keep within slack at index j
        # also small epsilon floor
        if delta > 1e-12:
            g[j] += delta
            slack[j:j+n] -= delta * f
            # keep non-negative slack due to numerical errors
            np.maximum(slack, 0.0, out=slack)

    # candidate mixing
    best = f
    best_score = float(evaluate_sequence(best.tolist()))
    # test mix ratios
    for t in np.linspace(0.0, 1.0, num=21):  # 0,0.05,...1
        candidate = (1.0 - t) * f + t * g
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = candidate.copy()

    # also test pure g
    sc_g = float(evaluate_sequence(g.tolist()))
    if np.isfinite(sc_g) and sc_g < best_score:
        best = g.copy()
        best_score = sc_g

    # return list
    return [float(v) for v in best]

    then mixes f and g and keeps the best mixture.
    """

            f[i] = f[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    n = f.size

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    conv_f = np.convolve(f, f)
    M = float(conv_f.max())

    # ------------------------------------------------------------------
    # 3. Build a secondary sequence g such that f * g ≤ M.
    #    We use a simple fractional greedy method:
    #       - keep a slack vector (initially M) for each convolution index
    #       - repeatedly add a fraction (here 50%) of the maximal
    #         admissible amount at each position, updating the slack.
    #       - after a few passes the slack is exhausted and g is near‑optimal.
    # ------------------------------------------------------------------
    slack = np.full(2 * n - 1, M, dtype=np.float64)   # slack[k] = M - (f*g)[k]
    g = np.zeros_like(f)

    # number of passes – a small constant is enough for the required accuracy
    for _ in range(6):
        # walk from the end to the start (tail‑first helps spreading the mass)
        for j in range(n - 1, -1, -1):
            # slice of slack that would be touched by an addition at position j
            sl = slack[j:j + n]                     # length n
            # maximal admissible addition at this position (respecting all constraints)
            # safe because f > 0 for every index of the binomial/parent sequence
            max_delta = float(np.min(sl / f))
            # we add only a fraction to keep room for later indices
            delta = max_delta * 0.5
            # respect the absolute upper bound per entry (1000) – here never reached
            if delta > 1e-12:

                sl -= delta * f                      # update the touched slack window
                # numerical safety
                np.maximum(sl, 0.0, out=sl)
                slack[j:j + n] = sl

    # ------------------------------------------------------------------
