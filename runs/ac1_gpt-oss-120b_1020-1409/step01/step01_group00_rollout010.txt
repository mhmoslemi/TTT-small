import numpy as np

# ------------------------------------------------------------
# Fast AC1 upper‑bound evaluation (O(n log n) via FFT)
# ------------------------------------------------------------
def _fast_score(arr: np.ndarray) -> float:
    n = arr.shape[0]
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    # length for a linear convolution of two length‑n arrays
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (s * s)


# ------------------------------------------------------------
# Simple generators for feasible start vectors
# ------------------------------------------------------------
def _generate_binomial_sequence(length: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i  (≈ 1/√i)"""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _offset_power_sequence(length: int, offset: float = 0.5) -> np.ndarray:
    """a_i = 1 / sqrt(i + offset)"""
    idx = np.arange(length, dtype=np.float64)
    a = 1.0 / np.sqrt(idx + offset)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


# ------------------------------------------------------------
# Perturbation primitives (pair transfer, block scaling)
# ------------------------------------------------------------
def _pair_transfer(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.01, 0.2)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _block_scale(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = arr.size
    w_low = max(1, n // 200)
    w_high = max(2, n // 20)
    width = int(rng.integers(w_low, w_high))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Focused move: shift mass away from indices that dominate the current peak
# ------------------------------------------------------------
def _focus_move(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = arr.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    k = int(np.argmax(conv))

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    inds = np.arange(i_min, i_max + 1, dtype=np.int64)
    js = k - inds
    contrib = arr[inds] * arr[js]
    total = contrib.sum()
    if total == 0.0:
        return arr.copy()
    probs = contrib / total
    donor = rng.choice(inds, p=probs)
    recipient = rng.integers(0, n)
    while recipient == donor:
        recipient = rng.integers(0, n)
    delta = arr[donor] * rng.uniform(0.01, 0.2)
    cand = arr.copy()
    cand[donor] = max(0.0, cand[donor] - delta)
    cand[recipient] = min(1000.0, cand[recipient] + delta)
    return cand

        elif move_type == 1:
            cand = _block_scale(best, rng)
        elif move_type == 2:
            cand = _focus_move(best, rng)
        else:
            # trivial random jitter on a small random subset
            cand = best.copy()
            mask = rng.random(n) < 0.03
            factor = float(np.exp(rng.normal(0.0, 0.04)))
            cand[mask] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _fast_score(cand)

        # accept improvements; simple simulated‑annealing style acceptance
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        else:
            # occasional uphill move to escape plateaus
            temp = 0.002 * (1.0 - step / num_iterations) + 1e-9
            prob = np.exp((best_score - cand_score) / temp)
            if rng.random() < prob:
                best = cand
                best_score = cand_score

        # occasional full re‑evaluation with the official evaluator
        if step % 500 == 0:
            true_score = float(evaluate_sequence(best.tolist()))
            if true_score < best_score:
                best_score = true_score  # keep the tighter bound we just observed

    # Final safety check with the official evaluator
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        # fallback to the simple binomial starter
        best = binom.copy()
    # Return plain Python list as required
    return [float(v) for v in best]
