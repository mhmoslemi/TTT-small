import numpy as np

# Helper: enforce a non‑increasing sequence in‑place
def enforce_monotonic_desc(seq: np.ndarray) -> None:
    """Clamp seq so that seq[i] >= seq[i+1] for all i."""
    for i in range(1, seq.size):
        if seq[i] > seq[i - 1]:
            seq[i] = seq[i - 1]

def propose_candidate(seed: int = 42, **kwargs) -> list:
    """
    Construct a step function whose AC1 upper bound is as low as possible.
    The routine builds a decreasing power‑law head and a constant tail,
    then refines the best candidate with cheap hill‑climbing while preserving
    monotonicity.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and a fallback parent sequence.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.shape[0])
        parent_seq = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        n = 256
        parent_seq = np.ones(n, dtype=np.float64)

    best_seq = parent_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 1) Grid search over a decreasing power‑law head + constant tail.
    # ------------------------------------------------------------------
    alphas = np.logspace(-4, 2, num=12, dtype=np.float64)   # 1e‑4 … 1e2
    m_fracs = [0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.40]  # head size fraction
    eps_factors = np.linspace(0.0, 1.0, num=9, dtype=np.float64)  # tail height as fraction of head end

    for alpha in alphas:
        for mf in m_fracs:
            m = int(round(n * mf))
            if m < 1:
                m = 1
            if m >= n:
                m = n - 1

            # power‑law head a_i = 1/sqrt(i+alpha)
            head = 1.0 / np.sqrt(np.arange(m, dtype=np.float64) + alpha)
            head_last = head[-1]

            tail_len = n - m

            # tail height = 0 (already covered by eps_factors[0])
            seq0 = np.concatenate([head, np.zeros(tail_len, dtype=np.float64)])
            seq0 = np.clip(seq0, 0.0, 1000.0)
            sc0 = float(evaluate_sequence(seq0.tolist()))
            if np.isfinite(sc0) and sc0 < best_score:
                best_seq, best_score = seq0, sc0

            # sweep tail height
            for ef in eps_factors[1:]:          # skip the zero case already evaluated
                eps = ef * head_last            # ensure monotonicity
                tail = np.full(tail_len, eps, dtype=np.float64)
                seq = np.concatenate([head, tail])
                seq = np.clip(seq, 0.0, 1000.0)
                sc = float(evaluate_sequence(seq.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_seq, best_score = seq, sc

    # ------------------------------------------------------------------
    # 2) Light hill‑climbing from the best candidate found.
    # ------------------------------------------------------------------
    # Block‑update parameters (kept modest to stay within time limits)
    min_width = max(1, n // 200)      # ~0.5 % of length
    max_width = max(2, n // 20)       # up to ~5 %
    sigma = 0.07                      # log‑normal noise scale

    for _ in range(800):  # reduced from 2000 to limit runtime
        cand = best_seq.copy()

        if rng.random() < 0.7:                 # multiplicative block perturbation
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor
        else:                                 # early‑late mass transfer
            i = int(rng.integers(0, n // 2))
            j = int(rng.integers(n // 2, n))
            delta = cand[i] * rng.random() * 0.2
            cand[i] -= delta
            cand[j] += delta

        # enforce constraints
        cand = np.clip(cand, 0.0, 1000.0)
        enforce_monotonic_desc(cand)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq, best_score = cand, sc

    # ------------------------------------------------------------------
    # 3) Final sanity – fallback to parent if nothing valid.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score) or best_seq.size == 0:
        best_seq = parent_seq

    return [float(x) for x in best_seq]
