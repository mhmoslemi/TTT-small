def donor_recipient_transfer(seq, frac=0.01):
    donor = int(np.argmax(seq))
    recipient = int(np.argmin(seq))
    if donor == recipient:
        return seq.copy()
    delta = seq[donor] * frac
    new_seq = seq.copy()
    new_seq[donor] = max(0.0, seq[donor] - delta)
    new_seq[recipient] = seq[recipient] + delta
    # Clip to 1000
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    return new_seq

import numpy as np


def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..length-1 (the classic binomial sequence)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _block_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = arr.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fast_convolve(arr: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of arr with itself using FFT (returns full length)."""
    # rfft works with real input and is faster than full complex FFT
    A = np.fft.rfft(arr, n=size)
    c = np.fft.irfft(A * A, n=size)
    return c[: arr.size * 2 - 1]
