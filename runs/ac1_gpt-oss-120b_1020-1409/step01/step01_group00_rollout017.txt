import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the classic binomial (discrete 1/√x) sequence.
# ----------------------------------------------------------------------
def _generate_binomial_sequence(length: int) -> np.ndarray:
    """Return a_i = C(2i,i)/4^i for i=0..length-1 (float64)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        f = float(i)
        a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
    return a


# ----------------------------------------------------------------------
# Fast AC1 bound evaluation using FFT (matrix‑free, O(n log n)).
# ----------------------------------------------------------------------
def _fft_score(arr: np.ndarray) -> float:
    """Return the AC1 upper bound for a non‑negative array `arr`."""
    n = arr.shape[0]
    if n == 0:
        return float('inf')
    s = arr.sum()
    if s < 0.01:
        return float('inf')
    # Length as next power of two ≥ 2n‑1 for clean FFT convolution.
    L = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    maxb = conv.max()
    return 2.0 * n * maxb / (s * s)
