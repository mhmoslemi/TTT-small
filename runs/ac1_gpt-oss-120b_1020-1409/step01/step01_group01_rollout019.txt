def compute_score(a):
    n = a.size
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    # convolution via FFT for speed:
    size = 1 << (2 * n - 1).bit_length()

    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    max_b = conv.max()

def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)

    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim ==1 and parent.size>0:
        n = int(parent.shape[0])
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_r_sequence(n)

    # Ensure nonnegative and clip
    np.clip(a, 0.0, 1000.0, out=a)

    def score(arr):
        sum_a = arr.sum()
        if sum_a < 0.01:
            return np.inf
        # FFT length
        size = 1 << ((2*n-1).bit_length())
        # Use real FFT for speed
        fa = np.fft.rfft(arr, size)
        conv = np.fft.irfft(fa * fa, size)[:2*n-1]
        max_b = conv.max()
        return 2.0 * n * max_b / (sum_a * sum_a)

    best_arr = a
    best_val = score(best_arr)
    # If best_val is nan or inf, fallback:
    if not np.isfinite(best_val):
        best_arr = np.ones(n, dtype=np.float64)
        best_val = score(best_arr)

    # parameters
    max_iter = 5000
    step_frac = 0.2  # fraction of source amount to move
    # Use decaying step fraction maybe
    for it in range(max_iter):
        # compute convolution & max index
        size = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(best_arr, size)
        conv = np.fft.irfft(fa * fa, size)[:2*n-1]
        idx_max = int(conv.argmax())
        # gradient for max index
        grad = np.zeros(n, dtype=np.float64)
        # all i where j = idx_max - i in range
        low = max(0, idx_max - (n-1))
        high = min(n-1, idx_max)
        # i from low to high:
        i_vals = np.arange(low, high+1)
        grad[i_vals] = 2.0 * best_arr[idx_max - i_vals]
        # pick source index with max grad (and >0)
        src_candidates = np.where(grad > 0)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[grad[src_candidates].argmax()]  # index with max grad
        # pick target index with min grad (prefers zero)
        tgt_candidates = np.where(grad == grad.min())[0]
        # exclude src
        tgt_candidates = tgt_candidates[tgt_candidates != src]
        if tgt_candidates.size == 0:
            break
        tgt = rng.choice(tgt_candidates)
        # compute delta
        amount = best_arr[src] * step_frac
        if amount <= 0:
            continue
        # ensure nonnegativity
        delta = min(amount, best_arr[src])
        cand = best_arr.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # clip to bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        # (sum preserved automatically)
        cand_val = score(cand)
        if cand_val < best_val:
            best_arr = cand
            best_val = cand_val
            # maybe increase step_frac a little
            step_frac = min(0.5, step_frac * 1.05)
        else:
            # reduce step
            step_frac *= 0.9
            if step_frac < 1e-6:
                break

    # final ensure finite score
    final_score = score(best_arr)
    if not np.isfinite(final_score):
        best_arr = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_arr]

import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the classic r‑sequence a_i = C(2i,i)/4^i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def _fast_score(a: np.ndarray) -> float:
    """Quadratic‑time‑free score: 2·n·max(a*a)/(sum a)^2."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    # length for a power‑of‑two FFT (>=2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def propose_candidate(seed: int = 42, **kwargs):
    """Return a step‑function that improves the AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Starting point – use the supplied parent if present, else r‑seq.
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_r_sequence(n)

    np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    # 2.  Helper: compute convolution, max index and a simple subgradient.
    # ------------------------------------------------------------------
    def _conv_and_max(x):
        fft_len = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(x, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())
        return conv, idx

    # ------------------------------------------------------------------
    # 3.  Main stochastic sub‑gradient descent.
    # ------------------------------------------------------------------
    best = a
    best_val = _fast_score(best)
    if not np.isfinite(best_val):
        best = np.ones(n, dtype=np.float64)
        best_val = _fast_score(best)

    # tuning parameters
    max_iter = 8000                # total loop count (adjustable)
    step_frac = 0.25               # initial fraction of mass to move
    target = 1.50274349            # break early if achieved

    for it in range(max_iter):
        conv, idx_max = _conv_and_max(best)

        # build subgradient of the max‑entry w.r.t. a
        grad = np.zeros(n, dtype=np.float64)
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        idxs = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idxs] = 2.0 * best[idx_max - idxs]

        # source = index with largest influence, target = smallest influence
        src = int(np.argmax(grad))
        # second‑best for target to avoid src == tgt
        order = np.argsort(grad)
        tgt = int(order[0] if order[0] != src else order[1])

        if grad[src] <= 0 or best[src] <= 0:
            # nothing useful to move – random perturbation as escape
            src = rng.integers(0, n)
            tgt = rng.integers(0, n)
            if src == tgt or best[src] == 0:
                continue

        if delta <= 0:
            continue
