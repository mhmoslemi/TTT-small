def compute_conv(a):

    n = len(a)
    m = 1 << (2*n-1 - 1).bit_length()  # power of two >= 2n-1
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)

import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Catalan‑like decreasing sequence: r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def convolve_self(a: np.ndarray) -> np.ndarray:
    """Return full convolution of a with itself using FFT (linear convolution)."""
    n = a.shape[0]
    size = 2 * n - 1
    # smallest power of two >= size for efficient FFT
    m = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[:size]

def improve_sequence(initial: np.ndarray, max_iters: int = 30,
                    eps_start: float = 0.01, eps_min: float = 1e-6) -> np.ndarray:
    """Hill‑climbing guided by the gradient of the AC1 bound."""
    best = initial.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.shape[0]

    for _ in range(max_iters):
        # convolution and index of maximal entry
        conv = convolve_self(best)
        k_max = int(np.argmax(conv))
        max_b = conv[k_max]
        S = best.sum()

        # contribution of each coordinate to the current maximal convolution
        idx = np.arange(n, dtype=np.int64)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        neighbor = np.empty(n, dtype=best.dtype)
        neighbor.fill(0.0)
        neighbor[mask] = best[j[mask]]

        # sign of derivative of the bound w.r.t. a_i (ignoring a positive factor)
        diff = neighbor * S - 2.0 * max_b
        improvement_indices = np.where(diff < 0)[0]
        if improvement_indices.size == 0:
            break

        eps = eps_start
        improved = False
        while eps > eps_min:
            cand = best.copy()
            cand[improvement_indices] += eps
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                improved = True
                break
            eps *= 0.5

        if not improved:
            break   # cannot push further

    return best

def propose_candidate(seed: int = 42, **kwargs):
    """Return a step‑function whose AC1 bound improves on the previous baseline."""
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
        init_candidate = np.clip(parent.astype(np.float64), 0.0, 1000.0)
        # also keep a Catalan‑like seed and pick the better one
        r_candidate = generate_r_sequence(n)
        score_init = float(evaluate_sequence(init_candidate.tolist()))
        score_r = float(evaluate_sequence(r_candidate.tolist()))
        if score_r < score_init:
            init_candidate = r_candidate
    else:
        n = 256
        init_candidate = generate_r_sequence(n)

    # perform a few guided improvement steps
    best = improve_sequence(init_candidate, max_iters=30,

    # safety fallback (should never be needed)
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = np.ones_like(best, dtype=np.float64)

    return [float(x) for x in best]
