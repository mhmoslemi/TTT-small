def compute_convolution(a, L=None):
    n = a.shape[0]
    if L is None:
        L = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n - 1]
    return conv
