import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities – matrix‑free, O(n log n) via FFT
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a power‑of‑two FFT (real input)."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=n_fft)                     # forward FFT (real)
    fb = fa * fa                                      # element‑wise product
    b = np.fft.irfft(fb, n=n_fft)                    # inverse FFT (real)
    return b[: 2 * n - 1]                            # truncate to true length


def _score_normed(a: np.ndarray) -> float:
    """
    Score for a vector that already sums to 1.
    Returns 2 * n * max( a*a ).
    """
    n = a.shape[0]
    b = _conv_full(a)
    maxb = float(b.max())
    return 2.0 * n * maxb


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step heights whose AC1 bound is low.
    The routine works without any external files; it only uses
    the supplied `height_sequence_1` (if present) as a fallback parent.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size from the provided parent construction.
    # --------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:                     # no parent available
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Normalise the parent to a probability vector – the score is
    # invariant under positive scaling, so we work with a sum of 1.
    parent = np.clip(parent, 0.0, None)
    if parent.sum() == 0.0:
        parent[:] = 1.0
    best = parent / parent.sum()
    best_score = _score_normed(best)

    # --------------------------------------------------------------
    # 1) Scan a small family of analytically‑motivated shapes.
    #    a_i = (i+α)^(-p)  with a few choices of p and α.
    #    Both the raw and a symmetrised version are examined.
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    powers = [0.4, 0.5, 0.6, 0.8, 1.0]   # exponent p
    alphas = np.logspace(-4.0, 2.0, num=30)   # shift α

    for p in powers:
        for a0 in alphas:
            cand = 1.0 / np.power(idx + a0, p)
            cand = cand / cand.sum()
            sc = _score_normed(cand)
            if sc < best_score:
                best = cand
                best_score = sc

            # symmetric variant (average with its reverse)
            cand_sym = 0.5 * (cand + cand[::-1])
            cand_sym = cand_sym / cand_sym.sum()
            sc = _score_normed(cand_sym)
            if sc < best_score:
                best = cand_sym
                best_score = sc

    # --------------------------------------------------------------
    # 2) Gradient‑free local search.
    #    The loop performs several cheap stochastic moves and a
    #    specialised gradient‑descent step that targets the currently
    #    worst convolution entry.
    # --------------------------------------------------------------
    min_width = max(1, n // 200)          # ≈0.5 % of the length
    max_width = max(2, n // 20)           # up to ≈5 % of the length
    sigma = 0.12                          # log‑normal multiplicative noise
    transfer = 0.20                       # max fraction transferred in a pairwise move
    lr_grad = 0.02                        # learning rate for the gradient step
    total_iters = 20000

    for it in range(total_iters):
        cand = best.copy()
        r = rng.random()

        # ---- multiplicative block perturbation (≈65 % of the time) ----
        if r < 0.65:
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor

        # ---- pairwise mass transfer (≈25 % of the time) ----
        elif r < 0.90:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            delta = best[i] * rng.random() * transfer
            cand[i] -= delta
            cand[j] += delta

        # ---- targeted gradient step (≈10 % of the time) ----
        else:
            # full convolution to locate the current bottleneck
            full = _conv_full(cand)
            k = int(np.argmax(full))            # index of the maximal entry
            # gradient of b_k = (a*a)_k w.r.t. a_i is 2*a_{k-i}
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            grad = np.zeros_like(cand)
            grad[i_range] = 2.0 * cand[k - i_range]
            cand = cand - lr_grad * grad         # move away from the peak

        # ---- occasional structural nudges ----
        if rng.random() < 0.03:                  # enforce (soft) symmetry
            cand = 0.5 * (cand + cand[::-1])
        if rng.random() < 0.02:                  # tiny three‑point smoothing
            kernel = np.array([0.25, 0.5, 0.25])
            cand = np.convolve(cand, kernel, mode='same')

        # ---- feasibility restore ----
        cand = np.clip(cand, 0.0, None)
        total = cand.sum()
        if total == 0.0:
            continue
        cand /= total                              # keep sum exactly 1

        # ---- accept if improved ----
        sc = _score_normed(cand)
        if sc < best_score:
            best = cand
            best_score = sc

        # ---- slow cooling of the stochastic parameters ----
        if (it + 1) % 3000 == 0:
            sigma *= 0.85
            lr_grad *= 0.85

    # --------------------------------------------------------------
    # 3) Final sanity check & conversion to plain Python list.
    # --------------------------------------------------------------
    if not np.isfinite(best_score):
        # fallback: clipped parent (already non‑negative)
        best = np.clip(parent, 0.0, 1000.0)
    # values are already in [0,1]; the evaluator will clip to [0,1000] anyway.
    return [float(v) for v in best.tolist()]
