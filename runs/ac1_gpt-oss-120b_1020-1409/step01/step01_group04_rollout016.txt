import numpy as np

# ----------------------------------------------------------------------
# Helper: fast AC1 bound for non‑negative sequences.
# For non‑negative a we have max(conv(a,a)) = sum_i a_i^2.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the exact value of evaluate_sequence without the convolution."""
    # seq is assumed already clipped to [0,1000]
    n = seq.size
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    sq = np.dot(seq, seq)          # sum_i a_i^2
    return float(2.0 * n * sq / (s * s))

def propose_candidate(seed: int = 42, **kwargs) -> list:
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound as defined by `evaluate_sequence`.
    The implementation avoids dense n‑by‑n matrices and uses only

    # ------------------------------------------------------------------

        n = int(parent.size)
    except Exception:

        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initialise candidates.

    # 1) Use the supplied parent (clipped to the allowed range).
    cand_parent = np.clip(parent, 0.0, 1000.0)
