def _perturb(vec, rng, target_sum):
    cand = vec.copy()
    n = cand.size
    if rng.random() < 0.5:
        # block perturb
        max_width = max(2, n // 20)
        width = int(rng.integers(1, max_width))
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:start+width] *= factor
    else:
        # pointwise perturb
        k = int(rng.integers(1, max(5, n // 200)))
        idx = rng.choice(n, size=k, replace=False)
        deltas = np.exp(rng.normal(0.0, 0.12, size=k))
        cand[idx] = cand[idx] * deltas
    # optional smoothing via mirroring
    if rng.random() < 0.1:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand

import numpy as np

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full convolution arr * arr (FFT, exact for real data)."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of 2 >= 2n-1
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact objective used by the official evaluator (scale‑invariant)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def _shape_powerlaw(n: int, p: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return (i + beta) ** (-p)


def _shape_exponential(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return r ** i


def _shape_combined(n: int, p: float, beta: float, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return (i + beta) ** (-p) * (r ** i)


def _shape_linear(n: int, exp: float) -> np.ndarray:
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    return (1.0 - t) ** exp

        start = int(rng.integers(0, n - w + 1))

    else:

    # occasional mild symmetrisation (helps stability)

    np.clip(cand, 0.0, 1000.0, out=cand)

    if s > 0.0:
