    # Determine the length n from the provided parent if possible.
    try:
        # height_sequence_1 is supplied by the harness (if any)
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = int(parent.size)
    except Exception:
        # Fallback length – any positive integer works; 256 is a reasonable default.
        n = 256

    # ------------------------------------------------------------------
    # Target L1‑norm (the evaluator is scale‑invariant, but we keep this for
    # consistency with the original framework).
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build a linear decreasing sequence: a_i = a0 * (1 - i/(n‑1)),
    # with a0 chosen so that the sum equals target_sum.
    if n <= 1:
        # Degenerate case – use a uniform value.
        a = np.full(n, target_sum / max(1, n), dtype=np.float64)
    else:
        a0 = 2.0 * target_sum / n                 # because sum = n*a0/2 = target_sum
        idx = np.arange(n, dtype=np.float64)
        a = a0 * (1.0 - idx / (n - 1))
        # Numerical safety: enforce the required bounds.
        np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    # Renormalise exactly to the target sum (eliminates tiny rounding error).
    s = a.sum()
    if s > 0.0:
        a *= (target_sum / s)

    # Ensure final clipping after renormalisation.
    np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    # Return a plain Python list as required by the evaluator.
    return [float(x) for x in a]
