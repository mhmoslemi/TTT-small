            seq = np.clip(seq, 0.0, 1000.0)
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score

    # random descending sequences
    for _ in range(200):
        # generate random exponential values
        vals = rng.exponential(scale=1.0, size=n)
        vals.sort()  # ascending
        seq = vals[::-1]  # descending
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best = seq
            best_score = score

    # hill climbing
    min_width = max(1, n // 500)
    max_width = max(2, n // 30)
    sigma = 0.05
    # adapt sigma schedule maybe decreasing
    for it in range(4000):
        cand = best.copy()
        # choose update type
        if rng.random() < 0.75:
            width = int(rng.integers(min_width, max_width+1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start+width] *= factor
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n-1))
            if j >= i: j += 1
            delta = best[i] * rng.random() * 0.15
            cand[i] -= delta
            cand[j] += delta

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # optionally tighten sigma when improvement
        # optionally reduce sigma gradually
        sigma *= 0.9999

    # final verification:
    if not (np.isfinite(best_score) and best.size > 0):

    return [float(x) for x in best]

import numpy as np


def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step‑function sequence with a low AC1 bound.

    The routine starts from the supplied parent (if any), scans a family of
    power‑law decays, samples random decreasing sequences, and finally performs
    a modest hill‑climbing loop with block‑wise multiplicative updates and
    pairwise mass transfers.  The best sequence found is returned as a plain
    Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size from the supplied parent construction.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Best so far --------------------------------------------------------
    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    idx = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1) Grid search over power‑law families a_i = (i+α)^(-p)
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.2, 1.5, 14)          # exponent candidates
    alphas = np.logspace(-5, 2, 15)            # offset candidates

    for p in p_vals:
        inv_pow = 1.0 / ((idx + alphas[:, None]) ** p)   # shape (len(alphas), n)
        # iterate over rows to keep memory modest
        for row in inv_pow:
            cand = np.clip(row, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score

    # ------------------------------------------------------------------
    # 2) Random decreasing sequences (exponential draws, sorted)
    # ------------------------------------------------------------------
    for _ in range(300):
        vals = rng.exponential(scale=1.0, size=n)
        vals.sort()
        cand = vals[::-1]                       # descending order
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # 3) Simple hill‑climbing from the current best.
    # ------------------------------------------------------------------
    min_width = max(1, n // 300)               # ~0.3 % of length
    max_width = max(3, n // 30)                # up to ~3 %
    sigma = 0.07                               # log‑normal scale factor STD

    for it in range(2500):
        cand = best.copy()

        if rng.random() < 0.70:               # block multiplicative perturbation
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor
        else:                                 # pairwise mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1                         # ensure i != j
            delta = best[i] * rng.random() * 0.20
            cand[i] = max(0.0, cand[i] - delta)
            cand[j] += delta

        # Occasionally enforce monotonicity (helps keep large values away
        # from each other and often reduces the peak of the convolution).
        if rng.random() < 0.15:
            cand = np.sort(cand)[::-1]

        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            sigma = max(sigma * 0.95, 0.01)     # tighten step size after improvement

        sigma = max(sigma * 0.9995, 0.01)

    # ------------------------------------------------------------------
