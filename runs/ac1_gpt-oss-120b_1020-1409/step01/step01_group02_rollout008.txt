def _full_convolution(a):
    n = a.size
    size = 2*n - 1
    # compute next power of 2 for FFT speed
    # but we can just use np.fft.rfft with length N = next_pow_of_2 >= size
    # Use np.fft.rfft for real input: N = 1 << (size-1).bit_length()
    N = 1 << (size-1).bit_length()
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:size]

    n = a.shape[0]

    fa = np.fft.rfft(a, n=N)
