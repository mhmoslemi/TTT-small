import numpy as np
import math
import time

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the sequence r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def propose_candidate(seed: int = 42, **kwargs):
    """Return a step‑function sequence aiming to lower the AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain initial sequence (use the provided parent if possible)
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_r_sequence(n)
    n = a.shape[0]

    # enforce hard bounds required by evaluator
    np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    # Pre‑compute convolution, sum and the current score
    # ------------------------------------------------------------------
    sum_a = float(a.sum())
    conv = np.convolve(a, a)                     # length 2n-1
    current_score = 2.0 * n * float(conv.max()) / (sum_a * sum_a)
    best_a = a.copy()
    best_score = current_score

    # ------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing parameters
    # ------------------------------------------------------------------
    T = max(1e-12, current_score * 0.1)           # starting temperature
    T_min = 1e-12
    decay = 0.9995
    max_iter = 12000                               # safety cap
    no_improve_limit = 2000
    no_improve = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    for it in range(max_iter):
        # --------------------------------------------------------------

        # --------------------------------------------------------------
        if rng.random() < 0.7:
            idx_max = int(np.argmax(conv))
            i_start = max(0, idx_max - (n - 1))
            i_end = min(n - 1, idx_max)
            ii = np.arange(i_start, i_end + 1, dtype=np.int64)
            jj = idx_max - ii
            contrib = a[ii] * a[jj]

            # avoid moving mass to the partner that creates the large product
            forbidden = int(jj[np.argmax(contrib)])
        else:

            forbidden = -1

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        dst = int(rng.integers(0, n))
