import numpy as np

def _arcsine_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 0.5 / i)
    return a

def _make_block_perturbation(seq, rng):
    n = seq.size
    # select random block width randomly between 1% and 20% of n
    width = max(1, int(rng.uniform(0.01, 0.2) * n))
    start = rng.integers(0, n - width + 1)
    scale = np.exp(rng.normal(0.0, 0.2))
    cand = seq.copy()
    cand[start:start+width] *= scale
    np.clip(cand, 0.0, 1000.0, out=cand)
    # optionally symmetrize
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    # renormalize sum to 1
    total = cand.sum()
    if total < 1e-12:
        return seq.copy()
    cand /= total
    return cand

import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _arcsine_sequence(n: int) -> np.ndarray:
    """Generate the arcsine‑type decreasing sequence of length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a cheap multiplicative perturbation on a random contiguous block.
    The resulting vector is re‑normalized to keep the sum = 1.
    """
    n = seq.size
    # width between 1 % and 20 % of the length
    width = max(1, int(rng.uniform(0.01, 0.20) * n))
    start = rng.integers(0, n - width + 1)
    scale = np.exp(rng.normal(0.0, 0.20))
    cand = seq.copy()
    cand[start:start + width] *= scale
    np.clip(cand, 0.0, 1000.0, out=cand)
    # occasional symmetrisation – helps when a good palindrome exists
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])
    total = cand.sum()
    if total > 0:
        cand /= total
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, max_steps: int = 2500, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that seeks to
    minimise the AC1 upper bound.  The routine works for the provided
    `height_sequence_1` if it exists; otherwise it falls back to the
    analytically‑derived arcsine sequence.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise – parent or arcsine, normalised to sum = 1
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim == 1
            and parent.size > 0
            and np.isfinite(parent).all()
        ):
            n = parent.size
            init = np.clip(parent, 0.0, 1000.0)
        else:
            raise ValueError
    except Exception:
        n = 256
        init = _arcsine_sequence(n)

    init_sum = init.sum()
    if init_sum < 1e-12:
        init = np.ones(n, dtype=np.float64)
        init_sum = n
    init = init / init_sum                               # scale to sum = 1

    # ------------------------------------------------------------------
    # 2. Evaluate the two obvious seeds (parent vs arcsine)
    # ------------------------------------------------------------------
    best_seq = init.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    arcsine = _arcsine_sequence(n)
    arcsine = arcsine / arcsine.sum()
    arcsine_score = float(evaluate_sequence(arcsine.tolist()))
    if arcsine_score < best_score:
        best_seq = arcsine.copy()
        best_score = arcsine_score

    current = best_seq.copy()

    # ------------------------------------------------------------------
    # 3. Main iterative improvement
    # ------------------------------------------------------------------
    steps_without_improve = 0
    temperature_start = 1.0
    temperature_end = 0.001
    for step in range(max_steps):
        # Linear cooling schedule
        temp = max(
            temperature_end,
            temperature_start * (1.0 - step / max_steps),
        )

        # Full auto‑convolution and index of the current maximal entry
        conv = np.convolve(current, current)
        k_max = int(np.argmax(conv))

        # Gradient‑like weights: contribution of each coordinate to conv[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        weights = np.zeros(n, dtype=np.float64)
        weights[mask] = 2.0 * current[idx[mask]]

        # ----------------------------------------------------------------
        #   Propose a new candidate
        # ----------------------------------------------------------------
        if weights.max() > 0 and rng.random() < 0.85:
            # Move mass from a high‑weight index to a low‑weight one
            src = int(np.argmax(weights))
            dst = int(np.argmin(weights))
            if src == dst:
                # pick a different destination uniformly at random
                dst = int(rng.integers(0, n - 1))
                if dst >= src:
                    dst += 1
                dst %= n
            # Fraction of source mass to relocate – between 5 % and 50 %
            frac = 0.05 + 0.45 * rng.random()
            delta = min(current[src] * frac, current[src])
            cand = current.copy()
            cand[src] -= delta
            cand[dst] += delta
        else:
            # Simple random transfer (fallback / exploration)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            frac = 0.05 + 0.45 * rng.random()
            delta = min(current[i] * frac, current[i])
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta

        # Occasionally apply a block perturbation for larger jumps
        if rng.random() < 0.02:
            cand = _block_perturb(cand, rng)

        # Project back into the feasible box and renormalise sum to 1
        np.clip(cand, 0.0, 1000.0, out=cand)
        total = cand.sum()
        if total < 1e-12:
            continue
        cand /= total

        # ----------------------------------------------------------------
        #   Evaluate and accept/reject
        # ----------------------------------------------------------------
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            score = float("inf")

        improved = score < best_score
        if improved:
            best_seq = cand.copy()
            best_score = score
            current = cand.copy()
            steps_without_improve = 0
        else:
            # Simulated‑annealing acceptance of worsening moves
            if rng.random() < np.exp((best_score - score) / max(temp, 1e-12)):
                current = cand.copy()
            steps_without_improve += 1

        # Early termination if stagnation persists
        if steps_without_improve > 400:
            break

    # ------------------------------------------------------------------
    # 4. Return the best candidate found (as a plain Python list)
    # ------------------------------------------------------------------
    if best_seq.size == 0:
        # Absolute safety net – a tiny uniform vector
        best_seq = np.ones(256, dtype=np.float64) / 256.0
    return [float(x) for x in best_seq]
