def fast_max_convolution(a):
    # a is numpy array
    n = a.shape[0]
    n_fft = 1 << ((2*n - 1).bit_length())
    # Use real FFT
    fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(fa * fa, n=n_fft)
    # conv length n_fft; take relevant part
    max_val = np.max(conv[:2*n-1])
    return max_val

import numpy as np

# ---------- Helper: fast convolution max and index (FFT, O(n log n)) ----------
def _fast_conv_max_idx(a: np.ndarray) -> tuple[float, int]:
    """Return (max_convolution_value, index_of_max) for a*a using FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(fa * fa, n=n_fft)            # real convolution
    conv = conv[: 2 * n - 1]                         # keep only needed part
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ---------- Helper: cheap bound using the fast convolution ----------
def _cheap_bound(a: np.ndarray, max_val: float) -> float:
    """Upper‑bound value 2·n·max/(sum²) using already‑known max."""
    n = a.shape[0]
    s = a.sum()
    if s == 0.0:
        return np.inf
    return float(2.0 * n * max_val / (s * s))
