def ac1_score(a):
    n = a.shape[0]
    total = a.sum()
    if total < 0.01:
        return np.inf
    m = 2*n-1
    nfft = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(fa * fa, nfft)[:m]
    max_b = conv.max()
    return float(2.0 * n * max_b / (total*total))

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    parent = _parent_sequence()
    n = int(parent.shape[0])
    # precompute nfft
    m = 2*n - 1
    nfft = 1 << ((m-1).bit_length())
    # score helper
    def score(a):
        total = a.sum()
        if total < 0.01:
            return np.inf
        a_fft = np.fft.rfft(a, nfft)
        conv = np.fft.irfft(a_fft * a_fft, nfft)[:m]
        max_b = conv.max()
        return float(2.0 * n * max_b / (total * total))
    # initial candidates
    best = parent.copy()
    best_score = score(best)
    arcsine = _arcsine_sequence(n)
    arcsine_score = score(arcsine)
    if arcsine_score < best_score:
        best = arcsine
        best_score = arcsine_score
    # ensure best sum non-zero
    if best.sum() < 0.01:
        best[0] = 0.01
    current = best.copy()
    current_score = best_score

    # main loop
    base_alpha = 0.07  # may adjust
    for step in range(num_steps):
        # compute conv of current
        a_fft = np.fft.rfft(current, nfft)
        conv = np.fft.irfft(a_fft * a_fft, nfft)[:m]
        # idx max
        idx_max = int(conv.argmax())
        # compute gradient slice
        lo = max(0, idx_max - (n-1))
        hi = min(n-1, idx_max)
        grad = np.zeros_like(current)
        if lo <= hi:
            idxs = idx_max - np.arange(lo, hi+1)
            grad[lo:hi+1] = 2.0 * current[idxs]
        # step size schedule
        alpha = base_alpha * (0.5 ** (step // 500))
        cand = current - alpha * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        total = cand.sum()
        if total < 0.01:
            cand[0] += 0.01 - total
            total = 0.01
        # maybe normalize total to original sum? Not needed.
        cand_score = score(cand)
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
        # acceptance for current
        if cand_score < current_score:
            current = cand
            current_score = cand_score
        else:
            # simulated annealing acceptance
            temp = 0.01 * (1.0 + step)  # maybe increasing, but let's set decaying small temp.
            # Accept if better by probability exp(-(cand_score - current_score)/temp)
            if temp > 0:
                prob = math.exp(-(cand_score - current_score) / temp)
                if rng.random() < prob:
                    current = cand
                    current_score = cand_score
        # occasional random perturb
        if rng.random() < 0.02:
            pert = _block_perturb(current, rng)
            pert_score = score(pert)
            if pert_score < best_score:
                best = pert.copy()
                best_score = pert_score
            if pert_score < current_score:
                current = pert
                current_score = pert_score
    # fallback
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)
    # convert to list
    return [float(x) for x in best]

def _block_perturb(base, rng):
    n = base.shape[0]
    # choose width random between n//200 and n//20, at least 1
    low = max(1, n // 200)
    high = max(low+1, n // 20)
    width = int(rng.integers(low, high))
    start = int(rng.integers(0, n - width + 1))
    cand = base.copy()
    scale = float(np.exp(rng.normal(0.0, 0.2)))
    cand[start:start+width] *= scale
    # optional symmetry bias
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

import numpy as np
import math

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _parent_sequence():
    """
    Return the provided parent sequence if it exists and is valid.
    Otherwise return a short uniform fallback.
    """
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    # fallback: a short uniform vector that always satisfies the evaluator
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """
    Generate a decreasing sequence defined by
        a_0 = 1,
        a_i = a_{i-1} * (1 - 0.5 / i)  for i >= 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _block_perturb(base, rng):
    """
    Apply a modest random multiplicative perturbation to a contiguous block.
    Occasionally impose a weak symmetry bias.
    """
    n = base.shape[0]
    low = max(1, n // 200)
    high = max(low + 1, n // 20)
    width = int(rng.integers(low, high))
    start = int(rng.integers(0, n - width + 1))
    cand = base.copy()
    scale = float(np.exp(rng.normal(0.0, 0.2)))
    cand[start:start + width] *= scale
    if rng.random() < 0.3:                     # occasional symmetry bias
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _ac1_score(a, n, nfft):
    """
    Compute the AC1 upper‑bound score for a non‑negative sequence `a`.
    The formulation matches `evaluate_sequence` but uses FFT for speed.
    """
    total = a.sum()
    if total < 0.01:
        return np.inf
    # convolution via FFT (real‑valued)
    a_fft = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(a_fft * a_fft, nfft)[: (2 * n - 1)]
    max_b = conv.max()
    return float(2.0 * n * max_b / (total * total))

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 upper bound.
    Returns a list of floats representing the best found sequence.
    """
    rng = np.random.default_rng(seed)
    num_steps = int(kwargs.get('num_steps', 4000))

    # ------------------------------------------------------------------
    # initialise candidates
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = int(parent.shape[0])
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two for FFT

    best = parent.copy()
    best_score = _ac1_score(best, n, nfft)

    arcsine = _arcsine_sequence(n)
    arcsine_score = _ac1_score(arcsine, n, nfft)
    if arcsine_score < best_score:
        best = arcsine
        best_score = arcsine_score

    # guarantee a tiny positive mass
    if best.sum() < 0.01:
        best[0] += 0.01 - best.sum()

    current = best.copy()
    current_score = best_score

    # ------------------------------------------------------------------
    # projected gradient descent with occasional random jumps
    # ------------------------------------------------------------------
    base_alpha = 0.07

    for step in range(num_steps):
        # full convolution (FFT) of the current iterate
        a_fft = np.fft.rfft(current, nfft)
        conv = np.fft.irfft(a_fft * a_fft, nfft)[:m]

        # index of the maximal convolution entry
        idx_max = int(conv.argmax())

        # gradient of the maximal entry with respect to each coefficient
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        grad = np.zeros_like(current)
        if lo <= hi:
            idxs = idx_max - np.arange(lo, hi + 1)
            grad[lo:hi + 1] = 2.0 * current[idxs]

        # diminishing step size
        alpha = base_alpha * (0.5 ** (step // 500))

        # projected gradient step
        cand = current - alpha * grad
        np.clip(cand, 0.0, 1000.0, out=cand)

        total = cand.sum()
        if total < 0.01:
            cand[0] += 0.01 - total
            total = 0.01

        cand_score = _ac1_score(cand, n, nfft)

        # record the best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # accept if it improves the current iterate; otherwise occasional
        # simulated‑annealing acceptance to escape shallow minima
        if cand_score < current_score:
            current = cand
            current_score = cand_score
        else:
            temp = max(1e-8, 0.5 / (step + 1))
            prob = math.exp(-(cand_score - current_score) / temp)
            if rng.random() < prob:
                current = cand
                current_score = cand_score

        # occasional random block perturbation to explore new regions
        if rng.random() < 0.02:
            pert = _block_perturb(current, rng)
            pert_score = _ac1_score(pert, n, nfft)
            if pert_score < best_score:
                best = pert.copy()
                best_score = pert_score
            if pert_score < current_score:
                current = pert
                current_score = pert_score

    # ------------------------------------------------------------------
    # safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
