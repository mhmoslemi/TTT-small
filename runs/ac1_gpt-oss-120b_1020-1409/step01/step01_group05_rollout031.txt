import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _fft_convolve_self(a, n_fft):
    """Full convolution of a with itself using an FFT of length n_fft."""
    # real FFT of the zero‑padded array
    A = np.fft.rfft(a, n_fft)
    # element‑wise product (â²) and inverse FFT
    conv = np.fft.irfft(A * A, n_fft)[:2 * a.size - 1]
    return conv


def _score_from_conv(conv, sum_a, n):
    """Exact AC1 upper‑bound score given a convolution array."""
    return float(2.0 * n * conv.max() / (sum_a * sum_a))


def _initial_candidate(n, parent_ok, parent_arr):
    """Return a normalised non‑negative starting vector."""
    if parent_ok:
        a = np.clip(parent_arr, 0.0, 1000.0)
        s = a.sum()
        if s > 0.0 and np.isfinite(s):
            a = a / s
        else:
            a = np.full(n, 1.0 / n)
    else:
        # decreasing power‑law (1/√i) – simple and cheap to compute
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / np.sqrt(idx)
        s = a.sum()
        a = a / s
    # enforce hard bounds once – they remain satisfied by the update rule
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _gradient_at_max(a, k):
    """Gradient of the maximal convolution entry w.r.t. a (mirror descent)."""
    n = a.size
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    idx = np.arange(low, high + 1, dtype=np.int64)
    partner = k - idx
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[partner]
    return g
