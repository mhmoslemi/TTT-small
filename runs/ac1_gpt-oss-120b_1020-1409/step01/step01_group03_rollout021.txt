import numpy as np

# ------------------------------------------------------------
# Convolution utilities (FFT based, linear memory)
# ------------------------------------------------------------
def _conv(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of a and b using real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    fft_len = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)[:n]
    return conv


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Convolution of x with itself (full length)."""
    return _conv(x, x)


# ------------------------------------------------------------
# Scoring – mirrors the evaluator's exact computation
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the AC1 upper bound for a candidate sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)


# ------------------------------------------------------------
# Generate the classic analytic series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Soft‑max weights over the convolution entries
# ------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return a probability vector w ≈ softmax(alpha * conv)."""
    # shift for numerical stability
    shifted = conv - conv.max()
    # exponentiate
    exp = np.exp(alpha * shifted)
    total = exp.sum()
    if total == 0.0:
        # fall back to a uniform weight (should not happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


# ------------------------------------------------------------
# Gradient of the objective (smooth approximation of max)
# ------------------------------------------------------------
def _gradient(x: np.ndarray, alpha: float):
    """
    Compute an approximate gradient of
        f(x) = 2 * n * max(conv(x,x)) / (sum(x)^2)
    using a soft‑max over the convolution entries.
    Returns (grad_f, current_max).
    """
    n = x.shape[0]
    # full self‑convolution
    conv = _conv_self(x)
    M = conv.max()

    # soft‑max weighting of each convolution entry
    w = _softmax_weights(conv, alpha)

    # convolution of the weight vector with x (two directions)
    conv_fw = _conv(w, x)                     # length = (2n-1)+n-1 = 3n-2
    conv_rev = _conv(w[::-1], x)              # same length

    # first n entries of forward part
    part1 = conv_fw[:n]

    # first n entries of reversed part (correspond to indices L-1-i)
    part2 = conv_rev[::-1][:n]

    # gradient of the max‑approximation (2 * (part1 + part2))
    grad_M = 2.0 * (part1 + part2)

    # full gradient of f(x) = (2n/S^2) * M - (4n*M/S^3) * sum(x)
    S = x.sum()
    if S == 0.0:
        # avoid division by zero – return zero gradient
        return np.zeros_like(x), M

    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    # note: the factor n cancels out for the direction, we omit it
    grad_f = 2.0 * inv_S2 * grad_M - 4.0 * M * inv_S3 * np.ones_like(x)
    return grad_f, M
