# Analytic Catalan‑type series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the length‑`length` truncation of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast convolution (full mode) via real FFT
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 upper‑bound (mirrors evaluate_sequence)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    return float(2.0 * seq.size * c.max() / (s * s))


# ------------------------------------------------------------
# One projected gradient step (returns improved candidate)
# ------------------------------------------------------------
def _grad_step(x: np.ndarray, n: int, rng: np.random.Generator):
    conv = _conv_self(x)
    M = conv.max()
    k = int(conv.argmax())
    S = x.sum()
    if S == 0.0:
        return x, float('inf')

    # gradient of M w.r.t. x at the active index k:
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    g = np.zeros_like(x)
    g[idx] = x[k - idx]                     # derivative contribution

    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

    lr = 1.0
    best_cand = x
    best_sc = _score(x)
    for _ in range(6):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        if rng.random() < 0.1:               # occasional symmetrisation
            cand = 0.5 * (cand + cand[::-1])
        sc = _score(cand)
        if sc < best_sc:
            best_cand = cand
            best_sc = sc
            break
        lr *= 0.5
    return best_cand, best_sc


# ------------------------------------------------------------
# Random transfer of a fraction of mass from i to j
# ------------------------------------------------------------
def _transfer_step(x: np.ndarray, n: int, rng: np.random.Generator):
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while i == j:
        j = rng.integers(0, n)
    frac = rng.random() * 0.5                # up to 50 % of source entry
    amount = frac * x[i]
    cand = x.copy()
    cand[i] -= amount
    cand[j] += amount
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Random block scaling perturbation
# ------------------------------------------------------------
def _block_perturb(x: np.ndarray, n: int, rng: np.random.Generator):
    width = max(1, n // rng.integers(12, 31))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))    # log‑normal scaling
    cand = x.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    if rng.random() < 0.15:                  # occasional mirror mix
        cand = 0.5 * (cand + cand[::-1])
    return cand

    return [float(v) for v in best]
