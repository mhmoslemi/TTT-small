import numpy as np

def _make_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0]=1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    return a

def _conv_self(x):
    n = x.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv

    m = c.max()
    return float(2.0 * seq.size * m / (s*s))

import numpy as np

# ------------------------------------------------------------
# Convolution (self) using real FFT – O(n log n) time, O(n) memory.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Fast surrogate for the evaluator: 2*n*max(conv)/sum^2 .
# Mirrors the evaluator’s clipping of the sequence.
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    m = c.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Classic “square‑root” decreasing series a_i = C(2i,i)/4^i .
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a
