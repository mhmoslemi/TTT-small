import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _parent_sequence():
    """
    Return the provided parent sequence `height_sequence_1` if it exists
    and looks valid, otherwise a short uniform fallback.
    """
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            # enforce evaluator bounds
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    # fallback – uniform vector that always satisfies the evaluator
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """
    Classical discrete arcsine seed:
        a_0 = 1,
        a_i = a_{i-1} * (1 - 0.5 / i)   for i >= 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _next_pow2(x):
    """Small helper: next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_conv_fft(seq, size):
    """Linear convolution via FFT (real inputs)."""
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * seq.size - 1]


def _score_from_seq(seq, fft_size):
    """Compute the AC1 upper bound using the fast convolution."""
    n = seq.size
    conv = _full_conv_fft(seq, fft_size)
    max_b = conv.max()
    sum_a = seq.sum()
    # the evaluator is scale‑invariant; we keep the exact formula
    return 2.0 * n * max_b / (sum_a * sum_a)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed=42, num_steps=4000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1
    upper bound.  Starts from the arcsine seed (or the provided parent)
    and performs a directed stochastic mass‑transfer optimisation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initial candidate
    # ------------------------------------------------------------------
    base = _parent_sequence()
    n = base.size

    # use the better of the parent and the arcsine seed
    arcsine = _arcsine_sequence(n)
    # keep a copy of the best candidate seen so far
    cand_parent_score = _score_from_seq(base, _next_pow2(2 * n - 1))
    cand_arcsine_score = _score_from_seq(arcsine, _next_pow2(2 * n - 1))

    if cand_arcsine_score < cand_parent_score:
        best = arcsine.copy()
        best_score = cand_arcsine_score
    else:
        best = base.copy()
        best_score = cand_parent_score

    current = best.copy()

    # FFT size (power of two) – same for the whole run
    fft_size = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # compute current convolution and locate the maximal entry
        conv = _full_conv_fft(current, fft_size)
        max_b = conv.max()
        k0 = int(conv.argmax())

        # indices that actually contribute to this maximal entry
        lo = max(0, k0 - (n - 1))
        hi = min(k0, n - 1)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)

        # gradient weights proportional to a_{k0 - i}
        grad_weights = current[k0 - idx_range]

        # pick a source index (where the gradient is large)
        total_grad = grad_weights.sum()
        if total_grad > 0 and rng.random() < 0.9:
            probs = grad_weights / total_grad
            i = int(rng.choice(idx_range, p=probs))
        else:
            i = int(rng.integers(lo, hi + 1))

        # pick a destination index where the counterpart weight is small
        inv_weights = 1.0 / (grad_weights + 1e-12)
        probs_j = inv_weights / inv_weights.sum()
        j = int(rng.choice(idx_range, p=probs_j))

        if i == j:
            continue

        # move a modest fraction of mass from i to j
        max_transfer = current[i] * 0.5
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)   # stay inside evaluator bounds

        # evaluate the candidate using the fast score routine
        new_score = _score_from_seq(cand, fft_size)

        # deterministic improvement
        if new_score < best_score:
            best = cand
            best_score = new_score
            current = cand
            continue

        # simulated‑annealing acceptance to escape shallow minima
        temperature = 1.0 / (step + 1.0)
        if np.exp((best_score - new_score) / temperature) > rng.random():
            current = cand

        # occasional large random jump (helps to escape plateaus)
        if rng.random() < 0.02:
            # block‑wise multiplicative perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            block = current.copy()
            factor = float(np.exp(rng.normal(0.0, 0.25)))
            block[start:stop] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            block_score = _score_from_seq(block, fft_size)
            if block_score < best_score:
                best = block
                best_score = block_score
                current = block

    # ------------------------------------------------------------------
    # final safety check – guarantee a valid return value
    # ------------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    # Convert to plain Python floats (the evaluator expects a list)
    return [float(x) for x in best.tolist()]
