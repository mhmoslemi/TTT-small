import numpy as np

# Fast computation of the maximal entry of the self‑convolution using FFT.
def _max_conv_fft(seq: np.ndarray) -> float:
    """Return max(seq * seq) where * is linear convolution."""
    n = seq.shape[0]
    size = 2 * n - 1
    # next power of two for zero‑padding (FFT length)
    fft_len = 1 << ((size - 1).bit_length())
    ft = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(ft * ft, n=fft_len)[:size]
    return float(conv.max())
