    Return a non‑negative step‑function (list of floats) whose AC1 bound
    is as low as possible.  The routine works without any external data
    except the optional array `height_sequence_1` supplied by the harness.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initialise from the supplied parent, if any.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise ValueError
        n = parent_arr.size
        base = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        # fallback: modest length, uniform values
        n = 256
        base = np.ones(n, dtype=np.float64)

    best = base.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 1) Scan a simple power‑law family – cheap global improvement.
    # ------------------------------------------------------------------
    alphas = np.logspace(-5, 1, num=num_alpha, dtype=np.float64)
    for alpha in alphas:
        cand = _power_seq(alpha, n)
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 2) Guided stochastic search.
    # ------------------------------------------------------------------
    for step in range(num_steps):
        cand = best.copy()
        move = rng.random()

        # ------------------------------------------------------------------
        # (a) Multiplicative block perturbation – broad exploration.
        # ------------------------------------------------------------------
        if move < 0.5:
            min_w = max(1, n // 200)          # ≈0.5 % of the length
            max_w = max(2, n // 20)           # up to ≈5 %
            w = int(rng.integers(min_w, max_w + 1))
            start = int(rng.integers(0, n - w + 1))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand[start:start + w] *= factor

        # ------------------------------------------------------------------
        # (b) Simple pairwise mass transfer – preserves total mass.
        # ------------------------------------------------------------------
        elif move < 0.7:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            delta = cand[i] * rng.random() * 0.2
            cand[i] -= delta
            cand[j] += delta

        # ------------------------------------------------------------------
        # (c) Targeted reduction of the current maximal convolution term.
        # ------------------------------------------------------------------
        elif move < 0.9:
            conv = np.convolve(cand, cand)
            kmax = int(np.argmax(conv))
            idx = np.arange(n)
            partner = kmax - idx
            ok = (partner >= 0) & (partner < n)
            involved = idx[ok]
            if involved.size:
                i_sel = int(rng.choice(involved))
                j_sel = int(kmax - i_sel)
                # choose the larger of the two as donor
                donor = i_sel if cand[i_sel] >= cand[j_sel] else j_sel
                excl = np.array([i_sel, j_sel], dtype=np.int64)
                rec_candidates = np.setdiff1d(idx, excl, assume_unique=True)
                if rec_candidates.size:
                    rec = int(rng.choice(rec_candidates))
                    delta = cand[donor] * rng.uniform(0.05, 0.2)
                    cand[donor] -= delta
                    cand[rec] += delta

        # ------------------------------------------------------------------
        # (d) Tiny random permutation of a short block – adds diversity.
        # ------------------------------------------------------------------
        else:
            blk_len = int(rng.integers(2, min(30, n // 10) + 1))
            start = int(rng.integers(0, n - blk_len + 1))
            block = cand[start:start + blk_len]
            rng.shuffle(block)
            cand[start:start + blk_len] = block

        # ------------------------------------------------------------------
        # Clip to admissible range and evaluate.
        # ------------------------------------------------------------------
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            if best_score < 1.50274349:      # target already reached
                break

    # ------------------------------------------------------------------
    # 3) Final sanity – return a valid list of floats.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score) or best.size == 0:
        best = np.clip(base, 0.0, 1000.0)
    return [float(v) for v in best]
