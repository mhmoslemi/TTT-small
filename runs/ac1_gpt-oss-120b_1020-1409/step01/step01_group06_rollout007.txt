import numpy as np

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum coefficient of the full self‑convolution of a non‑negative array."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact value of the official evaluator (without calling evaluate_sequence)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p); optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Small stochastic block scaling while preserving the L1‑norm."""
    cand = vec.copy()
    n = cand.size
    # block width in [1, n/12]
    lo_w = max(1, n // 200)
    hi_w = max(2, n // 12) + 1
    width = int(rng.integers(lo_w, hi_w))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    cand[start:start + width] *= factor
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    """
    Return a non‑negative step‑function (list of floats) whose AC1 upper bound
    improves over the naive value 2.0.
    The routine uses a mixture of analytic families and stochastic local search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and a viable starting vector (fallback to uniform)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
        init_vec = parent.copy()
    except Exception:
        n = 256
        init_vec = np.full(n, 1.0, dtype=np.float64)

    target_sum = np.sqrt(2.0 * n)          # scaling used by many constructions

    # ------------------------------------------------------------------
    # Normalise a candidate to the target L1‑norm and clip to the allowed range
    def _prepare(v):
        v = np.clip(v, 0.0, 1000.0, out=v)
        s = v.sum()
        if s > 0.0:
            v *= target_sum / s
        return v

    best_vec = np.full(n, target_sum / n, dtype=np.float64)   # uniform fallback
    best_vec = _prepare(best_vec)
    best_score = _score(best_vec, n)

    # ------------------------------------------------------------------
    # Try the (possibly) provided parent after normalisation
    parent_prepared = _prepare(init_vec.copy())
    sc = _score(parent_prepared, n)
    if sc < best_score:

    # ------------------------------------------------------------------
    # Grid search over power‑law families (quick deterministic sweep)
    p_vals = np.linspace(0.40, 0.60, 9)               # around the observed optimum

    for sym in (False, True):
        for p in p_vals:

                cand = _prepare(cand)
                sc = _score(cand, n)
                if sc < best_score:
                    best_score, best_vec = sc, cand.copy()

    # ------------------------------------------------------------------
    # Stochastic refinement loop (mix of perturbations, permutations, smoothing, fresh draws)
    total_iters = 1500
    for _ in range(total_iters):
        op = rng.integers(0, 5)   # 0‑block,1‑perm,2‑global‑scale,3‑smooth,4‑random‑draw
        if op == 0:

        elif op == 1:
            perm = rng.permutation(n)
            cand = best_vec[perm].copy()          # permutation keeps the L1‑norm
        elif op == 2:

            cand = _prepare(cand)
        elif op == 3:

        sc = _score(cand, n)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()

    # ------------------------------------------------------------------
    # Return a plain Python list – the harness will run the official evaluator.
    return [float(x) for x in best_vec]
