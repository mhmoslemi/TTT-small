import numpy as np

def propose_candidate(seed=42, num_iters=4000, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine length n and optional parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # Helper normalize
    def normalize(v):
        s = v.sum()
        if s > 0 and np.isfinite(s):
            return v / s
        else:
            return np.ones_like(v) / v.size

    # Initialize candidates
    candidates = []
    scores = []

    # 1. Parent (if any)
    if parent_ok:
        pseq = np.clip(parent_arr, 0.0, 1000.0)
        pseq = normalize(pseq)
        candidates.append(pseq)
        scores.append(float(evaluate_sequence(pseq.tolist())))

    # 2. Power-law family
    ids = np.arange(1, n+1, dtype=np.float64)
    for p in np.linspace(0.2, 3.0, 16):
        seq = ids ** (-p)
        seq = normalize(seq)
        candidates.append(seq)
        scores.append(float(evaluate_sequence(seq.tolist())))

    # 3. Exponential family
    for lam in np.linspace(0.0, 5.0 / n, 12):
        seq = np.exp(-lam * (ids - 1))
        seq = normalize(seq)
        candidates.append(seq)
        scores.append(float(evaluate_sequence(seq.tolist())))

    # Choose best initial sequence
    best_idx = int(np.argmin(scores))
    a = candidates[best_idx].copy()
    best = a.copy()
    best_score = scores[best_idx]

    # Optimization hyperparameters
    step0 = 0.30          # initial step for exponentiated updates
    top_k = 5
    weight_coef = 5.0     # exponent coefficient for softmax weights
    perturb_interval = 101
    block_min = max(1, n // 300)
    block_max = max(2, n // 30)

    for it in range(num_iters):
        # Full convolution
        conv = np.convolve(a, a)

        # Determine top indices (largest values)
        if top_k >= conv.size:
            ks = np.arange(conv.size)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[ks]
        # Softmax-like weights emphasizing larger values
        maxv = vals.max()
        w = np.exp(weight_coef * (vals - maxv))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.ones_like(vals) / len(vals)
        else:
            w = w / w_sum

        # Gradient accumulator
        g = np.zeros_like(a)
        for kk, wk in zip(ks, w):
            k = int(kk)
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx_range = np.arange(low, high + 1, dtype=np.int64)
            partner = k - idx_range
            # Accumulate weighted gradient contribution
            g[idx_range] += 2.0 * wk * a[partner]

        # Exponentiated gradient step with decaying step size
        step = step0 * (0.99 ** it)
        # Prevent overflow/underflow by capping the exponent
        exp_factor = np.exp(-step * g)
        a = a * exp_factor

        # Renormalise to keep L1 norm = 1 and enforce bounds
        a = np.clip(a, 0.0, 1000.0)
        a = normalize(a)

        # Occasional random block perturbation to escape plateaus
        if (it % perturb_interval) == 0 and rng.random() < 0.30:
            width = int(rng.integers(block_min, block_max + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.14)))
            pert = a.copy()
            pert[start:start+width] *= factor

        if np.isfinite(cur_score) and cur_score < best_score:

            best_score = cur_score

import numpy as np

def _normalize(v):
    """Scale vector to have L1‑norm 1, keeping non‑negativity."""
    s = v.sum()
    if s > 0 and np.isfinite(s):
        return v / s
    # fallback to uniform distribution on failure
    return np.ones_like(v, dtype=np.float64) / v.size

def propose_candidate(seed=42, num_iters=4000, **kwargs):
    """
    Search for a non‑negative step sequence with a low AC‑1 upper bound.
    Uses a soft‑max weighted exponentiated‑gradient descent on the
    maximal convolution entries together with occasional random block
    perturbations. Falls back to simple power‑law and exponential
    initialisations when a parent is unavailable.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and optional parent sequence.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = (
            parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # ------------------------------------------------------------------
    # Build a small pool of initial candidates (parent, power‑law, expo).
    # ------------------------------------------------------------------
    cand_seqs = []
    cand_scores = []

    if parent_ok:
        seq = np.clip(parent_arr, 0.0, 1000.0)
        seq = _normalize(seq)
        cand_seqs.append(seq)
        cand_scores.append(float(evaluate_sequence(seq.tolist())))

    # power‑law shapes: a_i ∝ i^{‑p}
    ids = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.2, 3.0, 16):
        seq = ids ** (-p)
        seq = _normalize(seq)
        cand_seqs.append(seq)
        cand_scores.append(float(evaluate_sequence(seq.tolist())))

    # exponential shapes: a_i ∝ exp(‑λ·i)
    for lam in np.linspace(0.0, 5.0 / n, 12):
        seq = np.exp(-lam * (ids - 1))
        seq = _normalize(seq)
        cand_seqs.append(seq)
        cand_scores.append(float(evaluate_sequence(seq.tolist())))

    # ------------------------------------------------------------------
    # Pick the best among the pool as the starting point.
    # ------------------------------------------------------------------
    best_idx = int(np.argmin(cand_scores))
    a = cand_seqs[best_idx].copy()
    best = a.copy()
    best_score = cand_scores[best_idx]

    # ------------------------------------------------------------------
    # Hyper‑parameters for the optimisation loop.
    # ------------------------------------------------------------------
    step0 = 0.30          # initial multiplicative step
    top_k = 5             # number of largest convolution entries to address
    weight_coef = 5.0     # soft‑max sharpness for the top entries
    perturb_interval = 101
    block_min = max(1, n // 300)
    block_max = max(2, n // 30)

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    for it in range(num_iters):
        # Full linear convolution (length 2n‑1)
        conv = np.convolve(a, a)

        # Identify the indices of the top‑k convolution values
        if top_k >= conv.size:
            ks = np.arange(conv.size, dtype=np.int64)
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[ks]

        # Soft‑max weights, emphasising the largest values
        maxv = vals.max()
        w = np.exp(weight_coef * (vals - maxv))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.ones_like(vals) / vals.size
        else:
            w = w / w_sum

        # Accumulate a (weighted) sub‑gradient of the selected entries
        g = np.zeros_like(a)
        for kk, wk in zip(ks, w):
            k = int(kk)
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx = np.arange(low, high + 1, dtype=np.int64)
            partner = k - idx
            g[idx] += 2.0 * wk * a[partner]

        # Exponentiated‑gradient update with a decaying step size
        step = step0 * (0.99 ** it)
        a = a * np.exp(-step * g)

        # Enforce non‑negativity, clip to the hard bound, and renormalise
        np.clip(a, 0.0, 1000.0, out=a)
        a = _normalize(a)

        # --------------------------------------------------------------
        # Occasional random block perturbation (exploration)
        # --------------------------------------------------------------
        if (it % perturb_interval) == 0 and rng.random() < 0.30:
            width = int(rng.integers(block_min, block_max + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.14)))
            pert = a.copy()
            pert[start:start + width] *= factor
            np.clip(pert, 0.0, 1000.0, out=pert)
            pert = _normalize(pert)
            score = float(evaluate_sequence(pert.tolist()))
            if np.isfinite(score) and score < best_score:
                best = pert.copy()
                best_score = score
                a = pert  # continue optimisation from the improvement

        # --------------------------------------------------------------
        # Record best solution seen so far.
        # --------------------------------------------------------------
        cur_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best = a.copy()
            best_score = cur_score

    # ------------------------------------------------------------------
    # Return the best sequence (plain Python floats).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
