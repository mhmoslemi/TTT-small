import numpy as np

def _power_seq(p, alpha, n):
    """Generate a_i = (i+alpha)^(-p) for i=0..n-1."""
    seq = (np.arange(n, dtype=np.float64) + alpha) ** (-p)
    return np.clip(seq, 0.0, 1000.0)


def _exp_seq(tau, n):
    """Generate a_i = exp(-i/tau)."""
    seq = np.exp(-np.arange(n, dtype=np.float64) / tau)
    return np.clip(seq, 0.0, 1000.0)


def _linear_desc_seq(n):
    """Linear decreasing from 1 to 0."""
    seq = 1.0 - np.arange(n, dtype=np.float64) / (n - 1)
    return np.clip(seq, 0.0, 1000.0)


def _permute_seq(seq, step):
    """Reorder seq by taking indices 0,step,2*step,… then 1,step+1,… etc."""
    n = seq.size
    order_parts = [np.arange(off, n, step) for off in range(step)]
    order = np.concatenate(order_parts)
    return seq[order]


def _evaluate(seq):
    """Helper to evaluate a candidate list."""
    # convert to plain Python list for the external evaluator
    return float(evaluate_sequence(seq.tolist()))

    best = np.clip(parent, 0.0, 1000.0)
