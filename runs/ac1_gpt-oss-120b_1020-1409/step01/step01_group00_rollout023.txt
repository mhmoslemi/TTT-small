import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """
    a_i = C(2i,i) / 4^i  for i=0..length-1.
    This gives an exact convolution a*a = 1 (up to truncation).
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        f = float(i)
        a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
    return a


def propose_candidate(seed: int = 42, max_steps: int = 2000, **kwargs):
    """
    Returns a list of non‑negative coefficients (step‑function heights)
    that aim to minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    The routine starts from the binomial sequence (or the provided
    `height_sequence_1` if present) and performs a simple hill‑climb that
    repeatedly transfers a small fraction of mass from the current largest
    entry to the current smallest entry.  Accepted moves strictly lower the
    bound; occasional random pair moves help escape plateaus.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise the base sequence.
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` is supplied by the harness when available.
        n = int(height_sequence_1.shape[0])
        base = np.array(height_sequence_1, dtype=np.float64, copy=True)
    except Exception:
        n = 256
        base = _generate_binomial_sequence(n)

    # Ensure all entries lie in the allowed interval.
    np.clip(base, 0.0, 1000.0, out=base)

    # ------------------------------------------------------------------
    # Baseline score.
    # ------------------------------------------------------------------
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    candidate = best.copy()

    # Target bound (for early stopping).  Slightly above the required
    # threshold to accommodate rounding noise.
    TARGET = 1.50274349

    # ------------------------------------------------------------------
    # Main hill‑climbing loop.
    # ------------------------------------------------------------------
    for step in range(max_steps):
        # Move a fraction of mass from the largest to the smallest entry.
        idx_max = int(np.argmax(candidate))
        idx_min = int(np.argmin(candidate))
        if idx_max == idx_min:
            break

        # Transfer 2 % of the donor value (adjustable fraction).
        delta = candidate[idx_max] * 0.02
        if delta <= 0.0:
            break

        cand = candidate.copy()
        cand[idx_max] -= delta
        cand[idx_min] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc
            candidate = cand
            if best_score <= TARGET:
                break
            continue  # successful move, try another

        # ------------------------------------------------------------------
        # If the deterministic move failed, try a random pair perturbation
        # every few iterations to possibly escape a local plateau.
        # ------------------------------------------------------------------
        if step % 37 == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and candidate[i] > 0.0:
                d = candidate[i] * rng.uniform(0.001, 0.03)
                cand2 = candidate.copy()
                cand2[i] = max(0.0, cand2[i] - d)
                cand2[j] = min(1000.0, cand2[j] + d)
                sc2 = float(evaluate_sequence(cand2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best = cand2.copy()
                    best_score = sc2
                    candidate = cand2
                    if best_score <= TARGET:
                        break

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
