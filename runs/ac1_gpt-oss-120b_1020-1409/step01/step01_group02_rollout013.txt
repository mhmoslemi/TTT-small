def _arcsine_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1,n):
        a[i] = a[i-1] * (1.0 - 0.5 / i)
    return a

def _objective(seq):
    # seq is numpy array
    n = seq.size
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    # compute convolution using FFT
    fft_len = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

def _parent_sequence():
    """Return the provided parent sequence if it exists, else a fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            np.clip(arr, 0.0, 1000.0, out=arr)
            return arr
    except Exception:  # pragma: no cover
        pass
    # fallback – a short arcsine vector
    return _arcsine_sequence(256)


def _arcsine_sequence(n):
    """Generate the truncated arcsine coefficients."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _objective(seq):
    """Exact AC1 upper bound (scale‑invariant) using FFT convolution."""
    n = seq.size
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=42, num_steps=20000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the AC1 bound.
    Uses a fast FFT‑based convolution and a gradient‑guided mass‑transfer
    heuristic with occasional random perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = parent.size

    arcsine = _arcsine_sequence(n)

    parent_score = _objective(parent)
    arcsine_score = _objective(arcsine)

    if arcsine_score < parent_score:
        best = arcsine.copy()
        best_score = arcsine_score
    else:
        best = parent.copy()
        best_score = parent_score

    current = best.copy()
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # ----- convolution and max index ---------------------------------
        fa = np.fft.rfft(current, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        idx_max = int(np.argmax(conv))

        # ----- gradient of the maximal entry ----------------------------
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        partner = idx_max - idx_range

        grad = np.zeros_like(current)
        mask = idx_range != partner
        grad[idx_range[mask]] = 2.0 * current[partner[mask]]
        grad[idx_range[~mask]] = current[partner[~mask]]  # diagonal terms

        # ----- choose source (high impact) and target (low impact) ------
        src_scores = grad * current
        i_src = int(np.argmax(src_scores))
        if current[i_src] <= 0.0:
            continue

        sorted_idx = np.argsort(grad)
        i_tgt = int(sorted_idx[0])
        if i_tgt == i_src:
            if sorted_idx.size > 1:
                i_tgt = int(sorted_idx[1])
            else:
                continue

        # ----- transfer a modest amount of mass -------------------------
        max_transfer = current[i_src] * 0.5
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[i_src] -= delta
        cand[i_tgt] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _objective(cand)

        # ----- accept if improvement ------------------------------------
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            current = cand
            continue

        temperature = 1.0 / (step + 1.0)
        if np.exp((best_score - cand_score) / temperature) > rng.random():
            current = cand
            continue

        # ----- occasional random block perturbation --------------------
        if rng.random() < 0.003:

            start = int(rng.integers(0, max(1, n - width + 1)))

    # safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)
