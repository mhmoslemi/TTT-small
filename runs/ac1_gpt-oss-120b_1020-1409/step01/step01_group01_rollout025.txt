def compute_objective(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a*fft_a, n=L)
    conv = conv[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a**2)
```

    import numpy as np
    rng = np.random.default_rng(seed)
    parent = globals().get('height_sequence_1')
    if isinstance(parent, np.ndarray) and parent.ndim==1 and parent.size>0:
        n = int(parent.shape[0])
    else:
        n = 256
    # initial seq: r-sequence
    a = np.empty(n, dtype=np.float64)
    r = 1.0
    a[0] = r
    for i in range(1, n):
        r = r * (2*i - 1) / (2*i)
        a[i] = r
    a = np.clip(a, 0.0, 1000.0)
    best_a = a.copy()
    best_score = compute_objective(best_a)
    # parametric search over exponents
    exponents = np.linspace(0.2, 1.5, 14)
    for alpha in exponents:
        cand = (np.arange(1, n+1, dtype=np.float64) ** (-alpha))
        cand = np.clip(cand, 0.0, 1000.0)
        score = compute_objective(cand)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
    # local hill climbing
    # define step size delta decreasing
    delta0 = best_a.max() * 0.1  # initial step size
    for it in range(4000):
        # pick i,j distinct
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # propose transfer delta
        # we can sample delta proportional to current values
        delta = delta0 * (rng.random() - 0.5)  # can be positive or negative; we interpret as adding to i, subtract from j
        # Actually we want to modify both i and j preserving sum: propose new values
        new_a = best_a.copy()
        # adjust i and j by delta, ensure nonnegative and <=1000
        vi = new_a[i] + delta
        vj = new_a[j] - delta
        if vi < 0 or vj < 0 or vi > 1000 or vj > 1000:
            continue
        new_a[i] = vi
        new_a[j] = vj
        score = compute_objective(new_a)
        if score < best_score:
            best_score = score
            best_a = new_a
        # optional annealing: reduce delta gradually
        delta0 *= 0.9999
    # fallback ensure valid
    if not np.isfinite(best_score):
        best_a = np.ones(n, dtype=np.float64)

def compute_objective(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    # Real FFT
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)
    max_b = conv[:2*n-1].max()
    sum_a = a.sum()
    if sum_a <= 0:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)
    parent = globals().get('height_sequence_1')
    if isinstance(parent, np.ndarray):
        n = parent.shape[0]
    else:
        n = 256

    # initial sequence: r sequence
    a = np.empty(n, dtype=np.float64)
    r = 1.0
    a[0] = r
    for i in range(1, n):
        r = r * (2*i - 1) / (2*i)
        a[i] = r
    a = np.clip(a, 0.0, 1000.0)

    def compute_obj(x):
        L = 1 << ((2*n-1).bit_length())
        fa = np.fft.rfft(x, n=L)
        conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
        maxb = conv.max()
        s = x.sum()
        if s <= 0:
            return np.inf
        return 2.0 * n * maxb / (s * s)

    best = a.copy()
    best_score = compute_obj(best)

    # explore parametric shapes
    exps = np.linspace(0.3, 1.2, 10)
    for alpha in exps:
        cand = (np.arange(1, n+1, dtype=np.float64) ** (-alpha))
        cand = np.clip(cand, 0.0, 1000.0)
        sc = compute_obj(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # hill climbing flatten extremes
    step = best.max() * 0.1
    for it in range(5000):
        i_high = int(best.argmax())
        i_low = int(best.argmin())
        if i_high == i_low:
            break
        delta = step * (rng.random() * 0.5 + 0.5)  # between 0.5*step and step
        # ensure delta not exceed bounds

            delta = best[i_high] * 0.9

        cand[i_high] -= delta

        # ensure nonnegative

        sc = compute_obj(cand)
        if sc < best_score:

            step *= 0.999

    return [float(x) for x in best]

    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size:

        a[i] = r
    a = np.clip(a, 0.0, 1000.0)

    # precompute FFT length
    L = 1 << ((2*n - 1).bit_length())
    def obj(x):

        conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
        maxb = conv.max()
        s = x.sum()
        if s <= 0:
            return np.inf
        return 2.0 * n * maxb / (s * s)

    best = a.copy()
    best_score = obj(best)

    # try simple power‑law families
    exps = np.linspace(0.3, 1.2, 12)
    idx = np.arange(1, n+1, dtype=np.float64)
    for alpha in exps:
        cand = idx ** (-alpha)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = obj(cand)
        if sc < best_score:
            best = cand
            best_score = sc

    # systematic transfer from large to small entries
    top_idx = np.argpartition(-best, 5)[:5]
    low_idx = np.argpartition(best, 5)[:5]

                cand[tgt] += delta
                if cand[src] < 0 or cand[tgt] > 1000:
                    continue
                sc = obj(cand)
                if sc < best_score:
                    best = cand
                    best_score = sc

    # batch transfer to a block of tail entries
