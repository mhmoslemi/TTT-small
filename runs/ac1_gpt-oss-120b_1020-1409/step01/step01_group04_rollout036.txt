import numpy as np

def _full_convolution(seq, fft_size):
    """Return full linear convolution of seq with itself using FFT."""
    fa = np.fft.rfft(seq, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)
    return conv[:2 * seq.shape[0] - 1]

def _score(seq, fft_size):
    n = seq.shape[0]
    conv = _full_convolution(seq, fft_size)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a <= 0.0:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

import numpy as np

def _full_convolution(seq: np.ndarray, fft_size: int) -> np.ndarray:
    """Linear convolution of seq with itself using FFT (returns length 2*len-1)."""
    fa = np.fft.rfft(seq, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)
    return conv[: 2 * seq.shape[0] - 1]


def _score(seq: np.ndarray, fft_size: int) -> float:
    """AC1 upper bound for a non‑negative sequence."""
    n = seq.shape[0]
    conv = _full_convolution(seq, fft_size)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a <= 0.0:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))
