    idx = np.arange(n, dtype=np.float64) + 1.0
    seq = idx ** (-p)
    total = seq.sum()
    if total > 0.0:
        seq /= total
    return seq


def _exp_seq(lam: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalized to sum = 1."""
    seq = np.exp(-lam * np.arange(n, dtype=np.float64))
    total = seq.sum()
    if total > 0.0:
        seq /= total
    return seq


def _combined_seq(p: float, lam: float, n: int) -> np.ndarray:
    """product of power law and exponential, then normalized."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    seq = (idx ** (-p)) * np.exp(-lam * np.arange(n, dtype=np.float64))
    total = seq.sum()
    if total > 0.0:
        seq /= total
    return seq

    strives to minimise the AC1 upper bound.  The routine builds a set of
    parametric candidates, refines the best one with cheap stochastic
    perturbations and finally returns the best list found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n from the supplied parent construction if present.
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])   # parent array supplied by the harness

    # Keep track of the best sequence encountered.
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)  # uniform start
    best_score = _ac1_score(best_seq)

    # ------------------------------------------------------------------
    # 1) Grid search over simple parametric families.
    # ------------------------------------------------------------------
    # Power‑law family a_i ∝ (i+1)^(-p)
    for p in np.linspace(0.1, 0.9, 17):          # 0.1 … 0.9  (inclusive)
        cand = _power_seq(p, n)
        sc = _ac1_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # Exponential decay family a_i ∝ exp(-lam·i)
    for lam in np.logspace(-6, -1, 15):          # 1e‑6 … 1e‑1
        cand = _exp_seq(lam, n)
        sc = _ac1_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # Combined power‑law × exponential family
    for p in np.linspace(0.2, 0.8, 7):           # coarse grid for p
        base = (np.arange(n, dtype=np.float64) + 1.0) ** (-p)
        for lam in np.logspace(-5, -2, 5):      # 1e‑5 … 1e‑2
            cand = base * np.exp(-lam * np.arange(n, dtype=np.float64))
            total = cand.sum()
            if total > 0.0:
                cand /= total
            sc = _ac1_score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc

    # ------------------------------------------------------------------
    # 2) Stochastic local refinement of the current best.
    # ------------------------------------------------------------------
    steps = 500
    max_width = max(1, n // 100)                # at most ~1 % of length
    for _ in range(steps):
        cand = best_seq.copy()
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        # multiplicative log‑normal perturbation of a contiguous block
        factor = float(np.exp(rng.normal(0.0, 0.05)))   # σ≈5 %
        cand[start:start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)   # keep inside evaluator limits
        s = cand.sum()
        if s > 0.0:
            cand /= s
        sc = _ac1_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc

    # ------------------------------------------------------------------
    # Defensive final check – guarantee a valid non‑empty list.
    # ------------------------------------------------------------------
    result = best_seq.tolist()
    if not result or any(not np.isfinite(x) for x in result):
        result = [1.0 / n] * n
    return result
