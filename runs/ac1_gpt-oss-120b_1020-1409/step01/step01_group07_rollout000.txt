def _uniform_sequence(n):
    """Return a length‑n array with equal non‑negative entries summing to 1."""
    if n <= 0:
        raise ValueError("Length must be positive")
    arr = np.empty(n, dtype=np.float64)
    arr.fill(1.0 / n)
    return arr
