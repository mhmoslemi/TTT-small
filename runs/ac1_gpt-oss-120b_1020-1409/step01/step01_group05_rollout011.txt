def greedy_set(n, rng, max_count):
    sum_counts = np.zeros(2*n-1, dtype=np.int16)
    positions = []
    used = np.zeros(n, dtype=bool)
    order = rng.permutation(n)  # random order
    for x in order:
        # try to add
        diag_ok = sum_counts[2*x] + 1 <= max_count
        if not diag_ok:
            continue
        # check collisions with existing
        # vectorized check: compute s = x + positions array
        s_arr = x + np.array(positions, dtype=np.int64)
        if np.any(sum_counts[s_arr] + 2 > max_count):

        positions.append(int(x))
        used[int(x)] = True
        sum_counts[2*x] += 1
        sum_counts[s_arr] += 2
    return positions

def build_sparse_sequence(n, rng, max_cnt=8, trials=30):
    best_positions = None
    best_score = float('inf')
    for t in range(trials):
        # random permutation order
        order = rng.permutation(n)
        sum_counts = np.zeros(2*n-1, dtype=np.int16)
        positions = []
        for x in order:
            # compute indices for checking: need to check all existing positions
            diag_ok = sum_counts[2*x] + 1 <= max_cnt
            if not diag_ok:
                continue
            # Use vectorized for quick check
            if positions:
                arr = np.array(positions, dtype=np.int32)
                s_arr = x + arr
                # check condition for each
                if np.any(sum_counts[s_arr] + 2 > max_cnt):
                    continue
                # accept
                sum_counts[s_arr] += 2
            # accept x
            sum_counts[2*x] += 1
            positions.append(int(x))
        # compute score bound
        m = len(positions)
        if m == 0:
            continue
        # compute max of sum_counts, which is the max ordered pairs per sum
        cur_maxcnt = int(sum_counts.max())
        # compute bound based on uniform equal weights
        bound_est = 2 * n * cur_maxcnt / (m * m)
        # evaluate precisely
        a = np.zeros(n, dtype=np.float64)
        a[positions] = 1.0 / m
        # evaluate using evaluate_sequence
        cur_score = float(evaluate_sequence(a.tolist()))
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            best_positions = positions.copy()
        # optional early exit if best_score < target
        if best_score < 1.50274349:
            break
    return best.tolist()

import numpy as np
import math

# ------------------------------------------------------------
# Helper – construct a sparse non‑negative step sequence.
# ------------------------------------------------------------
def _build_sparse(n: int, rng: np.random.Generator,
                  max_cnt: int = 8, trials: int = 80):
    """
    Try several random greedy packings of positions so that the number of
    ordered pair contributions to any convolution index never exceeds ``max_cnt``.
    Returns the best sequence (as a NumPy array) and its score.
    """
    best_seq = None
    best_score = float('inf')

    # target score for early stopping (slightly better than the required bound)
    target_score = 1.50274349

    # length of the convolution result
    conv_len = 2 * n - 1

    for _ in range(trials):
        order = rng.permutation(n)                     # random ordering of indices
        cnt = np.zeros(conv_len, dtype=np.int16)       # ordered‑pair counts per sum
        positions = []                                 # selected indices

        for x in order:
            diag = 2 * x
            if cnt[diag] + 1 > max_cnt:               # diagonal would overflow
                continue

            if positions:
                cur = np.array(positions, dtype=np.int32)
                sums = x + cur                         # candidate sums with existing
                if np.any(cnt[sums] + 2 > max_cnt):    # any off‑diagonal overflow?
                    continue
                # accept: update off‑diagonal contributions
                cnt[sums] += 2

            # accept: update diagonal contribution
            cnt[diag] += 1
            positions.append(int(x))

        m = len(positions)
        if m == 0:
            continue

        # equal weight on every kept position (sum = 1)
        seq = np.zeros(n, dtype=np.float64)
        seq[positions] = 1.0 / m

        # evaluate with the supplied objective
        score = float(evaluate_sequence(seq.tolist()))

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            if best_score < target_score:
                break

    return best_seq, best_score

def propose_candidate(seed=42, max_cnt=8, trials=80, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 bound
    is as low as possible.  The routine tries a sparse construction
    based on bounded convolution multiplicities and falls back to a
    simple uniform distribution or the provided parent sequence.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length of the problem (use the parent if present).
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.size)
        parent_ok = (
            parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
        )
    except Exception:
        n = 2000                     # generic fallback length
        parent_ok = False

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (sum = 1) – always feasible.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # If a parent sequence exists, normalise it and consider it.
    # ------------------------------------------------------------------
    if parent_ok:
        p = np.clip(parent_arr, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0 and np.isfinite(s):
            p = p / s
        else:
            p = uniform.copy()
        p_score = float(evaluate_sequence(p.tolist()))
        if p_score < best_score:
            best_seq, best_score = p, p_score

    # ------------------------------------------------------------------
    # Try the sparse construction – many attempts with different random
    # orders; keep the best result.
    # ------------------------------------------------------------------
    sparse_seq, sparse_score = _build_sparse(n, rng, max_cnt=max_cnt, trials=trials)
    if sparse_seq is not None and sparse_score < best_score:
        best_seq, best_score = sparse_seq, sparse_score

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
