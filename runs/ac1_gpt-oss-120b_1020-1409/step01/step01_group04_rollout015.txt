import numpy as np

# ----------------------------------------------------------------------
# Convolution using FFT (linear convolution, not circular)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return linear convolution a * a using real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective used by the evaluator (exactly the same formula)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Compute the AC1 upper bound for a non‑negative sequence a."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _conv_fft(a)
    max_b = conv.max()
    n = a.shape[0]
    return (2.0 * n * max_b) / (s * s)
