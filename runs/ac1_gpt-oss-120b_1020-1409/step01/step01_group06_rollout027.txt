def _conv_max_and_idx(arr):
    n = arr.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    idx = int(np.argmax(conv))
    maxv = float(conv[idx])
    return maxv, idx

def _conv_max_fft(arr):
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _conv_max_and_idx_fft(arr):

import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear‑time memory)
def _conv_max_and_idx_fft(arr: np.ndarray):
    """Return (max value, index) of the full convolution arr*arr."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _score_fft(arr: np.ndarray):
    """Exact evaluator formula, using the fast convolution."""
    n = arr.shape[0]
    maxb, _ = _conv_max_and_idx_fft(arr)
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * maxb / (s * s)

# Simple parametric families
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^{-p}, optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base

def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float):
    n = vec.size
    cand = vec.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.2)))   # multiplicative jitter
    cand[start:start + width] *= factor
    if rng.random() < 0.2:                         # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand
