def _conv_self(x):
    n = len(x)
    conv_len = 2*n-1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx*fx, n=fft_len)[:conv_len]
    return conv

def _score(x):
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(x)
    return 2.0 * len(x) * c.max() / (s*s)

def _subgradient(x, conv, M):
    # compute gradient of M wrt x (subgradient)
    # S indices where conv ~ M
    # We'll use mask of exact equality
    tol = 1e-12 # maybe relative
    # Actually we can find where conv is close to M within small epsilon: M - conv <= eps
    # Choose eps = max(1e-12, 1e-10 * M) maybe
    eps = max(1e-12, 1e-10 * M)
    idxs = np.where(conv >= M - eps)[0]
    g = np.zeros_like(x)
    n = len(x)
    for k in idxs:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # slice indices
        i_range = np.arange(i_low, i_high+1)
        # correspond reversed index: x[k - i] where i range
        g[i_range] += x[k - i_range]
    return g

def _gradient(x):
    n = len(x)
    conv = _conv_self(x)
    M = conv.max()
    g_m = _subgradient(x, conv, M)
    S = x.sum()
    coeff = 2.0 * n
    c = coeff / (S*S)
    d = 2.0 * coeff * M / (S*S*S)   # = 4nM/S^3
    grad = c * g_m - d
    return grad, M, S

def _subgrad_maxconv(x, conv, M):
    # create mask of indices where conv >= M - eps
    eps = max(1e-12, 1e-10 * abs(M))
    mask_indices = np.where(conv >= M - eps)[0]
    n = len(x)
    grad = np.zeros_like(x)
    for k in mask_indices:
        # compute overlapped region.
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # Use slice
        i = np.arange(lo, hi+1, dtype=int)
        grad[i] += x[k - i]
    return grad

import numpy as np

# ------------------------------------------------------------
# Helper: a simple analytic sequence (fallback initialization)
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return truncation of a_i = C(2i,i)/4^i (a_i decreases slowly)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast full convolution via real FFT (linear, O(n log n))
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact score (mirrors the global evaluate_sequence)
# ------------------------------------------------------------
def _score(x: np.ndarray) -> float:
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(x)
    return 2.0 * x.size * c.max() / (s * s)


# ------------------------------------------------------------
# Subgradient of the maximal convolution entry.
# ------------------------------------------------------------
def _subgrad_maxconv(x: np.ndarray, conv: np.ndarray, M: float) -> np.ndarray:
    """
    Return a subgradient of g(x)=max(conv(x,x)) at x.
    conv is the full self‑convolution of x, M=conv.max().
    """
    n = x.shape[0]
    # Treat all indices that are within a tiny tolerance of the max as active.
    eps = max(1e-12, 1e-10 * abs(M))
    active = np.where(conv >= M - eps)[0]

    grad = np.zeros_like(x)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution from x_{k-i}
        grad[idx] += x[k - idx]
    return grad


# ------------------------------------------------------------
# Gradient of the AC1 bound f(x) = 2 n * M / S^2
# ------------------------------------------------------------
def _grad_f(x: np.ndarray):
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    gM = _subgrad_maxconv(x, conv, M)          # ∂M/∂x (subgradient)
    S = x.sum()
    coeff = 2.0 * n
    c = coeff / (S * S)
    d = 2.0 * coeff * M / (S * S * S)          # scalar term from ∂S/∂x = 1
    grad = c * gM - d
    return grad, M, S
