import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the sequence r_i = C(2i,i)/4^i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a step‑function (list of non‑negative floats) that lowers the AC1 bound.
    The routine refines the base sequence by greedy block extensions from both
    ends with a decreasing schedule of epsilons, followed by a short random
    local search.
    """

    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
        best = parent.astype(np.float64).copy()
    else:
        n = 256
        best = generate_r_sequence(n)

    best = np.clip(best, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # ----- epsilon schedule -------------------------------------------------
    eps = 0.01
    epsilons = []
    while eps >= 1e-6:
        epsilons.append(eps)
        eps *= 0.5   # halve each step

    # ----- greedy block extensions (front & back) -------------------------
    max_passes = 2                     # repeat the whole schedule a few times
    for _ in range(max_passes):
        any_improved = False
        for eps in epsilons:
            # ---- extend from the front (indices 1,2,…) --------------------
            k = 0
            step = max(1, n // 10)
            while step >= 1:
                improved = False
                while True:
                    new_k = min(k + step, n - 1)          # never touch index 0
                    if new_k == k:
                        break

                    cand[1 + k:1 + new_k] += eps
                    np.clip(cand, 0.0, 1000.0, out=cand)
                    sc = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(sc) and sc < best_score - 1e-10:
                        best = cand
                        best_score = sc
                        k = new_k
                        improved = True
                        any_improved = True
                    else:
                        break

                    step //= 2

            k_tail = 0
            step = max(1, n // 10)

                    if new_k == k_tail:
                        break
                    cand = best.copy()
                    start = n - new_k

    for _ in range(200):
