import numpy as np

# ------------------------------------------------------------
# Helper: construct a monotone power‑law sequence.
# The sequence is decreasing from the first element to the last,
# scaled so that the maximum entry does not exceed 1000.
# ------------------------------------------------------------
def _power_sequence(n: int, p: int) -> np.ndarray:
    """
    Return a (n,) float ndarray with entries (n-i)^p,
    scaled to keep the maximum ≤ 1000.
    """
    # descending integers: n, n-1, ..., 1
    base = np.arange(n, 0, -1, dtype=np.float64) ** p
    max_val = base.max()
    # Scale only if the peak would exceed the evaluator bound.
    if max_val > 1000.0:
        scale = 1000.0 / max_val
    else:
        scale = 1.0
    seq = base * scale
    # Final safety clamp.
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq
