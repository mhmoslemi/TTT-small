def apply_gradient(a, k_max, step):
    n = a.shape[0]
    idx = np.arange(n)
    rev_idx = k_max - idx
    mask = (rev_idx >=0) & (rev_idx < n)
    rev = np.zeros_like(a)
    rev[mask] = a[rev_idx[mask]]
    grad = a + rev
    a2 = a - step * grad
    a2 = np.clip(a2, 0.0, None)
    s = a2.sum()
    if s > 0:
        a2 = a2 / s
    else:
        a2 = np.full_like(a, 1.0/n)
    return a2

import numpy as np

def _power_seq(alpha, n):
    """One‑parameter family a_i = 1 / sqrt(i + alpha)."""
    seq = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + alpha)
    return np.clip(seq, 0.0, 1000.0)


def _exp_seq(r, n):
    """Geometric sequence a_i = r**i, clipped."""
    seq = r ** np.arange(n, dtype=np.float64)
    return np.clip(seq, 0.0, 1000.0)


def _score(seq):
    """Exact score used by the internal evaluator."""
    a = np.clip(seq, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * len(a) * c.max() / (s * s))


def _gradient(a, k):
    """Sub‑gradient of max_{shift} convolution at shift k."""
    n = a.shape[0]
    idx = np.arange(n)
    rev_idx = k - idx
    mask = (rev_idx >= 0) & (rev_idx < n)
    rev = np.zeros_like(a)
    rev[mask] = a[rev_idx[mask]]
    return a + rev


def propose_candidate(seed=42, num_alpha=35, num_r=30,
                     num_steps=1500, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose
    AC1 bound is small.  The routine combines a cheap family scan,
    random block perturbations and a sub‑gradient descent on the
    dominant convolution lag.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size from the supplied parent construction.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:               # no parent available
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # initialise best candidate – clip then normalise to unit sum
    best = np.clip(parent, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.full(n, 1.0)
    best = best / best.sum()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 1) Scan simple analytic families (power‑law and exponential).
    # ------------------------------------------------------------------
    alphas = np.logspace(-5, 1, num=num_alpha, dtype=np.float64)
    for alpha in alphas:
        cand = _power_seq(alpha, n)
        cand = cand / cand.sum()
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc

    rs = np.linspace(0.5, 0.99, num=num_r, dtype=np.float64)
    for r in rs:
        cand = _exp_seq(r, n)
        cand = cand / cand.sum()
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 2) Stochastic improvement + sub‑gradient descent.
    # ------------------------------------------------------------------
    min_width = max(1, n // 300)      # ~0.3 % of length
    max_width = max(2, n // 25)       # up to ~4 %
    sigma = 0.06                      # multiplicative noise scale
    base_step = 0.08                  # sub‑gradient step magnitude

    for it in range(num_steps):
        cand = best.copy()

        # ----- random perturbation -----
        if rng.random() < 0.65:                # block multiplicative
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor
        else:                                  # pairwise mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            delta = cand[i] * rng.random() * 0.15
            cand[i] -= delta
            cand[j] += delta

        # enforce feasibility and renormalise
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / s

        # ----- compute dominant convolution lag -----
        conv = np.convolve(cand, cand)
        k_max = int(conv.argmax())

        # ----- sub‑gradient descent step -----
        step = base_step / np.sqrt(it + 1.0)
        grad = _gradient(cand, k_max)
        cand = cand - step * grad
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / s

        # ----- evaluate candidate -----
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc

        # occasional random restart to avoid stagnation
        if it % 250 == 0 and rng.random() < 0.08:
            rnd = rng.random(n)
            rnd = rnd / rnd.sum()
            sc = _score(rnd)
            if np.isfinite(sc) and sc < best_score:
                best, best_score = rnd, sc

    # ------------------------------------------------------------------
    # 3) Final sanity check and return.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.clip(parent, 0.0, 1000.0)
        best = best / best.sum()

    return [float(x) for x in best]
