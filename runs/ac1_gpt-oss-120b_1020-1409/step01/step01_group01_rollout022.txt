import numpy as np
def generate_r_sequence(n):
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Catalan‑type sequence r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 upper bound
    is smaller than the baseline r‑sequence.  The algorithm moves a fraction
    of the early (large) coefficients to the later part of the vector while
    keeping the total ℓ¹ norm unchanged.  Each move is kept only if it strictly
    improves the evaluator score.
    """
    # ----------------------------------------------------------------------
    # 1.  Base sequence
    # ----------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        a = parent.astype(np.float64).copy()
        n = int(a.shape[0])
    else:
        n = 256
        a = generate_r_sequence(n)

    a = np.clip(a, 0.0, 1000.0, out=a)

    # ----------------------------------------------------------------------
    # 2.  Initial evaluation
    # ----------------------------------------------------------------------
    best = a
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # 3.  Deterministic mass‑redistribution passes
    # ----------------------------------------------------------------------
    # indices that will donate mass (the “early” part)
    early_limit = max(1, n // 4)                 # first quarter of the vector
    tail_start = early_limit                     # where the mass is received
    tail_len = n - tail_start

    # aggressive to gentle scaling factors for successive passes
    step_factors = [0.5, 0.2, 0.1, 0.05, 0.02]

    for step in step_factors:
        improved_this_pass = False

        for i in range(early_limit):
            if best[i] <= 1e-12:
                continue

            delta = best[i] * step                 # mass taken from position i
            if delta <= 0.0:
                continue

            inc = delta / tail_len                  # uniform addition to the tail

            # build a candidate
            cand = best.copy()
            cand[i] -= delta
            cand[tail_start:] += inc
            np.clip(cand, 0.0, 1000.0, out=cand)

            # evaluate
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                improved_this_pass = True

        # if the whole pass made no progress we simply go to the next (smaller) step
        if not improved_this_pass:
            continue

    # ----------------------------------------------------------------------
    # 4.  Fallback safety (should never trigger)
    # ----------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 5.  Return the final coefficient list
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
