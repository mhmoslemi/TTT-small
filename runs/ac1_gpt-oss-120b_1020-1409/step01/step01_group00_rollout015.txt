import numpy as np

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    # Helper evaluator
    def score(arr):
        return float(evaluate_sequence(arr.tolist()))

    # Initial candidates
    candidates = []

    # binomial
    bin_seq = _generate_binomial_sequence(n)
    candidates.append(bin_seq)

    # power law shapes
    indices = np.arange(n, dtype=np.float64)
    for p in np.linspace(0.2, 1.0, 17):
        for delta in (0.0, 0.5, 1.0):
            seq = 1.0/np.power(indices + delta + 1.0, p)  # shift +1 to avoid zero
            candidates.append(seq)

    # exponential
    for lam in np.linspace(0.0001, 0.005, 20):
        seq = np.exp(-lam*indices)
        candidates.append(seq)

    # normal distribution around middle
    for sigma in (n/10., n/5., n/3.):
        seq = np.exp(-0.5*((indices - (n-1)/2)/sigma)**2)
        candidates.append(seq)

    # Clip to [0,1000], as per evaluator will clip anyway but okay.
    for i in range(len(candidates)):
        np.clip(candidates[i], 0.0, 1000.0, out=candidates[i])

    # Find best initial
    best = None
    best_score = np.inf
    for seq in candidates:
        sc = score(seq)
        if np.isfinite(sc) and sc < best_score:
            best = seq.copy()
            best_score = sc

    # Local search: pair perturbation and block scaling
    # Use functions similar to previous but reduce interval count for speed.
    for step in range(1500):
        # pair perturbation: move some mass from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j and best[i] > 0:
            delta = best[i] * rng.uniform(0.001, 0.02)  # smaller delta
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc

        # occasional block scaling
        if step % 200 == 0:
            width = rng.integers(max(1, n//300), max(2, n//30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.05))
            cand = best.copy()
            cand[start:start+width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc

        # occasional random jitter to avoid stuck
        if step % 500 == 0:
            # small random additive noise to all entries
            noise = rng.normal(0.0, 0.01, size=n)
            cand = best + noise
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = score(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc

    return best.tolist()

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine n
    try:
        n = int(height_sequence_1.shape[0])
        base = _generate_binomial_sequence(n)
    except Exception:
        n = 256
        base = _generate_binomial_sequence(n)

    # Ensure not overly dense.
    best = base
    best_score = evaluate_sequence(best.tolist())

    # kernel lengths to try
    for m in [3,5,7,9,11,13,15,21,31,41,61]:
        kernel = np.full(m, 1.0/m, dtype=np.float64)
        cand = np.convolve(base, kernel, mode='same')
        # no scaling
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

        # double smoother (apply kernel twice)
        cand2 = np.convolve(cand, kernel, mode='same')
        sc2 = evaluate_sequence(cand2.tolist())
        if np.isfinite(sc2) and sc2 < best_score:
            best = cand2
            best_score = sc2

    # Further local search using pair/block perturbations for limited steps.
    num_steps = 500
    for step in range(num_steps):
        # random small adjustment with around 2% of randomly chosen entry
        i = rng.integers(0, n)
        delta = best[i] * rng.uniform(0.0, 0.02)
        cand = best.copy()
        cand[i] = max(0.0, cand[i] - delta)
        # distribute delta uniformly across a random block of entries
        block_len = rng.integers(1, min(20, n//10))
        block_start = rng.integers(0, n - block_len + 1)
        add = delta / block_len
        cand[block_start:block_start+block_len] += add
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

        if step % 100 == 0:
            # occasional random smoothing with random kernel length to escape local minima
            m = rng.choice([3,5,7,9,11,13,15])
            kernel = np.full(m, 1.0/m)
            cand = np.convolve(best, kernel, mode='same')
            sc = evaluate_sequence(cand.tolist())
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc

    return best.tolist()

import numpy as np

# ------------------------------------------------------------
# Helper: binomial step‑function (C(2i,i)/4^i)
# ------------------------------------------------------------
def _generate_binomial_sequence(length: int) -> np.ndarray:
    """Return a_i = C(2i,i)/4^i for i=0..length-1 (float64)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a

    The routine starts from the classic binomial sequence, explores a
    family of uniform smoothing kernels, and then performs a brief

    """

    def _score(arr: np.ndarray) -> float:
        """Return the AC1 bound for `arr` (list of floats)."""

    uniform = np.ones(n, dtype=np.float64)
