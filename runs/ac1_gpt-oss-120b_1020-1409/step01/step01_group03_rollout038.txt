    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2*i)

import numpy as np

# ------------------------------------------------------------
# Helper: generate the classic a_i = C(2i,i)/4^i sequence.
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the length‑`length` truncation of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast linear convolution via FFT (real‑valued input).
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 bound (mirrors evaluate_sequence).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    conv = _conv_self(seq)
    m = conv.max()
    n = seq.size
    return 2.0 * n * m / (s * s)


# ------------------------------------------------------------
# Main entry point.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 bound
    is aggressively minimised.  The routine starts from the classic
    Catalan‑type series (or a provided parent) and then performs a
    many‑step stochastic hill‑climbing: mass is repeatedly transferred
    from large entries to small entries while keeping the total sum
    unchanged.  The algorithm evaluates the true bound after each trial
    (FFT‑based) and retains the best candidate found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size – use the provided parent if it exists.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256

    # ------------------------------------------------------------------
    # Initial candidate: the analytic series (optimal M=1) or the parent.
    # ------------------------------------------------------------------
    x = _make_series(n)
    x = np.clip(x, 0.0, 1000.0)

    if parent is not None:
        parent = np.clip(parent, 0.0, 1000.0)
        if _score(parent) < _score(x):
            x = parent.copy()

    # ------------------------------------------------------------------
    # Track the best solution seen so far.
    # ------------------------------------------------------------------
    best = x.copy()
    best_score = _score(best)

    # ------------------------------------------------------------------
    # Core optimisation loop.
    # ------------------------------------------------------------------
    max_iters = 4000                     # budget – linear in n
    patience = 0                         # stagnation counter
    target = 1.50274349                  # goal (stop a little earlier)

    for it in range(max_iters):
        # Current exact bound (reshape after each successful move).
        cur_score = _score(x)
        if cur_score < best_score:
            best = x.copy()
            best_score = cur_score
        if best_score < target:
            break                                            # achieved target

        # ------------------------------------------------------------------
        # Choose a transfer direction.
        #   70 % of the time move from the current maximum entry to the minimum.
        #   30 % of the time use the gradient of the current maximal convolution.
        # ------------------------------------------------------------------
        if rng.random() < 0.7:
            i = int(np.argmax(x))                # source – largest value
            j = int(np.argmin(x))                # target – smallest value
        else:
            # Gradient‑like step: locate the index of the maximal convolution.
            conv = _conv_self(x)
            k = int(conv.argmax())               # position where max occurs
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            sub = x[i_low:i_high + 1]            # slice that participates
            g = sub[::-1]                        # contribution weights
            i = i_low + int(np.argmax(g))        # source = biggest contribution
            j = i_low + int(np.argmin(g))        # target = smallest contribution
        if i == j:
            # fallback – pick a random different target.
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

        # ------------------------------------------------------------------
        # Determine a feasible transfer amount (preserve box constraints).
        # ------------------------------------------------------------------
        max_reduce = x[i]                       # cannot take more than we have
        max_increase = 1000.0 - x[j]            # cannot exceed upper bound
        max_delta = min(max_reduce, max_increase)
        if max_delta <= 0.0:
            patience += 1
            if patience > 600:
                break
            continue

        # Random magnitude – up to 50 % of the feasible amount.
        eps = max_delta * rng.random() * 0.5
        if eps <= 0.0:
            continue

        # ------------------------------------------------------------------
        # Build the candidate and evaluate its bound.
        # ------------------------------------------------------------------
        cand = x.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_score = _score(cand)

        # ------------------------------------------------------------------
        # Acceptance rule: always accept strict improvements; otherwise
        # accept with a tiny probability to escape plateaus.
        # ------------------------------------------------------------------
        if cand_score < cur_score or rng.random() < 0.001:
            x = cand
            cur_score = cand_score
            patience = 0
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        else:
            patience += 1

        # ------------------------------------------------------------------
        # Occasionally perform a random block perturbation (helps exploration).
        # ------------------------------------------------------------------
        if it % 64 == 0 and rng.random() < 0.12:
            width = max(1, n // rng.integers(10, 30))
            start = rng.integers(0, n - width + 1)
            factor = 1.0 + rng.normal(0.0, 0.06)
            block = x.copy()
            block[start:start + width] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            block_score = _score(block)
            if block_score < best_score:
                x = block
                best = block.copy()
                best_score = block_score
                patience = 0

        # ------------------------------------------------------------------
        # Early termination if we have been stagnant for a while.
        # ------------------------------------------------------------------
        if patience > 800:
            break

    # ------------------------------------------------------------------
    # Fallback safety net – guarantee a non‑empty, finite list.
    # ------------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    return [float(v) for v in best]
