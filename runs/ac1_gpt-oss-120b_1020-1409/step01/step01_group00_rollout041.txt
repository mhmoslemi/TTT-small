def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """
    a_i = C(2*i,i) / 4**i  for i = 0..length-1.
    This sequence satisfies (a*a)[k] ≈ 1 for all k < length
    (the discrete analogue of 1/√x).
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _conv_max(seq: np.ndarray) -> float:
    """Maximum of the full convolution seq*seq using an FFT (matrix‑free)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft = np.fft.rfft(seq, n=size)
    prod = fft * fft
    conv = np.fft.irfft(prod, n=size)               # real inverse FFT
    return float(np.max(conv[: 2 * n - 1]))


def propose_candidate(seed: int = 42, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 upper bound.
    The method starts from the classic binomial sequence and repeatedly
    injects small amounts of mass into the tail while keeping the peak of the
    self‑convolution bounded (scale‑invariant).  This greedily raises the
    ℓ¹‑norm without increasing the ‖f*f‖∞ term, thereby lowering the bound.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------
    # Determine problem size.
    # ----------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])          # provided by the harness
        parent_exists = True
    except Exception:                                 # fallback if not present
        n = 256
        parent_exists = False

    # ----------------------------------------------------
    # Initialise with the binomial construction.
    # ----------------------------------------------------
    base = _generate_binomial_sequence(n)
    np.clip(base, 0.0, 1000.0, out=base)

    # If a parent sequence is available and has a larger sum, start from it.
    if parent_exists:
        parent = np.array(height_sequence_1, dtype=np.float64)
        np.clip(parent, 0.0, 1000.0, out=parent)
        if np.sum(parent) > np.sum(base):
            cur = parent.copy()
        else:
            cur = base.copy()
    else:
        cur = base.copy()

    # ----------------------------------------------------
    # Core optimisation loop.
    # ----------------------------------------------------
    M = _conv_max(cur)                # current peak of f*f (≈1 for binomial)
    if M <= 0.0:
        M = 1.0
    best_seq = cur.copy()
    best_score = float(2 * n * M / (np.sum(best_seq) ** 2))

    max_steps = 3000                  # adjustable budget
    tail_start = n // 2               # indices >= this are considered “tail”

    for step in range(max_steps):
        # ----- inject mass into a tail entry ---------------------------------
