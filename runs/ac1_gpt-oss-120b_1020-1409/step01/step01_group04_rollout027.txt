def gradient_step(a, lr, rng, sigma_noise):
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    idx = k_star - np.arange(len(a))
    mask = (idx >= 0) & (idx < len(a))
    g = np.zeros_like(a)
    g[mask] = 2 * a[idx[mask]]
    # step
    a = a - lr * g
    # clip nonneg
    a = np.clip(a, 0.0, None)
    # apply multiplicative noise
    if sigma_noise > 0:
        a *= np.exp(rng.normal(0.0, sigma_noise, size=a.shape))
    # renormalize
    s = a.sum()
    if s < 1e-12:
        a = np.full_like(a, 1.0 / len(a))
        s = 1.0
    a /= s
    return a, conv.max()

def _subgradient_descent(a, rng, max_iter=200, lr=0.5, decay=0.005, noise_sigma=0.01):
    # a must be numpy array, nonneg
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total <= 0:
        a = np.full_like(a, 1.0)
        total = a.sum()
    a = a / total
    best_seq = a.copy()
    best_score = float('inf')
    for t in range(max_iter):
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        idx = k_star - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        step = lr / (1.0 + decay * t)
        a = a - step * g
        a = np.clip(a, 0.0, None)
        # noise
        if noise_sigma > 0:
            a *= np.exp(rng.normal(0.0, noise_sigma, size=a.shape))
        s = a.sum()
        if s <= 0:
            a = best_seq.copy()
            s = a.sum()
        a = a / s
        # occasional evaluation
        if t % 10 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
    # final evaluation
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
    return best_seq, best_score

def _optimize(a0):
    a = a0.copy()
    # subgradient descent
    for t in range(80):
        conv = np.convolve(a, a)
        k_star = np.argmax(conv)
        idx = k_star - np.arange(len(a))
        mask = (idx >=0) & (idx < len(a))
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        step = 0.5 / (1 + t * 0.02)
        a = a - step * g
        a = np.clip(a, 0.0, None)
        # small noise occasionally
        if t % 10 == 0:
            a *= np.exp(rng.normal(0.0, 0.02, size=a.shape))

        s = a.sum()

            a = a0.copy()
            s = a.sum()

    # hill climbing

import numpy as np

def _subgradient_descent(start_seq, rng, max_iter=200,
                         lr=0.7, decay=0.02, noise_sigma=0.01):
    """Projected sub‑gradient descent on  max_k (a*a)[k].

    Returns the best sequence found and its score.
    """
    a = np.clip(start_seq.astype(np.float64), 0.0, None)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0)
        s = a.sum()
    a = a / s

    best_seq = a.copy()
    best_score = float("inf")

    n = a.shape[0]
    idx_array = np.arange(n)

    for t in range(max_iter):
        conv = np.convolve(a, a)                     # full convolution
        k_star = int(np.argmax(conv))                # index of the current max

        # gradient of the max‑convolution (sub‑gradient)
        idx = k_star - idx_array                     # partner indices for each position
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # diminishing step size
        step = lr / (1.0 + decay * t)
        a = a - step * g
        a = np.clip(a, 0.0, None)

        # small multiplicative noise to escape flat regions
        if noise_sigma > 0.0:
            a *= np.exp(rng.normal(0.0, noise_sigma, size=n))

        s = a.sum()
        if s <= 0.0:                     # safeguard against total collapse
            a = best_seq.copy()
            s = a.sum()
        a = a / s

        # occasional evaluation (every 10 iterations)
        if t % 10 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()

    # final evaluation
    final_sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(final_sc) and final_sc < best_score:
        best_score = final_sc
        best_seq = a.copy()

    return best_seq, best_score
