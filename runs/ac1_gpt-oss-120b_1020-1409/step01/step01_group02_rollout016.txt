import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return the supplied parent sequence if it is valid, otherwise a fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    # simple safe fallback
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """Generate the classic decreasing arcsine coefficients."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _block_perturbation(base, rng):
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # occasional symmetry bias
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _score_from_conv(conv, sum_f, n):
    """Exact objective value 2*n*max(conv)/sum_f**2."""
    if sum_f < 0.01:
        return np.inf
    return float(2.0 * n * conv.max() / (sum_f * sum_f))

def propose_candidate(seed=42, num_steps=15000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 upper bound.
    The algorithm combines a strong analytic seed with greedy mass‑transfer
    moves guided by the location of the maximal auto‑correlation and occasional
    block‑wise perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise from the better of the provided parent or the arcsine seed
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = parent.size
    arcsine = _arcsine_sequence(n)

    # evaluate both seeds via the exact formula (no need to call evaluator)
    conv_parent = np.convolve(parent, parent)
    conv_arcsine = np.convolve(arcsine, arcsine)

    sum_parent = parent.sum()
    sum_arcsine = arcsine.sum()

    score_parent = _score_from_conv(conv_parent, sum_parent, n)
    score_arcsine = _score_from_conv(conv_arcsine, sum_arcsine, n)

    if score_arcsine < score_parent:
        best = arcsine.copy()
        best_score = score_arcsine
    else:
        best = parent.copy()
        best_score = score_parent

    # current working point (start from the best seed)
    current = best.copy()
    conv = np.convolve(current, current)
    sum_f = current.sum()          # unchanged by the mass‑transfer moves

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # ----------------------------------------------------------------
        # occasional large block perturbation to escape plateaus
        # ----------------------------------------------------------------

        counterpart = current[k - overlap]

        if counterpart.sum() == 0.0:

        probs = counterpart / counterpart.sum()

    # ------------------------------------------------------------------
    # final safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
