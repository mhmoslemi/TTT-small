import numpy as np


def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        f = float(i)
        a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
    return a


def _pair_random(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small random amount of mass from i to j."""
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.02)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _mass_spread(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Take a fraction of the largest entry and spread it to a few others."""
    n = arr.size
    i = int(np.argmax(arr))
    if arr[i] <= 0.0:
        return arr.copy()
    # fraction of mass to move, keep it modest
    frac = rng.uniform(0.05, 0.20)
    delta = arr[i] * frac
    # number of recipients (2‑5)
    rec_cnt = int(rng.integers(2, 6))
    # choose distinct recipients not equal to i
    idxs = np.arange(n)
    idxs = np.delete(idxs, i)
    if idxs.size == 0:
        return arr.copy()
    recipients = rng.choice(idxs, size=rec_cnt, replace=False)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    share = delta / rec_cnt
    for j in recipients:
        cand[j] = min(1000.0, cand[j] + share)
    return cand


def _smooth(arr: np.ndarray, lam: float = 0.15) -> np.ndarray:
    """One‑dimensional 3‑point smoothing (conv with [0.25,0.5,0.25])."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    sm = np.convolve(arr, kernel, mode='same')
    cand = arr * (1.0 - lam) + sm * lam
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _block_scaling(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = arr.size
    low = max(1, n // 250)
    high = max(2, n // 30)
    width = int(rng.integers(low, high))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.09)))
    cand = arr.copy()
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
