        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent_arr.ndim == 1
            and parent_arr.size > 0
            and np.isfinite(parent_arr).all()
        ):
            n = parent_arr.size
            parent_seq = np.clip(parent_arr, 0.0, 1000.0)
            # normalise to unit sum (scale‑invariant)
            s = parent_seq.sum()
            if s > 0 and np.isfinite(s):
                parent_seq = parent_seq / s
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Baseline candidates: uniform and (if available) the parent.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    val = _score(uniform)
    if val < best_val:
        best_val = val
        best_seq = uniform.copy()

    # parent (if any)
    if parent_seq is not None:
        val = _score(parent_seq)
        if val < best_val:
            best_val = val
            best_seq = parent_seq.copy()

    # ------------------------------------------------------------------
    # Sweep over geometric decays a_i = r^i (r = 1 - eps).
    # eps is chosen in a narrow band that yields a bound near the target.
    # ------------------------------------------------------------------
    eps_candidates = np.linspace(0.0001, 0.0015, 20)  # ~20 points
    for eps in eps_candidates:
        r = 1.0 - eps
        # log‑space generation avoids under‑/overflow
        log_r = np.log(r)
        a = np.exp(np.arange(n, dtype=np.float64) * log_r)
        s = a.sum()
        if s == 0 or not np.isfinite(s):
            continue
        a /= s
        # values already lie in [0,1000] because sum=1 and a_i ≤ 1
        val = _score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # Tiny local random search around the best geometric candidate.
    # This step is cheap (few hundred evaluations) and often yields a
    # noticeable improvement.
    # ------------------------------------------------------------------
    if best_seq is None:               # very unlikely, but keep safety net
        best_seq = uniform.copy()
        best_val = _score(best_seq)

    a = best_seq.copy()
    for _ in range(200):
        # pick a random contiguous block and scale it
        width = int(rng.integers(1, max(2, n // 25)))
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand = a.copy()
        cand[start : start + width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        total = cand.sum()
        if total == 0 or not np.isfinite(total):
            continue
        cand /= total          # keep L1‑norm = 1 (scale‑invariant)
        val = _score(cand)
        if val < best_val:
            best_val = val
            a = cand

    # ------------------------------------------------------------------
    # Return the best sequence found – plain Python floats.
    # ------------------------------------------------------------------
    return [float(x) for x in a]
