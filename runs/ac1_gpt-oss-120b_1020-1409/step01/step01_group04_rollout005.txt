import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ------------------------------------------------------------
def _eval(seq: np.ndarray) -> float:
    """
    Evaluate a candidate sequence using the provided evaluator.
    The evaluator expects a plain Python list of floats.
    """
    return float(evaluate_sequence(seq.tolist()))


def _alternating_arrangement(seq: np.ndarray) -> np.ndarray:
    """
    Rearrange ``seq`` into a high‑low alternating pattern.
    Large values go to even indices, the remaining (smaller) values
    are placed increasingly on odd indices.
    """
    n = seq.size
    # values sorted descending
    desc = np.sort(seq)[::-1]
    out = np.empty_like(seq)
    evens = np.arange(0, n, 2)
    odds = np.arange(1, n, 2)

    # most large numbers on even positions
    out[evens] = desc[: evens.size]

    # the rest (smaller numbers) on odd positions, ordered increasingly
    out[odds] = desc[evens.size :][::-1]
    return out

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_alpha: int = 30,
                     num_steps: int = 1500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts
    to minimise the AC1 bound evaluated by ``evaluate_sequence``.
    The routine uses a multi‑stage stochastic search:
      1. Scan a simple power‑law family and a high‑low alternating
         rearrangement of each member.
      2. Perform a randomized local search consisting of
         block scalings, pairwise mass transfers, swaps, and
         targeted swaps that aim to reduce the current maximal
         convolution coefficient.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and a sensible fallback initial vector.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # initialise best candidate (clipped to allowed range)
    best = np.clip(parent, 0.0, 1000.0)
    best_score = _eval(best)

    # ------------------------------------------------------------------
    # 1) Scan power‑law family and alternating variants.
    # ------------------------------------------------------------------
    alphas = np.logspace(-5, 1, num=num_alpha, dtype=np.float64)
    for alpha in alphas:
        seq = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + alpha)
        seq = np.clip(seq, 0.0, 1000.0)

        s = _eval(seq)
        if np.isfinite(s) and s < best_score:
            best, best_score = seq, s

        # high‑low alternating version of the same values
        alt = _alternating_arrangement(seq)
        alt = np.clip(alt, 0.0, 1000.0)
        s_alt = _eval(alt)
        if np.isfinite(s_alt) and s_alt < best_score:
            best, best_score = alt, s_alt

    # ------------------------------------------------------------------
    # 2) Stochastic local optimisation.
    # ------------------------------------------------------------------
    # Parameters controlling the perturbations
    min_width = max(1, n // 200)          # ~0.5 % of length
    max_width = max(2, n // 20)           # ~5 %
    sigma_block = 0.07                    # log‑normal block factor
    patience = 0
    max_patience = 300                    # early stop if stalled

    for step in range(num_steps):
        cand = best.copy()
        op = rng.integers(0, 4)  # choose perturbation type

        if op == 0:
            # ---- multiplicative block perturbation ----
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma_block)))
            cand[start:start + width] *= factor

        elif op == 1:
            # ---- pairwise mass transfer ----
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            delta = best[i] * rng.random() * 0.2
            cand[i] -= delta
            cand[j] += delta

        elif op == 2:
            # ---- random swap of two entries ----
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            cand[i], cand[j] = cand[j], cand[i]

        else:
            # ---- targeted swap aiming at the current maximal convolution ----
            # Compute convolution of the current best (used only for guidance)
            conv = np.convolve(best, best)
            kmax = int(np.argmax(conv))
            i_min = max(0, kmax - (n - 1))
            i_max = min(kmax, n - 1)
            i = int(rng.integers(i_min, i_max + 1))
            j = int(rng.integers(0, n - 1))
            if j == i:

            cand[i], cand[j] = cand[j], cand[i]

        # enforce bounds
        cand = np.clip(cand, 0.0, 1000.0)

        best = np.full(n, 1.0, dtype=np.float64)

    return [float(x) for x in best]
