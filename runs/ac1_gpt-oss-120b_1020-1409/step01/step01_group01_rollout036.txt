import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Generate the r‑sequence a_i = C(2i,i)/4^i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function that lowers the AC1 bound."""
    # ------------------------------------------------------------------
    # Determine problem size.
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256  # fallback size

    # ------------------------------------------------------------------
    # Initialisation – start from the classic r‑sequence.
    # ------------------------------------------------------------------
    best = generate_r_sequence(n)
    np.clip(best, 0.0, 1000.0, out=best)

    # Evaluate the starting point.
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):

        best_score = float(evaluate_sequence(best.tolist()))

    # distribute the removed mass uniformly over the remaining indices.
    # This reduces the maximal convolution while keeping the total sum
    # (hence the denominator) essentially unchanged.

            removed += (old - new)
            cand[i] = new

        # Distribute the removed mass uniformly over the tail.
        if removed > 0.0:
            add = removed / tail_cnt
            cand[early_cnt:] += add
            cand[:early_cnt] -= add   # cancel the addition on the early part

        # Enforce bounds.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Evaluate.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # keep the same factor and try another reduction
            continue
        else:
            # No improvement – try a milder reduction or stop.
            # When the factor is already close to 1 we stop.
            if factor > 0.9995:
                break
            # Make the next step less aggressive.
            factor = 1.0 - (1.0 - factor) * 0.5

    # ------------------------------------------------------------------
    # Small random fine‑tuning – occasional tiny perturbations may
    # escape tiny plateaus.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    tiny_eps = 1e-3
    for _ in range(30):
        i = rng.integers(0, n)
        # propose a tiny positive or negative change
        delta = tiny_eps * (rng.random() - 0.5) * 2.0
        cand = best.copy()
        cand[i] = np.clip(cand[i] + delta, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Final safety fallback (should never trigger).
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
