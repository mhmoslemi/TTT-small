def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        n = 256
    target_sum = np.sqrt(2.0 * n)

    base = np.arange(n, 0, -1, dtype=np.float64)  # n, n-1, ..., 1
    # Optionally we can add a small epsilon to avoid zeros, but not needed.
    # Scale to target sum
    s = base.sum()
    a = base * (target_sum / s)
    # Clip to [0, 1000] (no need)
    np.clip(a, 0.0, 1000.0, out=a)
    # Ensure non-negative (already)
    # Possibly improve with slight local refinement:
    # apply a random perturb small enough that does not increase max.
    # For safety, compute score and compare with uniform fallback.
    def _conv_max(arr):
        n_local = arr.shape[0]
        size = 1 << ((2 * n_local - 1).bit_length())
        fa = np.fft.rfft(arr, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n_local - 1]
        return float(conv.max())
    def _score_arr(arr):
        s_local = arr.sum()
        maxb = _conv_max(arr)
        return float(2.0 * n * maxb / (s_local * s_local))

    best_vec = a
    best_score = _score_arr(best_vec)

    # Uniform fallback
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    uniform_score = _score_arr(uniform)
    if uniform_score < best_score:
        best_vec = uniform
        best_score = uniform_score

    # Optionally refine by small random perturb (few iterations):
    for _ in range(200):
        cand = best_vec.copy()
        # Perturb random block
        width = rng.integers(1, max(2, n // 100))
        start = rng.integers(0, max(1, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.02))
        cand[start:start+width] *= factor
        np.clip(cand, 0.0, 1000.0, out=cand)
        # Renormalize to sum target
        s = cand.sum()
        if s > 0.0:
            cand *= target_sum / s
        sc = _score_arr(cand)
        if sc < best_score:
            best_vec = cand
            best_score = sc

    return [float(x) for x in best_vec]

import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (matrix‑free, O(n log n))
def _conv_max(arr: np.ndarray) -> float:
    """Maximum of the full linear convolution arr * arr via FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact evaluator without the safety wrapper."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape factories
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p).  If sym=True the sequence is added to its reverse."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


def _make_exponential(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-rate * idx)


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i/σ)^2)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(- (idx / sigma) ** 2)


# ----------------------------------------------------------------------
def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Small stochastic local change while keeping the L1‑norm fixed."""
    cand = vec.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 250), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.04)))
    cand[start:start + width] *= factor
    if rng.random() < 0.25:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand


# ----------------------------------------------------------------------

          families,
        * a modest stochastic refinement around the current best,
        * a uniform fallback.
    """
