    # Try to load parent

    best = a.copy()

    cur_score = best_score

    max_block = max(2, int(block_frac * n))

    for it in range(num_iters):
        if rng.random() < move_prob:
            # block scaling
            b_len = rng.integers(1, max_block+1)
            start = rng.integers(0, n - b_len + 1)

import numpy as np

# ------------------------------------------------------------
# Helper functions – all top‑level (no closures)
# ------------------------------------------------------------
def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """Single exponential‑gradient update targeting the current max convolution entry."""
    n = a.shape[0]
    conv = np.convolve(a, a)                     # O(n log n) via NumPy
    k = int(np.argmax(conv))                     # index of maximal entry
    low = max(0, k - (n - 1))
    high = min(k, n - 1)
    idx = np.arange(low, high + 1, dtype=np.int64)
    partner = k - idx
    # gradient of conv[k] w.r.t. a (g_j = 2*a_{k-j})
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[partner]

    # exponential move, stay non‑negative
    a = a * np.exp(-step * g)
    np.clip(a, 0.0, 1000.0, out=a)

    s = a.sum()
    if s == 0 or not np.isfinite(s):
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / s
    return a


def _random_modify(a: np.ndarray,
                   rng: np.random.Generator,
                   max_block: int,
                   sigma: float,
                   max_move_frac: float) -> np.ndarray:
    """Return a copy of *a* after a small stochastic perturbation."""
    n = a.shape[0]
    if rng.random() < 0.70:          # block‑wise multiplicative scaling
        blen = rng.integers(1, max_block + 1)
        start = rng.integers(0, n - blen + 1)
        factor = np.exp(rng.normal(0.0, sigma))
        b = a.copy()
        b[start:start + blen] *= factor
    else:                            # transfer a fraction of mass from i to j
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = rng.random() * max_move_frac * a[i]
        b = a.copy()
        b[i] -= delta
        b[j] += delta

    np.clip(b, 0.0, 1000.0, out=b)
    s = b.sum()
    if s == 0 or not np.isfinite(s):
        b = np.ones(n, dtype=np.float64) / n
    else:
        b = b / s
    return b

def propose_candidate(seed=42,
                     num_iters=6000,
                     block_frac=0.02,
                     sigma=0.15,
                     step0=0.30,
                     step_decay=0.996,
                     max_move_frac=0.30) -> list:
    """
    Search for a non‑negative step sequence that minimises the AC1 upper bound.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and optional parent sequence (height_sequence_1)
    # ----------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = (
            parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # ----------------------------------------------------------------------
    # Initial candidate (normalized to sum == 1)
    # ----------------------------------------------------------------------
    if parent_ok:
        a0 = np.clip(parent_arr, 0.0, 1000.0)
        s = a0.sum()
        if s > 0.0 and np.isfinite(s):
            a0 = a0 / s
        else:
            a0 = np.ones(n, dtype=np.float64) / n
    else:
        # power‑law start: a_i ∝ (i+1)^{-0.7}
        expo = 0.70
        a0 = np.arange(1, n + 1, dtype=np.float64) ** -expo
        a0 = a0 / a0.sum()

    # ----------------------------------------------------------------------
    # Best‑so‑far bookkeeping
    # ----------------------------------------------------------------------
    best = a0.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = a0.copy()
    cur_score = best_score

    max_block = max(2, int(block_frac * n))

    # ----------------------------------------------------------------------
    # Main stochastic optimisation loop
    # ----------------------------------------------------------------------
    for it in range(num_iters):
        # 1) Random perturbation of the current candidate
        cand = _random_modify(cur, rng, max_block, sigma, max_move_frac)

        # 2) One gradient polishing step aimed at the current max entry
        step = step0 * (step_decay ** it)
        cand = _gradient_step(cand, step)

        # 3) Evaluate
        new_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(new_score):
            continue

        # 4) Accept only improvements (simple hill climbing)
        if new_score < cur_score:
            cur = cand
            cur_score = new_score
            if new_score < best_score:
                best = cand.copy()
                best_score = new_score

        # 5) Every few thousand iterations restart from the best found so far
        if (it + 1) % 2000 == 0:
            cur = best.copy()
            cur_score = best_score

    # ----------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats)
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
