import numpy as np

def _conv_full(arr):
    n = arr.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _max_and_idx(arr):
    conv = _conv_full(arr)
    idx = int(conv.argmax())
    return float(conv.max()), idx, conv

def _gradient_at_max(arr, idx):
    n = arr.shape[0]
    g = np.zeros_like(arr)
    if idx < n:
        L = idx + 1
        g[:L] = arr[:L][::-1]
    else:
        L = 2*n - 1 - idx
        lo = n - L
        g[lo:] = arr[-L:][::-1]
    return g

def _score(arr, n):
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_full(arr).max()
    return float(2.0 * n * max_b / (s * s))

def propose_candidate(seed=42, **kwargs):

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():

        n = parent.size
    except Exception:

    target_sum = np.sqrt(2.0 * n)

    # baseline uniform
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    best_vec = uniform.copy()
    best_score = _score(best_vec, n)

    # grid search with power law (maybe narrower for speed)
    p_vals = np.linspace(0.2, 0.8, 13)  # more options
    beta_vals = np.logspace(-2, 2, 13)
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                idx = np.arange(n, dtype=np.float64)
                base = (idx + beta) ** (-p)
                if sym:
                    base = base + (idx[::-1] + beta) ** (-p)
                s = base.sum()
                if s == 0.0:
                    continue
                a = base * (target_sum / s)
                np.clip(a, 0.0, 1000.0, out=a)

                if sc < best_score:
                    best_score = sc
                    best_vec = a.copy()

    # Additional random perturbations
    for _ in range(800):
        cand = _block_perturb(best_vec, rng, target_sum)
        sc = _score(cand, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    for it in range(2000):

def _conv_full(arr):
    """Full linear convolution arr*arr using FFT (exact for real data)."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv
