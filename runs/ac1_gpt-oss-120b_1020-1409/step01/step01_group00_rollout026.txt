import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..length-1 (central binomial coefficients)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(seq: np.ndarray, m: int) -> np.ndarray:
    """Convolution of seq with itself via real FFT, length trimmed to 2*len-1."""
    fa = np.fft.rfft(seq, m)
    prod = fa * fa
    conv = np.fft.irfft(prod, m)[: 2 * seq.shape[0] - 1]
    return conv


def _score_from_seq(seq: np.ndarray, m: int) -> float:
    """Exact score used by the evaluator (fast via FFT)."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = _conv_fft(seq, m)
    max_b = conv.max()
    n = seq.shape[0]
    return float(2.0 * n * max_b / (s * s))
