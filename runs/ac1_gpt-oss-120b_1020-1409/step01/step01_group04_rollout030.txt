import numpy as np

def _normalize(seq, target_sum):
    s = seq.sum()
    if s <= 0: return seq
    return seq * (target_sum / s)

def _max_self_conv(seq, conv_len, n):
    # compute conv via FFT
    fft_seq = np.fft.rfft(seq, n=conv_len)
    conv = np.fft.irfft(fft_seq * fft_seq, n=conv_len)
    # max over first 2*n-1 entries
    return conv[:2*n-1].max()

def _score(seq, conv_len, n, target_sum):
    # seq assumed normalized to target_sum maybe
    max_b = _max_self_conv(seq, conv_len, n)
    sum_a = seq.sum()
    if sum_a <= 0: return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    rng = np.random.default_rng(seed)

    # Determine n
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:

        n = parent_arr.size
    except Exception:
        n = 256
        parent_arr = np.ones(n, dtype=np.float64)

    # Helper: conv length for FFT
    conv_len = 1 << ((2*n-1).bit_length())

    # Normalization target sum: we can use sum of parent or 1.0
    target_sum = max(parent_arr.sum(), 1.0)  # ensure >0
    # Normalize parent
    parent_arr = _normalize(parent_arr, target_sum)

    # current best
    best_seq = parent_arr.copy()
    best_score = _score(best_seq, conv_len, n, target_sum)

    # Helper: evaluate candidate and maybe update
    def try_update(cand):
        nonlocal best_seq, best_score
        cand = np.clip(cand, 0.0, 1000.0)

        cand = _normalize(cand, target_sum)
        sc = _score(cand, conv_len, n, target_sum)
        if np.isfinite(sc) and sc < best_score:
            best_seq = cand
            best_score = sc

    # 1. Scan param families

    # Power law a_i = (i+alpha)^(-p)
    alphas = np.logspace(-4, 1, 12) # maybe 12 values

        for p in ps:
            cand = 1.0 / np.power(idx + alpha, p)
            try_update(cand)

    # Exponential: a_i = exp(-beta * i)
    betas = np.logspace(-4, -0.3, 12)
    for beta in betas:
        cand = np.exp(-beta * idx)
        try_update(cand)

    cand = np.sin(np.pi * (idx + 0.5) / n)

    try_update(cand)

    half = n // 2

    try_update(cand)

        sigma = sigma_start * (sigma_end / sigma_start) ** t
        cand = best_seq.copy()
        if rng.random() < 0.6:  # block scaling
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, sigma))
            cand[start:start+width] *= factor
        else:
            # mass transfer between two positions
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            delta = cand[i] * rng.random() * sigma

            cand[j] = cand[j] + delta
        # optionally symmetric enforce
        if rng.random() < 0.05:

        # clip and normalize

def _normalize(seq, target_sum):
    s = seq.sum()
    if s == 0:
        return seq
    return seq * (target_sum / s)

    # Returns full self-convolution using FFT
    fft_seq = np.fft.rfft(seq, n=conv_len)
    conv = np.fft.irfft(fft_seq * fft_seq, n=conv_len)
    return conv[:2 * n - 1]

def _max_self_conv(seq, conv_len, n):
    conv = _full_conv(seq, conv_len, n)
    return conv.max()

import numpy as np


def _normalize(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale seq so that its sum equals target_sum (unless sum is zero)."""
    s = seq.sum()
    if s == 0.0:
        return seq
    return seq * (target_sum / s)


def _full_conv(seq: np.ndarray, conv_len: int, n: int) -> np.ndarray:
    """Full self‑convolution of seq using FFT (real‑valued)."""
    fft_seq = np.fft.rfft(seq, n=conv_len)
    conv = np.fft.irfft(fft_seq * fft_seq, n=conv_len)
    return conv[: 2 * n - 1]


def _max_self_conv(seq: np.ndarray, conv_len: int, n: int) -> float:
    return _full_conv(seq, conv_len, n).max()


def _score(seq: np.ndarray, conv_len: int, n: int) -> float:
    """Exact AC1 upper‑bound score for a non‑negative step sequence."""
    sum_a = seq.sum()
    if sum_a <= 0.0:
        return np.inf
    max_b = _max_self_conv(seq, conv_len, n)
    return 2.0 * n * max_b / (sum_a * sum_a)
