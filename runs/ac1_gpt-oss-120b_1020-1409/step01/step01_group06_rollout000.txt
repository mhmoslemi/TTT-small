def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    try: 
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size==0 or not np.isfinite(arr).all():
            raise Exception
        n = arr.size
    except Exception:
        n = 256
        arr = np.full(n, 1.0)

    target_sum = np.sqrt(2.0 * n)

    # helper functions
    def normalize(v):
        v = np.clip(v, 0.0, 1000.0, out=v)
        s = v.sum()
        if s > 0:
            v *= target_sum / s
        else:
            v[:] = target_sum / v.size
        return v

    def score(v):
        # compute max conv using FFT
        size = 1 << ((2 * n - 1).bit_length())
        fv = np.fft.rfft(v, n=size)
        conv = np.fft.irfft(fv * fv, n=size)[:2*n-1]
        return float(2.0 * n * conv.max() / (v.sum()**2))

    # initial candidates
    best = normalize(arr.copy())
    best_score = score(best)

    # uniform
    uniform = np.full(n, target_sum / n)
    uni_score = score(uniform)
    if uni_score < best_score:
        best = uniform.copy()
        best_score = uni_score

    # power-law grid search extended
    p_vals = np.linspace(0.1, 1.0, 19)   # 0.1 to 1.0 step 0.05
    beta_vals = np.logspace(-3, 3, 15)
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                idx = np.arange(n, dtype=np.float64)
                base = (idx + beta) ** (-p)
                if sym:
                    base = base + (idx[::-1] + beta) ** (-p)
                cand = normalize(base)
                sc = score(cand)
                if sc < best_score:
                    best_score = sc
                    best = cand.copy()

    # exponential grid search
    rates = np.linspace(0.90, 0.999, 12)  # decay factor
    for sym in (False, True):
        for r in rates:
            idx = np.arange(n, dtype=np.float64)
            base = r ** idx
            if sym:
                base = base + (r ** (n-1-idx))
            cand = normalize(base)
            sc = score(cand)
            if sc < best_score:
                best_score = sc
                best = cand.copy()

    # local random search
    total_iter = 3000
    for it in range(total_iter):
        cand = best.copy()
        # choose perturb type
        if rng.random() < 0.5:
            # block scaling
            width = int(rng.integers(max(1, n//200), max(2, n//12)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.06 * (1.0 - it/total_iter))))  # decrease temperature
            cand[start:start+width] *= factor
        else:
            # Gaussian noise
            sigma = 0.05 * (1.0 - it/total_iter)
            noise = rng.normal(0.0, sigma, size=n)
            cand += noise
        # optional mirror
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])
        # renorm
        cand = normalize(cand)
        sc = score(cand)
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    # Return list
    return [float(x) for x in best]

import numpy as np

# ----------------------------------------------------------------------
def _normalize(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and rescale to have L1 norm = target_sum."""
    np.clip(vec, 0.0, 1000.0, out=vec)
    s = vec.sum()
    if s > 0.0:
        vec *= target_sum / s
    return vec


def _score(vec: np.ndarray, n: int, conv_size: int) -> float:
    """Exact AC1 upper bound using FFT‑based convolution."""
    fv = np.fft.rfft(vec, n=conv_size)
    conv = np.fft.irfft(fv * fv, n=conv_size)[: 2 * n - 1]
    max_b = conv.max()
    s = vec.sum()
    return float(2.0 * n * max_b / (s * s))


def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^{-p}, optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


def _make_exponential(n: int, rate: float, sym: bool) -> np.ndarray:
    """a_i ∝ rate^{i}, optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = rate ** idx
    if sym:
        base = base + (rate ** (n - 1 - idx))
    return base


def _perturb_block(vec: np.ndarray, rng: np.random.Generator,
                   target_sum: float) -> np.ndarray:
    """Scale a random contiguous block."""
    cand = vec.copy()
    n = cand.size
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    cand[start:start + width] *= factor
    if rng.random() < 0.2:                     # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return _normalize(cand, target_sum)


def _perturb_noise(vec: np.ndarray, rng: np.random.Generator,
                   sigma: float, target_sum: float) -> np.ndarray:
    """Add small Gaussian noise to all entries."""
    cand = vec + rng.normal(0.0, sigma, size=vec.shape)
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    return _normalize(cand, target_sum)
