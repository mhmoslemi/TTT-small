import numpy as np

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full convolution arr * arr using an FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _score(arr: np.ndarray, n: int) -> float:
    """Exact evaluator formula (avoids the O(n²) np.convolve)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p); optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


# ----------------------------------------------------------------------
def _block_perturb(vec: np.ndarray, rng: np.random.Generator,
                   target_sum: float) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = vec.copy()
    n = cand.size
    max_width = max(2, n // 10)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.08)))
    cand[start:start + width] *= factor
    if rng.random() < 0.2:                      # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand

                base = _make_powerlaw(n, p, beta, sym)
                s = base.sum()
                if s == 0.0:
                    continue
                a = base * (target_sum / s)
                np.clip(a, 0.0, 1000.0, out=a)
                sc = _score(a, n)
                prototypes.append((sc, a))

    prototypes.sort(key=lambda x: x[0])     # best first
    top_k = 5                               # keep a few promising seeds
    for sc, vec in prototypes[:top_k]:
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()

    # ------------------------------------------------------------------
    # Random permutations of the best few seeds
    permutations_per_seed = 200
    for _, seed_vec in prototypes[:top_k]:
        for _ in range(permutations_per_seed):
            cand = rng.permutation(seed_vec)               # same histogram, shuffled
            # renormalise (tiny numerical drift may appear)
            s = cand.sum()
            if s != 0.0:
                cand = cand * (target_sum / s)
            sc = _score(cand, n)
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()

    # ------------------------------------------------------------------
    # Fine‑grained stochastic refinement (few hundred local moves)
    for _ in range(500):
        cand = _block_perturb(best_vec, rng, target_sum)
        sc = _score(cand, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # Return a plain Python list (the harness will apply its final checks)
    return [float(x) for x in best_vec]
