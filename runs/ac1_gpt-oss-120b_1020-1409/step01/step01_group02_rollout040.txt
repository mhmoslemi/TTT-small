import numpy as np

# ----------------------------------------------------------------------
# Helper sequences
# ----------------------------------------------------------------------
def _arcsine_sequence(n: int) -> np.ndarray:
    """Classic arcsine construction a_i = a_{i-1}*(1-0.5/i)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _lambda_sequence(n: int, lam: float) -> np.ndarray:
    """Generalised recurrence a_i = a_{i-1}*(1-lam/i).  Positive for 0<lam<1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - lam / i)
    return a

    """Return the parent construction if present, else None."""

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT, length 2*len(a)-1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact bound value 2*n*max(conv)/(sum(a)^2).  Mirrors evaluate_sequence."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _conv_full(a)
    return 2.0 * n * conv.max() / (s * s)

    """
