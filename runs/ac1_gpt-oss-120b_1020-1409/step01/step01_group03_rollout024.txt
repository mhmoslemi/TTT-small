def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim !=1 or parent.size ==0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        n = 256
        parent = None

    def _make_series(...): same as earlier.

    if parent is not None:
        x = np.clip(parent, 0.0, 1000.0)
    else:
        x = _make_series(n)

    best = x.copy()
    best_score = _score(best)

    # optional: symmetrize initial
    sym = 0.5*(x + x[::-1])
    sym_score = _score(sym)
    if sym_score < best_score:
        best = sym.copy(); best_score = sym_score; x = sym.copy()

    momentum = np.zeros_like(x)
    beta = 0.9

    stagnant = 0
    for step in range(num_steps):
        # gradient
        conv = _conv_self(x)
        M = np.max(conv)
        k = int(conv.argmax())
        S = np.sum(x)
        if S == 0.0:
            break
        # compute shifted a_{k-i}
        i_low = max(0, k-(n-1))
        i_high = min(n-1, k)
        idx = np.arange(i_low, i_high+1, dtype=int)
        a_shift = np.zeros_like(x)
        a_shift[idx] = x[k - idx]

        target = M / S
        grad = a_shift - target  # direction of increase in f; we need to go opposite
        # momentum update
        lr = 0.5  # base learning rate; could adapt
        momentum = beta * momentum - lr * grad
        cand = x + momentum
        np.clip(cand, 0.0, 1000.0, out=cand)

        # occasional symmetrization
        if rng.random() < 0.05:
            cand = 0.5 * (cand + cand[::-1])

        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            stagnant = 0
        else:
            stagnant += 1

        x = cand

        # random block perturbation
        if rng.random() < 0.2:
            width = rng.integers(1, max(2, n // rng.integers(10, 30)))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            block = x.copy()
            block[start:start+width] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            sc_block = _score(block)
            if sc_block < best_score:
                best = block.copy()
                best_score = sc_block
                x = block.copy()
                stagnant = 0

        # occasional smoothing
        if rng.random() < 0.05:
            smooth = (x + np.concatenate(([0], x[:-1])) + np.concatenate((x[1:], [0]))) / 3.0
            sc_smooth = _score(smooth)
            if sc_smooth < best_score:
                best = smooth.copy()
                best_score = sc_smooth
                x = smooth.copy()
                stagnant = 0

        # restart after stagnation
        if stagnant > 300:
            # combine best with random perturbation
            randvec = rng.random(n)  # uniform in [0,1)
            randvec = np.clip(randvec, 0.0, 1000.0)
            combo = best * 0.6 + randvec * 0.4
            np.clip(combo, 0.0, 1000.0, out=combo)
            x = combo
            stagnant = 0

        # early stop after many steps
        if stagnant > 800:

    # final fallback if best invalid
    if best.size ==0 or not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
    return [float(v) for v in best]

import numpy as np

# ------------------------------------------------------------
# Fast convolution (real FFT) for self‑convolution.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 upper bound (mirrors the provided evaluator).
# ------------------------------------------------------------
def _score(x: np.ndarray) -> float:
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(x)
    m = c.max()
    return 2.0 * x.size * m / (s * s)


# ------------------------------------------------------------
# Classical binomial series a_i = C(2i,i)/4^i.
# ------------------------------------------------------------
def _make_series(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# One projected gradient step (sub‑gradient of the max‑convolution).
# ------------------------------------------------------------
def _grad_step(x: np.ndarray) -> tuple[np.ndarray, float]:
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    # indices attaining (or very close to) the maximum
    idxs = np.where(conv >= M - 1e-12)[0]
    S = x.sum()
    if S == 0.0:
        return x, float("inf")
    # average of the shifted vectors a_{k-i}
    shift = np.zeros_like(x)
    for k in idxs:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=int)
        shift[i_range] += x[k - i_range]
    shift /= max(1, idxs.size)                 # avoid division by zero
    target = M / S
    # sub‑gradient of f = 2n*M/S^2 (direction of increase)
    grad = shift - target
    # line‑search on a few learning rates
    best = x
    best_sc = _score(x)
    lr = 1.0
    for _ in range(5):
        cand = x - lr * grad                     # move opposite to increase
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            break
        lr *= 0.5
    return best, best_sc
