def _gradient_softmax(x, tau):
    n = x.shape[0]
    conv = _conv_self(x)    # length 2n-1
    # compute softmax weights
    maxc = conv.max()
    # compute exp(tau*(conv - maxc))
    exp_term = np.exp(tau * (conv - maxc))
    sum_exp = exp_term.sum()
    p = exp_term / sum_exp   # shape (2n-1,)
    M_tau = maxc + np.log(sum_exp) / tau
    S = x.sum()
    # compute convolution p * x_rev
    x_rev = x[::-1]
    conv_px = _conv_general(p, x_rev)   # length 2n-1 + n -1 = 3n-2
    # gradient_i = (4*n / S**2) * conv_px[n-1 + i] - (4*n * M_tau / S**3)
    factor1 = (4.0 * n) / (S * S)
    factor2 = (4.0 * n * M_tau) / (S * S * S)
    # compute array of factor2 constant subtract
    grad = factor1 * conv_px[n-1:n-1+n] - factor2
    return grad, M_tau, maxc

def _conv_general(a, b):
    la = a.shape[0]
    lb = b.shape[0]
    conv_len = la + lb - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)[:conv_len]
    return conv
```

import numpy as np

# ------------------------------------------------------------
# Helper: Analytic series a_i = C(2i,i)/4^i  (optimal for M=1)
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the length‑`length` truncation of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# FFT‑based full convolution (real input)
# ------------------------------------------------------------
def _conv_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full convolution of 1‑D real arrays a and b."""
    la = a.shape[0]
    lb = b.shape[0]
    conv_len = la + lb - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)[:conv_len]
    return conv


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution of x (length 2*len(x)-1)."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact scoring (mirrors evaluate_sequence)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """2*n*max(conv(seq,seq))/(sum(seq)^2) with evaluator clipping."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    conv = _conv_self(seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ------------------------------------------------------------
# Gradient of the soft‑max approximation of the max‑convolution.
# ------------------------------------------------------------
def _gradient_softmax(x: np.ndarray, tau: float):
    """
    Return gradient of f(x)=2*n*M/||x||_1^2 where M is a smooth
    approximation of max(conv(x,x)) using temperature tau.
    Also returns the smooth max value and the hard max (for scoring).
    """
    n = x.shape[0]
    # full self‑convolution and its hard maximum
    conv = _conv_self(x)
    max_c = conv.max()

    # soft‑max weights (stabilised)
    diff = conv - max_c                     # ≤ 0
    exp_term = np.exp(tau * diff)          # max term = 1
    sum_exp = exp_term.sum()
    p = exp_term / sum_exp                  # probability vector, length 2n‑1

    # smooth max value
    smooth_max = max_c + np.log(sum_exp) / tau

    # sum of coefficients
    S = x.sum()
    if S == 0.0:
        return np.zeros_like(x), smooth_max, max_c

    # convolution of the weight vector with reversed x
    x_rev = x[::-1]
    conv_px = _conv_general(p, x_rev)       # length (2n‑1)+n‑1 = 3n‑2
    start = n - 1
    weighted = conv_px[start:start + n]     # ∑_k p_k * x_{k-i}

    factor1 = (4.0 * n) / (S * S)
    factor2 = (4.0 * n * smooth_max) / (S * S * S)

    grad = factor1 * weighted - factor2
    return grad, smooth_max, max_c
