def ac1_score(seq):
    n = seq.shape[0]
    sum_s = seq.sum()
    if sum_s < 0.01:
        return np.inf
    # Compute conv via FFT
    N = 1 << ((2*n - 1).bit_length())
    # compute FFT
    fft_seq = np.fft.rfft(seq, N)
    conv = np.fft.irfft(fft_seq * fft_seq, N)[:2*n-1]
    max_conv = conv.max()
    return 2.0 * n * max_conv / (sum_s * sum_s)

def gradient_move(seq, conv, kmax):
    lo = max(0, kmax - (n - 1))
    hi = min(n - 1, kmax)
    idx_range = np.arange(lo, hi + 1)
    # contributions
    contrib = seq[idx_range] * seq[kmax - idx_range]
    total = contrib.sum()
    if total <= 0 or rng.random() > 0.9:
        i = rng.integers(lo, hi+1)
    else:
        probs = contrib / total
        i = int(rng.choice(idx_range, p=probs))
    # Ensure we have enough mass to transfer
    max_transfer = seq[i] * 0.4  # up to 40% to avoid zero
    if max_transfer <= 0:
        return None, None
    delta = rng.random() * max_transfer
    # Choose j not equal i
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    # apply move
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand, delta

def block_move(seq):

    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    a = idx ** (-alpha)
    return a

def _compute_score(seq, n, N_fft):
    sum_s = seq.sum()
    if sum_s < 0.01:
        return np.inf, None, None, None
    fft_seq = np.fft.rfft(seq, N_fft)
    conv = np.fft.irfft(fft_seq * fft_seq, N_fft)[:2*n-1]
    max_idx = int(conv.argmax())
    max_val = conv[max_idx]
    score = 2.0 * n * max_val / (sum_s * sum_s)
    return float(score), conv, max_idx, max_val

def propose_candidate(seed=42, num_steps=6000, **kwargs):
    rng = np.random.default_rng(seed)

    parent = _parent_sequence()
    n = parent.size
    # ensure length reasonable >0
    # compute FFT size power of two at least 2*n-1
    N_fft = 1 << ((2*n - 1).bit_length())

    # generate initial candidates
    candidates = [parent]
    candidates.append(_arcsine_sequence(n))
    candidates.append(_powerlaw_sequence(n, alpha=0.5))
    candidates.append(np.full(n, 1.0, dtype=np.float64))

    best_seq = None
    best_score = np.inf
    # evaluate initial seeds using our own scoring (consistent with evaluator)
    for cand in candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        score, _, _, _ = _compute_score(cand, n, N_fft)
        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score

    if best_seq is None:
        # fallback
        best_seq = np.ones(256, dtype=np.float64)
        n = best_seq.size
        N_fft = 1 << ((2*n - 1).bit_length())

    # maintain a current working sequence
    current = best_seq.copy()
    current_score, current_conv, current_max_idx, _ = _compute_score(current, n, N_fft)
    # target bound for early exit
    target_bound = 1.50274349

    sum_target = best_seq.sum()
    # main stochastic optimisation loop
    for step in range(num_steps):
        # early stop if we already beat target
        if best_score <= target_bound:
            break

        temperature = 1.0 / (step + 1.0)

        # Decide move type
        if rng.random() < 0.75:
            # gradient‑guided mass transfer
            # compute contributing indices for current maximum
            lo = max(0, current_max_idx - (n - 1))
            hi = min(n - 1, current_max_idx)
            idx_range = np.arange(lo, hi + 1, dtype=int)
            # contribution of each index to the max autocorrelation
            contrib = current[idx_range] * current[current_max_idx - idx_range]
            total_contrib = contrib.sum()
            if total_contrib > 0 and rng.random() < 0.9:
                probs = contrib / total_contrib
                i = int(rng.choice(idx_range, p=probs))
            else:
                i = int(rng.integers(lo, hi + 1))
            if current[i] <= 0.0:
                continue
            # choose a different index to receive mass
            j = int(rng.integers(0, n))
            while j == i:
                j = int(rng.integers(0, n))
            # transfer a random fraction (up to 40%) of mass from i to j
            max_transfer = 0.4 * current[i]
            delta = rng.random() * max_transfer
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:
            # block multiplicative perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = np.exp(rng.normal(0.0, 0.12))
            cand = current.copy()
            cand[start:stop] *= factor
            # occasional symmetry bias
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            # re‑scale to preserve total mass (does not affect the objective)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand *= sum_target / cand_sum
            np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score, cand_conv, cand_max_idx, _ = _compute_score(cand, n, N_fft)
        if not np.isfinite(cand_score):
            continue

        # update best if improved
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score

        # accept move for the current state (simulated annealing)
        if cand_score < current_score:
            accept = True
        else:

            accept = rng.random() < accept_prob

            current = cand
            current_score = cand_score
            current_conv = cand_conv
            current_max_idx = cand_max_idx

    # safety fallback – should never be needed
    if best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_seq]
