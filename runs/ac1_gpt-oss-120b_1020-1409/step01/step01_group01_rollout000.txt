import numpy as np
import math

def generate_r_sequence(n: int) -> np.ndarray:
    """Generate the sequence r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 bound.  Uses the classic r‑sequence which gives a provably low bound.
    """
    # Determine problem size.
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256  # fallback length

    # Build the candidate sequence.
    candidate = generate_r_sequence(n)

    # Enforce evaluator's clipping range (0 ≤ x ≤ 1000).
    np.clip(candidate, 0.0, 1000.0, out=candidate)

    # Verify the candidate via the supplied evaluator; fall back if something
    # went wrong (should never happen for the r‑sequence).
    try:
        score = evaluate_sequence(candidate.tolist())
        if not (isinstance(score, float) and math.isfinite(score)):
            raise ValueError
    except Exception:
        # A very safe fallback: uniform ones.
        candidate = np.ones(n, dtype=np.float64)
        evaluate_sequence(candidate.tolist())  # ignore its score

    # Return as a plain Python list of floats.
    return [float(x) for x in candidate]
