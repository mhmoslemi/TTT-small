def _smooth_move(arr):
    n = arr.size
    sm = np.empty_like(arr)
    sm[:] = arr * 0.5
    # left neighbor contribution
    sm[1:] += arr[:-1] * 0.25
    # right neighbor
    sm[:-1] += arr[1:] * .25
    # preserve sum scaling
    s_orig = arr.sum()
    s_new = sm.sum()
    if s_new > 0:
        sm *= s_orig / s_new
    np.clip(sm, 0.0, 1000.0, out=sm)
    return sm
```

    n = arr.size

import numpy as np

# ----------------------------------------------------------------------
# Helper perturbations
# ----------------------------------------------------------------------
def _pair_transfer(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = arr.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    # fraction of donor's mass to move
    frac = rng.uniform(0.005, 0.05)
    delta = arr[i] * frac
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _block_scale(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor and renormalise."""
    n = arr.size
    min_w = max(1, n // 200)
    max_w = max(2, n // 20)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:stop] *= factor
    # renormalise to preserve the original L1 norm (scale invariance)
    orig_sum = arr.sum()
    new_sum = cand.sum()
    if new_sum > 0:
        cand *= orig_sum / new_sum
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _smooth_move(arr: np.ndarray) -> np.ndarray:
    """
    Apply a light smoothing kernel (0.25, 0.5, 0.25) and renormalise.
    This tends to flatten peaks while keeping the total mass unchanged.
    """
    n = arr.size
    sm = np.empty_like(arr)
    sm[:] = arr * 0.5
    sm[1:] += arr[:-1] * 0.25
    sm[:-1] += arr[1:] * 0.25
    # renormalise to the original sum (the objective is homogeneous)
    s_orig = arr.sum()
    s_new = sm.sum()
    if s_new > 0:
        sm *= s_orig / s_new
    np.clip(sm, 0.0, 1000.0, out=sm)
    return sm


def _gradient_transfer(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Approximate a gradient step: move mass from an index whose partner
    (with respect to the current maximal convolution index) is large
    to one whose partner is small.
    """
    n = arr.size
    # full convolution to locate the current worst hotspot
    conv = np.convolve(arr, arr)
    max_idx = int(np.argmax(conv))
    # partner indices for each position
    idx = np.arange(n)
    partner = max_idx - idx
    mask = (partner >= 0) & (partner < n)
    w = np.zeros_like(arr)
    w[mask] = arr[partner[mask]]

    # pick donor (largest w) and receiver (smallest w) among valid partners
    if not np.any(mask):
        return arr.copy()
    donor = int(np.argmax(w))
    receiver = int(np.argmin(w))
    if donor == receiver or arr[donor] <= 0.0:
        return arr.copy()

    # transfer a modest fraction of donor's mass
    frac = rng.uniform(0.02, 0.08)
    delta = arr[donor] * frac
    cand = arr.copy()
    cand[donor] = max(0.0, cand[donor] - delta)
    cand[receiver] = min(1000.0, cand[receiver] + delta)
    return cand
