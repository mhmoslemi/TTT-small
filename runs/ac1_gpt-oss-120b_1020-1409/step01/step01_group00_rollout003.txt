def _pair_transfer(arr,rng):
    n = len(arr)
    i = rng.integers(0,n)
    # choose j preferentially at edges
    if rng.random() < 0.3:
        # choose edges region uniform among first/last 5% of indices
        edge_width = max(1, n // 20)
        if rng.random() < 0.5:
            j = rng.integers(0, edge_width)
        else:
            j = rng.integers(n - edge_width, n)
    else:
        j = rng.integers(0,n)
    if i==j or arr[i] <= 0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.1) # larger range
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand

def _edge_block_scale(arr,rng):
    n = len(arr)
    width = int(rng.integers(1, max(2,n//200))) # small block
    # position near edges randomly left or right
    if rng.random() < 0.5:
        start = 0
    else:
        start = n - width
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.08)))
    cand[start:start+width] *= factor
    np.clip(cand,0,1000,out=cand)
    return cand

def _smooth(arr,rng):
    n = len(arr)
    idx = rng.integers(1,n-1) # not boundaries
    cand = arr.copy()
    # replace with weighted average of neighbors
    cand[idx] = (cand[idx-1] + cand[idx+1]) / 2.0
    return cand

import numpy as np

# ----------------------------------------------------------------------
# Helper: binomial (discrete 1/√x) sequence.
# ----------------------------------------------------------------------
def _generate_binomial_sequence(length: int) -> np.ndarray:
    """Return a_i = C(2i,i)/4^i for i=0..length-1 (float64)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a

# ----------------------------------------------------------------------
# Perturbation operators (all O(n) memory and time)
# ----------------------------------------------------------------------
def _pair_transfer(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.07)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _pair_transfer_edge(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer mass preferentially from interior to the edges."""
    n = arr.size
    i = rng.integers(0, n)                     # donor (anywhere)
    # receiver biased toward the first or last 5 % of indices
    edge_width = max(1, n // 20)
    if rng.random() < 0.5:
        j = rng.integers(0, edge_width)
    else:
        j = rng.integers(n - edge_width, n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.002, 0.12)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _block_scale(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = arr.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
    start = rng.integers(0, max(1, n - width + 1))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.07)))  # often between ~0.9–1.1
    cand = arr.copy()
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _edge_block_scale(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a small block attached to one of the edges."""
    n = arr.size
    width = int(rng.integers(1, max(2, n // 250)))
    if rng.random() < 0.5:
        start = 0
    else:
        start = n - width
    factor = float(np.exp(rng.normal(0.0, 0.09)))  # stronger edge push
    cand = arr.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _smooth_step(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Replace a random interior element by the average of its neighbours."""
    n = arr.size
    if n < 3:
        return arr.copy()
    idx = rng.integers(1, n - 1)
    cand = arr.copy()
    cand[idx] = (cand[idx - 1] + cand[idx + 1]) * 0.5
    return cand
