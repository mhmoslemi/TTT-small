import numpy as np

# ---------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full linear convolution arr * arr."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv)/sum(arr)^2 (no clipping)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    m = _conv_max(arr)
    return float(2.0 * n * m / (s * s))


def _clip(arr: np.ndarray):
    """Apply the evaluator's hard clipping in‑place."""
    np.clip(arr, 0.0, 1000.0, out=arr)
    return arr


# ---------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that
    aims to minimise the AC1 upper bound defined by `evaluate_sequence`.
    The routine:
      * obtains length n from the provided parent `height_sequence_1`
        (fallback to 256);
      * evaluates a uniform candidate (value 1.0) – this attains the
        theoretical lower bound 2.0;
      * optionally tries the parent vector (after clipping);
      * performs a tiny random local search around the current best
        candidate using a scale‑perturbation that preserves the L1‑norm.
    The best candidate found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ---- determine length -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        n = 256
        parent = None

    # ---- uniform candidate (value 1.0) ------------------------------------
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = _score(best, n)

    # ---- try the parent (if any) -----------------------------------------
    if parent is not None:
        cand = _clip(parent.copy())
        sc = _score(cand, n)
        if sc < best_score:
            best = cand
            best_score = sc

    # ---- tiny stochastic refinement ---------------------------------------
    for _ in range(200):
        cand = best.copy()
        # pick a random block and perturb multiplicatively
        i = int(rng.integers(0, n))
        factor = float(np.exp(rng.normal(0.0, 0.04)))
        cand[i] = np.clip(cand[i] * factor, 0.0, 1000.0)
        # re‑normalise L1‑norm to keep the scale comparable
        s = cand.sum()
        if s > 0.0:
            cand *= (best.sum() / s)
        cand = _clip(cand)
        sc = _score(cand, n)
        if sc < best_score:
            best = cand
            best_score = sc

    # ---- return as plain list --------------------------------------------
    return [float(x) for x in best]
