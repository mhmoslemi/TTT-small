def flatten_step(a, tau=0.05):
    B = np.convolve(a, a)
    k = np.argmax(B)
    n = a.size
    # gradient contributions
    grad = np.zeros_like(a)
    i_min = max(0, k - (n-1))
    i_max = min(n-1, k)
    # fill
    # vectorized: idx = np.arange(i_min, i_max+1)
    # grad[idx] = 2 * a[k - idx]
    idx = np.arange(i_min, i_max+1)
    grad[idx] = 2.0 * a[k - idx]
    # select indices with largest grad (top fraction) to reduce
    num_change = max(1, n // 20)  # 5% of entries
    top = np.argpartition(grad, -num_change)[-num_change:]  # indices of largest values
    bottom = np.argpartition(grad, num_change)[:num_change]  # smallest
    # apply multiplicative factor
    factor_down = 1.0 - tau
    factor_up = 1.0 + tau
    a_new = a.copy()
    a_new[top] *= factor_down
    a_new[bottom] *= factor_up
    # clip
    a_new = np.clip(a_new, 0, None)
    # renorm sum to 1
    sum_a = np.sum(a_new)
    if sum_a <= 0:
        a_new = np.ones_like(a_new) / a_new.size
    else:
        a_new /= sum_a
    return a_new
```

import numpy as np

def _flatten_step(a, step_factor):
    n = a.shape[0]
    B = np.convolve(a, a)
    k = int(np.argmax(B))
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    # number of indices to modify: at least 1 or fraction
    change_cnt = max(1, n // 30)  # about 3% of n
    # largest grads
    top_idx = np.argpartition(grad, -change_cnt)[-change_cnt:]
    # smallest grads
    low_idx = np.argpartition(grad, change_cnt)[:change_cnt]
    a_new = a.copy()
    a_new[top_idx] *= (1.0 - step_factor)
    a_new[low_idx] *= (1.0 + step_factor)
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0:
        a_new = np.ones_like(a_new) / n
    else:
        a_new /= s
    return a_new

    for _ in range(num_random):
        cand = rng.random(n)
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum  # sum = 1
        # scaling not needed, evaluator does scaling internally
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
    # local refinement
    a = best.copy()
    # schedule of step factors decreasing
    for it in range(num_iters):
        # decreasing step size
        factor = 0.1 * (1.0 - it / num_iters) + 0.001
        cand = _flatten_step(a, factor)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            a = cand.copy()
        else:
            # optionally occasional random perturbation to escape
            if rng.random() < 0.05:
                # add small noise
                noise = rng.normal(scale=0.01, size=n)
                cand = a + noise
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s == 0:
                    continue
                cand /= s
                a = cand
    # final check
    if not np.isfinite(best_score):
        best = parent_clip
    # convert to list of floats
    return [float(x) for x in best]

import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a power‑law sequence a_i = 1 / sqrt(i + alpha)
# The result is normalised so that its L1‑norm is 1 (scale does not matter
# for the evaluator because the score is homogeneous of degree 0).
# ----------------------------------------------------------------------
def _power_law(alpha: float, n: int) -> np.ndarray:
    seq = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + alpha)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total == 0.0:
        seq[:] = 1.0 / n
    else:
        seq /= total
    return seq


# ----------------------------------------------------------------------
# One iteration that tries to flatten the peak of the self‑convolution.
# It identifies the index k where a*a is maximal, then slightly reduces
# the entries that contribute most to that peak and slightly boosts the
# least contributing entries.  Afterwards the vector is re‑normalised.
# ----------------------------------------------------------------------
def _flatten_step(a: np.ndarray, step: float) -> np.ndarray:
    n = a.shape[0]
    conv = np.convolve(a, a)                 # length 2n‑1
    k = int(np.argmax(conv))                 # index of the current peak
    # gradient of the peak w.r.t. a_i (ignoring the factor 2, only ranking matters)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    # for each i in the support of the peak, contribution is a_{k-i}
    grad[idx] = a[k - idx]

    # decide how many coordinates to touch (≈5 % of the length)
    change_cnt = max(1, n // 20)

    # indices with largest / smallest gradient entries
    if change_cnt < n:
        top = np.argpartition(grad, -change_cnt)[-change_cnt:]
        low = np.argpartition(grad, change_cnt)[:change_cnt]
    else:                 # very small n – just use everything
        top = np.arange(n)
        low = np.arange(n)

    a_new = a.copy()
    a_new[top] *= (1.0 - step)   # pull down the heavy contributors
    a_new[low] *= (1.0 + step)   # push up the light contributors
    a_new = np.clip(a_new, 0.0, None)

    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s                # renormalise to L1‑norm 1 (scale is irrelevant)
    return a_new

# The function must return a plain Python list of finite floats.

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

        n = int(parent.size)
    except Exception:                     # no parent supplied
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Normalise the parent (scale does not affect the score)
    parent_norm = parent.copy()
    s_parent = parent_norm.sum()
    if s_parent == 0.0:
        parent_norm[:] = 1.0 / n
    else:
        parent_norm /= s_parent

    # ------------------------------------------------------------------
    # Initialise the best candidate with the (normalised) parent.
    # ------------------------------------------------------------------
    best = parent_norm.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 1) Scan the one‑parameter family a_i = 1/sqrt(i+alpha).
    # ------------------------------------------------------------------
    alphas = np.logspace(-6, 2, num=num_alpha, dtype=np.float64)
    for alpha in alphas:
        cand = _power_law(alpha, n)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # 2) Random search – generate many Dirichlet‑like random vectors.
    # ------------------------------------------------------------------

        cand = rng.gamma(shape=1.0, scale=1.0, size=n)
        total = cand.sum()
        if total == 0.0:
            continue
        cand /= total
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
