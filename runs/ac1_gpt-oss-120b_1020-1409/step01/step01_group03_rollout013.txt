import numpy as np

# ------------------------------------------------------------
# Fast full convolution using real FFT (O(n log n))
# ------------------------------------------------------------
def _conv_self_fft(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv_full = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv_full


# ------------------------------------------------------------
# Exact AC1 upper bound (mirrors evaluator)
# ------------------------------------------------------------
def _score(x: np.ndarray) -> float:
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self_fft(x)
    return 2.0 * x.size * c.max() / (s * s)

def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a
