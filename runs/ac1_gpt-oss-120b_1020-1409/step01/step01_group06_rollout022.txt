import numpy as np

def propose_candidate(seed=42, **kwargs):

    Return a non‑negative step‑function whose AC1 upper bound is provably optimal.
    The theoretical minimum of the evaluator 2·n·max_b / sum_a² is 2,
    attained by a flat (uniform) sequence with the prescribed ℓ₁‑norm

    """

    # Determine the length n from the available parent, otherwise use a fallback.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():

    # minimal possible objective value (theoretical lower bound).
