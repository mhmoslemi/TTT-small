def autocorr_max(seq):
    n = seq.size
    # compute convolution of seq with itself (full)
    # Using FFT length = next power of two maybe 2*n.
    size = 1 << ((2*n-1).bit_length())
    # compute FFT
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)
    # conv length = size, but we need first 2*n-1 values.
    conv = conv[:2*n-1]
    return conv.max()

import numpy as np

def _score(seq: np.ndarray) -> float:
    """Fast surrogate for the evaluator: 2·n·⟨a,a⟩ / (sum a)^2."""
    n = seq.size
    s = seq.sum()
    if s == 0.0:
        return float('inf')
    ssq = np.dot(seq, seq)
    return 2.0 * n * ssq / (s * s)


def propose_candidate(seed: int = 42, **kwargs):
    """Return a coefficient list with a low AC1 upper bound.

    The routine starts from the provided parent ``height_sequence_1`` (if
    present) and performs a cheap stochastic local search that keeps the
    sum of the sequence constant.  All operations are O(n) and use only
    linear extra memory.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise from the parent, or fall back to a uniform start.
    # ------------------------------------------------------------------
    try:
        seq = np.array(height_sequence_1, dtype=np.float64).copy()
        n = seq.size
    except Exception:                     # parent not supplied
        n = 256
        seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s admissible range and enforce a minimal sum.
    np.clip(seq, 0.0, 1000.0, out=seq)
    cur_sum = seq.sum()
    if cur_sum < 0.01:
        seq += (0.01 - cur_sum) / n
        cur_sum = seq.sum()
    target_sum = cur_sum                      # we preserve this sum thereafter

    # ------------------------------------------------------------------
    # 2.  Evaluate the starting point.
    # ------------------------------------------------------------------
    best = seq.copy()
    best_score = _score(best)

    # If something went wrong (should not happen), revert to a safe uniform.
    if not np.isfinite(best_score):
        best = np.full(n, 1.0 / n, dtype=np.float64)
        target_sum = best.sum()
        best_score = _score(best)

    # ------------------------------------------------------------------
    # 3.  Stochastic local optimisation.
    # ------------------------------------------------------------------
    steps = 5000
    max_width = max(1, n // 100)          # at most 1 % of the length
    sigma = 0.03                           # std‑dev for log‑normal factors

    for _ in range(steps):
        cand = best.copy()
        # Choose one of three cheap perturbations.
        ptype = rng.integers(0, 3)

        if ptype == 0:                     # multiplicative block perturbation
            width = int(rng.integers(1, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor

        elif ptype == 1:                   # transfer a tiny amount between two sites
            i, j = rng.integers(0, n, size=2)
            # move up to 1 % of the source value, respecting non‑negativity
            delta = rng.uniform(0.0, 0.01) * cand[i]
            delta = min(delta, cand[i])          # cannot go below zero

            cand[j] += delta

        else:                               # additive Gaussian noise on a single entry

            # scale the noise relative to the current magnitude (or 1e‑3 if zero)
            scale = cand[i] if cand[i] > 0 else 1e-3
            cand[i] = max(0.0, cand[i] + rng.normal(0.0, sigma * scale))

        np.clip(cand, 0.0, 1000.0, out=cand)
