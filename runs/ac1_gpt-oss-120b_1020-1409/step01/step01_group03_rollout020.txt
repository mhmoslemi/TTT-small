import numpy as np

# ------------------------------------------------------------
# Helper: classic a_i = C(2i,i)/4^i series (the previous state‑of‑the‑art)
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the length‑`length` truncation of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast convolution (real input) via FFT.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact score (mirrors evaluate_sequence).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)


# ------------------------------------------------------------
# One sub‑gradient step that tries to decrease the maximal
# convolution value while keeping the ℓ1‑norm fixed.
# ------------------------------------------------------------
def _subgrad_step(x: np.ndarray, target_sum: float, base_step: float):
    """Return a candidate obtained by a sub‑gradient descent step."""
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    # indices where the maximum is attained (within tiny tolerance)
    ks = np.where(np.isclose(conv, M, atol=1e-12))[0]

    # sub‑gradient of the max‑convolution (ignoring the factor 2,
    # which can be absorbed into the step size)
    g = np.zeros_like(x)
    for k in ks:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] += x[k - idx]

    # try a few decreasing step sizes and keep the best improvement
    best_cand = x
    best_sc = _score(x)
    step = base_step
    for _ in range(4):
        cand = x - step * g
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s > 0.0:
            cand *= target_sum / s          # project onto the ℓ1‑hyperplane
        sc = _score(cand)
        if sc < best_sc:
            best_cand = cand
            best_sc = sc
        step *= 0.5
    return best_cand, best_sc

def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):
    """
    Search for a non‑negative step‑function that improves the
    AC1 upper bound.  The method starts from the classic series,
    then performs a dedicated sub‑gradient optimisation on the
    maximal convolution value while keeping the total sum fixed.
    Random local block perturbations are interleaved to escape
    shallow minima.  The best sequence seen is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size; use the supplied parent if present.
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():

        parent = None

    # --------------------------------------------------------
    # Initial candidate: the classic (C(2i,i)/4^i) series.
    # --------------------------------------------------------
    x = _make_series(n)
    target_sum = x.sum()
    best = x.copy()
    best_score = _score(best)

    # If a parent exists and is already better, start from it.
    if parent is not None:
        parent = np.clip(parent, 0.0, 1000.0)
        p_score = _score(parent)
        if p_score < best_score:
            best = parent.copy()
            best_score = p_score
            target_sum = best.sum()
        x = best.copy()

    # --------------------------------------------------------
    # Optimisation loop.
    # --------------------------------------------------------
    step = 0.12                     # initial sub‑gradient step size
    stagnant = 0                    # counts iterations without improvement
    cur_score = best_score

    for it in range(num_steps):
        # ----- sub‑gradient descent on max‑convolution -----
        cand, cand_sc = _subgrad_step(x, target_sum, step)

        # ----- occasional random block perturbation -----
        if it % 13 == 0:
            width = max(1, n // rng.integers(15, 35))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            block = cand.copy()
            block[start:start + width] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            s = block.sum()
            if s > 0.0:
                block *= target_sum / s
            sc_block = _score(block)
            if sc_block < cand_sc:
                cand, cand_sc = block, sc_block

        # ----- accept if we made progress -----
        if cand_sc + 1e-15 < cur_score:
            x = cand
            cur_score = cand_sc
        # else keep the previous x (monotone descent)

        # ----- update global best -----
        if cur_score < best_score - 1e-15:
            best = x.copy()
            best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1

        # ----- step‑size decay -----
        step *= 0.9996

        # ----- early exit if stagnation persists -----
        if stagnant > 500:
