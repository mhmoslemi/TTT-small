import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """Binomial (central) coefficients a_i = C(2i,i)/4^i, i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a

def _pair_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small random fraction of mass from a donor to a receiver."""
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.04)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand

def _block_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = arr.size
    width = int(rng.integers(max(1, n // 250), max(2, n // 30)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _max_convolution_index(arr: np.ndarray, fft_len: int) -> int:
    """Return index of max entry of the self‑convolution of arr."""
    fa = np.fft.rfft(arr, n=fft_len)
    b = np.fft.irfft(fa * fa, n=fft_len)[: 2 * arr.size - 1]
    return int(np.argmax(b))
