import numpy as np
import time

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the sequence r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function that (deterministically) improves
    the AC1 upper bound.  The algorithm performs a simple projected gradient
    descent on the scale‑invariant objective
        f(a) = 2·n·max(a*a) / (sum(a)^2)
    using the exact sub‑gradient of the max‑convolution term.
    """
    # --------------------------------------------------------------------
    # 1. initialise sequence ------------------------------------------------
    # --------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_r_sequence(n)

    a = np.clip(a, 0.0, 1000.0)          # respect the evaluator's clipping

    # --------------------------------------------------------------------
    # 2. initial score ----------------------------------------------------
    # --------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------------
    # 3. gradient‑descent parameters ---------------------------------------
    # --------------------------------------------------------------------
    n = best.size
    indices = np.arange(n, dtype=np.int64)

    max_iter = 8000               # enough for a few thousand improvements
    lr = 0.025                     # base learning‑rate (change per step ~ lr)
    decay = 0.9995                # exponential cooling
    min_lr = 1e-9

    # --------------------------------------------------------------------
    # 4. main optimisation loop --------------------------------------------
    # --------------------------------------------------------------------
    for it in range(max_iter):
        # convolution and the location of the maximal entry
        conv = np.convolve(best, best)
        max_b = conv.max()
        k = int(conv.argmax())
        sum_a = best.sum()

        # compute g_i = ∂max_b/∂a_i = a_{|i-k|} (zero outside the range)
        diffs = np.abs(indices - k)
        mask = diffs < n
        g = np.empty_like(best)
        g[mask] = best[diffs[mask]]

        # sub‑gradient direction for the scale‑invariant objective
        d = 2.0 * max_b - sum_a * g      # proportional to -∇f

        # normalise step so that the largest absolute change is ≈ lr
        max_abs = np.max(np.abs(d))
        if max_abs == 0.0:
            break
        step = lr / max_abs
        cand = best + step * d
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            # on success allow a modest increase of learning rate
            lr = min(lr * 1.05, 0.1)
        else:
            # on failure shrink learning rate
            lr *= 0.5

        # global cooling
        lr = max(lr * decay, min_lr)

        # early exit if learning rate has collapsed
        if lr < 1e-12:
            break

    # --------------------------------------------------------------------
    # 5. cheap random polishing (optional, deterministic) ------------------
    # --------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    # try a few small random tweaks; accept only improvements
    for _ in range(200):
        j = rng.integers(0, n)
        delta = (rng.random() - 0.5) * 0.02 * best[j]  # ≤1 % relative change
        if delta == 0.0:
            continue
        cand = best.copy()
        cand[j] = max(0.0, cand[j] + delta)
        np.clip(cand, 0.0, 1000.0, out=cand)

        best = np.ones_like(best, dtype=np.float64)

    # return plain Python list as required by the harness
    return [float(x) for x in best]
