def _local_search(start_vec, rng, target_sum, n, best_score, steps=2500):
    best_vec = start_vec.copy()
    cur_vec = start_vec.copy()
    cur_score = best_score
    T = best_score * 0.05  # small temperature
    temp_decay = 0.995
    for step in range(steps):
        # perturbation
        new_vec = cur_vec.copy()
        # block scaling
        max_width = max(2, n // 30)
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.2))
        new_vec[start:start+width] *= factor
        # mass transfer
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            delta = rng.uniform(-0.05, 0.05) * new_vec[i]
            if new_vec[i] - delta >= 0 and new_vec[j] + delta >= 0:
                new_vec[i] -= delta
                new_vec[j] += delta
        # optional symmetrize
        if rng.random() < 0.2:
            new_vec = 0.5 * (new_vec + new_vec[::-1])
        # clip and renorm
        np.clip(new_vec, 0.0, 1000.0, out=new_vec)
        total = new_vec.sum()
        if total > 0.0:
            new_vec *= target_sum / total
        else:
            continue
        new_score = _score(new_vec, n)
        # acceptance
        if new_score < best_score:
            best_score = new_score
            best_vec = new_vec.copy()
            cur_vec = new_vec
            cur_score = new_score
        else:
            # Metropolis
            if rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
                cur_vec = new_vec
                cur_score = new_score
        # cooling schedule
        if step % 50 == 0:
            T *= temp_decay
            # keep T from going too low
            if T < 1e-6:

import numpy as np

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum of the full auto-convolution of arr (via FFT)."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two >= 2n-1
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact AC1 upper bound used by the official evaluator."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    return float(2.0 * n * _conv_max(arr) / (s * s))


def _clip_and_renorm(arr: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and scale to have L1‑norm = target_sum."""
    np.clip(arr, 0.0, 1000.0, out=arr)
    total = arr.sum()
    if total > 0.0:
        arr *= target_sum / total
    else:                                   # degenerate case – fall back to uniform
        arr[:] = target_sum / arr.size
    return arr


# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^{‑p}; optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


def _make_exponential(n: int, lam: float, sym: bool) -> np.ndarray:
    """a_i ∝ exp(-λ i); optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = np.exp(-lam * idx)
    if sym:
        base = base + np.exp(-lam * (idx[::-1]))
    return base


def _make_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π(i+½)/(n+1))^p (already symmetric)."""
    idx = np.arange(n, dtype=np.float64)
    base = np.sin(np.pi * (idx + 0.5) / (n + 1.0))
    return base ** p


def _make_triangle(n: int) -> np.ndarray:
    """Triangular shape: linearly increasing then decreasing."""
    idx = np.arange(n, dtype=np.float64)
    return np.minimum(idx + 1.0, n - idx)


# ----------------------------------------------------------------------
def _local_search(start_vec: np.ndarray,
                  rng: np.random.Generator,
                  target_sum: float,
                  n: int,
                  init_score: float,
                  steps: int = 2500) -> np.ndarray:
    """Simple simulated‑annealing style refinement."""
    best_vec = start_vec.copy()
    cur_vec = start_vec.copy()
    best_score = init_score
    cur_score = init_score

    T = best_score * 0.05                     # initial temperature
    decay = 0.995

    max_width = max(2, n // 30)

    for step in range(steps):
        cand = cur_vec.copy()

        # --- block scaling -------------------------------------------------
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.2))
        cand[start:start + width] *= factor

        # --- pairwise mass transfer ---------------------------------------
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            delta = rng.uniform(-0.05, 0.05) * cand[i]
            if cand[i] - delta >= 0.0 and cand[j] + delta >= 0.0:
                cand[i] -= delta
                cand[j] += delta

        # --- optional symmetrisation --------------------------------------
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])

        # --- clip, renorm --------------------------------------------------
        np.clip(cand, 0.0, 1000.0, out=cand)
        total = cand.sum()
        if total == 0.0:
            continue
        cand *= target_sum / total

        # --- evaluate ------------------------------------------------------
        new_score = _score(cand, n)

        # --- acceptance ----------------------------------------------------
        if new_score < best_score:
            best_score = new_score
            best_vec = cand.copy()
            cur_vec = cand
            cur_score = new_score
        else:
            # Metropolis move based on current temperature
            if rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
                cur_vec = cand
                cur_score = new_score

        # --- cooling -------------------------------------------------------
        if step % 50 == 0:
            T = max(T * decay, 1e-6)

    return best_vec

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose
    AC1 upper bound is aimed to be below 1.50274349.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine length n (use parent if present)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Initial candidates
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    best_vec = uniform.copy()
    best_score = _score(best_vec, n)

    # parent as a candidate
    pv = _clip_and_renorm(parent.astype(np.float64), target_sum)
    s = _score(pv, n)
    if s < best_score:
        best_vec = pv.copy()
        best_score = s

    # --------------------------------------------------------------
    # Parameter grids
    p_vals = np.linspace(0.2, 1.2, 11)
    beta_vals = np.logspace(-2.0, 2.0, 11)
    lam_vals = np.logspace(-3.0, 0.0, 12)
    sine_p_vals = np.linspace(0.2, 3.0, 14)

    # Power‑law family
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                a = _make_powerlaw(n, p, beta, sym)
                a = _clip_and_renorm(a, target_sum)
                sc = _score(a, n)
                if sc < best_score:
                    best_score, best_vec = sc, a.copy()

    # Exponential family
    for sym in (False, True):
        for lam in lam_vals:
            a = _make_exponential(n, lam, sym)
            a = _clip_and_renorm(a, target_sum)
            sc = _score(a, n)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()

    # Sine‑power family (already symmetric)
    for p in sine_p_vals:
        a = _make_sine_power(n, p)
        a = _clip_and_renorm(a, target_sum)
        sc = _score(a, n)
        if sc < best_score:
            best_score, best_vec = sc, a.copy()

    # Triangle shape
    a = _make_triangle(n)
    a = _clip_and_renorm(a, target_sum)
    sc = _score(a, n)
    if sc < best_score:
        best_score, best_vec = sc, a.copy()

    # --------------------------------------------------------------
    # Local refinement (simulated annealing)
    refined = _local_search(best_vec, rng, target_sum, n, best_score, steps=2500)
    refined_score = _score(refined, n)
    if refined_score < best_score:
        best_vec = refined
        best_score = refined_score

    # --------------------------------------------------------------
    # Final safety: ensure the sequence is finite and non‑negative
    final = [float(x) for x in best_vec]
    return final
