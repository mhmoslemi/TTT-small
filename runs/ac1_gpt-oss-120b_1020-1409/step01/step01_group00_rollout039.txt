import numpy as np

# ------------------------------------------------------------
# Helper utilities (all matrix‑free, linear memory usage)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility to get the next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _compute_bound(seq: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute the AC1 upper bound exactly as evaluate_sequence does,
    but without the extra safety checks.  `seq` is modified in‑place
    only to apply the [0,1000] clipping (a copy should be passed if the
    original data must stay unchanged).
    """
    np.clip(seq, 0.0, 1000.0, out=seq)
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    # convolution via FFT (size fft_len >= 2*n-1)
    fa = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (sum_a * sum_a)


def _generate_binomial_sequence(length: int) -> np.ndarray:
    """
    a_i = C(2i,i) / 4^i  (the coefficients of (1-x)^-1/2).
    They satisfy (a*a)[k] = 1 for all k < length.
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a

    from early (large) entries to late (small) entries.  Convolution
    is evaluated by a fast FFT‑based routine, keeping the whole process
    O(n log n) per iteration.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size (use parent if supplied)
    # --------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])      # provided by the harness
    except Exception:
        n = 256                                   # a safe fallback

    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------
    # 1️⃣  Gather a few cheap initial candidates
    # --------------------------------------------------------
    # a) binomial (near optimal theoretical constant)
    best = _generate_binomial_sequence(n)
    np.clip(best, 0.0, 1000.0, out=best)
    best_score = _compute_bound(best.copy(), n, fft_len)

    # b) power‑law with a small offset (reduces the huge first coefficient)
    for offset in (0.5, 1.0, 2.0):
        cand = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + offset)
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _compute_bound(cand.copy(), n, fft_len)
        if sc < best_score:
            best = cand
            best_score = sc

    # c) mild exponential decay (just another shape to test)
    for scale in (0.001, 0.002, 0.005, 0.01):
        cand = np.exp(-np.arange(n, dtype=np.float64) * scale)
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _compute_bound(cand.copy(), n, fft_len)
        if sc < best_score:
            best = cand
            best_score = sc

    # --------------------------------------------------------
    # 2️⃣  Hill‑climbing: move tiny mass packets from the early
    #     (large) part of the vector towards the late (tiny) part.
    # --------------------------------------------------------
    cur = best.copy()                     # current working point
    early_cut = int(0.15 * n)             # first 15 % are donors
    late_start = int(0.85 * n)            # last 15 % are receivers
    early_idxs = np.arange(early_cut, dtype=np.int64)
    late_idxs = np.arange(late_start, n, dtype=np.int64)

    stale = 0                              # counts iterations without improvement
    for step in range(num_steps):
        # ---- small pairwise mass transfer ---------------------------------
        if early_idxs.size and cur[early_idxs].max() > 0.0:
            donor = int(rng.choice(early_idxs))
        else:
            donor = int(rng.integers(n))
        recipient = int(rng.choice(late_idxs))
        if donor == recipient or cur[donor] <= 0.0:
            continue

        # transfer a modest fraction of the donor entry
        frac = rng.uniform(0.001, 0.02)
        delta = cur[donor] * frac
        cand = cur.copy()
        cand[donor] = max(0.0, cand[donor] - delta)
        cand[recipient] = min(1000.0, cand[recipient] + delta)

        sc = _compute_bound(cand, n, fft_len)
        if sc < best_score:
            best = cand
            best_score = sc
            cur = cand
            stale = 0
        else:
            stale += 1

        # ---- occasional block scaling (adds a bit of stochasticity) -----
        if step % 250 == 0:
            start = int(rng.integers(int(0.5 * n), n))
            max_len = max(2, int(0.04 * n))
            length = int(rng.integers(5, max_len))
            end = min(n, start + length)
            factor = float(np.exp(rng.normal(0.0, 0.05)))
            cand2 = cur.copy()
            cand2[start:end] *= factor
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            sc2 = _compute_bound(cand2, n, fft_len)
            if sc2 < best_score:
                best = cand2
                best_score = sc2
                cur = cand2
                stale = 0

        # ---- escape from a flat region ------------------------------------
        if stale > 2000:
            # a random jump that still respects the total mass
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and cur[i] > 0.0:
                delta = cur[i] * rng.uniform(0.02, 0.07)
                tmp = cur.copy()
                tmp[i] = max(0.0, tmp[i] - delta)
                tmp[j] = min(1000.0, tmp[j] + delta)
                sc = _compute_bound(tmp, n, fft_len)
                if sc < best_score:
                    best = tmp
                    best_score = sc
                    cur = tmp
            stale = 0

    # --------------------------------------------------------
    # Return plain Python list (required by the harness)
    # --------------------------------------------------------
    return [float(v) for v in best]
