import numpy as np

# ----------------------------------------------------------------------
# Utility: compute the AC1 upper bound score for a given sequence.
# ----------------------------------------------------------------------
def compute_score(seq: np.ndarray) -> float:
    """Return the objective value 2*n*max(conv(seq,seq))/sum(seq)^2.
    Values are clipped to the allowed range before evaluation.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:
        return float('inf')
    n = seq.size
    # length of FFT – next power of two for speed
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    return float(2.0 * n * max_b / (total * total))
