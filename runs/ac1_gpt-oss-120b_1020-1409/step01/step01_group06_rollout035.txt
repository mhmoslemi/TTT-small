def perturb(vec):
    cand = vec.copy()
    n = len(cand)
    # random operation type
    op = rng.integers(0,3)
    if op ==0: # block scale
        width = rng.integers(max(1, n//200), max(2, n//20))
        start = rng.integers(0, max(1, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.06))
        cand[start:start+width] *= factor
    elif op ==1: # swap mass
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        delta = rng.uniform(0, 0.1) * cand[i]
        cand[i] -= delta
        cand[j] += delta
    else: # random noise additive
        eps = rng.normal(0, 0.02, size=n)
        cand += cand * eps
    # mirror? optional
    if rng.random() < 0.2:
        cand = 0.5*(cand + cand[::-1])
    np.clip(cand,0.0,1000.0, out=cand)
    # renormalize to target sum
    s = cand.sum()
    if s>0:
        cand *= target_sum / s
    return cand

import numpy as np

# ----------------------------------------------------------------------
# Convolution max using FFT (O(n log n))
def _conv_max(arr: np.ndarray) -> float:
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())           # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Exact score used during search (mirrors evaluate_sequence)
def _score(arr: np.ndarray) -> float:
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * arr.shape[0] * max_b / (s * s))


# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p)   (optionally symmetrised)."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


# ----------------------------------------------------------------------
def _normalize_and_clip(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to [0,1000] and renormalise to target L1‑norm."""
    np.clip(vec, 0.0, 1000.0, out=vec)
    s = vec.sum()
    if s > 0.0:
        vec *= target_sum / s
        np.clip(vec, 0.0, 1000.0, out=vec)
    return vec


# ----------------------------------------------------------------------
def _perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Small stochastic neighbourhood move (keeps L1‑norm roughly fixed)."""
    cand = vec.copy()
    n = cand.shape[0]

    op = rng.integers(0, 3)
    if op == 0:                                  # block scaling
        width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.06)))
        cand[start:start + width] *= factor
    elif op == 1:                                # transfer between two positions
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        delta = float(rng.random() * cand[i] * 0.1)   # up to 10 % of a_i
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] += delta
    else:                                        # multiplicative noise
        eps = rng.normal(0.0, 0.02, size=n)
        cand += cand * eps

    if rng.random() < 0.2:                       # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])

    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
        np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
