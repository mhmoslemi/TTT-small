def _conv_self(x):
    n = x.shape[0]
    conv_len = 2*n-1
    fft_len = 1 << (conv_len-1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    c = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return c

import numpy as np

MAX_VAL = 1000.0  # upper bound enforced by the evaluator


def _make_series(length: int) -> np.ndarray:
    """Catalan‐type series a_i = C(2i,i)/4**i (optimal for M≈1)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full convolution x*x via real FFT (length 2*len(x)-1)."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, n=fft_len)
    c = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return c


def _fast_score(x: np.ndarray) -> float:
    """Score = 2*n*max(conv)/sum(x)^2 (same formula as the evaluator)."""
    s = x.sum()
    if s < 0.01:
        return np.inf
    m = _conv_self(x).max()
    return 2.0 * x.size * m / (s * s)


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    previous bound.  The routine starts from the provided parent
    (height_sequence_1) or from the Catalan series, then runs a
    matrix‑free Adam optimisation on the objective
        f(x) = max(conv(x,x)) / (sum x)^2
    while keeping all entries in [0,MAX_VAL].
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Acquire problem size and a starting vector.
    # ------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:
        parent = None
        n = 256

    if parent is not None:
        x = np.clip(parent, 0.0, MAX_VAL)
    else:
        x = _make_series(n)
        x = np.clip(x, 0.0, MAX_VAL)

    best = x.copy()
    best_score = _fast_score(best)

    # ------------------------------------------------------------
    # Adam optimiser parameters (matrix‑free, O(n) per step).
    # ------------------------------------------------------------
    lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    max_iters = 800
    no_improve = 0

    for t in range(max_iters):
        # ---- convolution, max index, and sums ----
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        S = x.sum()
        if S < 1e-12:
            break

        # ---- gradient of f = M / S^2 (scale‑invariant) ----
        g = np.zeros_like(x)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        if i_low <= i_high:
            idx = np.arange(i_low, i_high + 1, dtype=int)
            g[idx] = 2.0 * x[k - idx]          # dM/dx_i = 2 * x_{k-i}
        grad = g / (S * S) - (2.0 * M) / (S * S * S) * np.ones_like(x)

        # ---- Adam update (matrix‑free) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        x_new = x - step
        np.clip(x_new, 0.0, MAX_VAL, out=x_new)

        # ---- occasional random block perturbation (exploration) ----
        if t % 20 == 0 and rng.random() < 0.3:
            width = max(1, n // rng.integers(15, 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            block = x_new.copy()
            block[start:start + width] *= factor
            np.clip(block, 0.0, MAX_VAL, out=block)
            if _fast_score(block) < _fast_score(x_new):
                x_new = block

        # ---- light smoothing (helps flatten the shape) ----
        if rng.random() < 0.1:
            avg = (x_new + np.roll(x_new, 1)) * 0.5
            x_new = (x_new + avg) * 0.5
            np.clip(x_new, 0.0, MAX_VAL, out=x_new)

        # ---- evaluation and bookkeeping ----
        sc = _fast_score(x_new)
        if sc < best_score - 1e-12:
            best = x_new.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1

        x = x_new
        if no_improve > 150:
            break

    # ------------------------------------------------------------
    # Fallback safety net – guarantee a valid non‑empty list.
    # ------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.full(n if n > 0 else 1, 1.0, dtype=np.float64)
        best = np.clip(best, 0.0, MAX_VAL)

    return [float(v) for v in best]
