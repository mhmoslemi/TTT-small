import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..length-1 (the classic 1/√i sequence)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        f = float(i)
        a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
    return a

def _compute_score(arr: np.ndarray) -> float:
    """Exact same formula as evaluate_sequence but without the safety wrapper."""
    if arr.size == 0:
        return float('inf')
    conv = np.convolve(arr, arr)
    max_b = np.max(conv)
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * arr.size * max_b / (sum_a * sum_a))

def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 upper bound
    is better than the uniform baseline.  The method starts from the parent
    construction (if present) and repeatedly performs a gradient‑guided mass
    transfer that equalises the contribution to the maximal autocorrelation,
    interleaved with occasional block scalings and random pair moves.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise the sequence.
    # ------------------------------------------------------------------
    try:

        n_fallback = 256
        a = _generate_binomial_sequence(n_fallback)
    n = a.size
