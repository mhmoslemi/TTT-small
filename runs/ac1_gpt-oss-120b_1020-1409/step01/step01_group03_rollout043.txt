    else:

    # initialize
    def make_catalan(n):
        a = np.empty(n, dtype=np.float64)
        a[0]=1.0
        for i in range(1,n):
            a[i]=a[i-1] * (2*i-1) / (2.0*i)
        return a
    if parent is not None:
        init = np.clip(parent, 0.0, 1000.0)
        init_score = _score(init)
        catalan = _make_series(n)
        catalan_score = _score(catalan)
        if catalan_score < init_score:
            current = catalan.copy()
            current_score = catalan_score
        else:
            current = init.copy()
            current_score = init_score
    else:
        current = _make_series(n)
        current_score = _score(current)
    best = current.copy()
    best_score = current_score
    # Simulated annealing
    max_iter = 6000
    # initial temperature
    temp0 = best_score * 0.1 + 1e-3
    for it in range(max_iter):
        # temperature schedule
        t = temp0 * (0.995 ** it)
        # propose modification
        cand = current.copy()
        # choose modification type
        mtype = rng.integers(0, 3)
        if mtype == 0:
            # block scaling
            blk = max(1, n // rng.integers(15, 40))
            start = rng.integers(0, n - blk + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[start:start+blk] *= factor
        elif mtype == 1:
            # symmetric perturb
            if rng.random() < 0.5:
                perm = np.arange(n)[::-1]  # reversed
                cand = 0.5*(cand + cand[perm])
        else:
            # random jitter
            jitter = rng.normal(0.0, 0.3, size=n)  # adjust scale later
            cand += jitter
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _score(cand)
        # acceptance
        if cand_score < current_score or rng.random() < np.exp(-(cand_score - current_score) / max(t,1e-9)):
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        # early stop if best_score below target
        if best_score < 1.50274349:
            break
    # guarantee non-empty list
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
    return [float(v) for v in best]

def _gradient_step(x):
   conv = _conv_self(x)
   M = conv.max()
   k = int(conv.argmax())
   n = x.size
   # compute g_i = x[k-i] if 0 <= k-i < n else 0, plus x[i] for i == k-i (i.e., add again)
   g = np.zeros_like(x)
   i_low = max(0, k - (n - 1))
   i_high = min(k, n - 1)
   idx = np.arange(i_low, i_high + 1, dtype=int)
   g[idx] = x[k - idx]
   # add diagonal term
   if k % 2 == 0:
      i_mid = k // 2
      if i_mid < n:
         g[i_mid] += x[i_mid]  # adds extra a_i term
   S = x.sum()
   inv_S2 = 1.0 / (S * S)
   grad = 2.0 * inv_S2 * g - 2.0 * M * inv_S2 / S
   # direction -grad for descent
   return grad
```

import numpy as np

# ------------------------------------------------------------
# Helper: Catalan-like series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast convolution (real FFT)
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 upper‑bound (mirrors evaluate_sequence)
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Search for a non‑negative step‑function that improves the AC1 bound.
    Uses a simple simulated‑annealing / hill‑climbing scheme starting from
    either the provided parent `height_sequence_1` or the Catalan series.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and an initial candidate.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        parent = None
        n = 256

    # Catalan series (truncated)
    catalan = _make_series(n)
    best_seq = np.clip(catalan, 0.0, 1000.0)
    best_score = _score(best_seq)

    # Prefer the supplied parent if it is already better.
    if parent is not None:
        cand = np.clip(parent, 0.0, 1000.0)
        cand_score = _score(cand)
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score

    # Current working point for the meta‑search.
    cur_seq = best_seq.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Simulated annealing / hill climbing parameters.
    # ------------------------------------------------------------------
    target = 1.50274349          # the goal to beat
    max_iter = 8000
    temperature = max(0.1 * best_score, 0.001)

    # ------------------------------------------------------------------
    # Helper: perform a cheap projected gradient step (optional).
    # ------------------------------------------------------------------
    def _grad_step(x: np.ndarray) -> np.ndarray:
        """One projected gradient descent step on the bound."""
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        nloc = x.size

        # gradient of the maximal convolution element
        g = np.zeros_like(x)
        lo = max(0, k - (nloc - 1))
        hi = min(k, nloc - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = x[k - idx]
        if (k & 1) == 0:                     # diagonal term when k is even
            mid = k // 2
            if mid < nloc:
                g[mid] += x[mid]

        S = x.sum()
        if S == 0.0:
            return x
        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3
        lr = 0.5
        y = x - lr * grad
        np.clip(y, 0.0, 1000.0, out=y)
        return y

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    for it in range(max_iter):
        temp = temperature * (0.995 ** it)

        # --------------------------------------------------------------
        # Propose a new candidate.
        # --------------------------------------------------------------
        cand = cur_seq.copy()
        move_type = rng.integers(0, 3)

        if move_type == 0:
            # Sparse additive bump on a few random indices.
            max_cnt = max(2, n // 500)               # keep the number tiny
            cnt = rng.integers(1, max_cnt + 1)
            eps = rng.uniform(0.02, 0.20)            # modest bump
            idx = rng.choice(n, size=cnt, replace=False)
            cand[idx] += eps
        elif move_type == 1:
            # One cheap projected gradient descent step.
            cand = _grad_step(cand)
        else:
            # Occasional symmetrisation (helps flatten the convolution).
            if rng.random() < 0.3:
                cand = 0.5 * (cand + cand[::-1])

        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = _score(cand)

        # --------------------------------------------------------------
        # Acceptance rule (simulated annealing).
        # --------------------------------------------------------------
        if cand_score < cur_score or rng.random() < np.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                if best_score < target:
                    break

    # ------------------------------------------------------------
    # Safety net – guarantee a valid non‑empty list.
    # ------------------------------------------------------------
    if best_seq.size == 0 or not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)

    return [float(v) for v in best_seq]
