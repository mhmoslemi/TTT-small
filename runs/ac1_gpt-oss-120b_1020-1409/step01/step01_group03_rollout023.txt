import numpy as np

# ------------------------------------------------------------
# Helper: generate the classic Catalan (binomial) series.
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the length‑`length` truncation of a_i = C(2i,i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast linear convolution via FFT (real‑valued input).
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 upper‑bound (mirrors the evaluator).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)


# ------------------------------------------------------------
# One projected gradient step (objective = 2*n*max(conv)/sum^2).
# ------------------------------------------------------------
def _grad_step(x: np.ndarray, rng: np.random.Generator) -> tuple:
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    k = int(conv.argmax())
    S = x.sum()
    if S == 0.0:
        return x, float('inf')

    # gradient of the max‑convolution term (accounting for the factor 2)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    g = np.zeros_like(x)
    # contribution from the pair (i, k-i)
    g[idx] = x[k - idx]

    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

    # line‑search on a few learning rates
    lr = 1.0
    best_local = x
    best_local_score = _score(x)
    for _ in range(5):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)

        # occasional symmetry (helps flatten the autocorrelation)
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        sc = _score(cand)
        if sc < best_local_score:
            best_local = cand
            best_local_score = sc
            break
        lr *= 0.5
    return best_local, best_local_score


# ------------------------------------------------------------
# Random block perturbation (multiplicative jitter).
# ------------------------------------------------------------
def _perturb_block(x: np.ndarray, rng: np.random.Generator) -> tuple:
    n = x.shape[0]
    width = max(1, n // rng.integers(12, 33))
    start = rng.integers(0, max(1, n - width + 1))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.13))
    cand = x.copy()
    cand[start:stop] *= factor
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand, _score(cand)

# ------------------------------------------------------------

    """
    Return a list of non‑negative heights that (hopefully) beats the
    previous π/2 bound.  The routine starts from the Catalan series,
    optionally switches to a provided parent construction, then performs
    a long projected‑gradient descent combined with random block jitter.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        parent = np.clip(parent, 0.0, 1000.0)
        parent_score = _score(parent)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # Optimization loop.
    # ------------------------------------------------------------------
    x = best.copy()
    stagnant = 0         # number of successive steps without improvement
    max_stagnant = 1200  # generous patience

    for step in range(num_steps):
        # projected gradient move
        x_new, sc_new = _grad_step(x, rng)

    # Fallback guard – ensure a non‑empty, finite list.
    # ------------------------------------------------------------

    return [float(v) for v in best]
