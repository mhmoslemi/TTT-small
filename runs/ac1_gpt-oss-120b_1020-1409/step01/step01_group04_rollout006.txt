import numpy as np

# The evaluator is provided by the harness as `evaluate_sequence`.
# It must be called with a plain Python list of floats.
def _score(seq):
    """Return the AC1 upper‑bound score for a sequence."""
    return float(evaluate_sequence(seq))

def propose_candidate(seed=42,
                     num_steps=800,
                     block_fraction_min=0.005,
                     block_fraction_max=0.04,
                     sigma=0.09,
                     **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound
    is as low as possible.

    The routine starts from the supplied parent ``height_sequence_1`` (if
    present) and performs a very lightweight stochastic hill‑climbing.
    All temporary storage is O(n) and no dense n×n matrix is ever built.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Acquire a base sequence and its length.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is injected by the harness when a parent exists.
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:                     # no parent available
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Clip to the allowed range and compute the initial score.
    best = np.clip(parent, 0.0, 1000.0)
    best_score = _score(best.tolist())

    # ------------------------------------------------------------------
    # Parameterisation of the random block updates.
    # ------------------------------------------------------------------
    min_width = max(1, int(n * block_fraction_min))
    max_width = max(2, int(n * block_fraction_max))

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing.
    #
    # Two move types are used:
    #   * multiplicative perturbation on a random contiguous block,
    #   * a small mass transfer between two distinct positions.
    # The best‑found sequence is retained throughout the loop.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = best.copy()

        if rng.random() < 0.68:  # block scaling
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor
        else:                     # pairwise transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1            # guarantee i != j
            delta = float(best[i] * rng.random() * 0.15)
            cand[i] -= delta
            cand[j] += delta

        # Enforce feasibility.
        cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate and possibly accept.
        cur_score = _score(cand.tolist())
        if np.isfinite(cur_score) and cur_score < best_score:
            best = cand
            best_score = cur_score

    # ------------------------------------------------------------------
    # Fallback: if the parent was unusable, use a uniform vector.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.full(n, 1.0, dtype=np.float64)
        best_score = _score(best.tolist())

    # Return a plain Python list of floats (the harness expects this).
    return [float(x) for x in best]
