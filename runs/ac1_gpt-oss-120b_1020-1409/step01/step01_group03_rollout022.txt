import numpy as np

# ------------------------------------------------------------
# Convolution (full) via real FFT, reusable for any length.
# ------------------------------------------------------------
def _conv_full(x: np.ndarray) -> np.ndarray:
    """Return full linear convolution of x with itself."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 upper‑bound mirror of evaluate_sequence (works on ndarray).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    m = _conv_full(seq).max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# Analytic starter: a_i = C(2i,i)/4^i  (truncated, then clipped).
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Gradient of the AC1 bound w.r.t. the current sequence.
# ------------------------------------------------------------
def _gradient(x: np.ndarray) -> tuple:
    """Return (grad, max_conv, sum_x) for the current x."""
    n = x.shape[0]
    conv = _conv_full(x)
    M = conv.max()
    k = int(conv.argmax())
    S = x.sum()
    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S

    # subgradient of max‑convolution term (factor 2 already included)
    g = np.zeros_like(x)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    g[idx] = 2.0 * x[k - idx]

    # ∇f = 2·n·(g/S² – 2·M/S³)
    term1 = g * inv_S2
    term2 = 2.0 * M * inv_S3
    grad = 2.0 * n * (term1 - term2)
    return grad, M, S


# ------------------------------------------------------------
# Small smoothing operation (preserves total mass).
# ------------------------------------------------------------
def _smooth(x: np.ndarray, alpha: float = 0.15) -> np.ndarray:
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    sm = np.convolve(x, kernel, mode="same")
    total = sm.sum()
    if total > 0.0:
        sm *= x.sum() / total
    np.clip(sm, 0.0, 1000.0, out=sm)
    return (1.0 - alpha) * x + alpha * sm


# ------------------------------------------------------------
# Random block perturbation (scale a contiguous sub‑array).
# ------------------------------------------------------------
def _perturb_block(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = x.shape[0]
    width = max(1, n // rng.integers(12, 33))
    start = rng.integers(0, max(1, n - width + 1))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.13))
    y = x.copy()
    y[start:stop] *= factor
    # optional cheap symmetrisation
    if rng.random() < 0.2:
        y = 0.5 * (y + y[::-1])
    np.clip(y, 0.0, 1000.0, out=y)
    return y


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    previous bound.  The routine starts from the classic Catalan‑type series,
    optionally replaces it by the provided parent construction, then performs a
    projected sub‑gradient descent with occasional smoothing and block
    perturbations.  Early stopping is triggered after a long stagnation.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size from the optional parent array.
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
    except Exception:
        parent = None
        n = 256

    # --------------------------------------------------------
    # Initial candidate: analytic series (then clipped).
    # --------------------------------------------------------
    best = _make_series(n)
    best_score = _score(best)

    # Use parent if it is already better.
    if parent is not None:
        parent = np.clip(parent, 0.0, 1000.0)
        pscore = _score(parent)
        if pscore < best_score:
            best = parent.copy()
            best_score = pscore

    # --------------------------------------------------------
    # Optimisation loop.
    # --------------------------------------------------------
    x = best.copy()
    stagn = 0                     # consecutive non‑improving steps
    for step in range(num_steps):
        # --- gradient step -------------------------------------------------
        grad, M, S = _gradient(x)

        # simple back‑tracking line‑search
        best_local = x
        best_local_score = _score(x)
        for lr in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = x - lr * grad
            np.clip(cand, 0.0, 1000.0, out=cand)
            # occasional cheap symmetrisation
            if rng.random() < 0.15:
                cand = 0.5 * (cand + cand[::-1])
            # occasional smoothing
            if rng.random() < 0.10:
                cand = _smooth(cand, alpha=0.10)
            sc = _score(cand)
            if sc < best_local_score:
                best_local = cand
                best_local_score = sc
                break

        # --- random block perturbation (periodic) -------------------------
        if step % 17 == 0:
            cand2 = _perturb_block(best_local, rng)
            sc2 = _score(cand2)
            if sc2 < best_local_score:
                best_local = cand2
                best_local_score = sc2

        # --- accept globally if improved -----------------------------------
        if best_local_score + 1e-14 < best_score:
            best = best_local.copy()
            best_score = best_local_score
            stagn = 0
        else:
            stagn += 1

        x = best_local

        # --- early termination ------------------------------------------------
        if stagn > 500:
            break

    # --------------------------------------------------------
    # Final polishing: a few deterministic smoothing trials.
    # --------------------------------------------------------
    kernels = [
        np.array([0.5], dtype=np.float64),
        np.array([0.25, 0.5, 0.25], dtype=np.float64),
        np.array([0.2, 0.6, 0.2], dtype=np.float64),
    ]
    for ker in kernels:
        sm = np.convolve(best, ker, mode="same")
        total = sm.sum()
        if total > 0.0:
            sm *= best.sum() / total
        np.clip(sm, 0.0, 1000.0, out=sm)
        sc = _score(sm)
        if sc < best_score:
            best = sm
            best_score = sc

    # --------------------------------------------------------
    # Safety net – return a simple fallback if something went wrong.
    # --------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(v) for v in best]
