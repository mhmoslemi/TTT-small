def _arcsine_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 0.5 / i)
    return a

    for cand in seed_candidates:
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score

    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    seq = best_seq.copy()
    sum_a = seq.sum()
    conv = np.convolve(seq, seq)
    curr_score = float(2.0 * n * conv.max() / (sum_a * sum_a))

    # main loop
    for step in range(num_steps):
        idx_max = int(conv.argmax())
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        # contributions for each i in range
        if idx_range.size > 0:
            contrib = seq[idx_range] * seq[idx_max - idx_range]
            i = int(idx_range[np.argmax(contrib)])
            # ensure i has positive mass
            if seq[i] <= 1e-12:
                i = int(rng.integers(lo, hi + 1))
        else:
            i = int(rng.integers(0, n))

        # choose j outside the contributing range if possible
        if lo > 0:
            j = int(rng.integers(0, lo))
        elif hi < n - 1:
            j = int(rng.integers(hi + 1, n))
        else:
            # fallback random different index
            j = int(rng.integers(0, n))
            if j == i:
                continue

        # limit transfer amount
        delta_max = min(seq[i] * 0.5, 1000.0 - seq[j])
        if delta_max <= 0.0:
            continue
        delta = rng.random() * delta_max
        if delta <= 0.0:
            continue

        # compute new candidate
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # incremental conv update
        new_conv = conv.copy()
        # linear terms
        new_conv[i:i+n] -= 2.0 * delta * seq
        new_conv[j:j+n] += 2.0 * delta * seq
        # quadratic terms
        new_conv[2 * i] += delta * delta
        new_conv[2 * j] += delta * delta
        new_conv[i + j] -= 2.0 * delta * delta

        new_max = new_conv.max()
        new_score = float(2.0 * n * new_max / (sum_a * sum_a))

        # update best
        if new_score < best_score:
            best_seq = new_seq.copy()
            best_score = new_score

        # accept or reject for current state
        if new_score <= curr_score:
            seq = new_seq
            conv = new_conv
            curr_score = new_score
        else:
            # simulated annealing
            temperature = 1.0 / (step + 1.0)
            if np.exp((curr_score - new_score) / temperature) > rng.random():
                seq = new_seq
                conv = new_conv
                curr_score = new_score
        # occasional symmetrization
        if step % 500 == 0 and rng.random() < 0.5:
            sym_seq = 0.5 * (seq + seq[::-1])
            # recompute conv and score if beneficial
            sym_conv = np.convolve(sym_seq, sym_seq)
            sym_score = float(2.0 * n * sym_conv.max() / (sum_a * sum_a))
            if sym_score < best_score:
                best_seq = sym_seq.copy()
                best_score = sym_score
            if sym_score < curr_score:
                seq = sym_seq
                conv = sym_conv
                curr_score = sym_score

        # occasional block perturbation to escape stagnation
        if step % 1000 == 0 and rng.random() < 0.15:
            # random block multiplication
            width = int(rng.integers(max(1, n // 80), max(2, n // 6)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            block_seq = seq.copy()
            block_seq[start:start+width] *= factor
            np.clip(block_seq, 0.0, 1000.0, out=block_seq)
            block_sum = block_seq.sum()
            block_conv = np.convolve(block_seq, block_seq)
            block_score = float(2.0 * n * block_conv.max() / (block_sum * block_sum))
            if block_score < best_score:
                best_seq = block_seq.copy()
                best_score = block_score
            if block_score < curr_score or rng.random() < 0.3:
                seq = block_seq
                conv = block_conv
                sum_a = block_sum
                curr_score = block_score

    # final fallback
    if best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_seq]

# ----------------------------------------------------------------------
# Helper: arcsine‑type sequence (good analytic seed)
# ----------------------------------------------------------------------
def _arcsine_sequence(n: int) -> np.ndarray:
    """Generate a decreasing positive sequence a_i = a_{i-1}*(1-0.5/i)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a
