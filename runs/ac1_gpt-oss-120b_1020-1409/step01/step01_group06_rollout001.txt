def _perturb(vec, rng, target_sum, sigma, mirror_prob=0.3, swap_prob=0.2):
    n = vec.size
    cand = vec.copy()
    # block width
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))  # using same range
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:start+width] *= factor
    if rng.random() < mirror_prob:
        cand = 0.5 * (cand + cand[::-1])
    if rng.random() < swap_prob:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        cand[i], cand[j] = cand[j], cand[i]
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0:
        cand *= target_sum / s
    return cand

import numpy as np

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full convolution arr * arr using FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _score(arr: np.ndarray, n: int) -> float:
    """Exact evaluator score (without calling evaluate_sequence)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """Generate a non‑negative sequence a_i ∝ (i+β)^(-p). Symmetrised if requested."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


# ----------------------------------------------------------------------
def _perturb(vec: np.ndarray,
             rng: np.random.Generator,
             target_sum: float,
             sigma: float,
             mirror_prob: float = 0.3,
             swap_prob: float = 0.2) -> np.ndarray:
    """Return a copy of `vec` with a random block multiplied by exp(N(0, sigma)).
    Optionally mirror and swap two entries, then renormalise to `target_sum`."""
    n = vec.size
    cand = vec.copy()

    # random block
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:start + width] *= factor

    # optional mirroring
    if rng.random() < mirror_prob:
        cand = 0.5 * (cand + cand[::-1])

    # optional swap of two positions
    if rng.random() < swap_prob:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        cand[i], cand[j] = cand[j], cand[i]

    # clip and renormalise
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand
