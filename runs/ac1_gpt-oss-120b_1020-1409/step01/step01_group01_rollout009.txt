import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the r‑sequence r_i = C(2i,i)/4^i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def _fast_conv(arr: np.ndarray, size: int) -> np.ndarray:
    """Full convolution of arr with itself using FFT, size is a power of two."""
    fx = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * arr.size - 1]
    return conv


def _bound_fast(arr: np.ndarray, size: int) -> float:
    """Fast evaluation of the AC1 bound (same formula as evaluate_sequence)."""
    conv = _fast_conv(arr, size)
    max_b = float(conv.max())
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    n = arr.size
    return 2.0 * n * max_b / (s * s)
