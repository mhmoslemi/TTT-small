import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Generate the r‑sequence: r_i = C(2i,i)/4^i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose
    AC1 upper bound is as low as possible.
    """
    # ------------------------------------------------------------------
    # 1. Determine length and initialise the sequence.
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_r_sequence(n)

    a = np.clip(a, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 2. Helper to evaluate the current bound.
    # ------------------------------------------------------------------
    def bound(arr: np.ndarray) -> float:
        return float(evaluate_sequence(arr.tolist()))

    best_a = a.copy()
    best_score = bound(best_a)

    # ------------------------------------------------------------------
    # 3. Greedy slack‑filling passes.
    #
    #    For each epsilon we repeatedly add at most `eps` to every
    #    coordinate that can absorb it without raising the current
    #    convolution maximum `M`.  After each batch we recompute the
    #    convolution, the new slack and possibly accept the update.
    # ------------------------------------------------------------------
    epsilons = [
        0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05
    ]

    for eps in epsilons:
        # keep trying the same eps as long as we can make progress
        while True:
            # current convolution and its maximum
            conv = np.convolve(a, a)
            M = float(conv.max())
            a0 = a[0] if a[0] != 0 else 1.0  # a0 stays positive (r_0 = 1)

            # slack_i = amount we may increase a[i] without exceeding M
            # (only the first n entries of conv correspond to shifts i)
            slack = (M - conv[:n]) / (2.0 * a0)
            # keep only non‑negative slack
            slack = np.where(slack > 0, slack, 0.0)

            # delta_i = min(eps, slack_i); never touch a[0]
            delta = np.minimum(eps, slack)
            delta[0] = 0.0

            if not np.any(delta > 0):
                # nothing can be increased with this epsilon
                break

            candidate = a + delta
            # enforce the global bounds required by the evaluator
            np.clip(candidate, 0.0, 1000.0, out=candidate)

            cand_score = bound(candidate)
            if np.isfinite(cand_score) and cand_score < best_score:
                # improvement – accept
                a = candidate
                best_a = a.copy()
                best_score = cand_score
                # try to add the same epsilon again (more slack may remain)
                continue
            else:
                # no improvement possible with this batch
                break

    # ------------------------------------------------------------------
    # 4. Safety fallback – should never be needed.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best_a = np.ones(n, dtype=np.float64)

    # Return a plain Python list of floats as required.
    return [float(x) for x in best_a]
