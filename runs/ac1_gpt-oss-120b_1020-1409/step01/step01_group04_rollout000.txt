import numpy as np

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n and parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best = None
    best_score = np.inf

    # Helper to evaluate and update best
    def try_update(seq):
        nonlocal best, best_score
        seq = np.clip(seq, 0.0, 1000.0)   # ensure bounds
        sc = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = seq.copy()
        # return sc maybe
        return sc

    # Phase 1: scan families
    # compute indices array once
    idx = np.arange(n, dtype=np.float64)

    # power law family
    p_vals = np.linspace(0.2, 3.0, 40)
    for p in p_vals:
        cand = (idx + 1) ** (-p)
        try_update(cand)

    # exponential family
    lam_vals = np.linspace(0.0005, 0.02, 30)
    for lam in lam_vals:
        cand = np.exp(-lam * idx)
        try_update(cand)

    # linear decreasing (triangular) family
    for slope in np.linspace(0.0, 1.0, 25):
        cand = np.maximum(0.0, 1.0 - slope * idx)
        try_update(cand)

    # Use parent as baseline if not yet set or improvement
    if best is None:
        best = np.clip(parent, 0.0, 1000.0)
        best_score = float(evaluate_sequence(best.tolist()))

    # Phase 2: local search with simulated annealing
    # parameters
    num_iter = 2000
    init_temp = 0.02   # adjust scale of score
    decay = 0.995

    for it in range(num_iter):
        temp = init_temp * (decay ** it)
        cand = best.copy()
        # choose perturbation
        if rng.random() < 0.6:
            # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 20) + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand[start:start+width] *= factor
        else:
            # pairwise transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n-1))
            if j >= i:
                j += 1
            delta = best[i] * rng.random() * 0.3
            cand[i] -= delta
            cand[j] += delta

        # optional small symmetry enforcement
        if rng.random() < 0.05:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))

        if np.isfinite(sc):
            # accept if better or with prob
            if sc < best_score:
                best = cand
                best_score = sc
            else:
                # Metropolis
                if rng.random() < np.exp(-(sc - best_score) / max(temp, 1e-12)):
                    best = cand
        # else ignore

    # final sanity: ensure best valid
    if best is None or not np.isfinite(best_score):
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step function (list of floats) that aims to
    minimise the AC1 bound evaluated by `evaluate_sequence`.
    The routine uses a quick parametric scan followed by a simple
    simulated‑annealing style local search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and obtain a fallback parent sequence.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Initialise best with the parent (guaranteed feasible).
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Phase 1 – parametric scan of simple families.
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # Power‑law a_i = (i+1)^{-p}
    for p in np.linspace(0.2, 3.0, 40):
        cand = (idx + 1.0) ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # Exponential a_i = exp(-lambda * i)
    for lam in np.linspace(5e-4, 2e-2, 30):
        cand = np.exp(-lam * idx)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # Linear decreasing (triangular) a_i = max(0, 1 - s*i)
    for s in np.linspace(0.0, 1.0 / max(1, n - 1), 25):
        cand = np.maximum(0.0, 1.0 - s * idx)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # Phase 2 – local search (simulated annealing style).
    # ------------------------------------------------------------------
    num_iter = 2000
    init_temp = 0.02
    decay = 0.995

    min_width = max(1, n // 200)   # ≈0.5 % of n
    max_width = max(2, n // 20)    # ≈5 % of n
    sigma = 0.07                   # for log‑normal block factor

    for it in range(num_iter):
        # Temperature schedule (only used for probabilistic acceptance)
        temp = init_temp * (decay ** it)

        cand = best.copy()

        if rng.random() < 0.6:
            # Multiplicative block perturbation
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor
        else:

            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:

            delta = best[i] * rng.random() * 0.3
            cand[i] -= delta

        if rng.random() < 0.05:
