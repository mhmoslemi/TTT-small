import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return the provided parent sequence if valid, else a safe fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:
        pass
    # fallback – a short uniform vector that always satisfies the evaluator
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """Generate the arcsine (central binomial) coefficients."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _block_perturbation(base, rng):
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    # width ≈ [n/200 , n/12]
    lo_w = max(1, n // 200)
    hi_w = max(2, n // 12)
    width = int(rng.integers(lo_w, hi_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _conv_fft(seq, fft_len):
    """Linear convolution via FFT (real‑to‑real)."""
    # rfft works on real input and returns the positive‑frequency part
    ft = np.fft.rfft(seq, fft_len)
    prod = ft * ft
    conv = np.fft.irfft(prod, fft_len)
    # linear convolution length is 2*n-1, we keep the leading part
    return conv[: 2 * seq.size - 1]


def _score(seq, fft_len):
    """Compute the AC1 upper‑bound for a candidate sequence."""
    n = seq.size
    conv = _conv_fft(seq, fft_len)
    max_b = float(conv.max())
    s = float(seq.sum())
    # Protection against division‑by‑zero (should never happen for valid candidates)
    if s <= 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, num_steps=20000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 upper bound.
    The routine starts from a strong analytic seed (the arcsine sequence) and
    performs a large number of inexpensive stochastic adjustments guided by the
    location of the current maximal autocorrelation value.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial candidate selection
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = parent.size

    # Pre‑compute FFT length (next power of two ≥ 2*n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    best = parent.copy()
    best_score = _score(best, fft_len)

    arcsine = _arcsine_sequence(n)
    arcsine_score = _score(arcsine, fft_len)
    if arcsine_score < best_score:
        best = arcsine
        best_score = arcsine_score

    current = best.copy()
    last_improve = 0  # iteration index of last improvement

    # ------------------------------------------------------------------
    # Stochastic optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # Occasionally perform a larger block move to escape plateaus
        if rng.random() < 0.02:
            cand = _block_perturbation(current, rng)
            cand_score = _score(cand, fft_len)
            if cand_score < best_score:
                best, best_score, current = cand, cand_score, cand
                last_improve = step
                continue
            # SA acceptance for the block move
            temp = 1.0 / np.sqrt(step + 1.0)
            if np.exp((best_score - cand_score) / temp) > rng.random():
                current = cand
            continue

        # ----------------------------------------------------------------
        # Normal gradient‑guided mass‑redistribution move
        # ----------------------------------------------------------------
        conv = _conv_fft(current, fft_len)
        max_idx = int(conv.argmax())

        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        partner = max_idx - idx_range
        contrib = current[idx_range] * current[partner]
        total_contrib = contrib.sum()

        # choose source index i biased by its contribution to the maximum
        if total_contrib > 0.0 and rng.random() < 0.9:
            probs = contrib / total_contrib
            i = int(rng.choice(idx_range, p=probs))
        else:
            i = int(rng.integers(0, n))

        # choose a destination index j outside the contributing window, if possible
        if lo == 0 and hi == n - 1:
            # the whole array contributes – pick any other index
            while True:
                j = int(rng.integers(0, n))
                if j != i:
                    break
        else:
            complement = n - (hi - lo + 1)
            k = int(rng.integers(0, complement))
            if k < lo:
                j = k
            else:
                j = k + (hi - lo + 1)

        # Transfer a modest yet flexible fraction of mass from i to j
        if current[i] <= 0.0:
            continue
        max_transfer = min(current[i], 0.5 * current[i] + 1e-12)
        delta = rng.random() * max_transfer
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _score(cand, fft_len)

        # Keep the best ever seen
        if cand_score < best_score:
            best, best_score, current = cand, cand_score, cand
            last_improve = step
            continue

        # Simulated‑annealing style acceptance to allow occasional uphill moves
        temp = 1.0 / np.sqrt(step + 1.0)
        if np.exp((best_score - cand_score) / temp) > rng.random():
            current = cand

        # Early termination if no improvement for a long stretch
        if step - last_improve > 6000:
            break

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
