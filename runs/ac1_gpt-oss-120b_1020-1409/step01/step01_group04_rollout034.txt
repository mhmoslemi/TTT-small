        n = parent.size

    # Initialize best to parent
    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # define update function
    def consider(arr):
        nonlocal best, best_score
        arr = np.clip(arr, 0.0, 1000.0)
        # avoid zero sum (shouldn't happen)
        if np.sum(arr) < 1e-8:
            return
        sc = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = arr
            best_score = sc

    # baseline uniform
    consider(np.ones(n, dtype=np.float64))

    idx = np.arange(n, dtype=np.float64)

    # 1. Power-law family with variable exponent p and alpha
    alphas = np.logspace(-5, 1, num=25, dtype=np.float64)  # 25
    ps = np.linspace(0.2, 1.2, num=11, dtype=np.float64)  # 11
    for p in ps:

            consider(cand)

        consider(cand)

    cand = 1.0 - np.square(idx / (n - 1))
    cand = np.clip(cand, 0.0, None)
    consider(cand)

    # 5. Linear decreasing to zero
    cand = (n - 1 - idx) / (n - 1)
    consider(cand)

    # 6. Random permutations of best few candidates

    base_vectors = []

        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:                     # no parent available
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Helpers
    # ----------------------------------------------------------------------
    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    def consider(arr):
        """Clip, evaluate and keep the candidate if it improves the score."""
        nonlocal best, best_score
        arr = np.clip(arr, 0.0, 1000.0)
        if np.sum(arr) < 1e-8:   # avoid the degenerate zero‑sum case
            return
        sc = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = arr
            best_score = sc

    idx = np.arange(n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 1) Baseline candidates
    # ----------------------------------------------------------------------
    consider(np.ones(n, dtype=np.float64))                # uniform
    consider(parent)                                      # parent (fallback)

    # ----------------------------------------------------------------------
    # 2) Parametric families
    # ----------------------------------------------------------------------
    # Power‑law a_i = (i + alpha)^(-p)
    alphas = np.logspace(-5, 1, num=25, dtype=np.float64)
    ps = np.linspace(0.2, 1.2, num=11, dtype=np.float64)
    for p in ps:
        for alpha in alphas:
            cand = 1.0 / np.power(idx + alpha, p)
            consider(cand)

    # Exponential decay a_i = exp(-beta * i)
    betas = np.logspace(-5, -0.5, num=15, dtype=np.float64)
    for beta in betas:
        cand = np.exp(-beta * idx)
        consider(cand)

    # Half‑sine shape (positive on [0, n))
    cand = np.sin(np.pi * (idx + 0.5) / (2.0 * n))
    consider(cand)

    # Quadratic (parabolic) shape, non‑negative
    cand = 1.0 - np.square(idx / (n - 1))
    consider(np.clip(cand, 0.0, None))

    # Linear decreasing to zero
    cand = (n - 1 - idx) / (n - 1)
    consider(cand)

    # ----------------------------------------------------------------------
    # 3) Random permutations of promising bases
    # ----------------------------------------------------------------------
    base_vectors = [
        1.0 / np.sqrt(idx + 1.0),
        1.0 / np.power(idx + 1.0, 0.8),
        np.exp(-0.001 * idx),
        np.sin(np.pi * (idx + 0.5) / (2.0 * n)),
    ]
    for base in base_vectors:
        for _ in range(7):                     # a few shuffles per base
            perm = base[rng.permutation(n)]
            consider(perm)

    # ----------------------------------------------------------------------
    # 4) Simple hill‑climbing from the best found so far
    # ----------------------------------------------------------------------
    min_width = max(1, n // 200)
    max_width = max(2, n // 20)
    sigma = 0.06                               # log‑normal multiplicative noise
    for _ in range(2000):
        cand = best.copy()
        if rng.random() < 0.65:                # block‑wise scaling
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor
        else:                                   # pairwise mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            delta = best[i] * rng.random() * 0.15
            cand[i] -= delta
            cand[j] += delta
        if rng.random() < 0.07:                # occasional symmetry push
            cand = 0.5 * (cand + cand[::-1])
        consider(cand)

    # ----------------------------------------------------------------------
    # 5) Final sanity check and return
    # ----------------------------------------------------------------------
    if not np.isfinite(best_score) or best.size == 0:
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]
