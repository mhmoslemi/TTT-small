    for it in range(max_iter):

        cand = cur.copy()
        # choose random operation
        op = rng.integers(0,3)
        if op == 0:
            # pointwise multiplier at random index
            i = rng.integers(0, n)
            factor = np.exp(rng.normal(0, 0.2))
            cand[i] *= factor
        elif op == 1:
            # block multiplier
            width = max(1, n // rng.integers(8,30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0, 0.2))
            cand[start:start+width] *= factor
        else:
            # smoothing: convolve with small kernel
            ksize = 3
            kernel = np.array([0.25, 0.5, 0.25])
            cand = np.convolve(cand, kernel, mode='same')
        # enforce nonneg
        np.clip(cand, 0.0, 1000.0, out=cand)
        # enforce symmetry
        cand = (cand + cand[::-1]) * 0.5

        # evaluate
        cand_score = _score(cand)

        # acceptance: simulated annealing
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score - 1e-12:
                best, best_score = cand.copy(), cand_score
        else:
            # maybe accept with small probability if improve over best?

            # temperature schedule decreasing
            temp = max(0.001, (max_iter - it) / max_iter)
            if rng.random() < np.exp(-(cand_score - cur_score) / (temp * best_score)):

        # occasional re-symmetrize and clip again (already symmetrical)

    # final ensure best nonempty
    if best.size == 0 or not np.isfinite(best_score):

def _grad_step(x):
    conv = _conv_self(x)
    M = conv.max()
    # active indices - maybe choose top K conv positions
    K = min(5, conv.size)
    top_idx = np.argpartition(conv, -K)[-K:]
    g = np.zeros_like(x)
    for k in top_idx:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] += x[k - idx]
    g /= K
    S = x.sum()
    inv_S2 = 1.0 / (S * S) if S != 0 else 0.0
    inv_S3 = inv_S2 / S if S != 0 else 0.0
    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3
    # line search
    best_local = x
    best_local_score = _score(x)
    lr = 1.0
    for _ in range(5):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _score(cand)
        if sc < best_local_score:
            best_local = cand
            best_local_score = sc
            break
        lr *= 0.5
    return best_local, best_local_score

import numpy as np

def _conv_self(x):
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv

def _score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)

def _make_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2.0 * i - 1) / (2.0 * i)
    return a

def _grad_step(x, n):
    conv = _conv_self(x)
    M = conv.max()
    # indices achieving max (maybe many)
    max_idx = np.flatnonzero(conv == M)
    # average gradient over all max indices
    g = np.zeros_like(x)
    for k in max_idx:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] += x[k - idx]
    g /= max_idx.size
    S = x.sum()
    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3
    # simple backtracking line search
    best = x
    best_sc = _score(x)
    lr = 1.0
    for _ in range(5):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            break
        lr *= 0.5
    return best, best_sc

        width = max(1, n // rng.integers(10, 30))

import numpy as np

# ------------------------------------------------------------
# Convolution (full) using FFT – linear memory, O(n log n).
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact score used by the evaluator (mirrors evaluate_sequence).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)

# Classic binomial (central) series a_i = C(2i,i)/4^i.
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return a

# One gradient descent step – uses (averaged) active convolution indices.
# ------------------------------------------------------------
def _grad_step(x: np.ndarray, n: int):
    conv = _conv_self(x)
    M = conv.max()
    # all indices attaining the current maximum (often just one)
    max_idx = np.flatnonzero(conv == M)
    # averaged contribution of those indices
    g = np.zeros_like(x)
    for k in max_idx:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] += x[k - idx]
    g /= max_idx.size

    S = x.sum()
    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

    # simple back‑tracking line search
    best = x
    best_sc = _score(x)
    lr = 1.0
    for _ in range(5):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _score(cand)
        if sc < best_sc:
            best, best_sc = cand, sc
            break
        lr *= 0.5
    return best, best_sc
