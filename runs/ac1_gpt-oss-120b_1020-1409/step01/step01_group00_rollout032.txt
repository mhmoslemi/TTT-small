import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _compute_bound(a: np.ndarray, b: np.ndarray) -> float:
    """
    Return the AC1 upper bound for sequence a,
    given its full convolution b = a * a.
    """
    n = a.shape[0]
    max_b = float(b.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _apply_delta(a: np.ndarray, b: np.ndarray, i: int, j: int, d: float) -> None:
    """
    Perform a rank‑2 update of a and its convolution b when a[i] is decreased
    by d and a[j] is increased by d (d > 0, i != j).
    a is modified in‑place, as is b.
    """
    n = a.shape[0]

    # cross‑terms: 2 * (a * Δ)
    #   Δ has -d at i and +d at j
    #   contribution = -2*d * shift(a,i) + 2*d * shift(a,j)
    b[i:i + n] -= 2.0 * d * a
    b[j:j + n] += 2.0 * d * a

    # self‑terms: Δ * Δ
    b[2 * i] += d * d
    b[2 * j] += d * d
    b[i + j] -= 2.0 * d * d

    # update a
    a[i] -= d
    a[j] += d


def _donor_recipient(a: np.ndarray, b: np.ndarray) -> tuple:
    """
    Heuristic to choose a donor index (large impact on current max),
    and a recipient index that does not affect that max.
    Returns (donor, recipient) or (-1, -1) if no valid move.
    """
    n = a.shape[0]
    k_max = int(b.argmax())

    # range of i that contributes to b[k_max]
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)

    if lo > hi:
        return -1, -1

    idx = np.arange(lo, hi + 1, dtype=int)
    partner = k_max - idx
    prod = a[idx] * a[partner]
    donor = int(idx[prod.argmax()])

    # choose a recipient that does NOT appear in the sum for k_max
    if k_max < n - 1:
        recipient = n - 1                     # any index > k_max
    else:
        L = k_max - (n - 1)                    # first index NOT used
        if L == 0:
            recipient = n - 1
        else:
            recipient = L - 1                  # any index < L
    if recipient == donor:
        # fallback to a far opposite side
        recipient = (donor + n // 2) % n
        if recipient == donor:
            return -1, -1
    return donor, recipient


def _random_pair_move(a: np.ndarray, rng: np.random.Generator) -> tuple:
    """
    Produce a small random mass transfer between two distinct indices.
    Returns (i, j, delta) where delta > 0 will be moved from i to j.
    May return (-1, -1, 0) if no move is possible.
    """
    n = a.shape[0]
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j or a[i] <= 0.0:
        return -1, -1, 0.0
    max_transfer = min(a[i], 1000.0 - a[j])
    if max_transfer <= 0.0:
        return -1, -1, 0.0
    delta = max_transfer * float(rng.uniform(0.001, 0.03))
    return i, j, delta
