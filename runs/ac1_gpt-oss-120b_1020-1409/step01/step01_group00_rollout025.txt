import numpy as np

# ---------- convolution utilities ----------
def _conv_max_and_idx(x: np.ndarray, fft_len: int):
    """Return (max convolution value, index of the max) for x*x using FFT."""
    X = np.fft.rfft(x, n=fft_len)
    C = np.fft.irfft(X * X, n=fft_len)
    c = C[: 2 * x.shape[0] - 1]
    idx = int(np.argmax(c))
    return float(c[idx]), idx


def _bound_from_conv(x: np.ndarray, fft_len: int):
    """Bound = 2*n*max(x*x)/(sum(x)^2); also returns the max for reuse."""
    max_val, _ = _conv_max_and_idx(x, fft_len)
    s = x.sum()
    if s < 0.01:
        return np.inf, max_val
    n = x.shape[0]
    bound = 2.0 * n * max_val / (s * s)
    return bound, max_val


# ---------- perturbation helpers ----------
def _biased_pair_transfer(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a small fraction of mass from a low‑index region to a high‑index region.
    Keeps the total sum unchanged and respects the [0,1000] box.
    """
    n = arr.size
    region = max(1, n // 20)                     # ~5 % of the domain
    i = rng.integers(0, region)                 # donor (low index)
    j = rng.integers(n - region, n)             # receiver (high index)
    if arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.05)    # 0.1 % – 5 % of donor
    out = arr.copy()
    out[i] = max(0.0, out[i] - delta)
    out[j] = min(1000.0, out[j] + delta)
    return out


def _edge_block_scale(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:

    Scale a random block situated at one edge (left or right) by a log‑normal factor.
    """

    out[start:stop] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out

            cur_bound, _ = _bound_from_conv(cur, fft_len)

    # ---- final validation with the official scorer ----
    final = best.tolist()
    try:
        _ = float(evaluate_sequence(final))
    except Exception:
        # safe fallback (uniform vector) – never empty and always valid
        uniform = np.ones(n, dtype=np.float64)
        np.clip(uniform, 0.0, 1000.0, out=uniform)
        final = uniform.tolist()
    return final
