import numpy as np
import math

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum of the full linear convolution arr*arr using FFT (exact for real data)."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two >= 2n-1
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _fast_score(arr: np.ndarray) -> float:
    """Fast evaluation of the objective 2*n*max(conv)/(sum(arr)^2)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    n = arr.shape[0]
    return 2.0 * n * max_b / (s * s)

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) aiming to
    minimise the AC1 upper bound used by `evaluate_sequence`.
    The routine:
      * Starts from the provided parent `height_sequence_1` if available,
        otherwise from a uniform vector.
      * Performs a short stochastic search using random Dirichlet
        perturbations, power‑law families and a sinusoidal template,
        always keeping the L1‑norm fixed to sqrt(2·n).
      * Returns the best candidate found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n from the parent if it is a valid 1‑D array.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = int(parent.size)
        start_vec = parent.copy()
    except Exception:
        # Fall back to a uniform vector of length 256 (any positive length works).
        n = 256
        start_vec = np.full(n, 1.0, dtype=np.float64)

    # Normalise the starting vector to the canonical L1‑norm.
    target_sum = math.sqrt(2.0 * n)
    s = start_vec.sum()
    if s == 0.0:
        start_vec = np.full(n, target_sum / n, dtype=np.float64)
    else:
        start_vec *= target_sum / s
    start_vec = np.clip(start_vec, 0.0, 1000.0)

    best_vec = start_vec
    best_score = _fast_score(best_vec)

    # ------------------------------------------------------------------
    # Helper: generate a random positive vector with prescribed sum.
    def _random_dirichlet():
        # Exponential(1) draws (Gamma(1,1)) give a Dirichlet(1,...,1) after normalisation.
        rnd = rng.exponential(1.0, size=n)

        if rnd_sum == 0.0:

        # normalise to target sum
