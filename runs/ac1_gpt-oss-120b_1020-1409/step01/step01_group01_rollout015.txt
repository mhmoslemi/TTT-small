import numpy as np

# ------------------------------------------------------------
# Helper: generate the classic r‑sequence (central binomial /4^i)
# ------------------------------------------------------------
def generate_r_sequence(n: int) -> np.ndarray:
    """Return r_i = C(2i,i)/4^i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose
    AC1 upper bound is lower than the baseline r‑sequence.
    """
    # ------------------------------------------------------------------
    # Determine problem size: use the provided parent if present,
    # otherwise fall back to a modest length.
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256

    # ------------------------------------------------------------------
    # Baseline: the r‑sequence (always feasible)
    # ------------------------------------------------------------------
    baseline = generate_r_sequence(n)
    baseline = np.clip(baseline, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Pre‑compute ordering of indices from smallest to largest.
    # Adding mass to the smallest entries influences the max convolution
    # the least (they are far in the tail of the r‑sequence).
    # ------------------------------------------------------------------
    order_asc = np.argsort(baseline)  # ascending by value

    # ------------------------------------------------------------------
    # Search parameters.
    #   We add a tiny amount `eps` to the `K` smallest entries.
    #   Several (K, eps) pairs are tried; the best candidate is kept.
    # ------------------------------------------------------------------
    # Candidate list of (fraction_of_n, epsilon). Fractions are chosen
    # to give a sensible total added mass (≈2–5).
    candidate_specs = [
        (0.30, 0.001),   # 30% of entries, 1e‑3 each  → Δ≈1.8
        (0.40, 0.001),   # 40% → Δ≈2.4
        (0.50, 0.001),   # 50% → Δ≈3.0
        (0.60, 0.001),   # 60% → Δ≈3.6
        (0.70, 0.001),   # 70% → Δ≈4.2
        (0.80, 0.001),   # 80% → Δ≈4.8
        (0.90, 0.001),   # 90% → Δ≈5.4
        (0.50, 0.002),   # 50% with a slightly larger bump
        (0.30, 0.002),   # 30% with a larger bump
    ]

    best_seq = baseline.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Try each specification, keep the one with the lowest bound.
    # ------------------------------------------------------------------
    for frac, eps in candidate_specs:
        K = int(np.floor(frac * n))
        if K <= 0:
            continue
        cand = baseline.copy()
        # Add `eps` to the `K` smallest entries
        cand[order_asc[:K]] += eps
        np.clip(cand, 0.0, 1000.0, out=cand)
        # Evaluate – the evaluator already enforces sum > 0.01, clipping, etc.
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:   # safety net – any unexpected error
            score = np.inf
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand

    # ------------------------------------------------------------------
    # Very last safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)

    # Return as a plain Python list of floats
    return [float(x) for x in best_seq]
