- rng = np.random.default_rng(seed)

- Determine if parent exists: try reading global height_sequence_1.

- If parent_ok, shape parent_seq by clip 0-1000, sum normalized to 1.

- Initialize a = same as either parent or power law.

- Loop for num_iters:

    conv = np.convolve(a, a)
    # compute smoothing weights
    maxc = conv.max()
    # subtract max for stability
    exp_weights = np.exp(alpha * (conv - maxc))
    weight_sum = exp_weights.sum()
    if weight_sum == 0 or not np.isfinite(weight_sum):
        # fallback to current max index
        # compute gradient for max index
        k = int(np.argmax(conv))
        # compute g as before
        g = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high+1)
        partner = k - idx
        g[idx] = 2.0 * a[partner]
    else:
        w = exp_weights / weight_sum
        # compute gradient via convolution
        rev_a = a[::-1]
        conv_rev = np.convolve(w, rev_a)
        # select middle slice
        g = 2.0 * conv_rev[n-1 : n-1 + n]
    # step size decays
    step = step0 * (0.995 ** it)
    # update multiplicatively
    a = a * np.exp(-step * g)
    # clip no need because exponent ensures positivity but may become too large; clip to 1000
    np.clip(a, 0.0, 1000.0, out=a)
    # renormalize sum to 1
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        a = np.ones(n) / n
    else:
        a = a / s

    # occasionally do random block perturbation:
    if (it % 100 == 0) and rng.random() < 0.3:
        # choose block
        width = int(rng.integers(max(1, n//200), max(2, n//10)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.2)))
        a_pert = a.copy()
        a_pert[start:start+width] *= factor
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        a_pert = a_pert / a_pert.sum()
        cur_score = float(evaluate_sequence(a_pert.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best = a_pert.copy()
            best_score = cur_score
            a = a_pert
    # evaluate current candidate sometimes
    if (it % 10 == 0):
        cur_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best = a.copy()
            best_score = cur_score

    return float(2 * n * np.max(conv) / (s * s))
