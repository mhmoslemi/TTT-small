import numpy as np

def _generate_binomial_sequence(length):
    a = np.empty(length, dtype=np.float64)
    a[0]=1.0
    for i in range(1,length):
        i_f = float(i)
        a[i] = a[i-1] * ((2.0*i_f-1.0)*(2.0*i_f)) / (4.0*i_f*i_f)
    return a

def _compute_max_shift(arr, Lfft, n):

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..length-1, a discrete approximation of 1/sqrt(i)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _fft_max_shift(arr: np.ndarray, Lfft: int, n: int) -> int:
    """Return index k where (arr*arr)[k] is maximal (linear convolution)."""
    f = np.fft.rfft(arr, n=Lfft)
    conv = np.fft.irfft(f * f, n=Lfft)
    conv = conv[: 2 * n - 1]          # keep the linear‑convolution part
    return int(np.argmax(conv))


def _choose_recipient(arr: np.ndarray, max_idx: int, rng: np.random.Generator) -> int:
    """Pick an index that contributes little (or not at all) to the max shift."""
    n = arr.shape[0]
    # try a few random indices; accept if they are outside the active range
    # or if the product a[j]*a[max_idx-j] is tiny.
    for _ in range(10):
        j = int(rng.integers(0, n))
        partner = max_idx - j
        if partner < 0 or partner >= n:
            return j
        if arr[j] * arr[partner] < arr.max() * 0.01:
            return j
    # fallback: any random index
    return int(rng.integers(0, n))
