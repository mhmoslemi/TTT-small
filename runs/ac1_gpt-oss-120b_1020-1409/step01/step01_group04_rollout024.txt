def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step sequence that (hopefully) improves the
    AC1 upper bound.  The routine works in three stages:
      1. Scan simple parametric families (exponential and power‑law),
         optionally with a random permutation to break structure.
      2. Perform cheap stochastic hill climbing starting from the best
         candidate found.
      3. Return the best sequence as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size from the provided parent (height_sequence_1).
    # ------------------------------------------------------------------
    parent_arr = globals().get("height_sequence_1")
    if isinstance(parent_arr, np.ndarray) and parent_arr.ndim == 1 and parent_arr.size > 0:
        n = parent_arr.size
        parent = np.clip(parent_arr.astype(np.float64), 0.0, 1000.0)
    else:                                   # fallback in case no parent is present
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    target_score = 1.50274349  # the target we want to beat

    def try_update(candidate):

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score

    idx = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    betas = np.logspace(np.log10(5e-4), np.log10(2e-3), num=15, dtype=np.float64)
    for beta in betas:
        base = np.exp(-beta * idx)
        try_update(base)
        try_update(base[rng.permutation(n)])
        if best_score <= target_score:

    # ------------------------------------------------------------------
    # 2) Scan power‑law families a_i = 1/(i + alpha)^p.
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    max_width = max(2, n // 20)       # up to ~5 %
    sigma = 0.07                      # std‑dev for multiplicative noise

    for _ in range(num_steps):
        cand = best.copy()

        if rng.random() < 0.6:                     # block multiplicative perturbation
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor

            cand[j] += delta
