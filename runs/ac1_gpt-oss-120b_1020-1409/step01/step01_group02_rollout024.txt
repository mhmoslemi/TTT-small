def compute_score(seq):
    n = seq.shape[0]
    # compute convolution via FFT:
    m = 1
    while m < 2*n - 1:
        m <<= 1
    # Use rfft on seq and compute.
    F = np.fft.rfft(seq, m)
    conv = np.fft.irfft(F * F, m)[:2*n-1]
    max_c = np.max(conv)
    sum_seq = np.sum(seq)
    if sum_seq < 0.01:
        return np.inf
    # constant factor 2*n
    return 2.0 * n * max_c / (sum_seq * sum_seq)
