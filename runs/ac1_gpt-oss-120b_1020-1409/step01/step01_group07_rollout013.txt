import numpy as np

# Target bound – the loop can stop early if reached.
_TARGET_BOUND = 1.50274349


def _block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    max_width = max(1, n // 100)            # at most 1 % of the length
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.03)))   # ≈3 % log‑normal change
    cand = seq.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    tot = cand.sum()
    if tot > 0.0:
        cand /= tot
    return cand


def _swap_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two random positions (order‑only change)."""
    n = seq.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return seq
    cand = seq.copy()
    cand[i], cand[j] = cand[j], cand[i]
    # total sum unchanged → no renormalisation needed
    return cand


def _max_pair_indices(seq: np.ndarray, conv: np.ndarray):
    """Indices i,j (i≤j) that realise the maximal product contributing to the
    current maximal convolution entry."""
    n = seq.size
    k_max = int(np.argmax(conv))
    lag = k_max - (n - 1)                    # signed lag corresponding to conv index
    if lag >= 0:
        # product seq[i] * seq[i+lag]
        prod = seq[: n - lag] * seq[lag:]
        i = int(np.argmax(prod))
        return i, i + lag
    else:
        lag2 = -lag
        prod = seq[lag2:] * seq[: n - lag2]
        i = int(np.argmax(prod))
        return i + lag2, i


def _targeted_pair_perturb(seq: np.ndarray, conv: np.ndarray,
                           rng: np.random.Generator) -> np.ndarray:
    """Reduce the two entries that cause the current maximal convolution value."""
    i, j = _max_pair_indices(seq, conv)
    # Mostly decrease (90 % chance); occasionally allow a small increase to escape plateaus.
    if rng.random() < 0.9:
        factor = rng.uniform(0.94, 0.99)
    else:
        factor = rng.uniform(1.01, 1.06)
    cand = seq.copy()
    cand[i] *= factor
    cand[j] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    tot = cand.sum()
    if tot > 0.0:
        cand /= tot
    return cand


def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Initialise from the supplied parent if available
    # ------------------------------------------------------------------
    try:
        seq = np.array(height_sequence_1, dtype=np.float64).copy()
        n = seq.size
    except Exception:
        # Fallback: a mild inverse‑sqrt shape, normalised to unit sum.
        n = 256
        idx = np.arange(n, dtype=np.float64)
        seq = 1.0 / np.sqrt(idx + 1.0)
        s = seq.sum()
        if s > 0.0:
            seq /= s

    # Ensure the evaluator’s simple bounds.
    np.clip(seq, 0.0, 1000.0, out=seq)
    tot = seq.sum()
    if tot > 0.0:
        seq /= tot

    best = seq
    best_score = float(evaluate_sequence(best.tolist()))

    # If the parent is somehow rejected, fall back to uniform.
    if not np.isfinite(best_score):
        best = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # Pre‑compute the convolution of the current best – needed for the
    # targeted‑pair move.
    conv_best = np.convolve(best, best)

    # ------------------------------------------------------------------
    # 2️⃣  Stochastic local optimisation
    # ------------------------------------------------------------------
    STEPS = 1500
    for _ in range(STEPS):
        # Early exit if we have already crossed the target.
        if best_score < _TARGET_BOUND:
            break

        r = rng.random()
        if r < 0.4:
            cand = _block_perturb(best, rng)
        elif r < 0.8:
            cand = _targeted_pair_perturb(best, conv_best, rng)
        else:
            cand = _swap_perturb(best, rng)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            conv_best = np.convolve(best, best)

    # ------------------------------------------------------------------
    # 3️⃣  Defensive finalisation
    # ------------------------------------------------------------------
    result = best.tolist()
    if not result or any(not np.isfinite(x) for x in result):
        result = [1.0 / n] * n
    return result
