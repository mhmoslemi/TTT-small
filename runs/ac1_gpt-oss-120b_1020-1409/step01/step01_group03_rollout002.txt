# ------------------------------------------------------------
# Helper to generate the classic a_i = C(2i,i)/4^i series
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the truncation of a_i = (2i choose i)/4^i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast convolution of a real vector with itself via FFT
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Fast evaluation of the AC1 bound (same formula as evaluator)
# ------------------------------------------------------------
def _fast_score(x: np.ndarray) -> float:
    """Compute 2*n*max(conv)/sum^2 for a non‑negative vector."""
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_self(x)
    M = conv.max()
    return 2.0 * x.shape[0] * M / (s * s)

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 1500, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    AC1 upper bound.  The routine starts from the available parent
    `height_sequence_1` (if present) or from the analytic series, then
    performs a matrix‑free projected Adam optimisation with occasional
    random block perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size and initial vector
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
        init = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        init = _make_series(n)

    # --------------------------------------------------------
    # Keep best candidate seen so far
    # --------------------------------------------------------
    best = init.copy()
    best_score = _fast_score(best)

    # Also test the pure analytic series (it can be better than a bad parent)
    series = _make_series(n)
    series_score = _fast_score(series)
    if series_score < best_score:
        best = series.copy()
        best_score = series_score

    # --------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------
    x = best.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5               # base learning rate
    t = 0
    stagn = 0
    max_stagn = 250

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    for step in range(num_steps):
        # ----- convolution and max index(es) -----
        conv = _conv_self(x)
        M = conv.max()
        cur_score = 2.0 * n * M / (x.sum() ** 2)

        # ----- indices where convolution is (nearly) maximal -----
        # small tolerance to capture ties;
        # if none found (numerical oddities) fall back to the global argmax
        tol = 1e-8
        near = np.where(conv >= M * (1.0 - tol))[0]
        if near.size == 0:
            near = np.array([int(conv.argmax())])

        # ----- sub‑gradient of M w.r.t. x -----
        g = np.zeros_like(x)
        for k in near:
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            idx = np.arange(i_low, i_high + 1, dtype=int)
            g[idx] += x[k - idx]

        S = x.sum()
        if S == 0.0:      # defensive, should never happen
            break
        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

        # ----- Adam update (projected) -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = x - step_vec
        np.clip(cand, 0.0, 1000.0, out=cand)

        # occasional symmetrisation (helps flatten the autocorrelation)
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])

        cand_score = _fast_score(cand)

        # ----- Keep the best overall candidate -----
        if cand_score + 1e-14 < best_score:
            best = cand.copy()
            best_score = cand_score
            stagn = 0
        else:
            stagn += 1

        # ----- Accept the candidate for the next iteration if it improves locally -----
        if cand_score < cur_score:
            x = cand
        # else keep current x (moments stay as they are)

        # ----- Random block perturbation every few iterations -----
        if step % 33 == 0:
            block_width = max(1, n // rng.integers(8, 20))
            start = rng.integers(0, n - block_width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            block = x.copy()
            block[start:start + block_width] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            if rng.random() < 0.2:
                block = 0.5 * (block + block[::-1])
            block_score = _fast_score(block)

            if block_score + 1e-14 < best_score:
                best = block.copy()
                best_score = block_score
                stagn = 0

            if block_score < cur_score:
                x = block

        # ----- Adapt learning rate if we stall for a while -----
        if stagn > max_stagn:
            lr *= 0.5
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)

        # (optional) early exit if we already beat a strong target
        if best_score < 1.45:
            break

    # --------------------------------------------------------
    # Fallback safety net
    # --------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        # a simple uniform vector with the same L1 norm as the analytic series
        fallback = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)
        best = np.clip(fallback, 0.0, 1000.0)

    # Return as a plain Python list of floats
    return [float(v) for v in best]
