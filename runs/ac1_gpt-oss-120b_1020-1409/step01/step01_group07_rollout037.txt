import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _inverse_sqrt_initial(n):
    """
    Build the classic 1/sqrt(i+1) shape and normalise to sum == 1.
    """
    seq = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 1.0)
    total = seq.sum()
    if total == 0.0:
        seq = np.ones(n, dtype=np.float64)
        total = n
    seq /= total
    return seq


def _uniform_initial(n):
    """
    Uniform non‑negative step function, sum == 1.
    """
    return np.full(n, 1.0 / n, dtype=np.float64)


def _perturb_block(seq, rng, sigma=0.03, max_width_ratio=0.01):
    """
    Multiplicative block perturbation keeping the total sum ≈ 1.
    """
    n = seq.size
    max_width = max(1, int(n * max_width_ratio))
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, sigma)))  # log‑normal factor ~3 % std.
    cand = seq.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    total = cand.sum()
    if total > 0.0:
        cand /= total
    return cand
