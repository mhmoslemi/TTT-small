def greedy_lp_max_sum(f, M):
    n = len(f)
    cap = np.full(2*n - 1, M, dtype=np.float64)
    g = np.zeros_like(f)
    order = np.argsort(f)  # ascending
    for i in order:
        # slice of cap corresponding to constraints that g_i appears in
        start = i
        end = i + n
        cap_slice = cap[start:end]
        # compute bound = cap_slice / f, but avoid division by zero
        # where f == 0, bound infinite.
        with np.errstate(divide='ignore', invalid='ignore'):
            bound = np.where(f > 0, cap_slice / f, np.inf)
        # maximum feasible amount for g_i
        g_i = bound.min()
        if g_i <= 0 or np.isinf(g_i):
            continue
        g[i] = g_i
        # update cap
        cap[start:end] = cap_slice - f * g_i
        # ensure caps >= 0 (clamp)
        # But due to numerical, could become slightly negative; clip.
        np.maximum(cap[start:end], 0, out=cap[start:end])
    return g

import numpy as np

def generate_r_sequence(n):
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2*i - 1) / (2*i)
        seq[i] = r
    return seq

def greedy_feasible_g(f, M):
    n = f.shape[0]
    cap = np.full(2*n - 1, M, dtype=np.float64)
    g = np.zeros_like(f)
    order = np.argsort(f)  # ascending values; f > 0
    # precompute f_nonzero mask to avoid division by zero
    for i in order:
        start = i
        end = i + n
        cap_slice = cap[start:end]
        # compute bound per constraint
        # it's safe because f positive everywhere
        bound = cap_slice / f  # broadcasting: f is (n,) shape
        # bound could be negative due to cap < 0 from numerical issues, clip
        bound = np.where(bound >= 0, bound, np.inf)
        g_i = bound.min()
        if g_i <= 0 or not np.isfinite(g_i):
            continue
        g[i] = g_i
        # update caps: cap_slice -= f * g_i
        cap_slice -= f * g_i
        # ensure no negative due to rounding
        np.maximum(cap_slice, 0, out=cap_slice)
    return g

        # construct feasible g

        if sum_g == 0:

import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the classical r‑sequence (Catalan ratios)
# ----------------------------------------------------------------------
def _r_sequence(n: int) -> np.ndarray:
    """Return r_i = C(2i,i)/4^i for i=0..n‑1 (non‑negative, decreasing)."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


# ----------------------------------------------------------------------
# Greedy construction of a feasible g such that (f * g)_k <= M for all k.
# This solves (approximately) the LP: maximize sum(g) subject to
#   f * g <= M ,  g >= 0.
# ----------------------------------------------------------------------
def _greedy_feasible_g(f: np.ndarray, M: float) -> np.ndarray:
    n = f.shape[0]
    # capacity of each convolution bin; start with the uniform bound M
    cap = np.full(2 * n - 1, M, dtype=np.float64)
    g = np.zeros_like(f)

    # Process indices with smallest f first (they allow the largest g)
    order = np.argsort(f)  # ascending
    # avoid division by zero – treat zeros as “no restriction” (inf)
    safe_f = f.copy()
    safe_f[safe_f <= 0] = np.inf

    for i in order:
        start = i
        end = i + n
        cap_slice = cap[start:end]               # view, length n

        # maximal amount we could add at position i without breaking any bound
        # bound_j = cap_slice[j] / f[j]; if f[j]=0 -> inf (no restriction)
        bounds = cap_slice / safe_f               # element‑wise, length n
        g_i = bounds.min()                        # smallest bound governs
        if not np.isfinite(g_i) or g_i <= 0.0:
            continue

        g[i] = g_i
        # update the capacities for the affected bins
        cap_slice -= f * g_i
        # numerical safety – capacities must stay non‑negative
        np.maximum(cap_slice, 0.0, out=cap_slice)

    return g
