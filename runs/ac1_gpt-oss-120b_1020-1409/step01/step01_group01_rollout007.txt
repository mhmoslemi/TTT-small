import numpy as np

def generate_r_sequence(n):
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2*i - 1) / (2*i)
        seq[i] = r
    return seq

def propose_candidate(seed=42, **kwargs):
    parent = globals().get('height_sequence_1')
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        seq = parent.copy()
    else:
        # fallback length 256 if not provided
        n = 256
        seq = generate_r_sequence(n)
    # ensure within bounds
    np.clip(seq, 0.0, 1000.0, out=seq)

    # keep best
    best = seq
    best_score = float(evaluate_sequence(best.tolist()))

    # try tiny local improvements around the centre (only if length manageable)
    n = best.shape[0]
    # limit attempts for large n to avoid heavy compute
    if n <= 512:
        centre = n // 2
        epsilons = [0.02, 0.01, 0.005, 0.002, 0.001]
        for eps in epsilons:
            # expand a symmetric block around centre
            radius = 0
            while True:
                start = max(0, centre - radius)
                end = min(n, centre + radius + 1)
                candidate = best.copy()
                candidate[start:end] += eps
                np.clip(candidate, 0.0, 1000.0, out=candidate)
                score = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = candidate
                    best_score = score
                    radius += 1
                    # continue expanding
                else:
                    break
    # final safety: ensure finite score
    if not np.isfinite(best_score):
        best = np.ones_like(best, dtype=np.float64)

    # return as list of Python floats
    return [float(x) for x in best.tolist()]

import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Generate the classic r‑sequence a_i = C(2i,i)/4^i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq
