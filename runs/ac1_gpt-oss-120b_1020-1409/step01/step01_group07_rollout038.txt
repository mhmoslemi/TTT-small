def attenuate(seq, k_star, epsilon):
    n = len(seq)
    # Determine indices in overlapping region.
    i_start = max(0, k_star - (n-1))
    i_end = min(n-1, k_star)
    # compute mask of overlapping indices; also compute counterpart index j = k_star - i.
    # We'll apply scaling to the earlier half of this region: while i < j, scale down i and scale up j.
    cand = seq.copy()
    for i in range(i_start, i_end+1):
        j = k_star - i
        if i < j:
            # Scale i down and j up (conserve sum)
            delta = epsilon * cand[i]
            cand[i] = max(cand[i] - delta, 0.0)
            cand[j] = cand[j] + delta
    # Then normalize sum to 1.
    total = cand.sum()
    if total > 0:
        cand = cand / total
    return cand

import numpy as np

# ------------------------------------------------------------
# Fast evaluation (FFT based) – used only during the search.
# ------------------------------------------------------------
def _max_convolution_fft(seq: np.ndarray, L: int) -> float:
    """Return max of the full convolution seq * seq using FFT."""
    # Real FFT of the padded sequence
    f = np.fft.rfft(seq, n=L)
    # Element‑wise product (self‑convolution in frequency domain)
    conv = np.fft.irfft(f * f, n=L)
    # First 2*n-1 entries are the true convolution
    return float(conv[: 2 * seq.size - 1].max())


def _fast_score(seq: np.ndarray, L: int) -> float:
    """Fast approximation of the evaluator's objective."""
    s = seq.sum()
    if s == 0.0:
        return np.inf
    max_b = _max_convolution_fft(seq, L)
    return float(2.0 * seq.size * max_b / (s * s))
