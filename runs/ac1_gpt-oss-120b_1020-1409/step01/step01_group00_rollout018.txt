import numpy as np
import time


def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..length-1 (discrete analogue of 1/√x)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        fi = float(i)
        a[i] = a[i - 1] * ((2.0 * fi - 1.0) * (2.0 * fi)) / (4.0 * fi * fi)
    return a


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound
    is as small as can be found within modest time.  The routine starts
    from the parent vector `height_sequence_1` if present, otherwise from
    the binomial sequence, and performs a fast O(n) local search that
    repeatedly moves a small amount of mass from indices that contribute
    most to the current maximal autocorrelation toward random destinations.
    Occasional block scalings are tried as a larger‑scale perturbation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise the candidate.
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])          # parent vector provided
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:                                 # any failure → fallback
        n = 256
        a = _generate_binomial_sequence(n)

    np.clip(a, 0.0, 1000.0, out=a)

    # Full autocorrelation (self‑convolution) and auxiliary values.
    conv = np.convolve(a, a)               # length 2n‑1, dtype float64
    sum_a = a.sum()                        # sum stays constant for pair moves
    best_a = a.copy()
    best_conv = conv.copy()
    best_score = 2.0 * n * conv.max() / (sum_a * sum_a)

    # ------------------------------------------------------------------
    # Parameters for the search.
    # ------------------------------------------------------------------
    max_steps = 12000                # hard ceiling on loop count
    time_limit = 9.5                 # seconds, leave margin for the harness
    start_time = time.time()

    for step in range(max_steps):
        # Respect the overall time budget.
        if time.time() - start_time > time_limit:

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        k0 = int(conv.argmax())                     # lag with maximal value
        lo = k0 - (n - 1)

        contrib = a[idx] * a[k0 - idx]
