import numpy as np

# Basic utilities

def _make_series(length: int) -> np.ndarray:
    """Truncated series a_i = C(2i,i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself via real FFT."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound (matches evaluate_sequence)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    m = _conv_self(seq).max()
    return 2.0 * seq.size * m / (s * s)

def _lp_max_sum(f: np.ndarray, M: float):
    """
    Solve  max sum(g)   s.t. conv(f,g) <= M  (elementwise), 0 <= g <= 1000.
    Returns g (np.ndarray) if successful, otherwise None.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import diags
    except Exception:
        return None

    n = f.size
    # Build A where A[k, j] = f[k-j]  (zero outside range)
    # Using a tiled matrix for the diagonals of the Toeplitz operator.
    # Each diagonal i (offset i) is constant f[i].
    data = np.tile(f[:, None], (1, n))          # shape (n, n)
    offsets = np.arange(n)                      # 0 .. n-1
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)           # minimise -sum(g)  == maximise sum(g)
    bounds = [(0.0, 1000.0)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x

        lp_iters = 8
        f = best.copy()
        for _ in range(lp_iters):
            conv = _conv_self(f)
            M = conv.max()
            g = _lp_max_sum(f, M)
            if g is None:
                break
            sum_f = f.sum()
            sum_g = g.sum()
            if sum_g <= sum_f + 1e-12:
                break                      # can't increase the L1 norm under current M

            improved = False
            # try a few mixing proportions
            for t in (0.5, 0.3, 0.1, 0.05):
                cand = (1.0 - t) * f + t * g
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _score(cand)
                if sc < best_score - 1e-12:
                    best = cand.copy()
                    best_score = sc
                    f = cand.copy()
                    improved = True
                    break
            if not improved:
                break

    # --------------------------------------------------------
    # 2) Gradient‑descent + random perturbations for fine tuning.
    # --------------------------------------------------------
    f = best.copy()
    no_improve = 0
    max_steps = 2000
    for step in range(max_steps):
        conv = _conv_self(f)
        M = conv.max()
        k = int(conv.argmax())
        S = f.sum()
        if S == 0.0:
            break

        # gradient of 2*n*M / S^2  (treated as M / S^2, constant factor omitted)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g_vec = np.zeros_like(f)
        g_vec[idx] = f[k - idx]

        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g_vec - (2.0 * M) * inv_S3

        # line search with decreasing learning rates
        lr = 1.0
        cand = f
        cand_score = best_score
        for _ in range(5):
            trial = f - lr * grad
            np.clip(trial, 0.0, 1000.0, out=trial)
            if rng.random() < 0.12:                    # occasional symmetrisation
                trial = 0.5 * (trial + trial[::-1])
            sc = _score(trial)
            if sc < cand_score:
                cand = trial
                cand_score = sc
                break
            lr *= 0.5

        # occasional block scaling perturbation
        if step % 13 == 0:
            width = max(1, n // rng.integers(10, 30))
            start_idx = rng.integers(0, max(1, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.10))
            block = cand.copy()
            block[start_idx:start_idx + width] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            sc_block = _score(block)
            if sc_block < cand_score:
                cand = block
                cand_score = sc_block

        if cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        f = cand

        if no_improve > 400:           # stagnation cut‑off
            break

    # --------------------------------------------------------
    # Final safety net.
    # --------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(v) for v in best]
