        norm = np.linalg.norm(g)
        if norm == 0:
            break
        direction = g / norm

        # update
        a_new = best - step * direction
        a_new = np.clip(a_new, 0.0, 1000.0)
        sum_new = np.sum(a_new)

import numpy as np

def _full_conv(a, L):
    """Linear convolution of a with itself using an FFT of length L."""
    fa = np.fft.rfft(a, n=L)
    c = np.fft.irfft(fa * fa, n=L)
    return c[: 2 * a.shape[0] - 1]

        v = np.ones_like(v)
        s = v.shape[0]
    return v / s

        parent = np.ones(n, dtype=np.float64)
