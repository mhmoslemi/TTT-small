def solve_lp_for_g(a, M, active_set):
    n = len(a)
    m = len(active_set)
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_set):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # for i in lo..hi:
        # column j = k - i
        # coefficient a_i
        # we need to add entry (row_idx, j, a_i)
        # Actually conv(a,g)[k] = sum_i a_i * g_{k-i}
        for i in range(lo, hi+1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(a[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))
    # Constraint: A_ub @ g <= M
    c = -np.ones(n)   # maximize sum(g) => minimize -sum(g)
    bounds = [(0, 1000) for _ in range(n)]
    # use high-level solver:
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        g = a # fallback
    return g

import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return the provided parent sequence if it exists, else a safe fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    # fallback – a short uniform vector that always satisfies the evaluator
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """
    Truncated arcsine coefficients:
        a[0] = 1,
        a[i] = a[i-1] * (1 - 0.5 / i)   for i >= 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _lp_update(a, M, active_set):
    """
    Solve the linear program
        maximize   sum(g)
        subject to conv(a, g)[k] <= M   for k in active_set
                   0 <= g_i <= 1000.
    Returns the optimal g (as a NumPy array).  If the LP fails,
    the original vector a is returned unchanged.
    """
    n = a.shape[0]
    m = len(active_set)
    rows = []
    cols = []
    data = []

    # Build sparse matrix for the active convolution constraints.
    for row, k in enumerate(active_set):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # i runs from lo..hi, coefficient a[i] multiplies g_{k-i}
        i_vals = np.arange(lo, hi + 1, dtype=np.int64)
        j_vals = k - i_vals
        rows.extend([row] * i_vals.size)
        cols.extend(j_vals.tolist())
        data.extend(a[i_vals].tolist())

    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n), dtype=np.float64)
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # maximize sum(g)  -> minimize -sum(g)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    try:
        res = opt.linprog(
            c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True}
        )
    except Exception:  # pragma: no cover
        return a

    if not res.success:
        return a
    g = res.x
    return g

def propose_candidate(seed=42, num_steps=1500, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the AC1 upper bound.
    The algorithm mixes a gradient‑driven mass‑transfer heuristic with a few
    sparse linear‑program updates (alternating optimisation).  The routine
    returns the best sequence seen during the search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise from parent or from the analytic arcsine seed
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = parent.size

    # pick the better of parent and arcsine (both normalised to sum 1)
    best = parent.copy()
    best_sum = best.sum()
    if best_sum == 0.0:
        best = np.ones(n, dtype=np.float64)
        best_sum = n
    best = best / best_sum
    best_score = float(evaluate_sequence(best.tolist()))

    arcsine = _arcsine_sequence(n)
    arcsine = arcsine / arcsine.sum()
    arcsine_score = float(evaluate_sequence(arcsine.tolist()))
    if arcsine_score < best_score:
        best = arcsine
        best_score = arcsine_score

    # keep a copy of the best ever-found vector
    best_overall = best.copy()
    best_overall_score = best_score

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # --------------------------------------------------------------
        # 1) compute current autocorrelation data
        # --------------------------------------------------------------
        conv = np.convolve(best, best)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # --------------------------------------------------------------
        # 2) gradient‑like mass‑transfer step (focused on the max entry)
        # --------------------------------------------------------------
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution of each coordinate to the peak
        contrib = np.zeros_like(best)
        contrib[idx_range] = best[max_idx - idx_range]

        # step size slowly decreasing
        alpha = 0.5 * (1.0 - step / num_steps) + 0.05   # between ~0.55 and 0.05
        cand = best - alpha * contrib
        # small random jitter to keep exploration alive
        cand += rng.normal(scale=0.001, size=n)
        np.clip(cand, 0.0, 1000.0, out=cand)

        total = cand.sum()
        if total > 0.0:
            cand = cand / total
        else:
            cand = best.copy()

        # occasional symmetrisation (helps flatten the autocorrelation)
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])
            cand = cand / cand.sum()

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_overall_score:
            best_overall, best_overall_score = cand.copy(), cand_score

        # accept if improvement, otherwise occasional annealed acceptance
        cur_score = float(evaluate_sequence(best.tolist()))
        if np.isfinite(cand_score) and cand_score < cur_score:
            best = cand
        else:
            # simulated‑annealing style acceptance
            temperature = 0.5 / (step + 1.0)
            if np.isfinite(cand_score):
                prob = np.exp((cur_score - cand_score) / temperature)
                if rng.random() < prob:
                    best = cand

        # --------------------------------------------------------------
        # 3) sparse LP update every few iterations
        # --------------------------------------------------------------
        if (step + 1) % 300 == 0:
            # select a modest active set – the largest K convolution entries
            K = min(30, conv.size)
            top_idxs = np.argpartition(conv, -K)[-K:]
            active_set = list(np.unique(top_idxs))
            g = _lp_update(best, max_val, active_set)

            # rescale g to keep the total mass unchanged (sum = 1)
            sum_g = g.sum()
            if sum_g > 0.0:
                g = g / sum_g
            else:
                g = best.copy()

            # convex mixing of the current vector with the LP result
            t = 0.5
            cand_lp = (1.0 - t) * best + t * g
            np.clip(cand_lp, 0.0, 1000.0, out=cand_lp)

            if np.isfinite(cand_lp_score) and cand_lp_score < best_overall_score:
                best_overall, best_overall_score = cand_lp.copy(), cand_lp_score

            cur_score = float(evaluate_sequence(best.tolist()))
            if np.isfinite(cand_lp_score) and cand_lp_score < cur_score:
                best = cand_lp

    # ------------------------------------------------------------------
    if best_overall.size == 0 or not np.isfinite(best_overall_score):
        best_overall = np.ones(n, dtype=np.float64) / n

    return [float(x) for x in best_overall]
