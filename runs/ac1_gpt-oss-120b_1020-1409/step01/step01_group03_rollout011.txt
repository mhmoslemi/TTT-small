def _conv_self(seq):
    n = len(seq)
    conv_len = 2*n-1
    fft_len = 1 << (conv_len-1).bit_length()
    fx = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fx*fx, n=fft_len)[:conv_len]
    return conv

def _solve_lp(f, M):
    n = f.shape[0]
    A = _build_A(f)
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximize sum => minimize -sum
    bounds = [(0.0, 1000.0) for _ in range(n)]
    # call linprog
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    else:
        # fallback: return current f (maybe)
        return f.copy()

def propose_candidate(seed=42, num_iters=5, t=0.2, **kwargs):
    rng = np.random.default_rng(seed)
    # get parent if exists
    try: 
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim !=1 or parent.size ==0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        parent = None
        n = 256
    # initial candidate: use parent if exists else analytic series defined earlier
    if parent is not None:
        f = parent.copy()
    else:
        # analytic series from previous program
        f = np.empty(n, dtype=np.float64)
        f[0] = 1.0
        for i in range(1,n):
            f[i] = f[i-1] * (2*i-1) / (2.0*i)
    # clip
    np.clip(f, 0.0, 1000.0, out=f)
    # compute best
    best = f.copy()
    best_score = _score(best)
    # iterate
    for it in range(num_iters):
        conv = _conv_self(best)
        M = conv.max()
        # solve LP to get g
        g = _solve_lp(best, M)
        # mixing
        new_f = (1 - t) * best + t * g
        np.clip(new_f, 0.0, 1000.0, out=new_f)
        # evaluate
        sc = _score(new_f)
        if sc < best_score:
            best = new_f
            best_score = sc
        # maybe also random perturbation if stuck
        else:
            # random shuffle: pick a random block and multiply factor
            width = max(1, n // rng.integers(10, 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.2))
            block = best.copy()
            block[start:start+width] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            scb = _score(block)
            if scb < best_score:
                best = block
                best_score = scb
        # optionally small symmetrization
        if rng.random() < 0.1:
            # optional symmetric averaging
            best = 0.5 * (best + best[::-1])
            np.clip(best, 0.0, 1000.0, out=best)
            best_score = _score(best)
    # final fallback: if best invalid
    if best.size ==0 or not np.isfinite(best_score):
        best = np.ones(256)

import numpy as np
from numpy.fft import rfft, irfft

# Optional SciPy imports – if unavailable we simply skip the LP part.
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ------------------------------------------------------------
# Convolution and scoring (mirrors the evaluator exactly)
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself using real FFT."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = rfft(x, n=fft_len)
    conv = irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound used by the evaluator."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)


# ------------------------------------------------------------
# Analytic starting series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Truncated binomial series a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Build the sparse matrix representing the linear map g -> f * g
# ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Return a CSR matrix A of shape (2n-1, n) such that A @ g = conv(f, g).
    Each column j contains the vector f placed on the diagonal with offset j.
    """
    n = f.shape[0]
    # column indices: repeat 0..n-1, each n times
    col = np.tile(np.arange(n, dtype=np.int32), n)
    # row indices: offset + row index within the column
    offset = np.repeat(np.arange(n, dtype=np.int32), n)
    idx = np.tile(np.arange(n, dtype=np.int32), n)
    rows = offset + idx
    data = np.tile(f, n)
    return csr_matrix((data, (rows, col)), shape=(2 * n - 1, n))


# ------------------------------------------------------------
# Linear program: maximise sum(g) subject to conv(f,g) <= M and box bounds
# ------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray:
    """Return a feasible g with maximal ℓ1‑norm under the convolution bound."""
    if not _SCIPY_AVAILABLE:
        # SciPy missing – return the input as a dummy solution
        return f.copy()

    n = f.shape[0]
    # Build constraint matrix (sparse)
    A = _build_conv_matrix(f)

    # Right‑hand side: all rows share the same bound M (tiny slack for safety)
    b = np.full(2 * n - 1, np.nextafter(M, M + 1), dtype=np.float64)

    # Objective: maximise sum(g)  →  minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # Box bounds for each variable
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # Solve using HiGHS (sparse‑capable)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if res.success:
            return np.array(res.x, dtype=np.float64)
        # fall back on the current f if LP fails
        return f.copy()
    except Exception:  # pragma: no cover
        return f.copy()


# ------------------------------------------------------------
# Simple projected gradient descent on the AC1 bound
# ------------------------------------------------------------
def _projected_gradient_step(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One step of projected gradient descent on the bound."""
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    k = int(conv.argmax())

    # indices that contribute to the maximal convolution term
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1, dtype=int)

    # gradient of M w.r.t. x (sub‑gradient of the max)
    grad_M = np.zeros_like(x)
    grad_M[idx] = x[k - idx]

    S = x.sum()
    if S == 0.0:
        return x

    # derivative of 2n*M / S^2
    C = 2.0 * n
    grad = C * (grad_M / (S * S) - 2.0 * M * np.ones_like(x) / (S * S * S))

    # simple back‑tracking line search
    lr = 1.0
    best = x
    best_sc = _score(x)
    for _ in range(5):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        if rng.random() < 0.1:  # occasional symmetrisation
            cand = 0.5 * (cand + cand[::-1])
        sc = _score(cand)
        if sc < best_sc:
            best = cand
            best_sc = sc
            break
        lr *= 0.5
    return best
