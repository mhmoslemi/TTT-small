def gradient_step(x):

    conv = _conv_self(x)

    M = conv.max()

    k = conv.argmax()

    # compute g:

    n = len(x)

    g = np.zeros_like(x)

    i_low = max(0, k - (n - 1))

    i_high = min(n - 1, k)

    idx = np.arange(i_low, i_high + 1)

    g[idx] = x[k - idx]

    # gradient of f = M / S^2

    S = x.sum()

    if S == 0: return x

    inv_S2 = 1.0 / (S * S)

    inv_S3 = inv_S2 / S

    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

    # step size selection: we want to minimize score; compute projected step length.

    # We'll try step sizes from 0.8 down to 0.01 scaling.

    # Accept best candidate among a few.

    best_z = x

    best_sc = _score(x)

    for lr in [0.8, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01]:

        z = x - lr * grad

        np.clip(z, 0.0, 1000.0, out=z)

        # symmetrize optionally

        if True:  # always symmetrize

            z = (z + z[::-1]) * 0.5

        sc = _score(z)

        if sc < best_sc - 1e-12:

            best_z, best_sc = z, sc

            break

    return best_z, best_sc

def random_block(x, rng):

    n = len(x)

    width = rng.integers(1, max(2, n // 10))

    start = rng.integers(0, n - width + 1)

    factor = np.exp(rng.normal(0.0, 0.2))

    y = x.copy()

    y[start:start+width] *= factor

    np.clip(y, 0.0, 1000.0, out=y)

    y = (y + y[::-1]) * 0.5

    return y

def random_exchange(x, rng):

    n = len(x)

    i = rng.integers(0, n)

    j = rng.integers(0, n)

    if i == j: return x

    amt = x[i] * rng.random()  # move fraction of i's mass

    y = x.copy()

    y[i] -= amt

    y[j] += amt

    # clip

    np.clip(y, 0.0, 1000.0, out=y)

    y = (y + y[::-1]) * 0.5

    return y

def _gradient_step(x, rng):
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    k = int(conv.argmax())
    # build g
    g = np.zeros_like(x)
    if k < n:
        g[:k+1] = x[k::-1]
    else:
        start = k - (n - 1)
        g[start:] = x[start:][::-1]

    S = x.sum()
    if S == 0.0:
        return x.copy(), _score(x)

    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = 2.0 * inv_S2 * g - (2.0 * M) * inv_S3

    # choose learning rate
    # we base on ratio of S to norm of grad perhaps:
    # lr = 0.5 * S / max(np.linalg.norm(grad), 1e-8)  # not needed to be perfect

    lr = 0.5
    y = x - lr * grad
    # Clip to [0,1000]
    np.clip(y, 0.0, 1000.0, out=y)
    # Symmetrize
    y = (y + y[::-1]) * 0.5

    sc = _score(y)
    return y, sc

def _random_block(x, rng):
    n = x.shape[0]
    # width random between 1 and n//10 (max 10% size)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    y = x.copy()
    y[start:start+width] *= factor
    np.clip(y, 0.0, 1000.0, out=y)
    y = (y + y[::-1]) * 0.5
    return y

Random exchange:

def _random_exchange(x, rng):
    n = x.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return x
    # transfer some portion from i to j
    amount = x[i] * rng.random()
    y = x.copy()
    y[i] -= amount
    y[j] += amount
    np.clip(y, 0.0, 1000.0, out=y)
    y = (y + y[::-1]) * 0.5
    return y

def _make_series(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i - 1) / (2.0 * i)
    return a

        if step % 9 == 0:

def _gradient_step(x, rng):
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    k = int(conv.argmax())
    # compute g
    g = np.zeros_like(x)
    if k < n:
        # indices 0..k inclusive
        # x[k::-1] gives x[k], x[k-1], ... x[0]
        g[:k+1] = x[k::-1]
    else:
        start = k - (n - 1)
        # x[start:] reversed
        g[start:] = x[start:][::-1]
    S = x.sum()

    np.clip(y, 0.0, 1000.0, out=y)
    y = _symmetrize(y)

import numpy as np

# ------------------------------------------------------------
# Convolution (full) using real FFT – O(n log n), memory O(n).
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    c = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return c


# ------------------------------------------------------------
# Exact AC‑1 score (mirrors the evaluator).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    m = _conv_self(seq).max()
    return 2.0 * seq.size * m / (s * s)


# ------------------------------------------------------------
# Classic Catalan‑like series a_i = C(2i,i)/4^i.
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Simple symmetrisation (enforces a_i = a_{n-1-i}).
# ------------------------------------------------------------
def _symmetrize(x: np.ndarray) -> np.ndarray:
    return 0.5 * (x + x[::-1])


# ------------------------------------------------------------
# One projected gradient step based on the maximal convolution term.
# ------------------------------------------------------------
def _gradient_step(x: np.ndarray, rng: np.random.Generator) -> tuple:

    conv = _conv_self(x)
    M = conv.max()
    k = int(conv.argmax())

    # g_i = a_{k-i} on the overlapping region, zero elsewhere
    g = np.zeros_like(x)
    if k < n:
        g[: k + 1] = x[k::-1]
    else:
        start = k - (n - 1)
        g[start:] = x[start:][::-1]

    S = x.sum()
    if S == 0.0:
        return x.copy(), _score(x)

    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = 2.0 * inv_S2 * g - (2.0 * M) * inv_S3

    # modest fixed step size – works well for the scaled‑invariant problem
    lr = 0.2
    y = x - lr * grad
    np.clip(y, 0.0, 1000.0, out=y)
    y = _symmetrize(y)
    return y, _score(y)

    y = x.copy()
    y[start : start + width] *= factor
    np.clip(y, 0.0, 1000.0, out=y)
    return _symmetrize(y)

# ------------------------------------------------------------
