def _random_pair_perturbation(arr, rng):
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] == 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.05)
    new_arr = arr.copy()
    new_arr[i] = max(0.0, new_arr[i] - delta)
    new_arr[j] = min(1000.0, new_arr[j] + delta)
    return new_arr
```

def _block_scaling(arr, rng):
    n = arr.size
    min_width = max(1, n // 200)
    max_width = max(2, n // 20)
    width = int(rng.integers(min_width, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = np.exp(rng.normal(0.0, 0.07))
    new_arr = arr.copy()
    new_arr[start:start+width] *= factor
    np.clip(new_arr, 0.0, 1000.0, out=new_arr)
    return new_arr
```
