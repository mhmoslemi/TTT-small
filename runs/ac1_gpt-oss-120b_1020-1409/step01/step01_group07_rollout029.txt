import numpy as np

# ------------------------------------------------------------
# Helper: fast full convolution via FFT (linear, not circular)
# ------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, m: int, n: int) -> np.ndarray:
    """Return a[:2*n-1] = a * a using FFT of length m >= 2*n-1."""
    A = np.fft.rfft(a, n=m)
    b_full = np.fft.irfft(A * A, n=m)
    return b_full[: 2 * n - 1]


def _score_fft(a: np.ndarray, m: int, n: int) -> float:
    """Score = 2*n*max(conv)/(sum(a)^2) using FFT convolution."""
    conv = _full_convolution_fft(a, m, n)
    return 2.0 * n * conv.max() / (a.sum() * a.sum())
