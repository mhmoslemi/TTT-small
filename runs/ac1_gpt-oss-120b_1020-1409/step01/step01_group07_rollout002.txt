def _uniform_sequence(n):
    """Return a uniform non‑negative sequence of length n with sum = 1."""
    # Using float64 ensures the arithmetic matches the evaluator expectations.
    return np.full(n, 1.0 / n, dtype=np.float64)

def _pair_perturb(seq, rng):
    """Move a small amount of mass from one index to another (sum preserved)."""
    n = seq.size
    # pick two distinct indices
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return seq

    max_transfer = 0.03 * seq[i]          # at most 3 % of the donor
