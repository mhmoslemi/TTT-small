def project_onto_simplex(v, z=1.0):
    # Ensure nonnegative?
    # standard algorithm: shift to nonnegative by sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_onto_simplex(v, z=1.0):
    # v can be any real-valued np.ndarray
    # returns projection onto simplex {x: x >= 0, sum x = z}
    if np.all(v >= 0) and abs(v.sum() - z) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v, z=1.0):
    # projection onto simplex {x>=0, sum(x)=z}
    v = v.astype(np.float64, copy=False)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros
        return np.maximum(v, 0)
    rho = ind[cond][-1] - 1  # zero-based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _arcsine_sequence(n):
    """Generate the classic arcsine‐type seed a_i = a_{i-1}*(1-0.5/i)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0

def _project_simplex(v, z=1.0):
    """
    Euclidean projection of vector v onto the simplex {x >= 0, sum(x)=z}.
    O(n log n) due to sorting.
    """
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)

    w = np.maximum(v - theta, 0.0)
