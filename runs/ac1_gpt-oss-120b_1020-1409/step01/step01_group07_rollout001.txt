import numpy as np

def _inverse_sqrt_seq(n):
    """Generate a 1/√(i+1) shape, L1‑normalised."""
    seq = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 1.0)
    s = seq.sum()
    if s == 0.0:
        seq = np.ones(n, dtype=np.float64) / n
    else:
        seq /= s
    return seq

def _clip_and_normalise(seq):
    """Clip to [0,1000] and renormalise to unit L1 sum."""
    np.clip(seq, 0.0, 1000.0, out=seq)
    s = seq.sum()
    if s > 0.0:
        seq /= s
    else:
        # fallback to uniform if everything vanished
        seq[:] = 1.0 / seq.size
    # second clipping in case renorm blew past 1000
    np.clip(seq, 0.0, 1000.0, out=seq)
    s = seq.sum()
    if s > 0.0:
        seq /= s
    return seq

def _perturb_block(seq, rng, max_width):
    """Multiplicative perturbation of a random contiguous block."""
    n = seq.size
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.03)))
    cand = seq.copy()
    cand[start:start + width] *= factor
    return _clip_and_normalise(cand)

def _smooth_candidate(seq, rng):
    """Apply a short moving‑average smoothing."""
    n = seq.size
    # width 3 or 5 chosen randomly
    w = int(rng.choice([3, 5]))
    kernel = np.full(w, 1.0 / w, dtype=np.float64)
    cand = np.convolve(seq, kernel, mode='same')
    return _clip_and_normalise(cand)

def _targeted_shift(seq, rng, max_frac):
    """Move a small amount of mass from a region that contributes to the maximal autocorrelation."""
    n = seq.size
    conv = np.convolve(seq, seq)
    max_idx = int(np.argmax(conv))
    offset = max_idx - (n - 1)          # offset k where the maximum occurs
    i_start = max(0, -offset)
    i_end = min(n - 1, n - 1 - offset)  # inclusive range of indices participating
    if i_start > i_end:
        # Fell back to a block perturbation if the region is empty
        return _perturb_block(seq, rng, max(1, n // 100))
    # pick a source index that actually carries mass
    for _ in range(5):
        src = int(rng.integers(i_start, i_end + 1))
        if seq[src] > 0.0:
            break
    else:
        src = int(rng.integers(i_start, i_end + 1))
    dst = int(rng.integers(0, n))
    if dst == src:
        dst = (dst + 1) % n
    eps = seq[src] * rng.random() * max_frac
    if eps <= 0.0:
        return seq.copy()
    cand = seq.copy()
    cand[src] -= eps
    cand[dst] += eps
    return _clip_and_normalise(cand)
