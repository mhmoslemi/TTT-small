import numpy as np

# ---------- Helper functions -------------------------------------------------
def _max_and_conv(a: np.ndarray, n: int, fft_len: int):
    """Return (max_convolution, full_convolution_slice) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    sub = conv[: 2 * n - 1]
    return float(sub.max()), sub


def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact AC1 upper bound for a candidate (no clipping, just the formula)."""
    max_b, _ = _max_and_conv(a, n, fft_len)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ---------- Main entry point -------------------------------------------------
def propose_candidate(seed: int = 42, max_iter: int = 6000, **kwargs):
    """
    Return a list of non‑negative step‑function heights whose AC1 bound is low.
    The routine works without any external data other than the optional parent
    construction `height_sequence_1`.  It scans several analytic families,
    then performs a mixed stochastic / targeted hill‑climbing using FFT‑based
    convolution (O(n log n) per iteration).  A short final refinement uses
    the exact np.convolve evaluator to guarantee the reported score.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size from the supplied parent construction.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ----------------------------------------------------------------------
    # FFT parameters (single length used for all convolutions).
    # ----------------------------------------------------------------------
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))

    # ----------------------------------------------------------------------
    # Initial candidates from a few compact families.
    # ----------------------------------------------------------------------
    best = np.clip(parent, 0.0, 1000.0)
    best_score = _score(best, n, fft_len)

    # 1) Power‑law a_i = 1/√(i+α)
    alphas = np.logspace(-5, 1.0, num=30, dtype=np.float64)
    for alpha in alphas:
        cand = np.clip(1.0 / np.sqrt(np.arange(n, dtype=np.float64) + alpha),
                       0.0, 1000.0)
        cand_score = _score(cand, n, fft_len)
        if cand_score < best_score:
            best, best_score = cand, cand_score

    # 2) Exponential decay a_i = exp(-β i)
    for beta in np.linspace(0.0005, 0.05, 20):
        cand = np.clip(np.exp(-beta * np.arange(n, dtype=np.float64)),
                       0.0, 1000.0)
        cand_score = _score(cand, n, fft_len)
        if cand_score < best_score:
            best, best_score = cand, cand_score

    # 3) Linear drop from 1 to 0
    cand = np.clip(np.linspace(1.0, 0.0, n, dtype=np.float64), 0.0, 1000.0)
    cand_score = _score(cand, n, fft_len)
    if cand_score < best_score:
        best, best_score = cand, cand_score

    # ----------------------------------------------------------------------
    # Mixed stochastic optimisation.
    # ----------------------------------------------------------------------
    min_width = max(1, n // 200)          # ≈0.5 % of length
    max_width = max(2, n // 20)           # ≈5 % of length
    sigma = 0.07                          # multiplicative log‑normal noise
    init_temp = max(best_score * 0.05, 1e-4)

    for it in range(max_iter):
        # temperature decreasing linearly
        temp = init_temp * (1.0 - it / max_iter)
        if rng.random() < 0.6:  # targeted mass transfer
            # full convolution and peak location
            _, conv = _max_and_conv(best, n, fft_len)
            k_max = int(np.argmax(conv[: 2 * n - 1]))

            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            region_len = i_end - i_start + 1

            # If the region covers the whole vector, fall back to block update
            if region_len == n:
                width = int(rng.integers(min_width, max_width + 1))
                start = int(rng.integers(0, n - width + 1))
                factor = float(np.exp(rng.normal(0.0, sigma)))
                cand = best.copy()
                cand[start:start + width] *= factor
                if rng.random() < 0.08:               # occasional symmetrisation
                    cand = 0.5 * (cand + cand[::-1])
                cand = np.clip(cand, 0.0, 1000.0)
            else:
                # indices that participate in the maximal entry
                region = np.arange(i_start, i_end + 1, dtype=np.int64)
                partners = k_max - region
                prods = best[region] * best[partners]
                i_max = region[np.argmax(prods)]

                # pick a destination outside the region
                while True:
                    t = int(rng.integers(0, n))
                    if not (i_start <= t <= i_end):
                        break

                delta = best[i_max] * rng.random() * 0.15
                cand = best.copy()
                cand[i_max] -= delta
                cand[t] += delta
                cand = np.clip(cand, 0.0, 1000.0)

        else:                     # random multiplicative block perturbation
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand = best.copy()
            cand[start:start + width] *= factor
            if rng.random() < 0.08:                # occasional weak symmetry
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)

        cand_score = _score(cand, n, fft_len)

        # Metropolis acceptance (allows occasional uphill moves)
        if cand_score < best_score or rng.random() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
            best, best_score = cand, cand_score

    # ----------------------------------------------------------------------
    # Final refinement using the exact evaluator (np.convolve) to guarantee
    # the reported bound.
    # ----------------------------------------------------------------------
    try:
        exact_score = float(evaluate_sequence(best.tolist()))
    except Exception:                     # safety fallback
        exact_score = np.inf

    for _ in range(250):                  # modest number of exact evaluations
        # small random block tweak
        width = int(rng.integers(min_width, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, sigma)))
        cand = best.copy()
        cand[start:start + width] *= factor
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if cand_score < exact_score:
            best, exact_score = cand, cand_score

    # ----------------------------------------------------------------------
    # Sanity check – if everything failed, fall back to the parent.
    # ----------------------------------------------------------------------
    if not np.isfinite(exact_score) or best.size == 0:
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]
