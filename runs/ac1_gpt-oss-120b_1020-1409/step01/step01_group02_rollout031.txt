def _parent_sequence():
    """Return the provided parent sequence if it exists, else None."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:
        pass
    return None


def _arcsine_sequence(n):
    """Arcsine‑like decreasing sequence a_i ≈ i^{-1/2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _power_sequence(n, p):
    """a_i ∝ (i+1)^{-p} (normalized)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a

def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 upper bound.
    Uses a lightweight projected gradient descent on the maximal autocorrelation
    entry with occasional stochastic acceptance.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Obtain a starting length n and some candidate seeds
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    if parent is None:
        # fallback – a short uniform vector (will be resized later)
        n = 256
        parent = np.ones(n, dtype=np.float64) / n
    else:
        n = int(parent.size)

    # ------------------------------------------------------------------
    # 2.  Build a small pool of analytic seeds and pick the best one
    # ------------------------------------------------------------------
    candidates = []

    # normalized parent
    if parent.sum() > 0:
        cand = parent / parent.sum()
        candidates.append(cand)

    # arcsine seed
    arcsine = _arcsine_sequence(n)
    candidates.append(arcsine / arcsine.sum())

    # power‑law seeds for a few exponents
    for p in (0.4, 0.5, 0.6, 0.7):
        cand = _power_sequence(n, p)
        candidates.append(cand / cand.sum())

    # pick the best seed according to the official scorer
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        # evaluator is cheap for a handful of candidates
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # if everything failed, fall back to a simple uniform vector
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 3.  Initialise the optimisation state
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    conv = np.convolve(cur, cur)          # length 2*n-1
    sum_a = cur.sum()                     # invariant (≈1 after normalisation)
    sum_sq = sum_a * sum_a

    max_steps = int(min(60000, max(5000, n * 12)))
    target_score = 1.50274349

    # ------------------------------------------------------------------
    # 4.  Main projected gradient descent loop
    # ------------------------------------------------------------------
    no_improve = 0
    for step in range(max_steps):
        # keep copies to allow a rollback
        cur_old = cur.copy()
        conv_old = conv.copy()

        # locate the current maximal autocorrelation entry
        idx_max = int(conv.argmax())
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)

        # indices that actually affect this entry
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        rev_range = cur[idx_max - i_range]                 # a_{k-i}
        contrib = cur[i_range] * rev_range
        i_idx = int(i_range[np.argmax(contrib)])

        # pick a partner index with (almost) zero contribution to this max entry
        if lo > 0:
            j_idx = 0
        elif hi < n - 1:
            j_idx = n - 1
        else:
            # full range – choose index with minimal rev_range
            min_rev = rev_range.min()
            cand_js = i_range[rev_range == min_rev]
            if cand_js.size > 1:
                j_idx = int(cand_js[0] if cand_js[0] != i_idx else cand_js[1])
            else:
                j_idx = (i_idx + 1) % n

        # ensure i != j
        if j_idx == i_idx:
            j_idx = (i_idx + 1) % n

        if cur[i_idx] <= 0.0:
            # nothing to move – skip this iteration
            continue

        # transfer a modest fraction of mass from i to j
        delta = cur[i_idx] * 0.5
        if delta <= 0.0:
            continue

        # ----- update the convolution (matrix‑free) -----
        pre_a = cur.copy()                     # values before the move
        # linear part
        conv[i_idx:i_idx + n] -= 2.0 * delta * pre_a
        conv[j_idx:j_idx + n] += 2.0 * delta * pre_a
        # quadratic self‑terms
        conv[2 * i_idx] += delta * delta
        conv[2 * j_idx] += delta * delta
        # cross term
        conv[i_idx + j_idx] -= 2.0 * delta * delta

        # ----- update the sequence itself -----
        cur[i_idx] -= delta
        cur[j_idx] += delta

        # enforce non‑negativity (tiny negative round‑off may appear)
        if cur[i_idx] < 0.0:
            cur[i_idx] = 0.0

        # current score after the move
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / sum_sq

        # simulated‑annealing style acceptance
        temperature = max(0.01, 1.0 - step / max_steps)
        accept = False
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            accept = True
        elif np.exp((best_score - cur_score) / temperature) > rng.random():
            accept = True

        if not accept:
            # rollback
            cur = cur_old
            conv = conv_old
            no_improve += 1
        else:
            no_improve = 0

        # early exit if we have already reached the target bound
        if best_score <= target_score:
            break

        # safety break if we stagnate for a long time
        if no_improve >= 5000:
            break

    # ------------------------------------------------------------------
    # 5.  Safety fallback – should never be needed
    # ------------------------------------------------------------------
    try:
        best
    except NameError:
        best = np.ones(max(1, n), dtype=np.float64) / n

    # Return a plain Python list of floats
    return [float(x) for x in best]
