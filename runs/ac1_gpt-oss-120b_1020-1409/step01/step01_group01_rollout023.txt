import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Generate the classic r‑sequence  a_i = C(2i,i)/4^i  for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function that tries to minimise the AC1 bound."""
    # ---------------------------------------------------------------------
    # 1.  Initialise from the supplied parent or fall back to the r‑sequence.
    # ---------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = generate_r_sequence(n)

    np.clip(a, 0.0, 1000.0, out=a)          # keep inside evaluator limits
    n = a.size

    # ---------------------------------------------------------------------
    # 2.  Pre‑compute the full self‑convolution.
    # ---------------------------------------------------------------------
    b = np.convolve(a, a)                  # length 2*n-1
    M = b.max()                            # current max of the convolution

    # ---------------------------------------------------------------------
    # 3.  Greedy finite‑difference optimisation.
    #    For each index i we compute the largest δ that can be added to a[i]
    #    without raising any entry of b above the current bound M.
    #    We then add a fraction of that slack and update a and b online.
    # ---------------------------------------------------------------------
    MAX_PASSES = 5                          # safety bound for iterations
    REDUCTION_FACTOR = 0.5                  # take half of the admissible slack
    EPS_MIN = 1e-12

    for _ in range(MAX_PASSES):
        improved = False
        # recompute M at the start of the pass (may have changed in previous pass)
        M = b.max()
        for i in range(n):
            # -----------------------------------------------------------------
            # Compute admissible slack for position i.
            # -----------------------------------------------------------------
            # slice of b that will be affected by a change at index i
            b_slice = b[i:i + n]                     # length n
            # denominator 2*a (avoid division by zero)
            denom = 2.0 * a
            numerator = M - b_slice
            # candidate slack values – infinite where a is zero (no restriction)
            cand = np.where(denom > 0.0, numerator / denom, np.inf)
            slack = cand.min()
            if slack <= EPS_MIN:
                continue

            # -----------------------------------------------------------------
            # Apply a safe step (half of the slack) and update a, b, M.
            # -----------------------------------------------------------------
            delta = slack * REDUCTION_FACTOR
            # contribution of the existing a‑vector to the affected part of b
            term = 2.0 * delta * a                     # length n
            b[i:i + n] += term
            b[2 * i] += delta * delta                  # e_i * e_i term
            a[i] += delta                               # update the coefficient

            # update the current maximum of b (only the changed region can increase)
            slice_max = b[i:i + n].max()
            if slice_max > M:
                M = slice_max
            if b[2 * i] > M:
                M = b[2 * i]

            improved = True

        if not improved:

    # ---------------------------------------------------------------------

    # ---------------------------------------------------------------------
    np.clip(a, 0.0, 1000.0, out=a)

    # Verify that the sequence is acceptable for the external evaluator.
    try:
        score = float(evaluate_sequence(a.tolist()))
        if not np.isfinite(score):
            raise ValueError
    except Exception:
        # In the unlikely event of an invalid sequence, fall back to a simple vector.
        a = np.ones(n, dtype=np.float64)

    return [float(x) for x in a]
