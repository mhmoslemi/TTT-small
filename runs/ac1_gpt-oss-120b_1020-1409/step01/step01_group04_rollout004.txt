import numpy as np

import math

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _power_seq(alpha: float, p: float, n: int) -> np.ndarray:
    """
    Decreasing power‑law sequence a_i = 1/(i+alpha)^p, clipped to [0,1000].
    """
    i = np.arange(n, dtype=np.float64)
    seq = 1.0 / np.power(i + alpha, p)
    # Numerical safety – the evaluator also clips, but we stay inside limits.
    return np.clip(seq, 0.0, 1000.0)


def _enforce_monotone_desc(seq: np.ndarray) -> np.ndarray:
    """
    Force a non‑increasing (monotonically descending) sequence.
    Works in‑place‑free and respects the 0‑1000 range.
    """
    # max‑accumulate from the right side, then reverse back.
    return np.maximum.accumulate(seq[::-1])[::-1]
