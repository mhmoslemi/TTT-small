def _pair_move(arr, rng, scale_range=(0.01,0.2)):
    n = arr.size
    i = rng.integers(0, n)
    attempts=0
    while attempts<5 and arr[i] <= 0:
        i = rng.integers(0, n)
        attempts+=1
    j = rng.integers(0, n)
    # ensure not same
    if i == j:
        return arr.copy()
    frac = rng.uniform(*scale_range)
    delta = arr[i] * frac
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand
```

def _block_move(arr, rng, scale_range=(0.7,1.3)):
    n = arr.size
    width = int(rng.integers(max(1, n//200), max(2, n//20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = rng.uniform(*scale_range)
    cand = arr.copy()
    cand[start:start+width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _approx_score(arr):
    n = arr.size
    sum_a = arr.sum()
    if sum_a == 0:
        return np.inf
    L = 2*n - 1
    # compute convolution via FFT
    # rfft length L yields N/2+1 points.

    conv = np.fft.irfft(fft_a * fft_a, n=L)
    max_b = np.max(conv)

    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a

def _pair_move(arr: np.ndarray, rng: np.random.Generator,
               scale_range: tuple = (0.01, 0.30)) -> np.ndarray:
    """Transfer a fraction of mass from one index to another, keeping sum constant."""
    n = arr.size
    # pick donor with positive mass
    for _ in range(5):
        i = rng.integers(0, n)
        if arr[i] > 0.0:
            break
    else:
        return arr.copy()
    j = rng.integers(0, n)
    if i == j:
        return arr.copy()
    frac = rng.uniform(*scale_range)
    delta = arr[i] * frac
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand

def _block_move(arr: np.ndarray, rng: np.random.Generator,
                scale_range: tuple = (0.80, 1.25)) -> np.ndarray:
    """Scale a random contiguous block by a factor."""
    n = arr.size
    # width between n/200 and n/20
    min_w = max(1, n // 200)
    max_w = max(2, n // 20)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = rng.uniform(*scale_range)
    cand = arr.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _swap_move(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two entries, preserving the multiset."""
    n = arr.size
    i, j = rng.choice(n, size=2, replace=False)
    if i == j:
        return arr.copy()
    cand = arr.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand

def _approx_score(arr: np.ndarray) -> float:
    """
    Fast surrogate for evaluate_sequence using FFT.
    Returns 2 * n * max(conv) / (sum^2).
    """
    n = arr.size
    sum_a = arr.sum()
    if sum_a == 0.0:
        return np.inf
    L = 2 * n - 1
    # real FFT convolution
    fft_a = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed: int = 42, max_iters: int = 5000, **kwargs):
    """
    Return a list of non‑negative step‑function heights aiming to minimise
    the AC1 upper bound defined by evaluate_sequence.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initialise candidates.
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])          # provided by harness
        base = height_sequence_1.astype(np.float64).copy()
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    bin_seq = _generate_binomial_sequence(n)

    # Evaluate the two obvious starters.
    try:
        score_base = float(evaluate_sequence(base.tolist()))
    except Exception:
        score_base = np.inf
    try:
        score_bin = float(evaluate_sequence(bin_seq.tolist()))
    except Exception:
        score_bin = np.inf

    if score_bin < score_base:
        best = bin_seq.copy()
        best_score = score_bin
    else:
        best = base.copy()
        best_score = score_base

    best_approx = _approx_score(best)

    # ------------------------------------------------------------------
    # Main stochastic search – cheap FFT surrogate guides expensive evals.
    # ------------------------------------------------------------------
    no_improve_counter = 0
    for it in range(max_iters):
        # Choose move type.
        r = rng.random()
        if r < 0.55:
            cand = _pair_move(best, rng, scale_range=(0.02, 0.35))
        elif r < 0.80:
            cand = _block_move(best, rng, scale_range=(0.75, 1.30))
        else:
            cand = _swap_move(best, rng)

        approx = _approx_score(cand)

        # Require a relative improvement over the current best surrogate.
        if approx < best_approx * 0.9995:
            # Expensive exact evaluation.
            try:
                exact = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue
            if np.isfinite(exact) and exact < best_score:
                best = cand.copy()
                best_score = exact
                best_approx = approx
                no_improve_counter = 0
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1

        # Periodic diversification if stuck.
        if (it + 1) % 1000 == 0 or no_improve_counter >= 2000:
            # Small random perturbation of the current best.
            cand = _pair_move(best, rng, scale_range=(0.05, 0.60))
            approx = _approx_score(cand)
            try:
                exact = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue
            if np.isfinite(exact) and exact < best_score:
                best = cand.copy()
                best_score = exact
                best_approx = approx
            no_improve_counter = 0

    # ------------------------------------------------------------------
    # Final safeguard – ensure the returned sequence is valid.
    # ------------------------------------------------------------------
    # Clip to allowed range and coerce to python scalars.
    np.clip(best, 0.0, 1000.0, out=best)
    final = [float(v) for v in best]
    return final
