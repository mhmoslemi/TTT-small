def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise Exception
        n = parent_arr.size
        parent = np.clip(parent_arr, 0.0, 1000.0)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # scan power law p
    ps = np.linspace(0.5, 0.9, 41)  # step 0.01
    base = np.arange(1, n+1, dtype=np.float64)
    for p in ps:
        cand = np.power(base, -p)
        # optionally shift: cand = np.power(base + offset, -p)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Hill climbing
    min_width = max(1, n // 200)
    max_width = max(2, n // 20)
    sigma = 0.07
    for step in range(2000):
        cand = best.copy()
        r = rng.random()
        # subgradient step
        if r < 0.15:
            # compute convolution for gradient
            conv = np.convolve(cand, cand)
            k_max = int(np.argmax(conv))
            # compute gradient: for i where j=k_max-i valid
            grad = np.zeros_like(cand)
            # vectorized: find i array
            i = np.arange(n)
            j = k_max - i
            mask = (j >= 0) & (j < n)
            grad[mask] = 2 * cand[j[mask]]
            # step size adapt
            lr = 0.01 * (1.0 - step/2000)  # decaying
            cand = cand - lr * grad
        elif r < 0.85:  # multiplicative block
            width = int(rng.integers(min_width, max_width+1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start+width] *= factor
        else:  # pairwise transfer
            i_idx = int(rng.integers(0, n))
            j_idx = int(rng.integers(0, n-1))
            if j_idx >= i_idx:
                j_idx += 1
            delta = cand[i_idx] * rng.random() * 0.2
            cand[i_idx] -= delta
            cand[j_idx] += delta

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

    # final sanity
    if not np.isfinite(best_score) or best.size == 0:
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]

import numpy as np

def _power_seq(p, n):
    """Return a_i = (i+1)^{-p}, clipped to the allowed range."""
    idx = np.arange(1, n + 1, dtype=np.float64)   # i+1  (i starts at 0)
    seq = np.power(idx, -p)
    return np.clip(seq, 0.0, 1000.0)
