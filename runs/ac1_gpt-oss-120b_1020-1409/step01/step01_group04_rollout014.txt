def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)

    # get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
        cur = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        cur = np.full(n, 1.0, dtype=np.float64)

    # utilities
    m = np.fft.next_fast_len(2*n - 1)

    def conv_full(a):
        A = np.fft.rfft(a, n=m)
        c = np.fft.irfft(A * A, n=m)
        return c[:2*n-1]

    def score_of(a):
        sum_a = a.sum()
        if sum_a < 0.01:
            return np.inf
        conv = conv_full(a)
        max_b = conv.max()
        return 2.0 * n * max_b / (sum_a * sum_a)

    cur_score = score_of(cur)
    best = cur.copy()
    best_score = cur_score

    # iteration parameters
    max_iters = 800
    sigma = 0.05   # for multiplicative noise
    min_width = max(1, n // 250)   # maybe 1% length
    max_width = max(2, n // 20)    # 5% length
    transfer_frac = 0.2   # fraction of source to transfer

    for it in range(max_iters):
        cand = cur.copy()

        if rng.random() < 0.4:   # gradient-guided mass transfer
            conv = conv_full(cand)
            idx_max = int(np.argmax(conv))
            # compute gradient contributions
            i_start = max(0, idx_max - (n - 1))
            i_end = min(n - 1, idx_max)
            # indices in range [i_start, i_end]
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            grad = np.zeros(n, dtype=np.float64)
            # grad[i] = 2 * cand[idx_max - i] for i within range
            grad[idx] = 2.0 * cand[idx_max - idx]

            # pick source as index with largest grad and positive value
            src_candidates = np.where(cand > 1e-12)[0]  # non-zero
            if src_candidates.size == 0:
                continue
            # use grad to weight selection
            grad_src = grad[src_candidates]
            if grad_src.max() <= 0:
                src = rng.choice(src_candidates)
            else:
                src = src_candidates[np.argmax(grad_src)]
            # pick target as index with smallest grad (could be same as src, avoid)
            tgt_candidates = np.arange(n)
            # avoid same index
            tgt_candidates = tgt_candidates[tgt_candidates != src]
            grad_tgt = grad[tgt_candidates]
            tgt = tgt_candidates[np.argmin(grad_tgt)]

            # transfer amount
            delta = cand[src] * rng.random() * transfer_frac
            cand[src] -= delta
            cand[tgt] += delta
        else:
            # random block multiplicative perturbation
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor

        # clip to allowed range
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = score_of(cand)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # hill climbing acceptance
        if np.isfinite(cand_score) and cand_score < cur_score:
            cur = cand
            cur_score = cand_score
        # optionally, occasionally accept worse moves to escape local minima
        else:
            # probabilistic acceptance based on temperature schedule
            # simple: if random < 0.02 accept
            if rng.random() < 0.02:
                cur = cand
                cur_score = cand_score

    # final sanity check
    if not np.isfinite(best_score) or best.size == 0:
        best = cur.copy()
    return [float(x) for x in best]

import numpy as np

def _conv_full(a: np.ndarray, m: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of length m."""
    A = np.fft.rfft(a, n=m)
    c = np.fft.irfft(A * A, n=m)
    return c[: 2 * len(a) - 1]


def _score(a: np.ndarray, n: int, m: int) -> float:
    """Exact bound 2·n·max(a*a) / (sum(a)^2) with clipping."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _conv_full(a, m)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)
