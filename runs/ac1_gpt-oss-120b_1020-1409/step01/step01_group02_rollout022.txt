import numpy as np

# ----------------------------------------------------------------------
# Basic building blocks
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return the externally supplied parent sequence, or None."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:  # parent not present
        return None
    if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
        return None
    return np.clip(arr, 0.0, 1000.0)


def _arcsine_sequence(n):
    """Discrete arcsine weights a_i ≈ i^{-0.5}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    # normalise only to keep numbers moderate; scaling does not affect the score
    a /= a.sum()
    return a


def _power_sequence(n, p):
    """a_i = (i+1)^{-p}, normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= a_sum
    return a


def _exp_sequence(n, r):
    """a_i = r**i, normalised."""
    a = r ** np.arange(n, dtype=np.float64)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= a_sum
    return a


def _rev_exp_sequence(n, r):
    """a_i = r**(n-1-i), normalised."""
    a = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= a_sum
    return a


def _uniform_sequence(n):
    """Uniform distribution (sum=1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)

def _refine_by_gradient(seq, steps=800, alpha_start=0.5, alpha_end=0.01):
    """
    Greedy projected gradient descent on the maximum convolution entry.
    Returns (best_seq, best_score) seen during the walk.
    """
    n = seq.size
    # current sequence and its score
    cur = seq.copy()
    cur_score = float(evaluate_sequence(cur.tolist()))
    best_seq = cur.copy()
    best_score = cur_score

    for t in range(steps):
        conv = np.convolve(cur, cur)
        k = int(np.argmax(conv))

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = k - idx

        # gradient of the active max entry
        grad = cur[partner].copy()
        diag = idx == partner
        grad[diag] *= 2.0

        i = int(idx[grad.argmax()])     # strongest contributor
        j = int(idx[grad.argmin()])     # weakest contributor
        if i == j:
            break

        # step size that linearly decays
        alpha = alpha_start - (alpha_start - alpha_end) * (t / steps)
        delta = alpha * cur[i]
        if delta <= 0:
            break

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            break

        # accept only if we improve the current objective
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score

    return best_seq, best_score


def _hill_climb(seq, best_seq, best_score, rng, steps=500):
    """
    Light‑weight random hill‑climbing using small mass transfers.
    """
    cur = seq.copy()
    cur_score = best_score
    n = cur.size

    for _ in range(steps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = rng.random() * 0.1 * cur[i]  # move at most 10 % of source mass
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            continue
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
    return best_seq, best_score

def propose_candidate(seed=42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound as measured by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    parent = _parent_sequence()
    n = parent.size if parent is not None else 256

    # ------------------------------------------------------------------
    # Helper: evaluate and keep the best candidate seen so far
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def _consider(arr):
        nonlocal best_seq, best_score
        a = np.asarray(arr, dtype=np.float64)
        if a.shape != (n,):
            a = np.resize(a, n)
        np.clip(a, 0.0, 1000.0, out=a)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq = a.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # Initial deterministic candidates
    # ------------------------------------------------------------------
    if parent is not None:
        _consider(parent)
    _consider(_uniform_sequence(n))
    _consider(_arcsine_sequence(n))

    # ------------------------------------------------------------------
    # Parameterised families
    # ------------------------------------------------------------------
    for p in np.linspace(0.10, 2.00, 30):
        _consider(_power_sequence(n, p))
    for r in np.linspace(0.99, 0.30, 30):
        _consider(_exp_sequence(n, r))
        _consider(_rev_exp_sequence(n, r))

    # ------------------------------------------------------------------
    # Gradient‑based refinement (deterministic)
    # ------------------------------------------------------------------
    if best_seq is not None:
        refined_seq, refined_score = _refine_by_gradient(best_seq, steps=800,
                                                        alpha_start=0.5,
                                                        alpha_end=0.01)
        if refined_score < best_score:
            best_seq, best_score = refined_seq, refined_score

    # ------------------------------------------------------------------
    # Light random hill climbing to escape possible shallow minima
    # ------------------------------------------------------------------
    if best_seq is not None:
        best_seq, best_score = _hill_climb(best_seq, best_seq, best_score,
                                          rng, steps=500)

    # ------------------------------------------------------------------
    # Safety fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(n, dtype=np.float64) / n

    return [float(x) for x in best_seq]
