import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the sequence r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def propose_candidate(seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function whose AC1 bound is as low as possible.
    The function follows a cheap greedy hill‑climbing scheme that repeatedly
    adds a tiny amount to the cheapest coefficient (with respect to the current
    active maximal convolution index) while the bound improves.
    """
    # ---- initial sequence -------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        seq = parent.astype(np.float64).copy()
    else:
        n_fallback = 256
        seq = generate_r_sequence(n_fallback)

    seq = np.clip(seq, 0.0, 1000.0)
    if np.sum(seq) < 0.01:          # safeguard against degenerate start
        seq[0] = max(seq[0], 0.01)

    best = seq
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.shape[0]

    # ---- greedy improvement loop -----------------------------------------
    eps = 0.05                # step size for coefficient increments
    no_improve = 0
    max_no_improve = 500      # stop after this many consecutive failures
    while no_improve < max_no_improve and eps > 1e-6:
        # current maximal convolution index
        conv = np.convolve(best, best)
        max_idx = int(conv.argmax())

        # weight = influence on the current maximum (smaller is cheaper)
        weight = np.full(n, np.inf, dtype=np.float64)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            rev = max_idx - idx
            weight[idx] = best[rev]          # derivative ~ 2 * best[rev]

        # never touch the first coefficient (it would raise a0^2 quickly)
        if n > 0:
            weight[0] = np.inf
        # ignore entries already at the upper bound
        weight[best >= 1000.0] = np.inf

        i = int(weight.argmin())
        if weight[i] == np.inf:
            break  # no admissible index left

        cand = best.copy()
        cand[i] = min(cand[i] + eps, 1000.0)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            no_improve = 0
            # modestly increase step size after a success (but cap it)
            eps = min(eps * 1.1, 0.5)
        else:
            no_improve += 1
            # shrink step size after a failure
            eps = max(eps * 0.5, 1e-6)

    # ---- final safety check ------------------------------------------------
    if not np.isfinite(best_score):
        best = np.ones_like(best)

    return [float(x) for x in best]
