import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return the supplied parent sequence if it is valid, else None."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    return None


def _arcsine_sequence(n):
    """
    Truncated arcsine coefficients:
        a_0 = 1,
        a_i = a_{i-1} * (1 - 0.5 / i)  for i >= 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _score(seq, n):
    """
    Compute the AC1 upper bound for a sequence (already clipped).
    Return float('inf') on pathological inputs.
    """
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    M = np.max(conv)
    return float(2.0 * n * M / (s * s))


def _make_block_perturbation(base, rng):
    """Cheap random block‑wise perturbation (occasional big jump)."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetry bias
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=8000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing the AC1 upper bound.
    The routine starts from a good analytic seed (arcsine) or the provided
    parent, then performs a simple projected gradient‑descent style
    mass‑transfer optimisation with occasional random jumps.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise candidate
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    if parent is not None:
        n = parent.size
    else:
        # fallback length – a moderate size that fits comfortably in memory
        n = 4096
        parent = None

    # start from the better of parent and arcsine (if parent exists)
    cand_arcsine = _arcsine_sequence(n)
    best = cand_arcsine.copy()
    best_score = _score(best, n)

    if parent is not None:
        parent_score = _score(parent, n)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    current = best.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # occasional random big jump to escape plateaus
        if rng.random() < 0.02:
            big = _make_block_perturbation(current, rng)
            big_score = _score(big, n)
            if np.isfinite(big_score) and big_score < best_score:
                best, best_score, current, cur_score = big, big_score, big, big_score
                continue

        # convolution and location of maximal entry
        conv = np.convolve(current, current)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # indices that contribute to this maximal entry
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx_range = np.arange(lo, hi + 1, dtype=int)
        mirror = max_idx - idx_range                     # indices mirrored w.r.t. max_idx
        grad_weights = current[mirror]                   # proportional to influence on M

        # donor: pick index with large influence (weighted random)
        if grad_weights.sum() > 0.0:
            probs = grad_weights / grad_weights.sum()
            donor_local = int(rng.choice(idx_range, p=probs))
        else:
            donor_local = int(rng.choice(idx_range))

        # receiver: prefer an index that does NOT influence the current max
        outside = np.setdiff1d(np.arange(n), idx_range, assume_unique=True)
        if outside.size > 0:
            receiver = int(rng.choice(outside))
        else:
            # all indices influence the max – pick the one with minimal influence
            receiver_local = int(np.argmin(grad_weights))
            if receiver_local == donor_local:
                # fallback to a random different index
                choices = np.delete(idx_range, np.where(idx_range == donor_local))
                if choices.size == 0:
                    continue
                receiver = int(rng.choice(choices))
            else:
                receiver = int(idx_range[receiver_local])

        if donor_local == receiver:
            continue

        donor_val = current[donor_local]
        if donor_val <= 0.0:
            continue

        # step size decays linearly, bounded away from zero
        frac = 0.08 * (1.0 - step / float(num_steps)) + 0.002
        delta = donor_val * frac
        if delta <= 0.0:
            continue

        # propose mass transfer
        cand = current.copy()
        cand[donor_local] -= delta
        cand[receiver] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate new candidate
        new_score = _score(cand, n)

        # keep best ever found
        if np.isfinite(new_score) and new_score < best_score:
            best, best_score = cand, new_score

        # deterministic descent: accept only if it improves the current state
        if np.isfinite(new_score) and new_score < cur_score:
            current, cur_score = cand, new_score
        else:
            # occasional simulated‑annealing style acceptance
            temperature = 0.05 * (1.0 - step / float(num_steps)) + 1e-4
            if np.isfinite(new_score) and np.exp((cur_score - new_score) / temperature) > rng.random():
                current, cur_score = cand, new_score

        # early exit if we already beat the target bound
        if best_score <= 1.50274349:
            break

    # ------------------------------------------------------------------
    # safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
