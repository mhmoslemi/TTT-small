import numpy as np

# Helper functions (top‑level as required)

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..length‑1 (≈1/√i)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _generate_power_sequence(length: int, p: float, offset: float) -> np.ndarray:
    """a_i = 1/(i+offset)^p, normalized to have unit L1‑norm."""
    idx = np.arange(length, dtype=np.float64)
    seq = 1.0 / np.power(idx + offset, p)
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s == 0.0:
        return seq
    return seq / s


def _generate_exponential_sequence(length: int, decay: float) -> np.ndarray:
    """a_i = exp(-decay*i), normalized to unit L1‑norm."""
    idx = np.arange(length, dtype=np.float64)
    seq = np.exp(-decay * idx)
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s == 0.0:
        return seq
    return seq / s


def _pair_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from one index to another."""
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.05)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _block_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = arr.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, **kwargs):
    """
    Construct a non‑negative step‑function (list of floats) whose
    AC‑1 upper bound (as computed by evaluate_sequence) is low.
    The routine examines a few analytic families (power‑law,
    exponential, binomial) and then performs a lightweight stochastic
    refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])  # provided by the harness
    except Exception:
        n = 256  # safe fallback if parent not present

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    # 1) Binomial (the classic 1/√i shape)
    best = _generate_binomial_sequence(n).astype(np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # 2) Uniform baseline (keeps a sanity check)
    uniform = np.ones(n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # 3) Parent if available
    try:
        parent = np.maximum(height_sequence_1, 0.0).astype(np.float64)
        parent_score = float(evaluate_sequence(parent.tolist()))
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Analytic parametric search (power‑law and exponential families)
    # ------------------------------------------------------------------
    # Power‑law a_i = 1/(i+off)^p
    offsets = np.linspace(0.01, 10.0, 21)          # 21 values
    powers = np.linspace(0.30, 0.80, 15)           # 15 values
    for off in offsets:
        for p in powers:
            cand = _generate_power_sequence(n, p, off)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = cand.copy()
                best_score = sc

    # Exponential a_i = exp(-decay * i)
    decays = np.linspace(0.0002, 0.02, 20)        # 20 values
    for d in decays:
        cand = _generate_exponential_sequence(n, d)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # Light stochastic refinement (few hundred iterations)
    # ------------------------------------------------------------------
    num_steps = 500
    for step in range(num_steps):
        # Pairwise mass transfer
        cand = _pair_perturbation(best, rng)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

        # Block scaling occasional
        if step % 100 == 0:
            cand2 = _block_perturbation(best, rng)
            sc2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best = cand2
                best_score = sc2

    # ------------------------------------------------------------------
    # Return the best found vector as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best]
