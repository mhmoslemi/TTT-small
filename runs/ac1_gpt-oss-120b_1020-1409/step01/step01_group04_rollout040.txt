import numpy as np

# ----------------------------------------------------------------------
# Helper: fast full convolution via real FFT (linear, O(n log n))
# ----------------------------------------------------------------------
def full_convolve(x: np.ndarray) -> np.ndarray:
    """Return np.convolve(x, x) using FFT (real input)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    # FFT of the signal
    X = np.fft.rfft(x, n=size)
    # element‑wise product (auto‑convolution)
    Y = X * X
    # inverse FFT, keep only the first 2*n-1 entries
    conv = np.fft.irfft(Y, n=size)[: 2 * n - 1]
    return conv

            hi = min(n - 1, max_idx)
            idxs = np.arange(lo, hi + 1, dtype=np.int64)
            partners = max_idx - idxs
            contrib = cur[idxs] * cur[partners]

            if contrib.size > 0:
                k = max(1, int(frac_top * contrib.size))
                top = np.argpartition(-contrib, k - 1)[:k]
                i_reduce = idxs[top]

                cand = cur.copy()
                removed = np.sum(cand[i_reduce]) * eps
                cand[i_reduce] *= (1.0 - eps)

                # distribute the removed mass uniformly over the rest
                not_reduce = np.setdiff1d(np.arange(n), i_reduce, assume_unique=True)
                if not_reduce.size > 0:
                    cand[not_reduce] += removed / not_reduce.size
            else:                                   # fall back to a random block update
                width = int(rng.integers(block_min, block_max + 1))
                start = int(rng.integers(0, n - width + 1))
                factor = float(np.exp(rng.normal(0.0, 0.08)))
                cand = cur.copy()
                cand[start:start + width] *= factor
        else:                                       # pure stochastic perturbation
            if rng.random() < 0.7:                  # multiplicative block
                width = int(rng.integers(block_min, block_max + 1))
                start = int(rng.integers(0, n - width + 1))
                factor = float(np.exp(rng.normal(0.0, 0.07)))
                cand = cur.copy()
                cand[start:start + width] *= factor
            else:                                   # pairwise mass transfer
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n - 1))
                if j >= i:
                    j += 1
                delta = cur[i] * rng.random() * 0.15
                cand = cur.copy()
                cand[i] -= delta
                cand[j] += delta

        # ----- enforce feasibility -----
        cand = np.clip(cand, 0.0, 1000.0)

        # ----- evaluate and possibly accept -----
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score
            cur = cand
        else:
            # occasional non‑improving acceptance to escape local minima
            if rng.random() < 0.02:
                cur = cand

        # early stop if we have already crossed the target bound
        if best_score < target_bound:
            break

    # --------------------------------------------------------------
    # 3) Final sanity check – guarantee a valid return value
    # --------------------------------------------------------------
    if not np.isfinite(best_score) or best.size == 0:
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]
