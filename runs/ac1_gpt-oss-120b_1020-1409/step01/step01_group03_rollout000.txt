        cur_score = _score(x)

import numpy as np

# ------------------------------------------------------------
# Helper: generate a classic decreasing series (fallback init).
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return a_i = C(2i,i)/4^i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast convolution of a real vector with itself via FFT.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 bound (mirrors evaluate_sequence).
# ------------------------------------------------------------
def _score(x: np.ndarray) -> float:
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(x)
    m = c.max()
    return 2.0 * x.shape[0] * m / (s * s)
