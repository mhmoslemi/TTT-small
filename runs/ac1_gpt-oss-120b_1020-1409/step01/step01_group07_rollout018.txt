def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):

    sequence = [float(x) for x in sequence]

import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _power_law_sequence(n, p):
    """Generate a decreasing power‑law sequence a_i ∝ (i+1)^{-p},
    normalised so that the sum equals 1."""
    # avoid overflow for large p by using float64
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = 1.0 / np.power(idx, p)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq /= total
    return seq

def _block_perturb(seq, rng):
    """Multiply a random contiguous block by a log‑normal factor."""
    n = seq.size
    max_width = max(1, n // 100)               # ≤ 1 % of length
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.03)))   # ≈ 3 % multiplicative change
    cand = seq.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    total = cand.sum()
    if total > 0.0:
        cand /= total
    return cand

def _pair_transfer(seq, rng):
    """Move a small amount of mass from one index to another."""
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    # transfer at most 20 % of the source entry (or a tiny epsilon)
    max_delta = min(seq[i], 0.2 * seq[i])
    delta = float(rng.uniform(0.0, max_delta + 1e-12))
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    # numerical drift –  renormalise to keep the sum exactly 1
    total = cand.sum()
    if total > 0.0:
        cand /= total
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _exponent_candidate(n, rng, base_p=0.5):
    """Create a fresh power‑law candidate with a perturbed exponent."""
    # perturb around the current exponent (base_p) with a Gaussian step
    p = float(base_p + rng.normal(0.0, 0.2))
    p = max(0.01, min(5.0, p))          # keep within a sensible range
    cand = _power_law_sequence(n, p)
    return cand, p
