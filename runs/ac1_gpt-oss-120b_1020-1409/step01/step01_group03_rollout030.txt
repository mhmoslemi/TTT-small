        best_score = parent_score

    # gradient descent loop
    x = best.copy()
    for step in range(num_steps // 2):
        # compute gradient based on max conv index
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        # compute g vector: reversed slice
        # g_i = x_{k - i} if i in [max(0, k-n+1), min(k, n-1)], else 0
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g = np.zeros_like(x)
        g[idx] = x[k - idx]

        S = x.sum()
        if S == 0:
            break
        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        # derivative of f = 2*n * M / S^2 -> df = 2*n * (dM / S^2 - 2M dS / S^3)

        grad = (2.0 * n) * (g * inv_S2 - (2.0 * M) * inv_S3)

        lr = 1.0
        best_local = x

                best_local_score = sc
                break

            best_score = best_local_score

import numpy as np

# ------------------------------------------------------------
# FFT‑based self‑convolution (full length 2n‑1).
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 upper‑bound (mirrors the provided evaluator).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(seq)
    return 2.0 * seq.size * c.max() / (s * s)


# ------------------------------------------------------------
# Simple analytic families for quick candidate generation.
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i  (classical decreasing sequence)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _exp_decay(n: int, rate: float) -> np.ndarray:
    return np.exp(-rate * np.arange(n, dtype=np.float64))


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    d = np.minimum(np.arange(n, dtype=np.float64),
                   np.arange(n - 1, -1, -1, dtype=np.float64))
    return np.exp(-rate * d)


def _poly_decay(n: int, p: float) -> np.ndarray:
    return (np.arange(1, n + 1, dtype=np.float64)) ** -p


def _sym_poly_decay(n: int, p: float) -> np.ndarray:
    d = np.minimum(np.arange(n, dtype=np.float64),
                   np.arange(n - 1, -1, -1, dtype=np.float64)) + 1.0
    return d ** -p

    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size and initial parent (if any).
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
        parent = np.clip(parent, 0.0, 1000.0)
        parent_score = _score(parent)
    except Exception:
        parent = None
        n = 256
        parent_score = float("inf")

    # --------------------------------------------------------
    # Fallback analytic start (central‑binomial series).
    # --------------------------------------------------------
    best = _make_series(n)
    best_score = _score(best)

    # Prefer the parent if it is already better.
    if parent is not None and parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # --------------------------------------------------------
    # Helper: compute gradient of f = 2*n*max(conv)/sum^2.
    # --------------------------------------------------------
    def _gradient_step(x: np.ndarray):
        """Perform a single projected‑gradient step from x."""
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        # g_i = x_{k-i} whenever the index is valid
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g = np.zeros_like(x)
        g[idx] = x[k - idx]

        S = x.sum()
        if S == 0.0:
            return x, float("inf")

        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        # gradient of f (scaled by 2*n, factor irrelevant for direction)
        grad = (2.0 * n) * (g * inv_S2 - (2.0 * M) * inv_S3)

        # simple backtracking line‑search
        lr = 1.0
        cur_score = _score(x)
        best_local = x
        best_local_score = cur_score
        for _ in range(6):
            cand = x - lr * grad
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if sc < best_local_score:
                best_local = cand
                best_local_score = sc
                break
            lr *= 0.5
        return best_local, best_local_score

    # --------------------------------------------------------
    # Phase 1 – projected gradient descent (half of the budget).
    # --------------------------------------------------------
    x = best.copy()
    grad_iters = max(1, num_steps // 2)
    for step in range(grad_iters):
        x_new, sc_new = _gradient_step(x)

        # Accept improvement globally.
        if sc_new + 1e-14 < best_score:
            best = x_new.copy()
            best_score = sc_new

        # Occasionally mix a random analytic shape to escape stagnation.
        if step % 25 == 0:
            choice = rng.integers(0, 4)
            if choice == 0:
                shape = _exp_decay(n, rng.uniform(0.0005, 0.025))
            elif choice == 1:
                shape = _poly_decay(n, rng.uniform(0.2, 3.0))
            elif choice == 2:
                shape = _sym_exp_decay(n, rng.uniform(0.001, 0.03))
            else:
                shape = _sym_poly_decay(n, rng.uniform(0.3, 2.5))
            t = rng.random()
            cand = (1.0 - t) * x_new + t * shape
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _score(cand)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
                x_new = cand
        x = x_new

    # --------------------------------------------------------
    # Phase 2 – random convex combinations of analytic families.
    # --------------------------------------------------------
    rand_iters = max(1, num_steps - grad_iters)
    for _ in range(rand_iters):
        typ = rng.integers(0, 4)
        if typ == 0:
            shape = _exp_decay(n, rng.uniform(0.0005, 0.035))
        elif typ == 1:
            shape = _poly_decay(n, rng.uniform(0.1, 3.5))
        elif typ == 2:
            shape = _sym_exp_decay(n, rng.uniform(0.001, 0.04))
        else:
            shape = _sym_poly_decay(n, rng.uniform(0.2, 3.0))

        t = rng.random()
        cand = (1.0 - t) * best + t * shape
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # --------------------------------------------------------
    # Fallback safety net (should never trigger).
    # --------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(v) for v in best]
