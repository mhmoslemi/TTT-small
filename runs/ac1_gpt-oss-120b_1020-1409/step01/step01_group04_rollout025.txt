import numpy as np

# ---------- convolution (FFT, matrix‑free) ----------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, m)
    fb = fa * fa
    conv = np.fft.irfft(fb, m)[: 2 * n - 1]
    return conv

    seq = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + alpha)
    return np.clip(seq, 0.0, 1000.0)

            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.09)))
            cand = best.copy()
            cand[start:start + width] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best, best_score = cand, sc
            continue

        # 3. propose a mass‑transfer that reduces the offending product
        #    while keeping the total sum unchanged.
        min_val = min(best[i_max], best[j_max])
        epsilon = 0.10 * min_val                     # 10 % of the smaller partner
        epsilon = min(epsilon, best[i_max])           # safety clamps
        epsilon = min(epsilon, best[j_max])

        cand = best.copy()
        if i_max == j_max:
            # move a single amount from the central entry to a random other location
            r = int(rng.integers(0, n))
            while r == i_max:
                r = int(rng.integers(0, n))
            cand[i_max] = max(0.0, cand[i_max] - epsilon)
            cand[r] += epsilon
        else:
            # take epsilon from both contributors and distribute it equally elsewhere
            r = int(rng.integers(0, n))
            while r == i_max or r == j_max:
                r = int(rng.integers(0, n))
            s = int(rng.integers(0, n))
            while s == i_max or s == j_max or s == r:
                s = int(rng.integers(0, n))
            cand[i_max] = max(0.0, cand[i_max] - epsilon)
            cand[j_max] = max(0.0, cand[j_max] - epsilon)
            cand[r] += epsilon * 0.5
            cand[s] += epsilon * 0.5

        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            continue        # successful move – next iteration starts from the new state

        # 4. occasional broader perturbation (helps escape stagnation)
        if rng.random() < 0.015:
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.14)))
            cand = best.copy()
            cand[start:start + width] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best, best_score = cand, sc

    # ----- final sanity check -----
    if not np.isfinite(best_score) or best.size == 0:
        best = np.clip(np.ones(n, dtype=np.float64), 0.0, 1000.0)

    return [float(x) for x in best]
