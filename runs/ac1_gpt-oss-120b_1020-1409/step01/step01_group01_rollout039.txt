def compute_score(seq):
    n = len(seq)
    conv = np.convolve(seq, seq)
    maxb = np.max(conv)
    s = np.sum(seq)
    if s < 0.01:
        return float('inf')
    return 2.0 * n * maxb / (s * s)

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        a = parent.astype(np.float64).copy()
    else:
        n = 256
        a = np.empty(n, dtype=np.float64)
        r = 1.0
        a[0] = r
        for i in range(1, n):
            r = r * (2 * i - 1) / (2 * i)
            a[i] = r
    np.clip(a, 0.0, 1000.0, out=a)
    # ensure positive sum
    if np.sum(a) < 0.01:
        a[0] = 0.01

    best = a
    best_score = compute_score(best)

    n = best.shape[0]
    step_factor = 0.1
    min_step = 1e-6
    max_iters = 2000

    for iteration in range(max_iters):
        # recompute convolution and find last max index
        conv = np.convolve(best, best)
        max_b = conv.max()
        # find all indices where conv == max_b within tolerance
        # use np.isclose with relative tolerance
        mask = np.isclose(conv, max_b, atol=1e-12, rtol=0)
        idxs = np.flatnonzero(mask)
        k_max = idxs[-1]  # last index
        S = np.sum(best)
        thresh = 2.0 * max_b / S

        # compute dM/da approximated
        # dM_da[i] = 2*best[k_max - i] if 0 <= k_max - i < n else 0
        dM = np.zeros_like(best)
        start = max(0, k_max - (n - 1))
        # i range where i <= k_max and k_max - i < n => i >= k_max - (n-1)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        # assign values
        # for i in [i_start, i_end]: dM[i] = 2*best[k_max - i]
        # vectorized:
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        dM[idx_range] = 2.0 * best[k_max - idx_range]

        grad = dM - thresh

        # find indices with positive and negative grad
        pos_idx = np.argmax(grad)  # picks largest, may be negative if all negative
        neg_idx = np.argmin(grad)

        if grad[pos_idx] <= 0 or grad[neg_idx] >= 0:
            # no improving direction
            break

        # set delta as fraction of source amplitude
        delta = best[pos_idx] * step_factor
        if delta <= min_step:
            break
        # ensure delta does not exceed destination capacity (1000)
        # we will clip after setting.
        cand = best.copy()
        cand[pos_idx] -= delta
        cand[neg_idx] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = compute_score(cand)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # optionally increase step_factor a bit
            step_factor = min(step_factor * 1.1, 0.5)
        else:
            # reduce step factor
            step_factor *= 0.7
            if step_factor < min_step:
                break

    # fallback safety
    if not np.isfinite(best_score):
        best = np.ones_like(best, dtype=np.float64)

    return [float(x) for x in best]

import numpy as np

def _compute_score(seq: np.ndarray) -> float:
    """Score = 2 * n * max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function that improves the AC1 bound."""

    # ------------------------------------------------------------------

        r = 1.0
        a[0] = r
        for i in range(1, n):
            r = r * (2 * i - 1) / (2 * i)
            a[i] = r
    np.clip(a, 0.0, 1000.0, out=a)

    # sanity: ensure a non‑trivial sum
    if np.sum(a) < 0.01:
        a[0] = 0.01

    best = a
    best_score = _compute_score(best)
    n = best.shape[0]

    # ------------------------------------------------------------------
    # Gradient‑guided mass transfer.
    # ------------------------------------------------------------------
    max_iters = 2000
    min_delta = 1e-8
    max_delta_cap = 0.5     # never move more than this in one step

    for _ in range(max_iters):
        conv = np.convolve(best, best)
        max_b = conv.max()

        # choose the *last* index attaining the maximum (usually n‑1)
        max_idxs = np.flatnonzero(np.isclose(conv, max_b, atol=1e-12))
        if max_idxs.size == 0:
            break
        k_max = max_idxs[-1]

        total = np.sum(best)
        thr = 2.0 * max_b / total          # common term in the gradient

        # dM/da_i = 2 * best[k_max - i]   when 0 <= k_max-i < n
        dM = np.zeros_like(best)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        dM[idx_range] = 2.0 * best[k_max - idx_range]

        grad = dM - thr

        pos_candidates = np.where((grad > 0) & (best > min_delta))[0]
