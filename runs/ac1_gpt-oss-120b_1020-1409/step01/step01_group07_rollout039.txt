def proj_simplex(v):
    # v: 1D numpy array
    # returns projection onto simplex {x >=0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / np.arange(1, len(v)+1) > 0
    rho = np.where(rho_arr)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    # Input v: 1D numpy array
    # Returns projection onto simplex (non-negative, sum=1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0, return uniform
        return np.full_like(v, 1.0/len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    # ensure sum close to 1; adjust small numerical error
    s = w.sum()
    if s > 0:
        w = w / s
    else:
        w = np.full_like(v, 1.0/len(v))
    return w

import numpy as np

def _maxconv_fft(a):
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    return conv.max()

def _project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    # renorm to guard numeric drift
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0/v.size)
    return w

import numpy as np

def _maxconv_fft(a):
    """Fast max‑convolution |a*a| using zero‑padded FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power‑of‑two ≥ 2n‑1
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return conv.max()


def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0.0
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / v.size)
    return w

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # fallback – should never happen
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 2) Refine the current best by a projected sub‑gradient method.
    #    The sub‑gradient of the max‑convolution is 2·a_{k−i} for the
    #    active shift k.
    # ------------------------------------------------------------------
    max_iter = 250
    for it in range(max_iter):
        # FFT‑based self‑convolution to locate the active shift.
        m = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(best_seq, n=m)
        conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(best_seq)
        grad[mask] = 2.0 * best_seq[idx[mask]]

        # simple diminishing step size
        step = 0.6 / (it + 1)
        cand = best_seq - step * grad
        cand = _project_simplex(cand)
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()

        # occasional stochastic jitter to escape plateaus
        if (it + 1) % 50 == 0 and best_score > 1.55:
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            jitter = best_seq * noise
            jitter = _project_simplex(jitter)
            np.clip(jitter, 0.0, 1000.0, out=jitter)
            score = float(evaluate_sequence(jitter.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = jitter.copy()

    # ------------------------------------------------------------------
    # 3) Defensive finalisation.
    # ------------------------------------------------------------------
    result = best_seq.tolist()
    if not result or any(not np.isfinite(x) for x in result):
        result = [1.0 / n] * n
    return result
