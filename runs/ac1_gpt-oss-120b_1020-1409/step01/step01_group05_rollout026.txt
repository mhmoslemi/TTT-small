# Helper: max convolution value (and its index) for a real vector.
# ------------------------------------------------------------
def _conv_max(a: np.ndarray, n: int) -> tuple:
    """Return (max_conv, argmax) of a*a using an FFT (real) convolution."""
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    a_pad = np.zeros(L, dtype=np.float64)
    a_pad[:n] = a
    fa = np.fft.rfft(a_pad)
    conv = np.fft.irfft(fa * fa, n=L)
    conv = conv[: 2 * n - 1]                     # true length
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx
