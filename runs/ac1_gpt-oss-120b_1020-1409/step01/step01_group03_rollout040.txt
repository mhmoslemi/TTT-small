import numpy as np

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def _make_central_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..n-1 (positive, decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _taper_window(n: int, power: float = 2.0) -> np.ndarray:
    """Sinusoidal window w_i = sin(pi*(i+0.5)/n)^power, vanishes at ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    return w


def _conv_full(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself using FFT."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


def _score(x: np.ndarray) -> float:
    """Replicates the evaluator: 2*n*max(conv)/sum^2 with clipping."""
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float("inf")
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    m = conv.max()
    return 2.0 * n * m / (s * s)


def _grad_step(x: np.ndarray):
    """
    One projected sub‑gradient step for f = 2*n*max(conv)/sum^2.
    Returns (new_vector, new_score).
    """
    n = x.shape[0]
    conv = _conv_full(x)
    M = conv.max()
    k = int(conv.argmax())
    S = x.sum()
    if S == 0.0:
        return x, float("inf")

    # gradient of the active convolution entry
    gm = np.zeros_like(x)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    counterpart = k - idx
    gm[idx] = 2.0 * x[counterpart]

    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = gm * inv_S2 - (2.0 * M) * inv_S3  # sub‑gradient of the objective

    # simple backtracking line‑search
    best_vec = x
    best_val = _score(x)
    lr = 1.0
    for _ in range(5):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s > 0.0:
            cand = cand / s
        val = _score(cand)
        if val < best_val:
            best_vec, best_val = cand, val
            break
        lr *= 0.5
    return best_vec, best_val
