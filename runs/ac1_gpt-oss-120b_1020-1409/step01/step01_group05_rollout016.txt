        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_ok = (
            parent_arr.ndim == 1
            and parent_arr.size > 0
            and np.isfinite(parent_arr).all()
        )
        n = int(parent_arr.size)
    except Exception:
        parent_arr = None
        parent_ok = False
        n = 2000                     # fallback length

    # ----------------------------------------------------------------------
    # Helper to evaluate a candidate (list)
    # ----------------------------------------------------------------------
    def eval_seq(arr):
        return float(evaluate_sequence(arr.tolist()))

    best_seq = None
    best_score = np.inf

    # ----------------------------------------------------------------------
    # Normalise and possibly record a sequence
    # ----------------------------------------------------------------------
    def normalise_and_record(a):
        nonlocal best_seq, best_score
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.ones_like(a) / a.size
            s = a.sum()
        a = a / s                     # enforce sum == 1 (scale invariant)
        score = eval_seq(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
        return a

    # ----------------------------------------------------------------------
    # Initialise with parent if present
    # ----------------------------------------------------------------------
    if parent_ok:
        a = normalise_and_record(parent_arr)
    else:
        a = np.ones(n, dtype=np.float64) / n
        best_seq = a.copy()
        best_score = eval_seq(best_seq)

    # ----------------------------------------------------------------------
    # Generate a few parametric families and keep the best
    # ----------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # Power‑law family  (p from 0.2 to 2.0)
    for p in np.linspace(0.2, 2.0, 9):
        cand = (idx + 1.0) ** (-p)
        normalise_and_record(cand)

    # Exponential decay family
    for lam in np.linspace(0.0005, 0.01, 9):
        cand = np.exp(-lam * idx)
        normalise_and_record(cand)

    # Sinusoidal bump (positive on the whole interval)
    cand = np.sin(np.pi * (idx + 0.5) / n)
    normalise_and_record(cand)

    # Simple triangle (linear rise then fall)
    half = n // 2
    tri = np.concatenate([
        np.linspace(1.0, half, half, endpoint=False),
        np.linspace(half, 1.0, n - half)
    ])
    normalise_and_record(tri)

    # Random permutations of a power‑law base
    base = (idx + 1.0) ** -1.0
    for _ in range(5):
        cand = rng.permutation(base)
        normalise_and_record(cand)

    # ----------------------------------------------------------------------
    # Begin local optimisation from the best candidate so far
    # ----------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
    else:
        a = np.ones(n, dtype=np.float64) / n

    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two

    # hyper‑parameters
    num_iters = int(kwargs.get("num_iters", 500))
    step0 = 0.45
    tol_active = 1e-5            # relative tolerance for “active” max entries
    sum_target = 1.0

    for it in range(num_iters):
        # ------------------------------------------------------------------
        # Compute auto‑correlation using FFT (O(n log n))
        # ------------------------------------------------------------------
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]

        max_val = conv.max()
        # active indices where the convolution is within tol of the maximum
        active = np.where(conv >= max_val * (1.0 - tol_active))[0]

        # ------------------------------------------------------------------
        # Gradient of the sum of active entries (g = 2 * Σ a_{k-i})
        # ------------------------------------------------------------------
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        # ------------------------------------------------------------------
        # Multiplicative update with decaying step size
        # ------------------------------------------------------------------
        step = step0 * (0.99 ** it)
        a *= np.exp(-step * g)

        # Renormalise and enforce hard bounds
        a_sum = a.sum()
        if a_sum <= 0 or not np.isfinite(a_sum):
            a = np.ones(n, dtype=np.float64) / n
            a_sum = a.sum()
        a /= a_sum
        np.clip(a, 0.0, 1000.0, out=a)

        # ------------------------------------------------------------------
        # Occasionally inject a random block perturbation (exploration)
        # ------------------------------------------------------------------
        if (it % 50 == 0) and rng.random() < 0.3:
            width = int(rng.integers(max(1, n // 200), max(2, n // 10)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = float(np.exp(np.clip(rng.normal(0.0, 0.12), -0.5, 0.5)))
            pert = a.copy()
            pert[start:stop] *= factor
            pert_sum = pert.sum()
            if pert_sum > 0:
                pert /= pert_sum
            sc = eval_seq(pert)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = pert.copy()
                a = pert.copy()

        # ------------------------------------------------------------------
        # Periodic evaluation of the current iterate
        # ------------------------------------------------------------------
        if (it % 20) == 0:
            cur_score = eval_seq(a)
            if np.isfinite(cur_score) and cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()

        # Early exit if we already beat the target bound
        if best_score < 1.50274349:
            break

    # ----------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / n
    return [float(x) for x in best_seq]
