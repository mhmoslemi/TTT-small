import numpy as np

def propose_candidate(seed=42, num_iters=8000, top_k=3, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the AC1 upper bound.
    Uses a multiplicative mirror‑descent on the largest few convolution entries
    together with occasional random block perturbations.
    Returns a list of Python floats.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine initial sequence (parent if available)
    # ----------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.size)
        parent_ok = (
            parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # Normalisation target (scale‑invariant objective)
    sum_target = 1.0
    max_val = 1000.0

    if parent_ok:
        a = np.clip(parent_arr, 0.0, max_val)
        s = a.sum()
        if s > 0 and np.isfinite(s):
            a = a / s * sum_target
        else:
            a = np.ones(n, dtype=np.float64) / n * sum_target
    else:
        # power‑law start a_i = 1/sqrt(i+1)
        a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
        s = a.sum()
        a = a / s * sum_target

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best_score = np.inf

    # ----------------------------------------------------------------------
    # Parameters for the iterative process
    # ----------------------------------------------------------------------
    step0 = 0.70          # initial multiplicative step size
    decay = 0.998         # exponential decay per iteration
    no_improve = 0        # counter for stagnation
    stagnation_thresh = 500

    for it in range(num_iters):
        # ----- compute convolution and select top entries -----
        conv = np.convolve(a, a)
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            # unsorted top‑k indices
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]

        # ----- gradient of the sum of selected convolution entries -----
        g = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution 2*a_{k-i}
            g[idx] += 2.0 * a[k - idx]

        # ----- adaptive step size -----
        step = step0 * (decay ** it)
        if no_improve > stagnation_thresh:
            step *= 1.5  # increase step to escape plateau

        # ----- multiplicative mirror‑descent update -----
        # Clip exponent to avoid overflow/underflow extremes
        exp_factor = np.exp(-step * g)
        a = a * exp_factor

        # ----- enforce feasibility (non‑negative, bounds, sum) -----
        np.clip(a, 0.0, max_val, out=a)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.ones(n, dtype=np.float64) / n
            s = a.sum()
        a = a / s * sum_target

        # ----- evaluate current candidate -----
        cur_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score - 1e-12:
            best = a.copy()
            best_score = cur_score
            no_improve = 0
        else:
            no_improve += 1

        # ----- occasional random block perturbation -----
        if (it % 250 == 0) and rng.random() < 0.30:
            width = int(rng.integers(max(1, n // 200), max(2, n // 15)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.22)))  # log‑normal scale
            pert = a.copy()
            pert[start:start + width] *= factor
            np.clip(pert, 0.0, max_val, out=pert)
            pert_sum = pert.sum()
            if pert_sum > 0.0:
                pert = pert / pert_sum * sum_target
                pert_score = float(evaluate_sequence(pert.tolist()))
                if np.isfinite(pert_score) and pert_score < best_score:
                    best = pert.copy()
                    best_score = pert_score
                    a = pert.copy()
                    no_improve = 0

        # ----- early termination if a good bound is reached -----
        if best_score < 1.5:
            break

    # ----------------------------------------------------------------------
    # Return the best sequence found (plain Python floats)
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
