def _make_block_perturbation(base, rng):
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

import numpy as np

# -------------------------------------------------------------
# Helpers
# -------------------------------------------------------------
def _parent_sequence():
    """Return the provided parent sequence if it exists, else a fallback."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    # fallback – a short uniform vector that always satisfies the evaluator
    return np.ones(256, dtype=np.float64)


def _arcsine_sequence(n):
    """
    Generate the truncated arcsine coefficients:
        a_0 = 1,
        a_i = a_{i-1} * (1 - 0.5 / i)  for i >= 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _make_block_perturbation(base, rng):
    """A cheap O(n) random block‑wise perturbation (used for occasional jumps)."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # occasional symmetry bias
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, num_steps=6000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 upper bound.
    Starts from a sensible analytic seed (arcsine) and performs a
    gradient‑guided stochastic local search that redistributes mass away from
    the most offending convolution pair while occasionally applying a block
    perturbation for global exploration.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise candidates
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = parent.size

    # try arcsine seed of same length
    arcsine = _arcsine_sequence(n)

    # evaluate both candidates
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    arcsine_score = float(evaluate_sequence(arcsine.tolist()))
    if arcsine_score < best_score:
        best = arcsine
        best_score = arcsine_score

    current = best.copy()

    # ------------------------------------------------------------------
    # Core stochastic optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # occasional large jump to avoid stagnation
        if rng.random() < 0.07:
            cand = _make_block_perturbation(current, rng)
        else:
            # ----------------------------------------------------------
            # Pair‑mass‑transfer move guided by the current maximal entry
            # ----------------------------------------------------------
            conv = np.convolve(current, current)
            max_idx = int(np.argmax(conv))

            lo = max(0, max_idx - (n - 1))
            hi = min(n - 1, max_idx)
            idxs = np.arange(lo, hi + 1, dtype=int)
            partners = max_idx - idxs
            prods = current[idxs] * current[partners]

            # pick the pair that contributes most to the maximal convolution
            pair_loc = int(np.argmax(prods))
            i = int(idxs[pair_loc])
            j = int(partners[pair_loc])

            # pick two destination indices different from i and j,
            # and try to avoid creating a new peak at the same location
            u = int(rng.integers(0, n))
            while u == i or u == j:
                u = int(rng.integers(0, n))
            v = int(rng.integers(0, n))
            while v == i or v == j or v == u or (u + v) == max_idx:
                v = int(rng.integers(0, n))

            # amount of mass to move – a decaying fraction of the smaller entry
            min_val = min(current[i], current[j])
            if min_val <= 0.0:
                # fallback to a cheap perturbation if nothing can be moved
                cand = _make_block_perturbation(current, rng)
                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best, best_score, current = cand, score, cand
                continue

            frac = 0.5 / ((step + 1) ** 0.5)   # larger steps early, fine‑tune later
            if frac > 0.5:
                frac = 0.5
            delta_i = min_val * frac
            delta_j = min_val * frac

            cand = current.copy()
            cand[i] -= delta_i
            cand[j] -= delta_j
            cand[u] += delta_i
            cand[v] += delta_j
            np.clip(cand, 0.0, 1000.0, out=cand)

        # ------------------------------------------------------------------
        # Evaluate candidate
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            current = cand.copy()
            continue

        # simple simulated‑annealing acceptance to escape shallow minima
        temperature = 1.0 / (step + 1.0)
        if np.isfinite(score) and np.exp((best_score - score) / temperature) > rng.random():
            current = cand.copy()

    # ------------------------------------------------------------------
    # safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
