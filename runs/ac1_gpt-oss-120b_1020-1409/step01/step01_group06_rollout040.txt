def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)

    # get parent length
    try:
        arr_parent = np.asarray(height_sequence_1, dtype=np.float64)
        if arr_parent.ndim != 1 or arr_parent.size == 0 or not np.isfinite(arr_parent).all():
            raise Exception
        n = arr_parent.size
    except Exception:
        n = 256

    target_sum = np.sqrt(2.0 * n)

    # Fast conv max function
    def conv_max(a):
        n_local = a.shape[0]
        size = 1 << ((2*n_local - 1).bit_length())
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2*n_local - 1]
        return float(conv.max())

    def score(a):
        s = a.sum()
        if s == 0.0:
            return np.inf
        return float(2.0 * n * conv_max(a) / (s * s))

    # Baseline uniform
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    best_vec = uniform.copy()
    best_score = score(best_vec)

    # Powerlaw candidates as before (maybe just keep)
    # But we can skip as we have new method.

    # Sidon-based sparse candidates
    # Determine a list of K values to try, eg from 70 to 130 step 5
    K_vals = np.arange(70, min(n, 150)+1, 5)  # size range
    for K in K_vals:
        # Attempt some trials per K
        for trial in range(5):
            # Greedy random Sidon set
            indices = []
            sums_set = set()
            attempts = 0
            max_attempts = K * 20  # limit attempts
            while len(indices) < K and attempts < max_attempts:
                i = int(rng.integers(n))
                if i in indices:
                    attempts += 1
                    continue
                conflict = False
                # check sums with existing indices
                for j in indices:
                    s = i + j
                    if s in sums_set:
                        conflict = True
                        break
                if conflict:
                    attempts += 1
                    continue
                # also check double sum i+i
                if (2*i) in sums_set:
                    attempts += 1
                    continue
                # accept i
                # add new sums
                for j in indices:
                    sums_set.add(i + j)
                sums_set.add(2*i)
                indices.append(i)
                attempts = 0  # reset attempts
            if len(indices) < K:
                # failed to get full set, break
                continue
            # assign equal values
            v = target_sum / K
            cand = np.zeros(n, dtype=np.float64)
            cand[indices] = v
            sc = score(cand)
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()
            # early exit if good enough
            if best_score < 1.52:  # threshold
                break
        if best_score < 1.52:
            break

    # Additional refinement: simple local random moves to tweak best_vec
    # We'll try a few hundred random pairwise transfers to maybe improve further
    for _ in range(500):
        # pick source and target indices
        src = int(rng.integers(n))
        dst = int(rng.integers(n))
        if src == dst:
            continue
        # amount to move: up to 10% of source
        delta = best_vec[src] * rng.random() * 0.2
        if delta <= 0.0:
            continue
        cand = best_vec.copy()
        cand[src] -= delta
        cand[dst] += delta
        # clip

        # renorm sum to target

        sc = score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
        # early exit
        if best_score < 1.5:

    final = [float(x) for x in best_vec]

import numpy as np

# ----------------------------------------------------------------------
def _conv_full(arr: np.ndarray) -> np.ndarray:
    """Linear convolution arr*arr using FFT (exact)."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of full convolution arr*arr."""
    return float(_conv_full(arr).max())


def _score(arr: np.ndarray) -> float:
    """Exact value of the evaluator (without calling evaluate_sequence)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    n = arr.shape[0]
    return float(2.0 * n * _conv_max(arr) / (s * s))


def _make_powerlaw(n: int, p: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+β)^(-p) (asymmetric)."""
    idx = np.arange(n, dtype=np.float64)
    return (idx + beta) ** (-p)


def _sidon_vector(n: int, K: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Greedy construction of a sparse vector whose pairwise sums are (approximately) unique."""
    indices = []
    sums_set = set()
    attempts, max_attempts = 0, K * 25
    while len(indices) < K and attempts < max_attempts:
        i = int(rng.integers(n))
        if i in indices:
            attempts += 1
            continue
        conflict = False
        for j in indices:
            if (i + j) in sums_set:
                conflict = True
                break
        if conflict or (2 * i) in sums_set:
            attempts += 1
            continue
        # accept i
        for j in indices:
            sums_set.add(i + j)
        sums_set.add(2 * i)
        indices.append(i)
        attempts = 0
    if len(indices) < K:
        # fallback: fill remaining slots without uniqueness guarantee
        while len(indices) < K:
            i = int(rng.integers(n))
            if i not in indices:
                indices.append(i)
    v = target_sum / K
    vec = np.zeros(n, dtype=np.float64)
    vec[indices] = v
    return vec

            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand2 = arr.copy()
            cand2[start:start + width] *= factor
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            cand2 *= target_sum / cand2.sum()
            sc2 = _score(cand2)
            if sc2 < best_score:
                best_score = sc2
                arr = cand2
                best_vec = cand2.copy()
        if it % 500 == 0 and best_score < 1.5:
            break

    # ------------------------------------------------------------------
    # final result as a plain Python list (evaluator will clamp and validate)
    return [float(x) for x in best_vec]
