import numpy as np

def _generate_binomial_sequence(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i-1] * ((2.0*i_f-1.0)*(2.0*i_f)) / (4.0*i_f*i_f)

    max_b = conv.max()

import numpy as np


def _generate_binomial_sequence(length: int) -> np.ndarray:
    """Return a_i = C(2i,i)/4^i for i=0..length-1 (float64)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _score(arr: np.ndarray, n: int) -> float:
    """Exact AC1 upper bound used by the evaluator."""
    s = arr.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(arr, arr)
    return 2.0 * n * conv.max() / (s * s)

    Return a list of non‑negative floats (step‑function heights) with a
    low AC1 upper bound.  The routine starts from the parent sequence
    ``height_sequence_1`` if present, otherwise from a binomial sequence,
    and performs a cheap greedy descent that always moves mass away from
    the shift producing the current maximal autocorrelation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    try:
        n = int(height_sequence_1.shape[0])

    except Exception:                     # parent not available
        n = 256
