    Uses a soft‑max‑weighted multiplicative update that flattens the
    autocorrelation, with occasional random block perturbations.

            parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if parent_ok:
        a = np.clip(parent_arr, 0.0, 1000.0)
        s = a.sum()
        if s > 0 and np.isfinite(s):
            a = a / s

            a = np.ones(n, dtype=np.float64) / n
    else:

        tau = max(1.0, n / 10.0)
        a = np.exp(-np.arange(n, dtype=np.float64) / tau)
        a = a / a.sum()

    best = a.copy()
    # initialise best score using the official evaluator (once)
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Hyper‑parameters.
    # ------------------------------------------------------------------

    step_decay = 0.998    # gradual decay
    beta = 1.0            # soft‑max sharpness
    beta_factor = 1.02
    beta_max = 20.0
    target_score = 1.50274349  # early‑stop threshold
