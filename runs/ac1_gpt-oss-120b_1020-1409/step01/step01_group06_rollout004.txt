import numpy as np

# ----------------------------------------------------------------------
def _conv_max(arr: np.ndarray) -> float:
    """Maximum of full linear convolution arr * arr using FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray) -> float:
    """Exact evaluator (without calling evaluate_sequence)."""
    s = arr.sum()
    if s <= 0.0:
        return float("inf")
    n = arr.shape[0]
    return float(2.0 * n * _conv_max(arr) / (s * s))


# ----------------------------------------------------------------------
def _order_and_normalize(raw: np.ndarray, rng: np.random.Generator,
                         target_sum: float) -> np.ndarray:
    """
    Turn a raw non‑negative vector into a candidate:
    * sort descending,
    * interleave high / low values,
    * apply a random circular shift,
    * clip to [0,1000] and renormalise to target_sum.
    """
    # descending sort
    desc = np.sort(raw)[::-1]

    n = desc.size
    out = np.empty_like(desc)

    # interleave: big values on even positions, small on odd
    half = (n + 1) // 2
    out[::2] = desc[:half]                     # big half
    out[1::2] = desc[half:][::-1]              # small half reversed

    # random roll to move large values away from the edges
    shift = int(rng.integers(n))
    out = np.roll(out, shift)

    # clip and renormalise
    np.clip(out, 0.0, 1000.0, out=out)
    s = out.sum()
    if s > 0.0:
        out *= target_sum / s
    return out


def _block_perturb(vec: np.ndarray, rng: np.random.Generator,
                   target_sum: float, sigma: float) -> np.ndarray:
    """Multiplicative block perturbation with optional random swap."""
    cand = vec.copy()
    n = cand.size

    # block width in a reasonable range
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    if max_w <= min_w:
        width = min_w
    else:
        width = int(rng.integers(min_w, max_w))
    start = int(rng.integers(0, n - width + 1))

    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:start + width] *= factor

    # occasional simple swap to change ordering
    if rng.random() < 0.2:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        cand[i], cand[j] = cand[j], cand[i]

    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand

def propose_candidate(seed=42, **kwargs):
    """
    Construct a non‑negative step function that aims to minimise the
    AC1 upper bound (the evaluator returns ``2 * n * max(conv) / sum^2``).
    The method builds a variety of analytically‑generated shapes,
    applies a high/low interleaving ordering, optionally rotates them,
    and finally performs a modest stochastic refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n and a default candidate (fallback to uniform)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    target_sum = np.sqrt(2.0 * n)

    # guarantee a valid starting candidate from the provided parent
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 0.01:
        best_vec = np.full(n, target_sum / n, dtype=np.float64)
    else:
        best_vec = parent * (target_sum / parent.sum())
    best_score = _score(best_vec)

    # ------------------------------------------------------------------
    # Grid of analytical shapes (power‐law and exponential)
    p_vals = np.linspace(0.30, 1.50, 13)                 # power‑law exponents
    beta_vals = np.logspace(-2, 2, 12)                  # offsets
    gamma_vals = np.linspace(0.0005, 0.020, 12)         # exponential rates

    idx = np.arange(n, dtype=np.float64)

    # ----- power‑law shapes ------------------------------------------------
    for p in p_vals:
        for beta in beta_vals:
            raw = (idx + beta) ** (-p)
            cand = _order_and_normalize(raw, rng, target_sum)
            sc = _score(cand)
            if sc < best_score:
                best_score, best_vec = sc, cand.copy()
            # a couple of random shuffles as cheap alternatives
            for _ in range(2):
                cand_shuf = np.take(cand, rng.permutation(n))
                sc = _score(cand_shuf)
                if sc < best_score:
                    best_score, best_vec = sc, cand_shuf.copy()

    # ----- exponential shapes ---------------------------------------------
    for gamma in gamma_vals:
        raw = np.exp(-gamma * idx)
        cand = _order_and_normalize(raw, rng, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
        for _ in range(2):
            cand_shuf = np.take(cand, rng.permutation(n))
            sc = _score(cand_shuf)
            if sc < best_score:
                best_score, best_vec = sc, cand_shuf.copy()

    # ------------------------------------------------------------------
    # Stochastic refinement (hill‑climbing with decreasing step size)
    ITER = 1500
    for it in range(ITER):
        sigma = 0.12 * (1.0 - it / ITER) ** 1.5   # annealed magnitude
        cand = _block_perturb(best_vec, rng, target_sum, sigma)

        # occasional extra rotation to move large entries away from the ends
        if rng.random() < 0.15:
            cand = np.roll(cand, int(rng.integers(n)))

        sc = _score(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()

    # ------------------------------------------------------------------
    # Final safety check – the evaluator will clamp anyway, but we keep
    # the vector inside the expected bounds.
    final = [float(x) for x in best_vec]
    return final
