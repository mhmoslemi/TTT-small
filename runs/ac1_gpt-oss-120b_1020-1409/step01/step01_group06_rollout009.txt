def _guided_perturb(arr, rng, target_sum, size, factor_range):
    cand = arr.copy()
    # compute convolution and max index
    fa = np.fft.rfft(cand, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*len(arr)-1]
    k = int(conv.argmax())
    lo = max(0, k - (len(arr)-1))
    hi = min(len(arr)-1, k)
    # apply reduction factor to subrange [lo, hi]
    # choose a subblock within range to reduce more strongly
    width = int(rng.integers(max(1, (hi-lo+1)//4), max(2, (hi-lo+1)//2)))
    start = int(rng.integers(lo, hi - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))  # maybe around 0.9 typical
    cand[start:start+width] *= factor
    # renorm
    s = cand.sum()
    if s>0: cand *= target_sum / s
    return cand, k

    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # first 2*len(arr)-1 are the linear convolution values
    l = 2 * arr.shape[0] - 1
    return float(conv[:l].max())

def _normalize(vec, target):
    s = vec.sum()
    if s > 0:
        vec *= target / s
    else:
        # avoid zero division, set uniform
        vec[:] = target / vec.shape[0]
    return vec

    """Maximum value of the linear convolution arr*arr using FFT."""
    fa = np.fft.rfft(arr, n=fft_size)
    conv = np.fft.irfft(fa * fa, n=fft_size)
    length = 2 * arr.shape[0] - 1
    return float(conv[:length].max())


def _normalize(vec: np.ndarray, target: float) -> np.ndarray:
    """Scale vec to have L1‑norm = target (in‑place)."""
    s = vec.sum()
    if s > 0.0:
        vec *= target / s
    else:
        vec[:] = target / vec.shape[0]
    return vec

def propose_candidate(seed=42, **kwargs):
    """
    Construct a non‑negative step‑function (list of floats) that
    minimises the AC1 upper bound used by `evaluate_sequence`.
    The routine
    * tries a variety of analytic shapes,
    * keeps the best one,
    * then refines it with many cheap stochastic perturbations,
    * finally returns the best vector found.
    """
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = int(parent.size)
        use_parent = True
    except Exception:
        n = 256
        parent = None
        use_parent = False

    rng = np.random.default_rng(seed)

    target_sum = np.sqrt(2.0 * n)                # the normalisation enforced by the evaluator
    fft_size = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n‑1

    # ------------------------------------------------------------------
    # Helper to evaluate a vector quickly (max‑convolution only)
    def _max_conv(v):
        return _conv_full_max(v, fft_size)

    # ------------------------------------------------------------------
    # Start with a uniform fallback (score = 2.0)
    best_vec = np.full(n, target_sum / n, dtype=np.float64)
    best_max = _max_conv(best_vec)
    best_score = 2.0 * n * best_max / (target_sum * target_sum)

    # ------------------------------------------------------------------
    # Include the supplied parent (if any) as an initial candidate
    if use_parent:
        cand = parent.astype(np.float64, copy=True)
        cand = _normalize(cand, target_sum)
        cand_max = _max_conv(cand)
        cand_score = 2.0 * n * cand_max / (target_sum * target_sum)
        if cand_score < best_score:
            best_score, best_max, best_vec = cand_score, cand_max, cand

    # ------------------------------------------------------------------
    # Deterministic shape families (all will be L1‑normalised)
    idx = np.arange(n, dtype=np.float64)

    shape_makers = []

    # Triangle (linear ramp up then down)
    shape_makers.append(lambda i=idx: np.minimum(i + 1, n - i))

    # Hann window
    shape_makers.append(lambda i=idx: 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1))))

    # Half‑sine
    shape_makers.append(lambda i=idx: np.sin(np.pi * i / (n - 1)))

    # Exponential decays
    for a in (0.001, 0.003, 0.005, 0.01, 0.02, 0.05):
        shape_makers.append(lambda a=a, i=idx: np.exp(-a * i))

    # Gaussian‑like blobs
    for sf in (0.12, 0.15, 0.20, 0.25):
        sigma = sf * n
        shape_makers.append(
            lambda sigma=sigma, i=idx: np.exp(-0.5 * ((i - (n - 1) / 2.0) / sigma) ** 2)
        )

    # Power‑law family (both asymmetric and symmetric)
    p_vals = (0.3, 0.4, 0.5, 0.6, 0.7)
    beta_vals = (0.01, 0.1, 1.0, 10.0, 100.0)
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                if sym:
                    shape_makers.append(
                        lambda p=p, beta=beta, i=idx: (i + beta) ** (-p) + (i[::-1] + beta) ** (-p)
                    )
                else:
                    shape_makers.append(lambda p=p, beta=beta, i=idx: (i + beta) ** (-p))

    # ------------------------------------------------------------------
    # Evaluate all deterministic shapes
    for maker in shape_makers:
        vec = maker()
        if not np.isfinite(vec).all():
            continue
        vec = _normalize(vec, target_sum)
        m = _max_conv(vec)
        sc = 2.0 * n * m / (target_sum * target_sum)
        if sc < best_score:
            best_score, best_max, best_vec = sc, m, vec

    # ------------------------------------------------------------------
    # Stochastic refinement (gradient‑free block updates)
    max_iter = 3000
    for _ in range(max_iter):
        # current convolution (needed to locate the peak)
        fa = np.fft.rfft(best_vec, n=fft_size)
        conv = np.fft.irfft(fa * fa, n=fft_size)[: 2 * n - 1]
        cur_max = float(conv.max())
        cur_idx = int(conv.argmax())

        # decide move type
        if rng.random() < 0.2:          # guided move aiming at the current peak
            lo = max(0, cur_idx - (n - 1))
            hi = min(n - 1, cur_idx)
            rng_range = hi - lo + 1
            if rng_range > 0:
                w_low = max(1, rng_range // 4)
                w_high = max(2, rng_range // 2)
                width = int(rng.integers(w_low, w_high))
                start = int(rng.integers(lo, hi - width + 2))
                # bias factor < 1 to shrink the contribution to the peak
                factor = float(np.exp(rng.normal(-0.12, 0.04)))
                cand = best_vec.copy()
                cand[start:start + width] *= factor
            else:
                cand = best_vec.copy()
        else:                          # generic random block perturbation
            w_low = max(1, n // 200)
            w_high = max(2, n // 12)
            width = int(rng.integers(w_low, w_high))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.06)))
            cand = best_vec.copy()
            cand[start:start + width] *= factor
            if rng.random() < 0.3:      # occasional symmetrisation
                cand = 0.5 * (cand + cand[::-1])

        # occasional global symmetrisation (helps many shapes)
        if rng.random() < 0.1:
            cand = 0.5 * (cand + cand[::-1])

        # renormalise to the required L1 norm and enforce bound
        cand = _normalize(cand, target_sum)
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate candidate
        cand_max = _max_conv(cand)
        cand_score = 2.0 * n * cand_max / (target_sum * target_sum)

        if cand_score < best_score:
            best_score, best_max, best_vec = cand_score, cand_max, cand
            if best_score < 1.50:           # early stop when already excellent
                break

    # ------------------------------------------------------------------
    # Return a plain Python list – the harness will re‑evaluate it
    return [float(x) for x in best_vec]
