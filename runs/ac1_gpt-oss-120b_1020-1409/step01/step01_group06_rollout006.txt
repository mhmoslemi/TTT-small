def _conv_full(arr):
    n = arr.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv
```

Normalization function:

```
def _renorm(arr, target):
    s = arr.sum()
    if s > 0:
        arr *= target / s
    else:
        # avoid zero sum: fill uniform
        arr[:] = target / arr.size
    return arr
```

Block perturb function:

```
def _block_perturb(arr, rng, target):
    n = arr.size
    cand = arr.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.05)))  # slightly larger sigma?
    cand[start:start+width] *= factor
    if rng.random() < 0.4:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    _renorm(cand, target)

import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_full(arr: np.ndarray) -> np.ndarray:
    """Full linear convolution arr*arr using FFT (real, exact for given size)."""
    n = arr.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full convolution arr*arr."""
    return float(_conv_full(arr).max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact evaluator without calling the external function."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


def _renorm(arr: np.ndarray, target: float) -> None:
    """Scale arr in‑place so that its L1‑norm equals target (if possible)."""
    s = arr.sum()
    if s > 0.0:
        arr *= target / s
    else:
        # Fallback to uniform if all zeros (should never happen)
        arr[:] = target / arr.size

def _block_perturb(arr: np.ndarray, rng: np.random.Generator, target: float) -> np.ndarray:
    """Random block scaling, keeping total sum fixed."""

    start = int(rng.integers(0, n - width + 1))

    cand[start : start + width] *= factor

    """
    Identify the shift giving max convolution and attenuate the most
    contributing entries.

    n = arr.size
    conv = _conv_full(arr)
    max_idx = int(np.argmax(conv))

    # Contribution of each index to the max shift
    contrib = np.zeros(n, dtype=np.float64)
    i_min = max(0, max_idx - (n - 1))
    i_max = min(n - 1, max_idx)
    i_vals = np.arange(i_min, i_max + 1, dtype=np.int64)
    j_vals = max_idx - i_vals
    contrib[i_vals] = arr[i_vals] * arr[j_vals]
