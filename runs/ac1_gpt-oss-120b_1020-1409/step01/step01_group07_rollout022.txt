def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        n = int(height_sequence_1.shape[0])
        parent = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)
    # Ensure non-negative and <=1000 already.
    # Normalized sum to 1.
    def normalize(seq):
        s = seq.sum()
        if s == 0:
            return np.full_like(seq, 1.0/seq.size)
        seq = seq / s
        return seq
    # List of initial candidates
    candidates = []
    # parent normalized
    cand = normalize(parent.copy())
    candidates.append(cand)
    # uniform
    candidates.append(np.full(n, 1.0/n, dtype=np.float64))
    # inverse sqrt shape
    invsqrt = 1.0/np.sqrt(np.arange(1, n+1, dtype=np.float64))
    candidates.append(normalize(invsqrt))
    # exponential decays
    for alpha in [0.0005, 0.001, 0.002, 0.005, 0.01]:
        expdec = np.exp(-alpha * np.arange(n, dtype=np.float64))
        candidates.append(normalize(expdec))
    # Evaluate
    best_seq = None
    best_score = np.inf
    for seq in candidates:
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
    if best_seq is None:
        best_seq = np.full(n, 1.0/n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    # Gradient descent loop
    sum_target = best_seq.sum()
    # Ensure sum_target >= 0.01
    if sum_target < 0.01:
        sum_target = 0.01
        best_seq = best_seq * (sum_target / best_seq.sum())
    # Precompute FFT length L
    L = 1 << ((2*n - 1).bit_length())
    step = 0.5 / n   # initial step size
    decay = 0.995
    for it in range(500):
        # Compute convolution via FFT
        fa = np.fft.rfft(best_seq, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[:2*n-1]
        max_idx = int(np.argmax(conv_full))
        max_val = conv_full[max_idx]
        # Compute gradient for this max index
        g = np.zeros_like(best_seq)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end+1)
            g[i_range] = 2.0 * best_seq[max_idx - i_range]
        # Update with subgradient step
        new_seq = best_seq - step * g
        # Clip and renormalize
        np.clip(new_seq, 0.0, 1000.0, out=new_seq)
        s = new_seq.sum()
        if s > 0.0:
            new_seq *= (sum_target / s)
        # Evaluate
        score = float(evaluate_sequence(new_seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = new_seq
            # maybe increase step forward? (keep same)
        else:
            # if not improved, reduce step
            step *= 0.7
        # Decay step regardless
        step *= decay
        # Occasionally random perturb to escape stagnation
        if it % 50 == 0 and rng.random() < 0.3:
            width = int(rng.integers(1, max(2, n // 50) + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.02)))
            pert = best_seq.copy()
            pert[start:start+width] *= factor
            np.clip(pert, 0.0, 1000.0, out=pert)
            pert *= (sum_target / pert.sum())
            s2 = float(evaluate_sequence(pert.tolist()))
            if np.isfinite(s2) and s2 < best_score:
                best_score = s2
                best_seq = pert
    # Final check: ensure valid
    result = best_seq.tolist()
    if not result or any(not np.isfinite(x) for x in result):
        result = [1.0 / n] * n
    return result

import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _normalize(seq: np.ndarray, target: float = 1.0) -> np.ndarray:
    """Scale seq so that its sum equals target (preserves non‑negativity)."""
    s = seq.sum()
    if s == 0.0:
        return np.full_like(seq, target / seq.size, dtype=np.float64)
    seq = seq * (target / s)
    return seq


def _initial_candidates(n: int, parent_seq: np.ndarray) -> list:
    """
    Produce a short list of plausible starting shapes.
    All candidates are normalized to sum = 1.0.
    """
    cand = []

    # 1. parent (if available)
    if parent_seq is not None:
        cand.append(_normalize(parent_seq.copy()))

    # 2. uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3. inverse‑sqrt
    inv_sqrt = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    cand.append(_normalize(inv_sqrt))

    # 4. a few exponential decays
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01):
        exp_decay = np.exp(-lam * np.arange(n, dtype=np.float64))
        cand.append(_normalize(exp_decay))

    return cand


def _conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution a*a using FFT of length L (L >= 2*len(a)-1).
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    return conv


def _gradient_for_max(a: np.ndarray, idx_max: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry with respect to a.
    For the selected index idx_max the derivative is
        ∂b_idx_max/∂a_i = 2 * a_{idx_max-i}   (if the index lies inside the array)
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx_max - (n - 1))
    i_end = min(n - 1, idx_max)
    if i_start <= i_end:
        i = np.arange(i_start, i_end + 1)
        grad[i] = 2.0 * a[idx_max - i]
    return grad
