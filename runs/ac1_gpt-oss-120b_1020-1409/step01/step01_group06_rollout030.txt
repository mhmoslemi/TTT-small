import numpy as np

# ----------------------------------------------------------------------
# Efficient max of self‑convolution using FFT (exact for real non‑negative data)
def _conv_max(arr: np.ndarray, size: int) -> float:
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * arr.size - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
def _score(arr: np.ndarray, n: int, size: int) -> float:
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr, size)
    return float(2.0 * n * max_b / (s * s))

# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base

# ----------------------------------------------------------------------
def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    cand = vec.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    cand[start: start + width] *= factor
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand
