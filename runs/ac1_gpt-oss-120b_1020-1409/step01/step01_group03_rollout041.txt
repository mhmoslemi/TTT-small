import numpy as np

# ------------------------------------------------------------
# Helpers
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Classic decreasing series a_i = C(2i,i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself using FFT."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound (mirrors evaluate_sequence)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)


def _grad_step(x: np.ndarray, lr: float) -> np.ndarray:
    """
    One projected gradient step (no line‑search).
    Returns a new candidate (already clipped).
    """
    n = x.size
    conv = _conv_self(x)
    k = int(conv.argmax())
    M = conv[k]
    S = x.sum()
    if S == 0.0:
        return x.copy()

    # g_i = a_{k-i} on the overlap interval
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    g = np.zeros_like(x)
    g[idx] = x[k - idx]

    # descent direction (gradient of f without the constant factor)
    delta = g / (S * S) - M * x / (S * S * S)

    cand = x - lr * delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Returns a list of non‑negative heights whose AC1 bound improves on the
    previous best (target < 1.50274349).  The routine runs several short
    gradient‑descent runs from different initialisations and keeps the best
    candidate found.
    """
    rng = np.random.default_rng(seed)

    # ------- problem size -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
        parent_ok = True
    except Exception:
        parent = None
        n = 256
        parent_ok = False

    # ------- initialise a collection of starting points --------------------
    starts = []

    # parent (if available)
    if parent_ok:
        starts.append(np.clip(parent, 0.0, 1000.0))

    # classic analytic series
    starts.append(_make_series(n))

    # simple geometric decay (r close to 1 keeps numbers moderate)
    r = 0.9995
    geom = np.power(r, np.arange(n, dtype=np.float64))
    starts.append(geom)

    # random uniform seed
    rand = rng.random(n, dtype=np.float64)
    starts.append(rand)

    # small‑amplitude sine‑shaped bump (symmetric, edges near zero)
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    sine = np.sin(theta)
    starts.append(sine)

    best_seq = None
    best_score = float('inf')

    # ------- optimisation parameters ---------------------------------------
    max_outer = len(starts)          # one optimisation run per start
    inner_steps = 2000               # gradient steps per run
    base_lr = 1.0                    # initial learning rate for back‑tracking

    for run in range(max_outer):
        x = starts[run].copy()
        # keep a copy of the best inside this run
        run_best = x.copy()
        run_best_score = _score(x)

        stagnant = 0
        for step in range(inner_steps):
            # --------------------------------------------------------------
            # Gradient step with simple back‑tracking line search
            # --------------------------------------------------------------
            conv = _conv_self(x)
            M = conv.max()
            k = int(conv.argmax())
            S = x.sum()
            if S == 0.0:
                break

            # form g on the active overlap interval
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            idx = np.arange(i_low, i_high + 1, dtype=int)
            g = np.zeros_like(x)
            g[idx] = x[k - idx]

            # descent direction (gradient of f divided by 4*n)
            delta = g / (S * S) - M * x / (S * S * S)

            # back‑tracking line search
            lr = base_lr
            cur_score = 2.0 * n * M / (S * S)   # same as _score but cheap
            improved = False
            for _ in range(12):
                cand = x - lr * delta
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _score(cand)
                if sc + 1e-14 < cur_score:
                    x = cand
                    cur_score = sc
                    improved = True
                    break
                lr *= 0.5

            # --------------------------------------------------------------
            # If gradient step fails, try a random block perturbation
            # --------------------------------------------------------------
            if not improved:
                width = max(1, n // rng.integers(12, 33))
                start = rng.integers(0, n - width + 1)
                block = x.copy()
                factor = np.exp(rng.normal(0.0, 0.12))
                block[start:start + width] *= factor
                if rng.random() < 0.2:
                    block = 0.5 * (block + block[::-1])
                np.clip(block, 0.0, 1000.0, out=block)
                sc_block = _score(block)
                if sc_block + 1e-14 < cur_score:
                    x = block
                    cur_score = sc_block
                    improved = True

            # --------------------------------------------------------------
            # Update run‑best
            # --------------------------------------------------------------
            if cur_score < run_best_score - 1e-14:
                run_best = x.copy()
                run_best_score = cur_score
                stagnant = 0
            else:
                stagnant += 1

            # optional symmetrisation (helps flatten the autocorrelation)
            if rng.random() < 0.08:
                x = 0.5 * (x + x[::-1])

            # early stop if nothing improves for a while
            if stagnant > 250:
                break

        # --------------------------------------------------------------
        # Keep global best across all runs
        # --------------------------------------------------------------
        if run_best_score < best_score:
            best_seq = run_best
            best_score = run_best_score

        # stop early if we already beat the target bound
        if best_score < 1.50274349:
            break

    # --------------------------------------------------------------
    # Fallback safety net (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(256, dtype=np.float64)

    # Return as plain Python list of floats
    return [float(v) for v in best_seq]
