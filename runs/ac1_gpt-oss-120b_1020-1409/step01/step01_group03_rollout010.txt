import numpy as np

def _conv_self_fft(x):
    n = x.size
    conv_len = 2*n -1
    fft_len = 1 << ((conv_len-1).bit_length())
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx*fx, n=fft_len)[:conv_len]
    return conv

def _score(x):
    seq = np.clip(x,0.0,1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    conv = _conv_self_fft(seq)
    m = conv.max()
    return 2.0 * seq.size * m / (s*s)

def propose_candidate(seed=42, max_iters=3000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim!=1 or parent.size==0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        n = 256
        parent = None

    # Starting seq
    if parent is not None:
        seq = parent.copy()
    else:
        # use C(2i,i)/4^i series
        seq = np.empty(n, dtype=np.float64)
        seq[0]=1.0
        for i in range(1,n):
            seq[i] = seq[i-1] * (2*i-1) / (2*i)
        # Optionally add some small random noise
    # Ensure within bounds
    np.clip(seq,0.0,1000.0,inplace=True)

    best_seq = seq.copy()
    best_score = _score(best_seq)

    m = np.zeros_like(seq) # first moment
    v = np.zeros_like(seq) # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # main loop
    for it in range(max_iters):
        t+=1
        # compute conv and M
        conv = _conv_self_fft(seq)
        M = conv.max()
        # find indices close to max within tolerance
        tol = 1e-9
        max_idxs = np.where(conv >= M - tol)[0]  # probably single index
        # compute gradient contributions aggregated
        # g_i = average of a_{k-i} over all k in max_idxs
        g = np.zeros_like(seq)
        for k in max_idxs:
            i_low = max(0, k - (n-1))
            i_high = min(n-1, k)
            idx = np.arange(i_low, i_high+1, dtype=int)
            # For each idx j, k-j within range yields a_{k-j}
            g[idx] += seq[k - idx]
        # average if multiple max_idxs
        if len(max_idxs) > 0:
            g /= len(max_idxs)
        # compute gradient of f (ignoring constant 2n)
        S = seq.sum()
        if S == 0:
            break
        inv_S2 = 1.0 / (S*S)
        inv_S3 = inv_S2 / S
        # gradient = 2*g*inv_S2 - 2*M*inv_S3 (since gradient of M = 2*a_{k-i})
        # Actually we used g = a_{k-i}, so incorporate factor 2
        grad_f = 2.0 * g * inv_S2 - 2.0 * M * inv_S3

        # Adam update
        m = beta1 * m + (1-beta1) * grad_f
        v = beta2 * v + (1-beta2) * (grad_f*grad_f)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size
        lr = 0.05 / (np.sqrt(v_hat) + eps)
        cand = seq - lr * m_hat
        np.clip(cand, 0.0, 1000.0, out=cand)

        # symmetrize
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])

        sc = _score(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            # Reset momentum maybe?
        seq = cand

        # Occasionally random block perturb
        if it % 100 == 0:
            width = max(1, n // rng.integers(8,20))
            start = rng.integers(0, n-width+1)
            factor = np.exp(rng.normal(0.0, 0.2))
            block = seq.copy()
            block[start:start+width] *= factor
            np.clip(block,0.0,1000.0, out=block)
            # maybe symmetrize half block
            if rng.random()<0.5:
                block = 0.5 * (block + block[::-1])
            scb = _score(block)
            if scb < best_score:
                best_seq = block.copy()
                best_score = scb
                seq = block

    # final fallback if best_score invalid
    if best_seq.size == 0 or not np.isfinite(best_score):
        best_seq = np.ones(256, dtype=np.float64)

    return [float(v) for v in best_seq]

import numpy as np

_MAX_VAL = 1000.0


def _make_series(length: int) -> np.ndarray:
    """Return the classic a_i = C(2i,i)/4^i sequence of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself using an FFT (real inputs)."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    # Clip small negatives caused by round‑off.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _score(seq: np.ndarray) -> float:
    """Exact copy of the evaluator's scoring formula."""
    seq = np.clip(seq, 0.0, _MAX_VAL)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = _conv_self(seq)
    m = conv.max()
    return 2.0 * seq.size * m / (s * s)

    """Matrix‑free projected Adam optimiser for the AC1 bound."""
    seq = np.clip(initial.astype(np.float64), 0.0, _MAX_VAL)
    n = seq.size

    v = np.zeros_like(seq)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05
    best_seq = seq.copy()
    best_score = _score(best_seq)
    t = 0

    for it in range(max_iters):
        t += 1
        conv = _conv_self(seq)
        M = conv.max()

        tol = max(1e-12, M * 1e-12)
        max_idxs = np.where(conv >= M - tol)[0]

        g = np.zeros_like(seq)
