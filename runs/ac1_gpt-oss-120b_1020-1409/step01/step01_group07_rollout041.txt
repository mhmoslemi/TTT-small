   # Simple hill climbing for limited steps

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _normalize_to_one(arr: np.ndarray) -> np.ndarray:
    """Scale arr so that its L1‑norm equals 1 (if possible)."""
    total = arr.sum()
    if total <= 0.0:
        # fallback to uniform if something went wrong
        return np.full_like(arr, 1.0 / arr.size, dtype=np.float64)
    return arr / total


def _evaluate_score(arr: np.ndarray) -> float:
    """Return the AC1 upper‑bound score for a numpy array."""
    # The evaluator expects a plain list of scalars.
    return float(evaluate_sequence(arr.tolist()))


def _perturb_block(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative block perturbation, keeps the sum near 1."""
    n = seq.size
    # block width up to 1 % of n, at least 1
    max_width = max(1, n // 100)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    # log‑normal factor, ~3 % relative change
    factor = float(np.exp(rng.normal(0.0, 0.03)))
    cand = seq.copy()
    cand[start:start + width] *= factor
    # enforce evaluator bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    # renormalise to keep the scale fixed (scale independent for the score)
    cand = _normalize_to_one(cand)
    return cand

    """

    # Determine length (use the supplied parent if available)
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])          # parent supplied by harness
    except Exception:
        n = 256                                      # safe fallback length

    # Initial candidates

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    candidates = [uniform]

        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size == n:
            candidates.append(_normalize_to_one(parent))
    except Exception:
        pass  # parent not available or malformed

    # 3. Inverse‑square‑root shape (used in the reference solution)
