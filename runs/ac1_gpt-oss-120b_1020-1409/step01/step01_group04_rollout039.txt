import numpy as np

def _power_seq(alpha, p, n):
    seq = (np.arange(n, dtype=np.float64) + alpha) ** (-p)
    return np.clip(seq, 0.0, 1000.0)

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine n from height_sequence_1 if available
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim != 1 or parent_arr.size == 0:
            raise Exception
        n = parent_arr.size
    except Exception:
        n = 256
        parent_arr = np.ones(n, dtype=np.float64)

    best = np.clip(parent_arr, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # Scan parametric families
    alphas = np.logspace(-2, 2, num=15, dtype=np.float64)
    ps = np.linspace(0.55, 0.9, num=8, dtype=np.float64)  # exclude 0.5 since already parent maybe
    for p in ps:
        for alpha in alphas:
            cand = _power_seq(alpha, p, n)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = cand, score

    # Hill climbing
    min_width = max(1, n // 200)      # ~0.5%
    max_width = max(2, n // 20)       # up to ~5%
    sigma = 0.09
    num_steps = 3000

    for step in range(num_steps):
        cand = best.copy()
        if rng.random() < 0.68:  # block factor
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start+width] *= factor
        else:  # pairwise transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n-1))
            if j >= i:
                j += 1
            delta = best[i] * rng.random() * 0.15
            cand[i] -= delta
            cand[j] += delta

        # occasional symmetry enforce
        if rng.random() < 0.04:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

    if not np.isfinite(best_score) or best.size == 0:
        best = np.clip(parent_arr, 0.0, 1000.0)

    return [float(x) for x in best]

import numpy as np

# ----------------------------------------------------------------------
# Helper: generate power‑law step heights a_i = (i + alpha)^{-p}
# ----------------------------------------------------------------------
def _gen_power(alpha: float, p: float, n: int) -> np.ndarray:
    """Return array a_i = 1/(i+alpha)^p, clipped to the allowed range."""
    # i runs from 0 to n-1
    idx = np.arange(n, dtype=np.float64) + alpha
    # avoid div‑by‑zero; alpha > 0 in all calls
    seq = np.power(idx, -p, out=np.empty_like(idx))
    # evaluator caps values between 0 and 1000, enforce the same here
    return np.clip(seq, 0.0, 1000.0)
