            cand = np.clip(cand + jitter, 0.0, 1000.0)

        # Occasionally enforce symmetry (helps often).
        if rng.random() < 0.08:
            cand = 0.5 * (cand + cand[::-1])

        cand_score = _calc_score(cand)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
        else:
            # occasional uphill move to escape plateaus
            if rng.random() < 0.02:
                cur = cand
                cur_score = cand_score

    # ------------------------------------------------------------------
    # 3) Final short stochastic refinement (block scaling).
    # ------------------------------------------------------------------
    min_width = max(1, n // 200)
    max_width = max(2, n // 20)
    for _ in range(150):
        cand = best.copy()
        width = int(rng.integers(min_width, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.04))
        cand[start:start + width] *= factor
        cand = np.clip(cand, 0.0, 1000.0)

        # occasional symmetry enforcement
        if rng.random() < 0.1:
            cand = 0.5 * (cand + cand[::-1])

        score = _calc_score(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Safety fallback.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score) or best.size == 0:
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]
