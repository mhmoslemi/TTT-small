    m = 2*n - 1
    nfft = 1 << (m-1).bit_length()  # next power of two >= m
    A = np.fft.rfft(a, n=nfft)
    B = A * A
    conv = np.fft.irfft(B, n=nfft)[:m]
    max_idx = conv.argmax()
    max_val = conv[max_idx]
    sum_a = a.sum()

def compute_score_and_grad(a, nfft):
    n = a.shape[0]
    A = np.fft.rfft(a, n=nfft)
    B = A * A
    conv = np.fft.irfft(B, n=nfft)[:2*n - 1]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    sum_a = float(a.sum())
    if sum_a < 0.01:
        score = np.inf
    else:
        score = 2.0 * n * max_val / (sum_a * sum_a)
    # gradient of max_val:
    grad_M = np.zeros_like(a)
    lo = max_idx - (n - 1)
    if lo < 0:
        lo = 0
    hi = max_idx
    if hi >= n:
        hi = n - 1
    # Note: hi may be less than lo, but not typical
    if lo <= hi:
        idx = np.arange(lo, hi+1)
        rev_idx = max_idx - idx
        # rev_idx should be within [0,n-1]
        grad_M[idx] = 2.0 * a[rev_idx]
    # full gradient of score:
    if sum_a > 0:
        coeff = 2.0 * n / (sum_a * sum_a)
        grad_score = coeff * grad_M - (4.0 * n * max_val / (sum_a ** 3)) * np.ones_like(a)

        grad_score = np.zeros_like(a)
    return score, max_idx, max_val, grad_M, grad_score

import numpy as np

def _generate_binomial_sequence(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i-1] * ((2.0*i_f - 1.0) * (2.0*i_f)) / (4.0 * i_f * i_f)
    return a

def _pair_perturbation(arr, rng):
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.05)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand

def _block_perturbation(arr, rng):
    n = arr.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:start+width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _targeted_pair_perturbation(arr, rng, max_idx):
    n = arr.size
    lo = max_idx - (n - 1)
    if lo < 0:
        lo = 0
    hi = max_idx
    if hi >= n:
        hi = n - 1
    donors = np.arange(lo, hi + 1)
    donors = donors[arr[donors] > 0.0]
    if donors.size == 0:
        return _pair_perturbation(arr, rng)
    i = int(rng.choice(donors))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    delta = arr[i] * rng.uniform(0.001, 0.05)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand

def _project_and_normalize(arr, target_sum):
    np.clip(arr, 0.0, 1000.0, out=arr)
    total = arr.sum()
    if total > 0.0:
        arr *= (target_sum / total)
    else:
        # fallback uniform
        n = arr.size
        arr[:] = target_sum / n
    return arr

import numpy as np


def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..length-1, the classic “binomial” start."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _pair_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small random fraction of mass from one index to another."""
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.05)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _block_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = arr.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _targeted_pair_perturbation(arr: np.ndarray,
                               rng: np.random.Generator,
                               max_idx: int) -> np.ndarray:
    """Move mass from the overlap region that creates the maximal convolution."""
    n = arr.size
    lo = max_idx - (n - 1)
    if lo < 0:
        lo = 0
    hi = max_idx
    if hi >= n:
        hi = n - 1
    donors = np.arange(lo, hi + 1)
    donors = donors[arr[donors] > 0.0]
    if donors.size == 0:
        return _pair_perturbation(arr, rng)
    i = int(rng.choice(donors))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    delta = arr[i] * rng.uniform(0.001, 0.05)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _project_and_normalize(arr: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and renormalise the total mass to target_sum."""
    np.clip(arr, 0.0, 1000.0, out=arr)
    total = arr.sum()
    if total > 0.0:
        arr *= (target_sum / total)
    else:  # degenerate – fall back to a uniform distribution
        n = arr.size
        arr[:] = target_sum / n
    return arr


def _conv_max_and_score(a: np.ndarray, nfft: int):
    """
    Fast full convolution a*a via FFT, returning the full array,
    the index and value of its maximum, and the AC1 bound score.
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.shape[0] - 1]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    sum_a = float(a.sum())
    if sum_a < 0.01:
        score = np.inf
    else:
        n = a.shape[0]
        score = 2.0 * n * max_val / (sum_a * sum_a)
    return conv, max_idx, max_val, score
