import numpy as np, math, random

def generate_r_sequence(n):
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2*i - 1) / (2*i)
        seq[i] = r
    return seq

def evaluate_score(arr):
    n = arr.size
    b = np.convolve(arr, arr)
    max_b = b.max()
    sum_a = arr.sum()
    return 2 * n * max_b / (sum_a * sum_a)
