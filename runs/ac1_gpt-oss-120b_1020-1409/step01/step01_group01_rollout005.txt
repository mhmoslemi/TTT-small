def generate_r_sequence(n):
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def propose_candidate(seed=42, **kwargs):

    # Determine initial sequence
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:

        n = seq.shape[0]
    else:
        n = int(kwargs.get("n", 256))
        seq = generate_r_sequence(n)
    # Clip to allowed range
    np.clip(seq, 0.0, 1000.0, out=seq)

    # initial best
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # If initial invalid, fallback uniform
    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    max_iters = 150  # safety bound
    for _ in range(max_iters):
        # convolution and relevant statistics
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        max_B = float(conv[max_idx])
        sum_a = float(np.sum(seq))
        if sum_a < 1e-12:
            break
        # paired coefficients for each position relative to max_idx
        i = np.arange(n)
        partner_idx = max_idx - i
        partner = np.zeros_like(seq)
        valid = (partner_idx >= 0) & (partner_idx < n)
        partner[valid] = seq[partner_idx[valid]]

        # indices where increasing the entry reduces the objective
        threshold = max_B / sum_a
        inc_mask = partner < threshold
        if not np.any(inc_mask):
            break

        # choose step size proportional to threshold, limited to avoid large jumps
        eps = min(0.01, threshold * 0.5)
        if eps <= 0:
            break

        # produce candidate
        cand = seq.copy()
        cand[inc_mask] += eps
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:

            break
        cand = seq.copy()
        cand[inc_mask] += eps
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            seq = cand
            best_seq = cand.copy()
            best_score = cand_score
            continue

        # no improvement possible
        break

    # ------------------------------------------------------------------
    # optional random fine‑tuning (few cheap perturbations)
    # ------------------------------------------------------------------
    for _ in range(120):
        idx = rng.integers(0, n)
        delta = rng.uniform(-0.5, 0.5)
        cand = best_seq.copy()
        cand[idx] = max(0.0, min(1000.0, cand[idx] + delta))
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq = cand
            best_score = cand_score

    # ------------------------------------------------------------------
    # final safety check
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_seq]
