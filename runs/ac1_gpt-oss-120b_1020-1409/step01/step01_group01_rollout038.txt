def greedy_augment(seq, eps, direction='head'/'tail'):
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    k = 0
    step = max(1, n // 20)
    while step >= 1:

        while True:

            if new_k == k:
                break

            if direction == 'head':
                candidate[1 + k : 1 + new_k] += eps
            else:  # tail
                candidate[-new_k:] += eps
            np.clip(candidate, 0.0, 1000.0, out=candidate)
            score = evaluate_sequence(candidate.tolist())
            if np.isfinite(score) and score < best_score:
                best_seq = candidate
                best_score = score
                k = new_k
                improved = True
            else:
                break
        if not improved:
            step //= 2
    return best_seq, best_score

import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Generate the classic r‑sequence: r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def _score(seq: np.ndarray) -> float:
    """Wrapper around the supplied evaluator – returns a float score."""
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed: int = 42, **kwargs):
    """Return a step‑function sequence with a low AC1 bound.

    Strategy:
      * Start from the provided parent sequence if present, otherwise from the
        r‑sequence.
      * Greedily increase the sum of the coefficients while keeping the maximal
        autocorrelation low.
      * Perform two passes:
          - a head pass (indices increasing from the start)
          - a tail pass (indices decreasing from the end) repeated until
            no further improvement.
    """
    # ------------------------------------------------------------------
    # 1. Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        best = np.clip(parent.astype(np.float64).copy(), 0.0, 1000.0)
        n = int(best.shape[0])
    else:
        n = 256
        best = generate_r_sequence(n)
        best = np.clip(best, 0.0, 1000.0)

    best_score = _score(best)

    # ------------------------------------------------------------------
    # 2. Greedy head augmentation (mirrors the baseline program) -----
    # ------------------------------------------------------------------
    epsilons = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for eps in epsilons:
        k = 0
        step = max(1, n // 20)
        while step >= 1:
            improved = False
            while True:
                new_k = min(k + step, n - 1)          # never touch index 0
                if new_k == k:
                    break
                cand = best.copy()
                cand[1 + k : 1 + new_k] += eps
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _score(cand)
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc
                    k = new_k
                    improved = True
                else:
                    break
            if not improved:
                step //= 2

    # ------------------------------------------------------------------
    # 3. Greedy tail augmentation (repeated passes) -------------------
    # ------------------------------------------------------------------
    max_cycles = 10  # safety bound; will stop earlier if no improvement
    for _ in range(max_cycles):
        any_improved = False
        for eps in epsilons:
            k = 0
            step = max(1, n // 20)
            while step >= 1:
                improved = False
                while True:
                    new_k = min(k + step, n - 1)      # block size (excluding index 0)
                    if new_k == k:
                        break
                    cand = best.copy()
                    # augment the last `new_k` entries (i.e. indices n‑new_k … n‑1)
                    cand[-new_k:] += eps
                    np.clip(cand, 0.0, 1000.0, out=cand)
                    sc = _score(cand)
                    if np.isfinite(sc) and sc < best_score:
                        best = cand
                        best_score = sc
                        k = new_k
                        improved = True
                        any_improved = True
                    else:
                        break
                if not improved:
                    step //= 2
        if not any_improved:
            break   # no further improvement in this whole cycle

    # ------------------------------------------------------------------
    # 4. Fallback (should never trigger) -------------------------------
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    # Return a plain Python list of floats
    return [float(x) for x in best]
