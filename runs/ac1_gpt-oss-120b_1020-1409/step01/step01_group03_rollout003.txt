def _conv_self(x):
    n = x.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv

   max_lr = 10.0

        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())

import numpy as np

# ------------------------------------------------------------
# Helper: classic binomial series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast self–convolution via real FFT.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact AC1 upper bound (mirrors evaluate_sequence).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(seq)
    return 2.0 * seq.size * c.max() / (s * s)


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_steps: int = 3000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the previous bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size and initial candidate.
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
        init = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        init = np.clip(_make_series(n), 0.0, 1000.0)

    # --------------------------------------------------------
    # Keep the best solution seen so far.
    # --------------------------------------------------------
    best = init.copy()
    best_score = _score(best)

    # --------------------------------------------------------
    # Optimization state.
    # --------------------------------------------------------
    x = best.copy()
    lr = 1.0               # current learning rate
    min_lr = 1e-6
    max_lr = 10.0
    stagnant = 0           # steps without improvement of global best

    for step in range(max_steps):
        # ----- gradient computation -----
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        S = x.sum()
        if S < 1e-12:
            break  # degenerate, shouldn't happen

        # shift1[i] = x[k-i] where indices are valid
        shift1 = np.zeros_like(x)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        if i_low <= i_high:
            idx = np.arange(i_low, i_high + 1, dtype=int)
            shift1[idx] = x[k - idx]

        # shift2[i] = x[i-k] for i>=k (if k < n)
        shift2 = np.zeros_like(x)
        if k < n:
            shift2[k:] = x[: n - k]

        gM = shift1 + shift2                     # ∂M/∂a
        grad = gM / (S * S) - 2.0 * M / (S * S * S)   # ∂f/∂a up to a constant factor

        # ----- line search -----
        improved = False
        cand = None
        lr_try = lr
        for _ in range(8):
            cand = x - lr_try * grad
            np.clip(cand, 0.0, 1000.0, out=cand)

            # occasional symmetrisation (helps flatten the convolution)
            if rng.random() < 0.07:
                cand = 0.5 * (cand + cand[::-1])

            sc = _score(cand)
            if sc < best_score - 1e-12:   # better than global best
                improved = True
                break
            # also accept a move that improves over the current point
            if sc < _score(x) - 1e-12:
                improved = True
                break
            lr_try *= 0.5
            if lr_try < min_lr:
                break

        # ----- fallback random block perturbation if gradient step failed -----
        if not improved:
            width = max(1, n // rng.integers(15, 30))
            start = rng.integers(0, n - width + 1)
            block = x.copy()
            factor = np.exp(rng.normal(0.0, 0.12))
            block[start:start + width] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            sc = _score(block)
            if sc < best_score - 1e-12:
                cand = block
                sc = _score(cand)
                improved = True
            else:
                cand = x  # stay where we are
                sc = _score(cand)

        # ----- accept new point -----
        x = cand
        cur_score = sc

        # update global best
        if cur_score < best_score - 1e-12:
            best = x.copy()
            best_score = cur_score
            stagnant = 0
            lr = min(max_lr, lr * 1.05)   # gently increase step size on success
        else:
            stagnant += 1
            lr = max(min_lr, lr * 0.97)   # decay on failure

        # occasional extra symmetrisation (does not affect score directly)
        if rng.random() < 0.04:
            x = 0.5 * (x + x[::-1])

        # early termination if no progress for a while
        if stagnant > 500:
            break

    # --------------------------------------------------------
    # Safety fallback – ensure a valid, non‑empty result.
    # --------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.full(n, 0.1, dtype=np.float64)

    return [float(v) for v in best]
