    n = len(x) + len(y) - 1

import numpy as np

# ------------------------------------------------------------
# 1. Basic building blocks
# ------------------------------------------------------------

def _make_series(length: int) -> np.ndarray:
    """Truncate the classic a_i = C(2i,i)/4^i series."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (real FFT, O(n log n))."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors (FFT based)."""
    n = len(x) + len(y) - 1
    fft_len = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)[:n]
    return conv


def _bound(seq: np.ndarray) -> float:
    """Exact AC1 upper bound (mirrors evaluate_sequence)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    m = c.max()
    return float(2.0 * seq.size * m / (s * s))


# ------------------------------------------------------------
# 2. Sub‑gradient of the max‐convolution term
# ------------------------------------------------------------

def _subgrad(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of M = max_k (a*a)[k] with respect to a.
    Uses all indices attaining the maximum (within a tiny tolerance).
    Returns a vector g of length n where g_i = 2*sum_{k in K} a_{k-i}.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    M = conv.max()
    eps = max(1e-12, M * 1e-12)
    K = np.where(conv >= M - eps)[0]          # positions attaining the max

    ind = np.zeros(2 * n - 1, dtype=np.float64)
    ind[K] = 1.0                               # indicator of the active set

    a_rev = a[::-1]                           # reversed view
    full = _conv_fft(ind, a_rev)              # length 3n‑2
    start = n - 1
    g = 2.0 * full[start:start + n]           # extract the needed slice
    return g

    s = a.sum()
    g = _subgrad(a)
    avg_g = g.mean()
    # direction with zero‑sum (so total mass is unchanged before clipping)
    d = -g + avg_g

    max_abs = np.abs(d).max()
    if max_abs == 0.0:
        return a, _bound(a)

    # base step scale – we change each coordinate by at most ~½ of the mean value
    base_eta = 0.5 * (s / n) / max_abs

    best_local = a
    best_score = _bound(a)

    for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
        eta = base_eta * factor
        cand = a + eta * d
        np.clip(cand, 0.0, 1000.0, out=cand)

        # clipping may alter the total mass – re‑normalise to the original sum
        sc = cand.sum()
        if sc == 0.0:
            continue
        cand *= (s / sc)

        scnd = _bound(cand)
        if scnd < best_score:
            best_local = cand
            best_score = scnd
            break
    return best_local, best_score


# ------------------------------------------------------------
# 4. Main entry point
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, num_steps: int = 1200, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    AC1 upper bound.  The routine starts from the classic C(2i,i)/4^i
    series (or the provided parent if it is already better) and then
    runs a sum‑preserving projected gradient descent using the full
    sub‑gradient of the max‑convolution term.
    """
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        n = 256
        parent = None

    # ----- initial candidate --------------------------------------------
    a = _make_series(n)
    if parent is not None:
        p = np.clip(parent, 0.0, 1000.0)
        if _bound(p) < _bound(a):
            a = p.copy()

    best = a.copy()
    best_score = _bound(best)

    # ----- optimisation loop --------------------------------------------
    stagnant = 0
    for step in range(num_steps):
        cand, cand_score = _optim_step(best, rng)

        # occasional random block perturbation (helps escape shallow plateaus)
        if step % 13 == 0:
            width = max(1, n // rng.integers(12, 33))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            tmp = cand.copy()
            tmp[start:start + width] *= factor
            np.clip(tmp, 0.0, 1000.0, out=tmp)
            tmp *= (cand.sum() / tmp.sum())
            tmp_score = _bound(tmp)
            if tmp_score < cand_score:
                cand, cand_score = tmp, tmp_score

        if cand_score + 1e-14 < best_score:
            best = cand
            best_score = cand_score
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 250:
            break

    # ----- safety net ----------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(v) for v in best]
