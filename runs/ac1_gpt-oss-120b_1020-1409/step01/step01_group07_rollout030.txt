import numpy as np

def _analytic_score(c, d, n):
    """Fast O(n) upper‑bound computation for a_i = exp(-c*i/n) + d."""
    i = np.arange(n, dtype=np.float64)
    base = np.exp(-c * i / n)               # decreasing exponential part
    S = base.sum()                          # sum of the exponential part

    # convolution lengths
    k = np.arange(2 * n - 1, dtype=np.float64)

    # number of overlapping terms for each shift k
    L = np.where(k < n, k + 1, 2 * n - 1 - k)   # shape (2n-1,)

    # exponential factor for each shift (common to all i in the overlap)
    exp_term = np.exp(-c * k / n)

    # prefix sums of base for fast range‑sum queries
    pref = np.concatenate(([0.0], np.cumsum(base)))   # length n+1

    lo = np.maximum(0, k - n + 1).astype(np.int64)
    hi = np.minimum(k, n - 1).astype(np.int64)
    sum_region = pref[hi + 1] - pref[lo]   # sum of base over the overlap

    # total convolution value at each shift:
    #   base*base   → L * exp_term
    #   2*d*base*1 → 2*d * sum_region
    #   d*d*1*1    → d*d * L
    conv = L * exp_term + 2.0 * d * sum_region + d * d * L
    max_b = conv.max()

    total = S + n * d                      # full sum after adding the constant offset
    if total <= 0.0:
        return np.inf
    return 2.0 * n * max_b / (total * total)


def _perturb(seq, rng):
    """Small multiplicative block perturbation (keeps values non‑negative)."""
    n = seq.size
    max_width = max(1, n // 200)          # at most 0.5 % of the length
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.015)))   # ~1.5 % log‑normal change
    cand = seq.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
