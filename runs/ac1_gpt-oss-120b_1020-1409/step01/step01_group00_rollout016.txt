import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        f = float(i)
        a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
    return a

def _autoconv(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of seq with itself using a zero‑padded FFT."""
    n = seq.shape[0]
    L = 2 * n                     # length after zero padding
    fft = np.fft.rfft(seq, L)
    conv = np.fft.irfft(fft * fft, L)
    return conv[: 2 * n - 1]      # keep the linear part

def _score(seq: np.ndarray) -> float:
    """Exact objective used by the evaluator (no clipping needed here)."""
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    n = seq.shape[0]
    max_b = _autoconv(seq).max()
    return 2.0 * n * max_b / (s * s)
