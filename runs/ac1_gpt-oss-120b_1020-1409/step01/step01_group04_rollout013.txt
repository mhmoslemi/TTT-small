def propose_candidate(seed=42, **kwargs):

    import numpy as np, math

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
        base_seq = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        base_seq = np.ones(n, dtype=np.float64)

    target_sum = math.sqrt(2.0 * n)

    # Normalize base_seq to target_sum
    total = base_seq.sum()
    if total > 0:
        base_seq = base_seq * (target_sum / total)
    base_seq = np.clip(base_seq, 0.0, 1000.0)
    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # Helper functions
    def normalize(seq):
        tot = seq.sum()
        if tot <= 0:
            return seq
        seq = seq * (target_sum / tot)
        return np.clip(seq, 0.0, 1000.0)

    def max_lag(seq):
        n = seq.size
        L = 1 << ((2*n-1).bit_length())
        fft_seq = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(fft_seq * fft_seq, n=L)
        conv = conv[:2*n-1]
        idx = int(np.argmax(conv))
        val = float(conv[idx])
        return idx, val

    # Scan parameter families
    # power law
    alphas = np.linspace(0.0, 10.0, 31)
    ps = np.linspace(0.4, 1.8, 15)
    for a in alphas:
        for p in ps:
            seq = (np.arange(n, dtype=np.float64) + a) ** (-p)
            seq = np.clip(seq, 0.0, 1000.0)
            seq = normalize(seq)
            sc = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_seq, best_score = seq, sc

    # exponential family
    lambdas = np.logspace(-4, -0.3, 12)
    for lam in lambdas:
        seq = np.exp(-lam * np.arange(n, dtype=np.float64))
        seq = np.clip(seq, 0.0, 1000.0)
        seq = normalize(seq)
        sc = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq, best_score = seq, sc

    # hill climbing
    num_steps = 12000
    min_width = max(1, n // 250)
    max_width = max(2, n // 15)
    sigma = 0.06
    transfer_frac = 0.2

    for it in range(num_steps):
        cand = best_seq.copy()
        r = rng.random()

import numpy as np
import math

def _clip_and_normalize(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale seq to have the required sum and clip to allowed range."""
    total = seq.sum()
    if total > 0.0:
        seq = seq * (target_sum / total)
    return np.clip(seq, 0.0, 1000.0)


def _max_convolution(seq: np.ndarray):
    """Return (index, value) of the maximal entry of seq*seq (full convolution)."""
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    conv = conv[: 2 * n - 1]                      # true full convolution length
    idx = int(np.argmax(conv))
    val = float(conv[idx])
    return idx, val


def _score(seq: np.ndarray) -> float:
    """Exact score used by the evaluator (avoids calling it repeatedly)."""
    total = seq.sum()
    if total < 0.01:
        return float("inf")
    _, max_val = _max_convolution(seq)
    return 2.0 * seq.size * max_val / (total * total)
