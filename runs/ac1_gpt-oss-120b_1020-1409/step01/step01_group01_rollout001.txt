import numpy as np
import math

# ----------------------------------------------------------------------
# Helper: generate the classic r‑sequence a_i = C(2i,i)/4^i
def generate_r_sequence(n: int) -> np.ndarray:
    """Return a decreasing non‑negative sequence of length n."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

# ----------------------------------------------------------------------
# Helper: single‑pass greedy block augmentation (like the original program)
def greedy_augment(seq: np.ndarray, start_score: float) -> tuple[np.ndarray, float]:
    """
    Try to increase early entries (excluding index 0) by tiny epsilons.
    Returns (new_seq, new_score) where new_score <= start_score.
    """
    best_seq = seq.copy()
    best_score = start_score
    epsilons = [0.01, 0.005, 0.002, 0.001]          # decreasing steps
    n = len(best_seq)

    for eps in epsilons:
        k = 0                # size of already‑augmented block (excluding index 0)
        step = max(1, n // 20)
        while step >= 1:
            improved = False
            while True:
                new_k = min(k + step, n - 1)      # we never touch index 0
                if new_k == k:
                    break
                cand = best_seq.copy()
                cand[1 + k : 1 + new_k] += eps
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_seq = cand
                    best_score = sc
                    k = new_k
                    improved = True
                else:
                    break
            if not improved:
                step //= 2
    return best_seq, best_score

def propose_candidate(seed: int = 42, **kwargs):

    Returns a list of non‑negative step‑function heights.
    The routine starts from the classical r‑sequence (or the provided
    parent if it is better) and performs a short simulated‑annealing search
    followed by a deterministic greedy refinement.
    """

    # Determine problem size and a possible parent sequence
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256

    # ------------------------------------------------------------------
    # Build two initial candidates: r‑sequence and (if present) the parent
    r_seq = generate_r_sequence(n)
    best_seq = r_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.shape[0] == n:
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best_seq = parent.astype(np.float64).copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # Simulated annealing (very lightweight)
    rng = np.random.default_rng(seed)
    cur_seq = best_seq.copy()
    cur_score = best_score

    temperature = cur_score * 0.1 if cur_score > 0 else 1.0
    decay = 0.9995
    max_iter = 4000                     # tuned to stay well within time limits

    for _ in range(max_iter):
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose a relative move; scale by the larger of the two current values
        scale = max(cur_seq[i], cur_seq[j], 1e-8)
        delta = rng.normal(loc=0.0, scale=0.05 * scale)   # about ±5 % of the magnitude

        new_i = cur_seq[i] + delta
        new_j = cur_seq[j] - delta
        if new_i < 0.0 or new_j < 0.0 or new_i > 1000.0 or new_j > 1000.0:
            # reject moves that violate bounds immediately
            temperature *= decay
            continue

        cand[i] = new_i
        cand[j] = new_j

        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
