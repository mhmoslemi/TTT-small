import numpy as np


def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..length‑1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        f = float(i)
        a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
    return a


def _solve_lp_for_g(f: np.ndarray, M: float, rng: np.random.Generator,
                    m: int = 150) -> np.ndarray | None:
    """
    Solve   max 1·g   s.t. 0≤g≤1000,  (f*g)[k] ≤ M   for a small random set
    of m convolution indices k.  Returns the optimal g or None if the
    solver fails.
    """
    n = f.size
    # pick m distinct indices from the full convolution length (2n‑1)
    ks = rng.choice(2 * n - 1, size=min(m, 2 * n - 1), replace=False)

    # Build A (m×n) where  (f*g)[k] = Σ_j f[k‑j] g_j .
    A = np.zeros((ks.size, n), dtype=np.float64)
    for row, k in enumerate(ks):
        i0 = max(0, k - (n - 1))
        i1 = min(k, n - 1)
        # j runs from i0..i1, and the coefficient for g_j is f[k‑j].
        j = np.arange(i0, i1 + 1)
        A[row, i0:i1 + 1] = f[k - j]

    b_ub = np.full(ks.size, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  == minimise −Σ g
    bounds = [(0.0, 1000.0) for _ in range(n)]

    try:
        from scipy.optimize import linprog
        sol = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if sol.success:
            return sol.x
    except Exception:
        pass
    return None


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 bound
    is as low as possible within the allowed resources.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initial vector (use the provided parent if available)
    try:
        n = int(height_sequence_1.shape[0])
        a = height_sequence_1.astype(np.float64).copy()
    except Exception:
        n = 256
        a = _generate_binomial_sequence(n)

    np.clip(a, 0.0, 1000.0, out=a)

    # ------------------------------------------------------------------
    # helper to compute the bound directly (avoids repeated calls to
    # evaluate_sequence which are O(n²)).
    def _bound(vec, conv):
        s = vec.sum()
        if s < 0.01:
            return float('inf')
        return 2.0 * n * conv.max() / (s * s)

    # initial convolution and bound
    conv = np.convolve(a, a)
    best = a.copy()
    best_score = _bound(best, conv)

    # ------------------------------------------------------------------
    # 1) LP‑based mixing steps (few iterations, cheap constraints)
    for _ in range(4):                      # at most four mixing cycles
        M = conv.max()
        g = _solve_lp_for_g(best, M, rng, m=150)
        if g is None:
            break
        sum_best = best.sum()
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # rescale g to keep the total mass unchanged
        scale = sum_best / sum_g
        g = g * scale

        # convex combination – t = 0.2 is a safe step size
        t = 0.2
        cand = (1.0 - t) * best + t * g
        np.clip(cand, 0.0, 1000.0, out=cand)

        conv_cand = np.convolve(cand, cand)
        cand_score = _bound(cand, conv_cand)

        if cand_score < best_score:
            best = cand
            conv = conv_cand
            best_score = cand_score
        else:
            break

    # ------------------------------------------------------------------
    # 2) Targeted pair‑mass transfers (fast O(n) updates)
    #    Move mass from the index that contributes most to the current
    #    maximal convolution entry toward the globally smallest entry.
    max_iter = 5000
    stale = 0
    for _ in range(max_iter):
        k_max = int(conv.argmax())

        # indices that take part in the maximal coefficient
        i0 = max(0, k_max - (n - 1))
        i1 = min(k_max, n - 1)
        i_vals = np.arange(i0, i1 + 1)
        j_vals = k_max - i_vals
        prod = best[i_vals] * best[j_vals]

        donor = int(i_vals[np.argmax(prod)])

        # recipient = index with smallest coefficient (break ties arbitrarily)
        recipient = int(np.argmin(best))
        if recipient == donor:
            # second smallest if the smallest coincides with donor
            order = np.argsort(best)
            recipient = int(order[1] if n > 1 else donor)

        # transfer amount – modest fraction of donor, obeying the 1000‑clip
        delta = min(best[donor] * 0.5, 0.1)                 # ≤0.1 per step
        delta = min(delta, 1000.0 - best[recipient])
        if delta <= 0.0:
            stale += 1
            if stale > 300:
                break
            continue

        # incremental update of the full convolution (O(n))
        new_conv = conv.copy()
        new_conv[donor:donor + n] -= 2.0 * delta * best
        new_conv[recipient:recipient + n] += 2.0 * delta * best
        idx = 2 * donor
        if idx < new_conv.size:
            new_conv[idx] += delta * delta
        idx = 2 * recipient
        if idx < new_conv.size:
            new_conv[idx] += delta * delta
        idx = donor + recipient
        if idx < new_conv.size:
            new_conv[idx] -= 2.0 * delta * delta

        new_score = _bound(best, new_conv)   # sum unchanged

        if new_score < best_score:
            best[donor] -= delta
            best[recipient] += delta
            conv = new_conv
            best_score = new_score
            stale = 0
        else:
            stale += 1
            if stale > 200:
                # occasional random perturbation to escape possible plateaus
                i, j = rng.integers(0, n, size=2)
                if i != j and best[i] > 0.0:
                    d = best[i] * rng.uniform(0.001, 0.02)
                    d = min(d, 1000.0 - best[j])
                    if d > 0.0:
                        tmp_conv = conv.copy()
                        tmp_conv[i:i + n] -= 2.0 * d * best
                        tmp_conv[j:j + n] += 2.0 * d * best
                        k = 2 * i
                        if k < tmp_conv.size:
                            tmp_conv[k] += d * d
                        k = 2 * j
                        if k < tmp_conv.size:
                            tmp_conv[k] += d * d
                        k = i + j
                        if k < tmp_conv.size:
                            tmp_conv[k] -= 2.0 * d * d
                        tmp_score = _bound(best, tmp_conv)
                        if tmp_score < best_score:
                            best[i] -= d
                            best[j] += d
                            conv = tmp_conv
                            best_score = tmp_score
                            stale = 0
                if stale > 400:
                    break

    # ------------------------------------------------------------------
    # final safety check with the official evaluator (cheap – called once)
    try:
        final_score = float(evaluate_sequence(best.tolist()))
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
    except Exception:
        pass

    return [float(v) for v in best]
