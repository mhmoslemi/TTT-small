import numpy as np

# ----------------------------------------------------------------------
# Helper sequences
# ----------------------------------------------------------------------
def _parent_sequence():
    """Return height_sequence_1 if it exists and is valid, else None."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return np.clip(arr, 0.0, 1000.0)
    except Exception:
        pass
    return None


def _arcsine_sequence(n):
    """Arcsine‐type coefficients a_i = a_{i-1}·(1‑0.5/i)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _power_sequence(n, power):
    """Monotonically decreasing a_i = ((n‑1‑i)/(n‑1))**power, 0≤i<n."""
    idx = np.arange(n, dtype=np.float64)
    return ((n - 1 - idx) / (n - 1)) ** power

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimises
    the AC1 upper bound as evaluated by `evaluate_sequence`.
    The method builds a few analytic candidates (parent, arcsine,
    and a simple power‑law shape) and selects the one with the lowest
    score.  No large matrices are allocated; memory usage is linear in n.
    """
    rng = np.random.default_rng(seed)   # kept for potential future extensions

    # ------------------------------------------------------------------
    # Determine length n and optional parent candidate
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    if parent is not None:
        n = parent.size
    else:
        n = 256  # fallback length

    # ------------------------------------------------------------------
    # Assemble candidate sequences
    # ------------------------------------------------------------------
    candidates = []

    if parent is not None:
        candidates.append(parent)

    # arcsine candidate (same length)
    candidates.append(_arcsine_sequence(n))

    # power‑law candidates – the higher the power, the smaller the bound.
    # power = 2 already beats the target; we also include power = 3 for safety.
    candidates.append(_power_sequence(n, 2))
    candidates.append(_power_sequence(n, 3))

    # ------------------------------------------------------------------
    # Evaluate and pick the best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for cand in candidates:
        # ensure the sequence satisfies the evaluator's clipping rules
        np.clip(cand, 0.0, 1000.0, out=cand)

        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            score = float('inf')

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64)

    # Return a plain Python list of floats
    return [float(x) for x in best_seq]
