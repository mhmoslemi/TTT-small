def _smooth_perturb(vec, rng, target_sum):
    # apply moving average smoothing with random window size
    n = vec.size
    # choose window odd size 3 to 11
    w = int(rng.integers(3, 12))
    if w % 2 == 0:
        w += 1
    # compute convolution with uniform kernel
    kernel = np.ones(w, dtype=np.float64) / w
    smooth = np.convolve(vec, kernel, mode='same')
    # blend with original
    alpha = float(rng.uniform(0.1, 0.4))
    cand = (1 - alpha) * vec + alpha * smooth
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand

import numpy as np
import math

# ----------------------------------------------------------------------
def _conv_max_fft(arr: np.ndarray) -> float:
    """Maximum value of full convolution arr*arr using FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact objective (2*n*max_conv / sum^2)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max_fft(arr)
    return float(2.0 * n * max_b / (s * s))


def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p), optionally symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Scale a random contiguous block (log‑normal factor) and renorm."""
    n = vec.size
    cand = vec.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    cand[start:start + width] *= factor
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand


def _smooth_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Blend the vector with a short moving‑average filter."""
    n = vec.size
    w = int(rng.integers(3, 12))
    if w % 2 == 0:
        w += 1
    kernel = np.ones(w, dtype=np.float64) / w
    smooth = np.convolve(vec, kernel, mode='same')
    alpha = float(rng.uniform(0.1, 0.4))
    cand = (1 - alpha) * vec + alpha * smooth
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand
