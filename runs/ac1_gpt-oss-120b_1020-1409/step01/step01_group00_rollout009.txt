import numpy as np

# ----------------------------------------------------------------------
# Helper: compute convolution statistics (max value, its index) using FFT.
# ----------------------------------------------------------------------
def _conv_max_stats(a: np.ndarray) -> tuple[float, int]:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fa * fa, n=size)
    conv = conv_full[: 2 * n - 1]                # linear convolution length
    idx = int(np.argmax(conv))
    max_val = float(conv[idx])
    return max_val, idx


# ----------------------------------------------------------------------
# Helper: objective value 2·n·max(conv)/(sum(a)^2)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, max_val: float, sum_a: float) -> float:
    n = a.shape[0]
    return 2.0 * n * max_val / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Helper: gradient of the objective (without the zero‑sum projection)
# ----------------------------------------------------------------------
def _raw_gradient(a: np.ndarray, max_idx: int,
                  max_val: float, sum_a: float) -> np.ndarray:
    n = a.shape[0]
    # contribution part: 2*a[max_idx - i] when index valid, else 0
    idxs = max_idx - np.arange(n)
    valid = (idxs >= 0) & (idxs < n)
    contrib = np.zeros_like(a)
    contrib[valid] = 2.0 * a[idxs[valid]]

    # constants from the derivative
    const = (4.0 * n * max_val) / (sum_a ** 3)          # same for all i
    grad = (2.0 * n / (sum_a ** 2)) * contrib - const
    return grad


# ----------------------------------------------------------------------
# Simple random pair transfer (keeps the total sum unchanged).
# ----------------------------------------------------------------------
def _pair_move(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] <= 0.0:
        return arr
    delta = arr[i] * rng.uniform(0.001, 0.05)
    out = arr.copy()
    out[i] -= delta
    out[j] += delta
    np.clip(out, 0.0, 1000.0, out=out)
    return out
