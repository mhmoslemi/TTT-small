def autocorr(x):
    n = x.shape[0]
    conv_len = 2*n-1
    fft_len = 1 << (conv_len-1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx*fx, n=fft_len)[:conv_len]
    return conv
```

import numpy as np

# ------------------------------------------------------------
# Helper: classic series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the first `length` terms of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast full convolution via real FFT.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv

    The objective is 2·n·max(conv(a,a))/sum(a)^2, which is invariant to scaling;
    we fix sum(a)=1 and minimise the maximal autocorrelation entry.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size (use parent if available)
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        n = 256
        parent = None

    # --------------------------------------------------------

    inits = []
    if parent is not None:

    inits.append(_make_series(n))
    # Fill up with Dirichlet draws (alpha=0.5 gives sparse vectors)
