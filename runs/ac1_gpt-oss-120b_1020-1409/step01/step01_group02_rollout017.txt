def propose_candidate(seed=42, num_steps=3000, step_frac=0.3, symm_prob=0.05, **kwargs):
    rng = np.random.default_rng(seed)
    a = _load_parent()
    n = a.size
    # Normalize sum to 1
    total = float(a.sum())
    if total < 1e-12:
        a = arcsine_seq(n)
        total = a.sum()
    a = a / total   # sum=1

    # ensure constraints:
    a = np.clip(a, 0.0, 1000.0)

    def bound(seq):
        # compute bound directly: 2*n*max(conv)/sum(seq)^2 ; sum=1
        conv = np.convolve(seq, seq)
        return 2.0 * n * conv.max() / (seq.sum()**2)

    best = a.copy()
    best_score = bound(best)

    current = a.copy()
    curr_score = best_score

    for step in range(num_steps):
        # gradient and max index
        conv = np.convolve(current, current)
        max_idx = int(conv.argmax())
        # compute gradient (vector)
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        grad = np.zeros_like(current)
        # Use slicing for speed
        idx = np.arange(lo, hi+1)
        grad[lo:hi+1] = 2.0 * current[max_idx - idx]

        # find source among positive masses with largest gradient
        pos_mask = current > 1e-12
        if not np.any(pos_mask):
            break
        grad_pos = grad[pos_mask]
        source_candidates = np.nonzero(pos_mask)[0]
        src_idx = source_candidates[grad_pos.argmax()]

        # target lowest gradient (could be same as src)
        tgt_idx = int(np.argmin(grad))

        if src_idx == tgt_idx:
            # pick random other index
            tgt_idx = int(rng.integers(0, n))
            if tgt_idx == src_idx:
                continue

        # transfer amount fraction of src mass
        delta = step_frac * current[src_idx] * rng.random()
        # ensure not too tiny
        if delta < 1e-12:
            continue

        new = current.copy()
        new[src_idx] -= delta
        new[tgt_idx] += delta
        # enforce constraints
        np.clip(new, 0.0, 1000.0, out=new)

        # optionally symmetrize
        if rng.random() < symm_prob:
            new = 0.5 * (new + new[::-1])

        new_score = bound(new)

        # accept if improvement; else maybe accept with prob
        if new_score < best_score:
            best = new.copy()
            best_score = new_score

        # simple acceptance for any non-worse to continue search
        if new_score <= curr_score or rng.random() < 0.01:
            current = new
            curr_score = new_score

    # scale best back to original sum if needed? sum=1 but constraints allow any sum >0.01, we can return as is.
    # Also ensure values within [0,1000].
    best = np.clip(best, 0.0, 1000.0)

    return [float(x) for x in best]

    # load parent or fallback
    init = _load_parent()
    n = init.size
    # baseline candidate maybe arcsine shape with sum=1:
    # not needed because we will generate param sequences.

    best_seq = init.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # parameter search
    alphas = np.linspace(0.2, 1.0, 21)  # 21 values
    for alpha in alphas:
        idx = np.arange(1, n+1, dtype=np.float64)
        seq = (idx) ** (-alpha)
        seq = seq / seq.sum()  # sum=1
        # clip
        np.clip(seq, 0.0, 1000.0, out=seq)
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best_seq = seq.copy()
            best_score = score

    # now refinement via gradient descent
    current = best_seq.copy()
    curr_score = best_score
    # optionally random jitter for initial
    for step in range(num_steps):
        # compute convolution
        conv = np.convolve(current, current)
        max_idx = int(conv.argmax())
        # gradient
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        grad = np.zeros_like(current)
        idx_range = np.arange(lo, hi+1, dtype=int)
        grad[lo:hi+1] = 2.0 * current[max_idx - idx_range]

        # source index: max grad where current not too small
        # If gradient all zeros (e.g., near edges), break
        if not np.any(grad):
            break
        # choose source proportionally to gradient times current
        weighted = grad * current
        if weighted.sum() > 0:
            src = int(rng.choice(np.arange(n), p=weighted/weighted.sum()))
        else:
            src = int(rng.integers(0, n))
        # source must have positive mass
        if current[src] <= 1e-12:
            # find any positive
            positives = np.where(current > 1e-12)[0]
            if positives.size == 0:
                break
            src = int(rng.choice(positives))

        # target index: min grad (or random low)
        tgt = int(np.argmin(grad))
        if tgt == src:
            tgt = int(rng.integers(0, n))
            if tgt == src:
                continue

        # step size scaling decays
        max_step_frac = 0.6 / (1 + step/500)
        delta = max_step_frac * current[src] * rng.random()
        if delta < 1e-12:
            continue

        new = current.copy()
        new[src] -= delta
        new[tgt] += delta
        np.clip(new, 0.0, 1000.0, out=new)

        # optionally symmetrize rarely
        if rng.random() < 0.02:
            new = 0.5 * (new + new[::-1])

        new_score = float(evaluate_sequence(new.tolist()))
        if new_score < best_score:
            best_seq = new.copy()
            best_score = new_score

        # accept if better or occasionally accept worse to diversify
        if new_score < curr_score or rng.random() < 0.01:
            current = new
            curr_score = new_score

        # early stop if bound approx target < 1.50274
        if best_score < 1.50274349:
            break

    # final fallback: ensure nonempty and valid
    if best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64)

    # final clip
    np.clip(best_seq, 0.0, 1000.0, out=best_seq)

    return [float(x) for x in best_seq]

def _load_parent():
    """
    Return the provided parent sequence `height_sequence_1` if it exists
    and is valid, otherwise fall back to a short arcsine‐type sequence.
    The sequence is clipped to the evaluator's allowed range.
    """
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    # fallback – a compact arcsine construction of length 256
    return _arcsine_sequence(256)


def _arcsine_sequence(n):
    """
    Generate the classic discrete arcsine coefficients:
        a_0 = 1,
        a_i = a_{i-1} * (1 - 0.5 / i)   for i >= 1.
    The resulting vector is returned unchanged (no normalization).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _power_sequence(alpha, n):
    """
    Return a non‑negative sequence a_i ∝ (i+1)^{-alpha},
    then scaled to have unit ℓ¹‑norm.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-alpha)
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total > 0:
        a = a / total
    return a

    best_score = float(evaluate_sequence(best_seq.tolist()))

    # explore a modest grid centred around the arcsine exponent 0.5
    alphas = np.linspace(0.30, 0.70, 21)  # 21 candidates
    for alpha in alphas:
        cand = _power_sequence(alpha, n)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score

    # ------------------------------------------------------------------
    # phase 2 – projected gradient descent on the maximal convolution entry
    # ------------------------------------------------------------------
    current = best_seq.copy()
    curr_score = best_score

    for step in range(num_steps):
        # full auto‑convolution
        conv = np.convolve(current, current)
        max_idx = int(conv.argmax())

        # gradient of the maximal entry w.r.t. each coefficient
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        grad = np.zeros_like(current)
        if lo <= hi:
            idx_range = np.arange(lo, hi + 1, dtype=int)
            grad[lo : hi + 1] = 2.0 * current[max_idx - idx_range]

        # if the gradient is all zeros we are stuck – break early
        if not grad.any():
            break

        # source index: large gradient * current mass (probabilistic choice)
        weighted = grad * current
        if weighted.sum() > 0:
            src = int(rng.choice(np.arange(n), p=weighted / weighted.sum()))
        else:
            src_candidates = np.where(current > 1e-12)[0]
            if src_candidates.size == 0:
                break
            src = int(rng.choice(src_candidates))

        # ensure source still has mass
        if current[src] <= 0:
            continue

        # target index: minimal gradient (deterministic)
        tgt = int(np.argmin(grad))
        if tgt == src:
            tgt = int(rng.integers(0, n))
            if tgt == src:
                continue

        # step‑size decays with the iteration count
        max_frac = 0.6 / (1.0 + step / 500.0)          # ≤ 0.6, → smaller later
        delta = max_frac * current[src] * rng.random()  # move a fraction of src
        if delta <= 0.0:
            continue

        # perform the mass transfer and re‑clip
        new = current.copy()
        new[src] -= delta
        new[tgt] += delta
        np.clip(new, 0.0, 1000.0, out=new)

        # occasional mild symmetrisation (helps for many instances)
        if rng.random() < 0.02:
            new = 0.5 * (new + new[::-1])

        # evaluate the new candidate
        new_score = float(evaluate_sequence(new.tolist()))
        if np.isfinite(new_score) and new_score < best_score:
            best_seq = new.copy()
            best_score = new_score

        # accept the move if it improves or with a tiny probability to escape plateaus
        if new_score < curr_score or rng.random() < 0.01:
            current = new
            curr_score = new_score

        # early exit when we have already passed the target bound
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # safety fallback – should never trigger
    # ------------------------------------------------------------------
    if best_seq.size == 0:
        best_seq = np.ones(256, dtype=np.float64)

    # final clipping (ensures compliance with the evaluator)
    np.clip(best_seq, 0.0, 1000.0, out=best_seq)

    # return as plain Python floats
    return [float(x) for x in best_seq]
