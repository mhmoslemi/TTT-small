import numpy as np

def propose_candidate(seed=42, num_iters=3000, tol=1e-12, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine n and parent candidate
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # Define initial x (logits) from parent or generated
    if parent_ok:
        # Clip to [0,1000], renorm to sum 1
        init_a = np.clip(parent_arr, 0.0, 1000.0)
        sum_a = init_a.sum()
        if sum_a > 0:
            init_a = init_a / sum_a
        else:
            init_a = np.ones(n) / n
    else:
        # Use power law ~ 1/sqrt(i+1)
        i = np.arange(1, n+1, dtype=np.float64)
        init_a = 1.0 / np.sqrt(i)
        init_a = init_a / init_a.sum()
    # Ensure not zero to avoid log(0)
    eps = 1e-12
    init_a = np.maximum(init_a, eps)

    # Initialize log variables x
    x = np.log(init_a)

    # Adam hyperparameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best = init_a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    for it in range(num_iters):
        # Compute a from x (softmax)
        # To improve numerical stability, subtract max(x) before exp.
        x_shift = x - np.max(x)  # for stability
        a = np.exp(x_shift)
        sum_a = a.sum()
        a = a / sum_a

        # Compute convolution
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # Convert to score for logging optionally
        cur_score = 2 * n * max_conv  # since sum=1
        # Evaluate (maybe) to store best
        # Evaluate via evaluate_sequence for robust scoring
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score

        # Find active indices (within tolerance)
        # Use relative tolerance to include near max
        rel_tol = 1e-6
        active_mask = conv >= max_conv * (1 - rel_tol)
        active_idx = np.nonzero(active_mask)[0]
        if active_idx.size == 0:
            # fallback to only max
            active_idx = np.array([np.argmax(conv)], dtype=int)

        # Compute gradient g w.r.t a
        g = np.zeros_like(a)
        n_a = a.shape[0]
        for k in active_idx:
            low = max(0, k - (n_a - 1))
            high = min(k, n_a - 1)
            idx = np.arange(low, high + 1, dtype=np.int64)
            partner = k - idx
            # g[idx] += 2 * a[partner]
            np.add.at(g, idx, 2.0 * a[partner])
        # Convert to gradient w.r.t x
        grad_x = g * a

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Occasionally inject random noise to escape local minima
        if (it % 250 == 0) and rng.random() < 0.3:
            noise = rng.normal(0.0, 0.1, size=n)
            x += noise

    # Final return best after loop
    return [float(v) for v in best]

def propose_candidate(seed=42, num_iters=3000, rel_tol=1e-6, target=1.50274349, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine length and parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = (
            parent_arr.ndim == 1
            and n > 0
            and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # Initial a
    if parent_ok:
        a0 = np.clip(parent_arr, 0.0, 1000.0)
        s = a0.sum()
        if s > 0 and np.isfinite(s):
            a0 = a0 / s
        else:
            a0 = np.ones(n, dtype=np.float64) / n
    else:
        # power law decreasing
        i = np.arange(1, n + 1, dtype=np.float64)
        a0 = 1.0 / np.sqrt(i)
        a0 = a0 / a0.sum()

    eps = 1e-12
    a0 = np.maximum(a0, eps)

    # Logits for softmax
    x = np.log(a0)

    # Adam variables
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # Pre‑compute FFT size
    conv_len = 2 * n - 1
    n_fft = 1 << (conv_len - 1).bit_length()

    # Helper to compute convolution via FFT
    def conv_fft(vec):
        fft_vec = np.fft.rfft(vec, n=n_fft)
        prod = fft_vec * fft_vec
        res = np.fft.irfft(prod, n=n_fft)[:conv_len]
        return res

    # Evaluate initial candidate
    a = a0
    conv = conv_fft(a)
    max_conv = conv.max()
    best = a.copy()
    best_score = float(2 * n * max_conv)

    for it in range(1, num_iters + 1):
        # Softmax from x
        shift = x - np.max(x)
        a = np.exp(shift)
        a /= a.sum()

        # Convolution via FFT
        conv = conv_fft(a)
        max_conv = conv.max()

        # Gradient wrt a
        g = np.zeros_like(a)

import numpy as np

def propose_candidate(seed=42, num_iters=3000, rel_tol=1e-6,
                     target=1.50274349, **kwargs):
    """
    Search a non‑negative step sequence that minimises the AC‑1 upper bound.
    The algorithm works on the probability simplex (sum = 1) via a
    soft‑max parametrisation and an Adam optimiser that follows the gradient
    of the maximal convolution entry (active set).  It starts from the
    provided ``height_sequence_1`` when available, otherwise from a simple
    power‑law.  The best sequence encountered is returned as a Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and (optional) parent sequence.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = (
            parent_arr.ndim == 1
            and n > 0
            and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # ------------------------------------------------------------------
    # Initialise a probability vector a (sum = 1, all entries > 0).
    # ------------------------------------------------------------------
    if parent_ok:
        a0 = np.clip(parent_arr, 0.0, 1000.0)
        s = a0.sum()
        if s > 0 and np.isfinite(s):
            a0 = a0 / s
        else:
            a0 = np.ones(n, dtype=np.float64) / n
    else:
        i = np.arange(1, n + 1, dtype=np.float64)
        a0 = 1.0 / np.sqrt(i)            # 1/√i power‑law
        a0 = a0 / a0.sum()

    eps = 1e-12
    a0 = np.maximum(a0, eps)            # avoid log(0)

    # Logits for a soft‑max parametrisation (a = softmax(x))
    x = np.log(a0)

    # ------------------------------------------------------------------
    # Adam optimiser initialisation.
    # ------------------------------------------------------------------
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # ------------------------------------------------------------------
    # FFT‑based convolution (full length = 2*n-1).
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    n_fft = 1 << (conv_len - 1).bit_length()   # next power of two

    def conv_fft(vec):
        """Return full linear convolution of vec with itself via FFT."""
        f = np.fft.rfft(vec, n=n_fft)
        prod = f * f
        res = np.fft.irfft(prod, n=n_fft)[:conv_len]
        return res

    # ------------------------------------------------------------------
    # Initial score.
    # ------------------------------------------------------------------
    a = a0.copy()
    conv = conv_fft(a)
    max_conv = conv.max()
    best = a.copy()
    best_score = float(2 * n * max_conv)      # sum(a)=1 ⇒ score = 2·n·max(conv)

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    for it in range(1, num_iters + 1):
        # ----- soft‑max to obtain a -----
        shift = x - np.max(x)                 # stability
        a = np.exp(shift)
        a /= a.sum()

        # ----- convolution & current score -----
        conv = conv_fft(a)
        max_conv = conv.max()
        cur_score = float(2 * n * max_conv)

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            if best_score <= target:
                break

        # ----- active set of max‑convolution indices -----
        active = np.where(conv >= max_conv * (1 - rel_tol))[0]
        if active.size == 0:                  # fall back to the single max
            active = np.array([np.argmax(conv)], dtype=int)

        # ----- gradient of max(conv) w.r.t. a -----
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            partner = k - idx
            np.add.at(g, idx, 2.0 * a[partner])

        # ----- map gradient to logits (soft‑max Jacobian) -----
        mu = np.dot(g, a)                     # expectation of g under a
        grad_x = a * (g - mu)                 # ∂L/∂x = a·(g - μ)

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional random jitter to escape plateaus -----
        if (it % 250 == 0) and rng.random() < 0.3:
            x += rng.normal(0.0, 0.1, size=n)

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats).
    # ------------------------------------------------------------------
    return [float(v) for v in best]
