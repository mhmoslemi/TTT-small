def compute_score(seq):
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = seq.size
    size = 2 * n
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:2*n-1]
    max_b = conv.max()
    # Now compute bound
    return 2.0 * n * max_b / (total * total)

import numpy as np

def compute_score(seq):
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = seq.size
    size = 2 * n
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (total * total)

import numpy as np

def _score_fft(seq: np.ndarray) -> float:
    """Fast evaluation of the AC1 bound using FFT."""
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = seq.size
    size = 2 * n
    # linear convolution via FFT (real-valued)
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (total * total)
