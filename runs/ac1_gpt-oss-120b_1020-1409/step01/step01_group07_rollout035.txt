def propose_candidate(seed: int = 42, **kwargs):
    """
    Search for a non‑negative step sequence that minimizes
    2 * n * max(a*a) / (sum(a)**2) without exceeding the evaluator limits.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and initialise from the parent if present.
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 256
        parent = None

    # ------------------------------------------------------------------
    # Helper: fast bound using FFT (identical to evaluate_sequence apart
    # from the clipping step – we keep the sequence inside the allowed
    # range at all times, so the results match).
    # ------------------------------------------------------------------
    n_fft = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n‑1

    def fast_bound(arr: np.ndarray) -> float:
        """Return the AC1 bound for a non‑negative array `arr` (sum>0)."""
        f = np.fft.rfft(arr, n=n_fft)
        conv = np.fft.irfft(f * f, n=n_fft)[: 2 * n - 1]
        maxb = float(conv.max())
        s = float(arr.sum())
        return 2.0 * n * maxb / (s * s)

    # ------------------------------------------------------------------
    # Initialise a few starting points.
    # ------------------------------------------------------------------
    # 1) Normalised parent (or uniform fallback).
    if parent is not None and parent.size == n:
        seq0 = parent.copy()
        total = seq0.sum()
        if total == 0.0:
            seq0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            seq0 = seq0 / total
    else:
        seq0 = np.full(n, 1.0 / n, dtype=np.float64)

    # 2) Random Dirichlet (Gamma(1,1) → exponential) distribution.
    seq1 = rng.gamma(1.0, 1.0, size=n)
    seq1 /= seq1.sum()

    # Ensure admissible range (the evaluator will clip anyway).
    np.clip(seq0, 0.0, 1000.0, out=seq0)
    np.clip(seq1, 0.0, 1000.0, out=seq1)

    # Pick the better of the two starters.
    best_seq = seq0
    best_score = fast_bound(best_seq)
    score1 = fast_bound(seq1)
    if score1 < best_score:
        best_seq = seq1
        best_score = score1

    # ------------------------------------------------------------------
    # Simple hill‑climbing / simulated‑annealing loop.
    # ---------------------------------------------------------------
    steps = 3000                     # total move attempts
    sigma = 0.1                      # relative amount moved per step
    stagnant = 0                     # counts steps without improvement

    for _ in range(steps):

        if best_score <= 1.50274349:

        delta = sigma * src_val

        cand[j] += delta
