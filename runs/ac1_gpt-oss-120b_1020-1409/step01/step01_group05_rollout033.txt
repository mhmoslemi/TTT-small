import numpy as np

def compute_bound(a, n, size):
    a_sum = a.sum()
    if a_sum == 0:
        return np.inf
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (a_sum * a_sum)

def normalize(a):
    s = a.sum()
    if s <= 0:
        return np.full_like(a, 1.0 / len(a))
    a = a / s
    # clip to [0,1000] if needed
    np.clip(a, 0.0, 1000.0, out=a)
    return a

import numpy as np

def _normalize(vec: np.ndarray) -> np.ndarray:
    """Scale to have L1‑norm 1 and enforce the hard bounds."""
    s = vec.sum()
    if s <= 0 or not np.isfinite(s):
        vec = np.full_like(vec, 1.0 / vec.size)
        s = vec.sum()
    vec = vec / s
    np.clip(vec, 0.0, 1000.0, out=vec)
    return vec


def _bound(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Return the AC1 upper bound for a non‑negative vector."""
    s = vec.sum()
    if s <= 0:
        return np.inf
    # linear convolution via FFT (zero‑padded to avoid wrap‑around)
    f = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    m = conv.max()
    return 2.0 * n * m / (s * s)


def propose_candidate(seed: int = 42, num_iters: int = 2000,
                     grid_steps: int = 10, **kwargs):
    """
    Search for a non‑negative step sequence (list of floats) that minimizes
    the AC1 bound.  Uses a cheap parametric grid search (power‑law and geometric
    families) followed by a multiplicative gradient‑descent refinement with
    occasional stochastic perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n and optional parent sequence.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = (
            parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # FFT length: next power of two ≥ 2n‑1 (avoids circular wrap‑around)
    fft_len = 1 << ((2 * n - 1).bit_length())

    best = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Parent candidate (if present) after normalisation.
    # ------------------------------------------------------------------
    if parent_ok:
        candidate = _normalize(np.clip(parent_arr, 0.0, 1000.0))
        score = _bound(candidate, n, fft_len)
        if np.isfinite(score):
            best = candidate.copy()
            best_score = score

    # ------------------------------------------------------------------
    # 1) Grid search over power‑law families  a_i ∝ (i+β)^‑p
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.30, 1.20, grid_steps)
    beta_vals = np.linspace(0.5, 5.0, grid_steps)
    idx = np.arange(n, dtype=np.float64)

    for p in p_vals:
        for beta in beta_vals:
            a = 1.0 / np.power(idx + beta, p)
            a = _normalize(a)
            sc = _bound(a, n, fft_len)
            if sc < best_score:
                best = a.copy()
                best_score = sc

    # ------------------------------------------------------------------
    r_vals = np.linspace(0.5, 0.999, grid_steps)

        a = _normalize(a)
        sc = _bound(a, n, fft_len)
        if sc < best_score:

            best_score = cur_score

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best is None or not np.isfinite(best_score):
        best = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(x) for x in best]
