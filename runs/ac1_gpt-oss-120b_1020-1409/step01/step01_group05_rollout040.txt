def propose_candidate(seed=42, **kwargs):

        parent_ok = False

        s = seq.sum()
        if s <= 0 or not np.isfinite(s):
            seq = np.ones_like(seq) / seq.size
        else:
            seq = seq / s
        return seq

    best_seq = None

        parent_seq = normalize(parent_seq)

def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative sequence that minimises the AC1 bound.
    The search tries a small family of analytically‑motivated shapes
    (power‑law and exponential decays), keeps the best candidate,
    and finally refines it with a few steps of multiplicative gradient
    descent plus occasional random block perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and (optionally) a parent construction.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = (
            parent_arr.ndim == 1
            and n > 0
            and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # ------------------------------------------------------------------
    # Helper: bring a vector to the feasible set (non‑negative, sum=1,
    # clip to the allowed magnitude range).
    # ------------------------------------------------------------------
    def normalize(v):
        s = v.sum()
        if not np.isfinite(s) or s <= 0.0:
            v = np.ones_like(v) / v.size
            s = v.sum()
        v = v / s
        np.clip(v, 0.0, 1000.0, out=v)
        return v

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1️⃣  Parent (if available)
    # ------------------------------------------------------------------
    if parent_ok:
        seq = normalize(np.clip(parent_arr, 0.0, 1000.0))
        sc = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(sc):
            best_seq = seq.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # 2️⃣  Uniform baseline
    # ------------------------------------------------------------------
    uniform = normalize(np.ones(n, dtype=np.float64))
    sc = float(evaluate_sequence(uniform.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_seq = uniform.copy()
        best_score = sc

    # ------------------------------------------------------------------
    # 3️⃣  Power‑law families a_i ∝ (i+1)^{-p}
    # ------------------------------------------------------------------
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    power_exps = np.linspace(0.35, 0.75, 15)           # dense around the optimum
    for p in power_exps:
        a = i ** (-p)

        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq = a.copy()
            best_score = sc

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    r_vals = np.linspace(0.92, 0.998, 12)
    for r in r_vals:
        # geometric series, sum = 1‑r**n
        powers = r ** np.arange(n, dtype=np.float64)
        a = (1.0 - r) * powers
        total = 1.0 - r ** n
        if total > 1e-12:
            a = a / total
        else:
            a = normalize(np.ones(n, dtype=np.float64))
        a = normalize(a)

            best_score = sc

    # ------------------------------------------------------------------

    if best_seq is not None:
        a = best_seq.copy()
        sum_target = 1.0

        max_iters = 500

        for it in range(max_iters):

            conv = np.convolve(a, a)

            a = a * np.exp(-step * g)

            a = normalize(a)

                    best_score = scp

    # 6️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = normalize(np.ones(n, dtype=np.float64))

    # Return a plain Python list of floats
    return [float(x) for x in best_seq]
