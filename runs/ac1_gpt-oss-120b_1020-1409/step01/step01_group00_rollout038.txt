import numpy as np

def _bound(seq):
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    maxc = conv.max()
    return float(2 * n * maxc / (s * s))

def propose_candidate(seed: int = 42, num_steps: int = 2500, **kwargs):
    rng = np.random.default_rng(seed)

    # initialize
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 256
        seq = np.ones(n, dtype=np.float64)  # uniform sum = n
    n = seq.shape[0]

    # Clip to allowed range just in case
    np.clip(seq, 0.0, 1000.0, out=seq)

    best = seq
    best_score = _bound(best)

    # optional fallback uniform (sum=n)
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform_score = _bound(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    for step in range(num_steps):
        # compute convolution once
        conv = np.convolve(best, best)
        max_conv = conv.max()
        # indices attaining the max
        max_idxs = np.flatnonzero(conv == max_conv)
        d = int(rng.choice(max_idxs))

        # gradient weights g_i = 2 * best[d - i] when valid
        idx = np.arange(n)
        partner = d - idx
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[partner[mask]]

        # donor: large g, non‑zero mass
        donors = np.where(best > 1e-12)[0]
        if donors.size == 0:
            break
        donor_idx = donors[int(np.argmax(g[donors]))]

        # receiver: small g, not at upper bound
        receivers = np.where(best < 1000.0 - 1e-12)[0]
        if receivers.size == 0:
            break
        receiver_idx = receivers[int(np.argmin(g[receivers]))]

        if donor_idx != receiver_idx:
            # transfer a fraction of allowed amount
            cap = min(best[donor_idx], 1000.0 - best[receiver_idx])
            if cap > 1e-12:
                delta = cap * rng.uniform(0.02, 0.2)
                cand = best.copy()
                cand[donor_idx] -= delta
                cand[receiver_idx] += delta
                # ensure bounds after floating‑point noise
                np.clip(cand, 0.0, 1000.0, out=cand)
                sc = _bound(cand)
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc

        # occasional random perturbations for diversification
        if step % 100 == 0:
            # block scaling
            # choose a width roughly between n/300 and n/30
            low = max(1, n // 300)
            high = max(2, n // 30)
            width = int(rng.integers(low, high + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.06)))
            cand2 = best.copy()
            cand2[start:start+width] *= factor
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            sc2 = _bound(cand2)
            if np.isfinite(sc2) and sc2 < best_score:
                best = cand2
                best_score = sc2

    return [float(v) for v in best]

import numpy as np


def _bound(seq: np.ndarray) -> float:
    """
    Compute the AC1 upper bound for a non‑negative sequence.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    maxc = conv.max()
    return float(2.0 * n * maxc / (s * s))


def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):
    """
    Return a step‑function (list of floats) whose AC1 bound improves on the
    classic binomial construction.  The method starts from the provided
    `height_sequence_1` (if present) and performs a gradient‑guided local
    search that repeatedly moves a small amount of mass from indices that
    contribute most to the current maximal convolution value to indices that
    contribute least.
    """
    rng = np.random.default_rng(seed)

    # ----- initial sequence -------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: a modest uniform vector
        n = 256
        seq = np.ones(n, dtype=np.float64)
    n = seq.shape[0]
    np.clip(seq, 0.0, 1000.0, out=seq)

    best = seq
    best_score = _bound(best)

    # a simple uniform fallback, kept only if it happens to be better
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform_score = _bound(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # ----- main optimisation loop -----------------------------------------
    for step in range(num_steps):
        # convolution and the offset where the maximum is attained
        conv = np.convolve(best, best)
        max_conv = conv.max()
        d = int(np.argmax(conv))                 # offset of the current worst lag

        # gradient weights: g_i = 2 * best[d - i]  (0 if out of range)
        idx = np.arange(n)
        partner = d - idx
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[partner[mask]]

        # donor – large gradient, non‑zero mass
        donor_candidates = np.where(best > 1e-12)[0]
        if donor_candidates.size == 0:
            break
        donor_idx = donor_candidates[int(np.argmax(g[donor_candidates]))]

        # receiver – small gradient, room to increase
        receiver_candidates = np.where(best < 1000.0 - 1e-12)[0]
        if receiver_candidates.size == 0:
            break
        receiver_idx = receiver_candidates[int(np.argmin(g[receiver_candidates]))]

        if donor_idx != receiver_idx:
            # amount that can be transferred without violating bounds
            capacity = min(best[donor_idx], 1000.0 - best[receiver_idx])
            if capacity > 1e-12:
                delta = capacity * rng.uniform(0.02, 0.20)   # move 2‑20 % of the slack
                cand = best.copy()
                cand[donor_idx] -= delta
                cand[receiver_idx] += delta
                np.clip(cand, 0.0, 1000.0, out=cand)

                sc = _bound(cand)
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc

        # occasional random block scaling for additional exploration
        if step % 100 == 0:
            low = max(1, n // 300)
            high = max(2, n // 30)
            width = int(rng.integers(low, high + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.06)))
            cand2 = best.copy()
            cand2[start:start + width] *= factor
            np.clip(cand2, 0.0, 1000.0, out=cand2)

            sc2 = _bound(cand2)
            if np.isfinite(sc2) and sc2 < best_score:
                best = cand2
                best_score = sc2

    # -----------------------------------------------------------------------
    # Return a plain Python list of floats.
    return [float(v) for v in best]
