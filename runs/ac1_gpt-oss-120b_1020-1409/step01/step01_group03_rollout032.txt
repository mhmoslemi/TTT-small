import numpy as np

def _conv_full(x):
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv

def _score(x):
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_full(x)
    return 2.0 * x.size * c.max() / (s * s)

def _grad_step(x, rng):
    n = x.size
    conv = _conv_full(x)
    M = conv.max()
    k_inds = np.where(conv == M)[0]  # all max indices
    g = np.zeros_like(x)
    for k in k_inds:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] += x[k - idx]
    s = x.sum()
    if s == 0.0:
        return x, float('inf')
    inv_s2 = 1.0 / (s * s)
    inv_s3 = inv_s2 / s
    grad = (2.0 * n) * g * inv_s2 - (4.0 * n * M) * inv_s3
    # line-search
    lr = 1.0
    best_x = x
    best_score = _score(x)
    for _ in range(5):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        if rng.random() < 0.1:
            cand = 0.5 * (cand + cand[::-1])
        sc = _score(cand)
        if sc < best_score:
            best_x = cand
            best_score = sc
            break
        lr *= 0.5
    return best_x, best_score

import numpy as np

# ------------------------------------------------------------
# Convolution (full linear) using FFT – matrix‑free.
# ------------------------------------------------------------
def _conv_full(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact evaluator copy – used for internal decisions.
# ------------------------------------------------------------
def _score(x: np.ndarray) -> float:
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_full(x)
    return 2.0 * x.size * c.max() / (s * s)


# ------------------------------------------------------------
# Gradient step for the objective f = 2*n*max(conv)/sum^2.
# ------------------------------------------------------------
def _grad_step(x: np.ndarray, rng: np.random.Generator):
    n = x.size
    conv = _conv_full(x)
    M = conv.max()
    # all positions attaining the maximum (sub‑gradient)
    k_inds = np.where(conv == M)[0]
    g = np.zeros_like(x)
    for k in k_inds:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g[idx] += x[k - idx]

    s = x.sum()
    if s == 0.0:
        return x, float('inf')

    inv_s2 = 1.0 / (s * s)
    inv_s3 = inv_s2 / s
    # ∇f = 2*n*g/s^2 - 4*n*M/s^3
    grad = (2.0 * n) * g * inv_s2 - (4.0 * n * M) * inv_s3

    # simple back‑tracking line search
    lr = 1.0
    best_local = x
    best_score = _score(x)
    for _ in range(5):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        # occasional mild symmetrisation
        if rng.random() < 0.1:
            cand = 0.5 * (cand + cand[::-1])
        sc = _score(cand)
        if sc < best_score:
            best_local = cand
            best_score = sc
            break
        lr *= 0.5
    return best_local, best_score

def _random_block(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:

    min_w = max(1, n // 50)          # 2‑% of n at least
    max_w = max(1, n // 5)           # up to 20 %
    w = rng.integers(min_w, max_w + 1)

    factor = np.exp(rng.normal(0.0, 0.15))
    y = x.copy()
    y[start:start + w] *= factor
    if rng.random() < 0.2:
        y = 0.5 * (y + y[::-1])
    np.clip(y, 0.0, 1000.0, out=y)
    return y
