    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

def _conv_full_fft(a, size):
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * a.shape[0] - 1]
    return conv

def _subgradient(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        i = np.arange(i_start, i_end + 1)
        j = k - i
        g[i] = 2.0 * a[j]
    return g

def _scale_to_sum(a, target):
    np.clip(a, 0.0, None, out=a)
    s = a.sum()
    if s > 0.0:
        a *= target / s
    return a

import numpy as np

# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a (full) using a power‑of‑2 FFT."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """Subgradient of max_k (a*a)[k] at the active shift k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        i = np.arange(i_start, i_end + 1)
        j = k - i
        g[i] = 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clamp to non‑negative and renormalise to the exact L1‑norm."""
    np.clip(v, 0.0, None, out=v)
    s = v.sum()
    if s > 0.0:
        v *= target / s
    return v
