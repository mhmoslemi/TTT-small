def targeted_perturb(arr, rng):
    b = np.convolve(arr, arr)
    k = np.argmax(b)
    # indices i in 0..n-1 such that j = k-i is also in 0..n-1
    i_min = max(0, k - (n-1))
    i_max = min(n-1, k)
    indices = np.arange(i_min, i_max+1)
    # compute product contributions (score) maybe arr[i]*arr[k-i]
    contributions = arr[indices] * arr[k - indices]
    # get index with largest contribution (largest product)
    idx = indices[np.argmax(contributions)]
    # donor = idx, receiver random other index not same.
    donor = int(idx)
    recv = rng.integers(0, n)
    if donor == recv or arr[donor] <= 0:
        return arr.copy()
    # transfer fraction of donor value
    delta = arr[donor] * rng.uniform(0.001, 0.02)
    cand = arr.copy()
    cand[donor] = max(0.0, cand[donor] - delta)
    cand[recv] = min(1000.0, cand[recv] + delta)
    return cand
```

import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4^i for i=0..length-1 (negative‑free, fast recurrence)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _interleaved(arr: np.ndarray) -> np.ndarray:
    """Place largest, smallest, second‑largest, second‑smallest …"""
    vals = np.sort(arr)[::-1]               # descending
    n = vals.shape[0]
    out = np.empty_like(vals)
    left, right, pos = 0, n - 1, 0
    while left <= right:
        out[pos] = vals[left]; left += 1; pos += 1
        if left <= right:
            out[pos] = vals[right]; right -= 1; pos += 1
    return out


def _score(arr: np.ndarray) -> float:
    """Exact copy of the evaluator's objective (no extra calls)."""
    n = arr.shape[0]
    conv = np.convolve(arr, arr)
    max_b = float(conv.max())
    sum_a = float(arr.sum())
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
