import numpy as np

# ------------------------------------------------------------
# Helper: classic decreasing series a_i = C(2i,i)/4^i.
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast full convolution (real inputs) via FFT.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Build a sparse matrix T such that (T @ b) == conv(f, b).
# ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray):
    try:
        from scipy import sparse
    except Exception:
        return None
    n = f.size
    # each diagonal k (offset) has constant value f[k]
    diagonals = [np.full(n, f[k], dtype=np.float64) for k in range(n)]
    offsets = np.arange(0, n, dtype=int)
    T = sparse.diags(diagonals, offsets, shape=(2 * n - 1, n), format="csc")
    return T


# ------------------------------------------------------------
# Solve the linear program:
#   maximize   sum(b)
#   subject to T @ b <= M  ,  b >= 0
# ------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float):
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except Exception:
        return None

    n = f.size
    T = _build_conv_matrix(f)
    if T is None:
        return None

    A_ub = T
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximize sum(b)  ->  minimize -sum(b)
    bounds = [(0.0, None)] * n

    # 'highs' works with sparse A_ub
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success or res.x is None:
        return None
    return res.x

def propose_candidate(seed: int = 42, max_lp_iters: int = 6, mix: float = 0.5, **kwargs):
    """
    Returns a list of non‑negative heights that aim to minimise the
    AC1 upper bound.  The algorithm starts from the classic binomial series,
    then repeatedly solves a linear program that reshapes the sequence while
    keeping the worst‑case convolution unchanged, finally mixing the new shape
    back into the current candidate.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine length and an initial feasible sequence.
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
        init = parent.copy()
    except Exception:
        n = 256
        parent = None
        init = _make_series(n)

    # Ensure a valid start (clip, non‑negative).
    np.clip(init, 0.0, 1000.0, out=init)

    # Use the parent if it is already better than the analytic series.
    if parent is not None:
        # parent may already be optimal; keep it if it scores lower.
        try:
            parent_score = evaluate_sequence(parent.tolist())
            init_score = evaluate_sequence(init.tolist())
            if parent_score < init_score:
                init = parent.copy()
        except Exception:
            pass

    f = init.copy()
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # --------------------------------------------------------
    # Iterative LP‑based refinement.
    # --------------------------------------------------------
    for it in range(max_lp_iters):
        # current worst‑case convolution value
        conv_f = _conv_self(f)
        M = float(conv_f.max())

        # Solve LP to find a better shaped sequence with the same bound.
        b = _solve_lp(f, M)
        if b is None:
            break

        # Scale b to keep the total sum unchanged (conv(f,b) stays ≤ M).
        sum_f = float(f.sum())
        sum_b = float(b.sum())
        if sum_b == 0.0:
            break
        scale = sum_f / sum_b
        g = b * scale

        # Blend the new shape into the current one.
        f_new = (1.0 - mix) * f + mix * g
        np.clip(f_new, 0.0, 1000.0, out=f_new)

        # Evaluate the candidate.
        score_new = evaluate_sequence(f_new.tolist())
        if score_new < best_score:
            best_seq = f_new.copy()
            best_score = score_new

        # Stop early if no improvement.
        if abs(score_new - best_score) < 1e-12:
            break

        f = f_new

    # --------------------------------------------------------
    # Fallback: ensure we return a non‑empty finite list.
    # --------------------------------------------------------
    if best_seq.size == 0 or not np.isfinite(best_score):
        best_seq = np.ones(256, dtype=np.float64)

    return [float(v) for v in best_seq]
