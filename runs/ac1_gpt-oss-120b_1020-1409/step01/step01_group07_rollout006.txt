def _normalize(seq):
    """Clamp to [0,1000] and scale to have sum 1 (or leave if sum zero)."""
    # Clip in-place
    np.clip(seq, 0.0, 1000.0, out=seq)
    total = seq.sum()
    if total <= 0.0:
        # uniform fallback
        seq[:] = 1.0 / seq.size
    else:
        seq /= total
    return seq

import numpy as np

def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clamp to [0,1000] and scale to sum = 1 (or uniform if sum = 0)."""
    np.clip(seq, 0.0, 1000.0, out=seq)
    total = seq.sum()
    if total <= 0.0:
        seq[:] = 1.0 / seq.size
    else:
        seq /= total
    return seq
