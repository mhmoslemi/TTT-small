def _gen_power_seq(beta: float, alpha: float, n: int) -> np.ndarray:
    """Power‑law sequence a_i = 1/(i+alpha)^beta, normalized to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)      # i = 1…n
    seq = 1.0 / ((i + alpha) ** beta)
    total = seq.sum()
    if total <= 0.0:
        seq = np.ones_like(seq)
        total = seq.sum()
    seq = seq / total                               # sum == 1
    return np.clip(seq, 0.0, 1000.0)


def _gen_exp_seq(lam: float, n: int) -> np.ndarray:
    """Exponential decay sequence exp(-lam * i), normalized to sum = 1."""

    seq = np.exp(-lam * i)
    total = seq.sum()
    seq = seq / total
    return np.clip(seq, 0.0, 1000.0)
