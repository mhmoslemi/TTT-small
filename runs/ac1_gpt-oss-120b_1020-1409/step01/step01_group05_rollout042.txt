import numpy as np

def propose_candidate(seed=42, num_iters=5000, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine length n and parent sequence
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = parent_arr.ndim == 1 and n > 0 and np.isfinite(parent_arr).all()
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # Starting candidate
    if parent_ok:
        a = np.clip(parent_arr, 0.0, 1000.0)
        psum = a.sum()
        if psum > 0 and np.isfinite(psum):
            a = a / psum
        else:
            a = np.ones(n) / n
        best = a.copy()
        best_score = float(evaluate_sequence(best.tolist()))
    else:
        a = 1.0 / np.sqrt(np.arange(1, n+1, dtype=np.float64))
        a = a / a.sum()
        best = a.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    # Parameters
    step0 = 0.30
    beta = 2000.0  # softmax inverse temperature
    # main loop
    for it in range(num_iters):
        conv = np.convolve(a, a)   # length 2n-1
        # Soft-max weighting on convolution entries
        conv_max = conv.max()
        # shift to avoid overflow
        w = np.exp(beta * (conv - conv_max))
        # normalize weights to sum 1 to keep gradient magnitude reasonable
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback to hard max weight
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # gradient via correlation
        corr = np.correlate(a, w, mode='full')
        offset = len(w) - 1
        g = 2.0 * corr[offset: offset + n]
        # multiplicative update
        step = step0 * (0.99 ** it)
        a = a * np.exp(-step * g)

        # renormalize to sum 1, clip bounds
        a_sum = a.sum()
        if a_sum <= 0 or not np.isfinite(a_sum):
            a = np.ones(n) / n
            a_sum = a.sum()
        a = a / a_sum
        np.clip(a, 0.0, 1000.0, out=a)

        # occasional random block perturbation (as earlier)
        if (it % 250 == 0) and rng.random() < 0.15:
            width = int(rng.integers(max(1, n//200), max(2, n//20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.10)))
            pert = a.copy()
            pert[start:start+width] *= factor
            pert = np.clip(pert, 0.0, 1000.0)
            pert = pert / pert.sum()
            score = float(evaluate_sequence(pert.tolist()))
            if np.isfinite(score) and score < best_score:
                best = pert.copy()
                best_score = score
                a = pert   # continue from improved point

        # evaluate current a
        cur_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best = a.copy()
            best_score = cur_score

        # early stop if below target (maybe)
        if best_score < 1.5:
            break

    # final fallback if still not finite
    if not (np.isfinite(best_score) and best_score > 0):
        # default uniform
        best = np.ones(n) / n
    # convert to list of floats
    return [float(x) for x in best]
