def evaluate_sequence(sequence):
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def _conv_max(arr):
    n = arr.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n - 1]
    return float(conv.max())

def _score(arr, n):
    s = arr.sum()
    if s == 0.0:
        return float('inf')
    maxb = _conv_max(arr)
    return float(2.0 * n * maxb / (s * s))

def _make_powerlaw(n, p, beta, sym):
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base

def _block_perturb(vec, rng, target_sum):
    cand = vec.copy()
    n = cand.size
    # choose random block fraction
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.07)))  # larger variation
    cand[start:start+width] *= factor
    if rng.random() < 0.5:
        # optionally mirror half to increase symmetry
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)

    # get length n
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        n = 4498
        parent = None

    target_sum = np.sqrt(2.0 * n)

    # baseline uniform
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    best_vec = uniform.copy()
    best_score = _score(best_vec, n)

    # try parent as candidate (if exists)
    if parent is not None:
        cand = parent.astype(np.float64)
        # clip and renormalise
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s > 0.0:
            cand *= target_sum / s
        sc = _score(cand, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # grid search over power-law family (dense)
    p_vals = np.linspace(0.30, 0.80, 25)   # finer resolution
    beta_vals = np.logspace(-2, 2, 20)    # more samples
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                base = _make_powerlaw(n, p, beta, sym)
                s = base.sum()
                if s == 0.0:
                    continue
                a = base * (target_sum / s)
                np.clip(a, 0.0, 1000.0, out=a)
                sc = _score(a, n)
                if sc < best_score:
                    best_score = sc
                    best_vec = a.copy()

    # local stochastic search
    max_iter = 4000
    for it in range(max_iter):
        move = rng.random()
        if move < 0.2:
            # swap two entries (preserves sum)
            i, j = rng.integers(n, size=2)
            cand = best_vec.copy()
            cand[i], cand[j] = cand[j], cand[i]
        elif move < 0.5:
            # block perturb
            cand = _block_perturb(best_vec, rng, target_sum)
        elif move < 0.7:
            # rotate
            shift = int(rng.integers(n))
            cand = np.roll(best_vec, shift)
        else:
            # random mixture of two power-laws
            p1 = rng.uniform(0.30, 0.80)
            p2 = rng.uniform(0.30, 0.80)
            beta1 = 10**rng.uniform(-2, 2)
            beta2 = 10**rng.uniform(-2, 2)
            w = rng.random()
            idx = np.arange(n, dtype=np.float64)
            a1 = (idx + beta1) ** (-p1)
            a2 = (idx + beta2) ** (-p2)
            base = w * a1 + (1.0 - w) * a2
            s = base.sum()
            if s == 0.0:
                continue
            cand = base * (target_sum / s)
            np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

        # early stop if we hit theoretical limit maybe close to 1.5
        if best_score < 1.51:
            break

    final = [float(x) for x in best_vec]
    return final

import numpy as np

def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full linear convolution arr*arr via FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact evaluator formula (faster than evaluate_sequence)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p). If sym=True the sequence is symmetrised."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


def _make_exponential(n: int, gamma: float, sym: bool) -> np.ndarray:
    """a_i ∝ exp(-γ·i). Optional symmetrisation."""
    idx = np.arange(n, dtype=np.float64)
    base = np.exp(-gamma * idx)
    if sym:
        base = base + np.exp(-gamma * (n - 1 - idx))
    return base


def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Stretch a random block (keeps L1‑norm fixed)."""
    cand = vec.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.08)))   # moderate log‑normal change
    cand[start:start + width] *= factor
    if rng.random() < 0.5:                         # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand
