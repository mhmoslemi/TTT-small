def _pair_move(arr, rng):
    n = arr.shape[0]
    i = rng.integers(0, n)
    # ensure donor has some mass
    if arr[i] <= 0.0:
        return arr.copy()
    j = rng.integers(0, n)
    if i == j:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.002, 0.02)
    out = arr.copy()
    out[i] -= delta
    out[j] = min(1000.0, out[j] + delta)
    # ensure non-negative due to rounding
    if out[i] < 0:
        out[i] = 0.0
    return out
```

def _block_move(arr, rng):
    n = arr.shape[0]
    min_w = max(1, n // 200)
    max_w = max(2, n // 20)
    width = rng.integers(min_w, max_w+1)

def _guided_move(arr, rng):
    n = arr.shape[0]
    conv = np.convolve(arr, arr)
    k_max = int(np.argmax(conv))
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    if i_max < i_min:
        # should not happen
        return arr.copy()
    idxs = np.arange(i_min, i_max+1)
    products = arr[idxs] * arr[k_max - idxs]
    # find index with largest product
    best_local = int(idxs[np.argmax(products)])
    # partner index
    partner = k_max - best_local
    donor = best_local if arr[best_local] >= arr[partner] else partner
    # shift a small fraction
    delta = arr[donor] * rng.uniform(0.001, 0.015)
    recipient = rng.integers(0, n)
    # avoid trivial
    if recipient == donor:
        recipient = (recipient + n//2) % n
    out = arr.copy()
    out[donor] = max(0.0, out[donor] - delta)
    out[recipient] = min(1000.0, out[recipient] + delta)
    return out
```

import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        f = float(i)
        a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
    return a


def _exp_decay_sequence(length: int, alpha: float) -> np.ndarray:
    """Exponential decay a_i = exp(-alpha*i)."""
    i = np.arange(length, dtype=np.float64)
    return np.exp(-alpha * i)


def _power_law_sequence(length: int, p: float) -> np.ndarray:
    """Power‑law a_i = (i+1)^(-p)."""
    i = np.arange(1, length + 1, dtype=np.float64)
    return i ** (-p)


def _pair_move(arr: np.ndarray, rng: np.random.Generator, max_frac: float) -> np.ndarray:
    n = arr.shape[0]
    i = rng.integers(0, n)
    if arr[i] <= 0.0:
        return arr.copy()
    j = rng.integers(0, n)
    if i == j:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, max_frac)
    out = arr.copy()
    out[i] = max(0.0, out[i] - delta)
    out[j] = min(1000.0, out[j] + delta)
    return out


def _block_move(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = arr.shape[0]
    min_w = max(1, n // 200)
    max_w = max(2, n // 20)
    width = rng.integers(min_w, max_w + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.05))
    out = arr.copy()
    out[start:start + width] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _guided_move(arr: np.ndarray, rng: np.random.Generator, max_frac: float) -> np.ndarray:
    """Move a small amount from a coordinate that participates in the maximal
    autocorrelation to a far‑away coordinate."""
    n = arr.shape[0]
    conv = np.convolve(arr, arr)
    k_max = int(np.argmax(conv))
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    if i_max < i_min:
        return arr.copy()
    idxs = np.arange(i_min, i_max + 1)
    products = arr[idxs] * arr[k_max - idxs]
    best_local = int(idxs[np.argmax(products)])
    partner = k_max - best_local
    donor = best_local if arr[best_local] >= arr[partner] else partner
    delta = arr[donor] * rng.uniform(0.001, max_frac)

    # pick a recipient outside the interval that contributed to k_max
    rec = rng.integers(0, n)
    if i_min <= rec <= i_max:
        # simple deterministic fallback outside the interval
        rec = (i_max + 1) % n
    out = arr.copy()
    out[donor] = max(0.0, out[donor] - delta)
    out[rec] = min(1000.0, out[rec] + delta)
    return out


def _score(arr: np.ndarray) -> float:
    """Convenient wrapper for evaluate_sequence."""
    return float(evaluate_sequence(arr.tolist()))
