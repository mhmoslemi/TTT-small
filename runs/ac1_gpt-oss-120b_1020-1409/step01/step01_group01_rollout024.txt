    import numpy as np
    rng = np.random.default_rng(seed)
    parent = globals().get('height_sequence_1')
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:

    best = a

    # hill climbing moves
    max_iters = 5000
    no_improve = 0
    for _ in range(max_iters):
        # pick indices i (early) and j (later)
        i = rng.integers(0, n//2)
        j = rng.integers(n//2, n)
        # amount to transfer, up to a[i]
        delta = rng.random() * min(best[i] * 0.5, 0.1)  # e.g., up to 10% of a[i] or 0.1
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] = min(1000.0, cand[j] + delta)
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 2000:
            break
    # if best_score not improved, fallback to simple r-seq
    if not np.isfinite(best_score):
        best = np.clip(generate_r_sequence(n), 0.0, 1000.0)
