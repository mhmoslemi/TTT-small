import numpy as np

def propose_candidate(seed=42, num_iters=3000, top_k=3, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the AC1 bound.
    Uses a multi‑peak multiplicative gradient descent with occasional
    random perturbations.  Falls back to a simple power‑law start if the
    provided parent is unusable.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and optional parent sequence.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = (
            parent_arr.ndim == 1
            and n > 0
            and np.isfinite(parent_arr).all()
        )
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # ------------------------------------------------------------------
    # Helper to compute the evaluator score.
    # ------------------------------------------------------------------
    def eval_score(arr):
        # arr is a 1‑D numpy array
        return float(evaluate_sequence(arr.tolist()))

    # ------------------------------------------------------------------
    # Initial sequence generation.
    # ------------------------------------------------------------------
    if parent_ok:
        a = np.clip(parent_arr, 0.0, 1000.0)
        s = a.sum()
        if s > 0.0 and np.isfinite(s):
            a = a / s
        else:
            a = np.ones(n, dtype=np.float64) / n
    else:
        # Power‑law start a_i = (i+1)^(-p) with p=0.5
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** -0.5
        a = a / a.sum()

    # Store the best sequence seen so far.
    best = a.copy()
    best_score = eval_score(best)

    # ------------------------------------------------------------------
    # Hyper‑parameters.
    # ------------------------------------------------------------------
    step0 = 0.6                # initial multiplicative step size
    decay = 0.9985             # exponential decay per iteration
    block_prob = 0.18          # probability of a block scaling perturbation
    transfer_prob = 0.30       # probability of a pairwise transfer
    eps_transfer = (0.05, 0.20)  # fraction range taken from source index

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    for it in range(num_iters):
        # Full convolution (linear) and top‑k index extraction.
        conv = np.convolve(a, a)
        # indices of the largest `top_k` entries (unsorted)
        if top_k >= conv.size:
            top_idxs = np.arange(conv.size)
        else:
            # argpartition gives the k largest unsorted.
            top_idxs = np.argpartition(conv, -top_k)[-top_k:]

        # Build gradient as sum of contributions from the selected peaks.
        g = np.zeros_like(a)
        for k in top_idxs:
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx_range = np.arange(low, high + 1, dtype=np.int64)
            partner = k - idx_range
            # Add 2*a[partner] to each participating coefficient.
            g[idx_range] += 2.0 * a[partner]

        # Multiplicative update.
        step = step0 * (decay ** it)
        a = a * np.exp(-step * g)

        # Renormalise to keep the L1‑norm fixed (scale‑invariant objective).
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.ones(n, dtype=np.float64) / n
            s = a.sum()
        a = a / s
        np.clip(a, 0.0, 1000.0, out=a)

        # ------------------------------------------------------------------
        # Random block scaling perturbation (occasionally).
        # ------------------------------------------------------------------
        if (it % 200 == 0) and (rng.random() < block_prob):
            low_w = max(1, n // 250)
            high_w = max(low_w + 1, n // 15)
            width = int(rng.integers(low_w, high_w))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            a_pert = a.copy()
            a_pert[start:start + width] *= factor
            np.clip(a_pert, 0.0, 1000.0, out=a_pert)
            a_pert = a_pert / a_pert.sum()
            score_pert = eval_score(a_pert)
            if score_pert < best_score:
                best = a_pert.copy()
                best_score = score_pert
                a = a_pert  # continue from improvement

        # ------------------------------------------------------------------
        # Pairwise mass transfer perturbation (occasionally).
        # ------------------------------------------------------------------
        if (it % 50 == 0) and (rng.random() < transfer_prob):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 0.0:
                frac = rng.uniform(eps_transfer[0], eps_transfer[1])
                delta = min(a[i] * frac, a[i])
                a_pert = a.copy()
                a_pert[i] -= delta
                a_pert[j] += delta
                # No need to clip; positivity is preserved.
                a_pert = a_pert / a_pert.sum()
                score_pert = eval_score(a_pert)
                if score_pert < best_score:
                    best = a_pert.copy()
                    best_score = score_pert
                    a = a_pert

        # ------------------------------------------------------------------
        # Record current best.
        # ------------------------------------------------------------------
        cur_score = eval_score(a)
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
