def propose_candidate(seed=42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    The routine scans several compact parametric families (power‑law,
    exponential, sigmoid, linear) and performs a brief hill‑climbing
    refinement.  All candidates respect the bounds [0,1000] and the sum
    constraint required by the evaluator.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initialise a fallback candidate.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best_seq = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Helper to enforce constraints (non‑negative, max ≤1000, sum ≥0.01).
    # Scaling does not change the objective, only the clipping does.
    # ------------------------------------------------------------------
    def _normalize(arr):
        arr = np.maximum(arr, 0.0)
        max_val = arr.max()
        if max_val == 0.0:
            return arr
        arr = arr / max_val                     # max == 1
        total = arr.sum()
        # raise sum to the minimal admissible value if needed
        scale_up = max(0.01 / total, 1.0)
        # never exceed the absolute bound 1000 after scaling
        scale_up = min(scale_up, 1000.0)
        arr = arr * scale_up
        np.clip(arr, 0.0, 1000.0, out=arr)
        return arr

    # ------------------------------------------------------------------
    # Baseline: uniform sequence (already a good reference).
    # ------------------------------------------------------------------
    uniform = np.ones(n, dtype=np.float64)
    uniform = _normalize(uniform)
    sc = float(evaluate_sequence(uniform.tolist()))
    if sc < best_score:
        best_seq, best_score = uniform, sc

    # ------------------------------------------------------------------

            if i != j:
                delta = cand[i] * rng.random() * 0.2
                cand[i] -= delta
                cand[j] += delta

        # enforce bounds and a minimal total mass
        np.clip(cand, 0.0, 1000.0, out=cand)
        total = cand.sum()
        if total < 0.01:
            cand *= (0.01 / total)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq, best_score = cand, sc

    # ------------------------------------------------------------------
    # Final sanity – should never happen, but keep a safe fallback.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score) or best_seq.size == 0:
        best_seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_seq]
