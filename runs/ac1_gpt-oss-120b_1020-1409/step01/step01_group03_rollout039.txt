import numpy as np

# ------------------------------------------------------------
# Fast convolution of a vector with itself using real FFT.
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Exact scoring function mirroring evaluate_sequence.
# ------------------------------------------------------------
def _score(x: np.ndarray) -> float:
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(x)
    m = c.max()
    return 2.0 * x.size * m / (s * s)


# ------------------------------------------------------------
# Parameterised families of candidate sequences.
# ------------------------------------------------------------
def _exp_shape(n: int, c: float) -> np.ndarray:
    """a_i = exp(-c * i / n)  (i = 0..n-1)."""
    r = np.exp(-c / n)
    idx = np.arange(n, dtype=np.float64)
    return r ** idx


def _two_phase_shape(n: int, c1: float, c2: float, p_frac: float) -> np.ndarray:
    """Early exponential with rate c1, then slower exponential with rate c2.
    p_frac ∈ (0,1) is the fraction where the switch occurs."""
    m = int(p_frac * n)
    if m < 1:
        m = 1
    if m >= n:
        m = n - 1
    idx = np.arange(n, dtype=np.float64)

    early = np.exp(-c1 * idx[:m] / n)
    base = np.exp(-c1 * m / n)                     # value at the switch
    tail = base * np.exp(-c2 * (idx[m:] - m) / n)  # slower decay after m
    return np.concatenate((early, tail))

# One projected‑gradient step (matrix‑free, no closures).
# ------------------------------------------------------------
def _grad_step(x: np.ndarray) -> tuple[np.ndarray, float]:
    """Return a candidate obtained from a single gradient‑like move and its score."""
    conv = _conv_self(x)
    M = conv.max()
    k = int(conv.argmax())
    n = x.size

    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1, dtype=int)

    g = np.zeros_like(x)
    g[idx] = x[k - idx]                      # derivative of the max‑convolution term

    S = x.sum()
    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S

    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

    # try a few decreasing learning rates, keep the best improvement
    best_local = x
    best_score = _score(x)
    lr = 1.0
    for _ in range(4):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _score(cand)
        if sc < best_score:
            return cand, sc
        lr *= 0.5
    return x, best_score
