import numpy as np
import math

def _build_sequence(M: float, n: int):
    """Construct the maximal ℓ1‑sequence for a given convolution bound M."""
    a = np.zeros(n, dtype=np.float64)
    conv = np.zeros(2 * n - 1, dtype=np.float64)

    a[0] = 1.0                     # a₀ must be 1 because a₀² ≤ M and we start from the extremal case
    conv[0] = 1.0
    sqrtM = math.sqrt(M)

    for k in range(1, n):
        cross = conv[k]            # sum_{i=1}^{k-1} a_i a_{k-i}
        val = (M - cross) * 0.5    # maximal a_k respecting 2·a_k ≤ M–cross
        if val < 0.0:
            val = 0.0
        if val > sqrtM:
            val = sqrtM
        if val > 1000.0:           # hard bound from the evaluator
            val = 1000.0
        a[k] = val
        if val == 0.0:
            continue
        # update convolution contributions of the new element
        conv[k:k + k] += 2.0 * val * a[:k]   # indices k+i, i=0..k‑1
        conv[2 * k] += val * val
    return a, conv


def _score_from_sequence(a: np.ndarray, conv: np.ndarray) -> float:
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a <= 0.0:
        return np.inf
    n = a.size
    return float(2.0 * n * max_b / (sum_a * sum_a))
