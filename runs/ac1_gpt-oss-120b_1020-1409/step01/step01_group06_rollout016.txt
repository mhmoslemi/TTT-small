import numpy as np

# ----------------------------------------------------------------------
def _conv_full(arr: np.ndarray) -> np.ndarray:
    """Linear convolution arr * arr using FFT, returns full result."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score_arr(arr: np.ndarray, n: int) -> float:
    """Exact score as used by the official evaluator."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_full(arr).max()
    return float(2.0 * n * max_b / (s * s))


def _normalize(arr: np.ndarray, target: float) -> np.ndarray:
    """Rescale in‑place so that sum(arr) == target (if possible)."""
    s = arr.sum()
    if s > 0.0:
        arr *= target / s
    return arr


def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """Generate a_i ∝ (i+β)^(-p); optionally add the reversed copy."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base

    if rng.random() < 0.2:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    _normalize(cand, target)
    return cand


def _multiplicative_perturb(vec: np.ndarray, rng: np.random.Generator, target: float) -> np.ndarray:
    """Apply independent log‑normal factors to each entry, then renormalise."""
    factor = np.exp(rng.normal(0.0, 0.03, size=vec.size))
    cand = vec * factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    _normalize(cand, target)
    return cand


def _additive_perturb(vec: np.ndarray, rng: np.random.Generator, target: float) -> np.ndarray:
    """Add small Gaussian noise, clip and renormalise."""
    delta = rng.normal(0.0, 0.02 * target, size=vec.size)
    cand = vec + delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    _normalize(cand, target)
    return cand


def _gradient_step(vec: np.ndarray, rng: np.random.Generator, target: float,
                   n: int, it: int, max_it: int) -> np.ndarray:
    """One projected gradient descent step on the maximal convolution entry."""
    conv = _conv_full(vec)
    k = int(np.argmax(conv))                      # index of current maximum
    # gradient: ∂/∂a_i (a * a)_k = 2 * a_{k-i} (if index valid)
    j = k - np.arange(n)
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(vec)
    grad[mask] = vec[j[mask]]
    grad *= 2.0
    # step size shrinks linearly
    lr = 0.12 * (1.0 - it / max_it) * target / (np.max(grad) + 1e-12)
    cand = vec - lr * grad
    np.clip(cand, 0.0, 1000.0, out=cand)
    _normalize(cand, target)
    return cand


def _mass_transfer(vec: np.ndarray, rng: np.random.Generator, target: float) -> np.ndarray:
    """Transfer a small fraction of mass from a high‑contributing index to a random one."""
    cand = vec.copy()
    n = cand.size
    conv = _conv_full(cand)
    k = int(np.argmax(conv))
    i_idx = np.arange(n)
    j = k - i_idx
    mask = (j >= 0) & (j < n)
    prod = np.zeros_like(cand)
    prod[mask] = cand[i_idx[mask]] * cand[j[mask]]
    i_max = int(np.argmax(prod))
    # pick a destination distinct from source
    j_rand = int(rng.integers(0, n))
    while j_rand == i_max:
        j_rand = int(rng.integers(0, n))
    delta = min(cand[i_max], cand[i_max] * 0.02)   # move up to 2 % of the source mass
    cand[i_max] -= delta
    cand[j_rand] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    _normalize(cand, target)
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step function (as a Python list) whose
    AC1 upper bound is as low as possible.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Length of the problem; fall back to a modest size if the parent is missing.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    target_sum = np.sqrt(2.0 * n)          # any scaling cancels in the score

    # ------------------------------------------------------------------
    # Assemble a diverse set of seed vectors.
    seeds = []

    # uniform (baseline)
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    seeds.append(uniform)

    # normalised parent
    p0 = parent.copy()
    _normalize(p0, target_sum)
    seeds.append(p0)

    # power‑law family (asymmetric and symmetric)
    p_vals = np.linspace(0.25, 0.90, 9)
    beta_vals = np.logspace(-3, 2, 12)
    for sym in (False, True):
        for p in p_vals:
            for beta in beta_vals:
                a = _make_powerlaw(n, p, beta, sym)
                _normalize(a, target_sum)
                seeds.append(a)

    # exponential decay family
    lam_vals = np.logspace(-3, 0, 10)
    idx = np.arange(n, dtype=np.float64)
    for lam in lam_vals:
        a = np.exp(-lam * idx)
        _normalize(a, target_sum)
        seeds.append(a)

    # a few random Dirichlet‑like starts
    for _ in range(4):
        a = rng.exponential(scale=1.0, size=n)
        _normalize(a, target_sum)
        seeds.append(a)

    # ------------------------------------------------------------------
    # Pick the best among the seeds.
    best_vec = seeds[0].copy()
    best_score = _score_arr(best_vec, n)
    for vec in seeds[1:]:
        sc = _score_arr(vec, n)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()

    # ------------------------------------------------------------------
    # Long stochastic refinement (mix of several simple perturbations).
    max_iter = 8000
    for it in range(max_iter):
        typ = rng.integers(0, 5)               # five perturbation flavours
        if typ == 0:
            cand = _block_perturb(best_vec, rng, target_sum)
        elif typ == 1:
            cand = _multiplicative_perturb(best_vec, rng, target_sum)
        elif typ == 2:
            cand = _additive_perturb(best_vec, rng, target_sum)
        elif typ == 3:
            cand = _gradient_step(best_vec, rng, target_sum, n, it, max_iter)
        else:
            cand = _mass_transfer(best_vec, rng, target_sum)

        sc = _score_arr(cand, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # Return as plain Python list (the evaluator will clamp/finalise it).
    return [float(x) for x in best_vec]
