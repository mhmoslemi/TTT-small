def convolve_fft(x, y, size=None):
    n = len(x) + len(y) - 1
    N = 1 << (n-1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X*Y
    z = np.fft.irfft(Z, N)[:n]
    return z
```

def softmax(conv, alpha):
    cmax = conv.max()
    exp = np.exp(alpha * (conv - cmax))
    sumexp = exp.sum()
    w = exp / sumexp
    return w

def propose_candidate(seed=42, num_iters=3000, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine n and optional parent
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent_arr.size
        parent_ok = parent_arr.ndim ==1 and n>0 and np.isfinite(parent_arr).all()
    except Exception:
        parent_arr = None
        n = 2000
        parent_ok = False

    # init candidate a
    if parent_ok:
        a = np.clip(parent_arr, 0.0, 1000.0)
        s = a.sum()
        if s > 0 and np.isfinite(s):
            a = a / s
        else:
            a = np.ones(n) / n
    else:
        # use power law
        a = 1.0 / np.sqrt(np.arange(1, n+1, dtype=np.float64))
        a = a / a.sum()
    sum_target = 1.0

    # best candidate
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # optionally evaluate shape candidates as fallback for best
    def try_candidate(c):
        nonlocal best, best_score
        cand = np.clip(c, 0.0, 1000.0)
        cand_sum = cand.sum()
        if cand_sum ==0:
            return
        cand = cand / cand_sum * sum_target
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # evaluate some handcrafted shapes
    tri = triangular_shape(n)
    try_candidate(tri)
    hann = 0.5 - 0.5 * np.cos(2*np.pi * np.arange(n)/ (n-1))
    try_candidate(hann)
    # maybe sine shape
    sine = np.sin(np.pi * (np.arange(n)+0.5)/ n)
    try_candidate(sine)

    # gradient descent parameters
    step0 = 0.25
    decay = 0.997
    beta = 0.9   # momentum factor
    m = np.zeros_like(a)
    # iteration loop
    for it in range(num_iters):
        # compute convolution
        conv = np.convolve(a, a)
        # locate max
        k = int(np.argmax(conv))

        # gradient for max index
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high+1, dtype=np.int64)
        partner = k - idx
        # g = zeros and assign 2*a[partner]
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[partner]

        # momentum update
        m = beta * m + (1 - beta) * g

        # step decay
        step = step0 * (decay ** it)

        # multiplicative update (stay nonnegative)
        a = a * np.exp(-step * m)

        # renormalize and clip
        a = np.clip(a, 0.0, 1000.0)
        a_sum = a.sum()
        if a_sum == 0 or not np.isfinite(a_sum):
            a = np.ones(n) / n
            a_sum = a.sum()
        a = a / a_sum * sum_target

        # occasional random block perturbation
        if (it % 150 == 0) and rng.random() < 0.25:
            width = int(rng.integers(max(1, n//300), max(2, n//20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.15)))
            a_pert = a.copy()
            a_pert[start:start+width] *= factor
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_pert = a_pert / a_pert.sum() * sum_target
            sc = float(evaluate_sequence(a_pert.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = a_pert.copy()
                best_score = sc
                a = a_pert.copy()  # continue from improvement

        # occasional pairwise exchange

            i,j = rng.choice(n, size=2, replace=False)

            a_pair[i] -= delta

def normalize_seq(seq: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """Clip to [0,1000], renormalise to target_sum."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s == 0 or not np.isfinite(s):
        seq = np.ones_like(seq) / seq.size
        s = seq.sum()
    return seq * (target_sum / s)


def triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular hump (height proportional to distance from ends)."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    a = np.where(i <= half, i + 1.0, n - i)
    return normalize_seq(a)


def hann_shape(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return normalize_seq(a)


def sine_shape(n: int) -> np.ndarray:
    """Half‑sine window (positive, zero at ends)."""
    i = np.arange(n, dtype=np.float64)
