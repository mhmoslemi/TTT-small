def conv_max(seq, nfft=None):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    if nfft is None:
        nfft = 1 << ((2*n-1).bit_length())

```
def gradient_step(a, km, eta):
    n = a.size
    if km == 0:
        g = 2.0 * a
    else:
        g = np.zeros_like(a)
        g[:n-km] += a[km:]
        g[km:] += a[:n-km]
    a_new = a - eta * g
    np.clip(a_new, 0.0, None, out=a_new)  # enforce nonnegativity
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        # fallback uniform
        a_new = np.full(n, 1.0/n, dtype=np.float64)
    return a_new
```
