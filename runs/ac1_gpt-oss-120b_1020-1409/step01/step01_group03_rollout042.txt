def compute_ratio(r, n):
    if r <=0: return inf
    if r >= 1: return 2.0   # uniform case.
    # sum
    r_n = r**n
    S = (1 - r_n) / (1 - r)
    # compute max conv
    max_val = 0.0
    rp = 1.0
    for k in range(2*n -1):
        t = k+1 if k < n else 2*n-1 - k
        val = t * rp
        if val > max_val:
            max_val = val
        rp *= r
    # conv_max = max_val
    ratio = 2.0 * n * max_val / (S * S)
    return ratio

def _grad_step(x):
    conv = _conv_self_fft(x)
    M = conv.max()
    k = int(conv.argmax())
    S = x.sum()
    n = x.shape[0]
    # Compute g vector: g_i = x[k - i] if 0 <= k-i < n else 0
    # Pre-allocate zeros and fill within bounds:
    g = np.zeros_like(x)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    # indices in x that affect M: i in i_low..i_high
    # For these, j = k - i in range.
    idx = np.arange(i_low, i_high + 1, dtype=int)
    g[idx] = x[k - idx]  # this populates g at idx.
    inv_S2 = 1.0 / (S * S) if S != 0 else 0.0
    inv_S3 = inv_S2 / S if S !=0 else 0.0
    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3
    # Try line search
    best_local = x
    best_local_score = _score_fast(x)
    lr = 1.0
    for _ in range(7):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        if rng.random() < 0.1:
            cand = 0.5 * (cand + cand[::-1])
        sc = _score_fast(cand)
        if sc < best_local_score:
            best_local = cand
            best_local_score = sc
            break
        lr *= 0.5
    return best_local, best_local_score

            break

import numpy as np

# ------------------------------------------------------------
# Basic building blocks
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Catalan‑type series a_i = C(2i,i)/4^i for i=0..length‑1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself using an FFT (real input)."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


def _score_fast(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (uses same clipping rules)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    return 2.0 * seq.size * c.max() / (s * s)

# ------------------------------------------------------------
def _grad_step(x: np.ndarray, rng: np.random.Generator) -> tuple:
    """One projected gradient step; returns (new_x, new_score)."""
    n = x.shape[0]
    conv = _conv_self(x)
    M = conv.max()
    k = int(conv.argmax())
    S = x.sum()
    if S == 0.0:
        return x, float('inf')

    # build gradient of M w.r.t x (subgradient at the active max index)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    g = np.zeros_like(x)
    g[idx] = x[k - idx]

    inv_S2 = 1.0 / (S * S)
    inv_S3 = inv_S2 / S
    grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

    # line‑search with a few candidates
    best_local = x
    best_score = _score_fast(x)
    lr = 1.0
    for _ in range(6):
        cand = x - lr * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        # occasional cheap symmetrisation (helps a bit)
        if rng.random() < 0.12:
            cand = 0.5 * (cand + cand[::-1])
        sc = _score_fast(cand)
        if sc < best_score:
            best_local = cand
            best_score = sc
            break
        lr *= 0.5
    return best_local, best_score

def _optimize(start: np.ndarray, rng: np.random.Generator,
              max_steps: int = 4000) -> tuple:
    """Run many gradient steps on start; keep best ever seen."""
    x = start.copy()
    best = x.copy()
    best_score = _score_fast(best)
    stagn = 0

    for step in range(max_steps):

        # occasional block‑scale perturbation to escape plateaus

            block = x_new.copy()
            factor = np.exp(rng.normal(0.0, 0.12))
            block[st:st + width] *= factor
            np.clip(block, 0.0, 1000.0, out=block)
            sc_block = _score_fast(block)
            if sc_block < sc_new:

            best_score = sc_new
            stagn = 0
        else:
