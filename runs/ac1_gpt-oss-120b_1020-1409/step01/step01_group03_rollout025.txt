import numpy as np

# ------------------------------------------------------------
# Helper: classic decreasing series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return first `length` terms of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a

# FFT‑based self‑convolution (real inputs)
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


# ------------------------------------------------------------
# Fast bound computation, identical to `evaluate_sequence`
# ------------------------------------------------------------
def _score(x: np.ndarray) -> float:
    """2*n*max(conv(x,x))/sum(x)^2 with clipping as the evaluator."""
    x = np.clip(x, 0.0, 1000.0)
    s = x.sum()
    if s < 0.01:
        return float("inf")
    c = _conv_self(x)
    m = c.max()
    return 2.0 * x.size * m / (s * s)


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iters: int = 3000, **kwargs):
    """
    Return a non‑empty list of non‑negative heights that (hopefully) reduces the
    AC1 upper bound.  The routine starts from the provided parent sequence
    `height_sequence_1` (if present) or from the analytical series, then performs
    a projected gradient descent with occasional random block perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Load parent or fallback initialisation
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        x = np.clip(parent, 0.0, 1000.0)
        n = x.size
    except Exception:
        n = 256
        x = _make_series(n)
        x = np.clip(x, 0.0, 1000.0)

    if x.sum() < 0.01:
        x = np.ones(n, dtype=np.float64)

    best = x.copy()
    best_score = _score(best)
    curr = x.copy()
    curr_score = best_score
    stagnant = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    for it in range(max_iters):
        # Full convolution and identification of the active index
        conv = _conv_self(curr)
        M = conv.max()
        k = int(conv.argmax())
        S = curr.sum()
        if S == 0.0:
            break

        # Gradient of f = 2*n*M/S^2 (exact sub‑gradient)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g = np.zeros_like(curr)
        g[idx] = curr[k - idx]

        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

        # ----------------------------------------------------
        # Try a few learning rates (projected gradient)
        # ----------------------------------------------------
        improved = False
        cand = curr.copy()
        cand_score = curr_score
        lr = 1.0
        for _ in range(5):
            cand = curr - lr * grad
            np.clip(cand, 0.0, 1000.0, out=cand)
            if rng.random() < 0.1:                # occasional symmetrisation
                cand = 0.5 * (cand + cand[::-1])
            sc = _score(cand)
            if sc < cand_score:
                cand_score = sc
                improved = True
                break
            lr *= 0.5

        # ----------------------------------------------------
        # If gradient step failed, use a random block perturbation
        # ----------------------------------------------------
        if not improved:
            if it % 19 == 0:
                width = max(1, n // rng.integers(8, 20))
                start = rng.integers(0, max(1, n - width + 1))
                stop = start + width
                factor = np.exp(rng.normal(0.0, 0.12))
                block = curr.copy()
                block[start:stop] *= factor
                np.clip(block, 0.0, 1000.0, out=block)
                if rng.random() < 0.15:
                    block = 0.5 * (block + block[::-1])
                sc_block = _score(block)
                if sc_block < cand_score:
                    cand = block
                    cand_score = sc_block
                    improved = True

            # small random mass transfer between two entries
            if rng.random() < 0.05:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    delta = (rng.random() - 0.5) * 0.2 * curr[i]   # move up to ±10 % of a_i
                    new_i = np.clip(curr[i] - delta, 0.0, 1000.0)
                    new_j = np.clip(curr[j] + delta, 0.0, 1000.0)
                    cand2 = curr.copy()
                    cand2[i] = new_i
                    cand2[j] = new_j
                    sc2 = _score(cand2)
                    if sc2 < cand_score:
                        cand = cand2
                        cand_score = sc2
                        improved = True

        # ----------------------------------------------------
        # Update the global best if we improved
        # ----------------------------------------------------
        if cand_score + 1e-14 < best_score:
            best = cand.copy()
            best_score = cand_score
            stagnant = 0
        else:
            stagnant += 1

        # ----------------------------------------------------
        # Accept the candidate as current point if it improved locally
        # ----------------------------------------------------
        if cand_score < curr_score:
            curr = cand.copy()
            curr_score = cand_score
        else:
            # occasional restart from the best to escape flat regions
            if stagnant > 200 and rng.random() < 0.3:
                curr = best.copy()
                curr_score = best_score
                stagnant = 0

        # ----------------------------------------------------
        # Early stop if no progress for many iterations
        # ----------------------------------------------------
        if stagnant > 500:
            break

    # --------------------------------------------------------
    # Final safety check and return
    # --------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    return [float(v) for v in best]
