def compute_score(arr):
    n = arr.shape[0]
    # compute convolution using FFT
    m = 1 << ((2*n - 1).bit_length())  # next power of two
    fft_a = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
    max_conv = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_conv / (sum_a * sum_a))

import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the sequence r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq

def _conv_max(arr: np.ndarray) -> float:
    """Return the maximum element of the full convolution of arr with itself."""
    n = arr.shape[0]
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    fft_a = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2 * n - 1]
    return float(conv.max())

def compute_score(arr: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence, but computed via FFT."""
    n = arr.shape[0]
    sum_a = arr.sum()
    if sum_a < 0.01:
        return float('inf')
    max_conv = _conv_max(arr)
    return float(2.0 * n * max_conv / (sum_a * sum_a))

def propose_candidate(seed: int = 42, **kwargs):

    early_end = max(1, n // 3)
