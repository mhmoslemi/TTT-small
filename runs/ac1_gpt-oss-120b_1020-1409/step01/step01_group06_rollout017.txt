def make_powerlaw(n, p, beta, sym):
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base

def make_exponential(n, alpha, sym):
    idx = np.arange(n, dtype=np.float64)
    base = np.exp(-alpha * idx)
    if sym:
        base = base + np.exp(-alpha * (n-1 - idx))
    return base

import numpy as np

# ----------------------------------------------------------------------
# Convolution max via FFT (exact for real non‑negative data)
def _conv_max(arr: np.ndarray) -> float:
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _score(arr: np.ndarray, n: int) -> float:
    """Exact evaluator metric (clipping included)."""
    a = np.clip(arr, 0.0, 1000.0, out=None)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    max_b = _conv_max(a)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^(-p) , optionally symmetrised."""
    i = np.arange(n, dtype=np.float64)
    base = (i + beta) ** (-p)
    if sym:
        base = base + (i[::-1] + beta) ** (-p)
    return base


def _make_exponential(n: int, alpha: float, sym: bool) -> np.ndarray:
    """a_i ∝ exp(-α·i) , optionally symmetrised."""
    i = np.arange(n, dtype=np.float64)
    base = np.exp(-alpha * i)
    if sym:
        base = base + np.exp(-alpha * (n - 1 - i))
    return base


# ----------------------------------------------------------------------
def _normalize_clip(arr: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to [0,1000], scale to target L1‑norm, clamp again."""
    a = np.clip(arr, 0.0, 1000.0, out=None)
    s = a.sum()
    if s <= 0.0:
        # degenerate – fall back to uniform
        a = np.full_like(a, target_sum / a.size)
    else:
        a = a * (target_sum / s)
        a = np.clip(a, 0.0, 1000.0, out=None)
    return a


# ----------------------------------------------------------------------
def _perturb_block(vec: np.ndarray, rng: np.random.Generator,
                   target_sum: float, sigma: float) -> np.ndarray:
    """Scale a random contiguous block (log‑normal factor)."""
    n = vec.size
    cand = vec.copy()
    min_w = max(1, n // 200)
    max_w = max(2, n // 10)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:start + width] *= factor
    return _normalize_clip(cand, target_sum)


def _perturb_smooth(vec: np.ndarray, rng: np.random.Generator,
                    target_sum: float) -> np.ndarray:
    """Replace by a short moving‑average (kernel width 3‑7)."""
    n = vec.size
    width = int(rng.integers(3, 8))            # inclusive 3..7
    kernel = np.ones(width, dtype=np.float64) / width
    smooth = np.convolve(vec, kernel, mode='same')
    return _normalize_clip(smooth, target_sum)


def _perturb_mirror(vec: np.ndarray, target_sum: float) -> np.ndarray:
    """Average with its reversal (enforces near symmetry)."""
    cand = 0.5 * (vec + vec[::-1])
    return _normalize_clip(cand, target_sum)
