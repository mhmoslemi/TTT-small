def _arcsine_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (1.0 - 0.5 / i)
    return a

def _autocorr_fft(seq):
    n = seq.size
    # next power of two >= 2*n -1
    size = 1 << ((2*n - 1).bit_length())
    # compute rfft of zero-padded sequence
    freq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(freq * freq, n=size)
    return conv[:2*n-1]

def propose_candidate(...):
    rng = np.random.default_rng(seed)
    # Determine n and initial seq.
    seq = parent if valid else arcsine(n)
    seq = np.clip(seq, 0, 1000)
    if seq.sum() < 0.01:
        seq += 0.01 / len(seq)

    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())

    total_sum = seq.sum()  # for scaling after modifications

    max_iter = 2000  # maybe more

    for it in range(max_iter):
        # compute conv via FFT
        conv = _autocorr_fft(seq)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # compute product contributions
        idx_arr = max_idx - np.arange(n)
        valid = (idx_arr >= 0) & (idx_arr < n)
        prod = np.zeros_like(seq)
        prod[valid] = seq * seq[idx_arr[valid]]

        # find i with largest product
        i = int(np.argmax(prod))

        # choose partner j randomly
        j = int(rng.integers(n))

        if i == j:
            # skip if equal
            continue

        # transfer a fraction of mass from i to j
        # limit: up to half of seq[i]
        max_transfer = seq[i] * 0.5
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random()
        if delta <= 0:
            continue

        cand = seq.copy()
        cand[i] -= delta
        cand[j] += delta

        # smooth a bit later?

        # Clip and rescale to keep sum constant
        np.clip(cand, 0.0, 1000.0, out=cand)
        # adjust sum scaling
        sum_cand = cand.sum()
        if sum_cand <= 0:
            continue
        cand *= (total_sum / sum_cand)  # keep sum unchanged

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))

        # accept if better, or simulated annealing
        temperature = max(1e-4, 1.0 / (it + 1))
        improve = score < best_score
        if improve or (score < float('inf') and np.exp((best_score - score) / temperature) > rng.random()):
            seq = cand
            if improve:
                best_seq = cand.copy()
                best_score = score

        # occasional smoothing
        if rng.random() < 0.05:
            # apply smoothing pass
            smoothed = seq.copy()
            smoothed[1:-1] = 0.25*seq[:-2] + 0.5*seq[1:-1] + 0.25*seq[2:]
            # keep ends unchanged
            np.clip(smoothed, 0.0, 1000.0, out=smoothed)
            # rescale sum
            sum_sm = smoothed.sum()
            if sum_sm > 0:
                smoothed *= (total_sum / sum_sm)
                # evaluate
                s_score = float(evaluate_sequence(smoothed.tolist()))
                if s_score < best_score:
                    seq = smoothed
                    best_seq = smoothed.copy()
                    best_score = s_score

        # occasional block perturbation
        if rng.random() < 0.02:
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.1)))
            cand2 = seq.copy()
            cand2[start:start+width] *= factor
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            sum_c2 = cand2.sum()
            if sum_c2 > 0:
                cand2 *= (total_sum / sum_c2)
                s2_score = float(evaluate_sequence(cand2.tolist()))
                if s2_score < best_score:
                    seq = cand2
                    best_seq = cand2.copy()
                    best_score = s2_score

        # reduce some parameters (optional)

    # fallback
    if best_seq.size == 0 or not np.isfinite(best_score):
        best_seq = np.ones(256, dtype=np.float64)

    # return as list
    return [float(x) for x in best_seq]

import numpy as np

# ----------------------------------------------------------------------
# Helper generators
# ----------------------------------------------------------------------
def _arcsine_sequence(n: int) -> np.ndarray:
    """Arcsine‐type decreasing sequence: a[i] = a[i‑1]*(1‑0.5/(i+1))."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (1.0 - 0.5 / (i + 1))
    return a


def _semicircle_sequence(n: int) -> np.ndarray:
    """Half‑semicircle: a_i = sqrt(1‑((i+0.5)/n)^2)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    return a


def _cosine_sequence(n: int) -> np.ndarray:
    """Monotonically decreasing cosine: a_i = cos(pi*(i+0.5)/(2n))."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i + 0.5) / (2.0 * n))
    a[a < 0] = 0.0
    return a


def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^(-p)  (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


# ----------------------------------------------------------------------
# Convolution (FFT, linear memory)
# ----------------------------------------------------------------------
def _autocorr_fft(a: np.ndarray) -> np.ndarray:
    """Return full linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    freq = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(freq * freq, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Parent handling
# ----------------------------------------------------------------------
def _parent_sequence() -> np.ndarray | None:
    """Return height_sequence_1 if it exists and is valid, else None."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            arr.ndim == 1
            and arr.size > 0
            and np.isfinite(arr).all()
        ):
            return np.clip(arr, 0.0, 1000.0)
    except Exception:  # pragma: no cover
        pass
    return None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, max_iter: int = 2000, **kwargs) -> list:
    """
    Produce a non‑negative step function that (hopefully) improves the AC1 bound.
    The routine builds several analytic seeds, picks the best, then runs a
    lightweight stochastic refinement that respects memory limits.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and initial candidates
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    if parent is not None:
        n = parent.size
    else:
        n = 256                                     # arbitrary safe fallback size

    # generate a pool of analytic constructions
    candidates = []

    if parent is not None:
        candidates.append(parent.copy())

    candidates.append(_arcsine_sequence(n))
    candidates.append(_semicircle_sequence(n))
    candidates.append(_cosine_sequence(n))
    candidates.append(np.ones(n, dtype=np.float64))  # uniform

    # a few power‑law exponents around 0.5
    for p in (0.35, 0.40, 0.45, 0.55, 0.60):
        candidates.append(_power_law_sequence(n, p))

    # ------------------------------------------------------------------
    # Evaluate the pool and keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for cand in candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:                       # avoid degenerate sum
            cand += 0.01 / n
        try:
            sc = float(evaluate_sequence(cand.tolist()))
        except Exception:                           # defensive
            sc = np.inf
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Stochastic refinement (mass‑transfer + occasional smoothing)
    # ------------------------------------------------------------------
    if best_seq is None:                            # safety net
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    seq = best_seq.copy()
    total_mass = seq.sum()                          # keep this constant

    for step in range(max_iter):
        # ----- convolution and dominant index -----
        conv = _autocorr_fft(seq)
        k_max = int(np.argmax(conv))

        # ----- contribution products for the dominant index -----
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        prod = np.zeros_like(seq)
        prod[valid] = seq * seq[idx[valid]]

        # pick the pair responsible for the largest product
        i = int(np.argmax(prod))
        if prod[i] == 0.0:
            # nothing to improve – break early
            break

        # random partner index
        j = int(rng.integers(n))
        if i == j:
            continue

        # transfer a modest fraction of mass from i to j
        max_transfer = 0.5 * seq[i]
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * rng.random()
        cand = seq.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        # renormalise to keep total mass unchanged
        sum_cand = cand.sum()
        if sum_cand > 0.0:
            cand *= (total_mass / sum_cand)

        # ----- evaluate -----
        try:
            sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            sc = np.inf

        # ----- acceptance (simulated annealing) -----
        temperature = max(1e-4, 1.0 / (step + 1.0))
        if sc < best_score or (np.isfinite(sc) and np.exp((best_score - sc) / temperature) > rng.random()):
            seq = cand
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

        # ----- occasional smoothing pass -----
        if rng.random() < 0.06:
            smooth = seq.copy()
            # interior 3‑point averaging (linear, O(n))
            smooth[1:-1] = (
                0.25 * seq[:-2] + 0.5 * seq[1:-1] + 0.25 * seq[2:]
            )
            np.clip(smooth, 0.0, 1000.0, out=smooth)
            sum_smooth = smooth.sum()
            if sum_smooth > 0.0:
                smooth *= (total_mass / sum_smooth)
                try:
                    sc_s = float(evaluate_sequence(smooth.tolist()))
                except Exception:
                    sc_s = np.inf
                if sc_s < best_score:
                    seq = smooth
                    best_seq = smooth.copy()
                    best_score = sc_s

        # ----- occasional block perturbation (exploration) -----
        if rng.random() < 0.02:
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            blk = seq.copy()
            blk[start : start + width] *= factor
            np.clip(blk, 0.0, 1000.0, out=blk)
            sum_blk = blk.sum()
            if sum_blk > 0.0:
                blk *= (total_mass / sum_blk)
                try:
                    sc_b = float(evaluate_sequence(blk.tolist()))
                except Exception:
                    sc_b = np.inf
                if sc_b < best_score:
                    seq = blk
                    best_seq = blk.copy()
                    best_score = sc_b

    # ------------------------------------------------------------------
    # Final safety fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(256, dtype=np.float64)

    return [float(x) for x in best_seq]
