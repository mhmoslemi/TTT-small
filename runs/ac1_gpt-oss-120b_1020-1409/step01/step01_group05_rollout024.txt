import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step sequence that minimizes the AC1 upper bound.
    Uses a lightweight parametric sweep (power‑law and exponential families)
    followed by a short multiplicative gradient refinement.
    """
    # ----------------------------------------------------------------------
    # Determine problem size and optional parent sequence.
    # ----------------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent_arr.size)
        parent_ok = (
            parent_arr.ndim == 1
            and n > 0
            and np.isfinite(parent_arr).all()
        )
    except Exception:          # height_sequence_1 not present or malformed
        parent_arr = None
        n = 2000
        parent_ok = False

    # ----------------------------------------------------------------------
    # Helper: evaluate a numpy vector (will be normalised to sum 1 internally).
    # ----------------------------------------------------------------------
    def _score(vec):
        # enforce bounds, normalise L1 and evaluate
        vec = np.clip(vec, 0.0, 1000.0)
        s = vec.sum()
        if s <= 0 or not np.isfinite(s):
            return float('inf')
        vec = vec / s
        return float(evaluate_sequence(vec.tolist()))

    # ----------------------------------------------------------------------
    # Initialise best candidate (parent if available, else uniform).
    # ----------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    if parent_ok:
        pseq = np.clip(parent_arr, 0.0, 1000.0)
        s = pseq.sum()
        if s > 0:
            pseq = pseq / s
        else:
            pseq = np.full(n, 1.0 / n)
        sc = _score(pseq)
        if np.isfinite(sc):
            best_seq, best_score = pseq, sc

    # uniform fallback (always valid)
    uniform = np.full(n, 1.0 / n)
    sc = _score(uniform)
    if sc < best_score:
        best_seq, best_score = uniform, sc

    # ----------------------------------------------------------------------
    # Parameter sweep: power‑law a_i = (i+1)^{-p}
    # ----------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)      # 1 … n
    p_vals = np.linspace(0.45, 0.70, 30)            # reasonable range
    for p in p_vals:
        a = idx ** (-p)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_seq, best_score = a, sc

        a_rev = a[::-1]                     # large mass at the opposite end
        sc = _score(a_rev)
        if sc < best_score:
            best_seq, best_score = a_rev, sc

        a_sym = (a + a_rev) * 0.5           # symmetric blend
        sc = _score(a_sym)
        if sc < best_score:
            best_seq, best_score = a_sym, sc

    # ----------------------------------------------------------------------
    # Parameter sweep: exponential a_i = r^{i}
    # ----------------------------------------------------------------------
    i_arr = np.arange(n, dtype=np.float64)
    r_vals = np.linspace(0.90, 0.999, 20)
    for r in r_vals:
        a = r ** i_arr
        if a.sum() == 0:
            continue
        a = a / a.sum()
        sc = _score(a)
        if sc < best_score:
            best_seq, best_score = a, sc

        a_rev = a[::-1]
        sc = _score(a_rev)
        if sc < best_score:
            best_seq, best_score = a_rev, sc

    # ----------------------------------------------------------------------
    # Short multiplicative gradient refinement starting from the best so far.
    # ----------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        n2 = n * 2 - 1                     # length of full convolution
        steps = 300
        base_step = 0.15
        for it in range(steps):
            # convolution via FFT (linear time, memory O(n))
            fa = np.fft.rfft(a, n2)
            conv = np.fft.irfft(fa * fa, n2)

            k = int(np.argmax(conv))       # index of the current maximal entry
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx_range = np.arange(low, high + 1, dtype=np.int64)
            partner = k - idx_range
            g = np.zeros_like(a)
            g[idx_range] = 2.0 * a[partner]

            step = base_step * (0.99 ** it)

            s = a.sum()
            if s <= 0 or not np.isfinite(s):

                s = a.sum()
            a = a / s
            np.clip(a, 0.0, 1000.0, out=a)

            # occasional evaluation to capture improvement
            if it % 20 == 0:
                sc = _score(a)
                if sc < best_score:
                    best_seq, best_score = a.copy(), sc

    # ----------------------------------------------------------------------
    # Final safeguard: if somehow no valid sequence was found, return uniform.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Return a plain Python list of floats
    return [float(x) for x in best_seq]
