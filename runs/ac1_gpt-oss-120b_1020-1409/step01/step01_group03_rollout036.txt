import numpy as np

# ------------------------------------------------------------
# Helpers
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Central‑binomial (C(2i,i)/4^i) series, a classic decreasing shape."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self(x: np.ndarray) -> np.ndarray:
    """Full convolution via FFT (real input)."""
    n = x.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv


def _score_approx(seq: np.ndarray) -> float:
    """Fast version of the evaluator (identical clipping, FFT convolution)."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    m = _conv_self(seq).max()
    return 2.0 * seq.size * m / (s * s)


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Determine problem size from the supplied parent if possible.
    # --------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        parent = None
        n = 256                     # fallback length

    # --------------------------------------------------------
    # Initial candidate: either the parent (if good) or the classic series.
    # --------------------------------------------------------
    best = _make_series(n)
    best_score = _score_approx(best)

    if parent is not None:
        parent_clip = np.clip(parent, 0.0, 1000.0)
        try:
            parent_score = evaluate_sequence(parent_clip.tolist())
        except Exception:
            parent_score = float("inf")
        if parent_score < best_score:
            best = parent_clip.copy()
            best_score = parent_score

    # --------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------
    num_steps = 4000
    lr_start = 1.0
    lr_decay = 0.9995
    momentum = 0.9
    vel = np.zeros_like(best)           # momentum vector
    stagnant = 0                        # for early stopping

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    x = best.copy()
    lr = lr_start
    for step in range(num_steps):
        # ----- convolution and its maximal entry -----
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())
        S = x.sum()
        if S == 0.0:
            break

        # ----- gradient of the (unnormalised) objective -----
        # gradient of max‑convolution w.r.t. the active index k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g = np.zeros_like(x)
        g[idx] = x[k - idx]

        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3   # overall scale irrelevant

        # ----- momentum / line‑search step -----
        vel = momentum * vel - lr * grad
        cand = x + vel
        np.clip(cand, 0.0, 1000.0, out=cand)

        # occasional gentle symmetrisation (helps flatten peaks)
