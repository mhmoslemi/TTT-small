def _make_power_law(length, p):
    """Return a normalized non‑negative vector a_i = (i+1)^(-p)."""
    i = np.arange(1, length + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        a = np.ones(length, dtype=np.float64) / length
    else:
        a /= s
    return a
