def _block_perturb(vec, rng, target_sum):
    cand = vec.copy()
    n = cand.size
    # choose width relative to n
    min_w = max(1, n // 500)  # smaller width
    max_w = max(2, n // 20)   # larger width
    width = int(rng.integers(min_w, max_w+1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.05)))
    cand[start:start+width] *= factor
    # optionally mirror
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    # Renorm sum to target_sum
    s = cand.sum()
    if s > 0:
        cand *= target_sum / s
    return cand

import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT, size pre‑computed)
def _conv_max(arr: np.ndarray, size: int) -> float:
    """Maximum entry of the full linear convolution arr*arr using FFT."""
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # keep only the first 2*n-1 entries (the rest are padding zeros)
    n = arr.shape[0]
    return float(conv[: 2 * n - 1].max())


def _score(arr: np.ndarray, size: int) -> float:
    """Exact objective 2*n*max(conv)/(sum(arr)^2)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    m = _conv_max(arr, size)
    n = arr.shape[0]
    return float(2.0 * n * m / (s * s))


# ----------------------------------------------------------------------
# Shape families
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + base[::-1]
    return base


def _make_exponential(n: int, lam: float, sym: bool) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    base = np.exp(-lam * idx)
    if sym:
        base = base + base[::-1]
    return base


def _make_beta(n: int, a: float, b: float) -> np.ndarray:
    # Discrete Beta(α,β) on {0,…,n‑1}
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    base = np.power(x, a) * np.power(1.0 - x, b)
    return base


# ----------------------------------------------------------------------
# Simple stochastic moves
def _block_perturb(vec: np.ndarray, rng: np.random.Generator,
                   target_sum: float) -> np.ndarray:
    """Scale a random contiguous block and renormalise."""
    cand = vec.copy()
    n = cand.size
    min_w = max(1, n // 500)
    max_w = max(2, n // 20)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.05)))
    cand[start:start + width] *= factor
    if rng.random() < 0.3:                     # slight symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand


def _random_dirichlet_vec(n: int, rng: np.random.Generator,
                         target_sum: float) -> np.ndarray:
    """Draw a Dirichlet‑type vector, then permute."""
    alpha = rng.uniform(0.2, 4.0)
    g = rng.gamma(alpha, 1.0, size=n)
    vec = g / g.sum() * target_sum
    rng.shuffle(vec)
    return vec

    """

    # ------------------------------------------------------------------
    # Initial candidate – uniform (guaranteed valid)
    best_vec = np.full(n, target_sum / n, dtype=np.float64)

    # Exponential
