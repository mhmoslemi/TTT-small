import numpy as np

def _conv_max(arr):
    n = arr.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n - 1]
    return conv.max()

def _score(arr):
    s = arr.sum()
    if s == 0:
        return float('inf')
    max_b = _conv_max(arr)
    return float(2.0 * arr.shape[0] * max_b / (s * s))

def _make_exponential(n, rate):
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * idx)
    return a

def _make_linear(n):
    a = np.linspace(1.0, 0.0, n)
    return a

import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, exact for real non‑negative data)
def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full linear convolution arr * arr."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray) -> float:
    """Exact objective used by the official evaluator."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    n = arr.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Parameterised families of non‑negative sequences
def _make_exponential(n: int, rate: float) -> np.ndarray:
    """a_i = exp(-rate * i) , i = 0..n‑1."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-rate * idx)


def _make_linear(n: int) -> np.ndarray:
    """Linearly decreasing from 1 to 0."""
    return np.linspace(1.0, 0.0, n, dtype=np.float64)


def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """a_i ∝ (i+β)^{-p}; symmetrised if requested."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base
