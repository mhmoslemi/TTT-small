import numpy as np

def _generate_binomial_sequence(length: int) -> np.ndarray:
    """Return a_i = C(2i,i) / 4^i for i=0..length-1 (the classic binomial shape)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a

    Generate a non‑negative step‑function (list of floats) whose AC1 upper bound
    is substantially better than the simple binomial baseline.

    rng = np.random.default_rng(seed)

    conv = np.convolve(a, a)

    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # ------------------------------------------------------------------
    # 3.  Simple epsilon‑lifting of tiny tail entries ------------------------
    # ------------------------------------------------------------------
    epsilons = np.logspace(-3, -1, num=12)          # 0.001 … 0.1
    for eps in epsilons:
        cand = np.maximum(a, eps)                   # lift every entry below eps
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best = cand.copy()
            best_score = score
            a = best.copy()
            conv = np.convolve(a, a)

    # ------------------------------------------------------------------
    # 4.  Gradient‑free hill climbing (mass transfer) ----------------------
    # ------------------------------------------------------------------
    no_improve = 0
    for step in range(num_steps):
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # early steps move larger fractions, later steps finer adjustments
        if step < num_steps // 2:
            frac = rng.uniform(0.2, 0.9)
        else:
            frac = rng.uniform(0.01, 0.25)

        delta = a[i] * frac
        if a[j] + delta > 1000.0:
            delta = 1000.0 - a[j]
            if delta <= 0.0:

        inc_i = 2.0 * delta_i * a
        inc_j = 2.0 * delta_j * a
        conv[i:i + n] += inc_i
        conv[j:j + n] += inc_j

        conv[2 * i] += delta_i * delta_i
