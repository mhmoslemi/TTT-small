def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

import numpy as np

def _generate_binomial_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        i_f = float(i)
        a[i] = a[i-1] * ((2.0*i_f-1.0)*(2.0*i_f)) / (4.0*i_f*i_f)
    return a

def _generate_powerlaw_sequence(n, p=0.6, L=1):
    idx = np.arange(n, dtype=np.float64)
    arr = (idx + L) ** (-p)
    return arr

        if delta <= 0.0:
            break

        cand_seq = a.copy()
        cand_seq[donor] -= delta
        cand_seq[receiver] += delta
        np.clip(cand_seq, 0.0, 1000.0, out=cand_seq)

        sc = _evaluate(cand_seq)
        if np.isfinite(sc) and sc < best_score:
            a = cand_seq
            best = cand_seq
            best_score = sc

        # Occasionally apply a random block scaling to escape tiny plateaus
        if step % 250 == 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand_seq = a.copy()
            cand_seq[start:start + width] *= factor
            np.clip(cand_seq, 0.0, 1000.0, out=cand_seq)
            sc = _evaluate(cand_seq)
            if np.isfinite(sc) and sc < best_score:
                a = cand_seq
                best = cand_seq
                best_score = sc

    # Return a plain Python list of floats.
    return [float(v) for v in best]
