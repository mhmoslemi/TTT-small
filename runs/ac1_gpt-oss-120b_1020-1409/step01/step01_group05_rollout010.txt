            a_pert = np.clip(a_pert, 0.0, 1000.0)
            a_pert = a_pert / a_pert.sum()
            score_pert = float(evaluate_sequence(a_pert.tolist()))
            if np.isfinite(score_pert) and score_pert < best_score:
                best = a_pert.copy()
                best_score = score_pert
                a = a_pert
                last_improve = it

        # ---- Evaluate current candidate ----------------------------------
        cur_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(cur_score) and cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            last_improve = it

        # ---- Early stopping if stagnant ----------------------------------
        if it - last_improve > patience:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
