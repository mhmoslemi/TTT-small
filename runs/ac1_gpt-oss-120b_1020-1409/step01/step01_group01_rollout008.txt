def propose_candidate(seed=42, **kwargs):
    # Determine n
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256

    # Base init:
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size == n:
        base_seq = parent.astype(np.float64).copy()
        base_seq = np.clip(base_seq, 0.0, 1000.0)
    else:
        base_seq = generate_r_sequence(n)

    best_seq = base_seq.copy()
    best_score = compute_score(best_seq.tolist())  # compute using compute_score

    # Candidate search
    # define L candidates
    L_candidates = [int(n**0.25), int(n**0.33), int(n**0.5), int(n**0.66), int(n**0.75), n//4, n//2]
    L_candidates = sorted(set([l for l in L_candidates if l > 0 and l < n]))
    delta_candidates = [0.01, 0.05, 0.1, 0.2, 0.5, 1.0]  # additive
    scale_candidates = [1.01, 1.02, 1.05, 1.1, 1.2]      # multiplicative

    for L in L_candidates:
        # additive
        for delta in delta_candidates:
            cand = best_seq.copy()
            cand[:L] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = compute_score(cand.tolist())
            if np.isfinite(score) and score < best_score:
                best_seq = cand
                best_score = score
        # multiplicative
        for scale in scale_candidates:
            cand = best_seq.copy()
            cand[:L] *= scale
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = compute_score(cand.tolist())
            if np.isfinite(score) and score < best_score:
                best_seq = cand
                best_score = score

    # Also try suffix modifications similarly
    for L in L_candidates:
        # work on suffix
        start = n - L
        for delta in delta_candidates:
            cand = best_seq.copy()
            cand[start:] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = compute_score(cand.tolist())
            if np.isfinite(score) and score < best_score:
                best_seq = cand
                best_score = score
        for scale in scale_candidates:
            cand = best_seq.copy()
            cand[start:] *= scale
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = compute_score(cand.tolist())
            if np.isfinite(score) and score < best_score:
                best_seq = cand
                best_score = score

    # Now optional small random local search
    rng = np.random.default_rng(seed)
    for _ in range(2000):
        i = rng.integers(0, n)

        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    return [float(x) for x in best_seq]

import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Return the sequence r_i = C(2i,i)/4^i for i=0..n-1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def compute_score(seq: np.ndarray) -> float:
    """
    Fast computation of the AC1 upper bound:
        2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)
    Uses real FFT, works for any non‑negative sequence.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    total = seq.sum()
    if total < 0.01:
        return np.inf
    # next power of two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (total * total))


def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function that aims to minimize the AC1 bound."""
    # Determine length of the problem
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = int(parent.shape[0])
    else:
        n = 256

    # Base sequence: use the provided parent if sensible, otherwise the r‑sequence
    if isinstance(parent, np.ndarray) and parent.shape == (n,):
        base = np.clip(parent.astype(np.float64), 0.0, 1000.0)
    else:
        base = generate_r_sequence(n)
        base = np.clip(base, 0.0, 1000.0)

    best_seq = base.copy()
    best_score = compute_score(best_seq)

    # ----------------------------------------------------------------------
    # Grid search: add a constant or scale a prefix / suffix of the sequence.
    # ----------------------------------------------------------------------
    # Candidate lengths (roughly powers of n)
    L_candidates = sorted(
        set(
            int(v)
            for v in [
                n ** 0.25,
                n ** 0.33,
                n ** 0.5,
                n ** 0.66,
                n ** 0.75,
                n // 4,
                n // 3,
                n // 2,
            ]
        )
        if 0 < v < n
    )
    # Additive deltas (tiny positive bumps)
    delta_add = [0.001, 0.002, 0.003, 0.004, 0.005, 0.008, 0.010]
    # Multiplicative factors (slight scaling)
    delta_mul = [1.001, 1.002, 1.005, 1.010, 1.020, 1.050]

    # Helper to test a candidate and update the best known sequence.
    def test_candidate(cand: np.ndarray):
        nonlocal best_seq, best_score
        np.clip(cand, 0.0, 1000.0, out=cand)
        sc = compute_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_seq = cand.copy()
            best_score = sc

    # Prefix modifications
    for L in L_candidates:
        # additive
        for d in delta_add:
            cand = best_seq.copy()
            cand[:L] += d
            test_candidate(cand)
        # multiplicative
        for m in delta_mul:
            cand = best_seq.copy()
            cand[:L] *= m

    return [float(x) for x in best_seq]
