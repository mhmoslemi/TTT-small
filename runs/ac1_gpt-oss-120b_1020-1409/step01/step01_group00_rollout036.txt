def _pair_transfer(arr, rng):
    n = arr.size
    # choose donor index weighted by value (prob to pick larger a)
    probs = arr / arr.sum()
    i = rng.choice(n, p=probs)  # donor
    # choose recipient index inversely weighted (prefer smaller)
    inv = 1.0 / (arr + 1e-12)
    inv_probs = inv / inv.sum()
    j = rng.choice(n, p=inv_probs)
    if i == j or arr[i] <= 0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.05)
    cand = arr.copy()
    cand[i] -= delta
    cand[j] = min(1000.0, cand[j] + delta)
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

import numpy as np

# ----------------------------------------------------------------------
# Helper sequence generators
# ----------------------------------------------------------------------
def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..length-1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        f = float(i)
        a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
    return a


# ----------------------------------------------------------------------
# Simple perturbations (all O(n) memory)
# ----------------------------------------------------------------------
def _pair_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small random fraction from a high‑value index to a low‑value one."""
    n = arr.size
    i = rng.integers(0, n)       # donor
    j = rng.integers(0, n)       # receiver
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.05)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _block_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = arr.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _tail_shift(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a chunk from the front third to the last third of the vector."""
    n = arr.size
    i = rng.integers(0, n // 3)               # donor in first third
    j = rng.integers(2 * n // 3, n)           # receiver in last third
    if arr[i] <= 0.0:
        return arr.copy()
    delta = arr[i] * rng.uniform(0.001, 0.07)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand

            best_score = cand_score

        # simulated‑annealing acceptance for the current walk
        delta = cand_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            cur = cand
            cur_score = cand_score

    # ------------------------------------------------------------------
    # final sanity clamp – guarantees a non‑empty valid return
    # ------------------------------------------------------------------
    final = np.clip(best, 0.0, 1000.0)
    # make sure the sum is not accidentally too tiny
    if final.sum() < 0.01:
        final = _generate_binomial_sequence(final.size)

    return [float(v) for v in final]
