    # Ensure within bounds

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # main loop
    for it in range(300):
        # compute convolution via FFT
        L = 1 << ((2*n-1).bit_length())  # next power of two
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        M = conv.max()
        k0 = int(conv.argmax())
        S = a.sum()
        if S == 0:
            break
        # compute shifted a
        idx = k0 - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        a_shifted = np.empty_like(a)
        a_shifted[:] = 0.0
        a_shifted[valid] = a[idx[valid]]
        target = M / S
        # direction to reduce bound
        direction = target - a_shifted
        # step size: adapt
        # compute max feasible step to keep non-negative
        # for direction < 0 (decrease), max step = -a_i / direction_i
        neg = direction < 0
        if np.any(neg):
            max_step = np.min(-a[neg] / direction[neg])
        else:
            max_step = np.inf
        # Choose step fraction
        step = min(0.5, max_step * 0.5)  # ensure not too big
        # apply update
        cand = a + step * direction
        np.clip(cand, 0.0, 1000.0, out=cand)
        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            a = cand  # move to better point
        else:
            # try random perturbation fallback
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 0:
                delta = a[i] * rng.uniform(0.001, 0.02)
                cand2 = a.copy()
                cand2[i] = max(0.0, cand2[i] - delta)
                cand2[j] = min(1000.0, cand2[j] + delta)
                score2 = float(evaluate_sequence(cand2.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best = cand2.copy()
                    best_score = score2
                    a = cand2
        # early stop if B close to target
        if best_score < 1.503:  # near target
            break
    return [float(v) for v in best]

import numpy as np


def _generate_binomial_sequence(length: int) -> np.ndarray:
    """C(2i,i)/4**i for i=0..length‑1 (the discrete 1/√x analogue)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        f = float(i)
        a[i] = a[i - 1] * ((2.0 * f - 1.0) * (2.0 * f)) / (4.0 * f * f)
    return a

        i = rng.choice(n, p=prob_donor)

        prob_rec = 1.0 / (a + 1.0)          # +1 avoids extreme weights for zeros
        prob_rec /= prob_rec.sum()
        j = rng.choice(n, p=prob_rec)

        if i != j and a[i] > 0.0:
            frac = rng.uniform(0.001, 0.020)          # move 0.1‑2 % of donor mass
            delta = a[i] * frac
            cand = a.copy()
            cand[i] = max(0.0, cand[i] - delta)
            cand[j] = min(1000.0, cand[j] + delta)

            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand.copy()
                best_score = score
                a = cand
                if best_score <= 1.5028:
                    break

        # ----- occasional block scaling --------------------------------
        if step % 100 == 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.06)))   # modest log‑normal tweak
            cand2 = a.copy()
            cand2[start:start + width] *= factor
            np.clip(cand2, 0.0, 1000.0, out=cand2)

            sc2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best = cand2.copy()
                best_score = sc2
                a = cand2
                if best_score <= 1.5028:
                    break

    return [float(v) for v in best]
