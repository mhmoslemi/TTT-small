import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(arr: np.ndarray) -> float:
    """Maximum value of full convolution arr*arr using FFT (linear convolution)."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(arr: np.ndarray, n: int) -> float:
    """Exact AC1 upper bound score (identical to evaluate_sequence)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    maxb = _conv_max_fft(arr)
    return float(2.0 * n * maxb / (s * s))


def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Small stochastic local change, preserving L1 norm."""
    cand = vec.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    cand[start : start + width] *= factor
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand


def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = 0.5 * (base + base[::-1])
    return base


def _make_exponential(n: int, tau: float, sym: bool) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    base = np.exp(-idx / tau)
    if sym:
        base = 0.5 * (base + base[::-1])
    return base


def _make_combined(n: int, p: float, beta: float, tau: float, sym: bool) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    base = ((idx + beta) ** (-p)) * np.exp(-idx / tau)
    if sym:
        base = 0.5 * (base + base[::-1])
    return base

                        continue
                    a = base * (target_sum / s)
                    np.clip(a, 0.0, 1000.0, out=a)
                    sc = _score(a, n)
                    if sc < best_score:
                        best_score, best_vec = sc, a.copy()

    # --------------------------------------------------------------
    # Randomised search over the same families (broader parameter ranges)
    for _ in range(1800):
        family = rng.choice(["pow", "exp", "comb"])
        sym = bool(rng.integers(0, 2))

        if family == "pow":
            p = float(rng.uniform(0.1, 1.8))
            beta = 10.0 ** rng.uniform(-2.0, 2.0)
            base = _make_powerlaw(n, p, beta, sym)

        elif family == "exp":
            tau = float(rng.uniform(0.03 * n, 0.9 * n))
            base = _make_exponential(n, tau, sym)

        else:  # combined
            p = float(rng.uniform(0.1, 1.8))
            beta = 10.0 ** rng.uniform(-2.0, 2.0)
            tau = float(rng.uniform(0.03 * n, 0.9 * n))
            base = _make_combined(n, p, beta, tau, sym)

        s = base.sum()
        if s == 0.0:
            continue
        a = base * (target_sum / s)
        np.clip(a, 0.0, 1000.0, out=a)
        sc = _score(a, n)
        if sc < best_score:
            best_score, best_vec = sc, a.copy()

    # --------------------------------------------------------------
    # Local stochastic refinement around the current best
    for _ in range(500):
        cand = _block_perturb(best_vec, rng, target_sum)
        sc = _score(cand, n)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()

    # --------------------------------------------------------------
    # Final safety check using the official evaluator (guards against unexpected NaNs)
    # The evaluator will clamp to [0,1000] and recompute the exact score.
    # If it returns infinite, fall back to the uniform vector.
    try:
        from math import isfinite

        eval_score = evaluate_sequence(best_vec.tolist())
        if not isfinite(eval_score):
            best_vec = uniform.copy()
    except Exception:
        # In the unlikely event evaluate_sequence is unavailable, keep the current best
        pass

    # Return plain Python list of floats
    return [float(x) for x in best_vec]
