def _conv_self_fft(x):
    n = x.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len-1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    return conv

import numpy as np

# ------------------------------------------------------------
# Helper: classic decreasing series a_i = C(2i,i)/4^i
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the length‑`length` truncation of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Convolution via FFT (full linear convolution)
# ------------------------------------------------------------
def _conv_self_fft(x: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution of x with itself using length‑fft_len FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    conv_full = np.fft.irfft(fx * fx, n=fft_len)
    conv_len = 2 * x.shape[0] - 1
    return conv_full[:conv_len]


# ------------------------------------------------------------
# Exact AC1 upper bound (matches evaluate_sequence)
# ------------------------------------------------------------
def _score_fft(seq: np.ndarray, fft_len: int) -> float:
    """Compute 2*n*max(conv(seq,seq))/(sum(seq)**2) after clipping."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    conv = _conv_self_fft(seq, fft_len)
    m = conv.max()
    n = seq.shape[0]
    return 2.0 * n * m / (s * s)
