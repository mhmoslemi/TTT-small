import numpy as np

# --------------------------------------------------------------
def _conv_full(arr: np.ndarray) -> np.ndarray:
    """Linear convolution of a real array with itself via FFT."""
    n = arr.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # power‑of‑2 >= 2n‑1
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _conv_max(arr: np.ndarray) -> float:
    """Maximum value of the full self‑convolution."""
    return float(_conv_full(arr).max())

def _score(arr: np.ndarray, n: int) -> float:
    """Exact AC1 upper‑bound (identical to evaluate_sequence)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    return float(2.0 * n * _conv_max(arr) / (s * s))

        alpha = step_factor * max_alpha

        new_vec = best_vec + alpha * direction
        np.clip(new_vec, 0.0, 1000.0, out=new_vec)

        # renormalise to the prescribed L1‑norm
        s = new_vec.sum()
        if s > 0.0:
            new_vec *= target_sum / s

        new_score = _score(new_vec, n)
        if new_score < best_score:
            best_score = new_score
            best_vec = new_vec
            stale = 0
        else:
            stale += 1

        # occasional small random shake to escape plateaus
        if (it + 1) % 100 == 0:
            cand = best_vec * np.exp(rng.normal(0.0, 0.02, size=n))
            np.clip(cand, 0.0, 1000.0, out=cand)
            s = cand.sum()
            if s > 0.0:
                cand *= target_sum / s
            cand_score = _score(cand, n)
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand
                stale = 0

        if stale >= 250:          # no improvement for a while → stop
            break

    # ----------------------------------------------------------
    # Final safety: ensure the output meets the evaluator's expectations.
    # (The evaluator will clamp, renormalise and recompute the score.)
    return [float(x) for x in best_vec]
