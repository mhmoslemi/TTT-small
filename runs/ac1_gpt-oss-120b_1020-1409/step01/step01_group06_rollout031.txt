import numpy as np

# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (real input)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * a.shape[0] - 1]


def _conv_max(a: np.ndarray, size: int) -> float:
    """Maximum value of a*a."""
    return float(_conv_full(a, size).max())


def _normalize(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to [0,1000] and rescale to a prescribed L1‑norm."""
    np.clip(v, 0.0, 1000.0, out=v)
    s = v.sum()
    if s > 0.0:
        v *= target_sum / s
    else:
        v[:] = target_sum / v.size
    return v

    """Exact score used by the official evaluator (clamping already applied)."""
    s = v.sum()
    if s == 0.0:
        return float("inf")
    maxb = _conv_max(v, size)
    return float(2.0 * n * maxb / (s * s))

                    best_vec = cand.copy()

    # 2) Exponential decay / growth
    alphas = np.logspace(-3, -0.1, 12)
    for rev in (False, True):
        for a in alphas:
            base = _make_exponential(n, a, rev)
            cand = _normalize(base.copy(), target_sum)
            sc = _score(cand, size, n)
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()

    # 3) Gaussian bell
    sigmas = np.linspace(n / 20.0, n / 2.0, 10)
    for sig in sigmas:
        base = _make_gaussian(n, sig)
        cand = _normalize(base.copy(), target_sum)
        sc = _score(cand, size, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # 4) Powered sine window (smooth, vanishing ends)
    p_sine = np.linspace(0.5, 6.0, 12)
    for p in p_sine:
        base = _make_sine_pow(n, p)
        cand = _normalize(base.copy(), target_sum)
        sc = _score(cand, size, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # --------------------------------------------------------------
    # Gradient‑descent style refinement targeting the current worst convolution entry
    max_iter = 800
    for it in range(max_iter):
        conv = _conv_full(best_vec, size)
        k = int(conv.argmax())                          # index of maximal value
        idx = k - np.arange(n)                          # shifted indices
        mask = (idx >= 0) & (idx < n)
        shift = np.zeros_like(best_vec)
        shift[mask] = best_vec[idx[mask]]
        grad = best_vec + shift                         # ∇ conv_k = a + shifted(a)
        eta = 0.02 * (1.0 - it / max_iter)              # decreasing step size
        cand = best_vec - eta * grad
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = cand.sum()
        if s > 0.0:
            cand *= target_sum / s
        sc = _score(cand, size, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # --------------------------------------------------------------
    # Short stochastic refinement (a few block perturbations)
    for _ in range(200):
        cand = _block_perturb(best_vec, rng, target_sum)
        sc = _score(cand, size, n)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    # --------------------------------------------------------------
    # Return a plain Python list of finite floats
    return [float(x) for x in best_vec]
