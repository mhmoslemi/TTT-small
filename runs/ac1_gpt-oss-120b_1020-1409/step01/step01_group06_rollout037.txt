import numpy as np

# --------------------------------------------------------------
# Fast linear convolution (full) using FFT – returns the maximum value.
def _conv_max_fft(a: np.ndarray) -> float:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# --------------------------------------------------------------
# Exact score according to the official evaluator (no side‑effects).
def _score(a: np.ndarray, n: int) -> float:
    s = a.sum()
    if s == 0.0:
        return float("inf")
    return float(2.0 * n * _conv_max_fft(a) / (s * s))


# --------------------------------------------------------------
# Generate a smooth random, non‑negative candidate.
#   - low‑pass spectral envelope with random decay rate
#   - random phase spectrum
#   - absolute value to enforce non‑negativity
def _random_smooth(rng: np.random.Generator, n: int, target_sum: float) -> np.ndarray:
    # random decay for low‑pass filter
    decay = rng.uniform(0.01, 0.20)     # larger -> more low‑frequency content
    freqs = np.fft.rfftfreq(n, d=1.0)
    amp = np.exp(-decay * freqs)        # monotone decreasing envelope
    phase = rng.uniform(0.0, 2.0 * np.pi, size=amp.shape)
    spec = amp * (np.cos(phase) + 1j * np.sin(phase))
    cand = np.fft.irfft(spec, n=n)
    cand = np.abs(cand)                 # enforce non‑negativity
    # short random scaling to diversify magnitudes
    cand *= rng.uniform(0.5, 2.0)
    # final clipping before normalisation
    np.clip(cand, 0.0, 1000.0, out=cand)
    # normalise L1‑norm to the chosen target sum
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand

# Small local perturbation that respects the L1‑norm.
def _local_perturb(rng: np.random.Generator, vec: np.ndarray, target_sum: float) -> np.ndarray:
    cand = vec.copy()
    n = cand.size
    # select a contiguous block
    w = rng.integers(max(1, n // 250), max(2, n // 15))
    i = rng.integers(0, n - w + 1)
    factor = np.exp(rng.normal(0.0, 0.08))
    cand[i:i + w] *= factor
    # optional gentle smoothing by averaging with a shifted copy
    if rng.random() < 0.25:
        shift = rng.integers(1, n // 10)
        cand = 0.5 * (cand + np.roll(cand, shift))
    np.clip(cand, 0.0, 1000.0, out=cand)
    # renormalise to the target L1‑norm
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand
