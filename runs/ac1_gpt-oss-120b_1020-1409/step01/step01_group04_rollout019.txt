def evaluate_sequence(sequence: list) -> float:
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:

            return np.inf
        if np.isnan(x) or np.isinf(x):

import numpy as np

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # initial best = parent clipped
    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # Function to test candidate and update best
    def maybe_update(cand):
        nonlocal best, best_score
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # 1. Scan families
    alphas = np.logspace(-5, 2, 20)  # adjust ranges
    p_values = np.linspace(0.2, 1.5, 14)  # 0.2-1.5
    for p in p_values:
        for alpha in alphas:
            seq = 1.0 / np.power(np.arange(n, dtype=np.float64) + alpha, p)
            maybe_update(seq)

    # exponential family
    betas = np.linspace(0.2, 2.5, 20)  # beta parameter
    for beta in betas:
        r = np.exp(-beta / n)
        seq = r ** np.arange(n, dtype=np.float64)
        maybe_update(seq)

    # 2. hill climbing & annealing
    steps = 5000
    sigma0 = 0.15
    min_width = max(1, n // 500)
    max_width = max(2, n // 10)
    for it in range(steps):
        cand = best.copy()
        # decide update type
        if rng.random() < 0.8:
            width = int(rng.integers(min_width, max_width+1))
            start = int(rng.integers(0, n-width+1))
            # adapt sigma
            sigma = sigma0 * (1 - it/steps)
            factor = np.exp(rng.normal(0.0, sigma))
            cand[start:start+width] *= factor
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i: j += 1
            delta = best[i] * rng.random() * 0.2
            cand[i] -= delta
            cand[j] += delta
        # occasional symmetry enforcement
        if rng.random() < 0.05:
            cand = 0.5 * (cand + cand[::-1])
        maybe_update(cand)

    # final fallback check: ensure best non-empty and finite.
    if best_score is None or not np.isfinite(best_score) or best.size == 0:
        best = np.clip(parent, 0.0, 1000.0)

    # return as plain list of floats
    return [float(x) for x in best]
