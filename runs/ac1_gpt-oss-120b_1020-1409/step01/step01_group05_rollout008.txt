def propose_candidate(seed=42, num_iters=15000, **kwargs):

    # Determine n

        n = 2000
        parent_ok = False

    # fallback init: decreasing power law: a_i = (i+1)^-p for p=0.5 maybe.

        a_sum = a.sum()
        if a_sum>0:

        else:

        s = a.sum()
        if s>0:

        else:
            a = np.ones(n)/n
        best = a.copy()
        best_score = float(evaluate_sequence(best.tolist()))
    else:
        a = init_powerlaw(p=0.6)
        best = a.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    # Precompute sum target = 1.0
    sum_target = 1.0

    # iteration hyperparameters
    step0 = 0.6
    decay = 0.998  # slower decay
    perturb_interval = 100
    perturb_prob = 0.3
    # Additional smoothing frequency maybe 500

    for it in range(num_iters):
        # compute convolution using FFT for efficiency
        # use length next_pow2 >= 2*n-1
        # compute conv via real FFT
        N = 1 << ((2*n-1).bit_length())
        # FFT of a zero-padded length N
        A = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(A * A, n=N)[:2*n-1]
        # find index of max
        k = int(np.argmax(conv))

        # Compute gradient wrt a: g_i = 2*a_{k-i} if k-i in range else 0
        low = max(0, k-(n-1))
        high = min(k, n-1)
        idx_range = np.arange(low, high+1, dtype=np.int64)
        partner = k - idx_range
        g = np.zeros_like(a)
        # partner indices are within [0,n-1]
        g[idx_range] = 2.0 * a[partner]

        # multiplicative update
        step = step0 * (decay**it)
        factor = np.exp(-step * g)
        a = a * factor

        # renormalise to keep sum 1, enforce bounds
        a_sum = a.sum()

def compute_score(arr):
    s = arr.sum()
    if s < 0.01:
        return np.inf
    N = 1 << ((2*len(arr)-1).bit_length())
    A = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(A*A, n=N)[:2*len(arr)-1]
    max_b = conv.max()
    return 2.0 * len(arr) * max_b / (s*s)

import numpy as np

# ----------------------------------------------
# Helper: compute the AC1 upper bound for a vector
# ----------------------------------------------
def _ac1_score(vec, n, N):
    """Return 2*n*max(conv)/sum(vec)^2 using FFT (no clipping)."""
    s = vec.sum()
    if s < 0.01 or not np.isfinite(s):
        return np.inf
    # FFT convolution (real input)
    A = np.fft.rfft(vec, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (s * s)

def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the AC1 bound.
    Returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------

    else:

        n = 2000
        parent_ok = False

    # Initial candidate

    def _init_powerlaw(p=0.6):

        s = a.sum()
        if s > 0 and np.isfinite(s):
            a /= s
        else:

    if parent_ok:

    # --------------------------------------------------------------
