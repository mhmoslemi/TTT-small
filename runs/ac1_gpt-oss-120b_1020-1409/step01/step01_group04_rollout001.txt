def lag_update(seq, rng, reduction=0.2):

    L = 2*n-1
    m = 1 << (L - 1).bit_length()

    conv = np.fft.ifft(fft_seq*fft_seq).real[:L]

    i = np.arange(n)
    j = idx - i
    mask = (j >= 0) & (j < n)
    if not np.any(mask):
        return seq  # nothing to do
    seq2 = seq.copy()
    # Reduce mask entries
    reduction_factor = 1 - reduction
    seq2[mask] *= reduction_factor
    # Compute reduction sum
    total_reduction = np.sum(seq[mask] - seq2[mask])
    # Add equally to others
    other_idx = np.where(~mask)[0]
    if other_idx.size > 0:
        add = total_reduction / other_idx.size
        seq2[other_idx] += add
    # clip again outside
    seq2 = np.clip(seq2, 0.0, 1000.0)
    return seq2

import numpy as np

# ----------------------------------------------------------------------
# Helper: fast computation of the AC1 upper bound (same formula as the evaluator)
# ----------------------------------------------------------------------
def compute_score(seq):
    """Return the AC1 bound for a non‑negative sequence."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        return float('inf')
    n = seq.size
    L = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << ((L - 1).bit_length())
    fft_seq = np.fft.fft(seq, n=m)
    conv = np.fft.ifft(fft_seq * fft_seq).real[:L]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Lag‑reduction update (uses the convolution to identify the worst lag)
# ----------------------------------------------------------------------
def lag_update(seq, rng, reduction=0.2):
    """Reduce contributions on the lag where the autocorrelation is maximal."""
    n = seq.size
    L = 2 * n - 1
    m = 1 << ((L - 1).bit_length())
    fft_seq = np.fft.fft(seq, n=m)
    conv = np.fft.ifft(fft_seq * fft_seq).real[:L]
    idx = int(conv.argmax())

    i = np.arange(n)
    j = idx - i
    mask = (j >= 0) & (j < n)

    if not mask.any():
        return seq  # nothing to modify

    new_seq = seq.copy()
    # shrink the “hot” entries
    factor = 1.0 - reduction
    new_seq[mask] *= factor
    loss = float(np.sum(seq[mask] - new_seq[mask]))

    # distribute the lost mass over the remaining entries
    other = np.where(~mask)[0]
    if other.size > 0 and loss > 0.0:
        add = loss / other.size
        new_seq[other] += add

    return np.clip(new_seq, 0.0, 1000.0)
