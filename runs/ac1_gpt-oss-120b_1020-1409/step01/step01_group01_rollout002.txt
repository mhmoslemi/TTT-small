import numpy as np

def generate_r_sequence(n: int) -> np.ndarray:
    """Generate the classical r‑sequence r_i = C(2i,i)/4^i for i=0..n‑1."""
    seq = np.empty(n, dtype=np.float64)
    r = 1.0
    seq[0] = r
    for i in range(1, n):
        r = r * (2 * i - 1) / (2 * i)
        seq[i] = r
    return seq


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1
    # rfft works with real input and is slightly faster than fft
    fa = np.fft.rfft(a, n=size)
    c = np.fft.irfft(fa * fa, n=size)
    return c[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact replica of the evaluator’s objective (after clipping)."""
    np.clip(a, 0.0, 1000.0, out=a)
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)


def improve_sequence(a: np.ndarray, max_iters: int = 500,
                     target: float = 1.50274349) -> (np.ndarray, float):
    """Iteratively move mass from high‑influence to low‑influence coordinates."""
    best = a.copy()
    best_score = compute_score(best)
    step_factor = 0.5          # fraction of source entry moved each trial
    eps = 1e-12

    for _ in range(max_iters):
        # current convolution and its maximal index
        conv = full_convolution_fft(a)
        max_b = conv.max()
        k_star = int(np.argmax(conv))

        n = a.shape[0]

        # influence vector v_i = a_{k*‑i} when the index exists, else 0
        v = np.zeros_like(a)
        i_low = max(0, k_star - (n - 1))
        i_high = min(n - 1, k_star)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        v[idx] = a[k_star - idx]

        # source index: large influence and positive mass
        mask_positive = a > eps
        if not np.any(mask_positive):
            break
        v_src = np.where(mask_positive, v, -np.inf)
        i_max = int(np.argmax(v_src))

        # destination index: smallest influence (could be zero)
        i_min = int(np.argmin(v))

        # No beneficial move possible
        if v[i_max] <= v[i_min] + eps:
            break

        # Amount to transfer
        delta = step_factor * a[i_max]
        if delta < eps:
            break
        # Respect bounds
        delta = min(delta, a[i_max])                     # stay non‑negative
        delta = min(delta, 1000.0 - a[i_min])            # stay ≤ 1000
        if delta <= eps:
            break

        # Apply tentative transfer
        a[i_max] -= delta
        a[i_min] += delta
        np.clip(a, 0.0, 1000.0, out=a)

        new_score = compute_score(a)

        if new_score < best_score - 1e-12:
            # Accept improvement
            best_score = new_score
            best = a.copy()
            step_factor = min(step_factor * 1.1, 0.9)    # try larger moves
        else:
            # Reject and revert
            a[i_max] += delta
            a[i_min] -= delta
            np.clip(a, 0.0, 1000.0, out=a)
            step_factor *= 0.5                          # be more conservative

        if best_score <= target:
            break

    return best, best_score
