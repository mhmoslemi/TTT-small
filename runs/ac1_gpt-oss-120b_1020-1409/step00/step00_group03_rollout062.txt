def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)

    n = height_sequence_1.shape[0] (provided).
    best = None; best_score = np.inf.

    # ensure fallback initial sequence: uniform of ones.
    fallback = np.ones(n, dtype=np.float64)

    # evaluate fallback
    best = fallback
    best_score = float(evaluate_sequence(best.tolist())) # should be 2

    # search geometric r values
    # generate r values from 0.05 to 0.9999 (maybe 120 values evenly spaced on linear or log).
    r_vals = np.linspace(0.05, 0.9999, 200) # 200 points

    for r in r_vals:
        candidate = np.power(r, np.arange(n, dtype=np.float64))
        # Optionally maybe multiply to use max 1000 final? Not needed.
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = candidate.copy()

    # search reversed geometric (descending from 1 at end)
    for r in r_vals:
        candidate = np.power(r, np.arange(n-1, -1, -1, dtype=np.float64))
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = candidate.copy()

    # search simple power law: a_i = (i+1)**(-p) ; p from 0.1 to 3
    p_vals = np.linspace(0.1, 3.0, 100)
    indices = np.arange(1, n+1, dtype=np.float64)
    for p in p_vals:
        candidate = indices ** (-p)
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = candidate.copy()

    # optional random perturbations around best:
    # do few random local modifications.
    for _ in range(200):
        # simple perturbation: pick random slice and multiply by factor
        cand = best.copy()
        start = rng.integers(n)
        length = rng.integers(1, max(2, n//20))
        end = min(n, start+length)
        factor = np.exp(rng.normal(0, 0.1))
        cand[start:end] *= factor
        # clip
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    return [float(x) for x in best]

import numpy as np

# The evaluator is pre‑imported in the harness as `evaluate_sequence`.


def _score(seq):
    """Return the AC1 upper‑bound score for a candidate sequence."""
    # Convert to plain Python list as required by the evaluator.
    return float(evaluate_sequence(seq.tolist()))


def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step‑function height list with a low score.

    The routine builds a variety of parametric families (geometric,
    power‑law, sinusoidal) and keeps the best candidate seen.  After the
    systematic sweep a short random refinement stage is performed.
    A simple uniform fallback guarantees a valid result.
    """
    rng = np.random.default_rng(seed)

    # Length of the provided parent construction.
    n = int(height_sequence_1.shape[0])

    # ----- fallback: uniform vector (guarantees a finite score) -----
    best = np.ones(n, dtype=np.float64)
    best_score = _score(best)

    # ----- geometric families  a_i = r**i  (r in (0,1)) -----
    # Use a finer grid near 1 where improvement is possible.
    r_vals = np.concatenate((
        np.linspace(0.05, 0.90, 30, endpoint=False),
        np.linspace(0.90, 0.9999, 120, endpoint=False)
    ))
    idx = np.arange(n, dtype=np.float64)

    for r in r_vals:
        cand = np.power(r, idx)                     # decreasing geometric
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

        cand_rev = np.power(r, idx[::-1])           # increasing geometric
        sc = _score(cand_rev)
        if sc < best_score:
            best, best_score = cand_rev, sc

    # ----- power‑law families  a_i = (i+1) ** (-p)  -----
    p_vals = np.linspace(0.1, 3.0, 80)
    i1 = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        cand = i1 ** (-p)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

        cand_rev = i1 ** (-p)[::-1]
        sc = _score(cand_rev)
        if sc < best_score:
            best, best_score = cand_rev, sc

    # ----- sinusoidal families  a_i = sin(π * i / (n-1)) + offset -----
    # The offset guarantees non‑negativity.
    theta = np.pi * idx / (n - 1)
    sin_vals = np.sin(theta)
    for offset in (0.0, 0.1, 0.5):
        cand = sin_vals + offset
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # ----- short random refinement around the current best -----
    for _ in range(300):
        cand = best.copy()

        # Pick a random block and perturb its amplitude multiplicatively.
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 20))
        stop = min(n, start + width)

        factor = np.exp(rng.normal(0.0, 0.08))
        cand[start:stop] *= factor

        # Occasionally smooth the whole vector a little.
        if rng.random() < 0.25:
            # a simple moving‑average filter with window size 3
            cand[1:-1] = (cand[:-2] + cand[1:-1] + cand[2:]) / 3.0

        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # Return as a plain Python list of finite floats.
    return [float(x) for x in best]
