def propose_candidate(seed=42, **kwargs):
    """
    Construct a simple two‑block step function.
    The sequence consists of a block of ones at the left,
    a block of ones at the right, and zeros in between.
    This shape yields a convolution max equal to the block length,
    giving a score ≈ n/(2*block_len) which is well below the target bound.
    """

    # Guard against degenerate sizes.
    if n < 3:
        # Very small case – fall back to a uniform positive vector.
        cand = np.ones(n, dtype=np.float64)
        return cand.tolist()

    # Choose a small gap of zeros between the two blocks.
    # For odd lengths a gap of 1 works, for even use a gap of 2.
    gap = 1 if (n % 2) else 2
    block_len = (n - gap) // 2
    if block_len < 1:
        # If the gap would consume the whole vector, revert to uniform.
        cand = np.ones(n, dtype=np.float64)
        return cand.tolist()

    # Build the candidate sequence.
    cand = np.zeros(n, dtype=np.float64)
    cand[:block_len] = 1.0                # left block
    cand[-block_len:] = 1.0               # right block

    # Ensure compliance with the evaluator's clipping bounds.
    np.clip(cand, 0.0, 1000.0, out=cand)

    # Return as a plain Python list of floats.
    return cand.tolist()
