        if M == 0 or not np.isfinite(M):
            break
        if not use_lp:
            break
        # Build LP
        n_var = n
        x = cp.Variable(n_var, nonneg=True)
        f_param = cp.Parameter(n_var, nonneg=True)
        f_param.value = best
        # convolution: f * x
        conv_fx = cp.conv(f_param, x)  # length 2n-1
        constraints = [conv_fx <= M, x <= 1000.0]
        prob = cp.Problem(cp.Maximize(cp.sum(x)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-6, reltol=1e-6)
        except Exception:
            # LP failed; break
            break
        if x.value is None:
            break
        g = np.asarray(x.value, dtype=np.float64)
        # Clip and scale g to target sum
        g = np.clip(g, 0.0, 1000.0)
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g * (target_sum / sum_g)
        # Mix
        candidate = (1.0 - t) * best + t * g
        candidate = np.clip(candidate, 0.0, 1000.0)
        cand_score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = candidate
            best_score = cand_score
        # Additional local perturbation step as safety
        candidate2 = make_block_perturbation(best, rng)
        cand_score2 = float(evaluate_sequence(candidate2.tolist()))
        if np.isfinite(cand_score2) and cand_score2 < best_score:
            best = candidate2
            best_score = cand_score2
    # If LP not used or not enough improvement, perform simple sampling
    if best_score == float('inf') or best_score > 2.5:
        # fallback random block search
        for _ in range(2000):
            cand = make_block_perturbation(best, rng)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc
    return [float(x) for x in best]

import numpy as np

# Helper: simple block perturbation (fallback & occasional local move)
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # choose a block width (log‑uniform between n/200 and n/12)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # random multiplicative perturbation of the whole block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
