import numpy as np

# --------------------------------------------------------------
# Helper: a cheap O(n) random block perturbation of a non‑negative sequence.
# --------------------------------------------------------------
def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a copy of ``seq`` where a random contiguous block is scaled."""
    n = seq.size
    # choose a block size proportional to the length, at least 1 element
    block_len = int(max(1, n // int(rng.integers(50, 150))))
    start = int(rng.integers(0, max(1, n - block_len + 1)))
    stop = start + block_len
    factor = float(np.exp(rng.normal(0.0, 0.05)))   # modest multiplicative noise
    new = seq.copy()
    new[start:stop] *= factor
    # occasional symmetric smoothing (helps keep structure)
    if rng.random() < 0.3:
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)
