    n = a.size

    while size < 2*n-1:

    S = []

import numpy as np

# ------------------------------------------------------------
# Helper: fast max-convolution using FFT (matrix‑free)
# ------------------------------------------------------------
def _max_convolution(x: np.ndarray) -> float:
    """Return max_{k} (x*x)[k] using FFT, zero‑padded convolution."""
    n = x.size
    size = 1
    while size < 2 * n - 1:               # next power of two
        size <<= 1
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    return float(np.max(conv))


# ------------------------------------------------------------
# Greedy construction of a Sidon set (all sums i+j distinct)
# ------------------------------------------------------------
def _greedy_sidon_set(length: int, rng: np.random.Generator, trials: int = 8):
    """Return a list of indices S ⊂ [0,length) with pairwise distinct sums."""
    best = []
    # try several random orders – the best one is kept
    for _ in range(trials):
        order = rng.permutation(length)
        S = []
        sums = set()
        for i in order:
            # self‑sum must be fresh
            if i + i in sums:
                continue
            # check sums with already chosen indices
            conflict = False
            for j in S:
                if i + j in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # accept i
            for j in S:
                sums.add(i + j)
            sums.add(i + i)
            S.append(i)
        if len(S) > len(best):
            best = S
            # theoretical ceiling ≈ sqrt(2*length); stop early if reached
            if len(best) >= int(np.sqrt(2 * length)) + 2:
                break
    return best
