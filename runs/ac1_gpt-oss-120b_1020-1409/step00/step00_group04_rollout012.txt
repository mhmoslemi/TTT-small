import numpy as np

def _make_block_perturbation(base, rng):
    """Return a copy of ``base`` with a single random block perturbed."""
    cand = base.copy()
    n = cand.size

    # block width between 0.1 % and 5 % of the length (at least one element)
    lo = max(1, int(n * 0.001))
    hi = max(2, int(n * 0.05))
    width = int(rng.integers(lo, hi + 1))

    start = int(rng.integers(0, n - width + 1))
    stop = start + width

    # multiplicative perturbation (log‑normal factor)
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:stop] *= factor

    # occasional mirroring to keep the shape roughly symmetric
    if rng.random() < 0.33:
        cand = 0.5 * (cand + cand[::-1])

    # enforce the hard limits required by the evaluator
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Return a list of non‑negative step‑function heights.
    The routine starts from the supplied parent ``height_sequence_1`` (if
    available) and performs a cheap stochastic hill‑climb that never uses
    more than O(n) auxiliary memory.  The exact evaluator ``evaluate_sequence``
    (pre‑imported in the harness) is called for every proposal.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Obtain a sensible starting vector.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is provided at runtime in the harness.
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
    except Exception:
        # fallback: a modestly sized uniform vector
        base = np.full(256, 1.0, dtype=np.float64)

    # make sure the vector respects the evaluator's clipping rules
    np.clip(base, 0.0, 1000.0, out=base)

    # ensure a non‑trivial sum (the evaluator rejects sums < 0.01)
    if base.sum() < 0.01:
        base[:] = 0.01 / base.size

    # ------------------------------------------------------------------
    # 2️⃣  Keep the best solution seen so far.
    # ------------------------------------------------------------------
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 3️⃣  Simple stochastic local search (memory‑bounded, exact evaluation).
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = _make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))

        # accept only strictly improving, finite scores
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # 4️⃣  Return a plain Python list as required by the contract.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
