def propose_candidate(seed=42, num_trials=20, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 5000
    best = None
    best_score = np.inf
    for _ in range(num_trials):
        # generate binary vector
        cand = rng.integers(0,2,size=n, dtype=np.float64)
        # ensure sum >=0.01; if too small (maybe all zeros) re-generate 
        if cand.sum() < 0.01:
            cand[0]=1.0  # ensure at least one.
        # evaluate
        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = float(score)
    if best is None:
        # fallback: uniform positive small vector
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
    return best.tolist()
