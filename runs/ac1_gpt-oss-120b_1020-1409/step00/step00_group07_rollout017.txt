def propose_candidate(seed=42, num_iters=30, t=0.02, K=10):
    rng = np.random.default_rng(seed)
    f = np.asarray(height_sequence_1, dtype=np.float64).copy()
    f = np.clip(f, 0.0, 1000.0)
    if f.ndim != 1 or f.size == 0 or not np.isfinite(f).all():
        f = np.ones(256, dtype=np.float64)
    # normalize sum to 1 for stability
    total = np.sum(f)
    if total == 0:
        f = np.ones_like(f)
        total = np.sum(f)
    f = f / total

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    for it in range(num_iters):
        c = np.convolve(f, f)
        M = float(np.max(c))
        # find top K indices (with highest c)
        # Actually we can take indices with near max
        # Use argpartition to find K largest
        top_k_idx = np.argpartition(c, -K)[-K:]
        top_k_idx = np.unique(top_k_idx)  # ensure size <=K
        # Build matrix A of shape (len(top_k_idx), n)
        rows = []
        for k in top_k_idx:
            # For j from 0..n-1, coefficient for g_j is f_{k-j} if in range
            # We can construct vector using np.zeros and fill slice.
            row = np.zeros_like(f)
            # compute j range where k-j in [0,n-1]: j ∈ [k-(n-1), k]
            start = max(0, k - (len(f) - 1))
            end = min(len(f) - 1, k)
            # for j in [start, end], coefficient = f[k - j]
            # So row[start:end+1] = f[k - np.arange(start, end+1)]
            idxs = np.arange(start, end + 1)
            row[idxs] = f[k - idxs]
            rows.append(row)
        A = np.vstack(rows) if rows else np.zeros((0, f.size))
        # Solve LP: maximize sum(g) subject to A @ g <= M, g>=0
        g_var = cp.Variable(f.size)
        constraints = [g_var >= 0]
        if rows:
            constraints.append(A @ g_var <= M)
        # objective: maximize sum(g)
        obj = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500, abstol=1e-8, reltol=1e-8)
        except cp.error.SolverError:
            # fallback: break
            break
        if g_var.value is None:
            break
        g0 = np.array(g_var.value).flatten()
        # Renormalize to have sum 1
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0
        # convex combination
        f = (1 - t) * f + t * g
        # Ensure nonnegative and clip to bounds
        f = np.clip(f, 0.0, 1000.0)
        # renormalize sum to 1
        f = f / f.sum()
        # evaluate
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f.copy()
            best_score = score
    # return best as list of floats
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

import numpy as np

def _build_constraint_matrix(f, idxs):
    """
    Build a matrix A where each row corresponds to a convolution
    constraint (f * g)[k] <= M for k in idxs.
    Row k has entries f[k - j] for those j where 0 <= k - j < len(f).
    """
    n = f.size
    rows = []
    for k in idxs:
        row = np.zeros_like(f)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        j = np.arange(lo, hi + 1, dtype=int)
        row[j] = f[k - j]
        rows.append(row)
    if rows:
        return np.vstack(rows)
    return np.empty((0, n), dtype=f.dtype)


def propose_candidate(seed=42, num_iters=12, step=0.12,
                     active_frac=0.99, max_active=12, **kwargs):
    """
    Search for a non‑negative step‑function that lowers the AC1 upper bound.
    Uses a simple LP to enlarge the total mass while keeping the current
    convolution peak unchanged, then blends the solution back into the
    current candidate.

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

        np.asarray(globals().get('height_sequence_1', np.ones(256)),

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    f = parent.copy()
    f = np.clip(f, 0.0, None)                     # keep non‑negative
    s = f.sum()
    if s == 0.0:
        f = np.ones(n, dtype=np.float64) / n
    else:
        f = f / s                                 # normalise to L1 = 1

    # small random noise to break perfect symmetry
    f += rng.normal(scale=0.01, size=n)
    f = np.clip(f, 0.0, None)
    f = f / f.sum()

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
