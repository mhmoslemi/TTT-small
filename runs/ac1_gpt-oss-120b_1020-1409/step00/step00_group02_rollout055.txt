    # We'll attempt to build up to target elements.
    # Use simple greedy with random shuffle of possible indices.
    indices = []
    sums = set()
    attempts = 0
    max_attempts = target * 10 * n // target  # maybe something else.
    while len(indices) < target and attempts < max_attempts:
        i = int(rng.integers(0, n))
        # ensure not duplicate index
        if i in indices:
            attempts += 1
            continue
        # compute sums with existing indices plus self sum
        ok = True
        # self sum 
        s_self = i + i
        if s_self in sums:
            ok = False
        else:
            for j in indices:
                s = i + j
                if s in sums:
                    ok = False
                    break
        if ok:
            indices.append(i)
            sums.add(s_self)
            for j in indices[:-1]:
                sums.add(i + j)
        attempts += 1
    return indices

def generate_low_collision_set(n, target_m, L, rng):
    indices = []
    sums = {}
    max_attempts = target_m * 1000
    attempts = 0
    while len(indices) < target_m and attempts < max_attempts:
        i = int(rng.integers(0, n))
        if i in indices:
            attempts += 1
            continue
        # compute new sums counts
        ok = True
        # self sum
        s_self = i + i
        cnt = sums.get(s_self, 0) + 1
        if cnt > L:
            ok = False
        # for each existing j
        if ok:
            for j in indices:
                s = i + j
                cnt = sums.get(s, 0) + 1
                if cnt > L:
                    ok = False
                    break
        if not ok:
            attempts += 1
            continue
        # Accept
        indices.append(i)
        # update sums
        sums[s_self] = sums.get(s_self, 0) + 1
        for j in indices[:-1]:
            s = i + j
            sums[s] = sums.get(s, 0) + 1
        attempts = 0  # maybe reset attempts to give chance
    return indices

import numpy as np

# ----------------------------------------------------------------------
# Helper: build a set of positions whose pairwise sums collide at most L times
# ----------------------------------------------------------------------
def build_low_collision_set(n: int, L: int, rng: np.random.Generator,
                            max_entries: int = 300) -> list:
    """Return a list of indices (0 <= i < n) such that each unordered pair
    (i,j) (including i=j) contributes to a sum i+j no more than L times.
    The algorithm repeatedly scans a random permutation of the index range
    and greedily accepts admissible positions until a full pass adds nothing.
    """
    indices = []                # accepted positions in order of addition
    index_set = set()           # fast membership test
    sum_counts = {}             # unordered‑pair count per sum (i+j)

    while True:
        added_this_pass = False
        perm = rng.permutation(n)          # random order each pass
        for i in perm:
            if i in index_set:
                continue

            # self‑sum check
            s_self = i + i
            if sum_counts.get(s_self, 0) + 1 > L:
                continue

            # check all sums with already‑chosen indices
            ok = True
            for j in indices:
                s = i + j
                if sum_counts.get(s, 0) + 1 > L:
                    ok = False
                    break
            if not ok:
                continue

            # ---------- accept i ----------
            indices.append(i)
            index_set.add(i)
            # update self sum
            sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
            # update sums with previous indices
            for j in indices[:-1]:
                s = i + j
                sum_counts[s] = sum_counts.get(s, 0) + 1
            added_this_pass = True

            if len(indices) >= max_entries:
                return indices
        if not added_this_pass:
            break
    return indices

    """

    # ------------------------------------------------------------------
