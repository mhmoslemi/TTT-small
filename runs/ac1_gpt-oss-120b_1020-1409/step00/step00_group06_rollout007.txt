def max_convolution(arr):
    n = arr.shape[0]
    # compute next power of two at least 2n-1
    m = 1 << ((2*n - 1).bit_length())
    # using rfft for real FFT
    fa = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    # conv length m
    # only first 2*n-1 relevant
    return np.max(conv[:2*n-1])

import numpy as np

# ------------------------------------------------------------
# Helper routines – all top‑level, no closures or lambdas.
# ------------------------------------------------------------

def max_convolution(arr: np.ndarray, m: int) -> float:
    """
    Compute max of the full convolution arr * arr using FFT.
    `m` must be a power‑of‑two >= 2*len(arr)-1.
    """
    # real FFT, point‑wise multiplication, inverse FFT
    fa = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    n = arr.shape[0]
    # first 2*n-1 entries contain the true convolution
    return float(np.max(conv[: 2 * n - 1]))


def objective(arr: np.ndarray, m: int) -> float:
    """
    Return the AC1 upper‑bound (the score the evaluator uses)
    for a given non‑negative sequence `arr`.
    """
    s = float(np.sum(arr))
    if s == 0.0:
        return np.inf
    maxc = max_convolution(arr, m)
    n = arr.shape[0]
    return 2.0 * n * maxc / (s * s)


def run_search(init_seq: np.ndarray, steps: int, rng: np.random.Generator) -> tuple:
    """
    Simple stochastic descent that works with O(n) memory.
    Returns (best_sequence, best_score).
    """
    n = init_seq.shape[0]
    # FFT length – next power of two covering full convolution
    m = 1 << ((2 * n - 1).bit_length())

    cand = np.clip(init_seq.copy(), 0.0, 1000.0)
    best = cand.copy()
    best_score = objective(cand, m)

    # initial step size (relative to the typical magnitude)
    sigma = max(0.01 * np.mean(cand) if np.mean(cand) > 0 else 0.01, 1e-3)

    for _ in range(steps):
        op = rng.integers(0, 3)          # 0: perturb two entries, 1: swap, 2: block scale

        if op == 0:                      # move a little mass from j to i
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            delta = rng.normal(0.0, sigma)
            new_i = cand[i] + delta
            new_j = cand[j] - delta
            new_i = max(0.0, min(1000.0, new_i))
            new_j = max(0.0, min(1000.0, new_j))
            new = cand.copy()
            new[i] = new_i
            new[j] = new_j

        elif op == 1:                    # simple swap of two positions
            i, j = rng.choice(n, size=2, replace=False)
            new = cand.copy()
            new[i], new[j] = new[j], new[i]

        else:                            # multiplicative perturbation of a contiguous block
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            new = cand.copy()
            new[start:start + width] *= factor
            np.clip(new, 0.0, 1000.0, out=new)

        sc = objective(new, m)

        if np.isfinite(sc) and sc < best_score:
            best = new
            best_score = sc
            cand = new
            sigma = max(sigma * 0.99, 1e-8)   # slowly tighten the step size

    return best, best_score

def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs) -> list:
    """
    Returns a non‑empty list of floats.
    The routine tries several cheap initialisations and performs a
    lightweight stochastic search that only needs O(n) auxiliary memory.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Acquire the parent construction if it exists in the execution env.
    # -----------------------------------------------------------------
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent_arr = None

    # Fallback length if parent is absent or malformed
    n = parent_arr.shape[0] if (parent_arr is not None and
                                parent_arr.ndim == 1 and
                                parent_arr.size > 0 and
                                np.isfinite(parent_arr).all()) else 256

    # -----------------------------------------------------------------
    # Build a small set of starting sequences.
    # -----------------------------------------------------------------
    init_seqs = []

    if parent_arr is not None and parent_arr.ndim == 1 and parent_arr.size > 0 and np.isfinite(parent_arr).all():
        init_seqs.append(np.clip(parent_arr, 0.0, 1000.0))

    # Uniform vector – a classic baseline (scale chosen arbitrarily)
    init_seqs.append(np.full(n, 10.0, dtype=np.float64))

    # A mildly randomised Dirichlet‑like vector (sum kept roughly constant)
    rand_vec = rng.gamma(shape=0.8, scale=1.0, size=n)
    rand_vec /= np.sum(rand_vec)
    rand_vec *= 10.0 * n          # give it a comparable total mass
    init_seqs.append(np.clip(rand_vec, 0.0, 1000.0))

    # -----------------------------------------------------------------
    # Run a short search from each starter and keep the best result.
    # -----------------------------------------------------------------
    best_overall = None
    best_score = np.inf
    steps_per_run = max(1, num_steps // len(init_seqs))

    for idx, seq0 in enumerate(init_seqs):
        seq, sc = run_search(seq0, steps_per_run,
                             np.random.default_rng(seed + idx + 12345))
        if np.isfinite(sc) and sc < best_score:
            best_overall = seq
            best_score = sc

    # If everything failed, fall back to a simple constant sequence.
    if best_overall is None:
        best_overall = np.full(n, 1.0, dtype=np.float64)

    # Final safety clip – the evaluator repeats the same clipping.
    best_overall = np.clip(best_overall, 0.0, 1000.0)

    # Return a plain Python list of floats as required.
    return [float(v) for v in best_overall]
