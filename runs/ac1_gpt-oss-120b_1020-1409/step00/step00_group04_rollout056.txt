def propose_candidate(seed=42, num_iters=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # Load parent
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    L = 1 << ((2*n - 1).bit_length())
    # Helper score function using FFT
    def fast_score(arr):
        sum_a = arr.sum()
        if sum_a < 0.01:
            return np.inf
        # compute linear convolution via FFT
        freq = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(freq * freq, n=L)[:2*n-1]
        max_b = conv.max()
        return 2.0 * n * max_b / (sum_a * sum_a)
    # Initial candidate: random permutation of parent
    best = rng.permutation(parent)
    best_score = fast_score(best)
    # fallback best (valid) in case score inf
    fallback = best.copy()
    fallback_score = best_score
    # Simulated annealing loop
    for it in range(num_iters):
        # propose: swap two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]
        s = fast_score(cand)
        # accept if better
        if s < best_score:
            best = cand
            best_score = s
        else:
            # occasional uphill move
            # temperature schedule
            T = 0.1 * (1 - it / num_iters)  # linearly decreasing
            if rng.random() < np.exp(-(s - best_score) / max(T, 1e-12)):
                best = cand
                best_score = s
        # keep fallback as best seen valid
        if np.isfinite(s) and s < fallback_score:
            fallback = cand.copy()
            fallback_score = s
    # Evaluate final best using true evaluate_sequence for safety
    try:
        final_score = float(evaluate_sequence(best.tolist()))
        if np.isfinite(final_score):
            # keep best
            pass
        else:
            best = fallback
    except Exception:
        best = fallback
    return [float(x) for x in best]

def _fast_score(arr):
    n = arr.size
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    L = 1 << ((2*n - 1).bit_length())
    freq = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(freq * freq, n=L)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)

def _fast_score(arr: np.ndarray) -> float:
    """Return the AC1 upper bound for *arr* using an FFT‑based convolution."""
    n = arr.size
    sum_a = float(arr.sum())
    if sum_a < 0.01:
        return np.inf
    # length for a linear convolution (next power of two >= 2·n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    # real FFT, point‑wise square, inverse FFT → linear convolution
    freq = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(freq * freq, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed: int = 42, num_iters: int = 5000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 bound.
    Returns a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Load the parent construction if it exists, otherwise fall back.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = np.ones(256, dtype=np.float64)

    parent = np.clip(base, 0.0, 1000.0)
    if (parent.ndim != 1) or (parent.size == 0) or (not np.isfinite(parent).all()):
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # ------------------------------------------------------------------
    # Build a small set of diverse initial candidates.
    # ------------------------------------------------------------------
    init_candidates = []

    # 1. Random permutation of the parent values.
    init_candidates.append(rng.permutation(parent))

    # 2. Sorted descending (large values first).
    init_candidates.append(np.sort(parent)[::-1])

    # 3. Sorted ascending (large values last).
    init_candidates.append(np.sort(parent))

    # 4. Uniformly random positive vector with the same total mass.
    rnd = rng.random(n)
    rnd *= parent.sum() / rnd.sum()
    init_candidates.append(rnd)

    # 5. A smooth cosine‑based pattern, rescaled to the same sum.
    phi = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    idx = np.arange(n, dtype=np.float64)
    cos_pat = (np.cos(2.0 * np.pi * idx * phi) + 1.0) / 2.0
    cos_pat *= parent.sum() / cos_pat.sum()
    init_candidates.append(cos_pat)

    # Evaluate all starters and keep the best one.
    best = None
    best_score = np.inf
    for cand in init_candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _fast_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # Safety fall‑back (should never be needed).
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = _fast_score(best)

    fallback = best.copy()
    fallback_score = best_score

    # ------------------------------------------------------------------
    # Simple stochastic local search: swaps + mild block scalings.
    # ------------------------------------------------------------------
    for it in range(num_iters):
        # Choose a move type.
        if rng.random() < 0.65:                     # --- swap two entries ---
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = best.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                                        # --- scale a random block ---
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))  # gentle log‑normal
            cand = best.copy()
            cand[start : start + width] *= factor

        # Enforce the problem’s box constraints.
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _fast_score(cand)

        # Greedy acceptance of improvements.
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

        # Keep the overall best seen (even if not accepted).
        if np.isfinite(sc) and sc < fallback_score:
            fallback = cand.copy()
            fallback_score = sc

    # ------------------------------------------------------------------
    # Verify with the authoritative evaluator; fall back if necessary.
    # ------------------------------------------------------------------
    try:
        true_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(true_score):
            best = fallback
    except Exception:
        best = fallback

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
