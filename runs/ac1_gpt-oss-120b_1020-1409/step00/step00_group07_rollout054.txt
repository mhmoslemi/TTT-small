def propose_candidate(seed=42, num_steps=3000, **kwargs):
    rng = np.random.default_rng(seed)
    # load parent
    parent = np.asarray(height_sequence_1, dtype=np.float64) if 'height_sequence_1' in globals() else np.ones(256, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    # fallback if invalid
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    cur = parent.copy()
    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = cur.size

    for step in range(num_steps):
        # Occasionally random perturb
        if rng.random() < 0.05:
            # block perturbation similar to previous
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            proposal = cur.copy()
            proposal[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            if rng.random() < 0.35:
                proposal = 0.5 * (proposal + proposal[::-1])
            proposal = np.clip(proposal, 0.0, 1000.0)
        else:
            # targeted mass transfer
            conv = np.convolve(cur, cur)
            k0 = int(conv.argmax())  # shift with max
            i_low = max(0, k0 - (n - 1))
            i_high = min(n - 1, k0)
            # if whole range, fallback to random
            if i_low == 0 and i_high == n - 1:
                # choose two random indices
                src_idx = int(rng.integers(0, n))
                tgt_idx = int(rng.integers(0, n))
                if src_idx == tgt_idx:
                    continue
            else:
                i_range = np.arange(i_low, i_high + 1)
                partners = k0 - i_range
                # compute contributions
                contrib = cur[i_range] * cur[partners]
                # choose source as max contribution
                src_idx = int(i_range[np.argmax(contrib)])
                # choose target outside the range
                # build possible target indices list could be two ranges
                # pick side with probability proportional to size
                left_size = i_low
                right_size = n - 1 - i_high
                if left_size + right_size == 0:
                    continue
                if rng.random() < (left_size / (left_size + right_size)):
                    tgt_idx = int(rng.integers(0, i_low))
                else:
                    tgt_idx = int(rng.integers(i_high + 1, n))
            # define delta
            src_val = cur[src_idx]
            if src_val <= 0:
                continue
            # step fraction decreasing
            step_frac = 0.1 * (1.0 - step / num_steps) + 0.001
            delta = src_val * step_frac * rng.random()
            if delta <= 0:
                continue
            proposal = cur.copy()
            proposal[src_idx] = max(0.0, proposal[src_idx] - delta)
            proposal[tgt_idx] = min(1000.0, proposal[tgt_idx] + delta)
        # evaluate
        score = float(evaluate_sequence(proposal.tolist()))
        if np.isfinite(score) and score < best_score:
            best = proposal
            best_score = score
            cur = proposal
        # optionally accept if score improved relative to cur even if not best, for exploration
        elif np.isfinite(score) and score < float(evaluate_sequence(cur.tolist())):
            cur = proposal

    return [float(x) for x in best]

def _initial_sequence():
    """Return a valid starting sequence."""
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        seq = np.ones(256, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.ndim != 1 or seq.size == 0 or not np.isfinite(seq).all():
        seq = np.ones(256, dtype=np.float64)
    return seq


def _random_block(seq, rng):
    """Create a cheap random block perturbation of ``seq``."""
    n = seq.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand = seq.copy()
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _targeted_move(seq, rng, step_frac):
    """Move a fraction of mass from an index participating in the
    maximal convolution entry to an index outside that participation set."""
    n = seq.size
    conv = np.convolve(seq, seq)
    k0 = int(conv.argmax())               # shift with maximal value
    i_low = max(0, k0 - (n - 1))
    i_high = min(n - 1, k0)

    # If the whole array participates (k0 == n-1) fall back to a random move.
    if i_low == 0 and i_high == n - 1:
        src = int(rng.integers(0, n))
        dst = int(rng.integers(0, n - 1))
        while dst == src:
            dst = int(rng.integers(0, n - 1))
    else:
        # indices that affect the maximal shift
        idx_range = np.arange(i_low, i_high + 1)
        partners = k0 - idx_range
        contrib = seq[idx_range] * seq[partners]
        src = int(idx_range[np.argmax(contrib)])

        # choose a target outside the active range
        left = i_low
        right = n - (i_high + 1)
        if left + right == 0:  # shouldn't happen, but guard anyway
            dst = src
        elif left == 0:
            dst = int(rng.integers(i_high + 1, n))
        elif right == 0:
            dst = int(rng.integers(0, i_low))
        else:
            if rng.random() < left / (left + right):
                dst = int(rng.integers(0, i_low))
            else:
                dst = int(rng.integers(i_high + 1, n))

    src_val = seq[src]
    if src_val <= 0.0:
        return seq.copy()   # nothing to move

    # move a fraction of the source mass
    delta = src_val * step_frac * rng.random()
    cand = seq.copy()
    cand[src] = max(0.0, cand[src] - delta)
    cand[dst] = min(1000.0, cand[dst] + delta)
    return cand


def propose_candidate(seed=42, num_steps=5000, **kwargs):
    """Return a non‑negative step‑function that (heuristically) minimizes the
    AC1 upper bound.  The implementation uses a memory‑bounded local search
    mixing targeted mass moves with occasional random block perturbations."""
    rng = np.random.default_rng(seed)

    cur = _initial_sequence()
    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score
    n = cur.size

    for step in range(num_steps):
        # decreasing step size for the targeted move
        step_frac = 0.12 * (1.0 - step / num_steps) + 0.001

        # 5 % of the time we inject a random block perturbation to escape plateaus
        if rng.random() < 0.05:
            cand = _random_block(cur, rng)
        else:
            cand = _targeted_move(cur, rng, step_frac)

        # evaluate candidate
        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue

        # keep the overall best solution
        if score < best_score:
            best = cand
            best_score = score
            cur = cand
            cur_score = score
            continue

        # accept improvements over the current point (hill climbing)
        if score < cur_score:
            cur = cand
            cur_score = score

    # As a final safety net ensure the returned list is non‑empty and finite
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
