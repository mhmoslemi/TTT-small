import numpy as np

def propose_candidate(seed=42, **kwargs):

    Return a uniform non‑negative step function of the appropriate length.

    evaluator's constraints.

    # when a parent construction exists; otherwise fall back to a modest size.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # Any constant value works; pick 1.0 which is safely within [0,1000].
    const_val = 1.0
    candidate = np.full(n, const_val, dtype=np.float64)

    # Convert to plain Python list of floats as required by the evaluator.
