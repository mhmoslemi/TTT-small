import numpy as np

# evaluate_sequence is provided in the execution environment.


def _gradient_for_max(seq: np.ndarray, max_idx: int) -> np.ndarray:
    """Return the (approximate) gradient of max(b) w.r.t. seq,
    where b = np.convolve(seq, seq) and max_idx is an argmax of b.
    The gradient for the full‑convolution loss max(b) is
        ∂max(b)/∂a_i = 2 * a_{max_idx-i}   (zero if the partner index is out of range).
    """
    n = seq.size
    idx = np.arange(n)
    partner = max_idx - idx
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(seq)
    # factor 2 follows from derivative of a_i * a_j + a_j * a_i.
    grad[mask] = 2.0 * seq[partner[mask]]
    return grad


def propose_candidate(seed: int = 42, num_iters: int = 1500, **kwargs):
    """
    Return a list of non‑negative step‑function heights that (heuristically)
    minimize the AC1 upper bound used by `evaluate_sequence`.

    The algorithm starts from the supplied `height_sequence_1` (if available),
    then repeatedly moves mass from indices that strongly contribute to the
    current maximal convolution entry to those that contribute little.
    Occasional random multiplicative noise helps escape plateaus.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise a feasible sequence.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        # fallback: a modest uniform vector
        base = np.full(256, 1.0, dtype=np.float64)

    a = np.clip(base, 0.0, 1000.0)
    # Ensure a minimal total mass (the evaluator rejects sum < 0.01)
    if a.sum() < 0.01:
        a = np.ones_like(a, dtype=np.float64)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    n = a.size
    # step size for the deterministic mass‑transfer step
    step = 0.15

    for it in range(num_iters):

        # 2.  Compute the convolution and locate the current worst entry.
        # ------------------------------------------------------------------
        conv = np.convolve(a, a)           # length 2n‑1, O(n log n) via FFT internally
        max_idx = int(np.argmax(conv))

        # 3.  Gradient w.r.t. the maximal entry.
        # ------------------------------------------------------------------
        grad = _gradient_for_max(a, max_idx)

        # If the gradient is all zeros (unlikely) we break.
        if not np.any(grad):

        # 4.  Move mass from high‑gradient positions to low‑gradient ones.
        # ------------------------------------------------------------------
        avg_grad = grad.mean()

        delta = step * (avg_grad - grad)   # positive where grad is below average, negative otherwise
        cand = a + delta

        # Clip to the admissible box [0,1000].
        cand = np.clip(cand, 0.0, 1000.0)

        # Renormalise to keep the total mass unchanged (very cheap).
        sum_a = a.sum()
        sum_c = cand.sum()
        if sum_c > 0:
            cand *= (sum_a / sum_c)
        else:
            # all entries vanished – revert to current a and skip this iteration
            cand = a.copy()

        # ------------------------------------------------------------------
        # 5.  Evaluate and possibly accept.
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            a = cand    # continue optimisation from the improved point
        else:
            # keep the old a (the deterministic step might be too aggressive)
            a = a

        # ------------------------------------------------------------------
        # 6.  Occasional stochastic jitter to avoid stagnation.
        # ------------------------------------------------------------------
        if (it + 1) % 40 == 0:
            # small multiplicative log‑normal noise
            noise = np.exp(rng.normal(0.0, 0.025, size=n))
            jitter = a * noise
            jitter = np.clip(jitter, 0.0, 1000.0)
            # renormalise the jittered candidate
            jitter_sum = jitter.sum()
            if jitter_sum > 0:
                jitter *= (sum_a / jitter_sum)
                j_score = float(evaluate_sequence(jitter.tolist()))
                if np.isfinite(j_score) and j_score < best_score:
                    best = jitter
                    best_score = j_score
                    a = jitter

    # ------------------------------------------------------------------
    # 7.  Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
