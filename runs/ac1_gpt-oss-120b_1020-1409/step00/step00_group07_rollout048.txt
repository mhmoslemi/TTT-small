import numpy as np

def _as_valid_array(arr):
    """Convert input to a clean 1‑D NumPy array with the required bounds."""
    arr = np.asarray(arr, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
        return None
    # clip to the limits enforced by the evaluator
    return np.clip(arr, 0.0, 1000.0)

def _constant_array(length, value=1.0):
    """Return a constant, non‑negative array of the given length."""
    return np.full(int(length), float(value), dtype=np.float64)

def propose_candidate(seed=42, **kwargs):
    """
    Return a sequence of heights that (provably) attains the minimal possible
    AC1 upper bound 2.0 for the given evaluator.

    The routine prefers an existing parent construction (``height_sequence_1``)
    when it is valid, but also compares it with a simple constant sequence and
    returns the better of the two.
    """
    # ----------------------------------------------------------------------
    # 1.  Determine the working length and a sensible fallback.
    # ----------------------------------------------------------------------
    fallback_len = 256                     # used if the parent is unavailable
    try:
        # ``height_sequence_1`` is supplied by the harness when present.
        parent_raw = height_sequence_1
    except NameError:
        parent_raw = None

    parent_arr = _as_valid_array(parent_raw) if parent_raw is not None else None
    if parent_arr is None:
        # No usable parent – start from a small constant array.
        n = fallback_len
        parent_arr = _constant_array(n, 1.0)
    else:
        n = parent_arr.size

    # ----------------------------------------------------------------------
    # 2.  Build a competing constant candidate.
    # ----------------------------------------------------------------------
    const_arr = _constant_array(n, 1.0)    # any constant gives the optimal bound 2

    # ----------------------------------------------------------------------
    # 3.  Evaluate both candidates using the supplied scorer.
    # ----------------------------------------------------------------------
    # The evaluator expects a plain Python list.
    parent_score = float(evaluate_sequence(parent_arr.tolist()))
    const_score = float(evaluate_sequence(const_arr.tolist()))

    # ----------------------------------------------------------------------
    # 4.  Choose the best finite result.
    # ----------------------------------------------------------------------
    best_arr = parent_arr
    best_score = parent_score
    if np.isfinite(const_score) and const_score < best_score:
        best_arr = const_arr
        best_score = const_score

    # As a final safety net, ensure we always return a valid non‑empty list.
    if best_arr.size == 0 or not np.isfinite(best_score):
        best_arr = _constant_array(fallback_len, 1.0)

    # Convert to plain Python floats for the harness.
    return [float(x) for x in best_arr]
