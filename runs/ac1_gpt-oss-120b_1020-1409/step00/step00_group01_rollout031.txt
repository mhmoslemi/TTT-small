import numpy as np

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # try to fetch height_sequence_1 if defined, else fallback uniform
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
    except Exception:
        base = np.ones(256, dtype=np.float64)

    n = base.size
    # target sum for scaling (optional)
    target_sum = np.sqrt(2.0 * n)

    def evaluate(arr):
        return float(evaluate_sequence(arr.tolist()))

    # Prepare candidates list of arrays
    candidates = []

    # candidate: uniform
    uniform = np.full(n, target_sum / n)
    candidates.append(uniform)

    # exponential decreasing r in a range
    for r in np.linspace(0.9, 0.9999, 12):
        exp_seq = np.power(r, np.arange(n))
        # scale to target sum
        scale = target_sum / exp_seq.sum()
        candidates.append(exp_seq * scale)

    # exponential increasing (reverse)
    for r in np.linspace(0.9, 0.9999, 12):
        exp_seq = np.power(r, np.arange(n-1, -1, -1))
        scale = target_sum / exp_seq.sum()
        candidates.append(exp_seq * scale)

    # power-law
    for p in np.linspace(0.2, 2.0, 12):
        seq = 1.0 / (np.arange(1, n+1) ** p)
        scale = target_sum / seq.sum()
        candidates.append(seq * scale)

    # linear decreasing from max to zero
    for frac in np.linspace(0.5, 1.0, 5):
        a0 = target_sum * 2.0 * frac / n  # choose a0 such that sum approx target?
        # Actually linear: a_i = a0*(1 - i/(n-1))
        lin = a0 * (1 - np.arange(n) / (n-1))
        lin[lin < 0] = 0.0
        scale = target_sum / lin.sum() if lin.sum() > 0 else 1.0
        candidates.append(lin * scale)

    # sparse pattern: choose k roughly sqrt(n) up to maybe n/10
    sqrt_n = int(np.sqrt(n))
    for k_multiplier in [1, 2, 3]:
        k = min(n, max(2, sqrt_n * k_multiplier))
        # positions equally spaced
        positions = np.linspace(0, n-1, k, dtype=int)
        sparse = np.zeros(n, dtype=np.float64)
        sparse[positions] = 1.0
        scale = target_sum / sparse.sum()
        candidates.append(sparse * scale)

    # random perturbations from parent plus scaling
    for _ in range(10):
        pert = base.copy()
        # apply random block scaling similar to previous but heavier
        width = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, max(1, n - width + 1))
        stop = min(n, start + width)
        pert[start:stop] *= np.exp(rng.normal(0.0, 0.3))
        # random mirror
        if rng.random() < 0.5:
            pert = 0.5 * (pert + pert[::-1])
        # scale to target sum
        scale = target_sum / pert.sum()
        candidates.append(pert * scale)

    # Evaluate candidates, keep best
    best_seq = None
    best_score = np.inf
    for cand in candidates:
        # clip to bounds
        cand = np.clip(cand, 0.0, 1000.0)
        score = evaluate(cand)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()

    # optional local improvement on best using block perturb search
    if best_seq is not None:
        # run few local steps (maybe 200)
        seq = best_seq.copy()
        for _ in range(200):
            cand = seq.copy()
            # choose random block
            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))
            stop = min(n, start + width)
            cand[start:stop] *= np.exp(rng.normal(0.0, 0.12))
            if rng.random() < 0.35:

            cand = np.clip(cand, 0.0, 1000.0)
            score = evaluate(cand)
            if np.isfinite(score) and score < best_score:
                best_score = score
                seq = cand
        best_seq = seq

    # final fallback if none valid
    if best_seq is None:
        best_seq = uniform

    # return as list of floats
    return [float(x) for x in best_seq]

import numpy as np

# Helper: obtain a safe parent sequence.
def _parent_sequence():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
            raise Exception
        return arr
    except Exception:
        # fallback length 256 uniform vector
        return np.ones(256, dtype=np.float64)


# Helper: evaluate a candidate sequence with the provided evaluator.
def _evaluate(arr):
    # the evaluator expects a Python list of floats
    return float(evaluate_sequence(arr.tolist()))


# Helper: clip values and rescale to a target total sum (scale invariant for the objective).
def _scale_to_target(arr, target_sum):
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s > 0.0:
        arr = arr * (target_sum / s)
    else:
        arr = np.full_like(arr, target_sum / arr.size)
    return np.clip(arr, 0.0, 1000.0)


def propose_candidate(seed=42, **kwargs):
    """Search for a step‑function that lowers the AC1 bound."""
    rng = np.random.default_rng(seed)

    base = _parent_sequence()
    n = base.size
    target_sum = np.sqrt(2.0 * n)          # any non‑zero sum works; this keeps magnitudes modest.

    # ------------------------------------------------------------------
    # 1.  Generate a variety of parametrised candidates (all O(n) memory).
    # ------------------------------------------------------------------
    candidates = []

    # uniform baseline
    candidates.append(np.full(n, target_sum / n))

    # exponential decays (forward, backward, symmetric)
    for r in np.linspace(0.90, 0.9999, 12):
        exp_fwd = np.power(r, np.arange(n, dtype=np.float64))
        candidates.append(exp_fwd)

        exp_rev = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
        candidates.append(exp_rev)

        dist = np.minimum(np.arange(n, dtype=np.float64),
                          (n - 1) - np.arange(n, dtype=np.float64))
        sym_exp = np.power(r, dist)
        candidates.append(sym_exp)

    # mixtures of uniform + exponential
    uniform = np.full(n, target_sum / n)
    for r in np.linspace(0.90, 0.9999, 8):
        exp_seq = np.power(r, np.arange(n, dtype=np.float64))
        for alpha in np.linspace(0.0, 1.0, 5):
            mix = alpha * exp_seq + (1.0 - alpha) * uniform
            candidates.append(mix)

    # power‑law decays
    for p in np.linspace(0.1, 2.0, 10):
        pow_seq = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        candidates.append(pow_seq)

    # sparse patterns (approximately Sidon‑like spacing)
    sqrt_n = int(np.sqrt(n))
    for mult in (1, 2, 3):

        candidates.append(cand)

    # ------------------------------------------------------------------

    best_seq = None
