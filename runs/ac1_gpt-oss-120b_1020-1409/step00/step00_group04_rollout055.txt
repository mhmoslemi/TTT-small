def full_convolution(a):
    n = a.shape[0]

# ---- Helper: fast full convolution via FFT (real input) --------------------
def _full_convolve_fft(x, fft_len):
    """Return full convolution x*x using an FFT of length fft_len."""
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)
    return conv[: 2 * x.size - 1]
