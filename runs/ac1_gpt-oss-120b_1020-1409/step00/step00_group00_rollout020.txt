import numpy as np

# Helper: compute the AC1 upper bound for a non‑negative vector.
def _ac1_score(vec: np.ndarray) -> float:
    s = float(vec.sum())
    if s < 0.01:
        return np.inf
    conv = np.convolve(vec, vec)
    return 2.0 * vec.size * float(conv.max()) / (s * s)


def propose_candidate(seed: int = 42, num_iters: int = 2000, **kwargs):
    """
    Search for a non‑negative step‑function that lowers the AC1 bound.
    Uses a lightweight greedy move that reduces the maximal convolution entry.
    Falls back to a uniform distribution if nothing improves.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise from the provided parent (height_sequence_1) or uniform.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # fallback length – must match the real parent length if known.
        n = 4806
        parent = np.ones(n, dtype=np.float64)

    n = parent.size
    # Normalise to total mass 1 (scale does not affect the score).
    total = parent.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = parent / total

    # Also keep a uniform start – it is cheap to compute and often good.
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # Choose the better of the two as the current best.
    score_a = _ac1_score(a)
    score_u = _ac1_score(uniform)
    if score_u < score_a:
        a = uniform.copy()
        best_score = score_u
    else:
        best_score = score_a

    best = a.copy()

    # ------------------------------------------------------------------
    # 2.  Main greedy search.
    # ------------------------------------------------------------------
    for it in range(num_iters):
        # Convolution of the current best vector.
        conv = np.convolve(best, best)
        k_max = int(conv.argmax())          # index where the max occurs

        # The contributing indices satisfy i + j = k_max.
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)

        if i_min > i_max:
            continue

        # Randomly pick a contributing pair (i, j).
        i = rng.integers(i_min, i_max + 1)
        j = k_max - i                       # guaranteed 0 ≤ j < n

        # Amount we can move – keep both entries non‑negative.
        max_delta = min(best[i], best[j]) * 0.5
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()    # a small chunk of mass

        # Choose a target index distinct from i and j.
        t = rng.integers(0, n)
        while t == i or t == j:
            t = rng.integers(0, n)

        # Propose the new candidate.
        cand = best.copy()
        cand[i] -= delta
        cand[j] -= delta
        cand[t] += 2.0 * delta

        # Evaluate the candidate (fast local evaluation).
        cand_score = _ac1_score(cand)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            # Early exit if we beat the target bound.
            if best_score < 1.50274349:
                break

        # Every few iterations try a fresh random point (helps escape plateaus).
        if (it + 1) % 400 == 0:
            rand_vec = rng.random(n)
            rand_vec /= rand_vec.sum()
            rand_score = _ac1_score(rand_vec)
            if rand_score < best_score:
                best = rand_vec
                best_score = rand_score

    # ------------------------------------------------------------------
    # 3.  Final safety: clip to the allowed range and return as a list.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
