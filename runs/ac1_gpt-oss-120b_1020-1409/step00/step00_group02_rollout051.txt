import numpy as np

def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using a single FFT (full mode)."""
    n = len(x) + len(y) - 1
    L = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    Z = X * Y
    conv = np.fft.irfft(Z, n=L)[:n]
    return conv

def propose_candidate(seed=42, num_steps=1500, **kwargs):
    """Return a non‑negative sequence that (hopefully) beats the 1.5027 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # 1.  Initialise from the provided parent (height_sequence_1) or a
    #     simple fallback.  Clip to the allowed range and normalise.
    # ------------------------------------------------------------------ #
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:                         # height_sequence_1 not present
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    total = float(np.sum(base))
    if total < 0.01:                          # guard against near‑zero sum
        base = np.ones_like(base)
        total = float(np.sum(base))

    n = base.size
    a = base / total                           # internal representation : sum == 1

    # ------------------------------------------------------------------ #

        """Exact score via the supplied evaluator (handles any scaling)."""
        return float(evaluate_sequence(arr.tolist()))
