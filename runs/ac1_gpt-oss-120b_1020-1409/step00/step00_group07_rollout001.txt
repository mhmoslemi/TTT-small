def propose_candidate(seed=42, num_steps=200, **kwargs):
    rng = np.random.default_rng(seed)
    # Base candidate
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            base = np.ones(256, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)
    n = base.size
    
    # Generate initial candidate by rearranging extremes
    sorted_seq = np.sort(base)  # ascending
    # Interleave largest, smallest
    candidate = np.empty_like(sorted_seq)
    left, right = 0, n-1
    for i in range(n):
        if i % 2 == 0:
            candidate[i] = sorted_seq[right]
            right -= 1
        else:
            candidate[i] = sorted_seq[left]
            left += 1
    # Clip to allowed range
    candidate = np.clip(candidate, 0.0, 1000.0)
    
    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Also compare to using original base as candidate (fallback)
    base_score = float(evaluate_sequence(base.tolist()))
    if np.isfinite(base_score) and base_score < best_score:
        best = base
        best_score = base_score
    
    # local search with block perturbations (like previous)
    for step in range(num_steps):
        # Decide operation type: swap two indices or block perturbation
        if rng.random() < 0.5:
            # swap a few pairs
            cand = best.copy()
            # number of swaps proportion to n
            num_swaps = max(1, int(rng.integers(1, max(2, n // 200))))
            for _ in range(num_swaps):
                i, j = rng.integers(0, n, size=2)
                cand[i], cand[j] = cand[j], cand[i]
            cand = np.clip(cand, 0.0, 1000.0)
        else:
            # block perturbation similar to previous
            cand = best.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            # multiplicative factor sampled lognormal
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:stop] *= factor
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)
        # Optionally apply scaling to ensure sum not too small: if sum < small threshold, rescale.
        # But ensure sums remain bounded.
        # Evaluate.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    # Return best as list
    return [float(x) for x in best]

import numpy as np

def is_prime(p: int) -> bool:
    """Simple trial‑division primality test (p ≤ 10⁵ is cheap)."""
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    r = int(p**0.5)
    for d in range(3, r + 1, 2):
        if p % d == 0:
            return False
    return True


def legendre_binary(length: int) -> np.ndarray:
    """
    Produce a 0/1 sequence of given length based on the Legendre symbol.
    If length is not prime, the largest prime ≤ length is used as modulus.
    """
    p = length
    while p > 2 and not is_prime(p):
        p -= 1
    exp = (p - 1) // 2          # exponent for Legendre computation
    seq = np.empty(length, dtype=np.float64)
    for i in range(length):
        r = i % p
        if r == 0:
            seq[i] = 0.0
        else:
            leg = pow(r, exp, p)    # 1 for quadratic residue, p‑1 for non‑residue
            seq[i] = 1.0 if leg == 1 else 0.0
    return seq

    # ------------------------------------------------------------------
    candidates = []

    # 2) Legendre binary sequence (and a few rotations)
    try:

        candidates.append(leg)
        for _ in range(3):

            candidates.append(np.roll(leg, shift))

            right -= 1
