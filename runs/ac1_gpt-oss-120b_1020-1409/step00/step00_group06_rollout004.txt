import numpy as np

def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a small O(n) perturbation of ``seq``."""
    n = seq.shape[0]
    y = seq.copy()
    # block width between 1 and max(2, n//12)
    min_width = max(1, n // 200)
    max_width = max(2, n // 12)
    # rng.integers(low, high) uses half‑open interval [low, high)
    width = int(rng.integers(min_width, max_width + 1))
    # start position so that block fits
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    if stop > n:
        stop = n
    # modest multiplicative change
    scale = float(np.exp(rng.normal(0.0, 0.03)))
    y[start:stop] *= scale
    # occasional symmetrisation
    if rng.random() < 0.2:
        y = 0.5 * (y + y[::-1])
    np.clip(y, 0.0, 1000.0, out=y)
    return y
