def build_sidon_set(n, rng, max_size=None):
    if max_size is None:
        max_size = int(2 * np.sqrt(n))  # approx optimal.
    chosen = []
    sums = set()
    attempts = 0
    max_attempts = n * 10  # compute maybe large.
    while len(chosen) < max_size and attempts < max_attempts:
        i = int(rng.integers(0, n))
        if i in chosen:
            attempts += 1
            continue
        conflict = False
        # compute i + j for all j in chosen + i
        # first check i + i
        if i + i in sums:
            conflict = True
        else:
            for j in chosen:
                s = i + j
                if s in sums:
                    conflict = True
                    break
        if conflict:
            attempts += 1
            continue
        # add i
        for j in chosen:
            sums.add(i + j)
        sums.add(i + i)
        chosen.append(i)
        attempts = 0
    return chosen

def find_sidon_set(n, rng, target_m):
    best_set = []
    best_len = 0
    # Try multiple restarts
    for _ in range(5):
        chosen = []
        sums = set()
        attempts = 0
        max_attempts = n
        while attempts < max_attempts and len(chosen) < target_m:
            i = int(rng.integers(0, n))
            if i in chosen:
                attempts += 1
                continue
            conflict = False
            # check i+i
            if i + i in sums:
                conflict = True
            else:
                for j in chosen:
                    if i + j in sums:
                        conflict = True
                        break
            if conflict:
                attempts += 1
                continue
            # add i
            for j in chosen:
                sums.add(i + j)
            sums.add(i + i)
            chosen.append(i)
            attempts = 0  # reset after success
        if len(chosen) > best_len:
            best_len = len(chosen)
            best_set = chosen
        if best_len >= target_m:
            break
    return best_set

import numpy as np

def fast_convolve(a):
    """Linear convolution a*a using FFT (O(n log n), linear memory)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def subgradient_step(a, k_max, step):
    """
    Perform one sub‑gradient (actually a descent) step aimed at reducing
    the maximal convolution coefficient at shift k_max.
    a is assumed to have non‑negative entries and sum 1.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)

    # gradient of b_k = a*a at the active shift:
    #   ∂b_k/∂a_i = a_i + a_{k-i}   (if the index exists)
    grad = np.zeros_like(a)
    grad[mask] = a[idx[mask]] + a[j[mask]]

    # take a small step opposite to the gradient and stay non‑negative
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, np.inf)

    # renormalise to keep the sum equal to 1 (scale‑invariant for the objective)
    total = a_new.sum()
    if total > 0.0:
        a_new /= total
    else:
        # degenerate case – fall back to a flat distribution
        a_new.fill(1.0 / n)
    return a_new
