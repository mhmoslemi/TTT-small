import numpy as np

# ------------------------------------------------------------
# Helper utilities – all operate in O(n) memory and time
# ------------------------------------------------------------
def _array_length():
    """
    Return the length to use for the candidate.
    If the global `height_sequence_1` exists and is a non‑empty 1‑D
    NumPy array, its length is used; otherwise a modest default is chosen.
    """
    try:
        arr = globals().get("height_sequence_1")
        if isinstance(arr, np.ndarray) and arr.ndim == 1 and arr.size > 0:
            return int(arr.size)
    except Exception:
        pass
    return 256  # safe fallback length


def _uniform_sequence(n):
    """
    Produce a uniform non‑negative step function of length ``n``.
    The constant value is chosen so that the L1‑norm equals sqrt(2 n),
    which is a common normalisation but does not affect the objective.
    """
    const = np.sqrt(2.0) / np.sqrt(n)  # sum = sqrt(2 n)
    return np.full(n, const, dtype=np.float64)


def _block_perturb(x, rng):
    """
    Produce an O(n) memory non‑negative perturbation of ``x``.
    A random contiguous block is scaled multiplicatively by a log‑normal factor.
    Occasionally the whole vector is symmetrised to keep the search well spread.
    """
    cand = x.copy()
    n = cand.size
    # block width between 1% and 10% of n, never zero
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.06)))  # modest change
    cand[start:stop] *= factor
    # optional mild symmetrisation
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _evaluate(seq):
    """
    Wrapper around the provided evaluate_sequence. Returns ``np.inf`` on any
    unexpected failure (should never happen for well‑formed input).
    """
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:
        return float('inf')

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed=42, num_steps=800, **kwargs):
    """
    Return a list of non‑negative floats representing a step function that
    (approximately) minimises the AC1 upper bound.
    The routine uses a tiny stochastic hill‑climber that starts from the
    mathematically optimal uniform distribution.  If the uniform candidate
    happens to be invalid (extremely unlikely), a safe fallback of ``1.0`` is
    used.
    """
    rng = np.random.default_rng(seed)

    n = _array_length()
    best = _uniform_sequence(n)
    best_score = _evaluate(best)

    # Guard against pathological evaluation (e.g. sum < 0.01 after clipping)
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best_score = _evaluate(best)

    # Simple stochastic hill‑climbing – each iteration is O(n)
    for _ in range(num_steps):
        cand = _block_perturb(best, rng)
        score = _evaluate(cand)
        # Accept only strict improvements to avoid stagnation
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Final sanity‑check – guarantee a feasible answer
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    # Convert to plain Python floats as required by the evaluator
    return [float(v) for v in best]
