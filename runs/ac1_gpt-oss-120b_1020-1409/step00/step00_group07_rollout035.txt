import numpy as np

def _score(n: int, conv: np.ndarray, total: float) -> float:
    """Return the objective 2*n*max(conv)/(total**2)."""
    if total <= 0.0:
        return np.inf
    return float(2.0 * n * conv.max() / (total * total))


def propose_candidate(seed: int = 42, num_steps: int = 15000, **kwargs):
    """
    Hill‑climbing search for a coefficient vector that lowers the AC1 bound.

    The routine works with the globally provided ``height_sequence_1`` (if
    present) as a starting point, otherwise falls back to a small uniform
    vector.  It keeps an exact convolution array and updates it in O(n)
    time for each coordinate modification, thereby staying far below the
    quadratic‑memory limit.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise the sequence ------------------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:                     # parent not supplied
        a = np.ones(256, dtype=np.float64)

    # enforce problem constraints
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if a.sum() < 0.01:                    # avoid the evaluate_sequence failure case
        a[:] = 1.0

    # initial convolution and score
    conv = np.convolve(a, a)               # full linear convolution (length 2n‑1)
    total = a.sum()
    best_a = a.copy()
    best_score = _score(n, conv, total)

    # ------------------------------------------------------------------
    # optimisation parameters ------------------------------------------------
    sigma = 0.25                            # log‑scale step size (mutation strength)

    for step in range(num_steps):
        # locate the current most violating lag
        k_max = int(conv.argmax())

        # ------------------------------------------------------------------
        # choose an index to modify
        #  70 % of the time we pick the entry that contributes most to the
        #  maximal convolution value; otherwise a random index.
        if rng.random() < 0.7:
            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            i_candidates = np.arange(i_low, i_high + 1)
            j_vals = k_max - i_candidates
            contrib = a[i_candidates] * a[j_vals]
            i = i_candidates[np.argmax(contrib)]
        else:
            i = rng.integers(0, n)

        # ------------------------------------------------------------------
        # propose a multiplicative perturbation (log‑normal step)
        factor = np.exp(rng.normal(0.0, sigma))
        new_val = a[i] * factor
        new_val = np.clip(new_val, 0.0, 1000.0)
        delta = new_val - a[i]
        if abs(delta) < 1e-12:
            continue

        # keep a snapshot of the vector before the change – needed for a cheap revert
        old_a = a.copy()
        old_total = total

        # --------------------------------------------------------------- --
        # update the convolution in O(n) time
        #   (a+δe_i)*(a+δe_i) = a*a + 2δ·shift(a,i) + δ²·e_{2i}
        conv[i:i + n] += 2.0 * delta * old_a
        conv[2 * i] += delta * delta

        # apply the change to the sequence
        a[i] = new_val
        total += delta

        # evaluate the new objective (exact, no external call)
        cur_score = _score(n, conv, total)

        # ------------------------------------------------------------------
        # keep the move only if it improves the bound
        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # slowly shrink the mutation strength to fine‑tune later steps
            sigma *= 0.9995
        else:
            # revert everything
            a = old_a
            total = old_total
            conv[i:i + n] -= 2.0 * delta * old_a
            conv[2 * i] -= delta * delta

    # final safety clipping (again, just in case)
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
