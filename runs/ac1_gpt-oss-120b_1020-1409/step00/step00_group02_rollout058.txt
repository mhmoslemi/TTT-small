def propose_candidate(seed=42):
    import numpy as np
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    else:
        n = 256
    # Ensure n > 0
    n = max(1, n)

    best_seq = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # parameter grids
    deltas = np.logspace(np.log10(0.00005), np.log10(0.001), num=30) # 30 values
    c0_vals = np.linspace(0, 200, num=21)  # baseline from 0 to 200

    for delta in deltas:
        exp_factor = np.exp(-delta * np.arange(n, dtype=np.float64))
        for c0 in c0_vals:
            scale = max(0.0, 1000.0 - c0)  # ensure non-negative
            seq = scale * exp_factor + c0
            # Clip for safety
            seq = np.clip(seq, 0.0, 1000.0)
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq.copy()
    # Further local search: random block perturbations from best_seq
    for _ in range(200):
        # Choose random block perturbation similar to prior program
        candidate = best_seq.copy()
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        if rng.random() < 0.35:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = candidate
            best_score = score

    # Fallback: if best_score is inf, use uniform sequence
    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)
    return [float(x) for x in best_seq]

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function whose AC1 upper bound
    (2 n*max(conv)/sum**2) is as small as possible.
    The method scans a 2‑parameter family:
        a_i = (1000 - c0) * exp(-lam * i) + c0
    and then performs a short local refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Base length (use the provided construction if present)
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate and keep the best so far
    # ------------------------------------------------------------------
    best_seq = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Parameter grid search
    #   lam  : exponential decay rate (≈1‑r)
    #   c0   : constant offset added to every entry
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # lam spans roughly 3·10⁻⁵ … 3·10⁻³ (log‑space, 20 points)
    lam_grid = np.logspace(-4.5, -2.5, num=20)
    # c0 from 0 to 150 in steps of 10 (float to keep vectorised)
    c0_grid = np.arange(0, 151, 10, dtype=np.float64)

    for lam in lam_grid:
        # exponential part : exp(-lam * i)
        exp_part = np.exp(-lam * idx)

        for c0 in c0_grid:
            scale = 1000.0 - c0
            if scale <= 0.0:
                continue

            seq = scale * exp_part + c0
            # Numerical safety (should already be ≤1000)
            seq = np.clip(seq, 0.0, 1000.0)

            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq.copy()

    # ------------------------------------------------------------------
    # Small stochastic refinement around the current best
    # ------------------------------------------------------------------
    # number of extra attempts (adjustable, keeps runtime modest)
    for _ in range(300):
        # random block perturbation (similar to the baseline program)
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = best_seq.copy()
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        # occasional symmetrisation – cheap, sometimes helpful
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand

    # ------------------------------------------------------------------
    # Fallback guard – should never be needed but kept for safety
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in best_seq]
