import numpy as np

def _full_convolution(a, L):
    fft_a = np.fft.fft(a, L)
    conv = np.fft.ifft(fft_a * fft_a).real
    return conv[:2*a.size-1]

def _gradient_step(a, n, L, rng):
    # compute convolution and gradient step, return new a and score
    conv = _full_convolution(a, L)
    M = conv.max()
    k = int(conv.argmax())
    S = a.sum()
    # compute g
    g = np.zeros_like(a)
    i_start = max(0, k - (n-1))
    i_end = min(n-1, k)
    if i_start <= i_end:
        i_idx = np.arange(i_start, i_end+1)
        j_idx = k - i_idx
        g[i_idx] = 2.0 * a[j_idx]
    factor = 2.0 * n / (S * S)
    grad = factor * (g - (2.0 * M / S) * np.ones_like(a))
    # step size
    eta = 0.5
    noise = rng.normal(scale=0.01, size=n)
    a_new = a - eta * grad + noise
    a_new = np.clip(a_new, 0.0, 1000.0)
    new_score = 2.0 * n * M / (S * S)  # Using old M,S; we will recompute after
    return a_new, conv, M, S

def make_block_perturbation(base, rng):
    # similar to previous but adjust
    candidate = base.copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    candidate[start:stop] *= factor
    if rng.random() < 0.35:

    return np.clip(candidate, 0.0, 1000.0)

import numpy as np

# ------------------------------------------------------------
# Helper: fast full convolution using FFT (O(n log n) memory O(n))
# ------------------------------------------------------------
def _full_convolution(x, L):
    """Return full convolution x*x of length 2*len(x)-1 using size L FFT."""
    fx = np.fft.fft(x, n=L)
    conv = np.fft.ifft(fx * fx).real
    return conv[: 2 * x.size - 1]


# ------------------------------------------------------------
# Score computed exactly as evaluate_sequence but with FFT evaluation
# ------------------------------------------------------------
def _score(x, n, L):
    """Quadratic AC1 bound for a candidate array x."""
    S = x.sum()
    if S < 1e-12:
        return np.inf
    conv = _full_convolution(x, L)
    M = conv.max()
    return 2.0 * n * M / (S * S)


# ------------------------------------------------------------
# Simple perturbations ------------------------------------------------
def make_block_perturbation(base, rng):
    """Random multiplicative block perturbation (O(n) memory)."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def transfer_between_reverse_pair(base, rng):
    """Move a random fraction of mass from an index to its reverse."""
    cand = base.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    r = n - 1 - i
    if i == r:
        return cand
    amt = rng.random() * cand[i] * 0.5          # up to 50 % of the source
    cand[i] -= amt
    cand[r] += amt
    return np.clip(cand, 0.0, 1000.0)


# ------------------------------------------------------------
# Gradient step for the objective  J = 2 n * max(conv) / sum^2
# ------------------------------------------------------------
def _gradient_step(x, n, L, rng):
    """One gradient‑descent advancement (with tiny stochastic noise)."""
    # current convolution and its maximiser
    conv = _full_convolution(x, L)
    M = float(conv.max())
    k = int(conv.argmax())
    S = float(x.sum())

    # gradient of the maximal convolution term
    g = np.zeros_like(x)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k - i_idx
        g[i_idx] = 2.0 * x[j_idx]                 # d(conv_k)/d x_i = 2 * x_{k-i}

    # gradient of J (see analysis)
    factor = 2.0 * n / (S * S)
    grad = factor * (g - (2.0 * M / S) * np.ones_like(x))

    # step with modest learning rate and tiny random jitter
    eta = 0.5
    noise = rng.normal(scale=0.01, size=n)
    nxt = x - eta * grad + noise
    nxt = np.clip(nxt, 0.0, 1000.0)
    return nxt

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed=42, num_steps=800, **kwargs):
    """Search for a step‑function with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------
    # Initialise from the provided parent (if any), otherwise a simple base
    # --------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, np.ndarray) and parent.ndim == 1 \
            and parent.size > 0 and np.isfinite(parent).all():
        a = np.clip(parent.astype(np.float64), 0.0, 1000.0)
    else:
        a = np.ones(256, dtype=np.float64)

    n = a.size
    # break perfect symmetry so the gradient is not zero at start
    if np.allclose(a, a[0]):
        a += rng.normal(scale=0.01, size=n)
        a = np.clip(a, 0.0, 1000.0)

    # FFT length (next power‑of‑two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------------
    # Keep the best candidate seen so far
    # --------------------------------------------------------------------
    best = a.copy()
    best_score = float(_score(best, n, L))

    # --------------------------------------------------------------------
    # Main stochastic optimisation loop
    # --------------------------------------------------------------------
    for _ in range(num_steps):
        # 1. gradient‑based move
        cand_grad = _gradient_step(a, n, L, rng)
        score_grad = _score(cand_grad, n, L)

        # 2. random block perturbation (as in the reference program)
        cand_block = make_block_perturbation(a, rng)
        score_block = _score(cand_block, n, L)

        # 3. transfer mass between a position and its reverse
        cand_trans = transfer_between_reverse_pair(a, rng)
        score_trans = _score(cand_trans, n, L)

        # 4. current candidate (already evaluated via previous loop)
        score_curr = _score(a, n, L)

        # Choose the best among the four possibilities
        candidates = [
            (a,          score_curr),
            (cand_grad,  score_grad),
            (cand_block, score_block),
            (cand_trans, score_trans),
        ]
        candidates.sort(key=lambda p: p[1])
        a, cur_score = candidates[0]

        # Record global best
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score

        # Early stop if we are already below the target threshold
        if best_score < 1.5028:
            break

    # --------------------------------------------------------------------
    # Return a plain Python list as required by the harness
    # --------------------------------------------------------------------
    return [float(v) for v in best]
