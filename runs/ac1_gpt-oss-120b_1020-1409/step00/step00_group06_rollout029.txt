def propose_candidate(seed=42, num_steps=5000, **kwargs):
    rng = np.random.default_rng(seed)
    n = height_sequence_1.shape[0] if isinstance(height_sequence_1, np.ndarray) else len(height_sequence_1)
    # Validate parent
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    if parent.ndim !=1 or parent.size ==0 or not np.isfinite(parent).all():
        parent = np.ones(n, dtype=np.float64)
    # List to hold best sequence
    best = parent.copy()
    best_score = evaluate(best)
    # parametric candidates
    # geometric r values
    for r in np.linspace(0.95, 0.9999, num=50):
        cand = generate_geometric(n, r)
        score = evaluate(cand)
        if score < best_score:
            best = cand
            best_score = score
    # reversed geometric
    for r in np.linspace(0.95, 0.9999, num=20):
        cand = generate_geometric(n, r)[::-1]
        score = evaluate(cand)
        if score < best_score:
            best = cand
            best_score = score
    # linear decreasing
    cand = np.linspace(1.0, 0.0, n, dtype=np.float64)
    score = evaluate(cand)
    if score < best_score:
        best = cand
        best_score = score
    # quadratic decreasing
    cand = np.linspace(1.0, 0.0, n, dtype=np.float64) ** 2
    ...
    # random block perturbations refine
    for step in range(num_steps):
        # choose a block
        width = int(rng.integers(1, max(2, n // 50)))
        start = int(rng.integers(0, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.1))

            best = candidate
            best_score = score
    return [float(x) for x in best]

import numpy as np

# The evaluator `evaluate_sequence` is provided by the harness.

def _eval(seq):
    """Return the AC1 upper bound for a NumPy 1‑D array `seq`."""
    # Convert to plain Python list as required by the evaluator.
    score = evaluate_sequence(seq.tolist())
    return float(score)


def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Return a list of non‑negative coefficients that (hopefully) improve the
    AC1 upper bound.  The routine uses a few parametric families (exponential,
    linear, quadratic) and a memory‑bounded stochastic block refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Load the parent construction if it exists; otherwise fall back.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:               # height_sequence_1 not defined
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size

    # ------------------------------------------------------------------
    # 2.  Initialise the current best candidate.
    # ------------------------------------------------------------------
    best = parent.copy()
    best_score = _eval(best)

    # ------------------------------------------------------------------
    # 3.  Scan a few simple parametric families.
    # ------------------------------------------------------------------
    # (a) Exponential decay a[i] = exp(-alpha * i)
    #     Theoretical optimum is around alpha ~= 1.2 / n.
    alphas = np.linspace(0.00005, 0.0025, 30)  # covers the interesting range
    for alpha in alphas:
        cand = np.exp(-alpha * np.arange(n, dtype=np.float64))
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _eval(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc

    # (b) Reversed exponential (large values at the end)
    for alpha in alphas[::5]:   # fewer points for the reversed version
        cand = np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _eval(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc

    # (c) Linear decay from 1 to 0
    cand = np.linspace(1.0, 0.0, n, dtype=np.float64)
    cand = np.clip(cand, 0.0, 1000.0)
    sc = _eval(cand)
    if np.isfinite(sc) and sc < best_score:
        best, best_score = cand, sc

    # (d) Quadratic decay
    cand = (np.linspace(1.0, 0.0, n, dtype=np.float64)) ** 2
    cand = np.clip(cand, 0.0, 1000.0)
    sc = _eval(cand)
    if np.isfinite(sc) and sc < best_score:
        best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 4.  Stochastic block‑wise refinement (memory‑bounded local search).
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # Random block width – at most about 2 % of the length.
        max_width = max(2, n // 50)
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))

        # Multiplicative perturbation sampled from a log‑normal distribution.
        factor = float(np.exp(rng.normal(0.0, 0.07)))

        candidate = best.copy()
        candidate[start:start + width] *= factor

        # Occasionally symmetrise to explore mirrored structures.
        if rng.random() < 0.12:

        candidate = np.clip(candidate, 0.0, 1000.0)

    # ------------------------------------------------------------------
    return [float(x) for x in best]
