import numpy as np
import math

def _greedy_sidon_set(n: int, rng: np.random.Generator, attempts: int = 30):
    """Return a set of indices (list) with all pairwise sums distinct (Sidon set).
    Tries several random orders and keeps the largest found."""
    best_set = []
    max_sum = 2 * n - 1  # possible sums range 0 .. 2n-2
    for _ in range(attempts):
        cur_set = []
        used = np.zeros(max_sum, dtype=bool)          # which sums have appeared
        order = rng.permutation(n)                   # random order of candidates
        for i in order:
            # check the diagonal sum i+i
            if used[i + i]:
                continue
            # check sums with already chosen elements
            conflict = False
            for p in cur_set:
                if used[i + p]:
                    conflict = True
                    break
            if conflict:
                continue
            # accept i
            cur_set.append(i)
            used[i + i] = True
            for p in cur_set[:-1]:
                used[i + p] = True
        if len(cur_set) > len(best_set):
            best_set = cur_set
    return best_set


def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step‑function that beats the 1.5027 bound."""
    rng = np.random.default_rng(seed)

    # Length of the problem (height_sequence_1 is provided at runtime)

    # ---- 1. build a Sidon set (all pairwise sums distinct) -----------------

    large_val = 1000.0                         # maximal allowed height
