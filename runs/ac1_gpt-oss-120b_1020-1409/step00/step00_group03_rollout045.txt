    """Return a nonnegative step-function sequence minimizing AC1 bound.

    Uses a simple constant sequence which is optimal under the nonnegative
    constraint (Cauchy‑Schwarz) and falls back to the supplied parent if it
    yields an equal score.
    """
    # Determine length from height_sequence_1 if available.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = base.size
        parent = np.clip(base, 0.0, 1000.0)
    except Exception:
        # fallback length
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # constant candidate
    const = np.full(n, 1.0, dtype=np.float64)
    const_score = float(evaluate_sequence(const.tolist()))
    parent_score = float(evaluate_sequence(parent.tolist()))
    if const_score <= parent_score:
        best = const
    else:
        best = parent
    return [float(x) for x in best]
