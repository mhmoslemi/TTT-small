```python
def local_search_update(candidate: np.ndarray) -> np.ndarray:
    """Produce one step of local search block-perturbation."""
    n = len(candidate)
    # Pick a region to perturb (width=min(30, n/10) or random)
    width = min(30, n//10) if n//10 else 1
    start = np.random.randint(0, n - width + 1) if n - width >= 1 else 0
    stop = start + width
    # Apply exponential perturbation
    candidate_cpy = candidate.copy()
    candidate_cpy[start:stop] *= np.exp(np.random.normal(0.0, 0.1))
    # Sometimes mirror
    if np.random.random() < 0.3:

    # Clip to [0, 1000] and return
    candidate_cpy = np.clip(candidate_cpy, 0.0, 1000.0)
    return candidate_cpy
```

```python
def compute_grad_finite(
    a: np.ndarray, 
    h: float = 1e-5
) -> np.ndarray:
    """Produce gradient for a via multi-point finite differences."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        # One-sided differentiation at boundaries
        if i == 0 or i == n-1:
            perturb = np.copy(a)
            perturb[i] += h
            score_plus = evaluate_sequence(perturb.tolist())
            score_orig = evaluate_sequence(a.tolist())
            grad[i] = (score_plus - score_orig) / h
        elif i == n//2:  # Special center point
            # Use two-point symmetric derivative
            perturb_plus = np.copy(a)
            perturb_plus[i] += h
            perturb_minus = np.copy(a)
            perturb_minus[i] -= h
            score_plus = evaluate_sequence(perturb_plus.tolist())
            score_minus = evaluate_sequence(perturb_minus.tolist())
            grad[i] = (score_plus - score_minus) / (2 * h)
        else:
            # Central derivative
            perturb_plus = np.copy(a)
            perturb_plus[i] += h
            perturb_minus = np.copy(a)
            perturb_minus[i] -= h
            score_plus = evaluate_sequence(perturb_plus.tolist())
            score_minus = evaluate_sequence(perturb_minus.tolist())
            grad[i] = (score_plus - score_minus) / (2 * h)
    return grad
```
