import numpy as np

# ------------------------------------------------------------
# Helper: generate a set of indices whose pairwise sums appear
# at most C times (C = 2 by default).  This yields a sequence
# with a very small convolution peak.
# ------------------------------------------------------------
def generate_low_overlap_positions(n, C=2, max_passes=12, rng=None):
    """Return a list of positions 0 ≤ p < n such that each unordered
    sum p+q occurs at most C times (including the self‑sum p+p).

    The algorithm is greedy: it repeatedly scans the indices in a
    random order and inserts a position whenever the C‑collision
    constraint would not be violated.
    """
    if rng is None:
        rng = np.random.default_rng()
    positions = []                 # accepted indices
    pos_set = set()                # for fast membership test
    sum_counts = {}                # mapping sum → number of pairs giving this sum

    for _ in range(max_passes):
        # random ordering of all possible indices
        order = rng.permutation(n)
        added = False
        for p in order:
            if p in pos_set:
                continue

            # check the new self‑sum
            s_self = p + p
            if sum_counts.get(s_self, 0) + 1 > C:
                continue

            # check sums with already chosen positions
            feasible = True
            for q in positions:
                s = p + q
                if sum_counts.get(s, 0) + 1 > C:
                    feasible = False
                    break
            if not feasible:
                continue

            # all checks passed – commit the new position
            sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
            for q in positions:
                s = p + q
                sum_counts[s] = sum_counts.get(s, 0) + 1
            positions.append(p)
            pos_set.add(p)
            added = True
        if not added:          # a full pass added nothing → we are saturated
            break
    return positions
