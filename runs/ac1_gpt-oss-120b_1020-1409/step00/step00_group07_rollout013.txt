    cur = best_seq.copy()
    cur_score = _approx_score(cur)
    max_swaps = 100                       # modest budget, stays O(n log n)

    for _ in range(max_swaps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # try swapping positions i and j
        cur[i], cur[j] = cur[j], cur[i]
        new_score = _approx_score(cur)
        if new_score < cur_score:
            cur_score = new_score          # keep the improvement
        else:
            cur[i], cur[j] = cur[j], cur[i]   # revert

    # --------------------------------------------------------------
    # Final exact check between the local optimum and the original best.
    # --------------------------------------------------------------
    final_score = evaluate_sequence(cur.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_seq = cur
        best_score = final_score

    # --------------------------------------------------------------
    # Return a clean Python list respecting the evaluator's bounds.
    # --------------------------------------------------------------
    result = np.clip(best_seq, 0.0, 1000.0).tolist()
    if not result:               # safety net – should never happen
        result = [0.0]
    return result
