import numpy as np

# ----------------------------------------------------------------------
# Helper: construct a Sidon (B2) set of indices – sums of any two are distinct.
# Greedy algorithm, optionally shuffled order for a larger set.
# ----------------------------------------------------------------------
def sidon_positions(n, rng):
    """Return a list of positions forming a Sidon set (distinct pairwise sums)."""
    # Random order helps to reach a larger set size.
    order = np.arange(n, dtype=np.int64)
    rng.shuffle(order)

    positions = []
    used_sums = set()          # sums that already appear (i+j)

    for i in order:
        # quick checks: self‑sum and sums with existing positions must be new.
        if (2 * i) in used_sums:
            continue
        conflict = False
        for p in positions:
            if (i + p) in used_sums:
                conflict = True
                break
        if conflict:
            continue

        # Accept i: record all new sums.
        for p in positions:
            used_sums.add(i + p)
        used_sums.add(2 * i)
        positions.append(int(i))

    return positions
