def propose_candidate(seed=42, num_iters=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, np.ndarray):
        n = height_sequence_1.size
    else:
        n = 2000
    # Initialize sequence: decreasing 1/sqrt(i+1)
    a = 1.0 / np.sqrt(np.arange(1, n+1, dtype=np.float64))
    a = a / np.sum(a)  # normalize to sum 1
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # iterative optimization
    sum_target = 1.0
    step0 = 0.2
    for iteration in range(num_iters):
        # compute convolution
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))
        # gradient for that shift
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # indices on a that affect conv[k]
        idx = np.arange(low, high+1, dtype=int)
        # compute gradient vector
        g = np.zeros_like(a)
        # corresponding partner indices
        partner = k - idx
        # values for partner are in [0, n-1], valid.
        g[idx] = 2.0 * a[partner]
        # multiplicative update
        step = step0 * (0.99 ** iteration)
        a = a * np.exp(-step * g)
        # renormalize sum to target
        a = a / np.sum(a) * sum_target
        # clip bounds
        np.clip(a, 0.0, 1000.0, out=a)
        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
        # optional random perturbation occasionally to escape
        if iteration % 100 == 0 and rng.random() < 0.1:
            # random block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = np.exp(rng.normal(0.0, 0.12))
            a_block = a.copy()
            a_block[start:stop] *= factor
            a_block = np.clip(a_block, 0.0, 1000.0)
            a_block = a_block / np.sum(a_block) * sum_target
            # evaluate
            score_block = float(evaluate_sequence(a_block.tolist()))
            if np.isfinite(score_block) and score_block < best_score:
                best = a_block.copy()
                best_score = score_block
                a = a_block
    return [float(x) for x in best]
