import numpy as np

def compute_score(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())   # next power of 2
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a == 0:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

# ---------- Helper functions (top‑level, no closures) ----------

def _conv_full(a, fft_len):
    """Linear full convolution a * a using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # result length = 2*len(a)-1, keep only the valid part
    return conv[:2 * a.size - 1]


def _score(a, n, fft_len):
    """Exact AC1 score (fast version) for a non‑negative vector a."""
    s = a.sum()
    if s == 0.0:
        return np.inf
    b = _conv_full(a, fft_len)
    max_b = b.max()
    return 2.0 * n * max_b / (s * s)


def _make_block_perturbation(base, rng):
    """Return a modest O(n)‑memory perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:                     # optional mirror‑symmetry
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)
