import numpy as np

def _get_parent():
    """
    Return a valid one‑dimensional NumPy array to use as the parent sequence.
    If `height_sequence_1` is not available or invalid, fall back to an array
    of ones of length 256.
    """
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
            raise ValueError
        # Clip to the evaluator's admissible range.
        arr = np.clip(arr, 0.0, 1000.0)
        return arr
    except Exception:
        # Safe default length.
        return np.ones(256, dtype=np.float64)


def _uniform_candidate(base):
    """
    Build a uniform (constant) candidate whose total sum equals that of `base`.
    Guarantees a total sum of at least 0.01 to avoid the evaluator's Inf.
    """
    n = base.size
    total = float(base.sum())
    if total < 0.01:
        # Minimal admissible total sum.
        total = 0.01
    val = total / n
    # Respect the 0–1000 bound.
    val = min(max(val, 0.0), 1000.0)
    return np.full(n, val, dtype=np.float64)


def _random_block(candidate, rng):
    """
    Produce a small random block perturbation of `candidate`.
    The perturbation respects the O(n) memory budget.
    """
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    if start < stop:
        factor = float(np.exp(rng.normal(0.0, 0.05)))  # modest change
        candidate[start:stop] *= factor
    if rng.random() < 0.25:                     # occasional symmetrisation
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that (approximately)
    minimise the AC1 bound as evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------------
    # 1. Initialise from the supplied parent or a safe fallback.
    # -------------------------------------------------------------------
    parent = _get_parent()
    n = parent.size

    # -------------------------------------------------------------------
    # 2. Construct a uniform candidate – this is optimal for the bound.
    # -------------------------------------------------------------------
    uniform = _uniform_candidate(parent)

    # -------------------------------------------------------------------
    # 3. Evaluate the two obvious candidates.
    # -------------------------------------------------------------------
    best_seq = parent
    best_score = float(evaluate_sequence(parent.tolist()))
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if np.isfinite(uniform_score) and uniform_score < best_score:
        best_seq = uniform
        best_score = uniform_score

    # -------------------------------------------------------------------
    # 4. Light random search around the current best (optional,
    #    may discover tiny floating‑point improvements).
    # -------------------------------------------------------------------
    max_steps = 200
    for _ in range(max_steps):
        cand = best_seq.copy()
        cand = _random_block(cand, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # -------------------------------------------------------------------
    # 5. Return the best sequence found as a plain Python list.
    # -------------------------------------------------------------------
    return [float(x) for x in best_seq]
