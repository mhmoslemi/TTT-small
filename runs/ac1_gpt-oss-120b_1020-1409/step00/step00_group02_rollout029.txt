def _generate_sparse_set(n, target, max_mult, rng, tries=200):
    """
    Build a set S of indices (0 ≤ i < n) of size at least `target`
    such that each unordered sum i+j occurs at most `max_mult` times.
    Returns the largest set found (may be smaller than target).
    """

        sum_cnt = np.zeros(2 * n - 1, dtype=np.int8)

        # random order each attempt
        order = rng.permutation(n)

    best_score = float(evaluate_sequence(best.tolist()))

    for _ in range(300):
        # perturb a random block multiplicatively
        cand = best.copy()
        width = max(1, n // 200)
        start = rng.integers(0, n - width + 1)
        cand[start:start + width] *= np.exp(rng.normal(0.0, 0.08))
        cand = np.clip(cand, 0.0, 1000.0)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    return [float(x) for x in best]
