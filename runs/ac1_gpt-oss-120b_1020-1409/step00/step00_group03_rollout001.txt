def _linear_ramp(n):

    linearly from ≈0 to 1000.

    ramp = (np.arange(1, n + 1, dtype=np.float64) * 1000.0) / n
    # Safety clipping (should already be within bounds).
    return np.clip(ramp, 0.0, 1000.0)
