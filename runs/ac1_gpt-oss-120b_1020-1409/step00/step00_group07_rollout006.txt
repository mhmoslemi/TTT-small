def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length
    try:
        base = height_sequence_1
        if not isinstance(base, np.ndarray) or base.ndim != 1 or base.size == 0:
            raise Exception
    except Exception:
        # fallback to default length 256
        n = 256
    else:
        n = int(base.shape[0])
    # create uniform baseline
    best = np.full(n, 1.0, dtype=np.float64)
    best = np.clip(best, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))
    # optionally perform simple random perturbation attempts to maybe lower score (unlikely)
    # We'll try a few random block perturbations again but accept only if lower.
    num_trials = 200   # modest, keep O(n) memory
    for _ in range(num_trials):
        # create copy
        cand = best.copy()
        # pick random block width
        width = rng.integers(1, max(2, n // 20))
        start = rng.integers(0, max(1, n - width + 1))
        stop = start + width
        # scale block by random factor lognormal
        factor = np.exp(rng.normal(0.0, 0.05))
        cand[start:stop] *= factor
        # with probability, symmetrize
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        # clip
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    The method builds a uniform baseline of the appropriate length and then
    performs a modest number of cheap random perturbations, keeping any
    improvement.
    """
    rng = np.random.default_rng(seed)

    # Determine the length from the provided parent sequence if available.
    base = globals().get("height_sequence_1")
    if isinstance(base, np.ndarray) and base.ndim == 1 and base.size > 0:
        n = int(base.shape[0])
    else:
        n = 256  # safe fallback length

    # Uniform baseline – this is the theoretical optimum for the given metric.
    best = np.full(n, 1.0, dtype=np.float64)
    np.clip(best, 0.0, 1000.0, out=best)

    # Evaluate the baseline.
    best_score = float(evaluate_sequence(best.tolist()))

    # A small random local‑search loop (O(n) memory per iteration).
    # The number of trials is modest to stay well within resource limits.
    n_trials = 200
    for _ in range(n_trials):
        cand = best.copy()

        # Random block width and start, bounded away from zero length.
        width = int(rng.integers(1, max(2, n // 20)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        # Scale the chosen block by a log‑normal factor (small perturbation).
        factor = float(np.exp(rng.normal(0.0, 0.05)))
        cand[start:stop] *= factor

        # Occasionally symmetrize to keep the shape balanced.
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])

        # Enforce the problem’s bounds.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Exact evaluation of the candidate.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list of floats.
    return [float(x) for x in best]
