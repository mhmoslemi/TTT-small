import numpy as np

def propose_candidate(seed=42, **kwargs):
    """Construct a sparse non‑negative sequence whose autocorrelation peak
    (the max of the linear convolution with itself) is kept low.

    The construction builds a set of indices whose pairwise sums appear only
    a few times (parameter L_max).  With m such indices the bound evaluated as

        2 * n * max(b) / (sum(a)**2)  ≤  2 * n * L_max / m**2

    can be driven below the target 1.50274349 by choosing m large enough.
    """
    rng = np.random.default_rng(seed)

    # Length of the parent sequence (provided by the harness as height_sequence_1)
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256  # fallback length

    max_val = 1000.0                     # allowed upper bound for each entry
    target_score = 1.50274349            # required upper bound
    L_max = 4                            # maximal ordered pair count per sum

    # Minimum number of non‑zero entries needed to beat the target bound
    required_m = int(np.ceil(np.sqrt(2 * n * L_max / target_score))) + 5
    required_m = min(required_m, n)      # cannot exceed the length

    # Try a few random shuffles; if we fail we relax L_max a little.
    for attempt in range(20):
        marks = []                       # selected positions
        sum_counts = np.zeros(2 * n - 1, dtype=np.int16)  # ordered‑pair counts per sum
        order = rng.permutation(n)       # random candidate order

        for pos in order:
            pos = int(pos)
            self_idx = pos + pos

            # Check self‑pair (i,i)
            if sum_counts[self_idx] + 1 > L_max:
                continue

            if marks:
                prev = np.array(marks, dtype=np.int32)
                s = pos + prev                         # sums with existing marks
                # Each unordered pair contributes two ordered pairs.
                if np.any(sum_counts[s] + 2 > L_max):
                    continue

                # Accept the position: update counts for unordered pairs (2 each)
                sum_counts[s] += 2

            # Update the self‑pair count and record the new position
            sum_counts[self_idx] += 1
            marks.append(pos)

            if len(marks) >= required_m:
                break

        if len(marks) >= required_m:
            break                     # success
        # If not enough points, allow a larger L_max and retry
        L_max += 1

    # Build the final sequence: 1.0 at the selected indices, 0 elsewhere.
    seq = np.zeros(n, dtype=np.float64)
    if marks:
        seq[marks] = 1.0
    seq = np.clip(seq, 0.0, max_val)  # enforce the evaluator's limits
    return seq.tolist()
