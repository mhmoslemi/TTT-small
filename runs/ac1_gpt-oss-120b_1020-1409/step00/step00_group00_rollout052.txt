def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    parent = np.array(height_sequence_1, dtype=np.float64) if 'height_sequence_1' in globals() else None
    if parent is None or not isinstance(parent, np.ndarray):
        parent = np.ones(256, dtype=np.float64)
    # clip
    parent = np.clip(parent, 0.0, 1000.0)

    current = parent.copy()
    current_score = float(evaluate_sequence(current.tolist()))
    best = current.copy()
    best_score = current_score

    temp0 = 10.0
    temp = temp0

    for step in range(num_steps):
        # copy
        candidate = current.copy()
        # apply random move
        move_type = rng.integers(0, 4)  # 0: block scale, 1: mass transfer, 2: smoothing, 3: random shift (circular)
        if move_type == 0:
            # block scaling
            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))
            scale = np.exp(rng.normal(0.0, 0.1))  # small scaling
            candidate[start:start+width] *= scale
        elif move_type == 1:
            # mass transfer between two indices
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amount = candidate[i] * rng.random() * 0.5  # move up to half of i's mass
                candidate[i] -= amount
                candidate[j] += amount
        elif move_type == 2:
            # smoothing via averaging neighbor window
            win = rng.integers(3, 9)  # odd window size
            # apply convolution with uniform kernel and mix
            kernel = np.ones(win) / win
            smoothed = np.convolve(candidate, kernel, mode='same')
            # mix with original
            alpha = rng.random() * 0.5
            candidate = (1 - alpha) * candidate + alpha * smoothed
        elif move_type == 3:
            # rotate
            shift = rng.integers(1, n)
            candidate = np.roll(candidate, shift)

        candidate = np.clip(candidate, 0.0, 1000.0)
        # evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if not np.isfinite(score):
            continue

        # acceptance
        if score < best_score:
            best = candidate.copy()
            best_score = score
            current = candidate.copy()
            current_score = score
        else:
            # maybe accept based on temperature
            accept_prob = np.exp(-(score - current_score) / max(temp, 1e-8))
            if rng.random() < accept_prob:
                current = candidate.copy()
                current_score = score
        # schedule temperature drop
        temp *= 0.9995

    return [float(x) for x in best]
