import numpy as np

def random_binary_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a binary (0/1) vector of length *n* with about half ones."""

    idx = rng.permutation(n)[:k]      # random distinct positions
    vec[idx] = 1.0
    return vec
