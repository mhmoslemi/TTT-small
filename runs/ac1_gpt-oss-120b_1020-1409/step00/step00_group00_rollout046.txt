import numpy as np

# The evaluator evaluate_sequence is already available in the execution environment.

def _initial_length():
    """
    Determine the length of the parent sequence if it is available and valid.
    Falls back to 256 if not.
    """
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            return arr.size
    except Exception:
        pass
    return 256


def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Return a list of non‑negative step heights that (almost) minimizes the
    AC1 upper bound 2 * n * max(f*f) / (sum f)^2.
    The algorithm starts from a uniform sequence (which already attains the
    theoretical lower bound 2) and performs a lightweight simulated‑annealing
    perturbation loop; the uniform sequence is kept as a safe fallback.
    """
    rng = np.random.default_rng(seed)

    n = _initial_length()

    # Uniform start: this gives the exact lower bound 2 for the objective.
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # Current working point (starts equal to the uniform sequence).
    cur = best.copy()
    cur_score = best_score

    # Simple exponential cooling schedule.
    temperature = 1.0

    for _ in range(num_steps):
        # ---- propose a local perturbation ---------------------------------
        # Choose a random block width between n/200 and n/12 (inclusive).
        min_w = max(1, n // 200)
        max_w = max(2, n // 12)
        width = int(rng.integers(min_w, max_w + 1))

        # Random start position for the block.
        start = int(rng.integers(0, max(1, n - width + 1)))

        # Multiplicative scaling factor (log‑normal perturbation).
        factor = float(np.exp(rng.normal(0.0, 0.05)))

        cand = cur.copy()
        cand[start:start + width] *= factor

        # Occasionally symmetrize the candidate (helps to avoid pathological spikes).
        if rng.random() < 0.1:
            cand = 0.5 * (cand + cand[::-1])

        # Enforce the hard bounds required by the evaluator.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # ---- evaluate ------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue

        # ---- acceptance (simulated annealing) -----------------------------
        if score < cur_score or rng.random() < np.exp(-(score - cur_score) / max(temperature, 1e-12)):
            cur = cand
            cur_score = score
            # Keep the best overall solution seen so far.
            if score < best_score:
                best = cand
                best_score = score

        # ---- temperature decay ---------------------------------------------
        temperature *= 0.9995

    # Return the best candidate found (fallback is the uniform sequence).
    return [float(v) for v in best]
