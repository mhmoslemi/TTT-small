import numpy as np

def _next_pow2(x: int) -> int:
    """Small helper to get the next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return the maximum value of the self‑convolution a*a using FFT."""
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)
    # only the first 2*n-1 entries belong to the full convolution
    n = a.shape[0]
    return float(conv[: 2 * n - 1].max())


def propose_candidate(seed: int = 42, num_iters: int = 3000, **kwargs):
    """
    Produce a non‑negative step‑function height sequence whose AC1 bound
    (as evaluated by `evaluate_sequence`) is as low as can be found by a
    simple greedy local search.

    The algorithm starts from the uniform distribution (which gives an
    AC1 value of exactly 2) and repeatedly moves a small amount of mass from
    the pair of indices that currently contribute the most to the maximal
    convolution entry.  Moves are accepted only when they lower the maximal
    convolution value; a small random jitter is added every few hundred
    iterations to avoid stagnation.
    """
    rng = np.random.default_rng(seed)

    # Determine length n – use the provided parent construction if present.
    parent = globals().get("height_sequence_1", None)
    if isinstance(parent, np.ndarray) and parent.ndim == 1 and parent.size > 0:
        n = parent.size
    else:
        n = 256

    # Start from the uniform vector (sum = 1, values = 1/n).
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # Pre‑allocate FFT length (next power of two for speed).
    fft_len = _next_pow2(2 * n - 1)

    # Keep the best solution seen so far.
    best_a = a.copy()
    best_val = _conv_max(best_a, fft_len)

    # Main greedy loop.
    for it in range(num_iters):
        # Current convolution and index of its maximum.
        af = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(af * af, n=fft_len)
        conv = conv[: 2 * n - 1]
        k_max = int(np.argmax(conv))
        cur_max = float(conv[k_max])

        # Identify the pair (i, j) with the largest product a[i] * a[j]
        # among all i + j == k_max.
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k_max - idx_i
        products = a[idx_i] * a[idx_j]
        pid = int(np.argmax(products))
        i = int(idx_i[pid])
        j = int(idx_j[pid])

        # Determine a small amount of mass to move.
        small_frac = 0.15  # fraction of the smaller of the two entries.
        delta = min(a[i], a[j]) * small_frac
        if delta <= 0.0:
            # Nothing to move – apply a tiny random jitter instead.
            jitter = rng.normal(0.0, 1e-4, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a /= a.sum()
            continue

        # Choose two distinct target positions that are not i or j.
        r1 = rng.integers(n)
        while r1 == i or r1 == j:
            r1 = rng.integers(n)
        r2 = rng.integers(n)
        while r2 == i or r2 == j or r2 == r1:
            r2 = rng.integers(n)

        # Perform the transfer.
        a[i] -= delta
        a[j] -= delta
        a[r1] += delta
        a[r2] += delta

        # Evaluate the new maximum after the move.
        new_max = _conv_max(a, fft_len)

        if new_max < cur_max:
            # Accept the move.
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
        else:
            # Reject – revert the change.
            a[i] += delta
            a[j] += delta
            a[r1] -= delta
            a[r2] -= delta

        # Occasionally add a mild random perturbation to escape plateaus.
        if (it + 1) % 250 == 0:
            jitter = rng.normal(0.0, 5e-4, size=n)
            a = np.clip(a + jitter, 0.0, None)
            # Renormalise to keep the sum exactly 1 (scale‑invariant for the score).
            total = a.sum()
            if total > 0.0:
                a /= total
            else:
                a = np.full(n, 1.0 / n, dtype=np.float64)

    # Final scaling (optional) – keep values inside the allowed range.
    out = np.clip(best_a, 0.0, 1000.0)

    return [float(v) for v in out]
