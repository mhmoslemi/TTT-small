import numpy as np

def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a non‑negative step‑function sequence that minimizes the AC1
    upper bound evaluated by ``evaluate_sequence``.

    The construction uses a simple geometric decay a[i] = r**i with
    r = 1 - k/n for a few small integer values k.  The best candidate among
    these (plus the supplied parent) is returned as a plain Python list.
    """
    # ----------------------------------------------------------------------
    # Determine the length n and a parent fallback sequence.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.shape[0])
    parent = np.clip(parent, 0.0, 1000.0)

    # ----------------------------------------------------------------------
    # Evaluate a candidate and keep the best one seen so far.
    # ----------------------------------------------------------------------
    best_seq = parent
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ----------------------------------------------------------------------
    # Try geometric sequences a[i] = (1 - k/n) ** i for a few small k.
    # These have a provably low AC1 value; the loop keeps the best.
    # ----------------------------------------------------------------------
    max_k = min(15, n // 10)          # keep work linear in n
    for k in range(1, max_k + 1):
        eps = float(k) / n
        r = 1.0 - eps
        if r <= 0.0:
            continue                    # safety, should not happen for our range

        # a[i] = r**i  (vectorised, O(n) memory)
        seq = np.power(r, np.arange(n, dtype=np.float64))
        seq = np.clip(seq, 0.0, 1000.0)

        # Exact evaluation (full convolution) – may be a bit pricey but
        # n is only a few thousand.
        try:
            score = float(evaluate_sequence(seq.tolist()))
        except Exception:
            continue

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq

    # ----------------------------------------------------------------------
    # Return the best sequence found as a plain list of floats.
    # ----------------------------------------------------------------------
    return [float(x) for x in best_seq]
