    try:
        parent_score = float(evaluate_sequence(parent.tolist()))
        uniform_score = float(evaluate_sequence(uniform.tolist()))
        best = uniform if uniform_score <= parent_score else parent
    except Exception:
        # If evaluation fails for any reason, fall back to the uniform candidate.
        best = uniform

    # Return a plain Python list of floats as required by the harness.
    return [float(x) for x in best]
