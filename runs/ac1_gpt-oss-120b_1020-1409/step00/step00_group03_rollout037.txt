def propose_candidate(seed=42, num_steps=4000, **kwargs):
    rng = np.random.default_rng(seed)
    base = np.asarray(height_sequence_1, dtype=np.float64)
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        n = 2000
        base = np.ones(n, dtype=np.float64)
    n = base.size

    # initial candidate: sorted descending
    cand = np.sort(base)[::-1]
    cand = np.clip(cand, 0.0, 1000.0)
    # normalize sum to 1 to avoid scaling effect (optional)
    if cand.sum() == 0:
        cand = np.ones_like(cand) / n
    else:
        cand = cand / cand.sum()
    best = cand.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # also try uniform as fallback
    uniform = np.full(n, 1.0 / n)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform.copy()
        best_score = uniform_score

    # local search loop
    for step in range(num_steps):
        # decide move type
        if rng.random() < 0.7:
            # gradient-like move based on max convolution index
            conv = np.convolve(best, best)
            idx_max = int(conv.argmax())
            start = max(0, idx_max - (n - 1))
            end = min(n - 1, idx_max)
            # gradient contributions
            g = np.zeros_like(best)
            i_idx = np.arange(start, end + 1)
            j_idx = idx_max - i_idx
            g[i_idx] = best[j_idx]
            # step size schedule
            lr = 0.1 * (0.99 ** step)  # decaying
            cand2 = best - lr * g
            cand2 = np.clip(cand2, 0.0, None)
            s = cand2.sum()
            if s > 0:
                cand2 = cand2 * (best.sum() / s)
        else:
            # random block perturbation
            cand2 = best.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start2 = int(rng.integers(0, max(1, n - width + 1)))
            stop2 = min(n, start2 + width)
            cand2[start2:stop2] *= float(np.exp(rng.normal(0.0, 0.12)))
            if rng.random() < 0.35:
                cand2 = 0.5 * (cand2 + cand2[::-1])
            # re-normalize sum to keep scale
            s2 = cand2.sum()
            if s2 > 0:
                cand2 = cand2 * (best.sum() / s2)

        cand2 = np.clip(cand2, 0.0, 1000.0)
        score2 = float(evaluate_sequence(cand2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best = cand2
            best_score = score2

        # occasional global perturb (reset to sorted descending of parent)
        if step % 500 == 0 and rng.random() < 0.2:
            redo = np.sort(base)[::-1]
            redo = np.clip(redo, 0.0, 1000.0)
            sredo = redo.sum()
            if sredo > 0:
                redo = redo / sredo  # normalize
            score_redo = float(evaluate_sequence(redo.tolist()))
            if score_redo < best_score:
                best = redo
                best_score = score_redo

    return [float(x) for x in best]

def reduce_max_move(p, idx_max, rng):
   # identify best contributing pair
   start = max(0, idx_max - (n - 1))
   end = min(n - 1, idx_max)
   i_range = np.arange(start, end+1)
   j_range = idx_max - i_range
   contrib = p[i_range] * p[j_range]
   max_idx = int(np.argmax(contrib))
   i = int(i_range[max_idx])
   j = int(j_range[max_idx])
   # choose target index t (not i or j)
   t = int(rng.integers(0, n))
   while t == i or t == j:
       t = int(rng.integers(0, n))
   # delta = fractional mass from i and j
   delta = min(p[i], p[j]) * 0.2
   if delta <= 0:
       return p  # no move
   # reproducible new p
   newp = p.copy()
   newp[i] -= delta
   newp[j] -= delta
   newp[t] += 2 * delta
   # clip for negative due to floating errors
   newp = np.clip(newp, 0.0, None)
   # renorm
   newp_sum = newp.sum()
   if newp_sum > 0:
       newp = newp / newp_sum
   return newp
```

import numpy as np

# -------------------------------------------------------------
# Helper: small multiplicative block perturbation (same as before)
# -------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a non‑negative O(n) perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # pick a block width proportional to n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

    # -------------------------------------------------

    best_score = float(evaluate_sequence(best.tolist()))
