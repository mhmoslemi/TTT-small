    # scale init to have sum, keep as is.
    seq = init.copy()
    n = seq.size
    sum0 = seq.sum()
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(seq.tolist()))
    step_size = 0.1 * sum0 / n  # initial step-size relative to average value.
    for itr in range(num_iter):
        conv = np.convolve(seq, seq)
        k_star = int(conv.argmax())
        # compute gradient vector g
        i_min = max(0, k_star - (n-1))
        i_max = min(n-1, k_star)
        i_range = np.arange(i_min, i_max+1)
        j_range = k_star - i_range
        g = np.zeros_like(seq)
        g[i_range] = 2.0 * seq[j_range]  # product contributions
        # Since each pair counted twice, we already have factor 2; ok.
        # compute tentative update
        new_seq = seq - step_size * g
        # ensure nonnegativity
        new_seq = np.maximum(new_seq, 0.0)
        # renormalize sum to sum0
        cur_sum = new_seq.sum()
        if cur_sum > 0:
            new_seq *= (sum0 / cur_sum)
        else:
            # fallback to uniform
            new_seq = np.full(n, sum0 / n)
        # clip to max 1000
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        new_score = float(evaluate_sequence(new_seq.tolist()))
        if np.isfinite(new_score) and new_score < best_score:
            best_seq = new_seq
            best_score = new_score
            seq = new_seq
            # maybe increase step size gradually
            step_size *= 1.05
        else:
            # reduce step size
            step_size *= 0.5
            # keep old seq unchanged
        # optional break if step_size becomes too small
        if step_size < 1e-12:
            break
    return [float(x) for x in best_seq]
