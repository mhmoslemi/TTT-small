import numpy as np

def propose_candidate(seed=42, num_steps=4000, **kwargs):
    """
    Hill‑climbing search for a sparse binary sequence whose
    self‑convolution peak is small relative to the squared sum.
    Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)

    # Length of the runtime vector, fall back to a modest size if unavailable.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    # Target for early termination (the benchmark you must beat).
    target_score = 1.50274349

    # Number of ones (active entries). 7 % of n works well for n≈7k.
    k = max(3, int(0.07 * n))

    # -----------------------------------------------------------------
    # Initialise a random binary vector with exactly k ones.
    # -----------------------------------------------------------------
    candidate = np.zeros(n, dtype=np.float64)
    init_idx = rng.choice(n, size=k, replace=False)
    candidate[init_idx] = 1.0

    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))

    # If the initial vector is rejected for any reason, fall back to a
    # uniform positive vector (this is always valid but has score = 2).
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64) * (np.sqrt(2.0 * n) / n)
        best_score = float(evaluate_sequence(best.tolist()))

    # -----------------------------------------------------------------
    # Simple hill‑climbing: repeatedly swap a 1 with a 0.
    # Accept only score‑improving moves.
    # -----------------------------------------------------------------
    for _ in range(num_steps):
        cur = best.copy()

        # Current positions of ones and zeros.
        ones = np.flatnonzero(cur)
        zeros = np.where(cur == 0)[0]

        if ones.size == 0 or zeros.size == 0:

            cur[:] = 0.0
            cur[rng.choice(n, size=k, replace=False)] = 1.0
        else:

            rm = rng.choice(ones)
            add = rng.choice(zeros)
            cur[rm] = 0.0
            cur[add] = 1.0

        score = float(evaluate_sequence(cur.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cur
            best_score = score
