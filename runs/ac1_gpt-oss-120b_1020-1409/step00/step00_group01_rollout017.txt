import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims at a low AC1
    upper bound.  The evaluator `evaluate_sequence` is used to compare a few
    simple constructions; the best (lowest) score is returned.
    """
    rng = np.random.default_rng(seed)

    # Determine a base length.  Use the provided parent sequence if available.
    try:
        base = height_sequence_1   # type: ignore[name-defined]
        if isinstance(base, np.ndarray) and base.ndim == 1 and base.size > 0:
            n = int(base.size)
        else:
            raise Exception
    except Exception:
        n = 256

    # Helper to evaluate a candidate array.
    def score_of(arr):
        s = evaluate_sequence(arr.tolist())
        return float(s) if np.isfinite(s) else np.inf

    # Small collection of simple candidates.
    candidates = []

    # Uniform candidates with a few different amplitudes.
    for val in (1.0, 0.5, 2.0, 10.0, 100.0):
        candidates.append(np.full(n, val, dtype=np.float64))

    # Random uniform values in the admissible range.
    for _ in range(10):
        cand = rng.random(n) * 1000.0   # values in [0,1000)
        # assure the sum is not too tiny
        if cand.sum() < 0.01:
            cand[0] = 0.01
        candidates.append(cand)

    # Evaluate all candidates, keep the best.
    best_seq = None
    best_sc = np.inf
    for cand in candidates:
        sc = score_of(cand)
        if sc < best_sc:
            best_sc = sc
            best_seq = cand

    # As a safety net, fall back to a guaranteed feasible uniform vector.
    if best_seq is None:
        best_seq = np.full(n, 1.0, dtype=np.float64)

    # Return a plain Python list of floats (required by the harness).
    return [float(x) for x in best_seq]
