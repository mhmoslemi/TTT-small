import numpy as np


def propose_candidate(seed=42, num_steps=500, **kwargs):
    """Search for a step‑function that minimizes the AC1 bound.

    The search starts from the inverse‑square‑root shape a_i = 1/sqrt(i+1),
    which is known to give a low value of the objective.
    A small local stochastic optimisation (block scaling, swaps and
    optional symmetrisation) is performed and the best candidate seen is
    returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the working length.
    # ------------------------------------------------------------------
    try:
        # Runtime provides the parent construction as a NumPy array.
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
    except Exception:          # fall back when no parent is present
        n = 256
        parent = None

    # ------------------------------------------------------------------
    # Initialise with the inverse‑square‑root profile.
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(idx + 1.0)               # a_i = 1/sqrt(i+1)
    a = np.clip(a, 0.0, 1000.0)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Local stochastic optimisation.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = best.copy()

        # 70 % of the time perform a multiplicative block perturbation,
        # 30 % of the time swap two random entries.
        if rng.random() < 0.7:
            # block width between n/200 and n/20 (inclusive)
            w_min = max(1, n // 200)
            w_max = max(2, n // 20)
            width = int(rng.integers(w_min, w_max + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = float(np.exp(rng.normal(0.0, 0.07)))  # log‑normal factor
            cand[start:stop] *= factor
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            cand[i], cand[j] = cand[j], cand[i]

        # Occasionally average with the reversed version to encourage symmetry.
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
