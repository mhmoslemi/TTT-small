import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a symmetric exponentially decaying step sequence.
    The decay parameter is tuned (by a simple deterministic search) so that
    the AC1 upper bound returned by `evaluate_sequence` falls below the
    target 1.50274349.  The routine keeps the best candidate found and falls
    back to a uniform sequence if something goes wrong.
    """
    # ------------------------------------------------------------------
    # Determine the length n of the problem instance.
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
            raise ValueError
        n = int(arr.size)
    except Exception:
        # fallback length if the runtime array is unavailable or invalid
        n = 256

    # Pre‑compute distances from the nearest edge (used for the symmetric shape)
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)          # distance to the nearer border

    # Target score – we stop as soon as we beat it.
    target_score = 1.50274349

    # ------------------------------------------------------------------
    # Simple deterministic search over the exponential decay constant.
    # The objective J ≈ 0.184 * n * c for small c, so decreasing c lowers the
    # score.  We start from a modest c and shrink it until the bound is met.
    c = 0.003            # initial decay (larger -> higher score)
    best_seq = None
    best_score = float("inf")

    for _ in range(12):                     # at most a handful of evaluations
        seq = np.exp(-c * dist)              # symmetric exponential shape
        seq = np.clip(seq, 0.0, 1000.0)       # enforce evaluator bounds

        # Exact evaluation of the AC1 bound
        score = float(evaluate_sequence(seq.tolist()))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

        if score <= target_score:
            break                           # good enough – stop early

        # Reduce the decay to lower the bound (smaller c → flatter profile)
        c *= 0.6
        if c < 1e-5:                         # avoid absurdly tiny c
            break

    # ------------------------------------------------------------------
    # Safety fallback: if for some reason we have no valid candidate,
    # return a simple positive uniform vector (which yields a bound of 2.0).
    if best_seq is None or not np.isfinite(best_score):
        best_seq = np.ones(n, dtype=np.float64)

    return best_seq.tolist()
