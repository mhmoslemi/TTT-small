def reduce_max_conv(a, rng, step=0.1):
    conv = np.convolve(a, a)
    k_max = int(conv.argmax())
    # compute contributions array size n
    contrib = np.zeros_like(a)
    for i in range(len(a)):
        j = k_max - i
        if 0 <= j < len(a):
            contrib[i] = a[i] * a[j]
    # find top t indices
    t = max(1, len(a)//200) # maybe size of block
    # get indices sorted descending
    top_idx = np.argpartition(contrib, -t)[-t:]
    # multiplicative factor per index
    factor = np.exp(rng.normal(0, 0.1)) # as earlier maybe
    # reduce large contributions
    for idx in top_idx:
        a[idx] = max(0.0, a[idx] * factor)
    # scale to maintain sum (or not needed)
    # optionally mirror
    if rng.random() < 0.5:
        a = 0.5 * (a + a[::-1])
    # clip to bounds
    return np.clip(a, 0.0, 1000.0)

    idx = np.arange(n)

def gradient_step(a):
    # compute convolution via FFT
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[:2*n-1]
    k_max = int(conv.argmax())
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[mask] * a[partner[mask]]
    # Choose top fraction
    # If there are no contributions (should not happen), return a unchanged
    if np.all(contrib == 0):
        return a
    # scaling factor
    factor_reduce = 0.9
    # Identify indices with contribution > median
    thresh = np.median(contrib[mask])  # compute median of nonzero contributions
    high = contrib > thresh
    a_new = a.copy()
    a_new[high] *= factor_reduce
    # renorm
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = a_new.sum()
    if s == 0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new

import numpy as np

# fast computation of the maximum entry of a*a using FFT (O(n log n), O(n) memory)
def _max_conv_peak(x: np.ndarray, fft_len: int, n: int) -> float:
    """Return max_{k} (x*x)[k] for a real vector x (full convolution)."""
    X = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(X * X, n=fft_len)
    # convolution length is 2*n-1, the rest is zero‑padding artefacts
    return float(conv[: 2 * n - 1].max())
