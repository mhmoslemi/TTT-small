import numpy as np

def build_sidon_set(n):
    """
    Greedy construction of a Sidon (B2) set of indices in [0, n).
    The set satisfies that all unordered sums i + j (i <= j) are distinct.
    """
    high = []
    used = set()
    for i in range(n):
        conflict = False
        # check sums with existing elements
        for j in high:
            if (i + j) in used:
                conflict = True
                break
        if conflict or (2 * i) in used:
            continue
        # accept i
        for j in high:
            used.add(i + j)
        used.add(2 * i)
        high.append(i)
    return high


def propose_candidate(seed=42, **kwargs):
    """
    Generate a candidate step‑function heights vector that aims to minimise
    the AC1 upper bound evaluated by `evaluate_sequence`.
    Strategy:
      * Build a Sidon set of indices (few collisions in the convolution).
      * Place the maximum allowed height (1000) at those indices.
      * Fill the remaining positions with a modest constant (tuned automatically).
    """
    rng = np.random.default_rng(seed)

    # Determine length from the provided parent construction if present
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256  # fallback length

    # Build the sparse high‑amplitude support
    high_idxs = build_sidon_set(n)

    # Search a small range of baseline values; they are cheap to evaluate.
    best_seq = None
    best_score = float("inf")
    for baseline in range(5, 31):          # try baselines 5 … 30
        seq = np.full(n, float(baseline), dtype=np.float64)
        if high_idxs:
            seq[high_idxs] = 1000.0        # maximal height on the Sidon positions
        # Exact evaluation (the harness provides evaluate_sequence)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # If for any reason no candidate succeeded, fall back to a simple uniform vector.
    if best_seq is None or best_score == float("inf"):
        fallback = np.ones(n, dtype=np.float64)
        best_seq = fallback

    # Return plain Python list of floats as required.
    return [float(x) for x in best_seq]
