    if not isinstance(sequence, list):

    low_w = max(1, n // 200)
    high_w = max(2, n // 12)
    width = int(rng.integers(low_w, high_w))
    # Random start position for the block.
    start = int(rng.integers(0, max(1, n - width + 1)))
    # Multiplicative perturbation (log‑normal factor, modest variance).
    scale = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:start + width] *= scale
    # Occasionally symmetrize the candidate.
    if rng.random() < 0.15:
        cand = 0.5 * (cand + cand[::-1])
    # Enforce the allowed range [0, 1000].
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """Return a list of heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the supplied parent if it exists; otherwise use a
    # uniform fallback.
    # ------------------------------------------------------------------
    try:
        parent_arr = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)
    except NameError:
        parent_arr = None

    if parent_arr is not None and parent_arr.size > 0:
        n = parent_arr.size
    else:
        # Fallback length – any positive length works; 4806 matches the
        # typical parent size.
        n = 4806
        parent_arr = np.full(n, 1.0, dtype=np.float64)

    # Uniform baseline (guarantees the theoretical lower bound = 2.0).
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = evaluate(best.tolist())

    # If the supplied parent is better, adopt it.
    parent_score = evaluate(parent_arr.tolist())
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent_arr.copy()
        best_score = parent_score

    # ------------------------------------------------------------------
    # Simple stochastic local search: many cheap block perturbations.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = make_candidate(best, rng)
        cand_score = evaluate(cand.tolist())
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score

    # Final safety clipping (should already be satisfied).
    np.clip(best, 0.0, 1000.0, out=best)
    return [float(x) for x in best]
