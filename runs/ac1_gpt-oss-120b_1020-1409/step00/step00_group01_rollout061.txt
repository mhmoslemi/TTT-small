def generate_sidon_set(n, target_m, seed=12345):
    rng = np.random.default_rng(seed)
    S = set()
    sum_set = set()
    attempts = 0
    max_attempts = n * 10
    while len(S) < target_m and attempts < max_attempts:
        cand = int(rng.integers(0, n))
        if cand in S:
            attempts += 1
            continue
        ok = True
        # check collisions
        # candidate self-sum
        double = cand + cand
        if double in sum_set:
            ok = False
        else:
            for a in S:
                s = cand + a
                if s in sum_set:
                    ok = False
                    break
        if ok:
            # Add new sums
            sum_set.add(double)
            for a in S:
                sum_set.add(cand + a)
            S.add(cand)
        attempts += 1
    return sorted(S)
```

def reduce_peak(a, kmax, rng, alpha=0.1):
    lo = max(0, kmax - (n-1))
    hi = min(kmax, n-1)
    indices = np.arange(lo, hi+1)
    js = kmax - indices
    contrib = a[indices] * a[js]
    # sort descending
    top = np.argsort(contrib)[-max(1, int(0.01*len(contrib))):]  # top 1%
    for idx in top:
        i = int(indices[idx])
        j = int(js[idx])
        if i == j:
            a[i] *= (1 - alpha)
        else:
            a[i] *= (1 - alpha)
            a[j] *= (1 - alpha)
    # renormalize to keep sum same
    total = a.sum()
    if total > 0:
        a *= (target_sum / total)
    return a
