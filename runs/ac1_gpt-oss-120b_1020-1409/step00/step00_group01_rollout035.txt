def compute_score(seq):
    n = seq.size
    # compute FFT length as power of two >= 2*n-1
    N = 1 << ((2 * n - 1).bit_length())

    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_c / (sum_a * sum_a))

def propose_candidate(seed=42, **kwargs):
    """
    Construct a linearly increasing step‑function (0 → 1000) whose AC1
