def proj_simplex(v):
    # v is 1-D np.ndarray
    # returns projection onto simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

def project_simplex(v):
   # projection onto simplex sum(v) = 1, v >=0.
   # reference Duchi et al.
   n = v.shape[0]
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
   if rho.size == 0:
       # all negative, set uniform
       return np.full_like(v, 1.0 / n)
   rho = rho[-1]
   theta = cssv[rho] / (rho + 1)
   w = np.maximum(v - theta, 0)
   return w

import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                     # the last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
