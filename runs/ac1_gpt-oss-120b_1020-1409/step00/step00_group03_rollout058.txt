import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required)
# ----------------------------------------------------------------------
def _ratio_array(length: int) -> np.ndarray:
    """
    Return a decreasing linear ramp from 1 down to 0 of size ``length``.
    The first element is 1, the last is 0 (if length > 1).
    """
    if length <= 1:
        return np.ones(1, dtype=np.float64)
    idx = np.arange(length, dtype=np.float64)
    return (length - 1 - idx) / (length - 1)


def _candidate_from_power(ratio: np.ndarray, p: float) -> np.ndarray:
    """Build a candidate sequence a_i = ratio_i ** p."""
    # power on a non‑negative array is safe; ``out`` avoids an extra allocation.
    out = np.empty_like(ratio)
    np.power(ratio, p, out=out)
    return out


def _block_perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a cheap O(n) block‑wise perturbation of ``base``.
    Mirrors the idea of the previous baseline code.
    """
    n = base.size
    cand = base.copy()
    # pick a block width between n/200 and n/12 (inclusive)
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation on the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps keep the shape balanced)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the evaluator's hard limits
    return np.clip(cand, 0.0, 1000.0)
