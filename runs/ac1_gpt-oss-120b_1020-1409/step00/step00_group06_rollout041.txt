def _transfer_mass(arr, i, j, delta):
    """Move ``delta`` from index ``i`` to ``j`` respecting bounds."""
    new_arr = arr.copy()
    new_arr[i] -= delta
    new_arr[j] += delta
    # enforce the hard constraints
    np.clip(new_arr, 0.0, 1000.0, out=new_arr)
    return new_arr


def propose_candidate(seed=42, max_iters=3000, **kwargs):
    """
    Heuristic search that repeatedly moves mass away from the
    lag producing the largest convolution value.
    Returns a list of non‑negative coefficients.
    """
    rng = np.random.default_rng(seed)
    a = _initial_candidate()
    n = a.size

    # keep the best we have seen
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # main optimisation loop
    for _ in range(max_iters):
        # convolution of the current candidate (cheap, np.convolve uses FFT)
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # indices that actually contribute to this maximal lag
        start = max(0, k_max - (n - 1))
        end = min(n - 1, k_max)
        # pick a source index (high value) inside the active window
        if start > end:
            # degenerate – pick any index
            src = int(rng.integers(0, n))
        else:
            # bias towards larger entries
            window = a[start:end + 1]
            if window.max() == 0:
                src = int(rng.integers(start, end + 1))
            else:
                probs = window / window.sum()
                src = start + int(rng.choice(len(window), p=probs))

        # pick a destination index outside the active window if possible
        mask = np.ones(n, dtype=bool)
        if start <= end:
            mask[start:end + 1] = False
        outside = np.where(mask)[0]
        if outside.size > 0:
            dst = int(rng.choice(outside))
