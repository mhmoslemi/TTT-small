import numpy as np

# ----------------------------------------------------------------------
# Helper: convolution via FFT (real‑valued, returns full length 2n-1)
def fft_convolve_self(a):
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return c

# Helper: exact objective (identical to the provided evaluate_sequence)
def objective(a):
    n = a.size
    conv = fft_convolve_self(a)
    max_b = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))
