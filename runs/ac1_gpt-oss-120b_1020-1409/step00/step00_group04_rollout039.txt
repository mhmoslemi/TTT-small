def generate_sidon_set(n, target_m):
    sums_set = set()
    indices = []
    for x in range(n):
        conflict = False
        # check for each y in indices
        for y in indices:
            if x + y in sums_set:
                conflict = True
                break
        # also need to check double sum x + x
        if not conflict and (x + x) in sums_set:
            conflict = True
        if not conflict:
            # add the sums
            for y in indices:
                sums_set.add(x + y)
            sums_set.add(x + x)
            indices.append(x)
            if len(indices) >= target_m:

def generate_sidon_full(n, target_m, rng, max_trials=5):
    # attempt up to max_trials random permutations
    indices = list(range(n))
    for _ in range(max_trials):
        rng.shuffle(indices)   # in place
        chosen = []
        sums = set()
        for x in indices:
            conflict = False
            # Quick check: x+x
            if (x + x) in sums:
                continue
            for y in chosen:
                if (x + y) in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # accept
            for y in chosen:
                sums.add(x + y)
            sums.add(x + x)
            chosen.append(x)
            if len(chosen) >= target_m:
                return chosen
    # fallback: return whatever we have (maybe less)
    return chosen
```
