import numpy as np

def block_perturb(seq, rng, sigma=0.1):
    n = seq.shape[0]
    if n == 0:
        return seq
    width = int(rng.integers(1, max(2, n // 20)))
    start = int(rng.integers(0, n - width + 1))
    factor = np.exp(rng.normal(0.0, sigma))
    new = seq.copy()
    new[start:start+width] *= factor
    if rng.random() < 0.3:
        new = 0.5 * (new + new[::-1])
    np.clip(new, 0.0, 1000.0, out=new)
    return new

def propose_candidate(seed=42, num_steps=200, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)
    n = base.shape[0]
    candidate = np.full(n, 1.0, dtype=np.float64)
    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))
    for _ in range(num_steps):
        pert = block_perturb(best, rng)
        score = float(evaluate_sequence(pert.tolist()))
        if np.isfinite(score) and score < best_score:
            best = pert
            best_score = score
    # ensure valid fallback if best_score still inf (should not)
    if not np.isfinite(best_score):
        best = np.clip(base, 0.0, 1000.0)
        if np.sum(best) < 0.01:
            best = np.ones_like(best)
    return [float(x) for x in best]
