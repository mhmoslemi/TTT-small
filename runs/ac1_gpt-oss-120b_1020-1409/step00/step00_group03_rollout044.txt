    n = _get_length()

    base = np.full(n, 1.0, dtype=np.float64)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    for _ in range(num_trials):
        # multiplicative perturbation on a random contiguous block
        start = int(rng.integers(0, n))
