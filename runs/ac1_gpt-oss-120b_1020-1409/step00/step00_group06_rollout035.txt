import numpy as np

# evaluate_sequence is provided by the harness


def _build_sidon_set(n, rng, attempts=5):
    """Return a list of indices forming a Sidon set (all i+j distinct)."""
    best_set = []
    for _ in range(attempts):
        perm = rng.permutation(n)
        chosen = []
        sums_used = set()
        for idx in perm:
            if idx == 0:               # avoid index zero to keep uniqueness simple
                continue
            conflict = False
            for j in chosen:
                if idx + j in sums_used:
                    conflict = True
                    break
            if conflict or (idx + idx) in sums_used:
                continue
            # accept idx
            chosen.append(int(idx))
            # record all new sums (including self)
            for j in chosen:
                sums_used.add(idx + j)
        if len(chosen) > len(best_set):
            best_set = chosen
    return best_set


def propose_candidate(seed=42, **kwargs):
    """Construct a low‑AC1 bound step‑function using a Sidon set and a small baseline."""
    rng = np.random.default_rng(seed)

    # Determine length n from the supplied parent if possible
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    # --------------------------------------------------------------
    # 1. Build a large Sidon set (adds O(1) contributions to max)
    # --------------------------------------------------------------
    sidon_indices = _build_sidon_set(n, rng, attempts=7)
    m = len(sidon_indices)                     # number of large spikes

    # --------------------------------------------------------------
    # 2. Choose a uniform tiny baseline ε.
    #    A simple analytically‑motivated choice is ε = 2 / max(m,1),
    #    which balances the contribution of the baseline to the score.
    # --------------------------------------------------------------
    eps = 2.0 / max(m, 1)
    # keep ε modest to avoid a too‑large baseline contribution to max
    eps = min(eps, 0.05)

    # --------------------------------------------------------------
    # 3. Form the candidate sequence.
    #    - baseline = ε on every position
    #    - spikes of weight 1 on the Sidon positions
    # --------------------------------------------------------------
    candidate = np.full(n, eps, dtype=np.float64)
    if m:
        candidate[sidon_indices] = 1.0
    candidate = np.clip(candidate, 0.0, 1000.0)

    # --------------------------------------------------------------
    # 4. Evaluate and optionally try a few simple scalings of ε.
    # --------------------------------------------------------------
    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    for factor in (0.5, 1.0, 1.5, 2.0):
        a = np.full(n, eps * factor, dtype=np.float64)
        if m:
            a[sidon_indices] = 1.0
        a = np.clip(a, 0.0, 1000.0)
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = a, sc

    # --------------------------------------------------------------
    # 5. Fallback: if something went wrong, return a simple uniform vector.
    # --------------------------------------------------------------
    if not np.isfinite(best_score):
        fallback = np.ones(n, dtype=np.float64)
        best = fallback

    return [float(x) for x in best]
