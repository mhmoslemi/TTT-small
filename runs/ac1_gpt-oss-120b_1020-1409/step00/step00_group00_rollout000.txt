import numpy as np

def propose_candidate(seed=42, num_steps=1000, **kwargs):
    """Return a non‑negative step‑function that (almost) minimizes the AC1 bound.

    The bound `2 * n * max(f*f) / (sum(f)^2)` is minimized by a constant
    sequence, so we start from that.  A small random block search is kept as a
    cheap fallback in case the provided `height_sequence_1` already beats the
    constant.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – use the supplied parent if it looks sane.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)   # generic fallback

    n = int(parent.size)

    # ---- baseline: uniform sequence (theoretical optimum for the bound) ----
    best_seq = np.full(n, 1.0, dtype=np.float64)      # any positive constant works
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ---- compare with the original parent (it may already be good) ----
    parent_seq = np.clip(parent, 0.0, 1000.0)
    parent_score = float(evaluate_sequence(parent_seq.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best_seq = parent_seq
        best_score = parent_score

    # ---- cheap random block perturbations – keep any genuine improvement ----
    # The bound is convex in the ℓ₁‑norm sense, so improvements are rare;
    # we limit the number of trials to stay fast.
    max_trials = min(num_steps, 200)
    for _ in range(max_trials):
        # Choose a random contiguous block and scale it slightly.
        width = max(1, int(n // rng.integers(50, 150)))
        start = int(rng.integers(0, n - width + 1))
        cand = best_seq.copy()
        scale = float(np.exp(rng.normal(0.0, 0.05)))   # mild perturbation
        cand[start:start + width] *= scale
        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best_seq]
