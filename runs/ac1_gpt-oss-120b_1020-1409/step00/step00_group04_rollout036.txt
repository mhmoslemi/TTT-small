import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _score(arr):
    """Exact score using the provided evaluator."""
    # Convert to Python list as required by evaluate_sequence
    return float(evaluate_sequence(arr.tolist()))


def _make_uniform(n):
    """Uniform step heights with sum = sqrt(2*n) (the normalization used in the paper)."""
    val = np.sqrt(2.0 * n) / n
    return np.full(n, val, dtype=np.float64)


def _random_move(x, rng, max_frac=0.25):
    """Transfer a random fraction of mass from one index to another."""
    n = x.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)

    if x[i] <= 0.0:
        return x  # nothing to move

    frac = rng.random() * max_frac
    delta = x[i] * frac
    y = x.copy()
    y[i] = max(0.0, y[i] - delta)
    y[j] = np.clip(y[j] + delta, 0.0, 1000.0)
    return y


def _targeted_move(x, rng, reduce_frac=0.10):
    """Reduce the two entries that contribute most to the current maximal
    convolution coefficient and redistribute their mass elsewhere."""
    n = x.size
    conv = np.convolve(x, x)
    k_max = int(np.argmax(conv))

    # range of i where i + j = k_max
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1)
    partners = k_max - idx

    # product contributions (ordered)
    prod = x[idx] * x[partners]
    # pick the ordered index with largest product
    pos = int(np.argmax(prod))
    i = idx[pos]
    j = partners[pos]

    # how much to take away
    if i == j:
        delta = x[i] * reduce_frac
        if delta <= 0.0:
            return x
        y = x.copy()
        y[i] = max(0.0, y[i] - delta)
        # give it to a random other position
        k = rng.integers(0, n)
        while k == i:
            k = rng.integers(0, n)
        y[k] = np.clip(y[k] + delta, 0.0, 1000.0)
        return y
    else:
        delta_i = x[i] * reduce_frac
        delta_j = x[j] * reduce_frac
        if delta_i <= 0.0 and delta_j <= 0.0:
            return x
        y = x.copy()
        y[i] = max(0.0, y[i] - delta_i)
        y[j] = max(0.0, y[j] - delta_j)
        total = (x[i] - y[i]) + (x[j] - y[j])
        # dump the removed mass to a random index
        k = rng.integers(0, n)
        while k == i or k == j:
            k = rng.integers(0, n)
        y[k] = np.clip(y[k] + total, 0.0, 1000.0)
        return y

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Returns a list of non‑negative step heights that (heuristically) minimize
    the AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and define a sensible starting point
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        # Fallback: a modest sized uniform vector
        parent = _make_uniform(256)

    n = parent.size

    # Uniform vector with the canonical normalization (sum = sqrt(2*n))
    uniform = _make_uniform(n)

    # Pick the better of the supplied parent or the uniform start
    cand_parent = uniform if _score(uniform) <= _score(parent) else parent
    best = cand_parent.copy()
    best_score = _score(best)

    # Current working solution (start from the best found so far)
    current = best.copy()
    current_score = best_score

    # ------------------------------------------------------------------
    # Simple simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    temperature = 1.0
    decay = 0.9995

    for step in range(num_steps):
        # decide which move to try
        if rng.random() < 0.6:
            # focused move on the current worst convolution coefficient
            trial = _targeted_move(current, rng, reduce_frac=0.10)
        else:
            # generic random transfer
            trial = _random_move(current, rng, max_frac=0.30)

        # safety: clip after the move
        trial = np.clip(trial, 0.0, 1000.0)

        # reject degenerate vectors (sum must be > 0.01)
        if trial.sum() < 0.01:
            continue

        trial_score = _score(trial)

        # Acceptance rule (always accept improvements)
        if np.isfinite(trial_score) and (trial_score < current_score or
                                         rng.random() < np.exp(-(trial_score - current_score) / max(temperature, 1e-12))):
            current = trial
            current_score = trial_score

            if trial_score < best_score:
                best = trial.copy()
                best_score = trial_score

        # cool down
        temperature *= decay

        # early exit if we already beat the target bound
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Return the best candidate found (as a plain list of floats)
    # ------------------------------------------------------------------
    # Ensure a safe fallback: if something went wrong, return the uniform vector
    if not np.isfinite(best_score) or best_score == float('inf'):
        best = uniform.copy()
    return [float(v) for v in best]
