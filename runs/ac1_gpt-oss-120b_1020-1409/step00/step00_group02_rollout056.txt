import numpy as np
import math

def _generate_arcsine_sequence(n: int) -> np.ndarray:
    """Return a non‑negative sequence of length n approximating the arcsine law.
    The continuous arcsine density f(x)=1/(π√(x(1-x))) on (0,1) satisfies
    f*f = uniform on (0,2). Discretising with mid‑points gives a sequence whose
    self‑convolution is almost constant, yielding a low AC1 bound.
    """
    i = np.arange(n, dtype=np.float64)
    # mid‑point in (0,1)
    x = (i + 0.5) / n
    # raw values before normalisation
    a = 1.0 / (math.pi * n) / np.sqrt(x * (1.0 - x))
    # normalise to sum exactly 1 (scaling cancels in the score but keeps numbers stable)
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        # fallback to uniform if something went wrong
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total
    # enforce the evaluator's hard bounds
    return np.clip(a, 0.0, 1000.0)
