    used = np.zeros(2*n, dtype=bool)
    positions = []

    candidates = list(range(n))

        idx += 1
        # compute sums

                collides = True
                break

            collides = True

            for pos in positions:
                used[cand + pos] = True
                used[pos + cand] = True
            # also set self sum
            used[cand + cand] = True
        else:
            attempts += 1
    return np.array(positions, dtype=int)

import numpy as np

def _fast_conv(a: np.ndarray) -> np.ndarray:
    """Return full linear convolution a*a using an FFT of size power‑of‑2."""
    n = a.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (same formula as evaluate_sequence)."""
    n = a.size
    conv = _fast_conv(a)
    m = conv.max()
    s = a.sum()
    # safeguard against degenerate sum, though the search never lets this happen
    if s <= 0.0:
        return np.inf
    return float(2.0 * n * m / (s * s))


def _block_perturb(parent: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block, mirrored with prob 0.35."""
    cand = parent.copy()
    n = cand.size
    # block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _peak_perturb(state: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Reduce the contribution to the current maximal convolution peak."""
    n = state.size
    conv = _fast_conv(state)
    k = int(np.argmax(conv))                     # index of current peak
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    if i_max < i_min:                            # should never happen
        return state.copy()
    i = int(rng.integers(i_min, i_max + 1))
    j = k - i
    if i == j:
        # single diagonal entry – we move a fraction of it elsewhere
        delta = state[i] * rng.random() * 0.3
        if delta <= 0.0:
            return state.copy()
        new = state.copy()
        new[i] -= delta
        # pick a distinct position to receive the mass
        p = int(rng.integers(0, n))
        while p == i:
            p = int(rng.integers(0, n))
        new[p] = np.clip(new[p] + delta, 0.0, 1000.0)
        return new
    # two distinct indices participating in the peak
    # move a small amount from both to two other locations
    delta_i = state[i] * rng.random() * 0.15
    delta_j = state[j] * rng.random() * 0.15
    delta = min(delta_i, delta_j)
    if delta <= 0.0:
        return state.copy()
    new = state.copy()
    new[i] -= delta
    new[j] -= delta
    # choose two receivers distinct from i and j
    exclude = {i, j}
    p = int(rng.integers(0, n))
    while p in exclude:
        p = int(rng.integers(0, n))
    q = int(rng.integers(0, n))
    while q in exclude or q == p:
        q = int(rng.integers(0, n))
    new[p] = np.clip(new[p] + delta, 0.0, 1000.0)
    new[q] = np.clip(new[q] + delta, 0.0, 1000.0)
    return new

    # ensure a tiny but positive total mass
    if a.sum() < 0.01:
        a = np.full_like(a, 1.0)

    best = a.copy()
    best_score = float(_score(best))

    # -----------------------------------------------------------------
    # main search loop -------------------------------------------------
    for step in range(num_steps):
        # 50 % chance to use each perturbation type
        if rng.random() < 0.5:
            cand = _block_perturb(a, rng)
        else:
            cand = _peak_perturb(a, rng)

        # quick evaluation (identical to evaluate_sequence up to floating error)
        score = _score(cand)
        if not np.isfinite(score):
            continue

        # accept improvements immediately
        if score < best_score:
            best = cand
            best_score = score
            a = cand
            continue

        # occasional uphill move to escape local minima
        # acceptance probability decays exponentially with step index
        prob = np.exp(-step / (num_steps / 5.0)) * 0.002
        if rng.random() < prob:
            a = cand

    # final verification with the official scorer (guarantees a valid return)
    final_score = float(evaluate_sequence(best.tolist()))
    # (the scorer may clip values; our candidate is already clipped)
    _ = final_score  # silence unused variable warning

    return [float(x) for x in best]
```
