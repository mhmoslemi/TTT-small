def simplex_projection(v, s):
    """Project v onto simplex {w >=0, sum(w) = s}."""
    if s < 0:
        raise
    v_shift = v - np.max(v)  # maybe not required
    u = np.sort(v)[::-1]  # descending
    css = np.cumsum(u) - s
    rho = np.where(u - css / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, num_iters=200, step0=0.001, decay=0.99, **kwargs):
    """
    Gradient‐descent / simplex projection search for a low AC1 bound.
    Returns a non‑empty list of non‑negative coefficients.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Load the parent sequence (provided as height_sequence_1) and
    #     guarantee a valid starting vector.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:                     # fallback if the variable is absent
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    a = parent.copy()

    # --------------------------------------------------------------
    # 2.  Fix the total sum (scale invariance allows any positive sum)
    # --------------------------------------------------------------
    sum_target = max(a.sum(), 1e-6)          # avoid division by zero
    a *= sum_target / a.sum()

    # --------------------------------------------------------------
    # 3.  Pre‑compute FFT length (next power of two >= 2*n-1)
    # --------------------------------------------------------------
    n_fft = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # 4.  Keep the best candidate seen so far.
    # --------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    step = step0

    for it in range(num_iters):
        # ---- convolution via FFT (O(n log n)) ----
        fa = np.fft.rfft(a, n=n_fft)
        conv_full = np.fft.irfft(fa * fa, n=n_fft)[:2 * n - 1]

        # ---- index where the convolution attains its maximum ----
        k_max = int(np.argmax(conv_full))

        # ---- sub‑gradient of the max‑convolution w.r.t. a ----
        idx1 = k_max - np.arange(n)
        idx2 = np.arange(n) - k_max

        g = np.zeros_like(a)

        valid1 = (idx1 >= 0) & (idx1 < n)
        g[valid1] += a[idx1[valid1]]

        valid2 = (idx2 >= 0) & (idx2 < n)
        g[valid2] += a[idx2[valid2]]

        # ---- gradient descent step ----
        a = a - step * g

        # ---- small random jitter to avoid stagnation ----
        a += rng.normal(0.0, 0.001 * sum_target, size=n)

        # ---- enforce non‑negativity and bound ----
        a = np.clip(a, 0.0, 1000.0)

        if cur_sum > 0:
            a *= sum_target / cur_sum
        else:                                 # very unlikely, re‑initialise uniformly
            a = np.full(n, sum_target / n, dtype=np.float64)

        a = np.clip(a, 0.0, 1000.0)

        # ---- occasional exact evaluation (every 10 steps) ----
        if (it + 1) % 10 == 0 or it == num_iters - 1:

    # ------------------------------------------------------------------
