    n = a.shape[0]

    A = np.fft.rfft(a, m)

import numpy as np
import math

# ----------------------------------------------------------------------
# Helper: convolution using a power‑of‑two FFT (O(n log n), linear memory)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full (non‑circular) convolution a*a using a real FFT."""
    n = a.shape[0]
    # length needed for linear convolution
    m = 1 << ((2 * n - 1).bit_length())       # next power of two
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# Fast but exact‑same objective as evaluate_sequence
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2 * n * max(conv) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(a)
    max_b = float(conv.max())
    n = a.shape[0]
    return 2.0 * n * max_b / (s * s)

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative heights whose AC1 upper bound is
    (hopefully) lower than the naive constant sequence.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the supplied parent (height_sequence_1) or fall back
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        cur = np.clip(base, 0.0, 1000.0)
    except Exception:
        # fallback: short uniform positive vector
        cur = np.full(256, 1.0, dtype=np.float64)

    n = cur.shape[0]

    # Work with a copy that always satisfies the constraints
    cur = np.clip(cur, 0.0, 1000.0)
    cur_sum = cur.sum()
    if cur_sum < 0.01:
        # ensure a minimal positive sum
        cur[:] = 1.0
        cur_sum = cur.sum()

    best = cur.copy()
    best_score = _objective(best)

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ------------------------------------------------------------------
    num_steps = int(kwargs.get("num_steps", 12000))
    # Temperature starts at the initial objective and cools to zero
    T0 = best_score

    for step in range(num_steps):
        # current temperature (linear cooling)
        t = 1.0 - step / num_steps
        temp = T0 * t

        # ----- identify the shift that currently attains the maximum -----
        conv = _conv_fft(cur)
        k_max = int(conv.argmax())
        # indices that actually contribute to this convolution entry
        i_lo = max(0, k_max - (n - 1))
        i_hi = min(k_max, n - 1)

        # ----- propose a small mass relocation -----
        i = rng.integers(i_lo, i_hi + 1)                # source index
        if cur[i] <= 1e-12:
            continue                                      # nothing to move

        # destination index – choose preferably outside the active range
        if rng.random() < 0.7:
            # try to stay away from the active interval
            while True:
                j = rng.integers(0, n)
                if j < i_lo or j > i_hi:
                    break
        else:
            j = rng.integers(0, n)
        if j == i:
            continue

        # move a random fraction (up to 30%) of the source mass
        delta = cur[i] * rng.random() * 0.30
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # enforce the hard box constraints
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _objective(cand)

        # ----- acceptance test (metropolis) -----
        improve = cand_score < best_score
        if cand_score < _objective(cur) or rng.random() < math.exp(
            -(cand_score - _objective(cur)) / max(temp, 1e-12)
        ):
            cur = cand
            if improve:
                best = cand
                best_score = cand_score

        # occasional mild smoothing to help escape flat plateaus
        if step % 500 == 0 and step > 0:
            # 3‑point averaging that preserves the total sum exactly
            left = np.concatenate([[0.0], cur[:-1]])
            right = np.concatenate([cur[1:], [0.0]])
            cur = 0.5 * cur + 0.25 * (left + right)
            np.clip(cur, 0.0, 1000.0, out=cur)

    # ------------------------------------------------------------------
    # Fallback sanity – ensure a valid return even if the search failed
    # ------------------------------------------------------------------
    if not np.isfinite(best_score) or best_score == float("inf"):
        # simple uniform fallback (length matches the parent if possible)
        fallback_len = n if n > 0 else 256
        best = np.full(fallback_len, 1.0, dtype=np.float64)

    # Convert to a plain Python list of floats as required by the harness
    return [float(x) for x in best.tolist()]
