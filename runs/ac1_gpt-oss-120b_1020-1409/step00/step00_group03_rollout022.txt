import numpy as np

# Helper: simple O(n) block perturbation (fallback random move)
def make_block_perturbation(base, rng):
    """Return a one‑step non‑negative perturbation of ``base``."""
    cand = np.array(base, copy=True, dtype=np.float64)
    n = cand.size
    # Choose a block width on a log‑scale
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Scale the block by a log‑normal factor
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # Optional mild symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# Helper: full linear convolution via FFT (O(n log n))
def full_convolution_fft(arr):
    """
    Compute the linear (full) convolution of ``arr`` with itself using FFT.
    Returns a NumPy array of length 2*len(arr)-1.
    """
    n = arr.shape[0]
    size = 2 * n                     # zero‑pad to at least 2n
    fft_a = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fft_a * fft_a, size)
    return conv[: 2 * n - 1]         # keep the true linear part
