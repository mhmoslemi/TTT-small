import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function that attains the theoretical minimum
    of the AC1 upper bound: a constant sequence.
    If a parent sequence `height_sequence_1` is available it is used only to

    try:

    """
    Return a non‑negative step‑function that attains the theoretical minimum
    of the AC1 upper bound.  The optimum is reached by a constant sequence,
    because for non‑negative vectors
        2·n·max(f*f)/(sum f)^2 >= 2,

    """

    try:

        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
