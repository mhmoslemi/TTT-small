import numpy as np

def uniform_sequence(length, value=1.0):
    """Return a constant non‑negative sequence of given length."""
    return np.full(length, float(value), dtype=np.float64)


def perturb_sequence(seq, rng):
    """
    Apply a modest random multiplicative perturbation to a subset of entries,
    then renormalise to preserve the total sum.
    """
    n = seq.size
    # Choose a modest fraction of coordinates to modify.
    k = max(1, n // 50)
    idx = rng.choice(n, size=k, replace=False)
    factor = float(np.exp(rng.normal(0.0, 0.1)))
    new_seq = seq.copy()
    new_seq[idx] *= factor
    # Clip to the evaluator's admissible range.
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    # Preserve the overall sum (scale invariance of the objective).
    total = seq.sum()
    cur = new_seq.sum()
    if cur > 0.0:
        new_seq *= total / cur
    return new_seq


def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Produce a candidate step‑function that (heuristically) minimises the
    AC1 upper bound as measured by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # Determine the problem size from the provided parent, if any.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            n = 256
        else:
            n = parent.size
    except Exception:
        n = 256

    # Start from the uniform sequence – it attains the theoretical lower bound J=2.
    best = uniform_sequence(n, value=1.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # Optionally consider the supplied parent as an alternate start.
    try:
        parent_clipped = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                                 0.0, 1000.0)
        if parent_clipped.ndim == 1 and parent_clipped.size == n:
            parent_score = float(evaluate_sequence(parent_clipped.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent_clipped.copy()
                best_score = parent_score
    except Exception:

    # Simple stochastic local search: keep any perturbation that improves the score.
    for _ in range(num_steps):
