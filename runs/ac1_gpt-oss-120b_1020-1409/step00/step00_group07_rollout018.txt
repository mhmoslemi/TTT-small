def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimize
    the AC1 upper bound evaluated by ``evaluate_sequence``.
    The routine works with the runtime variable ``height_sequence_1`` if it
    exists; otherwise it falls back to a modest length.
    """
    # ----------------------------------------------------------------------
    # Determine problem size (use the provided parent if available)
    # ----------------------------------------------------------------------
    base = globals().get('height_sequence_1')
    if isinstance(base, np.ndarray) and base.ndim == 1 and base.size > 0 \
            and np.isfinite(base).all():
        n = int(base.size)
    else:
        n = 256                       # safe fallback length

    # ----------------------------------------------------------------------
    # Start from a uniform sequence – this is provably optimal for the
    # evaluator (ratio = 2) and guarantees a finite score.
    # ----------------------------------------------------------------------
    best_arr = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best_arr.tolist()))

    # ----------------------------------------------------------------------
    # If the given parent is valid, evaluate it and keep the better one.
    # ----------------------------------------------------------------------
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = float(evaluate_sequence(parent.tolist()))
            if parent_score < best_score:
                best_arr = parent
                best_score = parent_score
    except Exception:
        pass  # ``height_sequence_1`` may be missing – ignore.

    # ----------------------------------------------------------------------
    # Simple stochastic block perturbation local search (O(n) memory).
    # ----------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for _ in range(num_steps):

        # Random contiguous block gets a multiplicative perturbation.
        width = rng.integers(max(1, n // 200), max(2, n // 12))

        # Respect the problem bounds.
        cand = np.clip(cand, 0.0, 1000.0)

        # Exact evaluation of the candidate.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score
