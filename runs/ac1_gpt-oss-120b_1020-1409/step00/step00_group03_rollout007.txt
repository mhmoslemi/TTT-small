import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a nonnegative step-function (list of floats) that minimizes the
    AC1 upper bound as evaluated by `evaluate_sequence`.  The uniform sequence
    achieves the theoretical optimum (value 2) for non‑negative coefficients,
    and is compared against any provided parent sequence `height_sequence_1`.
    """
    # Obtain a parent sequence if available; otherwise fall back to a simple
    # all‑ones vector.  The parent is clipped to the allowed range.
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
    except NameError:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # Uniform candidate (any positive constant works, here 1.0)
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform = np.clip(uniform, 0.0, 1000.0)

    # Evaluate both candidates using the supplied evaluator.
    # (We keep the scores in case the parent is already optimal.)
    try:
        parent_score = float(evaluate_sequence(parent.tolist()))
    except Exception:
        parent_score = float('inf')
    try:
        uniform_score = float(evaluate_sequence(uniform.tolist()))
    except Exception:
        uniform_score = float('inf')

    # Choose the better (lower) finite score; fall back to the parent if needed.
    if np.isfinite(uniform_score) and (uniform_score < parent_score or not np.isfinite(parent_score)):
        best = uniform
    else:
        best = parent

    # Return a plain Python list of floats.
    return [float(x) for x in best]
