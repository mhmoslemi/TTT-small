def reduce_max(seq, rng):
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_idx = int(conv.argmax())
    # Build list of i where j valid and both >0
    # we can vectorize:
    i = np.arange(n)
    j = max_idx - i
    valid = (j >=0) & (j < n) & (seq > 0) & (seq[j] > 0)
    if not valid.any():
        return seq.copy()
    # Compute contributions
    contributions = seq[i[valid]] * seq[j[valid]]
    # Choose index with largest contribution
    idx = np.argmax(contributions)
    i_sel = i[valid][idx]
    j_sel = j[valid][idx]
    # Determine delta
    delta = min(seq[i_sel], seq[j_sel]) * rng.uniform(0.02, 0.2)
    if delta <= 1e-12:
        return seq.copy()
    new_seq = seq.copy()
    new_seq[i_sel] -= delta
    new_seq[j_sel] -= delta
    # add to the smallest index
    min_idx = int(np.argmin(new_seq))
    new_seq[min_idx] += 2*delta
    # Clip and ensure non-negative
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq

import numpy as np

def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.array(base, dtype=np.float64, copy=True)
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _targeted_move(seq, rng):
    """Reduce the current convolution maximum by moving a small amount of mass."""
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())                     # shift with maximal value
    i = np.arange(n, dtype=np.int64)
    j = k - i
    mask = (j >= 0) & (j < n) & (seq > 0) & (seq[j] > 0)
    if not np.any(mask):
        return make_block_perturbation(seq, rng)
    # contribution of each ordered pair to the maximal entry
    contrib = seq[i[mask]] * seq[j[mask]]
    idx = int(np.argmax(contrib))
    i_sel = i[mask][idx]
    j_sel = j[mask][idx]

    # take a small proportion from the two partners
    max_delta = min(seq[i_sel], seq[j_sel])
    if max_delta == 0:
        return make_block_perturbation(seq, rng)
    factor = rng.uniform(0.01, 0.12)
    delta = max_delta * factor
    new_seq = seq.copy()
    new_seq[i_sel] -= delta
    new_seq[j_sel] -= delta

    # dump the removed mass onto the smallest entry (least likely to raise the max)
    min_idx = int(np.argmin(new_seq))
    new_seq[min_idx] += 2.0 * delta
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq

    """Return a step‑function that (usually) beats the 2.0 bound."""
    rng = np.random.default_rng(seed)

    # ----- initialise from the supplied parent, if possible -----
    parent_arr = globals().get('height_sequence_1')
    if (
        isinstance(parent_arr, np.ndarray)
        and parent_arr.ndim == 1
        and parent_arr.size > 0
        and np.isfinite(parent_arr).all()
    ):
        parent = np.clip(parent_arr.astype(np.float64), 0.0, 1000.0)
        n = parent.size
    else:
        # generic fallback length
        n = 256
        parent = np.full(n, 1.0, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ----- a simple linear‑decreasing candidate (max = 1000 at the left) -----
    lin = 1000.0 * (np.arange(n, 0, -1, dtype=np.float64) / n)
    lin_score = float(evaluate_sequence(lin.tolist()))
    if np.isfinite(lin_score) and lin_score < best_score:
        best = lin
        best_score = lin_score

    # ----- short local search to polish the solution -----
    no_improve = 0
    for _ in range(num_steps):
        # most of the time try a targeted move aimed at the current maximum
        if rng.random() < 0.7:
            cand = _targeted_move(best, rng)
        else:
            cand = make_block_perturbation(best, rng)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            break

    return [float(v) for v in best]
