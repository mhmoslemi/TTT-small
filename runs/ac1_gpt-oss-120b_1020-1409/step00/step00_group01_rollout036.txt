        n = 256

    # start from the extremal r‑sequence (max convolution = 1)
    best = generate_r_sequence(n)
    best = np.clip(best, 0.0, 1000.0)

    # initial score
    best_score = float(evaluate_sequence(best.tolist()))

    # greedy refinement: add tiny amounts to early entries where they
    # increase the sum faster than the max convolution.
    epsilons = [0.01, 0.005, 0.002, 0.001]  # decreasing step sizes
    for eps in epsilons:
        # k = number of indices (excluding index 0) already augmented
        k = 0
        step = max(1, n // 20)  # start with a coarse step
        while step >= 1:
            improved = False
            # try to extend the augmented block by `step` entries
            while True:
                new_k = min(k + step, n - 1)          # we never touch index 0
                if new_k == k:
                    break
                candidate = best.copy()
                # add eps only to the *new* part of the block
                candidate[1 + k : 1 + new_k] += eps
                np.clip(candidate, 0.0, 1000.0, out=candidate)
                score = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = candidate
                    best_score = score
                    k = new_k
                    improved = True
                else:
                    break
            if not improved:
                step //= 2

    # safety fallback (should never happen)
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
```
