import numpy as np

def propose_candidate(seed=42, **kwargs):
    """Return a uniform (or fallback) nonnegative step‑function.

    The returned sequence minimizes the AC1 upper bound 2·n·max(a*a)/(∑a)²,
    which attains its theoretical minimum value 2 when all entries are equal.
    """
    # The parent sequence may be provided as a global variable.
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base_arr = None

    # Validate the parent if present.
    valid_base = (
        base_arr is not None
        and base_arr.ndim == 1
        and base_arr.size > 0
        and np.isfinite(base_arr).all()
    )
    if valid_base:
        # Apply the same clipping the evaluator uses.
        base_arr = np.clip(base_arr, 0.0, 1000.0)

    # Score the parent (if any) – keep it as a fallback.
    if valid_base:
        base_score = float(evaluate_sequence(base_arr.tolist()))
    else:
        base_score = np.inf

    # ------------------------------------------------------------
    # Construct a uniform candidate with the same total mass as the parent.
    # ------------------------------------------------------------
    if valid_base:
        n = base_arr.size
        sum_a = float(np.sum(base_arr))
        # Ensure the sum meets the evaluator's minimal threshold.
        if sum_a < 0.01:
            avg = 0.01 / n
        else:
            avg = sum_a / n
        candidate_arr = np.full(n, avg, dtype=np.float64)
    else:
        # No viable parent – use a modest default length and value.
        n = 256
        avg = 1.0  # any positive value works; sum = n > 0.01
        candidate_arr = np.full(n, avg, dtype=np.float64)

    # Score the uniform candidate.
    candidate_score = float(evaluate_sequence(candidate_arr.tolist()))

    # Choose the better of parent and uniform candidate.
    if candidate_score < base_score:
        best_arr = candidate_arr
    else:
        best_arr = base_arr if valid_base else candidate_arr

    # Return a plain Python list of floats.
    return [float(v) for v in best_arr]
