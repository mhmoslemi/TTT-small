import numpy as np

def _uniform_candidate(length: int, value: float = 1.0) -> np.ndarray:
    """Return a uniform non‑negative vector of given length."""
    return np.full(length, float(value), dtype=np.float64)


def _perturb(candidate: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Create an O(n)‑memory random block perturbation of ``candidate``."""
    n = candidate.size
    # choose a block width between roughly 0.5 % and 5 % of the array
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.12))  # moderate log‑normal scaling
    new = candidate.copy()
    new[start:stop] *= factor
    # occasional symmetric smoothing (helps escape shallow minima)
    if rng.random() < 0.33:
        new = 0.5 * (new + new[::-1])
    return np.clip(new, 0.0, 1000.0)

    """
    Return a list of non‑negative step‑function heights.
    The routine builds a uniform candidate (the theoretical optimum for the
    given evaluator) and performs a brief random walk to capture any
    rounding‑driven improvement.
    """

    try:
        n = int(height_sequence_1.shape[0])
