import numpy as np

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """Return a non‑negative step‑function aiming to minimize the AC1 bound.

    The routine works with linear memory.  It starts from a uniform vector,
    optionally improves it with cheap block perturbations, and falls back to a
    simple uniform candidate if anything goes wrong.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (length of the sequence)
    # ------------------------------------------------------------------
    parent = globals().get("height_sequence_1")
    if isinstance(parent, (list, np.ndarray)):
        try:
            parent_arr = np.asarray(parent, dtype=np.float64).reshape(-1)
            n = parent_arr.size
        except Exception:
            parent_arr = None
            n = 256
    else:
        parent_arr = None
        n = 256

    # ------------------------------------------------------------------
    # Initial candidate: uniform positive values (any constant gives AC1 = 2)
    # ------------------------------------------------------------------
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # If a parent sequence is available, evaluate it as a possible starter.
    if parent_arr is not None:
        cand = np.clip(parent_arr, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Simple stochastic local search (block multiplicative perturbations)
    # ------------------------------------------------------------------
    # Width of a perturbed block is kept between 0.1 % and 10 % of n.
    w_min = max(1, n // 1000)
    w_max = max(2, n // 10)

    for _ in range(num_steps):
        cand = best.copy()

        # Choose a random contiguous block and rescale it.
        width = int(rng.integers(w_min, w_max + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        factor = np.exp(rng.normal(0.0, 0.18))        # modest multiplicative noise
        cand[start:stop] *= factor

        # Occasionally enforce symmetry (helps keep the autocorrelation flat).
        if rng.random() < 0.22:
            cand = 0.5 * (cand + cand[::-1])

        # Clip to allowed range.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Evaluate and keep improvement.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Final safeguard: ensure a finite, non‑empty result.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.full(n, 1.0, dtype=np.float64)

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
