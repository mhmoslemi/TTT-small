import numpy as np

def _geometric_score(r: float, n: int) -> float:
    """Analytic AC1 upper bound for the geometric sequence a_i = r**i."""
    if not (0.0 < r < 1.0):
        return np.inf
    # sum of the geometric series
    r_n = r ** n
    sum_a = (1.0 - r_n) / (1.0 - r)

    # maximum entry of the convolution a*a.
    # For each shift k (0 .. 2n-2) the count of contributing pairs is
    #   cnt(k) = n - |k-(n-1)|
    # and each term equals r**k, so the value is cnt(k) * r**k.
    max_b = 0.0
    pow_r = 1.0                     # r**k
    center = n - 1
    for k in range(2 * n - 1):
        cnt = n - abs(k - center)
        val = cnt * pow_r
        if val > max_b:
            max_b = val
        pow_r *= r
    # AC1 upper bound as defined in the evaluator
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 42, **kwargs) -> list:
    """
    Produce a non‑negative step‑function whose AC1 bound
    is (significantly) better than the uniform baseline.
    """
    # Length of the problem (height_sequence_1 is provided by the harness)
    try:
        n = int(len(height_sequence_1))
    except NameError:
        # safety fallback – should never happen in the official harness
        n = 256

    # ------------------------------------------------------------
    # 1) Scan a one‑parameter family a_i = r**i  (0 < r < 1)
    # ------------------------------------------------------------
    best_r = 0.5
    best_score = np.inf
    # wide enough range for the parameter t = (1-r)*n
    num_samples = 1200
    t_vals = np.linspace(0.01, 5.0, num_samples)
    for t in t_vals:
        r = 1.0 - t / n
        if r <= 0.0 or r >= 1.0:
            continue
        sc = _geometric_score(r, n)
        if sc < best_score:
            best_score, best_r = sc, r

    # also test the analytically suggested point t = 2 (r = 1-2/n)
    t = 2.0
    r = 1.0 - t / n
    if 0.0 < r < 1.0:
        sc = _geometric_score(r, n)
        if sc < best_score:
            best_score, best_r = sc, r

    # ------------------------------------------------------------
    # 2) Build the candidate sequence with the best r found.
    # ------------------------------------------------------------
    seq = best_r ** np.arange(n, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)          # respects the evaluator's caps

    # ------------------------------------------------------------
    # 3) Verify with the exact evaluator; fall back to a uniform
    #    vector if something unexpected happens.
    # ------------------------------------------------------------
    try:
        exact_score = float(evaluate_sequence(seq.tolist()))
        if not np.isfinite(exact_score):
            raise ValueError
    except Exception:
        # fallback – uniform sequence (score = 2.0, always valid)
        seq = np.ones(n, dtype=np.float64)

    return [float(x) for x in seq]
