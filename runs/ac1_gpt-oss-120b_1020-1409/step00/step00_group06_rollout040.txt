def evaluate_sequence(sequence: list) -> float:
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)

    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

import numpy as np

def _initial_inverse_sqrt(n: int) -> np.ndarray:
    """
    Return a non‑negative step function with a_i = 1000 / sqrt(i+1),
    clipped to [0, 1000].
    """
    a = 1000.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def propose_candidate(seed: int = 42, num_iter: int = 2000, **kwargs):
    """
    Memory‑bounded local optimisation that starts from the 1/√(i) profile.
    It repeatedly reduces the dominant contribution to the convolution peak
    and redistributes the removed mass uniformly, keeping the L1‑norm
    (and hence the denominator of the bound) essentially constant.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Determine the length n and a fallback parent sequence.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                     # height_sequence_1 may be absent
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # ------------------------------------------------------------------
    # 2.  Prepare the initial candidate – the classic inverse‑sqrt shape.
    # ------------------------------------------------------------------
    best = _initial_inverse_sqrt(n)
    best_score = float(evaluate_sequence(best.tolist()))

    # a simple safeguard: keep the provided parent if it is better
    parent_score = float(evaluate_sequence(parent.tolist()))
    if parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # ------------------------------------------------------------------
    # 3.  Iterative improvement.
    # ------------------------------------------------------------------
    for _ in range(num_iter):
        # Convolution and location of the current peak
        conv = np.convolve(best, best)
        k = int(conv.argmax())                     # offset with maximal value
        # product contributions for this offset (vectorised)
        i_idx = np.arange(n, dtype=np.int64)
        j_idx = k - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        prod = np.zeros_like(best)
        prod[mask] = best[mask] * best[j_idx[mask]]

        # Index that contributes most to the peak
        i_max = int(prod.argmax())
        j_max = k - i_max

        # ------------------------------------------------------------------
        # Propose a mass‑reallocation step:
        #   * Reduce the two dominant entries by a fraction.
        #   * Redistribute the extracted mass uniformly over the whole
        #     vector (so the total sum stays (almost) unchanged).
        # ------------------------------------------------------------------
        cand = best.copy()

            total_removed = delta

            cand[j_max] = max(0.0, cand[j_max] - delta_j)

        # Uniform redistribution of the removed mass
        addition = total_removed / n
        cand += addition
        np.clip(cand, 0.0, 1000.0, out=cand)

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
