import numpy as np

def make_block_perturbation(base, rng):
    """Return an O(n) memory non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # width between n/200 and n/12  (at least 1)
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # optional slight symmetry
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

                cand *= (total / s)
                np.clip(cand, 0.0, 1000.0, out=cand)
                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best, best_score, a = cand.copy(), score, cand.copy()
            continue

        # --------------------------------------------------------------
        # Targeted mass‑transfer step aimed at the dominant convolution shift
        # --------------------------------------------------------------
        conv = np.convolve(a, a)                     # full convolution
        K = int(conv.argmax())                       # shift with maximal value

        # indices that actually contribute to this shift
        start = max(0, K - (n - 1))
        stop = min(n - 1, K)
        idx = np.arange(start, stop + 1)             # i such that K-i is in range
        mate = K - idx                               # counterpart j = K-i
        contrib = a[idx] * a[mate]                   # product a_i * a_{K-i}

        # Choose a donor index weighted by its contribution to the peak
        if contrib.sum() > 0.0:
            probs = contrib / contrib.sum()
            donor = rng.choice(idx, p=probs)
        else:
            donor = rng.integers(0, n)

        # Choose a recipient index that currently carries little mass
        # (probability proportional to 1/(a+eps) encourages low‑value slots)
        inv = 1.0 / (a + 1e-12)
        rec_probs = inv / inv.sum()
        recipient = rng.choice(n, p=rec_probs)
        # avoid selecting the same position
        if recipient == donor:
            # pick a random different index if the probabilistic draw collided
            recipient = rng.integers(0, n - 1)
            if recipient >= donor:
                recipient += 1

        # Amount to move: a fraction of the donor's current value,
        # limited by the recipient's remaining capacity (1000 cap).
        frac = max(0.02, 0.3 * (1.0 - it / num_iters))   # decaying step size
        max_transfer = a[donor] * frac
        cap = 1000.0 - a[recipient]
        delta = min(max_transfer, cap)
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[donor] -= delta
        cand[recipient] += delta
        # sum stays exactly the same, non‑negativity already ensured
        # evaluate candidate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score, a = cand.copy(), score, cand.copy()
            # after a successful improvement we keep the new state
        # occasional acceptance of a slightly worse move to escape plateaus
        elif rng.random() < 0.001:
            a = cand.copy()   # move anyway (no score update)

    # ------------------------------------------------------------------
    # Ensure a valid fallback: if nothing improved, return the original parent
    # ------------------------------------------------------------------
    if best_score == float('inf') or not np.isfinite(best_score):
        best = parent.copy()
    return [float(x) for x in best]
