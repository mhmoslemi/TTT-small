def _build_spike_set(n, K, target_m, rng):
    """Greedy construction of a set of indices such that each sum i+j
    appears at most K times (unordered pairs). Returns a list of positions."""
    positions = []
    pos_set = set()
    sum_counts = {}          # maps sum -> number of unordered pairs producing it
    attempts = 0
    max_attempts = target_m * 100  # give up after many failed tries

    while len(positions) < target_m and attempts < max_attempts:
        p = int(rng.integers(0, n))
        if p in pos_set:
            attempts += 1
            continue

        # check self‑sum (2*p) against the limit
        s_self = 2 * p
        if sum_counts.get(s_self, 0) + 1 > K:
            attempts += 1
            continue

        # check sums with all existing positions
        bad = False
        for q in positions:
            s = p + q
            if sum_counts.get(s, 0) + 1 > K:
                bad = True
                break
        if bad:
            attempts += 1
            continue

        # all constraints satisfied – add the new position
        # update counts for the self‑pair
        sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
        # update counts for pairs with previous positions
        for q in positions:
            s = p + q
            sum_counts[s] = sum_counts.get(s, 0) + 1

        positions.append(p)
        pos_set.add(p)
        attempts = 0    # reset attempts after a successful insertion

    return positions
