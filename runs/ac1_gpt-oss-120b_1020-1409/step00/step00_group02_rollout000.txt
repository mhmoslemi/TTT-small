import numpy as np

def make_block_perturbation(base, rng):
    """Return a non‑negative O(n)‑memory perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # choose a modest block width (at least 2 elements)
    min_w = max(2, n // 200)
    max_w = max(min_w + 1, n // 20)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal perturbation
    factor = np.exp(rng.normal(0.0, 0.07))
    cand[start:stop] *= factor
    # occasional symmetric smoothing
    if rng.random() < 0.3:
        seg = cand[start:stop]
        cand[start:stop] = 0.5 * (seg + seg[::-1])
    # enforce the evaluator's bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
