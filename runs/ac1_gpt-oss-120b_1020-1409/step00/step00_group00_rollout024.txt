import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (identical to evaluate_sequence
# up to rounding).  Uses FFT for O(n log n) instead of O(n^2).
def _fast_score(a: np.ndarray) -> float:
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    # convolution length = 2*n-1
    m = 2 * n - 1
    # real FFT is sufficient because a is real
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    max_b = float(conv.max())
    return 2.0 * n * max_b / (s * s)

    the AC1 bound evaluated by `evaluate_sequence`.  The routine builds a
    family of analytically‑motivated seed sequences (inverse‑square‑root)
    and then performs a cheap stochastic local search that only ever keeps
    O(n) auxiliary memory.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n (fallback to a modest size if height_sequence_1 is
    # unavailable or malformed).
    try:
        n = int(height_sequence_1.shape[0])
        if n <= 0:
            raise Exception()
    except Exception:
        n = 1024
        # a dummy placeholder (will never be used for length‑dependent ops)
        dummy = np.ones(n, dtype=np.float64)
        globals()["height_sequence_1"] = dummy

    # ------------------------------------------------------------------
    # ----- 1. Build a few deterministic seed candidates ----------------
    best = None
    best_score = float("inf")

    # Uniform fallback (guaranteed feasible)
    uniform = np.ones(n, dtype=np.float64)
    best = uniform
    best_score = _fast_score(best)

    # Power‑law family a_i = 1 / sqrt(i + beta)
    idx = np.arange(n, dtype=np.float64)
    betas = [0.0, 0.3, 0.5, 1.0, 2.0, 5.0]
    for beta in betas:
        cand = 1.0 / np.sqrt(idx + beta + 1.0)  # +1 avoids division by zero when beta=0
        cand = np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _fast_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # Exponential family a_i = r**i (r very close to 1)
    rs = [0.9996, 0.9998, 0.9999]
    for r in rs:
        cand = r ** idx
        cand = np.clip(cand, 0.0, 1000.0, out=cand)
        sc = _fast_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc

    # ------------------------------------------------------------------
    # ----- 2. Stochastic local improvement -----------------------------
    # Number of proposals – configurable via kwargs, default 1500.
    max_iters = int(kwargs.get("num_steps", 1500))
    n = best.size
    for _ in range(max_iters):
        # start from the current best
        cand = best.copy()

        # ---- a) block multiplicative perturbation ----------------------
        # choose a random block width (log‑spaced to keep O(n) work)
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        factor = float(np.exp(rng.normal(0.0, 0.07)))  # modest scaling
        cand[start:stop] *= factor

        # ---- b) occasional symmetrisation (helps flatten convolution)-
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])

        # ---- c) occasional smoothing (averaging with neighbours) ------
        if rng.random() < 0.1:
            # three‑point moving average, respecting boundaries
            left = np.empty_like(cand)
            right = np.empty_like(cand)
            left[1:] = cand[:-1]
            left[0] = cand[0]
            right[:-1] = cand[1:]
            right[-1] = cand[-1]
            cand = (cand + left + right) / 3.0

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate
        sc = _fast_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # ------------------------------------------------------------------
    # Return as a plain Python list (required by the harness)
    return [float(v) for v in best]
