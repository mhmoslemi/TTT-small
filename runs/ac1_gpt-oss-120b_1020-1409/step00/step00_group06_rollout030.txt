import numpy as np

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # read parent if exists
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    n = parent.size

    # list of initial candidates
    candidates = []

    # uniform candidate (same as prior)
    uniform = np.full(n, 1.0, dtype=np.float64)
    candidates.append(uniform)

    # linear increasing from 0 to 1 scaling to max 1000
    base = np.arange(n, dtype=np.float64)  # 0..n-1
    max_val = 1000.0
    scale = max_val / base[-1] if n > 1 else 0.0
    linear_inc = base * scale
    candidates.append(linear_inc)

    # linear decreasing (reverse)
    linear_dec = linear_inc[::-1].copy()
    candidates.append(linear_dec)

    # quadratic shape: (i/(n-1))^2 * max_val
    if n > 1:
        t = np.linspace(0.0, 1.0, n, dtype=np.float64)
        quad = (t**2) * max_val
        candidates.append(quad)
        quad_rev = quad[::-1]
        candidates.append(quad_rev)

    # exponential shape: exp(alpha * i) scaled to max
    if n > 1:
        # choose small alpha such that exp(alpha*(n-1))=max_val
        # we want min value ~1 maybe; set a0=1 then scale factor
        alpha = np.log(max_val / 1.0) / (n - 1)
        expo = np.exp(alpha * base)
        candidates.append(expo)
        candidates.append(expo[::-1].copy())

    # Evaluate each initial candidate
    best = None
    best_score = np.inf
    for cand in candidates:
        # ensure finite
        if not np.isfinite(cand).all():
            continue
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand.copy()

    # If still none, fallback to parent
    if best is None:
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    # Local search
    for _ in range(num_steps):
        # Choose one of small perturbations
        # We'll randomly choose a mode
        mode = rng.integers(0, 3)
        if mode == 0:
            # block perturbation same as before
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = best.copy()
            factor = float(np.exp(rng.normal(0.0, 0.1)))
            cand[start:stop] *= factor
            if rng.random() < 0.3:
                cand[start:stop] = 0.5 * (cand[start:stop] + cand[start:stop][::-1])
        elif mode == 1:
            # random swap of two points
            cand = best.copy()
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # small global Gaussian perturbation
            cand = best * np.exp(rng.normal(0.0, 0.02, size=n))

        # Clip to allowed range
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(x) for x in best]
