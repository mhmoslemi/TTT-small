def propose_candidate(seed=42, **kwargs):
    import numpy as np
    rng = np.random.default_rng(seed)
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        n = base_arr.size
        if n == 0:
            raise Exception
    except Exception:
        n = 4498
        base_arr = np.ones(n, dtype=np.float64)
    # Greedy attempts to build a large subset with limited pair sums
    best_set = []
    best_max_count = None
    best_M = 0
    max_allowed = 2  # allowed max ordered pair count per sum
    attempts = 30
    for _ in range(attempts):
        perm = rng.permutation(n)
        selected = []
        counts = np.zeros(2*n-1, dtype=np.int16)  # small int enough
        for idx in perm:
            # check feasibility
            feasible = True
            # self pair
            if counts[2*idx] + 1 > max_allowed:
                feasible = False
            else:
                # check with existing selected
                for j in selected:
                    s = idx + j
                    if counts[s] + 2 > max_allowed:
                        feasible = False
                        break
            if feasible:
                # update counts
                counts[2*idx] += 1
                for j in selected:
                    s = idx + j
                    counts[s] += 2
                selected.append(int(idx))
        M = len(selected)
        if M == 0:
            continue
        current_max = int(counts.max())
        # compute simple bound score (ignoring filler baseline)
        bound_score = 2.0 * n * current_max / (M**2)
        # choose better set based on lower bound_score, then higher M
        if best_set == [] or bound_score < 2.0 * n * best_max_count / (best_M**2):
            best_set = selected
            best_max_count = current_max
            best_M = M
    # after attempts, we have best_set
    if not best_set:
        # fallback: uniform
        candidate = np.full(n, 1.0, dtype=np.float64)
        return candidate.tolist()
    # try different filler values
    filler_vals = [0.0, 1.0, 5.0, 10.0, 20.0]  # ensure within bound
    best_candidate = None
    best_score = float('inf')
    for filler in filler_vals:
        candidate = np.full(n, filler, dtype=np.float64)
        candidate[best_set] = 1000.0
        # sometimes we may exceed bounds; but filler small, okay
        # Evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_candidate = candidate.copy()
    # As final fallback if no candidate improved (e.g., evaluation fails), use uniform
    if best_candidate is None:
        best_candidate = np.full(n, 1.0, dtype=np.float64)

import numpy as np

# Helper: build a large index set with limited ordered‑pair sum multiplicities.
def build_sparse_set(n, max_allowed, rng, attempts=30):
    """
    Returns (indices, max_count, size) where `indices` is a list of positions,
    `max_count` is the maximal ordered‑pair count for any sum (<= max_allowed),
    and `size` is the number of selected positions.
    """
    best_set = []
    best_max = 0
    best_len = 0

    for _ in range(attempts):
        perm = rng.permutation(n)
        selected = []
        # counts for sums 0 .. 2n-2 (ordered pairs)
        counts = np.zeros(2 * n - 1, dtype=np.int16)

        for idx in perm:
            # feasibility test
            if counts[2 * idx] + 1 > max_allowed:
                continue
            feasible = True
            for j in selected:
                s = idx + j
                if counts[s] + 2 > max_allowed:
                    feasible = False
                    break
            if not feasible:
                continue

            # commit the addition
            counts[2 * idx] += 1
            for j in selected:
                s = idx + j
                counts[s] += 2
            selected.append(int(idx))

        cur_len = len(selected)
        cur_max = int(counts.max())
        # keep the set with the lowest bound 2*n*cur_max/cur_len**2
        if cur_len > 0:
            if best_len == 0:
                best_set, best_max, best_len = selected, cur_max, cur_len
            else:
                cur_bound = 2.0 * n * cur_max / (cur_len ** 2)
                best_bound = 2.0 * n * best_max / (best_len ** 2)
                if cur_bound < best_bound:
                    best_set, best_max, best_len = selected, cur_max, cur_len

    return best_set, best_max, best_len
