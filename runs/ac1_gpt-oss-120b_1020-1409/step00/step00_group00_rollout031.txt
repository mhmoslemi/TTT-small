def generate_sidon_set(n, target_k, rng):
    sum_used = np.zeros(2*n, dtype=bool)
    positions = []
    # iterate over random permutation of range(n)
    perm = rng.permutation(n)
    for i in perm:
        ok = True
        # check sum with self
        if sum_used[i + i]:
            ok = False
        else:
            # check with existing positions
            for j in positions:
                if sum_used[i + j]:
                    ok = False
                    break
        if ok:
            positions.append(i)
            # mark sums
            sum_used[i + i] = True
            for j in positions[:-1]:  # previous positions
                sum_used[i + j] = True
            if len(positions) >= target_k:
                break
    return positions
```

import numpy as np

def generate_sidon_set(n, target, rng):
    sum_used = np.zeros(2*n, dtype=bool)
    pos = []
    # random order to try variety
    candidates = rng.permutation(n)
    for i in candidates:
        # Check sum collision
        if sum_used[i+i]:
            continue
        conflict = False
        for j in pos:
            if sum_used[i + j]:
                conflict = True
                break
        if conflict:
            continue
        # add
        pos.append(i)
        # mark
        sum_used[i+i] = True
        for j in pos[:-1]:
            sum_used[i + j] = True
        if len(pos) >= target:
            break
    return np.array(pos, dtype=int)
```

import numpy as np

# ----------------------------------------------------------------------
# Helper to build a large set of indices whose pairwise sums appear only
# a few times (a bounded‑additive‑energy set).  The function returns up to
# `target_k` indices, never exceeding `max_rep` unordered representations
# for any sum i+j.
# ----------------------------------------------------------------------
def build_limited_sum_set(n, target_k, max_rep, rng):
    """
    Greedy construction of a set A⊂{0,…,n‑1} with
    at most `max_rep` unordered pairs for each possible sum.
    Returns a NumPy array of chosen indices (length ≤ target_k).
    """
    # number of possible sums = 0 … 2*(n‑1)
    sum_counts = np.zeros(2 * n, dtype=np.int16)   # unordered pair counts
    chosen = []                                    # Python list of ints

    # Try a random order of candidates; repeat if we cannot reach the target.
    candidates = rng.permutation(n)
    for i in candidates:
        # self‑pair (i,i) would increase the count of sum 2*i.
        if sum_counts[i + i] >= max_rep:
            continue
        ok = True
        for j in chosen:
            if sum_counts[i + j] >= max_rep:
                ok = False
                break
        if not ok:
            continue

        # accept i
        chosen.append(i)
        sum_counts[i + i] += 1
        for j in chosen[:-1]:
            sum_counts[i + j] += 1
        if len(chosen) >= target_k:
            break

    return np.array(chosen, dtype=np.int32)
