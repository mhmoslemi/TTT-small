import numpy as np

# ------------------------------------------------------------
# Helper: maximum ordered‑pair count of a binary sequence.
# ------------------------------------------------------------
def max_convolution_count(vec: np.ndarray) -> int:
    """
    Return max_k sum_i vec[i] * vec[k - i] for a binary (0/1) vector.
    Computed with an FFT of length a power‑of‑two ≥ 2·len(vec).
    """
    n = vec.size
    size = 1
    while size < 2 * n:
        size <<= 1                     # next power of two
    # real FFT, element‑wise square, inverse FFT
    f = np.fft.rfft(vec, n=size)
    conv = np.fft.irfft(f * f, n=size)
    # convolution length is 2*n‑1; round to nearest integer safely
    return int(np.rint(conv[: 2 * n - 1]).max())


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed=42, num_trials=3000, **kwargs):
    """
    Search for a non‑negative step‑function that lowers the AC1 bound.
    A sparse binary construction is used: we pick a set of positions,
    put 1.0 there and 0 elsewhere.  The score of such a vector is
        2·n·max_b / (m²)      with  max_b = max_convolution_count,
    where n = length of the sequence and m = number of ones.
    The routine performs many random trials (and a few size variations)
    keeping the best valid candidate.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain the baseline sequence (parent) if it exists.
    # ------------------------------------------------------------------
    try:
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        n = parent.size
    except NameError:
        # Fallback: a modest uniform sequence.
        n = 256
        parent = np.ones(n, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------
    # Helper to turn (m, max_c) into the exact evaluator score.
    # ------------------------------------------------------------
    def score_from_counts(m: int, max_c: int) -> float:
        # For a binary vector with m ones the sum is m,
        # max_b equals max_c (ordered‑pair count) because 1·1 = 1.
        return 2.0 * n * max_c / (float(m) * float(m))

    # ------------------------------------------------------------
    # Random sparse‑set search.
    # ------------------------------------------------------------
    # We explore a range of densities.
    min_frac = 0.04   # 4 % of the positions
    max_frac = 0.15   # 15 %
    for _ in range(num_trials):
        # Random density for this trial.
        frac = rng.random() * (max_frac - min_frac) + min_frac
        m = max(2, int(n * frac))          # at least two ones

        # Random subset of positions.
        idx = rng.choice(n, size=m, replace=False)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = 1.0

        # Compute the peak of the (ordered) self‑convolution.
        max_c = max_convolution_count(cand)

        # Quick score estimate; cheaper than a full evaluate_sequence call.
        cand_score = score_from_counts(m, max_c)

        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # Early exit if we already beat the target bound.
            if best_score < 1.5:
                break

    # ------------------------------------------------------------
    # Final verification with the authoritative evaluator.
    # ------------------------------------------------------------
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score) or final_score > best_score:
        # Something went wrong – fall back to the original parent.
        best = parent

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
