import numpy as np

def _next_pow2(x: int) -> int:
    """Small helper to get the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑2 FFT (real‑valued)."""
    # rfft works on real input and is a little faster than full FFT.
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * a.size - 1]
    # Numerical noise can produce tiny negatives – clip to 0.
    return np.maximum(conv, 0.0)


def _score(a: np.ndarray, size: int) -> float:
    """
    Compute the objective 2 * n * max(a*a) / (sum(a)**2) using FFT.
    a must be non‑negative; the function does not clip – callers must ensure it.
    """
    n = a.size
    conv = _conv_full(a, size)
    max_b = float(conv.max())
    s = float(a.sum())
    if s == 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, num_steps: int = 5000, **kwargs):
    """
    Return a non‑empty list of non‑negative heights that (hopefully) improve
    on the naïve uniform bound.  The routine uses a cheap greedy/annealing
    scheme that repeatedly moves mass away from the shift where the current
    autocorrelation attains its maximum.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial vector ----------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (
            parent.ndim != 1
            or parent.size == 0
            or not np.isfinite(parent).all()
        ):
            a = np.ones(256, dtype=np.float64)          # fallback
        else:
            a = np.clip(parent, 0.0, 1000.0)
    else:
        a = np.ones(256, dtype=np.float64)

    if a.sum() < 1e-12:                     # guarantee a positive sum
        a[:] = 1.0

    n = a.size
    size = _next_pow2(2 * n - 1)            # FFT length (power of two)

    best = a.copy()
    best_score = _score(best, size)

    # ------------------------------------------------------------------
    # Main optimisation loop --------------------------------------------
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # current convolution information
        fft_a = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
        conv = np.maximum(conv, 0.0)
        k_max = int(np.argmax(conv))
        max_val = float(conv[k_max])
        cur_score = 2.0 * n * max_val / (a.sum() * a.sum())

        # Early exit if we already beat the target.
        if best_score < 1.50274349:
            break

        # ----------------------------------------------------------------
        # Identify the pair (i,j) that contributes most to this maximal shift.
        # ----------------------------------------------------------------
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_max - i_range
        products = a[i_range] * a[j_range]
        idx = int(np.argmax(products))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # ----------------------------------------------------------------
        # Build a candidate by moving a fraction of mass from (i,j) to
        # positions that do **not** affect the current maximal shift.
        # ----------------------------------------------------------------

        red_frac = rng.random() * 0.3 + 0.1
        delta_i = a[i] * red_frac
        delta_j = a[j] * red_frac
        total_delta = delta_i + delta_j

        cand = a.copy()
        cand[i] = max(cand[i] - delta_i, 0.0)
        cand[j] = max(cand[j] - delta_j, 0.0)

        # Regions that are *outside* the contributing window.
        left_len = i_start                     # indices 0 … i_start‑1
        right_len = n - 1 - i_end              # indices i_end+1 … n‑1

            cand[dest_left] += amount
            allocated += amount
        if right_len > 0:

            amount = total_delta - allocated   # whatever is left
            cand[dest_right] += amount
            allocated += amount

    return [float(x) for x in best]
