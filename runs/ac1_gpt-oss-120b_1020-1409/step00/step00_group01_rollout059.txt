import numpy as np

def _initial_candidate(n: int) -> np.ndarray:
    """Decreasing 1/√i shape – a classic low‑AC1 construction."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    return np.clip(a, 0.0, 1000.0)


def _max_convolution_index(a: np.ndarray) -> int:
    """Return the index where a*a attains its maximum."""
    conv = np.convolve(a, a)
    return int(conv.argmax())


def _gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution value w.r.t. ``a``.
    For the active shift k the gradient component i is a_{k‑i} (zero if out‑of‑range).
    """
    n = a.size
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad[idx] = a[k - idx]
    return grad


def propose_candidate(seed: int = 42, num_steps: int = 800, **kwargs):
    """
    Memory‑bounded local search that starts from the
    1/√i construction and follows a sub‑gradient descent on the
    active convolution peak.  Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – use the provided parent if available.
    try:
        n = int(height_sequence_1.shape[0])  # type: ignore[name-defined]
    except Exception:
        n = 256

    # Initial candidate (the analytically good shape).
    a = _initial_candidate(n)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    stale = 0

    for _ in range(num_steps):
        # Locate the currently active convolution index.
        k = _max_convolution_index(a)

        # Sub‑gradient that would reduce the active peak while keeping
        # the total sum unchanged (direction sums to zero).
        grad = _gradient_at_max(a, k)
        mean_grad = grad.mean()
        direction = -grad + mean_grad

        # Step size – small and slightly randomised.
        eta = 0.05 * (0.5 + rng.random())
        cand = a + eta * direction

        # Occasional random block perturbation to escape plateaux.
        if rng.random() < 0.1:
            width = int(rng.integers(1, max(2, n // 20)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.08)))
            cand[start:start + width] *= factor

        # Enforce problem constraints.
        cand = np.clip(cand, 0.0, 1000.0)

        # Exact evaluation – the only place the upper‑bound is computed.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            a = cand
            stale = 0
        else:
            stale += 1
            # Occasionally accept a worse move to keep exploring.

                a = cand
