def generate_candidate_max_K(n, rng, target_K, max_pair_mult, max_trials):
    best_seq = None
    best_K = 0
    for _ in range(10):  # restarts
        sum_counts = np.zeros(2*n-1, dtype=np.int16)
        has_square = np.zeros(2*n-1, dtype=np.bool_)
        pos = []
        pos_set = set()
        attempts = 0
        while len(pos) < target_K and attempts < max_trials:
            p = int(rng.integers(0, n))
            if p in pos_set:
                attempts += 1
                continue
            # check constraints
            conflict = False
            for q in pos:
                s = p + q
                if sum_counts[s] >= max_pair_mult:
                    conflict = True
                    break
            if conflict:
                attempts += 1
                continue
            # accept
            pos.append(p)
            pos_set.add(p)
            for q in pos[:-1]:   # existing q
                s = p + q
                sum_counts[s] += 1
            has_square[2*p] = True  # square will exist
            attempts = 0  # reset attempts maybe to allow more tries
        if len(pos) > best_K:
            best_K = len(pos)
            best_seq = np.zeros(n, dtype=np.float64)
            best_seq[pos] = 1.0
        # if we reached target_K, break early
        if best_K >= target_K:
            break
    return best_seq, best_K

import numpy as np

def generate_binary_sparse(n, rng, max_sum_mult, target_k, max_trials=200000):
    sum_counts = np.zeros(2*n - 1, dtype=np.int16)
    positions = []
    pos_set = set()
    trials = 0
    while len(positions) < target_k and trials < max_trials:
        p = int(rng.integers(0, n))
        if p in pos_set:
            trials += 1
            continue
        # check constraints
        ok = True
        for q in positions:
            s = p + q
            if sum_counts[s] >= max_sum_mult:
                ok = False
                break
        if not ok:
            trials += 1
            continue
        # Accept
        for q in positions:
            s = p + q
            sum_counts[s] += 1
        positions.append(p)
        pos_set.add(p)
        trials = 0
    # Build array
    seq = np.zeros(n, dtype=np.float64)
    if positions:
        seq[positions] = 1.0
    return seq, len(positions)

    if not np.isfinite(best_score):

        # fallback to uniform
        best_seq = np.full(n, 1.0, dtype=np.float64)

    # Define search parameters
    # Desired max unordered multiplicity per sum (2 ensures M ≤5)
    max_sum_mult = 2

import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a binary (0/1) sequence with bounded sum‑multiplicity.
# The constraint:
#   for each possible sum s = i + j (i < j) the number of unordered pairs
#   (i, j) with that sum never exceeds max_unordered_mult.
# This keeps the maximum entry of the autocorrelation (convolution) low.
# ----------------------------------------------------------------------
def _build_sparse_binary(n, rng, max_unordered_mult, target_k, max_trials=200_000):
    """
    Greedy random construction.
    Returns a NumPy array of length n containing 0.0/1.0 and the actual count.
    """
    # unordered‑pair counters for each possible sum (0 … 2*n‑2)
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

    positions = []          # list of selected indices
    pos_set = set()         # O(1) membership test
    trials = 0

    while len(positions) < target_k and trials < max_trials:
        p = int(rng.integers(0, n))          # candidate index

        if p in pos_set:                     # already chosen
            trials += 1
            continue

        # verify that adding p would not violate the per‑sum bound
        ok = True
        for q in positions:
            s = p + q
            if sum_counts[s] >= max_unordered_mult:
                ok = False
                break

        if not ok:
            trials += 1
            continue

        # accept p: update counters for all new unordered pairs
        for q in positions:
            s = p + q
            sum_counts[s] += 1

        positions.append(p)
        pos_set.add(p)
        trials = 0                     # reset after a successful insertion

    # materialise the sequence
    seq = np.zeros(n, dtype=np.float64)
    if positions:
        seq[positions] = 1.0
    return seq, len(positions)


# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative step‑function heights.
    The routine first tries the supplied parent `height_sequence_1`,
    then searches for a sparse binary pattern with low autocorrelation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) initialise from the provided construction (fallback to uniform)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size
    except Exception:                      # height_sequence_1 not present
        n = 256
        parent = np.ones(n, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    best_seq = parent.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = float("inf")

    if not np.isfinite(best_score) or best_seq.sum() < 0.01:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 2) search for a sparse pattern with low maximum convolution entry
    # ------------------------------------------------------------------
    # target: at most 2 unordered pairs per sum (max_unordered_mult = 2)
    # which guarantees max(conv) ≤ 5 (two ordered pairs + possible square)
    max_unordered_mult = 2
    # we aim for roughly 200 spikes, but never exceed a safe bound
    target_k = min(200, n // 20)

    for _ in range(12):                     # a few independent restarts
        cand_seq, k_obtained = _build_sparse_binary(
            n, rng, max_unordered_mult, target_k, max_trials=200_000
        )
        if k_obtained == 0:
            continue

        # evaluate – the evaluator does all necessary clipping & checks
        try:
            cand_score = float(evaluate_sequence(cand_seq.tolist()))
        except Exception:
            cand_score = float("inf")

        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq = cand_seq
            best_score = cand_score
            # early exit if we already beat a generous threshold
            if best_score < 1.6:
                break

    # ------------------------------------------------------------------
    # 3) final sanity: ensure the returned list is non‑empty and finite
    # ------------------------------------------------------------------
    if best_seq.sum() < 0.01 or not np.isfinite(best_score):
        # final fallback – a simple uniform vector (guaranteed finite score)
        best_seq = np.full(n, 1.0, dtype=np.float64)

    return [float(x) for x in best_seq]
