import numpy as np

# ---------- helper routines (top‑level, no closures) ----------

def _conv_full_fft(v: np.ndarray) -> np.ndarray:
    """Return the full linear convolution v*v using a power‑of‑2 FFT."""
    n = v.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    fv = np.fft.rfft(v, size)                       # real FFT
    conv = np.fft.irfft(fv * fv, size)[: 2 * n - 1]  # inverse, truncate
    return conv


def _score(v: np.ndarray) -> float:
    """
    Compute the exact objective used by the evaluator:
        2 * n * max(v*v) / (sum(v)**2)
    Returns np.inf for illegal vectors.
    """
    s = v.sum()
    if s < 0.01:
        return np.inf
    conv = _conv_full_fft(v)
    m = conv.max()
    n = v.shape[0]
    return float(2.0 * n * m / (s * s))

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights that (hopefully)
    improve on the baseline bound of 2.0.
    The routine uses a cheap exponential family search followed by a
    lightweight hill‑climbing refinement that never requires dense O(n²)
    matrices.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Base sequence: use the supplied parent if it looks sane,
    #    otherwise fall back to a uniform vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    best_seq = parent.copy()
    best_score = float(_score(best_seq))

    n = best_seq.size
    indices = np.arange(n, dtype=np.float64)

    # 2) Scan a simple exponential family a_i = exp(-alpha * i)
    #    (alpha from 1e‑5 to 1e‑2).  This already beats the uniform bound.
    # ------------------------------------------------------------------
    alphas = np.logspace(-5, -2, num=80)   # 1e‑5 … 1e‑2
    for alpha in alphas:

            continue
        sc = float(_score(cand))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # ------------------------------------------------------------------

    #    convolution entry and move a tiny amount of mass away from the
    #    pair that contributes most to it.
    # ------------------------------------------------------------------
    a = best_seq.copy()

    if a.sum() < 0.01:
        a = np.ones(n, dtype=np.float64)

    # keep the best we have seen so far
    best_seq = a.copy()
    best_score = _score(a)

    for _ in range(2000):
        # full convolution (FFT, O(n log n))
        conv = _conv_full_fft(a)
        k_max = int(np.argmax(conv))
