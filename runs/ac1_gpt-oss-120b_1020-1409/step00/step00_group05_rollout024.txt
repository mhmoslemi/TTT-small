def make_block_perturbation(base, rng):
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative factor for each element in segment
    factor = float(np.exp(rng.normal(0.0, 0.05)))  # small sigma
    cand[start:stop] *= factor
    # optionally random symmetry flip as before.
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function sequence that achieves a low AC1 bound.
    Uses a linear ramp (1, 2, …, n) scaled to fit the [0, 1000] range.
    If the ramp somehow evaluates to an invalid score, falls back to a uniform
    vector of ones.
    """
    # Determine the length n from the provided parent if possible.
    try:
        n = int(len(height_sequence_1))
        if n <= 0:
            raise ValueError
    except Exception:
        n = 256  # safe default

    # Linear ramp: values 1..n scaled so the maximum is exactly 1000.
    scale = 1000.0 / n
    candidate = np.arange(1, n + 1, dtype=np.float64) * scale
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Verify the candidate with the authoritative evaluator.
    # Keep it as the best (only) candidate unless it is invalid.
    try:
        score = evaluate_sequence(candidate.tolist())
    except Exception:
        score = float("inf")

    if not np.isfinite(score):
        # Fallback: a simple uniform positive vector.
        candidate = np.ones(n, dtype=np.float64)
        candidate = np.clip(candidate, 0.0, 1000.0)

    # Return a plain Python list of floats as required.
    return [float(x) for x in candidate]
