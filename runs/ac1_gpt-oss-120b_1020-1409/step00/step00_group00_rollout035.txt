    conv = compute_conv(a)
    m = conv.argmax()
    # compute contributions list
    contrib = np.zeros_like(a)

def propose_candidate(seed=42, num_steps=2000):

    n = height_sequence_1.shape[0] if 'height_sequence_1' in globals() else 4806
    # initialize base

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # precompute FFT size
    mfft = int(1 << ((2*n-1).bit_length()))
    for step in range(num_steps):
        # compute convolution of best
        fft = np.fft.rfft(best, mfft)
        conv = np.fft.irfft(fft*fft, mfft)[:2*n-1]
        max_idx = int(conv.argmax())
        # compute contributions to bin max_idx
        # create array of indices i where j = max_idx - i within bounds
        i_vals = np.arange(n)
        j_vals = max_idx - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        # compute contribution product for each i
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[j_vals[mask]]
        # find index with max contribution
        i_max = int(contrib.argmax())
        j_max = max_idx - i_max
        # choose target indices to receive mass
        p, q = rng.choice(n, size=2, replace=False)
        # ensure p,q different from i_max,j_max? we can just pick any.
        # Choose fractions to move
        # Use step size adaptively decreasing
        alpha = 0.05 * (0.99 ** step)  # decreasing schedule
        # compute amounts to move
        delta_i = alpha * best[i_max]
        delta_j = alpha * best[j_max] if j_max != i_max else 0.0
        # ensure we don't go negative
        delta_i = min(delta_i, best[i_max])
        delta_j = min(delta_j, best[j_max])
        # apply changes
        candidate = best.copy()
        candidate[i_max] -= delta_i
        if j_max != i_max:
            candidate[j_max] -= delta_j
        candidate[p] += delta_i
        if j_max != i_max:
            candidate[q] += delta_j
        # clip
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        # compute score
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

import numpy as np

def _conv_max(arr, fft_len):
    """Return max of the linear convolution arr*arr using FFT of length fft_len."""
    # Real FFT convolution (O(n log n))
    f = np.fft.rfft(arr, n=fft_len)
    c = np.fft.irfft(f * f, n=fft_len)[: 2 * arr.size - 1]
    return c.max(), c

def _make_uniform(n):
    """Uniform non‑negative vector (sum = 1)."""
    a = np.full(n, 1.0, dtype=np.float64)
    return a

def propose_candidate(seed=42, num_steps=5000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 bound.
    Returns a list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Base sequence ----------------------------------------------------
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.size)          # parent provided by the harness
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        # fallback: a modest length if the harness does not provide it
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Ensure parent is a valid candidate (non‑negative, finite, non‑empty)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(n, dtype=np.float64)

    # Clip to the evaluator's allowed range
    parent = np.clip(parent, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Initialise the search (uniform vector, but keep the parent as a
    # possible fallback)
    # ------------------------------------------------------------------
    a = _make_uniform(n)                # sum = n
    a *= (parent.sum() / a.sum())       # rescale to the same total as the parent
    a = np.clip(a, 0.0, 1000.0)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # also consider the raw parent as an initial candidate
    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # FFT length – next power of two ≥ 2·n‑1 (used for every convolution)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Main optimisation loop – sub‑gradient style moves with a tiny
    # simulated‑annealing acceptance rule.
    # ------------------------------------------------------------------
    temperature = 0.1  # initial temperature for occasional uphill steps

    for step in range(num_steps):
        # --------------------------------------------------------------
        # Identify the most offending convolution index (the one attaining
        # the current maximum)

        cur_max, conv = _conv_max(best, fft_len)

        mask = (partner >= 0) & (partner < n)
