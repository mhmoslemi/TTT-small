import numpy as np

def make_block_perturbation(base, rng):

def gradient_step(a, step_scale=0.5):
    n = a.shape[0]
    c = np.convolve(a, a)
    max_c = np.max(c)
    k0 = int(np.argmax(c))
    # compute shifted version
    idx = np.arange(n)
    j = k0 - idx
    mask = (j >= 0) & (j < n)
    b = np.where(mask, a[j], 0.0)
    sum_a = a.sum()
    if sum_a == 0:
        return a.copy()
    factor1 = (4.0 * n) / (sum_a * sum_a)
    factor2 = (4.0 * n * max_c) / (sum_a * sum_a * sum_a)
    grad = factor1 * b - factor2
    step = step_scale * sum_a / (4.0 * n)
    new_a = a - step * grad
    new_a = np.clip(new_a, 0.0, 1000.0)
    return new_a

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # initialize
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # fallback in case best_score is infinite (shouldn't)
    if not np.isfinite(best_score):
        best = np.ones_like(parent) * (1.0 / parent.size)
        best_score = float(evaluate_sequence(best.tolist()))
    # probabilities
    prob_gd = 0.4
    for step in range(num_steps):
        if rng.random() < prob_gd:
            cand = gradient_step(best, step_scale=0.5)
        else:
            cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
    return [float(x) for x in best]

def gradient_flatten_step(a, orig_sum, step_size=0.1):
    n = len(a)
    # compute convolution
    c = np.convolve(a, a)
    mu = (orig_sum**2) / (2 * n - 1)   # same as c.mean() but avoid recompute.
    diff = c - mu
    diff_rev = diff[::-1]
    conv_rev = np.convolve(a, diff_rev)
    idx = (2 * n - 2) - np.arange(n)
    grad = 4.0 * conv_rev[idx]
    # Step size scaling: step = step_size * orig_sum / (np.max(np.abs(grad)) + 1e-12)

    step = step_size * orig_sum / n

import numpy as np

def make_block_perturbation(base, rng):
    candidate = base.copy()
    n = candidate.size
    # pick a block width at least 1% of n, at most ~1/12 of n
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

def renormalize(vec, target):
    s = vec.sum()
    if s > 0.0:
        return vec * (target / s)
    return vec

def gradient_flatten_step(a, target_sum, step_scale=0.2):
    n = a.shape[0]
    # compute convolution and its deviation from the uniform level
    c = np.convolve(a, a)
    # theoretical mean of convolution
    mu = (target_sum * target_sum) / (2 * n - 1)
    diff = c - mu
    # convolution of a with reversed diff
    conv = np.convolve(a, diff[::-1])
    idx = (2 * n - 2) - np.arange(n)
    grad = 4.0 * conv[idx]
    # direction of descent (negative gradient)
    norm = np.linalg.norm(grad) + 1e-12
    direction = grad / norm
    step = step_scale * target_sum / n
    a_new = a - step * direction
    a_new = np.clip(a_new, 0.0, 1000.0)
    a_new = renormalize(a_new, target_sum)
    return a_new

import numpy as np

def make_block_perturbation(base, rng):
    """Create a small O(n) random perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def renormalize(vec, target):
    """Scale ``vec`` so that its sum equals ``target`` (if possible)."""
    s = vec.sum()
    if s > 0.0:
        return vec * (target / s)
    return vec


def gradient_flatten_step(a, target_sum, step_scale=0.2):
    """
    One gradient‑descent step that tries to flatten the autocorrelation
    (minimise the L2 variance of a*a around its mean).
    """
    n = a.shape[0]
    # full convolution a*a
    conv = np.convolve(a, a)
    # theoretical mean of the convolution (independent of a)
    mu = (target_sum * target_sum) / (2 * n - 1)
    diff = conv - mu                                 # deviation from uniform level
    # convolution of a with the reversed deviation
    rev = diff[::-1]
    tmp = np.convolve(a, rev)                       # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)                # pick the needed slice
    grad = 4.0 * tmp[idx]                           # gradient of the L2 loss
    # take a unit step in the negative gradient direction
    norm = np.linalg.norm(grad) + 1e-12
    direction = grad / norm
    step = step_scale * target_sum / n
    a_new = a - step * direction
    a_new = np.clip(a_new, 0.0, 1000.0)
    a_new = renormalize(a_new, target_sum)
    return a_new
