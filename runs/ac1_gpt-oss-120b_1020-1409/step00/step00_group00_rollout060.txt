import numpy as np

def make_block_perturbation(base, rng):
    candidate = np.copy(base)
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.12))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

import numpy as np

def make_block_perturbation(base, rng):
    """Return a cheap O(n) perturbation of ``base``."""
    cand = np.copy(base)
    n = cand.size
    # random block width, at least 1 and up to n/12 (≈400 for n≈4800)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal tweak
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= factor
    # occasional symmetrisation – cheap and keeps values bounded
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    # ------------------------------------------------------------------
