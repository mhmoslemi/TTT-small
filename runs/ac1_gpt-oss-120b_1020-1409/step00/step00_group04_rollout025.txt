def propose_candidate(seed=42, attempts=20, target_k=170, s_max=2):
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    best_mask = None
    best_score = np.inf
    for attempt in range(attempts):
        # random order for processing maybe shuffle candidate positions.
        mask = np.zeros(n, dtype=np.int8)
        sum_counts = {}
        selected = []
        # random permutation of indices to try selection order
        indices = rng.permutation(n)
        for idx in indices:
            # attempt to add idx if not exceed s_max
            # compute future sums
            ok = True
            # compute self sum
            s_self = idx + idx
            if sum_counts.get(s_self, 0) >= s_max:
                ok = False
            else:
                # check sums with selected
                for j in selected:
                    s = idx + j
                    if sum_counts.get(s, 0) >= s_max:
                        ok = False
                        break
            if not ok:
                continue
            # Add this index
            mask[idx] = 1
            selected.append(idx)
            # Update sum_counts
            # self sum
            sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
            for j in selected[:-1]:  # previous selected indices only
                s = idx + j
                sum_counts[s] = sum_counts.get(s, 0) + 1
            # If reached target_k break
            if len(selected) >= target_k:
                break
        # Possibly add more by random attempts albeit may not achieve target_k.
        k = len(selected)
        if k == 0:
            continue
        # Evaluate candidate score
        candidate = mask.astype(np.float64)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_mask = mask.copy()

    if best_mask is None:
        # fallback use parent
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
        return [float(x) for x in parent]
    # Optionally we could scale up values to increase sum while preserving max? Not needed because binary works.
    # Return as list of floats
    result = best_mask.astype(np.float64)
    # Clip to bounds (already 0/1)
    return [float(x) for x in result]

import numpy as np

# ------------------------------------------------------------
# Helper: fast estimation of the AC1 bound for a binary mask.
# ------------------------------------------------------------
def _estimate_score(mask, n, s_max):
    """
    Compute 2*n*max_convolution/(sum^2) using only unordered‑pair counts.
    mask : 0/1 NumPy array
    n    : length of the sequence
    s_max: allowed maximum unordered pair count per sum
    """
    # unordered pair counters (including self‑pairs)
    sum_counts = {}
    self_counts = {}
    selected = np.nonzero(mask)[0]
    k = selected.size
    if k == 0:
        return np.inf

    # Build the sum_counts dictionaries (same logic as in the construction)
    for idx in selected:
        s_self = idx + idx
        sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
        self_counts[s_self] = self_counts.get(s_self, 0) + 1
        for j in selected:
            if j == idx:
                continue
            s = idx + j
            sum_counts[s] = sum_counts.get(s, 0) + 1

    # ordered pair count = 2*unordered - self (self counted once)
    max_ordered = 0
    for s, unordered in sum_counts.items():
        ordered = 2 * unordered - self_counts.get(s, 0)
        if ordered > max_ordered:
            max_ordered = ordered

    return 2.0 * n * max_ordered / (k * k)


# ------------------------------------------------------------
# Core construction: greedy sparse mask with bounded sum multiplicities.
# ------------------------------------------------------------
def _build_mask(n, rng, s_max=5, target_score=1.5, attempts=300):
    best_mask = None
    best_score = np.inf

    # Upper bound on how many elements we would ever need to try.
    # The theoretical limit k^2 ≤ 2*s_max*(2n-1) gives a safe ceiling.
    max_possible_k = int(np.sqrt(2 * s_max * (2 * n - 1))) + 1

    for _ in range(attempts):
        mask = np.zeros(n, dtype=np.int8)
        sum_counts = {}
        self_counts = {}
        selected = []

        # Random permutation of all indices – a different order each attempt.
        order = rng.permutation(n)

        for idx in order:
            # Quick abort if we have already reached the theoretical ceiling.
            if len(selected) >= max_possible_k:
                break

            # Check whether adding idx would exceed the per‑sum limit.
            ok = True
            s_self = idx + idx
            if sum_counts.get(s_self, 0) >= s_max:
                ok = False
            else:
                for j in selected:
                    s = idx + j
                    if sum_counts.get(s, 0) >= s_max:
                        ok = False
                        break

            if not ok:
                continue

            # Accept the index.
            mask[idx] = 1
            selected.append(idx)

            # Update counters.
            sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
            self_counts[s_self] = self_counts.get(s_self, 0) + 1
            for j in selected[:-1]:               # all previous elements
                s = idx + j
                sum_counts[s] = sum_counts.get(s, 0) + 1

        k = len(selected)
        if k == 0:
            continue

        # Estimate the score without a full convolution.
        # This is exact for a binary mask because max convolution == max ordered.
        max_ord = 0
        for s, unordered in sum_counts.items():
            ordered = 2 * unordered - self_counts.get(s, 0)
            if ordered > max_ord:
                max_ord = ordered
        score = float(2 * n * max_ord / (k * k))

        if score < best_score:
            best_score = score
            best_mask = mask.copy()
            if best_score <= target_score:
                break

    return best_mask, best_score
