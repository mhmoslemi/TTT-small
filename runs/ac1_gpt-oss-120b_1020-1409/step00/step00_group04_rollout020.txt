def propose_candidate(seed=42, num_iters=2000, support_min=80, support_max=150, **kwargs):
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1)
    best = np.clip(height_sequence_1.astype(float), 0, 1000)
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Initial random candidate
    for _ in range(num_iters):
        m = rng.integers(support_min, support_max+1)  # inclusive
        # pick random indices unique
        idx = rng.choice(n, size=m, replace=False)
        candidate = np.zeros(n, dtype=float)
        candidate[idx] = 1.0  # value 1
        # compute fast score
        score = compute_score(candidate)
        # verify with evaluator for best candidate only
        if score < best_score:
            # evaluate exact to be safe
            exact = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = candidate
                best_score = exact
    return best.tolist()

def build_sidon_set(n, target_m):
    # Greedy: start empty sets
    taken = []
    sums = set()
    # iterate candidate indices in random order
    rng = np.random.default_rng(seed)
    candidates = np.arange(n)
    rng.shuffle(candidates)
    for i in candidates:
        conflict = False
        for j in taken:
            if (i + j) in sums:
                conflict = True
                break
        if not conflict:
            # check also self sum
            if (i + i) in sums:
                conflict = True
        if not conflict:
            # add
            for j in taken:
                sums.add(i + j)
            sums.add(i + i)
            taken.append(i)
            if len(taken) >= target_m:
                break
    return taken

import numpy as np

# The evaluator is provided by the harness:
#   evaluate_sequence(sequence: list) -> float

def _fast_score(vec, n, L):
    """Return the AC1 upper bound for a non‑negative vector `vec` (numpy array)."""
    # Convolution via FFT (linear convolution, length 2*n-1)
    f = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_val = conv.max()
    s = vec.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_val / (s * s))


def propose_candidate(seed=42, num_iters=3000,
                      min_support=500, max_support=800, **kwargs):
    """
    Return a list of non‑negative step‑function heights whose AC1 bound
    improves on the supplied parent when possible.

    The search constructs sparse binary vectors and keeps the best one
    encountered.  Convolution values are evaluated with a fast FFT trick;
    the exact evaluator is consulted only for candidates that beat the
    current best score.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Parent vector (fallback)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:                         # pragma: no cover
        # If the runtime does not supply `height_sequence_1`, fall back to ones.
        parent = np.ones(256, dtype=np.float64)

    best_vec = parent.copy()
    best_score = float(evaluate_sequence(best_vec.tolist()))

    n = len(parent)
    # Length for the FFT (next power of two >= 2*n-1)
    L = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2.  Random sparse candidates
    # ------------------------------------------------------------------
    for _ in range(num_iters):
        # Random support size in the chosen interval
        m = int(rng.integers(min_support, max_support + 1))
        # Random, non‑repeating indices
        idx = rng.choice(n, size=m, replace=False)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = 1.0                     # binary heights; scale is irrelevant

        # Fast (approximate) score
        approx = _fast_score(cand, n, L)

        # Only bother with the exact (expensive) evaluation if we are better
        if approx < best_score:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_vec = cand
                best_score = exact

    # ------------------------------------------------------------------
    # 3.  Simple local refinement (optional, cheap)
    # ------------------------------------------------------------------
    # Try a few random swaps of a 1↔0 to see if the max can be lowered.
    # This uses the fast scorer; any improvement is verified exactly.
    for _ in range(200):
        ones = np.flatnonzero(best_vec > 0)
        zeros = np.flatnonzero(best_vec == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i_out = rng.choice(ones)
        i_in  = rng.choice(zeros)
        cand = best_vec.copy()
        cand[i_out] = 0.0
        cand[i_in]  = 1.0
        approx = _fast_score(cand, n, L)
        if approx < best_score:
            exact = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best_vec = cand
                best_score = exact

    # Return a plain Python list of floats as required by the harness.
    return [float(v) for v in best_vec]
