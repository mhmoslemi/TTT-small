    rng = np.random.default_rng(seed)
    try:

        n = 256
        base = np.ones(n, dtype=np.float64)
    base = np.clip(base, 0.0, 1000.0)
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size
    # parameters
    beta = 0.6          # reduction factor for high-contributing entries
    frac = 0.1          # fraction of entries to modify per iteration
    for it in range(num_iters):
        # compute convolution of current best
        conv = np.convolve(best, best)
        max_idx = int(conv.argmax())
        # compute contributions for this shift
        i = np.arange(n)
        j = max_idx - i
        valid = (j >= 0) & (j < n)
        contrib = np.zeros_like(best)
        contrib[valid] = best[i[valid]] * best[j[valid]]
        # select top contributing positions
        num_sel = max(1, int(frac * n))
        # Use argpartition for speed
        sel_idx = np.argpartition(contrib, -num_sel)[-num_sel:]
        # make candidate copy
        cand = best.copy()
        # reduce selected contributions
        cand[sel_idx] *= beta
        # keep total L1 sum unchanged by scaling
        sum_before = best.sum()
        sum_after = cand.sum()
        if sum_after > 0:
            cand *= (sum_before / sum_after)
        # clip to allowed range
        np.clip(cand, 0.0, 1000.0, out=cand)
        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
        # occasional random perturbation to escape plateaus
        if it % 100 == 0 and it > 0:
            # small random jitter
            jitter = rng.normal(scale=0.05, size=n)
            cand2 = best * (1.0 + jitter)
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand2
                best_score = score2
    return best.tolist()

    Heuristic min‑max search for a step‑function that lowers the AC1 bound.
    The algorithm repeatedly identifies the shift where the convolution attains

    plateaus.  The returned list is guaranteed to satisfy the evaluator’s

    base = np.clip(base, 0.0, 1000.0)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # Target bound – stop early if we reach it.
    target_score = 1.50274349

    # ----------------------------------------------------------------------
    # Main optimisation loop

    for it in range(num_iters):
        # Dynamically change aggressiveness during the run
        if it < num_iters // 2:
            beta = 0.6       # reduction factor for high‑impact entries
            frac = 0.10      # fraction of entries to touch
        else:

        # Full convolution of the current candidate
        conv = np.convolve(best, best)
        max_idx = int(conv.argmax())               # shift with the largest value
