# evaluate_sequence is provided by the harness


def _make_power_sequence(n: int, power: int) -> np.ndarray:
    """Generate a decreasing step‐function a_i = 1000 * (1 - i/(n-1))**power."""
    if n <= 1:
        return np.full(n, 1000.0 if n == 1 else 0.0, dtype=np.float64)

    idx = np.arange(n, dtype=np.float64)
    # (1 - i/(n-1)) is guaranteed in [0,1]; raise to integer power
    seq = 1000.0 * ((1.0 - idx / (n - 1)) ** power)
    return np.clip(seq, 0.0, 1000.0)
