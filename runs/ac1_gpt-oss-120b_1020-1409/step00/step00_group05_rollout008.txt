import numpy as np

def _uniform_array(length, value=1.0):
    """Return a constant non‑negative array of given length."""
    return np.full(length, float(value), dtype=np.float64)


def _make_block_perturbation(base, rng):
    """
    Produce an O(n)‑memory perturbation of ``base``.
    A random contiguous block is multiplied by a log‑normal factor;
    with some probability the result is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # block width between 1 and roughly n/20 (but at least 1)
    max_width = max(2, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.10)))  # modest multiplicative change
    cand[start:start + width] *= factor
    # occasional mirror‑symmetry smoothing
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (approximately)
    minimizes the AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise a parent sequence.  If the runtime supplies
    # ``height_sequence_1`` we use its length; otherwise fall back to 256.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = _uniform_array(256, value=1.0)

    n = parent.size

    # ------------------------------------------------------------------
    # The theoretical optimum for non‑negative entries is a constant array,
    # which yields a score of exactly 2.  Start from that optimum as a safe
    # fallback, then optionally replace it with the provided parent if it is
    # already as good.
    # ------------------------------------------------------------------
    best = _uniform_array(n, value=1.0)
    best_score = float(evaluate_sequence(best.tolist()))

    parent_score = float(evaluate_sequence(parent.tolist()))
