import numpy as np

def propose_candidate(seed=42, num_steps=5000, **kwargs):
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            n = 256
        else:
            n = parent.size
    else:
        n = 256
    # We'll use n length.
    # Use base value c = 1.0 (any positive, scaling invariant)
    base = 1.0

    half = n // 2
    # Initialize anti-symmetric deltas
    delta = rng.uniform(-base * 0.4, base * 0.4, size=half)
    # Ensure positivity after adding/subtracting
    a = np.empty(n, dtype=np.float64)
    a[:half] = base + delta
    a[-half:] = base - delta[::-1]   # assign reversed to maintain anti-symmetry
    if n % 2 == 1:
        a[half] = base  # middle element

    a = np.clip(a, 0.0, 1000.0)  # ensure within bounds

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    for _ in range(num_steps):
        # pick random pair index
        i = rng.integers(0, half)
        # propose new delta for this pair
        new_delta = delta[i] + rng.normal(scale=base * 0.1)
        # clip new_delta to [-base, base] to keep values nonnegative
        new_delta = max(-base, min(base, new_delta))
        # construct candidate
        cand = best.copy()
        cand[i] = base + new_delta
        cand[n-1-i] = base - new_delta
        # if odd and i == half (not possible), skip
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            delta[i] = new_delta

    return [float(x) for x in best]
