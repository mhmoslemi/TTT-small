def propose_candidate(seed=42, num_trials=200, max_allowed=2, **kwargs):

    Search for a sparse 0/1 step sequence with a low
    2·n·max(conv)/sum^2 value.  The routine builds a set of indices
    whose pairwise sums (ordered) never exceed ``max_allowed``.
    """
    rng = np.random.default_rng(seed)

    # ---- fallback parent -------------------------------------------------
    parent_arr = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)

        except Exception:
            parent_arr = None
    if parent_arr is None:
        parent_arr = np.ones(256, dtype=np.float64)

    best = parent_arr.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ---- main stochastic construction ------------------------------------
    for _ in range(num_trials):
        # target number of ones (m) – larger m gives smaller score
        m_target = int(rng.integers(100, 180))

        order = rng.permutation(n)
