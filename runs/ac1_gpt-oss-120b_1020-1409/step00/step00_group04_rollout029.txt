def _determine_length():

    try:

        return int(arr.shape[0])

        # Fallback length if the parent sequence is unavailable.

def _initial_power_law(n):
    """
    Produce a decreasing power‑law sequence a_i = 1 / sqrt(i+1).
    This shape gives a flat self‑convolution and typically yields
    an AC1 bound close to π/2 ≈ 1.57, well below the previous 2.0.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(idx)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Construct a step‑function heights vector and perform a tiny
    stochastic local search to improve the autocorrelation bound.
    The returned list contains non‑negative floats and is guaranteed
    to be a valid input for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    n = _determine_length()
    best = _initial_power_law(n)

    # Exact evaluation of the initial candidate.
    best_score = float(evaluate_sequence(best.tolist()))

    # Small local‑search loop – cheap enough for the given size.
    for _ in range(num_steps):
        # Choose between a swap move and a block‑scaling move.
        if rng.random() < 0.5:
            # ---- swap two random entries ------------------------------------
            i, j = rng.integers(0, n, size=2)
            if i == j:
