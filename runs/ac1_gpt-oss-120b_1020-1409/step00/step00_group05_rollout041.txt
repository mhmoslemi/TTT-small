def conv_full(a):
    n = a.size
    size = 1 << ((2*n -1).bit_length())
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa*fa, size)
    return c[:2*n-1]

        fa = np.fft.rfft(arr, size)
        c = np.fft.irfft(fa * fa, size)
        return c[:2*arr.size - 1]

    n = arr.shape[0]

    c = np.fft.irfft(fa * fa, size)
