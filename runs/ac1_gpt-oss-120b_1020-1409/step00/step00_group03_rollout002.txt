def _initial_sequence():
    """Return a starting non‑negative sequence.
    Prefer the runtime variable `height_sequence_1` if it exists and is valid,
    otherwise fall back to a uniform sequence of length 256."""
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        if seq.ndim != 1 or seq.size == 0:
            raise ValueError
        seq = np.clip(seq, 0.0, 1000.0)
        return seq
    except Exception:
        # 256 is a small safe default length used by the reference code.
        return np.full(256, 1.0, dtype=np.float64)

def propose_candidate(seed=42, **kwargs):
    """Return a list of non‑negative heights that (approximately) minimise the
    AC1 upper bound as defined by `evaluate_sequence`.

    The routine keeps the best candidate seen during a short stochastic
    search and uses a uniform sequence as a guaranteed fallback.
    """
    rng = np.random.default_rng(seed)

    # Number of random perturbation steps – can be overridden via kwargs.
    num_iter = int(kwargs.get("num_iter", 30))

    base = _initial_sequence()
    n = base.size

    # Uniform sequence (value 1.0) attains the theoretical lower bound 2.0.
    uniform = np.full(n, 1.0, dtype=np.float64)
    best = uniform.copy()
    # Evaluate once; the call always returns a finite float for valid data.
    best_score = float(evaluate_sequence(best.tolist()))

    # Try the supplied parent if it exists and has the same length.
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size == n:
            parent_score = float(evaluate_sequence(parent.tolist()))
            if parent_score < best_score:
                best = parent.copy()
                best_score = parent_score
    except Exception:
        pass

    # Simple block‑wise stochastic perturbation.
    for _ in range(num_iter):
        cand = best.copy()

        low = max(1, n // 200)
        high = max(2, n // 12) + 1  # rng.integers upper bound is exclusive
        width = int(rng.integers(low, high))

        stop = min(n, start + width)

        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor

        if rng.random() < 0.35:

        best_score = float(evaluate_sequence(best.tolist()))
