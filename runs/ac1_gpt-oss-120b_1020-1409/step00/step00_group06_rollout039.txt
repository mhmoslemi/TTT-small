import numpy as np

def _score(sequence: np.ndarray) -> float:
    """Return the AC1 upper‑bound score for a non‑negative sequence."""
    n = sequence.size
    s = sequence.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(sequence, sequence)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def _initial_candidates(rng: np.random.Generator, n: int):
    """Generate two simple start vectors: uniform and a clipped parent if present."""
    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # parent from the runtime variable `height_sequence_1` (if it exists)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = uniform.copy()
    if parent.ndim != 1 or parent.size != n or not np.isfinite(parent).all():
        parent = uniform.copy()
    else:
        parent = np.clip(parent, 0.0, 1000.0)

    total = parent.sum()
    if total > 0:
        parent = parent / total
    else:
        parent = uniform.copy()

    return [uniform, parent]


def propose_candidate(seed: int = 42, num_steps: int = 3000, step_frac: float = 0.05, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 bound.

    The routine works without any external files.  It uses the global
    ``height_sequence_1`` (if present) as an alternative starting point and
    falls back to a uniform distribution.  A simple hill‑climbing scheme moves a
    small amount of mass from the index that contributes most to the current
    worst convolution entry to an index that does not affect that entry.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem (use the length of height_sequence_1 if available)
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    # Choose the best of the trivial initial candidates
    best_vec = None
    best_score = np.inf
    for cand in _initial_candidates(rng, n):
        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_vec = cand.copy()

    a = best_vec  # current working vector

    # Hill‑climbing loop
    stagnant = 0
    for _ in range(num_steps):
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))

        # indices that actually participate in this convolution entry
        i_start = max(0, max_idx - (n - 1))
        i_end = min(max_idx, n - 1)
        if i_start > i_end:
            break

        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        partner = max_idx - i_vals
        contrib = a[i_vals] * a[partner]

        src_local = int(np.argmax(contrib))
        source_idx = int(i_vals[src_local])

        # Choose a destination that does **not** affect the current worst entry
        if max_idx < n - 1:
            dest_idx = int(rng.integers(max_idx + 1, n))
        else:
            if i_start > 0:
                dest_idx = int(rng.integers(0, i_start))
            else:
                dest_idx = int(rng.integers(0, n))

        if dest_idx == source_idx:
            dest_idx = (dest_idx + 1) % n

        # Move a fraction of the mass from source to destination
        delta = float(min(a[source_idx] * step_frac, a[source_idx]))
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[source_idx] -= delta
        a_new[dest_idx] += delta

        total = a_new.sum()
        if total <= 0.0:
            continue
        a_new /= total

        new_score = _score(a_new)
        if np.isfinite(new_score) and new_score < best_score:
            a = a_new
            best_score = new_score
            stagnant = 0
        else:
            stagnant += 1

        # Early stop if we are not improving for a while
        if stagnant >= 500:
            break

    # Final safety: clip to the evaluator's allowed range
    final = np.clip(a, 0.0, 1000.0)
    return [float(x) for x in final]
