import numpy as np

def propose_candidate(seed: int = 42, num_iter: int = 2000, **kwargs):
    """
    Search for a non‑negative step‑function that reduces the AC1 upper bound.
    The algorithm keeps the total sum (hence the objective is scale invariant)
    and iteratively transfers a small amount of mass from indices that
    heavily contribute to the current maximal convolution entry to indices that
    contribute little.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1.  Starting sequence -----------------------------------------------
    # ----------------------------------------------------------------------
    # `height_sequence_1` is supplied by the harness; use it when possible.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:               # fallback if the variable is missing
        parent = np.ones(256, dtype=np.float64)

    # Basic safety checks (the evaluator also sanitises the sequence)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    a = np.clip(parent, 0.0, 1000.0)

    # Small random perturbation to break symmetry (important for the first step)
    a *= np.exp(rng.normal(0.0, 0.05, size=a.shape))
    a = np.clip(a, 0.0, 1000.0)

    total_sum = a.sum()
    if total_sum < 0.01:                     # guard against near‑zero sum
        a[:] = 0.01 / a.size
        total_sum = a.sum()

    n = a.size
    L = 2 * n - 1                 # length of the full convolution

    # Full convolution of the current iterate (used for the objective)
    conv = np.convolve(a, a, mode='full')
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # 2.  Main iterative improvement ----------------------------------------
    # ----------------------------------------------------------------------
    # Parameters controlling the magnitude of each transfer.
    step_frac = 0.10               # move at most 10 % of the source value

    for it in range(num_iter):
        # a) Locate the index where the convolution attains its maximum
        k_max = int(np.argmax(conv))

        # b) Build the “partner” vector for the current k_max:
        #    partner[i] = a[k_max - i]   (zero where the index would be out of range)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        partner = np.zeros_like(a)
        partner[mask] = a[idx[mask]]

        # c) Contribution of each position to this maximal entry
        contrib = a * partner

        # d) Choose a source (large contribution) and a destination (small contribution)
        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        # If the selection collapsed (e.g. all equal), pick two random distinct indices
        if i_src == i_dst:
            i_src = int(rng.integers(0, n))
            i_dst = int(rng.integers(0, n))
            while i_dst == i_src:
                i_dst = int(rng.integers(0, n))

        # e) Determine a feasible transfer amount δ
        delta = step_frac * a[i_src]
        # Respect the upper bound 1000 on the destination
        delta = min(delta, 1000.0 - a[i_dst])
        # Ensure we do not drive the source negative
        delta = min(delta, a[i_src])
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # f) Apply the transfer and update the convolution incrementally.
        #    We keep a copy of the current state in case the move is not
        #    improving.
        # ------------------------------------------------------------------
        a_old = a.copy()
        conv_old = conv.copy()

        # Transfer mass
        a[i_src] -= delta
        a[i_dst] += delta

        # Incremental convolution update:
        #   conv_new = conv + 2·δ·(shift_dst - shift_src) + Δ_self
        shift_src = np.zeros(L, dtype=np.float64)
        shift_src[i_src:i_src + n] = a_old
        shift_dst = np.zeros(L, dtype=np.float64)
        shift_dst[i_dst:i_dst + n] = a_old

        conv += 2.0 * delta * (shift_dst - shift_src)

        # Quadratic self‑interaction term (δ²)
        conv[2 * i_src] += delta * delta
        conv[2 * i_dst] += delta * delta
        conv[i_src + i_dst] -= 2.0 * delta * delta

        # g) Evaluate the new objective (the sum is unchanged)
        max_conv = conv.max()
        new_score = float(2.0 * n * max_conv / (total_sum * total_sum))

        # Accept only if we strictly improve the best known score
        if np.isfinite(new_score) and new_score < best_score:
            best = a.copy()
            best_score = new_score
        else:
            # Revert the trial move
            a = a_old
            conv = conv_old

        # Re‑compute convolution from scratch every 100 iterations
        # to avoid numerical drift.
        if (it + 1) % 100 == 0:
            conv = np.convolve(a, a, mode='full')

        # Early exit if we have already crossed the target bound
        if best_score < 1.5028:          # a tiny safety margin
            break

    # ----------------------------------------------------------------------
    # 3.  Return the best sequence found (as a plain Python list) ----------
    # ----------------------------------------------------------------------
    return [float(v) for v in best]
