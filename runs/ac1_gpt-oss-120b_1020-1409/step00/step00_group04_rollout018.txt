import numpy as np

def propose_candidate(seed=42, max_trials=2000, **kwargs):
    """
    Generate a non‑negative step function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.

    The strategy searches for sparse binary patterns whose self‑convolution
    has a small maximum entry.  For a binary vector with K ones the bound
    equals 2·n·M / K² where M is the largest ordered pair count in the
    convolution.  By picking K moderately large (≈ n/20…n/10) random subsets
    usually yield M close to its average K²/(2n), giving a bound close to 1.

    The function keeps the best candidate seen (including the provided
    `height_sequence_1` as a fallback) and stops early if a target bound
    is reached.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise parent (fallback) sequence.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        # If the runtime variable is missing or invalid, use a simple uniform vector.
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)
    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Random sparse‑binary search.
    # ------------------------------------------------------------------
    # Reasonable range for the number of ones.  The lower bound ensures a
    # non‑trivial sum, the upper bound keeps the convolution reasonably flat.
    k_low = max(2, n // 30)          # ≈ 2 % of the length
    k_high = max(k_low + 1, n // 10) # ≈ 10 % of the length

    target = 1.50274349  # early‑stop threshold

    for _ in range(max_trials):
        K = int(rng.integers(k_low, k_high + 1))

        # Choose distinct positions for the ones.
        idx = rng.choice(n, size=K, replace=False)

        candidate = np.zeros(n, dtype=np.float64)
        candidate[idx] = 1.0  # amplitude 1 is sufficient; scaling cancels out

        # Exact evaluation (the only reliable way to respect the evaluator’s
        # exact definition).
        score = float(evaluate_sequence(candidate.tolist()))

        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            if best_score < target:
                break

    # Return a plain Python list of floats, as required.
    return [float(v) for v in best]
