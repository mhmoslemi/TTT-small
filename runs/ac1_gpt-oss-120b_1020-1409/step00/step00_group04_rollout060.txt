    positions = []
    sums = set()
    attempts = 0
    max_attempts = n * 10 maybe.
    while len(positions) < target_m and attempts < max_attempts:
        # random candidate uniformly
        x = rng.integers(0, n)
        # check collision
        if x in positions:
            attempts+=1
            continue
        collides=False
        # compute sums with current positions
        for y in positions:
            if (x+y) in sums:
                collides=True
                break
        if collides:
            attempts+=1
            continue
        # also check self sum
        if (x+x) in sums:
            attempts+=1
            continue
        # passed, add
        for y in positions:
            sums.add(x+y)
        sums.add(x+x)
        positions.append(x)
        attempts=0 # reset after successful addition maybe
    return positions

import numpy as np
import math

# ------------------------------------------------------------
# Helper: generate a Sidon set (all pairwise sums distinct)
# ------------------------------------------------------------
def generate_sidon_positions(n, rng, max_attempts=20000):
    """
    Greedy construction of a Sidon set within {0,…,n-1}.
    Returns a list of positions. Expected size ≈ √n.
    """
    positions = []
    used_sums = set()
    attempts = 0
    target_size = int(math.sqrt(n)) + 2  # a bit generous
    while attempts < max_attempts and len(positions) < target_size:
        x = int(rng.integers(0, n))
        if x in positions:
            attempts += 1
            continue
        conflict = False
        # check new sums against existing ones
        for y in positions:
            if (x + y) in used_sums:
                conflict = True
                break
        if conflict or (x + x) in used_sums:
            attempts += 1
            continue
        # accept x
        for y in positions:
            used_sums.add(x + y)
        used_sums.add(x + x)
        positions.append(x)
        attempts = 0  # reset after a success
    return positions

    # --------------------------------------------------------
    try:
        parent_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, V)
        parent_score = _score(parent_seq)
        if parent_score < best_score:
            best_score = parent_score
            best_seq = parent_seq
    except Exception:
        pass

    # Defensive final fallback (should never be reached)
    if best_seq is None:
        best_seq = np.full(n, 1.0, dtype=np.float64)

    # Return a plain Python list of floats as required
    return [float(x) for x in best_seq]
