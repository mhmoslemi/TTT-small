import numpy as np

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that (locally) minimizes
    the AC1 upper bound as evaluated by the provided `evaluate_sequence` function.
    """
    rng = np.random.default_rng(seed)

    # Determine the sequence length, preferring the supplied parent if present.
    try:
        n = len(height_sequence_1)       # type: ignore[name-defined]
        if n == 0:
            raise ValueError
    except Exception:
        n = 256  # safe fallback length

    # Start from the uniform sequence – it is the theoretical minimizer of the bound.
    best = np.full(n, 1.0, dtype=np.float64)

    # Evaluate the initial candidate (should be finite and equal to 2.0).
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = float('inf')

    # Simple stochastic hill‑climbing; accept only strict improvements.
    for _ in range(num_steps):
        cand = best.copy()
        # Perturb a single coordinate multiplicatively by a small log‑normal factor.
        idx = rng.integers(0, n)
        factor = np.exp(rng.normal(0.0, 0.05))  # modest changes

        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue  # ignore invalid candidates

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list as required.
