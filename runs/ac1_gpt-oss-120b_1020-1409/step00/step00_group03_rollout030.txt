def propose_candidate(seed=42, num_steps=500, alpha=0.1, **kwargs):
    rng = np.random.default_rng(seed)
    # try using parent if exists and valid
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    candidate = np.clip(parent, 0.0, 1000.0)
    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # simple smoothing iteration: candidate <- (1-alpha)*candidate + alpha*mean
    mean_val = candidate.mean()
    for _ in range(num_steps):
        # compute new candidate by moving values toward current mean
        new_candidate = (1 - alpha) * candidate + alpha * mean_val
        # ensure bounds
        np.clip(new_candidate, 0.0, 1000.0, out=new_candidate)
        score = float(evaluate_sequence(new_candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = new_candidate.copy()
            best_score = score
        # update candidate anyway
        candidate = new_candidate
        mean_val = candidate.mean()
    return [float(x) for x in best]

import numpy as np

# The evaluator `evaluate_sequence` is provided by the harness.

def propose_candidate(seed=42, num_iters=200, alpha=0.2, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by `evaluate_sequence`.

    Strategy
    --------
    * Use the length of the supplied `height_sequence_1` if it exists,
      otherwise fall back to a modest size.
    * The uniform sequence is analytically optimal for the ratio
      2·n·max(a∗a)/(∑a)², because it minimises the sum of squares for a fixed
      total mass.
    * Start from the parent (if valid) and iteratively smooth it towards the
      uniform mean; this inexpensive O(n) process guarantees we never worsen
      the best score seen.
    * Return the best sequence observed (uniform or a smoothed variant).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (n) and obtain a sensible parent candidate.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if (parent.ndim != 1 or parent.size == 0 or
                not np.isfinite(parent).all()):
            raise ValueError
    except Exception:
        # If the runtime does not provide a valid parent, use a small default.
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)

    # ------------------------------------------------------------------
    # Build the uniform candidate – analytically optimal.
    # ------------------------------------------------------------------
    uniform_val = 1.0                     # any constant works; 1.0 is safe
    uniform = np.full(n, uniform_val, dtype=np.float64)

    # ------------------------------------------------------------------
    # Clip the parent to the admissible [0, 1000] range.
    # ------------------------------------------------------------------
    parent = np.clip(parent, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Initialise the best‑so‑far record.
    # ------------------------------------------------------------------
    best = uniform.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Consider the raw parent as a possible better candidate.
    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # ------------------------------------------------------------------
    # Simple smoothing iterations: move each entry toward the uniform mean.
    # This is O(n) per iteration and never increases the objective.
    # ------------------------------------------------------------------
    target_mean = uniform.mean()           # constant = uniform_val
    candidate = parent.copy()
    for _ in range(num_iters):
        # Exponential moving average toward the uniform mean.
        candidate = (1.0 - alpha) * candidate + alpha * target_mean
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate.copy()
            best_score = score

    # ------------------------------------------------------------------
    # Return a plain Python list of floats as required by the harness.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
