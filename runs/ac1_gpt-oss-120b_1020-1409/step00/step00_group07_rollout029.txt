def perturb_swap(seq, rng, eps=0.05):
    # move a fraction eps of value from i to j
    n = seq.size
    i = rng.integers(n)
    j = rng.integers(n)
    while j == i:
        j = rng.integers(n)
    # amount at most eps * seq[i]
    amount = seq[i] * rng.uniform(0, eps)
    if amount <= 0:
        return seq
    new = seq.copy()
    new[i] = max(new[i] - amount, 0.0)
    new[j] = min(new[j] + amount, 1000.0)
    return new

def _transfer_perturb(seq: np.ndarray, rng: np.random.Generator, frac: float = 0.10) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = seq.size
    if n < 2:
        return seq.copy()
    i = rng.integers(n)
    j = rng.integers(n - 1)
    if j >= i:
        j += 1                     # ensure i != j
    # amount to move is a random fraction of the donor cell
    amount = seq[i] * rng.uniform(0.0, frac)
    if amount == 0.0:
        return seq.copy()
    out = seq.copy()
    out[i] = max(out[i] - amount, 0.0)
    out[j] = min(out[j] + amount, 1000.0)
    return out

# Main entry point required by the harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients that (approximately)
    minimise the AC1 upper bound.  The routine works entirely with O(n) memory
    and uses only the supplied evaluate_sequence for exact scoring.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size.  If the parent construction `height_sequence_1`
    # exists we use its length, otherwise fall back to a modest default.
    # ------------------------------------------------------------------
    try:
        base_len = int(height_sequence_1.shape[0])          # type: ignore[name-defined]
    except Exception:                                      # pragma: no cover
        base_len = 256

    # ------------------------------------------------------------------
    # Initialise a uniform candidate (any positive constant works because the
    # score is scale‑invariant).  Uniform values are known to be close to the
    # theoretical optimum for non‑negative vectors.
    # ------------------------------------------------------------------
    candidate = np.ones(base_len, dtype=np.float64)
    candidate = np.clip(candidate, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Keep the best solution seen so far.  The provided parent construction
    # is also examined – it may already be a good solution.
    # ------------------------------------------------------------------
    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)  # type: ignore[name-defined]
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent
                best_score = parent_score
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # Stochastic local search: repeatedly transfer a small amount of mass
    # between two positions and occasionally apply a mild multiplicative
    # scaling to a short contiguous block.  Each proposal uses O(n) time/
    # memory and is accepted only if it improves the exact score.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = _transfer_perturb(best, rng, frac=0.08)

        # occasional block‑wise multiplicative perturbation (helps escape plateaus)
        if rng.random() < 0.15:
            width = int(rng.integers(1, max(2, base_len // 60)))
            start = int(rng.integers(0, base_len - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.04)))
            cand[start:start + width] *= factor

        # enforce box constraints
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
