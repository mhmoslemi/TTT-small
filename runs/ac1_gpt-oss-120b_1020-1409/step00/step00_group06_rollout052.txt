import numpy as np

# -------------------------------------------------------------
# Helper: generate a simple uniform candidate of length n.
# -------------------------------------------------------------
def uniform_candidate(n, value=1.0):
    """Return an n‑vector filled with *value* (clipped to allowed range)."""
    cand = np.full(n, float(value), dtype=np.float64)
    return np.clip(cand, 0.0, 1000.0)


# -------------------------------------------------------------
# Helper: a cheap block‑scaled perturbation (O(n) memory).
# -------------------------------------------------------------
def block_perturb(a, rng):
    """Multiply a random contiguous block of *a* by a log‑normal factor."""
    n = a.size
    # block width between 1% and 10% of n (at least 1)
    lo = max(1, n // 100)
    hi = max(lo, n // 10)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.07)))   # modest perturbation
    b = a.copy()
    b[start:stop] *= factor
    # optional mild symmetrisation
    if rng.random() < 0.25:
        b[start:stop] = 0.5 * (b[start:stop] + b[start:stop][::-1])
    return np.clip(b, 0.0, 1000.0)


# -------------------------------------------------------------
# Main entry point required by the harness.
# -------------------------------------------------------------
def propose_candidate(seed=42, max_iters=150, **kwargs):
    """
    Return a non‑empty list of coefficients.  The routine starts from a
    uniform candidate (using the mean of the supplied parent when available)
    and performs a few cheap perturbations, always keeping the best valid
    candidate seen.  Memory usage stays O(n).
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # Determine problem size and a reasonable baseline value.
    # ---------------------------------------------------------
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = parent.size if parent.ndim == 1 and parent.size > 0 else 256
    else:
        parent = None
        n = 256

    # If a parent exists, initialise around its average; otherwise use 1.0.
    if parent is not None and parent.size == n:
        base_val = float(parent.mean())
    else:
        base_val = 1.0

    # ---------------------------------------------------------
    # Start from the uniform candidate and remember the best.
    # ---------------------------------------------------------
    best = uniform_candidate(n, base_val)
    # Guard against the (unlikely) case sum is too small.
    if best.sum() < 0.01:
        best[:] = 0.01 / n
    best_score = float(evaluate_sequence(best.tolist()))

    # ---------------------------------------------------------
    # Very short local search – each step is O(n) and cheap.
    # ---------------------------------------------------------
    for _ in range(max_iters):
        cand = block_perturb(best, rng)
        # Quick sanity check: avoid evaluating obviously bad sum.
        if cand.sum() < 0.01:
            continue
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ---------------------------------------------------------
    # Return the best candidate found (as a plain Python list).
    # ---------------------------------------------------------
    return [float(x) for x in best]
