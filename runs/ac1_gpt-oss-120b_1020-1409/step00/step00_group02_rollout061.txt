def fast_score(arr):
    n = arr.size
    # use FFT size = next power of 2 >= 2*n-1
    m = 1 << (2*n-1).bit_length()
    # but we cannot use huge size perhaps >? n ~ 7k, 2n-1=~14k, next power of 2 = 16384, fine.
    fft_a = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
    max_b = np.max(conv.real)
    sum_a = np.sum(arr)
    if sum_a < 1e-12:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

import numpy as np

# ----------------------------------------------------------------------
# Fast (FFT based) proxy for the official evaluator.
# Returns the same AC1‐upper‑bound formula but is much quicker than
# np.convolve for many trial candidates.
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return np.inf
    # next power of two for convolution length 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    # real FFT convolution (seq * seq)
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a sequence of non‑negative heights (step function) that
    improves the AC1 upper bound.  The routine builds a family of
    candidates based on a controllable exponential shape, evaluates them
    cheaply with an FFT based surrogate and keeps the best one using the
    official evaluator for final verification.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the supplied parent if it exists, otherwise fall back.
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(base_seq, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    n = best.size
    idx = np.arange(n, dtype=np.float64)          # 0 … n‑1
    idx1 = idx + 1.0                               # 1 … n

    # ------------------------------------------------------------------
    # 1) Systematic search over a two‑parameter family:
    #    a_i = r^i * (i+1)^p   (scaled to respect the 0‑1000 bound)
    # ------------------------------------------------------------------
    # r in (0.95, 0.99995) – finer near 1 where the best ratios appear
    r_candidates = np.concatenate((
        np.linspace(0.95, 0.999, 15, endpoint=False),
        np.linspace(0.999, 0.99995, 15, endpoint=False),
    ))
    p_candidates = [0.0, 0.5, 1.0, 1.5, 2.0]

    for r in r_candidates:
        # pre‑compute r**i (more stable than np.power for many i)
        log_r = np.log(r)
        exp_part = np.exp(idx * log_r)      # r**i

        for p in p_candidates:
            # shape = r**i * (i+1)^p
            shape = exp_part * (idx1 ** p)

            # normalise to satisfy the 0‑1000 bound (scaling does not affect the score)
            max_val = shape.max()
            if max_val > 0.0:
                scale = min(1.0, 1000.0 / max_val)
                shape = shape * scale

            # cheap proxy score
            sc = _fast_score(shape)
            if sc < best_score:
                # verify with the official (exact) evaluator
                exact = float(evaluate_sequence(shape.tolist()))
                if np.isfinite(exact) and exact < best_score:
                    best = shape.copy()
                    best_score = exact

    # ------------------------------------------------------------------
    # 2) Small stochastic refinement around the current best.
    # ------------------------------------------------------------------
    for _ in range(250):
        # multiplicative Gaussian perturbation (median around 1.0)
        perturb = best * (1.0 + rng.normal(0.0, 0.02, size=n))
        perturb = np.clip(perturb, 0.0, 1000.0)

        # cheap evaluation first
        sc = _fast_score(perturb)
        if sc < best_score:
            exact = float(evaluate_sequence(perturb.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = perturb.copy()
                best_score = exact

    # ------------------------------------------------------------------
    # Return a plain Python list of floats (as required by the harness).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
