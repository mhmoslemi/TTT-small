def generate_sidon_set(n, rng, max_trials=10):
    best_set = []
    best_size = -1
    for t in range(max_trials):
        order = rng.permutation(n)
        current = []
        sums = set()
        for x in order:
            # compute new sums
            conflict = False
            # candidate sums list could compute quickly using loop
            for y in current:
                if (x + y) in sums:
                    conflict = True
                    break
            if conflict or (x + x) in sums:
                continue
            # else add
            for y in current:

import numpy as np

# ------------------------------------------------------------
# Helper: generate a (near) Sidon set inside [0, n)
# ------------------------------------------------------------
def generate_sidon_set(n, rng, attempts=100):
    """
    Return a list of indices forming a Sidon set (no duplicate pair sums)
    using a greedy algorithm over several random orderings.
    """
    best_set = []
    best_len = -1
    max_sum = 2 * n                     # enough for all i+j up to 2n-2
    for _ in range(attempts):
        order = rng.permutation(n)
        cur = []
        # boolean mask of used sums i+j
        sum_used = np.zeros(max_sum, dtype=np.bool_)
        for x in order:
            # self‑sum collision?
            if sum_used[x + x]:
                continue
            # collision with existing elements?
            conflict = False
            for y in cur:
                if sum_used[x + y]:
                    conflict = True
                    break
            if conflict:
                continue
            # accept x, mark new sums
            sum_used[x + x] = True
            for y in cur:
                sum_used[x + y] = True
            cur.append(int(x))
        if len(cur) > best_len:
            best_len = len(cur)
            best_set = cur
            # optional early stop when we are already close to the
            # theoretical upper bound sqrt(2n)
            if best_len >= int(np.sqrt(2 * n)) + 5:
                # keep trying – a later random order may be better
                pass
    return best_set

def propose_candidate(seed=42, num_trials=120, epsilon=0.01, **kwargs):
    """
    Construct a sequence that beats the baseline AC1 bound.
    1. Build a Sidon set of “large” entries (height 1.0).
    2. Fill all other positions with a tiny positive value (epsilon).
    3. Return the best of this construction and the provided parent.
    """
    rng = np.random.default_rng(seed)

    # Parent (fallback) – may be absent in some environments
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = np.ones(256, dtype=np.float64)
    n = base.size
    fallback = np.clip(base, 0.0, 1000.0)
    fallback_score = float(evaluate_sequence(fallback.tolist()))

    # ----------------------------------------------------------------
    # Build candidate: Sidon spikes of height 1, background epsilon
    # ----------------------------------------------------------------
    sidon_indices = generate_sidon_set(n, rng, attempts=num_trials)

    candidate = np.full(n, float(epsilon), dtype=np.float64)
    if sidon_indices:
        candidate[sidon_indices] = 1.0
    candidate = np.clip(candidate, 0.0, 1000.0)

    cand_score = float(evaluate_sequence(candidate.tolist()))

    # ----------------------------------------------------------------
    # Choose the best valid candidate
    # ----------------------------------------------------------------
    if np.isfinite(cand_score) and cand_score < fallback_score:
        return candidate.tolist()
    # if the Sidon construction failed to improve, fall back to parent
    if np.isfinite(fallback_score):
        return fallback.tolist()
    # final safety net – a uniform tiny sequence (always valid)
    final = np.full(n, float(epsilon), dtype=np.float64)
    return final.tolist()
