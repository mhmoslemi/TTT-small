import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a compact nonnegative step‑function that achieves a low AC1 upper bound.
    Uses a linearly increasing sequence scaled so that the maximum height is 1000.
    Falls back to a uniform sequence if the reference height_sequence_1 is unavailable.
    """
    # Determine length from the provided parent if possible.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        # Fallback length – any positive length works.
        n = 256

    # Linear ramp: a_i = (i+1) * (1000 / n), guaranteeing 0 < a_i <= 1000.
    scale = 1000.0 / n
    candidate = scale * np.arange(1, n + 1, dtype=np.float64)

    # Clip just in case numerical noise pushes values slightly out of bounds.
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Return as a plain Python list of floats.
    return [float(x) for x in candidate]
