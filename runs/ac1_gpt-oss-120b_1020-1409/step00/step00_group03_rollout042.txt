import numpy as np

def _max_convolution(vec: np.ndarray) -> float:
    """Return max of the full convolution of vec with itself using FFT."""
    n = vec.size
    # next power of two for efficient FFT
    L = 1 << ((2 * n - 1).bit_length())
    # real FFT is enough because the input is real
    f_vec = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(f_vec * f_vec, n=L)
    # keep only the first 2*n-1 entries (full convolution length)
    return float(conv[: 2 * n - 1].max())


def propose_candidate(seed: int = 42, num_samples: int = 3000, **kwargs):
    """
    Return a non‑negative height sequence whose AC1 bound is (hopefully) lower
    than the naïve uniform construction.
    The algorithm draws many random Dirichlet samples (Γ‑draws) and keeps the
    one with the smallest 2·n·max(conv)/sum² value.
    """
    rng = np.random.default_rng(seed)

    # length of the problem – use the provided parent if present
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256

    # start with the uniform vector (its score is exactly 2.0)
    best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = 2.0 * n * _max_convolution(best_vec)  # sum = 1 → denominator = 1

    # random Dirichlet (Gamma) draws – shape=0.5 gives slightly sparse vectors
    for _ in range(num_samples):
        # draw positive gamma numbers and normalize to unit L1‑norm
        # (the scaling of the vector does not affect the final score)
        g = rng.gamma(0.5, 1.0, size=n)
        vec = g / g.sum()
        # evaluate the bound (sum is 1, so denominator is 1)
        cur_score = 2.0 * n * _max_convolution(vec)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = vec

    # ensure the result respects the evaluator's clipping limits
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # final fallback – if something went wrong, return a simple uniform vector
    if best_vec.ndim != 1 or best_vec.size == 0 or not np.isfinite(best_vec).all():
        best_vec = np.ones(n, dtype=np.float64) / n

    return [float(v) for v in best_vec]
