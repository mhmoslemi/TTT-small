import numpy as np

# ------------------------------------------------------------
# Helper: generate the classic a_i = C(2i,i)/4^i sequence.
# ------------------------------------------------------------
def _make_series(length: int) -> np.ndarray:
    """Return the length‑`length` truncation of a_i = (2i choose i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a_i = a_{i-1} * (2*i-1) / (2*i)
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Fast linear convolution via FFT (real‑valued input).
# ------------------------------------------------------------
def _conv_self(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    # length of the full convolution
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << (conv_len - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(fx * fx, n=fft_len)[:conv_len]
    # small negative round‑off errors are harmless for the max
    return conv


# ------------------------------------------------------------
# Compute the AC1 upper bound exactly (mirrors evaluate_sequence).
# ------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return 2*n*max(conv(seq,seq))/sum(seq)^2 with the same clipping as the evaluator."""
    # clip as the evaluator would do
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    c = _conv_self(seq)
    m = c.max()
    return 2.0 * seq.size * m / (s * s)

# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, num_steps: int = 1200, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    previous 2.0 bound.  The method starts from the analytically optimal
    “square‑root” series and then performs a cheap projected gradient descent
    that moves weight from entries that dominate the maximal convolution
    towards entries that contribute little, thereby flattening the convolution.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size.  Prefer the provided parent construction.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:
        # fallback length when the parent is unavailable
        n = 256
        parent = None

    # ------------------------------------------------------------------
    # Initialise with the series a_i = C(2i,i)/4^i (optimal for M=1).
    # ------------------------------------------------------------------
    best = _make_series(n)
    best_score = _score(best)

    # If a parent exists and is already better, start from it.
    if parent is not None:
        parent = np.clip(parent, 0.0, 1000.0)
        parent_score = _score(parent)
        if parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # Helper: compute gradient of  f = M / S^2   (M = max convolution).
    # ------------------------------------------------------------------
    def _grad_step(x: np.ndarray):
        """One projected gradient step; returns new candidate and its score."""
        # convolution and its maximal entry
        conv = _conv_self(x)
        M = conv.max()
        k = int(conv.argmax())          # index of the maximal convolution term
        S = x.sum()
        if S == 0.0:
            return x, float('inf')

        # build the vector g_i = a_{k-i} (0 if out of range)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        g = np.zeros_like(x)
        g[idx] = x[k - idx]

        # gradient of f = M / S^2  (see analysis)
        inv_S2 = 1.0 / (S * S)
        inv_S3 = inv_S2 / S
        grad = (2.0 * inv_S2) * g - (2.0 * M) * inv_S3

        # try a few learning‑rates, keep the best improvement
        lr = 1.0
        best_local = x
        best_local_score = _score(x)
        for _ in range(5):
            cand = x - lr * grad
            # projection onto the feasible box
            np.clip(cand, 0.0, 1000.0, out=cand)
            # optional slight symmetrisation (helps flattening)
            if rng.random() < 0.15:
                cand = 0.5 * (cand + cand[::-1])
            sc = _score(cand)
            if sc < best_local_score:
                best_local = cand
                best_local_score = sc
                break
            lr *= 0.5
        return best_local, best_local_score

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    x = best.copy()
    stagnant = 0        # count iterations without improvement
    for step in range(num_steps):
        # Gradient descent move
        x_new, sc_new = _grad_step(x)

        # Random block perturbation (occasionally)
        if step % 17 == 0:
            width = max(1, n // rng.integers(12, 33))
            start = rng.integers(0, max(1, n - width + 1))
            stop = start + width
            factor = np.exp(rng.normal(0.0, 0.13))
            block = x_new.copy()
            block[start:stop] *= factor
            if rng.random() < 0.2:
                block = 0.5 * (block + block[::-1])
            np.clip(block, 0.0, 1000.0, out=block)
            sc_block = _score(block)
            if sc_block < sc_new:
                x_new, sc_new = block, sc_block

        # Accept if we improved globally
        if sc_new + 1e-14 < best_score:
            best = x_new.copy()
            best_score = sc_new
            stagnant = 0
        else:
            stagnant += 1

        x = x_new

        # early stop if nothing improves for a while
        if stagnant > 250:
            break

    # ------------------------------------------------------------
    # Ensure we return a valid non‑empty list.
    # ------------------------------------------------------------
    if best.size == 0 or not np.isfinite(best_score):
        best = np.ones(256, dtype=np.float64)
    return [float(v) for v in best]
