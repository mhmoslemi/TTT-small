    # If best is still None (should not), fallback to uniform.

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # random block width between n/200 and n/12
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation on the block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetry to keep shape balanced
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def conv_max_info(vec: np.ndarray):
    """Return (max_value, index_of_max) of the full convolution vec*vec."""
    conv = np.convolve(vec, vec)
    max_val = conv.max()
    max_idx = int(conv.argmax())
    return max_val, max_idx, conv


def gradient_of_j(vec: np.ndarray, n: int, M: float, k: int, s: float):
    """Derivative of the objective J w.r.t. vec (size n)."""
    # gradient of the max‑convolution term
    grad_M = np.zeros_like(vec)
    i_low = max(0, k - (n - 1))
    i_high = min(k, n - 1)
    i = np.arange(i_low, i_high + 1, dtype=np.int64)
    j = k - i

    neq = i != j
    # off‑diagonal contributions
    if np.any(neq):
        ii = i[neq]
        jj = j[neq]
        grad_M[ii] += vec[jj]
        grad_M[jj] += vec[ii]

    # diagonal contributions (i == j)
    eq = i == j
    if np.any(eq):
        ii = i[eq]
        grad_M[ii] += 2.0 * vec[ii]

    # derivative of J = 2n * M / s^2
    C = 2.0 * n
    inv_s2 = 1.0 / (s * s)
    inv_s3 = inv_s2 / s
    grad_J = C * (grad_M * inv_s2 - 2.0 * M * inv_s3)
    return grad_J


def geometric_candidate(n: int, c: float) -> np.ndarray:
    """Geometric decay a_i = (1 - c/n)^i, clipped to non‑negative."""
    r = 1.0 - c / n
    if r <= 0.0:
        r = 0.0
    exponents = np.arange(n, dtype=np.float64)
    return r ** exponents


def improve_by_gradient(a: np.ndarray, max_iter: int, rng: np.random.Generator) -> np.ndarray:
    """Simple projected gradient descent on the objective J."""
    n = a.size
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = best.copy()
    cur_score = best_score

    # initial step size proportional to the mean entry
    step = max(1e-3, cur.mean())

    for _ in range(max_iter):
        M, k, _ = conv_max_info(cur)
        s = cur.sum()
        if s == 0.0:
            break
        grad = gradient_of_j(cur, n, M, k, s)

        # normalize step to avoid too large jumps
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        alpha = step / norm

        cand = cur - alpha * grad
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
            step *= 1.1  # success – try a slightly larger step
        else:
            step *= 0.5  # back‑off
        # safety stop
        if step < 1e-12:
            break
    return best


def improve_by_random_search(a: np.ndarray, rng: np.random.Generator, steps: int) -> np.ndarray:
    """Cheap block‑wise random hill‑climbing."""
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    for _ in range(steps):
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score
    return best
