import numpy as np

def _initial_length():
    """
    Determine the length of the parent sequence if it exists and is valid.
    Falls back to a modest default length.
    """
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0:
            raise ValueError
        return int(arr.shape[0])
    except Exception:
        # Safe fallback when the parent is missing or malformed.
        return 256


def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (as a Python list) that minimizes
    the AC1 upper bound evaluated by `evaluate_sequence`.

    The theoretical minimum of the bound is 2, attained by any uniform
    non‑zero sequence.  We therefore return a constant vector of length

    n = _initial_length()

    score = evaluate_sequence(candidate.tolist())
