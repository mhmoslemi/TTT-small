def propose_candidate(seed=42, num_iters=2000, step_size=0.01, **kwargs):
    rng = np.random.default_rng(seed)
    # Get parent length
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        n = 256
        a = np.full(n, 1.0)
    else:
        n = parent.size
        # Initialize with random positive values
        a = rng.random(n) * 2.0  # uniform between 0 and 2
        # Ensure sum not too small
        if np.sum(a) < 0.01:
            a = np.ones(n)
    # Scale to keep sum constant (e.g., 1)
    a = a / np.sum(a)  # sum = 1
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # iterate
    for it in range(num_iters):
        conv = np.convolve(a, a)
        idx = int(np.argmax(conv))
        max_val = conv[idx]
        # compute gradient
        grad = np.zeros_like(a)
        # vectorized: for i in range(n) if 0 <= idx-i < n: grad[i] = 2*a[idx-i]
        # create an array of indices j = idx - np.arange(n)
        i_arr = np.arange(n)
        j_arr = idx - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad[mask] = 2.0 * a[j_arr[mask]]
        # step
        a_new = a - step_size * grad
        # ensure nonnegative
        a_new = np.clip(a_new, 0.0, None)
        # renormalize sum to 1
        total = np.sum(a_new)
        if total < 1e-12:
            # fallback to random perturbation
            a_new = rng.random(n)
            total = np.sum(a_new)
        a_new = a_new / total
        # evaluate
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_new.copy()
            best_score = score
        # accept update for next iteration: maybe keep a_new even if not improved? For search
        a = a_new
        # optionally adjust step size (e.g., decay)
        step_size *= 0.9995
    # Return best as list
    return [float(x) for x in best]

    n = len(height_sequence_1)
    #initialize uniform
    a = np.ones(n, dtype=np.float64)
    a = a / np.sum(a)  # sum = 1

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # apply random moves
    for step in range(steps):
        # compute conv and max index
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        # Identify indices involved in max pair
        # returns all i where the pair (i, max_idx-i) contributes
        # We'll randomly adjust one of them
        i_candidates = np.arange(n)
        j = max_idx - i_candidates
        valid = (j >= 0) & (j < n)
        # contributions weight
        contrib = a[i_candidates] * a[j]
        # pick index with probability proportional to contribution
        if np.any(valid):
            probs = contrib * valid
            if probs.sum() > 0:
                probs = probs / probs.sum()
                i = rng.choice(n, p=probs)
                j = max_idx - i
                # Transfer a small amount from i or j to a random other index
                delta = rng.random() * min(a[i], a[j]) * 0.5
                # reduce both i and j (since both contribute) or maybe just one
                a[i] = max(a[i] - delta, 0.0)
                a[j] = max(a[j] - delta, 0.0)
                # distribute to a random index not in pair
                k = rng.integers(0, n)
                while k == i or k == j:
                    k = rng.integers(0, n)
                a[k] += 2 * delta
        else:
            # fallback random perturbation
            candidate = a.copy()
            # pick two random indices
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.random() * min(candidate[i], candidate[j]) * 0.5
            candidate[i] -= delta
            candidate[j] -= delta
            k = rng.integers(0, n)
            candidate[k] += 2 * delta
            a = candidate
        # ensure sum normalized
        total = np.sum(a)
        if total == 0:
            a = np.ones(n)
            total = n
        a = a / total
        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
    return [float(x) for x in best]

import numpy as np

def _convolution_max(a: np.ndarray) -> float:
    """Return the maximum value of the full convolution a*a."""
    # np.convolve returns an array of length 2*len(a)-1
    return float(np.max(np.convolve(a, a)))


def propose_candidate(seed: int = 42, *, num_steps: int = 1500, **kwargs):
    """
    Stochastic optimiser that tries to lower the AC1 bound.
    Returns a non‑negative list of coefficients (the best found).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise a candidate sequence.
    # ------------------------------------------------------------------
    parent = None
    try:                         # height_sequence_1 is supplied at run time
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:            # pragma: no cover
        pass

    if (
        parent is None
        or parent.ndim != 1
        or parent.size == 0
        or not np.isfinite(parent).all()
    ):
        n = 256
        a = np.ones(n, dtype=np.float64) / n          # uniform fallback
    else:
        n = int(parent.size)
        # start from a uniform distribution – it gives the known bound 2.
        a = np.ones(n, dtype=np.float64) / n

    # keep the best sequence seen so far
    best = a.copy()
    best_score = float(_convolution_max(best) * 2 * n / (np.sum(best) ** 2))

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    step_factor = 0.12          # fraction of the smallest participating weight
    for _ in range(num_steps):
        # full convolution and index of its maximum
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))

        # contributions that form the current maximum
        i_arr = np.arange(n)
        j_arr = max_idx - i_arr
        mask = (j_arr >= 0) & (j_arr < n) & (a > 0) & (a[j_arr] > 0)

        if not np.any(mask):
            # nothing to tweak – break early
            break

        # contribution of each participating pair (ordered, i contributes twice
        # when i != j, but that scaling does not affect the direction of the move)
        contrib = a[mask] * a[j_arr[mask]]
        if contrib.sum() == 0:
            break

        # pick a pair proportionally to its contribution
        prob = contrib / contrib.sum()
        chosen_i = rng.choice(np.where(mask)[0], p=prob)
        chosen_j = max_idx - chosen_i

        # move a small amount of mass away from this pair
        delta = step_factor * min(a[chosen_i], a[chosen_j])
        if delta <= 0:
            continue
        a[chosen_i] -= delta
        a[chosen_j] -= delta

        # distribute the removed mass to two random locations (they may coincide)
        k1 = rng.integers(0, n)
        k2 = rng.integers(0, n)
        share = delta * 0.5
        a[k1] += share
        a[k2] += share

        # re‑normalise to keep the sum exactly 1 (scaling does not affect the score)
        total = a.sum()
        if total <= 0:
            a = np.ones(n, dtype=np.float64) / n
        else:
            a /= total

        # evaluate the new candidate
        cur_score = float(_convolution_max(a) * 2 * n / (np.sum(a) ** 2))

        if np.isfinite(cur_score) and cur_score < best_score:
            best = a.copy()
            best_score = cur_score

        # occasional random “kick’’ to escape shallow plateaus
        if rng.random() < 0.03:
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(a[i], a[j]) * 0.3
            a[i] -= eps
            a[j] -= eps
            k = rng.integers(0, n)
            a[k] += 2 * eps
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
