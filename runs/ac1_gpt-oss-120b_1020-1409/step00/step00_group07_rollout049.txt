import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that tries to
    minimise the AC1 upper bound evaluated by `evaluate_sequence`.
    The routine works with the parent sequence `height_sequence_1` if it
    exists, otherwise falls back to a uniform vector.
    A tiny random local‑search loop is added for potential improvement,
    while keeping memory usage O(n).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine a valid starting sequence (parent or generic fallback)
    # ------------------------------------------------------------------
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if base_arr.ndim != 1 or base_arr.size == 0 or not np.isfinite(base_arr).all():
            raise ValueError
    except Exception:
        # fallback length – any reasonable size works; 256 matches the
        # original script's fallback.
        base_arr = np.ones(256, dtype=np.float64)

    # Clip to the admissible range [0, 1000]
    base_arr = np.clip(base_arr, 0.0, 1000.0)
    n = base_arr.size

    # ------------------------------------------------------------------
    # 2. Uniform candidate (the theoretical optimum for the given
    #    objective under the non‑negativity and box constraints)
    # ------------------------------------------------------------------
    total = np.sum(base_arr)
    # Guarantee a positive sum (the evaluator rejects sum < 0.01)
    if total < 0.01:
        mean_val = 0.01 / n
    else:
        mean_val = total / n
    uniform = np.full(n, mean_val, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3. Pick the better of the parent and the uniform candidate
    # ------------------------------------------------------------------
    best_seq = base_arr
    best_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(best_score):
        best_score = float('inf')

    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if np.isfinite(uniform_score) and uniform_score < best_score:
        best_seq = uniform
        best_score = uniform_score

    # ------------------------------------------------------------------
    # 4. Very cheap stochastic refinement – stay O(n) memory.
    #    Only a few hundred trials are cheap and can occasionally beat
    #    the uniform solution on noisy inputs.
    # ------------------------------------------------------------------
    max_steps = 200
    max_width = max(2, n // 20)          # moderate block size
    for _ in range(max_steps):
        # Random block size and location
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        # Random multiplicative perturbation (log‑normal factor)
        factor = float(np.exp(rng.normal(0.0, 0.10)))
        cand = best_seq.copy()
        cand[start:start + width] *= factor
        # Optional mild symmetrisation (helps for some patterns)
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_seq = cand
            best_score = cand_score

    # ------------------------------------------------------------------
    # 5. Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]
