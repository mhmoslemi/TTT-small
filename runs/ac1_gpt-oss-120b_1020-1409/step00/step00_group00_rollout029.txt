import numpy as np

def _score(arr: np.ndarray) -> float:
    """
    Compute the same objective as evaluate_sequence but with a NumPy array.
    Returns np.inf for invalid inputs.
    """
    # Clip to the domain enforced by the evaluator.
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return float('inf')
    # Full convolution (np.convolve uses an efficient FFT path for large arrays).
    conv = np.convolve(arr, arr)
    max_conv = float(conv.max())
    n = arr.size
    return 2.0 * n * max_conv / (s * s)


def propose_candidate(seed=42, num_steps=3000, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that strives to
    minimise the AC1 upper bound used by `evaluate_sequence`.

    The routine starts from a analytically motivated shape
        a[i] = 1 / sqrt(i+1)
    and then performs a lightweight stochastic local search that
    respects the O(n) memory budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Candidate initialisation
    # ------------------------------------------------------------------
    # Try to use the provided parent construction when it exists.
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:  # height_sequence_1 not defined or bad
        parent_arr = None

    # Default length when no parent is present.
    n = parent_arr.shape[0] if (parent_arr is not None and parent_arr.ndim == 1) else 256

    # Analytic start‑shape:  a[i] = 1 / sqrt(i+1)
    idx = np.arange(n, dtype=np.float64)
    base = 1.0 / np.sqrt(idx + 1.0)

    # Small random perturbation to break exact symmetry.
    base += rng.normal(scale=0.01, size=n)
    base = np.clip(base, 0.0, 1000.0)

    # Choose the better of the analytic start and the parent (if any).
    best = base.copy()
    best_score = _score(best)

    if parent_arr is not None and parent_arr.ndim == 1 and parent_arr.size > 0:
        parent = np.clip(parent_arr, 0.0, 1000.0)
        parent_score = _score(parent)
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # ------------------------------------------------------------------
    # 2. Stochastic improvement loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        cand = best.copy()
        op = rng.integers(0, 3)  # three simple move types

        if op == 0:
            # --- block multiplicative perturbation -----------------
            # Width adapted to the problem size.
            min_w = max(1, n // 300)
            max_w = max(2, n // 15)
            width = int(rng.integers(min_w, max_w + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            # Mild log‑normal factor (≈ 0.8 – 1.2).
            factor = np.exp(rng.normal(0.0, 0.08))
            cand[start:stop] *= factor

        elif op == 1:
            # --- move a fraction of mass from an early index to a later one ---
            # With 50 % probability move forward, otherwise move backward.
            forward = rng.integers(0, 2) == 0
            if forward:
                # early → later
                i = rng.integers(0, max(1, n // 5))
                j = rng.integers(max(i + 1, n // 2), n)
            else:
                # later → early
                i = rng.integers(max(0, n - n // 5), n)
                j = rng.integers(0, min(n // 2, i))
            # Transfer at most 20 % of the source value.
            delta = rng.random() * 0.2 * cand[i]
            cand[i] -= delta
            cand[j] += delta

        else:
            # --- tiny global Gaussian shake (helps to explore around a plateau) ---
            cand += rng.normal(scale=0.005, size=n)

        # Keep within the admissible box.
        cand = np.clip(cand, 0.0, 1000.0)

        cur_score = _score(cand)

                best = permuted
                best_score = perm_score

    # ------------------------------------------------------------------
    # 3. Return a plain Python list as required by the harness.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
