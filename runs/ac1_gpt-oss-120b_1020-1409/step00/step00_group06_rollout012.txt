def evaluate(seq):
    # expects a list or 1D array
    return float(evaluate_sequence(seq.tolist() if isinstance(seq, np.ndarray) else seq))

    rng = np.random.default_rng(seed)
    candidate = initialize_sequence()
    best = candidate.copy()
    best_score = evaluate(best)
    # optional simple improvements: try symmetric block scaling maybe
    # But uniform is optimal, we might just exit

    for _ in range(num_steps):
        # make a copy
        pert = best.copy()
        # choose random subsegment
        n = pert.size
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 10))
        stop = min(n, start + width)
        scale = np.exp(rng.normal(0.0, 0.1))
        pert[start:stop] *= scale

        score = evaluate(pert)
        if np.isfinite(score) and score < best_score:
            best = pert
            best_score = score
    return [float(x) for x in best]

    rng = np.random.default_rng(seed)

    # Start from a uniform sequence of the appropriate length.
    current = _initial_sequence()
    best = current.copy()
    best_score = _score(best)

    # Optional cheap local perturbations – they rarely improve the bound
    # because the uniform sequence is the global minimizer, but they cost
    # little and keep the logic general.
    for _ in range(num_steps):
        cand = best.copy()
        n = cand.size
        # Random block size and position.
        width = int(rng.integers(1, max(2, n // 20)))
        start = int(rng.integers(0, n - width + 1))
        stop = start + width
        # Random multiplicative perturbation.
        scale = np.exp(rng.normal(0.0, 0.07))
        cand[start:stop] *= scale
        # Optional symmetry to keep the shape balanced.
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])
        # Enforce the allowed range.
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    # Return a plain Python list of floats as required by the harness.
    return [float(x) for x in best]
