import numpy as np

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that seeks a low
    autocorrelation upper bound as measured by the pre‑imported
    ``evaluate_sequence`` function.

    The routine builds a simple uniform candidate (which attains the
    theoretical optimum of 2.0 for non‑negative vectors) and then
    performs a tiny stochastic refinement that never uses more than O(n)
    additional memory.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Acquire a parent sequence if the harness provided one; otherwise fall back.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # ------------------------------------------------------------------
    # Build an initial uniform candidate having the same total mass as the parent.
    # The uniform distribution is analytically optimal for the non‑negative case
    # (its score equals 2.0).
    # ------------------------------------------------------------------
    total = float(np.sum(parent))
    total = max(total, 0.01)                     # respect the evaluator's sum bound
    uniform_val = total / n
    best = np.full(n, uniform_val, dtype=np.float64)
    best = np.clip(best, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Tiny stochastic refinement: modest Gaussian noise, renormalisation,
    # and acceptance if the score improves.  This loop is O(n) per iteration
    # and runs for a modest number of steps, keeping memory usage linear.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # generate a candidate by adding small noise
        cand = best + rng.normal(scale=0.02, size=n)
        # enforce bounds required by the evaluator
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = np.sum(cand)
        if s < 0.01:                # keep the sum above the minimal threshold
            continue
        # renormalise to the original total mass (scale invariance of the score)
        cand *= (total / s)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Return the best candidate found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
