import numpy as np

def propose_candidate(seed=42, **kwargs):
    # Determine length
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 256
    # compute zero block length as one third
    t = n // 3
    # remaining non-zero length
    remaining = n - t
    left_len = remaining // 2
    right_len = remaining - left_len
    # use constant height 1.0 (could use any positive)
    seq = np.concatenate([
        np.full(left_len, 1.0, dtype=np.float64),
        np.zeros(t, dtype=np.float64),
        np.full(right_len, 1.0, dtype=np.float64)])
    # Ensure length is n (in case rounding issues)
    if seq.size != n:
        # adjust by trimming or padding zeros at end
        seq = seq[:n]
        if seq.size < n:
            seq = np.concatenate([seq, np.zeros(n - seq.size, dtype=np.float64)])
    # Validate with evaluator; fallback to uniform if invalid
    try:
        score = float(evaluate_sequence(seq.tolist()))
    except Exception:
        # fallback uniform 1.0
        seq = np.ones(n, dtype=np.float64)
    return seq.tolist()

def _linear_ramp():
    """
    Return a linear ramp from 0 to 1000 with the same length as
    the provided parent construction `height_sequence_1`.
    If that variable is unavailable, fall back to a length of 256.

    try:

        n = 256                                # safety fallback
    # linspace already respects the max‑height bound (≤1000)
    return np.linspace(0.0, 1000.0, n, dtype=np.float64)


def propose_candidate(seed=42, **kwargs):
    """

    seq = _linear_ramp()
