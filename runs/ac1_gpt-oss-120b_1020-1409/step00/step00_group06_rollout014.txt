import numpy as np

def make_block_perturbation(base, rng):
    """Generate a single O(n) memory non‑negative perturbation of ``base``."""
    candidate = np.array(base, copy=True, dtype=np.float64)
    n = candidate.size
    # choose a block width in a reasonable range
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative perturbation of the block
    factor = np.exp(rng.normal(0.0, 0.12))
    candidate[start:stop] *= float(factor)
    # occasional mirroring for added diversity
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce the evaluator's bounds
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Obtain a parent sequence (fallback to a simple constant if needed)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # ------------------------------------------------------------------
    # 2. Uniform candidate – provably optimal for the given objective
    # ------------------------------------------------------------------
    total = np.sum(parent)
    if total < 0.01:                 # ensure the evaluator does not reject it
        uniform_val = 0.01 / n
    else:
        uniform_val = total / n
    uniform_val = max(0.0, min(1000.0, uniform_val))

    best = np.full(n, uniform_val, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 3. Light stochastic refinement (won't worsen the score)
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = make_block_perturbation(best, rng)
        # guarantee a non‑trivial sum to avoid infinite penalty
        if np.sum(cand) < 0.01:

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score

    # ------------------------------------------------------------------
    # 4. Return the best-found sequence as a plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best]
