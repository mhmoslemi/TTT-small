import numpy as np

def _clip_and_score(arr: np.ndarray):
    """Clip to the allowed range and return the AC1 bound and the convolution."""
    # Follow exactly the logic of the provided `evaluate_sequence`.
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return np.inf, None
    conv = np.convolve(arr, arr)
    max_b = conv.max()
    n = arr.size
    score = float(2.0 * n * max_b / (s * s))
    return score, conv


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator):
    """A cheap O(n) block‑scale perturbation similar to the previous baseline."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _mass_transfer(candidate: np.ndarray, rng: np.random.Generator):
    """Transfer a random fraction of mass from one index to another."""
    n = candidate.size
    i, j = rng.integers(0, n, size=2)
    if i == j or candidate[i] <= 0.0:
        return candidate  # nothing to do
    # transfer up to 50 % of the source amount
    delta = rng.random() * 0.5 * candidate[i]
    new = candidate.copy()
    new[i] -= delta
    new[j] += delta
    np.clip(new, 0.0, 1000.0, out=new)
    return new


def _gradient_step(candidate: np.ndarray, conv: np.ndarray, rng: np.random.Generator):
    """One very small gradient‑like step aimed at decreasing the bound."""
    n = candidate.size
    km = int(conv.argmax())            # index where the max occurs
    max_b = conv.max()
    s = candidate.sum()
    if s < 1e-12:
        return candidate
    # dM/da_i = 2 * a_{km-i}  (zero if the partner index is out of range)
    partner = np.arange(n) - km            # partner = km - i, but we need partner[i] = km - i
    partner = km - np.arange(n)
    valid = (partner >= 0) & (partner < n)
    dM = np.zeros_like(candidate)
    dM[valid] = 2.0 * candidate[partner[valid]]
    # gradient of the score B = 2*n*max_b / s^2
    factor = 2.0 * n / (s * s)
    grad = factor * dM - (4.0 * n * max_b / (s ** 3)) * candidate
    # take a tiny step opposite to the gradient
    step = -0.02 * grad                     # 0.02 is a hand‑tuned scale
    new = candidate + step
    np.clip(new, 0.0, 1000.0, out=new)
    return new

            cand = _gradient_step(best, best_conv, rng)

        score, conv = _clip_and_score(cand)

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            best_conv = conv
            no_improve = 0
            # early exit if we beat a reasonable target
            if best_score < 1.55:
                break
        else:
            no_improve += 1
            # occasional random restart if we are stuck for a while
            if no_improve >= 250 and step < num_steps // 2:
                restart = rng.random(n)
                restart = restart * (target_sum / restart.sum())
                rscore, rconv = _clip_and_score(restart)
                if np.isfinite(rscore) and rscore < best_score:
                    best = restart
                    best_score = rscore
                    best_conv = rconv
                no_improve = 0

    # ------------------------------------------------------------------
    # Return a plain Python list – the harness expects that format.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
