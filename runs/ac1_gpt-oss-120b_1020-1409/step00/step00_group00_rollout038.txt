    # 3) Linear decreasing a_i = n-i
    seq = (n - idx)            # idx = 0,…,n‑1
    seq = seq / seq.sum()
    score = float(evaluate_sequence(seq.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = seq.copy()

    # 4) Half‑sine shape
    seq = np.sin(np.pi * (idx + 0.5) / n)
    seq = seq / seq.sum()
    score = float(evaluate_sequence(seq.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = seq.copy()

    # 5) Down‑ward cosine shape (0 at both ends)
    if n > 1:
        seq = (np.cos(np.pi * idx / (n - 1)) + 1.0) / 2.0
    else:
        seq = np.ones(1, dtype=np.float64)
    seq = seq / seq.sum()
    score = float(evaluate_sequence(seq.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = seq.copy()

    # ----- local refinement ---------------------------------------------------
    seq = best_seq.copy()
    max_iters = 800                      # modest number of targeted moves
    for _ in range(max_iters):
        conv = np.convolve(seq, seq)
        k = int(np.argmax(conv))                     # index with maximal convolution

        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range

        # contribution of each pair (i, j) to this maximal entry
        prod = seq[i_range] * seq[j_range]
        double = (i_range != j_range)
        weight = np.where(double, 2.0, 1.0)
        weighted = prod * weight

        # pick the index that participates most strongly
        imax = i_range[np.argmax(weighted)]

        # pick a location that currently holds the smallest mass
        donor = int(np.argmin(seq))
        if donor == imax:                     # avoid self‑transfer
            # second smallest if possible
            sorted_idx = np.argsort(seq)
            donor = int(sorted_idx[1]) if n > 1 else imax

        # move a modest fraction of mass from imax to donor
        frac = 0.03 + 0.02 * rng.random()     # between 3% and 5%
        amount = seq[imax] * frac
        if amount <= 0.0:
            continue
        seq[imax] -= amount
        seq[donor] += amount
        if seq[imax] < 0.0:                  # safeguard against numerical issues
            seq[donor] += seq[imax]
            seq[imax] = 0.0

        # evaluate the new sequence
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
        else:
            # revert the move
            seq[donor] -= amount
            seq[imax] += amount

    # ----- return the best found sequence ------------------------------------
    return [float(v) for v in best_seq]
