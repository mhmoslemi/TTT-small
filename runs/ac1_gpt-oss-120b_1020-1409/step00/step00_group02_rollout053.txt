    rng = np.random.default_rng(seed)
    # initialization: use parent if valid else uniform type
    base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    # Ensure sum >= 0.01; base likely has large sum.
    # Normalize maybe? No need.
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # start iterative greedy local search with targeted adjustments
    n = best.size
    for iteration in range(num_iters):
        # compute convolution using FFT.
        # Using real FFT for speed.
        fft_len = 1 << (2 * n - 1).bit_length()  # next power of 2 >= 2n-1
        # Actually we can compute using np.fft.rfft of length m.
        a_fft = np.fft.rfft(best, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
        # find index of max
        k = int(np.argmax(conv))
        # Determine involved indices.
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # among these, find i with largest a_i * a_{k-i}
        # compute products vectorized
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        products = best[i_vals] * best[j_vals]
        # find index of max product
        max_pair_idx = int(np.argmax(products))
        i_max = i_vals[max_pair_idx]
        j_max = j_vals[max_pair_idx]
        # Choose a random scaling factor between 0.7 and 0.95
        factor = float(0.7 + 0.25 * rng.random())
        # Apply scaling to both i_max and j_max (if identical, scale once)
        candidate = best.copy()
        candidate[i_max] *= factor
        candidate[j_max] *= factor
        # Ensure nonnegative, clip
        np.clip(candidate, 0.0, 1000.0, out=candidate)
        # Evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
        # Occasionally do random block perturbation to escape.
        if rng.random() < 0.05:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _generate_sidon_set(length: int) -> list:
    """
    Greedy construction of a Sidon set (all pairwise sums distinct)
    within ``[0, length)``.  Returns a list of selected indices.
    """
    used = set()
    pos = []
    # theoretical upper bound for a Sidon set in this interval
    target = int(np.floor(np.sqrt(2 * length - 1)))
    for i in range(length):
        # quick reject if i+i already taken
        if i + i in used:
            continue
        conflict = False
        for s in pos:
            if i + s in used:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        pos.append(i)
        used.add(i + i)
        for s in pos[:-1]:
            used.add(i + s)
        if len(pos) >= target:
            break
    return pos

    and returns whichever candidate achieves the lower score.

    n = base.size

    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------

    fft_len = _next_pow2(2 * n - 1)

    for _ in range(num_steps):
        # current convolution and its dominant index
        A = np.fft.rfft(best, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
        peak_idx = int(np.argmax(conv))

        # indices that contribute to this peak
        i_start = max(0, peak_idx - (n - 1))
