    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # occasional symmetry bias
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=4000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the AC1 upper bound.
    The algorithm starts from a good analytic seed (the arcsine sequence)
    and performs a simple gradient‑like stochastic local search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initial candidates
    # ------------------------------------------------------------------
    parent = _parent_sequence()
    n = parent.size

    # start from the better of the parent or the arcsine construction
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    arcsine = _arcsine_sequence(n)
    arcsine_score = float(evaluate_sequence(arcsine.tolist()))
    if arcsine_score < best_score:
        best = arcsine
        best_score = arcsine_score

    current = best.copy()

    # ------------------------------------------------------------------
    # main stochastic optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # occasional big jump to avoid stagnation
        if rng.random() < 0.03:
            cand = _make_block_perturbation(current, rng)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score, current = cand, score, cand
                continue

        # compute full auto‑convolution and locate its maximal entry
        conv = np.convolve(current, current)
        idx_max = int(np.argmax(conv))

        # indices that actually influence this maximal entry
        lo = max(0, idx_max - (n - 1))
        hi = min(n - 1, idx_max)
        idx_range = np.arange(lo, hi + 1, dtype=int)

        # gradient magnitude for each contributing coordinate (2 * a[idx_max - i])
        grad_weights = current[idx_max - idx_range]
        if grad_weights.sum() > 0 and rng.random() < 0.9:
            probs = grad_weights / grad_weights.sum()
            i = int(rng.choice(idx_range, p=probs))
        else:                     # fallback to a uniform draw
            i = int(rng.integers(0, n))

        # pick a random partner index to exchange mass with
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # move a modest fraction of mass from i to j
        max_transfer = min(current[i], 0.5 * current[i] + 1e-12)
        delta = rng.random() * max_transfer
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score, current = cand, score, cand
            continue

        # simple simulated‑annealing acceptance to escape shallow minima
        temperature = 1.0 / (step + 1.0)
        if np.isfinite(score) and np.exp((best_score - score) / temperature) > rng.random():
            current = cand

    # ------------------------------------------------------------------
    # safety fallback – should never be needed
    # ------------------------------------------------------------------
    if best.size == 0:
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
