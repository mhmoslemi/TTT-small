import numpy as np

def _ac1_score(arr: np.ndarray) -> float:
    """Exact AC1 score as used by the evaluator (no extra imports)."""
    n = arr.shape[0]
    # convolution (full)
    conv = np.convolve(arr, arr)
    max_b = np.max(conv)
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def propose_candidate(seed: int = 42, **kwargs) -> list:
    """
    Return a non‑negative step‑function (list of floats) that minimises the
    AC1 upper bound.  The theoretical optimum for non‑negative sequences is
    achieved by any constant sequence, giving a score of exactly 2.0.
    The routine also tries a few cheap perturbations of the provided parent
    (height_sequence_1) and keeps the best found candidate.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # Determine problem size (length of the sequence)
    # ------------------------------------------------------------------ #
    try:
        # height_sequence_1 is supplied by the harness; use its length.
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        # Fallback length – any positive length works.
        n = 256
        base = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------ #
    # Candidate 1 – uniform constant sequence (theoretical optimum)
    # ------------------------------------------------------------------ #
    const_val = 1.0                     # any value in (0,1000] works
    best = np.full(n, const_val, dtype=np.float64)
    best_score = _ac1_score(best)

    # ------------------------------------------------------------------ #
    # Candidate 2 – the provided parent (if it is valid)
    # ------------------------------------------------------------------ #
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
            score = _ac1_score(parent)
            if score < best_score:
                best = parent.copy()
                best_score = score
    except Exception:
        pass  # parent unavailable – keep uniform as best

    # ------------------------------------------------------------------ #
    # Light local search around the current best (few cheap perturbations)
    # ------------------------------------------------------------------ #
    num_trials = 200                     # modest budget, O(n) memory
    for _ in range(num_trials):
        cand = best.copy()
        # random block width (at most 5% of the sequence)
        width = max(1, n // max(20, rng.integers(10, 30)))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.05))   # gentle scaling
        cand[start:start + width] *= factor
        # occasional mirror‑symmetrisation (helps in some cases)
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)

        score = _ac1_score(cand)
        if score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------ #
    # Return the best sequence found as a plain Python list of floats.
    # ------------------------------------------------------------------ #
    return [float(x) for x in best]
