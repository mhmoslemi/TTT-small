def compute_score(arr):
    n = arr.shape[0]
    L = 1 << ((2 * n - 1 - 1).bit_length())
    ft = np.fft.fft(arr, L)
    conv = np.real(np.fft.ifft(ft * ft))[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def gradient_step(arr, L, lr_factor=0.02):
    ft = np.fft.fft(arr, L)
    conv = np.real(np.fft.ifft(ft * ft))[:2*len(arr)-1]
    mean = conv.mean()
    diff = conv - mean
    ft_rev = np.fft.fft(diff[::-1], L)
    conv2 = np.real(np.fft.ifft(ft * ft_rev))
    # g_i = 4 * conv2[2n-2 - i]
    n = len(arr)
    g = 4.0 * conv2[2*n-2 - np.arange(n)]
    # step size adapt
    max_abs = np.max(np.abs(g))
    if max_abs == 0:
        step = 0.0
    else:
        step = lr_factor / max_abs
    new_arr = arr - step * g
    # Clip
    new_arr = np.clip(new_arr, 0.0, np.inf)
    total = new_arr.sum()
    if total > 0:
        new_arr = new_arr / total
    else:
        # fallback to uniform
        new_arr = np.ones_like(arr) / arr.size
    return new_arr

import numpy as np

def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Convolution a*a using FFT on a power‑of‑two size."""
    fft_a = np.fft.fft(a, size)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv


def _score(a: np.ndarray, fft_size: int) -> float:
    """Exact AC1 upper bound for a (scale‑invariant)."""
    n = a.size
    conv = _full_convolution(a, fft_size)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is forced to be >0 elsewhere
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _gradient_step(a: np.ndarray, fft_size: int, lr_factor: float = 0.02) -> np.ndarray:
    """One sub‑gradient step aimed at flattening the convolution."""
    n = a.size
    # a*a
    fft_a = np.fft.fft(a, fft_size)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]

    # deviation from the mean (target flat vector)
    mean = conv.mean()
    diff = conv - mean                     # length 2n‑1

    # gradient of sum (conv‑mean)²  w.r.t a   (4 * (diff * a_rev) convolution)
    fft_diff_rev = np.fft.fft(diff[::-1], fft_size)
    conv2 = np.real(np.fft.ifft(fft_a * fft_diff_rev))   # length ≥ 3n‑2

    # extract the needed n components
    g = 4.0 * conv2[2 * n - 2 - np.arange(n)]

    # adaptive step size
    max_abs = np.max(np.abs(g))
    step = lr_factor / max_abs if max_abs > 0 else 0.0

    new_a = a - step * g
    new_a = np.clip(new_a, 0.0, np.inf)

    total = new_a.sum()
    if total > 0:
        new_a /= total
    else:
        new_a = np.ones_like(a) / n
    return new_a
