import numpy as np

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by `evaluate_sequence`.
    The routine uses a tiny local‑search around a uniform baseline and
    falls back to the uniform solution if nothing improves.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Determine the problem size (length of the sequence)
    # ------------------------------------------------------------
    try:
        # `height_sequence_1` is supplied by the harness if a parent exists.
        n = int(len(height_sequence_1))
    except Exception:
        # No parent; pick a modest default length.
        n = 256

    # ------------------------------------------------------------
    # Baseline: a uniform positive sequence.
    # For any non‑negative vector the AC1 bound is at least 2,
    # and the uniform vector attains this lower bound.
    # ------------------------------------------------------------
    best = np.full(n, 1.0, dtype=np.float64)          # uniform heights = 1
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------
    # If a parent sequence is available, keep it as a candidate.
    # It may already be optimal; we compare its score.
    # ------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
        if (parent.ndim == 1 and parent.size > 0 and
                np.isfinite(parent).all()):
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent
                best_score = parent_score

    # ------------------------------------------------------------
    # Very cheap stochastic local search.
    # Each proposal perturbs a random block of the current best.
    # ------------------------------------------------------------
    for _ in range(num_steps):
        cand = best.copy()

        # Choose a block width proportional to n (but at least 1 element).
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        # Multiplicative perturbation with a log‑normal factor.
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor

        # Occasionally enforce a mild symmetry (helps in some cases).
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])

        # Clip to the admissible range.
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Evaluate the candidate.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------
    # Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------
    return best.tolist()
