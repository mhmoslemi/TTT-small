        trial = best.copy()
        trial[start:start + width] *= factor
        # Clip to the admissible [0,1000] interval
        np.clip(trial, 0.0, 1000.0, out=trial)

        score = float(evaluate_sequence(trial.tolist()))
        if np.isfinite(score) and score < best_score:
            best = trial
            best_score = score

    # Return the best candidate found (guaranteed non‑empty and valid)
    return best.tolist()
