import numpy as np

# The evaluator is pre‑imported in the harness as evaluate_sequence.
# It will be used directly; we must not redefine it.

def _initial_array():
    """Return a default non‑negative array if height_sequence_1 is unavailable."""
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0 or not np.isfinite(arr).all():
            raise ValueError
        return arr
    except Exception:
        return np.ones(256, dtype=np.float64)

def _score(seq):
    """Exact score using the supplied evaluator (returns a Python float)."""
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed=42, **kwargs):
    """Return a sequence of non‑negative heights with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Base length and fallback parent.
    # ------------------------------------------------------------------
    parent = _initial_array()
    n = parent.size
    best_seq = parent.copy()
    best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # 2. Structured families (power‑law and exponential) – cheap grid search.
    # ------------------------------------------------------------------
    # Power‑law a_i = (i + offset)^{-p}
    exponents = [0.45, 0.5, 0.55]                # p values around the analytic optimum
    offsets   = [0.0, 0.5, 1.0, 2.0, 5.0]       # small shifts to avoid the singularity at i=0
    idx = np.arange(n, dtype=np.float64) + 1.0   # i+1, common part of all formulas

    for p in exponents:
        for off in offsets:
            a = (idx + off) ** (-p)
            a = np.clip(a, 0.0, 1000.0)
            s = _score(a)
            if s < best_score:
                best_score, best_seq = s, a.copy()
            # reversed version (mirrored shape)
            ar = a[::-1]
            s = _score(ar)
            if s < best_score:
                best_score, best_seq = s, ar.copy()

    # Exponential a_i = exp(-lam * i)
    lambdas = [0.0005, 0.001, 0.002, 0.004]
    for lam in lambdas:
        a = np.exp(-lam * (np.arange(n, dtype=np.float64)))
        a = np.clip(a, 0.0, 1000.0)
        s = _score(a)
        if s < best_score:
            best_score, best_seq = s, a.copy()
        ar = a[::-1]
        s = _score(ar)
        if s < best_score:
            best_score, best_seq = s, ar.copy()

    # ------------------------------------------------------------------
    # 3. Simple stochastic refinement (few hundred cheap moves).
    # ------------------------------------------------------------------
    for _ in range(300):
        # pick a random contiguous block and scale it slightly
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 20))
        stop = min(n, start + width)
        factor = np.exp(rng.normal(0.0, 0.05))
        cand = best_seq.copy()
        cand[start:stop] *= factor

        # occasional tiny swap to break symmetry
        if rng.random() < 0.15:
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        cand = np.clip(cand, 0.0, 1000.0)
        s = _score(cand)
        if np.isfinite(s) and s < best_score:
            best_score, best_seq = s, cand.copy()

    # ------------------------------------------------------------------
    # 4. Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
