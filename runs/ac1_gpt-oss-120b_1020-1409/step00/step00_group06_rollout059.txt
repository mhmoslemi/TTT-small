import numpy as np

def propose_candidate(seed=42, max_iters=1500, reduce_factor=0.9, top_percent=20, redistribute=True, **kwargs):
    rng = np.random.default_rng(seed)
    # Start
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a = np.clip(a, 0.0, 1000.0)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            raise ValueError
    except Exception:
        # fallback uniform vector of ones
        a = np.ones(256, dtype=np.float64)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = a.size
    no_improve_counter = 0
    for it in range(max_iters):
        # compute convolution
        b = np.convolve(a, a)
        k_max = int(np.argmax(b))
        # compute overlapping indices
        start = max(0, k_max - (n - 1))
        end = min(n - 1, k_max)
        i_idx = np.arange(start, end+1)
        j_idx = k_max - i_idx
        products = a[i_idx] * a[j_idx]
        # Determine threshold for top contribution
        if products.size == 0:
            # cannot improve, break
            break
        # Use percentiles
        th = np.percentile(products, 100 - top_percent)  # top X%
        mask_i = products >= th
        # indices to reduce
        reduce_i = i_idx[mask_i]
        reduce_j = j_idx[mask_i]
        # create boolean mask for reduction
        reduce_mask = np.zeros_like(a, dtype=bool)
        reduce_mask[reduce_i] = True
        reduce_mask[reduce_j] = True
        # Apply reduction factor
        a_new = a.copy()
        a_new[reduce_mask] *= reduce_factor
        # Clip after reduction
        a_new = np.clip(a_new, 0.0, 1000.0)
        # Optional redistribution to keep sum
        if redistribute:
            sum_before = a.sum()
            sum_after = a_new.sum()
            diff = sum_before - sum_after
            if diff > 1e-12:
                # distribute diff uniformly over not-reduced entries
                non_red_indices = np.where(~reduce_mask)[0]
                if non_red_indices.size > 0:
                    add_each = diff / non_red_indices.size
                    a_new[non_red_indices] += add_each

                    a_new = np.clip(a_new, 0.0, 1000.0)
        # Evaluate
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_new.copy()
            best_score = score
            a = a_new

import numpy as np

# -------------------------------------------------------------
# Helper: small random block perturbation (memory O(n))
# -------------------------------------------------------------
def block_perturbation(base, rng):
    """Return a non‑negative O(n) perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # pick a block whose width is a modest fraction of n
    width = int(rng.integers(max(1, n // 150), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation of the block
    cand[start:stop] *= np.exp(rng.normal(0.0, 0.08))
    # occasional mirroring to keep a bit of symmetry
    if rng.random() < 0.33:
        cand[start:stop] = 0.5 * (cand[start:stop] + cand[start:stop][::-1])
    # enforce the hard bounds required by the evaluator
    return np.clip(cand, 0.0, 1000.0)

    performs a focused “peak‑flattening’’ search.
    """

    # -----------------------------------------------------------------

    except Exception:                     # parent missing or invalid
        a = np.ones(256, dtype=np.float64) * 0.5

    n = a.size

    # -----------------------------------------------------------------
    # 2) keep the best candidate seen so far
    # -----------------------------------------------------------------
    best = a.copy()

    if not np.isfinite(best_score):

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    t = np.arange(n, dtype=np.float64) / (n - 1)

    hann = hann * 1000.0                                 # respect the upper bound
    # ensure a tiny offset so sum > 0.01

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    no_improve = 0                         # stop‑early counter
    for it in range(max_iters):
        # ---- compute full autocorrelation ----
        conv = np.convolve(a, a)           # length 2n-1, O(n) memory

        start = max(0, k_max - (n - 1))
