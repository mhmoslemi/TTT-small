import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a monotone power‑law ramp sequence.
    The exponent is chosen from a small list to minimise the
    AC1 upper bound 2·n·max(conv)/sum².
    """
    # Length of the problem (use the provided parent if present)
    parent_arr = globals().get('height_sequence_1')
    if isinstance(parent_arr, np.ndarray) and parent_arr.ndim == 1 and parent_arr.size > 0:
        n = int(parent_arr.size)
        parent = np.clip(parent_arr.astype(np.float64), 0.0, 1000.0)
    else:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Start with the parent as a valid fallback
    best_seq = parent
    best_score = float(evaluate_sequence(parent.tolist()))

    # Try several power‑law ramps a[i] = (i/(n-1))**p * 1000
    # Theoretically, larger p give smaller scores (≈2·C(p) where C(p)↓).
    for p in (1, 2, 3, 4, 5, 6):
        t = np.linspace(0.0, 1.0, n, dtype=np.float64)          # normalized positions
        seq = (t ** p) * 1000.0                                 # ramp, max value exactly 1000
        seq = np.clip(seq, 0.0, 1000.0)                          # safety clipping

        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq

    # Return a plain Python list of floats as required by the harness
    return [float(v) for v in best_seq]
