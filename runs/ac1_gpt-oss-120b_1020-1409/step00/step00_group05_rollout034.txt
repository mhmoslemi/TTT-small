    # compute contributions vector g:
    n = a.size
    left = max(0, k_max - (n-1))
    right = min(n-1, k_max)
    idx = np.arange(left, right+1)
    g = np.zeros_like(a)
    g[idx] = a[k_max - idx]
    # step: new_a = a - alpha * g
    new_a = a - alpha * g
    # ensure nonnegativity
    new_a = np.maximum(0.0, new_a)
    # mass lost = sum(a) - sum(new_a)
    lost = a.sum() - new_a.sum()
    # redistribute lost uniformly
    if lost > 0:
        new_a += lost / n
    # clip to 1000
    new_a = np.clip(new_a, 0.0, 1000.0)
    return new_a
