def _convolution_max(arr: np.ndarray) -> float:
    """Linear convolution max using FFT (O(n log n))."""
    n = arr.size
    # length for zero‑padded FFT (next power of two >= 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    # keep only the linear part (first 2n‑1 samples)
    return float(np.max(conv[: 2 * n - 1]))

def _objective(arr: np.ndarray) -> float:

    s = arr.sum()
    if s < 0.01:

    m = _convolution_max(arr)
    return 2.0 * arr.size * m / (s * s)
