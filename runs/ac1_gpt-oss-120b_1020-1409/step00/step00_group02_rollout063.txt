import numpy as np

def _make_powerlaw_candidate(n: int, alpha: float) -> np.ndarray:
    """
    Return a non‑negative step function of length ``n`` with entries proportional
    to (i+1)^(-alpha), normalised to have ℓ1‑norm 1.
    """
    # indices 1 … n (avoid zero‑power issues)
    idx = np.arange(1, n + 1, dtype=np.float64)
    # power‑law weights
    w = np.power(idx, -alpha)
    # normalise to sum == 1  (scale does not affect the score)
    w /= w.sum()
    # enforce the hard bounds required by the evaluator
    np.clip(w, 0.0, 1000.0, out=w)
    return w

def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a step‑function that empirically beats the uniform 2‑bound.
    Several power‑law exponents are tried; the one with the smallest
    AC1‑upper‑bound (as reported by ``evaluate_sequence``) is returned.
    """
    rng = np.random.default_rng(seed)

    # Determine the length of the parent construction if present.
    try:
        n = int(height_sequence_1.shape[0])   # type: ignore[name-defined]
    except Exception:
        # Fallback: a modest length when the parent is unavailable.
        n = 256

    # Candidate exponents – values very close to 1 give a rapidly decreasing
    # sequence which strongly reduces the convolution peak.
    exponents = [0.98, 0.99, 0.995, 0.997, 0.999]

    best_vec = None
    best_score = np.inf

    # Try each exponent, keep the best (few evaluations keep runtime modest).
    for a in exponents:
        cand = _make_powerlaw_candidate(n, a)
        # Exact evaluation (the harness supplies ``evaluate_sequence``).
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            score = np.inf
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand

    # If all attempts failed (should not happen), fall back to a uniform vector.
    if best_vec is None:
        best_vec = np.ones(n, dtype=np.float64)
        best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Return a plain Python list of floats as required.
    return best_vec.tolist()
