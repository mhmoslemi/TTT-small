def propose_candidate(seed=42, num_const=300, num_patterns=100, **kwargs):

    # --- initial candidate (parent if present) ---------------------------
    if arr is not None:

    else:

    # Helper: evaluate and possibly update the best sequence
    def try_candidate(seq):
        nonlocal best_seq, best_score
        # seq is a NumPy 1‑D array (float64)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # Ensure a minimal positive constant so that the sum constraint passes.
    min_c = max(1e-12, 0.01 / n)
