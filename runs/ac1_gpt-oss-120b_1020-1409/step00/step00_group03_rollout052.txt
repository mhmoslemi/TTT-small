import numpy as np

def evaluate_score(seq):
    n = seq.shape[0]
    s = np.sum(seq)
    if s == 0:
        return np.inf
    c = np.convolve(seq, seq)
    return 2.0 * n * np.max(c) / (s * s)

def make_block_perturbation(base, rng):
    n = base.size
    candidate = base.copy()
    # choose random block width between n//200 and n//12
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.12))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def gradient_step(seq, lr):
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    # compute gradient: g_i = 2 * seq[k - i] if valid else 0
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)
    grad = np.zeros_like(seq)
    grad[valid] = seq[partner[valid]]
    # factor 2 not needed for direction; using 1 step size.
    cand = seq - lr * grad
    np.clip(cand, 0.0, 1000.0, out=cand)
    s = np.sum(cand)
    if s > 0:
        cand *= (np.sum(seq) / s)
    return cand

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # try to use provided parent
    parent = None
    if 'height_sequence_1' in globals():
        h = globals()['height_sequence_1']
        if isinstance(h, np.ndarray) and h.ndim == 1:
            parent = np.clip(h.astype(np.float64), 0.0, 1000.0)
    if parent is None or parent.size == 0 or not np.isfinite(parent).all():
        n = 256
        parent = np.ones(n, dtype=np.float64)
    n = parent.size
    best = parent.copy()
    best_score = evaluate_score(best)

    # Keep a copy of original sum for gradient normalization
    orig_sum = np.sum(best)

    for step in range(num_steps):
        if step % 50 == 0:
            # random permutation
            cand = np.random.permutation(best)
            np.clip(cand, 0.0, 1000.0, out=cand)
        elif step % 15 == 0:
            # gradient step with decreasing learning rate
            lr = 0.5 * (1.0 - step / num_steps) + 0.01
            cand = gradient_step(best, lr)
        else:
            # block perturbation
            cand = make_block_perturbation(best, rng)

        # keep sum close to original (helps denominator)
        s = np.sum(cand)
        if s > 0:
            cand *= (orig_sum / s)
        np.clip(cand, 0.0, 1000.0, out=cand)

        score = evaluate_score(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # update original sum to new sum for future normalization
            orig_sum = np.sum(best)

    # fallback to uniform if still bad
    if not np.isfinite(best_score) or best_score > 1e9:
        best = np.full(n, 1.0, dtype=np.float64)
    return [float(x) for x in best]

import numpy as np

def _candidate_score(seq: np.ndarray) -> float:
    """Exact score matching `evaluate_sequence`."""
    n = seq.shape[0]
    s = np.sum(seq)
    if s == 0.0:
        return np.inf
    conv = np.convolve(seq, seq)
    return 2.0 * n * np.max(conv) / (s * s)


def _block_perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = base.size
    cand = base.copy()
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _gradient_step(vec: np.ndarray, lr: float) -> np.ndarray:
    """One step that reduces the current maximal convolution entry."""
    n = vec.size
    conv = np.convolve(vec, vec)
    k = int(np.argmax(conv))                     # index of the peak
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)
    grad = np.zeros_like(vec)
    grad[valid] = vec[partner[valid]]            # ∂max/∂a_i ≈ a_{k-i}
    cand = vec - lr * grad
    np.clip(cand, 0.0, 1000.0, out=cand)
    # keep the total mass unchanged (scaling does not affect the score)
    tot = np.sum(cand)
    if tot > 0.0:
        cand *= (np.sum(vec) / tot)
    return cand


def propose_candidate(seed: int = 42, num_steps: int = 2500, **kwargs) -> list:
    """Search for a step‑function that lowers the AC1 bound."""
    rng = np.random.default_rng(seed)

    # ---- initialise ---------------------------------------------------------
    try:
        parent = height_sequence_1
        if isinstance(parent, np.ndarray) and parent.ndim == 1:
            parent = np.clip(parent.astype(np.float64), 0.0, 1000.0)
        else:
            raise Exception
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = _candidate_score(best)
    current_mass = np.sum(best)          # total sum – kept constant during normalisation

    # ---- search -------------------------------------------------------------
    for step in range(num_steps):
        # choose a move type
        if step % 50 == 0:                # full random permutation
            cand = np.random.permutation(best)
            np.clip(cand, 0.0, 1000.0, out=cand)

        elif step % 15 == 0:              # gradient descent on the peak
            lr = 0.5 * (1.0 - step / num_steps) + 0.01
            cand = _gradient_step(best, lr)

        elif step % 7 == 0:               # fresh random Dirichlet draw
            # Dirichlet with α≈1 gives a spread of values, scaling does not matter
            raw = rng.dirichlet(np.ones(n))
            cand = raw * current_mass
            np.clip(cand, 0.0, 1000.0, out=cand)

        else:                             # modest block perturbation
            cand = _block_perturb(best, rng)

        # normalise mass (does not change the score) and enforce bounds
        tot = np.sum(cand)
        if tot > 0.0:
            cand *= (current_mass / tot)
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc = _candidate_score(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            current_mass = np.sum(best)

    # ---- fallback -----------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.full(n, 1.0, dtype=np.float64)

    return [float(v) for v in best]
