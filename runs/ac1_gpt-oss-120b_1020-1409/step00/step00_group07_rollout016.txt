import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a simple linear ramp sequence that achieves a low AC1 upper bound.
    The construction uses the length of the provided `height_sequence_1`
    (if available) and scales the ramp to stay within the [0, 1000] limits.
    """
    # Try to obtain the length from the parent construction.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(parent.size)
    except Exception:
        # Fallback length if the parent is missing.
        n = 256

    # Guard against pathological lengths.
    if n < 2:
        n = 2

    # Scale so that the last entry hits the clipping bound exactly.
    scale = 1000.0 / (n - 1)

    # Linear ramp: a_i = i * scale  (i = 0, 1, ..., n-1)
    candidate = np.arange(n, dtype=np.float64) * scale

    # Enforce evaluator bounds absolutely (clip to [0, 1000]).
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Return as a plain Python list of floats.
    return [float(x) for x in candidate]
