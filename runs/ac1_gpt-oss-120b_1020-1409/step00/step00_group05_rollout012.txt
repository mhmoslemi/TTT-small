    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as measured by `evaluate_sequence`.  The algorithm simply
    compares a constant sequence with the provided parent sequence (if any) and
    chooses the better one.
    """

    # ----------------------------------------------------------------------

        # Validate the parent – it must be a finite 1‑D array.

    except Exception:
        # Fallback: use a modest length when no parent is available.
        n = 256

    # the theoretical lower bound 2.0 for the objective, because max(conv)=∑a_i²
    # and the Cauchy–Schwarz inequality makes the ratio 2·n·∑a_i²/(∑a_i)² ≥ 2.
