import numpy as np

# --------------------------------------------------------------
# Helper: convolution via FFT (real input, real output)
# --------------------------------------------------------------
def _convolve_fft(x: np.ndarray) -> np.ndarray:
    """Return full convolution x*x using real FFT (O(n log n))."""
    n = x.shape[0]
    # length needed for full convolution
    m = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fx * fx, n=m)[: 2 * n - 1]
    return conv

        new_cur[j_max] -= delta
        new_cur[p] += delta
        new_cur[q] += delta

        # respect the hard 0–1000 bounds (unlikely to be hit here)
        np.clip(new_cur, 0.0, 1000.0, out=new_cur)

        # --------------------------------------------------------------
        # evaluate the new vector (fast using FFT)
        # --------------------------------------------------------------
        new_conv = _convolve_fft(new_cur)
        new_max = new_conv[np.argmax(new_conv)]
        new_score = 2.0 * n * new_max / (np.sum(new_cur) ** 2)

        # keep the change only if it improves the objective
        if new_score < cur_score:
            cur = new_cur
            cur_score = new_score
            if new_score < best_score:
                best = new_cur.copy()
                best_score = new_score

    # --------------------------------------------------------------
    # final safety: make absolutely sure the returned vector is valid
    # --------------------------------------------------------------
    if not np.isfinite(best_score) or best.size == 0:
        # as a last resort return the plain uniform construction
        best = np.full(n, target_sum / n, dtype=np.float64)

    # convert to plain Python floats as required by the harness
    return [float(v) for v in best]
