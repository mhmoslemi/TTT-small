import numpy as np

# ----------------------------------------------------------------------
# Helper: fast max of self‑convolution using FFT (linear convolution)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, fft_len: int) -> float:
    """Return max value of a*a (linear convolution) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) whose
    AC1 upper bound is as low as possible.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length n and a reference array if provided by the harness
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is supplied in the execution environment
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # fallback length
        n = 256
        base = np.full(n, 1.0)

    # ------------------------------------------------------------------
    # Pre‑compute FFT length (next power of two ≥ 2n‑1)
    # ------------------------------------------------------------------
    fft_len = 1
    while fft_len < 2 * n - 1:
        fft_len <<= 1

    # ------------------------------------------------------------------
    # Scoring function (fast, using FFT)
    # ------------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        total = vec.sum()
        if total <= 1e-12:
            return np.inf
        mx = _max_conv(vec, fft_len)
        return 2.0 * n * mx / (total * total)

    # ------------------------------------------------------------------
    # Store best candidate found (fast‑score and exact‑score)
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Utility: evaluate with the official scorer (exact, safe)
    # ------------------------------------------------------------------
    def _exact_score(vec: np.ndarray) -> float:
        # the harness provides evaluate_sequence; use it for final check
        return float(evaluate_sequence(vec.tolist()))

    # ------------------------------------------------------------------
    # Candidate generators
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64) / (n - 1)  # in [0,1]

    # 1. Uniform (reference)
    candidates = [np.ones(n, dtype=np.float64)]

    # 2. Normalised parent (if available)
    parent_norm = base.copy()
    if parent_norm.max() > 0:
        parent_norm = parent_norm / parent_norm.max()
        candidates.append(parent_norm)

    # 3. Exponential decays a_i = exp(-c * idx)
    for c in np.linspace(0.2, 2.0, 10):
        candidates.append(np.exp(-c * idx))

    # 4. Power decays a_i = (1 - idx)^p
    for p in (0.2, 0.5, 1.0, 2.0, 3.0):

    # 5. Mixed exponential + offset a_i = eps + (1-eps)*exp(-c*idx)
    for eps in (0.0, 0.05, 0.10, 0.20):
        for c in (0.5, 1.0, 1.5):
            candidates.append(eps + (1.0 - eps) * np.exp(-c * idx))

    # 6. Random uniform draws
    for _ in range(5):
        candidates.append(rng.random(n))

    # 7. Random exponential draws (scale ≈0.5)
    for _ in range(5):
        cand = rng.exponential(scale=0.5, size=n)
        candidates.append(cand / max(cand.max(), 1e-12))

    # 8. Interleaved sorted values (low/high alternation)
    rnd = rng.random(n)
    srt = np.sort(rnd)
    inter = np.empty_like(srt)
    half = n // 2
    inter[0::2][:half] = srt[:half]
    inter[1::2][: n - half] = srt[half:]
    if n % 2:

    candidates.append(inter)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    for cand in candidates:
        # ensure non‑negative and bounded (values are already ≤1)
        cand = np.clip(cand, 0.0, 1.0)

        if fast < best_score:

            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_vec = cand.copy()

    # ------------------------------------------------------------------
    # Local refinement: simple hill‑climbing by moving a random amount
    # from one index to another while keeping the sum constant.
    # ------------------------------------------------------------------
    if best_vec is None:
        # should never happen, but keep a safe fallback
        best_vec = np.ones(n, dtype=np.float64)
        best_score = _exact_score(best_vec)

    # number of refinement steps (kept modest for runtime safety)
    for _ in range(300):
        i, j = rng.integers(0, n, size=2)
        if i == j or best_vec[i] <= 0.0:

        new_vec = best_vec.copy()
        new_vec[i] -= delta
        new_vec[j] += delta
        # stay inside the [0,1] box (scaling does not affect the ratio)

        new_score = _score(new_vec)
        if new_score < best_score:
            exact = _exact_score(new_vec)
            if np.isfinite(exact) and exact < best_score:
                best_score = exact
                best_vec = new_vec

    # ------------------------------------------------------------------
    # Return the best vector found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return best_vec.tolist()
