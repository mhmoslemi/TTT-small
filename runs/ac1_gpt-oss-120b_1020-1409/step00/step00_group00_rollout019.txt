import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a monotone linear ramp that yields a low AC1 upper bound.
    The ramp goes from 0 up to the allowed maximum 1000 and uses the
    length of the provided `height_sequence_1` when available.
    """
    # Determine the length to use.
    try:
        # height_sequence_1 is supplied by the harness.
        n = int(height_sequence_1.shape[0])
        if n < 2:
            raise Exception
    except Exception:
        # Fallback to a modest size if the parent is missing or invalid.
        n = 256

    # Scale factor so the last element is exactly 1000.
    scale = 1000.0 / max(1, n - 1)

    # Linear ramp: 0, scale, 2*scale, ..., (n‑1)*scale
    seq = np.arange(n, dtype=np.float64) * scale

    # Ensure bounds (clipping is a safety net for rounding errors).
    seq = np.clip(seq, 0.0, 1000.0)

    # Return a plain Python list of floats as required.
    return [float(v) for v in seq]
