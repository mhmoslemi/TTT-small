def compute_fft_max(seq):
   n=len(seq)
   size = 1 << ((2*n-1).bit_length())
   fft_seq = np.fft.rfft(seq, size)
   conv = np.fft.irfft(fft_seq * fft_seq, size)[:2*n-1]
   return np.max(conv)

import numpy as np

# ---------- Helper functions (linear memory) ----------
def _fft_conv_max(x: np.ndarray) -> float:
    """Return max value of the full convolution x*x using a power‑of‑2 FFT."""
    n = x.size
    # length of linear convolution = 2*n-1, choose next power of two
    size = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    # numerical noise may produce tiny negatives
    conv = np.maximum(conv, 0.0)
    return float(conv.max())


def _objective(x: np.ndarray) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = x.size
    s = float(x.sum())
    if s < 1e-12:
        return np.inf
    maxb = _fft_conv_max(x)
    return 2.0 * n * maxb / (s * s)


# ---------- Main search routine ----------
def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """
    Return a non‑empty list of non‑negative coefficients.
    The routine starts from a random Dirichlet‑like point and performs a
    simple greedy reduction of the dominant convolution bin.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise a starting vector.
    #    Prefer the provided parent if it exists and looks sane.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = None

    if (
        parent is None
        or parent.ndim != 1
        or parent.size == 0
        or not np.isfinite(parent).all()
    ):
        # fallback: uniform vector (sum = sqrt(2*n) keeps the scale used in papers)
        n = 256
        base = np.ones(n, dtype=np.float64)
    else:
        base = parent.copy()
        n = base.size

    # Ensure bounds and a positive total mass
    base = np.clip(base, 0.0, 1000.0)
    if base.sum() < 0.01:
        base = np.ones_like(base)

    # Random start: exponential (Dirichlet) draw, then renormalise to the same sum
    cur = rng.exponential(scale=1.0, size=n)
    cur *= base.sum() / cur.sum()
    cur = np.clip(cur, 0.0, 1000.0)

    # Keep the best point seen so far (exactly evaluated with the provided evaluator)
    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Pre‑compute FFT size once (power of two ≥ 2*n-1)
    fft_size = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Greedy improvement loop.
    #    At each step we look at the convolution bin with maximal value,
    #    pick the pair (i,j) contributing most to it and shift a small
    #    amount of mass away from that pair towards the smallest entry.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # Full convolution and the index of its maximum bin
        fx = np.fft.rfft(cur, fft_size)
        conv = np.fft.irfft(fx * fx, fft_size)[: 2 * n - 1]
        conv = np.maximum(conv, 0.0)
        k_max = int(conv.argmax())

        # Indices that actually appear in this bin
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        i_vals = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_vals = k_max - i_vals

        # Contribution of each ordered pair (i,j) to this bin
        prod = cur[i_vals] * cur[j_vals]
        max_pair_idx = int(prod.argmax())
        i0 = int(i_vals[max_pair_idx])
        j0 = int(j_vals[max_pair_idx])

        # Decide a transfer amount (up to 30 % of the smaller entry)
        limit = min(cur[i0], cur[j0])
        if limit <= 0.0:
            continue
        delta = limit * rng.random() * 0.30
        if delta <= 0.0:
            continue

        # Receiver: the current smallest entry (not i0 or j0)
        mask = np.ones(n, dtype=bool)
        mask[i0] = False
        mask[j0] = False
        receiver = int(np.argmin(cur[mask]))
        # map back to original index
        recv_idx = int(np.flatnonzero(mask)[receiver])

        # Apply the mass move (preserve total sum)
        cand = cur.copy()
        cand[i0] -= delta
        cand[j0] -= delta
        cand[recv_idx] += 2.0 * delta  # both subtractions go to the same receiver
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Fast approximate objective (sufficient for a greedy decision)
        approx_score = _objective(cand)

        # Accept only if it strictly improves the approximate score
        cur_score = _objective(cur)
        if approx_score < cur_score:
            cur = cand
            # Full exact check only for the current best
            exact = float(evaluate_sequence(cur.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = cur.copy()
                best_score = exact

    # ------------------------------------------------------------------
    # 3. Return the best vector found (as a plain Python list).
    # ------------------------------------------------------------------
    return [float(v) for v in best]
