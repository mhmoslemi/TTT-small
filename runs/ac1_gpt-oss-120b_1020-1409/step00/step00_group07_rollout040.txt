import numpy as np

# ----------------------------------------------------------------------
# Helper – exact score according to the supplied evaluator (re‑implemented)
# ----------------------------------------------------------------------
def _score(arr: np.ndarray) -> float:
    """Return the objective 2*n*max(conv)/sum^2 for a non‑negative array."""
    n = arr.size
    conv = np.convolve(arr, arr)          # length 2*n-1
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:                      # keep same guard as evaluate_sequence
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))
