import numpy as np

def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Return an O(n)-memory non‑negative perturbation of ``base``.
    A random contiguous block is multiplied by a log‑normal factor,
    optionally symmetrised, and then clipped to the allowed range.
    """
    cand = base.copy()
    n = cand.size
    # block width between 1 and roughly n/20 (but at least 1)
    max_width = max(2, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.10)))
    cand[start:stop] *= factor
    # occasional mirror symmetrisation to keep the shape balanced
    if rng.random() < 0.20:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the problem’s clipping bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    the AC1 upper bound evaluated by ``evaluate_sequence``.
    The algorithm starts from a constant sequence (the theoretical optimum)
    and performs a lightweight stochastic local search, keeping the best
    candidate seen.  Memory usage is linear in the sequence length.
    """

        n = int(height_sequence_1.shape[0])
    except Exception:
        pass

    # Initial score (exact evaluation).
