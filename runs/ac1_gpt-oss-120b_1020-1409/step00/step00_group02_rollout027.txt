import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a Sidon (B_2) set of indices – all sums i+j are unique.
# ----------------------------------------------------------------------
def generate_sidon_set(n: int, K: int, rng: np.random.Generator):
    """Return up to K indices in [0, n) whose pairwise sums are all distinct."""
    selected = []
    sums = set()
    # random order helps to avoid pathological early choices
    order = rng.permutation(n)
    for i in order:
        # check whether i+i collides
        if (i + i) in sums:
            continue
        # check all i+j with already chosen j
        conflict = False
        for j in selected:
            if (i + j) in sums:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        sums.add(i + i)
        for j in selected:
            sums.add(i + j)
        selected.append(i)
        if len(selected) >= K:
            break
    return selected
