def compute_score(arr):
    # Compute 2 * n * max(conv(arr,arr)) / sum(arr)**2 using FFT.
    n = arr.size
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return np.inf
    # compute convolution using FFT (full convolution)
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)
