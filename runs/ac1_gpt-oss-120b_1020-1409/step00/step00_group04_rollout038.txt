import numpy as np

def gen_powerlaw(alpha, offset, length):
    """a_i = (i + offset + 1)^{-alpha}."""
    idx = np.arange(length, dtype=np.float64)
    return (idx + offset + 1) ** (-alpha)


def block_perturb(arr, rng):
    """Small O(n) perturbation used for local refinement."""
    n = arr.size
    # random block width roughly between n/200 and n/12
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    out = arr.copy()
    out[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        out = 0.5 * (out + out[::-1])
    return np.clip(out, 0.0, 1000.0)


def propose_candidate(seed=42, num_trials=500, **kwargs):
    """Search for a step‑function that lowers the AC1 bound.

    The routine tries a few analytic families (power‑law decays) and many
    random Dirichlet‑like vectors, then refines the best candidate
    with cheap block perturbations.
    """
    rng = np.random.default_rng(seed)

    # start from the provided parent if it exists, otherwise fall back
    # ------------------------------------------------------------------

        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

            cand = np.clip(cand, 0.0, 1000.0)
