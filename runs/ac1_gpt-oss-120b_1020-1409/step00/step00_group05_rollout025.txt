def fast_score(seq):
    # compute sum, then convolution via FFT, find max, compute ratio 2*n*max/sum**2.
    # uses np.fft.rfft for real inputs.
    n = seq.size
    fa = np.fft.rfft(seq, n*2-1)
    fc = fa * fa
    conv = np.fft.irfft(fc, n*2-1)
    # conv length = 2n-1
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    max_b = conv[:2 * n - 1].max()
    return 2.0 * n * max_b / (sum_seq * sum_seq)

import numpy as np

# ------------------------------------------------------------
# fast approximation of the evaluator (uses FFT, O(n log n) time)
# ------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the AC1 upper bound for a given non‑negative sequence."""
    n = seq.shape[0]
    total = float(seq.sum())
    if total < 0.01 or not np.isfinite(total):
        return np.inf
    # length for a power‑of‑two FFT (covers full convolution)
    L = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)          # full convolution
    max_b = float(conv[: 2 * n - 1].max())
    return 2.0 * n * max_b / (total * total)

def propose_candidate(seed: int = 42, num_trials: int = 400, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimizes the AC1 upper bound defined by `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    num_trials : int
        Approximate total number of random candidates examined.
    **kwargs
        Ignored – present for interface compatibility.

    Returns
    -------
    list[float]
        Candidate height sequence (non‑negative, finite).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Base candidate: the supplied parent (if any) – otherwise a simple uniform.
    # ------------------------------------------------------------------
    try:

        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception()
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best_seq = parent.copy()
    best_score = _fast_score(best_seq)

    # keep a reference score from the official evaluator (may be slightly different)
    ref_score = float(evaluate_sequence(parent.tolist()))
    if ref_score < best_score:
        best_seq = parent.copy()
        best_score = ref_score

    # ------------------------------------------------------------------
    # 1) Random sparse masks  (binary values scaled to unit L1 norm)
    # ------------------------------------------------------------------
    # each mask uses a fixed density; we try several densities.
    densities = [0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85]
    trials_per_density = max(1, num_trials // len(densities))

    for dens in densities:
        for _ in range(trials_per_density):
            mask = (rng.random(n) < dens).astype(np.float64)
            s = mask.sum()
            if s < 1.0:
                continue                     # need at least one non‑zero entry
            weight = 1.0 / s                  # ensures L1‑norm = 1
            cand = mask * weight
            cand_score = _fast_score(cand)
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score

    # ------------------------------------------------------------------
    # 2) Local refinement: swap a few ones/zeros of the current best mask
    #    (keeps total L1 norm constant, only improves the pair‑sum distribution)
    # ------------------------------------------------------------------
    # Work with a binary mask derived from the current best sequence.
    # If the best sequence is not binary we still treat non‑zeros as a mask.
    mask = (best_seq > 0).astype(np.float64)
    s = mask.sum()
    if s >= 2:  # need both a 1 and a 0 to swap
        weight = 1.0 / s
        for _ in range(200):
            # Choose a random 1‑position and a random 0‑position
            ones = np.flatnonzero(mask)
            zeros = np.flatnonzero(1.0 - mask)
            i = rng.choice(ones)
            j = rng.choice(zeros)
            # Propose swap
            mask[i] = 0.0
            mask[j] = 1.0
            cand = mask * weight
            cand_score = _fast_score(cand)
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                # keep the new mask; continue from here
            else:
                # revert swap
                mask[i] = 1.0
                mask[j] = 0.0

    # ------------------------------------------------------------------
    # Final verification with the official evaluator (ensures clipping etc.)
    # ------------------------------------------------------------------
    # The evaluator clips negatives to 0 and caps at 1000, both of which
    # our construction already respects.
    final_score = float(evaluate_sequence(best_seq.tolist()))

    # cached best, keep the previous best (the score is monotone in the same
    # data, so this branch should never fire).
    if final_score > best_score:
        # recompute using the official score of the earlier best_seq
        best_seq = best_seq  # unchanged; placeholder for clarity
    # Return as plain Python list of floats
    return [float(x) for x in best_seq]
