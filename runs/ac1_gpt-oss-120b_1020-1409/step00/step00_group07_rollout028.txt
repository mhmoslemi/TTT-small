def conv_full_fft(a):

        j = k - i
        g[i] += a[j]
        g[j] += a[i]

import numpy as np

# ----- convolution (linear, zero‑padded) using FFT -----
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using a power‑of‑two FFT size."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=L)
    B = A * A                                      # element‑wise square → convolution in time
    conv = np.fft.irfft(B, n=L)[:2 * n - 1]        # keep the linear part
    return conv


# ----- gradient with respect to a set of convolution indices -----
def _gradient_for_indices(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the (soft‑max) objective restricted to the given k‑indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in idxs:
        # i runs over all positions that participate in a[i] * a[k-i]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        # each unordered pair (i, j) contributes 2*a_j to ∂b[k]/∂a_i
        g[i] += a[j]
        g[j] += a[i]
    return g


# ----- main entry point -----
def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """Memory‑bounded local optimisation of the AC1 bound."""
    rng = np.random.default_rng(seed)

    # ---- initialise from the provided parent if possible ----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)   # fallback uniform vector

    a = np.clip(parent, 0.0, 1000.0).copy()
    if a.sum() < 0.01:               # guard against degenerate start‑up
        a = np.ones_like(a)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    n = a.shape[0]
    K = 5                              # how many top convolution entries are used each step

    for _ in range(num_steps):
        # ----- current autocorrelation and its prominent indices -----
        b = _full_convolution_fft(a)
        if K >= b.size:
            top_idx = np.arange(b.size)
        else:
            top_idx = np.argpartition(b, -K)[-K:]

        # ----- compute a descent direction -----
        g = _gradient_for_indices(a, top_idx)
        d = -g
        d -= d.mean()                  # keep the direction inside the hyperplane Σa = const

        # ----- maximal safe step respecting non‑negativity -----
        neg = d < 0
        if np.any(neg):
            max_step = np.min(-a[neg] / d[neg])
            step = 0.5 * max_step       # modest step to stay away from the boundary
        else:
            step = 0.0

        # ----- propose a new candidate -----
        if step > 0.0:
            cand = a + step * d
        else:
            # small stochastic move when the gradient gives no direction
            cand = a * np.exp(rng.normal(0.0, 0.04, size=n))

        cand = np.clip(cand, 0.0, 1000.0)

        # occasional extra jitter to escape shallow plateaus
        if rng.random() < 0.03:
            cand = np.clip(cand * np.exp(rng.normal(0.0, 0.02, size=n)),
                           0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

        # accept the candidate as the new search point if it does not degrade
        if score <= best_score * 1.001:
            a = cand

    return [float(x) for x in best]
