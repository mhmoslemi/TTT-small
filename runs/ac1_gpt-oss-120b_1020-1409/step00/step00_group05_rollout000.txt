import numpy as np

# evaluate_sequence is provided by the harness.

def _uniform_candidate(length, value=1.0):
    """Return a uniform non‑negative candidate of given length."""
    return np.full(length, float(value), dtype=np.float64)


def _clip_and_finite(arr):
    """Clip to the allowed range and guarantee a finite numpy array."""
    arr = np.clip(arr, 0.0, 1000.0)
    # Force dtype to float64 for the evaluator.
    return arr.astype(np.float64, copy=False)


def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a list of non‑negative step‑function heights that (locally) minimize
    the AC1 upper bound evaluated by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # Determine the problem size.
    try:
        n = int(height_sequence_1.shape[0])  # provided by the harness
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = _clip_and_finite(parent)
        if parent.ndim != 1 or parent.size != n:
            raise ValueError
    except Exception:
        # Fallback: synthetic length if the parent is absent or malformed.
        n = 256
        parent = None

    # Start from a uniform vector – it is optimal for the given objective.
    best = _uniform_candidate(n, value=1.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # If a valid parent exists, evaluate it and keep it if it is better.
    if parent is not None:
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent.copy()
            best_score = parent_score

    # Simple local search: blend with the uniform vector and add tiny noise.
    # The objective is scale‑invariant, so we keep the magnitude unchanged.
    uniform_vec = _uniform_candidate(n, value=1.0)

    for _ in range(num_steps):
        # Blend current best with the uniform vector.
        blend = rng.uniform(0.01, 0.05)          # mixing proportion
        candidate = (1.0 - blend) * best + blend * uniform_vec

        # Add small zero‑mean Gaussian noise scaled to the current magnitude.
        noise_scale = 0.02 * (candidate.max() if candidate.max() > 0 else 1.0)
        candidate += rng.normal(0.0, noise_scale, size=n)

        # Ensure feasibility.
        candidate = _clip_and_finite(candidate)

        # Evaluate and keep if improved.
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    # Guarantee a valid return even if something went wrong.
    if best_score == float('inf') or not np.isfinite(best_score):
        best = _uniform_candidate(n, value=1.0)
    return [float(v) for v in best]
