def propose_candidate(seed=42, num_steps=500, **kwargs):
    rng = np.random.default_rng(seed)
    # Initialize parent
    try:
        A = np.asarray(height_sequence_1, dtype=np.float64)
        if A.ndim != 1 or A.size == 0 or not np.isfinite(A).all():
            raise ValueError
        parent = np.clip(A, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64) * 0.5
    n = parent.size

    # initial candidates: we will try few random seeds and keep best
    def score(arr):
        # perform clipping inside just in case
        arr = np.clip(arr, 0.0, 1000.0)
        total = arr.sum()
        if total < 0.01:
            return np.inf
        b = np.convolve(arr, arr)
        return 2.0 * n * b.max() / (total * total)

    best = parent.copy()
    best_score = score(best)

    # also try uniform candidate (maybe better)
    uniform = np.full(n, 1.0, dtype=np.float64) # arbitrary sum 1 but actual values not scaled
    uniform = np.clip(uniform, 0.0, 1000.0)
    uniform_score = score(uniform)
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # random Dirichlet candidate with concentration 0.5 (more spread)
    rand = rng.dirichlet(np.ones(n)*0.5) * 1000.0  # scaling to fill max?
    rand_score = score(rand)
    if rand_score < best_score:
        best = rand
        best_score = rand_score

    cand = best.copy()
    cand_score = best_score

    # main hill climbing loop
    for step in range(num_steps):
        n = cand.size
        # compute convolution and max shift
        b = np.convolve(cand, cand)
        max_idx = int(np.argmax(b))
        max_val = b[max_idx]

        # compute contributions for that shift
        i_min = max(0, max_idx - (n - 1))
        i_max = min(n - 1, max_idx)
        i_arr = np.arange(i_min, i_max + 1)

        prod = cand[i_arr] * cand[j_arr]

        # index of largest product
        max_pair_idx = int(np.argmax(prod))
        i_star = i_arr[max_pair_idx]
        j_star = j_arr[max_pair_idx]

        # step size as fraction of smallest of the two entries
        delta = 0.05 * min(cand[i_star], cand[j_star])

def _clipped_score(arr):
    """Return the exact AC1 score for a non‑negative array."""
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(arr, arr)
    return 2.0 * arr.size * conv.max() / (s * s)


def _initial_candidates(rng, n):
    """Generate a few diverse start vectors of length n."""
    # 1) the supplied parent, if any
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():

    uniform = np.full(n, 1.0, dtype=np.float64)

    exponential = rng.exponential(scale=500.0, size=n)

    return [parent, uniform, dirichlet, exponential]

        j_arr = k_max - i_arr
        prod = cand[i_arr] * cand[j_arr]

        pair_idx = int(np.argmax(prod))
        i_star = int(i_arr[pair_idx])
        j_star = int(j_arr[pair_idx])
