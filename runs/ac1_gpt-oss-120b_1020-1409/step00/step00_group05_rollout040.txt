import numpy as np, math

def compute_score(a):
    n = a.size
    # compute convolution via FFT
    fft_len = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=42, num_steps=500, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 1024
    # Initial distribution: geometric with r = 1 - 1/n
    r = 1.0 - 1.0 / n
    # raw = r^i; ensure using np.arange
    i = np.arange(n, dtype=np.float64)
    raw = np.power(r, i)
    a = raw / raw.sum()
    # Clip and ensure positivity (already)
    best = a.copy()
    best_score = compute_score(best)
    # local search annealing
    T = 0.01
    move_rate = 0.2
    for step in range(num_steps):
        # choose source index weighted by a (higher should move)
        # Use rng.choice with probabilities proportional to a
        src = rng.choice(n, p=a / a.sum())
        # choose target index uniformly different from src
        tgt = src
        while tgt == src:
            tgt = rng.integers(n)
        delta = a[src] * move_rate * rng.random()
        if delta <= 0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # Ensure no negative due to floating errors
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum  # renormalize to sum 1
        score = compute_score(cand)
        if not np.isfinite(score):
            continue
        if score < best_score or rng.random() < math.exp((best_score - score) / max(T, 1e-12)):
            a = cand
            if score < best_score:
                best_score = score
                best = cand.copy()
        # cooling
        T *= 0.9995
        # optionally reset T if too low?
    # final fallback: ensure nonnegative, sum>0.
    best = np.clip(best, 0.0, 1000.0)
    # If best sum < 0.01 revert to uniform
    if best.sum() < 0.01:
        best = np.ones(n, dtype=np.float64)
        best /= best.sum()
    return best.tolist()

import math
import numpy as np

def _fft_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT (O(n log n))."""
    n = a.size
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact AC1 upper‑bound for a non‑negative sequence a."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _fft_convolution(a)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, num_steps: int = 5000, **kwargs):
    """
    Return a list of non‑negative heights (length equals the provided
    ``height_sequence_1`` if it exists) that aims to minimise the AC1 bound.
    The routine starts from a simple exponential seed and performs a fast
    greedy/annealed local search that moves mass away from indices that
    contribute most to the current maximal convolution entry.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – use the supplied parent if available.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 1024  # a reasonable fallback length

    # ---- 1. Initialise with a geometric (exponential) distribution ----
    # r = 1 - 1/n is the analytically optimal exponential for this family.
    r = 1.0 - 1.0 / n
    raw = np.power(r, np.arange(n, dtype=np.float64))
    a = raw / raw.sum()                     # sum(a) == 1
    a = np.clip(a, 0.0, None)               # ensure exact non‑negativity

    best = a.copy()
    best_score = _score(best)

    # ---- 2. Greedy / simulated‑annealing style improvement ----
    move_rate = 0.20        # maximum fraction of source mass moved per step
    temperature = 0.01      # initial temperature for occasional uphill moves

    idx_array = np.arange(n, dtype=np.int64)

    for step in range(num_steps):
        # Convolution and the index of its current maximum
        conv = _fft_convolution(a)
        k_max = int(conv.argmax())

        # Contributions of each a[i] to the maximal entry conv[k_max].
        partner = k_max - idx_array
        valid = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[valid] = a[valid] * a[partner[valid]]

        # Choose a source (large contribution) and a destination (small contribution)
        src = int(contrib.argmax())
        dst = int(contrib.argmin())
        if src == dst:
            # pick a different destination uniformly at random
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)

        # Amount to move from src to dst
        delta = a[src] * move_rate * rng.random()
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # Numerical safety: clamp and renormalise
        cand = np.clip(cand, 0.0, None)
        total = cand.sum()
        if total <= 0.0:
            continue
        cand /= total

        cand_score = _score(cand)
        if not np.isfinite(cand_score):
            continue

        # Accept if better, otherwise occasionally accept uphill moves
        if cand_score < best_score or rng.random() < math.exp((best_score - cand_score) / max(temperature, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()

        # Temperature cooling (very slow)
        temperature = max(temperature * 0.9995, 1e-8)

    # ---- 3. Final sanity checks and return ----
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:                     # extreme fallback – uniform unit mass
        best = np.ones(n, dtype=np.float64)
        best /= best.sum()
    return best.tolist()
