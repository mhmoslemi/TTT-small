import numpy as np
import math

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Generate a step‑function that (heuristically) lowers the AC1 upper bound.
    The search starts from a set of exponential decays and then performs a
    short, memory‑bounded local refinement.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Base sequence: use the supplied parent if it exists and is valid.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:                     # height_sequence_1 missing or bad
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # 1) Scan a family of exponentially decaying sequences.
    #    a_i = exp(-c * i)  with c ranging from 5e‑5 to 5e‑3.
    # ----------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    decays = np.geomspace(5e-5, 5e-3, num=40)   # geometric spacing over a useful range
    for c in decays:
        cand = np.exp(-c * idx)                # decreasing exponential
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ----------------------------------------------------------------------
    # 2) Small stochastic block perturbations (always keep the best so far).
    # ----------------------------------------------------------------------
    for _ in range(num_steps):
        cand = best.copy()

        # pick a random contiguous block (O(n) work)
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        # multiplicative log‑normal tweak
        factor = float(math.exp(rng.normal(0.0, 0.08)))
        cand[start:stop] *= factor

        if rng.random() < 0.25:

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ----------------------------------------------------------------------
