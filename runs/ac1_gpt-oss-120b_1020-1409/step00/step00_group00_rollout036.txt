import numpy as np

# ----------------------------------------------------------------------
# Helper: construct a sparse index set while bounding the convolution peak.
# ----------------------------------------------------------------------
def build_sparse_set(rng, n, target_m, max_load, max_attempts):
    """
    Greedily selects up to target_m distinct indices in [0, n) such that
    the full convolution of the corresponding 0/1 vector never exceeds max_load.
    Returns the list of chosen indices and the resulting convolution counts.
    """
    # convolution size = 2*n-1
    conv_counts = np.zeros(2 * n - 1, dtype=np.int16)

    # bookkeeping for chosen indices
    chosen = np.empty(target_m, dtype=np.int32)
    chosen_mask = np.zeros(n, dtype=bool)
    m = 0
    attempts = 0

    while attempts < max_attempts and m < target_m:
        idx = rng.integers(0, n)
        if chosen_mask[idx]:
            attempts += 1
            continue

        # sums with already chosen indices (ordered pairs give a contribution of 2)
        if m > 0:
            existing = chosen[:m]
            sums = idx + existing                     # vector of sums
            # will these sums exceed the allowed load?
            if np.any(conv_counts[sums] + 2 > max_load):
                attempts += 1
                continue

        # self‑pair contribution (only one ordered pair when i=j)
        self_sum = idx * 2
        if conv_counts[self_sum] + 1 > max_load:
            attempts += 1
            continue

        # --- accept idx -------------------------------------------------
        if m > 0:
            conv_counts[idx + existing] += 2          # (idx,ex) and (ex,idx)
        conv_counts[self_sum] += 1                    # (idx,idx)
        chosen[m] = idx
        chosen_mask[idx] = True
        m += 1
        attempts = 0        # reset after a successful insertion

    return chosen[:m], conv_counts
