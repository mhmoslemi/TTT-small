import cvxpy as cp

def solve_lp(f, M, idxs):
    n = f.size
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in idxs:
        # compute convolution coefficients for f * g at index k
        # g term appears with coefficient f[i] for i such that 0<=j=k-i<n => j in range
        # This is essentially the inner product of f[ max(0, k-(n-1)) : min(n-1, k) + 1] with g[ k - i ]
        # It can be expressed as sum_i f[i] * g[k - i] where indices valid.
        # Equivalent to dot( f_slice, g_slice_rev )? but for constructing constraint we need vector of length n with given weights.
        # We'll compute weight vector w of length n, zero-initialized, then for i where 0<=i<n and 0<=k-i<n set w[i] = f[i]; but the term is f[i]*g[k-i]; we need w[j] for g_j.
        # Let's re-index: term involves g[j] where j = k-i. So weight on g[j] is f[i] = f[k-j] .
        # So weight vector w[j] = f[k-j] if 0 <= k-j < n and 0 <= j < n.
        # So simply w = np.zeros(n); for j from max(0, k-(n-1)) to min(k, n-1): w[j] = f[k-j]
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # w[lo:hi+1] = f[k - np.arange(lo, hi+1)]
        # We can precompute using slicing.
        w = np.zeros(n)
        idxs_g = np.arange(lo, hi+1)
        w[idxs_g] = f[k - idxs_g]
        constraints.append(w @ g <= M)
    # objective: maximize sum(g)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False, max_iters=2000)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return g.value
    else:
        return None
```

import numpy as np


def make_block_perturbation(base, rng):
    """Return a modest O(n) non‑negative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # optional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _next_pow2(x):
    """Small utility: next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _score(seq, fft_len):
    """Exact objective value using FFT‑based convolution."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.fft.irfft(np.fft.rfft(seq, n=fft_len) * np.fft.rfft(seq, n=fft_len), n=fft_len)
    conv = conv[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)

    The routine blends a simple greedy “linear‑program” phase with
    gradient‑descent fine‑tuning.  All intermediate structures are O(n)

    else:
        a = np.ones(256, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(2 * n - 1)

    cur_score = _score(a, fft_len)
    best = a.copy()
    best_score = cur_score

    # ----- greedy “LP” phase -----------------------------------------
    mix = 0.2                     # how much of the new direction is mixed in
    for _ in range(num_lp_steps):
        # convolution of the current sequence with itself
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
        max_b = conv.max()
        k_max = int(conv.argmax())

        # weight vector w such that (a * g)[k_max] = w·g
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        w = np.zeros(n, dtype=np.float64)
