import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using FFT (complexity O(n log n))
# ----------------------------------------------------------------------
def _full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return linear convolution x * y (length len(x)+len(y)-1)."""
    n = x.size + y.size - 1
    # next power of two for efficient FFT
    n_fft = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=n_fft)
    fy = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(fx * fy, n=n_fft)[:n]
    return conv.real
