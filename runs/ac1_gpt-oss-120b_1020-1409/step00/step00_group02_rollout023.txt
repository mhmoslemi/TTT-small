def make_block_perturbation(base, rng):
    """Return a simple O(n)–memory non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # choose a random block width and location
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative Gaussian perturbation on the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps keep the shape balanced)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def geometric_candidate(n, r):
    """Geometric decreasing sequence, normalised so that its sum equals 1."""
    # avoid underflow/overflow by using log‑space
    exponents = np.arange(n, dtype=np.float64)
    raw = np.exp(exponents * np.log(r))
    total = raw.sum()
    if total == 0.0:
        raw = np.ones(n, dtype=np.float64)
        total = n
    cand = raw / total          # now sum(cand) == 1
    return np.clip(cand, 0.0, 1000.0)

            sum_b = b_opt.sum()
            if sum_b > 0.0:
                # Rescale so that the new vector has the same total mass as ``best``
                scale = best.sum() / sum_b
                # ---- construct the piecewise‑constant candidate g ----
                g = np.zeros_like(best)
                for j in range(m_blocks):
                    start = block_starts[j]
                    end = min(start + block_sz, n)
                    g[start:end] = b_opt[j] * scale
                # ---- convex combination (t ∈ (0,1)) ----
                t = 0.28
                cand = (1.0 - t) * best + t * g
                cand = np.clip(cand, 0.0, 1000.0)
                sc = score_of(cand)
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc

        # ---- a few cheap random block perturbations as a fallback ----
        for _ in range(4):
            cand = make_block_perturbation(best, rng)
            sc = score_of(cand)
            if np.isfinite(sc) and sc < best_score:
                best = cand
                best_score = sc

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
```
