import numpy as np

# The evaluator is provided by the harness.
# It will be called as evaluate_sequence(seq: list) -> float

def _initial_vector():
    """Return a reasonable starting vector.
    Prefer the provided parent `height_sequence_1`; otherwise fall back to a uniform vector."""
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            return np.clip(parent, 0.0, 1000.0)
    # fallback: uniform vector of length 256 (any non‑empty length works)
    return np.ones(256, dtype=np.float64)


def _max_convolution_index(arr):
    """Return the index of the maximal entry of the full self‑convolution."""
    conv = np.convolve(arr, arr)
    return int(np.argmax(conv))


def _perturb_towards_flat(arr, idx_max, rng):
    """
    Produce a neighbour of `arr` that tries to lower the value at `idx_max`.
    It moves a small amount of mass from the pair (i, j) contributing to that
    index to two random locations.
    """
    n = arr.size
    # admissible i range for which j = idx_max - i lies inside the array
    i_lo = max(0, idx_max - (n - 1))
    i_hi = min(n - 1, idx_max)
    # pick a random i in that range
    i = rng.integers(i_lo, i_hi + 1)
    j = idx_max - i
    cand = arr.copy()

    if 0 <= j < n:
        # move at most 30 % of the smaller of the two entries
        limit = min(cand[i], cand[j]) * 0.30
        if limit > 0.0:
            delta = rng.random() * limit          # 0 ≤ delta ≤ limit
            # remove delta from both ends of the pair
            cand[i] -= delta
            cand[j] -= delta
            # redistribute the removed mass to two random sites
            r1 = rng.integers(0, n)
            r2 = rng.integers(0, n)
            cand[r1] += delta
            cand[r2] += delta
    # occasional broader perturbation to keep the search lively
    if rng.random() < 0.04:
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        scale = float(np.exp(rng.normal(0.0, 0.09)))
        cand[start:stop] *= scale
    # enforce the problem bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Stochastic local search that tries to flatten the self‑convolution.
    Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)

    # starting point
    best = _initial_vector()
    best_score = float(evaluate_sequence(best.tolist()))
    # guarantee a finite starting score
    if not np.isfinite(best_score):
        best = np.ones_like(best)
        best_score = float(evaluate_sequence(best.tolist()))

    target = 1.50274349  # stop early if we already beat the target

    for _ in range(num_steps):
        if best_score <= target:
            break

        # locate the currently largest convolution entry
        idx_max = _max_convolution_index(best)

        # generate a neighbour that tries to reduce that peak
        cand = _perturb_towards_flat(best, idx_max, rng)

        # evaluate the neighbour
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score

    # final safety clip (already applied) and conversion to plain list
    return [float(x) for x in best]
