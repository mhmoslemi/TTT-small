import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a simple monotone power‑law sequence.
    A cubic (or quadratic) shape gives a very low AC1 bound:
        a_i = 1000 * ((i+1)/n) ** p   with p ≥ 2.
    The function tests p = 2 and p = 3 and returns the better one.
    """
    # length of the provided construction (or a default if missing)
    try:
        n = int(len(height_sequence_1))
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size != n or not np.isfinite(base).all():
            raise Exception
    except Exception:
        # fallback length if the parent is not available
        n = 256
        base = np.ones(n, dtype=np.float64)

    # indices 1 … n  (float for precise division)
    idx = np.arange(1, n + 1, dtype=np.float64)

    best_vec = None
    best_score = np.inf

    # try a few powers – higher powers give even smaller scores
    for p in (2.0, 3.0, 4.0):
        # scaled so that the maximum entry is exactly 1000 (within allowed range)
        vec = (idx / n) ** p * 1000.0
        # safety clipping (should be unnecessary but harmless)
        np.clip(vec, 0.0, 1000.0, out=vec)

        # evaluate – evaluate_sequence is provided by the harness
        try:
            score = float(evaluate_sequence(vec.tolist()))
        except Exception:          # any unexpected error – treat as invalid
            score = np.inf

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = vec

    # if for some reason all candidates failed, fall back to the parent sequence
    if best_vec is None or not np.isfinite(best_score):
        best_vec = np.clip(base, 0.0, 1000.0)

    # return as a plain list of Python floats (required by the harness)
    return [float(x) for x in best_vec]
