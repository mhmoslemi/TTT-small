    seq = np.clip(seq, 0., 1000.)
    sum_a = np.sum(seq)
    if sum_a < 0.01:

def gradient_move(seq, rng):
    n = seq.size
    conv = np.convolve(seq, seq)
    s_max = int(np.argmax(conv))
    idx = np.arange(n)
    partner = s_max - idx
    mask = (partner >= 0) & (partner < n)
    # compute contributions
    contrib = np.zeros(n, dtype=seq.dtype)
    contrib[mask] = seq[idx[mask]] * seq[partner[mask]]
    i_high = int(np.argmax(contrib))
    # low candidates: mask false or minimal contribution
    zero_mask = ~mask
    if np.any(zero_mask):
        low_candidates = np.where(zero_mask)[0]
    else:
        # choose among low contributions
        low_candidates = np.where(contrib <= np.percentile(contrib, 10))[0]
        if low_candidates.size == 0:
            low_candidates = np.where(contrib == contrib.min())[0]
    i_low = int(rng.choice(low_candidates))
    # transfer a fraction from high to low
    amount = seq[i_high] * rng.uniform(0.01, 0.3)
    amount = min(amount, seq[i_high])
    amount = min(amount, 1000.0 - seq[i_low])
    if amount <= 0:
        # fallback to random small move
        return seq.copy()
    new_seq = seq.copy()
    new_seq[i_high] -= amount
    new_seq[i_low] += amount
    # enforce bounds just in case
    new_seq[i_high] = max(new_seq[i_high], 0.0)
    new_seq[i_low] = min(new_seq[i_low], 1000.0)
    return new_seq

import numpy as np


def compute_score(seq):
    """Exact replica of evaluate_sequence (but without the extra checks)."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = np.sum(seq)
    if total < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    n = seq.size
    return float(2.0 * n * max_b / (total * total))


def block_perturbation(seq, rng):
    """Random multiplicative perturbation of a contiguous block."""
    n = seq.size
    max_width = max(2, int(n * 0.05))
    width = rng.integers(2, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    new_seq = seq.copy()
    new_seq[start:start + width] *= factor
    if rng.random() < 0.1:                     # occasional symmetrisation
        new_seq = 0.5 * (new_seq + new_seq[::-1])
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq


def gradient_move(seq, rng):
    """Move a little mass from a high‑contributing index to a low one."""
    n = seq.size
    conv = np.convolve(seq, seq)
    s_max = int(np.argmax(conv))

    idx = np.arange(n)
    partner = s_max - idx
    mask = (partner >= 0) & (partner < n)

    contrib = np.zeros(n, dtype=seq.dtype)
    contrib[mask] = seq[idx[mask]] * seq[partner[mask]]

    i_high = int(np.argmax(contrib))

    zero_mask = ~mask
    if np.any(zero_mask):
        low_candidates = np.where(zero_mask)[0]
    else:
        low_candidates = np.where(contrib == np.min(contrib))[0]
    i_low = int(rng.choice(low_candidates))
    if i_low == i_high:                         # ensure a genuine transfer
        i_low = (i_high + 1) % n

    amount = seq[i_high] * rng.uniform(0.01, 0.3)
    amount = min(amount, seq[i_high])
    amount = min(amount, 1000.0 - seq[i_low])
    if amount <= 0.0:
        return seq.copy()

    new_seq = seq.copy()
    new_seq[i_high] -= amount
    new_seq[i_low] += amount
    new_seq[i_high] = max(new_seq[i_high], 0.0)
    new_seq[i_low] = min(new_seq[i_low], 1000.0)
    return new_seq
