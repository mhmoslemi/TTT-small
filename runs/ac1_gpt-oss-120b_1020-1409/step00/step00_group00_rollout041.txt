   # Initialize a random vector as basis for search:
   a = rng.random(n)  # uniform [0,1)

   sum_a = np.sum(a)

   fa = np.fft.rfft(a, n=16384)
   b = np.fft.irfft(fa * fa, n=16384)[:2*n-1]
   max_val = b.max()
   cur_score = 2 * n * max_val / (sum_a * sum_a)
   if cur_score < best_score:
       best = a.copy()
       best_score = cur_score

   L = 1 << ((2*n - 1).bit_length())

       max_val = b[max_idx]
       # compute score
       cur_score = 2 * n * max_val / (sum_a * sum_a)
       if cur_score < best_score:
