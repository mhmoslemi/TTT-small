import numpy as np

def propose_candidate(seed=42, **kwargs):
    # Get parent sequence if available, else fallback.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)
    # Validate parent vector.
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    # Compute total sum, enforce minimal sum for evaluator.
    total = float(np.sum(parent))
    if total < 0.01:
        total = 0.01
    # Uniform candidate: same total sum distributed equally.
    uniform_val = total / n
    # Clip to allowed range [0,1000].
    if uniform_val < 0.0:
        uniform_val = 0.0
    elif uniform_val > 1000.0:
        uniform_val = 1000.0
    candidate = np.full(n, uniform_val, dtype=np.float64)
    # Ensure sum after clipping meets minimal requirement.
    if candidate.sum() < 0.01:
        # Adjust first entry to reach 0.01.
        deficit = 0.01 - candidate.sum()
        candidate[0] = candidate[0] + deficit
        if candidate[0] > 1000.0:
            candidate[0] = 1000.0
    # Evaluate both candidates; keep the better.
    try:
        best = candidate
        best_score = float(evaluate_sequence(best.tolist()))
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent
            best_score = parent_score
    except Exception:
        # In case evaluation fails, fall back to uniform.
        best = candidate
    # Return as plain Python list of floats.
    return [float(x) for x in best]
