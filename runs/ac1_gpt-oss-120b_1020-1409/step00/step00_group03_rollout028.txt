def propose_candidate(seed=42, num_trials=2000, density=0.45, target=1.50274349, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)
    else:
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    # If base already good, return it.
    for _ in range(num_trials):
        # random sparse/dense vector based on density
        mask = (rng.random(n) < density).astype(np.float64)
        # Ensure at least one element nonzero
        if mask.sum() == 0:
            mask[rng.integers(n)] = 1.0
        candidate = mask  # values 0 or 1
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate.copy()
            if best_score < target:
                break
    return [float(x) for x in best_seq]
```

import numpy as np

# ------------------------------------------------------------
# Helper: construct a binary sequence where the ordered
# convolution count at any index does not exceed a limit L.
# ------------------------------------------------------------
def build_limited_convolution_sequence(n, L, rng):
    """
    Greedy construction of a 0/1 sequence of length n.
    For any sum index k, the number of ordered pairs (i,j) with
    a_i = a_j = 1 and i + j = k never exceeds L.
    Returns the sequence as a float NumPy array.
    """
    counts = np.zeros(2 * n - 1, dtype=np.int32)   # ordered pair counts per sum
    positions = []                                 # selected indices

    # Random permutation of all indices
    for idx in rng.permutation(n):
        # Check self‑pair contribution
        self_sum = idx + idx
        if counts[self_sum] + 1 > L:
            continue

        # Contributions with already selected positions
        if positions:
            sums = idx + np.array(positions, dtype=np.int32)
            # each unordered pair adds two ordered contributions (i,j) & (j,i)
            if np.any(counts[sums] + 2 > L):
                continue

        # All checks passed – accept idx
        if positions:
            sums = idx + np.array(positions, dtype=np.int32)
            counts[sums] += 2
        counts[self_sum] += 1
        positions.append(idx)

    # Build final sequence (value 1.0 at selected positions)
    seq = np.zeros(n, dtype=np.float64)
    if positions:
        seq[positions] = 1.0
    return seq

    parent construction if needed.
    """
    rng = np.random.default_rng(seed)

    try:
