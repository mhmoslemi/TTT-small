def reduce_max_pairwise(a, beta=0.1):
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    # create mask for indices i where j=k-i in [0,n)
    n = a.shape[0]
    i_vals = np.arange(n)
    j_vals = k - i_vals
    valid = (j_vals >= 0) & (j_vals < n)
    # compute reduction factor for those i where valid and both entries >0
    # Use partner values
    partner = np.zeros_like(a)
    partner[valid] = a[j_vals[valid]]
    # reduction factor (ensuring nonneg)
    factor = 1.0 - beta * (partner / (np.max(partner) + 1e-12))
    factor = np.clip(factor, 0.5, 1.0)  # limit decrease
    a_new = a * factor
    # renormalize sum to original or not needed.
    return a_new

def reduce_peaked_pairwise(a, rng, beta=0.08):
    n = a.size
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    # picking a random i among indices that produce valid j

    j = k - valid_i
