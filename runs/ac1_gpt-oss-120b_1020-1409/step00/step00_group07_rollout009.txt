        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

        factor = float(np.exp(rng.normal(0.0, 0.08)))
        new = arr.copy()
        new[start:start+width] *= factor
        # optional reflection symmetrization
        if rng.random() < 0.4:
            segment = new[start:start+width]
            new[start:start+width] = 0.5 * (segment + segment[::-1])

        factor = float(np.exp(rng.normal(0.0, 0.05)))
        new = arr * factor

def block_mutation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a random factor."""
    n = arr.size
    # Choose a block width between roughly n/200 and n/12
    min_w = max(1, n // 200)
    max_w = max(min_w + 1, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    # Choose a start index so the block fits
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.08)))
    new = arr.copy()
    new[start:start + width] *= factor
    # Occasionally blend the block with its mirror image
    if rng.random() < 0.4:
        block = new[start:start + width]
        new[start:start + width] = 0.5 * (block + block[::-1])
    return np.clip(new, 0.0, 1000.0)


def global_scale_mutation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale the whole vector by a modest random factor."""
    factor = float(np.exp(rng.normal(0.0, 0.05)))

            cand = global_scale_mutation(cand, rng)

        # Exact evaluation (the heavy part, but required for correct scoring)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

        # Periodic diversification: try a fresh mutation of the original parent.
        if (step + 1) % 250 == 0:
            cand2 = block_mutation(base, rng)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand2
                best_score = score2

    # ------------------------------------------------------------------
    # Return the best vector found as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
```
