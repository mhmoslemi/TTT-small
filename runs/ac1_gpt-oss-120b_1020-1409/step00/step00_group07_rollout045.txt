# evaluate_sequence is provided by the harness.

def _initial_length():
    """Return a sensible length for the step-function."""
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        if seq.ndim == 1 and seq.size > 0:

    except Exception:

    # fallback length when no parent is present

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that (approximately)
    minimizes the AC1 upper bound evaluated by `evaluate_sequence`.

    The algorithm starts from the uniform sequence – the theoretical minimizer
    of the bound (value 2) – and performs cheap random perturbations that keep
    the total sum unchanged, accepting only improvements.
    """
    rng = np.random.default_rng(seed)

    n = _initial_length()
    # uniform start gives the provable lower bound 2
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # Small utility to keep the sum fixed after a perturbation
    target_sum = best.sum()

    for _ in range(num_steps):
        cand = best.copy()

        # Choose a random block size and location
        block_len = rng.integers(max(1, n // 200), max(2, n // 12))
        start = rng.integers(0, n - block_len + 1)
        stop = start + block_len

        # Apply a multiplicative perturbation on the block
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor

        # Optional symmetric averaging (mirroring) for extra diversity
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])

        # Enforce bounds and keep the total sum unchanged
        cand = np.clip(cand, 0.0, 1000.0)
        cur_sum = cand.sum()
        if cur_sum > 0.0:
            cand *= (target_sum / cur_sum)

        # Evaluate; keep the candidate only if it improves the score
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # early exit if we ever beat the theoretical optimum (unlikely)
            if best_score < 1.999999:
                break

    # Return a plain Python list as required
    return [float(x) for x in best]
