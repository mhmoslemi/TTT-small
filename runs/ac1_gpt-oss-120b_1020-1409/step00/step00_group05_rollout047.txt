import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative sequence that (provably) minimizes the AC1 upper bound.
    The optimal shape is a uniform vector; we start from the provided parent,
    compare its score, and then perform a short cheap stochastic refinement.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1️⃣ Try to use the parent construction ``height_sequence_1`` if available.
    # ----------------------------------------------------------------------
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            parent = np.clip(arr, 0.0, 1000.0)
    except Exception:
        parent = None

    # ----------------------------------------------------------------------
    # 2️⃣ Build a uniform candidate of the same length (or a default length).
    # ----------------------------------------------------------------------
    n = parent.shape[0] if parent is not None else 256
    uniform = np.full(n, 1.0, dtype=np.float64)          # any constant works

    # Scale uniform to the same total mass as the parent – scaling does NOT
    # affect the objective, but keeps numbers in a familiar range.
    if parent is not None:
        s_parent = float(np.sum(parent))
        s_uniform = float(np.sum(uniform))
        if s_uniform > 0.0:
            uniform = uniform * (s_parent / s_uniform)

    # ----------------------------------------------------------------------
    # 3️⃣ Evaluate parent vs. uniform and keep the better one.
    # ----------------------------------------------------------------------
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(p_score) and p_score < best_score:
            best = parent
            best_score = p_score

    # ----------------------------------------------------------------------
    # 4️⃣ Light‑weight stochastic refinement (optional – rarely beats uniform).
    # ----------------------------------------------------------------------
    num_steps = kwargs.get('num_steps', 200)   # modest budget, linear memory
    for _ in range(num_steps):
        cand = best.copy()
        # Random block width between n/200 and n/12 (inclusive, at least 1)
        low = max(1, n // 200)
        high = max(2, n // 12)
        width = int(rng.integers(low, high + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
