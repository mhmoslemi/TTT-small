        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # ----- exponential decreasing (λ close to 1) -----
    # ε ≈ 1.256 / n gives the optimal trade‑off for a pure geometric law
    eps_opt = 1.256 / n
    lam = 1.0 - eps_opt
    if lam > 0.0:
        raw = np.exp(np.arange(n, dtype=np.float64) * np.log(lam))
        if raw.sum() > 0.0:
            seq = raw / raw.sum()
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score, best_seq = score, seq.copy()

    # ----- exponential increasing (reverse of the above) -----
    if lam > 0.0:
        raw = np.exp(np.arange(n - 1, -1, -1, dtype=np.float64) * np.log(lam))
        seq = raw / raw.sum()
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score, best_seq = score, seq.copy()

    # ----- linear decreasing (weights n, n‑1, …, 1) -----
    raw = np.arange(n, 0, -1, dtype=np.float64)
    seq = raw / raw.sum()
    score = float(evaluate_sequence(seq.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score, best_seq = score, seq.copy()

    # ----- linear increasing (weights 1, 2, …, n) -----
    raw = np.arange(1, n + 1, dtype=np.float64)
    seq = raw / raw.sum()
    score = float(evaluate_sequence(seq.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score, best_seq = score, seq.copy()

    # ----- uniform (fallback) -----
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
    return best_seq, best_score


def propose_candidate(seed: int = 42, num_steps: int = 3000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the AC1 upper bound evaluated by the supplied `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length n from the provided parent sequence, if any.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256                     # safe generic length

    # ------------------------------------------------------------------
    # Start from a few analytically motivated candidates.
    # ------------------------------------------------------------------
    best_seq, best_score = _init_sequence(n)

    # ------------------------------------------------------------------
    # Simple stochastic refinement – always keep an improvement.
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # ---------- single‑coordinate log‑normal perturbation ----------
        cand = best_seq.copy()
        idx = rng.integers(0, n)
        factor = float(np.exp(rng.normal(0.0, 0.08)))   # modest multiplicative change
        cand[idx] = max(cand[idx] * factor, 0.0)

        tot = cand.sum()
        if tot == 0.0:
            continue
        cand = cand / tot

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score, best_seq = sc, cand.copy()

        # ----- occasional block‑wise move (adds diversity) -----
        if step % 50 == 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            block = best_seq.copy()
            block[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            if rng.random() < 0.35:                     # optional mirroring
                block = 0.5 * (block + block[::-1])
            block = np.clip(block, 0.0, 1000.0)
            block = block / block.sum()
            sc = float(evaluate_sequence(block.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score, best_seq = sc, block.copy()

    # ------------------------------------------------------------------
    # The contract requires a non‑empty list of finite floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best_seq]
