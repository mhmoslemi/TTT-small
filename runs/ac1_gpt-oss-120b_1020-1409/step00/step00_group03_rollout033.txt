import numpy as np

def generate_sidon_set(n, rng, target=None):
    """Return a list of indices forming a Sidon set (distinct pairwise sums)."""
    # start with random permutation.
    cand = np.arange(n, dtype=np.int64)
    rng.shuffle(cand)
    selected = []
    sums = set()
    for p in cand:
        # check self sum
        if p + p in sums:
            continue
        conflict = False
        for q in selected:
            if p + q in sums:
                conflict = True
                break
        if conflict:
            continue
        # accept
        selected.append(int(p))
        sums.add(p + p)
        for q in selected[:-1]:
            sums.add(p + q)
        if target is not None and len(selected) >= target:
            break
    return selected

def propose_candidate(seed=42, max_attempts=20, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length n
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256
    best_seq = None
    best_score = np.inf
    V = 1000.0

    # try several random Sidon constructions
    for _ in range(max_attempts):
        sel = generate_sidon_set(n, rng)
        K = len(sel)
        if K < 2:
            continue
        # compute epsilon using optimal ratio V/epsilon = K*n/(2*n - K**2)
        eps = V * (2 * n - K * K) / (K * n)
        if eps <= 0:
            continue
        seq = np.full(n, eps, dtype=np.float64)
        seq[sel] = V
        np.clip(seq, 0.0, 1000.0, out=seq)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
        # early exit if we already beat target
        if best_score < 1.50274349:
            break

    # if no Sidon candidate found (unlikely) fall back to block perturbation
    if best_seq is None:
        # fallback to parent (or uniform)
        if 'height_sequence_1' in globals():
            parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
                parent = np.ones(256, dtype=np.float64)
        else:
            parent = np.ones(256, dtype=np.float64)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        # simple perturbation loop
        for _ in range(500):
            cand = make_block_perturbation(best_seq, rng)
            s = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(s) and s < best_score:
                best_seq = cand
                best_score = s
                if best_score < 1.50274349:
                    break

    return [float(x) for x in best_seq]

def make_block_perturbation(base, rng):
    """Return a nonnegative block perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate
```

import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a Sidon set (all pairwise sums are distinct)
# ----------------------------------------------------------------------
def _generate_sidon_set(n, rng):
    """Return a list of indices with distinct pairwise sums."""
    cand = np.arange(n, dtype=np.int64)
    rng.shuffle(cand)
    selected = []
    sums = set()
    for p in cand:
        # self‑sum must be new
        if p + p in sums:
            continue
        # check sums with previously selected members
        conflict = False
        for q in selected:
            if p + q in sums:
                conflict = True
                break
        if conflict:
            continue
        # accept p
        selected.append(int(p))
        sums.add(p + p)
        for q in selected[:-1]:
            sums.add(p + q)
    return selected

# ----------------------------------------------------------------------
# Optional cheap perturbation (fallback)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a non‑negative block perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, max_attempts=20, **kwargs):
    """
    Build a sparse Sidon‑type sequence:
        large entries (value 1000) at positions with distinct pairwise sums,
        all other positions filled with a small constant chosen to minimize
        the AC1 bound.
    If the construction fails to produce a valid candidate, fall back to a
    simple block‑perturbation search starting from the provided parent.
    """
    rng = np.random.default_rng(seed)

    # length of the problem instance
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).shape[0])
    else:
        n = 256

    V = 1000.0                     # max allowed height for the “large” entries
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Try several random Sidon constructions; keep the best one found.
    # ------------------------------------------------------------------
    for _ in range(max_attempts):
        sel = _generate_sidon_set(n, rng)
        K = len(sel)
        if K < 2:
            continue

        # optimal small value eps = V * (2n - K^2) / (K n)
        eps = V * (2 * n - K * K) / (K * n)
        if eps <= 0.0:
            continue

        seq = np.full(n, eps, dtype=np.float64)
        seq[sel] = V
        np.clip(seq, 0.0, 1000.0, out=seq)

        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
        # early exit if we have already beaten the target bound
        if best_score < 1.50274349:
            break

    # ------------------------------------------------------------------
    # Fallback: use the parent construction and perform cheap local search.
    # ------------------------------------------------------------------
    if best_seq is None:
        if 'height_sequence_1' in globals():
            parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)
            if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():

            parent = np.ones(256, dtype=np.float64)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

        for _ in range(300):

                best_score = sc
                if best_score < 1.50274349:

    # ------------------------------------------------------------------
