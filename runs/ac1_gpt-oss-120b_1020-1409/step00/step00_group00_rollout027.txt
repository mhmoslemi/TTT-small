import numpy as np

def _build_set(n, capacity, rng):
    """Greedy construction of a set S⊂{0,…,n‑1} such that
       the ordered sum‑multiplicity of any k = i+j (i,j∈S) does not exceed *capacity*.
       Returns the list of indices, the maximal ordered count and the size of the set."""
    indices = []
    # ordered pair count for each possible sum (range 0 … 2n‑2)
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

    # random order of candidates – deterministic given the RNG seed
    order = np.arange(n, dtype=np.int32)
    rng.shuffle(order)

    for idx in order:
        # self‑pair (i,i) contributes a single count
        if sum_counts[idx + idx] + 1 > capacity:
            continue
        # check all mixed pairs (idx,j) where j is already selected
        ok = True
        for j in indices:
            s = idx + j
            if sum_counts[s] + 2 > capacity:   # (idx,j) and (j,idx)
                ok = False
                break
        if not ok:
            continue
        # accept idx
        indices.append(idx)
        sum_counts[idx + idx] += 1
        for j in indices[:-1]:                 # previous elements only
            s = idx + j
            sum_counts[s] += 2
    max_b = int(sum_counts.max())
    return indices, max_b, len(indices)
