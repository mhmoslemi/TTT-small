import numpy as np

def propose_candidate(seed=42, num_trials=5, **kwargs):
    """
    Construct a sparse non‑negative sequence whose convolution peak is low.
    Uses a greedy Sidon (B_2) set construction with several random trials
    and returns the best result found.
    """
    # Determine the length of the sequence (use the provided parent if present)
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256  # fallback length if the parent is unavailable

    rng = np.random.default_rng(seed)

    best_positions = []
    best_m = -1
    # Theoretical maximum size of a Sidon set in [0, n-1] satisfies
    #   m(m+1)/2 <= 2*n-1  =>  m <= floor((sqrt(8*n-7)-1)/2)
    max_possible = int(np.floor((np.sqrt(8 * n - 7) - 1) / 2))

    for _ in range(num_trials):
        order = rng.permutation(n)                # random ordering of indices
        positions = []                             # chosen indices
        used_sums = set()                          # all i+j with i<=j in positions

        for x in order:
            # Check whether adding x would create a duplicate sum
            conflict = False
            for p in positions:
                if (p + x) in used_sums:
                    conflict = True
                    break
            if not conflict and (2 * x) in used_sums:
                conflict = True

            if not conflict:
                # Register the new sums produced by x
                for p in positions:
                    used_sums.add(p + x)
                used_sums.add(2 * x)
                positions.append(int(x))

        if len(positions) > best_m:
            best_m = len(positions)
            best_positions = positions
            if best_m >= max_possible:   # cannot do better, stop early
                break

    # Build the candidate sequence: 1.0 on the Sidon positions, 0 elsewhere
    if best_m <= 0:
        cand = np.ones(n, dtype=np.float64)       # safe fallback
    else:
        cand = np.zeros(n, dtype=np.float64)
        cand[best_positions] = 1.0

    # Validate with the provided evaluator; fallback to uniform ones if needed
    try:
        sc = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(sc):
            raise ValueError
    except Exception:
        cand = np.ones(n, dtype=np.float64)

    return [float(v) for v in cand]
