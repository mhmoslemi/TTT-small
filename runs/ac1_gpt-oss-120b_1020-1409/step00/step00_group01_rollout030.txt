def convolve_fft(a,b):
    n = len(a)
    size = 1 << (2*n-1).bit_length() # next power of 2
    # compute FFT
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    fc = fa * fb
    c = np.fft.irfft(fc, size)[:2*n-1]
    return c

def perturb(seq):
    cand = seq.copy()
    # Choose random segment length ~ n/100 to n/10, start random.
    n = len(cand)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative factor per element: exp(normal(0, sigma))
    sigma = 0.1
    factor = np.exp(rng.normal(0.0, sigma, size=width))
    cand[start:stop] *= factor
    # Apply random global shuffle of a subset of indices
    if rng.random() < 0.3:
        # Random swapping of some elements
        idx = rng.choice(n, size=int(0.05*n), replace=False)
        np.random.shuffle(cand[idx])  # not directly; we need to permute values across idx
        # Actually we can permute the whole array optionally
        cand = cand[rng.permutation(n)]  # but that's full permutation, may degrade sum? will preserve same values.
    # Clip to bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def targeted_perturb(seq):
    # compute convolution and find max index
    conv = np.convolve(seq, seq)
    max_idx = np.argmax(conv)
    n = len(seq)
    # Determine indices that contribute
    contributes = np.arange(n)
    partner = max_idx - contributes
    mask = (partner >= 0) & (partner < n)
    # For those indices, we will scale them down by factor <1 with some probability,
    # and compensate by scaling up others not in mask.
    cand = seq.copy()
    # Choose donors from mask (where product large)
    donor_idx = rng.choice(np.where(mask)[0], size= int(0.02*n), replace=False)
    # Choose recipients from complement
    recipient_idx = rng.choice(np.where(~mask)[0], size= donor_idx.size, replace=False)
    # Transfer mass
    for d, r in zip(donor_idx, recipient_idx):
        amount = 0.1 * cand[d]  # reduce 10%
        cand[d] -= amount
        cand[r] += amount
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def rearrange_alternating(arr):
    """Return a new array placing sorted values alternately from ends."""
    sorted_vals = np.sort(arr)[::-1]  # descending
    n = len(arr)
    out = np.empty_like(arr)
    left = 0
    right = n - 1
    toggle = True
    for v in sorted_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out

import numpy as np

def _alternating(seq: np.ndarray) -> np.ndarray:
    """Reorder a 1‑D array by placing the largest, smallest, 2nd‑largest,
    2nd‑smallest, … alternating values."""
    n = seq.shape[0]
    out = np.empty_like(seq)
    half = (n + 1) // 2
    out[::2] = seq[:half]          # largest half, left to right
    out[1::2] = seq[half:][::-1]   # smallest half, right to left
    return out

    integer exponents p.  Both the plain and an alternating reordering are
    evaluated via the supplied ``evaluate_sequence``; the best candidate is
    returned as a list of floats.

        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256                       # safe fallback length
