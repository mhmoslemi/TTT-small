# ---------- helper: full convolution via FFT (O(n log n)) ----------
def full_convolution(arr: np.ndarray) -> np.ndarray:
    """
    Return the linear convolution of ``arr`` with itself using real FFT.
    Length of the result is 2*len(arr)-1.
    """
    n = arr.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << (size - 1).bit_length()
    fft_a = np.fft.rfft(arr, n=m)
    prod = fft_a * fft_a
    conv = np.fft.irfft(prod, n=m)
    return conv[:size]

            best = candidate
            best_score = cand_score

        # accept improvement for the current walk
        if cand_score < current_score:
            current = candidate
            current_score = cand_score

    # -----------------------------------------------------------------
    # fallback safety: if nothing valid was found, return a uniform vector
    # -----------------------------------------------------------------
    if best_score == float('inf') or not np.isfinite(best_score):
        best = np.full(n, 1.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # return as a plain Python list of floats
    return [float(v) for v in best]
