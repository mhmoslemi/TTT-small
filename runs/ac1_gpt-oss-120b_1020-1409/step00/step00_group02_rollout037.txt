def interleave_sorted(arr):
    sorted_desc = np.sort(arr)[::-1]  # descending
    result = np.empty_like(arr)
    left = 0
    right = len(arr) - 1
    toggle = True
    for val in sorted_desc:
        if toggle:
            result[left] = val
            left += 1
        else:
            result[right] = val
            right -= 1
        toggle = not toggle
    return result

import numpy as np


def propose_candidate(seed=42, **kwargs):
    """
    Return a linear ramp sequence (0 → 1000) of the same length as

    """

    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    seq = np.linspace(0.0, 1000.0, num=n, dtype=np.float64)

    seq = np.clip(seq, 0.0, 1000.0)

    else:
