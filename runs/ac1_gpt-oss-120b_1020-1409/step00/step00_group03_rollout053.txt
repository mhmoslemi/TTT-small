def _make_power_sequence(alpha, n):
    """Return 1/sqrt(i+alpha) for i=0..n-1, clipped to allowed range."""
    seq = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + alpha)
    # values are naturally <=1000 for alpha >= 1e-6; clip for safety
    return np.clip(seq, 0.0, 1000.0)

    """Search a non‑negative step sequence with low AC1 bound.

    The search starts from a family a_i = 1/sqrt(i+alpha) and
    performs a cheap hill‑climbing using multiplicative block perturbations.
    The best sequence found (or the parent fallback) is returned as a list.
    """
    rng = np.random.default_rng(seed)

    # Determine the length from the provided parent construction if possible.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:

        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # Initialise best with the parent (already clipped by the evaluator).
    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 1) Scan a few values of the shape parameter alpha.
    # ------------------------------------------------------------------
    # log‑space grid from 1e-5 up to 10 (covers flat to sharply decaying shapes)
    alphas = np.logspace(-5, 1, num=num_alpha, dtype=np.float64)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score

    # 2) Simple hill‑climbing on the currently best vector.
    # ------------------------------------------------------------------
    # Width of a randomly chosen block. Chosen so that it scales roughly
    # between 0.5 % and 5 % of the total length.
    min_width = max(1, n // 200)

    # ------------------------------------------------------------------
    # Block‑update parameters.
    min_width = max(1, n // 200)      # roughly 0.5 % of the length
    max_width = max(2, n // 20)       # up to ≈5 %
    sigma = 0.07                      # log‑normal multiplicative noise

    for _ in range(num_steps):
        cand = best.copy()

        if rng.random() < 0.7:        # multiplicative block perturbation
            width = int(rng.integers(min_width, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor
            if rng.random() < 0.08:  # occasional weak symmetry enforcement
                cand = 0.5 * (cand + cand[::-1])
        else:                         # pairwise mass transfer
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1                # ensure i != j
            delta = best[i] * rng.random() * 0.2
            cand[i] -= delta
            cand[j] += delta

        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

    # ------------------------------------------------------------------
    # 3) Final sanity check and return.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score) or best.size == 0:
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best]
