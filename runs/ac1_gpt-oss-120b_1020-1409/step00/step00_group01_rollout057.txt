import numpy as np

def _score_vector(vec):
    """Return the evaluator score for a numpy vector."""
    # The evaluator expects a plain Python list.
    return float(evaluate_sequence(vec.tolist()))


def propose_candidate(seed=42, max_iters=20, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that aims to
    minimise the AC1 bound used by `evaluate_sequence`.

    The method builds an exponential‑decay vector a_i = (1‑ε)^i,
    searches for the optimal ε using a golden‑section line search,
    and finally refines the result with a few cheap local perturbations.
    """
    # ------------------------------------------------------------------
    # Determine problem size and obtain a fallback parent sequence.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0:
            raise ValueError
        n = parent.size
    except Exception:
        # No provided parent – fall back to a modest uniform vector.
        n = 256
        parent = np.ones(n, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    best_vec = parent.copy()
    best_score = _score_vector(best_vec)

    # ------------------------------------------------------------------
    # Helper: produce the exponential candidate for a given epsilon.
    # ------------------------------------------------------------------
    def _make_exp(eps):
        """Return (1‑eps)^i for i=0..n‑1 as a float64 array."""
        # log1p(-eps) is accurate for small eps.
        log_r = np.log1p(-eps)          # = log(1‑eps) < 0
        idx = np.arange(n, dtype=np.float64)
        vec = np.exp(idx * log_r)       # (1‑eps)^i
        return np.clip(vec, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Golden‑section search for ε ∈ (0, 0.01) that minimises the score.
    # ------------------------------------------------------------------
    phi = (1.0 + 5.0 ** 0.5) / 2.0          # golden ratio
    lo = max(1e-6, 0.5 / n)               # lower bound, ≈ 1/(2n)
    hi = 0.01                              # generous upper bound

    # Initial interior points
    c = hi - (hi - lo) / phi
    d = lo + (hi - lo) / phi
    fc = _score_vector(_make_exp(c))
    fd = _score_vector(_make_exp(d))

    for _ in range(max_iters):
        if fc < fd:
            hi, d, fd = d, c, fc
            c = hi - (hi - lo) / phi
            fc = _score_vector(_make_exp(c))
        else:
            lo, c, fc = c, d, fd
            d = lo + (hi - lo) / phi
            fd = _score_vector(_make_exp(d))

    # Test the three candidates from the final interval.
    eps_candidates = [ (lo + hi) / 2.0, c, d ]
    for eps in eps_candidates:
        vec = _make_exp(eps)
        sc = _score_vector(vec)
        if np.isfinite(sc) and sc < best_score:
            best_vec = vec
            best_score = sc

    # ------------------------------------------------------------------
    # Light local refinement: random multiplicative block perturbations.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    for _ in range(200):
        cand = best_vec.copy()
        # Choose a random block width (between ~n/200 and ~n/12)
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        # Apply a modest random scaling factor.
        factor = float(np.exp(rng.normal(0.0, 0.05)))
        cand[start:stop] *= factor
        # Occasionally symmetrise to keep the shape balanced.
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score_vector(cand)
        if np.isfinite(sc) and sc < best_score:
            best_vec = cand
            best_score = sc

    # ------------------------------------------------------------------
    # Return a clean Python list of floats.
    # ------------------------------------------------------------------
    return [float(v) for v in best_vec]
