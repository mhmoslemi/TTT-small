def propose_candidate(seed=42, **kwargs):
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    # We'll use deterministic r values.
    best = None
    best_score = float('inf')
    # Ensure candidate is float python list.
    # Precompute indices: idx = np.arange(n)
    idx = np.arange(n, dtype=np.float64)
    # Search for ratio r values.
    for r in np.geomspace(0.5, 0.9999, num=200):
        a = np.exp(idx * np.log(r))   # r**idx
        # Clip to [0,1000] optional.
        # Evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
    # Possibly add random perturbation around best to maybe improve further
    # Do few local updates via random transfer.
    rng = np.random.default_rng(seed)
    for _ in range(200):
        # propose perturbation: move some mass from one index to another scaled appropriately.
        cand = best.copy()
        # select two distinct indices i,j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # transfer amount delta = fraction of a[i] (if >0)
        if cand[i] > 0:
            delta = cand[i] * rng.uniform(0.0, 0.1)
            cand[i] -= delta
            cand[j] += delta
        # clip
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand
    # fallback if best still None: use uniform vector of 1's
    if best is None:
        best = np.ones(n, dtype=np.float64)
    return [float(x) for x in best]

def simplex_projection(v):
    # Returns projection onto simplex {x >= 0, sum=1}
    # v is np.ndarray
    u = np.sort(v)[::-1]  # descending
    css = np.cumsum(u) - 1
    rho = np.where(u - css / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        # all negative
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

def _conv_max(a: np.ndarray, fft_len: int):
    """Return (max coefficient, index of max, full convolution) for a*a."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * a.size - 1]
    max_c = conv.max()
    k_star = int(conv.argmax())
    return max_c, k_star, conv


def _gradient(a: np.ndarray, k_star: int):
    """Sub‑gradient of the active convolution coefficient with respect to a."""
    n = a.size
    grad = np.zeros_like(a)
    idx = np.arange(n)
    j = k_star - idx
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k_star % 2 == 0:
        i0 = k_star // 2
        if i0 < n:
            grad[i0] += a[i0]          # double contribution on the diagonal
    return grad


def propose_candidate(seed=42, num_iters=1500, **kwargs):
    """Generate a step‑function height list with a small AC1 upper bound.

    The routine performs a projected gradient descent that directly reduces the
    maximal convolution coefficient while keeping the vector non‑negative and
    normalised (sum = 1).  The final candidate is clipped to the required bound
    [0, 1000] and verified with ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    #  Initialise from the supplied parent if it exists.
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    if base.ndim != 1 or n == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)
        n = base.size

    parent = np.clip(base, 0.0, 1000.0)

    # best so far (fallback)
    try:
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float('inf')

    # ----------------------------------------------------------------------
    #  Random start (break symmetry) – normalise to unit sum.
    # ----------------------------------------------------------------------
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    if s > 0.0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # FFT length (next power of two for speed)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------------------
    #  Main optimisation loop.
    # ----------------------------------------------------------------------
    for it in range(num_iters):
        max_c, k_star, _ = _conv_max(a, fft_len)
