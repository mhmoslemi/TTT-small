def propose_candidate(seed=42, num_steps=300, block_min_frac=0.001, block_max_frac=0.05, sigma=0.1, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        base = height_sequence_1
    except NameError:
        base = None
    if isinstance(base, np.ndarray) and base.ndim == 1 and base.size > 0:
        n = base.shape[0]
    else:
        n = 256

    # initial candidate uniform value 1.0 (or maybe average of base)
    uniform_val = 1.0
    best = np.full(n, uniform_val, dtype=np.float64)
    best = np.clip(best, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))

    # If base exists and its score better than uniform, start from base.
    if base is not None:
        # Clip base as per earlier code
        parent = np.clip(np.asarray(base, dtype=np.float64), 0.0, 1000.0)
        # Ensure vertical dimension 1-D
        if parent.ndim == 1 and parent.size > 0:
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best = parent.copy()
                best_score = parent_score

    for _ in range(num_steps):
        # pick a random block length
        max_len = max(2, n // 20)
        block_len = int(rng.integers(1, max_len+1))
        start = int(rng.integers(0, max(1, n - block_len + 1)))
        candidate = best.copy()
        # multiplicative noise
        noise = np.exp(rng.normal(0.0, sigma, size=block_len))
        candidate[start:start+block_len] *= noise
        # optionally mirror
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    return [float(x) for x in best]
