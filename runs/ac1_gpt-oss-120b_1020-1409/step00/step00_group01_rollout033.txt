import numpy as np

# evaluate_sequence is provided by the harness.

def _alt_high_low(arr):
    """Interleave large and small values to break large‑large overlaps."""
    vals = np.sort(arr)                     # ascending
    n = vals.size
    half = (n + 1) // 2                     # size of the smaller half
    low = vals[:half]                       # smallest half (ascending)
    high = vals[half:][::-1]                # largest half (descending)

    out = np.empty_like(arr)
    even_idx = np.arange(0, n, 2)
    odd_idx = np.arange(1, n, 2)

    out[even_idx] = high[:even_idx.size]
    out[odd_idx] = low[:odd_idx.size]
    return out

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """Search for a non‑negative step function with low AC1 upper bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Base sequence (fallback to uniform if parent is missing/invalid)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    if base.sum() < 0.01:                     # guard against degenerate parent
        base = np.ones_like(base)

    n = base.size

    # ------------------------------------------------------------------
    # 2. Initial candidates
    # ------------------------------------------------------------------
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # a) deterministic interleaving of high / low values
    cand = _alt_high_low(base)
    sc = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best, best_score = cand, sc

    # ------------------------------------------------------------------
    # 3. Stochastic local search
    # ------------------------------------------------------------------
    current = best.copy()
    for _ in range(num_steps):
        # copy current solution
        candidate = current.copy()

        r = rng.random()
        if r < 0.4:                                 # whole array shuffle
            rng.shuffle(candidate)
        elif r < 0.7:                               # localized multiplicative tweak
            width = int(rng.integers(1, max(2, n // 100)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.08)))
            candidate[start:start + width] *= factor
            candidate = np.clip(candidate, 0.0, 1000.0)
        else:                                       # swap two entries
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                candidate[i], candidate[j] = candidate[j], candidate[i]

        # occasional global scaling (does not affect the ratio but keeps values in range)
        if rng.random() < 0.05:
            scale = float(np.exp(rng.normal(0.0, 0.03)))
            candidate *= scale
            candidate = np.clip(candidate, 0.0, 1000.0)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = candidate, score
            current = candidate
        elif np.isfinite(score) and score < float(evaluate_sequence(current.tolist())):
            # accept a modest improvement as new current position
            current = candidate

    # ------------------------------------------------------------------
    # 4. Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.full(n, 1.0, dtype=np.float64)

    return [float(x) for x in best]
