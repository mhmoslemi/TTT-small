import numpy as np

# ----------------------------------------------------------------------
# Helper: try to build a sparse 0/1 sequence with limited convolution peaks.
# ----------------------------------------------------------------------
def build_sparse_sequence(n, k_target, b_target, rng, max_restarts=20):
    """
    Construct a length‑n array with exactly k_target ones such that
    every entry of the self‑convolution (ordered pairs) is at most b_target.
    Returns the array or None if construction fails.
    """
    for _ in range(max_restarts):
        selected = []                     # list of chosen indices
        # convolution buckets: index t = i + j (0 ≤ t ≤ 2n‑2)
        count = np.zeros(2 * n - 1, dtype=np.int16)

        # random order of all positions
        cand = rng.permutation(n)
        for i in cand:
            # self‑pair contribution (i,i) adds 1 to bucket 2*i
            if count[2 * i] + 1 > b_target:
                continue

            if selected:
                sel_arr = np.array(selected, dtype=np.int32)
                sums = i + sel_arr               # buckets for (i,j) and (j,i)
                # each unordered pair contributes 2 to its bucket
                if (count[sums] + 2 > b_target).any():
                    continue

            # accept i
            if selected:
                count[i + sel_arr] += 2
            count[2 * i] += 1
            selected.append(i)

            if len(selected) >= k_target:
                # success – build the sequence
                seq = np.zeros(n, dtype=np.float64)
                seq[selected] = 1.0
                # sanity check (should always hold)
                if count.max() <= b_target:
                    return seq
                # otherwise fall through to restart
    return None


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) gives a low
    AC1 upper bound.  The routine tries a sparse construction; if it fails
    it falls back to a constant vector.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    try:
        n = int(height_sequence_1.shape[0])   # supplied parent length
    except Exception:
        n = 256                               # safe fallback size

    target_ratio = 1.50274349

    # Try a few (k, b) parameter pairs, preferring larger k (more mass)
    # and small b (few collisions).  The theoretical ratio bound is
    #   2 * n * b / k^2 .
    param_list = [
        (300, 10),
        (250, 10),
        (200, 12),
        (150, 12),
        (120, 14),
    ]

    for k_target, b_target in param_list:
        # quick feasibility test
        if 2 * n * b_target / (k_target * k_target) >= target_ratio * 1.1:
            continue
        seq = build_sparse_sequence(n, k_target, b_target, rng)
        if seq is None:
            continue
        # Evaluate with the exact evaluator (already imported by the harness)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < target_ratio:
            return seq.tolist()

    # ------------------------------------------------------------------
    # Fallback: simple constant sequence (always valid, ratio = 2).
    # ------------------------------------------------------------------
    fallback = np.ones(n, dtype=np.float64)
    return fallback.tolist()
