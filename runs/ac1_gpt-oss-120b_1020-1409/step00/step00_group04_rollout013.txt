import numpy as np

def _is_prime(num: int) -> bool:
    """Simple deterministic primality test for small integers."""
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    limit = int(num ** 0.5) + 1
    for d in range(3, limit, 2):
        if num % d == 0:
            return False
    return True


def _sidon_indices(p: int) -> np.ndarray:
    """
    Construct a Sidon set of size p using the Bose‑Chowla construction:
    A = { i * p + (i*i % p) : i = 0 .. p-1 }.
    All elements lie in [0, p**2) and any unordered sum occurs at most once.
    """
    i = np.arange(p, dtype=np.int64)
    return i * p + (i * i) % p
