import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Produce a uniform, non‑negative step‑function of the same length as the
    supplied ``height_sequence_1`` (or a short fallback length).  For any
    non‑negative sequence the AC1 upper bound satisfies
        2 * n * max(conv(a,a)) / (sum(a)^2) >= 2,
    with equality when all entries are equal.  Hence the uniform vector is
    optimal and yields a score of exactly 2, improving on any non‑uniform
    parent that the baseline search might return.
    """
    # Length to use – preferably the length of the provided parent.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        # Fallback length if the runtime array is missing.
        n = 256

    # Uniform positive values; any constant works as long as it stays in
    # the allowed range [0,1000].  Using 1.0 keeps the sum comfortably above
    # the 0.01 lower bound.
    seq = np.full(n, 1.0, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)

    # The evaluator expects a plain Python list.
    return seq.tolist()
