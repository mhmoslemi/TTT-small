def propose_candidate(seed=42, **kwargs):
    import numpy as np

    # Get parent if exists
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # deterministic shape: geometric sequence with r ~ 1 - 2/n
    epsilon = 2.0 / n
    r = 1.0 - epsilon
    candidate = np.power(r, np.arange(n, dtype=np.float64))
    # ensure within bounds
    candidate = np.clip(candidate, 0.0, 1000.0)

    # Evaluate
    try:
        score = float(evaluate_sequence(candidate.tolist()))
    except Exception:
        score = float('inf')

    # fallback uniform
    fallback = np.full(n, 1.0, dtype=np.float64)
    try:
        fallback_score = float(evaluate_sequence(fallback.tolist()))
    except Exception:
        fallback_score = float('inf')

    if fallback_score < score and np.isfinite(fallback_score):
        best = fallback
    else:
        best = candidate

    return [float(x) for x in best]

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function sequence (list of floats) that
    strives to minimise the AC1 upper bound evaluated by
    ``evaluate_sequence``.
    The routine proceeds in three cheap phases:
      1. Scan a family of exponential decays.
      2. Perform a modest block‑perturbation local search.
      3. Apply a few gradient‑descent steps that directly reduce the
         offending convolution peak.
    A fallback to a uniform sequence is kept for robustness.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Base length – use the provided parent if it exists and is sane.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # Start from a simple uniform candidate (scale does not matter).
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Phase 1 – scan exponential shapes (a_i = exp(-eps*i)).
    # ------------------------------------------------------------------
    # Parameter x = eps * n, we explore a range that contains the
    # theoretical optimum (x ≈ 1.2 gives the smallest AC1 for this family).
    for x in np.linspace(0.2, 5.0, 50):
        eps = x / n
        cand = np.exp(-eps * np.arange(n, dtype=np.float64))
        cand = np.clip(cand, 0.0, 1000.0)

        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best, best_score = cand, sc

        # also test the flipped version (mirrored shape)
        cand_rev = cand[::-1]
        sc = float(evaluate_sequence(cand_rev.tolist()))
        if sc < best_score:
            best, best_score = cand_rev, sc

    # ------------------------------------------------------------------
    # Phase 2 – cheap block perturbations around the current best.
    # ------------------------------------------------------------------
    width_min = max(1, n // 200)
    width_max = max(2, n // 12)
    for _ in range(200):
        cand = best.copy()
        w = int(rng.integers(width_min, width_max))
        s = int(rng.integers(0, max(1, n - w + 1)))
        e = s + w
        # multiplicative log‑normal perturbation
        cand[s:e] *= float(np.exp(rng.normal(0.0, 0.15)))
        # occasional symmetrisation
        if rng.random() < 0.4:
            cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc

    # ------------------------------------------------------------------
    # Phase 3 – a few gradient‑type steps that directly attack the
    #            largest convolution entry.
    # ------------------------------------------------------------------
    eta = 0.01
    for _ in range(30):
        conv = np.convolve(best, best)
        kmax = int(np.argmax(conv))

        # indices that actually contribute to conv[kmax]
        start = max(0, kmax - (n - 1))
        end = min(n, kmax + 1)
        idx = np.arange(start, end, dtype=np.int64)

        # gradient of conv[kmax] w.r.t. each a_i involved:
        # d/da_i (sum_j a_j * a_{kmax-j}) = 2 * a_{kmax-i}
        grad = np.zeros_like(best)
        grad[idx] = 2.0 * best[kmax - idx]

        # simple multiplicative‑additive descent; keep non‑negative.
        cand = best - eta * grad
        cand = np.clip(cand, 0.0, 1000.0)

        sc = float(evaluate_sequence(cand.tolist()))

        if np.isfinite(sc) and sc < best_score:

            if eta < 1e-6:

    # ------------------------------------------------------------------
    if not np.isfinite(best_score):

        best = np.clip(best, 0.0, 1000.0)

    # Return a plain Python list as required by the harness.
    return [float(v) for v in best]
