def perturb(candidate, rng):
    n = len(candidate)
    width = max(1, int(rng.integers(n // 200, n // 20)))  # random width
    start = rng.integers(0, n - width + 1)
    end = start + width

    new = candidate.copy()
    new[start:end] *= factor
    if rng.random() < 0.1:
        new[start:end] = (new[start:end] + new[start:end][::-1]) * 0.5

    Produce a cheap O(n) perturbation of a non‑negative vector.

    A random contiguous block is scaled by a log‑normal factor;
    optionally the block is mirrored.
    """
    n = x.size
    # Choose a block width between n/200 and n/20, at least 1.
    lo = max(1, n // 200)
    hi = max(lo + 1, n // 20)
    width = int(rng.integers(lo, hi))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width

    y = x.copy()
    factor = float(np.exp(rng.normal(0.0, 0.02)))  # mild log‑normal scaling
    y[start:stop] *= factor

    if rng.random() < 0.1:                     # occasional symmetrisation
        block = y[start:stop]
        y[start:stop] = 0.5 * (block + block[::-1])

    np.clip(y, 0.0, 1000.0, out=y)
    return y

def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the AC1 upper bound used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------
    # 1️⃣  Obtain a usable starting vector.
    # --------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(height_sequence_1, dtype=np.float64)
        if not _is_valid_array(cand):
            cand = None
    else:
        cand = None

    if cand is None:
        # fallback: a modestly sized constant vector (always feasible)
        cand = np.full(256, 1.0, dtype=np.float64)

    # --------------------------------------------------------------------
    # 2️⃣  Compute the baseline score of the candidate we just obtained.
    # --------------------------------------------------------------------
    try:
        best = cand
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:
        best_score = float("inf")
        best = cand

    # --------------------------------------------------------------------
    # 3️⃣  Try a simple constant vector of the same length – it attains the
    #      theoretical lower bound of 2 for non‑negative inputs.
    # --------------------------------------------------------------------
    const_vec = np.full_like(best, 1.0)   # any positive constant works
    const_score = float(evaluate_sequence(const_vec.tolist()))
    if const_score < best_score:
        best, best_score = const_vec, const_score

    # --------------------------------------------------------------------
    # 4️⃣  Perform a light local‑search around the current best.
    # --------------------------------------------------------------------
    # Limit the number of expensive evaluations to keep runtime modest.
    max_steps = 30
    for _ in range(max_steps):
        trial = _block_perturb(best, rng)
        try:
            score = float(evaluate_sequence(trial.tolist()))
        except Exception:
            continue
        if score < best_score:
            best, best_score = trial, score

    # --------------------------------------------------------------------
    # 5️⃣  Return the best sequence found as a plain Python list.
    # --------------------------------------------------------------------
    return [float(v) for v in best]
