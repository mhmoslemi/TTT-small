    fft_x = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fft_x * fft_x, n=size)
    return conv[:m]  # real values

import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _is_prime(num: int) -> bool:
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num ** 0.5)
    for d in range(3, r + 1, 2):
        if num % d == 0:
            return False
    return True


def _largest_prime_leq(limit: int) -> int:
    for cand in range(limit, 1, -1):
        if _is_prime(cand):
            return cand
    return 2


def _build_sidonic_candidate(n: int, p: int, low_val: float) -> np.ndarray:
    """
    Construct a step‑function of length ``n``.
    * ``p`` indices (a Bose–Chowla Sidon set) receive the maximal height 1000.
    * All remaining entries receive ``low_val`` (a small positive number).
    If a pair of Sidon indices would sum to ``n‑1`` (the full‑overlap shift)
    we turn one of them into a low entry to avoid a large high–high term.
    """
    high = 1000.0
    a = np.full(n, low_val, dtype=np.float64)

    # Bose–Chowla Sidon set:  i * p + (i*i mod p) ,  i = 0..p‑1
    idxs = []
    for i in range(p):
        idx = i * p + (i * i) % p
        if idx < n:
            idxs.append(idx)
    idx_set = set(idxs)

    # remove one element of any pair that would collide at shift n‑1
    to_remove = set()
    for i in idx_set:
        j = n - 1 - i
        if j in idx_set:
            # keep the smaller index (arbitrary choice) and drop the other
            if i > j:
                to_remove.add(i)
            else:
                to_remove.add(j)
    idx_set.difference_update(to_remove)

    for i in idx_set:
        a[i] = high
    return a

# Main entry point expected by the harness
# --------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Build a compact candidate using a Sidon set of large values (1000)
    and a small background level.  Several background levels are tried
    and the best (lowest evaluated bound) is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length of the vector.  Use the provided parent if present.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            n = 256
        else:
            n = parent.size
    else:
        n = 256

    # ------------------------------------------------------------------
    # Choose a Sidon set size.  The Bose–Chowla construction works for any
    # prime p with p² ≤ n.  We take the largest such prime (≈ sqrt(n)).
    # ------------------------------------------------------------------
    sqrt_n = int(np.sqrt(n))
    p = _largest_prime_leq(sqrt_n)
    # safety: ensure p² ≤ n
    while p * p > n:
        p = _largest_prime_leq(p - 1)

    # ------------------------------------------------------------------
    # Try a few low background levels and keep the best.
    # ------------------------------------------------------------------
    low_candidates = [5, 8, 10, 12, 15, 18, 20, 25]
    best_score = float("inf")
    best_vec = None

    for low in low_candidates:
        cand = _build_sidonic_candidate(n, p, float(low))
        # clip to the evaluator's allowed range (should already be fine)
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            score = float("inf")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # If for any reason we did not obtain a valid candidate, fall back
    # to a uniform vector (which yields a bound of exactly 2).
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 500.0, dtype=np.float64)  # 500 → sum = 0.5*1000*n

    # ------------------------------------------------------------------
    # Return the best candidate as a plain Python list of floats.
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
