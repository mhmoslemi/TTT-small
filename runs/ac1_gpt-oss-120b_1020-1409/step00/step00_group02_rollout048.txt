import numpy as np

# ---------------------------------------------------------------
# Helper utilities
# ---------------------------------------------------------------
def _list_score(arr):
    """Exact score from the provided evaluator."""
    # `evaluate_sequence` is supplied by the harness.
    return float(evaluate_sequence(arr.tolist()))


def _smooth(arr, radius):
    """Replace each entry with the average over a window of size 2*radius+1."""
    if radius <= 0:
        return arr
    # Uniform kernel for sliding average.
    kernel = np.ones(2 * radius + 1, dtype=np.float64) / (2 * radius + 1)
    # ``mode='same'`` keeps the original length.
    return np.convolve(arr, kernel, mode='same')


def _block_perturb(arr, rng):
    """Create an O(n)‑memory multiplicative perturbation of a sub‑block."""
    cand = arr.copy()
    n = cand.size
    # Choose a block width between n/200 and n/12 (inclusive).
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Multiplicative log‑normal factor (small amplitude).
    factor = float(np.exp(rng.normal(0.0, 0.08)))
    cand[start:stop] *= factor
    # Optional mild symmetrisation.
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑empty list of non‑negative heights that (approximately)
    minimizes the AC1 upper bound evaluated by ``evaluate_sequence``.
    The algorithm starts from the provided ``height_sequence_1`` (if
    available), applies cheap smoothing / block perturbations and keeps
    the best candidate seen.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Initialise from the external parent construction if possible.
    # -----------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                     # height_sequence_1 missing or bad
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    # Clamp to the allowed range [0,1000].
    parent = np.clip(parent, 0.0, 1000.0)

    # Ensure a minimally acceptable L1 norm (required by the evaluator).
    if parent.sum() < 0.01:
        parent[:] = 0.01 / parent.size

    best = parent.copy()
    best_score = _list_score(best)

    # -----------------------------------------------------------------
    # Simple refinement loop (linear memory, cheap operations).
    # -----------------------------------------------------------------
    for _ in range(num_steps):
        # Randomly pick a perturbation type.
        if rng.random() < 0.7:
            # Smoothing: random radius 1–4.
            radius = int(rng.integers(1, 5))
            cand = _smooth(best, radius)
        else:
            # Block‑wise multiplicative perturbation.
            cand = _block_perturb(best, rng)

        cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate and keep the improvement.
        score = _list_score(cand)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # The theoretical lower bound (in exact arithmetic) is 2.
            # Stop early if we are already at that limit (within tiny epsilon).
            if best_score <= 2.000001:
                break

    # Convert to plain Python floats for the harness.
    return [float(v) for v in best]
