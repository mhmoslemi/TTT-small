    # 1. Obtain a base sequence if the harness supplied one.
    # ------------------------------------------------------------------
    try:

    except Exception:

    # ------------------------------------------------------------------
    # 2. Determine the length of the problem.
    # ------------------------------------------------------------------
    if base is not None:
        n = base.size
    else:

        n = 5582

    #    Uniform sequence with sum sqrt(2n) yields the exact bound 2.0.
    # ------------------------------------------------------------------

        if s > 0:

        else:
            scaled = best.copy()

        uniform_score = float(evaluate_sequence(best.tolist()))
        scaled_score = float(evaluate_sequence(scaled.tolist()))
        if scaled_score < uniform_score:
            best = scaled
            best_score = scaled_score
        else:
            best_score = uniform_score
    else:
        best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 4. Stochastic local search – cheap block perturbations.
    #    Each iteration creates O(n) auxiliary data and calls the exact

    # ------------------------------------------------------------------
