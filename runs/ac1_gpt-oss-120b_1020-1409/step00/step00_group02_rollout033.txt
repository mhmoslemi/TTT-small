    rng = np.random.default_rng(seed)
    n = height_sequence_1.shape[0] if defined else 256

    # init: uniform may be decent baseline.
    parent = np.ones(n, dtype=np.float64)

    try:
        arr0 = np.asarray(height_sequence_1, dtype=np.float64)
        if arr0.ndim == 1 and arr0.size == n and np.isfinite(arr0).all():
            parent = arr0.copy()
    except NameError:
        pass

    # clip
    parent = np.clip(parent, 0.0, 1000.0)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # track for early exit
    for step in range(num_steps):
        # randomly select i,j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # compute delta as fraction of a_i
        ai = best[i]
        if ai <= 1e-12:
            continue
        delta = ai * rng.uniform(0.01, 0.05)  # small fraction
        # propose new
        cand = best.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] = min(1000.0, cand[j] + delta)
        # ensure sum doesn't change much: if clipping affected, maybe adjust
        # compute score
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

        # optional occasional large perturbation using block scaling
        if step % 100 == 0:
            # apply block perturbation similar to previous but better acceptance
            width = int(max(1, n // rng.integers(50, 200))) # random small width
            start = rng.integers(0, n - width + 1)
            stop = start + width
            factor = np.exp(rng.normal(0.0, 0.07))
            cand2 = best.copy()
            cand2[start:stop] *= factor
            cand2 = np.clip(cand2, 0.0, 1000.0)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand2
                best_score = score2

    return [float(x) for x in best]

    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):

import numpy as np


def _generate_binomial_sequence(length: int) -> np.ndarray:
    """
    Generate the sequence a_i = C(2i,i) / 4^i for i=0..length-1.
    This satisfies (a * a)[k] = 1 for all k < length (up to rounding).
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    # recurrence: a_i = a_{i-1} * ((2i-1)*(2i)) / (4*i*i)
    for i in range(1, length):
        i_f = float(i)
        a[i] = a[i - 1] * ((2.0 * i_f - 1.0) * (2.0 * i_f)) / (4.0 * i_f * i_f)
    return a


def _pair_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a small random amount of mass from one index to another
    while keeping the vector non‑negative and respecting the 0‑1000 clip.
    """
    n = arr.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j or arr[i] <= 0.0:
        return arr.copy()
    # shift a fraction of the donor value
    delta = arr[i] * rng.uniform(0.001, 0.05)
    cand = arr.copy()
    cand[i] = max(0.0, cand[i] - delta)
    cand[j] = min(1000.0, cand[j] + delta)
    return cand


def _block_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Scale a random contiguous block by a log‑normal factor.
    """
    n = arr.size
    # block width between roughly n/200 and n/20
    width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = arr.copy()
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
