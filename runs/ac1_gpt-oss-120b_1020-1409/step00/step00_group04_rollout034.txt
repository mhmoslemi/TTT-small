import numpy as np

def convolve_self(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]
    return conv

def compute_score(a):
    n = a.shape[0]
    b = convolve_self(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

import numpy as np

def _convolve_self(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a FFT (O(n log n))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact AC1 upper bound for a (scale‑invariant)."""
    n = a.shape[0]
    b = _convolve_self(a)
    max_b = float(b.max())
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """Search for a non‑negative step sequence with a low AC1 bound.

    The routine keeps O(n) memory, uses FFT‑based convolutions, and
    performs guided pairwise mass transfers that tend to lower the
    maximum convolution entry.
    """
    rng = np.random.default_rng(seed)

    # ---- initialise ---------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        a = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # fallback: a small uniform vector
        a = np.ones(256, dtype=np.float64)

    # work with a scale‑invariant representation (sum == 1)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a)
        total = a.sum()
    a = a / total

    n = a.shape[0]
    best = a.copy()
    best_score = float(_score(best))

    # ---- main optimisation loop --------------------------------------------
    for step in range(num_steps):
        # current convolution and the most offending index
        conv = _convolve_self(a)
        k_max = int(np.argmax(conv))

        # all i that actually contribute to this k_max
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        idx = np.arange(i_min, i_max + 1, dtype=np.int64)
        partners = k_max - idx
        contrib = a[idx] * a[partners]
        total_contrib = contrib.sum()
        if total_contrib == 0.0:
            # degenerate – fall back to a tiny random block perturbation
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, max(1, n - width + 1))
            stop = start + width
            cand = a.copy()
            cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand / cand.sum()
            sc = float(_score(cand))
            if sc < best_score:
                best, best_score, a = cand, sc, cand
            continue

        # pick a contributing index, weighted by its contribution size
        probs = contrib / total_contrib
        i = rng.choice(idx, p=probs)
        j = k_max - i                     # partner of i for this peak

        # pick a destination index that currently carries little mass
        inv = 1.0 / (a + 1e-12)
        t = rng.choice(n, p=inv / inv.sum())
        if t == i:

        max_transfer = a[i] * 0.5
        if max_transfer <= 0.0:

        delta = rng.uniform(0.0, max_transfer)

        cand = a.copy()
        cand[i] -= delta

        if np.any(cand < 0.0):

        cand = cand / cand.sum()          # keep scale‑invariant

        sc = float(_score(cand))

        else:

            T = max(0.01, 1.0 - step / num_steps)    # linear cooling
            if rng.random() < np.exp((best_score - sc) / T):
                a = cand

    # -------------------------------------------------------------------------
    # Return the best vector found (as a plain Python list)
    # -------------------------------------------------------------------------
    return [float(v) for v in best]
