def convolve_full(a):
    n = a.size
    size = 2*n - 1
    # find next power of two for efficient FFT
    m = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[:size]
    return conv

import numpy as np

def propose_candidate(seed=42, num_steps=400, **kwargs):
    """Find a non‑negative step sequence with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Obtain a starting vector (uniform mixed with the provided parent)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # uniform distribution, sum = 1
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # blend a little with the parent (already normalized)
    parent_norm = parent / max(parent.sum(), 1e-12)
    cur = 0.5 * uniform + 0.5 * parent_norm
    cur_sum = cur.sum()
    if cur_sum > 0:
        cur /= cur_sum
    else:
        cur = uniform.copy()

    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score

    # --------------------------------------------------------------
    # 2. Simple simulated‑annealing optimisation (pairwise mass moves)
    # --------------------------------------------------------------
    start_temp = 0.02  # initial temperature (fraction of score)
    for step in range(num_steps):
        # linear cooling schedule
        temp = start_temp * (1.0 - step / num_steps)

        # pick an index i
        i = rng.integers(0, n)

        # with 60 % probability move mass between mirrored positions,
        # otherwise pick a completely random partner.
        if rng.random() < 0.6:
            j = n - 1 - i
            if j == i:
                j = (j + 1) % n
        else:
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

        # amount to transfer: a fraction of the smaller entry
        amt = min(cur[i], cur[j]) * rng.random() * 0.5
        if amt <= 0:
            continue

        # random direction of the transfer
        if rng.random() < 0.5:
            new_i = cur[i] - amt
            new_j = cur[j] + amt
        else:
            new_i = cur[i] + amt
            new_j = cur[j] - amt

        if new_i < 0 or new_j < 0:
            continue

        cand = cur.copy()
        cand[i] = new_i
        cand[j] = new_j

        # enforce bounds and renormalise (scale invariance)
        np.clip(cand, 0.0, 1000.0, out=cand)
        total = cand.sum()
        if total <= 0:
            continue
        cand /= total

        # evaluate the candidate
        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue

        # accept if improvement or with Metropolis probability
        if score < cur_score or (temp > 0 and rng.random() < np.exp((cur_score - score) / temp)):
            cur = cand
            cur_score = score
            if score < best_score:
                best = cand.copy()
                best_score = score
                if best_score < 1.50274349:   # target reached – stop early
                    break

        # occasional tiny random perturbation to escape plateaus
        if step % 100 == 0 and rng.random() < 0.03:
            noise = rng.normal(0.0, 0.005, size=n)
            cur = np.clip(cur + noise, 0.0, 1000.0)
            cur_sum = cur.sum()
            if cur_sum > 0:
                cur /= cur_sum
                cur_score = float(evaluate_sequence(cur.tolist()))
                if cur_score < best_score:
                    best = cur.copy()
                    best_score = cur_score

    # --------------------------------------------------------------
