def fft_convolve(x, y):
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length() # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:n]
    return conv

import numpy as np

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using real FFT (linear, zero‑padded)."""
    n = x.shape[0] + y.shape[0] - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    z = np.fft.irfft(X * Y, n=N)[:n]
    return z


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    n = base.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed: int = 42, num_gradient: int = 300,
                     num_block: int = 500, **kwargs):
    """Generate a non‑negative step‑function approximating a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # 1.  Initialise from the provided parent (if any) or a uniform start #
    # ------------------------------------------------------------------ #
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():

    except Exception:               # fallback when parent is absent/invalid

    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-8:               # avoid division by zero
        a = np.ones_like(a)

    # The objective is scale‑invariant; keep a normalised for numerical health.
    a = a / a.sum()

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    n = a.shape[0]

    # ------------------------------------------------------------------ #
    # 2. Gradient‑descent phase on a smooth approximation of the max     #
    # ------------------------------------------------------------------ #
    step = 0.5                 # initial learning rate (in the normalised space)
    beta = 12.0                # temperature for soft‑max of the autocorrelation

    for _ in range(num_gradient):
        # autocorrelation a*a (full)
        conv_aa = _fft_convolve(a, a)                 # length 2n‑1
        maxb = conv_aa.max()

        # soft‑max weights p_k ≈ δ_{k, argmax}
        w = np.exp(beta * (conv_aa - maxb))

        # take a step and re‑project onto the simplex (non‑negative, sum=1)
