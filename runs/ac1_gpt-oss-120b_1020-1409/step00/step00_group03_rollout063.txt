import numpy as np

# The evaluator is supplied in the execution environment
# from the problem statement:
#   evaluate_sequence(sequence: list) -> float

def _initial_candidate(n: int) -> np.ndarray:
    """Return a feasible starting vector (uniform distribution)."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _random_transfer(rng, a, max_transfer=0.05):
    """Transfer a random amount of mass from one index to another."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # amount to move: a fraction of the source entry (or zero if source is tiny)
    amount = rng.random() * max_transfer * a[i]
    if amount > a[i]:
        amount = a[i]
    a[i] -= amount
    a[j] += amount
    return a

    for step in range(num_steps):
        # linear cooling
        t = step / float(num_steps)
        temperature = temp_start * (1 - t) + temp_end * t

        # Propose a new point by moving a small amount of mass

        s = candidate.sum()
