    for _ in range(40):                       # ~30‑40 iterations give high precision
        c = b - (b - a) * invphi
        d = a + (b - a) * invphi
        rc = _ratio_for_r(c, n)
        rd = _ratio_for_r(d, n)
        if rc < rd:
            b = d
        else:
            a = c

    r_opt = (a + b) * 0.5

    # ---------- Build the candidate sequence ----------
    seq = np.power(r_opt, np.arange(n, dtype=np.float64))
    seq = np.clip(seq, 0.0, 1000.0)            # safety against the evaluator limits

    # Ensure the sum is not pathological (should be fine for our construction)
    if np.sum(seq) < 0.01:
        seq = np.ones(n, dtype=np.float64)

    # Final sanity check – if something went wrong fall back to uniform ones
    try:
        sc = evaluate_sequence(seq.tolist())
        if not np.isfinite(sc):
            seq = np.ones(n, dtype=np.float64)
    except Exception:
        seq = np.ones(n, dtype=np.float64)

    # Return a plain Python list as required
    return seq.tolist()
