def _build_sum_unique_set(n: int, rng: np.random.Generator, target: int) -> list:
    """
    Greedy construction of a set S ⊂ {0,…,n-1} such that all sums
    i+j (i ≤ j, i,j∈S) are distinct.
    Returns the selected indices (list of ints). The routine tries a few
    random orderings and keeps the longest set found.
    """
    best_set = []
    # The theoretical maximum number of distinct sums is 2*n‑1.
    # We stop early when a set already reaches that limit.
    for _ in range(7):                     # a few random trials
        order = rng.permutation(n)         # random ordering of candidates
        used = set()                       # occupied sums
        cur = []
        for idx in order:
            # Quick reject if the self‑sum already taken
            if (idx + idx) in used:
                continue
            # Check all sums with the current set
            conflict = False
            for s in cur:
                if (idx + s) in used:
                    conflict = True
                    break
            if conflict:
                continue
            # Accept idx: record all new sums
            for s in cur:
                used.add(idx + s)
            used.add(idx + idx)
            cur.append(idx)
            # stop early if we already have enough elements
            if len(cur) >= target:
                # we still try to add more, but the loop will finish soon
                pass
        if len(cur) > len(best_set):
            best_set = cur
        # If the set is already maximal w.r.t. the available sum space, exit
        if len(best_set) * (len(best_set) + 1) // 2 >= 2 * n - 1:
            break
    return best_set


def _block_perturb(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:

            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # stop early if we already beat the target bound
            if best_score < 1.5028:
                break

    # ------------------------------------------------------------------
    # Very light local polishing (optional, inexpensive).
    # ------------------------------------------------------------------
    for _ in range(50):
        cand = _block_perturb(best, rng)
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return plain Python list as required by the harness.
    return [float(v) for v in best]
