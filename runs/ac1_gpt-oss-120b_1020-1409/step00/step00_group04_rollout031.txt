def generate_sidon(n, target_m):
    sums = set()
    indices = []
    for i in range(n):
        # early break if reached target
        if len(indices) >= target_m:
            break
        # compute sums if i added
        conflict = False
        if i + i in sums:
            conflict = True
        else:
            for j in indices:
                s = i + j
                if s in sums:
                    conflict = True
                    break
        if not conflict:
            indices.append(i)
            sums.add(i + i)
            for j in indices[:-1]:  # all previous j
                sums.add(i + j)
    return indices

import math
import numpy as np


def _generate_sidon_set(length):
    """Greedy construction of a B2 (Sidon) set in {0,…,length-1}.
    Guarantees that all unordered sums i+j (including i=i) are distinct.
    """
    sums_seen = set()
    indices = []
    for i in range(length):
        # self‑sum must be new
        if i + i in sums_seen:
            continue
        conflict = False
        for j in indices:
            if i + j in sums_seen:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        indices.append(i)
        sums_seen.add(i + i)
        for j in indices[:-1]:
            sums_seen.add(i + j)
    return indices

    autocorrelation upper bound.

        maximal allowed height 1000.
    The Sidon property ensures that at any convolution index at most
    one unordered pair of spikes contributes, so the maximal
    convolution value is 2·1000².
    """
    # ------------------------------------------------------------------
    # Determine the length (use the provided parent if possible)
    # ------------------------------------------------------------------

        n = int(base.size)
    except Exception:
        # fallback length
        n = 256
        base = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
