    positions = []
    used_sums = set()
    attempts = 0
    max_attempts = n * 10  # limit to avoid infinite loops
    while len(positions) < m_target and attempts < max_attempts:
        cand = int(rng.integers(0, n))
        # ensure not already chosen
        if cand in positions:
            attempts += 1
            continue
        # compute sums with existing positions including self (cand + cand)
        bad = False
        for p in positions:
            s = p + cand
            if s in used_sums:
                bad = True
                break
        if not bad:
            # check self sum
            s_self = cand + cand
            if s_self in used_sums:
                bad = True
        if not bad:
            positions.append(cand)
            # add sums
            for p in positions: # need to add for new cand with all, including itself
                used_sums.add(p + cand)
        attempts += 1
    return positions
```

def build_positions(n, max_m, K, rng):
    sum_counts = np.zeros(2*n-1, dtype=np.int16)
    positions = []
    attempts = 0
    max_attempts = n*10
    while len(positions) < max_m and attempts < max_attempts:
        cand = int(rng.integers(0, n))
        if cand in positions:
            attempts += 1
            continue
        # Check if adding cand would exceed K for any sum
        ok = True
        for p in positions:
            s = p + cand
            if sum_counts[s] >= K:
                ok = False
                break
        if ok and sum_counts[cand + cand] < K:
            # Add cand
            for p in positions:
                sum_counts[p + cand] += 1
            sum_counts[cand + cand] += 1
            positions.append(cand)
        attempts += 1
    return positions
```

import numpy as np

def _greedy_sparse_candidate(n: int, K: int, max_val: float = 1000.0) -> np.ndarray:
    """
    Build a sparse non‑negative vector of length ``n``.
    At most ``K`` products may share the same convolution index
    (i.e. each sum i+j may appear at most K times).
    All non‑zero entries are set to ``max_val``.
    """
    # count how many times each possible sum (i+j) has been used
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    positions = []
    for idx in range(n):
        # self‑sum must respect the bound
        if sum_counts[idx + idx] >= K:
            continue
        # check sums with already chosen positions
        ok = True
        for p in positions:
            if sum_counts[p + idx] >= K:
                ok = False
                break
        if not ok:
            continue
        # accept this index
        for p in positions:
            sum_counts[p + idx] += 1
        sum_counts[idx + idx] += 1
        positions.append(idx)

    vec = np.zeros(n, dtype=np.float64)
    if positions:
        vec[positions] = max_val
    return vec
