import numpy as np

# ------------------------------------------------------------
# Fast convolution (full) using FFT – O(n log n) memory and time
# ------------------------------------------------------------
def _full_convolve_fft(x: np.ndarray) -> np.ndarray:
    """Return np.convolve(x, x) using real FFT (full mode)."""
    n = x.shape[0]
    size = 2 * n - 1
    xf = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(xf * xf, n=size)
    return conv


# ------------------------------------------------------------
# Compute the bound 2*n*max(b)/(sum(a)^2) without the extra
# clipping that evaluate_sequence performs (our candidates are
# already clipped to the allowed range).
# ------------------------------------------------------------
def _bound_fft(a: np.ndarray, sum_a: float) -> float:
    """Bound for a given sequence a (already non‑negative, ≤1000)."""
    if sum_a < 0.01:
        return np.inf
    conv = _full_convolve_fft(a)
    max_b = np.max(conv)
    return 2.0 * a.shape[0] * max_b / (sum_a * sum_a)

def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """Return a list of non‑negative heights whose AC1 bound is small."""
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------
    # Determine length n (use the provided parent if it exists)
    # ----------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # ----------------------------------------------------------------
    # 1. Coarse grid search over power‑law exponents and offsets
    # ----------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64) + 1.0  # 1‑based indices
    best_seq = None
    best_score = np.inf

    exponents = [0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.65]
    offsets = [0.0, 0.5, 1.0, 2.0, 4.0]

    for p in exponents:
        for off in offsets:
            seq = (idx + off) ** (-p)
            # clip only for safety (values are far below 1000)
            seq = np.clip(seq, 0.0, 1000.0)
            s = np.sum(seq)
            score = _bound_fft(seq, s)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()

    # ----------------------------------------------------------------
    # 2. Refine the best candidate with a very simple projected‑gradient
    #    descent on the quantity max(b) while keeping the L1 norm fixed.
    # ----------------------------------------------------------------
    a = best_seq.copy()
    sum_target = np.sum(a)  # constant throughout the refinement
    size_fft = 2 * n - 1

    # pre‑compute current bound (used for acceptance test)
    cur_score = _bound_fft(a, sum_target)

    for step in range(num_steps):
        # learning‑rate schedule (slowly decreasing)
        eta = 0.5 * (1.0 - step / num_steps) + 0.01

        # convolution and location of the current maximal entry
        xf = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(xf * xf, n=size_fft)
        max_idx = int(np.argmax(conv))

        # gradient of the maximal entry w.r.t. a_i :
        #   ∂b_max / ∂a_i = a_{max_idx‑i}  (if the index lies inside the vector)
        w = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j = max_idx - i_range
        w[i_range] = a[j]

        # take a small step opposite to the gradient, keep non‑negativity,
        # then renormalise to the original L1 norm (scale‑invariant objective)
        a_new = a - eta * w
        a_new = np.clip(a_new, 0.0, 1000.0)

        s_new = np.sum(a_new)
        if s_new > 0.0:
            a_new *= (sum_target / s_new)
        else:
            a_new = a.copy()

        # evaluate the new bound (fast, no need to call evaluate_sequence yet)
        new_score = _bound_fft(a_new, sum_target)

        # keep the best ever seen
        if new_score < best_score:

            best_seq = a_new.copy()

        # accept the move only if it improves the current bound
        if new_score < cur_score:
            a = a_new
            cur_score = new_score

            a *= (sum_target / np.sum(a))
            cur_score = _bound_fft(a, sum_target)

    # ----------------------------------------------------------------
    # Return the best sequence found as a plain Python list
    # ----------------------------------------------------------------
    return [float(x) for x in best_seq]
