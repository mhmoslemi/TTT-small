def propose_candidate(seed=42, num_iters=15, top_k=10, t=0.1, **kwargs):
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0., 1000.)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)
    # start with uniform normalized maybe sum=1 for numeric stability
    a = parent.copy()
    # ensure sum >0:
    if np.sum(a) < 0.01:
        a = np.ones_like(a)
    # optional scaling to ensure reasonable magnitude e.g., keep values <= 1
    # you can rescale to have sum = 1 (or any)
    # We'll just use as-is.

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    n = a.size
    for iter in range(num_iters):
        conv = np.convolve(a, a, mode='full')
        M = np.max(conv)
        # get top indices (distinct) with largest conv values
        # we can use np.argpartition to get top_k+some more. Use negative.
        idxs = np.argpartition(-conv, range(min(top_k, conv.size)))[:top_k]
        # Also ensure include central index
        central = n - 1
        if central not in idxs:
            idxs = np.append(idxs, central)
        # Build LP
        K = idxs.size
        A_ub = np.zeros((K, n))
        for row_i, k in enumerate(idxs):
            # a * b coefficient at index k
            # For j from max(0, k-(n-1)) to min(n-1, k)
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            # slice of length (j_end - j_start + 1)
            # indices j_start..j_end inclusive
            # corresponding a indices are k - j
            A_ub[row_i, j_start:j_end+1] = a[k - np.arange(j_start, j_end+1)]
        # bounds for b variables
        bounds = [(0.0, 1000.0)] * n
        # objective: maximize sum of b -> minimize negative sum
        c = -np.ones(n)
        # RHS: M for each constraint
        b_ub = np.full(K, M)
        # solve
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fallback: break
            break
        b = res.x
        sum_a = np.sum(a)
        sum_b = np.sum(b)
        if sum_b == 0:
            # skip
            break
        # rescale b to have same sum as a
        g = b * (sum_a / sum_b)
        # update a with mixing
        a = (1 - t) * a + t * g
        # clip to bounds
        a = np.clip(a, 0.0, 1000.0)
        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
        # optional early stop if score close to target
        if best_score < 1.6:
            # maybe break early
            pass
    # return best as list of floats
    return [float(x) for x in best]
