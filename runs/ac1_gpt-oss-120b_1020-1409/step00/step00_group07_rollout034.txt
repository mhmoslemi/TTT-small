def spaced_arrangement(values):
    indices = np.arange(len(values))
    # sort values descending
    sorted_vals = np.sort(values)[::-1]
    n = len(values)
    k = len(sorted_vals)
    # we will place them into sequence positions using step = n // k
    step = max(1, n // k)
    seq = np.zeros(n)
    pos = 0
    for v in sorted_vals:
        seq[pos] = v
        pos = (pos + step) % n
        # if collision (position already filled) just find next free
        while seq[pos] != 0:
            pos = (pos + 1) % n
    return seq
```

import numpy as np
import math

def _fast_score(seq: np.ndarray, L: int) -> float:
    """Return the AC1 upper bound using FFT (no clipping)."""
    n = seq.size
    total = seq.sum()
    if total < 0.01:
        return np.inf
    # convolution via FFT (real‐to‐real)
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (total * total)


def _alternating_arrangement(values: np.ndarray) -> np.ndarray:
    """Place large values alternating from left and right to separate them."""
    n = values.size
    sorted_vals = np.sort(values)[::-1]      # descending
    out = np.empty_like(values)
    left, right = 0, n - 1
    toggle = True
    idx = 0
    while left <= right:
        if toggle:
            out[left] = sorted_vals[idx]
            left += 1
        else:
            out[right] = sorted_vals[idx]
            right -= 1
        toggle = not toggle
        idx += 1
    return out


def propose_candidate(seed: int = 42, max_iters: int = 3000, **kwargs):
    """
    Return a list of non‑negative step heights with a low AC1 bound.
    The routine starts from the provided parent (height_sequence_1) and
    performs a simple stochastic optimisation that swaps entries to
    decorrelate large values.  The score is evaluated with an FFT‑based
    proxy for speed; the final candidate is verified with the official
    evaluator before returning.
    """
    rng = np.random.default_rng(seed)

    # ----- initialise parent sequence ---------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:                     # no parent supplied by harness
        parent = np.ones(256, dtype=np.float64)

    # safety clipping (the evaluator also clips, but we stay consistent)
    parent = np.clip(parent, 0.0, 1000.0)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    L = 1 << ((2 * n - 1).bit_length())   # power‑of‑two length for FFT

    # ----- baseline candidates -----------------------------------------
    best_seq = parent.copy()
    best_score = _fast_score(best_seq, L)

    # try a deterministic alternating layout (often reduces peaks)
    alt = _alternating_arrangement(parent)
    alt_score = _fast_score(alt, L)
    if alt_score < best_score:
        best_seq, best_score = alt, alt_score

    # ----- stochastic hill‑climbing ------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score

    for _ in range(max_iters):
        # propose a swap of two random positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = current_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]

        cand_score = _fast_score(cand, L)

        # accept if better, otherwise with a tiny chance to escape plateaus
        if cand_score < current_score or rng.random() < 0.01:
            current_seq, current_score = cand, cand_score

        # keep global best
        if cand_score < best_score:
            best_seq, best_score = cand, cand_score

        # occasional random full reshuffle to explore far regions
        if rng.random() < 0.005:
            reshuffled = parent.copy()
            rng.shuffle(reshuffled)
            reshuffled_score = _fast_score(reshuffled, L)
            if reshuffled_score < best_score:
                best_seq, best_score = reshuffled, reshuffled_score
            # allow the walk to jump to that configuration
            if reshuffled_score < current_score:
                current_seq, current_score = reshuffled, reshuffled_score

    # ----- final verification with the official evaluator ----------------
    try:
        # the evaluator is pre‑imported in the harness
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if not np.isfinite(final_score):
            # fallback to a safe uniform sequence
            best_seq = np.full(n, 1.0, dtype=np.float64)
    except Exception:
        # any unexpected failure – return a simple non‑empty list
        best_seq = np.full(max(1, n), 1.0, dtype=np.float64)

    # ensure the result respects the required bounds
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
