import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    """Simple uniform vector (sum = n)."""
    return np.full(n, 1.0, dtype=np.float64)


def _dirichlet_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex."""
    # Dirichlet with all parameters = 1 gives a uniform distribution on the simplex
    return rng.dirichlet(np.ones(n, dtype=np.float64))


def _gradient_step(a: np.ndarray) -> np.ndarray:
    """
    One projected gradient step aimed at reducing the current maximal
    convolution entry.
    """
    n = a.size
    # exact convolution (full) – NumPy will use an FFT internally for large n
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # gradient of b[k_max] w.r.t. a_i is 2 * a_{k_max-i} (zero outside the support)
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[partner[mask]]

    # step size proportional to the current average magnitude
    step = 0.1 * a.mean()
    cand = a - step * grad
    cand = np.clip(cand, 0.0, 1000.0)

    # renormalise to keep the total sum constant (scaling does not affect the score)
    total = cand.sum()
    if total > 0.0:
        cand *= (a.sum() / total)
    return cand

    return float(evaluate_sequence(seq.tolist()))

        cand = _dirichlet_candidate(n, rng)  # starts on the simplex (sum = 1)

        # apply a handful of cheap gradient steps to push the max convolution down
        for __ in range(gd_steps):
            cand = _gradient_step(cand)

        score = _evaluate(cand)
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            # early exit if we have already crossed a reasonable target
            if best_score < 1.6:
                break

    # ------------------------------------------------------------
    # Return a plain Python list of floats (required by the harness)
    # ------------------------------------------------------------
    return [float(x) for x in best]
```
