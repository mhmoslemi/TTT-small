import numpy as np

def propose_candidate(seed=42, num_iters=2000, lr=0.1, **kwargs):
    rng = np.random.default_rng(seed)
    # get length
    try:
        n = int(height_sequence_1.shape[0])
    except NameError:
        # fallback length 256
        n = 256
    # initialize uniform
    a = np.full(n, 1.0 / n, dtype=np.float64)
    # maybe incorporate parent if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n and np.isfinite(parent).all():
            parent = np.clip(parent, 0.0, 1000.0)
            # normalize to sum 1
            if parent.sum() > 0.0:
                a = parent / parent.sum()
    except Exception:
        pass
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # main loop
    for it in range(num_iters):
        conv = np.convolve(a, a)
        max_val = conv.max()
        # active indices (exact equality)
        active = np.where(conv == max_val)[0]
        # compute subgradient
        g = np.zeros_like(a)
        # for each active lag, add contributions
        for k in active:
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            idx = np.arange(start, end + 1)
            g[idx] += a[k - idx]
        # multiplicative update
        a = a * np.exp(-lr * g)
        # renormalize to sum 1 (to keep scaling)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
            s = a.sum()
        a /= s
        # occasional random jitter to escape flat region
        if it % 200 == 0 and it > 0:
            jitter = rng.normal(loc=1.0, scale=0.02, size=n)
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        # enforce bounds (1000) after scaling - should be fine
        a = np.clip(a, 0.0, 1000.0)
        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
        # decay learning rate
        lr *= 0.9995
    # final fallback: use parent if no improvement
    try:
        parent_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent_seq.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent_seq
            best_score = parent_score
    except Exception:
        pass
    # return list
    return [float(x) for x in best]

import numpy as np

def propose_candidate(seed=42, num_iters=1500, lr_start=5.0, **kwargs):
    """
    Gradient based flattening of the auto‑convolution.
    Starts from a (possibly) jittered uniform distribution (or the provided
    parent sequence) and iteratively moves mass from indices that contribute
    most to the current peak towards the edges.
    Returns a list of non‑negative coefficients.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Determine length and initialise a feasible sequence (sum == 1)
    # ----------------------------------------------------------------------
    try:
        # height_sequence_1 is supplied by the harness
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
        base = np.clip(base, 0.0, 1000.0)
        if base.sum() > 0.0:
            a = base / base.sum()
        else:
            raise ValueError
    except Exception:
        # fallback: uniform vector of the appropriate length (default 256)
        try:
            n = int(height_sequence_1.shape[0])  # use length if parent exists
        except Exception:
            n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 2. Small random jitter to avoid the uniform stationary point
    # --------------------------------------------------------------
    jitter = rng.uniform(0.9, 1.1, size=n)
    a = a * jitter
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum

    # --------------------------------------------------------------
    # 3. Main optimisation loop – minimise variance of the convolution
    # --------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    lr = lr_start

    target_const = 1.0 / (2 * n - 1)          # target value of a*a for sum == 1

    for it in range(num_iters):
        # full linear convolution
        conv = np.convolve(a, a)               # length 2n‑1
        # deviation from the flat target
        w = conv - target_const

        # sub‑gradient of  Σ (conv‑target)^2  w.r.t. a
        # g_j = 4 * Σ_i a_i * w_{i+j}   (cross‑correlation, mode='valid')
        g = 4.0 * np.correlate(a, w, mode='valid')

        # multiplicative update – keeps positivity automatically
        a = a * np.exp(-lr * g)

        # normalise to keep the scale fixed (the score is scale‑invariant)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # occasional random jitter to escape tiny plateaux
        if (it + 1) % 300 == 0:
            jitter = rng.normal(1.0, 0.02, size=n)
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # decay learning rate slowly
        lr *= 0.9995

        # evaluate – keep the best valid candidate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score

    # ----------------------------------------------------------------------
    # 4. Light‑weight local random perturbations as a safety net
    # ----------------------------------------------------------------------
    def random_block_perturb(x):
        """Scale a random contiguous block and renormalise."""
        y = x.copy()
        blk_len = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - blk_len + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        y[start:start + blk_len] *= factor
        y = np.clip(y, 0.0, None)
        tot = y.sum()
        if tot == 0.0:
            return np.full(n, 1.0 / n, dtype=np.float64)
        y /= tot
        return y

    # a few cheap probes around the best found point
    probe_best = best.copy()
    probe_score = best_score
    for _ in range(200):
        cand = random_block_perturb(probe_best)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < probe_score:
            probe_best = cand
            probe_score = sc

    if probe_score < best_score:
        best = probe_best
        best_score = probe_score

    # ----------------------------------------------------------------------
    # 5. Final fallback to parent if it happened to be better
    # ----------------------------------------------------------------------
    try:
        parent_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                             0.0, 1000.0)
        pscore = float(evaluate_sequence(parent_seq.tolist()))
        if np.isfinite(pscore) and pscore < best_score:
            best = parent_seq / max(parent_seq.sum(), 1e-12)   # normalised
            best_score = pscore
    except Exception:
        pass

    # Return as a plain Python list of floats
    return [float(v) for v in best]
