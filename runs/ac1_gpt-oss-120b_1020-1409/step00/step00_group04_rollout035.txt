def generate_sidon_set(n, target_m, rng):
    indices = []
    sum_set = set()
    attempts = 0
    max_attempts = target_m * 50  # some attempts
    while len(indices) < target_m and attempts < max_attempts:
        i = int(rng.integers(0, n))
        # ensure i not already used
        if i in indices:
            attempts += 1
            continue
        conflict = False
        if (i + i) in sum_set:
            conflict = True
        else:
            for j in indices:
                s = i + j
                if s in sum_set:
                    conflict = True
                    break
        if not conflict:
            # add
            for j in indices:
                sum_set.add(i + j)
            sum_set.add(i + i)
            indices.append(i)
        attempts += 1
    return np.array(indices, dtype=int)

def _generate_sidon_indices(n, target_m, rng):
    """Return a list of indices forming a Sidon set (unique pairwise sums)."""
    indices = []
    sum_set = set()
    attempts = 0
    max_attempts = target_m * 1000
    while len(indices) < target_m and attempts < max_attempts:
        i = int(rng.integers(0, n))
        if i in indices:
            attempts += 1
            continue
        # test uniqueness of sums
        conflict = False
        # self sum
        if (i + i) in sum_set:
            conflict = True
        else:
            for j in indices:
                if (i + j) in sum_set:
                    conflict = True
                    break
        if not conflict:
            # add new sums
            for j in indices:
                sum_set.add(i + j)
            sum_set.add(i + i)
            indices.append(i)
        attempts += 1
    return np.array(indices, dtype=int)
