import numpy as np

def _make_polynomial_ramp(n: int, power: int) -> np.ndarray:
    """
    Return a non‑negative step function of length n with values proportional to (i+1)^power,
    scaled so that the maximum entry is exactly 1000 (ensuring no clipping by the evaluator).
    """
    # indices 1 … n as float
    idx = np.arange(1, n + 1, dtype=np.float64)
    # scaling factor so that max(idx**power) * scale == 1000
    scale = 1000.0 / (float(n) ** power)
    return (idx ** power) * scale
