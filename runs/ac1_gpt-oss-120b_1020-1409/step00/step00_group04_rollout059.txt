import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Search for a step–function that lowers the AC1 upper bound.
    The search explores a simple two‑region family:
        - a leading block with a geometric decay a_i = r**i
        - a trailing flat region with constant height eps
    Various block lengths, decay factors and epsilons are tried
    (both on a deterministic grid and via a small random walk).
    The best candidate found is returned as a list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Base fallback (the provided parent construction, if any)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is supplied by the harness when available
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        # a generic non‑empty fallback
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    # ------------------------------------------------------------------
    # 2.  Deterministic grid search over the two‑region family
    # ------------------------------------------------------------------
    # decay factors (r close to 1 gives a slowly decreasing block)
    r_vals = np.linspace(0.990, 0.9999, 20)            # 20 values
    # flat tail heights
    eps_vals = np.linspace(0.01, 0.60, 12)            # 12 values
    # block lengths as fractions of n (avoiding the trivial 0 / n cases)
    m_fractions = np.linspace(0.20, 0.80, 7)          # 7 values
    m_vals = (m_fractions * n).astype(int)

    for m in m_vals:
        if m <= 0 or m >= n:
            continue
        # pre‑allocate the tail (filled later with the block)
        tail = np.empty(n, dtype=np.float64)
        for eps in eps_vals:
            tail[:] = eps                                 # flat tail everywhere
            for r in r_vals:
                # geometric block on the left side
                block = r ** np.arange(m, dtype=np.float64)
                tail[:m] = block
                # evaluate
                score = float(evaluate_sequence(tail.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = tail.copy()
                    best_score = score

    # ------------------------------------------------------------------
    # 3.  Small random exploration around promising regions
    # ------------------------------------------------------------------
    for _ in range(300):
        # random block size (10 %–90 % of the length)
        m = rng.integers(int(0.10 * n), int(0.90 * n))
        # random flat tail height (0.01–0.60)
        eps = rng.random() * 0.59 + 0.01
        # random decay factor (0.99–1.0)
        r = 1.0 - rng.random() * 0.01
        cand = np.full(n, eps, dtype=np.float64)
        cand[:m] = r ** np.arange(m, dtype=np.float64)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found
    # ------------------------------------------------------------------
    return [float(x) for x in best]
