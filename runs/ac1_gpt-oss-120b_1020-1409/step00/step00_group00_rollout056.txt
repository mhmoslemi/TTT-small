import numpy as np

def alternating_extremes(vals):
    n = vals.shape[0]
    res = np.empty(n, dtype=np.float64)
    left = 0
    right = n-1
    toggle = True
    for v in vals:
        if toggle:
            res[left] = v
            left += 1
        else:
            res[right] = v
            right -= 1
        toggle = not toggle
    return res

def make_block_perturbation(base, rng):
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    if rng.random() < 0.35:

def propose_candidate(seed=42, num_steps=1000, **kwargs):
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 256
    n = base_len
    # generate random values
    vals = rng.exponential(scale=1.0, size=n)
    vals = np.clip(vals, 0.0, 1000.0)
    sorted_vals = np.sort(vals)[::-1]
    candidate = alternating_extremes(sorted_vals)

    # ensure sum not too small
    if np.sum(candidate) < 0.01:
        candidate = np.ones(n, dtype=np.float64)
    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = best.copy()
    cur_score = best_score
    for step in range(num_steps):
        if rng.random() < 0.8:
            # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            cand = make_block_perturbation(cur, rng)
        # clip already done in block perturbation
        if np.sum(cand) < 0.01:
            continue
        score = float(evaluate_sequence(cand.tolist()))
        if score < cur_score:
            cur = cand
            cur_score = score
        if score < best_score:
            best = cand
            best_score = score
    # final fallback check
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
    return [float(x) for x in best]

import numpy as np

def alternating_extremes(vals: np.ndarray) -> np.ndarray:
    """Place the sorted values alternating from the left and right ends."""
    n = vals.shape[0]
    out = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    toggle = True
    for v in vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _lp_step(f: np.ndarray, M: float, idxs: np.ndarray) -> np.ndarray | None:
    """Solve the linear programme that maximises sum(g) subject to f*g ≤ M on idxs."""
    n = f.size
    k = len(idxs)
    A = np.zeros((k, n), dtype=np.float64)
    for row, kk in enumerate(idxs):
        lo = max(0, kk - (n - 1))
        hi = min(kk, n - 1)
        js = np.arange(lo, hi + 1, dtype=np.int64)
        # g[j] contributes f[kk-j]
        A[row, lo:hi + 1] = f[kk - js]
    b = np.full(k, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                     # maximise sum(g)
    bounds = [(0.0, 1000.0)] * n                           # respect problem limits
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, num_steps: int = 0, **kwargs):
    """
    Generate a step‑function (non‑negative vector) that tries to minimise
    the AC1 upper bound used by `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length (use the provided parent if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(height_sequence_1))
    else:
        n = 256  # arbitrary fallback length

    # ------------------------------------------------------------------
    # Initial construction – a broad random shape with large values spread
    # ------------------------------------------------------------------
    vals = rng.exponential(scale=1.0, size=n)
    vals = np.clip(vals, 0.0, 1000.0)
    sorted_vals = np.sort(vals)[::-1]                 # descending
    f = alternating_extremes(sorted_vals)            # start vector

    # Ensure a non‑zero sum (required by the evaluator)
    if np.sum(f) < 0.01:
        f = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # LP‑based refinement (only if SciPy is available)
    # ------------------------------------------------------------------
    if _SCIPY_OK:
        max_iter = 8                      # a handful of LP steps
        top_k = 10                        # constraints kept from the active set
        mix = 0.25                        # convex‑combination weight
        for _ in range(max_iter):
            conv = np.convolve(f, f)                # full autocorrelation
            M = float(np.max(conv))
            # pick indices where the current bound is tight
            if top_k >= conv.size:
                active = np.arange(conv.size)
            else:
                active = np.argpartition(-conv, top_k)[:top_k]
            g = _lp_step(f, M, active)
            if g is None or np.sum(g) < 1e-12:
                break
            # move a fraction toward the LP solution
            f = (1.0 - mix) * f + mix * g
            # the convex combination respects the 0‑1000 box automatically
    else:
        # If SciPy is missing we simply keep the random start.
        pass

    # ------------------------------------------------------------------
    # Local greedy search with swaps (few hundred cheap evaluations)
    # ------------------------------------------------------------------
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = best.copy()
    cur_score = best_score

    local_steps = 200
    for _ in range(local_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = cur.copy()
        cand[i], cand[j] = cand[j], cand[i]
        # quick validation – sum cannot be too small
        if np.sum(cand) < 0.01:
            continue
        score = float(evaluate_sequence(cand.tolist()))
        if score < cur_score:
            cur = cand
            cur_score = score
        if score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Final safeguard – fall back to a simple uniform vector if needed
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
