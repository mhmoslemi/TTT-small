    size = 2 * n - 1

    conv = np.fft.irfft(fa * fa, fft_len)[:size]

def propose_candidate(seed=42, num_steps=2000, **kwargs):

    parent = None

        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:

        else:
            a = a / s

    a = r ** np.arange(n, dtype=np.float64)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    step = 0.05

        max_b = conv.max()
        score = float(2 * n * max_b / (a.sum() ** 2))
        if score < best_score and np.isfinite(score):
            best = a.copy()
            best_score = score
            # early stop if below target bound

        # compute gradient for the argmax index

import numpy as np

def full_convolve(a):
    n = a.shape[0]
    size = 2*n - 1
    fft_len = 1 << ((size - 1).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:size]
    return conv

def gradient_l2(a, b, S):
    # b = convolution of a with a (full)
    n = a.shape[0]
    C = S*S / (2*n - 1)
    diff = b - C
    a_rev = a[::-1]
    size = len(diff) + n -1 # diff length 2n-1 + n-1 = 3n-2
    fft_len = 1 << ((size - 1).bit_length())
    fd = np.fft.rfft(diff, fft_len)
    fr = np.fft.rfft(a_rev, fft_len)
    conv_full = np.fft.irfft(fd * fr, fft_len)[:size]
    start = n - 1
    g = 2.0 * conv_full[start:start + n]
    return g, C
```

import numpy as np

# ----- convolution utilities (FFT, O(n log n) memory) -----
def full_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 2 * n - 1
    fft_len = 1 << ((size - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:size]
    return conv


def score_fast(a: np.ndarray) -> float:
    """Exact AC1 score using the same formula that evaluate_sequence uses."""
    n = a.shape[0]
    conv = full_convolve(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def propose_candidate(seed: int = 42, num_steps: int = 1500, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes the AC1 upper bound.
    The method starts from the analytic shape a_i ∝ 1/√(i+1) (which gives a
    convolution that is almost flat) and then performs cheap stochastic
    improvements while keeping memory linear in the length of the sequence.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem (use the provided construction if present)
    try:
        n = int(height_sequence_1.shape[0])          # type: ignore[name-defined]
    except Exception:
        n = 6000                                      # fallback length

    # ---- 1. initial analytic shape a_i = 1 / sqrt(i+1) ----
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    a = a / a.sum()                                   # normalise (scale does not matter)

    # ---- 2. keep the best candidate seen so far ----
    best = a.copy()
    best_score = score_fast(best)

    # optional: try the parent construction as a second starting point
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)  # type: ignore[name-defined]
        if parent.shape == (n,) and np.isfinite(parent).all() and parent.ndim == 1:

            sc = score_fast(parent)

    except Exception:

    target = 1.50274349          # the bound we would like to beat
