import numpy as np

# ----- helper functions ----------------------------------------------------
def _random_dirichlet(rng, size, total):
    """Generate a random probability vector of given size with given total."""
    v = rng.random(size)
    v /= v.sum()
    return v * total


def _make_transfer(rng, arr, i, j):
    """Transfer a random fraction of mass from index i to index j."""
    if arr[i] <= 0.0:
        return arr.copy()
    frac = rng.uniform(0.01, 0.12)          # 1‑12 % of the source mass
    delta = arr[i] * frac
    out = arr.copy()
    out[i] = max(out[i] - delta, 0.0)
    out[j] = min(out[j] + delta, 1000.0)
    return out

    """
    Return a list of non‑negative heights that (heuristically) lower the
    AC1 upper bound.  The function works with the optional global
    ``height_sequence_1`` if it exists; otherwise it falls back to a uniform
    start vector.
    """

    # -------- initial sequence ---------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError

        cur = np.clip(parent, 0.0, 1000.0)
    except Exception:
        n = 256
        cur = np.full(n, 1.0, dtype=np.float64)

    if cur.sum() < 0.01:
        cur = np.full(n, 1.0, dtype=np.float64)

    best = cur.copy()

        conv = np.convolve(best, best)

    return [float(v) for v in best]
