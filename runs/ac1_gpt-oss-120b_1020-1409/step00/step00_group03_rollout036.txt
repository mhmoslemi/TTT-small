import numpy as np

# evaluate_sequence is provided by the harness.


def _initial_sequence(n, rng):
    """
    Produce a random positive starting vector of length n.
    The vector is normalised to sum to 1 (scale does not affect the
    objective, but keeps numbers in a safe range).
    """
    # log‑normal style random values, always >0
    a = np.exp(rng.normal(0.0, 0.5, size=n))
    s = a.sum()
    if s > 0.0:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _gradient_step(a, tmax, eta):
    """
    One sub‑gradient descent step for the active shift `tmax`.
    The derivative of (a*a)_tmax w.r.t. a_i equals 2*a_{tmax-i}
    (zero when the index is out of range).  We ignore the factor 2
    and use a step size `eta`.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, tmax - (n - 1))
    i_end = min(n - 1, tmax)
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    grad[idx] = a[tmax - idx]
    # take a step opposite to the gradient
    a = a - eta * grad
    # enforce non‑negativity (clip to zero)
    np.clip(a, 0.0, None, out=a)
    return a


def propose_candidate(seed=42, num_iters=2000, eta=0.07, noise_scale=0.015, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that
    attempts to minimise the AC1 upper bound.
    The algorithm:
        * start from a random positive vector (sum normalised to 1)
        * repeatedly:
            – compute the full convolution
            – locate the shift with the maximal entry
            – perform a sub‑gradient step decreasing that entry
            – inject a tiny multiplicative noise to avoid stagnation
            – renormalise to keep the sum stable
        * keep the best vector seen (lowest finite score)
    If anything goes wrong a uniform fallback is returned.
    """
    rng = np.random.default_rng(seed)

    # Length of the sequence – use the provided parent if available.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # ----- initialise -----
    a = _initial_sequence(n, rng)

    # keep a copy of the best candidate so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ----- main optimisation loop -----
    for _ in range(num_iters):
        # convolution (full, linear)
        conv = np.convolve(a, a)
        tmax = int(np.argmax(conv))

        # sub‑gradient descent step targeting the current maximal shift
        a = _gradient_step(a, tmax, eta)

        # tiny stochastic perturbation to keep the search alive
        if noise_scale > 0.0:
            a *= np.exp(rng.normal(0.0, noise_scale, size=n))

        # renormalise (scale does not matter for the score, but avoids
        # numerical under/overflow and keeps values comfortably <1000)
        total = a.sum()
        if total <= 0.0:
            # pathological case – re‑initialise
            a = _initial_sequence(n, rng)
            total = a.sum()
        a = a / total

        # evaluate candidate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()

        # optional: slowly decay the step size (helps convergence)
        eta *= 0.9995

    # ----- fallback safety -----
    if not np.isfinite(best_score):
        # fallback to a simple uniform positive vector
        best = np.ones(n, dtype=np.float64) / n
    # guarantee the output format required by the harness
    return [float(x) for x in best]
