import numpy as np

# fast evaluation using FFT (identical to evaluate_sequence up to rounding)
def _fast_score(arr: np.ndarray, size: int) -> float:
    """Return 2*n*max(conv)/sum^2 using FFT, where conv = arr*arr."""
    n = arr.shape[0]
    freq = np.fft.rfft(arr, size)
    conv = np.fft.irfft(freq * freq, size)[: 2 * n - 1]
    max_b = np.max(conv)
    s = np.sum(arr)
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, num_steps: int = 200, **kwargs):
    """Search for a sequence with a low AC1 upper bound.

    The method follows the LP‑style update described in the statement:
    we keep a current sequence ``f`` and, using the shift where the
    self‑convolution attains its maximum, we construct a helper vector ``g``
    that respects the linear constraint  f * g ≤ M .  A convex combination
    (1‑t)·f + t·g (t small) never increases the maximum convolution.
    Repeating the step reduces the bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise source sequence ``f``
    # ------------------------------------------------------------------
    try:
        # parent sequence is supplied in the execution environment
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        # fallback: uniform vector (any non‑zero length works)
        parent = np.full(256, 1.0, dtype=np.float64)

    n = parent.shape[0]
    f = parent.copy()

    # exact baseline score (used for acceptance)
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # pre‑compute FFT length (next power of two)
    size = 1 << int(np.ceil(np.log2(2 * n - 1)))

    # mixing parameter – small enough to keep g·g term negligible
    t_mix = 0.1
    eps = 1e-12

    for step in range(num_steps):
        # --------------------------------------------------------------
        # 1.  Compute current maximum convolution and the shift that attains it
        # --------------------------------------------------------------
        freq_f = np.fft.rfft(f, size)
        conv_f = np.fft.irfft(freq_f * freq_f, size)[: 2 * n - 1]
        max_b = np.max(conv_f)
        k_star = int(np.argmax(conv_f))               # one of the maximisers

        # --------------------------------------------------------------
        # 2.  Build the coefficient vector h_i = f_{k*‑i} (zero if out of range)
        # --------------------------------------------------------------
        idx = np.arange(n)
        partner = k_star - idx
        valid = (partner >= 0) & (partner < n)
        h = np.zeros_like(f)
        h[valid] = f[partner[valid]]

        # --------------------------------------------------------------
        # 3.  Construct g respecting  ⟨h, g⟩ ≤ max_b  and maximizing Σ g_i
        # --------------------------------------------------------------
        order = np.argsort(h, kind="mergesort")   # ascending, ties kept stable
        g = np.zeros_like(f)
        budget = max_b

        # allocate to zero‑cost slots first (no effect on the constraint)
        for i in order:
            if h[i] <= eps:
                g[i] = 1000.0            # max allowed – will be clipped later
                continue

            # allocate as much as possible respecting the budget and a
            # per‑coordinate cap of 1000 (the evaluator also caps)
            possible = budget / h[i]
            if possible <= 0.0:
                continue
            alloc = min(possible, 1000.0)
            g[i] = alloc
            budget -= alloc * h[i]
            if budget <= eps:
                break

        # any residual budget is spent on the cheapest positive coefficient
        if budget > eps:
            # first index in ordered list with positive h
            for i in order:
                if h[i] > eps:
                    g[i] += budget / h[i]   # may exceed 1000 – evaluator will clip
                    break

        # --------------------------------------------------------------
        # 4.  Form the new candidate f' = (1‑t)·f + t·g  and clip to [0,1000]
        # --------------------------------------------------------------
        new_f = (1.0 - t_mix) * f + t_mix * g
        new_f = np.clip(new_f, 0.0, 1000.0)

        # --------------------------------------------------------------
        # 5.  Fast bound evaluation; if promising, verify with the exact scorer
        # --------------------------------------------------------------
        fast_val = _fast_score(new_f, size)
        if fast_val + 1e-9 < best_score:                     # clear improvement
            exact_val = float(evaluate_sequence(new_f.tolist()))
            if np.isfinite(exact_val) and exact_val < best_score:
                best = new_f.copy()
                best_score = exact_val
                f = new_f.copy()          # continue search from the improved point
                continue

        # --------------------------------------------------------------
        # 6.  Occasionally perform a tiny random mass‑transfer move (exploration)
        # --------------------------------------------------------------
        if step % 20 == 0:
            i, j = rng.choice(n, size=2, replace=False)
            if f[i] > 0.0:
                delta = rng.random() * f[i] * 0.05          # move ≤5 % of source mass
                cand = f.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = np.clip(cand, 0.0, 1000.0)

                fast_val = _fast_score(cand, size)
                if fast_val + 1e-9 < best_score:
                    exact_val = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(exact_val) and exact_val < best_score:
                        best = cand.copy()
                        best_score = exact_val
                        f = cand.copy()

    # final output – plain Python list of floats
    return [float(v) for v in best]
