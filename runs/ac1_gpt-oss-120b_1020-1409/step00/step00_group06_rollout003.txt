import numpy as np

# Helper: produce a non‑negative descending version of an array
def descending_version(arr: np.ndarray) -> np.ndarray:
    """Return a copy of ``arr`` sorted in descending order and clipped to [0,1000]."""
    arr = np.clip(arr, 0.0, 1000.0)
    return np.sort(arr)[::-1]


def propose_candidate(seed: int = 42, **kwargs) -> list:
    """
    Generate a step‑function height sequence that (heuristically) minimizes
    the AC1 upper bound used by ``evaluate_sequence``.

    The routine starts from the provided ``height_sequence_1`` (if available),
    sorts it descending (which is known to reduce the maximum auto‑convolution),
    and then performs a cheap block‑perturbation local search while preserving
    the descending order.  A simple uniform fallback is kept as a safety net.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Load the parent construction if present, otherwise fall back.
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 2. Initial candidates
    # ----------------------------------------------------------------------
    # a) Descending version of the parent (the main heuristic)
    best = descending_version(base)

    # b) Very simple uniform fallback (guarantees a finite score)
    fallback = np.full_like(best, 1.0, dtype=np.float64)

    # ----------------------------------------------------------------------
    # 3. Evaluate both and keep the best so far.
    # ----------------------------------------------------------------------
    best_score = float(evaluate_sequence(best.tolist()))
    fallback_score = float(evaluate_sequence(fallback.tolist()))
    if np.isfinite(fallback_score) and fallback_score < best_score:
        best = fallback
        best_score = fallback_score

    n = best.size
    # Parameters for the cheap block perturbation
    min_width = max(1, n // 200)
    max_width = max(2, n // 12)
    sigma = 0.12                     # log‑normal spread of the multiplicative factor
    steps = 800                      # a modest number of evaluations (O(n log n) each)

    # ----------------------------------------------------------------------
    # 4. Local search: cheap block perturbations with re‑sorting.
    # ----------------------------------------------------------------------
    for _ in range(steps):
        width = int(rng.integers(min_width, max_width + 1))
        start = int(rng.integers(0, n - width + 1))
        stop = start + width

        cand = best.copy()
        factor = np.exp(rng.normal(0.0, sigma))
        cand[start:stop] *= factor

        # Occasionally blend with its mirror to keep the profile roughly symmetric
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])

        # Clip to the admissible interval and restore descending order
        cand = descending_version(cand)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ----------------------------------------------------------------------
    # 5. Return the best candidate found as a plain Python list.
    # ----------------------------------------------------------------------
    return best.tolist()
