Pseudo:

def compute_score(arr):
    # arr: 1D np.ndarray of floats, assumed nonnegative
    # Clip
    arr = np.clip(arr, 0.0, 1000.0)
    S = arr.sum()
    if S < 0.01:
        return np.inf
    n = arr.size
    L = 1 << (int(np.ceil(np.log2(2*n - 1))))
    # Use real FFT
    fft_arr = np.fft.rfft(arr, n=L)
    fft_conv = fft_arr * fft_arr
    conv = np.fft.irfft(fft_conv, n=L)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (S*S)

def compute_score(arr, fft_len, n):
    # assumes arr is 1-D np.ndarray
    a = np.clip(arr, 0.0, 1000.0)
    S = a.sum()
    if S < 0.01:
        return np.inf
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S*S)

import numpy as np

# -------------------------------------------------------------
# Helper: fast evaluation of the exact objective used by the
# harness (identical to evaluate_sequence but with FFT).
# -------------------------------------------------------------
def _fast_score(arr, fft_len, n):
    """Return the AC1 upper bound for a non‑negative array."""
    a = np.clip(arr, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # convolution via FFT (linear convolution)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (s * s))


# -------------------------------------------------------------
# Core search routine
# -------------------------------------------------------------
def propose_candidate(seed=42, num_steps=5000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    AC1 bound.  The routine works with the supplied
    `height_sequence_1` (length around 7 000) as a starting point.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Obtain the parent vector (the runtime provides height_sequence_1)
    # -----------------------------------------------------------------
    try:
        # height_sequence_1 is available in the execution environment
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                     # pragma: no cover
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    # Normalise to unit sum – the objective is homogeneous, any scale works.
    total = parent.sum()
    if total < 0.01:                       # pragma: no cover
        parent = np.ones(n, dtype=np.float64)
        total = parent.sum()
    cur = parent / total

    # Pre‑compute the FFT length (next power of two ≥ 2n‑1).
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))

    # -----------------------------------------------------------------
    # Initialise best solution
    # -----------------------------------------------------------------
    best = cur.copy()
    best_score = _fast_score(best, fft_len, n)

    # -----------------------------------------------------------------
    # Simple simulated‑annealing / hill‑climbing loop
    # -----------------------------------------------------------------
    # temperature controls acceptance of worsening moves
    temperature = best_score * 0.2
    sigma = 0.3                     # scale of multiplicative block perturbations

    for step in range(num_steps):
        cand = best.copy()

        # ---- random block scaling ------------------------------------
        if rng.random() < 0.7:
            # pick a block width between n/200 and n/12
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor
        else:
            # ---- small mass transfer between two positions -------------
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                move = cand[i] * rng.random() * 0.5
                cand[i] -= move
                cand[j] += move

        # optional optional symmetrisation (helps flattening)
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        # enforce bounds and renormalise
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 0.01:                       # pragma: no cover
            continue
        cand /= s

        # evaluate candidate
        cand_score = _fast_score(cand, fft_len, n)

        # keep the best ever seen
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # tighten the search as we improve
            sigma *= 0.97
            temperature = max(temperature * 0.97, 1e-6)

        # accept move for the current state (SA rule)
        cur_score = _fast_score(cur, fft_len, n)
        if cand_score <= cur_score:
            cur = cand
        else:
            # Metropolis criterion
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                cur = cand

    # -----------------------------------------------------------------
    # Final fallback – if for any reason we ended with an invalid
    # vector, revert to a uniform one (score = 2).
    # -----------------------------------------------------------------
    if not np.isfinite(best_score) or best_score > 2.5:   # pragma: no cover
        best = np.full(n, 1.0 / n, dtype=np.float64)

    # Return plain Python list as required by the harness
    return [float(x) for x in best]
