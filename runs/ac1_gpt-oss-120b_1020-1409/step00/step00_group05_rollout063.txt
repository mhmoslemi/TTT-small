def flatten_step(a, grad, step=0.01):
    # allocate small mass
    mass = step * a.sum()
    # pick candidates to reduce: indices with highest grad (top K)
    # pick indices to increase: lowest grad (bottom K)
    # number K = max(1, n // 1000) maybe.
    K = max(1, n // 500)  # e.g., 10 per thousand.
    # sort indices by grad
    idx_desc = np.argsort(-grad)  # descending
    idx_asc = np.argsort(grad)   # ascending
    # Determine amount to move per selected index proportionally
    # For each large index, reduce by reduction = min(a[idx]*0.5, mass / K) (some proportion)
    reduction_total = 0.0
    for i in idx_desc[:K]:
        dec = min(a[i] * 0.5, mass / K)
        a[i] -= dec
        reduction_total += dec
    # distribute equally to small indices
    inc = reduction_total / K
    for i in idx_asc[:K]:
        a[i] += inc
    # clip a to [0,1000]
    np.clip(a,0,1000,inplace=True)
    return a

def full_conv(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def propose_candidate(seed=42, num_iters=2000):
    rng = np.random.default_rng(seed)
    # initialize a from parent if valid else uniform
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        a = parent.copy()
        a = np.clip(a, 0.0, 1000.0)
    except Exception:
        a = np.ones(256, dtype=np.float64)
    
    n = a.shape[0]
    # Normalize sum to sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, target_sum / n)
    else:
        a *= target_sum / a_sum
    
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    for iteration in range(num_iters):
        # compute convolution via FFT
        conv = full_conv(a)
        # find index of max
        k_max = int(np.argmax(conv))
        # compute gradient contributions (2* partner)
        partner = k_max - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[partner[mask]]
        # Determine indices with high and low gradient
        # number to adjust
        num_adj = max(1, n // 500)   # about 0.2% of entries
        # indices sorted descending and ascending by grad
        idx_desc = np.argsort(-grad)
        idx_asc = np.argsort(grad)
        # compute total mass to move: a small fraction of sum
        move_frac = 0.02  # 2% of total sum each iteration
        move_mass = move_frac * a.sum()
        # distribute from high to low
        total_removed = 0.0
        # we will remove proportionally from high indices
        for idx in idx_desc[:num_adj]:
            # remove at most half of current value to keep nonneg
            max_removable = a[idx] * 0.5
            rem = min(max_removable, move_mass / num_adj)
            a[idx] -= rem
            total_removed += rem
        # distribute equally to low indices
        if total_removed > 0:
            inc = total_removed / num_adj
            for idx in idx_asc[:num_adj]:
                a[idx] += inc
        # ensure bounds and normalization
        np.clip(a, 0.0, 1000.0, out=a)
        # Renormalize sum to target_sum to keep scale invariant
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        # compute score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
        # occasional random perturbation to escape local minima
        if iteration % 100 == 0:
            # small random multiplicative jitter
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            a *= jitter
            np.clip(a, 0.0, 1000.0, out=a)
            # renorm
            s = a.sum()
            if s > 0:
                a *= target_sum / s
    
    return [float(x) for x in best]

import numpy as np

# ----------------------------------------------------------------------
# Helper: linear convolution via FFT (exact for non‑negative real data)
def _conv_max(arr: np.ndarray) -> float:
    """Return max value of full convolution arr*arr using FFT."""
    n = arr.shape[0]
    # size = next power of two >= 2n-1 for exact linear convolution
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _score(arr: np.ndarray, n: int) -> float:
    """Exact value of the evaluator (without calling evaluate_sequence)."""
    s = arr.sum()
    if s == 0.0:
        return float("inf")
    max_b = _conv_max(arr)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def _make_powerlaw(n: int, p: float, beta: float, sym: bool) -> np.ndarray:
    """Generate a non‑negative sequence a_i ∝ (i+β)^(-p) (symmetrised if requested)."""
    idx = np.arange(n, dtype=np.float64)
    base = (idx + beta) ** (-p)
    if sym:
        base = base + (idx[::-1] + beta) ** (-p)
    return base


# ----------------------------------------------------------------------
def _block_perturb(vec: np.ndarray, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Small stochastic local change, keeping the L1‑norm fixed."""
    cand = vec.copy()
    n = cand.size
    # random block width and location
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    cand[start : start + width] *= factor
    # optional mirroring (helps symmetry)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    # renormalise to the required L1‑norm
    s = cand.sum()
    if s > 0.0:
        cand *= target_sum / s
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Construct a non‑negative step function whose AC1 upper bound is
    substantially below the naive value 2.0.

    The routine
    * tries a family of power‑law shapes a_i ∝ (i+β)^(-p) (optionally symmetrised),
    * normalises each to sum = sqrt(2·n),
    * keeps the best one,
    * then performs a short stochastic local search around that best candidate.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain the length n (use the provided parent if valid, otherwise fall back)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        n = parent.size
    except Exception:
        n = 256
        parent = np.ones(n, dtype=np.float64)

    target_sum = np.sqrt(2.0 * n)          # the normalisation used by the evaluator

    # ------------------------------------------------------------------
    # Uniform vector (objective = 2.0) – serve as a fallback
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    best_vec = uniform.copy()
    best_score = _score(best_vec, n)

    # ------------------------------------------------------------------
