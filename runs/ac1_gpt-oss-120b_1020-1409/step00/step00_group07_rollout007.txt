def propose_candidate(seed=42, num_iters=30, t=0.2, frac_active=0.02, **kwargs):
    import numpy as np, scipy.optimize, scipy.sparse

    rng = np.random.default_rng(seed)

    # Initialize f
    try:
        f = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if f.ndim != 1 or f.size == 0 or not np.isfinite(f).all():
            raise Exception
    except Exception:
        n = 256
        f = np.full(n, 1.0)  # uniform 1
    n = f.size
    f = np.clip(f, 0.0, 1000.0)

    # Ensure not all zeros: if sum==0 set uniform 1
    if f.sum() < 1e-6:
        f = np.full(n, 1.0)

    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))

    for it in range(num_iters):
        # compute self convolution
        conv_f = full_convolution(f, f)
        M = conv_f.max()
        # find active shifts: top fraction
        k_vals = np.argpartition(-conv_f, int(frac_active * conv_f.size))[:int(frac_active * conv_f.size)]
        # ensure at least one shift
        if k_vals.size == 0:
            k_vals = np.array([np.argmax(conv_f)], dtype=int)
        # Build constraint matrix
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(k_vals):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            # corresponding f indices for g_i coefficient
            f_coeff = f[k - i_range]
            rows.extend([row_idx] * i_range.size)
            cols.extend(i_range.tolist())
            data.extend(f_coeff.tolist())
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(k_vals.size, n))
        b_ub = np.full(k_vals.size, M, dtype=np.float64)

        # Linear program: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)

        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                    bounds=[(0.0, None)] * n,
                                    method='highs', options={'presolve': True, 'time_limit': 5.0})
        if not res.success:
            # fallback: skip this iteration
            continue
        g = res.x
        # blend
        f_new = (1 - t) * f + t * g
        # clip
        f_new = np.clip(f_new, 0.0, 1000.0)
        # evaluate
        score = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best_f = f_new
            best_score = score
        # update f for next iteration
        f = f_new

    # final safety: if best_score still infinite, fallback to uniform
    if not np.isfinite(best_score):
        best_f = np.full(n, 1.0)
    return [float(x) for x in best_f]

def full_convolution(a, b):
    n = a.size + b.size - 1
    size = 1 << (n - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv
```

import scipy.sparse


def full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = a.size + b.size - 1
    # next power of two for efficient FFT
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv

    """Iteratively improve a non‑negative step sequence to lower the AC1 bound."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if isinstance(base, np.ndarray) and base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
        f = np.asarray(base, dtype=np.float64).copy()
    else:
        n = 256
        f = np.full(n, 1.0, dtype=np.float64)

    f = np.clip(f, 0.0, 1000.0)
    if f.sum() == 0.0:
        f[:] = 1.0

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------
    # Main optimisation loop: solve a linear programme that
    # increases the total mass while obeying the active convolution
    # constraints, then blend the result back into f.
    # --------------------------------------------------------------
    for _ in range(num_iters):
        conv_f = full_convolution(f, f)
        M = conv_f.max()

        # pick a modest set of shifts where the bound is tight
        num_active = max(1, int(frac_active * conv_f.size))
        idx = np.argpartition(-conv_f, num_active - 1)[:num_active]
        active_shifts = np.sort(idx)

        for row_idx, k in enumerate(active_shifts):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)

        res = scipy.optimize.linprog(
