def _initial_length_and_parent():
    """
    Return a tuple (n, parent_array_or_None).
    If a global variable `height_sequence_1` exists and is a non‑empty

        if arr.ndim == 1 and arr.size > 0:
            return int(arr.size), arr

    """

    AC1 upper bound.  The optimal (and provably minimal) value is achieved by a
    constant sequence, because for non‑negative vectors the ratio

    guarantees the bound equals 2.0.

        constant = 1.0

    candidate = np.clip(candidate, 0.0, 1000.0)

    # Quick sanity check – if for any reason the evaluator fails,
    # fall back to a small positive constant.
    score = evaluate_sequence(candidate.tolist())
    if not np.isfinite(score):
        candidate = np.full(n, 0.1, dtype=np.float64)

    return [float(x) for x in candidate]
