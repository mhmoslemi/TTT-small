def _evaluate(seq):

    return float(evaluate_sequence(seq.tolist()))


def _block_perturbation(base, rng):
    """Simple O(n) non‑negative perturbation used as a fallback."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))

    if rng.random() < 0.35:

    return np.clip(cand, 0.0, 1000.0)


def _solve_lp(f, max_b, idxs):
    """Linear program: maximise Σg under f∗g ≤ max_b at chosen indices."""
    if not _HAS_SCIPY or len(idxs) == 0:
        return None

    n = f.shape[0]
    K = len(idxs)

    data = []
    rows = []
    cols = []
    for row, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            data.append(f[i])
            rows.append(row)
            cols.append(j)

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))
    c = -np.ones(n)                     # maximise Σg  ⇔ minimise –Σg
    bounds = [(0.0, 1000.0)] * n        # respect the global height limits

            b_ub=np.full(K, max_b),
            bounds=bounds,
            method='highs',

        )

            return res.x
    except Exception:  # pragma: no cover
        pass
    return None

    Starts from the supplied parent construction (if present) and repeatedly
    tries to increase the L1‑mass while keeping the current maximum of the
    self‑convolution bounded.  A cheap block‑perturbation fallback guarantees
    a valid return even when SciPy is unavailable.
    """

        cand_score = _evaluate(cand)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            cur = cand.copy()

    return [float(v) for v in best]
