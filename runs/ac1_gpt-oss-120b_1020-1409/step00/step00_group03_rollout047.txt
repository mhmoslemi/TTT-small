def _build_sparse_candidate(n, rng, target_m=180, max_mult=4):
    pos = []
    sum_counts = defaultdict(int)
    attempts = 0
    max_attempts = n * 20
    while len(pos) < target_m and attempts < max_attempts:
        cand = int(rng.integers(1, n-1))  # avoid edges

            attempts += 1
            continue

        ok = True
        s_self = cand + cand
        if sum_counts[s_self] + 1 > max_mult:

        if ok:

                if sum_counts[s] + 1 > max_mult:
                    ok = False
                    break
        if ok:
            # accept
            pos.append(cand)
            sum_counts[s_self] += 1
            for p in pos[:-1]:
                sum_counts[p + cand] += 1
        attempts += 1
    arr = np.zeros(n, dtype=np.float64)
    for p in pos:
        arr[p] = 1.0
    return arr

def propose_candidate(seed=42, num_steps=250, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256
    # Build heuristic sparse candidate

    # Evaluate
    best_arr = candidate
    best_score = float(evaluate_sequence(best_arr.tolist()))
    # fallback to provided parent if better
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.isfinite(parent).all():
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score) and parent_score < best_score:
                best_arr = parent
                best_score = parent_score
    # Simple local improvement: swap a 1 and a 0
    for _ in range(num_steps):
        idx_one = np.where(best_arr > 0)[0]
        idx_zero = np.where(best_arr == 0)[0]
        if idx_one.size == 0 or idx_zero.size == 0:
            break
        i_remove = int(rng.choice(idx_one))
        i_add = int(rng.choice(idx_zero))
        new_arr = best_arr.copy()
        new_arr[i_remove] = 0.0
        new_arr[i_add] = 1.0
        new_score = float(evaluate_sequence(new_arr.tolist()))
        if np.isfinite(new_score) and new_score < best_score:
            best_arr = new_arr
            best_score = new_score
    # Ensure non-empty and finite
    if best_arr.size == 0 or not np.isfinite(best_score):

import numpy as np
from collections import defaultdict

def _build_sparse_candidate(n, rng, target_m=180, max_mult=4):
    """
    Construct a length‑n non‑negative vector with a limited number of
    one‑valued spikes. The construction guarantees that for every possible
    convolution index k the number of unordered pairs (i,j) with i+j=k does not
    exceed ``max_mult``. This bounds the maximum convolution entry and hence
    the AC1 upper bound.
    """
    positions = []                     # chosen indices
    sum_counts = defaultdict(int)      # map k=i+j -> number of unordered pairs
    attempts = 0
    max_attempts = n * 20              # generous guard against endless loops

    while len(positions) < target_m and attempts < max_attempts:
        cand = int(rng.integers(1, n - 1))   # avoid the two edges (k=0,2n‑2)
        if cand in positions:
            attempts += 1
            continue

        # check whether adding cand would violate the multiplicity bound
        ok = True
        s_self = cand + cand
        if sum_counts[s_self] + 1 > max_mult:
            ok = False
        if ok:
            for p in positions:
                s = p + cand
                if sum_counts[s] + 1 > max_mult:
                    ok = False
                    break

        if ok:
            # accept candidate
            positions.append(cand)
            sum_counts[s_self] += 1
            for p in positions[:-1]:
                sum_counts[p + cand] += 1
        attempts += 1

    # build the actual sequence
    arr = np.zeros(n, dtype=np.float64)
    for p in positions:
        arr[p] = 1.0
    return arr
