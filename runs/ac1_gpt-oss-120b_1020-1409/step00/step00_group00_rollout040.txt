import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a step‑function height sequence that minimizes the AC1 upper bound.
    The strategy builds on the analytic form a_i ~= 1/sqrt(i) (which yields a
    near‑flat self‑convolution) and adds a tiny uniform offset to increase the
    total sum without inflating the maximal convolution term.
    A small amount of local random mass transfer is then applied to polish the
    result.
    """
    # ----------------------------------------------------------------------
    # Determine the problem size (length of the sequence)
    # ----------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
            raise Exception
        n = base_seq.size
    except Exception:
        # fallback size if the parent array is unavailable
        n = 256

    # ----------------------------------------------------------------------
    # Core analytic candidate: a_i = 1 / sqrt(i+1)  (+ constant offset)
    # ----------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)          # i = 1 … n
    base = 1.0 / np.sqrt(idx)                           # decreasing power‑law

    # Initialise the best candidate with the pure power‑law
    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ----------------------------------------------------------------------
    # Simple baseline (uniform) – always a safe fallback
    # ----------------------------------------------------------------------
    uniform = np.ones(n, dtype=np.float64)
    score_uniform = float(evaluate_sequence(uniform.tolist()))
    if score_uniform < best_score:
        best_seq = uniform
        best_score = score_uniform

    # ----------------------------------------------------------------------
    # Scan a small grid of constant offsets (delta) added to the power‑law.
    # Adding a tiny uniform component raises the total sum while only mildly
    # increasing the maximal convolution term, thus lowering the final ratio.
    # ----------------------------------------------------------------------
    deltas = np.linspace(5e-04, 3.5e-03, 31)  # 31 values from 0.0005 to 0.0035
    for delta in deltas:
        cand = base + delta
        # evaluate original orientation
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
        # evaluate the reversed orientation (might give a slightly different
        # convolution profile)
        cand_rev = cand[::-1]
        score_rev = float(evaluate_sequence(cand_rev.tolist()))
        if np.isfinite(score_rev) and score_rev < best_score:
            best_seq = cand_rev
            best_score = score_rev

    # ----------------------------------------------------------------------
    # Light local optimisation: move a small random fraction of mass from one
    # index to another.  The total sum is preserved, and the move is accepted
    # only if it improves the score.
    # ----------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    max_iter = 500
    for _ in range(max_iter):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        src_val = best_seq[i]
        if src_val <= 0.0:
            continue
        # transfer up to 20 % of the source value
        frac = rng.random() * 0.2
        delta = src_val * frac
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # enforce the problem’s bound constraints
        if cand[i] < 0.0:
            continue
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score

    # Return a plain Python list of floats, as required by the harness.
    return [float(x) for x in best_seq]
