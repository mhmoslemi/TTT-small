    return np.exp(-lam * np.arange(length, dtype=np.float64))

    """a_i = exp(-lam * (i-centre)^2)"""
    return np.exp(-lam * distances * distances)


# ----- main proposal -------------------------------------------------------
def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Search for a non‑negative step‑function with a low AC1 upper bound.
    The routine first scans a few analytically‑motivated families
    (symmetric exponential, right‑hand exponential and a Gaussian)
    and then refines the best result with cheap block perturbations.
    """
    rng = np.random.default_rng(seed)

    # ---- obtain a parent sequence (height_sequence_1 if present) ----------
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is None:
        n = 256
        parent = np.ones(n, dtype=np.float64)
    else:
        n = parent.size

    parent = np.clip(parent, 0.0, 1000.0)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ---- pre‑compute distances for symmetric families --------------------
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    distances = np.abs(idx - centre)

    # ---- scan analytical families ---------------------------------------
    # decay rates (lam) from ~5e‑5 to 1e‑3 (log‑spaced for coverage)
    lam_vals = np.logspace(np.log10(5e-5), np.log10(1e-3), num=40)

    for lam in lam_vals:
        # symmetric exponential
        cand = generate_symmetric_exp(distances, lam)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

        # right‑hand exponential
        cand = generate_right_exp(n, lam)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

        # Gaussian (uses the same distances)
        cand = generate_gaussian(distances, lam)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = cand, score

    # ---- cheap local refinement -----------------------------------------
    for _ in range(num_steps):
        candidate = best.copy()

        # pick a modest block and perturb multiplicatively
        start = int(rng.integers(0, n))
        width = int(rng.integers(1, max(2, n // 200)))  # O(n)‑bounded width
        stop = min(n, start + width)

        factor = float(np.exp(rng.normal(0.0, 0.03)))  # small log‑normal tweak
        candidate[start:stop] *= factor

        # optional gentle symmetrisation (helps if the current best is symmetric)
        if rng.random() < 0.15:
            candidate = 0.5 * (candidate + candidate[::-1])

        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = candidate, score

    # ---- final safety fallback -------------------------------------------
    if not np.isfinite(best_score) or best_score == float('inf'):
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
