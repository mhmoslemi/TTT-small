import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Construct a non‑negative step sequence that aims to minimize the
    AC1 upper bound used by `evaluate_sequence`.

    The method builds a set of indices whose pairwise sums never exceed a
    prescribed limit C (maximum number of contributions to any convolution
    entry).  With equal unit weights the convolution entry values are exactly the
    number of contributions, so the bound becomes 2·n·C / |P|² where |P| is the
    number of chosen positions.  By trying several limits C and several random
    orderings we keep the best (lowest) score found.
    """
    rng = np.random.default_rng(seed)

    # Obtain the parent length; fall back to a modest size if unavailable.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:                 # pragma: no cover
        n = 256

    best_seq = None
    best_score = float('inf')

    # Upper bound on allowed contributions per convolution index.
    max_C = 6               # try a few values, cheap enough.
    attempts_per_C = 8      # different random orderings per C.

    for C in range(1, max_C + 1):
        for _ in range(attempts_per_C):
            # Greedy construction of a set with sum‑multiplicity ≤ C.
            indices = np.arange(n, dtype=np.int32)
            rng.shuffle(indices)

            collisions = np.zeros(2 * n - 1, dtype=np.int16)
            positions = []                 # chosen indices, kept in list order

            for i in indices:
                # Fast reject based on the square term i+i.
                if collisions[i + i] + 1 > C:
                    continue

                # Check all existing positions.
                feasible = True
                for j in positions:
                    s = i + j
                    if collisions[s] + 1 > C:
                        feasible = False
                        break
                if not feasible:
                    continue

                # Accept this index.
                positions.append(i)
                collisions[i + i] += 1
                # Update pairwise sums (i with each older element).
                for j in positions[:-1]:
                    collisions[i + j] += 1

            if not positions:                 # nothing built – try next attempt
                continue

            # Build the candidate sequence (unit weight on each chosen index).
            seq = np.zeros(n, dtype=np.float64)
            seq[positions] = 1.0

            # Exact evaluation (the authoritative function is imported by the harness).
            score = evaluate_sequence(seq.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.copy()

    # Fallback: uniform sequence if all attempts failed (should not happen).
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Return as a plain Python list of floats.
    return [float(v) for v in best_seq]
