def compute_full_convolution(a):
    n = a.size
    L = 1 << ((2*n - 1).bit_length())
    a_pad = np.zeros(L, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    conv = np.fft.irfft(A * A, n=L)[:2*n-1]
    return conv

def compute_gradient(a, w):
    n=a.size
    L = 1 << ((3*n - 2).bit_length())
    a_rev = a[::-1]
    ar_pad = np.zeros(L, dtype=np.float64)
    ar_pad[:n] = a_rev
    w_pad = np.zeros(L, dtype=np.float64)
    w_pad[:len(w)] = w
    AR = np.fft.rfft(ar_pad)
    WR = np.fft.rfft(w_pad)
    conv_rev_w = np.fft.irfft(AR * WR, n=L)[:3*n-2]
    grad = 2 * conv_rev_w[n-1:n-1+n]  # length n
    return grad

def propose_candidate(seed=42, num_steps=500, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        n = int(height_sequence_1.shape[0])
    except NameError:
        n = 256
    indices = np.arange(1, n+1, dtype=np.float64)
    a = np.power(indices, -0.5)
    a = a / a.sum()
    # Keep best
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Setup parameters
    beta = 5.0
    step = 0.5
    # Precompute FFT length
    L = 1 << ((3*n - 2).bit_length())
    for iter_idx in range(num_steps):
        # Convolution a*a
        a_pad = np.zeros(L, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        max_val = conv.max()
        # smoothing weights
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        a_rev = a[::-1]
        ar_pad = np.zeros(L, dtype=np.float64)
        ar_pad[:n] = a_rev
        w_pad = np.zeros(L, dtype=np.float64)
        w_pad[:len(w)] = w
        AR = np.fft.rfft(ar_pad)
        WR = np.fft.rfft(w_pad)
        conv_rev_w = np.fft.irfft(AR * WR, n=L)[:3*n-2]
        grad = 2.0 * conv_rev_w[n-1:n-1+n]
        # multiplicative update
        a = a * np.exp(-step * grad)
        # occasional random noise
        if rng.random() < 0.05:
            a = a + rng.normal(scale=0.01, size=n)
        a = np.clip(a, 0.0, None)
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.ones(n, dtype=np.float64) / n
            sum_a = a.sum()
        a = a / sum_a
        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
        # adjust parameters
        if (iter_idx + 1) % 100 == 0:
            beta = min(beta * 1.5, 30.0)
            step = step * 0.8
        # early stop if good enough
        if best_score < 1.50275:
            break
    return [float(x) for x in best]

import numpy as np

def _score_of(seq, n, L):
    """
    Compute the AC1 upper‑bound score for a 1‑D NumPy array ``seq``.
    The array may be un‑normalised – the formula is homogeneous.
    """
    a = np.asarray(seq, dtype=np.float64)
    a_pad = np.zeros(L, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(a.sum())
    if s == 0.0:
        return np.inf
    return 2.0 * n * max_b / (s * s)

    """
    Generate a sequence of non‑negative coefficients that minimises the
    AC1 upper‑bound.  The routine builds a 1/√‑law initial shape, tries a
    number of random permutations, and finally refines the best candidate
    with a few gradient‑descent steps performed in the Fourier domain.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

    except NameError:

    idx = np.arange(1, n + 1, dtype=np.float64)          # i = 1 … n
    base = np.power(idx, -0.5)                          # 1/√i
    base = base / base.sum()                            # normalised, sum = 1

    # ------------------------------------------------------------------
    # 2. Prepare buffers for FFT‑based convolutions
    # ------------------------------------------------------------------
    L_conv = 1 << ((2 * n - 1).bit_length())            # ≥ 2n‑1

    best = base.copy()

    for _ in range(num_trials):
        perm = rng.permutation(n)
        cand = base[perm]
        sc = _score_of(cand, n, L_conv)
        if sc < best_score:
            best = cand.copy()

    # ------------------------------------------------------------------
    # 5. Gradient‑descent refinement (smooth max‑approximation)
    # ------------------------------------------------------------------
    beta = 5.0          # controls smoothness of the max‑approximation
    step = 0.5          # multiplicative update strength

    for _ in range(num_grad_steps):
        a = best
        # ----- convolution a*a (FFT) -----
        a_pad = np.zeros(L_conv, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        w = np.exp(beta * (conv - mx))
