def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Setup parent if available
    parent = np.asarray(eval('height_sequence_1') if 'height_sequence_1' in globals() else None, dtype=np.float64)
    maybe_valid = (parent.ndim == 1 and parent.size >0 and np.isfinite(parent).all())
    if maybe_valid:
        parent_seq = np.clip(parent, 0.0, 1000.0)
        parent_score = float(evaluate_sequence(parent_seq.tolist()))
    else:
        parent_seq = None
        parent_score = float('inf')
    # constant candidate
    n = parent_seq.size if parent_seq is not None else 256
    const_seq = np.full(n, 1.0, dtype=np.float64)
    const_seq = np.clip(const_seq, 0.0, 1000.0)
    const_score = float(evaluate_sequence(const_seq.tolist()))
    # Choose best
    if const_score < parent_score:
        best = const_seq
    else:
        best = parent_seq if parent_seq is not None else const_seq
    return best.tolist()
