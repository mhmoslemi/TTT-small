import numpy as np

def make_block_perturbation(base, rng):
    """Return a non‑negative O(n) perturbation of ``base``."""
    candidate = np.array(base, copy=True)
    n = candidate.size
    # Choose a block width between 1% and 10% of n (inclusive)
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # Multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # Occasionally enforce symmetry (helps for some lengths)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # Clip to the allowed range
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Return a list of non‑negative step heights that (approximately) minimizes
    the AC1 upper bound evaluated by ``evaluate_sequence``.
    The routine uses a simple hill‑climbing search with O(n) memory.
    """
    rng = np.random.default_rng(seed)

    # Determine the problem size.
    try:
        # ``height_sequence_1`` is supplied by the harness when available.
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0:
            n = 256
        else:
            n = base_seq.size
    except Exception:
        n = 256

    # Initialise with a uniform vector – this attains the theoretical lower bound 2.
    best = np.full(n, 1.0, dtype=np.float64)  # any positive constant works
    best_score = float(evaluate_sequence(best.tolist()))

    # Perform a bounded number of stochastic perturbations.
    for _ in range(num_steps):
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        # Accept strictly better candidates.
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Sanity fallback – in the extremely unlikely event the above fails.
    if best_score is None or not np.isfinite(best_score):
        best = np.full(n, 1.0, dtype=np.float64)

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
