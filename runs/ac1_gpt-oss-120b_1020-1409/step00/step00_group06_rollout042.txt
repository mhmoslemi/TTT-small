def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length n

    # Ensure bounds

    # Function to compute score
    def score(arr):
        return float(evaluate_sequence(arr.tolist()))
    
    # Starting best as uniform fallback if needed later
    best_seq = base
    best_score = score(best_seq)

    # Evaluate if height_sequence_1 exists and maybe better
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        parent_score = score(parent)
        if np.isfinite(parent_score) and parent_score < best_score:
            best_seq = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    # Perform ternary search on epsilon additive offset
    # Upper bound maybe ~5.0 (since values are ~1, but we can safely use 5)
    eps_low = 0.0
    eps_high = 5.0

    # Function to evaluate epsilon candidate quickly
    def eval_eps(eps):
        cand = base + eps
        # clip not needed, values remain small (<1005)
        return score(cand)

    # Ternary search for minimum AC1 over epsilon
    for _ in range(30):
        e1 = eps_low + (eps_high - eps_low) / 3.0
        e2 = eps_high - (eps_high - eps_low) / 3.0
        s1 = eval_eps(e1)
        s2 = eval_eps(e2)
        if s1 < s2:
            eps_high = e2
        else:
            eps_low = e1

    eps_opt = (eps_low + eps_high) / 2.0
    cand_opt = base + eps_opt
    cand_opt = np.clip(cand_opt, 0.0, 1000.0)
    cand_score = score(cand_opt)
    if np.isfinite(cand_score) and cand_score < best_score:
        best_seq = cand_opt
        best_score = cand_score

    # Optional fine-grained random local search around best_seq
    # Use a small number of iterations to polish
    for _ in range(500):
        # Randomly pick a block width around 1% of n
        width = rng.integers(1, max(2, n // 100))
        start = rng.integers(0, n - width + 1)
        stop = start + width
        # Multiplicative perturbation
        factor = np.exp(rng.normal(0.0, 0.05, size=width))
        new_seq = best_seq.copy()
        new_seq[start:stop] *= factor
        # keep nonnegative, clip
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        new_score = score(new_seq)
        if np.isfinite(new_score) and new_score < best_score:

            best_score = new_score

    # Ensure final sequence respects constraints
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Return as list of floats
    return [float(x) for x in best_seq]

    # -------------------------------------------------
