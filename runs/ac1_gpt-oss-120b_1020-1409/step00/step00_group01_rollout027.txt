import numpy as np

def exponential_sequence(alpha, n):
    r = 1.0 - alpha
    # Using np.power with dtype float
    # Exponential may underflow for large exponent; but r close to 1 so not extreme
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx, dtype=np.float64)
    return seq

def approximate_score(alpha, n):
    r = 1.0 - alpha
    # compute values needed
    # compute r_pow_k for k=0..n-1 using geometric progression
    k = np.arange(n, dtype=np.float64)
    r_pow = np.power(r, k)
    # compute f = (k+1) * r_pow
    f = (k+1) * r_pow
    M = f.max()
    delta = 1.0 - np.power(r, n)
    # avoid division by zero
    if delta == 0:
        return np.inf
    return 2.0 * n * (alpha**2) * M / (delta*delta)

def make_block_perturbation(base, rng):
    candidate = np.array(base, copy=True)
    n = candidate.size
    # block width between n/200 and n/12, as before
    min_width = max(1, n // 200)
    max_width = max(2, n // 12)
    width = int(rng.integers(min_width, max_width+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.08))  # smaller sigma for fine tuning
    candidate[start:stop] *= factor
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # clip to allowed range
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a pure exponential decreasing sequence.
# ----------------------------------------------------------------------
def exponential_sequence(alpha, n):
    """Return a length‑n non‑negative sequence a_i = (1‑alpha)^i."""
    r = 1.0 - alpha
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx, dtype=np.float64)
    return seq

# ----------------------------------------------------------------------
# Helper: fast analytic surrogate for evaluate_sequence (exact for pure exponentials).
# ----------------------------------------------------------------------
def approximate_score(alpha, n, idx):
    """
    Compute 2*n*max(conv)/sum^2 analytically for the exponential shape.
    This avoids an O(n²) convolution.
    """
    r = 1.0 - alpha
    # r**k for k = 0..n‑1
    r_pow = np.power(r, idx, dtype=np.float64)
    # (k+1) * r^k  (overlap length on the left side of the convolution)
    f = (idx + 1.0) * r_pow
    M = f.max()                     # max_{k} L_k * r^k
    delta = 1.0 - np.power(r, n)    # 1 - r^n
    if delta <= 0.0:
        return np.inf
    return 2.0 * n * (alpha * alpha) * M / (delta * delta)

# ----------------------------------------------------------------------
# Helper: simple block‑wise multiplicative perturbation (memory‑bounded).
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    cand = np.array(base, copy=True)
    n = cand.size
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    w = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = start + w
    factor = np.exp(rng.normal(0.0, 0.08))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_alpha=500, perm_steps=80,
                     local_steps=500, **kwargs):
    """Return a list of non‑negative heights with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Use the provided parent if it exists; otherwise fall back.
    # ------------------------------------------------------------------
    try:
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 1) Scan a family of pure exponentials using the cheap surrogate.
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    alphas = np.geomspace(1e-6, 0.01, num=num_alpha)  # log‑space sweep
    best_alpha = None
    best_alpha_val = np.inf

    for a in alphas:
        val = approximate_score(a, n, idx)
        if val < best_alpha_val:
            best_alpha_val = val
            best_alpha = a

    if best_alpha is not None:
        cand = exponential_sequence(best_alpha, n)
        cand = np.clip(cand, 0.0, 1000.0)
        exact = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(exact) and exact < best_score:
            best = cand
            best_score = exact

    # ------------------------------------------------------------------
    # 2) Random permutations of the best exponential (same multiset of values).
    # ------------------------------------------------------------------
    for _ in range(perm_steps):
        perm = rng.permutation(n)
        cand = best[perm]
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # 3) Fine‑grained block perturbations (local search).
    # ------------------------------------------------------------------
    for _ in range(local_steps):
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
