import numpy as np

def _conv_max(arr: np.ndarray) -> float:
    """Return the maximum of the full self‑convolution of a nonnegative array."""
    n = arr.size
    # power‑of‑two length for efficient FFT
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(conv.max())


def compute_score(arr: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv)/sum(arr)^2 (scale‑invariant)."""
    s = arr.sum()
    if s == 0.0:
        return float('inf')
    n = arr.size
    max_b = _conv_max(arr)
    return 2.0 * n * max_b / (s * s)

            vi, vj = new_a[i], new_a[j]
            new_vi, new_vj = vi * factor, vj * factor
            removed = (vi - new_vi) + (vj - new_vj)
            new_a[i], new_a[j] = new_vi, new_vj
            # give the removed mass to a random other index
            k = rng.integers(0, nlen)
            while k == i or k == j:
                k = rng.integers(0, nlen)
            new_a[k] = np.clip(new_a[k] + removed, 0.0, 1000.0)

        # quick surrogate evaluation
        cand_score = compute_score(new_a)

        # accept if it improves the best (checked with the exact evaluator)
        if cand_score < best_score:
            exact = float(evaluate_sequence(new_a.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = new_a
                best_score = exact
                a = new_a
                continue

        # simulated‑annealing style uphill moves
        temp = max(0.001, 1.0 - step / num_steps)   # linear cooling
        cur_score = compute_score(a)
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / (temp + 1e-12)):
            a = new_a

    # final safety: if nothing valid was found, fall back to the parent
    if not np.isfinite(best_score):
        best = np.clip(parent, 0.0, 1000.0)

    return [float(x) for x in best.tolist()]
