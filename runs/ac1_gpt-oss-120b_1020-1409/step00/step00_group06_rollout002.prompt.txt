    Returns np.inf if the input is invalid.

    if not isinstance(sequence, list):

def compute_grad_finite(
    a: np.ndarray, 
    h: float = 1e-5
) -> np.ndarray:
    """Produce gradient for a via multi-point finite differences."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        # One-sided differentiation at boundaries
        if i == 0 or i == n-1:
            perturb = np.copy(a)
            perturb[i] += h
            score_plus = evaluate_sequence(perturb.tolist())
            score_orig = evaluate_sequence(a.tolist())
            grad[i] = (score_plus - score_orig) / h
        elif i == n//2:  # Special center point
            # Use two-point symmetric derivative
            perturb_plus = np.copy(a)
            perturb_plus[i] += h
            perturb_minus = np.copy(a)
            perturb_minus[i] -= h
            score_plus = evaluate_sequence(perturb_plus.tolist())
            score_minus = evaluate_sequence(perturb_minus.tolist())
            grad[i] = (score_plus - score_minus) / (2 * h)
        else:
            # Central derivative
            perturb_plus = np.copy(a)
            perturb_plus[i] += h
            perturb_minus = np.copy(a)
            perturb_minus[i] -= h
            score_plus = evaluate_sequence(perturb_plus.tolist())
            score_minus = evaluate_sequence(perturb_minus.tolist())
            grad[i] = (score_plus - score_minus) / (2 * h)
    return grad
