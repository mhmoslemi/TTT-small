import numpy as np

def _linear_ramp(n: int) -> np.ndarray:
    """Return a non‑negative step function: a linear ramp from 0 to 2."""
    # Values lie in [0, 2] → safely inside the [0,1000] bound.
    return np.linspace(0.0, 2.0, n, dtype=np.float64)


def _safe_parent() -> np.ndarray:
    """Return a sanitized copy of the supplied parent if it exists."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        # No parent supplied – fall back to a neutral constant vector.
        return np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        return np.ones(256, dtype=np.float64)

    # Enforce the problem’s hard bounds.
    return np.clip(parent, 0.0, 1000.0)


def propose_candidate(seed=42, **kwargs):
    """
    Produce a non‑negative step‑function sequence whose AC1 upper bound
    is (4*(n-2))/(3*(n-1)) ≈ 1.333 for any length n≥3.
    The function also compares this construction with the provided parent
    (if any) and returns the better of the two.
    """
    rng = np.random.default_rng(seed)  # kept for possible extensions
    n = height_sequence_1.shape[0] if 'height_sequence_1' in globals() else 256

    # Primary construction: linear ramp 0 → 2.
    linear = _linear_ramp(n)

    # Parent fallback (sanitized).
    parent = _safe_parent()
    if parent.size != n:
        # If the parent length differs, truncate/pad it to match n.
        if parent.size > n:
            parent = parent[:n]
        else:
            pad = np.full(n - parent.size, 0.0, dtype=np.float64)
            parent = np.concatenate([parent, pad])

    # Evaluate both candidates and keep the best.
    best_seq = None
    best_score = float('inf')
    for cand in (linear, parent):
        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = float(score)

    # In the unlikely event both are invalid, fall back to a uniform vector.
    if best_seq is None:
        best_seq = np.full(n, 1.0, dtype=np.float64)

    return [float(x) for x in best_seq]
