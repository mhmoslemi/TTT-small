import numpy as np

def _arcsine_sequence(length: int) -> np.ndarray:
    """Generate a non‑negative step function whose self‑convolution is nearly flat.

    The continuous arcsine density f(x)=1/π√(x(1‑x)) has the property that
    f*f is uniform on [0,2].  Discretising the reciprocal √(x(1‑x)) yields a
    sequence that makes the discrete convolution close to constant, thus
    minimizing the AC1 bound.
    """
    # Mid‑point discretisation to avoid the singular end‑points.
    idx = np.arange(length, dtype=np.float64)
    x = (idx + 0.5) / length                # x ∈ (0,1)
    a = 1.0 / np.sqrt(x * (1.0 - x))        # arcsine shape
    return np.clip(a, 0.0, 1000.0)           # safety clamp

    try:
        score = float(evaluate_sequence(candidate.tolist()))
        if not np.isfinite(score):
            raise ValueError
    except Exception:
        # Uniform fallback – always valid and yields AC1 = 2.
        candidate = np.ones(n, dtype=np.float64)

    # Return a plain Python list of floats (the harness expects this form).
    return [float(v) for v in candidate]
