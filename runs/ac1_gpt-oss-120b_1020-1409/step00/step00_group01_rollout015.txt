import numpy as np

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """
    Generate a non‑negative step‑function sequence aiming to minimize
    the AC1 upper bound used by `evaluate_sequence`.

    The algorithm starts from the available `height_sequence_1` (or a
    uniform fallback), creates an anti‑symmetric ordering that tends to
    reduce the peak of the self‑convolution, and then performs a bounded
    local search with cheap perturbations (block scaling, smoothing,
    shuffling, and random re‑ordering).  All operations use O(n) memory.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain initial parent sequence (may be provided by the harness)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    n = base.size
    if n == 0:
        return [0.0]

    # ------------------------------------------------------------------
    # Initial candidates: the raw parent and an anti‑symmetric ordering
    # ------------------------------------------------------------------
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # anti‑symmetric arrangement (small values paired with large ones)
    sorted_vals = np.sort(base)
    cand = np.empty_like(sorted_vals)
    half = n // 2
    cand[:half] = sorted_vals[:half]                     # left part: small → large
    cand[-half:] = sorted_vals[-half:][::-1]             # right part: large → small
    if n % 2:
        cand[half] = sorted_vals[half]                  # middle element (if any)
    cand = np.clip(cand, 0.0, 1000.0)
    cand_score = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(cand_score) and cand_score < best_score:
        best = cand
        best_score = cand_score

    # ------------------------------------------------------------------
    # Local search utilities
    # ------------------------------------------------------------------
    sigma = 0.30            # multiplicative log‑scale noise magnitude
    decay = 0.9995          # exponential decay of sigma
    smooth_kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    # ------------------------------------------------------------------
    # Main iterative improvement loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        r = rng.random()
        # ------------------------------------------------------------------
        # 1) Block multiplicative perturbation (most common)
        # ------------------------------------------------------------------
        if r < 0.60:
            candidate = best.copy()
            low = max(1, n // 200)
            high = max(low + 1, n // 12 + 1)
            width = int(rng.integers(low, high))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, sigma))
            candidate[start:start + width] *= factor
            # occasional small circular shift
            if rng.random() < 0.10:
                shift = int(rng.integers(-n // 4, n // 4 + 1))
                candidate = np.roll(candidate, shift)
            candidate = np.clip(candidate, 0.0, 1000.0)

        # ------------------------------------------------------------------
        # 2) Local smoothing (low‑pass filter)
        # ------------------------------------------------------------------
        elif r < 0.80:
            candidate = np.convolve(best, smooth_kernel, mode='same')
            # Re‑normalise to keep the total sum comparable
            s_orig = best.sum()
            s_new = candidate.sum()
            if s_new > 0:
                candidate *= s_orig / s_new
            candidate = np.clip(candidate, 0.0, 1000.0)

        # ------------------------------------------------------------------
        # 3) Shuffle current best (breaks accidental symmetry)
        # ------------------------------------------------------------------
        elif r < 0.95:
            perm = rng.permutation(n)
            candidate = best[perm]

        # ------------------------------------------------------------------
        # 4) Random re‑ordering of the original parent (exploratory)
        # ------------------------------------------------------------------
        else:
            perm = rng.permutation(base.size)
            candidate = base[perm]

        # ------------------------------------------------------------------
        # Evaluate and accept if improvement
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

        # ------------------------------------------------------------------
        # Annealing of sigma and early stopping
        # ------------------------------------------------------------------
        sigma = max(sigma * decay, 0.02)
        if best_score < 1.6:    # generous bound to stop early
            break

    # ------------------------------------------------------------------
    # Return the best found sequence as a plain Python list of floats
    # ------------------------------------------------------------------
    return best.tolist()
