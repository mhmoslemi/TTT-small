def propose_candidate(seed=42):
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1) if exists else 256
    best_arr = np.clip(height_sequence_1, 0, 1000).astype(np.float64) # fallback
    
    best_score = float('inf')
    if best_arr.size > 0:
        best_score = float(evaluate_sequence(best_arr.tolist()))
        best = best_arr.copy()
    else:
        best = np.ones(n, dtype=np.float64) * 1000.0
        best_score = float(evaluate_sequence(best.tolist()))
    
    # generate exponential candidates
    for c_param in np.linspace(0.5, 2.5, 20):
        delta = c_param / n
        r = max(0.0, 1.0 - delta)
        cand = 1000.0 * (r ** np.arange(n))
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # double exponential random samples
    for _ in range(100):
        m_frac = rng.uniform(0.1, 0.9)
        m = int(m_frac * n)
        r1 = 1.0 - rng.uniform(0.0001, 0.001) # close to 1
        r2 = 1.0 - rng.uniform(0.005, 0.02)   # faster decay
        base = 1000.0
        pref = base * (r1 ** np.arange(m))
        suffix = base * (r1 ** m) * (r2 ** np.arange(n - m))
        cand = np.concatenate([pref, suffix])
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # local search
    for _ in range(200):
        # create block perturbation as previous but on best
        width = max(1, int(rng.integers(max(1, n // 200), max(2, n // 20))))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        cand = best.copy()
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.05)))  # smaller magnitude
        # optional symmetry toggle
        if rng.random() < 0.2:
            cand = 0.5 * (cand + cand[::-1])
        # clip
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(x) for x in best]

import numpy as np

def _make_sidon_set(n, k_target, rng):
    """Greedy construction of a Sidon set of size up to k_target.
    Returns a list of indices.  All pairwise sums are distinct."""
    # shuffle candidate indices
    candidates = np.arange(n, dtype=np.int64)
    rng.shuffle(candidates)
    selected = []
    used_sums = set()
    for idx in candidates:
        # check whether adding idx would duplicate a sum
        conflict = False
        for s in selected:
            if (idx + s) in used_sums:
                conflict = True
                break
        if conflict or (idx + idx) in used_sums:
            continue
        # accept this index
        selected.append(idx)
        # record its sums with all previously selected (including itself)
        for s in selected:
            used_sums.add(idx + s)
        if len(selected) >= k_target:
            break
    return selected


def propose_candidate(seed=42, **kwargs):
    """Construct a sequence that aims for a low AC1 upper bound.
    The strategy places a Sidon set of large values (height 1000) and fills
    the remaining positions with a modest constant height.  The Sidon
    property limits the number of large–large products that can appear in
    any convolution entry, keeping the max convolution small while the
    overall sum stays large."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Base sequence (fallback)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the harness
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = base.size
    except Exception:  # pragma: no cover
        n = 256
        base = np.full(n, 1000.0, dtype=np.float64)

    # initial best
    best = base
    best_score = float('inf')
    try:
        sc = float(evaluate_sequence(best.tolist()))
        if np.isfinite(sc):
            best_score = sc
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # Parameters for the Sidon‑large / small pattern
    # ------------------------------------------------------------------
    max_k = int(np.sqrt(n))            # theoretical Sidon size bound
    small_vals = [4, 6, 8, 10, 12, 14, 16, 18, 20]

    # Try various sizes of the large‑value Sidon set
    for k in range(max_k, max_k - 5, -1):
        if k <= 0:
            continue
        for attempt in range(3):
            large_idx = _make_sidon_set(n, k, rng)
            if len(large_idx) < k:
                continue    # failed to reach desired size, try again
            for sv in small_vals:
                cand = np.full(n, float(sv), dtype=np.float64)
                cand[large_idx] = 1000.0
                cand = np.clip(cand, 0.0, 1000.0)
                try:
                    sc = float(evaluate_sequence(cand.tolist()))
                except Exception:  # pragma: no cover
                    continue
                if np.isfinite(sc) and sc < best_score:
                    best = cand
                    best_score = sc
                    if best_score < 1.5028:   # target reached, early exit
                        return [float(x) for x in best]

    # ------------------------------------------------------------------
    # Light local refinement (block perturbations)
    # ------------------------------------------------------------------
    for _ in range(200):
        width = max(1, int(rng.integers(max(1, n // 200), max(2, n // 20))))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        cand = best.copy()
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.05)))
        cand = np.clip(cand, 0.0, 1000.0)
        try:
            sc = float(evaluate_sequence(cand.tolist()))
        except Exception:  # pragma: no cover
            continue
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            if best_score < 1.5028:
                break

    return [float(x) for x in best]
