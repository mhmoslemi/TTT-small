    if a.sum() < 0.01:
        a = np.ones(n)/n

    best = a.copy()

    i_idx = np.arange(n)
    # main loop
    for it in range(num_steps):
        # compute convolution
        c = np.convolve(a, a)
        t_max = int(np.argmax(c))

        # compute gradient
        j = t_max - i_idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # vectorized gather
        # Using fancy indexing; need to create temporary array of a[j[mask]]
        j_valid = j[mask]

        # step size schedule
        step = 0.5 / (1.0 + it / 1000.0)  # decreasing

        a = a - step * g
        # clip negatives to zero
        np.maximum(a, 0.0, out=a)

        if rng.random() < 0.05:
            noise = rng.normal(0.0, 0.001, size=n)
            a += noise
            np.maximum(a, 0.0, out=a)

        # renormalize to sum 1
        total = a.sum()
        if total > 0:

        else:
            a = np.ones(n) / n

        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:

            best = a.copy()

                pass

        # optional early stop after many no improvements

        # but skip for simplicity

    # fallback if best invalid

    # Convert to list

def _initial_vector(seed: int, rng: np.random.Generator, n: int) -> np.ndarray:
    """Return a non‑negative start vector of length ``n`` with sum == 1."""
    # Try to use the supplied parent construction if it exists.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size == n:
            vec = np.clip(parent, 0.0, 1000.0)
            total = vec.sum()

                return vec / total
    except Exception:

    # Random start – at least a tiny positive value everywhere.
    vec = rng.random(n) + 1e-6
    return vec / vec.sum()


def _gradient_step(a: np.ndarray, t_max: int, step: float, idx: np.ndarray) -> np.ndarray:
    """
    Perform a single projected gradient step aimed at reducing the dominant
    convolution bin `t_max`.  The gradient of the bin sum with respect to `a`
    is g_i = 2 * a_{t_max-i} (zero when the partner index is out of range).
    """
    n = a.size
    j = t_max - idx              # partner indices
    mask = (j >= 0) & (j < n)    # valid partners
    g = np.zeros_like(a)
    # vectorised copy of the partner values
    g[mask] = 2.0 * a[j[mask]]
    # descent and non‑negativity projection
    a_new = a - step * g
    np.maximum(a_new, 0.0, out=a_new)
    # renormalise to preserve the scale‑invariant objective
    total = a_new.sum()
    if total > 0.0:
        a_new /= total
    else:
        a_new.fill(1.0 / n)
    return a_new


def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """
    Search for a step‑function coefficient vector that minimizes the
    AC1 upper bound as defined by ``evaluate_sequence``.
    The algorithm uses a few thousand gradient‑type updates starting from a
    random (or parent‑derived) point and keeps the best candidate seen.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – derived from the provided parent if possible.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 256  # a safe fallback size

    idx = np.arange(n)                     # reused index array

    # Initialise a feasible vector (non‑negative, sum == 1).
    a = _initial_vector(seed, rng, n)

    # Remember the best solution found so far.
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Simple stagnation counter for early termination.
    stagnant = 0
    max_stagnant = 200

    for it in range(num_steps):
        # Full convolution and the index of the largest entry.
        conv = np.convolve(a, a)
        t_max = int(np.argmax(conv))

        # Adaptive step size – larger early, shrinking later.
        step = 0.5 / (1.0 + it / 1000.0)

        # Gradient‑type update that reduces the dominant bin.
        a = _gradient_step(a, t_max, step, idx)

        # Occasional random jitter to escape shallow plateaus.
        if rng.random() < 0.04:
            jitter = rng.normal(0.0, 0.001, size=n)
            a += jitter
            np.maximum(a, 0.0, out=a)
            a /= a.sum()

        # Evaluate the candidate.
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Early exit if no improvement for a while.
        if stagnant >= max_stagnant:
            break

    # Final safety net – ensure a non‑empty, finite result.
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best /= best.sum()

    # Return a plain Python list as required by the harness.
    return [float(v) for v in best]
