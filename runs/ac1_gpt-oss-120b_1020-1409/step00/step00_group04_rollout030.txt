def linear_convolution(a):
    n = a.shape[0]
    size = 1 << (2*n-1).bit_length()  # next power of 2 >= 2n-1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv
```

def linear_convolution(a):
    n = a.shape[0]
    size = int(2**np.ceil(np.log2(2*n-1)))  # next power of 2
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa*fa, n=size)[:2*n-1]
    return conv

import numpy as np

def _linear_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using FFT (O(n log n))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """Exact objective 2 * n * max(conv) / sum(a)^2."""
    n = a.shape[0]
    conv = _linear_convolution(a)
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
