def generate_sidon(n, attempts=10, rng=None):
    best_set = []
    for attempt in range(attempts):
        order = np.arange(n)
        rng.shuffle(order)
        used = np.zeros(2*n-1, dtype=bool)
        cur = []
        for i in order:
            # quickly test if any sum collides
            if used[i+i]:
                continue
            conflict = False
            for j in cur:
                if used[i+j]:
                    conflict = True
                    break
            if conflict:
                continue
            # Add i
            for j in cur:
                used[i+j] = True
                used[j+i] = True  # same index as i+j
            # set self-sum
            used[i+i] = True
            cur.append(i)
        if len(cur) > len(best_set):
            best_set = cur
    return np.array(best_set, dtype=int)

import numpy as np
from math import ceil, sqrt

# ------------------------------------------------------------
# Helper: generate a Sidon (B_2) set – all sums i+j (i<=j) are unique.
# ------------------------------------------------------------
def _sidon_set(n: int, rng: np.random.Generator, attempts: int = 200,
               target_len: int = None) -> np.ndarray:
    """
    Return indices of a large Sidon set within {0,…,n-1}.
    If target_len is given the search stops early when a set of that size
    is found.  The algorithm shuffles the candidate order and greedily adds
    points while keeping the uniqueness of all pairwise sums.
    """
    best = np.empty(0, dtype=int)
    best_len = 0
    max_sums = 2 * n - 1                     # possible values of i+j

    # theoretical maximum size m satisfying m(m+1)/2 <= 2n-1
    max_theory = int((sqrt(1 + 8 * max_sums) - 1) // 2)

    for _ in range(attempts):
        order = np.arange(n, dtype=int)
        rng.shuffle(order)

        used = np.zeros(max_sums, dtype=bool)   # used[s] == True if sum s already taken
        cur = []

        for i in order:
            # quick reject if i+i already used (collision with any earlier pair)
            if used[i + i]:
                continue
            # check collisions with existing elements
            coll = False
            for j in cur:
                if used[i + j]:
                    coll = True
                    break
            if coll:
                continue

            # add i to the set
            used[i + i] = True
            for j in cur:
                used[i + j] = True
            cur.append(i)

        cur_len = len(cur)
        if cur_len > best_len:
            best_len = cur_len
            best = np.array(cur, dtype=int)

        # early stop if we have reached a size that guarantees the target bound
        if target_len is not None and best_len >= target_len:
            break
        # early stop if we hit the theoretical maximum
        if best_len >= max_theory:
            break

    return best

# ------------------------------------------------------------
def propose_candidate(seed: int = 42, **kwargs):
    """Return a non‑negative step‑function (list) with a low AC1 bound."""
    rng = np.random.default_rng(seed)

    # Determine the problem length n.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = int(parent.size)
    except Exception:
        n = 256                     # safe fallback length

    # Target bound – we need a set large enough that 4n / m^2 <= 1.50274349
    target_score = 1.50274349
    target_m = int(ceil(sqrt(4 * n / target_score)))

    # Build a large Sidon set.
    indices = _sidon_set(n, rng, attempts=200, target_len=target_m)

    # Construct the binary candidate (1.0 at Sidon indices, 0 elsewhere).
    candidate = np.zeros(n, dtype=np.float64)
    if indices.size:
        candidate[indices] = 1.0

    # Safety: guarantee a non‑zero sum (should already hold).
    if candidate.sum() < 0.01:
        candidate[0] = 1.0

    # Evaluate the Sidon candidate.
    sid_score = float(evaluate_sequence(candidate.tolist()))

    # Fallback: uniform vector (guaranteed bound 2.0).
    uniform = np.ones(n, dtype=np.float64)
    uni_score = float(evaluate_sequence(uniform.tolist()))

    # Choose the better (lower) score.
    if np.isfinite(sid_score) and sid_score < uni_score:
        best = candidate
    else:
        best = uniform

    # Clip to the bounds enforced by the evaluator (extra safety).
    best = np.clip(best, 0.0, 1000.0)

    return [float(v) for v in best]
