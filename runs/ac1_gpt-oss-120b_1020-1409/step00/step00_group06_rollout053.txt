def propose_candidate(seed=42, **kwargs):
    """
    Search for a non‑negative step‑function whose AC1 upper bound
    (2·n·max(convolution)/sum²) is as low as possible.

    The search uses an exponential‑decay family a_i = exp(-α·i)
    and a tiny hill‑climbing on the parameter α.  It falls back to the
    provided `height_sequence_1` (or a constant vector) if nothing
    improves.
    """
    # ---- initial data -------------------------------------------------
    # `height_sequence_1` is supplied by the harness; use it as a fallback.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.shape[0]

    # ---- baseline (parent) --------------------------------------------
    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ---- helpers -------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)               # shared index array

    def eval_alpha(alpha):
        """Generate the exponential candidate and evaluate its score."""
        seq = np.exp(-alpha * idx)                     # a_i = e^{-α i}
        # all values are already non‑negative and << 1000
        score = float(evaluate_sequence(seq.tolist()))

    # ---- coarse grid ---------------------------------------------------
