import numpy as np

# Linear convolution via FFT (full, linear) – O(n log n) memory and time.
def _max_convolution(seq, pad_len):
    """Return max element of seq * seq (full convolution) using FFT."""
    # rfft pads automatically to length pad_len
    r = np.fft.rfft(seq, n=pad_len)
    conv = np.fft.irfft(r * r, n=pad_len)   # linear convolution in real domain
    # full convolution length is 2*len(seq)-1
    return conv[: 2 * len(seq) - 1].max()


def _fast_score(seq, n, pad_len):
    """Fast approximation of evaluate_sequence (same formula, no clipping)."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    max_b = _max_convolution(seq, pad_len)
    return float(2.0 * n * max_b / (sum_a * sum_a))
