def perturb_seq(seq, rng):
   # returns new candidate
   n = seq.size
   # Choose a random block size and factor
   block = rng.integers(max(1, n//200), max(2, n//12))
   start = rng.integers(0, n-block+1)
   factor = np.exp(rng.normal(0.0, 0.12))
   new = seq.copy()
   new[start:start+block] *= factor
   if rng.random() < 0.35:
       new = 0.5 * (new + new[::-1])
   np.clip(new, 0.0, 1000.0, out=new)

def _make_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative O(n) perturbation of ``base``."""
    n = base.size
    # choose a block size between n/200 and n/12 (at least 1)
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand = base.copy()
    cand[start:start + width] *= factor
    # occasional mirroring to keep some symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
