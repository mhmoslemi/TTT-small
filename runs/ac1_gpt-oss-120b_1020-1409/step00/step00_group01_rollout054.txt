    n = len(height_sequence_1) if defined else 256
    target_sum = math.sqrt(2 * n) # as per earlier scaling
    # start with random Dirichlet (like RNG gamma)
    a = rng.random(n)  # uniform [0,1)
    a = a / a.sum() * target_sum

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    lr = 0.1  # step size
    for it in range(num_iters):
        c = np.convolve(a, a)
        smax = int(np.argmax(c))
        # compute gradient g_i = a[smax - i] if index valid else 0
        indices = np.arange(n)
        pair_idx = smax - indices
        mask = (pair_idx >=0) & (pair_idx < n)
        g = np.zeros_like(a)
        g[mask] = a[pair_idx[mask]]
        # For i=j case (i == smax-i) where smax even and i = smax/2, product derivative should be 2*a_i, we can approximate by double:
        if smax % 2 == 0:

            if 0 <= i_mid < n:

        # Scale gradient to preserve sum: we want to move down entries with large g. Use direction d = -g (or - (g - avg_g)).

        d = -g
        # adjust step size to keep sum unchanged: after update a' = a + lr * d, sum changes by lr * sum(d) = -lr * sum(g).
        # To keep sum constant, we can compute offset = - (sum after step - target_sum) / n and add equal offset to each entry.

        a_new = a + lr * d

import math
import numpy as np

def propose_candidate(seed=42, num_iters=4000, **kwargs):
    """Iterative peak‑reduction search for a low AC1 bound.

    The algorithm repeatedly finds the shift with the largest convolution value,
    diminishes the two coefficients that realise this peak and redistributes the
    removed mass to positions that do not affect the current peak.
    The sequence is kept non‑negative, bounded by 1000 and normalised to the
    conventional sum sqrt(2 n) so that the returned score equals the AC1 bound.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise – use the provided parent if it exists and is valid.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        if parent.sum() < 0.01:
            # degenerate parent – fall back to uniform
            a = np.full(n, 1.0, dtype=np.float64)
        else:
            a = parent.copy()
    else:
        n = 256
        a = np.full(n, 1.0, dtype=np.float64)

    # Normalise the sum to sqrt(2 n) – the standard “energy” normalisation.
    target_sum = math.sqrt(2.0 * n)
    if a.sum() > 0.0:
        a *= (target_sum / a.sum())
    else:
        a = rng.random(n)
        a *= (target_sum / a.sum())

    # ------------------------------------------------------------------
    # Helper to compute the AC1 score (identical to evaluate_sequence).
    # ------------------------------------------------------------------
    def ac1_score(seq):
        conv = np.convolve(seq, seq)
        max_b = float(conv.max())
        s = float(seq.sum())
        if s < 0.01:
            return float('inf')
        return 2.0 * n * max_b / (s * s)

    best = a.copy()
    best_score = ac1_score(best)

    # ------------------------------------------------------------------
    # Main iterative improvement.
    # ------------------------------------------------------------------
    reduction_factor = 0.8          # how strongly we shrink the peak pair
    for _ in range(num_iters):
        # current convolution and the shift giving the worst value
        conv = np.convolve(a, a)
        smax = int(conv.argmax())

        # contributions of each index to that peak (a[i] * a[smax-i] when defined)
        idx = np.arange(n)
        partner = smax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=np.float64)
        i_valid = idx[mask]
        j_valid = partner[mask]
        contrib[i_valid] = a[i_valid] * a[j_valid]

        # pick the index with the largest contribution
        i_star = int(contrib.argmax())
        j_star = smax - i_star

        # shrink the two involved coefficients
        old_i = a[i_star]
        a[i_star] = old_i * reduction_factor
        removed = old_i - a[i_star]

        if 0 <= j_star < n and j_star != i_star:
            old_j = a[j_star]
            a[j_star] = old_j * reduction_factor
            removed += old_j - a[j_star]

        # ------------------------------------------------------------------
        # Redistribute the removed mass to positions that do *not* participate
        # in the current peak (their partner index is out of range).  This
        # preferentially moves weight away from the offending shift.
        # ------------------------------------------------------------------
        out_of_range = (smax - np.arange(n) < 0) | (smax - np.arange(n) >= n)
        candidates = np.where(out_of_range)[0]
        if candidates.size == 0:
            # all indices affect the peak – fall back to a random location
            candidates = np.arange(n)

        # distribute the whole removed amount in one go; clipping will keep
        # the values inside [0,1000].
        add_idx = rng.choice(candidates)
        a[add_idx] += removed

        # enforce the hard bounds and restore the desired total sum
        np.clip(a, 0.0, 1000.0, out=a)
        cur_sum = a.sum()
        if cur_sum > 0.0:
            a *= (target_sum / cur_sum)

        # occasional tiny jitter to escape tiny plateaus
        if rng.random() < 0.02:
            a *= np.exp(rng.normal(0.0, 0.02, size=n))
            np.clip(a, 0.0, 1000.0, out=a)
            a *= (target_sum / a.sum())

        # evaluate the new candidate
        score = ac1_score(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
            # early stop if we have already beaten the target bound
            if best_score < 1.50274349:
                break

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid non‑empty sequence.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score) or best_score > 2.0:
        best = np.full(n, target_sum / n, dtype=np.float64)

    return [float(x) for x in best]
