def compute_ac1(a):
    n = len(a)

    L = 1 << ((2*n -1).bit_length())

def fast_ac1(a):
    n = len(a)
    L = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.irfft(A*A, n=L)[:2*n-1]
    max_b = np.max(B)
    sum_a = np.sum(a)
    return 2.0 * n * max_b / (sum_a**2)

    n = len(height_sequence_1) if defined else 256

    num_steps = 5000

    for step in range(num_steps):
        # copy
        cand = current.copy()
        # select block
        w = rng.integers(1, max(2, n // 10))  # width
        start = rng.integers(0, n - w + 1)
        # apply multiplicative factor

        cand_score = fast_ac1(cand)

            current = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
        # cooling
        temperature *= 0.9995  # small cooling

    # fallback: if best_score not finite, use uniform
    return best.tolist()

import numpy as np

def fast_ac1(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.irfft(A*A, n=L)[:2*n-1]
    max_b = np.max(B)
    sum_a = np.sum(a)
    return 2.0 * n * max_b / (sum_a * sum_a)

import numpy as np

def _fast_ac1(a: np.ndarray) -> float:
    """Fast approximate AC1 score using FFT (same formula as the evaluator)."""
    n = a.shape[0]
    # Length for a linear convolution (next power of two for speed)
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    max_b = np.max(B)
    s = np.sum(a)
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, num_steps: int = 5000, **kwargs):
    """
    Search for a non‑negative step‑function that lowers the AC1 bound.
    Returns a list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise a starting sequence ------------------------------------------------
    try:
        # use the provided construction if it exists
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        a = np.clip(base, 0.0, 1000.0)
    except Exception:
        # fallback: a modest length, all ones (scale is irrelevant)
        n = 256
        a = np.ones(n, dtype=np.float64)

    n = a.shape[0]
    cur = a.copy()
    cur_score = _fast_ac1(cur)

    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # simulated‑annealing style search ----------------------------------------------
    temperature = 1.0
    cooling = 0.9995  # geometric cooling

    for _ in range(num_steps):
        cand = cur.copy()
        op = rng.integers(0, 4)

        if op == 0:
            # multiplicative block perturbation
            width = int(rng.integers(1, max(2, n // 20)))
            start = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.1 * temperature))
            cand[start:start + width] *= factor

        elif op == 1:
            # additive Gaussian noise on a block
            width = int(rng.integers(1, max(2, n // 20)))
            start = int(rng.integers(0, n - width + 1))
            noise = rng.normal(0.0, 0.1 * temperature, size=width)
            cand[start:start + width] += noise

        elif op == 2:
            # shuffle a small subset of entries
            k = int(rng.integers(1, max(2, n // 50)))
            idx = rng.choice(n, size=k, replace=False)
            perm = rng.permutation(k)
            cand[idx] = cand[idx[perm]]

        else:
            # full redraw from a Dirichlet (keeps the total roughly the same)
            cand = rng.dirichlet(np.ones(n)) * n

        # enforce bounds
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = _fast_ac1(cand)

        # accept if better or with a probability that slowly cools down
        if cand_score < cur_score or rng.random() < np.exp(
            (cur_score - cand_score) / max(temperature, 1e-12)
        ):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score

        temperature *= cooling

    # ------------------------------------------------------------------
    # final safety check with the exact evaluator
    try:
        final_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(final_score):
            raise ValueError
    except Exception:
        # safe fallback – a uniform vector works and yields a finite score
        best = np.ones_like(a)

    return [float(x) for x in best]
