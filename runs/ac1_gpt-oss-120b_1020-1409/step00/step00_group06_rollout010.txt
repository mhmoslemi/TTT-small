import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimize the AC1 upper bound evaluated by `evaluate_sequence`.
    The strategy is simple: compare the provided parent sequence (if any)
    with a uniform constant sequence and return the one with the lower
    evaluated score. No heavy memory usage or external solvers are needed.
    """
    # Helper to safely evaluate a sequence and return (score, ok)
    def _score(seq):
        try:
            sc = evaluate_sequence(seq)
            if np.isfinite(sc):
                return float(sc), True
        except Exception:
            pass
        return float('inf'), False

    # Try to obtain the parent sequence `height_sequence_1`
    parent_arr = None
    try:
        # ``height_sequence_1`` is injected by the harness if available.
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent_arr = None

    # Determine the length to use
    if parent_arr is not None and parent_arr.ndim == 1 and parent_arr.size > 0:
        n = int(parent_arr.size)
    else:
        # Fallback length (used in the reference implementation)
        n = 256

    # Prepare candidate sequences
    candidates = []
    # 1. Parent sequence (if valid)
    if parent_arr is not None and parent_arr.ndim == 1 and parent_arr.size == n:
        # Clip to the allowed range as the evaluator would do
        parent_clipped = np.clip(parent_arr, 0.0, 1000.0)
        cand_parent = parent_clipped.tolist()
        score_parent, ok_parent = _score(cand_parent)
        if ok_parent:
            candidates.append((score_parent, cand_parent))

    # 2. Uniform constant sequence (value 1.0, any positive constant yields bound 2)
    const_seq = np.full(n, 1.0, dtype=np.float64)
    const_seq = np.clip(const_seq, 0.0, 1000.0)
    cand_const = const_seq.tolist()
    score_const, ok_const = _score(cand_const)
    if ok_const:
        candidates.append((score_const, cand_const))

    # 3. Optional tiny random perturbation of the constant (does not improve bound
    #    theoretically, but may avoid pathological ties).
    rng = np.random.default_rng(seed)
    perturb = const_seq.copy()
    # Apply a modest log‑normal perturbation on a random block (O(n) memory only)
    block_width = max(1, n // 100)
    start = int(rng.integers(0, n - block_width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.08)))
    perturb[start:start + block_width] *= factor
    perturb = np.clip(perturb, 0.0, 1000.0)
    cand_pert = perturb.tolist()
    score_pert, ok_pert = _score(cand_pert)
    if ok_pert:
        candidates.append((score_pert, cand_pert))

    # Choose the best-scoring candidate
    if not candidates:
        # As a last resort, return a simple uniform sequence of length n
        return [1.0] * n

    best_seq = min(candidates, key=lambda x: x[0])[1]
    # Ensure the returned list contains plain Python floats
    return [float(x) for x in best_seq]
