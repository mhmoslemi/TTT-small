import numpy as np

def make_block_perturbation(base, rng):
    candidates = base.copy()
    n = candidates.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidates[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidates = 0.5 * (candidates + candidates[::-1])
    return np.clip(candidates, 0.0, 1000.0)

def exponential_sequence(n, c):
    # c is positive; compute r = 1 - c/n, ensure 0<r<1
    p = c / n
    r = 1.0 - p
    if r <= 0.0:
        # degenerate: only first entry 1, rest zero
        seq = np.zeros(n, dtype=np.float64)
        seq[0] = 1.0
        return seq
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx)
    return seq

def log_perturb(seq, sigma, rng):
    # multiplicative log-space Gaussian perturbation
    # add small epsilon to avoid log(0)
    eps = 1e-12
    log_seq = np.log(seq + eps)
    log_seq += rng.normal(0.0, sigma, size=log_seq.shape)
    pert = np.exp(log_seq)
    return np.clip(pert, 0.0, 1000.0)

def make_block_perturbation(base, rng):
    """Apply a localized multiplicative perturbation and optional mirroring."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def exponential_sequence(n, c):
    """
    Return a monotone decreasing sequence a_i = (1 - c/n)^i  (length n).
    If c>=n the sequence degenerates to a single 1 at the start.
    """
    p = c / n
    r = 1.0 - p
    if r <= 0.0:
        seq = np.zeros(n, dtype=np.float64)
        seq[0] = 1.0
        return seq
    idx = np.arange(n, dtype=np.float64)
    seq = np.power(r, idx)
    return seq


def log_perturb(seq, sigma, rng):
    """Multiplicative log‑space Gaussian perturbation."""
    eps = 1e-12
    log_seq = np.log(seq + eps)
    log_seq += rng.normal(0.0, sigma, size=log_seq.shape)
    pert = np.exp(log_seq)
    return np.clip(pert, 0.0, 1000.0)
