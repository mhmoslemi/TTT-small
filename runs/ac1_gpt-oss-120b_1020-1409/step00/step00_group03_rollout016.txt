def block_perturb(base, rng, target_sum):
    # Prepare copy
    cand = base.copy()
    n = cand.size
    # Choose block width randomly between 1% and 30% of n
    min_w = max(1, n // 200)  # 0.5%
    max_w = max(2, n // 12)   # ~8%
    width = int(rng.integers(min_w, max_w+1))
    start = int(rng.integers(0, max(0, n - width + 1)))
    stop = start + width
    scale = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= scale
    # Optional symmetrize
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # Clip and renorm to target_sum
    cand = np.clip(cand, 0.0, 1000.0)
    cur_sum = cand.sum()
    if cur_sum > 0:
        cand *= (target_sum / cur_sum)
    else:
        # fallback to uniform if sum zero (should not happen)
        cand = np.full_like(cand, target_sum / n)
    # final clipping after renorm (again)
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

import numpy as np

def _block_perturb(base, rng, target_sum):
    """Return a copy of *base* with a random block scaled and re‑normalized."""
    cand = base.copy()
    n = cand.size
    # block width between ~0.5% and ~8% of the length
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps escape flat regions)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # enforce bounds, then re‑normalise to keep the total sum unchanged
    cand = np.clip(cand, 0.0, 1000.0)
    cur_sum = cand.sum()
    if cur_sum > 0.0:
        cand *= (target_sum / cur_sum)
    else:  # degenerate – fall back to uniform distribution
        cand = np.full_like(cand, target_sum / n)
    cand = np.clip(cand, 0.0, 1000.0)
    return cand
