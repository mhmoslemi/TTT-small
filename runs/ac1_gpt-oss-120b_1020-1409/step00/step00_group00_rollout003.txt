import numpy as np

def _make_perturbation(base, rng):
    """Return a non‑negative O(n) perturbation of ``base``."""
    cand = np.copy(base)
    n = cand.size
    # choose a block width between n/200 and n/12 (at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.11)))
    cand[start:stop] *= factor
    # occasional symmetry blend (helps escape shallow plateaus)
    if rng.random() < 0.32:
        cand = 0.5 * (cand + cand[::-1])
    # enforce problem bounds
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Return a list of non‑negative coefficients that (approximately) minimize
    the AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise a parent sequence.
    # Prefer the supplied ``height_sequence_1`` if it exists and looks sane.
    # Otherwise fall back to a uniform vector of modest length.
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # Uniform fallback of length 256 – any positive length works.
        parent = np.ones(256, dtype=np.float64)

    # Ensure we have a non‑empty candidate.
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Uniform vector of the same length provides the known theoretical optimum (ratio = 2).
    uniform = np.ones(best.size, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing using block perturbations.
    # Each iteration uses O(n) memory and a single exact evaluation.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = _make_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # The theoretical lower bound is 2; stop early if we reach it.
            if best_score <= 2.000001:
                break

    # ------------------------------------------------------------------
    # Final safeguard: if for any reason the best candidate is worse than
    # the uniform vector, return the uniform one (guaranteed to give 2.0).
    # ------------------------------------------------------------------
    if best_score > uniform_score:
        best = uniform

    # Return a plain Python list as required.
    return [float(x) for x in best]
