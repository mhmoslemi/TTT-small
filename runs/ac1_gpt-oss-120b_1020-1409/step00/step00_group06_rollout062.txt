def solve_lp_sub(f, ks, M):
    n = f.size
    m = len(ks)
    if m == 0:
        return None
    A = np.zeros((m, n), dtype=np.float64)
    for row_idx, k in enumerate(ks):
        start = max(0, k - (n - 1))
        stop = min(k, n - 1)
        # indices i where start<=i<=stop
        i = np.arange(start, stop + 1, dtype=np.int64)
        A[row_idx, i] = f[k - i]
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    bounds = [(0.0, 1000.0) for _ in range(n)]
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog

def make_block_perturbation(base, rng):
    """Return one O(n)-memory non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # choose a random block length and position
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring to keep symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def build_constraint_matrix(f, ks):
    """Build A where each row corresponds to a shift k in ks:
       (f * g)[k] = sum_i f[i] * g[k-i] <= M.
    """
    n = f.size
    m = len(ks)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(ks):
        start = max(0, k - (n - 1))
        stop = min(k, n - 1)
        i = np.arange(start, stop + 1, dtype=np.int64)
        A[row, i] = f[k - i]
    return A


def solve_lp_for_g(f, ks, M):
    """Solve   max sum(g)  s.t.   f * g (at shifts ks) <= M,   g >= 0.
       Returns the optimal g (numpy array) or None on failure.
    """
    n = f.size
    if len(ks) == 0:
        return None
    A = build_constraint_matrix(f, ks)
    b = np.full(len(ks), float(M), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximize sum(g)  <=> minimize -sum(g)
    bounds = [(0.0, 1000.0) for _ in range(n)]  # obey the evaluation clamping limits
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def active_shifts(conv, max_active=20):
    """Select up to max_active shift indices with largest convolution values."""
    if conv.size == 0:
        return np.array([], dtype=int)
    # take the indices of the largest values (including the absolute maximum)
    idx = np.argsort(conv)[-max_active:]
    return np.sort(idx)  # sorted for nicer matrix structure
