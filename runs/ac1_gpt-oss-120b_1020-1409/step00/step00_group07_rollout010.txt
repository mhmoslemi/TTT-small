import numpy as np

# The evaluator evaluate_sequence is provided by the harness.
# The parent construction (if any) is available as height_sequence_1.

def _select_positions(n: int, K: int, rng: np.random.Generator) -> list:
    """
    Greedy construction of a Sidon‑like set of indices:
    every unordered sum i+j (including i+i) is unique.
    Returns a list of K distinct positions or an empty list if it fails.
    """
    indices = np.arange(n, dtype=np.int64)
    rng.shuffle(indices)
    positions = []
    used_sums = set()          # stores all i+j for already chosen i, j (i≤j)

    for i in indices:
        if len(positions) >= K:
            break
        i = int(i)
        # the square i+i must be unique
        if (2 * i) in used_sums:
            continue
        # all cross sums i+j must be unique
        conflict = False
        for j in positions:
            if (i + j) in used_sums:
                conflict = True
                break
        if conflict:
            continue
        # accept this index
        positions.append(i)
        used_sums.add(2 * i)                 # its square
        for j in positions[:-1]:             # cross sums with earlier points
            used_sums.add(i + j)
    return positions if len(positions) == K else []
