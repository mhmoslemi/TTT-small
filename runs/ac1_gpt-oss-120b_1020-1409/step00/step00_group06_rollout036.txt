def project_to_simplex(v):
    """Project onto simplex {x >=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def greedy_sidon(order):

                break
        if not conflict:
            # add new sums
            sums.add(i + i)
            for j in indices:
                sums.add(i + j)
            indices.append(i)
    return indices

import numpy as np

MAX_VAL = 1000.0  # upper bound enforced by evaluate_sequence


def _greedy_sidset(order):
    """Create a Sidon‑type set using the given order of indices.

    The returned list `inds` satisfies that all sums i+j (i<=j) are distinct.
    """
    inds = []
    used = set()
    for i in order:
        # check self‑sum conflict
        if i + i in used:
            continue
        conflict = False
        for j in inds:
            if i + j in used:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        inds.append(i)
        used.add(i + i)
        for j in inds[:-1]:
            used.add(i + j)
    return inds


def propose_candidate(seed=42, num_attempts=300, **kwargs):
    """Return a height sequence aiming to lower the AC1 bound.

    The algorithm builds a large Sidon‑type subset of indices and puts the
    maximal allowed height (1000) on those positions.  With distinct
    pairwise sums the convolution peak is bounded by 2·MAX_VAL², while the
    ℓ₁‑norm grows linearly with the number of selected indices, yielding an
    AC1 value ≈4·n/|S|².  Randomised greedy construction is repeated and the
    best candidate is kept.
    """
    rng = np.random.default_rng(seed)

    # use the provided parent if available; otherwise fall back to a small vector
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
        n = base.size
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    best_score = float("inf")
    best_inds = None

    # try several random orderings to obtain a large Sidon set
    for _ in range(num_attempts):
        order = rng.permutation(n)
        inds = _greedy_sidset(order)
        m = len(inds)
        if m == 0:
            continue

        # compute convolution peak by explicit ordered pair accumulation (O(m²))
        conv = np.zeros(2 * n - 1, dtype=np.float64)
        prod = MAX_VAL * MAX_VAL
        for i in inds:
            for j in inds:
                conv[i + j] += prod

        max_b = conv.max()
        sum_a = MAX_VAL * m
        if sum_a < 0.01:
            continue
        score = 2.0 * n * max_b / (sum_a * sum_a)

        if score < best_score:
            best_score = score
            best_inds = inds
            # early exit if we already beat the target threshold
            if best_score < 1.50274349:
                break

    # fallback: uniform vector (guaranteed to be feasible)
    if best_inds is None or len(best_inds) == 0:
        a = np.full(n, MAX_VAL / n, dtype=np.float64)
    else:
        a = np.zeros(n, dtype=np.float64)
        a[best_inds] = MAX_VAL

    # final safety clipping (enforces evaluate_sequence requirements)
    a = np.clip(a, 0.0, MAX_VAL)

    # return as a plain Python list of floats
    return [float(v) for v in a]
