    # FFT of the real array
    fft = np.fft.rfft(arr)
    # Convolution via pointwise multiplication in frequency domain
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_b = conv.max()
    sum_a = arr.sum()
    # Avoid division by zero – the caller guarantees sum_a > 0.
    return 2.0 * n * max_b / (sum_a * sum_a)


def _binary_candidate(n: int, rng: np.random.Generator, p: float) -> np.ndarray:
    """Generate a {0,1} vector with marginal probability p."""
    return (rng.random(n) < p).astype(np.float64)


def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose AC1 upper bound
    is hopefully below 1.5.  The routine builds a short list of random binary
    candidates, uses a cheap FFT‑based approximation to filter them, and finally
    evaluates the best one exactly with `evaluate_sequence`.
    """
    # Parameters – can be overridden via kwargs (e.g. num_trials=200, prob=0.5)
    num_trials = int(kwargs.get("num_trials", 200))
    prob = float(kwargs.get("prob", 0.5))

    rng = np.random.default_rng(seed)

    # Base length and a safe fallback parent (the provided construction, if any)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        if base.ndim != 1 or n == 0 or not np.isfinite(base).all():
            raise Exception
    except Exception:
        # Fallback to a simple constant sequence of length 1024
        n = 1024
        base = np.ones(n, dtype=np.float64)

    # Start with the parent as a guaranteed valid candidate.
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Quick sanity: if the parent is already very good, we may return early.
    # (Score 2 is the typical uniform bound; we aim for < 1.5.)
    if best_score <= 1.5:
        return [float(x) for x in best]

    # Main random search – each trial uses a cheap FFT approximation first.
    for _ in range(num_trials):
        cand = _binary_candidate(n, rng, prob)
        # Ensure a non‑trivial sum; otherwise the evaluator would reject.
        if cand.sum() < 0.01:
            continue

        approx = _approx_ac1(cand)
        # Only spend the expensive exact evaluation if the approximation looks promising.
        if approx < best_score:
            try:
                exact = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue
            if np.isfinite(exact) and exact < best_score:
                best = cand
                best_score = exact

    # As a last resort, if the random search never improved, fall back to a very sparse Sidon‑like pattern.
    if best_score > 1.5:
        # Construct a simple Sidon set (size ≈ sqrt(n)) and put a 1 at each position.
        sidon = np.zeros(n, dtype=np.float64)
        used_sums = set()
        for pos in range(n):
            # Greedy acceptance: check that all new sums are unique.
            collision = False
            for existing in np.flatnonzero(sidon):
                s = pos + existing
                if s in used_sums:
                    collision = True
                    break
            if not collision:
                # Accept this position.
                sidon[pos] = 1.0
                # Record new sums (including the self‑sum).
                for existing in np.flatnonzero(sidon):
                    used_sums.add(pos + existing)
                # Stop early once we have a reasonable number of ones.
                if sidon.sum() >= int(np.sqrt(n)):
                    break
        # Evaluate the Sidon candidate exactly.
        try:
            exact = float(evaluate_sequence(sidon.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = sidon
                best_score = exact
        except Exception:
            pass

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
