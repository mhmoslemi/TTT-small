```python
def build_large_set(n, k_target, max_mult, rng):
    positions = []
    sum_counts = {}
    attempts = 0
    while len(positions) < k_target and attempts < 10000:
        cand = int(rng.integers(0, n))
        if cand in positions:
            attempts += 1
            continue
        # compute new sums:
        good = True
        # self-sum
        s = cand + cand
        cnt = sum_counts.get(s, 0) + 1
        if cnt > max_mult:
            good = False
        else:
            # check with existing
            for p in positions:
                s1 = cand + p
                if sum_counts.get(s1, 0) + 1 > max_mult:
                    good = False
                    break
        if not good:
            attempts += 1
            continue
        # update counts
        sum_counts[cand + cand] = sum_counts.get(cand + cand, 0) + 1
        for p in positions:
            s1 = cand + p
            sum_counts[s1] = sum_counts.get(s1, 0) + 1
        positions.append(cand)
        attempts = 0 # reset as we succeeded, maybe to allow more tries
    return positions
```

import numpy as np


def _build_set(n, target_k, max_mult, rng):
    """Greedy construction of a set of indices whose pairwise sums appear at most `max_mult` times."""
    positions = []
    sum_counts = {}
    attempts = 0
    max_attempts = 10000
    while len(positions) < target_k and attempts < max_attempts:
        cand = int(rng.integers(0, n))
        if cand in positions:
            attempts += 1
            continue

        # check that adding `cand` respects the multiplicity bound
        good = True
        s_self = cand + cand
        if sum_counts.get(s_self, 0) + 1 > max_mult:
            good = False
        else:
            for p in positions:
                s = cand + p
                if sum_counts.get(s, 0) + 1 > max_mult:
                    good = False
                    break

        if not good:
            attempts += 1
            continue

        # record the new sums
        sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
        for p in positions:
            s = cand + p
            sum_counts[s] = sum_counts.get(s, 0) + 1

        positions.append(cand)
        attempts = 0  # reset counter after a successful addition
    return positions, sum_counts

    Return a non‑negative step‑function (list of floats) whose
