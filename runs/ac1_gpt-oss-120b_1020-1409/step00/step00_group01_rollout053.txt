try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

def full_convolution_fft(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    fft_len = 1 << (out_len - 1).bit_length()
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[:out_len]
    return conv

def conv_two_fft(a, b):
    n = a.shape[0]
    out_len = 2 * n - 1
    fft_len = 1 << (out_len - 1).bit_length()
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(A * B, fft_len)[:out_len]
    return conv

def make_block_perturbation(base, rng):
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return full convolution a*a using FFT (real signals)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    fft_len = 1 << (out_len - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[:out_len]
    # numerical noise may produce tiny negative values
    return np.maximum(conv, 0.0)


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b via FFT (real signals)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    fft_len = 1 << (out_len - 1).bit_length()
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(A * B, fft_len)[:out_len]
    return np.maximum(conv, 0.0)
