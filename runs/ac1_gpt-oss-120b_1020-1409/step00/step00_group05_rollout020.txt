    fallback = np.ones(n, dtype=np.float64)
    fallback_score = float(evaluate_sequence(fallback.tolist()))
    best = fallback.copy()
    best_score = fallback_score
    # determine K range
    min_k = max(2, n // 20)  # ~225
    max_k = max(min_k+1, n // 4)  # ~1120
    for _ in range(num_trials):
        k = int(rng.integers(min_k, max_k+1))
        # choose distinct indices
        idx = rng.choice(n, size=k, replace=False)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = 1.0
        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:

            best_score = score
            # break early if we beat target threshold
            if best_score < 1.50274349:

import numpy as np

# ------------------------------------------------------------
# Helper: greedy construction of a sparse 0/1 sequence.
# The construction keeps the ordered pair count for each
# convolution index (i+j) below a prescribed limit L.
# ------------------------------------------------------------
def _greedy_build(n: int, L: int, rng: np.random.Generator):
    """
    Return a tuple (indices, sum_counts) where:
      - indices is an array of positions set to 1,
      - sum_counts[t] = number of ordered pairs (i,j) with i+j=t
        for the chosen indices (i.e. the convolution result).
    The algorithm respects sum_counts[t] + added ≤ L for every t.
    """
    sum_counts = np.zeros(2 * n - 1, dtype=np.int32)
    indices = []                     # selected positions
    order = rng.permutation(n)      # random scan order

    for i in order:
        # self‑pair contribution (i,i) adds 1 to sum_counts[2*i]
        if sum_counts[i + i] + 1 > L:
            continue

        # check contributions with previously selected positions
        ok = True
        for j in indices:
            s = i + j
            # (i,j) and (j,i) together add 2 to sum_counts[s]
            if sum_counts[s] + 2 > L:
                ok = False
                break
        if not ok:
            continue

        # accept i: update the counters
        sum_counts[i + i] += 1
        for j in indices:
            s = i + j
            sum_counts[s] += 2
        indices.append(i)

    return np.array(indices, dtype=int), sum_counts


# ------------------------------------------------------------
# Main entry point required by the harness.
# ------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """
    Construct a non‑negative step‑function (list of floats) whose
    autocorrelation upper bound is low.
    The routine builds several sparse binary candidates using the
    greedy construction above and keeps the best one.
    """
    rng = np.random.default_rng(seed)

    # length of the problem (provided as height_sequence_1 if present)
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256                     # fallback length

    # -----------------------------------------------------------------
    # baseline: uniform sequence (score = 2) – serves as a safe fallback
    # -----------------------------------------------------------------
    baseline = np.ones(n, dtype=np.float64)
    best_seq = baseline.copy()
    best_score = float(evaluate_sequence(baseline.tolist()))

    # -----------------------------------------------------------------
    # Search parameters
    # -----------------------------------------------------------------
    target_score = 1.50274349          # desired upper bound
    L_values = list(range(4, 13))      # allowed max ordered count per sum
    attempts_per_L = 5                 # how many random greedy tries per L

    for L in L_values:
        if best_score < target_score:
            break                     # already good enough

        for _ in range(attempts_per_L):
            idx, sc = _greedy_build(n, L, rng)
            k = idx.size
            if k < 2:                # need at least two non‑zero entries
                continue

            cand = np.zeros(n, dtype=np.float64)
            cand[idx] = 1.0            # binary amplitudes

            # exact evaluation (the harness uses the same function)
            score = float(evaluate_sequence(cand.tolist()))

            if np.isfinite(score) and score < best_score:
                best_seq = cand
                best_score = score
                # early exit if we beat the target
                if best_score < target_score:
                    break
        # end attempts loop
    # end L loop

    # -----------------------------------------------------------------
    # Ensure we always return a non‑empty, finite sequence.
    # -----------------------------------------------------------------
    if best_seq.size == 0:
        best_seq = baseline.copy()

    # Convert to plain Python floats for the harness
    return [float(v) for v in best_seq]
