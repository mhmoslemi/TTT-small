import numpy as np

def make_block_perturbation(arr, rng):
    n = arr.size
    # Ensure width at least 1 and at most n//10 for speed (tiny)
    # width distribution: log-uniform maybe
    max_width = max(1, n // 20) # small
    width = int(rng.integers(1, max_width+1))
    start = int(rng.integers(0, n - width + 1))
    factor = np.exp(rng.normal(0.0, 0.08))
    new = arr.copy()
    new[start:start+width] *= factor
    # Optionally occasionally symmetrize (similar to prior)
    if rng.random() < 0.2:
        new = 0.5 * (new + new[::-1])
    np.clip(new, 0.0, 1000.0, out=new)
    return new

def propose_candidate(seed=42, num_steps=200, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine parent fallback
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim !=1 or parent_arr.size==0 or not np.isfinite(parent_arr).all():
            raise ValueError
    except Exception:
        parent_arr = np.full(256, 1.0, dtype=np.float64)
    n = parent_arr.shape[0]

    # Generate power‑law candidates for a few exponents.
    exponents = [0.5, 0.55, 0.45]  # empirical mix
    best = None
    best_score = np.inf

    for p in exponents:
        cand = (np.arange(n, dtype=np.float64) + 1.0) ** (-p)
        np.clip(cand, 0.0, 1000.0, out=cand)
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best = cand
            best_score = s

    # also consider the parent as initial best
    s_parent = float(evaluate_sequence(parent_arr.tolist()))
    if np.isfinite(s_parent) and s_parent < best_score:
        best = parent_arr.copy()
        best_score = s_parent

    # Local stochastic refinements
    for _ in range(num_steps):
        cand = make_block_perturbation(best, rng)
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best = cand
            best_score = s

    # Return as plain python list of floats
    return [float(v) for v in best]

def make_block_perturbation(arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Return a new non‑negative candidate derived from ``arr``.
    A random contiguous block is multiplied by a log‑normal factor.
    Occasionally the whole vector is symmetrised to encourage diversity.
    """
    n = arr.size
    # choose a block size that is a small fraction of n (but at least 1)
    max_width = max(1, n // 20)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.08)))  # modest log‑normal jitter

    new[start:start + width] *= factor

        new = 0.5 * (new + new[::-1])

    # enforce the hard domain limits required by the evaluator
    np.clip(new, 0.0, 1000.0, out=new)
    return new
