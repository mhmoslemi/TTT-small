import numpy as np

def _sidon_indices(n, rng, max_attempts=30, min_size=0):
    """
    Find a set of indices S ⊂ {0,…,n-1} such that all unordered sums i+j (i≤j)
    are distinct.  The algorithm is a simple greedy search with random order.
    Returns the largest set found over ``max_attempts`` runs (or a set of at
    least ``min_size`` if possible).
    """
    best_set = []
    best_len = 0
    # theoretical upper bound m satisfying m(m+1)/2 ≤ 2n-1
    m_max = int(np.floor((-1.0 + np.sqrt(1.0 + 8.0 * (2 * n - 1))) / 2.0))

    for _ in range(max_attempts):
        order = rng.permutation(n)               # random order of candidates
        used = np.zeros(2 * n - 1, dtype=bool)   # used[sum] == True if sum already taken
        cur = []
        for i in order:
            if used[2 * i]:                       # conflict with its own square
                continue
            if cur:
                s = i + np.array(cur, dtype=int)
                if used[s].any():
                    continue
            # no conflicts – accept i
            if cur:
                used[i + np.array(cur, dtype=int)] = True
            used[2 * i] = True
            cur.append(int(i))
            # early exit if we already hit the theoretical maximum
            if len(cur) >= m_max:
                break
        if len(cur) > best_len:
            best_set = cur
            best_len = len(cur)
            if best_len >= m_max:
                break
        if best_len >= min_size and min_size > 0:
            # sufficient size reached, stop early
            break
    return best_set

    non‑zero term.  Hence the maximum convolution entry equals 1000², while the
    total sum is 1000·|S|, giving a score 2·n·10⁶/(1000·|S|)² = 2·n/|S|².
    For |S| ≥ 60 (and n≈2500) the resulting score is well below the target
    bound 1.50274349.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size.
    if 'height_sequence_1' in globals():
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 0:

        else:
            n = 256
    else:
        n = 256

    # Desired minimal set size to beat the target.  Derived from
    #   2·n/|S|² < 1.50274349  ⇒  |S| > sqrt(2·n/1.5027) .
    min_needed = int(np.ceil(np.sqrt(2.0 * n / 1.50274349))) + 1
    # Ensure a modest lower bound even for very small n.
    min_needed = max(min_needed, 60)

    # Find a Sidon set.

    # If the greedy search failed to reach the required size, fall back to
    # the uniform constant‑1000 sequence (score = 2.0).  This still produces a
    # valid candidate.
    if len(indices) < min_needed:
        candidate = np.full(n, 1000.0, dtype=np.float64)
    else:
        candidate = np.zeros(n, dtype=np.float64)
        candidate[indices] = 1000.0

    # Enforce the evaluator’s clipping bounds (redundant but safe).
