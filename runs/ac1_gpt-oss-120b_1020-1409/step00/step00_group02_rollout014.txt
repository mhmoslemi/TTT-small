def propose_candidate(seed=42, num_steps=2000):
    rng = np.random.default_rng(seed)
    # use parent as start (height_sequence_1)
    parent = np.asarray(height_sequence_1, dtype=np.float64)
    # sanitize
    if parent.ndim !=1 or parent.size==0 or not np.isfinite(parent).all():
        parent = np.full(256, 1.0)
    best = np.clip(parent, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))
    n = best.size

    for step in range(num_steps):
        # decide perturbation type
        if rng.random() < 0.15:
            # random shuffle entire vector
            cand = best.copy()
            rng.shuffle(cand)
        else:
            # targeted reduction
            # compute shift where autocorrelation is max for best
            conv = np.convolve(best, best)
            max_idx = int(np.argmax(conv))
            shift = max_idx - (n - 1)
            cand = best.copy()
            # Try to find a pair (i,i+shift) within bounds
            if shift == 0:
                # just pick two distinct indices
                i, j = rng.choice(n, size=2, replace=False)
                # reduce both
                r = rng.uniform(0.6, 0.95)
                lost = (cand[i] + cand[j]) * (1 - r)
                cand[i] *= r
                cand[j] *= r
                # add lost to random other index
                k = rng.choice([idx for idx in range(n) if idx not in (i,j)])
                cand[k] += lost
            else:
                # pick i such that i+shift in range
                # Determine valid i range
                low = max(0, -shift)
                high = min(n, n - shift)
                if high - low <= 0:
                    # fallback: random shuffle
                    cand = best.copy()
                    rng.shuffle(cand)
                else:
                    i = rng.integers(low, high)
                    j = i + shift
                    r = rng.uniform(0.6, 0.95)
                    lost = (cand[i] + cand[j]) * (1 - r)
                    cand[i] *= r
                    cand[j] *= r
                    # add lost to another random index not i,j
                    # pick from all indices else
                    k = rng.integers(0, n)
                    while k == i or k == j:
                        k = rng.integers(0, n)
                    cand[k] += lost
        # clip to bounds and ensure nonnegative
        cand = np.clip(cand, 0.0, 1000.0)
        # Evaluate
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
        # occasional random restart: with small prob, replace best with parent or random
        if rng.random() < 0.02:
            # restart from parent with small random modifications to avoid stagnation
            cand = parent.copy()
            # apply simple perturbation
            width = max(1, n // 20)
            start = rng.integers(0, n - width + 1)
            cand[start:start+width] *= np.exp(rng.normal(0, 0.1))
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < best_score:
                best = cand
                best_score = cand_score
    return [float(x) for x in best]

def _linear_ramp(n: int) -> np.ndarray:
    """Return a non‑negative ramp a_i = α·i with max value ≤ 1000."""
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    # α = 1000/(n‑1) guarantees a_{n‑1}=1000, a_0=0.
    scale = 1000.0 / (n - 1)
    ramp = np.arange(n, dtype=np.float64) * scale
    return np.clip(ramp, 0.0, 1000.0)
