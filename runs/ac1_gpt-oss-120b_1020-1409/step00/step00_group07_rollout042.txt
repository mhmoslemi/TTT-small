def gradient(seq, beta):
    conv = np.convolve(seq, seq)
    # compute p softmax over conv
    max_conv = np.max(conv)  # for numerical stability? but we need M_s anyway
    # compute p using log-sum-exp
    # compute stable exponents:
    exps = np.exp(beta * (conv - max_conv))
    sum_exps = np.sum(exps)
    p = exps / sum_exps
    # compute g_partial (2 * cross-correlation)
    seq_rev = seq[::-1]
    g_full = np.convolve(seq_rev, p)
    start = len(seq) - 1
    g_partial = 2.0 * g_full[start:start+len(seq)]

    sum_seq = np.sum(seq)
    M_s = (np.log(sum_exps) / beta) + max_conv  # because M_s = (1/beta) log sum exp(beta*conv) = max_conv + logsumexp / beta

    # compute gradient vector
    factor = 2.0 * len(seq) / (sum_seq**2)
    term1 = factor * g_partial
    term2 = 4.0 * len(seq) * M_s / (sum_seq**3)  # scalar

    grad = term1 - term2
    return grad, M_s, sum_seq

    size = len(a)+len(b)-1
    N = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    c = np.fft.irfft(C, n=N)[:size]
    return c

import numpy as np

# ---------- convolution using FFT (real input) ----------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real 1‑D arrays using FFT."""
    size = a.size + b.size - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    c = np.fft.irfft(C, n=N)
    return c[:size]


# ---------- scoring ---------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Exact score used by the evaluator (avoid the external call)."""
    conv = _conv_fft(seq, seq)
    max_val = np.max(conv)
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    return float(2.0 * seq.size * max_val / (s * s))


# ---------- one gradient‑descent optimisation run -------------------------
def _optimise(start_seq: np.ndarray, steps: int = 200) -> tuple[np.ndarray, float]:
    """Run a short (matrix‑free) gradient‑descent on the smoothed max."""
    n = start_seq.size
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_sc = _score(seq)

    # parameters for the smooth‑max
    beta0 = 10.0          # start smoothing factor
    beta_inc = 0.15       # increase per iteration

    rng = np.random.default_rng(0)

    for it in range(steps):
        beta = beta0 + it * beta_inc

        # full convolution (needed for max and for the gradient)
        conv = _conv_fft(seq, seq)
        max_conv = np.max(conv)

        # smoothed max (log‑sum‑exp trick)
        shift = conv - max_conv                # ≤ 0, safe for exp
        exp_shift = np.exp(beta * shift)
        sum_exp = np.sum(exp_shift)
        probs = exp_shift / sum_exp            # soft‑max over convolution entries
        smooth_max = max_conv + np.log(sum_exp) / beta

        # gradient of the smooth‑max w.r.t. the sequence
        rev = seq[::-1]
        cross = _conv_fft(rev, probs)          # length = 3n‑2
        start = n - 1
        grad_smooth = 2.0 * cross[start:start + n]

        s = np.sum(seq)
        if s < 1e-12:
            break

        # full objective gradient = d(2 n * M / S²)/da
        factor = 2.0 * n / (s * s)
        term1 = factor * grad_smooth
        term2 = 4.0 * n * smooth_max / (s * s * s)
        grad = term1 - term2

        # step – normalise the direction to avoid overshooting
        avg = s / n
        norm = np.linalg.norm(grad) + 1e-12
        step = avg * 0.5 / norm
        cand = seq - step * grad
        cand = np.clip(cand, 0.0, 1000.0)

        sc = _score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_seq = cand.copy()

        # accept the candidate if it improves, otherwise keep current seq
        if sc < best_sc * 0.9999:
            seq = cand
        else:
            # occasional random jitter to escape flat regions
            if rng.random() < 0.03:
                jitter = np.exp(rng.normal(0.0, 0.02, size=n))
                seq = np.clip(seq * jitter, 0.0, 1000.0)

    return best_seq, best_sc
