    n = len(seq)
    b = np.convolve(seq, seq)
    k = int(np.argmax(b))

import numpy as np

# ----------------------------------------------------------------------
# Helper: exact score (identical to evaluate_sequence but without repeats)
def _score(arr: np.ndarray) -> float:
    """Return the objective 2*n*max(conv)/sum^2 for a non‑negative array."""
    s = float(arr.sum())
    if s < 0.01:
        return float("inf")
    conv = np.convolve(arr, arr)
    max_b = float(conv.max())
    return 2.0 * arr.size * max_b / (s * s)

# Simple block perturbation (same idea as the reference program)
def _block_perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    # pick a block width (geometric progression‑like)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional mirroring
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Reduce the current convolution peak by weakening the strongest pair
def _reduce_peak(base: np.ndarray, rng: np.random.Generator,
                 reduction: float = 0.90, inc_frac: float = 0.02) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    conv = np.convolve(cand, cand)
    k = int(np.argmax(conv))                     # index of the largest entry
    i0 = max(0, k - (n - 1))
    i1 = min(n - 1, k)
    i_vals = np.arange(i0, i1 + 1)
    j_vals = k - i_vals
    prod = cand[i_vals] * cand[j_vals]
    idx = int(np.argmax(prod))
    i = i_vals[idx]
    j = j_vals[idx]

    # shrink the two contributors
    removed_i = cand[i] - cand[i] * reduction
    removed_j = cand[j] - cand[j] * reduction
    cand[i] *= reduction
    cand[j] *= reduction
    removed = removed_i + removed_j

    # spread the removed mass to a few random locations
    inc_cnt = max(1, int(inc_frac * n))
    inc_idx = rng.choice(n, size=inc_cnt, replace=False)
    inc_add = removed / inc_cnt
    cand[inc_idx] += inc_add

    return np.clip(cand, 0.0, 1000.0)

    Return a list of non‑negative heights that (hopefully) improves on the
    baseline 2.0 bound.  The routine mixes cheap block perturbations,
    random swaps and a small deterministic peak‑reduction step.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise from the supplied parent if it exists
    parent = np.asarray(
        globals().get("height_sequence_1", np.ones(256, dtype=np.float64)),
        dtype=np.float64,
    )
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)

    best = parent.copy()
    best_score = _score(best)

    n = best.size

    # ------------------------------------------------------------------
    for step in range(num_steps):
        # choose a cheap modification strategy
        mode = rng.integers(0, 3)

        if mode == 0:

                cand[i], cand[j] = cand[j], cand[i]

            cand = _reduce_peak(best, rng)
