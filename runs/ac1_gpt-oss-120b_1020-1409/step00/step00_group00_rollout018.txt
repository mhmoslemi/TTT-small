import numpy as np

def _sidon_set(n: int, rng: np.random.Generator) -> np.ndarray:
    """Greedy Sidon (B2) set construction with a random order."""
    perm = rng.permutation(n)
    selected = []
    sums = set()
    for i in perm:
        # test for collisions with existing sums
        coll = (i + i) in sums
        if not coll:
            for j in selected:
                if (i + j) in sums:
                    coll = True
                    break
        if not coll:
            selected.append(i)
            sums.add(i + i)
            for j in selected[:-1]:
                sums.add(i + j)
    return np.array(selected, dtype=np.int64)


def _build_candidate(selected: np.ndarray, n: int,
                     big_val: float, eps: float) -> np.ndarray:
    """Create the step‑function: `big_val` on Sidon indices, `eps` elsewhere."""
    a = np.full(n, eps, dtype=np.float64)
    a[selected] = big_val
    return a


def propose_candidate(seed: int = 42, num_steps: int = 0, **kwargs):
    """Return a non‑negative sequence with a low AC1 upper bound.

    The method builds a large Sidon set of indices, puts the maximal allowed
    height (1000) on those positions and a modest constant on the remaining
    entries.  Several random attempts are tried and the best sequence according
    to `evaluate_sequence` is returned.
    """
    rng = np.random.default_rng(seed)

    # parent fallback (provided by the harness if available)
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    n = best.size
    BIG = 1000.0           # maximal admissible height

    # Try a few random Sidon constructions; keep the best scoring one.
    for _ in range(12):
        sel = _sidon_set(n, rng)
        k = len(sel)

        # Choose epsilon so that large‑epsilon contributions stay below
        # the 2·BIG² term in the convolution peak.
        # A safe bound is eps ≤ 2*BIG / k, and we also cap it at 6.
        if k > 0:
            eps_bound = (2.0 * BIG) / k
            eps = min(6.0, eps_bound)
        else:
            eps = 1.0
        eps = max(1.0, float(eps))

        cand = _build_candidate(sel, n, BIG, eps)

        # Exact evaluation (may be expensive, but only a few calls)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    return [float(x) for x in best]
