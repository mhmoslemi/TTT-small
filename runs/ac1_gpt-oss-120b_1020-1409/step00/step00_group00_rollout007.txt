    n = parent.size
    # compute mean of parent (clip)
    mean_val = np.clip(parent.mean(), 0.0, 1000.0)

    # Evaluate both uniform and parent, choose the one with lower score.
    score_parent = float(evaluate_sequence(parent.tolist()))
    score_uniform = float(evaluate_sequence(uniform.tolist()))
    best_seq = uniform if score_uniform < score_parent else parent
    # Also try some small gradient descent to refine.
    # Perform few iterations: shift a bit towards uniform.
    best = best_seq.copy()
    best_score = min(score_parent, score_uniform)
    for i in range(200):
        # compute a small random perturbation that preserves sum (e.g., subtract from random i, add to j)
        idx1 = rng.integers(0, n)
        idx2 = rng.integers(0, n)
        if idx1 == idx2:

        # amount proportional to current magnitude
        amount = rng.random() * best[idx1] * 0.1
        new = best.copy()
        if new[idx1] >= amount:
            new[idx1] -= amount
            new[idx2] += amount
        new = np.clip(new, 0.0, 1000.0)
        score = float(evaluate_sequence(new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = new
            best_score = score
    return [float(x) for x in best]

import numpy as np

def propose_candidate(seed=42, **kwargs):
    rng = np.random.default_rng(seed)
    # Acquire parent sequence if exists
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_raw.ndim != 1 or parent_raw.size == 0 or not np.isfinite(parent_raw).all():
            raise ValueError
    except Exception:
        # fallback: simple constant sequence of length 256
        parent_raw = np.full(256, 1.0, dtype=np.float64)
    # Clip to allowed bounds
    parent = np.clip(parent_raw, 0.0, 1000.0)
    n = parent.size
    # Uniform candidate based on mean of parent
    mean_val = parent.mean()
    if mean_val <= 0.0:
        mean_val = max(0.01 / n, 0.0)
    uniform = np.full(n, mean_val, dtype=np.float64)
    # Compute scores
    best_seq = parent
    best_score = float(evaluate_sequence(best_seq.tolist()))
    # evaluate uniform
    uni_score = float(evaluate_sequence(uniform.tolist()))
    if uni_score < best_score:
        best_seq = uniform
        best_score = uni_score
    # Try symmetrized version of parent (average with reversed)
    sym = 0.5 * (parent + parent[::-1])
    sym_score = float(evaluate_sequence(sym.tolist()))
    if sym_score < best_score:
        best_seq = sym
        best_score = sym_score
    # Small hill climbing steps that preserve sum
    best = best_seq.copy()
    sum_best = best.sum()
    # Determine a small step size relative to typical entry magnitude
    avg = sum_best / n if n else 0.0
    for _ in range(200):
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # transfer a small amount from i to j
        # amount is a random fraction of the source value
        if best[i] <= 0.0:
            continue
        amt = rng.random() * best[i] * 0.05  # up to 5% transfer
        cand = best.copy()
        cand[i] -= amt
        cand[j] += amt
        # clip to bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        # ensure sum not too small (should stay close)
        if cand.sum() < 0.01:
            continue
        cand_score = float(evaluate_sequence(cand.tolist()))
        if cand_score < best_score:
            best = cand
            best_score = cand_score
    # Return list of Python floats
    return [float(x) for x in best]
