def gradient_move(a, rng):
   # compute convolution
   c = np.convolve(a, a)
   k_max = int(np.argmax(c))
   n = a.size
   g = a.copy()  # start with a_i term
   # compute reversed part contributions
   idx = k_max - np.arange(n)
   # valid mask where idx in [0,n)
   mask = (idx >= 0) & (idx < n)
   # where mask true, add a[idx]
   g[mask] += a[idx[mask]]
   # pick indices
   i_high = int(np.argmax(g))
   i_low = int(np.argmin(g))
   # Determine transfer amount delta: small fraction of a[i_high]
   delta = min(a[i_high], max(1e-8, a[i_high] * rng.uniform(0.01, 0.2)))
   # Transfer
   new_a = a.copy()
   new_a[i_high] -= delta
   new_a[i_low] += delta
   # Clip to [0,1000] but ensure sum unchanged (maybe negligible). Since delta remove from high and add to low, sum unchanged.
   new_a = np.clip(new_a, 0.0, 1000.0)
   return new_a

def random_pair_transfer(a, rng):
    n = a.size

    new_a[j] += amount
    new_a = np.clip(new_a, 0.0, 1000.0)

import numpy as np

# Helper: simple random shuffle of the sequence
def random_shuffle(arr, rng):
    new_arr = arr.copy()
    rng.shuffle(new_arr)
    return new_arr

# Helper: move a random fraction of mass from one index to another
def random_pair_transfer(arr, rng):
    n = arr.size
    i, j = rng.choice(n, size=2, replace=False)
    if arr[i] == 0.0:
        return arr.copy()
    # transfer at most 20 % of the donor value
    amount = arr[i] * rng.uniform(0.0, 0.2)
    new_arr = arr.copy()
    new_arr[i] -= amount
    new_arr[j] += amount
    return np.clip(new_arr, 0.0, 1000.0)

# Helper: a deterministic move aimed at reducing the current maximal convolution value
def gradient_move(arr, rng):
    n = arr.size
    # find the index where the convolution attains its maximum
    conv = np.convolve(arr, arr)
    k_max = int(np.argmax(conv))

    # gradient of the maximal convolution w.r.t. each entry:
    # g_i = a_i + a_{k_max-i}   (the second term is added only if the index is valid)
    g = arr.copy()
    idx = k_max - np.arange(n)               # potential counterpart indices
    mask = (idx >= 0) & (idx < n)            # where the counterpart lies inside the array
    g[mask] += arr[idx[mask]]

    # choose a source (large gradient) and a sink (small gradient)
    src = int(np.argmax(g))
    dst = int(np.argmin(g))
    if src == dst or arr[src] == 0.0:
        return arr.copy()

    # transfer a modest fraction of the source to the destination
    delta = arr[src] * rng.uniform(0.01, 0.2)
    new_arr = arr.copy()
    new_arr[src] -= delta
    new_arr[dst] += delta
    return np.clip(new_arr, 0.0, 1000.0)

    """

        cur = np.ones(cur.size, dtype=np.float64)

    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # main optimisation loop – each iteration uses O(n) auxiliary memory.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # choose a proposal strategy
        r = rng.random()
        if r < 0.3:
            cand = random_pair_transfer(cur, rng)
        elif r < 0.6:
            cand = gradient_move(cur, rng)
        else:

        if np.isfinite(score) and score < best_score:
