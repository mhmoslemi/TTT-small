import numpy as np

def _make_uniform(parent):
    """
    Return a uniform array with the same total mass as ``parent``.
    Each entry is clipped to the allowed range [0, 1000].
    """
    n = parent.size
    total = float(parent.sum())
    if total < 0.01:               # ensure the evaluator does not reject it
        total = 0.01
    val = total / n
    # Clip to the admissible interval.
    if val < 0.0:
        val = 0.0
    elif val > 1000.0:
        val = 1000.0
    return np.full(n, val, dtype=np.float64)
