import numpy as np

def _full_convolution(a: np.ndarray, n_fft: int) -> np.ndarray:
    """Convolution a*a (full) using real FFT; returns length 2*len(a)-1 array."""
    # rfft works on real input and returns N/2+1 complex values.
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)
    return conv[: 2 * a.size - 1]


def _score_from_array(a: np.ndarray, n_fft: int) -> float:
    """Fast approximation of the evaluator score (identical formula)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _full_convolution(a, n_fft)
    max_b = conv.max()
    return 2.0 * a.size * max_b / (s * s)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # Power‑law candidate a_i = 1/sqrt(i+1)

    power = np.clip(power, 0.0, 1000.0)

    # Choose the better of the two as the starting point
    n_fft = 1 << ((2 * n - 1).bit_length())
    best = parent.copy()
    best_score = _score_from_array(best, n_fft)
    pw_score = _score_from_array(power, n_fft)
    if np.isfinite(pw_score) and pw_score < best_score:
        best = power.copy()
        best_score = pw_score

    # ------------------------------------------------------------------

    current = best.copy()
