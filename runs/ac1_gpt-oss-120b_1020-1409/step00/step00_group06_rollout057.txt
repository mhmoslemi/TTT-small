def propose_candidate(seed=42, max_iters=2000, step=0.5, random_step_scale=0.1, patience=200):

    # try to read parent
    try:

    n = a.shape[0]

        X = np.fft.rfft(x, n=m)
        c = np.fft.irfft(X * X, n=m)[:size]

    # Precompute score
    best = a.copy()
    S = best.sum()
    conv = conv_self(best)
    M = conv.max()
    best_score = float(2 * n * M / (S**2))
    # may also compute via evaluate_sequence
    best_score_eval = float(evaluate_sequence(best.tolist()))
    # Use whichever (they should be same)
    best_score = best_score_eval

    no_improve = 0

    for it in range(max_iters):
        # compute convolution, sum
        conv = conv_self(a)
        M = conv.max()
        k_star = int(conv.argmax())
        S = a.sum()
        if S < 0.01:
            # adjust small positive to avoid division by zero
            a += 1.0
            S = a.sum()
        # compute gradient
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        dM = np.zeros_like(a)
        dM[valid] = 2.0 * a[idx[valid]]
        if k_star % 2 == 0:
            i_mid = k_star // 2
            if i_mid < n:
                dM[i_mid] = a[i_mid]
        grad = 2.0 * n * (dM * S - 2.0 * M) / (S**3)

        # typical step size that decreases
        alpha = step / (1.0 + it * 0.001)  # slowly shrink

        cand = a - alpha * grad

        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate
        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:

            best_score = cand_score
            a = cand
            no_improve = 0
            continue

        width = max(1, n // 200)
        start = rng.integers(0, n - width + 1)
