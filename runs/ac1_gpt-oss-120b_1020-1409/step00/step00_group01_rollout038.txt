def evaluate_sequence(sequence: list) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]

import numpy as np

def convolve_self_fft(a):
    n = a.shape[0]
    size = 1 << (2*n - 1).bit_length()
    # use FFT length size (next power of two)
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)[:2*n-1]
    return conv

def objective(a):
    n = a.size
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = convolve_self_fft(a)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed=42, num_steps=500, **kwargs):
    rng = np.random.default_rng(seed)
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)
    # Normalize sum to 1 to keep scale consistent (optional)
    a = parent.copy()
    # Ensure positive sum
    if np.sum(a) < 0.01:
        a = np.ones_like(a)
    # Normalize to sum 1 (any scale works)
    a = a / np.sum(a)
    # Keep best
    best = a.copy()
    best_score = objective(best)
    n = a.size
    for step in range(num_steps):
        # compute current conv and max index
        conv = convolve_self_fft(a)
        max_b = np.max(conv)
        k_max = int(np.argmax(conv))
        # choose modification type
        if rng.random() < 0.6:
            # gradient slice scaling
            i_start = max(0, k_max - (n - 1))
            i_end = min(n-1, k_max)
            length = i_end - i_start + 1
            # scaling factor random between 0.90 and 0.99
            s = float(rng.uniform(0.90, 0.99))
            slice_sum = np.sum(a[i_start:i_end+1])
            removed = (1.0 - s) * slice_sum
            a[i_start:i_end+1] *= s
            if length < n:
                delta = removed / (n - length)
                # distribute to outside indices
                # create mask for outside
                mask = np.ones(n, dtype=bool)
                mask[i_start:i_end+1] = False
                a[mask] += delta
        else:
            # random block perturbation akin to previous
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.08)))
            a[start:stop] *= factor
            if rng.random() < 0.3:
                a = 0.5 * (a + a[::-1])
        # clip and renormalize to stay within bounds and keep sum 1
        np.clip(a, 0.0, 1000.0, out=a)
        total = np.sum(a)
        if total < 1e-12:
            a = np.ones_like(a) / n
            total = np.sum(a)
        a = a / total  # renormalize sum to 1 (scale invariant)
        # evaluate
        score = objective(a)
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
    # return list of floats scaled back to original magnitude? scaling doesn't matter.
    # We keep values as chosen (sum=1). Ensure not all zero.
    return [float(x) for x in best * 1.0]  # no scaling
