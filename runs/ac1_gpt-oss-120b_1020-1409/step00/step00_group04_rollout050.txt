def max_convolution_fft(x):
    n = x.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    # use rfft
    fx = np.fft.rfft(x, n=N)
    prod = fx * fx
    conv = np.fft.irfft(prod, n=N)
    # conv is length N, we need first 2*n-1 elements
    return float(conv[:2*n-1].max())
```

import numpy as np

def propose_candidate(seed=42, num_steps=5000, **kwargs):
    """Random‑subset search for a low autocorrelation upper bound."""
    rng = np.random.default_rng(seed)

    # ----------- obtain a valid parent (fallback if missing) -------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        parent = np.clip(base, 0.0, 1000.0)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # target from the statement – stop early if we reach it
    TARGET = 1.50274349

    # ----------- FFT parameters (matrix‑free convolution) ---------------
    # length needed for full convolution is 2*n-1; use next power of two
    N_fft = 1 << ((2 * n - 1).bit_length())

    def approx_score(vec):
        """Fast approximate score using FFT based convolution."""
        # real‑FFT of the vector (zero‑padded to N_fft)
        f = np.fft.rfft(vec, n=N_fft)
        prod = f * f                      # element‑wise product = FFT of a*a
        conv = np.fft.irfft(prod, n=N_fft)[: 2 * n - 1]
        max_c = conv.max()
        s = vec.sum()
        if s == 0.0:
            return np.inf
        return 2.0 * n * max_c / (s * s)

    # ----------- random subset search ------------------------------------
    # we focus on relatively dense subsets where theory predicts J < 1.5
    K_low = 500
    K_high = min(800, n - 1)

    for _ in range(num_steps):
        K = int(rng.integers(K_low, K_high + 1))
        # sample K distinct indices uniformly
        idx = rng.choice(n, size=K, replace=False)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = 1.0

        # cheap approximation first; if promising, evaluate exactly
        score_est = approx_score(cand)
        if score_est < best_score:
