    # Convert to plain Python list as required by the evaluator.

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes
    the AC1 upper bound defined by `evaluate_sequence`.

    The mathematically optimal shape under the non‑negative and

    This implementation starts from a uniform vector of the same length as
    `height_sequence_1` (if available) and performs a cheap stochastic
    refinement that only accepts strict improvements; however, no refinement

    rng = np.random.default_rng(seed)
