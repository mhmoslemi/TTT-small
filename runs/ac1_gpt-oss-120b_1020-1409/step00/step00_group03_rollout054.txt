import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a monotone power‑law step function.
    The shape a_i = (i/(n-1))**p * 1000 (with p=5) gives a very low AC1 bound.
    Falls back to the provided height_sequence_1 (or a uniform vector) if needed.
    """
    # -------------------------------------------------------------
    # Determine the length n from the runtime parent if available.
    # -------------------------------------------------------------
    n = 256                     # safe default
    parent = np.full(n, 1.0)    # fallback parent
    try:
        arr = height_sequence_1            # runtime variable
        if isinstance(arr, np.ndarray) and arr.ndim == 1 and arr.size > 0:
            parent = np.clip(arr.astype(np.float64), 0.0, 1000.0)
            n = int(parent.shape[0])
    except Exception:
        pass  # keep defaults

    # -------------------------------------------------------------
    # Build the power‑law candidate (p > 1 reduces the AC1 bound).
    # -------------------------------------------------------------
    p = 5.0                                   # exponent (tunable)
    # avoid division by zero for n==1 (should never happen here)
    denom = max(1, n - 1)
    idx = np.arange(n, dtype=np.float64) / denom
    cand = np.power(idx, p) * 1000.0
    cand = np.clip(cand, 0.0, 1000.0)

    # -------------------------------------------------------------
    # Quick sanity check: keep the better of candidate and parent.
    # -------------------------------------------------------------
    try:
        sc_cand = float(evaluate_sequence(cand.tolist()))
        sc_par  = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(sc_cand) and (not np.isfinite(sc_par) or sc_cand < sc_par):
            best = cand
        else:
            best = parent
    except Exception:
        best = cand

    return best.tolist()
