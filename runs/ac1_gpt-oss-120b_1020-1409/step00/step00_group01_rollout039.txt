def fast_score(seq):
    s = seq.sum()
    if s <= 0:
        return np.inf
    # compute convolution via FFT
    fft_len = 1 << ((2*len(seq) - 1).bit_length())
    # use rfft for real inputs
    X = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(X * X, n=fft_len)[:2*len(seq)-1]
    max_b = conv.max()
    return 2.0 * len(seq) * max_b / (s * s)
```

import numpy as np

# optional SciPy imports – fall back to random search if unavailable
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself using FFT (O(n log n))."""
    n = x.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    X = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(X * X, n=fft_len)[: 2 * n - 1]
    return conv


def _fast_score(x: np.ndarray) -> float:
    """Score = 2*n*max(conv) / (sum(x)^2) using the fast FFT convolution."""
    s = float(x.sum())
    if s <= 0.0:
        return np.inf
    conv = _full_convolution(x)
    max_b = float(conv.max())
    return 2.0 * x.size * max_b / (s * s)


def _solve_lp(f: np.ndarray, M: float, idxs) -> np.ndarray | None:
    """Linear program: maximize sum(g) subject to f*g <= M on selected shifts."""
    n = f.size
    rows = len(idxs)
    # Build sparse A matrix (rows x n)
    data = []
    row_ind = []
    col_ind = []
    for r, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # f[i] multiplies g[j] where j = k-i
        # skip zero weights to keep matrix sparse
        for i in range(i_start, i_end + 1):
            w = f[i]
            if w != 0.0:
                j = k - i
                row_ind.append(r)
                col_ind.append(j)
                data.append(w)
    if rows == 0:
        return None
    A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)        # minimize -sum(g)  -> maximize sum(g)
    bounds = [(0.0, 1000.0) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")  # type: ignore[arg-type]
    if res.success:
        return np.asarray(res.x, dtype=np.float64)
    return None
