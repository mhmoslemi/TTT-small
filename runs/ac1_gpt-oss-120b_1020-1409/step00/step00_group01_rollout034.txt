def propose_candidate(seed=42, num_steps=2000, **kwargs):
    rng = np.random.default_rng(seed)
    # Initialize: use parent if given else uniform
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        a = np.clip(base, 0.0, 1000.0).copy()
    except NameError:
        n = 256
        a = np.full(n, 1.0, dtype=np.float64)
    # ensure non-zero sum
    if a.sum() < 0.01:
        a = np.ones_like(a)
    # baseline score
    best_score = float(evaluate_sequence(a.tolist()))
    best = a.copy()
    # parameters
    reduction_factor = 0.2  # fraction of value to move
    for step in range(num_steps):
        # compute conv and get max index
        conv = np.convolve(a, a)
        max_val = conv.max()
        k = int(conv.argmax())
        # compute contributions at k
        i_start = max(0, k - (len(a) - 1))
        i_end = min(k, len(a) - 1)
        i_indices = np.arange(i_start, i_end + 1)
        j_indices = k - i_indices
        contrib = a[i_indices] * a[j_indices]
        # find the pair with largest contribution
        idx = np.argmax(contrib)
        i = i_indices[idx]
        j = j_indices[idx]
        # decide amount to move
        di = reduction_factor * a[i]
        dj = reduction_factor * a[j]
        # ensure we don't zero out too much
        di = min(di, a[i])
        dj = min(dj, a[j])
        # pick a recipient with minimal value
        recipient = np.argmin(a)
        # create candidate copy
        cand = a.copy()
        cand[i] -= di
        cand[j] -= dj
        cand[recipient] += di + dj
        # clip to bounds
        np.clip(cand, 0.0, 1000.0, out=cand)
        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand
            a = cand  # continue from improved
        # else optionally random perturb occasionally to escape local minima
        elif rng.random() < 0.05:
            # random small Gaussian perturb for all entries
            cand = a * np.exp(rng.normal(0.0, 0.05, size=a.shape))
            np.clip(cand, 0.0, 1000.0, out=cand)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = cand
                a = cand
    return [float(x) for x in best]

import numpy as np

def propose_candidate(seed=42, max_iter=2000, reduction=0.2, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            raise Exception
    except NameError:
        a = np.ones(256, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.ones_like(a)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # maintain current conv for efficiency
    conv = np.convolve(a, a)
    for it in range(max_iter):
        # locate k where conv largest
        k = int(conv.argmax())
        max_val = conv[k]
        # indices that contribute
        i_start = max(0, k - (a.size - 1))
        i_end = min(k, a.size - 1)
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k - i_idx
        prod = a[i_idx] * a[j_idx]
        # strongest product
        best_pair = int(prod.argmax())
        i = int(i_idx[best_pair])
        j = int(j_idx[best_pair])
        # amount to move
        delta_i = reduction * a[i]
        delta_j = reduction * a[j] if j != i else 0.0
        # ensure we don't exceed available
        delta_i = min(delta_i, a[i])
        delta_j = min(delta_j, a[j])
        # target index: location where conv is minimal
        target = int(conv.argmin())
        # also make sure target not equal i or j
        while target in (i, j):
            # pick alternative low conv
            # random low index
            target = rng.integers(a.size)
        cand = a.copy()
        cand[i] -= delta_i
        if j != i:
            cand[j] -= delta_j
        cand[target] += delta_i + delta_j
        np.clip(cand, 0.0, 1000.0, out=cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand
            a = cand
            conv = np.convolve(a, a)
            continue
        # optional random perturbation to escape
        if rng.random() < 0.02:
            perturb = np.exp(rng.normal(0.0, 0.05, size=a.shape))
            cand2 = a * perturb
            np.clip(cand2, 0.0, 1000.0, out=cand2)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best = cand2
                a = cand2
                conv = np.convolve(a, a)
    return [float(x) for x in best]
