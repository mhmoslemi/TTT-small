    # get parent if exists

    # start with uniform baseline

    # optionally incorporate parent but uniform likely better
    # compute baseline score
    best_score = float(evaluate_sequence(best.tolist()))

    # also evaluate parent for fallback
    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # main search
    for _ in range(num_steps):
        candidate = best.copy()
        # choose operation
        op = rng.integers(0,3)
        if op == 0:
            # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.08)))
            candidate[start:start+width] *= factor
        elif op == 1:
            # swap two positions
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            candidate[i], candidate[j] = candidate[j], candidate[i]
        else:
            # mirror averaging to reduce reverse pairing
            i = int(rng.integers(0, n))
            j = n - 1 - i
            avg = (candidate[i] + candidate[j]) * 0.5
            candidate[i] = avg
            candidate[j] = avg

        # clip to bounds
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    # fallback if best_score is inf
    if not np.isfinite(best_score):
        # simple uniform fallback
        best = np.full(n, 1.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
    return [float(x) for x in best]

        else:
            # --- mirror averaging (reduces reverse‑pair overlap) -------------
            i = int(rng.integers(0, n))
            j = n - 1 - i
            avg = (cand[i] + cand[j]) * 0.5
            cand[i] = avg
            cand[j] = avg

        # clip to the allowed range
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate exact score
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Fallback in the (unlikely) event that nothing finite was found.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = np.full(n, 1.0, dtype=np.float64)

    return [float(x) for x in best]
```
