def compute_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    # Next power of 2 size:
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]
    max_b = np.max(conv)
    score = 2 * n * max_b / (sum_a * sum_a)
    return float(score)

import numpy as np

# ---------------- helper utilities ----------------

def _conv_score(seq: np.ndarray):
    """Return (score, full convolution) for a non‑negative sequence."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    if n == 0:
        return np.inf, np.array([], dtype=np.float64)

    total = np.sum(a)
    if total < 1e-12:
        return np.inf, np.array([], dtype=np.float64)

    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]    # a*a (full convolution)
    max_b = np.max(conv)
    score = 2.0 * n * max_b / (total * total)
    return float(score), conv


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator):
    """Scale a random contiguous block of `base`."""
    n = base.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.08)))   # modest random factor
    cand = base.copy()
    cand[start:start + width] *= factor
    if rng.random() < 0.2:                         # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _make_transfer_perturbation(base: np.ndarray, rng: np.random.Generator):
    """Move a fraction of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    src = base[i]
    if src <= 0.0:
        return base.copy()
    delta = float(rng.random() * src * 0.5)   # move up to 50 % of source
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _make_targeted_perturbation(base: np.ndarray, conv: np.ndarray,
                               rng: np.random.Generator):
    """Reduce the largest convolution contribution."""
    n = base.size
    k_max = int(np.argmax(conv))                     # index with max b_k
    # indices i that participate in this sum:
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    idx = np.arange(i_low, i_high + 1, dtype=np.int64)
    partner = k_max - idx
    contrib = base[idx] * base[partner]
    i_star = idx[int(np.argmax(contrib))]            # worst contributor
    factor = 0.7 + 0.2 * rng.random()                # between 0.7 and 0.9
    delta = base[i_star] * (1.0 - factor)
    cand = base.copy()
    cand[i_star] *= factor
    # add the removed mass to a random low‑value position
    low_candidates = np.where(cand < cand.mean())[0]
    if low_candidates.size == 0:
        j = int(rng.integers(0, n))
    else:
        j = int(rng.choice(low_candidates))
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _exp_candidate(k: float, n: int):
    """Exponential decay a_i = r**i with r = exp(-k/n)."""
    c = k / n
    r = np.exp(-c)
    return np.power(r, np.arange(n, dtype=np.float64))


def _linear_candidate(n: int):
    """Linearly decreasing from 1000 to a small value."""
    return np.linspace(1000.0, 0.0, n, dtype=np.float64)


# ---------------- main entry point ----------------

def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """Return a non‑negative height sequence with a low AC1 bound.

    The routine starts from the supplied `height_sequence_1` (if present),
    tries a few deterministic shapes, then performs a bounded local search
    using cheap FFT‑based scoring.
    """
    rng = np.random.default_rng(seed)

    # ----- initialise from the parent construction -----
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                     # absent or malformed
        parent = None

    if (parent is None or parent.ndim != 1 or parent.size == 0
            or not np.isfinite(parent).all()):
        n = 256
        best = np.full(n, 1.0, dtype=np.float64)
    else:
        n = parent.size
        best = np.clip(parent, 0.0, 1000.0)

    best_score, best_conv = _conv_score(best)

    # ----- deterministic candidates (exponential and linear) -----
    for k in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0):
        cand = _exp_candidate(k, n)
        cand = np.clip(cand, 0.0, 1000.0)
        s, _ = _conv_score(cand)
        if s < best_score:
            best, best_score, best_conv = cand, s, _

    cand = _linear_candidate(n)
    cand = np.clip(cand, 0.0, 1000.0)
    s, _ = _conv_score(cand)
    if s < best_score:
        best, best_score, best_conv = cand, s, _

    # ----- main stochastic local search -----
    for _ in range(num_steps):
        choice = rng.integers(0, 3)   # 0=block,1=transfer,2=targeted
        if choice == 0:
            cand = _make_block_perturbation(best, rng)
        elif choice == 1:
            cand = _make_transfer_perturbation(best, rng)
        else:
            cand = _make_targeted_perturbation(best, best_conv, rng)

        s, conv = _conv_score(cand)
        if s < best_score:
            best, best_score, best_conv = cand, s, conv

    # ----- final small smoothing passes (optional) -----
    for w in (3, 5):
        kernel = np.ones(w, dtype=np.float64) / w
        smooth = np.convolve(best, kernel, mode='same')
        smooth = np.clip(smooth, 0.0, 1000.0)
        s, _ = _conv_score(smooth)
        if s < best_score:
            best, best_score = smooth, s

    return [float(x) for x in best]
