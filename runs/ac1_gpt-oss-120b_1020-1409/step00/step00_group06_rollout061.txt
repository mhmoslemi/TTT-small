        def is_prime(num):
            if num < 2:
                return False
            if num % 2 == 0:
                return num == 2
            r = int(np.sqrt(num))
            for p in range(3, r+1, 2):
                if num % p == 0:
                    return False
            return True
        q = max_q

    # Build Sidon positions using construction i*q + (i*i)%q for i in range(q)

    a = np.full(n, b, dtype=np.float64)

import numpy as np

def propose_candidate(seed=42, num_trials=200, **kwargs):
    """
    Generate an anti‑symmetric step‑function sequence.
    The construction uses pairs (i, n‑1‑i) with opposite perturbations,
    which tends to flatten the self‑convolution and lower the AC1 bound.
    """
    rng = np.random.default_rng(seed)

    # Determine length from the provided parent if present.
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        n = 5582

    # Base level (any positive constant works because the objective is scale‑invariant).
    base = 1.0

    best_seq = None
    best_score = np.inf
    half = n // 2

    for _ in range(num_trials):
        # Random anti‑symmetric perturbations for each mirrored pair.
        # delta_i ∈ [‑base, base] guarantees non‑negativity after adding/subtracting.
        deltas = rng.uniform(-base, base, size=half)

        a = np.empty(n, dtype=np.float64)

        # Left half gets base + delta, right half gets base – delta in reverse order.
        a[:half] = base + deltas
        a[-half:] = base - deltas[::-1]

        # If length is odd, the centre element stays at the base value.
        if n % 2:
            a[half] = base

        # Enforce hard bounds required by the evaluator.
        np.clip(a, 0.0, 1000.0, out=a)

        # Evaluate the candidate.
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            continue

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            # Early exit if we already beat the target bound.
            if best_score < 1.5:
                # keep searching for an even better one but we have a good candidate.
                pass

    # Fallback: if all trials failed, reuse a safe uniform candidate.
    if best_seq is None:
        # Uniform sequence (any constant works; we use the parent if available).
        try:
            parent = np.asarray(height_sequence_1, dtype=np.float64)
            parent = np.clip(parent, 0.0, 1000.0)
            if parent.ndim == 1 and parent.size > 0:
                best_seq = parent
        except Exception:
            best_seq = np.full(n, base, dtype=np.float64)

    return best_seq.tolist()
