def _make_block_perturbation(base, rng):
    """Return a non‑negative block perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width between n/200 and n/12, at least 1
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetry averaging
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _random_swap(arr, rng):
    """Swap two random positions in a copy of ``arr``."""
    a = arr.copy()
    n = a.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    a[i], a[j] = a[j], a[i]
    return a


def _smooth_subsegment(arr, rng):
    """Replace a random sub‑segment by a simple local average."""
    a = arr.copy()
    n = a.size
    start = rng.integers(0, n)
    max_width = min(80, n - start)          # keep it modest
    width = rng.integers(1, max_width + 1)
    seg = a[start:start + width]
    # pad with edge values and average with neighbours
    padded = np.pad(seg, (1, 1), mode='edge')
    avg = (padded[:-2] + padded[2:]) * 0.5
    a[start:start + width] = avg
    return a


def _spread_sorted(arr):
    """Return a copy where sorted values are evenly spaced (modular stepping)."""
    n = arr.size
    vals = np.sort(arr)                     # ascending
    step = (n // 2) + 1                     # coprime to n (n is prime)
    out = np.empty_like(arr)
    pos = 0
    for v in vals:
        out[pos] = v
        pos = (pos + step) % n
    return out

    """Multiplicative log‑normal noise applied to every entry."""
    a = arr.astype(np.float64) * np.exp(rng.normal(0.0, sigma, size=arr.shape))
    return np.clip(a, 0.0, 1000.0)
