import numpy as np

def build_low_max_candidate(n, rng, L=2, attempts=20):
    best_arr = None
    best_score = np.inf
    for _ in range(attempts):
        order = rng.permutation(n)
        counts = np.zeros(2*n-1, dtype=np.int16)
        positions = []
        for idx in order:
            # quickly reject if self sum already maxed
            if counts[idx+idx] >= L:
                continue
            conflict = False
            for p in positions:
                if counts[idx + p] >= L:
                    conflict = True
                    break
            if conflict:
                continue
            # add idx
            positions.append(idx)
            counts[idx+idx] += 1
            for p in positions[:-1]:
                s = idx + p
                counts[s] += 1
        if not positions:
            continue
        arr = np.zeros(n, dtype=np.float64)
        arr[positions] = 1.0   # value 1 is enough, scaling irrelevant
        score = evaluate_sequence(arr.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_arr = arr.copy()
    return best_arr, best_score

import numpy as np

# ----------------------------------------------------------------------
# Helper: construct a sparse binary vector whose self‑convolution
# has a bounded maximum (each sum index is used at most `max_pairs`
# unordered times).  The construction is greedy and repeated a few
# times with different random orders; the best result (lowest score)
# is kept.
# ----------------------------------------------------------------------
def _greedy_sparse_candidate(length, rng, max_pairs=2, attempts=40):
    """Return a binary array of shape (length,) and its score.

    The returned array contains only 0.0 or 1.0 values.  At most
    `max_pairs` unordered pairs (i,j) share the same i+j sum,
    guaranteeing that the convolution peak is at most `2*max_pairs`.
    """
    best_arr = None
    best_score = np.inf

    # capacity bound: each unordered pair contributes to exactly one sum.
    # Using `max_pairs` as the limit on that capacity makes the problem
    # tractable while allowing a large number of non‑zero entries.
    for _ in range(attempts):
        order = rng.permutation(length)
        # counts of unordered pairs for each possible sum index (0 … 2*length‑2)
        pair_counts = np.zeros(2 * length - 1, dtype=np.int16)
        positions = []                     # indices kept with value 1

        for idx in order:
            # self‑pair i+i must also respect the limit
            if pair_counts[idx + idx] >= max_pairs:
                continue

            # check all sums with already chosen positions
            conflict = False
            for p in positions:
                if pair_counts[idx + p] >= max_pairs:
                    conflict = True
                    break
            if conflict:
                continue

            # accept this index
            positions.append(idx)
            pair_counts[idx + idx] += 1
            for p in positions[:-1]:                # previous positions only
                pair_counts[idx + p] += 1

        if not positions:
            continue

        arr = np.zeros(length, dtype=np.float64)
        arr[positions] = 1.0                     # height 1 is sufficient
        score = evaluate_sequence(arr.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_arr = arr.copy()

    return best_arr, best_score

def propose_candidate(seed=42, **kwargs):
    """Return a list of non‑negative heights with a low AC1 bound.

    The routine builds a sparse binary step‑function using a greedy
    algorithm that limits the number of unordered pairs per convolution
    index.  If the greedy phase fails to produce a valid vector, a simple
    exponential decay fallback is used.
    """
    rng = np.random.default_rng(seed)

    # Determine the problem size from the supplied parent (if any).
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256
        else:
            n = base.size
    except Exception:
        n = 256

    # --------------------------------------------------------------
    # Attempt the sparse construction (max 2 unordered pairs per sum)
    # --------------------------------------------------------------
    candidate, score = _greedy_sparse_candidate(
        length=n,
        rng=rng,
        max_pairs=2,          # each sum may host up to two unordered pairs
        attempts=50           # more attempts give a larger set with high prob.
    )

    # --------------------------------------------------------------
    # Fallback: a simple exponential‑decay profile (never worse than 2)
    # --------------------------------------------------------------
    if candidate is None or not np.isfinite(score):
        # exponential decay a_i = r^i, r chosen near the optimum for this n
        t_opt = 0.0002                     # empirically good for n≈7000
        r = np.exp(-t_opt)
        candidate = np.power(r, np.arange(n), dtype=np.float64)
        # scale to the allowed maximum (does not affect the score)
        candidate *= 1000.0 / candidate.max()

    # Ensure bounds and a tiny positive sum (the evaluator checks this)
    candidate = np.clip(candidate, 0.0, 1000.0)
    if candidate.sum() < 0.01:
        candidate = np.ones(n, dtype=np.float64)

    return [float(x) for x in candidate]
