def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # width between n/200 and n/12, at least 1 element
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional mirroring which often reduces peaks
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _gradient_step(seq: np.ndarray, orig_sum: float, lr: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    One gradient‐descent step that flattens the convolution.
    The loss is the squared deviation of the full convolution from its average.
    """
    n = seq.size
    # full self‑convolution
    conv = np.convolve(seq, seq)          # length 2n‑1
    # target = average of convolution (scale‑invariant)
    target = (np.sum(seq) ** 2) / (2 * n - 1)
    diff = conv - target                  # length 2n‑1
    # gradient of Σ(diff²) w.r.t. seq  (valid correlation gives length n)
    grad = 4.0 * np.correlate(diff, seq, mode='valid')
    # limit the maximal absolute change to a fraction of the mean value
    mean_val = np.mean(seq)
    max_change = max(lr * mean_val, 1e-8)          # avoid zero
    scale = max_change / (np.max(np.abs(grad)) + 1e-12)
    cand = seq - scale * grad
    np.clip(cand, 0.0, 1000.0, out=cand)

    # renormalise to keep the total sum (scale‑invariant objective)
    s = np.sum(cand)
    if s > 0.0:
        cand *= (orig_sum / s)
    else:                     # extremely unlikely, fall back to previous sequence
        cand = seq.copy()
    return cand
