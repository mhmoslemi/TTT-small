def _make_candidate(n, x=2.0, tail_frac=0.5, delta=0.3):
    """
    Build a candidate sequence of length n.

    Parameters
    ----------
    n : int
        Length of the sequence.
    x : float, default 2.0
        Controls the decay rate of the exponential backbone.
        The per‑step decay is epsilon = x / n.
    tail_frac : float, default 0.5
        Fraction of the vector (starting from the end) that receives a
        constant offset ``delta``.
    delta : float, default 0.3
        Size of the offset added on the tail.

    Returns
    -------
    np.ndarray
        The generated non‑negative sequence, clipped to the permitted
        range [0, 1000].
    """
    epsilon = x / n
    r = np.exp(-epsilon)                # decay factor
    idx = np.arange(n, dtype=np.float64)

    start = int(np.floor(tail_frac * n))
    cand = base.copy()
    cand[start:] += delta               # boost the tail
    return np.clip(cand, 0.0, 1000.0)
