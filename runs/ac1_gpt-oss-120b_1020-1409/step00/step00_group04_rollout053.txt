    Return a non‑negative step‑function (list of floats) that attempts to
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    The routine builds a uniform sequence (which attains the theoretical

    keeping any improvement.

    # constant vector.  The parent may be undefined in the execution
    # environment, so we guard against NameError.
    # ------------------------------------------------------------------

    # This gives the analytically minimal value of the objective (J = 2.0).
    # ------------------------------------------------------------------

    # candidate feasible.  Any improvement replaces the incumbent.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # block size: between 1 and about n/30 elements
        max_block = max(2, n // 30)
        block_len = int(rng.integers(1, max_block + 1))
        start = int(rng.integers(0, n - block_len + 1))
        stop = start + block_len

        # multiplicative factor ≈ 1 (log‑normal perturbation)
        factor = float(np.exp(rng.normal(0.0, 0.04)))

        cand = best.copy()
        cand[start:stop] *= factor
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise to keep the total mass (important for the objective)
        s = np.sum(cand)
        if s > 0.0:
            cand *= (total / s)
        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Defensive fallback – in the extremely unlikely event the best score
    # became non‑finite, revert to the uniform construction.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = uniform

    return [float(x) for x in best]
