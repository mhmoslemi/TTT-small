                    used[i + s] = True

import numpy as np

def _build_sidon_candidate(n, rng, max_trials=100):
    """
    Try to construct a large Sidon (B2) set in {0,…,n-1}.
    Returns a binary numpy array of length n (1.0 at chosen positions).
    """
    # target size that would already beat the required bound (≈1.5)
    required_m = int(np.ceil(np.sqrt(2 * n / 1.5)))
    best_set = []
    best_len = 0

    for _ in range(max_trials):
        # random order of indices
        order = rng.permutation(n)
        used = np.zeros(2 * n, dtype=bool)   # marks sums already taken
        cur = []
        for i in order:
            # self‑sum must be free
            if used[i + i]:
                continue
            # check all sums with already selected indices
            conflict = False
            for s in cur:
                if used[i + s]:
                    conflict = True
                    break
            if conflict:
                continue
            # accept i
            for s in cur:
                used[i + s] = True
            used[i + i] = True
            cur.append(i)
            # early stop if we already meet a sufficiently large size
            if len(cur) >= required_m and len(cur) > best_len:
                break
        if len(cur) > best_len:
            best_len = len(cur)
            best_set = cur
            # if we hit the theoretical maximum (≈sqrt(2n)) we can stop
            if best_len * best_len >= 2 * n:
                break

    # build the binary candidate vector
    cand = np.zeros(n, dtype=np.float64)
    if best_set:
        cand[best_set] = 1.0
    return cand


def propose_candidate(seed=42, **kwargs):
    """
    Return a candidate list of non‑negative heights.
    Tries to build a sparse Sidon set; falls back to a uniform vector.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    try:
        base = height_sequence_1  # supplied parent (may be missing)
        n = int(base.shape[0])
    except Exception:
        n = 256  # safe default if the parent is unavailable

    # Start with a simple uniform fallback (score = 2)
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # Try to improve by constructing a Sidon set
    candidate = _build_sidon_candidate(n, rng, max_trials=200)
    if candidate.size and np.any(candidate):
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    # As an extra safety net, if the Sidon construction failed to improve,
    # we keep the uniform vector.
    return [float(x) for x in best]
