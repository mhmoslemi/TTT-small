import numpy as np

def _is_prime(num: int) -> bool:
    """Return True if *num* is prime (simple deterministic test)."""
    if num < 2:
        return False
    if num % 2 == 0:
        return num == 2
    r = int(num ** 0.5)
    for d in range(3, r + 1, 2):
        if num % d == 0:
            return False
    return True


def _largest_prime_leq(x: float) -> int:
    """Largest prime ≤ ⌊x⌋ (x is small – a simple scan is fine)."""
    k = int(np.floor(x))
    while k > 2:
        if _is_prime(k):
            return k
        k -= 1
    return 2


def _sidon_indices(p: int) -> np.ndarray:
    """
    Singer construction of a Sidon set of size *p* inside [0, p²‑1]:
        i = a·p + (a² mod p) , a = 0,…,p‑1.
    """
    inds = np.empty(p, dtype=np.int64)
    for a in range(p):
        inds[a] = a * p + (a * a) % p
    return inds


def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound used by ``evaluate_sequence``.
    The construction uses a Singer Sidon set to keep the convolution
    maximum small while keeping the ℓ¹ norm reasonably large.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Length of the problem – the harness provides ``height_sequence_1``.
    # ------------------------------------------------------------------
    try:
        n = int(height_sequence_1.shape[0])
    except Exception:
        # Very unlikely, but keep a safe fall‑back.
        n = 256

    # ---------------------------------------------------------------
    # Build a Sidon set of size ≈ √n.
    # ---------------------------------------------------------------
    p = _largest_prime_leq(np.sqrt(n))          # a prime ≤ √n
    idx = _sidon_indices(p)                     # candidate indices
    idx = idx[idx < n]                           # keep those fitting in the vector
    m = idx.size                                 # number of “large’’ entries

    # ---------------------------------------------------------------
    # Choose a small value `s` so that the central convolution entry
    # (which mixes many small–small and large–small products) is ≈ 1.
    # Solve (n‑2m)·s² + 2·m·s = 1  →  s = (√(m² + n‑2m) – m)/(n‑2m)
    # Then back off a little to stay safely below the threshold.
    # ---------------------------------------------------------------
    a_coeff = n - 2 * m
    if a_coeff <= 0:               # degenerate case – fall back to a trivial vector
        cand = np.ones(n, dtype=np.float64)
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand.tolist()

    s_opt = (np.sqrt(m * m + a_coeff) - m) / a_coeff
    # safety factor – a few percent below the theoretical limit
    s_opt *= 0.98

    # also respect the obvious limits  s ≤ 1/m  and  s ≤ 1/√n
    s_opt = min(s_opt, 1.0 / m, 1.0 / np.sqrt(n))

    # ---------------------------------------------------------------
