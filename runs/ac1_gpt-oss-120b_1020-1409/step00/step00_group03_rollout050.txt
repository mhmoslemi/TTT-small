    selected = []
    used_sums = set()
    for candidate in range(n):
        # check if adding candidate creates duplicate sums
        ok = True
        for s in selected:
            if (candidate + s) in used_sums:
                ok = False
                break
        if (candidate + candidate) in used_sums:
            ok=False

            selected.append(candidate)

                # but sum set includes diagonal; duplicates already prevented.
        if len(selected) >= m:

    return selected

import numpy as np

# target that we want to beat
_TARGET = 1.50274349


def _greedy_sidon(n: int, need: int, order: np.ndarray) -> list:
    """
    Greedy construction of a Sidon‑type set (all unordered sums i+j are distinct)
    using the supplied order of indices. Returns the selected indices (may be fewer
    than *need* if the order prevents reaching it).
    """
    selected = []
    used_sums = set()                         # sums already realized
    for idx in order:
        # check whether adding idx would repeat a sum
        if (idx + idx) in used_sums:
            continue
        conflict = False
        for p in selected:
            if (idx + p) in used_sums:
                conflict = True
                break
        if conflict:
            continue
        # accept idx
        for p in selected:                    # sums with previous elements
            used_sums.add(idx + p)
        used_sums.add(idx + idx)              # its own double
        selected.append(idx)
        if len(selected) >= need:
            break
    return selected


def _make_candidate(n: int, indices: list) -> np.ndarray:
    """Return a length‑n array with 1.0 at the given indices, 0 elsewhere."""
    cand = np.zeros(n, dtype=np.float64)
    if indices:
        cand[indices] = 1.0
    # guarantee a non‑zero sum (safety)
    if cand.sum() < 0.01:
        cand[0] = 1.0
    return cand
