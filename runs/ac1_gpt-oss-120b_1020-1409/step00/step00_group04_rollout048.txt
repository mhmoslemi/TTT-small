        # decay step size
        eta *= decay
        # early stop if eta small
        if eta < 1e-6:
            break
    return [float(x) for x in best]

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _initial_vector():
    """Return a non‑negative start vector.
    Uses the runtime provided `height_sequence_1` when possible,
    otherwise falls back to a modest constant vector."""
    try:
        v = np.asarray(height_sequence_1, dtype=np.float64)
        if v.ndim != 1 or v.size == 0 or not np.isfinite(v).all():
            raise ValueError
    except Exception:
        v = np.ones(256, dtype=np.float64)
    v = np.clip(v, 0.0, 1000.0)
    # guarantee a minimal sum (evaluator rejects <0.01)
    s = v.sum()
    if s < 0.01:
        v *= 0.01 / max(s, 1e-12)
    return v


def _block_perturb(x, rng):
    """Sparse multiplicative block perturbation of ``x``."""
    n = x.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    y = x.copy()
    y[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        y = 0.5 * (y + y[::-1])
    np.clip(y, 0.0, 1000.0, out=y)
    # keep total mass roughly constant (scale back)
    s = y.sum()
    if s > 0:
        y *= x.sum() / s
    return y


def _gradient_step(a, eta):
    """One gradient‑descent step on the AC1 upper‑bound objective.
    Returns the updated vector."""
    n = a.size
    conv = np.convolve(a, a)               # length 2n‑1
    M = conv.max()
    k_star = int(conv.argmax())

    S = a.sum()
    if S <= 0:
        return a.copy()  # degenerate, shouldn't happen

    # compute g_i = a_i + a_{k*–i} for valid pairs, else 0
    idx = np.arange(n)
    j = k_star - idx
    valid = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    if valid.any():
        g[valid] = a[valid] + a[j[valid]]

    # subgradient of J = 2 n M / S**2
    subgrad = g - (2.0 * M / S) * a

    # take a step and project back into the feasible box
    a_new = a - eta * subgrad
    np.clip(a_new, 0.0, 1000.0, out=a_new)

    # keep the total sum away from zero (scaling does not change J)
    s_new = a_new.sum()
    if s_new < 0.01:
        a_new *= 0.01 / max(s_new, 1e-12)
    else:
        a_new *= a.sum() / s_new  # preserve the original sum (scale‑invariant)

    return a_new

                     decay=0.999, prob_perturb=0.07, **kwargs):
    """Return a list of non‑negative heights with a low AC1 upper bound.

    The routine mixes a simple block‑perturbation local search with a
    gradient‑descent step that directly targets the objective.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    a = _initial_vector()
    n = a.size
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    cur = a.copy()
    cur_score = best_score
    eta = float(eta0)

    # ------------------------------------------------------------------
    # iterative improvement
    # ------------------------------------------------------------------
    for it in range(max_iter):
        # Occasionally try a random block move (helps escape plateaus)
        if rng.random() < prob_perturb:
            cand = _block_perturb(cur, rng)
        else:

        # Exact evaluation (the evaluator does all clipping/summing checks)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best = cand.copy()
                best_score = score
                # early exit if we beat the target
                if best_score <= 1.50274349:

        if eta < 1e-8:

    # ------------------------------------------------------------------
    if not np.isfinite(best_score) or best.size == 0:

        best_score = float(evaluate_sequence(best.tolist()))

    return [float(v) for v in best]
