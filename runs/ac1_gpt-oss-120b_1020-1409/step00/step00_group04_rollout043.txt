    rng = np.random.default_rng(seed)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)
    # Ensure non-negative & clip
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size

    # if parent is dense uniform maybe can't improve; start with random sparse candidate

    # We'll init f as parent if parent has zeros else use sparse.
    if np.count_nonzero(parent) < n // 10:
        f = parent.copy()
    else:
        # create sparse random candidate
        f = np.zeros(n, dtype=np.float64)
        k = max(2, n // 20)  # 5% nonzero
        indices = rng.choice(n, size=k, replace=False)
        values = rng.uniform(0.1, 1000.0, size=k)
        f[indices] = values
    f = np.clip(f, 0.0, 1000.0)

    best = f.copy()

            active = idx_part

import numpy as np
import math

# try to import a solver; if unavailable we will fall back to random search
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full (2·n‑1) linear convolution of a with itself using FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _constraint_weights(f: np.ndarray, idx: int) -> np.ndarray:
    """
    Row of the Toeplitz matrix representing (f * g)[idx] = w·g.
    Only the part overlapping the support of f is non‑zero.
    """
    n = f.size
    w = np.zeros(n, dtype=np.float64)
    # i runs over indices of f that overlap when the shift is idx
    start_i = max(0, idx - (n - 1))
    end_i = min(idx, n - 1)
    i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
    j_vals = idx - i_vals                     # corresponding indices in g
    w[j_vals] = f[i_vals]
    return w


def _solve_lp(f: np.ndarray, M: float, active: np.ndarray):
    """Linear program: maximise sum(g) subject to (f*g)[k] <= M for k in active."""
    if cp is None:                     # pragma: no cover
        return None
    n = f.size
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active:
        w = _constraint_weights(f, int(k))
        # tiny tolerance avoids strict feasibility issues
        constraints.append(w @ g <= M + 1e-12)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=3000)
    except Exception:                 # pragma: no cover
        return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    if g.value is None:
        return None
    return np.array(g.value, dtype=np.float64)


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small O(n) perturbation used as a fallback when the LP fails."""
    cand = base.copy()
    n = cand.size
    # random block width between 1% and 15% of n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
