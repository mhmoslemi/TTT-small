import numpy as np

def _initial_uniform(n):
    """Return a constant non‑negative vector of length n."""
    return np.full(n, 1.0, dtype=np.float64)

def _initial_parent():
    """Return the provided parent sequence, clipped to the allowed range."""
    arr = np.asarray(globals().get("height_sequence_1", []), dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        # fallback of reasonable size if the parent is missing
        arr = np.ones(256, dtype=np.float64)
    np.clip(arr, 0.0, 1000.0, out=arr)
    return arr

def _perturb(candidate, rng):
    """Generate a small O(n) perturbation of ``candidate``."""
    n = candidate.size
    # random block width (at most n//20, at least 1)
    width = int(rng.integers(1, max(2, n // 20)))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative perturbation with modest log‑normal factor
    factor = float(np.exp(rng.normal(0.0, 0.05)))
    pert = candidate.copy()
    pert[start:stop] *= factor
    # optional mirroring to keep symmetry
    if rng.random() < 0.3:
        pert = 0.5 * (pert + pert[::-1])
    np.clip(pert, 0.0, 1000.0, out=pert)
    return pert

def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative sequence that (heuristically) minimizes the AC1 bound."""
    rng = np.random.default_rng(seed)

    # base candidates: uniform vector and the given parent
    n = int(globals().get("height_sequence_1", np.ones(1)).size)
    uniform = _initial_uniform(n)
    parent = _initial_parent()

    # evaluate both and keep the better one
    score_uniform = float(evaluate_sequence(uniform.tolist()))
    score_parent = float(evaluate_sequence(parent.tolist()))
    if score_uniform <= score_parent:
        best = uniform
        best_score = score_uniform
    else:
        best = parent
        best_score = score_parent

    # modest local search – a few cheap perturbations
    for _ in range(200):
        cand = _perturb(best, rng)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
