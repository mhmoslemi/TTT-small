def simplex_projection(v):
    # v: 1D numpy array (float). Return projected onto {x >=0, sum x =1}.
    # Using algorithm of Wang & Carreira-Perpinán (2013).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        # all negative, return uniform? set all to 1/n
        return np.full_like(v, 1.0/len(v))
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    g = np.zeros_like(p)
    # i range: max(0, idx_max - (n-1)) to min(n-1, idx_max)

def simplex_projection(v):
    # from https://doi.org/10.1007/s10107-009-0267-2
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        # set to uniform
        w = np.full_like(v, 1.0/len(v))
        return w
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

def _simplex_projection(v):
    """Euclidean projection of vector v onto the probability simplex."""
    # Algorithm from https://arxiv.org/abs/1101.6081
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:                         # all entries are zero or negative
        return np.full_like(v, 1.0 / v.size)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(p, size_fft):
    """Linear convolution p * p using an FFT of length size_fft."""
    ft = np.fft.rfft(p, size_fft)
    conv_full = np.fft.irfft(ft * ft, size_fft)
    return conv_full[: 2 * p.size - 1]

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Obtain a starting length (use the provided parent if it exists)
    # ------------------------------------------------------------------
    try:

        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.size
    except Exception:                         # fallback when no parent is available
        n = 256
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initialise a random probability vector (simplex) – breaking symmetry
    # ------------------------------------------------------------------
    init = rng.exponential(scale=1.0, size=n)
    p = init / init.sum()                     # sum == 1

    # --------------------------------------------------------------
    # Helper data for FFT‑based convolutions (size is next power of two)
    # --------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Keep the best candidate seen so far (including the supplied parent)
    # --------------------------------------------------------------

        """Exact score used by evaluate_sequence (sum == 1)."""
        return float(2.0 * n * max_conv)

    # evaluate the random start
    conv_vals = _conv_self(p, fft_len)
    best_score = score_from_max(conv_vals.max())
    best_p = p.copy()

    # also try the parent (normalised) as a fallback
    parent_clip = np.clip(parent, 0.0, 1000.0)
    s = parent_clip.sum()
    if s > 1e-12:

        conv_parent = _conv_self(p_parent, fft_len)
        score_parent = score_from_max(conv_parent.max())
        if score_parent < best_score:
            best_score = score_parent
            best_p = p_parent.copy()

    # --------------------------------------------------------------
    # Main optimisation loop – subgradient descent on max‑convolution
    # --------------------------------------------------------------
    step0 = 0.5                               # initial learning rate
    for it in range(num_iters):
        conv_vals = _conv_self(p, fft_len)
        max_val = conv_vals.max()
        idx_max = int(conv_vals.argmax())

        # store the best point found
        cur_score = score_from_max(max_val)
        if cur_score < best_score:

            best_p = p.copy()

    # ------------------------------------------------------------------
    out = np.clip(best_p, 0.0, 1000.0).tolist()
    return out
