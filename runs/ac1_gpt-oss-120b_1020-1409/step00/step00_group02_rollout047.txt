import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Produce a monotone linear ramp that achieves a low AC1 upper bound.
    The ramp runs from 0 to 1000 (inclusive) and has the same length as
    the supplied `height_sequence_1` when available; otherwise a fallback
    length of 256 is used.
    """
    # Determine the target length.
    try:
        n = int(height_sequence_1.shape[0])   # provided by the harness
    except Exception:
        n = 256                               # fallback length

    if n <= 1:
        return [0.0]

    # Linear ramp: a_i = i * (1000 / (n-1)), i = 0..n-1
    scale = 1000.0 / (n - 1)
    ramp = np.arange(n, dtype=np.float64) * scale

    # Safety clamp (should already be within bounds)
    ramp = np.clip(ramp, 0.0, 1000.0)

    return ramp.tolist()
