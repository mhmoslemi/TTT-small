    try:
        s = float(evaluate_sequence(candidate.tolist()))
        if not np.isfinite(s):
            raise RuntimeError
    except Exception:
        candidate = np.ones(n, dtype=np.float64)

    # Return a plain Python list.
    return [float(x) for x in candidate]
