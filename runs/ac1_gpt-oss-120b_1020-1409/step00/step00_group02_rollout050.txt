import numpy as np

def _shape_power(n, s):
    """Return a non‑negative weight vector of length n based on power decay."""
    i = np.arange(n, dtype=np.float64)
    # symmetric power decay towards the center
    w = (i + 1.0) ** (-s) + (n - i) ** (-s)
    return w

def _shape_alternating(n, c):
    """Return a symmetric alternating weight vector, c in [0,1)."""
    # even index -> +c, odd -> -c
    sign = 1.0 - 2.0 * (np.arange(n) % 2)
    w = 1.0 + c * sign
    # clip to non‑negative (c<1 guarantees positivity)
    return np.clip(w, 0.0, None)

    # evaluator expects a plain Python list

    """Return a step‑function that (empirically) beats the uniform 2.0 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0:
            n = base.size
        else:
            n = 256
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Baseline uniform vector (average value = 1, sum = n)
    uniform = np.ones(n, dtype=np.float64)

    best_seq = uniform.copy()
    best_score = _evaluate(best_seq)

    # ------------------------------------------------------------------
    # Parameter grids
    s_vals = [0.3, 0.6, 0.9, 1.2, 1.5, 2.0]
    c_vals = [0.2, 0.4, 0.6, 0.8]

    # Shapes based on a power law
    for s in s_vals:
        w = _shape_power(n, s)
        # Normalise to have the same total mass as the uniform vector (sum = n)
        w = w / w.sum() * n
        for mix in mix_vals:
            cand = (1.0 - mix) * uniform + mix * w
            cand = np.clip(cand, 0.0, 1000.0)
            score = _evaluate(cand)

    for c in c_vals:

        w = w / w.sum() * n

import numpy as np

def _shape_power(n, s):
    """Symmetric power‑law weights: large at the ends, small in the middle."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-s) + (n - i) ** (-s)
    return w

def _shape_exponential(n, s):
    """Symmetric exponential decay towards the centre."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-s * i / n) + np.exp(-s * (n - 1 - i) / n)
    return w

def _shape_alternating(n, c):
    """Alternating pattern 1±c (c<1 guarantees non‑negativity)."""
    sign = 1.0 - 2.0 * (np.arange(n) % 2)   # +1 on even, –1 on odd
    w = 1.0 + c * sign
    return np.clip(w, 0.0, None)

def _evaluate(seq):
    """Helper that forwards a NumPy array to the supplied evaluator."""
    # The evaluator expects a plain Python list.
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed=42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) whose AC1 score
    improves on the trivial uniform bound (score = 2).  The method builds a
    few analytically‑motivated probability‑like shapes, mixes them with the
    uniform vector, picks the best by exact evaluation and finishes with a
    short block‑wise local search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the problem instance.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0:
            n = int(base.size)
        else:
            n = 256
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Uniform vector (average value = 1, total sum = n).  This yields a score
    # of exactly 2 and will serve as a fallback.
    uniform = np.ones(n, dtype=np.float64)

    best_seq = uniform.copy()
    best_score = _evaluate(best_seq)

    # ------------------------------------------------------------------
    # Parameter grids for the analytic shapes.
    power_params = [0.3, 0.6, 0.9, 1.2]      # exponents for power‑law shape
    expo_params  = [2.0, 4.0]                # exponents for exponential shape
    alt_params   = [0.2, 0.5, 0.8]           # alternating amplitude (c < 1)
    mix_factors  = [0.0, 0.5, 1.0]           # 0 = uniform, 1 = shape only

    # ------------------------------------------------------------------
    # Helper to normalise a weight vector so that its sum equals n (the
    # uniform vector’s sum).  This keeps the overall scale identical across
    # all candidates – scaling does not affect the score.
    def _normalise_to_n(w):
        return w / w.sum() * n

    # ------------------------------------------------------------------
    # Power‑law shapes.
    for s in power_params:
        w = _shape_power(n, s)
        w = _normalise_to_n(w)
        for mix in mix_factors:
            cand = (1.0 - mix) * uniform + mix * w
            cand = np.clip(cand, 0.0, 1000.0)
            score = _evaluate(cand)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Exponential shapes.
    for s in expo_params:
        w = _shape_exponential(n, s)
        w = _normalise_to_n(w)
        for mix in mix_factors:
            cand = (1.0 - mix) * uniform + mix * w
            cand = np.clip(cand, 0.0, 1000.0)
            score = _evaluate(cand)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Alternating patterns.
    for c in alt_params:
        w = _shape_alternating(n, c)
        w = _normalise_to_n(w)
        for mix in mix_factors:
            cand = (1.0 - mix) * uniform + mix * w
            cand = np.clip(cand, 0.0, 1000.0)
            score = _evaluate(cand)
            if np.isfinite(score) and score < best_score:

                best_seq = cand.copy()
