import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function sequence that improves the
    AC1 upper bound.  The construction is based on the continuous kernel
    a(x)=x^{-1/2}, whose discrete analogue yields a nearly flat
    self‑convolution and a score close to π/2 ≈ 1.57, better than the
    uniform (score 2.0) baseline.

    The length of the sequence is taken from the provided runtime array
    `height_sequence_1`.  If that array is unavailable or malformed, a
    simple fallback of ones (length 256) is used.
    """
    # Determine length safely
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise ValueError
    except Exception:
        # Fallback length if parent not present or empty
        n = 256

    # a_i = 1 / sqrt(i+1)  –  vectorised, O(n) memory and time
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = 1.0 / np.sqrt(idx)

    # Enforce evaluator's hard bounds (they are already satisfied)
    seq = np.clip(seq, 0.0, 1000.0)

    # Ensure a minimal total mass (should always hold)
    if np.sum(seq) < 0.01:
        seq = np.ones(n, dtype=np.float64)

    # Return as a plain Python list of floats
    return seq.tolist()
