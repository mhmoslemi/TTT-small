import numpy as np

# -------------------------------------------------------------
# Helper: fast linear convolution of a real vector with itself
# -------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return a*a (full linear convolution) using a FFT of size power‑of‑2."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)                     # real FFT
    prod = fa * fa                                  # element‑wise square
    conv = np.fft.irfft(prod, n=size)[: 2 * n - 1]   # keep the linear part
    return conv

    # break perfect symmetry – a tiny random perturbation
    a += 0.002 * rng.random(a.size)
    a = np.clip(a, 0.0, None)
    a = a / a.sum()

    n = a.size
    best = a.copy()
    best_score = float("inf")

    # -------------------------------------------------------------
    # main optimisation loop
    # -------------------------------------------------------------
    for step in range(num_steps):
        conv = _conv_self(a)                 # O(n log n)
        sum_a = a.sum()
        max_c = conv.max()
        score = 2.0 * n * max_c / (sum_a * sum_a)

        # keep the best seen vector
        if score < best_score:
            best_score = score
            best = a.copy()
            # early exit if we already beat the target (optional)
            if best_score < 1.50274349:
                break

        # ---------------------------------------------------------
        # choose a convolution index that is above the average
        # ---------------------------------------------------------
        avg_c = (sum_a * sum_a) / (2 * n - 1)          # mean of full convolution
        excess = conv - avg_c
        excess[excess < 0] = 0.0
        total_ex = excess.sum()
        if total_ex == 0.0:          # already perfectly flat
            break

        # weighted random choice of a too‑large index
        r = rng.random() * total_ex
        k = np.searchsorted(np.cumsum(excess), r)
        # all i that contribute to this k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        if i_low > i_high:
            continue

        # pick a random contributing pair (i , j = k‑i)
        i = rng.integers(i_low, i_high + 1)
        j = k - i
        if not (0 <= j < n):
            continue

        # amount to move – a decreasing fraction of the smaller entry
        frac = 0.06 * (0.995 ** step)          # start ~6 %, shrink slowly
        delta = min(a[i], a[j]) * frac
        if delta <= 0.0:
            continue

        # pick two destination slots that are *not* i or j
        # (if the range covers the whole vector we simply pick any other)
        l1 = rng.integers(0, n)
        while l1 == i or l1 == j:
            l1 = rng.integers(0, n)
        l2 = rng.integers(0, n)
        while l2 == i or l2 == j or l2 == l1:
            l2 = rng.integers(0, n)

        # perform the mass transfer
        a[i] -= delta
        a[j] -= delta
        a[l1] += delta
        a[l2] += delta

        # keep everything inside the allowed box and re‑normalise
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s <= 0.0:                # defensive fallback
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    # -------------------------------------------------------------
    # final sanity: ensure the result is a plain Python list of floats
    # -------------------------------------------------------------
    return [float(x) for x in best]
```
