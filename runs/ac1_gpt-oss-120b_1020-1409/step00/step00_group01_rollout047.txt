def propose_candidate(seed=42, **kwargs):

    # compute scores
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))
    if parent is not None:
        parent_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(parent_score) and parent_score < best_score:
            best = parent
            best_score = parent_score
    # optionally try a few random perturbations near uniform or parent for safety
    for _ in range(200):
        # perturb maybe random small multiplicative factor in O(n) time
        candidate = best.copy()
        # choose block
        width = max(1, int(n * rng.uniform(0.001, 0.01))) # small width
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.05))
