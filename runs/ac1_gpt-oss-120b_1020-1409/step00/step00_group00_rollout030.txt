import numpy as np

# The evaluator `evaluate_sequence` is already available in the execution environment.

def _build_sparse_mask(n, K, max_allowed, rng, max_attempts=12):
    """
    Greedy construction of a binary mask of length n with exactly K ones.
    The convolution of the mask with itself (ordered pair counts) never exceeds max_allowed.
    Returns the mask (numpy uint8) on success, otherwise None.
    """
    conv_len = 2 * n - 1
    for _ in range(max_attempts):
        mask = np.zeros(n, dtype=np.uint8)
        conv = np.zeros(conv_len, dtype=np.int16)          # ordered‑pair counts
        indices = []                                      # positions already chosen

        # random order in which we try positions
        order = rng.permutation(n)
        for pos in order:
            # check self‑contribution
            if conv[2 * pos] + 1 > max_allowed:
                continue
            # check contributions with existing positions
            good = True
            for q in indices:
                if conv[pos + q] + 2 > max_allowed:   # two ordered pairs (p,q) and (q,p)
                    good = False
                    break
            if not good:
                continue

            # accept this position
            for q in indices:
                conv[pos + q] += 2
            conv[2 * pos] += 1
            mask[pos] = 1
            indices.append(pos)

            if len(indices) >= K:
                break

        if len(indices) >= K:
            return mask
    return None
