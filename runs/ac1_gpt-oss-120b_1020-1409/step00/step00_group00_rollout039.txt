def fast_score(arr, n):
    # arr: 1D np.ndarray
    N = 1 << ((2*n-1).bit_length())
    # compute convolution using FFT real
    # Use rfft for real input - returns N/2+1 complex numbers
    fa = np.fft.rfft(arr, N)
    conv = np.fft.irfft(fa * fa, N)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))

import numpy as np

# ------------------------------------------------------------
# Helper: fast convolution (full) using FFT – O(n log n) memory
# ------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.size
    # length needed for full convolution
    length = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, length)
    conv = np.fft.irfft(fx * fx, length)[: 2 * n - 1]
    return conv


def _score_from_max(n: int, max_val: float) -> float:
    """Exact AC1 upper bound from the maximal convolution entry."""
    return float(2.0 * n * max_val)


def _current_score(seq: np.ndarray) -> float:
    """Score using a fast convolution (identical to evaluate_sequence)."""
    conv = _full_convolution(seq)
    return _score_from_max(seq.size, conv.max())

def propose_candidate(seed: int = 42, num_steps: int = 2000, **kwargs):
    """
    Search for a non‑negative step‑function that reduces the
    AC1 upper bound.  The routine stays O(n) in memory and
    uses only FFT‑based convolutions.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Prepare a safe fallback (the supplied parent or a uniform vector)
    # ------------------------------------------------------------------
    try:
        parent = np.clip(
            np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0
        )
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
    except Exception:
        parent = None

    n = height_sequence_1.shape[0] if parent is not None else 256
    # start from a uniform distribution (sum = 1, shape = 1/n)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1) Break symmetry with a few random block perturbations
    # ------------------------------------------------------------------
    for _ in range(30):
        width = int(rng.integers(1, max(2, n // 20)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.1)))
        a[start : start + width] *= factor
        if rng.random() < 0.35:                     # occasional symmetrisation
            a = 0.5 * (a + a[::-1])
        a = np.clip(a, 0.0, 1000.0)
        total = a.sum()
        if total > 0:
            a /= total

    # ------------------------------------------------------------------
    # Keep the best sequence seen so far
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = _current_score(best)

    # ------------------------------------------------------------------
    # 2) Guided hill‑climbing: move mass from indices that participate
    #    in the current maximal convolution entry to indices that do not.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # full convolution and its maximum
        conv = _full_convolution(a)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # indices that affect the current maximum (i + j = k_max)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        mask = np.arange(n)
        in_mask = (mask >= low) & (mask <= high)
        out_mask = ~in_mask

        # if every index participates, fall back to a random pair move
        if not out_mask.any():
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
        else:
            i = int(rng.choice(mask[in_mask]))
            j = int(rng.choice(mask[out_mask]))

        src_val = a[i]
        if src_val <= 0.0:
            continue

        # transfer a modest fraction of mass from i to j
        delta = src_val * rng.uniform(0.01, 0.10)   # 1‑10 % move
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # enforce bounds and renormalise (preserves total sum)
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        a_candidate /= a_candidate.sum()

        cand_score = _current_score(a_candidate)

        # accept only if the AC1 bound improves
        if cand_score < best_score:
            a = a_candidate
            best = a_candidate.copy()
            best_score = cand_score

        # early exit if we already beat the target comfortably
        if best_score <= 1.6:
            break

    # ------------------------------------------------------------------
    # 3) Final tiny random refinements (optional)
    # ------------------------------------------------------------------
    for _ in range(200):

        width = int(rng.integers(1, max(2, n // 100)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        block = a[start : start + width].copy()
        block *= float(np.exp(rng.normal(0.0, 0.03)))
        a_candidate = a.copy()
        a_candidate[start : start + width] = block
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)

        cand_score = _current_score(a_candidate)

    # ------------------------------------------------------------------
    if best_score == float("inf") or not np.isfinite(best).all():
        if parent is not None:
            best = parent.copy()
        else:
            best = np.ones(n, dtype=np.float64) / n

    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
