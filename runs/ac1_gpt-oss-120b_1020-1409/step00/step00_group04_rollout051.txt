    rng = np.random.default_rng(seed)

    best = _initial_sequence()
    # exact evaluation – returns a finite float if the sequence is valid
    best_score = float(evaluate_sequence(best.tolist()))

    n = best.size
    # local search: move a small amount of mass from one index to another
    for _ in range(num_steps):
        i, j = rng.choice(n, size=2, replace=False)

        # randomly decide direction of the transfer
        if rng.random() < 0.5:
            amount = rng.uniform(0.0, min(best[i], 1000.0 - best[j]))
            if amount == 0.0:
                continue
            cand = best.copy()
            cand[i] -= amount
            cand[j] += amount
        else:
            amount = rng.uniform(0.0, min(best[j], 1000.0 - best[i]))
            if amount == 0.0:
                continue
            cand = best.copy()
            cand[j] -= amount
            cand[i] += amount

        # evaluate candidate; keep it only if it is strictly better
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score - 1e-12:
            best = cand
            best_score = score
            # stop early if we already beat the (unattainable) target
            if best_score < 1.5:
                break

    # return a plain Python list of floats as required
    return [float(x) for x in best]
