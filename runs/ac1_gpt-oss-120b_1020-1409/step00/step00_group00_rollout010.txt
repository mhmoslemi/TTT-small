import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a candidate sequence that (theoretically) minimizes the AC1 upper bound.
    The strategy: use a uniform (constant) sequence with the same total sum as
    the available parent (or a simple fallback). For non‑negative vectors the
    bound 2·n·max(b)/(sum(a)^2) is minimized by equal entries, achieving the
    optimal value 2.0.
    """
    # ----------------------------------------------------------------------
    # Obtain a parent sequence if it exists and is valid.
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        # Clamp to the allowed interval before using its sum.
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        # Fallback: a modest sized constant vector.
        parent = np.ones(256, dtype=np.float64)

    n = parent.size

    # ----------------------------------------------------------------------
    # Build a uniform candidate with the same total mass as the parent.
    # ----------------------------------------------------------------------
    total = float(np.sum(parent))
    # Ensure a non‑trivial total to avoid the evaluator's inf return.
    if total < 0.01:
        total = 0.01
    avg = total / n
    # Respect the per‑element upper bound.
    if avg > 1000.0:
        avg = 1000.0

    candidate = np.full(n, avg, dtype=np.float64)

    # ----------------------------------------------------------------------
    # Choose the better of the uniform candidate and the (clipped) parent.
    # ----------------------------------------------------------------------
    best = candidate
    best_score = float(evaluate_sequence(best.tolist()))

    parent_score = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent
        best_score = parent_score

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
