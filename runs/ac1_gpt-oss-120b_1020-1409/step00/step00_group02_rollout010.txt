import numpy as np

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by ``evaluate_sequence``.
    The routine works with the globally provided ``height_sequence_1`` if
    present, otherwise falls back to a default length.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise Exception
    except Exception:
        n = 256                      # safe fallback size

    # ------------------------------------------------------------------
    # Build elementary candidates
    # ------------------------------------------------------------------
    # Uniform candidate – provably optimal (score = 2.0)
    uniform = np.full(n, 1.0, dtype=np.float64)
    uniform = np.clip(uniform, 0.0, 1000.0)

    # Parent candidate if the global variable exists and is sane
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = None

    # ------------------------------------------------------------------
    # Initialise the best‑so‑far record using exact evaluation
    # ------------------------------------------------------------------
    best = uniform
    best_score = float(evaluate_sequence(best.tolist()))

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(p_score) and p_score < best_score:
            best = parent
            best_score = p_score

    # ------------------------------------------------------------------
    # Simple stochastic local search (O(n) memory, O(num_steps * n) time)
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        cand = best.copy()

        # Random block perturbation – O(n) work only on a sub‑interval
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor

        # Optional symmetric smoothing
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])

        # Apply the required bounds
        cand = np.clip(cand, 0.0, 1000.0)

        # Exact evaluation of the candidate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ------------------------------------------------------------------
    # Return the final candidate as a plain Python list of floats
    # ------------------------------------------------------------------
    return [float(v) for v in best]
