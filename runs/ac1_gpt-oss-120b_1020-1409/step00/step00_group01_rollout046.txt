def propose_candidate(seed=42, num_steps=0, **kwargs):

    # initialize base as constant vector
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            raise ValueError
    except Exception:
        # fallback: use length 256? but we need n; we can choose 256 if no height_sequence_1.
        n = 256
        init = np.full(n, 1.0, dtype=np.float64)
    # Determine length n
    n = init.shape[0]
    # constant candidate
    candidate = np.full(n, 1.0, dtype=np.float64)
    # Evaluate and keep best
    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # optional local search: block perturb
    for _ in range(num_steps):
        # produce perturbation of best: select random block width small
        width = max(1, int(rng.integers(1, n//10+1)))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.05))
        candidate = best.copy()
        candidate[start:start+width] *= factor
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score - 1e-12:
            best = candidate
            best_score = score
    # Return best as list
    return [float(x) for x in best]
