import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def monotone_desc(arr):
    """Return a non‑increasing copy of ``arr``."""
    # ``np.sort`` returns ascending order; reverse for descending.
    return np.sort(arr)[::-1]


def exponential_shape(alpha, n):
    """Geometric decay a_i = exp(-alpha * i)."""
    return np.exp(-alpha * np.arange(n, dtype=np.float64))


def powerlaw_shape(p, n):
    """Power‑law decay a_i = (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def linear_shape(frac, n):
    """Linear decay from 1 to 1-frac."""
    return np.linspace(1.0, max(0.0, 1.0 - frac), n, dtype=np.float64)


def safe_score(vec):
    """Evaluate a candidate, returning ``inf`` on any failure."""
    try:
        sc = float(evaluate_sequence(vec.tolist()))
        if np.isfinite(sc):
            return sc
    except Exception:
        pass
    return np.inf

# Main entry point expected by the harness
# -------------------------------------------------------------
def propose_candidate(seed=42, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the AC1 bound."""
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Determine problem size (fallback to a modest length if unavailable)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        n = base.size if base.size else 256
        # Use the given parent as an initial reference candidate
        start_vec = np.clip(base, 0.0, 1000.0)
    else:
        n = 256
        start_vec = np.ones(n, dtype=np.float64)

    # -------------------------------------------------------------
    # Keep the best candidate seen so far
    # -------------------------------------------------------------
    best = monotone_desc(start_vec.copy())
    best_score = safe_score(best)

    # -------------------------------------------------------------
    # 1️⃣  Grid search over a few simple parametric families
    # -------------------------------------------------------------
    # Exponential decay (alpha roughly c / n, c≈1.2 gives good values)
    alphas = np.logspace(np.log10(0.5 / n), np.log10(5.0 / n), num=30)
    for a in alphas:
        cand = monotone_desc(exponential_shape(a, n))
        sc = safe_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # Power‑law decay
    p_vals = np.linspace(0.1, 2.0, num=30)
    for p in p_vals:
        cand = monotone_desc(powerlaw_shape(p, n))
        sc = safe_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # Simple linear decay
    for f in np.linspace(0.0, 0.99, num=20):
        cand = monotone_desc(linear_shape(f, n))
        sc = safe_score(cand)
        if sc < best_score:
            best, best_score = cand, sc

    # -------------------------------------------------------------
    # 2️⃣  Local stochastic refinement (memory‑bounded, O(n) per step)
    # -------------------------------------------------------------
    local_steps = 400
    for _ in range(local_steps):
        cand = best.copy()

        # --- block‑wise multiplicative perturbation ---
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        factor = float(np.exp(rng.normal(0.0, 0.05)))  # modest change
