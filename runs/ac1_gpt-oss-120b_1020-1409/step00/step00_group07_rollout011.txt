def max_convolution(seq, n, L):
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    conv = conv[:2*n-1]   # ignore padded rest
    # due to floating errors, conv may have small negative: clip to 0.
    return conv.max()

def gradient_step(seq, rng):
    # compute convolution via FFT
    conv = compute_full_conv(seq)
    maxb = conv.max()
    kmax = conv.argmax()
    # compute gradient g
    n = seq.size
    g = np.zeros_like(seq)
    # i from max(0, kmax - (n-1)) to min(n-1, kmax)
    lo = max(0, kmax - (n-1))
    hi = min(n-1, kmax)
    i_vals = np.arange(lo, hi+1)
    j_vals = kmax - i_vals
    g[i_vals] = seq[j_vals]  # gradient
    # step magnitude
    # Choose eps random 0 to 0.2
    eps = rng.uniform(0.01, 0.2)
    # Build candidate as seq - eps * g (but ensure nonneg)
    delta = eps * g
    candidate = seq - delta
    candidate = np.clip(candidate, 0.0, 1000.0)
    # To maintain sum (optional), we can add the removed mass to random other positions:
    mass_removed = seq.sum() - candidate.sum()
    if mass_removed > 0:
        # add mass uniformly random across all indices
        add = rng.uniform(0.0, 1.0, size=seq.size)
        add = add / add.sum() * mass_removed
        candidate += add
    # clip again
    candidate = np.clip(candidate, 0.0, 1000.0)
    return candidate
```

import numpy as np

# ---------- helper functions ----------
def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """Return full linear convolution seq*seq using FFT (O(n log n))."""
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    return conv[: 2 * n - 1]


def _score(seq: np.ndarray) -> float:
    """Exact AC1 upper bound (same formula as evaluate_sequence)."""
    S = np.sum(seq)
    if S < 0.01:
        return np.inf
    maxb = np.max(_full_convolution(seq))
    return float(2.0 * seq.size * maxb / (S * S))

def propose_candidate(seed: int = 42, **kwargs):
    """
    Return a list of non‑negative step‑function heights whose AC1 bound
    is (hopefully) below the target 1.5027.
    The function follows the iterative LP‑based scheme described in the
    problem statement, with a cheap random‑search fallback.
    """
    rng = np.random.default_rng(seed)

    # ---- initial sequence -------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            base = np.ones(256, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    n = base.shape[0]

    # start from a non‑uniform random vector (exponential distribution)
    cur = rng.exponential(scale=1.0, size=n)
    cur = cur / np.sum(cur) * np.sum(base)           # keep overall mass similar
    cur = np.clip(cur, 0.0, 1000.0)

    best = cur.copy()
    best_score = _score(best)

    # ---- try a sparse‑LP improvement if CVXPY is available -----------------
    try:
        import cvxpy as cp
        cvxpy_ok = True
    except Exception:
        cvxpy_ok = False

    if cvxpy_ok:
        # number of active convolution points used as constraints
        K = 12
        # mixing weight (how much of the LP solution we blend in)
        t_mix = 0.5
        max_lp_iters = 20

        for _ in range(max_lp_iters):
            conv = _full_convolution(best)
            M = np.max(conv)

            # pick K indices with largest convolution values
            if K >= conv.size:
                active = np.arange(conv.size)
            else:
                active = np.argpartition(-conv, K - 1)[:K]

            # ----- build and solve the linear program -----------------------
            g = cp.Variable(n, nonneg=True)
            cons = []
            for k in active:
                k = int(k)
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                j_start = k - i_end
                j_end = k - i_start
                f_slice = best[i_start : i_end + 1]          # constant coefficients
                g_slice = g[j_start : j_end + 1]              # variable slice
                cons.append(cp.sum(cp.multiply(f_slice, g_slice)) <= M + 1e-12)

            prob = cp.Problem(cp.Maximize(cp.sum(g)), cons)
            try:

                prob.solve(solver=cp.SCS, warm_start=True, max_iters=2000)
