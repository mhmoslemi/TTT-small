import numpy as np

def linear_convolution(a):
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, size)
    C = np.fft.irfft(A * A, size)                  # circular convolution
    return C[: 2 * n - 1]                           # truncation → linear convolution


def max_lag_and_pair(a):
    """Return the lag where a*a attains its maximum and one pair (i,j) that
    realizes the largest product a[i]*a[j] for that lag."""
    conv = linear_convolution(a)
    k = int(np.argmax(conv))                        # lag with maximal value
    n = a.size
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    idx = np.arange(low, high + 1)                  # all i with valid partner
    partner = k - idx                               # j = k‑i
    products = a[idx] * a[partner]                  # contributions to this lag
    i_max = int(idx[np.argmax(products)])           # i that gives the biggest product
    j_max = int(k - i_max)                          # its partner
    return k, i_max, j_max


def random_block_perturbation(seq, rng):
    """A cheap O(n) perturbation used for exploration."""
    n = seq.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    out = seq.copy()
    out[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                         # occasional symmetrisation
        out = 0.5 * (out + out[::-1])
    np.clip(out, 0.0, 1000.0, out=out)
    return out

    """
    Produce a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the provided parent if possible.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0:
            a = np.clip(base, 0.0, 1000.0).copy()
        else:
            a = np.full(256, 1.0, dtype=np.float64)
    else:
        a = rng.uniform(0.5, 500.0, size=256).astype(np.float64)

    n = a.size
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    for _ in range(max_iters):
        # Identify the lag that is currently the bottleneck and a dominant pair.
        k, i_max, j_max = max_lag_and_pair(a)

        # Transfer a fraction of the two offending entries to a low‑impact slot.
        delta_i = a[i_max] * step_frac

        if delta_i + delta_j < 1e-9:               # nothing sensible to move

            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:

                best = cand.copy()
                best_score = score
