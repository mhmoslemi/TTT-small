def generate_sidonic_set(n, max_size):
    # greedy random with randomness seeded.
    rng = np.random.default_rng(seed)
    indices = np.arange(n)
    rng.shuffle(indices)
    S = []
    used_sums = set()
    for i in indices:
        # compute sums with existing S and itself
        collision = False
        for j in S:
            if (i + j) in used_sums:
                collision = True
                break
        if collision or (i + i) in used_sums:
            continue
        S.append(i)
        used_sums.add(i + i)
        for j in S[:-1]:  # previous elements
            used_sums.add(i + j)
        if len(S) >= max_size:
            break
    return np.array(S)

import numpy as np

def generate_sidonic_set(n, target, rng):
    # Random order of indices
    idx = rng.permutation(n)
    chosen = []
    used = set()
    for i in idx:
        # check collisions
        conflict = False
        # self sum
        s = i + i
        if s in used:
            conflict = True
        else:
            for j in chosen:
                if i + j in used:
                    conflict = True
                    break
        if conflict:
            continue
        # add i
        chosen.append(i)
        used.add(i + i)
        for j in chosen[:-1]:
            used.add(i + j)
        if len(chosen) >= target:
            break
    return np.array(chosen, dtype=np.int64)

import numpy as np

def generate_sidonic_set(length, target, rng):
    """
    Greedy construction of a Sidon set (all pairwise sums distinct).
    Returns up to ``target`` indices from ``0 .. length-1``.
    """
    # random order of candidate indices
    order = rng.permutation(length)
    chosen = []
    used_sums = set()
    for i in order:
        # check self-sum
        s = i + i
        if s in used_sums:
            continue
        conflict = False
        # check sums with already chosen elements
        for j in chosen:
            if i + j in used_sums:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        chosen.append(i)
        used_sums.add(s)
        for j in chosen[:-1]:
            used_sums.add(i + j)
        if len(chosen) >= target:
            break
    return np.array(chosen, dtype=np.int64)

    # Target number of “large” entries – roughly sqrt(n/2)
