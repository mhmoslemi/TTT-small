import numpy as np

def _uniform_candidate(length: int, total: float) -> np.ndarray:
    """Return a length‑`length` array with constant entries summing to `total`."""
    if length <= 0:
        return np.array([], dtype=np.float64)
    # Constant height; avoids division by zero.
    const = total / float(length)
    # Clip to the allowed range [0, 1000] – the clip is cheap and safe.
    const = max(0.0, min(1000.0, const))
    return np.full(length, const, dtype=np.float64)

    Produce a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound used by ``evaluate_sequence``.
    The strategy is to flatten the input into a uniform sequence, which is
    optimal for the bound under the non‑negativity constraints.
