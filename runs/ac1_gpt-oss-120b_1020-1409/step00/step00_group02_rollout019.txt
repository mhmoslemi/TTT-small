        n = len(height_sequence_1)        # use the supplied construction length
    except Exception:
        n = 256                           # fallback length if not provided

    # linear ramp 1, 2, …, n  scaled to fit the 0‑1000 range
    scale = 1000.0 / n
    seq = np.arange(1, n + 1, dtype=np.float64) * scale
    # safety clipping (should be a no‑op but guards against rounding)
    seq = np.clip(seq, 0.0, 1000.0)

    return seq.tolist()
