import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to minimise
    the AC1 upper bound evaluated by `evaluate_sequence`.

    The implementation:
      * determines the problem length from the provided `height_sequence_1`,
        falling back to a default length if that array is unavailable or invalid;
      * starts from the uniform vector (which is theoretically optimal for the
        given objective);
      * performs a cheap stochastic local search that respects the O(n) memory
        bound; any improvement is kept, otherwise the uniform starter is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------- #
    # Determine the length of the sequence we must produce.
    # --------------------------------------------------------------------- #
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = base.size
        else:
            raise ValueError
    except Exception:
        # Fallback length if the parent is missing or malformed.
        n = 256

    # --------------------------------------------------------------------- #
    # Uniform starter – this shape attains the theoretical lower bound of 2.
    # --------------------------------------------------------------------- #
    best = np.full(n, 1.0, dtype=np.float64)      # any positive constant works
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------------- #
    # Helper: generate a cheap block‑wise perturbation of a candidate.
    # --------------------------------------------------------------------- #
    def make_block_perturbation(vec):
        cand = vec.copy()
        # Choose a block width that scales with the problem size.
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        # Random multiplicative factor (log‑normal perturbation).
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor
        # Occasionally symmetrise the candidate – cheap exploration.
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        # Enforce the evaluator's hard bounds.
        np.clip(cand, 0.0, 1000.0, out=cand)
        return cand

    # --------------------------------------------------------------------- #
    # Simple stochastic descent – limited to a few hundred trials to keep runtime
    # modest while staying within linear memory usage.
    # --------------------------------------------------------------------- #
    for _ in range(500):
        cand = make_block_perturbation(best)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # Return a plain Python list as required.
    return [float(v) for v in best]
