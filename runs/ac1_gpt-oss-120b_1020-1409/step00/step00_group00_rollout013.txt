import numpy as np

def _score(seq):
    """Return the AC1 upper bound for a NumPy 1‑D array."""
    # evaluate_sequence expects a plain Python list of numbers
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed=42, num_steps=200, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that (approximately)
    minimizes the AC1 upper bound.

    The implementation starts from a uniform sequence – provably optimal for the
    bound – and performs a cheap random block perturbation search that only
    accepts improvements.  This guarantees a valid fallback (the uniform
    sequence) while staying within O(n) memory and time per iteration.
    """
    rng = np.random.default_rng(seed)

    # Determine the length of the problem.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256  # safe fallback if the global is unavailable

    # Uniform initialization (optimal under the bound).
    base = np.full(n, 1.0, dtype=np.float64)
    best = base.copy()
    best_score = _score(best)

    # Random local search – each step uses only O(n) auxiliary memory.
    for _ in range(num_steps):
        # Choose a random contiguous block.
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        # Perturb the block multiplicatively.
        candidate = best.copy()
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        candidate[start:stop] *= factor

        # Optional symmetry to keep the shape balanced.
        if rng.random() < 0.35:
            candidate = 0.5 * (candidate + candidate[::-1])

        # Enforce the problem’s bounds.
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # Evaluate and accept only genuine improvements.
        score = _score(candidate)
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    # Return a plain Python list of floats (the harness expects this).
    return [float(x) for x in best]
