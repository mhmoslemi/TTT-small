    except Exception:
        # Reasonable default length when the parent is unavailable.
        n = 256

    # Linear ramp: a_i = 1000 * i / (n-1), i = 0 … n-1
    # np.linspace includes both endpoints.
    candidate = np.linspace(0.0, 1000.0, n, dtype=np.float64)

    # Verify that the candidate is acceptable; otherwise use a safe fallback.
    try:
        score = float(evaluate_sequence(candidate.tolist()))
        if not np.isfinite(score):
            raise ValueError("Non-finite score")
    except Exception:
        # Uniform fallback (gives AC1 = 2) – always valid.
        candidate = np.full(n, 1000.0 / n, dtype=np.float64)

    # Return a plain Python list of floats as required.
    return [float(x) for x in candidate]
