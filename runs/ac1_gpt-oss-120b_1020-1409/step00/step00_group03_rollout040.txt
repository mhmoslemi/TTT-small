def full_convolution(a):
    n = a.size
    size = 2*n - 1
    # next power of two for fast FFT maybe.
    m = 1 << ((size-1).bit_length())
    # compute FFT
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    # due to numerical errors, clip small negative
    return conv

        step = step * 0.9995

    return [float(x) for x in best_a]

def _full_convolution(a):
    n = a.shape[0]
    size = 2*n - 1
    m = 1 << ((size-1).bit_length())
    # rfft works for real input
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    # Clip small negative due to roundoff
    conv = np.clip(conv, 0, None)
    return conv

import numpy as np

# The evaluator `evaluate_sequence` is already available in the execution environment.

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full convolution x*x using FFT (real‑to‑real)."""
    n = x.shape[0]
    size = 2 * n - 1
    # next power of two for fast FFT
    m = 1 << ((size - 1).bit_length())
    fx = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fx * fx, n=m)[:size]
    # small negative round‑off errors are clipped to zero
    return np.clip(conv, 0.0, None)


def _initial_shape(n: int) -> np.ndarray:
    """
    A well‑known shape a_i = 1/sqrt(i+1) which gives a relatively low
    max‑convolution (≈π) and therefore a low AC1 value.
    The vector is scaled to have the same ℓ₁‑norm that the evaluator
    implicitly uses (any scaling leaves the ratio unchanged).
    """
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    # scale to the customary sum √(2n) – the exact value does not matter,
    # the ratio is scale‑invariant, but keeping a fixed sum eases the LP.
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    return a


def propose_candidate(seed=42, **kwargs):
    """
    Return a list of non‑negative coefficients (length same as
    `height_sequence_1`) that attempts to minimise the AC1 bound.
    The core idea:
      * start from the analytic 1/√(i) shape,
      * iteratively enlarge the ℓ₁‑norm while keeping the maximum
        of the convolution bounded, using a small linear program,
      * keep the best sequence seen.
    If CVXPY is unavailable or the LP fails, the routine falls back to the
    analytic shape.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem (always the length of the given parent)
    n = len(height_sequence_1)

    # ---------- 1.  Analytic start (1/√i) ----------
    f = _initial_shape(n)
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ---------- 2.  Try an LP‑based improvement ----------
    try:
        import cvxpy as cp

        # parameters for the iteration
        iters = 3               # number of LP iterations
        mix = 0.5               # convex mixing weight (new = (1‑mix)*old + mix*g)

        for _ in range(iters):
            # current maximum of the self‑convolution
            conv_f = _full_convolution(f)
            M = float(conv_f.max())
            if not np.isfinite(M) or M <= 0:
                break

            # Linear program: maximise sum(g)  subject to  f * g <= M,  g >= 0
            f_param = cp.Parameter(n, value=f, name="f")
            g = cp.Variable(n, nonneg=True)

            # full convolution of the constant f and the variable g
            conv_fg = cp.conv(f_param, g)            # length 2n‑1
            constraints = [conv_fg <= M]

            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            # a modest iteration limit; the problem is small enough.
            prob.solve(solver="ECOS", max_iters=2000, feastol=1e-7,
                       reltol=1e-7, abstol=1e-7, verbose=False)

            if g.value is None:
                # LP failed – quit the refinement loop
                break

            g_val = np.asarray(g.value, dtype=np.float64)
            # keep values in the allowed range
            g_val = np.clip(g_val, 0.0, 1000.0)

            # convex combination of the old and the new vector
            candidate = (1.0 - mix) * f + mix * g_val
            candidate = np.clip(candidate, 0.0, 1000.0)

            # evaluate the candidate
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate.copy()
                best_score = score

            # prepare for the next iteration
            f = candidate.copy()

    except Exception:
        # either cvxpy is missing or something went wrong – we simply skip the LP.
        pass

    # ---------- 3.  Final safety fallback ----------
    # If the LP never succeeded and the analytic start is not good enough,
    # try a cheap random perturbation of the analytic shape.
    if best_score > 1.8:                               # heuristic threshold
        candidate = f.copy()
        for _ in range(200):
            # multiply a random block by a log‑normal factor
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            candidate[start:stop] *= factor
            if rng.random() < 0.35:
                candidate = 0.5 * (candidate + candidate[::-1])
            candidate = np.clip(candidate, 0.0, 1000.0)

            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate.copy()
                best_score = score

    # ---------- 4.  Return ----------
    # Ensure the output is a plain Python list of floats.
    return [float(v) for v in best]
