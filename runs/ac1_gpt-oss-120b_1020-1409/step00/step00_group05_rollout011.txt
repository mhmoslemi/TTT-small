import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a simple constant step‑function candidate.
    The evaluator's lower bound (by Cauchy‑Schwarz) is 2, which is achieved
    by any uniform non‑negative sequence.  This implementation therefore

        n = int(height_sequence_1.shape[0])

        # Fallback length if the parent is missing.
        n = 256

    # Uniform value; 1.0 is safe and yields the theoretical optimum 2.0.
