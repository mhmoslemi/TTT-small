def construct_sidon_set(n, max_size=None):
    sums = set()
    indices = []
    for i in range(n):
        if max_size is not None and len(indices) >= max_size:
            break
        conflict = False
        for j in indices:
            if i + j in sums:
                conflict = True
                break
        if conflict:
            continue
        # also check i+i sum
        if i + i in sums:
            continue
        # add i
        for j in indices:
            sums.add(i + j)
            sums.add(j + i)  # not needed separate but okay
        sums.add(i + i)
        indices.append(i)
    return indices
```

import numpy as np

# ----------------------------------------------------------------------
# Helper: construct a Sidon set (all pairwise sums are distinct)
# ----------------------------------------------------------------------
def construct_sidon_set(n, max_size=None):
    """
    Greedy construction of a Sidon set S ⊂ {0,…,n‑1}.
    No two unordered pairs (i,j) with i≤j have the same sum i+j.
    Returns a list of indices.
    """
    sums = set()
    S = []
    # limit the size if requested, otherwise let it grow as much as possible
    limit = max_size if max_size is not None else n
    for i in range(n):
        if len(S) >= limit:
            break
        # i+i must be new
        if (i + i) in sums:
            continue
        # check all existing sums i+j
        conflict = False
        for j in S:
            if (i + j) in sums:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        for j in S:
            sums.add(i + j)
        sums.add(i + i)
        S.append(i)
    return S
