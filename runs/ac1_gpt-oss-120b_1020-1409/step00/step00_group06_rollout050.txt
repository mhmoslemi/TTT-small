import numpy as np

def propose_candidate(seed=42, num_steps=500, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that (approximately) minimizes
    the AC1 upper bound 2·n·max(conv(a,a)) / (sum(a)^2).

    The routine builds a uniform sequence sharing the total mass of the provided
    `height_sequence_1` (or a fallback) and then performs a lightweight, memory‑
    bounded local search with multiplicative block perturbations.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Obtain a valid parent sequence (the harness supplies height_sequence_1)
    # ----------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    # Force the allowed interval [0,1000] used by the evaluator.
    parent = np.clip(parent, 0.0, 1000.0)

    # ----------------------------------------------------------------------
    # 2. Uniform baseline – the theoretical minimiser under the constraints.
    #    It keeps the total mass of the parent but spreads it evenly.
    # ----------------------------------------------------------------------
    n = parent.size
    total = float(parent.sum())
    if total < 0.01:                # avoid the trivial infeasible case
        total = 0.01
    avg = total / n
    best = np.full(n, avg, dtype=np.float64)
    best = np.clip(best, 0.0, 1000.0)

    # Evaluate once – this is the reference score.
    best_score = float(evaluate_sequence(best.tolist()))

    # ----------------------------------------------------------------------
    # 3. Very cheap stochastic local search.
    #    Each iteration perturbs a random block multiplicatively.
    #    Only improvements are kept.
    # ----------------------------------------------------------------------
    for _ in range(num_steps):
        # Choose a random block size (1 … n//20) and a start index.
        width = int(rng.integers(1, max(2, n // 20) + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))

        cand = best.copy()

        # Multiplicative perturbation of the selected block.
        factor = float(np.exp(rng.normal(0.0, 0.06)))   # modest log‑normal noise
        cand[start:start + width] *= factor

        # Occasionally symmetrise the block (helps occasional smoothing).
        if rng.random() < 0.2:
            block = cand[start:start + width]
            cand[start:start + width] = 0.5 * (block + block[::-1])

        # Enforce the hard bounds required by the evaluator.
        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score

    # ----------------------------------------------------------------------
    # 4. Return the best sequence found as a plain Python list.
    # ----------------------------------------------------------------------
    return [float(x) for x in best]
