import numpy as np

def propose_candidate(seed=42, num_steps=2000, sigma=0.03, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that minimizes the
    AC1 upper bound as evaluated by `evaluate_sequence`.

    The routine uses a simple hill‑climbing search starting from a uniform
    sequence of length matching `height_sequence_1` (or 256 as a fallback).
    """
    rng = np.random.default_rng(seed)

    # Determine length n from the provided parent if possible.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = int(base.shape[0])
    except Exception:
        n = 256

    # Uniform initial candidate (any constant works; 1.0 is convenient).
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # Ensure the initial candidate is valid; otherwise fall back to a tiny constant.
    if not np.isfinite(best_score):
        best = np.full(n, 0.01, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))

    # Simple stochastic local search.
    for _ in range(num_steps):
        # Perturb the current best candidate.
        cand = best + rng.normal(0.0, sigma, size=n)
        cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate the perturbed candidate.
        score = float(evaluate_sequence(cand.tolist()))
        # Keep it if it improves the AC1 bound.
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # Early exit if we somehow beat the target bound.
            if best_score < 1.50274349:
                break

    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
