def compute_ac(a):
    n = len(a)
    S = np.sum(a)
    if S < 1e-12: return np.inf
    # compute convolution via FFT
    size = 2*n-1
    m = 1 << ((size-1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa*fa, n=m)[:size]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (S*S)
```

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ---------- helper functions -------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute a*a (full convolution) using an FFT of size a power of two.
    Returns an array of length 2*len(a)-1.
    """
    n = a.size
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]
    return conv


def evaluate_ac(a: np.ndarray) -> float:
    """
    Compute the exact AC1 upper bound using the provided evaluator.
    a must be a 1‑D real array.
    """
    # evaluate_sequence is supplied by the harness
    return float(evaluate_sequence(a.tolist()))


def build_constraint_matrix(f: np.ndarray, ks: np.ndarray, M: float) -> csr_matrix:
    """
    Build the sparse matrix A_ub for the constraints  f * g <= M  at the
    shifts listed in ks.  Each row corresponds to one shift k.
    """
    n = f.size
    rows, cols, data = [], [], []
    for row, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals                     # indices of g participating
        coeffs = f[i_vals]                       # f[i] multiplies g[j]
        rows.extend([row] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(coeffs.tolist())
    return csr_matrix((data, (rows, cols)), shape=(len(ks), n))


# ---------- main search routine ----------------------------------------------
def propose_candidate(seed=42, max_outer_iters=12, t=0.2, thresh=0.99, **kwargs):
    """
    Return a list of non‑negative step heights whose AC1 bound is (hopefully)
    smaller than that of a uniform sequence.
    The algorithm follows the iterative linear‑programming scheme described in
    the problem statement, using only a handful of active convolution constraints.
    """
    rng = np.random.default_rng(seed)

    # ----- initial vector ----------------------------------------------------
    if "height_sequence_1" in globals():
        f = np.asarray(height_sequence_1, dtype=np.float64).copy()
        f = np.clip(f, 0.0, 1000.0)
        if f.ndim != 1 or f.size == 0 or not np.isfinite(f).all():
            f = np.full(256, 1.0, dtype=np.float64)
    else:
        f = np.full(256, 1.0, dtype=np.float64)

    n = f.size
    # Normalise to the conventional L1‑norm √(2n); scaling does not affect the score.
    target_sum = np.sqrt(2.0 * n)
    cur_sum = f.sum()
    if cur_sum == 0.0:
        f = np.full_like(f, target_sum / n)
    else:
        f = f * (target_sum / cur_sum)

    best = f.copy()
    best_score = evaluate_ac(best)

    # ----- iterative improvement ---------------------------------------------
    for _ in range(max_outer_iters):
        # current convolution and its maximum
        conv = full_convolution(f)
        M = conv.max()

        # choose the "active" shifts where the convolution is close to its peak
        active = np.where(conv >= M * thresh)[0]
        if active.size == 0:                         # safeguard
            active = np.array([np.argmax(conv)], dtype=int)

        # build the (sparse) linear constraints  f * g <= M
        A_ub = build_constraint_matrix(f, active, M)
        b_ub = np.full(len(active), M, dtype=np.float64)

        # linear programme: maximize sum(g)  (i.e. minimise -sum(g))
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:            # any solver failure ⇒ stop iterating
            break

        if not res.success:
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # Rescale g so that it has the same L1‑norm as the current f.
        scale = sum_f / sum_g
        g_scaled = g * scale

        # Convex combination update.
        f = (1.0 - t) * f + t * g_scaled
        np.clip(f, 0.0, 1000.0, out=f)   # enforce the problem’s bounds

        # occasional tiny random perturbation to avoid stagnation
        if _ % 5 == 0:
            width = max(1, n // 200)
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12, size=width))
            f[start:start + width] *= factor
            np.clip(f, 0.0, 1000.0, out=f)

        # evaluate and keep the best candidate seen so far
        cur_score = evaluate_ac(f)
        if np.isfinite(cur_score) and cur_score < best_score:
            best = f.copy()
            best_score = cur_score

    # final safety clip and conversion to plain Python list
    np.clip(best, 0.0, 1000.0, out=best)
    return [float(v) for v in best]
