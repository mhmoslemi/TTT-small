import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Return a simple uniform non‑negative step function.
    The sequence length matches the provided parent construction
    ``height_sequence_1`` when available; otherwise a fallback length of 256
    is used.  All coefficients lie in the allowed interval [0, 1000] and the
    sequence is guaranteed to be finite, non‑empty, and valid for the
    ``evaluate_sequence`` scorer.
    """
    # Determine length from the existing construction if possible.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError
        n = parent.shape[0]
    except Exception:
        # Fallback to a modest length when no parent is supplied.
        n = 256

    # Uniform value – any positive number ≤1000 works; 1.0 keeps the numbers small.
    candidate = np.full(n, 1.0, dtype=np.float64)

    # Ensure the array respects the evaluator’s clipping range.
    np.clip(candidate, 0.0, 1000.0, out=candidate)

    # Return a plain Python list of floats as required.
    return [float(x) for x in candidate]
