def propose_candidate(seed=42, num_steps=200, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        n = 5582
        parent = np.full(n, 1.0, dtype=np.float64)
    n = parent.size
    # start best as uniform 1.0 maybe.
    best = np.full(n, 1.0, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    # evaluate parent and maybe update best
    score_parent = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(score_parent) and score_parent < best_score:
        best = parent.copy()
        best_score = score_parent
    # iterative search
    for step in range(num_steps):
        # generate candidate
        # use smoothing of best or parent, random scaling:
        choice = rng.integers(0, 3)
        if choice == 0:
            # random block perturbation (like earlier)
            candidate = best.copy()
            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))
            stop = min(n, start + width)
            factor = np.exp(rng.normal(0.0, 0.12))
            candidate[start:stop] *= factor
        elif choice == 1:
            # smoothing
            candidate = (best + np.roll(best, 1)) * 0.5
        else:
            # uniform random values, scaled to average of best
            avg = best.mean()
            candidate = rng.random(n) * avg * 2 # up to 2*avg
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    return best.tolist()

import numpy as np

def _score(seq):
    """Return the AC1 bound for a sequence (list or 1‑D array)."""
    # evaluate_sequence is provided by the harness
    return float(evaluate_sequence(list(seq)))


def propose_candidate(seed=42, num_steps=12, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that aims to
    minimise the AC1 upper bound evaluated by ``evaluate_sequence``.
    The routine works with the parent sequence ``height_sequence_1`` if
    it exists; otherwise it falls back to a uniform construction.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Acquire a base length n.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        # No parent provided – use the documented length.
        n = 5582
        base = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Initialise the best candidate.
    #    A constant array is optimal for the ratio
    #    (its bound equals 2, the theoretical minimum).
    # ------------------------------------------------------------------
    best = np.full(n, 1000.0, dtype=np.float64)          # uniform at the upper bound
    best_score = _score(best)

    # ------------------------------------------------------------------
    # 3. Consider the supplied parent as a possible improvement.
    # ------------------------------------------------------------------
    parent = np.clip(base, 0.0, 1000.0)
    parent_score = _score(parent)
    if np.isfinite(parent_score) and parent_score < best_score:
        best = parent.copy()
        best_score = parent_score

    # ------------------------------------------------------------------
    # 4. Simple stochastic local search (very cheap – O(n) per step).
    #    Only a few steps are performed to stay well within time limits.
    # ------------------------------------------------------------------
    for _ in range(num_steps):
        # Choose a perturbation type at random.
        mode = rng.integers(0, 3)

        if mode == 0:
            # Block scaling: multiply a random contiguous block by a log‑normal factor.
            cand = best.copy()
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:stop] *= factor

        elif mode == 1:
            # Simple smoothing by averaging with the right neighbour.
            cand = (best + np.roll(best, -1)) * 0.5

        else:
            # Randomised values around the current mean.
            mean_val = best.mean()
            cand = rng.random(n) * mean_val * 2.0

        # Enforce bounds.
        cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate and keep if improvement.
        cand_score = _score(cand)
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score

    # ------------------------------------------------------------------
    # 5. Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(x) for x in best]
