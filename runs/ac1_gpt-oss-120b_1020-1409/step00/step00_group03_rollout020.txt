            candidate = make_block_perturbation(best, rng)
        else:
            # targeted shift:
            candidate = best.copy()
            # compute convolution
            conv = np.convolve(candidate, candidate)
            k_max = int(conv.argmax())
            # find indices involved in that max:
            # pairs (i,j) such that i+j=k_max. For each i compute j=k_max-i in [0,n-1].
            # We'll compute contributions a_i * a_j.
            # Choose one index i where a_i is > epsilon and also a_j > epsilon.
            # We'll take the larger of a_i and a_j, and reduce it a bit, add that mass to a random zero index.
            # Implementation: find all i where 0 <= i < n, 0 <= j = k_max - i < n.
            # Choose candidate indices.
            idxs = np.arange(n)
            js = k_max - idxs
            mask = (js >= 0) & (js < n)
            idx_i = idxs[mask]
            idx_j = js[mask]
            # compute contributions products
            prods = candidate[idx_i] * candidate[idx_j]
            # find index pair with max product
            if len(prods)==0:
                # fallback random
                candidate = make_block_perturbation(candidate, rng)
            else:
                # find argmax
                max_idx = int(np.argmax(prods))
                i = int(idx_i[max_idx])
                j = int(idx_j[max_idx])
                # decide to move a small epsilon delta from larger of a_i,a_j.
                # delta = epsilon fraction of that value
                # choose random delta fraction up to 0.1 of the value.
                # ensure values remain nonnegative.
                # also choose target index k not in {i,j} to receive delta.
                delta = candidate[max(i,j)] * rng.uniform(0.01, 0.1)
                # reduce from that entry
                src = max(i,j)
                candidate[src] = max(candidate[src] - delta, 0.0)
                # pick random target index not src
                target = int(rng.integers(0, n))
                while target == src:
                    target = int(rng.integers(0, n))
                candidate[target] = min(candidate[target] + delta, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    return best.tolist()

import numpy as np

# ----------------------------------------------------------------------
# Helper: simple block perturbation (same idea as the baseline program)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = np.array(base, dtype=np.float64).copy()
    n = candidate.size
    # choose a block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal scaling of the block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring to keep a bit of symmetry
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce evaluator limits
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


# ----------------------------------------------------------------------
# Helper: directed reduction of the current convolution peak
# ----------------------------------------------------------------------
def reduce_max_convolution(seq, rng):
    """Try to lower the largest convolution entry while keeping the L1 sum."""
    n = seq.size
    conv = np.convolve(seq, seq)                 # full convolution, length 2n‑1
    k_max = int(conv.argmax())                   # index where max occurs

    # all i contributing to this index satisfy 0 ≤ i < n and 0 ≤ k_max-i < n
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx_i = np.arange(i_min, i_max + 1, dtype=int)
    idx_j = k_max - idx_i

    # contribution of each pair (i, j) to the peak
    contrib = seq[idx_i] * seq[idx_j]
    if contrib.size == 0:
        return None
    max_pos = int(contrib.argmax())
    i = int(idx_i[max_pos])
    j = int(idx_j[max_pos])

    # pick a small amount to move from the larger participants
    min_val = seq[i] if i == j else min(seq[i], seq[j])
    if min_val <= 0.0:
        return None
    epsilon = min_val * rng.uniform(0.01, 0.10)   # 1‑10% of the smaller entry

    new_seq = seq.copy()
    new_seq[i] -= epsilon
    if i != j:
        new_seq[j] -= epsilon

    # place the removed mass at a random other position
    target = int(rng.integers(0, n))
    while target == i or target == j:
        target = int(rng.integers(0, n))
    add_amount = epsilon * (2 if i != j else 1)
    new_seq[target] += add_amount

    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq

    """Search for a coefficient list with a low AC1 upper bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    best = parent.copy()

    # ------------------------------------------------------------------
