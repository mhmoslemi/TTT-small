import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Construct a simple linear ramp (0 → 1000) which yields a
    provably low AC‑1 bound (≈4/3 for large n).  If a parent sequence
    `height_sequence_1` exists and happens to be better, keep it.
    Perform a few tiny random refinements as a fallback.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length n and a fallback parent sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)  # safe default
    parent = np.clip(parent, 0.0, 1000.0)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    n = parent.size
    if n < 2:                     # degenerate case – make a tiny array
        n = max(2, n)
        parent = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Build the linear ramp: a[i] = (i/(n‑1))*1000,  i=0…n‑1
    # ------------------------------------------------------------------
    max_height = 1000.0
    scale = max_height / (n - 1) if n > 1 else 0.0
    ramp = np.arange(n, dtype=np.float64) * scale
    ramp = np.clip(ramp, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Choose the better of parent and ramp as the starting candidate.
    # ------------------------------------------------------------------
    ramp_score = float(evaluate_sequence(ramp.tolist()))
    parent_score = float(evaluate_sequence(parent.tolist()))

    if np.isfinite(ramp_score) and ramp_score < parent_score:
        best = ramp
        best_score = ramp_score
    else:
        best = parent
        best_score = parent_score

    # ------------------------------------------------------------------
    # Tiny local search (optional – harmless and may improve a little).
    # ------------------------------------------------------------------
    steps = 200  # modest budget; each step evaluates the whole sequence.
    for _ in range(steps):
        cand = best.copy()

    # ------------------------------------------------------------------
