import math
import numpy as np

def make_block_perturbation(base, rng):
    """Create a cheap O(n) block perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width between 1 and about n/10
    width = int(rng.integers(1, max(2, n // 10)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(math.exp(rng.normal(0.0, 0.05)))  # small multiplicative jitter
    cand[start:stop] *= factor
    # occasional symmetry to keep the shape smooth
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    # This scaling makes ∑a = √(2 n) and yields the theoretical lower bound 2.
