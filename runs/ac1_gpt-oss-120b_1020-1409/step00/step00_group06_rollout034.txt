import numpy as np

def _conv_max_and_arg(a, fft_len):
    """Return (max value, argmax index) of full convolution a*a using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx
