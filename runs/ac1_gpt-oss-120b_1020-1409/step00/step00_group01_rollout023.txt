def redistribute_mass(seq, max_inds, rng):
    n = seq.shape[0]
    contrib = np.zeros(n, dtype=seq.dtype)
    for k in max_inds:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        contrib[idxs] += seq[k - idxs]
    # find where contribution highest, and lowest
    i_max = int(np.argmax(contrib))
    i_min = int(np.argmin(contrib))
    if seq[i_max] < 1e-12:
        return seq
    # Determine how much to move.
    #  percent of larger value to move, between small and large
    move_fraction = float(rng.uniform(0.02, 0.2))
    delta = seq[i_max] * move_fraction
    seq[i_max] -= delta
    seq[i_min] += delta
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq
```

import numpy as np

# ----------------------------------------------------------------------
# Helper: apply a small random block multiplicative perturbation
# ----------------------------------------------------------------------
def _block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = seq.size
    # choose a block width between 1 and n//100 (at least 1)
    max_width = max(2, n // 100)
    width = int(rng.integers(1, max_width))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.07)))  # modest log‑normal change
    seq[start:start + width] *= factor
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


# ----------------------------------------------------------------------
# Helper: move a fraction of mass from the index that contributes most
# to the current maximal autocorrelation towards the index that contributes
# the least.
# ----------------------------------------------------------------------
def _redistribute_mass(seq: np.ndarray, max_inds: np.ndarray,
                       rng: np.random.Generator) -> np.ndarray:
    n = seq.size
    # contribution of each coefficient to the (set of) maximal lag(s)
    contrib = np.zeros_like(seq)

    for k in max_inds:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        contrib[idx] += seq[k - idx]

    # pick donors / receivers
    i_max = int(np.argmax(contrib))
    i_min = int(np.argmin(contrib))

    if i_max == i_min or seq[i_max] <= 1e-12:
        return seq

    # transfer a random fraction of the donor's mass
    frac = float(rng.uniform(0.02, 0.15))
    delta = seq[i_max] * frac
    seq[i_max] -= delta
    seq[i_min] += delta
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, num_steps=2000, **kwargs):
    """Return a non‑negative step‑function (list of floats) that aims
    to minimise the AC1 bound evaluated by `evaluate_sequence`.
    The routine mixes small block perturbations with a simple
    redistribution heuristic that flattens the autocorrelation."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise from the provided parent construction (if available)
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                # fallback when the variable does not exist
        parent = np.ones(256, dtype=np.float64)

    # sanitise the parent
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    seq = np.clip(parent, 0.0, 1000.0)
    n = seq.size
    if seq.sum() < 0.01:               # guarantee a legal sum
        seq.fill(0.01 / n)

    # best solution encountered so far
    b = np.convolve(seq, seq)
    max_b = b.max()
    sum_a = seq.sum()
    best_score = 2.0 * n * max_b / (sum_a * sum_a)
    best_seq = seq.copy()

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    for step in range(num_steps):
        # evaluate current sequence
        b = np.convolve(seq, seq)
        max_b = b.max()
        sum_a = seq.sum()
        score = 2.0 * n * max_b / (sum_a * sum_a)

        # keep the best one seen so far
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

        # ----------------------------------------------------------------
        # Decide how to modify the sequence
        # ----------------------------------------------------------------
        if rng.random() < 0.25:                     # occasional block perturbation
            seq = _block_perturb(seq, rng)
        else:
            # indices achieving (near‑) maximal autocorrelation
            tol = max_b * 1e-12 + 1e-15
            max_inds = np.where(b >= max_b - tol)[0]
            seq = _redistribute_mass(seq, max_inds, rng)

    # final safety clip and conversion
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
