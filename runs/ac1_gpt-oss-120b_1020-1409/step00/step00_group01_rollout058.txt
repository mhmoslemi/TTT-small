import numpy as np


def _max_convolution(seq: np.ndarray, conv_len: int) -> float:
    """
    Return the maximum entry of the full auto‑convolution of ``seq``.
    The implementation uses an FFT of size ``conv_len = 2*len(seq)-1``,
    i.e. O(n log n) time and O(n) memory.
    """
    # real FFT of the zero‑padded sequence
    fft_seq = np.fft.rfft(seq, n=conv_len)
    # element‑wise product corresponds to convolution in the time domain
    conv = np.fft.irfft(fft_seq * fft_seq, n=conv_len)
    # ``conv`` is real; the maximum is the quantity we need
    return float(conv.max())
