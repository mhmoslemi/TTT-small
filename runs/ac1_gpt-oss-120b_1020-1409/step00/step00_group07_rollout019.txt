def propose_candidate(seed=42, num_steps=10, K=200, t=0.5, **kwargs):
    import numpy as np
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    rng = np.random.default_rng(seed)

    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        # ensure shape n.
    else:
        parent = np.ones(256, dtype=np.float64)

    # Clip to bounds
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.full(256, 1.0)
    parent = np.clip(parent, 0.0, 1000.0)

    # Ensure nonzero sum
    if parent.sum() < 0.01:
        parent = np.ones_like(parent) * 0.1

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    f = best.copy()
    n = f.size

    # Determine K at most 200 (or maybe 500)
    max_k = min(200, len(np.convolve(f, f)))
    # We'll compute conv each iteration anew.

    for iteration in range(num_steps):
        conv = np.convolve(f, f)
        M = conv.max()
        # Determine active indices
        # Choose indices where conv is among top K values.
        # Use argpartition:
        if max_k < conv.size:
            idx = np.argpartition(-conv, max_k - 1)[:max_k]
        else:
            idx = np.arange(conv.size)
        # Build constraint matrix
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(idx):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # j = i from i_start..i_end => coefficient f_i (where i = k - j)
            # Actually coefficient for g_j is f_{k-j}
            # So for j in i_start..i_end: coeff = f[k - j]
            # We'll loop over j.
            for j in range(i_start, i_end + 1):
                rows.append(row_idx)
                cols.append(j)
                data.append(f[k - j])
        A_ub = csr_matrix((data, (rows, cols)), shape=(len(idx), n))
        b_ub = np.full(len(idx), M, dtype=np.float64)

        c_obj = -np.ones(n, dtype=np.float64)  # maximize sum(g)

        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'disp': False})
        if not res.success:
            # break or continue with fallback perturbation
            break
        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0 or not np.isfinite(sum_g):
            break
        scale = sum_f / sum_g
        g = g * scale

        # Mix new candidate
        t_mix = 0.5  # could be smaller for stability
        f_new = (1 - t_mix) * f + t_mix * g
        f_new = np.clip(f_new, 0.0, 1000.0)
        score_new = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score_new) and score_new < best_score:
            best = f_new.copy()
            best_score = score_new
        # Update f for next iteration
        f = f_new

    return [float(x) for x in best]

    from scipy.sparse import csr_matrix
    _has_scipy = True
except Exception:
    _has_scipy = False


def make_block_perturbation(base, rng):
    """Simple O(n) random local perturbation."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def build_constraint_matrix(f, idx):
    """Return a sparse matrix A where (A g)[k] = sum_i f_i g_{k-i} for k in idx."""
    n = f.shape[0]
    rows, cols, data = [], [], []
    for row, k in enumerate(idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(row)
            cols.append(j)
            data.append(f[k - j])
    if rows:
        return csr_matrix((data, (rows, cols)), shape=(len(idx), n))
    return csr_matrix((0, n))

        M = conv_f.max()

        k_take = min(top_k, conv_f.size)
        if k_take < conv_f.size:

        else:
