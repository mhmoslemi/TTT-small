import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj_and_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    obj = (cmax + eps * np.log(ssum)) * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    g *= dx
    return obj, g

def _proj(v, n):
    target = n * 0.5
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    h = np.clip(v - theta, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += d / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _pgd(h0, n, eps, n_iter, deadline):
    """Projected gradient descent with backtracking"""
    h = h0.copy()
    step = 1.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        _, g = _smoobj_and_grad(h, eps)
        # Backtracking line search
        f0, _ = _smoobj_and_grad(h, eps)
        for _ls in range(30):
            h_try = _proj(h - step * g, n)
            f_try, _ = _smoobj_and_grad(h_try, eps)
            if f_try <= f0 - 0.5 * step * np.dot(g, h_try - h):
                break
            step *= 0.5
        else:
            step = 1e-6
            h_try = _proj(h - step * g, n)
            f_try, _ = _smoobj_and_grad(h_try, eps)
        h = h_try
    return h

def _fista(h0, n, eps, n_iter, deadline):
    """FISTA (accelerated projected gradient)"""
    h = h0.copy()
    z = h.copy()
    t_val = 1.0
    step = 1.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        fz, gz = _smoobj_and_grad(z, eps)
        # FISTA update
        v = z - (1.0 / step) * gz
        h_new = _proj(v, n)
        fh_new, _ = _smoobj_and_grad(h_new, eps)
        # Backtracking on step
        while fh_new > fz + np.dot(gz, h_new - z) + 0.5 * step * np.dot(h_new - z, h_new - z):
            step *= 0.5
            v = z - (1.0 / step) * gz
            h_new = _proj(v, n)
            fh_new, _ = _smoobj_and_grad(h_new, eps)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        z = _proj(h_new + ((t_val - 1.0) / t_new) * (h_new - h), n)
        h = h_new
        t_val = t_new
        step = max(step * 1.05, 1e-4)  # slowly increase step
    return h
