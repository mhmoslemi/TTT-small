For np.correlate(h, 1-h, mode='full'):
- Result has 2n-1 entries
- Entry at index k (0 to 2n-2) corresponds to lag = k - (n-1)
- For lag = l (positive): corr = sum_{i=0}^{n-1-l} h[i+l] * (1-h[i])
- For lag = -l (negative): corr = sum_{i=l}^{n-1} h[i] * (1-h[i-l])

For positive lag l:
corr = sum_{i=0}^{n-1-l} h[i+l] - sum_{i=0}^{n-1-l} h[i+l]*h[i]
= (number of 1s from position l to n-1) - (autocorr at lag l, restricted)

For binary h with exactly n/2 ones, if the sequence is "well-distributed", the number of 1s in any contiguous block of length n-l is approximately (n-l)/2.

So corr at positive lag l ≈ (n-l)/2 - autocorr(h, l) where autocorr(h,l) = sum h[i]*h[i+l].

To minimize corr, we want to MAXIMIZE autocorr at each lag. But we want to minimize the MAX over all lags.

This is similar to designing a binary sequence with minimum maximum autocorrelation (or minimum maximum "partial autocorrelation").

This is related to:
- Barker sequences (minimize max sidelobe of autocorrelation)
- m-sequences (have ideal periodic autocorrelation)

But our problem is slightly different because we're looking at the "partial" autocorrelation (not wrapping around) and also the "one-sided" count of 1s.

For a truly good sequence, we'd want:
- The 1s to be as evenly distributed as possible (so count in any window is ≈ window_length/2)
- The autocorrelation to be as low as possible at all lags (so 1s don't cluster at specific shifts)

These two goals can conflict, but for "Gold-like" sequences they work well together.

import numpy as np
import time

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h):
    n = len(h)
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    diff = target - int(h_b.sum())
    if diff > 0:
        order = np.argsort(-h)[:diff]
        h_b[order] = 1.0
    elif diff < 0:
        order = np.argsort(h)[:abs(diff)]
        h_b[order] = 0.0
    return h_b

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = rng.choice(n, size=target, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def grad_descent(h_init, time_budget, seed, lr=0.05, top_m=5):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n / 2.0
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_start = time.time()
    deadline = t_start + time_budget
    stall = 0
    step = 0
    while time.time() < deadline - 0.3:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_top = np.argsort(corr)[::-1][:top_m]
        # Compute gradient w.r.t. top-m lags
        grad = np.zeros(n)
        for k in idx_top:
            lag = k - (n - 1)
            if lag >= 0:
                grad[lag:] += (1.0 - h[:n - lag])
                grad[:n - lag] -= h[lag:]
            else:
                L = -lag
                grad[:n - L] += (1.0 - h[L:])
                grad[L:] -= h[:n - L]
        grad /= top_m
        v = 0.8 * v - lr * grad
        h_new = project_feasible(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
        h = h_new
        step += 1
        if stall > 100:
            lr *= 0.8
            if lr < 0.001:
                h = project_feasible(best_h + rng.randn(n) * 0.03, target)
                best_h = h.copy()
                best_c5 = compute_c5(h)
                v[:] = 0
                lr = 0.05
                stall = 0
    return best_h, best_c5

def ga_binary(h_init, time_budget, seed, pop_size=35):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n // 2
    pop = [h_init.copy()]
    for i in range(1, pop_size):
        if i < pop_size // 3:
            h_new = h_init.copy()
            k = rng.randint(1, 5)
            for _ in range(k):
                oi = np.where(h_new == 1.0)[0]
                zi = np.where(h_new == 0.0)[0]
                h_new[oi[rng.randint(len(oi))]] = 0.0
                h_new[zi[rng.randint(len(zi))]] = 1.0
            pop.append(h_new)
        else:
            pop.append(make_random_binary(n, seed + i * 17))
    fitness = np.array([compute_c5(h) for h in pop])
    best_c5 = float(fitness.min())
    best_h = pop[int(np.argmin(fitness))].copy()
    t_start = time.time()
    deadline = t_start + time_budget
    while time.time() < deadline - 0.5:
        order = np.argsort(fitness)
        pop = [pop[i] for i in order]
        fitness = fitness[order]
        if fitness[0] < best_c5:
            best_c5 = float(fitness[0])
            best_h = pop[0].copy()
        n_elite = max(2, pop_size // 5)
        new_pop = [pop[i].copy() for i in range(n_elite)]
        while len(new_pop) < pop_size:
            c1 = pop[rng.randint(0, min(8, len(pop)))]
            c2 = pop[rng.randint(0, min(8, len(pop)))]
            mask = rng.random(n) > 0.5
            child = np.where(mask, c1, c2).astype(float)
            s = int(child.sum())
            if s > target:
                oi = np.where(child == 1.0)[0]
                child[rng.choice(oi, s - target, replace=False)] = 0.0
            elif s < target:
                zi = np.where(child == 0.0)[0]
                child[rng.choice(zi, target - s, replace=False)] = 1.0
            for _ in range(rng.randint(2, 5)):
                oi = np.where(child == 1.0)[0]
                zi = np.where(child == 0.0)[0]
                child[oi[rng.randint(len(oi))]] = 0.0
                child[zi[rng.randint(len(zi))]] = 1.0
            new_pop.append(child)
        pop = new_pop[:pop_size]
        fitness = np.array([compute_c5(h) for h in pop])
    return best_h, best_c5

import numpy as np
import time

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h):
    n = len(h)
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    diff = target - int(h_b.sum())
    if diff > 0:
        order = np.argsort(-h)[:diff]
        h_b[order] = 1.0
    elif diff < 0:
        order = np.argsort(h)[:abs(diff)]
        h_b[order] = 0.0
    return h_b

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = rng.choice(n, size=target, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def grad_descent(h_init, time_budget, seed, lr=0.04, top_m=5):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n / 2.0
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_start = time.time()
    deadline = t_start + time_budget
    stall = 0
    while time.time() < deadline - 0.3:
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_top = np.argsort(corr)[::-1][:top_m]
        grad = np.zeros(n)
        for k in idx_top:
            lag = k - (n - 1)
            if lag >= 0:
                grad[lag:] += (1.0 - h[:n - lag])
                grad[:n - lag] -= h[lag:]
            else:
                L = -lag
                grad[:n - L] += (1.0 - h[L:])
                grad[L:] -= h[:n - L]
        grad /= top_m
        v = 0.8 * v - lr * grad
        h_new = project_feasible(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 80:
            lr *= 0.85
            if lr < 0.002:
                h = project_feasible(best_h + rng.randn(n) * 0.04, target)
                best_h = h.copy()
                best_c5 = compute_c5(h)
                v[:] = 0.0
                lr = 0.04
                stall = 0
    return best_h, best_c5

def sa_fractional(h_init, time_budget, seed, t_init=0.04, nscale=0.05):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n / 2.0
    h = project_feasible(h_init.copy(), target)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    t_start = time.time()
    deadline = t_start + time_budget
    T = t_init
    stall = 0
    while time.time() < deadline - 0.3:
        k = rng.randint(3, 12)
        idx = rng.choice(n, k, replace=False)
        h_new = h.copy()

        h_new = project_feasible(h_new, target)
        new_c5 = compute_c5(h_new)
        delta = new_c5 - cur_c5

            h = h_new
            cur_c5 = new_c5

                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:

            stall = 0

            cur_c5 = compute_c5(h)
            T = t_init
            stall = 0
    return best_h, best_c5

def ga_binary(h_init, time_budget, seed, pop_size=35):

    target = n // 2
    pop = [h_init.copy()]

                h_new[oi[rng.randint(len(oi))]] = 0.0

            pop.append(h_new)
        else:
            pop.append(make_random_binary(n, seed + i * 17))
    fitness = np.array([compute_c5(h) for h in pop])

    t_start = time.time()

            new_pop.append(child)
        pop = new_pop[:pop_size]

    return best_h, best_c5
