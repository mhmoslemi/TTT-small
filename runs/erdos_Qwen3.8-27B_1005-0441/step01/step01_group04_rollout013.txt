            c5 = _eval_c5(h_r)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_r.copy()

    if time.time() < t0 + budget_s - 5:
        h_final = _refine_mm(best_h, 400)
        h_final = _fix_sum(h_final, len(h_final))
        c5 = _eval_c5(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
