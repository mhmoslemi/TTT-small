def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

_g_temp = 0.01

def _soft_obj(x):
    n = _g_n
    t = max(_g_temp, 1e-10)
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    k = max(5, n // 4)
    idx = np.argsort(corr)[-k:]
    top = corr[idx]

    corr = np.correlate(x, 1.0 - x, mode="full")

    target = float(n) / 2.0
    s = np.sum(x)

        x = np.full(n, 0.5)
        return x

        x = np.clip(x, 0.0, 1.0)
        s = np.sum(x)

    gen = 0
    stall = 0

        gen += 1
        order = np.argsort(vals)
        half = pop // 2
        # Elitism: keep top half
        new_pop = [pop_list[order[i]].copy() for i in range(half)]
        # Generate offspring
        while len(new_pop) < pop:
            # Tournament select
            ts = np.random.choice(half, size=min(3, half), replace=False)
            parent_i = order[ts[np.argmin([vals[order[t]] for t in ts])]]
            parent = pop_list[parent_i]
            # Crossover
            if np.random.rand() < 0.4 and half >= 2:
                ts2 = np.random.choice(half, size=min(2, half), replace=False)
                parent2 = pop_list[order[ts2[1]]]
                a = np.random.beta(5, 5)
                child = a * parent + (1-a) * parent2
            else:
                child = parent.copy()
            # Mutate
            mt = np.random.rand()
            if mt < 0.6:
                child += sigma * np.random.randn(n)
            elif mt < 0.8:
                si = np.random.randint(0, n-3)
                ln = np.random.randint(2, max(3, n//8))
                child[si:si+ln] += np.random.uniform(-0.15, 0.15)
            else:
                # Swap two blocks
                b1 = np.random.randint(0, n//2)
                b2 = np.random.randint(0, n//2)
                ln1 = np.random.randint(1, max(2, n//10))
                ln2 = np.random.randint(1, max(2, n//10))
                child[b1:b1+ln1] += 0.1 * np.random.randn(ln1)
                child[b2:b2+ln2] -= 0.1 * np.random.randn(ln2)
            new_pop.append(_project(child, n))
        
        pop_list = new_pop
        vals = [_compute_c5(p) for p in pop_list]
        
        nb = int(np.argmin(vals))
        if vals[nb] < best_v - 1e-13:
            best_h = pop_list[nb].copy()
            best_v = vals[nb]
            stall = 0
            sigma = max(0.015, sigma * 0.97)
        else:
            stall += 1
            sigma = min(0.12, sigma * 1.03)
        
        if stall > 60:

                pop_list[idx] = _project(h_best + (0.05+0.1*np.random.rand())*np.random.randn(n), n)
            vals = [_compute_c5(p) for p in pop_list]
            stall = 0
