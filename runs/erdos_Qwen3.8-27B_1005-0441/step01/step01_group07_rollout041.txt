def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        if proj.sum() > target_sum:
            lo = mid
        else:

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def make_binary(frac_h, target_ones):
    n = len(frac_h)
    h = np.zeros(n)
    idx = np.argsort(frac_h)[::-1][:target_ones]
    h[idx] = 1.0
    return h

def make_pattern(n, seed, ptype):
    rng = np.random.RandomState(seed)
    target = n // 2
    if ptype == 0:
        return np.full(n, 0.5)
    elif ptype == 1:
        phi = (1.0 + math.sqrt(5.0)) / 2.0
        pos_set = set()
        positions = []
        for i in range(4 * n):
            p = int((i * phi * 1.618) % n)
            if p not in pos_set and len(positions) < target:
                pos_set.add(p)
                positions.append(p)
        if len(positions) < target:
            for x in range(n):
                if x not in pos_set and len(positions) < target:
                    positions.append(x)
        h = np.zeros(n)
        h[positions] = 1.0
        return h
    elif ptype == 2:
        return make_binary(rng.rand(n), target)
    elif ptype == 3:
        h = np.tile([1.0, 0.0], n // 2 + 1)[:n].copy()
        for _ in range(max(1, n // 6)):
            h[rng.randint(n)] = 1.0 - h[rng.randint(n)]
        return make_binary(h, target)
    elif ptype == 4:
        freq = rng.randint(1, n // 4 + 1)
        x = np.arange(n)
        h = (np.sin(2.0 * math.pi * freq * x / n + rng.rand() * 6.28) > 0.0).astype(float)
        return make_binary(h, target)
    elif ptype == 5:
        nb = rng.randint(3, max(4, n // 8 + 1))
        h = np.zeros(n)
        for bi in range(nb):
            start = int(bi * n / nb)
            end = int((bi + 1) * n / nb)
            if bi % 2 == 0:
                h[start:end] = 1.0
        return make_binary(h, target)
    return np.full(n, 0.5)

def focused_grad_desc(h_init, target_sum, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5_value(h)
    v = np.zeros(n)
    lr = 0.05
    t0 = time.time()
    deadline = t0 + time_budget
    no_imp = 0
    while time.time() < deadline - 0.3:
        if no_imp > 150:
            h = project_feasible(best_h + rng.randn(n) * 0.025, target_sum)
            v[:] = 0.0
            lr = 0.05
            no_imp = 0
        elif no_imp > 60:
            lr = max(lr * 0.93, 0.002)
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_k = int(np.argmax(corr))
        lag = worst_k - (n - 1)
        if lag == 0:
            mc = corr.max()
            shifted = np.clip(40.0 * (corr - mc), -300, 300)
            exps = np.exp(shifted)
            wts = exps / exps.sum()
            grad = np.zeros(n)
            for k in range(len(corr)):
                if wts[k] < 1e-10:
                    continue
                kk = k - (n - 1)
                if kk >= 0:
                    grad[kk:] += wts[k] * (1.0 - h[:n-kk])
                    grad[:n-kk] -= wts[k] * h[kk:]
                else:
                    L = -kk
                    grad[:n-L] += wts[k] * (1.0 - h[L:])
                    grad[L:] -= wts[k] * h[:n-L]
        else:
            grad = np.zeros(n)
            if lag > 0:
                grad[lag:] += (1.0 - h[:n-lag])
                grad[:n-lag] -= h[lag:]
            else:
                L = -lag
                grad[:n-L] += (1.0 - h[L:])
                grad[L:] -= h[:n-L]
        v = 0.85 * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5_value(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_imp = 0
        else:
            no_imp += 1
        h = h_new
    return best_h, best_c5

def binary_search_focused(h_init, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n // 2
    h = make_binary(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5_value(h)
    t0 = time.time()
    deadline = t0 + time_budget
    stall = 0
    while time.time() < deadline - 0.3:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if stall > 4000:
            h = best_h.copy()
            for _ in range(rng.randint(2, 6)):
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                h[ones[rng.randint(len(ones))]] = 0.0
                h[zeros[rng.randint(len(zeros))]] = 1.0
            stall = 0
            continue
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_k = int(np.argmax(corr))
        lag = worst_k - (n - 1)
        if lag > 0:
            bad = np.where((h == 1.0))[0]
            bad = bad[bad + lag < n]
            bad = bad[h[bad + lag] == 0.0]
            if len(bad) > 0:
                i = bad[rng.randint(len(bad))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        elif lag < 0:
            L = -lag
            bad = np.where((h == 1.0))[0]
            bad = bad[bad - L >= 0]
            bad = bad[h[bad - L] == 0.0]
            if len(bad) > 0:
                i = bad[rng.randint(len(bad))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5_value(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
    return best_h, best_c5

def sa_binary(h_init, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n // 2
    h = make_binary(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5_value(h)
    t0 = time.time()
    deadline = t0 + time_budget
    T = 0.3
    T_min = 0.005
    cool = 0.99985
    while time.time() < deadline - 0.3:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5_value(h)
        delta = c5_new - best_c5
        if delta < 0 or rng.rand() < math.exp(-delta / max(T, 1e-7)):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T * cool, T_min)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    best_h = None
    best_c5 = 1.0
    best_n = 70

    if 'initial_h_values' in globals() and initial_h_values is not None:
        hp = np.asarray(initial_h_values, dtype=float)
        best_h = project_feasible(hp.copy(), len(hp) / 2.0)
        best_c5 = compute_c5_value(best_h)
        best_n = len(best_h)

    n_list = [70, 64, 72, 80, 96, 100, 112, 128]
    if best_n not in n_list:
        n_list.insert(0, best_n)

    for n in n_list:
        if time.time() > deadline - 50:
            break
        target_sum = n / 2.0
        target_ones = n // 2
        local_best = None
        local_best_c5 = 1.0

        for ptype in range(6):
            if time.time() > deadline - 50:
                break
            h_pat = make_pattern(n, seed + n * 100 + ptype * 17, ptype)
            h_pat = project_feasible(h_pat, target_sum)
            t_gd = min(10, max(3, int((budget_s * 0.5) / (len(n_list) * 6))))
            h_gd, c5_gd = focused_grad_desc(h_pat, target_sum, t_gd, seed + n * 1000 + ptype * 7)
            if c5_gd < local_best_c5:
                local_best_c5 = c5_gd
                local_best = h_gd.copy()
            h_bin = make_binary(h_gd, target_ones)
            t_bs = min(12, max(4, int((budget_s * 0.5) / (len(n_list) * 6))))
            h_b, c5_b = binary_search_focused(h_bin, t_bs, seed + n * 2000 + ptype * 11)
            if c5_b < local_best_c5:
                local_best_c5 = c5_b
                local_best = h_b.copy()

        if local_best is not None:
            t_sa = min(35, max(8, int((deadline - time.time() - 40) * 0.4)))
            if t_sa >= 5:
                h_sa, c5_sa = sa_binary(local_best, t_sa, seed + 5000 + n * 3)
                if c5_sa < local_best_c5:
                    local_best_c5 = c5_sa
                    local_best = h_sa.copy()
            t_fb = min(25, max(5, int((deadline - time.time() - 30) * 0.5)))
            if t_fb >= 4:
                h_fb, c5_fb = binary_search_focused(local_best, t_fb, seed + 9000 + n * 5)
                if c5_fb < local_best_c5:
                    local_best_c5 = c5_fb
                    local_best = h_fb.copy()

        if local_best is not None and local_best_c5 < best_c5:
            best_h = local_best
            best_c5 = local_best_c5
            best_n = n

    if best_h is not None and time.time() < deadline - 8:
        nf = len(best_h)
        tf = nf / 2.0
        rng_f = np.random.RandomState(seed + 12345)
        h_frac = project_feasible(best_h + rng_f.randn(nf) * 0.02, tf)
        t_rem = deadline - time.time() - 2
        if t_rem > 4:
            h_p, c5_p = focused_grad_desc(h_frac, tf, min(t_rem, 15), seed + 14000)
            if c5_p < best_c5:
                best_h = h_p
                best_c5 = c5_p
        h_bc = make_binary(best_h, nf // 2)
        c5_bc = compute_c5_value(h_bc)
        if c5_bc < best_c5:
            best_h = h_bc
            best_c5 = c5_bc

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5_value(best_h)
        best_n = 70

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5_value(best_h)
    return best_h, best_c5, len(best_h)
