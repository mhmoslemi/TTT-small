import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def compute_active_gradient(h, n, active_indices):
    """Gradient of sum of overlaps at active lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    h_safe = h.copy()
    for idx in active_indices:
        # idx is index into the full correlation array (length 2n-1)
        # lag m = idx - (n-1)
        m = idx - (n - 1)
        # grad[j] += dx * [(1-h[j-m])*1[0<=j-m<n] - h[j+m]*1[0<=j+m<n]]
        for j in range(n):
            g = 0.0
            jm = j - m
            if 0 <= jm < n:
                g += (1.0 - h_safe[jm])
            jp = j + m
            if 0 <= jp < n:

def soft_max_obj(h, n, beta):
    overlaps = compute_overlaps(h)
    m = overlaps.max()
    shifted = beta * (overlaps - m)
    shifted -= shifted.max()
    logsumexp = m + np.log(np.sum(np.exp(shifted))) / beta
    return logsumexp

def soft_max_grad(h, n, beta):
    overlaps = compute_overlaps(h)
    m = overlaps.max()
    shifted = beta * (overlaps - m)
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    # Gradient is weighted sum of individual overlap gradients
    grad = np.zeros(n)
    for idx in range(len(overlaps)):
        if weights[idx] > 1e-8:
            m_lag = idx - (n - 1)
            grad += weights[idx] * grad_lag(h, n, m_lag)
    return grad, m + np.log(np.sum(exp_shifted)) / beta

    h_cur = project_feasible(h.copy(), n)
    best = h_cur.copy()
    best_val = compute_c5(h_cur)

            return g

                    best_val = c5
        except Exception:
            pass
    return best, best_val

    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        g[idx] += (1.0 - h[idx - m]) * dx
    lo2 = max(0, -m)
    hi2 = min(n - 1, n - 1 - m)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        g[idx] -= h[idx + m] * dx
    return g

def topk_grad(h, n, K):
    overlaps = compute_overlaps(h)
    top_idx = np.argsort(overlaps)[::-1][:K]
    g = np.zeros(n)
    for idx in top_idx:
        g += grad_lag(h, n, idx - (n - 1))
    g /= K
    return g, overlaps

def soft_obj_and_grad(h, n, beta):
    ov = compute_overlaps(h)
    m = ov.max()
    s = beta * (ov - m)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    val = m + np.log(e.sum()) / beta
    g = np.zeros(len(h))
    for idx in range(len(ov)):
        if w[idx] > 1e-10:
            g += w[idx] * grad_lag(h, n, idx - (n - 1))
    return val, g

def weighted_grad(h, n, weights):
    """Compute weighted sum of gradients for all lags."""
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-8:
            continue
        m = idx - (n - 1)
        lo1, hi1 = max(0, m), min(n-1, m+n-1)
        if lo1 <= hi1:
            i = np.arange(lo1, hi1+1)
            g[i] += w * (1.0 - h[i - m])
        lo2, hi2 = max(0, -m), min(n-1, n-1-m)
        if lo2 <= hi2:
            i = np.arange(lo2, hi2+1)
            g[i] -= w * h[i + m]
    g *= dx
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def project_feasible(h):
    n = len(h)
    target = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hs = np.clip(h + mid, 0.0, 1.0)
        if np.sum(hs) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag_vec(h, n, m):
    dx = 2.0 / n
    g = np.zeros(n)
    lo = max(0, m)
    hi = min(n - 1, m + n - 1)
    if lo <= hi:
        i = np.arange(lo, hi + 1)
        g[i] += (1.0 - h[i - m]) * dx
    lo = max(0, -m)
    hi = min(n - 1, n - 1 - m)
    if lo <= hi:
        i = np.arange(lo, hi + 1)
        g[i] -= h[i + m] * dx
    return g

def topk_grad(h, n, K):
    ov = compute_overlaps(h)
    top = np.argsort(ov)[::-1][:K]
    g = np.zeros(n)
    for idx in top:
        g += grad_lag_vec(h, n, idx - (n - 1))
    g /= K
    return g, ov

def soft_vals(h, n, beta):
    ov = compute_overlaps(h)
    m = ov.max()
    s = beta * (ov - m)
    s -= s.max()
    e = np.exp(s)
    val = m + np.log(e.sum()) / beta
    w = e / e.sum()
    return val, w

def soft_grad(h, n, beta):
    _, w = soft_vals(h, n, beta)
    g = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] > 1e-9:
            g += w[idx] * grad_lag_vec(h, n, idx - (n - 1))
    return g

def slsqp_opt(h, n, t0, budget, maxiter=200, betas=(20,100,500,2000)):
    h = project_feasible(h.copy())
    best, best_c5 = h.copy(), compute_c5(h)
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bnd = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() - t0 > budget * 0.85:
            break
        def _obj(x, n=n, b=beta):
            return soft_vals(x, n, b)[0]
        def _objg(x, n=n, b=beta):
            return soft_grad(x, n, b)
        try:
            res = minimize(_obj, h, jac=_objg, method='SLSQP',
                          bounds=bnd, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h = project_feasible(res.x)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best, best_c5 = h.copy(), c5
        except Exception:
            pass
    return best, best_c5

def lbfgs_opt(h, n, t0, budget, beta=500, mu=50.0):
    h = project_feasible(h.copy())
    best, best_c5 = h.copy(), compute_c5(h)
    def _obj(x, n=n, b=beta, mu=mu):
        xc = np.clip(x, 0, 1)
        v, _ = soft_vals(xc, n, b)
        d = np.sum(xc) - n * 0.5
        return v + mu * d * d
    def _objg(x, n=n, b=beta, mu=mu):

        g = soft_grad(xc, n, b)
        g += 2 * mu * (np.sum(xc) - n * 0.5)
        g[x < 0] = 0
        g[x > 1] = 0
        return g

        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best, best_c5 = h2.copy(), c5
    except Exception:
        pass
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def project_feasible(h):
    n = len(h)
    target = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hs = np.clip(h + mid, 0.0, 1.0)
        if np.sum(hs) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag_vec(h, n, m):
    dx = 2.0 / n
    g = np.zeros(n)
    lo = max(0, m)
    hi = min(n - 1, m + n - 1)
    if lo <= hi:
        i = np.arange(lo, hi + 1)
        g[i] += (1.0 - h[i - m]) * dx
    lo = max(0, -m)
    hi = min(n - 1, n - 1 - m)
    if lo <= hi:
        i = np.arange(lo, hi + 1)
        g[i] -= h[i + m] * dx
    return g

def topk_grad(h, n, K, ov):
    top = np.argsort(ov)[::-1][:K]
    g = np.zeros(n)
    for idx in top:
        g += grad_lag_vec(h, n, idx - (n - 1))
    g /= K
    return g

def soft_vals(h, n, beta):
    ov = compute_overlaps(h)
    m = ov.max()
    s = beta * (ov - m)
    s -= s.max()
    e = np.exp(s)
    val = m + np.log(e.sum()) / beta
    w = e / e.sum()
    return val, w

    for idx in range(len(w)):
