import numpy as np
import time

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5_from_ov(ov):
    return float(np.max(ov))

def project_binary(h, n):
    """Project h to binary with exactly n/2 ones"""
    target = n // 2
    h = h.copy()
    # Sort indices by h value descending
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def init_binary(n, rng):
    """Create random binary h with n/2 ones"""
    h = np.zeros(n)
    positions = rng.choice(n, size=n//2, replace=False)
    h[positions] = 1.0
    return h

def swap_overlaps(ov, h, ia, ja, n):
    """Compute new overlap array after swapping h[ia](=1) and h[ja](=0)
    Uses old h values for the delta computation."""
    new_ov = ov.copy()
    for k in range(-(n-1), n):
        idx = k + n - 1
        delta = 0.0
        # ia as first index
        if 0 <= ia + k < n:
            delta -= (1.0 - h[ia + k])
        # ja as first index
        if 0 <= ja + k < n:
            delta += (1.0 - h[ja + k])
        # ia as second index (i = ia-k)
        ik = ia - k
        if 0 <= ik < n:
            delta += h[ik]
        # ja as second index (i = ja-k)
        jk = ja - k
        if 0 <= jk < n:
            delta -= h[jk]
        new_ov[idx] += delta
    return new_ov

def optimize_binary(h, n, deadline, rng):
    """Swap-based local search with periodic perturbation"""
    best_h = h.copy()
    ov = compute_overlaps(h, n)
    best_c5 = get_c5_from_ov(ov)
    stalled = 0
    it = 0
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    while time.time() < deadline:
        it += 1
        # Find best swap
        best_new_c5 = best_c5
        best_ia, best_ja = -1, -1
        best_new_ov = ov.copy()
        
        # Try all swaps
        for il in range(len(ones)):
            ia = ones[il]
            for jl in range(len(zeros)):
                ja = zeros[jl]
                new_ov = swap_overlaps(ov, h, ia, ja, n)
                c5 = np.max(new_ov)
                if c5 < best_new_c5 - 1e-15:
                    best_new_c5 = c5
                    best_ia, best_ja = ia, ja
                    best_new_ov = new_ov
        
        if best_ia >= 0 and best_new_c5 < best_c5 - 1e-15:
            # Accept the swap
            h[best_ia], h[best_ja] = h[best_ja], h[best_ia]
            ov = best_new_ov
            best_c5 = best_new_c5
            best_h = h.copy()
            stalled = 0
            # Update ones and zeros
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            stalled += 1
            # Try perturbation
            if stalled > 50 and stalled % 10 == 0:
                num_flips = rng.randint(2, n//4)
                flip_ones = rng.choice(ones, size=min(num_flips, len(ones)), replace=False)
                flip_zeros = rng.choice(zeros, size=min(num_flips, len(zeros)), replace=False)
                for fi in range(len(flip_ones)):
                    h[flip_ones[fi]], h[flip_zeros[fi]] = h[flip_zeros[fi]], h[flip_ones[fi]]
                ov = compute_overlaps(h, n)
                c5 = get_c5_from_ov(ov)
                best_h = h.copy()
                best_c5 = c5
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                stalled = 0
    
    return best_h, best_c5

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(ov):
    return float(np.max(ov))

def make_binary(h, n):
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin
