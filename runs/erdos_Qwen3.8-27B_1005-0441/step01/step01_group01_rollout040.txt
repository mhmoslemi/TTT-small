import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(get_corr(h)))

def fw_one_step(h, n, top_m, rng):
    corr = get_corr(h)
    top_idx = np.argsort(corr)[-top_m:]
    effect = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        j = np.arange(n)
        jp = j + k
        jm = j - k
        m1 = (jp >= 0) & (jp < n)
        m2 = (jm >= 0) & (jm < n)
        t1 = np.ones(n)
        t1[m1] = 1.0 - h[jp[m1]]
        t2 = np.zeros(n)
        t2[m2] = h[jm[m2]]
        effect += t1 - t2
    j_inc = int(np.argmin(effect))
    j_dec = int(np.argmax(effect))
    if j_inc == j_dec or abs(effect[j_inc] - effect[j_dec]) < 1e-15:
        return h, False
    best_h = h
    best_c = float(np.max(corr))
    for delta in [0.08, 0.04, 0.02, 0.01, 0.005]:
        h_new = h.copy()
        h_new[j_inc] = min(1.0, h_new[j_inc] + delta)
        h_new[j_dec] = max(0.0, h_new[j_dec] - delta)
        h_new = project_feasible(h_new, n)
        c_new = float(np.max(get_corr(h_new)))
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if best_h is not h:
        return best_h, True
    # Try a random pair
    for _ in range(3):
        a, b = rng.integers(0, n, 2)
        if a == b:
            continue
        for delta in [0.03, 0.015, 0.008]:
            h_new = h.copy()
            h_new[a] = min(1.0, h_new[a] + delta)
            h_new[b] = max(0.0, h_new[b] - delta)
            h_new = project_feasible(h_new, n)
            c_new = float(np.max(get_corr(h_new)))
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                break
    return best_h, best_h is not h

def fw_search(h, n, t_end, rng, max_iters=20000):
    h = h.copy()
    best_c = compute_c5(h)
    stagnant = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        new_h, improved = fw_one_step(h, n, 5, rng)
        if improved:
            h = new_h
            c = compute_c5(h)
            if c < best_c:
                best_c = c
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 200:
                mask = rng.random(n) < 0.15
                h = h + 0.02 * rng.standard_normal(n) * mask
                h = project_feasible(h, n)
                stagnant = 0
        if stagnant > 1000:
            break
    return h, best_c

def soft_obj(h, alpha):
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -400.0, 400.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def lbbf_optim(h, n, t_end, alpha, penalty):
    def obj(x):
        return soft_obj(x, alpha) + penalty * (np.sum(x) - n * 0.5) ** 2
    res = minimize(obj, h.copy(), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-14})
    hc = project_feasible(res.x, n)
    return hc, compute_c5(hc)

def coord_descent_polish(h, n, t_end, rng, rounds=200):
    h = h.copy()
    best_c = compute_c5(h)
    for r in range(rounds):
        if time.time() > t_end:
            break
        corr = get_corr(h)
        k_star = int(np.argmax(corr))
        k = k_star - (n - 1)
        j = np.arange(n)
        jp = j + k
        jm = j - k
        m1 = (jp >= 0) & (jp < n)
        m2 = (jm >= 0) & (jm < n)
        t1 = np.ones(n)
        t1[m1] = 1.0 - h[jp[m1]]
        t2 = np.zeros(n)
        t2[m2] = h[jm[m2]]
        eff = t1 - t2
        j_inc = int(np.argmin(eff))
        j_dec = int(np.argmax(eff))
        if j_inc == j_dec:
            continue
        dx = 2.0 / n
        # 1D: minimize max(corr + d*delta_e, ...) over d
        # Simple bisection
        delta_max = min(h[j_dec], 1.0 - h[j_inc])
        best_d = 0.0
        lo, hi = 0.0, delta_max
        for _ in range(20):
            mid = (lo + hi) / 2
            h_try = h.copy()
            h_try[j_inc] += mid
            h_try[j_dec] -= mid
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c:
                best_c = c_try
                best_d = mid
                lo = mid
            else:
                hi = mid
        if best_d > 0:
            h[j_inc] += best_d
            h[j_dec] -= best_d
            h = project_feasible(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(160)
        b = 40
        h0[:b] = 1.0
        h0[-b:] = 1.0
    h0 = project_feasible(h0, len(h0))
    n = len(h0)
    h_best = h0.copy()
    c5_best = compute_c5(h_best)

    # Phase 1: L-BFGS-B with increasing softness
    time_budget_phase1 = t0 + budget_s * 0.20
    for alpha in [50.0, 150.0, 400.0]:
        if time.time() > time_budget_phase1:
            break
        for pen in [500.0, 5000.0]:
            if time.time() > time_budget_phase1:
                break
            hc, cc = lbbf_optim(h_best, n, t_end, alpha, pen)
            if cc < c5_best:
                c5_best = cc
                h_best = hc

    # Phase 2: Frank-Wolfe peak reduction (main workhorse)
    time_budget_phase2 = t0 + budget_s * 0.65
    h_best, c5_best = fw_search(h_best, n, time_budget_phase2, rng, max_iters=30000)

    # Phase 3: More L-BFGS-B refinement
    time_budget_phase3 = t0 + budget_s * 0.75
    for alpha in [200.0, 500.0, 1000.0]:
        if time.time() > time_budget_phase3:
            break
        hc, cc = lbbf_optim(h_best, n, t_end, alpha, 10000.0)
        if cc < c5_best:
            c5_best = cc
            h_best = hc

    # Phase 4: Coordinate descent polish
    time_budget_phase4 = t0 + budget_s * 0.85
    h_best, c5_best = coord_descent_polish(h_best, n, time_budget_phase4, rng, rounds=500)

        h_best, c5_best = fw_search(h_best, n, t_end - 80, rng, max_iters=10000)
        if time.time() < t_end - 60:
            h_best, c5_best = coord_descent_polish(h_best, n, t_end - 60, rng, rounds=300)

    # Phase 6: Try a second FW pass from perturbed best

        hp, cp = fw_search(h_pert, n, t_end - 40, rng, max_iters=5000)
        if cp < c5_best:
            c5_best = cp
            h_best = hp

    # Final feasibility check
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
