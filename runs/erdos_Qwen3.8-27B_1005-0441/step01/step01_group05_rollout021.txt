def _repair(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-12:
            break

def _slsqp_polish(h0, n, tmax):
    h = h0.copy()
    dx = 2.0 / n
    target = n * 0.5
    
    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        # power mean trick
        p = 50
        return (np.sum(corr**p) / len(corr))**(1.0/p) * dx
    
    def obj_grad(x):
        # numerical gradient
        eps = 1e-7
        g = np.zeros(n)
        f0 = obj(x)
        for i in range(n):
            x[i] += eps
            g[i] = (obj(x) - f0) / eps
            x[i] -= eps
        return g
    
    # Use SLSQP
    bounds = [(0,1)]*n
    cons = [{'type':'eq','fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    
    best = h0.copy()
    best_v = _c5(best)
    
    t0 = time.time()
    for trial in range(5):
        if time.time() - t0 > tmax:
            break
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter':200, 'ftol':1e-14})
            h_new = _repair(res.x)
            v = _c5(h_new)
            if v < best_v:
                best = h_new
                best_v = v
        except:
            break
    return best, best_v

from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def _repair(h):
    n = len(h)
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = t - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0
            if not m.any(): break
            h[m] += d / m.sum()
        else:
            m = h > 0.0
            if not m.any(): break
            h[m] -= (-d) / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _block_mut(h, rng, strength):
    n = len(h)
    h2 = h.copy()
    a = rng.randint(0, n - 5)
    L = rng.randint(5, min(40, n - a))
    phase = rng.rand() * np.pi * 2
    bump = strength * np.abs(np.sin(np.linspace(0, np.pi, L) + phase)) * rng.choice([-1, 1.0])
    h2[a:a+L] += bump
    return _repair(h2)

    rng = np.random.RandomState(seed)
    t0 = time.time()

    pop[0] = _repair(h0.copy())
    for i in range(1, P):
        s = 0.02 + 0.12 * rng.rand()

    costs = np.array([_c5(pop[i]) for i in range(P)])
    order = np.argsort(costs)
    pop, costs = pop[order], costs[order]
    
    best_h, best_c = pop[0].copy(), costs[0]
    sigma = 0.06
    stale = 0
    gen = 0

        # Slowly decrease sigma

        sigma = max(sigma, 0.003)

            else:
                off[i] = _block_mut(pop[pi], rng, sigma * 0.5)

def _polish(h0, n, tmax, seed=0):
    dx = 2.0 / n
    t0 = time.time()
    best, best_v = h0.copy(), _c5(h0)
    
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        p = 100.0
        return (np.sum(c**p) / len(c)) ** (1.0/p) * dx
    
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    
    for trial in range(8):
        if time.time() - t0 > tmax - 2:
            break
        try:
            r = minimize(obj, best, method='SLSQP', bounds=bounds,
                        constraints=con, options={'maxiter': 300, 'ftol': 1e-15})
            h2 = _repair(r.x)
            v2 = _c5(h2)
            if v2 < best_v - 1e-12:
                best, best_v = h2, v2
        except Exception:
            break
    return best, best_v

            m = h < 1.0
            if not m.any():
                break
            h[m] += d / m.sum()
        else:
            m = h > 0.0
            if not m.any():
                break
            h[m] -= (-d) / m.sum()
        h[:] = np.clip(h, 0.0, 1.0)
    return h

def _block_mut(h, rng, strength):
    n = len(h)
    h2 = h.copy()
    a = rng.randint(0, max(1, n - 5))
    L = rng.randint(5, min(40, n - a))
    sign = 1.0 if rng.rand() < 0.5 else -1.0
    bump = sign * strength * np.abs(np.sin(np.linspace(0, np.pi, L)))
    h2[a:a + L] += bump
    return _repair(h2)

def _es(h0, n, seed, tmax, P, mu):
    global _g_n
    rng = np.random.RandomState(seed)
    t0 = time.time()
    pop = np.empty((P, n))
    pop[0] = _repair(h0.copy())
    for i in range(1, P):
        s = 0.02 + 0.12 * rng.rand()
        pop[i] = _repair(pop[0] + s * rng.randn(n))
    costs = np.array([_c5(pop[i]) for i in range(P)])
    order = np.argsort(costs)
    pop, costs = pop[order], costs[order]
    best_h, best_c = pop[0].copy(), costs[0]
    sigma = 0.06
    stale = 0
    while time.time() - t0 < tmax - 2:
        off = np.empty((mu, n))
        for i in range(mu):
            r = rng.rand()
            if r < 0.50:
                pi = i if i < mu else rng.randint(0, mu)
                off[i] = _repair(pop[pi] + sigma * rng.randn(n))
            elif r < 0.75:
                pi = i if i < mu else rng.randint(0, mu)
                off[i] = _block_mut(pop[pi], rng, sigma * 0.6)
            else:
                p1 = i if i < mu else rng.randint(0, mu)
                p2 = rng.randint(0, mu)
                if p1 == p2:
                    p2 = (p2 + 1) % P
                cp = rng.randint(1, n - 1)
                child = np.concatenate([pop[p1][:cp], pop[p2][cp:]])
                off[i] = _repair(child + sigma * 0.3 * rng.randn(n))
        off_c = np.array([_c5(off[i]) for i in range(mu)])
        keep = P - mu
        all_p = np.vstack([pop[:keep], off])
        all_c = np.concatenate([costs[:keep], off_c])
        o2 = np.argsort(all_c)[:P]
        pop, costs = all_p[o2], all_c[o2]
        if costs[0] < best_c - 1e-11:
            best_c = costs[0]
            best_h = pop[0].copy()
            stale = 0
        else:
            stale += 1
        sigma *= 0.999
        if sigma < 0.003:
            sigma = 0.003
        if stale > 20:
            sigma = max(0.05, sigma * 2.0)
            nd = max(2, P // 8)
            for i in range(nd):
                j = P - 1 - i
                s = 0.05 + 0.2 * rng.rand()
                pop[j] = _repair(best_h + s * rng.randn(n))
                costs[j] = _c5(pop[j])
            o3 = np.argsort(costs)
            pop, costs = pop[o3], costs[o3]
            stale = 0
    return best_h, best_c

def _eq_fun(x):
    return x.sum() - _g_n * 0.5

def _eq_jac(x):
    return np.ones(_g_n)

def _polish(h0, n, tmax):
    global _g_n
    _g_n = n
    dx = 2.0 / n
    t0 = time.time()
    best, best_v = h0.copy(), _c5(h0)
    p = 50.0
    def obj(x):
        xc = np.clip(x, 0.0, 1.0)
        c = np.correlate(xc, 1.0 - xc, mode="full")
        return float((np.sum(c ** p) / len(c)) ** (1.0 / p) * dx)
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    for trial in range(10):
        if time.time() - t0 > tmax - 2:
            break
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=con, options={'maxiter': 300, 'ftol': 1e-15})
            h2 = _repair(res.x)
            v2 = _c5(h2)
            if v2 < best_v - 1e-13:
                best, best_v = h2, v2
        except Exception:
            break
    return best, best_v

def _upsample(h, n_new):
    n_old = len(h)
    x_o = np.arange(n_old) * 2.0 / n_old
    x_n = np.arange(n_new) * 2.0 / n_new
    h_new = np.clip(np.interp(x_n, x_o, h), 0.0, 1.0)
    return _repair(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_s = time.time()
    def rem():
        return budget_s - (time.time() - t_s)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(160, 0.5)
    n = len(h0)
    best_h, best_c = h0.copy(), _c5(h0)

    # Phase 1: ES main at current n
    if rem() > 200:
        t1 = min(400, rem() - 100)
        h1, c1 = _es(best_h, n, seed, t1, P=400, mu=60)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Polish
    if rem() > 50:
        h2, c2 = _polish(best_h, n, min(90, rem() - 25))
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Try alternate resolutions
    for n_alt in [80, 100, 120]:
        if rem() < 50 or n_alt == n:
            continue
        h_alt = _upsample(best_h, n_alt)
        c_alt = _c5(h_alt)
        if c_alt < best_c:
            best_h, best_c = h_alt, c_alt
        if rem() > 40:
            h_a2, c_a2 = _es(h_alt, n_alt, seed + n_alt * 100, min(55, rem() - 15),
                             P=250, mu=40)
            if c_a2 < best_c:
                best_h, best_c = h_a2, c_a2

    # Phase 4: Second ES round with different seed/params
    if rem() > 120:
        t4 = min(250, rem() - 40)
        h4, c4 = _es(best_h, n, seed + 54321, t4, P=500, mu=80)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Final polish
    if rem() > 20:
        h5, c5 = _polish(best_h, n, max(12, rem() - 5))
        if c5 < best_c:
            best_h, best_c = h5, c5

    return best_h, best_c, len(best_h)
