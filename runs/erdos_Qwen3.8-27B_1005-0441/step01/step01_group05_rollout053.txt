    """SLSQP optimization with smooth max."""
    t_start = time.time()
    if temps is None:
        temps = [0.02, 0.005, 0.001, 0.0003]
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)},)
    dx = 2.0 / n
    h = h_init.copy()

        if time.time() - t_start > max_time - 5:
            break
        def obj(x):

            m = np.max(corr)
            sh = (corr - m) / temp
            ms = np.max(sh)
            return (m + temp * (ms + np.log(np.sum(np.exp(sh - ms))))) * dx

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 160
_g_temp = 0.01
_g_penalty = 0.0

def _obj_smooth(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    t = max(_g_temp, 1e-12)
    sh = (corr - m) / t
    ms = np.max(sh)
    return (m + t * (ms + np.log(np.sum(np.exp(sh - ms))))) * dx

def _obj_pen(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    t = max(_g_temp, 1e-12)
    sh = (corr - m) / t
    ms = np.max(sh)
    smooth = (m + t * (ms + np.log(np.sum(np.exp(sh - ms))))) * dx
    if _g_penalty > 0:
        smooth += _g_penalty * (np.sum(h) - n / 2.0) ** 2
    return smooth

def _cons_eq_fun(h):
    return np.sum(h) - _g_n / 2.0

def _cons_eq_jac(h):
    return np.ones(_g_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (n / 2.0) / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_phase(h_init, n, max_time, temps, iters_per_temp=8):
    global _g_n, _g_temp
    t_start = time.time()
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _cons_eq_fun, 'jac': _cons_eq_jac},)
    h = h_init.copy()
    best_val = compute_c5(h)
    for temp in temps:
        if time.time() - t_start > max_time - 5:
            break
        _g_temp = temp
        for it in range(iters_per_temp):
            if time.time() - t_start > max_time - 3:
                break
            try:
                res = minimize(_obj_smooth, h, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 600, 'ftol': 1e-14})
                h_new = _project(res.x, n)
                v = compute_c5(h_new)
                if v < best_val:
                    best_val = v
                h = h_new.copy()
            except Exception:
                break
    return h, best_val

def _lbfgs_phase(h_init, n, max_time, temps):
    global _g_n, _g_temp, _g_penalty
    t_start = time.time()
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    h = h_init.copy()
    best_val = compute_c5(h)
    _g_penalty = 500.0
    for temp in temps:
        if time.time() - t_start > max_time - 5:
            break
        _g_temp = temp
        try:
            res = minimize(_obj_pen, h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'gtol': 1e-12})
            h_new = _project(res.x, n)
            v = compute_c5(h_new)
            if v < best_val:
                best_val = v
            h = h_new.copy()
            _g_penalty = min(_g_penalty * 5.0, 1e6)
        except Exception:
            break
    _g_penalty = 0.0
    return h, best_val

def _perturb(h, scale):
    n = len(h)
    h_p = h + scale * np.random.randn(n)
    return _project(h_p, n)

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(160, 0.5)
    n = len(h0)
    h0 = _project(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with broad temperature sweep
    if rem() > 200:
        mt = min(170, rem() - 60)
        h_opt, c5_opt = _slsqp_phase(h0.copy(), n, mt,
                                      [0.03, 0.01, 0.003, 0.001, 0.0003, 0.0001], 8)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: L-BFGS-B refinement with increasing penalty
    if rem() > 120:
        mt = min(100, rem() - 50)
        h_opt, c5_opt = _lbfgs_phase(best_h.copy(), n, mt,
                                      [0.01, 0.003, 0.0005, 0.0001])
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: SLSQP fine polish
    if rem() > 100:
        mt = min(80, rem() - 40)
        h_opt, c5_opt = _slsqp_phase(best_h.copy(), n, mt,
                                      [0.001, 0.0003, 0.0001, 0.00003], 10)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Small random restarts
    restart = 0
    while rem() > 50 and restart < 40:
        scale = 0.008 + 0.015 * (restart % 6) / 5.0
        h_pert = _perturb(best_h, scale)
        mt = min(18, rem() - 25)
        if mt < 8:
            break
        h_opt, c5_opt = _slsqp_phase(h_pert, n, mt,
                                      [0.001, 0.0003, 0.0001], 5)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        restart += 1

    # Phase 5: Upsample and optimize at higher resolution
    n_up = min(n + 60, 300)
    if rem() > 70 and n_up > n:
        h_up = _upsample(best_h, n_up)
        mt = min(55, rem() - 30)
        h_opt, c5_opt = _slsqp_phase(h_up, n_up, mt,
                                      [0.003, 0.001, 0.0003, 0.0001], 6)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            n = n_up

    # Phase 6: Final ultra-fine polish
    if rem() > 15:
        mt = max(10, rem() - 5)
        h_opt, c5_opt = _slsqp_phase(best_h.copy(), len(best_h), mt,
                                      [0.0001, 0.00003, 0.00001], 8)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final feasibility check
    best_h = _project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
