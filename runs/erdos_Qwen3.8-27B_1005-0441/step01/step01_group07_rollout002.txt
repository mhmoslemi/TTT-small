import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def get_grad(h, dx, top_m, tau):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    mc = corr.max()
    if tau > 1:
        shifted = np.clip((corr - mc) * tau, -500, 0)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
    else:
        idx = np.argsort(corr)[::-1][:top_m]
        weights = np.zeros(len(corr))
        weights[idx] = 1.0 / top_m
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            a = slice(0, n - lag)
            b = slice(lag, n)
            grad[a] += w * (1.0 - h[b])
            grad[b] -= w * h[a]
        else:
            L = -lag
            a = slice(L, n)
            b = slice(0, n - L)
            grad[a] += w * (1.0 - h[b])
            grad[b] -= w * h[a]
    return grad * dx

def line_search(h, grad, dx, target, lr, tau, top_m, max_iters=5):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = lr
    for _ in range(max_iters):
        h_try = project_feasible(h - step * grad, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
        step *= 0.7
    return best_h, best_c5

def optimize_fista(h_init, time_budget, seed, use_smooth=True):

    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n

    best_h = h.copy()
    best_c5 = compute_c5(h)
    y = h.copy()
    t_k = 1.0
    beta = 0.9
    tau = 40.0 if use_smooth else 0
    top_m = 3
    lr_init = 0.1
    t0 = time.time()

    no_improve = 0
    lr = lr_init
    while time.time() < deadline - 0.2:
        if no_improve > 300:

            y = h.copy()
            t_k = 1.0
            lr = lr_init
            tau = 40.0 if use_smooth else 0
            no_improve = 0
        elif no_improve > 100:
            lr = max(lr * 0.8, 0.002)
        grad = get_grad(y, dx, top_m, tau)
        h_new = project_feasible(y - lr * grad, target)
        c5_new = compute_c5(h_new)

        h = h_new
        t_k = t_new

def upsample(h_in, factor=2):
    n = len(h_in)
    n2 = n * factor
    h2 = np.zeros(n2)
    for i in range(n):
        h2[i * factor] = h_in[i]
        if i + 1 < n:
            for j in range(1, factor):
                h2[i * factor + j] = h_in[i] + (h_in[i + 1] - h_in[i]) * j / factor
        else:
            h2[i * factor:] = h_in[i]
    return h2

def binary_local_search(h_frac, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    target_sum = int(n // 2)
    h = (h_frac >= 0.5).astype(float)
    diff = target_sum - int(h.sum())
    if diff > 0:
        idxs = np.argsort(h)[:diff]
        h[idxs] = 1.0
    elif diff < 0:
        idxs = np.argsort(h)[diff:]
        h[idxs] = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    while time.time() < deadline - 0.5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5
