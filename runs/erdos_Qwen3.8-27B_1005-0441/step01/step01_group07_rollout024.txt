def binary_simulated_annealing(h_binary, time_budget, seed):
    """Simulated annealing: random swaps with temperature annealing."""
    rng = np.random.RandomState(seed)
    n = len(h_binary)
    h = h_binary.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    T_init = 0.01  # initial temperature (in C₅ units)
    T_final = 0.0001
    cooling = 0.999
    
    t0 = time.time()
    deadline = t0 + time_budget
    T = T_init
    
    while time.time() < deadline - 0.1:
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0; h[j] = 1
        c5_new = compute_c5(h)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:

            if rng.random() < math.exp(-delta / T):
                # accept
                pass
            else:
                # reject
                h[i] = 1; h[j] = 0
        
        T *= cooling
        if T < T_final:
            T = T_init * 0.5  # restart with lower initial temp
    
    return best_h, best_c5

def binary_ga(h_init, pop_size, time_budget, seed):
    """Genetic algorithm on binary sequences."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n // 2
    
    # Initialize population
    pop = [h_init.copy()]
    while len(pop) < pop_size:
        h_new = np.zeros(n)
        positions = rng.permutation(n)[:target]
        h_new[positions] = 1
        pop.append(h_new)
    
    # Evaluate
    c5s = [compute_c5(h) for h in pop]
    best_idx = np.argmin(c5s)
    best_c5 = c5s[best_idx]
    best_h = pop[best_idx].copy()
    
    t0 = time.time()
    deadline = t0 + time_budget
    gen = 0
    
    while time.time() < deadline - 0.5:
        gen += 1
        # Sort by fitness
        order = np.argsort(c5s)
        pop = [pop[i] for i in order]
        c5s = [c5s[i] for i in order]
        best_h = pop[0].copy()
        best_c5 = c5s[0]
        
        # Elitism: keep top 10%
        elite_size = max(5, pop_size // 10)
        new_pop = [pop[i].copy() for i in range(elite_size)]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            # Tournament selection
            a = pop[rng.choice(elite_size * 2)]
            b = pop[rng.choice(elite_size * 2)]
            if c5s[a] <= c5s[b]:
                parent = a
            else:
                parent = b
            
            # Mutation: random swap
            h_new = pop[parent].copy()
            num_mut = rng.randint(1, 4)
            for _ in range(num_mut):
                ones = np.where(h_new == 1)[0]
                zeros = np.where(h_new == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i], h_new[j] = 0, 1
            
            new_pop.append(h_new)
        
        # Crossover (optional, for binary with sum constraint it's tricky)
        # Actually let's do: take two parents, randomly assign each position from one or the other
        # Then repair to maintain sum constraint
        
        pop = new_pop
        c5s = [compute_c5(h) for h in pop]
    
    return best_h, best_c5
