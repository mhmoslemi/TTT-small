```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def to_binary(h_frac, n):
    """Convert fractional to binary maintaining sum constraint."""
    h = (h_frac >= 0.5).astype(float)
    target = n // 2
    diff = target - int(h.sum())
    if diff > 0:
        # Need more 1s
        order = np.argsort(h_frac)[::-1]
        for i in range(diff):
            if h[order[i]] == 0:
                h[order[i]] = 1.0
    elif diff < 0:
        # Need fewer 1s
        order = np.argsort(h_frac)
        for i in range(-diff):
            if h[order[i]] == 1:
                h[order[i]] = 0.0
    return h

    best_c5 = compute_c5(h)

    sweeps = 0
    while improved and sweeps < max_sweeps:
        improved = False
        sweeps += 1
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_c5_raw(h):
    """Return max of correlation (without dx factor)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max())

def make_binary(h_frac, n):
    h = (h_frac >= 0.5).astype(float)
    target = n // 2
    cur = int(h.sum())
    if cur < target:
        idx = np.where(h == 0.0)[0]
        order = np.argsort(h_frac[idx])[::-1]
        for i in range(target - cur):
            h[idx[order[i]]] = 1.0
    elif cur > target:
        idx = np.where(h == 1.0)[0]
        order = np.argsort(h_frac[idx])
        for i in range(cur - target):
            h[idx[order[i]]] = 0.0
    return h

def local_search(h, deadline, rng):
    """Steepest descent local search with swap moves."""
    n = len(h)
    best_raw = compute_c5_raw(h)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0.0
        best_a, best_b = -1, -1
        for a in ones:
            for b in zeros:
                h[a], h[b] = 0.0, 1.0
                raw = compute_c5_raw(h)
                if raw < best_raw - 1e-12:
                    best_delta = raw - best_raw
                    best_a, best_b = a, b
                    best_raw = raw
                h[a], h[b] = 1.0, 0.0
        if best_a >= 0:
            h[best_a], h[best_b] = 0.0, 1.0
            improved = True
    return h, best_raw * (2.0 / n)

def il_search(h_init, deadline, rng, n_perturb_range=(2, 8)):
    """Iterated Local Search."""
    n = len(h_init)
    h = h_init.copy()
    h, best_c5 = local_search(h, deadline, rng)
    T = 0.01  # initial temperature
    t0 = time.time()
    
    while time.time() < deadline - 10:
        # Perturb
        k = rng.randint(*n_perturb_range)
        h_pert = h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(k):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_pert[a], h_pert[b] = 0.0, 1.0
            # Update ones and zeros
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
        
        # Local search on perturbed
        h_trial, c5_trial = local_search(h_pert, deadline, rng)
        
        # Accept
        if c5_trial < best_c5 - 1e-12:
            h = h_trial
            best_c5 = c5_trial
            T = max(T * 0.95, 0.001)
        elif rng.rand() < np.exp(-(c5_trial - best_c5) / T):
            h = h_trial
            T = max(T * 0.99, 0.001)
        else:
            T = max(T * 0.98, 0.001)
        
        # Cool
        T *= 0.999
    
    return h, best_c5
```

```python
import numpy as np
import time
```
```python
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_c5_raw(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max())

def make_binary(h_frac, n):
    h = (h_frac >= 0.5).astype(float)
    target = n // 2
    cur = int(h.sum())
    if cur < target:
        idx = np.where(h == 0.0)[0]
        order = np.argsort(h_frac[idx])[::-1]
        for i in range(target - cur):
            h[idx[order[i]]] = 1.0
    elif cur > target:
        idx = np.where(h == 1.0)[0]
        order = np.argsort(h_frac[idx])
        for i in range(cur - target):
            h[idx[order[i]]] = 0.0
    return h

def local_search(h, deadline, rng):
    n = len(h)
    best_raw = compute_c5_raw(h)
    while time.time() < deadline:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_improve = 1e-12
        best_a, best_b = -1, -1
        for a in ones:
            for b in zeros:
                h[a] = 0.0
                h[b] = 1.0
                raw = compute_c5_raw(h)
                if best_raw - raw > best_improve:
                    best_improve = best_raw - raw
                    best_a, best_b = a, b
                h[a] = 1.0
                h[b] = 0.0
        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            best_raw = compute_c5_raw(h)
        else:
            break
    return h, best_raw
```
