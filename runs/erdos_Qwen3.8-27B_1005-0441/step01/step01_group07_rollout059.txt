    lo, hi = -2.0, 2.0

            lo = mid
        else:
            hi = mid

def binary_to_feasible(h_frac, n):

    target = n // 2

    t = n // 2
    h = (hf >= 0.5).astype(float)
    s = int(h.sum())

    stall = 0
    t0 = time.time()
    while time.time() < t0 + budget - 0.3:
        corr = np.correlate(h, 1.0 - h, mode='full')
        sh = np.clip((corr - corr.max()) / tau, -40, 0)
        w = np.exp(sh); w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12: continue
            lag = k - (n - 1)
            if lag >= 0:
                g[lag:] += wk * (1.0 - h[:n - lag])
                g[:n - lag] -= wk * h[lag:]
            else:
                L = -lag
                g[:n - L] += wk * (1.0 - h[L:])
                g[L:] -= wk * h[:n - L]
        v[:] = beta * v - lr * (g * dx)
        h = project_feasible(h + v, n / 2.0)
        c = c5_of(h)
        if c < best_c - 1e-15:
            best, best_c = h.copy(), c; stall = 0
        else:
            stall += 1
        tau = min(tau * 1.003, 400.0)
        if stall > 500:
            h = project_feasible(best + rng.randn(n) * 0.04, n / 2.0)
            v[:] = 0; tau, lr = tau0, lr0; stall = 0
        elif stall > 150:
            lr = max(lr * 0.92, 0.003)
    return best, best_c

def greedy_bin(h_init, budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    dx = 2.0 / n
    h = frac_to_bin(h_init, n).copy()
    bh = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = float(corr.max() * dx)
    t0 = time.time()
    stall = 0
    while time.time() < t0 + budget - 0.3:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        cm = float(corr.max())
        bd, bi, bj = 0.0, -1, -1
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                nm = float(np.correlate(h, 1.0 - h, mode='full').max())
                h[j] = 0
                d = nm - cm
                if d < bd - 1e-14:
                    bd, bi, bj = d, i, j
            h[i] = 1
        if bi >= 0:
            h[bi], h[bj] = 0, 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            c = float(corr.max() * dx)
            if c < bc: bc, bh = c, h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = 0, 1
                corr = np.correlate(h, 1.0 - h, mode='full')
                stall = 0
    return bh, bc

def sa_bin(h_init, budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = frac_to_bin(h_init, n).copy()
    bh, bc = h.copy(), c5_of(h)
    T = 0.8
    t0 = time.time()
    while time.time() < t0 + budget - 0.2:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0, 1
        nc = c5_of(h)
        d = nc - bc
        if nc <= bc or (T > 0.005 and rng.random() < math.exp(max(-d / T, -700))):
            if nc < bc: bc, bh = nc, h.copy()
        else:
            h[i], h[j] = 1, 0
        T = max(0.003, T * 0.99998)
    return bh, bc
