    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        if resid > 0:
            fr = 1.0 - h
            idx = np.where(fr > 0)[0]
        else:
            fr = h
            idx = np.where(fr > 0)[0]
        if len(idx) > 0:
            order = np.argsort(fr[idx])[:max(1, int(np.ceil(abs(resid))))]
            idx_sel = idx[order]
            h[idx_sel] += resid / len(idx_sel)
            np.clip(h, 0.0, 1.0, out=h)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -400.0, 400.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def coord_descent(h, n, max_sweeps=50, deadline=None):

            break

            for s in step_sizes:

                        continue
                    h[i] = v
                    c = compute_c5(h)
                    if c < best_c:
                        best_c = c
                        best_v = v

def subgradient_step(h, n, lr, max_steps=100, deadline=None):

    best_h = h.copy()

        if deadline and time.time() > deadline:
            break
        dx_val = 2.0 / n
        corr_all = np.correlate(h, 1.0 - h, mode='full') * dx_val
        worst_idx = np.argmax(corr_all)
        worst_k = worst_idx - (n - 1)  # actual lag
        # Subgradient of max at this point: gradient of corr[worst_idx] wrt h
        # corr[j] = sum_i h[i] * (1 - h[i + j - (n-1)])
        # d corr[j] / d h[p] = (1 - h[p + j - (n-1)]) - h[p - (j - (n-1))] if in range
        # Simpler: use finite difference
        grad = np.zeros(n)

            h[p] = h_old - eps
            c_minus = np.correlate(h, 1.0 - h, mode='full')[worst_idx] * dx_val
            h[p] = h_old
            grad[p] = (c_plus - c_minus) / (2 * eps)

        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

For m < 0 (m' = -m > 0):

- p appears as h[i-m'] when i = p+m': term -h[p+m'] if p+m' <= n-1, i.e., p <= n-1-m'

So grad[p] = [p >= m'] * (1 - h[p-m']) - [p <= n-1-m'] * h[p+m']

def grad_worst_lag(h, worst_idx, n):
    dx_val = 2.0 / n
    m = worst_idx - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        # grad[p] = (1 - h[p+m]) * [p <= n-1-m] - h[p-m] * [m <= p]
        for p in range(n - m):
            grad[p] += (1.0 - h[p + m])
        for p in range(m, n):
            grad[p] -= h[p - m]
    else:
        mp = -m  # positive
        for p in range(mp, n):
            grad[p] += (1.0 - h[p - mp])
        for p in range(n - mp):
            grad[p] -= h[p + mp]
    return grad * dx_val

            h_best, c5_best = coord_descent(h_best, n, max_sweeps=15, deadline=dl5)

    # Phase 6: Perturbation restarts (10%)
    dl6 = t0 + budget_s * 0.90
    bounds = [(0.0, 1.0)] * n
    for attempt in range(5):
        if time.time() > dl6:
            break
        pert = 0.03 + 0.03 * rng.random()
        h_p = h_best + pert * rng.standard_normal(n)
        h_p = project_feasible(h_p, n)
        def obj6(x):
            return smooth_max_corr(x, 300.0) + 3000.0 * (np.sum(x) - n * 0.5) ** 2
        res = minimize(obj6, h_p.copy(), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 300, 'ftol': 1e-15})
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
            h_best, c5_best = coord_descent(h_best, n, max_sweeps=10, deadline=dl6)

    # Phase 7: Final polish
    dl7 = t0 + budget_s - 10
    if time.time() < dl7:
        h_best, c5_best = coord_descent(h_best, n, max_sweeps=20, deadline=dl7)
        h_best, c5_best = subgradient_descent(h_best, n, deadline=dl7)
        h_best, c5_best = coord_descent(h_best, n, max_sweeps=10, deadline=dl7)

    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
