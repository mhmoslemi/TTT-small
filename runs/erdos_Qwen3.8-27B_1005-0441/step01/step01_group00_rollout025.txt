                if val < best_val - 1e-14:
                    best_val = val
                    best_h = h_new
                    h = h_new
                    break

    if best_h is None:
        best_h = _proj(rng.rand(50), 50)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
