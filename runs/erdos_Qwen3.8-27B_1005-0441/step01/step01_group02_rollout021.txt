import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-13:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx

def project_polytope(h):
    n = len(h)
    target = n / 2.0
    # Bisection on tau for sum(clip(h-tau, 0, 1)) = target
    lo = h.min() - target
    hi = h.max()
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0.0, 1.0)

def gradient_momentum(h, beta, step_size, momentum, max_iter, t_budget):
    """Optimize with projected gradient + momentum"""
    n = len(h)
    h = project_polytope(h.copy().astype(float))
    v = np.zeros(n)
    best = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    lr = step_size
    for it in range(max_iter):
        if time.time() - t0 > t_budget:

        g = smooth_grad(h, beta)
        # Project gradient to sum=0 subspace (tangent to the constraint)
        g = g - g.mean()
        v = momentum * v - lr * g
        h_new = project_polytope(h + v)
        c_new = compute_c5(h_new)
        if c_new < best - 1e-14:
            best = c_new
            hb = h_new.copy()
        h = h_new
        # Adaptive step: if getting stuck, reduce lr

    return hb, best

def _set_coord(h, j, val):

    h_try[others] += diff / others.sum()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-13:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx

def project(h):
    n = len(h)
    target = n / 2.0
    lo = h.min() - target
    hi = h.max()
    for _ in range(70):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def _lbfgs_obj(x, beta, lam, target):
    hx = np.clip(x, 0.0, 1.0)
    return smooth_obj(hx, beta) + lam * (hx.sum() - target) ** 2

def _lbfgs_grad(x, beta, lam, target):
    hx = np.clip(x, 0.0, 1.0)
    g = smooth_grad(hx, beta).copy()
    g += 2.0 * lam * (hx.sum() - target)
    return g

def multi_scale_opt(h, t_budget, seed_val=0):
    n = len(h)
    h = project(h.copy().astype(float))
    best = compute_c5(h)
    hb = h.copy()
    v = np.zeros(n)
    t0 = time.time()
    scales = [(3, 0.12, 0.9, 400), (8, 0.06, 0.85, 500), (20, 0.03, 0.8, 500),
              (50, 0.015, 0.7, 500), (120, 0.008, 0.5, 400), (300, 0.004, 0.3, 300), (600, 0.002, 0.2, 200)]
    for beta, lr0, mom, maxit in scales:
        if time.time() - t0 > t_budget - 3:
            break
        lr = lr0
        for it in range(maxit):
            if time.time() - t0 > t_budget - 3:
                break
            g = smooth_grad(h, beta)
            g -= g.mean()
            v = mom * v - lr * g
            h_new = project(h + v)
            c_new = compute_c5(h_new)
            if c_new < best - 1e-15:
                best = c_new
                hb = h_new.copy()
            elif it > 50 and c_new > best + 0.002:
                v *= 0.5
            h = h_new
            if it % 150 == 149:
                lr *= 0.6
        v = np.zeros(n)
    return hb, best

def lbfgsb_phase(h, beta, t_budget):
    n = len(h)
    target = n / 2.0
    h = project(h.copy().astype(float))
    best = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    for lam in [0.5, 2.0, 8.0, 30.0]:
        if time.time() - t0 > t_budget - 3:
            break
        try:
            res = minimize(_lbfgs_obj, h.copy(), method='L-BFGS-B',
                          jac=_lbfgs_grad, args=(beta, lam, target),
                          options={'maxiter': 250, 'ftol': 1e-15, 'gtol': 1e-9})
            h_new = project(res.x)
        except Exception:
            break
        c = compute_c5(h_new)
        if c < best:
            best = c
            hb = h_new.copy()
        h = h_new
        if abs(h.sum() - target) < 1e-9:
            break
    return hb, best

def pairwise_search(h, n_trials, t_budget, seed_val=0):
    n = len(h)
    best = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    rng = np.random.RandomState(seed_val)
    for _ in range(n_trials):
        if time.time() - t0 > t_budget:
            break
        i, j = rng.choice(n, 2, replace=False)
        h_try = hb.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]
        h_try = project(h_try)
        c_try = compute_c5(h_try)
        if c_try < best - 1e-14:
            best = c_try
            hb = h_try
        for amt in [0.02, 0.05, 0.1]:
            h_try = hb.copy()
            tmp = h_try[i]
            h_try[i] = np.clip(tmp + amt, 0, 1)
            h_try[j] = np.clip(tmp - amt, 0, 1)
            h_try = project(h_try)
            c_try = compute_c5(h_try)
            if c_try < best - 1e-14:
                best = c_try
                hb = h_try
                break
    return hb, best

def fine_coord(h, t_budget):
    n = len(h)
    best = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    for p in range(4):
        for j in range(n):
            if time.time() - t0 > t_budget:
                return hb, best
            for delta in [0.001, -0.001, 0.002, -0.002, 0.0005, -0.0005, 0.003, -0.003]:
                h_try = hb.copy()
                h_try[j] = np.clip(h_try[j] + delta, 0, 1)
                h_try = project(h_try)
                c_try = compute_c5(h_try)
                if c_try < best - 1e-15:
                    best = c_try
                    hb = h_try
    return hb, best
