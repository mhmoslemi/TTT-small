def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999999
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-8
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:

        if diff > 0 and h[idx] < 1.0:
            h[idx] = min(1.0, h[idx] + diff)
        elif diff < 0 and h[idx] > 0.0:
            h[idx] = max(0.0, h[idx] + diff)
    return h

def corridor_k(k, n):
    """Valid i range for lag k: max(0,-k) <= i <= min(n-1, n-1-k)"""
    lo = max(0, -k)
    hi = min(n - 1, n - 1 - k)
    return lo, hi

def grad_dc(h, n):
    """Compute dc_k/dh_j for all k,j. Returns matrix of shape (2n-1, n)."""
    dx = 2.0 / n
    g = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        lo, hi = corridor_k(k, n)
        if lo > hi:
            continue
        for i in range(lo, hi + 1):
            # dc_k/dh[i+k] += (1 - h[i])
            j1 = i + k
            if 0 <= j1 < n:
                g[m, j1] += (1.0 - h[i])
            # dc_k/dh[i] += -h[i+k]
            j2 = i
            if 0 <= j2 < n:
                g[m, j2] -= h[i + k]
    return g * dx

def smooth_obj(h, n, beta, t_ref, pw):
    dx = 2.0 / n
    I = np.correlate(h, 1.0 - h, mode="full") * dx
    shifted = beta * (I - t_ref)
    m = shifted.max()
    smax = t_ref + (m + np.log(np.sum(np.exp(shifted - m)))) / beta
    sp = pw * (np.sum(h) - n / 2.0) ** 2
    return smax + sp

def smooth_grad(h, n, beta, t_ref, pw):
    dx = 2.0 / n
    I = np.correlate(h, 1.0 - h, mode="full") * dx
    shifted = beta * (I - t_ref)
    m = shifted.max()
    z = shifted - m
    exp_z = np.exp(z)
    p = exp_z / exp_z.sum()
    g = grad_dc(h, n)
    dg = p @ g
    dg += 2.0 * pw * (np.sum(h) - n / 2.0)
    return dg

    return smooth_obj(h, n, beta, t_ref, pw), smooth_grad(h, n, beta, t_ref, pw)
