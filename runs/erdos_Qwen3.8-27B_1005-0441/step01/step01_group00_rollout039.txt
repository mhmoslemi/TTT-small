import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0.0, 1.0)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_single(k, h, n):
    m = k - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    else:
        p = -m
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g

def _topk_smmax(h, active, eps):
    c = _corr(h)
    ca = c[active]
    cmax = ca.max()
    shifted = ca - cmax
    return cmax + eps * np.log(np.exp(shifted / eps).sum())

def _topk_grad(h, active, eps, n):
    c = _corr(h)
    ca = c[active]
    cmax = ca.max()
    shifted = ca - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for idx, k in enumerate(active):
        wk = s[idx] / ssum
        if wk < 1e-14:
            continue
        g += wk * _grad_single(k, h, n)
    return g * (2.0 / n)

def _run_lbfgs(h, n, active, eps, pen, maxit):
    dx = 2.0 / n

    def obj(x):
        v = _topk_smmax(x, active, eps) * dx

    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10, 'maxcor': 25})
        return _proj(res.x, n)
    except Exception:
        return _proj(h, n)

    best_val = _c5_val(h)
    lr = 0.03
    stall = 0
    for it in range(iters):

            break
        c = _corr(h)
        k1 = int(np.argmax(c))
        c2 = c.copy()
        c2[k1] = -np.inf
        k2 = int(np.argmax(c2))
        g = _grad_single(k1, h, n) * (2.0 / n)
        g += 0.4 * _grad_single(k2, h, n) * (2.0 / n)
        h_new = _proj(h - lr * g, n)
        v = _c5_val(h_new)
        if v < best_val - 1e-13:

            stall = 0
        else:
            stall += 1
            lr *= 0.985
            if stall > 300 or lr < 1e-9:
                break
    return best_h, best_val
