            lo = mid

            hi = mid

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

    grad = np.zeros(n)

        w = weights[k]
        if w < 1e-12:

        elif lag > 0:

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_weighted_grad(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            L = lag
            grad[:n-L] += w * (1.0 - h[L:])
            grad[L:] += -w * h[:n-L]
        else:
            M = -lag
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] += -w * h[M:]
    return grad * dx

import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_weighted_grad(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            L = lag
            grad[:n-L] += w * (1.0 - h[L:])
            grad[L:] += -w * h[:n-L]
        else:
            M = -lag
            grad[M:] += w * (1.0 - h[:n-M])
            grad[:n-M] += -w * h[M:]
    return grad * dx

def continuous_minimax(h_init, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_lags = 2 * n - 1
    p = np.ones(n_lags) / n_lags
    eta = 10.0
    lr = 0.12
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    while time.time() < deadline - 0.5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        scaled = np.clip(eta * corr, -50, 50)
        p = np.exp(scaled)
        p /= p.sum()
        grad = compute_weighted_grad(h, p, dx)
        step = lr / (1.0 + 0.015 * no_improve)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-11:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
            eta = min(eta * 1.008, 80.0)
        else:
            no_improve += 1
            if no_improve > 250:
                h = best_h + rng.randn(n) * 0.03
                h = project_feasible(h, target_sum)
                p = np.ones(n_lags) / n_lags
                eta = 10.0
                lr = 0.12
                no_improve = 0
            elif no_improve > 60:
                lr = max(lr * 0.97, 0.008)
        h = h_new
    return best_h, best_c5

def to_binary(h, target_count):
    n = len(h)
    idx = np.argsort(h)[::-1][:target_count]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def greedy_refine(h_bin, time_limit=5.0):
    h = h_bin.copy()
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, best_c5

def binary_sa(h_bin, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    temp = 0.025

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        if r < 0.65:

            j = zeros[rng.randint(len(zeros))]

            c5_new = compute_c5(h)
            d = c5_new - best_c5
            if d < -1e-14:

                best_h = h.copy()

            elif d < temp and rng.random() < math.exp(-d / max(temp, 1e-12)):

            else:

            h[i1], h[i2] = 0.0, 0.0
            h[j1], h[j2] = 1.0, 1.0

        if temp < 0.001 and rng.random() < 0.005:
            temp = 0.025
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    deadline = t0 + budget_s
    n = 70
    if 'initial_h_values' in globals() and initial_h_values is not None:
