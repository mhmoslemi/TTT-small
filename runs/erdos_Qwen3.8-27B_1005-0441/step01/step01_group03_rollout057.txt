If we think of it as a binary sequence of length n with n/2 ones, we want to minimize the maximum autocorrelation-like quantity. This is related to "Golay sequences" or "Zadoff-Chu sequences" in signal processing, but those are complex-valued. For binary, it's related to "optimal binary sequences" with low peak autocorrelation.

For a binary sequence with n/2 ones, the "overlap" at lag K counts the number of 1-followed-by-0 patterns at distance K (in both directions). We want all these counts to be as equal as possible (minimizing the max).

The total number of 1-0 pairs at distance K (for K > 0) is at most n/2 (since each 1 can have at most one 0 at distance K in each direction, but actually it's just the number of i where h[i]=1 and h[i+K]=0, plus h[i+K]=1 and h[i]=0... hmm).

Wait no, that's the "Hamming distance" at lag K. Let me re-examine.

correlate(h, 1-h, 'full') at the position corresponding to lag K:
= sum_i h[i] * (1-h[i-K]) (for appropriate range of i)

This counts the number of positions where h is 1 and h shifted by K is 0. It does NOT count the reverse (h is 0 and h shifted by K is 1). So it's a directed count.

Hmm wait, but the max is over all K from -(n-1) to n-1. So for K and -K, we get:
- K: count of (h[i]=1, h[i-K]=0)
- -K: count of (h[i]=1, h[i+K]=0) = count of (h[i+K]=1, h[i]=0) (by reindexing)

So max over all K includes both directions. The overlap at |K| in the symmetric sense is:
sum_i [h[i](1-h[i+K]) + (1-h[i])h[i+K]] = sum_i [h[i]≠h[i+K]]

This is the Hamming distance at lag K. And the directed version is:
sum_i h[i](1-h[i+K]) ≤ sum_i [h[i]≠h[i+K]]

Since we take max over all K (positive and negative), and for binary:
max(K>0) overlap[K] ≤ max(K>0) hamming_distance[K]

But they're not the same. The max of the directed correlation could be less than the Hamming distance.

Anyway, the key insight is: we want a binary sequence of length n with n/2 ones that minimizes the maximum number of 1s that have a 0 at distance K (for any K).

def find_best_swap(h, n):
    """Find the single swap that most reduces max overlap."""
    g = 1.0 - h
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    ov = np.correlate(h, g, mode='full')  # shape (2n-1,)
    current_max = ov.max()
    
    # For each (a, b) pair, compute delta for each lag
    # delta(a,b,k) = -g[a-k]*valid(a-k) + g[b-k]*valid(b-k) + h[a+k]*valid(a+k) - h[b+k]*valid(b+k) - 2*delta(0,k) + delta(a-b,k) + delta(b-a,k)
    # where k ranges from -(n-1) to n-1
    
    a_list = ones  # shape (n1,)
    b_list = zeros  # shape (n0,)
    n1, n0 = len(a_list), len(b_list)
    
    # k ranges: index m = 0 to 2n-2, k = m - (n-1) ranges from -(n-1) to n-1
    k_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For each pair (a, b):
    # delta[m] = -g[a-k[m]] (if 0<=a-k[m]<n) + g[b-k[m]] (if 0<=b-k[m]<n) 
    #            + h[a+k[m]] (if 0<=a+k[m]<n) - h[b+k[m]] (if 0<=b+k[m]<n)
    #            - 2*(k[m]==0) + (k[m]==a-b) + (k[m]==b-a)
    
    # Vectorize: create arrays of shape (n1, n0, 2n-1)
    # a[:, None, :] - b[None, :, :] ... this might use too much memory for large n
    # For n=46: 23 * 23 * 91 * 8 bytes ≈ 3.7 MB. Fine.
    
    a_ext = a_list[:, None, None]  # (n1, 1, 1)
    b_ext = b_list[None, :, None]  # (1, n0, 1)
    k_ext = k_vals[None, None, :]  # (1, 1, 2n-1)
    
    a_minus_k = a_ext - k_ext  # (n1, n0, 2n-1)
    b_minus_k = b_ext - k_ext
    a_plus_k = a_ext + k_ext
    b_plus_k = b_ext + k_ext
    
    valid_ak = (a_minus_k >= 0) & (a_minus_k < n)
    valid_bk = (b_minus_k >= 0) & (b_minus_k < n)
    valid_apk = (a_plus_k >= 0) & (a_plus_k < n)
    valid_bpk = (b_plus_k >= 0) & (b_plus_k < n)
    
    # g[a-k], g[b-k], h[a+k], h[b+k]
    # Need to index into g and h
    g_vals = g  # shape (n,)
    h_vals = h
    
    ga_k = np.where(valid_ak, g_vals[np.clip(a_minus_k, 0, n-1)], 0)
    gb_k = np.where(valid_bk, g_vals[np.clip(b_minus_k, 0, n-1)], 0)
    ha_k = np.where(valid_apk, h_vals[np.clip(a_plus_k, 0, n-1)], 0)
    hb_k = np.where(valid_bpk, h_vals[np.clip(b_plus_k, 0, n-1)], 0)
    
    delta_0 = -2.0 * (k_ext == 0)
    delta_ab = (k_ext == (a_ext - b_ext))
    delta_ba = (k_ext == (b_ext - a_ext))
    
    delta = -ga_k + gb_k + ha_k - hb_k + delta_0 + delta_ab + delta_ba
    
    # New overlap for each (a,b) pair: ov + delta
    new_ov = ov[None, None, :] + delta  # (n1, n0, 2n-1)
    new_max = new_ov.max(axis=2)  # (n1, n0)
    
    idx = np.unravel_index(new_max.argmin(), new_max.shape)
    min_val = new_max.min()
    
    if min_val < current_max - 1e-12:
        return int(a_list[idx[0]]), int(b_list[idx[1]]), float(min_val)
    return None, None, float(current_max)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

    b = (h >= 0.5).astype(float)

    target = n // 2

    d0 = -2.0 * (k_ext == 0)
    dab = (k_ext == (a_ext - b_ext)).astype(float)
    dba = (k_ext == (b_ext - a_ext)).astype(float)
    delta = -ga_k + gb_k + ha_k - hb_k + d0 + dab + dba
    new_max = (ov[None, None, :] + delta).max(axis=2)
    idx = np.unravel_index(new_max.argmin(), new_max.shape)
    min_val = float(new_max.min())
    if min_val < current_max - 1e-12:
        return int(a_list[idx[0]]), int(b_list[idx[1]]), min_val
    return None, None, current_max

def local_search(h, n, deadline, rng):
    k_vals = np.arange(-(n - 1), n)
    c5 = get_c5(h, n)
    h_best = h.copy()
    c5_best = c5
    no_improve = 0
    g = 1.0 - h
    while time.time() < deadline:
        g = 1.0 - h
        a, b, c5_new = find_best_swap(h, n, k_vals, g)
        if a is not None:
            h[a] = 0.0
            h[b] = 1.0
            c5 = c5_new
            no_improve = 0
            if c5 < c5_best - 1e-12:
                c5_best = c5
                h_best = h.copy()
        else:
            no_improve += 1
            if no_improve in (15, 35, 70, 130):
                nswaps = no_improve // 10 + 1
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                for _ in range(nswaps):
                    aa = oi[rng.randint(len(oi))]
                    bb = zi[rng.randint(len(zi))]
                    h[aa], h[bb] = 0.0, 1.0
                c5 = get_c5(h, n)
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
            elif no_improve > 250:
                break
    return h_best, c5_best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    max_time = t0 + budget_s - 5
    best_h = None
    best_c5 = 1.0
    best_n = 0
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    if h0 is not None:
        n0 = len(h0)
        n_list = [n0]
    else:
        n0 = 46
        n_list = [46]
    extra = [44, 48, 50, 42, 52, 40, 54, 38, 56, 36, 60, 64, 34, 68, 72, 76, 80]
    for n in extra:
        if n % 2 == 0 and n not in n_list:
            n_list.append(n)
    time_first = (max_time - t0) * 0.50
    time_other = (max_time - t0) * 0.40
    time_polish = (max_time - t0) * 0.10
    # Phase 1: optimize initial with multiple starts
    if h0 is not None:
        h_start = to_binary(h0, n_list[0], rng)
    else:
        h_start = block_init(n_list[0])
    n_starts_first = 4
    ts = time_first / n_starts_first
    for s in range(n_starts_first):
        if time.time() > t0 + time_first:
            break
        if s == 0:
            h_s = h_start.copy()
        elif s == 1:
            h_s = block_init(n_list[0])
        elif s == 2:
            h_s = random_init(n_list[0], rng)
        else:
            h_s = random_init(n_list[0], rng)
        dl = min(time.time() + ts, t0 + time_first)
        h_opt, c5_opt = local_search(h_s, n_list[0], dl, rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_list[0]
    # Phase 2: other n values
    time_per_n = time_other / max(1, len(n_list) - 1)
    for n in n_list[1:]:
        if time.time() > t0 + time_first + time_other - 3:
            break
        ava = min(time_per_n, t0 + time_first + time_other - 3 - time.time())
        if ava < 4:
            ava = 4
        dl_n = time.time() + ava
        ts2 = dl_n - time.time()
        for s in range(3):
            if time.time() > dl_n:
                break
            if s == 0:
                h_s = block_init(n)
            else:
                h_s = random_init(n, rng)
            dl = min(time.time() + ts2 / (3 - s), dl_n)
            h_opt, c5_opt = local_search(h_s, n, dl, rng)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n
    # Phase 3: final polish
    if best_h is not None and time.time() < max_time - 2:
        h_opt, c5_opt = local_search(best_h.copy(), best_n, max_time, rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    if best_h is None:
        best_h = block_init(46)
        best_n = 46
        best_c5 = get_c5(best_h, best_n)
    return best_h, float(best_c5), best_n
