    corr = np.correlate(h, 1.0 - h, mode='full')

    target = n / 2.0

        if abs(diff) < 1e-14:

            i = int(np.argmax(room))

            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub

    # x is h values (already satisfying bounds, sum constraint handled by penalty)
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(corr)
    # Log-sum-exp for smoothness
    shifted = corr - cmax
    lse = cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    # Penalty for sum constraint
    sum_pen = 1e4 * (np.sum(x) - n/2.0)**2
    # Penalty for bounds (shouldn't trigger if L-BFGS-B handles them)
    return lse + sum_pen

def _optimize_lse(h0, alpha, method='L-BFGS-B', max_iter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_obj, h0, args=(alpha,), method=method,
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-14})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h0.copy()

def _grad_lse(corr, alpha):
    """Gradient of logsumexp(alpha * corr) / alpha w.r.t. corr, then chain rule."""
    cmax = np.max(corr)
    shifted = corr - cmax
    max_s = np.max(shifted)
    if max_s * alpha > 500:
        shifted = shifted - max_s
        cmax += max_s
    weights = np.exp(alpha * shifted)
    weights /= np.sum(weights)
    return weights  # d/d(corr) of the smoothed max

def _subgrad_step(h, learning_rate):
    """One subgradient step for the max overlap problem."""
    n = len(h)
    dx = 2.0 / n
    corr = _corr_values(h)
    cmax = np.max(corr)
    active = np.where(corr > cmax - 1e-8)[0]
    # Subgradient w.r.t. h: for each active lag k, gradient of corr[k] w.r.t. h
    # corr[k] = sum_i h[i] * (1-h[i+k]) * dx (approximate)
    # d(corr[k])/d(h[j]) = dx * [(1-h[j+k])_if_exists + (-h[j-k])_if_exists]
    # Wait, let me be more careful.
    
    # For np.correlate(h, 1-h, 'full'), the result at index m (0-indexed, length 2N-1):
    # result[m] = sum_i h[i] * (1-h[i + m - (N-1)]) for valid i
    # Let lag = m - (N-1)
    # result[m] = sum_{i: valid} h[i] * (1 - h[i+lag])
    # = sum_{i:valid} h[i] - sum_{i:valid} h[i]*h[i+lag]
    
    # d result[m] / d h[j]:
    # From sum h[i]: contributes dx if j is in range
    # From -sum h[i]*h[i+lag]: 
    #   if j = i: contributes -h[j+lag] * dx
    #   if j = i+lag: contributes -h[j-lag] * dx
    # So total: dx * (1 - h[j+lag] - h[j-lag]) (with boundary: missing terms are 0)
    
    grad = np.zeros(n)
    N = n
    for m in active:
        lag = m - (N - 1)
        for j in range(N):
            g = dx
            # -h[j+lag] if j+lag in [0, N-1]
            if 0 <= j + lag < N:
                g -= h[j + lag]
            # -h[j-lag] if j-lag in [0, N-1] (this is the other factor)
            # Wait, I need to be more careful.
            # result[m] = sum_i h[i] * (1-h[i+lag]) for valid (i, i+lag)
            # = sum_i h[i] - sum_i h[i]*h[i+lag]
            # d/dh[j] of sum_i h[i] = 1 (if j in range)
            # d/dh[j] of sum_i h[i]*h[i+lag]:
            #   term where i=j: h[i+lag] = h[j+lag] (if valid)
            #   term where i+lag=j, i.e., i=j-lag: h[i] = h[j-lag] (if valid)
            if 0 <= j + lag < N:
                g -= h[j + lag]
            if 0 <= j - lag < N:
                g -= h[j - lag]
            grad[j] += g / len(active)
    
    h_new = h - learning_rate * grad
    h_new = _project_box_sum(h_new, n / 2.0)
    return h_new, cmax

def _subgrad_descent(h0, n_iters=200, lr_start=0.05):
    h = h0.copy()
    best = _eval_c5(h)
    best_h = h.copy()
    for it in range(n_iters):

            best = c5
            best_h = h_new.copy()
        h = h_new

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-16:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-16:
                break
            h[i] -= sub
    return h

def _project_box_sum(h, target):
    def s(clip_shift):
        return np.sum(np.clip(h - clip_shift, 0.0, 1.0))
    lo, hi = -2.0, 2.0
    if s(lo) < target:
        return np.clip(h - lo, 0, 1)
    if s(hi) > target:
        return np.clip(h - hi, 0, 1)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if s(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _lse_smooth(corr, alpha):
    cmax = np.max(corr)
    shifted = corr - cmax
    ms = np.max(shifted)
    if ms * alpha > 500:
        shifted -= ms
        cmax += ms
    if alpha < 1:
        return cmax
    return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha

        if k >= 0:

                grad[k:] += 1.0 - h[:n - k]

            else:

                grad[:] -= h

    t = 0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (bin(i).count('1') % 2 == 0) else 0.0
    return _project_box_sum(h, n / 2.0)

def _perturb(h, rng, scale):
    return _project_box_sum(h + rng.normal(0, scale, len(h)), len(h) / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    def tl():
        return budget_s - (time.time() - t0)

    # Load initial
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if h_init is not None:
        h_init = _fix_sum(h_init, len(h_init))
        c5 = _eval_c5(h_init)
        best_c5 = c5
        best_h = h_init.copy()

    # Phase 1: SLSQP with increasing alpha on initial
    if h_init is not None and tl() > 150:
        for alpha in [30, 80, 200, 500, 1000, 2000]:
            if tl() < 50:
                break
            h_opt = _optimize_slsqp(h_init, alpha, 250)
            c5 = _eval_c5(h_opt)
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 2: Subgradient descent on best
    if best_h is not None and tl() > 60:
        for lr in [0.04, 0.02, 0.01]:
            if tl() < 20:
                break
            h_sg, c5_sg = _subgrad_descent(best_h, 400, lr)
            h_sg = _fix_sum(h_sg, len(h_sg))
            c5_sg = _eval_c5(h_sg)
            if c5_sg < best_c5 - 1e-9:
                best_c5 = c5_sg
                best_h = h_sg.copy()

    # Phase 3: Try various n with structured starts
    n_list = [16, 20, 24, 30, 40, 50, 60, 70, 90, 100, 120, 160, 200]
    for n in n_list:
        if tl() < 25:
            break
        starts = [
            _random_h(n, rng),
            _periodic_h(n, 3, 2.0/3),
            _periodic_h(n, 4, 0.5),
            _periodic_h(n, 5, 2.0/5),
            _sin_h(n, 1, 0.4),
            _sin_h(n, 2, 0.35),
            _sin_h(n, 3, 0.3),
            _thue_morse_h(n, rng),
        ]
        for h0 in starts:
            if tl() < 15:
                break
            if n <= 60:
                h_opt = _optimize_slsqp(h0, 200, 300)
            else:
                h_opt = _fix_sum(h0, n)
                h_opt, _ = _subgrad_descent(h_opt, 200, 0.03)
            c5 = _eval_c5(h_opt)
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 4: Perturbation + re-opt on best
    if best_h is not None and tl() > 50:
        n = len(best_h)
        for scale in [0.06, 0.03, 0.015, 0.005]:
            for _ in range(8):
                if tl() < 15:
                    break
                h_p = _perturb(best_h, rng, scale)
                if n <= 100:
                    h_opt = _optimize_slsqp(h_p, 300, 250)
                else:
                    h_opt, _ = _subgrad_descent(h_p, 200, 0.02)
                c5 = _eval_c5(h_opt)
                if c5 < best_c5 - 1e-9:
                    best_c5 = c5
                    best_h = h_opt.copy()

    # Phase 5: Final subgradient polish with very small step
    if best_h is not None and tl() > 20:
        h_sg, _ = _subgrad_descent(best_h, 500, 0.005)
        h_sg = _fix_sum(h_sg, len(h_sg))
        c5 = _eval_c5(h_sg)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_sg

    # Phase 6: Final SLSQP
    if best_h is not None and tl() > 15 and len(best_h) <= 120:
        h_opt = _optimize_slsqp(best_h, 1500, 500)
        h_opt = _fix_sum(h_opt, len(h_opt))
        c5 = _eval_c5(h_opt)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_opt

    if best_h is None:
        best_h = _project_box_sum(np.full(50, 0.5), 25.0)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
