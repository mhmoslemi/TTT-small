numpy.correlate(a, v, mode='full'):
The output has length len(a) + len(v) - 1 = 2n - 1 (when a and v have the same length n).

The definition is: c[k] = sum_n a[n+k] * conj(v[n])

where the sum is over all n for which both n+k and n are valid indices.

The output array is indexed from 0 to 2n-2. The relationship between the output index and k is:
- Output index 0 corresponds to k = -(n-1)
- Output index n-1 corresponds to k = 0
- Output index 2n-2 corresponds to k = n-1

So for output index p: k = p - (n-1)

c[p] = sum_n h[n + p - (n-1)] * (1-h[n])

Let me verify with a simple example. If n=3, h = [a, b, c], v = [1-a, 1-b, 1-c]:
- p=0, k=-2: c[0] = sum h[n-2]*(1-h[n]) → only n=2: h[0]*(1-h[2]) = a*(1-c)
- p=1, k=-1: c[1] = sum h[n-1]*(1-h[n]) → n=1: h[0]*(1-h[1]) + n=2: h[1]*(1-h[2]) = a*(1-b) + b*(1-c)

For p=0, k=-2: c[0] = h[0]*(1-h[2])
∂c[0]/∂h[0] = 1-h[2]
∂c[0]/∂h[2] = -h[0]
∂c[0]/∂h[1] = 0

General formula: c[p] = sum_{n: valid} h[n+k]*(1-h[n]) where k = p-(n-1)
∂c[p]/∂h[j]:
- Terms where n+k = j, i.e., n = j-k: contribution (1-h[n]) = (1-h[j-k]) if 0≤j-k<n
- Terms where n = j: contribution -h[n+k] = -h[j+k] if 0≤j+k<n

So ∂c[p]/∂h[j] = (1-h[j-k]) * I(0≤j-k<n) - h[j+k] * I(0≤j+k<n)

where k = p-(n-1).

F = (1/beta) * log(sum_p exp(beta * c[p]))) * dx (plus constant m*dx which doesn't affect gradient)

where k_p = p-(n-1).

∂F/∂h[j] = dx * sum_{ℓ=-(n-1)}^{n-1} w[ℓ+n-1] * [(1-h[j-ℓ]) * I(0≤j-ℓ<n) - h[j+ℓ] * I(0≤j+ℓ<n)]

First term: sum_ℓ w[ℓ+n-1] * (1-h[j-ℓ]) * I(0≤j-ℓ<n)
Let i = j-ℓ, so ℓ = j-i. Range: 0≤i<n and -(n-1)≤j-i≤n-1.
= sum_{i=0}^{n-1} (1-h[i]) * w[j-i+n-1] * I(0≤j-i+n-1<2n-1)

The index j-i+n-1 ranges from (0-(n-1)+n-1)=0 to ((n-1)-0+n-1)=2n-2. So it's always valid!

First term = sum_{i=0}^{n-1} (1-h[i]) * w[j-i+n-1]

Second term: sum_ℓ w[ℓ+n-1] * h[j+ℓ] * I(0≤j+ℓ<n)
Let i = j+ℓ, so ℓ = i-j. Range: 0≤i<n and -(n-1)≤i-j≤n-1.

Again, i-j+n-1 ranges from (0-(n-1)+n-1)=0 to ((n-1)-0+n-1)=2n-2. Always valid!

Second term = sum_{i=0}^{n-1} h[i] * w[i-j+n-1]

So: ∂F/∂h[j] = dx * [sum_i (1-h[i]) * w[j-i+n-1] - sum_i h[i] * w[i-j+n-1]]

Note that w[j-i+n-1] and w[i-j+n-1] are related: w[j-i+n-1] = w_[(j-i)+(n-1)] and w[i-j+n-1] = w_[(i-j)+(n-1)].

Second term = (A^T @ h)[j] (since A[i,j] = w[i-j+n-1])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, n, beta):
    dx = 2.0 / n
    c = get_corr(h)
    m = c.max()
    s = beta * (c - m)
    s -= s.max()
    return (m + np.log(np.sum(np.exp(s))) / beta) * dx

def build_weight_matrix(n, w):
    idx = np.arange(n)
    W_idx = idx[:, None] - idx[None, :] + (n - 1)
    return w[W_idx]

def smooth_grad(h, n, beta):
    dx = 2.0 / n
    c = get_corr(h)
    m = c.max()
    s = beta * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    M = build_weight_matrix(n, w)
    return dx * (M @ (1.0 - h) - M.T @ h)

def cstr_fun(x):
    return np.sum(x) - len(x) / 2.0

def cstr_jac(x):
    return np.ones(len(x))

def proj_grad_opt(h, n, t_end):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    betas = [5, 20, 100, 500, 2000, 8000, 30000]
    for beta in betas:
        if time.time() > t_end:
            break
        h_cur = best_h.copy()
        lr0 = 200.0 / (beta + 50)
        for it in range(400):
            if time.time() > t_end:
                break
            g = smooth_grad(h_cur, n, beta)
            gn = np.linalg.norm(g)
            if gn < 1e-13:
                break
            f_cur = smooth_obj(h_cur, n, beta)
            step = lr0
            h_new = h_cur.copy()
            f_new = f_cur
            for _ in range(40):
                h_try = project_feasible(h_cur - step * g, n)
                f_try = smooth_obj(h_try, n, beta)
                if f_try < f_cur - 1e-16:
                    h_new = h_try
                    f_new = f_try
                    break
                step *= 0.5
            else:
                break
            h_cur = h_new
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
    # Subgradient polish on exact max
    if time.time() < t_end:
        h_cur = best_h.copy()
        for it in range(200):
            if time.time() > t_end:
                break
            c = get_corr(h_cur)
            m = c.max()
            active = c > m - 1e-7
            if np.sum(active) == 0:
                active = c > m - 0.01
            w = active.astype(float)
            w /= w.sum()
            M = build_weight_matrix(n, w)
            dx = 2.0 / n
            g = dx * (M @ (1.0 - h_cur) - M.T @ h_cur)
            step = 0.1 / (1.0 + it * 0.1)

            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h_cur = h_try
    return best_h, best_c5

def binary_local_search(h, n, t_end, max_sweeps=80):
    h_bin = (h > 0.5).astype(float)
    target_ones = n // 2
    cur_ones = int(np.sum(h_bin))
    if cur_ones > target_ones:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:cur_ones - target_ones]:
            h_bin[ones[idx]] = 0.0
    elif cur_ones < target_ones:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:target_ones - cur_ones]:
            h_bin[zeros[idx]] = 1.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng_l = np.random.RandomState(7)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        n_tries = min(8000, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = ones[rng_l.randint(len(ones))]
            j = zeros[rng_l.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_t = compute_c5(h_bin)

                best_h = h_bin.copy()
                improved = True
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            break
    return best_h, best_c5

def make_block_init(n, n_blocks, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    bs = n // n_blocks
    for b in range(n_blocks):
        s = b * bs
        e = s + bs if b < n_blocks - 1 else n
        if rng.rand() > 0.5:
            h[s:e] = 1.0
    return project_feasible(h, n)

def slsqp_polish(h, n, beta, t_end):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cstr_fun, 'jac': cstr_jac}]
    try:
        res = minimize(smooth_obj, h, args=(n, beta), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-13})
        if time.time() > t_end:
            return h, compute_c5(h)
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h, compute_c5(h)
