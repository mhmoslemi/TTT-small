```python
def projected_gradient_step(h, lr, n):
    c = np.correlate(h, 1-h, mode="full")
    max_idx = np.argmax(c)
    k = max_idx - (n-1)  # actual lag
    
    # Compute gradient of c[k]
    grad = np.zeros(n)
    # For each m, grad[m] = (1-h[m+k]) if m+k<n - h[m-k] if m-k>=0
    if k >= 0:
        grad[:n-k] += (1 - h[k:])  # from (1-h[m+k]) term
        grad[k:] += -h[:n-k]       # from -h[m-k] term
    else:
        k_abs = -k
        grad[k_abs:] += (1 - h[:n-k_abs])
        grad[:n-k_abs] += -h[k_abs:]
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0, 1)
    # Fix sum
    target = n / 2
    diff = target - np.sum(h_new)
    h_new += diff / n
    h_new = np.clip(h_new, 0, 1)
    # Re-fix sum
    h_new = project_feasible(h_new, n)
    return h_new, c[max_idx] * (2.0/n)
```

```python
def gradient_refine(h, n, lr, max_iters):
    h = h.copy()
    for _ in range(max_iters):
        c = np.correlate(h, 1-h, mode="full")
        k_star = c.argmax()
        # Gradient of c at lag k_star
        k = k_star - (n-1)
        grad = np.zeros(n)
        if k >= 0:
            grad[:n-k] += (1 - h[k:])
            grad[k:] -= h[:n-k]
        else:
            ka = -k
            grad[ka:] += (1 - h[:n-ka])
            grad[:n-ka] -= h[ka:]

        h_new = project_feasible(h_new, n)

        if c_new < c[k_star]:
            h = h_new
        else:
            # Halve learning rate
            lr /= 2
            if lr < 1e-10:
                break
    return h
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-9
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    h_bin = (h > 0.5).astype(float)
    need = int(n / 2 - h_bin.sum())
    if need > 0:
        zero_pos = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_pos])
        h_bin[zero_pos[order[:need]]] = 1.0
    elif need < 0:
        need = -need
        one_pos = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_pos])
        h_bin[one_pos[order[:need]]] = 0.0
    return h_bin

def build_S(h, n):
    d = np.arange(-(n-1), n)
    i = np.arange(n)
    idx_fwd = i[:, None] + d[None, :]
    idx_bwd = i[:, None] - d[None, :]
    valid_fwd = (idx_fwd >= 0) & (idx_fwd < n)
    valid_bwd = (idx_bwd >= 0) & (idx_bwd < n)
    safe_fwd = np.clip(idx_fwd, 0, n-1)
    safe_bwd = np.clip(idx_bwd, 0, n-1)
    return np.where(valid_fwd, h[safe_fwd], 0.0) + np.where(valid_bwd, h[safe_bwd], 0.0)

    S = build_S(h, n)
    best_c5 = c.max() * dx

        if len(one_pos) == 0 or len(zero_pos) == 0:

        best_gain = 1e-12
        best_i, best_j = -1, -1
        best_new_max = c.max()

            deltas = S[zero_pos]  # (len(zero), 2n-1)

            new_maxes = new_c.max(axis=1)
            best_local = new_maxes.min()
            if best_local < best_new_max - best_gain * (1/n):  # scale for comparison
                best_gain = best_new_max - best_local
                best_i = i
                best_j = zero_pos[new_maxes.argmin()]
                best_new_max = best_local

                best_j = zero_pos[np.random.randint(len(zero_pos))]
            h[best_i] = 0
            h[best_j] = 1
            c = np.correlate(h, 1-h, mode="full")
            S = build_S(h, n)
            cur_max = c.max()
            if cur_max < best_c5:
                best_c5 = cur_max * dx
            # Annealing: reset temperature
            if temperature > 0:
                temperature *= 0.9
        
        else:
            if temperature <= 0:
                break
