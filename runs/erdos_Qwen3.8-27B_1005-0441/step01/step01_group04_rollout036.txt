    h = np.array(h, dtype=float)

    idx = np.argsort(-h)[:target]

Actually, I want to double-check: is the polytope {h ∈ [0,1]^n : sum h = n/2} really vertexed at 0/1 vectors when n/2 is an integer?

The polytope P = {h : 0 ≤ h[i] ≤ 1 for all i, sum h[i] = n/2}.
A vertex of P is a point where n linearly independent constraints are tight. The constraints are: n/2 from sum=const, and n from 0≤h[i]≤1. At a vertex, we need n independent tight constraints. The sum constraint is 1. We need n-1 more, which means n-1 of the h[i] are at 0 or 1. The remaining h[i] is determined by the sum constraint: h[i] = n/2 - sum of the others. If the others are all 0/1, then h[i] = n/2 - (number of 1s among the others), which is an integer between 0 and 1 only if it's 0 or 1. Since there are n-1 others each 0 or 1, the sum is an integer between 0 and n-1, so n/2 - sum is in [n/2 - (n-1), n/2] = [1 - n/2, n/2]. For this to be in [0,1], we need n/2 - sum ∈ {0, 1} (since it's an integer). So sum = n/2 or sum = n/2 - 1. If sum = n/2, then h[i] = 0. If sum = n/2 - 1, then h[i] = 1. Both cases give a 0/1 vector. ✓

So yes, all vertices are 0/1 vectors. The minimum of max of bilinear functions over P is at a vertex. Confirmed.

        h[i], h[j] = h[j], h[i]
        oi = ones.index(i)
        oj = zeros.index(j)
        ones[oi] = j
        zeros[oj] = i
        new_c5 = _eval_c5(h)
        if new_c5 <= curr_c5 or rng.random() < np.exp(-(new_c5 - curr_c5) / max(t, 1e-10)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            ones[ones.index(j)] = i
            zeros[zeros.index(i)] = j
    return best_h, best_c5

def _hill_climb(h, rng, max_sweeps=40):
    n = len(h)
    h = h.copy()
    curr_c5 = _eval_c5(h)
    for _ in range(max_sweeps):
        improved = False
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            best_j = -1
            best_c5_val = curr_c5
            for j in zeros:
                if j == i:
                    continue
                h[i], h[j] = h[j], h[i]
                c5 = _eval_c5(h)
                h[i], h[j] = h[j], h[i]
                if c5 < best_c5_val:
                    best_c5_val = c5
                    best_j = j
            if best_j >= 0:
                oi = ones.index(i)
                oj = zeros.index(best_j)
                h[i], h[best_j] = h[best_j], h[i]
                ones[oi] = best_j
                zeros[oj] = i
                curr_c5 = best_c5_val
                improved = True
        if not improved:
            break
    return h, curr_c5

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h

def _optimize_cont(h0, max_iter=200):
    n = len(h0)
    dx = 2.0 / n
    alpha = 200.0
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr)
        shifted = corr - cmax
        return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    def eq(h):
        return float(np.sum(h) - n / 2.0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': eq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    def remaining():
        return budget_s - (time.time() - t0)

    candidates = []
    h_init_raw = None
    try:
        h_init_raw = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init_raw is not None:
        n0 = len(h_init_raw)
        candidates.append(_make_binary(h_init_raw, n0))

    for n in [60, 70, 80, 90, 100]:
        candidates.append(_random_binary(n, rng))
        candidates.append(_golden_binary(n, rng))
        candidates.append(_thue_morse_binary(n, rng))
        candidates.append(_spaced_binary(n, rng, 0))

    # Phase 1: SA on each candidate with limited budget
    n_cands = len(candidates)
    for idx in range(n_cands):
        if remaining() < 25:
            break
        h_cand = candidates[idx]
        n = len(h_cand)
        iters = max(2000, int(80000 / max(n_cands - idx, 1)))
        h_opt, c5 = _sa_search(h_cand, rng, min(iters, 20000), t_init=0.012, t_min=0.0003)
        c5 = _eval_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Phase 2: focused SA from best with perturbations
    if remaining() > 50 and best_h is not None:
        for r in range(8):
            if remaining() < 20:
                break
            if r == 0:
                hs = best_h.copy()
            else:
                hs = best_h.copy()
                for _ in range(r * 8):
                    ones_p = np.where(hs == 1)[0]
                    zeros_p = np.where(hs == 0)[0]
                    i = ones_p[rng.integers(len(ones_p))]
                    j = zeros_p[rng.integers(len(zeros_p))]
                    hs[i], hs[j] = hs[j], hs[i]
            h_opt, c5 = _sa_search(hs, rng, 12000, t_init=0.006, t_min=0.0001)
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 3: Hill climb polish
    if remaining() > 15 and best_h is not None:
        h_pol, c5_pol = _hill_climb(best_h, rng, max_sweeps=40)
        c5_pol = _eval_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: More SA with different temperatures
    if remaining() > 30 and best_h is not None:
        for ti in [0.004, 0.001, 0.025, 0.04]:
            if remaining() < 12:
                break
            h_opt, c5 = _sa_search(best_h, rng, 10000, t_init=ti, t_min=0.0001)
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 5: Continuous optimization attempt
    if remaining() > 25 and best_h is not None:
        h_cont = best_h.astype(float) + 0.04 * rng.random(len(best_h))
        h_opt_c = _optimize_cont(h_cont, 200)
        h_opt_c = _fix_sum(h_opt_c, len(h_opt_c))
        c5_c = _eval_c5(h_opt_c)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_opt_c.copy()

    # Phase 6: Final hill climb
    if remaining() > 8 and best_h is not None:
        h_f, c5_f = _hill_climb(best_h, rng, max_sweeps=25)
        c5_f = _eval_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if best_h is None:
        best_h = np.full(80, 0.5)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
