from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_and_grad(h):
    """Compute all correlation values and provide gradient info."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr, dx

def subgradient(h, active_set):
    """Compute subgradient of max over active_set."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_set:
        d = k - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i + d < n:
                g += (1.0 - h[i + d])
            if 0 <= i - d < n:
                g -= h[i - d]
            grad[i] += g
    grad /= len(active_set)
    grad *= dx
    return grad

def project_onto_simplex_box(h, target):
    """Project h to [0,1]^n with sum = target."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        # Find free coordinates
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not free.any():
            # Hard to adjust, use all
            free = np.ones(n, dtype=bool)
        nfree = free.sum()
        step = diff / nfree
        h[free] = np.clip(h[free] + step, 0.0, 1.0)
    # Final exact adjustment
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-15:
        # Distribute among coordinates with most room
        if diff > 0:
            order = np.argsort(-h)  # most room to increase: those nearest 0
            # Actually, to increase sum, increase those that can go up
            order = np.argsort(h)  # smallest first (most room to go up)
        else:
            order = np.argsort(-h)  # largest first (most room to go down)
        acc = 0.0
        for idx in order:
            if abs(acc) < 1e-15:
                break
            room = (1.0 - h[idx]) if diff > 0 else h[idx]
            adj = min(abs(diff - acc), room)
            if diff > 0:
                h[idx] += adj
            else:
                h[idx] -= adj
            acc += adj if diff > 0 else -adj
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta

def smooth_c5_grad(h, beta):

    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    corr = raw * dx

            # Spread over all
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / free.sum(), 0.0, 1.0)
    # Exact fix
    diff = target - h.sum()

        # Distribute to coordinates with most room
        if diff > 0:

        else:

        remaining = diff
        for i in idx:

                adj = min(remaining, 1.0 - h[i])
                h[i] += adj
                remaining -= adj
            elif remaining < 0:
                adj = max(remaining, -h[i])
                h[i] += adj
                remaining -= adj
            if abs(remaining) < 1e-15:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    val = mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta
    return val

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        lo1, hi1 = max(0, -d), min(n, n - d)
        grad[lo1:hi1] += wk * (1.0 - h[lo1 + d:hi1 + d])
        lo2, hi2 = max(0, d), min(n, n + d)
        grad[lo2:hi2] -= wk * h[lo2 - d:hi2 - d]
    return grad * dx

    best_c = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):

            s = h[i]  # current value

            j = (i + 1) % n

            lo = max(0.0, s_total - 1.0)
            hi = min(1.0, s_total)
            if hi - lo < 1e-12:
                continue
            
            best_t = h[i]
            best_val = best_c
            
            # Grid search (fine)
            for t in np.linspace(lo, hi, 200):
                h[i] = t
                h[j] = s_total - t
                if h[j] < -1e-12 or h[j] > 1.0 + 1e-12:
                    continue
                c = compute_c5(h)
                if c < best_val - 1e-14:
                    best_val = c
                    best_t = t
            # Refine with scalar minimization
            h[i] = best_t
            h[j] = s_total - best_t
            
            if best_val < best_c - 1e-14:
                best_c = best_val
                h[i] = best_t
                h[j] = s_total - best_t
                improved = True
            # Restore if no improvement
            h[i] = best_t
            h[j] = s_total - best_t
        if not improved:
            break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not free.any():
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] = np.clip(h[free] + diff / free.sum(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(h) if diff > 0 else np.argsort(-h)
        rem = diff
        for i in order:
            if abs(rem) < 1e-14:
                break
            if rem > 0:
                a = min(rem, 1.0 - h[i]); h[i] += a; rem -= a
            else:
                a = max(rem, -h[i]); h[i] += a; rem -= a
    return h

def _smooth(x, beta, target, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    v = mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta
    s = x.sum() - target
    return v + 0.5 * mu * s * s

def _smooth_grad(x, beta, target, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        lo1, hi1 = max(0, -d), min(n, n - d)
        g[lo1:hi1] += wk * (1.0 - x[lo1 + d:hi1 + d])
        lo2, hi2 = max(0, d), min(n, n + d)
        g[lo2:hi2] -= wk * x[lo2 - d:hi2 - d]
    g *= dx
    g += mu * (x.sum() - target)
    return g

def _golden(h, i, j, st, a, b):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i] = c; h[j] = st - c; fc = compute_c5(h)
    h[i] = d; h[j] = st - d; fd = compute_c5(h)
    for _ in range(50):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            h[i] = c; h[j] = st - c; fc = compute_c5(h)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            h[i] = d; h[j] = st - d; fd = compute_c5(h)
    t = (a + b) / 2.0
    h[i] = t; h[j] = st - t
    return t, compute_c5(h)

def _coord_desc(h, target, nsweeps=50, tlimit=None, t0=None):
    n = len(h)
    bc = compute_c5(h)
    for sw in range(nsweeps):
        if tlimit is not None and time.time() - t0 > tlimit:
            break
        imp = False
        for i in range(n):
            if tlimit is not None and time.time() - t0 > tlimit:
                break
            j = (i + 1) % n
            st = h[i] + h[j]
            lo, hi = max(0.0, st - 1.0), min(1.0, st)
            if hi - lo < 1e-10:
                continue
            best_t = h[i]
            bv = bc
            ts = np.linspace(lo, hi, 80)
            for t in ts:
                h[i] = t; h[j] = st - t
                c = compute_c5(h)
                if c < bv - 1e-14:
                    bv = c; best_t = t
            w = (hi - lo) / 80.0
            a, b = max(lo, best_t - 2 * w), min(hi, best_t + 2 * w)
            if b - a > 1e-8:
                gt, gv = _golden(h, i, j, st, a, b)
                if gv < bv - 1e-14:
                    bv = gv; best_t = gt
            h[i] = best_t; h[j] = st - best_t
            if bv < bc - 1e-14:
                bc = bv; imp = True
        if not imp:
            break
    return h, bc

def _subgrad_step(h, target, step, tol_frac=0.015):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mv = np.max(corr)
    tol = max(0.004, tol_frac * mv)
    active = np.where(corr >= mv - tol)[0]
    g = np.zeros(n)
    for k in active:
        d = k - (n - 1)
        lo1, hi1 = max(0, -d), min(n, n - d)
        g[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
        lo2, hi2 = max(0, d), min(n, n + d)
        g[lo2:hi2] -= h[lo2 - d:hi2 - d]
    g /= len(active)
    g *= dx
    g -= g.mean()
    return project(h - step * g, target), mv, len(active)
