import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m == 0:
            g += wk * (1.0 - 2.0 * h)
        elif m > 0:
            g[:n - m] += wk * (1.0 - h[m:])
            g[m:] += wk * (-h[:n - m])
        else:
            p = -m
            g[p:] += wk * (1.0 - h[:n - p])
            g[:n - p] += wk * (-h[p:])
    return g * dx

def _proj(h, n):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(3000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-12:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            np.clip(h + d / n, 0.0, 1.0, out=h)
            break
        h[free] += d / nf
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_matrix(h):
    n = len(h)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m = k - (n - 1)
        if m == 0:
            G[k] = 1.0 - 2.0 * h
        elif m > 0:
            G[k, :n - m] = 1.0 - h[m:]
            G[k, m:] += -h[:n - m]
        else:
            p = -m
            G[k, p:] = 1.0 - h[:n - p]
            G[k, :n - p] += -h[p:]
    return G

def _lp_step(h, n, t0, budget_s):
    c = _corr(h)
    G = _grad_matrix(h)
    rhs = G @ h - c
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n * 0.5]
    for k in range(2 * n - 1):
        cons.append(G[k] @ h_var - t_var <= rhs[k])
    prob = cp.Problem(cp.Minimize(t_var), cons)
    h_new = None
    for solver in [cp.GLOP, cp.GLPK]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    h_new = h_var.value
    if h_new is None:
        return None
    h_new = _proj(h_new, n)
    c_old_max = np.max(c)
    c_new_max = np.max(_corr(h_new))
    if c_new_max < c_old_max - 1e-10:
        return h_new
    for alpha in [0.5, 0.25, 0.1]:
        h_d = _proj((1 - alpha) * h + alpha * h_new, n)
        if np.max(_corr(h_d)) < c_old_max - 1e-10:
            return h_d
    return None

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _upsample(h_old, n_new):
    x_old = np.arange(len(h_old)) * (2.0 / len(h_old)) + 1.0 / len(h_old)
    x_new = np.arange(n_new) * (2.0 / n_new) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return _proj(h_new, n_new)
