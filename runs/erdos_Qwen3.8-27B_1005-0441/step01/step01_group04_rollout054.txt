def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            pool = room > 1e-12
            if not pool.any():
                break
            total_room = room[pool].sum()
            if total_room < 1e-14:
                break
            h[pool] = np.minimum(1.0, h[pool] + diff * room[pool] / total_room)
        else:
            pool = h > 1e-12
            if not pool.any():
                break
            total = h[pool].sum()
            if total < 1e-14:
                break

                    h_opt2 = _fix_sum(h_opt2, n)
                    c52 = _eval_c5(h_opt2)
                    if c52 < best_c5:
                        best_c5 = c52
                        best_h = h_opt2.copy()
                    else:
                        best_c5 = c5
                        best_h = h_opt.copy()
                else:
                    best_c5 = c5
                    best_h = h_opt.copy()

    # Perturbation rounds on best
    if best_h is not None and remaining(120):
        for scale in [0.06, 0.04, 0.02, 0.01]:
            if not remaining(50):
                break
            for _ in range(4):
                if not remaining(35):
                    break
                h_pert = _fix_sum(best_h + rng.normal(0, scale, len(best_h)), len(best_h))
                h_opt = _opt_lse(h_pert, 300, 400)
                h_opt = _fix_sum(h_opt, len(h_opt))
                h_opt = _opt_mm(h_opt, 300)
                h_opt = _fix_sum(h_opt, len(h_opt))
                c5 = _eval_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

    # Final polish: coord descent + SLSQP
    if best_h is not None and remaining(40):
        h_cd = _coord_descent(best_h, 40)
        c5 = _eval_c5(h_cd)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd.copy()
        if remaining(20):
            h_opt = _opt_mm(best_h, 500)
            h_opt = _fix_sum(h_opt, len(h_opt))
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    if best_h is None:
        best_h = _fix_sum(np.full(50, 0.5), 50)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
