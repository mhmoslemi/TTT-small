import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def circ_corr(h):
    n = len(h)
    h1 = 1.0 - h
    return np.real(np.fft.ifft(np.fft.fft(h) * np.conj(np.fft.fft(h1))))

def circ_corr_grad(h, m):
    n = len(h)
    j = np.arange(n)
    return (1.0 - h[(j - m + n) % n]) - h[(j + m) % n]

def project_sum(h, target):
    # Project h to sum = target while maintaining [0,1]
    # Simple approach: add uniform offset, then clip, iterate
    for _ in range(50):
        offset = (target - np.sum(h)) / len(h)
        h_new = h + offset
        if h_new.min() >= 0 and h_new.max() <= 1:
            return h_new
        h = np.clip(h_new, 0.0, 1.0)
    return h

def subgradient_descent(h, n_steps, t_max):
    n = len(h)
    target = float(n) / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()

        if time.time() - t0 > t_max:
            break
        c = circ_corr(h)
        m_star = int(np.argmax(c))
        g = circ_corr_grad(h, m_star)
        
        # Adaptive step size
        alpha = 0.01 / (1.0 + step * 0.01)
        
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def circ_corr(h):
    n = len(h)
    h1 = 1.0 - h
    return np.real(np.fft.ifft(np.fft.fft(h) * np.conj(np.fft.fft(h1))))

def circ_corr_grad(h, m):
    n = len(h)
    j = np.arange(n)
    return (1.0 - h[(j - m + n) % n]) - h[(j + m) % n]

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        # Add/subtract from non-bound elements
        offset = diff / len(h)
        h_new = h + offset
        if h_new.min() >= -1e-12 and h_new.max() <= 1.0 + 1e-12:
            h = np.clip(h_new, 0.0, 1.0)
            break
        h = np.clip(h_new, 0.0, 1.0)
    return h

def _global_n():
    pass  # placeholder

_n = 160
_temp = 0.01

def _ls_obj(h):
    dx = 2.0 / _n
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = max(_temp, 1e-10)
    m = float(np.max(corr))
    return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx

def _ls_eq(h):
    return np.sum(h) - _n / 2.0

def _ls_eq_jac(h):
    return np.ones(_n)

def slsqp_optimize(h_init, n, max_time):
    global _n, _temp
    t0 = time.time()
    _n = n
    target = float(n) / 2.0
    best = h_init.copy()
    best_val = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _ls_eq, 'jac': _ls_eq_jac},)
    
    temps = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004]
    for temp in temps:
        if time.time() - t0 > max_time - 5:
            break
        _temp = temp
        h = best.copy()
        for it in range(3):
            if time.time() - t0 > max_time - 2:
                break
            try:
                res = minimize(_ls_obj, h, method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
                v = compute_c5(h_new)
                if v < best_val - 1e-12:
                    best = h_new.copy()
                    best_val = v
                h = h_new.copy()
            except Exception:
                break
    return best, best_val

def subgrad_optimize(h_init, n, max_time):
    t0 = time.time()
    target = float(n) / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_steps = 5000
    
    for step in range(n_steps):
        if time.time() - t0 > max_time:
            break
        c = circ_corr(h)
        m_star = int(np.argmax(c))
        g = circ_corr_grad(h, m_star)
        
        alpha = 0.05 / (1.0 + step * 0.005)
        h_new = h - alpha * g
        h_new = project_feasible(h_new, target)
        h = h_new
        
        if step % 50 == 0 or step == n_steps - 1:
            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
    
    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    h_new = project_feasible(h_new, float(n_new) / 2.0)
    return h_new

def binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.permutation(n)[:n//2]
    h[idx] = 1.0

def perturb(h, scale, seed):

    n = len(h)
    h_p = h + scale * rng.randn(n)
    return project_feasible(h_p, float(n) / 2.0)

_comp_n = 160
_comp_temp = 0.01

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def circ_corr(h):
    n = len(h)
    h1 = 1.0 - h
    return np.real(np.fft.ifft(np.fft.fft(h) * np.conj(np.fft.fft(h1))))

def circ_corr_grad(h, m):
    n = len(h)
    j = np.arange(n)
    return (1.0 - h[(j - m + n) % n]) - h[(j + m) % n]

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        offset = diff / len(h)
        h_new = h + offset
        if h_new.min() >= -1e-14 and h_new.max() <= 1.0 + 1e-14:
            h = np.clip(h_new, 0.0, 1.0)
            break
        h = np.clip(h_new, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode="full")
    t = max(_comp_temp, 1e-10)
    m = float(np.max(corr))
    return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx
