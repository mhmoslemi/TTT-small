    return _fix_sum(h, n)

def _smooth_h(n, freq, phase, rng):
    x = np.linspace(0, 2 * np.pi * freq, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(x + phase) + 0.15 * np.sin(2 * x + phase * 2)
    return _fix_sum(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    candidates = []
    if h_init is not None:
        h_init = _fix_sum(h_init, len(h_init))
        candidates.append(h_init)
        best_c5 = _eval_c5(h_init)
        best_h = h_init.copy()

    for n in [47, 50, 60, 70, 80, 90, 100]:
        candidates.append(_thue_morse(n))
        candidates.append(_golden_h(n, rng))
        candidates.append(_smooth_h(n, 1, rng.random() * 6, rng))
        candidates.append(_smooth_h(n, 2, rng.random() * 6, rng))

    for h0 in candidates:
        if time.time() - t0 > budget_s - 80:
            break
        n = len(h0)
        c0 = _eval_c5(h0)
        if c0 < best_c5:
            best_c5 = c0
            best_h = h0.copy()

        for alpha, lam in [(50, 50), (200, 200), (500, 500)]:
            if time.time() - t0 > budget_s - 30:
                break
            h_opt = _lbfgsb_opt(h0, alpha, lam, max_iter=150)
            h_opt = _fix_sum(h_opt, n)
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

        if time.time() - t0 > budget_s - 30:
            continue
        h_pg, c5_pg = _proj_grad_step(h0, 300, t0, budget_s, n_steps=60)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Polish the best with more aggressive steps
    if best_h is not None and time.time() - t0 < budget_s - 40:
        n = len(best_h)
        for alpha in [100, 300, 1000]:
            if time.time() - t0 > budget_s - 20:
                break
            h_pg, c5_pg = _proj_grad_step(best_h, alpha, t0, budget_s, n_steps=50)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

        # L-BFGS-B final
        if time.time() - t0 < budget_s - 15:
            h_fin = _lbfgsb_opt(best_h, 300, 500, max_iter=300)
            h_fin = _fix_sum(h_fin, n)
            c5_f = _eval_c5(h_fin)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_fin.copy()

        # Binary refinement
        if time.time() - t0 < budget_s - 10:
            h_bin = (best_h > 0.5).astype(float)
            h_bin = _fix_sum(h_bin, n)
            h_bs, c5_bs = _binary_swap_opt(h_bin, t0, budget_s, n_swaps=200)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs.copy()
            else:
                # Try soft version around the binary
                h_soft = best_h + rng.normal(0, 0.02, n)
                h_soft = _fix_sum(h_soft, n)
                h_so, c5_so = _proj_grad_step(h_soft, 500, t0, budget_s, n_steps=30)
                if c5_so < best_c5:
                    best_c5 = c5_so
                    best_h = h_so.copy()

    if best_h is None:
        best_h = np.full(50, 0.5)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
