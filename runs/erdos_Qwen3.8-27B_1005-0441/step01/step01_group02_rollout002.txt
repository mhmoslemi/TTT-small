import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(800):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    return h

def _smooth_max_obj(h, beta):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx)

def _grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx

def _hardest_lag_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ki = int(np.argmax(corr))
    d = (n - 1) - ki
    grad = np.zeros(n)
    lo1 = max(0, -d)
    hi1 = min(n, n - d)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
    lo2 = max(0, d)
    hi2 = min(n, d + n)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 - d:hi2 - d]
    return grad * dx

def _sum_con(x):
    return x.sum() - x.size / 2.0

def _sum_jac(x):
    return np.ones(x.size)

def _slsqp(h, beta, maxiter=100):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_max_obj, h, method='SLSQP',
                      bounds=bounds,
                      constraints={'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac},
                      args=(beta,),
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return _project(res.x)
    except Exception:
        return _project(h.copy())

def _nm_polish(h, maxiter=300):
    try:
        res = minimize(compute_c5, h, method='Nelder-Mead',
                      options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-13})
        return _project(res.x)
    except Exception:
        return _project(h.copy())

def _local_opt(h0, t_budget, beta=30.0):
    h = _project(h0.copy().astype(float))
    best = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    if time.time() - t0 < t_budget * 0.6:
        h = _slsqp(h, beta, maxiter=200)
        c = compute_c5(h)
        if c < best:
            best = c
            hb = h.copy()
    if time.time() - t0 < t_budget * 0.85:
        hn = _nm_polish(hb, maxiter=min(400, int((t_budget - (time.time() - t0)) * 3)))
        cn = compute_c5(hn)
        if cn < best:
            best = cn
            hb = hn
    return hb, best

def _hardest_lag_step(h, lr):
    n = len(h)
    grad = _hardest_lag_grad(h)
    h_new = h - lr * grad
    h_new -= (h_new.sum() - n / 2.0) / n
    return _project(h_new)

def _evolve(pop, pop_size, t_budget, rng):
    t0 = time.time()
    n = len(pop[0])
    scores = [compute_c5(p) for p in pop]
    best_h = pop[int(np.argmin(scores))].copy()
    best_c = float(min(scores))
    gen = 0
    patience = 0
    max_patience = 12
    while time.time() - t0 < t_budget and patience < max_patience:
        gen += 1
        improved_this_gen = False
        order = np.argsort(scores)
        for i in range(pop_size):
            if time.time() - t0 > t_budget - 3:
                break
            c1 = int(order[int(rng.randint(0, min(5, pop_size)))])
            c2 = int(order[int(rng.randint(0, min(5, pop_size)))])
            parent = pop[c1] if scores[c1] <= scores[c2] else pop[c2]
            other = int(order[int(rng.randint(3, pop_size))])
            mask = rng.rand(n) > 0.5
            child = np.where(mask, parent, pop[other]).astype(float)
            mut = 0.12 / (1.0 + 0.15 * gen)
            child += mut * rng.randn(n)
            for _ in range(4):
                lr = 0.04 / (1.0 + 0.3 * gen)
                child = _hardest_lag_step(child, lr)
            child = _project(child)
            child = _slsqp(child, 20.0, maxiter=25)
            c = compute_c5(child)
            if c < best_c - 1e-10:
                best_c = c
                best_h = child.copy()
                improved_this_gen = True
            if i < pop_size:
                pop[i] = child
                scores[i] = c
        pop[int(np.argmin(scores))] = best_h.copy()
        scores[int(np.argmin(scores))] = best_c
        order = np.argsort(scores)
        if improved_this_gen:
            patience = 0
        else:
            patience += 1
            for i in range(pop_size // 2):
                pop[order[i]] = best_h + 0.08 * rng.randn(n)
                pop[order[i]] = _project(pop[order[i]])
                scores[order[i]] = compute_c5(pop[order[i]])
    return best_h, best_c

def _pair_swap(h, t_budget, rng):
    n = len(h)
    best = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    stagnation = 0
    while time.time() - t0 < t_budget and stagnation < 6:
        found = False
        for _ in range(200):
            if time.time() - t0 > t_budget - 2:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            for frac in [0.15, 0.25, 0.4, 0.6]:
                amt = frac * (hb[i] - hb[j])
                if abs(amt) < 1e-10:
                    continue
                ht = hb.copy()
                ht[i] -= amt
                ht[j] += amt
                if ht[i] < -0.001 or ht[i] > 1.001 or ht[j] < -0.001 or ht[j] > 1.001:
                    continue
                ht = _project(ht)
                ct = compute_c5(ht)
                if ct < best - 1e-11:
                    hb = ht
                    best = ct
                    found = True
                    break
            if found:
                break
        if found:
            stagnation = 0
        else:
            stagnation += 1
            if time.time() - t0 > t_budget * 0.4:
                hb = _project(hb + 0.04 * rng.randn(n))
    return hb, best
