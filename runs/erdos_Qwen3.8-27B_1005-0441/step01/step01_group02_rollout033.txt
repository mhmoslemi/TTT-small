def compute_c5(h):

            best = C
            best_h = h.copy()
            stall_count = 0
        else:
            if C - last_best < 1e-12:
                stall_count += 1
            last_best = C
        g = subgrad_topm(h, m)
        v = gamma * v - alpha * g
        improved = False
        a = alpha
        for _ in range(12):
            h_try = project(h + a * v, target)
            C_try = compute_c5(h_try)
            if C_try < C - 1e-15:
                h = h_try
                alpha = min(a * 1.03, 0.3)
                improved = True
                break
            a *= 0.4
        if not improved:
            alpha *= 0.7
            v *= 0.5
        it += 1
        if it % kick_every == 0 and stall_count > 60:
            n_kick = max(3, n // 15)
            idx = np.random.choice(n, size=n_kick, replace=False)
            h[idx] += 0.04 * np.random.randn(n_kick)
            h = project(h, target)
            v = np.zeros(n)
            alpha = alpha0
            stall_count = 0
        if alpha < 1e-10:
            break
    return best_h, best

def nm_polish(h0, maxiter=400):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return compute_c5(project(x, target))
    try:
        res = minimize(obj, h0.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-13})
        h_new = project(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0.copy(), compute_c5(h0)

def coord_descent(h0, t_budget):
    n = len(h0)
    target = n / 2.0
    h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    deltas = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.015, -0.015]
    while time.time() - t0 < t_budget:
        improved = False
        for j in range(n):
            for delta in deltas:
                h_try = h.copy()
                h_try[j] += delta
                if h_try[j] < 0 or h_try[j] > 1:
                    continue
                for k2 in range(n):
                    if k2 == j:
                        continue
                    h_try[k2] -= delta
                    if h_try[k2] < 0 or h_try[k2] > 1:
                        h_try[k2] = h[k2]
                        continue
                    c_try = compute_c5(h_try)
                    if c_try < best - 1e-13:
                        h = h_try.copy()
                        best = c_try
                        improved = True
                        break
                    h_try[k2] = h[k2]
            if time.time() - t0 > t_budget:
                break
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    if h_init is None:
        n = 70
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0
    n = len(h_init)
    target = n / 2.0
    h = project(h_init.copy(), target)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: Quick SLSQP
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 40:
        h1 = slsqp_step(h, beta=25.0, maxiter=120)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1

    # Phase 2: Projected gradient with momentum (main workhorse)
    t_rem = budget_s - (time.time() - t_start)
    pg_bud = min(400, t_rem - 250)
    if pg_bud > 50:
        hp, cp = pg_optimize(best_h.copy(), pg_bud, m=3, alpha0=0.04, gamma=0.85)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp

    # Phase 3: SLSQP with higher beta
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 60:
        h2 = slsqp_step(best_h.copy(), beta=50.0, maxiter=100)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 50:
        h3 = slsqp_step(best_h.copy(), beta=80.0, maxiter=80)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3

    # Phase 4: More PG with different m
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 60:
        hp4, cp4 = pg_optimize(best_h.copy(), min(200, t_rem - 150), m=5, alpha0=0.02, gamma=0.8)
        if cp4 < best_c5:
            best_c5 = cp4
            best_h = hp4

    # Phase 5: NM polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 50:
        hn, cn = nm_polish(best_h, maxiter=400)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn

    # Phase 6: Perturbation restarts
    t_rem = budget_s - (time.time() - t_start)
    n_restarts = min(int(t_rem / 50), 4)
    for i in range(n_restarts):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 50:
            break
        if i == 0:
            h_p = project(best_h + 0.06 * rng.randn(n), target)
        elif i == 1:
            h_p = project(best_h + 0.12 * rng.randn(n), target)
        elif i == 2:
            h_p = project(np.clip(best_h + 0.05 * rng.randn(n), 0, 1), target)
        else:
            flip = rng.choice(n, size=n // 10, replace=False)
            h_p = best_h.copy()
            h_p[flip] = 1.0 - h_p[flip]
            h_p = project(h_p, target)
        h_r = slsqp_step(h_p, beta=30.0, maxiter=80)
        cr = compute_c5(h_r)
        t_rem = budget_s - (time.time() - t_start)
        if t_rem > 25:
            hr, rr = pg_optimize(h_r, min(25, t_rem - 15), m=3, alpha0=0.03)
            if rr < cr:
                cr = rr
                h_r = hr
        if cr < best_c5:
            best_c5 = cr
            best_h = h_r

    # Phase 7: Coordinate descent
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 15:
        hc, cc = coord_descent(best_h, min(25, t_rem - 5))
        if cc < best_c5:
            best_c5 = cc
            best_h = hc

    # Phase 8: Final NM
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 15:
        hn, cn = nm_polish(best_h, maxiter=300)
        if cn < best_c5:
            best_c5 = cn
            best_h = hn

    best_h = project(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, n)
