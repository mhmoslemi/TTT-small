def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx), corr

def project_to_binary(h):
    """Project fractional h to binary with exactly n/2 ones."""
    n = len(h)
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    # Adjust to get exactly target ones
    diff = target - int(h_bin.sum())
    if diff > 0:
        # Need more ones: add the largest fractional values
        idx = np.argsort(h)[::-1]
        for i in range(diff):
            h_bin[idx[i]] = 1.0
    elif diff < 0:
        # Need fewer ones: remove the smallest ones
        idx = np.argsort(h)
        for i in range(-diff):
            h_bin[idx[i]] = 0.0
    return h_bin

def swap_delta(corr, h, i, j, n):
    """Compute change in corr when swapping h[i]:1->0 and h[j]:0->1."""
    # lag = n-1-m, so m = n-1-lag
    # For each lag, delta = -(1-h[i+lag]) + h[i-lag] + (1-h[j+lag]) - h[j-lag]
    # (terms with out-of-bounds indices are 0)
    lags = np.arange(-(n-1), n)  # -n+1 to n-1
    delta = np.zeros(2*n-1)
    
    # Term: -(1-h[i+lag]) where i+lag is in [0, n-1]
    valid = (i + lags >= 0) & (i + lags < n)
    delta[valid] -= (1 - h[i + lags[valid]])
    
    # Term: +h[i-lag] where i-lag is in [0, n-1]
    valid = (i - lags >= 0) & (i - lags < n)
    delta[valid] += h[i - lags[valid]]
    
    # Term: +(1-h[j+lag]) where j+lag is in [0, n-1]
    valid = (j + lags >= 0) & (j + lags < n)
    delta[valid] += (1 - h[j + lags[valid]])
    
    # Term: -h[j-lag] where j-lag is in [0, n-1]
    valid = (j - lags >= 0) & (j - lags < n)
    delta[valid] -= h[j - lags[valid]]
    
    return corr + delta

    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5, corr = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    iterations = 0
    while time.time() < deadline - 1 and iterations < max_iter:
        iterations += 1
        best_improvement = 0
        best_i, best_j = -1, -1
        best_new_c5 = best_c5
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                new_corr = swap_delta(corr, h, int(i), int(j), n)
                new_max = new_corr.max()
                new_c5_val = new_max * dx
                if new_c5_val < best_new_c5 - 1e-15:
                    best_new_c5 = new_c5_val
                    best_improvement = best_c5 - best_new_c5
                    best_i, best_j = i, j
        
        if best_i == -1:
            break  # No improving move
        
        # Apply best swap
        h[best_i] = 0.0
        h[best_j] = 1.0
        best_c5 = best_new_c5
        corr, _ = compute_c5(h)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    
    return h, best_c5

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h):
    n = len(h)
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    diff = target - int(h_b.sum())
    if diff > 0:
        idx = np.argsort(-h)[:diff]
        h_b[idx] = 1.0
    elif diff < 0:
        idx = np.argsort(h)[:(-diff)]
        h_b[idx] = 0.0
    return h_b

def init_binary(n, seed):
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    alpha = 1.0 / phi
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * alpha) % 1.0 < 0.5 else 0.0
    target = n // 2
    diff = target - int(h.sum())
    if diff > 0:
        idx = np.where(h == 0.0)[0]
        h[idx[:diff]] = 1.0
    elif diff < 0:
        idx = np.where(h == 1.0)[0]
        h[idx[:(-diff)]] = 0.0
    return h

def steepest_binary(h, t_end, restart_every=5):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), c5
    no_improve = 0
    while time.time() < t_end - 0.5 and no_improve < restart_every * 8:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 1e-14
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                d = c5 - new_c5
                if d > best_delta:
                    best_delta = d
                    best_pair = (int(i), int(j))
        if best_pair is None:
            no_improve += 1
            if no_improve >= restart_every:
                h = best_h + np.zeros(n)
                rng = np.random.RandomState(int(time.time() * 1000) % 100000)
                for _ in range(rng.randint(3, 8)):
                    ii = np.where(h == 1.0)[0][rng.randint(n // 2)]
                    jj = np.where(h == 0.0)[0][rng.randint(n // 2)]
                    h[ii] = 0.0
                    h[jj] = 1.0
                c5 = compute_c5(h)
                no_improve = 0
        else:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            c5 -= best_delta
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            no_improve = 0
    return best_h, best_c5

def sa_binary(h, t_end, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    c5 = best_c5
    T = 0.004
    T_floor = 0.00003
    cool = 0.99975
    while time.time() < t_end - 0.5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_trials = rng.randint(40, 100)
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < c5:
                c5 = new_c5
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            elif rng.rand() < math.exp(-max(0, new_c5 - c5) / T):
                c5 = new_c5
            else:
                h[i] = 1.0
                h[j] = 0.0
                continue
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        T = max(T * cool, T_floor)
    return best_h, best_c5

def continuous_optimize(h_init, t_end, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h, best_c5 = h.copy(), compute_c5(h)
    v = np.zeros(n)
    lr = 0.06
    beta = 0.8
    tau = 30.0
    while time.time() < t_end - 1.0:
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full')
        mc = corr.max()
        shifted = np.clip(tau * (corr - mc), -500, 500)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-13:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
            else:
                L = -lag
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
        grad *= dx
        v = beta * v - lr * grad
        h = project_feasible(h + v, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        tau = min(tau * 1.004, 500.0)
        if int(tau / 100) % 3 == 0:
            lr = max(lr * 0.93, 0.003)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    best_h = None
    best_c5 = 1.0
    best_n = 70

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
        n0 = len(h_init)
    else:
        n0 = 70
        h_init = init_binary(n0, seed)
    h_bin0 = to_binary(h_init)

    # Phase 1: Steepest binary on initial (n0)
    t1 = deadline - 90
    h1, c5_1 = steepest_binary(h_bin0, min(t1, t0 + 60), restart_every=4)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n0

    # Phase 2: SA from best binary
    t_rem = deadline - time.time()
    t2_end = t0 + min(t_rem * 0.20, 60)
    h2, c5_2 = sa_binary(best_h, t2_end, seed + 10)
    if c5_2 < best_c5:
        best_h, best_c5, best_n = h2, c5_2, n0

    # Phase 3: Continuous relax + re-binarize + steepest
    t_rem = deadline - time.time()
    t3_end = t0 + min(t_rem * 0.15, 50)
    h3, c5_3 = continuous_optimize(best_h, t3_end, seed + 20)
    if c5_3 < best_c5:
        best_h, best_c5, best_n = h3, c5_3, n0
    h3b = to_binary(h3)
    c5_3b = compute_c5(h3b)
    if c5_3b < best_c5:
        best_h, best_c5, best_n = h3b, c5_3b, n0
    t_rem = deadline - time.time()
    if t_rem > 15:
        t32_end = t0 + min(t_rem * 0.3, 30)
        h32, c5_32 = steepest_binary(h3b, t32_end, restart_every=3)
        if c5_32 < best_c5:
            best_h, best_c5, best_n = h32, c5_32, n0

    # Phase 4: Try other n values with binary optimization
    n_list = [64, 80, 96, 128, 160]
    if n0 not in n_list:
        n_list.append(n0 + 1)
    for n_try in n_list:
        t_rem = deadline - time.time()
        if t_rem < 35:
            break
        t_alloc = min(t_rem * 0.22, 90)
        h_try = init_binary(n_try, seed + n_try * 7)
        h_sb, c5_sb = steepest_binary(h_try, t0 + min(200, t0 + time.time() + t_alloc - 10), restart_every=4)
        if c5_sb < best_c5:
            best_h, best_c5, best_n = h_sb, c5_sb, n_try
        t_rem = deadline - time.time()
        if t_rem > 20:
            t_sa_end = t0 + min(t_rem * 0.25, 35)
            h_sa, c5_sa = sa_binary(h_sb, t_sa_end, seed + n_try + 500)
            if c5_sa < best_c5:
                best_h, best_c5, best_n = h_sa, c5_sa, n_try

    # Phase 5: Second continuous + binary round on best
    t_rem = deadline - time.time()
    if t_rem > 20:
        n_b = len(best_h)
        is_bin = (best_h * (1.0 - best_h)).sum() < 1e-6
        if is_bin:
            t5_end = t0 + min(t_rem * 0.3, 40)
            h5, c5_5 = steepest_binary(best_h, t5_end, restart_every=3)
            if c5_5 < best_c5:
                best_h, best_c5, best_n = h5, c5_5, n_b
            t_rem = deadline - time.time()
            if t_rem > 15:
                t52_end = t0 + min(t_rem * 0.3, 25)
                h52, c5_52 = sa_binary(best_h, t52_end, seed + 5000)
                if c5_52 < best_c5:
                    best_h, best_c5, best_n = h52, c5_52, n_b
        else:
            t5_end = t0 + min(t_rem * 0.3, 40)
            h5, c5_5 = continuous_optimize(best_h, t5_end, seed + 7000)
            if c5_5 < best_c5:
                best_h, best_c5, best_n = h5, c5_5, n_b

    # Phase 6: Final polish
    t_rem = deadline - time.time()
    if t_rem > 8:
        n_f = len(best_h)
        if (best_h * (1.0 - best_h)).sum() < 1e-6:
            h_f, c5_f = sa_binary(best_h, deadline - 1, seed + 9999)
            if c5_f < best_c5:
                best_h, best_c5, best_n = h_f, c5_f, n_f
        else:
            h_f, c5_f = continuous_optimize(best_h, deadline - 1, seed + 8888)
            if c5_f < best_c5:
                best_h, best_c5, best_n = h_f, c5_f, n_f

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
