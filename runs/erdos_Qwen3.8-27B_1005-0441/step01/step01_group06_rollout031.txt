import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def soft_obj_func(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    s = (c - m) * beta
    return (m + np.log1p(np.sum(np.exp(s) - 1.0)) / beta) * dx

def slsqp_pass(h0, n, maxiter, ftol, beta):
    nx = n
    target = n / 2.0
    def obj(x):
        return soft_obj_func(x, nx, beta)
    def cf(x):
        return np.sum(x) - target
    def cj(x):
        return np.ones(nx)
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * nx,
                   constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, nx)
    return h0

def to_binary(h, n):
    hb = (h > 0.5).astype(float)
    target = n / 2.0
    diff = int(round(target - np.sum(hb)))
    if diff > 0:
        order = np.argsort(-h)[:diff]
        hb[order] = 1.0
    elif diff < 0:
        order = np.argsort(h)[:abs(diff)]
        hb[order] = 0.0
    return project_feasible(hb, n)

def binary_search(h0, n, rng, t_deadline):
    h = to_binary(h0, n)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    stall = 0
    temp = 0.05
    for it in range(50000):
        if time.time() > t_deadline:
            break
        diff = np.sum(h) - n / 2.0
        if abs(diff) > 1e-10:
            h = project_feasible(h, n)
        
        # Pick a random one and a random zero, try swapping
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = 0.0, 1.0
        new_c5 = compute_c5(h)
        if new_c5 < c5 - 1e-14:
            c5 = new_c5
            best_h = h.copy()
            best_c5 = c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            stall = 0
        else:
            # Accept with probability exp(-(new_c5-c5)/temp) (Simulated Annealing)
            if rng.rand() < np.exp(-max(0, new_c5 - c5) / max(temp, 1e-10)):
                c5 = new_c5
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[a], h[b] = 1.0, 0.0
            stall += 1
        if stall > 2000:
            temp *= 0.7
            stall = 0
        if temp < 1e-6:
            break
    
    return best_h, best_c5
