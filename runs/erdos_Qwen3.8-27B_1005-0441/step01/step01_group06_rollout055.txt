from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h[mask] = np.minimum(h[mask], 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h[mask] = np.maximum(h[mask], 0.0)
    return h

def smooth_max_obj(h, n, beta, active_idx):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    if active_idx is not None:
        c = c[active_idx]
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    return m + np.log(np.sum(np.exp(shifted))) / beta

def constraint_fun(x):
    return np.sum(x) - len(x) / 2.0

def constraint_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, n, beta, active_idx, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        res = minimize(smooth_max_obj, h0, args=(n, beta, active_idx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def make_thue_morse(n):
    n = n if n % 2 == 0 else n - 1
    idx = np.arange(n, dtype=np.int64)
    tm = np.zeros(n, dtype=np.float64)
    max_bit = max(1, int(np.ceil(np.log2(max(n, 2)))))
    for bit in range(max_bit):
        tm += ((idx >> bit) & 1).astype(np.float64)
    tm = (tm % 2.0)
    half = n // 2
    ones = int(np.sum(tm))
    if ones > half:
        one_pos = np.where(tm == 1.0)[0]
        to_flip = one_pos[-(ones - half):]
        tm[to_flip] = 0.0
    elif ones < half:
        zero_pos = np.where(tm == 0.0)[0]
        to_flip = zero_pos[:half - ones]
        tm[to_flip] = 1.0
    return tm

def make_balanced(n, pattern, rng):
    if n % 2 != 0:
        n -= 1
    half = n // 2
    if pattern == 'tm':
        return make_thue_morse(n)
    elif pattern == 'alt':
        h = np.zeros(n)
        h[::4] = 1.0
        h[1::4] = 1.0
        return h
    elif pattern == 'block':
        b = max(1, n // 16)
        h = np.zeros(n)
        for i in range(0, n, 2 * b):
            h[i:i + b] = 1.0
        return project_feasible(h, n)
    elif pattern == 'quasi':
        h = np.zeros(n)
        for i in range(n):
            if (2 * i) % (n // 2 + 1) < (n // 4) + (n // 8):
                h[i] = 1.0
        return project_feasible(h, n)
    else:
        h = np.zeros(n)
        pos = rng.choice(n, size=half, replace=False)
        h[pos] = 1.0
        return h
