    return float(np.max(corr) * dx)

def _project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] = np.clip(proj[mask] + diff/cnt, 0.0, 1.0)
    return proj

def _mw_optimize(h0, iters, alpha0, temp_init, temp_rate, seed=0):

    h = h0.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    for t in range(iters):

        M = np.max(c_raw)
        c5 = M * 2.0 / n

            best_c5 = c5
            best_h = h.copy()
        temp = max(temp_init * (temp_rate ** t), 0.005)
        w = np.exp((c_raw - M) / temp)
        sw = w.sum()
        w /= sw
        g = np.zeros(n)
        for ki in range(2 * n - 1):
            wi = w[ki]
            if wi < 1e-13:
                continue
            lag = ki - (n - 1)
            if lag == 0:
                g -= wi * 2.0 * h
            elif lag > 0:
                g[:n-lag] -= wi * h[lag:]
                g[lag:] -= wi * h[:-lag]
            else:
                la = -lag
                g[:n-la] -= wi * h[la:]
                g[la:] -= wi * h[:-la]
        alpha = alpha0 / (1.0 + 0.005 * t)

        M = c.max()
        a = 300.0
        return float(M + np.log(np.sum(np.exp(a * (c - M)))) / a)

        return np.sum(x) - n / 2.0

                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    out = np.clip(res.x, 0.0, 1.0)
    out = _project(out, n)
    return out

def _project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] = np.clip(proj[mask] + diff / cnt, 0.0, 1.0)
    return proj

def _compute_weighted_grad(h, w):
    n = len(h)
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-13:
            continue
        lag = ki - (n - 1)
        if lag == 0:
            g -= wi * 2.0 * h
        else:
            al = abs(lag)
            g[:n - al] -= wi * h[al:]
            g[al:] -= wi * h[:n - al]
    return g

def _mw_optimize(h0, iters, alpha0, temp_init, temp_rate, use_momentum=True, momentum=0.85):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    v = np.zeros(n)
    for t in range(iters):
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        M = np.max(c_raw)
        c5 = M * 2.0 / n
        if c5 < best_c5 - 1e-14:

            best_h = h.copy()
        temp = max(temp_init * (temp_rate ** t), 0.003)

        sw = w.sum()
        if sw < 1e-30:
            w = np.zeros(2 * n - 1)
            w[n - 1] = 1.0
        else:
            w /= sw
        g = _compute_weighted_grad(h, w)
        gn = np.linalg.norm(g)
        if gn < 1e-15:

        g = g / gn
        if use_momentum:
            v = momentum * v + g
        alpha = alpha0 / (1.0 + 0.003 * t)
        h = _project(h - alpha * v, n)
    return best_h, best_c5

        M = c.max()
        return float(M + np.log(np.sum(np.exp(a_smooth * (c - M)))) / a_smooth)
    def con_eq(x):
        return np.sum(x) - n / 2.0

                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        out = np.clip(res.x, 0.0, 1.0)
        return _project(out, n)
    except Exception:
        return h0.copy()

def _minimax_polish(h0, max_iter=500):
    n = len(h0)
    dx = 2.0 / n
    def mm_obj(x):
        h = x[:-1]
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        a = 400.0
        M = c.max()
        return float(M + np.log(np.sum(np.exp(a * (c - M)))) / a)

        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full') * dx

        out = np.clip(res.x[:-1], 0.0, 1.0)
