    rng = np.random.RandomState(seed)
    n_ones = n // 2

            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            nc = compute_corr(h)
            ne = float(nc.max() * dx)
            if ne < cur_e or rng.random() < np.exp(-(ne - cur_e) / max(T, 1e-10)):
                cur_e = ne
                if ne < best_e:
                    best_e = ne
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = 1.0, 0.0
                no_improve += 1
    return best_h, best_e

def targeted_refine(h, budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_h = h.copy()
    best_e = compute_c5(h)
    t0 = time.time()
    end = t0 + budget
    while time.time() < end - 0.1:
        corr = compute_corr(h)
        max_val = corr.max()
        top_lags = np.argsort(corr)[::-1][:min(8, len(corr))]
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_try = min(200, len(ones) * len(zeros))
        for _ in range(n_try):
            tl = top_lags[rng.randint(len(top_lags))]
            lag = tl - (n - 1)
            cands_i, cands_j = [], []
            if lag >= 0:
                for pos in range(n - lag):
                    if h[pos] == 1.0 and h[pos + lag] == 0.0:
                        cands_i.append(pos)
                        cands_j.append(pos + lag)
            else:
                L = -lag
                for pos in range(L, n):
                    if h[pos] == 1.0 and h[pos - L] == 0.0:
                        cands_i.append(pos)
                        cands_j.append(pos - L)
            if not cands_i:
                cands_i = ones.tolist()
                cands_j = zeros.tolist()
            ci = cands_i[rng.randint(len(cands_i))]
            cj = cands_j[rng.randint(len(cands_j))]
            if h[ci] == 1.0 and h[cj] == 0.0:
                h[ci], h[cj] = 0.0, 1.0
                ne = compute_c5(h)
                if ne < best_e:
                    best_e = ne
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[ci], h[cj] = 1.0, 0.0
        if not improved:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            ne = compute_c5(h)
            if ne < best_e:
                best_e = ne
                best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
    return best_h, best_e

def grad_fractional(h, budget, seed):
    n = len(h)
    ts = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h.copy(), ts)
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros(n)
    beta = 0.7
    lr = 0.05
    tau = 30.0
    t0 = time.time()
    end = t0 + budget
    while time.time() < end - 0.1:
        corr = compute_corr(h)
        mc = corr.max()
        sh = np.clip(tau * (corr - mc), -300, 300)
        w = np.exp(sh)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-13:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                g[lag:] += w[k] * (1.0 - h[:n - lag])
                g[:n - lag] -= w[k] * h[lag:]
            else:
                L = -lag
                g[:n - L] += w[k] * (1.0 - h[L:])
                g[L:] -= w[k] * h[:n - L]
        g *= dx
        v = beta * v - lr * g
        h = project_feasible(h + v, ts)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        tau = min(tau * 1.004, 400.0)
        lr = max(lr * 0.9997, 0.002)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    best_h = None
    best_c = float('inf')
    best_n = 70
    h_given = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_given = np.asarray(initial_h_values, dtype=float)
    n_list = [70, 60, 80, 90, 100]
    if h_given is not None:
        n_given = len(h_given)
        n_list.sort(key=lambda x: abs(x - n_given))
    for n in n_list:
        t_rem = deadline - time.time()
        if t_rem < 20:
            break
        alloc = min(t_rem * 0.5, 250)
        for mode in ['gold', 'alt', 'random']:
            t_rem = deadline - time.time()
            if t_rem < 15:
                break
            if h_given is not None and len(h_given) == n and mode == 'gold':
                hb = (h_given >= 0.5).astype(float)
                while hb.sum() > n // 2:
                    hb[np.argmax(hb)] = 0.0
                while hb.sum() < n // 2:
                    hb[np.argmin(hb)] = 1.0
            else:
                hb = make_binary(n, seed + n * 31 + hash(mode) % 997, mode)
            sb = min(alloc / 3.5, 45)
            if sb < 5:
                continue
            hb2, cb2 = sa_binary(hb, sb, seed + n * 17 + (ord(mode[0]) % 100))
            cb2 = compute_c5(hb2)
            if cb2 < best_c:
                best_c = cb2
                best_h = hb2.copy()
                best_n = n
        t_rem = deadline - time.time()
        if t_rem > 20 and best_h is not None and len(best_h) == n:
            tr = min(t_rem * 0.4, 40)
            hr, cr = targeted_refine(best_h, tr, seed + n * 73)
            cr = compute_c5(hr)
            if cr < best_c:
                best_c = cr
                best_h = hr
                best_n = n
        t_rem = deadline - time.time()
        if t_rem > 15 and best_h is not None and len(best_h) == n:
            h_init = project_feasible(best_h + 0.08 * np.random.RandomState(seed + n * 91).randn(n), n / 2.0)
            hf, cf = grad_fractional(h_init, min(t_rem * 0.3, 35), seed + n * 97)
            cf = compute_c5(hf)
            if cf < best_c:
                best_c = cf
                best_h = hf
                best_n = n
    if best_h is None:
        best_h = make_binary(70, seed, 'gold')
        best_c = compute_c5(best_h)
        best_n = 70
    best_c = compute_c5(best_h)
    return best_h, best_c, best_n
