        if s > target:

        else:

def gradient_at_lags(h, lags):
    """Compute gradient of sum of corr_k over given lags (indexed in full corr array)"""
    n = len(h)
    grad = np.zeros(n)
    for k in lags:
        lag = k - (n - 1)
        # grad[j] += (1 - h[j+lag]) * [0<=j+lag<n] - h[j-lag] * [0<=j-lag<n]
        j = np.arange(n)
        valid_plus = (j + lag >= 0) & (j + lag < n)
        valid_minus = (j - lag >= 0) & (j - lag < n)
        grad[j[valid_plus]] += (1.0 - h[j[valid_plus] + lag])
        grad[j[valid_minus]] += (-h[j[valid_minus] - lag])
    return grad

            n_top = 4
        else:
            n_top = 3

        grad, _ = gradient_top_lags(h, n_top)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        momentum = mu * momentum - alpha * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new < c5_cur + 1e-15:
            h = h_new
            if c5_new < best_val:
                best = h.copy()
                best_val = c5_new
        else:
            momentum *= 0.5
            alpha *= 0.92

        if it % 400 == 399 and alpha > 0.005:
            h = project_feasible(h + 0.025 * rng.randn(n), target)
            momentum = np.zeros(n)
            alpha = max(alpha, 0.06)

        if it % 1000 == 999:
            alpha *= 0.75

    return best, best_val

def _perturb(h, scale, rng):
    h_p = h + scale * rng.randn(len(h))
    h_p = np.clip(h_p, 0.0, 1.0)
    s = np.sum(h_p)
    target = float(len(h)) / 2.0
    if s > 1e-12:
        h_p = h_p * target / s
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    target = float(n_new) / 2.0
    h_new = project_feasible(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def _rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(160, 0.5)

    n = len(h0)
    target = n / 2.0
    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Projected gradient on initial
    if _rem() > 200:
        h_pg, c5_pg = _projected_gradient(h0, target, min(350, _rem() - 150), seed)
        if c5_pg < best_c5:
            best_h = h_pg
            best_c5 = c5_pg

    # Phase 2: SLSQP polish with decreasing temperature
    if _rem() > 60:
        for temp in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            if _rem() < 40:
                break
            h_new = _do_slsqp(best_h, n, temp, 200)
            v = compute_c5(h_new)
            if v < best_c5:
                best_h = h_new
                best_c5 = v

    # Phase 3: Restart from perturbed best
    if _rem() > 100:
        for restart in range(5):
            if _rem() < 50:
                break
            scale = 0.04 + 0.03 * (restart % 3)
            h_pert = _perturb(best_h, scale, rng)
            h_pg, c5_pg = _projected_gradient(h_pert, target, min(80, _rem() - 40), seed + 500 + restart)
            if c5_pg < best_c5:
                best_h = h_pg
                best_c5 = c5_pg
            if _rem() > 40:
                h_new = _do_slsqp(best_h, n, 0.005, 150)
                v = compute_c5(h_new)
                if v < best_c5:
                    best_h = h_new
                    best_c5 = v

    # Phase 4: Try n=100
    if _rem() > 60:
        n2 = 100
        h2 = _upsample(best_h, n2)
        for temp in [0.05, 0.01, 0.003]:
            if _rem() < 25:
                break
            h2 = _do_slsqp(h2, n2, temp, 200)
            v = compute_c5(h2)
            if v < best_c5:
                best_h = h2
                best_c5 = v
                n = n2
                target = n / 2.0

    # Phase 5: Try n=200
    if _rem() > 80:
        n3 = 200
        h3 = _upsample(best_h, n3)
        h_pg3, c5_3 = _projected_gradient(h3, n3 / 2.0, min(60, _rem() - 40), seed + 2000)
        if c5_3 < best_c5:
            best_h = h_pg3
            best_c5 = c5_3
        if _rem() > 30:
            h3 = _do_slsqp(best_h if len(best_h) == n3 else _upsample(best_h, n3), n3, 0.01, 200)
            v = compute_c5(h3)
            if v < best_c5:
                best_h = h3
                best_c5 = v

    # Phase 6: Final polish
    if _rem() > 25:
        n_final = len(best_h)
        h_new = _do_slsqp(best_h, n_final, 0.001, 300)
        v = compute_c5(h_new)
        if v < best_c5:
            best_h = h_new
            best_c5 = v

    return best_h, best_c5, len(best_h)
