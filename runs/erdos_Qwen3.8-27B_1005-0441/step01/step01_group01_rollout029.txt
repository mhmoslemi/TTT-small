    if time.time() - t0 < budget_s - 40:
        n2 = n + 40 if n < 200 else n - 40
        if n2 >= 80:
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n2)
            h_up = np.interp(x_new, x_old, h_best)
            h_up = project_feasible(h_up, n2)
            c5_up = compute_c5(h_up)
            if c5_up < c5_best:
                c5_best = c5_up
                h_best = h_up
                n = n2
                bounds = [(0.0, 1.0)] * n
            # Quick L-BFGS on new grid
            if time.time() - t0 < budget_s - 20:
                pen = 3000.0
                res = minimize(penalty_obj_only, h_best.copy(),
                               jac=penalty_grad_only, args=(500, pen, n),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 1000, 'ftol': 1e-16})
                hc = project_feasible(res.x, n)
                c5c = compute_c5(hc)
                if c5c < c5_best:
                    c5_best = c5c
                    h_best = hc.copy()

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
