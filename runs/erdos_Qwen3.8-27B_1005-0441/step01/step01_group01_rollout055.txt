```python
    d = np.arange(-(n-1), n)  # lags
    # g[d]: coefficient of (x - h[j]) in the change for each lag
```

```python
def _project(h, n):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_N_SMALL = 160

def _sum_eq(x):
    return np.sum(x) - _N_SMALL * 0.5

def _sum_eq_jac(x):
    return np.ones(_N_SMALL)

def _obj_smooth(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -400.0, 400.0)
    return float((m + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _opt_coord(h, j, corr, n):
    x0 = h[j]
    idx0 = n - 1
    k_vals = np.arange(-(n-1), n)
    jm = j - k_vals
    jp = j + k_vals
    vm = (jm >= 0) & (jm < n)
    vp = (jp >= 0) & (jp < n)
    g = np.zeros(2*n-1)
    h_jm = np.where(vm, h[np.clip(jm, 0, n-1)], 0.0)
    h_jp = np.where(vp, h[np.clip(jp, 0, n-1)], 0.0)
    g = np.where(vm, 1.0 - h_jm, 0.0) - np.where(vp, h_jp, 0.0)
    
    # Coarse grid
    x_grid = np.linspace(0.0, 1.0, 200)
    vals = corr[None, :] + (x_grid[:, None] - x0) * g[None, :]
    vals[:, idx0] = corr[idx0] + x_grid * (1.0 - x_grid) - x0 * (1.0 - x0)
    maxes = vals.max(axis=1)
    idx = int(maxes.argmin())
    
    # Fine grid
    step = x_grid[1] - x_grid[0]
    lo = max(0.0, x_grid[idx] - step)
    hi = min(1.0, x_grid[idx] + step)
    if hi - lo < 1e-12:
        return float(x_grid[idx])
    x_fine = np.linspace(lo, hi, 500)
    vals = corr[None, :] + (x_fine[:, None] - x0) * g[None, :]
    vals[:, idx0] = corr[idx0] + x_fine * (1.0 - x_fine) - x0 * (1.0 - x0)
    maxes = vals.max(axis=1)
    idx = int(maxes.argmin())
    return float(x_fine[idx])

def _update_corr(h, j, old_x, new_x, corr, n):
    delta = new_x - old_x
    if abs(delta) < 1e-15:
        return
    idx0 = n - 1
    corr[idx0] += new_x * (1.0 - new_x) - old_x * (1.0 - old_x)
    k_vals = np.arange(-(n-1), n)
    k_vals = k_vals[k_vals != 0]
    m_vals = n - 1 + k_vals
    jm = j - k_vals
    jp = j + k_vals
    vm = (jm >= 0) & (jm < n)
    vp = (jp >= 0) & (jp < n)
    corr[m_vals[vm]] += delta * (1.0 - h[jm[vm]])
    corr[m_vals[vp]] -= delta * h[jp[vp]]

def _slsqp(h, alpha, n, maxiter):
    global _N_SMALL
    _N_SMALL = n
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}
    try:
        res = minimize(_obj_smooth, h, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=constraint,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return _project(res.x, n)
    except:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    global _N_SMALL
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n_init = 160
        h = np.zeros(n_init)
        h[:n_init//4] = 1.0
        h[-(n_init//4):] = 1.0
        h = _project(h, n_init)
    
    n = len(h)
    _N_SMALL = n
    h = _project(h, n)
    c5_best = _compute_c5(h)
    h_best = h.copy()
    
    # Phase 1: SLSQP (15%)
    t1 = t0 + budget_s * 0.15
    for alpha in [50.0, 150.0, 400.0]:
        if time.time() > t1:
            break
        h_try = _slsqp(h.copy(), alpha, n, maxiter=400)
        c5_try = _compute_c5(h_try)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try.copy()
            h = h_try.copy()
    
    # Phase 2: Coordinate descent (70%)
    t2 = t0 + budget_s * 0.85
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    while time.time() < t2:
        for j in range(n):
            old_x = h[j]
            x_new = _opt_coord(h, j, corr, n)
            if abs(x_new - old_x) > 1e-14:
                _update_corr(h, j, old_x, x_new, corr, n)
                h[j] = x_new
        h = _project(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr))
        if c5_now < c5_best:
            c5_best = c5_now
            h_best = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 50:
            h = _project(h + 0.005 * rng.standard_normal(n), n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
    
    # Phase 3: Final SLSQP (10%)
    t3 = t0 + budget_s * 0.95
    if time.time() < t3:
        for alpha in [300.0, 800.0]:
            if time.time() > t3:
                break
            h_try = _slsqp(h_best.copy(), alpha, n, maxiter=500)
            c5_try = _compute_c5(h_try)
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try.copy()
                # Quick coord descent polish
                h = h_try.copy()
                corr = np.correlate(h, 1.0 - h, mode='full')
                for _ in range(20):
                    for j in range(n):
                        old_x = h[j]
                        x_new = _opt_coord(h, j, corr, n)
                        if abs(x_new - old_x) > 1e-14:
                            _update_corr(h, j, old_x, x_new, corr, n)
                            h[j] = x_new
                    h = _project(h, n)
                    corr = np.correlate(h, 1.0 - h, mode='full')
                c5_try = _compute_c5(h)
                if c5_try < c5_best:
                    c5_best = c5_try
                    h_best = h.copy()
    
    h_best = _project(h_best, len(h_best))
    c5_best = _compute_c5(h_best)
    return h_best, c5_best, len(h_best)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_N_SMALL = 160

def _sum_eq(x):
    return np.sum(x) - _N_SMALL * 0.5

def _sum_eq_jac(x):
    return np.ones(_N_SMALL)

def _obj_smooth(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -400.0, 400.0)
    return float((m + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _opt_coord(h, j, corr, n):
    x0 = h[j]
    idx0 = n - 1
    k_vals = np.arange(-(n - 1), n)
    jm = j - k_vals
    jp = j + k_vals
    vm = (jm >= 0) & (jm < n)
    vp = (jp >= 0) & (jp < n)
    h_jm = np.where(vm, h[np.clip(jm, 0, n - 1)], 0.0)
    h_jp = np.where(vp, h[np.clip(jp, 0, n - 1)], 0.0)
    g = np.where(vm, 1.0 - h_jm, 0.0) - np.where(vp, h_jp, 0.0)

    x_grid = np.linspace(0.0, 1.0, 200)
    vals = corr[None, :] + (x_grid[:, None] - x0) * g[None, :]
    vals[:, idx0] = corr[idx0] + x_grid * (1.0 - x_grid) - x0 * (1.0 - x0)
    maxes = vals.max(axis=1)
    idx = int(maxes.argmin())

    step = x_grid[1] - x_grid[0]
    lo = max(0.0, x_grid[idx] - step)
    hi = min(1.0, x_grid[idx] + step)
    if hi - lo < 1e-12:
        return float(x_grid[idx])
    x_fine = np.linspace(lo, hi, 500)
    vals = corr[None, :] + (x_fine[:, None] - x0) * g[None, :]
    vals[:, idx0] = corr[idx0] + x_fine * (1.0 - x_fine) - x0 * (1.0 - x0)
    maxes = vals.max(axis=1)
    idx = int(maxes.argmin())
    return float(x_fine[idx])

def _update_corr(h, j, old_x, new_x, corr, n):
    delta = new_x - old_x
    if abs(delta) < 1e-15:
        return
    idx0 = n - 1
    corr[idx0] += new_x * (1.0 - new_x) - old_x * (1.0 - old_x)
    k_vals = np.arange(-(n - 1), n)
    k_vals = k_vals[k_vals != 0]
    m_vals = n - 1 + k_vals
    jm = j - k_vals
    jp = j + k_vals
    vm = (jm >= 0) & (jm < n)
    vp = (jp >= 0) & (jp < n)
    corr[m_vals[vm]] += delta * (1.0 - h[jm[vm]])
    corr[m_vals[vp]] -= delta * h[jp[vp]]

def _slsqp(h, alpha, n, maxiter):
    global _N_SMALL
    _N_SMALL = n
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}
    try:
        res = minimize(_obj_smooth, h, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=constraint,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return _project(res.x, n)
    except Exception:
        return h

def _coords_pass(h, corr, n, order=None):
    if order is None:
        order = range(n)
    for j in order:
        old_x = h[j]
        x_new = _opt_coord(h, j, corr, n)
        if abs(x_new - old_x) > 1e-14:
            _update_corr(h, j, old_x, x_new, corr, n)
            h[j] = x_new
    return h, corr

def run(seed=42, budget_s=1100, **kwargs):
    global _N_SMALL
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        ni = 160
        h = np.zeros(ni)
        h[:ni // 4] = 1.0
        h[-(ni // 4):] = 1.0
        h = _project(h, ni)

    n = len(h)
    _N_SMALL = n
    h = _project(h, n)
    c5_best = _compute_c5(h)
    h_best = h.copy()

    # Phase 1: SLSQP refinement (15% budget)
    t1 = t0 + budget_s * 0.15
    for alpha in [50.0, 150.0, 400.0]:
        if time.time() > t1:
            break
        h_try = _slsqp(h.copy(), alpha, n, 400)
        c5_try = _compute_c5(h_try)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try.copy()
            h = h_try.copy()

    # Phase 2: Coordinate descent (70% budget)
    t2 = t0 + budget_s * 0.85
    corr = np.correlate(h, 1.0 - h, mode='full')
    stall = 0
    pass_count = 0
    while time.time() < t2:
        order = np.arange(n)
        if pass_count % 3 == 1:
            rng.shuffle(order)
        h, corr = _coords_pass(h, corr, n, order)
        h = _project(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr))
        if c5_now < c5_best - 1e-9:
            c5_best = c5_now
            h_best = h.copy()
            stall = 0
        else:
            stall += 1
        pass_count += 1
        if stall > 80:
            h = _project(h + 0.003 * rng.standard_normal(n), n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0
            pass_count = 0

    # Phase 3: Final SLSQP + coord polish (10% budget)
    t3 = t0 + budget_s * 0.95
    if time.time() < t3:
        for alpha in [300.0, 700.0]:
            if time.time() > t3:
                break
            h_try = _slsqp(h_best.copy(), alpha, n, 500)
            c5_try = _compute_c5(h_try)
            if c5_try < c5_best:
                c5_best = c5_try
                h_best = h_try.copy()
            # Quick coord polish on best
            hp = h_best.copy()
            cp = np.correlate(hp, 1.0 - hp, mode='full')
            for _ in range(30):
                if time.time() > t3:
                    break
                hp, cp = _coords_pass(hp, cp, n)
                hp = _project(hp, n)
                cp = np.correlate(hp, 1.0 - hp, mode='full')
            c5p = float(np.max(cp)) * (2.0 / n)
            if c5p < c5_best:
                c5_best = c5p
                h_best = hp.copy()

    h_best = _project(h_best, len(h_best))
    c5_best = _compute_c5(h_best)
    return h_best, c5_best, len(h_best)
```
