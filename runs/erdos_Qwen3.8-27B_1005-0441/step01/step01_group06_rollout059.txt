import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.99999999
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h[mask] = np.minimum(h[mask], 1.0)
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h[mask] = np.maximum(h[mask], 0.0)
    return np.clip(h, 0.0, 1.0)

def lse_objective(x, N, beta, penalty):
    dx = 2.0 / N
    c = np.correlate(x, 1.0 - x, mode="full")
    c_max = c.max()
    shifted = beta * (c - c_max)
    S = np.sum(np.exp(shifted))
    obj = dx * (c_max + np.log(S) / beta) + penalty * (np.sum(x) - N / 2.0) ** 2
    return obj

def lse_gradient(x, N, beta, penalty):
    dx = 2.0 / N
    c = np.correlate(x, 1.0 - x, mode="full")
    c_max = c.max()
    shifted = beta * (c - c_max)
    S = np.sum(np.exp(shifted))
    w = np.exp(shifted) / S
    grad = np.zeros(N)
    for i in range(2 * N - 1):
        s = i - (N - 1)
        w_i = w[i]
        if w_i < 1e-15:
            continue
        if s >= 0:
            grad[s:N] += w_i * (1.0 - x[0:N - s])
            grad[0:N - s] -= w_i * x[s:N]
        else:
            ns = -s
            grad[0:N - ns] += w_i * (1.0 - x[ns:N])
            grad[ns:N] -= w_i * x[0:N - ns]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - N / 2.0)
    return grad

def optimize_lbfacb(h_start, N, t0, budget_s):
    h = project_feasible(h_start.copy(), N)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * N
    
    # Continuation: increase beta and penalty
    schedule = [
        # (beta, penalty, maxiter)
        (5, 10, 60),
        (20, 50, 60),
        (100, 200, 60),
        (500, 1000, 80),
        (2000, 5000, 80),
        (10000, 20000, 100),
    ]
    
    for beta, penalty, maxiter in schedule:
        if time.time() - t0 > budget_s * 0.85:
            break
        try:
            res = minimize(lse_objective, h, args=(N, beta, penalty),
                          method='L-BFGS-B', jac=lse_gradient,
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                                   'iprint': -1, 'maxfun': maxiter * 3})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, N)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def binary_local_search(h, N, t0, budget_s, max_iters=50000):
    """Local search on binary patterns: swap 1 and 0 to reduce max correlation."""
    h_bin = (h > 0.5).astype(float)
    # Fix count
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    target = N // 2
    while len(ones) > target:
        idx = ones.pop()
        h_bin[idx] = 0.0
        zeros.append(idx)
    while len(ones) < target:
        idx = zeros.pop()
        h_bin[idx] = 1.0
