def logsumexp_obj(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    if beta < 10:

    else:
        # For large beta, use top-k approximation
        topk = min(int(10 + 50 / beta), len(c))
        sorted_c = np.sort(c)[::-1]
        top = sorted_c[:topk]
        return m + np.log(np.sum(np.exp(beta * (top - m)))) / beta

    best = h.copy()
    best_c5 = compute_c5(h)[0]

    beta_schedule = [5, 20, 100, 500, 2000, 10000]

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = c.max()
    cs = c - m
