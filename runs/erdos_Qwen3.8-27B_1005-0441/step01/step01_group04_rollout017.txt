import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _subgrad_vec(h, m):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if m >= 0:
        if n - m > 0:
            grad[:n - m] += (1.0 - h[m:])
        if m < n:
            grad[m:] -= h[:n - m]
    else:
        mp = -m
        if n - mp > 0:
            grad[mp:] += (1.0 - h[:n - mp])
        if mp < n:
            grad[:n - mp] -= h[mp:]
    return grad * dx

def _grad_c5(h, alpha=5.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    weights = np.exp(alpha * (corr - cmax))
    wsum = weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] / wsum < 1e-6:
            continue
        m = k - (n - 1)
        grad += (weights[k] / wsum) * _subgrad_vec(h, m)
    return grad, cmax

def _project(h, target):
    lo = -max(h) - target
    hi = (1.0 - min(h)) + target
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0

            h[i] = max(0.0, h[i] + diff)
    return h

def _subgrad_descent(h, budget_s=30.0, alpha_soft=5.0):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()

            h_curr = h_new
    return best_h, best_c5

def _coord_descent(h, budget_s=25.0):

def _slsqp_soft(h, alpha=400, max_iter=500):
    n = len(h)
    dx = 2.0 / n
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full') * dx
        cmax = np.max(corr)
        shifted = corr - cmax
        return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    def eq(v):
        return np.sum(v) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': eq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        return res.x
    except Exception:
        return h.copy()

def _slsqp_minimax(h, max_iter=400):
    n = len(h)
    dx = 2.0 / n
    t0_init = _eval_c5(h)
    x0 = np.concatenate([h, [t0_init]])
    def mm_obj(x):
        return x[-1]
    def mm_ineq(x):
        hv = x[:-1]
        t = x[-1]
        corr = np.correlate(hv, 1.0 - hv, mode='full') * dx
        return t - corr
    def eq_x(x):
        return np.sum(x[:-1]) - n / 2.0
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(mm_obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': mm_ineq},
                                    {'type': 'eq', 'fun': eq_x}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        return res.x[:-1]
    except Exception:
        return h.copy()
