```python
import numpy as np
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _proj(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        mk = -k
        g[:n-mk] += (1.0 - h[mk:])
        g[mk:] -= h[:n-mk]
    return g

def _grad_softmax(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = c - c.max()
    w = np.exp(alpha * cs)
    w /= w.sum()
    # Only process significant weights
    idx = np.where(w > 1e-10)[0]
    grad = np.zeros(n)
    for m in idx:
        grad += w[m] * _grad_lag(h, m - (n-1))
    return grad

def _grad_exact(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    maxv = c.max()
    active = np.where(c >= maxv - 1e-10)[0]
    grad = np.zeros(n)
    for m in active:
        grad += _grad_lag(h, m - (n-1))
    return grad / len(active)

def _optimize_fista(h, alpha, t0, budget_frac):
    n = len(h)
    deadline = t0 + budget_frac
    y = h.copy()
    z = h.copy()
    t_mom = 1.0
    best = h.copy()
    best_c5 = _eval_c5(h)
    
    for it in range(5000000):
        if time.time() > deadline:
            break
        g = _grad_softmax(y, alpha)
        # Adaptive step: use 1/L where L ~ alpha
        step = 1.0 / (alpha + 1.0)
        new_z = _proj(y - step * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = new_z + (t_mom - 1.0) / t_new * (new_z - z)
        y = _proj(y, n)  # Keep y in feasible set
        z = new_z
        t_mom = t_new
        
        if it % 100 == 0:
            c5 = _eval_c5(z)
            if c5 < best_c5:
                best_c5 = c5
                best = z.copy()
    
    c5 = _eval_c5(z)
    if c5 < best_c5:
        best_c5 = c5
        best = z.copy()
    return best, best_c5

def _optimize_subgrad(h, t0, deadline):

def _coord_descent(h, n_sweeps=10):
    n = len(h)
    h = h.copy()
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            # Find optimal h[j] given others
            # f(t) = max_k (a_k * t + b_k)
            c = np.correlate(h, 1.0 - h, mode='full')
            # For coordinate j: c_m = sum_i h[i+k_m]*(1-h[i])
            # dc_m/dh[j] = a_m (computed below)
            # c_m at h[j] = t: c_m(t) = c_m(old) + a_m * (t - old)
            
            # Compute a_m = dc_m/dh[j] for all m
            a = np.zeros(2*n-1)
            for m in range(2*n-1):
                k = m - (n-1)
                # dc_m/dh[j] = (1-h[j-k])*[valid] - h[j+k]*[valid]
                g = 0.0
                if 0 <= j-k <= n-1:
                    g += (1.0 - h[j-k])
                if 0 <= j+k <= n-1:
                    g -= h[j+k]
                a[m] = g
            
            b = c - a * old
            # Minimize max_m (a[m]*t + b[m]) over t in [0,1]
            # Ternary search
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi-lo)/3.0
                m2 = hi - (hi-lo)/3.0
                f1 = np.max(a*m1 + b)
                f2 = np.max(a*m2 + b)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo+hi)/2.0
            h[j] = t_opt
            if abs(t_opt - old) > 1e-8:
                improved = True
        h = _proj(h, n)
        if not improved:
            break
    return h
