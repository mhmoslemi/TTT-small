        if not time_ok(0.80):
            break
        for alpha in [100.0, 250.0]:
            if not time_ok(0.82):
                break

            def obj_alt(x, a=alpha):
                dx = 2.0 / len(x)
                corr = np.correlate(x, 1.0 - x, mode='full')
                return smooth_max(corr, a) * dx

            res = minimize(obj_alt, sp.copy(), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 300, 'ftol': 1e-13})
            hc = project_feasible(res.x, n)
            c5c, _ = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 5: Random perturbations + local refine
    n_trials = 12
    for trial in range(n_trials):
        if not time_ok(0.90):
            break
        scale = 0.02 + 0.04 * rng.random()
        h_pert = h_best + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)

        def obj_p(x):
            dx = 2.0 / len(x)
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max(corr, 300.0) * dx

        res = minimize(obj_p, h_pert, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 150, 'ftol': 1e-13})
        hc = project_feasible(res.x, n)
        c5c, _ = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 6: Upscale to finer grid (n=240 or n=320)
    if time_ok(0.92):
        n2 = 240
        if n2 > n:
            x_old = np.linspace(0.0, 2.0, n)
            x_new = np.linspace(0.0, 2.0, n2)
            h_up = np.interp(x_new, x_old, h_best)
            h_up = project_feasible(h_up, n2)

            def obj_up(x):
                dx = 2.0 / len(x)
                corr = np.correlate(x, 1.0 - x, mode='full')
                return smooth_max(corr, 200.0) * dx

            res = minimize(obj_up, h_up, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n2,
                           options={'maxiter': 300, 'ftol': 1e-13})
            hc = project_feasible(res.x, n2)
            c5c, _ = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()
                n = n2

    # Phase 7: Final high-alpha polish
    if time_ok(0.96):
        nf = len(h_best)

        def obj_fin(x):
            dx = 2.0 / len(x)
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max(corr, 800.0) * dx

        res = minimize(obj_fin, h_best.copy(), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * nf,
                       options={'maxiter': 250, 'ftol': 1e-14})
        hc = project_feasible(res.x, nf)
        c5c, _ = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Ensure final feasibility
    h_best = project_feasible(h_best, len(h_best))
    c5_best, _ = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
