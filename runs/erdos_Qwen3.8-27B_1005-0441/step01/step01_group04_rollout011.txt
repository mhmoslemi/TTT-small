from scipy.optimize import minimize, linprog
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute evenly among those with room
        if diff > 0:
            headroom = 1.0 - h
            if headroom.max() < 1e-15:
                break
            # Add proportionally
            mask = headroom > 1e-12
            total_headroom = headroom[mask].sum()
            add = min(diff / total_headroom, 1.0)
            h[mask] += headroom[mask] * add
        else:
            mask = h > 1e-12
            total = h[mask].sum()
            if total < 1e-15:
                break
            sub = min(-diff / total, 1.0)
            h[mask] -= h[mask] * sub
    # Final clip and fix
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Add/remove from the element with most room
        if diff > 0:
            i = np.argmax(1.0 - h)
            h[i] = min(1.0, h[i] + diff)
        else:
            i = np.argmin(h)
            h[i] = max(0.0, h[i] + diff)
    return h

    g = 1.0 - h

            js = j + s
            if 0 <= js < n:
                A[row, j] = g[js] * dx
        A[row, n] = -1.0
    b = np.zeros(n_c)
    A_eq = np.zeros((1, n + 1))

                  bounds=bounds, method='highs')
    if res.success:

    return h.copy()

            best_h = h_new.copy()
        if abs(c5 - best_c5) < tol and it > 5:
            break
        h = h_new
    return best_h, best_c5

def _softmax_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def _eq_sum(h):
    return float(np.sum(h) - len(h) / 2.0)

class _ObjFunc:
    def __init__(self, alpha):
        self.alpha = alpha
    def __call__(self, h):
        return _softmax_obj(h, self.alpha)

def _slsqp_opt(h0, alpha, max_iter=500):

def _proj_subgrad(h, n, lr, n_iter=200):
