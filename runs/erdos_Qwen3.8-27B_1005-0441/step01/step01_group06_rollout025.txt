Let me reconsider. The correlation c_L = sum_n h[n+L]*(1-h[n]).

If I change h[idx] from old to new:
- The change in c_L is: (new - old) * (1-h[idx-L]) [from h[n+L] term] + h[idx+L] * (1-new - (1-old)) [from (1-h[n]) term]

  = (new - old) * (1 - h[idx-L] - h[idx+L])

So Δc_L = (new - old) * (1 - h[idx-L] - h[idx+L])

This means c_L changes linearly with h[idx]! The slope for c_L is (1 - h[idx-L] - h[idx+L]).

So if I know the current c vector, I can compute the new c vector as:
c_new[L] = c_old[L] + (h[idx]-old) * (1 - h[idx-L] - h[idx+L])

And the new max is max_L c_new[L].

This is O(n) per candidate evaluation! So the coordinate update is O(n * num_candidates).

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] += diff / cnt
            h[mask] = np.minimum(h[mask], 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] += diff / cnt
            h[mask] = np.maximum(h[mask], 0.0)
    return np.clip(h, 0.0, 1.0)

def soft_max_val(c, beta):
    m = c.max()
    if beta < 1e10:
        shifted = beta * (c - m)
        shifted -= shifted.max()
        return m + np.log(np.sum(np.exp(shifted))) / beta
    return m

def soft_max_grad_weight(c, beta):
    """Returns weights w such that grad of soft_max = sum w[p] * c[p]'s gradient."""
    m = c.max()
    if beta < 1e10:
        shifted = beta * (c - m)
        shifted -= shifted.max()
        e = np.exp(shifted)
        return e / e.sum()
    else:
        w = np.zeros(len(c))
        w[c == m] = 1.0
        w /= w.sum()
        return w

def lse_obj(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    return soft_max_val(c, beta) * dx

        else:
            g += wp * (1.0 - 2.0 * x)
    
    return g * dx

    s = beta * (c - m)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / beta

    s = beta * (c - m)
