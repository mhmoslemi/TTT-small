                    best_h = h_res
            except Exception:
                pass

    # Final micro-polish
    if best_h is not None and time.time() - t0 < budget_s * 0.995:
        n = len(best_h)
        try:
            h_res = _slsqp_optify(best_h, 0.0008, n, 500, 1e-14)
            if _c5(h_res) < best_val:
                best_val = _c5(h_res)
                best_h = h_res
        except Exception:
            pass
        try:
            h_res = _slsqp_optify(best_h, 0.0003, n, 300, 1e-14)
            if _c5(h_res) < best_val:
                best_val = _c5(h_res)
                best_h = h_res
        except Exception:
            pass

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    best_h = _proj_sum(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
