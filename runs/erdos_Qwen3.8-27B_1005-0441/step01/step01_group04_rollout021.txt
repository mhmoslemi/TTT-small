import numpy as np

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_dx(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on lambda."""
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjust
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_p < 1.0
            h_p[mask] = np.clip(h_p[mask] + diff / mask.sum(), 0, 1)
        else:
            mask = h_p > 0.0
            h_p[mask] = np.clip(h_p[mask] + diff / mask.sum(), 0, 1)
    return h_p

def _coord_optimize(h, n_iter=20, relax=0.5):
    """Coordinate descent with exact 1D minimax per coordinate."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    
    for it in range(n_iter):
        for i in range(n):
            # Compute B_k for all lags
            corr = _corr_dx(h)
            # For lag d = k - (n-1):
            # B_k = (1-h[i-d])*[0<=i-d<n] - h[i+d]*[0<=i+d<n]
            # where d = k-(n-1), so i-d = i-k+(n-1), i+d = i+k-(n-1)
            
            # Let's work with d directly
            # d ranges from -(n-1) to n-1
            # For a given d: B_d = (1-h[i-d])*[0<=i-d<n] - h[i+d]*[0<=i+d<n]
            
            # Vectorize over d:
            d_vals = np.arange(-(n-1), n)  # length 2n-1
            i_minus_d = i - d_vals  # where h[i-d] lives
            i_plus_d = i + d_vals   # where h[i+d] lives
            
            valid_minus = (i_minus_d >= 0) & (i_minus_d < n)
            valid_plus = (i_plus_d >= 0) & (i_plus_d < n)
            
            B = np.zeros(2*n-1)
            B[valid_minus] = (1.0 - h[i_minus_d[valid_minus]])
            B[valid_plus] -= h[i_plus_d[valid_plus]]
            
            # A_k = c[k] - h[i] * B_k
            hi_val = h[i]
            A = corr - hi_val * B
            
            # Minimize max_k (A_k + x * B_k) over x in [0,1]
            # Try x=0 and x=1
            f0 = np.max(A)
            f1 = np.max(A + B)
            best_x = 0.0 if f0 < f1 else 1.0
            best_val = min(f0, f1)
            
            # Check interior: for pairs (p, q) with B_p>0, B_q<0
            pos = B > 1e-10
            neg = B < -1e-10
            if np.any(pos) and np.any(neg):
                # For each pos line and neg line, find intersection
                # x* = (A_q - A_p) / (B_p - B_q)
                Bp = B[pos]
                Ap = A[pos]
                Bn = B[neg]
                An = A[neg]
                
                # x* = (A_n - A_p) / (B_p - B_n) for all pairs
                # Vectorize: compute all pairs
                # This could be large (80*80=6400), manageable
                x_star = (An[:, None] - Ap[None, :]) / (Bp[None, :] - Bn[:, None])
                valid = (x_star >= 0) & (x_star <= 1)
                if np.any(valid):
                    val_star = Ap[None, :] + x_star * Bp[None, :]
                    val_star = np.where(valid, val_star, np.inf)
                    min_idx = np.unravel_index(np.argmin(val_star), val_star.shape)
                    if val_star[min_idx] < best_val:
                        best_val = val_star[min_idx]
                        best_x = x_star[min_idx]
            
            # Update with relaxation
            h[i] = (1 - relax) * h[i] + relax * best_x
            h[i] = np.clip(h[i], 0.0, 1.0)
        
        # Project to maintain sum constraint
        h = _project_sum(h, target)
    
    return h

def _subgradient_step(h, alpha):
    """One subgradient step on the max of correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    active = np.where(corr >= cmax - 1e-8)[0]

        if len(i_plus_d) > 0:

    grad *= dx / len(active)
    return h - alpha * grad, cmax

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_p < 1.0
            if mask.sum() > 0:
                h_p[mask] = np.clip(h_p[mask] + diff / mask.sum(), 0, 1)
        else:
            mask = h_p > 0.0
            if mask.sum() > 0:
                h_p[mask] = np.clip(h_p[mask] + diff / mask.sum(), 0, 1)
    return h_p

            B = np.zeros(2 * n - 1)
            B[vm] = 1.0 - h[imd[vm]]
            B[vp] -= h[ipd[vp]]
            hi_val = h[i]
            A = corr - hi_val * B
            f0 = np.max(A)
            f1 = np.max(A + B)
            best_x = 0.0 if f0 <= f1 else 1.0

            pos = B > 1e-10

def _subgradient_optimize(h, n_steps=300):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best = h.copy()
    best_c5 = _eval_c5(h)
    for it in range(n_steps):
        alpha = 0.02 / (1.0 + 0.005 * it)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr)
        thresh = cmax - max(0.005, 0.02 * cmax)
        active = np.where(corr >= thresh)[0]
        grad = np.zeros(n)
        for k_idx in active:
            d = k_idx - (n - 1)
            lo1 = max(0, d)
            hi1 = min(n, n + d)
            if lo1 < hi1:
                grad[lo1:hi1] += (1.0 - h[lo1 - d:hi1 - d])
            lo2 = max(0, -d)
            hi2 = min(n, n - d)
            if lo2 < hi2:
                grad[lo2:hi2] -= h[lo2 + d:hi2 + d]
        grad *= dx / max(1, len(active))
        h_new = h - alpha * grad
        h_new = _project_sum(h_new, target)
        c5 = _eval_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
        h = h_new
    return best, best_c5

    h = np.zeros(n)
    p = max(3, n // 15)
    for i in range(n):
        h[i] = 1.0 if (i % p) < p // 2 else 0.0
    return _project_sum(h, n / 2.0)
