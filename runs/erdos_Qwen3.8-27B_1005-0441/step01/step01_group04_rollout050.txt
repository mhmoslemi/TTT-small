import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _lse_obj(x, alpha, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    ms = np.max(shifted)
    exp_s = np.exp(alpha * (shifted - ms))
    Z = np.sum(exp_s)
    obj = cmax + (ms + np.log(Z)) / alpha
    d = np.sum(x) - target
    obj += 1e6 * d * d
    return obj

def _lse_grad(x, alpha, target):
    n = len(x)
    dx = 2.0 / n
    v = 1.0 - x
    corr = np.correlate(x, v, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    ms = np.max(shifted)
    exp_s = np.exp(alpha * (shifted - ms))
    Z = np.sum(exp_s)
    w = exp_s / Z
    n_full = 2 * n - 1
    s_arr = np.arange(n_full) - (n - 1)
    grad = np.zeros(n)
    for p in range(n):
        fwd = (p + s_arr >= 0) & (p + s_arr < n)
        bwd = (p - s_arr >= 0) & (p - s_arr < n)
        a = np.zeros(n_full)
        a[fwd] = v[p + s_arr[fwd]]
        a[bwd] -= x[p - s_arr[bwd]]
        grad[p] = np.dot(w, a) * dx
    d = np.sum(x) - target
    grad += 2e6 * d
    return grad

def _optimize_lse(h0, alpha, max_iter=400):
    n = len(h0)
    target = n / 2.0
    try:
        res = minimize(_lse_obj, h0, jac=_lse_grad, method='L-BFGS-B',
                       args=(alpha, target),
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15,
                                'gtol': 1e-9, 'maxls': 50})
        return res.x
    except Exception:
        return h0.copy()

def _block_cd(h, n_rounds=20):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    n_full = 2 * n - 1
    s_arr = np.arange(n_full) - (n - 1)
    xs_grid = np.linspace(0.0, 1.0, 151)
    for _ in range(n_rounds):
        changed = False
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        for p in range(n):
            hp = h[p]
            fwd = (p + s_arr >= 0) & (p + s_arr < n)
            bwd = (p - s_arr >= 0) & (p - s_arr < n)
            a = np.zeros(n_full)
            a[fwd] = (1.0 - h[p + s_arr[fwd]])
            a[bwd] -= h[p - s_arr[bwd]]
            b = corr - hp * a
            scores = np.max(a[None, :] * xs_grid[:, None] + b[None, :], axis=1)
            x_new = xs_grid[int(np.argmin(scores))]
            if abs(x_new - hp) > 1e-9:
                h[p] = x_new
                changed = True
        if changed:
            h = _project_sum(h, target)
        else:
            break
    return h

def _subgrad_polish(h, n_steps=2000, lr=0.01):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    n_full = 2 * n - 1
    s_arr = np.arange(n_full) - (n - 1)
    for it in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        s_star = k_star - (n - 1)
        g = np.zeros(n)
        fwd = (s_arr == s_star)
        v = 1.0 - h
        for p in range(n):
            term = 0.0
            ps = p + s_star
            if 0 <= ps < n:
                term += v[ps]
            pm = p - s_star
            if 0 <= pm < n:
                term -= h[pm]
            g[p] = term * dx
        lr_t = lr / (1.0 + 0.01 * it)
        h = _project_sum(h - lr_t * g, target)
    return h

def _try_n(h_best, n_new, rng, time_left):
    if time_left() < 30:
        return h_best
    if n_new <= 0 or n_new > 200:
        return h_best
    h0 = np.interp(np.linspace(0, 1, n_new, endpoint=False),

    if time_left() > 15:

        h_best = _project_sum(h_best, len(h_best) / 2.0)
    return h_best
