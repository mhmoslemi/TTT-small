import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_values(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _smoothed_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = _corr_values(h)
    cmax = corr.max()
    shifted = corr - cmax
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()
    m_idx = np.arange(n)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-12:
            continue
        s = k - (n - 1)
        ms = m_idx - s
        mp = m_idx + s
        valid_ms = (ms >= 0) & (ms < n)
        valid_mp = (mp >= 0) & (mp < n)
        h_ms = np.where(valid_ms, h[np.clip(ms, 0, n - 1)], 0.0)
        h_mp = np.where(valid_mp, h[np.clip(mp, 0, n - 1)], 0.0)
        grad += weights[k] * dx * ((1.0 - h_ms) * valid_ms.astype(float) - h_mp * valid_mp.astype(float))
    return grad

def _project(v, target):
    lo = -np.max(v) - 2.0
    hi = -np.min(v) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def _fista(h0, alpha, target, n_iter, step_size, deadline):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    t_mom = 1.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        grad = _smoothed_grad(h, alpha)
        y = h + (t_mom - 1.0) / (t_mom + 1.0) * (h - h_prev)
        h_new = _project(y - step_size * grad, target)
        h_prev = h.copy()
        h = h_new
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
    return h

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-16:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-16:
                break
            h[i] -= sub
    return h

def _slsqp_smooth(h0, alpha, max_iter):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = _corr_values(x)
        cmax = corr.max()
        return cmax + np.log(np.sum(np.exp(alpha * (corr - cmax)))) / alpha

    def obj_grad(x):
        return _smoothed_grad(x, alpha)

    def eq_fun(x):
        return np.sum(x) - target

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': eq_fun}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h0.copy()

def _slsqp_minimax(h0, max_iter):
    n = len(h0)
    target = n / 2.0
    t0 = _eval_c5(h0)
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def mm_obj(x):
        return x[-1]

    def mm_ineq(x):
        h = x[:-1]
        t = x[-1]
        corr = _corr_values(h)
        return t - corr

    def eq_x(x):
        return np.sum(x[:-1]) - target

    try:
        res = minimize(mm_obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': mm_ineq},
                                    {'type': 'eq', 'fun': eq_x}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return res.x[:-1]
    except Exception:
        return h0.copy()

def _coord_descent(h, n_sweeps, deadline):
    n = len(h)
    h = h.copy()
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            old = h[i]
            best_val = old
            best_obj = _eval_c5(h)
            for delta in np.linspace(-0.08, 0.08, 17):
                nv = old + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                h[i] = nv
                obj = _eval_c5(h)
                if obj < best_obj:
                    best_obj = obj
                    best_val = nv
            h[i] = best_val
        h = _fix_sum(h, n)
    return h
