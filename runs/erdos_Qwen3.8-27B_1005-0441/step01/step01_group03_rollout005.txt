def compute_weighted_grad(h, n, weights):
    """weights is array of length 2n-1, indexed by correlator index"""
    g = np.zeros(n)
    for idx in range(2*n - 1):
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        s = idx - (n - 1)  # actual shift
        # grad_s[m] = (1-h[m+s])*I(0≤m+s<n) - h[m-s]*I(0≤m-s<n)
        # First term: m ranges over 0..n-1-s if s≥0, or 0..n-1 if s<0 (m+s≥0 means m≥-s)
        if s >= 0:
            # m+s < n => m < n-s, m ≥ 0
            for m in range(n - s):
                g[m] += w * (1.0 - h[m + s])
            # m-s ≥ 0 => m ≥ s, m < n
            for m in range(s, n):
                g[m] -= w * h[m - s]
        else:
            ss = -s  # positive
            # m+s < n is always true (m < n, s < 0), m+s ≥ 0 => m ≥ ss
            for m in range(ss, n):
                g[m] += w * (1.0 - h[m - ss])
            # m-s < n => m + ss < n => m < n-ss, m ≥ 0
            for m in range(n - ss):
                g[m] -= w * h[m + ss]
    return g
