import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        cnt = float(np.count_nonzero(mask))
        if cnt > 0:
            hp = np.clip(hp + diff / cnt, 0.0, 1.0)
    return hp

def ov_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[:L] += 1.0 - h[lag:]
            if lag > 0:
                g[lag:] -= h[:L]
    else:
        m = -lag
        L = n - m
        if L > 0:
            g[m:] += 1.0 - h[:L]
            g[:L] -= h[m:]
    return g

def lse_grad(h, n, beta):
    dx = 2.0 / n
    ov = ov_full(h, n)
    mx = float(np.max(ov))
    sh = np.clip(beta * (ov - mx), -300, 300)
    e = np.exp(sh)
    s = float(np.sum(e))
    if s < 1e-300:
        return mx, mx, np.zeros(n)
    w = e / s
    g = np.zeros(n)
    c = n - 1
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-9:
            continue
        g += wk * grad_lag(h, k - c, n)
    g *= dx
    lse = mx + np.log(s) / beta
    return lse, mx, g

def optimize(h, n, deadline, rng, b0=3.0, b1=400.0, lr0=0.25):
    h = h.copy()
    ov = ov_full(h, n)
    best = h.copy()

    stall = 0
    it = 0
    t_s = time.time()
    t_total = max(1.0, deadline - t_s)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_s) / t_total)
        beta = b0 + (b1 - b0) * frac * frac
        lse_v, _, g = lse_grad(h, n, beta)
        alpha = lr
        h_new = h
        lse_new = lse_v

            if lv < lse_v - 1e-15:
                h_new = cand
                lse_new = lv
                break
            alpha *= 0.5
        h = h_new
        it += 1
        c5 = float(np.max(ov_full(h, n)))
        if c5 < best_c5 - 1e-8:
            best_c5 = c5
            best = h.copy()
            stall = 0
        else:

        if stall == 50:

                lr = lr0

def make_structured(n, pattern='two_block'):
    h = np.zeros(n)
    if pattern == 'two_block':
        q = n // 4
        h[:q] = 1.0
        h[n // 2:n // 2 + q] = 1.0
    elif pattern == 'single_block':
        h[:n // 2] = 1.0
    elif pattern == 'alt':
        h[::2] = 1.0
    elif pattern == 'three':
        q = n // 6
        h[:q] = 1.0
        h[n//3:2*n//3] = 1.0
        h[-q:] = 1.0
    return project_feasible(h, n)

def upscale(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    out = np.interp(xt, xf, h)
    return project_feasible(out, n_to)
