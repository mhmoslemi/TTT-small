def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(v, target_sum):
    # Project v onto {0 <= h <= 1, sum(h) = target_sum}
    # Using bisection on lambda
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # Fix any small errors
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute evenly
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(v, target_sum):
    n = len(v)
    lo, hi = -5.0, 5.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        free = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
        if len(free) > 0:
            h[free] = np.clip(h[free] + diff / len(free), 0.0, 1.0)
    return h

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / temp
    shifted -= np.max(shifted)
    return (m + temp * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / temp
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        s = idx - (n - 1)
        if s == 0:
            grad += w[idx] * (1.0 - 2.0 * h)
        elif s > 0:
            g = np.zeros(n)
            g[:n-s] += (1.0 - h[s:])
            g[s:] += (-h[:n-s])
            grad += w[idx] * g
        else:
            mm = -s
            g = np.zeros(n)
            g[mm:] += (1.0 - h[:n-mm])
            g[:n-mm] += (-h[mm:])
            grad += w[idx] * g
    return grad * dx

def slsqp_optimize(h_init, max_time, seed=0):
    n = len(h_init)
    target = float(n) / 2.0
    best = h_init.copy()
    best_val = compute_c5(best)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    
    def eq_fun(h):
        return np.sum(h) - target
    def eq_jac(h):
        return np.ones(n)
    
    cons = ({"type": "eq", "fun": eq_fun, "jac": eq_jac},)
    
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for temp in temps:
        if time.time() - t0 > max_time - 10:
            break
        h = best.copy()
        for it in range(3):
            if time.time() - t0 > max_time - 5:
                break
            try:
                res = minimize(smooth_obj, h, jac=lambda x: smooth_obj_grad(x, temp),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 300, 'ftol': 1e-14})
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new = project_feasible(h_new, target)
                v = compute_c5(h_new)
                if v < best_val - 1e-12:
                    best = h_new.copy()
                    best_val = v
                h = h_new

                break

    t = max(_g_temp, 1e-10)
    sh = (corr - m) / t
    sh -= np.max(sh)
    return (m + t * np.log(np.sum(np.exp(sh)))) * dx

def _smooth_grad(h):
    dx = 2.0 / _g_n
    n = _g_n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    t = max(_g_temp, 1e-10)
    sh = (corr - m) / t
    sh -= np.max(sh)
    w = np.exp(sh)
    w /= np.sum(w)
    grad = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for idx in active:
        s = int(idx) - (n - 1)
        wi = w[idx]
        if s == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += wi * (1.0 - h[s:])
            grad[s:] += (-wi) * h[:n-s]
        else:
            mm = -s
            grad[mm:] += wi * (1.0 - h[:n-mm])
            grad[:n-mm] += (-wi) * h[mm:]
    return grad * dx

def _eq_fun(h):
    return np.sum(h) - _g_n / 2.0

def _eq_jac(h):
    return np.ones(_g_n)

def slsqp_opt(h_init, max_time, sd=0):
    global _g_n, _g_temp
    n = len(h_init)
    _g_n = n
    target = float(n) / 2.0
    best = h_init.copy()
    best_val = compute_c5(best)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = ({"type": "eq", "fun": _eq_fun, "jac": _eq_jac},)
    temps = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004]
    for temp in temps:
        if time.time() - t0 > max_time - 10:
            break
        _g_temp = temp
        h = best.copy()
        for it in range(3):
            if time.time() - t0 > max_time - 3:
                break
            try:
                res = minimize(_smooth_obj, h, jac=_smooth_grad,
                              method='SLSQP', bounds=bounds,
                              constraints=cons,
                              options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_feasible(np.clip(res.x, 0.0, 1.0), target)
                v = compute_c5(h_new)
                if v < best_val - 1e-13:
                    best = h_new.copy()
                    best_val = v
                h = h_new.copy()
            except Exception:
                break
    return best, best_val

def subgrad_opt(h_init, n_iter, lr0=0.4, top_k=6):
    n = len(h_init)
    dx = 2.0 / n
    target = float(n) / 2.0
    h = h_init.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    for t in range(n_iter):
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        k = min(top_k, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        w = c[top_idx] - c[top_idx].min() + 1e-10
        w = w ** 4
        w /= w.sum()
        grad = np.zeros(n)
        for jj in range(len(top_idx)):
            idx = int(top_idx[jj])
            s = idx - (n - 1)
            wi = w[jj]
            if s == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif s > 0:
                grad[:n-s] += wi * (1.0 - h[s:])
                grad[s:] += (-wi) * h[:n-s]
            else:
                mm = -s
                grad[mm:] += wi * (1.0 - h[:n-mm])
                grad[:n-mm] += (-wi) * h[mm:]
        lr = lr0 / (1.0 + 0.004 * t)
        h = project_feasible(h - lr * grad, target)
        if (t + 1) % 500 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best = h.copy()
    return best, best_c5

def resample(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(xn, xo, h)
    return project_feasible(h_new, float(n_new) / 2.0)

def make_binary(n, sd):
    rng = np.random.RandomState(sd)
    h = np.zeros(n)
    h[rng.permutation(n)[:n // 2]] = 1.0
    return h

def make_ramp(n, sd):
    rng = np.random.RandomState(sd)
    h = np.zeros(n)
    a = int(rng.uniform(0, n * 0.3))
    b = int(rng.uniform(n * 0.6, n))
    h[a:b] = 1.0
    return project_feasible(h, float(n) / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(160, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Subgradient warmup
    if rem() > 400:
        h, c5 = subgrad_opt(best_h, 4000, lr0=0.5, top_k=8)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: SLSQP at original n
    if rem() > 250:
        h, c5 = slsqp_opt(best_h, max_time=min(150, rem() - 220), sd=seed)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: Different n values
    for tn in [n0 - 40, n0 + 40, n0 * 2]:
        if rem() < 100 or tn < 40 or tn > 400 or tn == len(best_h):
            continue
        h_up = resample(best_h, tn)
        h_up, c5s = subgrad_opt(h_up, 2000, lr0=0.3, top_k=5)
        if rem() < 40:
            continue
        h, c5 = slsqp_opt(h_up, max_time=min(60, rem() - 30), sd=seed + tn)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Alternative initializations at best n
    nc = len(best_h)
    inits = [
        make_binary(nc, seed + 1),
        make_ramp(nc, seed + 2),
        make_ramp(nc, seed + 3),
        project_feasible(best_h + 0.15 * np.random.randn(nc), nc / 2.0),
    ]
    for ii, hi in enumerate(inits):
        if rem() < 50:
            break
        h, c5 = slsqp_opt(hi, max_time=min(35, rem() - 30), sd=seed + 200 + ii)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 5: Random restarts
    nc = len(best_h)
    for r in range(50):
        if rem() < 25:
            break
        scale = 0.04 + 0.06 * (r % 6) / 5.0
        h_p = project_feasible(best_h + scale * np.random.randn(nc), nc / 2.0)
        mt = min(20, rem() - 15)
        if mt < 8:
            break
        h, c5 = slsqp_opt(h_p, max_time=mt, sd=seed + 5000 + r)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: Subgradient + SLSQP refinement cycle
    if rem() > 60:
        h, c5 = subgrad_opt(best_h, 3000, lr0=0.2, top_k=8)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        if rem() > 30:
            h, c5 = slsqp_opt(best_h, max_time=max(10, rem() - 15), sd=seed + 9999)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Phase 7: Final SLSQP polish
    if rem() > 15:
        h, c5 = slsqp_opt(best_h, max_time=max(8, rem() - 5), sd=seed + 7777)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    return best_h, best_c5, len(best_h)
