import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        free = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(free.sum())
        if cnt > 0:
            hp[free] = np.clip(hp[free] + diff / cnt, 0.0, 1.0)
    return hp

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def weighted_gradient(h, n, weights):
    """Compute gradient = sum_k weights[k] * grad_k where weights is for index n-1+k."""
    g = np.zeros(n)
    for k in range(-(n - 1), n):
        idx = n - 1 + k
        w = weights[idx]
        if w < 1e-10:
            continue
        if k >= 0:
            if n - k > 0:
                g[k:] += w * (1.0 - h[:n - k])
                g[:n - k] -= w * h[k:]
        else:
            p = -k
            if n - p > 0:
                g[:n - p] += w * (1.0 - h[p:])
                g[p:] -= w * h[:n - p]
    return g

def soft_subgrad(h, n, deadline, rng):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    T = 0.03
    lr_base = 0.15
    it = 0
    no_improve = 0
    while time.time() < deadline:
        overlaps = compute_overlaps(h, n)
        c5 = np.max(overlaps)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        exp_vals = np.exp(overlaps / T - np.max(overlaps) / T)
        weights = exp_vals / exp_vals.sum()
        g = weighted_gradient(h, n, weights)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        lr = lr_base / (1.0 + 0.005 * it)
        h_new = h - lr * g / max(g_norm, 0.01)
        h_new = project_feasible(h_new, n)
        h = h_new
        T = max(0.005, T * 0.9995)
        it += 1
        if no_improve > 200:
            h = best_h + rng.randn(n) * (0.08 + 0.04 * rng.rand())
            h = project_feasible(h, n)
            no_improve = 0
            T = 0.04
            it = 0
    return best_h, best_c5

def binary_local_search(h_bin, n, deadline, rng, max_rounds=500):
    dx = 2.0 / n
    h = h_bin.copy()
    c5_cur = get_c5(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h < 0.5)[0].tolist())
    round_it = 0
    stuck = 0
    while time.time() < deadline and round_it < max_rounds:
        round_it += 1
        improved = False
        one_list = list(ones)
        zero_list = list(zeros)
        rng.shuffle(one_list)
        rng.shuffle(zero_list)
        for i in one_list:
            for j in zero_list:
                h[i] = 0.0
                h[j] = 1.0
                c5_new = get_c5(h, n)
                h[i] = 1.0
                h[j] = 0.0
                if c5_new < c5_cur - 1e-14:
                    h[i] = 0.0
                    h[j] = 1.0
                    ones.remove(i)
                    ones.add(j)
                    zeros.remove(j)
                    zeros.add(i)
                    c5_cur = c5_new
                    improved = True
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    break
            if improved and time.time() > deadline:
                break
        if not improved:
            stuck += 1
            if stuck > 5:
                break
        else:
            stuck = 0
    return best_h, best_c5

def coordinate_polish(h, n, deadline, rng):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    no_improve = 0
    it = 0
    while time.time() < deadline and no_improve < 200:
        improved = False
        for j in range(n):
            old = h[j]
            for candidate in [0.0, 1.0, 0.1, 0.3, 0.5, 0.7, 0.9]:
                if abs(candidate - old) < 1e-9:
                    continue
                h[j] = candidate
                # Adjust to maintain sum
                hp = project_feasible(h, n)
                c5_new = get_c5(hp, n)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = hp.copy()
                    h = hp.copy()
                    improved = True
                    no_improve = 0
                else:
                    h[j] = old
            if time.time() > deadline:
                return best_h, best_c5
        if not improved:
            no_improve += 1
        it += 1
    return best_h, best_c5

def try_n(n, deadline, rng, start_h=None):
    dx = 2.0 / n
    if start_h is not None:
        h = project_feasible(start_h.copy(), n)
    else:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = project_feasible(h, n)
    # Continuous optimization
    t_rem = max(deadline - time.time(), 1.0)
    d1 = time.time() + t_rem * 0.55
    bh, bc5 = soft_subgrad(h, n, d1, rng)
    # Binarize and local search
    if time.time() < deadline:
        h_bin = (bh > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        d2 = min(time.time() + t_rem * 0.25, deadline)

        bp, cp = coordinate_polish(bh, n, max(deadline - 2, time.time() + 1), rng)
        if cp < bc5:
            bh, bc5 = bp, cp
    return bh, bc5
