Let me also add a "dual" approach: since the problem is min_h max_k C(k)*dx, we can use the von Neumann minimax or a simple weight-based approach:

- Outer: update w_k ∝ exp(C_k * dx / tau) for some temperature tau

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    r = target - np.sum(h)
    if abs(r) > 1e-10:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = np.sum(m)
        if c > 0:
            h[m] += r / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def active_grad(h, n, deltas):
    g = np.zeros(n)
    na = max(len(deltas), 1)
    for delta in deltas:
        d = abs(int(delta))
        if d == 0:
            g += 1.0 - 2.0 * h
        elif d < n:
            if delta > 0:
                g[:n - d] += 1.0 - h[d:]
                g[d:] -= h[:n - d]
            else:
                g[d:] += 1.0 - h[:n - d]
                g[:n - d] -= h[d:]
        elif d == n:
            g[d:] += 1.0 - h[:0]  # no valid terms, skip
    return g / na

def top_deltas(c5v, n, k):
    idx = np.argsort(c5v)[::-1][:k]
    return (idx - (n - 1)).tolist()

def subgrad_run(h, n, t0, deadline, k0=5):

    best_c = compute_c5(h)
    best_h = h.copy()
    lr = 0.05
    pat = 0

        c5m = float(np.max(c5v))
        if c5m < best_c - 1e-12:
            best_c = c5m
            best_h = h.copy()
            pat = 0

            pat = 0

        kk = k0 + min(pat // 200, 15)
        dl = top_deltas(c5v, n, min(kk, 2 * n - 1))
        g = active_grad(h, n, dl)
        h = project_feasible(h - lr * g, n)
    return best_h, best_c

def slsqp_refine(h, n, alpha):
    target = 0.5 * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return (m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / len(x))
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(len(x))
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                   options={'maxiter': 1500, 'ftol': 1e-14})
    return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n))

def binary_search(h, n, deadline):
    hb = (h >= 0.5).astype(float)
    tgt = n // 2
    # Fix count
    while np.sum(hb) > tgt:
        j = np.argmax(hb * (1 - hb))  # least "confident" one
        if hb[j] < 0.5:
            j = np.argmax(hb)
        hb[j] = 0.0

        if hb[j] > 0.5:
            j = np.argmin(hb)
        hb[j] = 1.0

        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        best_gain = 1e-12
        best_pair = None
        for a in ones:
            for b in zeros:

                c = compute_c5(hb)
                gain = c5b - c
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (a, b)
                hb[a] = 1; hb[b] = 0
        if best_pair:
            a, b = best_pair
            hb[a] = 0; hb[b] = 1
            c5b -= best_gain
            improved = True
    return hb, c5b

            if a >= n:
                continue
            if d > 0:
                g[:n - a] += 1.0 - h[a:]
                g[a:] -= h[:n - a]
            else:
                g[a:] += 1.0 - h[:n - a]
                g[:n - a] -= h[a:]
    return g / na

def _todel(c5v, n, k):
    idx = np.argsort(c5v)[::-1][:k]
    return (idx - (n - 1)).tolist()

def _sg(h, n, deadline, k0=5):
    h = h.copy()
    bc = _c5(h)
    bh = h.copy()
    lr = 0.08
    pat = 0
    while time.time() < deadline:
        cv = _c5v(h)
        cm = float(np.max(cv))
        if cm < bc - 1e-13:
            bc = cm
            bh = h.copy()
            pat = 0
        else:
            pat += 1
        if pat > 300:
            lr *= 0.65
            pat = 0
            if lr < 1e-8:
                break
        kk = min(k0 + min(pat // 100, 20), 2 * n - 1)
        dls = _todel(cv, n, kk)
        g = _agrad(h, n, dls)
        h = _proj(h - lr * g, n)
    return bh, bc

def _sobj(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(_SGA * (c - m), -400.0, 400.0)
    return (m + np.log(np.sum(np.exp(s))) / _SGA) * (2.0 / len(x))

def _scon(x):
    return np.sum(x) - _SGT

def _sjac(x):
    return np.ones(len(x))

def _slsqp(h, n, alpha, maxit=1200):
    global _SGT, _SGA
    _SGT = 0.5 * n
    _SGA = alpha
    res = minimize(_sobj, h.copy(), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                   options={'maxiter': maxit, 'ftol': 1e-14})
    hc = _proj(res.x, n)
    return hc, _c5(hc)

def _lbobj(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    s = np.clip(_SGA * (c - m), -400.0, 400.0)
    base = (m + np.log(np.sum(np.exp(s))) / _SGA) * (2.0 / len(x))
    pen = 200.0 * (np.sum(x) - _SGT) ** 2
    return base + pen

def _lbgf(h, n, alpha, maxit=600):
    global _SGT, _SGA
    _SGT = 0.5 * n
    _SGA = alpha
    res = minimize(_lbobj, h.copy(), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxit, 'ftol': 1e-15})
    hc = _proj(res.x, n)
    return hc, _c5(hc)

def _binsearch(h, n, deadline):
    hb = (h >= 0.5).astype(float)
    t = n // 2
    diff = int(np.sum(hb)) - t
    if diff > 0:
        for _ in range(diff):
            if time.time() > deadline:
                break
            idx = list(np.where(hb == 1)[0])
            bj, bc = idx[0], _c5(hb)
            for j in idx:
                hb[j] = 0.0
                c = _c5(hb)
                if c < bc:
                    bc = c
                    bj = j
                hb[j] = 1.0
            hb[bj] = 0.0
    elif diff < 0:
        for _ in range(-diff):
            if time.time() > deadline:
                break
            idx = list(np.where(hb == 0)[0])
            bj, bc = idx[0], _c5(hb)
            for j in idx:
                hb[j] = 1.0
                c = _c5(hb)
                if c < bc:
                    bc = c
                    bj = j
                hb[j] = 0.0
            hb[bj] = 1.0
    c5b = _c5(hb)
    imp = True
    while imp and time.time() < deadline:
        imp = False
        ones = list(np.where(hb == 1)[0])
        zeros = list(np.where(hb == 0)[0])
        bg, bp = 1e-14, None
        for a in ones:
            for b in zeros:
                hb[a] = 0.0
                hb[b] = 1.0
                c = _c5(hb)
                g = c5b - c
                if g > bg:
                    bg = g
                    bp = (a, b)
                hb[a] = 1.0
                hb[b] = 0.0
        if bp is not None:
            hb[bp[0]] = 0.0
            hb[bp[1]] = 1.0
            c5b -= bg
            imp = True
    return hb, c5b

def _twoblock(n):
    h = np.zeros(n)
    b = n // 4
    h[:b] = 1.0
    h[-b:] = 1.0
    return _proj(h, n)

def _threblock(n):
    h = np.zeros(n)
    b = n // 6
    h[:b] = 1.0
    h[n // 2 - b:n // 2 + b] = 1.0
    h[-b:] = 1.0
    return _proj(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = _twoblock(160)
    n = len(h0)
    h = _proj(h0.copy(), n)
    c5_best = _c5(h)
    h_best = h.copy()
    print(f"Init n={n}: C5={c5_best:.6f}", flush=True)

    dl1 = t0 + budget_s * 0.22
    h1, c1 = _sg(h, n, dl1, k0=5)
    if c1 < c5_best:
        c5_best, h_best = c1, h1
    print(f"P1 sg: C5={c5_best:.6f}", flush=True)

    dl2 = t0 + budget_s * 0.40
    if time.time() < dl2:
        for a in [60.0, 250.0, 600.0]:
            if time.time() > dl2:
                break
            hc, c = _slsqp(h_best, n, a, maxit=1000)
            if c < c5_best:
                c5_best, h_best = c, hc
    print(f"P2 slsqp: C5={c5_best:.6f}", flush=True)

    dl2b = t0 + budget_s * 0.50
    if time.time() < dl2b:
        hc, c = _lbgf(h_best, n, 200.0, maxit=500)
        if c < c5_best:
            c5_best, h_best = c, hc

    dl3 = t0 + budget_s * 0.58
    if time.time() < dl3:
        h3, c3 = _sg(h_best, n, dl3, k0=10)
        if c3 < c5_best:
            c5_best, h_best = c3, h3
            print(f"P3 sg2: C5={c5_best:.6f}", flush=True)

    dl4 = t0 + budget_s * 0.72
    if time.time() < dl4:
        hb, c5b = _binsearch(h_best, n, dl4)
        if c5b < c5_best:
            c5_best, h_best = c5b, hb
    print(f"P4 bin: C5={c5_best:.6f}", flush=True)

    dl5 = t0 + budget_s * 0.85
    for starter_fn in [_twoblock, _threblock]:
        if time.time() > dl5:
            break
        s_h = starter_fn(n)
        s_h, s_c = _sg(s_h, n, min(time.time() + budget_s * 0.03, dl5), k0=5)
        if time.time() < dl5 - 5:
            s_h, s_c = _slsqp(s_h, n, 200.0, maxit=800)
        if s_c < c5_best:
            c5_best, h_best = s_c, s_h
            print(f"P5 alt: C5={c5_best:.6f}", flush=True)

    dl6 = t0 + budget_s - 30
    if time.time() < dl6:
        n2 = 2 * n
        if n2 <= 400:
            xo = np.arange(n, dtype=float)
            xn = np.linspace(0.0, float(n - 1), n2)
            hu = np.interp(xn, xo, h_best)
            hu = _proj(hu, n2)
            hu, cu = _sg(hu, n2, min(time.time() + budget_s * 0.04, dl6), k0=5)
            if time.time() < dl6 - 10:
                hu, cu = _slsqp(hu, n2, 300.0, maxit=800)
            if cu < c5_best:
                c5_best, h_best, n = cu, hu, n2
                print(f"P6 up n2: C5={c5_best:.6f}", flush=True)

    h_best = _proj(h_best, len(h_best))
    c5_best = _c5(h_best)
    return h_best, c5_best, len(h_best)
