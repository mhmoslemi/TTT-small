            # Distribute to non-fixed coords
            fixed_set = set(coords)
            free = [i for i in range(n) if i not in fixed_set]
            if not free:
                h[:] = h_save
                return 1e6
            for idx in free:
                adj = diff / len(free)
                h[idx] = np.clip(h[idx] + adj, 0.0, 1.0)
        return _eval_c5(h)
    x0 = np.array([h[c] for c in coords])
    try:
        res = minimize(obj, x0, method='Nelder-Mead',
                       options={'maxiter': max_iter, 'xatol': 1e-10, 'fatol': 1e-12})
        for k, idx in enumerate(coords):
            h[idx] = np.clip(res.x[k], 0.0, 1.0)
        h = _fix_sum(h, n)
        if _eval_c5(h) < _eval_c5(h_save):
            return h
    except Exception:
        pass
    h[:] = h_save
    return h_save

def _random_h(n, rng):
    h = rng.random(n)
    return _fix_sum(h, n)

def _interp_h(h_small, n_large):
    """Upsample by nearest-neighbor / linear interpolation."""
    n_small = len(h_small)
    x_small = np.arange(n_small)
    x_large = np.linspace(0, n_small - 1, n_large)
    h_large = np.interp(x_large, x_small, h_small)
    return _fix_sum(h_large, n_large)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    
    n = 80
    if h_init is not None:
        h_best = _fix_sum(h_init, len(h_init))
        n = len(h_best)
    else:
        h_best = _fix_sum(np.full(80, 0.5), 80)
        n = 80
    
    c5_best = _eval_c5(h_best)
    
    # Phase 1: SLSQP warm-start
    if time.time() - t0 < budget_s - 400:
        for alpha in [200, 500]:
            if time.time() - t0 > budget_s - 400:
                break
            h_new = _slsqp_refine(h_best, alpha, 200)
            c = _eval_c5(h_new)
            if c < c5_best:
                c5_best = c
                h_best = h_new
    
    # Phase 2: Pairwise random swaps (main phase)
    swap_deadline = t0 + budget_s * 0.45
    while time.time() < swap_deadline:
        i, j = rng.integers(0, n, size=2)
        i, j = int(i), int(j)
        if i == j:
            continue
        h_try = _swap_optimize(h_best.copy(), i, j)
        if h_try is not None:
            c = _eval_c5(h_try)
            if c < c5_best - 1e-12:
                c5_best = c
                h_best = h_try
    
    # Phase 3: Full scan passes
    scan_deadline = t0 + budget_s * 0.60
    while time.time() < scan_deadline:
        improved = False
        for i in range(n):
            if time.time() > scan_deadline:
                break
            for j in range(i + 1, n):
                h_try = _swap_optimize(h_best, i, j)
                if h_try is not None:
                    c = _eval_c5(h_try)
                    if c < c5_best - 1e-12:
                        c5_best = c
                        h_best = h_try.copy()
                        improved = True
        if not improved:
            break
    
    # Phase 4: SLSQP refinement
    if time.time() - t0 < budget_s - 100:
        h_new = _slsqp_refine(h_best, 500, 300)
        c = _eval_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new
    
    # Phase 5: More random swaps
    swap2_deadline = t0 + budget_s * 0.80
    while time.time() < swap2_deadline:
        i, j = rng.integers(0, n, size=2)
        i, j = int(i), int(j)
        if i == j:
            continue
        h_try = _swap_optimize(h_best.copy(), i, j)
        if h_try is not None:
            c = _eval_c5(h_try)
            if c < c5_best - 1e-12:
                c5_best = c
                h_best = h_try
    
    # Phase 6: NM on random subsets
    nm_deadline = t0 + budget_s * 0.90
    while time.time() < nm_deadline:
        subset_size = int(rng.integers(4, 9))
        coords = rng.choice(n, size=subset_size, replace=False).tolist()
        h_try = _nm_subspace(h_best.copy(), coords, max_iter=80)
        c = _eval_c5(h_try)
        if c < c5_best - 1e-12:
            c5_best = c
            h_best = h_try
    
    # Phase 7: Final SLSQP + swap polish
    if time.time() - t0 < budget_s - 20:
        h_new = _slsqp_refine(h_best, 800, 300)
        c = _eval_c5(h_new)
        if c < c5_best:
            c5_best = c
            h_best = h_new
    
    if time.time() - t0 < budget_s - 10:
        for _ in range(100):
            if time.time() > t0 + budget_s - 10:
                break
            i, j = rng.integers(0, n, size=2)
            i, j = int(i), int(j)
            if i == j:
                continue
            h_try = _swap_optimize(h_best.copy(), i, j)
            if h_try is not None:
                c = _eval_c5(h_try)
                if c < c5_best - 1e-12:
                    c5_best = c
                    h_best = h_try
    
    # Final validation
    h_best = _fix_sum(h_best, n)
    c5_best = _eval_c5(h_best)
    return h_best, c5_best, n
