    perturb_sizes = [3, 4, 5, 6, 8, 10, 12]
    idx = 0
    stagnation = 0
    while time.time() < t0 + time_limit - 5.0:
        remaining = t0 + time_limit - time.time()
        sa_dur = min(18.0, remaining * 0.12)
        if sa_dur < 5.0:
            break
        h_curr, c5_curr = sa_phase(h_bin, sa_dur, rng.randint(0, 2**31 - 1))
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 6:
            h_bin = frac_to_bin(best_h + 0.3 * rng.randn(n))
            stagnation = 0
        n_swaps = perturb_sizes[idx % len(perturb_sizes)]
        idx += 1
        h_pert = best_h.copy()
        ones_p = np.where(h_pert == 1.0)[0]
        zeros_p = np.where(h_pert == 0.0)[0]
        for _ in range(n_swaps):
            if len(ones_p) == 0 or len(zeros_p) == 0:
                break
            a = rng.randint(len(ones_p))
            b = rng.randint(len(zeros_p))
            h_pert[ones_p[a]] = 0.0
            h_pert[zeros_p[b]] = 1.0
            ones_p[a] = zeros_p[b]
            zeros_p[b] = ones_p[a]
        h_bin = h_pert
    return best_h, best_c5

def ga_binary(h_best, time_limit, seed, pop_size=40):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    target = n // 2
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    # Population
    pop = [best_h.copy()]
    for _ in range(pop_size - 1):
        h_try = best_h.copy()
        ones_p = np.where(h_try == 1.0)[0]
        zeros_p = np.where(h_try == 0.0)[0]
        for _ in range(rng.randint(5, 15)):
            a = rng.randint(len(ones_p))
            b = rng.randint(len(zeros_p))
            h_try[ones_p[a]] = 0.0
            h_try[zeros_p[b]] = 1.0
            ones_p[a] = zeros_p[b]
            zeros_p[b] = ones_p[a]
        pop.append(h_try)
    fits = [compute_c5(p) for p in pop]
    t0 = time.time()
    gen = 0
    while time.time() < t0 + time_limit - 3.0:
        # Sort by fitness
        order = np.argsort(fits)
        keep = order[:pop_size // 2]
        new_pop = [pop[keep[i]].copy() for i in range(len(keep))]
        fits_new = [fits[keep[i]] for i in range(len(keep))]
        # Generate offspring via swap mutation from top parents
        while len(new_pop) < pop_size and time.time() < t0 + time_limit - 3.0:
            parent = new_pop[rng.randint(len(new_pop))]
            child = parent.copy()
            ones_c = np.where(child == 1.0)[0]
            zeros_c = np.where(child == 0.0)[0]
            n_mut = 1 + rng.randint(0, 3)
            for _ in range(n_mut):
                if len(ones_c) == 0 or len(zeros_c) == 0:
                    break
                a = rng.randint(len(ones_c))
                b = rng.randint(len(zeros_c))
                child[ones_c[a]] = 0.0
                child[zeros_c[b]] = 1.0
                ones_c[a] = zeros_c[b]
                zeros_c[b] = ones_c[a]
            new_pop.append(child)
            fits_new.append(compute_c5(child))
        # Check improvement
        min_f = min(fits_new)
        min_idx = fits_new.index(min_f)
        if min_f < best_c5:
            best_c5 = min_f
            best_h = new_pop[min_idx].copy()
        pop = new_pop
        fits = fits_new
        gen += 1
    return best_h, best_c5

def quick_grad(h_init, dur, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t0 = time.time()
    lr = 0.05
    while time.time() < t0 + dur - 0.5:
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full')
        mc = corr.max()
        shifted = np.clip(50.0 * (corr - mc), -400, 400)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-14:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
            else:
                L = -lag
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
        grad *= dx
        v = 0.8 * v - lr * grad
        h = project_feasible(h + v, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr = max(lr * 0.999, 0.005)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = np.full(70, 0.5)
    n = len(h_init)
    target = n / 2.0
    h_init = project_feasible(h_init, target)
    best_h = frac_to_bin(h_init)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick gradient to improve fractional start
    t_rem = deadline - time.time()
    if t_rem > 100:
        h_frac1, c5_g = quick_grad(h_init, min(50, t_rem * 0.06), seed)
        h_bin1 = frac_to_bin(h_frac1)
        c5_b1 = compute_c5(h_bin1)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_bin1
        # Keep the fractional for ILS seeding
        h_for_ils = h_frac1
    else:
        h_for_ils = h_init

    # Phase 2: Long ILS (main workhorse)
    t_rem = deadline - time.time()
    if t_rem > 40:
        ils_dur = min(550, t_rem * 0.65)
        h_ils, c5_ils = ils_binary(h_for_ils, ils_dur, seed + 100)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils

    # Phase 3: Try second ILS with different perturb from best
    t_rem = deadline - time.time()
    if t_rem > 60:
        rng2 = np.random.RandomState(seed + 500)
        h_seed2 = frac_to_bin(best_h + 0.4 * rng2.randn(n))
        ils_dur2 = min(200, t_rem * 0.45)
        h_ils2, c5_ils2 = ils_binary(h_seed2, ils_dur2, seed + 600)
        if c5_ils2 < best_c5:
            best_c5 = c5_ils2
            best_h = h_ils2

    # Phase 4: GA for diversity
    t_rem = deadline - time.time()
    if t_rem > 80:
        ga_dur = min(200, t_rem * 0.5)
        h_ga, c5_ga = ga_binary(best_h, ga_dur, seed + 1000, pop_size=50)
        if c5_ga < best_c5:
            best_c5 = c5_ga
            best_h = h_ga

    # Phase 5: Final SA polish
    t_rem = deadline - time.time()
    if t_rem > 15:
        h_pol, c5_pol = sa_phase(best_h, min(60, t_rem - 3), seed + 9000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Final verification
    best_c5 = compute_c5(best_h)
    # Ensure binary
    best_h = (best_h >= 0.5).astype(float)
    # Fix sum if needed
    s = int(best_h.sum())
    target_int = n // 2
    if s != target_int:
        if s < target_int:
            zeros_idx = np.where(best_h == 0)[0]
            for k in zeros_idx[:target_int - s]:
                best_h[k] = 1.0
        else:
            ones_idx = np.where(best_h == 1)[0]
            for k in ones_idx[:s - target_int]:
                best_h[k] = 0.0
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
