import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        target = float(n) / 2.0
        if abs(s - target) < 1e-14:
            break
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def _sum_constraint_jac(h):
    return np.ones(len(h))

_g_n = 160
_g_temp = 0.01

def _lse_obj(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    t = max(_g_temp, 1e-12)
    return m + t * np.log(np.sum(np.exp((corr - m) / t)))

def _slsqp(h_init, n, max_time, seed=0):
    global _g_n, _g_temp
    t0 = time.time()
    _g_n = n
    best = h_init.copy()
    best_val = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_constraint_jac},)
    
    temps = [0.03, 0.01, 0.003, 0.001, 0.0003, 0.0001, 0.00003, 0.00001]
    for temp in temps:
        if time.time() - t0 > max_time - 3:
            break
        _g_temp = temp
        h_cur = best.copy()
        for _ in range(3):
            if time.time() - t0 > max_time - 1:
                break

                             bounds=bounds, constraints=cons,

                h_new = _project(res.x, n)
                v = compute_c5(h_new)
                if v < best_val - 1e-12:
                    best = h_new.copy()
                    best_val = v
                h_cur = h_new.copy()

                break
    return best, best_val

    rng = np.random.RandomState(seed)

    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

    """Two blocks of 1s separated by gaps."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2

    pos2 = (pos + half + int(n * ratio2)) % n

    return h

    best = h.copy()

        k_star = np.argmax(corr)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        tgt = float(n) / 2.0
        if abs(s - tgt) < 1e-14:
            break
        h = h * tgt / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _sum_eq(h):
    return np.sum(h) - len(h) / 2.0

def _sum_eq_jac(h):
    return np.ones(len(h))

_g_n = 160
_g_temp = 0.01

def _lse_obj(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    t = max(_g_temp, 1e-13)
    shifted = (corr - m) / t
    shifted = np.clip(shifted, -500, 500)
    return m + t * np.log(np.sum(np.exp(shifted)))

def _slsqp(h_init, n, max_time, seed=0):
    global _g_n, _g_temp
    t0 = time.time()
    _g_n = n
    best = h_init.copy()
    best_val = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac},)

    temps = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001, 0.00003]
    for temp in temps:
        if time.time() - t0 > max_time - 3:
            break
        _g_temp = temp
        h_cur = best.copy()
        n_iters = 4 if temp > 0.01 else 5
        for _ in range(n_iters):
            if time.time() - t0 > max_time - 1:
                break
            try:
                res = minimize(_lse_obj, h_cur, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 600, 'ftol': 1e-15})
                h_new = _project(res.x, n)
                v = compute_c5(h_new)
                if v < best_val - 1e-13:
                    best = h_new.copy()
                    best_val = v
                h_cur = h_new.copy()
            except Exception:
                break
    return best, best_val

def _perturb(h, scale, s):
    rng = np.random.RandomState(s)
    h_p = h + scale * rng.randn(len(h))
    return _project(h_p, len(h))

def _binary_init(n, s):
    rng = np.random.RandomState(s)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _block_init(n, s, n_blocks):
    rng = np.random.RandomState(s)
    h = np.zeros(n)
    n_ones = n // 2
    block_sizes = np.full(n_blocks, n_ones // n_blocks)
    rem = n_ones - n_ones // n_blocks * n_blocks
    for i in range(rem):
        block_sizes[i] += 1
    used = []
    for bs in block_sizes:
        start = rng.choice(n, 1, replace=False)[0]
        for i in range(bs):
            idx = (start + i) % n
            if idx not in used:
                h[idx] = 1.0
                used.append(idx)
    return _project(h, n)

def _grad_project(h, n, lr, n_steps):
    best = h.copy()
    best_val = compute_c5(best)
    eps = 1e-6
    for _ in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        grad = np.zeros(n)
        for j in range(n):
            h_p = h.copy()
            h_p[j] += eps
            cp = np.correlate(h_p, 1.0 - h_p, mode="full")[k_star]
            h_m = h.copy()
            h_m[j] -= eps
            cm = np.correlate(h_m, 1.0 - h_m, mode="full")[k_star]
            grad[j] = (cp - cm) / (2.0 * eps)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad = grad / gnorm
        h_new = _project(h - lr * grad, n)
        v = compute_c5(h_new)
        if v < best_val - 1e-14:
            best = h_new.copy()
            best_val = v
            h = h_new.copy()
        else:
            lr *= 0.7
    return best, best_val

def _coord_search(h, n, n_rounds, s):
    rng = np.random.RandomState(s)
    best = h.copy()
    best_val = compute_c5(best)
    for _ in range(n_rounds):
        idx = rng.randint(n)
        delta = 0.01 * rng.choice([-1.0, 1.0])
        h_p = best.copy()
        h_p[idx] += delta
        h_p = _project(h_p, n)
        v = compute_c5(h_p)
        if v < best_val - 1e-14:
            best = h_p
            best_val = v
    return best, best_val

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def _rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    else:
        n0 = 80
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize initial construction
    if _rem() > 120:
        h_opt, c5_opt = _slsqp(h0.copy(), n0, min(220, _rem() - 60), seed)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Try multiple n values
    n_candidates = sorted(set([64, 80, 96, 120, 144, 160, n0]))
    for n_try in n_candidates:
        if _rem() < 70:
            break
        if n_try == n0:
            continue
        x_old = np.arange(n0, dtype=float) * 2.0 / n0
        x_new = np.arange(n_try, dtype=float) * 2.0 / n_try
        h_init = _project(np.interp(x_new, x_old, best_h), n_try)
        h_opt, c5_opt = _slsqp(h_init, n_try, min(100, _rem() - 40), seed + n_try)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            n0 = n_try

    # Phase 3: Random restarts at best resolution
    n_cur = len(best_h)
    nr = 0
    while _rem() > 50 and nr < 40:
        scale = 0.03 + 0.06 * (nr % 7) / 6.0
        h_pert = _perturb(best_h, scale, seed + 200 + nr)
        mt = min(25, _rem() - 25)
        if mt < 12:
            break
        h_opt, c5_opt = _slsqp(h_pert, n_cur, mt, seed + 200 + nr)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        nr += 1

    # Phase 4: Structured initializations at current n
    for bi in range(4):
        if _rem() < 40:
            break
        if bi == 0:
            h_init = _binary_init(n_cur, seed + 500 + bi)
        elif bi == 1:
            h_init = _block_init(n_cur, seed + 500 + bi, 3)
        elif bi == 2:
            h_init = _block_init(n_cur, seed + 500 + bi, 5)
        else:
            h_init = _block_init(n_cur, seed + 500 + bi, 2)
        h_opt, c5_opt = _slsqp(h_init, n_cur, min(40, _rem() - 20), seed + 500 + bi)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 5: Gradient projection polish
    if _rem() > 15:
        h_opt, c5_opt = _grad_project(best_h, n_cur, lr=0.005, n_steps=15)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 6: Coordinate search fine-tuning
    if _rem() > 10:
        h_opt, c5_opt = _coord_search(best_h, n_cur, int(min(500, _rem() * 20)), seed + 900)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 7: Final SLSQP polish
    if _rem() > 20:
        h_opt, c5_opt = _slsqp(best_h.copy(), n_cur, max(10, _rem() - 5), seed + 9999)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final coordinate search
    if _rem() > 5:

        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    return best_h, best_c5, len(best_h)
