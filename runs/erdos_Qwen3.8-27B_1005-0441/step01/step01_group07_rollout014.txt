def compute_c5(h):

    return float(corr.max() * dx)

def project_binary(h, target_ones):
    """Project to binary with exactly target_ones ones."""
    h = np.array(h, dtype=float)
    ones = int(np.sum(h >= 0.5))
    if ones == target_ones:
        return (h >= 0.5).astype(float)
    # Adjust
    if ones > target_ones:
        # Remove some ones (those with lowest values)
        idx = np.argsort(h)[:ones - target_ones]
        h[idx] = 0
    else:
        # Add some ones (those with highest values among zeros)
        zero_idx = np.where(h < 0.5)[0]
        idx = zero_idx[np.argsort(h[zero_idx])[::-1][:target_ones - ones]]
        h[idx] = 1
    return (h >= 0.5).astype(float)

def simulated_annealing(h, time_budget, seed, T_initial=0.1, T_final=0.001):

    while time.time() < deadline - 0.1:
        # Temperature schedule
        elapsed = time.time() - t0
        frac = min(elapsed / total_time, 1.0)
        T = T_initial * (T_final / T_initial) ** frac
        
        # Random move: swap a 1 and a 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]

        zero_mask = h_bin == 0

        order = np.argsort(h_frac[candidates])[::-1][:diff]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def ensure_binary(h, target_ones):
    """Ensure h is binary with exactly target_ones ones."""
    n = len(h)
    h = (h >= 0.5).astype(float)
    cur = int(h.sum())
    if cur == target_ones:
        return h
    if cur > target_ones:
        # Remove (cur - target_ones) ones: choose positions with lowest "importance"
        # Just remove random ones for simplicity
        ones_idx = np.where(h == 1.0)[0]
        rng = np.random.RandomState(0)
        to_remove = rng.choice(ones_idx, cur - target_ones, replace=False)
        h[to_remove] = 0.0
    else:
        zeros_idx = np.where(h == 0.0)[0]
        rng = np.random.RandomState(0)
        to_add = rng.choice(zeros_idx, target_ones - cur, replace=False)
        h[to_add] = 1.0
    return h

def sa_optimize(h, time_budget, seed):
    """Simulated annealing for binary h."""
    rng = np.random.RandomState(seed)
    n = len(h)
    target_ones = n // 2
    h = ensure_binary(h, target_ones)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    t0 = time.time()
    deadline = t0 + time_budget
    T = 0.03
    cool = 0.99995
    
    while time.time() < deadline - 0.05:
        T = max(0.0003, T * cool)
        
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)

            if d < 0 or rng.random() < np.exp(-d / T):

                else:
                    h[i1], h[i2], h[j1], h[j2] = 1.0, 1.0, 0.0, 0.0

def structured_binary(n, seed):
    """Generate structured binary sequences."""
    rng = np.random.RandomState(seed)
    target = n // 2
    h = np.zeros(n)
    # Try a "Golay-like" construction: alternate with perturbations

        else:
            # Find nearest available
            for d in range(1, n):
                p2 = (pos + d) % n
                if h[p2] == 0:

    elif cur > target:
