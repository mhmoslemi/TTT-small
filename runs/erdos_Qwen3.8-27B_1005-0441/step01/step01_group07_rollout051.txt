def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_smooth(h, dx, tau, top_m):
    n = len(h)
    omh = 1.0 - h
    corr = compute_full_corr(h)
    if tau > 0:
        mc = corr.max()
        shifted = np.clip(tau * (corr - mc), -500, 500)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
    else:
        idx = np.argsort(corr)[::-1][:top_m]
        weights = np.zeros(len(corr))
        weights[idx] = 1.0 / top_m
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid

    n = len(h)

def binary_fast_search(h_init, time_budget, seed, mode='sa'):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n // 2
    h = (h_init >= 0.5).astype(float)
    # Fix sum
    diff = target_sum - int(h.sum())
    if diff > 0:
        idx = np.argsort(h)[:diff]
        h[idx] = 1.0
    elif diff < 0:
        idx = np.argsort(h)[-diff:]
        h[idx] = 0.0
    
    corr = compute_full_corr(h)
    dx = 2.0 / n
    best_c5 = corr.max() * dx
    best_h = h.copy()
    
    T = 0.03
    T_floor = 0.0005
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    max_no_improve = 50000  # restart threshold
    
    while time.time() < deadline - 0.3:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Apply swap
        do_swap(corr, h, i, j)
        c5_new = corr.max() * dx
        
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            if mode == 'sa':
                delta = c5_new - best_c5
                if rng.rand() < math.exp(-delta / T):
                    no_improve += 1
                else:
                    undo_swap(corr, h, i, j)
                    no_improve += 1
            else:
                undo_swap(corr, h, i, j)
                no_improve += 1
        
        if mode == 'sa':
            T = max(T_floor, T * 0.99995)
        
        if no_improve > max_no_improve:
            # Restart with perturbation
            for _ in range(10 + rng.randint(20)):
                ones2 = np.where(h == 1.0)[0]
                zeros2 = np.where(h == 0.0)[0]
                a = ones2[rng.randint(len(ones2))]
                b = zeros2[rng.randint(len(zeros2))]
                do_swap(corr, h, a, b)
            best_h = h.copy()
            best_c5 = corr.max() * dx
            no_improve = 0
            T = 0.03
    
    return best_h, best_c5
