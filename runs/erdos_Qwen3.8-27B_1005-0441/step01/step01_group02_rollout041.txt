    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, h0, jac=_slsqp_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _project(res.x)
    except Exception:
        return _project(h0.copy())

def _pgd_worst(h, n_iter, t_limit):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = compute_c5(h)
    step = 0.04
    beta = 25.0
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > t_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mc = corr.max()
        w = np.exp(beta * (corr - mc))
        ws = w.sum()
        if ws < 1e-300:
            break
        w /= ws
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if w[k_idx] < 1e-12:
                continue
            d = (n - 1) - k_idx
            grad += w[k_idx] * _vec_grad(h, d)
        h_new = _project(h - step * grad)
        c_new = compute_c5(h_new)
        if c_new < best - 1e-15:
            best = c_new
            h = h_new
            step = min(step * 1.03, 0.12)
        else:
            step *= 0.65
            if step < 1e-6:
                step = 0.04
                beta = min(beta * 1.3, 80.0)
    return h, best

def _coord_descent(h, max_rounds, rng, t_limit):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    for rnd in range(max_rounds):
        if time.time() - t0 > t_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - t0 > t_limit:
                break
            j = (i + 1) % n
            for delta in [0.002, 0.005, 0.01, -0.002, -0.005, -0.01]:
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                if 0.0 <= h_try[i] <= 1.0 and 0.0 <= h_try[j] <= 1.0:
                    c = compute_c5(h_try)
                    if c < best - 1e-14:
                        h = h_try
                        best = c
                        improved = True
                        break
            if improved:
                break
        if not improved:
            # Random swaps
            any_imp = False
            for _ in range(20):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for delta in [0.005, 0.01, -0.005, -0.01]:
                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    if 0.0 <= h_try[i] <= 1.0 and 0.0 <= h_try[j] <= 1.0:
                        c = compute_c5(h_try)
                        if c < best - 1e-14:
                            h = h_try
                            best = c
                            any_imp = True
                            break
                if any_imp:
                    break
            if not any_imp:
                break
    return h, best

def _nm_polish(h, t_limit):
    n = len(h)
    t0 = time.time()

    def nm_obj(x):
        return compute_c5(_project(x))

    maxiter = min(800, int(200 * max(0, t_limit - (time.time() - t0))))
    maxiter = max(maxiter, 50)
    try:
        res = minimize(nm_obj, h.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-12})
        return _project(res.x), compute_c5(_project(res.x))
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        n_fallback = 70
        h_init = np.zeros(n_fallback)
        h_init[:n_fallback // 2] = 1.0

    h_init = _project(h_init.copy())
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP from initial
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 80:
        h1 = _slsqp_opt(best_h, 20.0, 250)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: PGD
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 60:
        h2, c2 = _pgd_worst(best_h, 3000, min(180, t_rem - 10))
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Coordinate descent
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 40:
        h3, c3 = _coord_descent(best_h, 8000, min(150, t_rem - 5), rng)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: NM polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        h4, c4 = _nm_polish(best_h, min(80, t_rem - 5))
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Phase 5: Restart loops with perturbation
    t_rem = budget_s - (time.time() - t_start)
    scales = [0.06, 0.10, 0.04, 0.14]
    ri = 0
    while t_rem > 90 and ri < len(scales):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 90:
            break
        scale = scales[ri]
        h_p = _project(best_h + scale * rng.randn(n))
        t_sub = min(70, t_rem - 15)
        # Quick SLSQP
        if t_sub > 25:
            h_p = _slsqp_opt(h_p, 30.0, 120)
        # PGD
        t_sub = min(40, t_sub - (time.time() - t_start) if False else t_sub)
        t_rem2 = budget_s - (time.time() - t_start)
        if t_rem2 > 35:
            h_p, cp = _pgd_worst(h_p, 1500, min(35, t_rem2 - 10))
        # CD
        t_rem2 = budget_s - (time.time() - t_start)
        if t_rem2 > 25:
            h_p, cp = _coord_descent(h_p, 4000, min(20, t_rem2 - 5), rng)
        h_p = _project(h_p)
        cp = compute_c5(h_p)
        if cp < best_c5:
            best_c5, best_h = cp, h_p
        ri += 1

    # Phase 6: Final SLSQP + CD
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 25:
        h6 = _slsqp_opt(best_h, 50.0, 150)
        c6 = compute_c5(h6)
        if c6 < best_c5:
            best_c5, best_h = c6, h6

    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 10:
        h7, c7 = _coord_descent(best_h, 5000, min(30, t_rem - 3), rng)
        if c7 < best_c5:
            best_c5, best_h = c7, h7

    best_h = _project(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
