from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _softmax_val(corr, alpha):
    cmax = np.max(corr)
    s = np.sum(np.exp(alpha * (corr - cmax)))
    return cmax + np.log(s + 1e-30) / alpha

def _penalty_obj(x, alpha, lam):
    h = np.clip(x, 0.0, 1.0)
    corr = _corr_array(h)
    smax = _softmax_val(corr, alpha)
    se = np.sum(h) - len(x) / 2.0
    return smax + lam * se * se

def _slsqp_obj(x, alpha):
    h = np.clip(x, 0.0, 1.0)
    corr = _corr_array(h)
    return _softmax_val(corr, alpha)

def _slsqp_eq(x):
    return np.sum(x) - len(x) / 2.0

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            i = int(np.argmax(room))
            add = min(diff, room[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h

def _opt_lbfgs(h0, alpha, lam, max_iter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_penalty_obj, h0, args=(alpha, lam), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
        return res.x
    except Exception:
        return h0.copy()

def _opt_slsqp(h0, alpha, max_iter, ftol):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, h0, args=(alpha,), method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _slsqp_eq}],
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0.copy()

def _rand_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _relax_binary(hb, t, rng):
    h = (1.0 - t) * hb + t * rng.random(len(hb))
    return _fix_sum(h, len(hb))

def _rand_cont(n, rng):
    return _fix_sum(rng.random(n), n)

def _block_h(n, rng):
    nb = int(rng.integers(3, max(4, n // 10)))
    h = np.zeros(n)
    for b in range(nb):
        w = n // nb
        s, e = b * w, min((b + 1) * w, n)
        h[s:e] = rng.random()
    return _fix_sum(h, n)

def _sin_h(n, freq, rng):
    x = np.linspace(0, 2.0 * np.pi * freq, n, endpoint=False)
    amp = rng.uniform(0.2, 0.45)
    ph = rng.uniform(0, 2.0 * np.pi)
    h = 0.5 + amp * np.sin(x + ph)
    return _fix_sum(h, n)

def _saw_h(n, freq, rng):
    x = np.linspace(0, freq, n, endpoint=False)
    h = 1.0 - 2.0 * np.abs(x - np.round(x))
    h = (h - h.min()) / (h.max() - h.min() + 1e-15)
    h = 0.5 + rng.uniform(0.2, 0.4) * (2.0 * h - 1.0)
    return _fix_sum(h, n)

def _interp_resample(h, new_n):
    old_n = len(h)
    h_res = np.interp(np.linspace(0, 1, new_n, endpoint=False),
                      np.linspace(0, 1, old_n, endpoint=False), h)
    return _fix_sum(h_res, new_n)
