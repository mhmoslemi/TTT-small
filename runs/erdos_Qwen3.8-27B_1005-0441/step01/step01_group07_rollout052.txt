import numpy as np
import time, math

def project_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} via bisection on shift."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binary_heat(h, target_ones):
    """Binarize h, ensuring exactly target_ones ones."""
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target_ones:
        # Remove excess: set largest fractional h values to 0
        order = np.argsort(-h)
        for i in order:
            if h_bin[i] == 1 and n_ones > target_ones:
                h_bin[i] = 0
                n_ones -= 1
    elif n_ones < target_ones:

        order = np.argsort(-h)
        for i in order:

def continuous_optimize(h_init, time_budget, seed, beta=0.8, lr_init=0.05, tau_init=50.0, tau_max=500.0):
    """Continuous gradient-based optimization with softmax-smoothed max."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr_init
    tau = tau_init
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    
    while time.time() < deadline - 0.3:
        corr = compute_corr(h)
        mc = corr.max()
        # Softmax weights on the top overlaps
        shifted = np.clip((corr - mc) * tau, -100, 100)
        exps = np.exp(shifted)
        wsum = exps.sum()
        if wsum < 1e-30:
            break
        weights = exps / wsum
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-12:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
            else:
                L = -lag
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
        grad *= dx
        
        # Nesterov-like update
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        tau = min(tau * 1.003, tau_max)
        if no_improve > 200:
            lr = max(lr * 0.8, 0.001)
            tau = tau_init
        if no_improve > 500:
            # Restart
            h = project_feasible(best_h + rng.randn(n) * 0.03, target_sum)
            v[:] = 0
            lr = lr_init
            tau = tau_init
            no_improve = 0
    
    return best_h, best_c5

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binarize(h, target):
    order = np.argsort(-h)
    out = np.zeros(len(h))
    out[order[:target]] = 1.0
    return out

def binary_steepest_descent(h_bin, max_rounds=200):
    n = len(h_bin)
    h = h_bin.copy()
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    corr = compute_corr(h)
    c5 = float(corr.max() * 2.0 / n)
    for _ in range(max_rounds):
        best_delta = -0.5
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = float(compute_corr(h).max() * 2.0 / n)
                d = c - c5
                if d < best_delta:
                    best_delta = d
                    best_i, best_j = i, j
                h[i] = 1.0
                h[j] = 0.0
        if best_i < 0 or best_delta >= 0:
            break
        ones[ones.index(best_i)] = best_j
        zeros[zeros.index(best_j)] = best_i
        h[best_i] = 0.0
        h[best_j] = 1.0
        c5 += best_delta
    return h, c5

def sa_binary(h_init, time_budget, seed, T_init=0.012, T_min=0.0004, cool=0.9992):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    c5 = float(compute_corr(h).max() * 2.0 / n)
    best_h = h.copy()
    best_c5 = c5
    T = T_init
    t0 = time.time()
    deadline = t0 + time_budget
    stall = 0
    while time.time() < deadline - 0.3:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5_new = float(compute_corr(h).max() * 2.0 / n)
        delta = c5_new - c5
        if delta <= 0 or rng.rand() < math.exp(-delta / max(T, 1e-7)):
            c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        T = max(T * cool, T_min)
        if stall > 3000:
            T = T_init * 0.4
            stall = 0
    return best_h, best_c5

def grad_optimize(h_init, time_budget, seed, lr=0.035, beta=0.82, tau=100.0, tau_max=600.0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t0 = time.time()
    deadline = t0 + time_budget
    no_imp = 0
    while time.time() < deadline - 0.2:
        corr = compute_corr(h)
        mc = corr.max()
        shifted = np.clip((corr - mc) * tau, -70, 70)
        exps = np.exp(shifted)
        ws = exps.sum()
        weights = exps / ws
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-10:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
            else:
                L = -lag
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
        grad *= dx
        v = beta * v - lr * grad
        h = project_feasible(h + v, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            no_imp = 0
        else:
            no_imp += 1
        tau = min(tau * 1.003, tau_max)
        if no_imp > 120:
            lr = max(lr * 0.82, 0.002)
        if no_imp > 500:
            h = project_feasible(best_h + rng.randn(n) * 0.025, target)
            v[:] = 0.0
            lr = 0.035
            tau = 100.0
            no_imp = 0
    return best_h, best_c5

def multi_target_grad(h_init, time_budget, seed, M=3, lr=0.03, beta=0.8):
    """Focus gradient on top-M shifts only."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t0 = time.time()
    deadline = t0 + time_budget
    no_imp = 0
    while time.time() < deadline - 0.2:
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:M]
        weights = np.zeros(len(corr))
        for idx in top_idx:
            weights[idx] = 1.0 / M
        grad = np.zeros(n)
        for k in top_idx:
            w = weights[k]
            lag = k - (n - 1)
            if lag >= 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
            else:
                L = -lag
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
        grad *= dx
        v = beta * v - lr * grad
        h = project_feasible(h + v, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 200:
            lr = max(lr * 0.7, 0.001)
        if no_imp > 600:
            h = project_feasible(best_h + rng.randn(n) * 0.02, target)
            v[:] = 0.0
            lr = 0.03
            no_imp = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    n = 70
    target = 35
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float).copy()
        n = len(h_init)
        target = n // 2
    else:
        h_init = np.zeros(n)
        h_init[:target] = 1.0
    best_h = project_feasible(h_init, float(target))
    best_c5 = compute_c5(best_h)

    # Phase 1: Binarize + steepest descent + SA
    h_bin = binarize(best_h, target)
    t_rem = deadline - time.time()
    if t_rem > 30:
        h_sd, c5_sd = binary_steepest_descent(h_bin, 500)
        if c5_sd < best_c5:
            best_c5 = c5_sd
            h_bin = h_sd
            best_h = project_feasible(h_bin, float(target))
    t_rem = deadline - time.time()
    if t_rem > 30:
        t1 = min(t_rem * 0.25, 200)
        hb1, c5b1 = sa_binary(h_bin, t1, seed, T_init=0.01, cool=0.9993)
        if c5b1 < best_c5:
            best_c5 = c5b1
            h_bin = hb1
            best_h = project_feasible(hb1, float(target))

    # Phase 2: Continuous refinement (softmax)
    t_rem = deadline - time.time()
    if t_rem > 30:
        t2 = min(t_rem * 0.25, 250)
        hc1, c5c1 = grad_optimize(best_h, t2, seed + 1, lr=0.03, tau=120.0)
        if c5c1 < best_c5:
            best_h, best_c5 = hc1, c5c1

    # Phase 3: Multi-target gradient
    t_rem = deadline - time.time()
    if t_rem > 30:
        t3 = min(t_rem * 0.2, 180)
        hc2, c5c2 = multi_target_grad(best_h, t3, seed + 50, M=3, lr=0.025)
        if c5c2 < best_c5:
            best_h, best_c5 = hc2, c5c2

    # Phase 4: Binarize + SA + steepest again
    t_rem = deadline - time.time()
    if t_rem > 30:
        h_bin2 = binarize(best_h, target)
        h_sd2, c5_sd2 = binary_steepest_descent(h_bin2, 300)
        t4 = min(t_rem * 0.15, 120)
        hb2, c5b2 = sa_binary(h_sd2, t4, seed + 200, T_init=0.008, cool=0.9995)
        h_sd3, c5_sd3 = binary_steepest_descent(hb2, 300)
        c5_best_bin = min(c5_sd2, c5b2, c5_sd3)
        h_best_bin = h_sd2 if c5_sd2 == c5_best_bin else (hb2 if c5b2 == c5_best_bin else h_sd3)
        if c5_best_bin < best_c5:
            best_c5 = c5_best_bin
            best_h = project_feasible(h_best_bin, float(target))

    # Phase 5: Final continuous polish
    t_rem = deadline - time.time()
    if t_rem > 20:
        t5 = min(t_rem * 0.3, 200)
        hc3, c5c3 = grad_optimize(best_h, t5, seed + 500, lr=0.02, tau=150.0, tau_max=400.0)
        if c5c3 < best_c5:
            best_h, best_c5 = hc3, c5c3

    # Phase 6: Diverse restarts
    for trial in range(4):
        t_rem = deadline - time.time() - 10
        if t_rem < 40:
            break
        rng_t = np.random.RandomState(seed + 7700 + trial * 193)
        h_try = best_h + rng_t.randn(n) * 0.08
        h_try = project_feasible(h_try, float(target))
        h_try_bin = binarize(h_try, target)
        t_alloc_sa = min(t_rem * 0.3, 80)
        t_alloc_gr = min(t_rem * 0.35, 80)
        hb, c5hb = sa_binary(h_try_bin, t_alloc_sa, seed + 12000 + trial * 11)
        hb_sd, c5hb_sd = binary_steepest_descent(hb, 200)
        h_cont = project_feasible(hb_sd, float(target))
        hc, c5hc = grad_optimize(h_cont, t_alloc_gr, seed + 13000 + trial * 11, lr=0.025, tau=130.0)
        c5cand = min(c5hb, c5hb_sd, c5hc)
        if c5hb == c5cand:
            if c5canc < best_c5:
                best_c5 = c5canc
                best_h = project_feasible(hb, float(target))
        elif c5hb_sd == c5cand:
            if c5canc < best_c5:
                best_c5 = c5canc
                best_h = project_feasible(hb_sd, float(target))
        else:
            if c5canc < best_c5:
                best_h, best_c5 = hc, c5canc

    # Phase 7: Very tight final polish
    t_rem = deadline - time.time() - 1
    if t_rem > 3:
        hp, c5p = grad_optimize(best_h, min(t_rem, 25), seed + 99999,
                                lr=0.006, beta=0.85, tau=400.0, tau_max=1000.0)
        if c5p < best_c5:
            best_h, best_c5 = hp, c5p

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
