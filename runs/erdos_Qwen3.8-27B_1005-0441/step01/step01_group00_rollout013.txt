import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[:n-m] += wk * (1.0 - h[m:])
            g[m:] -= wk * h[:n-m]
        else:
            p = -m
            g[p:] += wk * (1.0 - h[:n-p])
            g[:n-p] -= wk * h[p:]
    return g * dx

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def _pgd(h0, n, eps, max_iter, lr_init, t_deadline):
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_f = _smoobj(h, eps)
    f_prev = best_f
    stall = 0
    for it in range(max_iter):
        if time.time() > t_deadline:
            break
        g = _smograd(h, eps)
        alpha = lr
        improved = False
        for _ in range(15):
            h_new = _proj(h - alpha * g, n)
            f_new = _smoobj(h_new, eps)
            if f_new <= f_prev - 1e-5 * alpha * np.dot(g, h - h_new):
                h = h_new
                f_prev = f_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            h = _proj(h - alpha * g, n)
            f_prev = _smoobj(h, eps)
        if f_prev < best_f - 1e-14:
            best_f = f_prev
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall == 100:
                lr *= 0.7
                stall = 0
            elif stall == 300:
                lr *= 0.5
                stall = 0
            if lr < 1e-12:
                break
        if it % 200 == 0:
            lr *= 0.98
    c5b = _c5(best_h)
    c5c = _c5(_proj(h, n))
    if c5c < c5b:
        best_h = _proj(h, n)
        c5b = c5c
    return best_h, c5b

def _slsqp_step(h0, n, eps, maxiter, t_deadline):
    bounds = [(0.0, 1.0)] * n
    def _ef(x): return x.sum() - n * 0.5
    def _ej(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': _ef, 'jac': _ej}]

        res = minimize(_smoobj, h0.copy(), args=(eps,),

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return _proj(res.x, n)
    except:
        return _proj(h0, n)
