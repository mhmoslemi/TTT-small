import time

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            if 1.0 - h[i] < 1e-15:
                break
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmin(h))
            if h[i] < 1e-15:
                break
            h[i] = max(0.0, h[i] + diff)
    return h

def _sa_objective(h, n):
    # Returns C5 with penalty for constraint violation
    penalty = 50.0 * (np.sum(h) - n / 2.0) ** 2
    return _eval_c5(h) + penalty

def _sa_move(h, rng, delta, n):
    # Random mass transfer: move delta from j to i
    i = rng.integers(n)
    j = rng.integers(n)
    if i == j:
        return None
    if h[j] < delta or h[i] + delta > 1.0:
        return None
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def _sa_run(h_init, rng, n, t_start, iters=5000):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    T = 0.01
    T_end = 0.0001
    cooling = (T_end / T) ** (1.0 / iters)

    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = np.inf

    def _tl():
        return t0 + budget_s

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None:
        h_init = _fix_sum(h_init, len(h_init))
        c5 = _eval_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5

    candidates = []
    if best_h is not None:
        candidates.append((best_h.copy(), best_c5))

    for n in [50, 60, 70, 80]:
        if time.time() > _tl() - 200:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            de_res = differential_evolution(
                _de_obj, bounds, seed=int(rng.integers(100000)),
                maxiter=100, popsize=25, tol=1e-10,
                mutation=(0.4, 0.9), recombination=0.7
            )
            h_de = _fix_sum(de_res.x, n)
            c5_de = _eval_c5(h_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
            candidates.append((h_de, c5_de))
        except Exception:
            pass

    candidates.sort(key=lambda x: x[1])
    candidates = candidates[:8]

    seen_seasons = set()
    for h_cand, c5_cand in candidates:
        if time.time() > _tl() - 80:
            break
        n = len(h_cand)
        key = (n, tuple(np.round(h_cand, 3)))
        if key in seen_seasons:
            continue
        seen_seasons.add(key)
        max_rnd = 60 if n <= 60 else 35
        h_ref, c5_ref = _interchange_refine(h_cand, n, max_rnd, _tl() - 60)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    for n in [50, 60, 70, 80]:
        if time.time() > _tl() - 100:
            break
        for pat in range(4):
            if time.time() > _tl() - 60:
                break
            h0 = _pattern_start(n, pat)
            h0 = _fix_sum(h0 + rng.normal(0, 0.05, n), n)
            max_rnd = 40 if n <= 60 else 25
            h_ref, c5_ref = _interchange_refine(h0, n, max_rnd, _tl() - 50)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
        if time.time() > _tl() - 60:
            break
        h0 = _random_start(n, rng)
        max_rnd = 40 if n <= 60 else 25
        h_ref, c5_ref = _interchange_refine(h0, n, max_rnd, _tl() - 50)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    if best_h is not None and time.time() < _tl() - 50:
        n = len(best_h)
        h_nml = _nm_refine(best_h, n, 300, _tl() - 40)
        c5_nm = _eval_c5(h_nml)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nml.copy()

    if best_h is not None and time.time() < _tl() - 30:
        n = len(best_h)
        h_sl = _slsqp_refine(best_h, n, _soft_obj_a, 300, _tl() - 25)
        c5_sl = _eval_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    if best_h is not None and time.time() < _tl() - 20:
        n = len(best_h)
        h_sl2 = _slsqp_refine(best_h, n, _soft_obj_b, 200, _tl() - 15)
        c5_sl2 = _eval_c5(h_sl2)
        if c5_sl2 < best_c5:
            best_c5 = c5_sl2
            best_h = h_sl2.copy()

    if best_h is not None and time.time() < _tl() - 10:
        n = len(best_h)
        h_final, c5_final = _interchange_refine(best_h, n, 15, _tl() - 5)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if best_h is None:
        best_h = _fix_sum(np.full(60, 0.5), 60)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
