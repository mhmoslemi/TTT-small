def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -400.0, 400.0)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    fval = (cmax + np.log(exp_s.sum()) / alpha) * dx
    m_idx = np.arange(2 * n - 1) - (n - 1)
    w = p * dx
    grad = np.zeros(n)
