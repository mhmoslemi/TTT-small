import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        nf = np.sum(free)
        if nf == 0:
            break
        h[free] += d / nf
        np.clip(h, 0.0, 1.0, out=h)
    return h

def corr_grad(h, k):
    """Gradient of correlation at index k w.r.t. h."""
    n = len(h)
    shift = k - (n - 1)
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        ns = n - shift
        grad[0:ns] = 1.0 - h[shift:n]
        grad[shift:n] -= h[0:ns]
    else:
        s = -shift
        ns = n - s
        grad[s:n] = 1.0 - h[0:ns]
        grad[0:ns] -= h[s:n]
    return grad

    """Objective and gradient for LSE-smoothed max with sum penalty."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exps = np.exp(shifted)
    logsum = np.log(np.sum(exps))
    f = m + logsum / alpha
    
    # Weights for LSE gradient
    w = exps / np.sum(exps)  # sum to 1
    
    # Gradient: sum_k w_k * grad_corr_k(h)
    # This is O(n^2) total
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-12:  # Skip negligible weights
            g += w[k] * corr_grad(h, k)
    
    # Penalty gradient
    s_penalty = pen * (np.sum(h) - n * 0.5)
    penalty = s_penalty ** 2 / (2 * pen) if pen > 0 else 0
    f += penalty
    if pen > 0:
        g += s_penalty * np.ones(n)
    
    return f, g

            res = minimize(slsqp_obj, h_best.copy(), method='SLSQP', bounds=bounds,
                          constraints=constraint, jac=slsqp_grad,
                          args=(a,), options={'maxiter': 600, 'ftol': 1e-14})
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc

    # Phase 3: Diverse starting points
    starts = [
        make_periodic(n, 3, 1),
        make_periodic(n, 4, 2),
        make_periodic(n, 5, 2),
        make_periodic(n, 7, 3),
        make_periodic(n, 8, 3),
        make_gauss(n, [n*0.15, n*0.5, n*0.85], [n*0.1, n*0.12, n*0.1]),
        make_gauss(n, [n*0.25, n*0.75], [n*0.2, n*0.2]),
        make_blocks(n, [(0, n//4), (n//2 - n//8, n//4), (3*n//4, n//4)]),
    ]
    for _ in range(4):
        hr = rng.random(n)
        starts.append(project_feasible(hr, n))

    for sp in starts:
        if remaining() < 45:
            break
        for alpha, pen, mi in [(200, 1e5, 200), (1500, 1e5, 300)]:
            if remaining() < 35:
                break
            hc, c5c = try_lse(sp.copy(), alpha, pen, mi)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc

    # Phase 4: Directed perturbations targeting high-overlap regions
    ntrials = max(4, int(remaining() / 30))
    for _ in range(ntrials):
        if remaining() < 25:
            break
        corr = compute_corr(h_best)
        m = np.max(corr)
        # Create perturbation weighted toward active lags
        active = corr > m * 0.95
        pert = 0.015 + 0.02 * rng.random()
        noise = rng.standard_normal(n)
        hp = h_best + pert * noise
        hp = project_feasible(hp, n)
        hc, c5c = try_lse(hp, 2000, 1e5, 250)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc

    # Phase 5: Final high-precision polish
    if remaining() > 30:
        hc, c5c = try_lse(h_best.copy(), 5000, 1e6, 500)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc
        if remaining() > 15:
            res = minimize(lambda x: obj_grad_lse(x, 5000, 0)[0], h_best.copy(),
                          method='SLSQP', bounds=bounds, constraints=constraint,
                          jac=lambda x: obj_grad_lse(x, 5000, 0)[1],
                          options={'maxiter': 500, 'ftol': 1e-15})
            hc2 = project_feasible(res.x, n)
            c5c2 = compute_c5(hc2)
            if c5c2 < c5_best:
                c5_best = c5c2
                h_best = hc2

    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
