import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    r = (corr - m) / temp
    r_max = np.max(r)
    exp_r = np.exp(r - r_max)
    s = np.sum(exp_r)
    w = exp_r / s
    f = (m + temp * (r_max + np.log(s))) * dx
    grad = np.zeros(n)
    for idx in range(len(w)):
        k = idx - (n - 1)
        wk = w[idx]
        if wk < 1e-13:
            continue
        dc = np.zeros(n)
        if k >= 0:
            nA = n - k
            if nA > 0:
                dc[k:] += (1.0 - h[:nA])
                dc[:nA] -= h[k:]
        else:
            nA = n + k
            if nA > 0:
                dc[:nA] += (1.0 - h[-k:])
                dc[-k:] -= h[:nA]
        grad += wk * dc
    grad *= dx / temp
    return f, grad

def project(h, S):
    lo = float(np.min(h)) - 2.0
    hi = float(np.max(h)) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hq = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hq) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def pg_optimize(h0, temp, max_iters, S):
    best = h0.copy()
    best_f = smooth_max_grad(best, temp)[0]
    for _ in range(max_iters):
        f, g = smooth_max_grad(best, temp)
        gnorm = np.dot(g, g)
        if gnorm < 1e-16:
            break
        lr = min(0.5, 0.5 / np.sqrt(gnorm))
        improved = False
        for _ in range(40):
            h_new = project(best - lr * g, S)
            f_new = smooth_max_grad(h_new, temp)[0]
            if f_new < f - 0.01 * lr * gnorm:
                best = h_new
                f = f_new
                best_f = min(best_f, f_new)
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return best, compute_c5(best)

def slsqp_optimize(h0, temp, max_iters, S):
    n = len(h0)
    def _obj(hv):
        return smooth_max_grad(hv, temp)[0]
    def _grad(hv):
        return smooth_max_grad(hv, temp)[1]
    def _con_fun(hv):
        return np.sum(hv) - S
    def _con_jac(hv):
        return np.ones(n)
    cons = ({'type': 'eq', 'fun': _con_fun, 'jac': _con_jac},)
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iters, 'ftol': 1e-14})
        h_new = project(res.x, S)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new / 2.0)

def make_pulse_init(n, rng, npulses, width_frac):
    h = np.zeros(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    centers = np.linspace(0.0, 2.0, npulses + 2)[1:-1]
    for c in centers:
        h += np.exp(-(((x - c) % 2.0) ** 2) / (2.0 * (width_frac * 2.0) ** 2))
    if np.max(h) > 0:
        h = h / np.max(h)
    return project(h, n / 2.0)

def make_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

        if c5 < best_c5:
            best_h, best_c5 = h_c.copy(), c5

    # Phase 4: Random restarts with small perturbations
    nrestart = 0
    while time_left() > 25 and nrestart < 200:
        scale = 0.02 + 0.04 * (nrestart % 8) / 7.0
        h_p = np.clip(best_h + scale * rng.randn(best_n), 0.0, 1.0)
        h_p = project(h_p, best_n / 2.0)
        h_c, c5 = pg_optimize(h_p, 0.003, 100, best_n / 2.0)
        if time_left() > 12:
            h_c, c5 = slsqp_optimize(h_c, 0.0005, 150, best_n / 2.0)
        if c5 < best_c5:
            best_h, best_c5 = h_c.copy(), c5
        nrestart += 1

    # Phase 5: Final fine polish at multiple temperatures
    if time_left() > 15:
        for tp in [0.0002, 0.00005]:
            if time_left() < 8:
                break
            h_c, c5 = pg_optimize(best_h, tp, 200, best_n / 2.0)
            if time_left() > 10:
                h_c, c5 = slsqp_optimize(h_c, tp, 300, best_n / 2.0)
            if c5 < best_c5:
                best_h, best_c5 = h_c.copy(), c5

    # Final validation
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
