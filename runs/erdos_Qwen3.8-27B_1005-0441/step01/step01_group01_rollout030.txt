                if c5c < c5_best - 1e-14:
                    c5_best = c5c
                    h_best = hc.copy()
            except Exception:
                pass

    # Phase 3: Many perturbation restarts with L-BFGS-B
    n_restart = 0
    while elapsed() < budget_s * 0.65:
        n_restart += 1
        scale = 0.01 + 0.08 * rng.random()
        h_start = h_best + scale * rng.standard_normal(n)
        h_start = project_feasible(h_start, n)
        try:
            res = minimize(
                smooth_obj, h_start, args=(300.0, 2000.0),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 200, 'ftol': 1e-16}
            )
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best - 1e-14:
                c5_best = c5c
                h_best = hc.copy()
        except Exception:
            pass

    # Phase 4: Structured starting points
    for pat in [1, 2, 3, 4]:
        if elapsed() > budget_s * 0.78:
            break
        h_struct = make_structured(n, pat, rng)
        for alpha in [150.0, 400.0]:
            if elapsed() > budget_s * 0.80:
                break
            try:
                res = minimize(
                    smooth_obj, h_struct, args=(alpha, 2000.0),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 250, 'ftol': 1e-16}
                )
                hc = project_feasible(res.x, n)
                c5c = compute_c5(hc)
                if c5c < c5_best - 1e-14:
                    c5_best = c5c
                    h_best = hc.copy()
            except Exception:
                pass

    # Phase 5: Very high alpha fine-tuning
    if elapsed() < budget_s * 0.88:
        for alpha in [1000.0, 2000.0]:
            if elapsed() > budget_s * 0.92:
                break
            try:
                res = minimize(
                    smooth_obj, h_best.copy(), args=(alpha, 50000.0),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 300, 'ftol': 1e-18}
                )
                hc = project_feasible(res.x, n)
                c5c = compute_c5(hc)
                if c5c < c5_best - 1e-14:
                    c5_best = c5c
                    h_best = hc.copy()
            except Exception:
                pass

    # Phase 6: Coordinate descent polish (swap-based to preserve sum)
    if elapsed() < budget_s - 20:
        for cd_round in range(8):
            if elapsed() > budget_s - 20:
                break
            for p in range(n):
                if elapsed() > budget_s - 20:
                    break
                if h_best[p] <= 1e-10 or h_best[p] >= 1.0 - 1e-10:
                    continue
                orig = h_best[p]
                vals = np.unique(np.concatenate([[0.0, 1.0], np.linspace(0, 1, 31)]))
                best_v, best_c = orig, c5_best
                for v in vals:
                    if abs(v - orig) < 1e-6:
                        continue
                    h_test = h_best.copy()
                    h_test[p] = v
                    h_test = project_feasible(h_test, n)
                    c = compute_c5(h_test)
                    if c < best_c - 1e-13:
                        best_c = c
                        best_v = v
                h_best[p] = best_v
            h_best = project_feasible(h_best, n)
            c5c = compute_c5(h_best)
            if c5c < c5_best:
                c5_best = c5c

    # Final
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
