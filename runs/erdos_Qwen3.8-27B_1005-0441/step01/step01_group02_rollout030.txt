    if time.time() - t0 < t_budget - 10:

            res = minimize(_nm_wrap, hb.copy(), method='Nelder-Mead',
                           options={'maxiter': 300, 'xatol': 1e-11, 'fatol': 1e-13})
            h_nm = _project(res.x)
            c_nm = compute_c5(h_nm)
            if c_nm < best:
                best = c_nm
                hb = h_nm
        except Exception:
            pass
    return hb, best

def _coord_descent(h, best_c5, n_iters=3):
    n = len(h)
    h_now = h.copy()
    c_now = best_c5
    for _ in range(n_iters):
        improved = True
        while improved:
            improved = False
            for j in range(n):
                for delta in [0.001, -0.001, 0.003, -0.003, 0.008, -0.008]:
                    h_try = h_now.copy()
                    h_try[j] += delta
                    k = (j + 1) % n
                    h_try[k] -= delta
                    if 0.0 <= h_try[j] <= 1.0 and 0.0 <= h_try[k] <= 1.0:
                        c_try = compute_c5(h_try)
                        if c_try < c_now - 1e-13:
                            c_now = c_try
                            h_now = h_try
                            improved = True
                            break
                if improved:
                    break
    return h_now, c_now

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is not None:
        best_h = _project(h_init.copy())
        best_c5 = compute_c5(best_h)
        n_main = len(h_init)
    else:
        n_main = 70
        h_init = np.zeros(n_main)
        h_init[:n_main // 2] = 1.0
        best_h = _project(h_init)
        best_c5 = compute_c5(best_h)

    # Phase 1: Main optimization from initial
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 100:
        h_opt, c_opt = _full_optimize(best_h, min(400, t_rem), rng)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt

    # Phase 2: Perturbed restarts from best
    t_rem = budget_s - (time.time() - t_start)
    n_restarts = min(int(t_rem / 70), 8) if t_rem > 70 else 0
    for i in range(n_restarts):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 70:
            break
        scale = 0.06 + 0.02 * (i % 4)
        h_pert = _project(best_h + scale * rng.randn(len(best_h)))
        hp, cp = _full_optimize(h_pert, min(55, t_rem - 5), rng)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp

    # Phase 3: Try other n values
    t_rem = budget_s - (time.time() - t_start)
    n_curr = len(best_h)
    for n_try in [50, 55, 60, 80, 90, 100, 120]:
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 50:
            break
        if n_try == n_curr:
            continue
        h_try = np.zeros(n_try)
        h_try[:n_try // 2] = 1.0
        h_try = _project(h_try + 0.25 * rng.randn(n_try))
        ht, ct = _full_optimize(h_try, min(40, t_rem - 5), rng)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht

    # Phase 4: Second round of restarts with smaller perturbations
    t_rem = budget_s - (time.time() - t_start)
    n_round2 = min(int(t_rem / 50), 5) if t_rem > 50 else 0
    for i in range(n_round2):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 50:
            break
        n_cur = len(best_h)
        h_pert = _project(best_h + 0.03 * rng.randn(n_cur))
        hp, cp = _full_optimize(h_pert, min(40, t_rem - 5), rng)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp

    # Phase 5: Coordinate descent polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 15:
        h_cd, c_cd = _coord_descent(best_h, best_c5, n_iters=5)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 6: Final NM
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 20:
        try:
            res = minimize(_nm_wrap, best_h.copy(), method='Nelder-Mead',
                           options={'maxiter': 500, 'xatol': 1e-12, 'fatol': 1e-14})
            h_fin = _project(res.x)
            c_fin = compute_c5(h_fin)
            if c_fin < best_c5:
                best_c5 = c_fin
                best_h = h_fin
        except Exception:
            pass

    # Final check and return
    best_h = _project(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
