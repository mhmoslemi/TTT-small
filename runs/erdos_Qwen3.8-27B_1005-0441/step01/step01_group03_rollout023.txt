        if 0 <= i < n and 0 <= i+k < n:
            affected.add(i)

def to_binary(h, n):
    """Project h to binary with exactly n//2 ones."""
    target = n // 2
    # Sort by value, take top target as 1
    idx = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=np.float64)
    b[idx[:target]] = 1.0
    return b

                npert = rng.randint(1, 4)
                for _ in range(npert):
                    pp = rng.permutation(ones_list)
                    zz = rng.permutation(zeros_list)
                    i0 = pp[rng.randint(len(pp))]
                    j0 = zz[rng.randint(len(zz))]
                    d = delta_swap(h, n, i0, j0)
                    cur_ov = cur_ov + d
                    cur_c5 = np.max(cur_ov) * dx
                    h[i0] = 0.0
                    h[j0] = 1.0
                    ones_list.remove(i0)
                    zeros_list.remove(j0)
                    ones_list.append(j0)
                    zeros_list.append(i0)
                stall = 0
        sweep += 1
        if sweep % 300 == 0:
            cur_ov = ov_arr(h)
            cur_c5 = np.max(cur_ov) * dx
    return best_h, best_c5

def proj_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
    cnt = float(np.sum(mask))
    if cnt > 0:
        hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def refine_continuous(h_binary, n, deadline, rng):
    h = h_binary.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = c5_val(h, n)
    lr = 0.08
    stall = 0
    while time.time() < deadline:
        ov = ov_arr(h)
        kmax = int(np.argmax(ov))
        lag = kmax - (n - 1)
        g = np.zeros(n)
        for j in range(n):
            jl = j - lag
            if 0 <= jl < n:
                g[j] += 1.0 - h[jl]
            jr = j + lag
            if 0 <= jr < n:
                g[j] -= h[jr]
        h_new = h - lr * g * (2.0 / n)
        h_new = proj_feasible(h_new, n)
        c_new = c5_val(h_new, n)
        if c_new < best_c5 - 1e-14:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            stall += 1
            h = h_new
        if stall > 200:
            lr *= 0.7
            stall = 0
            if lr < 0.001:
                h = best_h + rng.randn(n) * 0.05
                h = proj_feasible(h, n)
                lr = 0.08
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    # Phase 1: Binary LS from initial (35% budget)
    if h0 is not None:
        n0 = len(h0)
    else:
        n0 = 46
        h0 = np.ones(n0) * 0.5
    d1 = min(t0 + budget_s * 0.35, t0 + budget_s - 60)
    hb = to_binary(h0, n0)
    bh, bc5 = binary_ls(hb, n0, d1, rng, max_sweeps=3000)
    if bc5 < best_c5:
        best_h, best_c5, best_n = bh, bc5, n0

    # Phase 2: Multiple n values with random restarts (45% budget)
    n_candidates = [44, 46, 48, 50, 52, 54, 56]
    n_candidates = [x for x in n_candidates if x != n0]
    d2 = t0 + budget_s * 0.80
    per_n = (d2 - time.time()) / max(len(n_candidates), 1)
    for nn in n_candidates:
        if time.time() > d2 - 30:
            break
        d_try = min(time.time() + per_n, d2 - 20)
        best_n_c5 = 1.0
        best_n_h = None
        n_trials = max(2, int((d_try - time.time()) / 2.5))
        for trial in range(n_trials):
            if time.time() > d_try:
                break
            hh = rng.rand(nn)
            hb2 = to_binary(hh, nn)
            bh2, bc52 = binary_ls(hb2, nn, min(time.time() + 2.5, d_try), rng, max_sweeps=800)
            if bc52 < best_n_c5:
                best_n_c5 = bc52
                best_n_h = bh2
        if best_n_h is not None and best_n_c5 < best_c5:
            best_c5 = best_n_c5
            best_n = nn
            best_h = best_n_h

    # Phase 3: More restarts on best_n (10% budget)
    if best_h is not None and time.time() < t0 + budget_s - 40:
        d3 = t0 + budget_s * 0.90
        bh3, bc53 = binary_ls(best_h.copy(), best_n, d3, rng, max_sweeps=5000)
        if bc53 < best_c5:
            best_c5 = bc53
            best_h = bh3

    # Phase 4: Continuous refinement (10% budget)
    if best_h is not None and time.time() < t0 + budget_s - 10:
        d4 = t0 + budget_s - 10
        h_cont = best_h.copy().astype(np.float64)
        bh4, bc54 = refine_continuous(h_cont, best_n, d4, rng)
        if bc54 < best_c5:
            best_c5 = bc54
            best_h = bh4

    # Final validation
    if best_h is not None:
        h_final = proj_feasible(best_h, best_n) if best_h.min() < -1e-10 or best_h.max() > 1.0 + 1e-10 else best_h
        c5_final = c5_val(h_final, best_n)
        return h_final, float(c5_final), best_n

    # Fallback
    h_fb = to_binary(h0 if h0 is not None else np.ones(46) * 0.5, n0)
    return h_fb, c5_val(h_fb, n0), n0
