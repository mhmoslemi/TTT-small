import numpy as np
import time

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def to_binary(h, n):
    if n % 2 == 0:
        target = n // 2
        idx = np.argpartition(h, n - target)[n - target:]
        hb = np.zeros(n)
        hb[idx] = 1.0
    else:
        target = (n - 1) // 2
        idx = np.argpartition(h, n - target - 1)[n - target - 1:]
        idx_sorted = idx[np.argsort(h[idx])]
        hb = np.zeros(n)
        hb[idx_sorted[0]] = 0.5
        hb[idx_sorted[1:]] = 1.0
    return hb

def proj_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        frac = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = float(np.sum(frac))
        if cnt > 0:
            h[frac] += (target - s) / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_swap_search(h, n, deadline, rng):
    best_h = h.copy()
    best_c5v = get_c5(h, n)
    tabu = np.zeros((n, n), dtype=np.int8)
    tabu_stack = []
    stalled = 0
    sweep = 0
    while time.time() < deadline:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_oi = -1
        best_oj = -1
        cm = get_c5(h, n)
        for i in ones:
            for j in zeros:
                if tabu[i][j]:
                    continue
                h[i], h[j] = h[j], h[i]
                ov = np.correlate(h, 1.0 - h, mode='full')
                nc = float(np.max(ov)) * (2.0 / n)
                h[i], h[j] = h[j], h[i]
                d = nc - cm
                if d < best_delta - 1e-15:
                    best_delta = d
                    best_oi = int(i)
                    best_oj = int(j)
        if best_oi >= 0:
            h[best_oi], h[best_oj] = h[best_oj], h[best_oi]
            tabu[best_oi][best_oj] = 1
            tabu_stack.append((best_oi, best_oj))
            if len(tabu_stack) > 40:
                old = tabu_stack.pop(0)
                tabu[old[0]][old[1]] = 0
            nc = get_c5(h, n)
            if nc < best_c5v - 1e-12:
                best_h = h.copy()
                best_c5v = nc
            stalled = 0
        else:
            stalled += 1
            if stalled <= 2:
                a = int(rng.choice(ones))
                b = int(rng.choice(zeros))
                if not tabu[a][b]:
                    h[a], h[b] = h[b], h[a]
                    tabu[a][b] = 1
                    tabu_stack.append((a, b))
                    if len(tabu_stack) > 40:
                        old = tabu_stack.pop(0)
                        tabu[old[0]][old[1]] = 0
            elif stalled > 2:
                k = min(int(rng.randint(3, 7)), len(ones) // 2, len(zeros) // 2)
                oi = rng.choice(ones, size=k, replace=False)

                for aa, bb in zip(oi, oj):
                    h[int(aa)], h[int(bb)] = h[int(bb)], h[int(aa)]
                tabu[:] = 0
                tabu_stack = []
                stalled = 0
        sweep += 1
    return best_h, best_c5v

def grad_lag(h, lag, n):
    g = np.zeros(n)
    for j in range(n):
        jl = j - lag
        if 0 <= jl < n:
            g[j] += (1.0 - h[jl])
        jr = j + lag
        if 0 <= jr < n:
            g[j] -= h[jr]
    return g

def continuous_refine(h, n, deadline, rng):
    best_h = h.copy()
    best_c5v = get_c5(h, n)
    dx = 2.0 / n
    lr = 0.03
    stalled = 0
    it = 0
    while time.time() < deadline:
        ov = get_overlaps(h, n)
        top_idx = np.argsort(ov)[-3:]
        for ki in top_idx:
            if lr < 0.001:
                break
            lag = int(ki) - (n - 1)
            if lag == 0:
                continue
            g = grad_lag(h, lag, n)
            h_new = h - lr * g * dx
            h_new = proj_sum(h_new, n)
            nc = get_c5(h_new, n)
            if nc < best_c5v - 1e-13:
                best_h = h_new.copy()
                best_c5v = nc
                h = h_new
                stalled = 0
        it += 1
        if best_c5v == get_c5(h, n):
            stalled += 1
        else:
            stalled = 0
        if stalled > 80:
            lr *= 0.6
            stalled = 0
            if lr < 0.001:
                h = best_h + rng.randn(n) * 0.04
                h = proj_sum(h, n)
                lr = 0.03
        if stalled > 300:
            break
    return best_h, best_c5v
