        try:
            res = minimize(_smoobj, h.copy(), args=(eps,), jac=_smograd,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-14})
            hr = _proj(res.x, n)
            v = _c5(hr)
            if v < best:
                best = v
                h = hr
        except Exception:
            continue
    return h, best

def _subgrad_polish(h0, n, t_limit, t0):
    h = h0.copy()
    best = _c5(h)
    lr = 0.003
    stall = 0
    for _ in range(5000):
        if time.time() - t0 > t_limit:
            break
        c = _corr(h)
        k_star = int(np.argmax(c))
        m = k_star - (n - 1)
        g = np.zeros(n)
        if m >= 0:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
        else:
            p = -m
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
        h_new = _proj(h - lr * g, n)
        v = _c5(h_new)
        if v < best - 1e-12:
            best = v
            h = h_new
            stall = 0
            lr = min(lr * 1.02, 0.01)
        else:
            h = h_new
            stall += 1
            lr *= 0.93
            if lr < 1e-8 or stall > 200:
                break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = np.inf

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        h_init = _proj(h_init, len(h_init))
    except Exception:
        h_init = np.full(70, 0.5)

    n_base = len(h_init)

    # Coordinate descent on base n
    h_cd, v_cd = _coord_descent(h_init, n_base, nsweeps=25, t_limit=budget_s * 0.35, t0=t0)
    if v_cd < best_val:
        best_val = v_cd
        best_h = h_cd.copy()

    # SLSQP polish on base
    if time.time() - t0 < budget_s * 0.55:
        h_sp, v_sp = _slsqp_polish(best_h, n_base, [0.03, 0.01, 0.005, 0.002],
                                    budget_s * 0.55, t0)
        if v_sp < best_val:
            best_val = v_sp
            best_h = h_sp.copy()

    # Try other n values via interpolation
    for n_try in [n_base - 7, n_base + 7, n_base + 14, n_base + 21]:
        if time.time() - t0 > budget_s * 0.78:
            break
        if n_try < 30 or n_try > 200:
            continue
        x_orig = np.linspace(0, 2, n_base)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = _proj(h_interp, n_try)
        # Coordinate descent
        h_t, v_t = _coord_descent(h_interp, n_try, nsweeps=12,
                                   t_limit=budget_s * 0.82, t0=t0)
        if v_t < best_val:
            best_val = v_t
            best_h = h_t.copy()
        # Quick SLSQP
        if time.time() - t0 < budget_s * 0.85:
            h_t2, v_t2 = _slsqp_polish(h_t, n_try, [0.01, 0.004],
                                        budget_s * 0.85, t0)
            if v_t2 < best_val:
                best_val = v_t2
                best_h = h_t2.copy()

    # Random restarts on best_h
    if time.time() - t0 < budget_s * 0.90:
        n_b = len(best_h)
        for _ in range(3):
            if time.time() - t0 > budget_s * 0.90:
                break
            h_r = best_h + rng.randn(n_b) * 0.02
            h_r = _proj(h_r, n_b)
            h_r2, v_r = _coord_descent(h_r, n_b, nsweeps=8,
                                        t_limit=budget_s * 0.91, t0=t0)
            if v_r < best_val:
                best_val = v_r
                best_h = h_r2.copy()

    # Final subgradient polish
    if best_h is not None and time.time() < budget_s * 0.93:
        h_sg, v_sg = _subgrad_polish(best_h, len(best_h), budget_s * 0.95, t0)
        if v_sg < best_val:
            best_val = v_sg
            best_h = h_sg

    # Very final SLSQP with tiny eps
    if best_h is not None and time.time() < budget_s * 0.96:
        n_f = len(best_h)
        bounds = [(0.0, 1.0)] * n_f
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n_f * 0.5,
                 'jac': lambda: np.ones(n_f)}]
        try:
            res = minimize(_smoobj, best_h.copy(), args=(0.001,),
                           jac=_smograd, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-15})
            h_f = _proj(res.x, n_f)
            v_f = _c5(h_f)
            if v_f < best_val:
                best_val = v_f
                best_h = h_f
        except Exception:
            pass

    if best_h is None:
        best_h = h_init.copy()
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
