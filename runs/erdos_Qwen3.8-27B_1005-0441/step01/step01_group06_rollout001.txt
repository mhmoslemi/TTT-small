import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-9
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def pnorm_objective(x, n, p):
    h = np.clip(x, 0, 1)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    cp = np.maximum(c, 0.0)
    return (np.sum(cp**p))**(1.0/p) * dx

def optimize_continuous(h_init, n, t0, budget_s):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    bounds = [(0.0, 1.0)] * n
    
    for p in [4, 8, 16, 32, 64, 128, 256, 512, 1024]:
        if time.time() - t0 > budget_s * 0.85:
            break
        try:
            maxiter = 200 if n <= 200 else 100
            res = minimize(pnorm_objective, h, args=(n, p), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                h = best_h.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def fix_binary_sum(h_bin, n):
    target = n // 2
    cur = int(np.sum(h_bin))
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if cur > target:
        # Remove (cur - target) ones
        for i in range(cur - target):
            h_bin[ones[i]] = 0
    elif cur < target:
        # Add (target - cur) zeros to ones
        for i in range(target - cur):
            h_bin[zeros[i]] = 1
    return h_bin

def binary_local_search(h_bin, n, t0, budget_s, max_time=120):

    h = fix_binary_sum(h, n)
    c5 = compute_c5(h)
    dx = 2.0 / n
    
    deadline = min(t0 + budget_s * 0.85, t0 + max_time)
    improved = True
    
    while improved and time.time() < deadline:
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        best_improve = 1e-10
        best_flip = None
        
        # Try all pairs: flip one 0 to 1 and one 1 to 0
        for zi in zeros:
            for oi in ones:
                h[zi] = 1
                h[oi] = 0
                c = np.correlate(h, 1.0 - h, mode="full")
                new_c5 = np.max(c) * dx
                h[zi] = 0
                h[oi] = 1

            h[zi] = 1
            h[oi] = 0
            c5 = new_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def pnorm_obj(x, n, p):
    h = np.clip(x, 0, 1)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    cp = np.maximum(c, 0.0)
    return (np.sum(cp**p))**(1.0/p) * dx

def optimize_lbgfs(h_init, n, t0, deadline_frac):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for p in [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]:
        if time.time() - t0 > (time.time() - t0) + (deadline_frac - 1):
            break
        maxiter = 300 if n <= 200 else 150
        try:
            res = minimize(pnorm_obj, h, args=(n, p), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:

                h = best_h.copy()
        except Exception:
            pass
    return best_h, best_c5

            best_c5, best_h = c5_7, h7

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        val = evaluate_erdos_solution(best_h)
        if val < final_c5:
            final_c5 = val
    except Exception:
        pass

    return (best_h, final_c5, n_final)
