```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.999999
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.min(h[mask] + diff/cnt, 1.0)
        else:
            mask = h > 1e-9
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.max(h[mask] + diff/cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, beta, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    t = beta * (c - m)
    t -= t.max()
    lse = np.log(np.sum(np.exp(t))) / beta
    s = np.sum(x) - n / 2.0
    return (m + lse) * dx + mu * s * s

def slsqp_optimize(h_init, n, t0, budget_s):
    h = project_feasible(h_init, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    betas = [3, 10, 30, 100, 300, 1000, 3000, 10000]
    for beta in betas:
        if time.time() - t0 > budget_s * 0.80:
            break
        maxiter = 200 if n <= 200 else 100
        try:
            res = minimize(lse_obj, h, args=(n, beta, 100.0),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best, best_c5 = h_new.copy(), c5
                h = h_new
        except Exception:
            pass
    
    return best, best_c5
```

```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def make_obj(n, beta, mu):
    dx = 2.0 / n
    def f(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        t = beta * (c - m)
        t -= t.max()
        return (m + np.log(np.sum(np.exp(t))) / beta) * dx + mu * (np.sum(x) - n/2.0)**2
    return f

def make_grad(n, beta, mu):

    def g(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        t = beta * (c - m)
        t -= t.max()
        e = np.exp(t)
        w = e / e.sum()
        total = np.zeros(n)
        for k in range(2*n - 1):
            wk = w[k]
            if wk < 1e-14:

            s = k - (n - 1)
            if s >= 0:
                total[0:n-s] += wk * (1.0 - x[s:n])

                total[0:n-t2] += -wk * x[t2:n]
        total *= dx
        total += 2.0 * mu * (np.sum(x) - n/2.0)
```

```python
            total[0:n - t2] += -wk * x[t2:n]

    total += 2.0 * mu * (np.sum(x) - n / 2.0)

def lbfobs_opt(h_init, n, t0, budget_s, max_frac=0.75):
    h = project_feasible(h_init, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    betas = [5, 20, 100, 500, 2000, 10000]
    mus = [1000, 5000, 20000]
    for mu in mus:
        for beta in betas:
            if time.time() - t0 > budget_s * max_frac:
                return best, best_c5
            maxiter = 300 if n <= 200 else 150
            try:
                res = minimize(obj_fn, h, args=(n, beta, mu), jac=grad_fn,
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'disp': 0})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = project_feasible(res.x, n)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best, best_c5 = h_new.copy(), c5
                    h = h_new
            except Exception:
                pass
    return best, best_c5

def subgrad_polish(h, n, t0, budget_s, max_iter=200):
    h = project_feasible(h, n).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    step = 0.01
    for it in range(max_iter):
        if time.time() - t0 > budget_s:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        m = c.max()
        # Get top few and their gradients
        top_k = 5
        idx = np.argsort(c)[-top_k:]
        # Compute average gradient of top-k correlation values
        avg_grad = np.zeros(n)
        for ki in idx:
            s = ki - (n - 1)
            if s >= 0:
                avg_grad[0:n - s] += (1.0 - h[s:n])
                avg_grad[s:n] += -h[0:n - s]
            else:
                t2 = -s
                avg_grad[t2:n] += (1.0 - h[0:n - t2])
                avg_grad[0:n - t2] += -h[t2:n]

            h = h_new
            step = min(step * 1.1, 0.05)
        else:
            step *= 0.7
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def obj_fn(x, n, beta, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    t = beta * (c - m)
    t -= t.max()
    return (m + np.log(np.sum(np.exp(t))) / beta) * dx + mu * (np.sum(x) - n / 2.0) ** 2

def grad_fn(x, n, beta, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    t = beta * (c - m)
    t -= t.max()
    e = np.exp(t)

    total = np.zeros(n)

        if wk < 1e-14:

        s = k - (n - 1)
        if s >= 0:

            total[s:n] += -wk * x[0:n - s]
        else:
            t2 = -s
            total[t2:n] += wk * (1.0 - x[0:n - t2])
```
