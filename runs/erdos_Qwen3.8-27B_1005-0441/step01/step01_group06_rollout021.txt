def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = float(n) / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.minimum(h, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:

            h[mask] += diff / cnt
            h = np.maximum(h, 0.0)
