import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def get_grad_max(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(c)
    L = k_star - n + 1
    grad = np.zeros(n)
    idx1 = np.arange(n) + L
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += (1.0 - h[np.clip(idx1, 0, n-1)[valid1]])
    idx2 = np.arange(n) - L
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] += -h[np.clip(idx2, 0, n-1)[valid2]]
    return grad, k_star

def soft_obj(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    sh = beta * (c - m)
    sh -= sh.max()
    return (m + np.log(np.sum(np.exp(sh))) / beta) * dx

def lp_obj(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    return float(np.sum(c**p)**(1.0/p) * dx)

def cons_fun(x):
    return np.sum(x) - len(x)/2.0

def cons_jac(x):
    return np.ones(len(x))

def make_bounds(n):
    return [(0.0, 1.0)]*n

def run_subgrad(h, n, t0, tmax, maxit=8000):
    best = h.copy()
    best_c = compute_c5(h)
    vel = np.zeros(n)
    mu = 0.9
    lr = 0.005
    for it in range(maxit):
        if time.time() - t0 > tmax:
            break
        g, _ = get_grad_max(h)
        vel = mu*vel - lr*g
        h_new = project_feasible(h + vel, n)
        c5 = compute_c5(h_new)
        if c5 < best_c:
            best = h_new.copy()
            best_c = c5
        elif it > 200 and it % 100 == 0:
            lr *= 0.92
        h = h_new
    return best, best_c

def slsqp_opt(h, n, t0, tmax, beta=1000, maxit=250, obj_type='soft'):
    bounds = make_bounds(n)
    cons = [{'type':'eq','fun':cons_fun,'jac':cons_jac}]
    if obj_type == 'soft':
        fun = lambda x: soft_obj(x, n, beta)
    else:
        p = obj_type if isinstance(obj_type, int) else 8
        fun = lambda x: lp_obj(x, n, p)
    try:
        res = minimize(fun, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h, n)

def binary_local_search(h, n, t0, tmax, max_rounds=30):
    h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_rounds):
        if time.time() - t0 > tmax:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.time() - t0 > tmax:
                break
            old = h[i]
            h[i] = 1.0 if old < 0.5 else 0.0
            h = project_feasible(h, n)
            c5 = compute_c5(h)
            if c5 < best_c - 1e-10:
                best_c = c5
                improved = True
            else:
                h[i] = old
                h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c
