def _project(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _compute_c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode="full").max() * 2.0 / n)

def _get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

    e = np.exp(np.clip((corr - m) / max(temp, 1e-8), -500, 0))
    return e / e.sum()

def _grad(corr_weights, h):

    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    w = corr_weights
    for k in range(len(w)):

            grad[:n+s] -= wk * h[-s:]  # h[0:-s]
    return dx * grad

        s = n - 1 - k
        if s >= 0:
            grad[:n - s] += wk * (1.0 - h[s:n])
            grad[s:n] += -wk * h[:n - s]
        else:
            sp = -s
            grad[sp:n] += wk * (1.0 - h[:n - sp])
            grad[:n - sp] += -wk * h[sp:n]
    return dx * grad

def _pgd(h, temps, max_time):
    t0 = time.time()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    lr = 0.5
    for temp in temps:
        if time.time() - t0 > max_time - 3:
            break
        h = best_h.copy()
        n_iter = max(30, int(150.0 * temp / 0.5))
        stall = 0
        for i in range(n_iter):
            if time.time() - t0 > max_time - 2:
                break
            corr = _get_corr(h)
            w = _softmax_weights(corr, temp)
            g = _grad(w, h)
            if np.max(np.abs(g)) < 1e-10:
                break
            alpha = lr
            accepted = False
            for _ in range(40):
                h_new = _project(h - alpha * g, target)
                c5_new = _compute_c5(h_new)
                if c5_new < _compute_c5(h) * (1.0 - 1e-12):
                    h = h_new
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                lr = max(lr * 0.6, 0.01)
                stall += 1
                if stall > 5:
                    break
            else:
                stall = 0
                lr = min(lr * 1.05, 3.0)
            if _compute_c5(h) < best_c5 - 1e-12:
                best_c5 = _compute_c5(h)
                best_h = h.copy()
    return best_h, best_c5

def _local_search(h, n_trials):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = _compute_c5(h)
    for _ in range(n_trials):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        for mode in range(3):
            h_new = best.copy()
            if mode == 0:
                h_new[i], h_new[j] = best[j], best[i]
            elif mode == 1:
                h_new[i] = 1.0 - best[i]
                h_new[j] = 1.0 - best[j]
            else:
                d = best[i] - best[j]
                h_new[i] = best[j]
                h_new[j] = best[i]
            h_new = _project(h_new, target)
            c5 = _compute_c5(h_new)
            if c5 < best_c5 - 1e-12:
                best = h_new
                best_c5 = c5
                break
    return best, best_c5

def _perturb(h, scale):
    return _project(h + scale * np.random.randn(len(h)), len(h) / 2.0)

def _interp(h, n_new):
    n = len(h)
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new / 2.0)

def _slsqp_polish(h, max_time):
    n = len(h)
    t0 = time.time()
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    dx = 2.0 / n
    temp = 0.001

    def _obj_slsqp(x):
        c = _get_corr(x)
        m = c.max()
        e = np.exp(np.clip((c - m) / temp, -500.0, 0.0))
        return (m + temp * np.log(e.sum())) * dx

    def _obj_slsqp_grad(x):
        c = _get_corr(x)
        w = _softmax_weights(c, temp)
        return _grad(w, x)

    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)},)
    t_sched = [0.01, 0.003, 0.001, 0.0003, 0.0001]
    for temp in t_sched:
        if time.time() - t0 > max_time - 5:
            break
        try:
            res = minimize(_obj_slsqp, best_h, jac=_obj_slsqp_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_new = _project(np.clip(res.x, 0.0, 1.0), n / 2.0)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best_h, best_c5

def _binary_init(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return _project(h, n / 2.0)

def _two_pulse_init(n, gap, rng):
    h = np.zeros(n)
    half = n // 4
    g = int(n * gap)
    p1 = rng.randint(0, max(1, n - 2 * half - g - 1))
    for i in range(half):
        h[(p1 + i) % n] = 1.0
        h[(p1 + half + g + i) % n] = 1.0
    return _project(h, n / 2.0)

def _smooth_init(n, width, rng):
    h = np.zeros(n)
    c = rng.randint(n // 4, 3 * n // 4)
    x = np.arange(n, dtype=float)
    for center in [c, (c + n // 2) % n]:
        h += np.exp(-0.5 * ((x - center) % n / width) ** 2)
    return _project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def _rem():
        return max(0.0, budget_s - (time.time() - t0))

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(160, 0.5)
    n = len(h0)
    best_h = _project(h0, n / 2.0)
    best_c5 = _compute_c5(best_h)

    # Phase 1: PGD on initial construction
    if _rem() > 120:
        h_opt, c5_opt = _pgd(best_h, [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005],
                             max_time=min(120, _rem() - 30))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Alternative initializations
    inits = [_binary_init(n, rng), _two_pulse_init(n, 0.2, rng),
             _two_pulse_init(n, 0.35, rng), _smooth_init(n, 20, rng),
             _perturb(best_h, 0.15), _perturb(best_h, 0.3)]
    for h_init in inits:
        if _rem() < 40:
            break
        h_opt, c5_opt = _pgd(h_init, [0.3, 0.1, 0.03, 0.01],
                             max_time=min(35, _rem() - 15))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Multi-resolution
    n_cur = len(best_h)
    for target_n in [max(80, n_cur // 2), min(320, n_cur * 2)]:
        if _rem() < 60 or target_n == n_cur:
            continue
        h_res = _interp(best_h, target_n)
        h_opt, c5_opt = _pgd(h_res, [0.3, 0.1, 0.03, 0.01, 0.005],
                             max_time=min(80, _rem() - 20))
        c5_opt_check = _compute_c5(h_opt)
        if c5_opt_check < best_c5:
            best_h, best_c5 = h_opt, c5_opt_check
        n_cur = len(best_h)

    # Phase 4: Basin hopping
    n_cur = len(best_h)
    rh = best_h.copy()
    for i in range(40):
        if _rem() < 25:
            break
        scale = 0.02 + 0.05 * ((i % 7) / 6.0)
        h_pert = _perturb(rh, scale)
        h_opt, c5_opt = _pgd(h_pert, [0.1, 0.03, 0.01, 0.003],
                             max_time=min(22, _rem() - 10))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            rh = h_opt.copy()
        elif i < 15 and rng.rand() < 0.25:
            rh = h_opt.copy()

    # Phase 5: SLSQP polish
    if _rem() > 30:
        h_opt, c5_opt = _slsqp_polish(best_h, max_time=min(80, _rem() - 15))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 6: Local search + final PGD
    if _rem() > 20:
        h_ls, c5_ls = _local_search(best_h, 500)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls
        h_opt, c5_opt = _pgd(best_h, [0.005, 0.002, 0.001, 0.0005],
                             max_time=max(8, _rem() - 5))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final feasibility check
    best_h = _project(np.clip(best_h, 0.0, 1.0), len(best_h) / 2.0)
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, len(best_h)
