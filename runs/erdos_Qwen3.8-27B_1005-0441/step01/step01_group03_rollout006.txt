                        ct = get_c5(ht, n)
                        if ct < best_c5 - 1e-13:
                            best_c5 = ct
                            best_h = ht.copy()
                            h = ht
                            stale = 0
                            break
                if stale > 8:
                    h = best_h + rng.randn(n) * 0.02
                    h = project_feasible(h, n)
                    stale = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 46

    best_h = None
    best_c5 = 1.0
    best_n = 0

    if h0 is not None:
        h_start = project_feasible(np.clip(h0, 0, 1).copy(), n0)

        # Phase 1: main gradient optimization (45%)
        d1 = t0 + budget_s * 0.45
        bh, bc5 = optimize_main(h_start, n0, d1, rng)
        best_h, best_c5, best_n = bh, bc5, n0

        # Phase 2: coordinate refinement (30%)
        d2 = t0 + budget_s * 0.75
        bh, bc5 = refine_coord(best_h, best_n, d2, rng)
        if bc5 < best_c5:
            best_h, best_c5 = bh, bc5

        # Phase 3: try other resolutions (18%)
        d3 = t0 + budget_s * 0.93
        for n_try in [50, 52, 56]:
            if time.time() > d3 - 20:
                break
            sub_deadline = min(time.time() + (d3 - time.time()) / 3.0, d3 - 5)
            h_try = project_feasible(rng.rand(n_try), n_try)
            # start from structured
            hs = np.zeros(n_try)
            hs[:n_try // 2] = 1.0
            if n_try % 2 == 1:
                hs[n_try // 2] = 0.5
            hs = project_feasible(hs, n_try)
            bh2, bc5_2 = optimize_main(hs, n_try, sub_deadline, rng)
            if time.time() < d3 - 5:
                bh2, bc5_2 = refine_coord(bh2, n_try, d3 - 5, rng)
            if bc5_2 < best_c5:
                best_h, best_c5, best_n = bh2, bc5_2, n_try
    else:
        best_n = n0
        h_start = project_feasible(rng.rand(best_n), best_n)
        best_h, best_c5 = optimize_main(h_start, best_n, t0 + budget_s * 0.7, rng)
        if time.time() < t0 + budget_s - 10:
            best_h, best_c5 = refine_coord(best_h, best_n, t0 + budget_s - 10, rng)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c5 = get_c5(best_h, best_n)
    dx_f = 2.0 / best_n
    assert abs(np.sum(best_h) * dx_f - 1.0) < 1e-8, f"Sum constraint violated: {np.sum(best_h)*dx_f}"
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, float(best_c5), best_n
