import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_temp = 0.01
_g_deadline = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _obj(h):
    dx = 2.0 / _g_n
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = max(_g_temp, 1e-10)
    m = np.max(corr)
    return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx

def _cons_eq(h):
    return np.sum(h) - _g_n / 2.0

def _cons_eq_jac(h):
    return np.ones(_g_n)

def _time_left():
    return _g_deadline - time.time()

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = float(n) / 2.0
    if s > 1e-12:
        h = h * target / s
    h = np.clip(h, 0.0, 1.0)
    return h

def _slqp_polish(h, n, temps, iters_per_temp):
    global _g_n, _g_temp
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    best = h.copy()
    best_val = compute_c5(best)
    for temp in temps:
        if _time_left() < 3:
            break
        _g_temp = temp
        for _ in range(iters_per_temp):
            if _time_left() < 2:
                break
            try:
                res = minimize(_obj, best, method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-12})
                h_new = _project(res.x, n)
                v = compute_c5(h_new)
                if v < best_val - 1e-12:
                    best = h_new.copy()
                    best_val = v
            except Exception:
                break
    return best, best_val

    h_new = h + sigma * rng.standard_normal(n)

    return _project(h, n)

    bps = np.insert(bps, 0, 0.0)
    bps = np.append(bps, 2.0)
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = np.zeros(n)
    for i in range(n):
        for j in range(k):
            if bps[j] <= x[i] < bps[j + 1]:
                h[i] = rng.uniform(0.05, 0.95)
                break
    return _project(h, n)

def _gen_binary(n, rng):
    h = np.zeros(n)
    n_on = n // 2
    idx = rng.choice(n, size=n_on, replace=False)
    h[idx] = 1.0
    return h

def _gen_two_pulse(n, rng):
    h = np.zeros(n)

    gap = int(n * rng.uniform(0.1, 0.4))
