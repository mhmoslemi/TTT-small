import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def soft_max(c, beta):
    m = c.max()
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def obj_and_grad(x, n, beta, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    sm = soft_max(c, beta)
    s = np.sum(x) - n / 2.0
    val = sm + penalty * s * s
    
    # Gradient
    m = c.max()
    weights = np.exp(beta * (c - m))
    weights /= weights.sum()
    
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for j in range(n):
        vp = (j + lags >= 0) & (j + lags < n)
        vm = (j - lags >= 0) & (j - lags < n)
        s1 = np.dot(weights[vp], 1.0 - x[j + lags[vp]])
        s2 = np.dot(weights[vm], x[j - lags[vm]])
        grad[j] = dx * (s1 - s2)
    
    grad += 2.0 * penalty * s
    return val, grad

def optimize_lbnb(h_init, n, t0, budget, max_frac=0.85):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    bounds = [(0.0, 1.0)] * n
    
    for beta in [5, 10, 30, 100, 300, 1000, 3000]:
        penalty = 1.0
        for _ in range(3):  # increase penalty
            if time.time() - t0 > budget * max_frac:
                return best_h, best_c5
            res = minimize(lambda x: obj_and_grad(x, n, beta, penalty)[0],
                          h,
                          jac=lambda x: obj_and_grad(x, n, beta, penalty)[1],
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h = project_feasible(res.x, n)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            penalty *= 3
    
    return best_h, best_c5

    # After swap: h'[i]=0, h'[j]=1, others same
    # overlap'(k) - overlap(k) = dx * [h'[i]*(1-h'[i+k]) - h[i]*(1-h[i+k]) 
    #                                + h'[j]*(1-h'[j+k]) - h[j]*(1-h[j+k])
    #                                + for l where l+k=i: h'[l]*(1-h'[i]) - h[l]*(1-h[i])
    #                                + for l where l+k=j: h'[l]*(1-h'[j]) - h[l]*(1-h[j])]

    #   h'[i]*(1-h'[i+k]) - h[i]*(1-h[i+k]) = 0 - (1-h[i+k]) = -(1-h[i+k])
    #   (valid if i+k in [0,n))

    #   (valid if j+k in [0,n))

    #   h'[l]*(1-h'[i]) - h[l]*(1-h[i]) = h[l]*(1-0) - h[l]*(1-1) = h[l] - 0 = h[l] = h[i-k]

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def soft_max(c, beta):
    m = c.max()
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    sm = soft_max(c, beta)
    s = np.sum(x) - n / 2.0
    return sm + pen * s * s

def lbnb_grad(x, n, beta, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = c.max()
    w = np.exp(beta * (c - m))
    w /= w.sum()
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for j in range(n):
        vp = (j + lags)
        vm = (j - lags)
        mvp = (vp >= 0) & (vp < n)
        mvm = (vm >= 0) & (vm < n)
        s1 = np.dot(w[mvp], 1.0 - x[vp[mvp]])
        s2 = np.dot(w[mvm], x[vm[mvm]])
        grad[j] = dx * (s1 - s2)
    s = np.sum(x) - n / 2.0
    grad += 2.0 * pen * s
    return grad

But that uses global state which is ugly. Alternatively, I can use scipy's `args` parameter for the objective, and for the gradient, I can create a custom class.

Actually, looking at scipy docs, `minimize` with method='L-BFGS-B' accepts:
```python
res = minimize(fun, x0, args=(), jac=None, ...)
```

If `jac` is a boolean, it computes the Jacobian numerically. If it's a callable, it calls `jac(x, *args)`.

So I can do:
```python
def lbnb_fun(x, n, beta, pen):
    return lbnb_obj(x, n, beta, pen)

def lbnb_jac(x, n, beta, pen):

res = minimize(lbnb_fun, h, args=(n, beta, pen),

```

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def soft_max_fn(c, beta):
    m = c.max()
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def lbnb_objective(x, n, beta, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    sm = soft_max_fn(c, beta)
    s = np.sum(x) - n / 2.0
    return sm + pen * s * s

def lbnb_gradient(x, n, beta, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    m = c.max()
    w = np.exp(beta * (c - m))
    w /= w.sum()
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for j in range(n):
        vp = j + lags
        vm = j - lags
        mvp = (vp >= 0) & (vp < n)
        mvm = (vm >= 0) & (vm < n)

        grad[j] = dx * (s1 - s2)
    s = np.sum(x) - n / 2.0
    grad += 2.0 * pen * s
    return grad
