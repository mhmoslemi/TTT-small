        n_best = len(best_h)
        for lr_try in [0.005, 0.003, 0.002]:
            if time.time() - t0 > budget_s * 0.92:
                break
            h_polished, v = _subgrad_optim(best_h, n_best, t0, budget_s,
                                            max_iter=6000, lr_init=lr_try)
            if v < best_val:
                best_val = v
                best_h = h_polished.copy()

    # Random perturbation restarts
    if best_h is not None and time.time() - t0 < budget_s * 0.88:
        n_best = len(best_h)
        for attempt in range(4):
            if time.time() - t0 > budget_s * 0.90:
                break
            noise = rng.randn(n_best) * 0.02
            h_pert = _proj(best_h + noise, n_best)
            h_pert = _slsqp_optim(h_pert, 0.01, n_best, t0, budget_s, 400)
            h_pert, v = _subgrad_optim(h_pert, n_best, t0, budget_s, max_iter=2000, lr_init=0.003)
            if v < best_val:
                best_val = v
                best_h = h_pert.copy()

    # Final aggressive subgradient
    if best_h is not None and time.time() - t0 < budget_s * 0.95:
        n_best = len(best_h)
        h_final, v = _subgrad_optim(best_h, n_best, t0, budget_s,
                                     max_iter=10000, lr_init=0.003)
        if v < best_val:
            best_val = v
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
