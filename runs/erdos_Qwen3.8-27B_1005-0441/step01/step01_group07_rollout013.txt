```python
def continuous_local_search(h, time_limit=10.0):
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        improved = False
        best_delta = 1e-12
        best_pair = None
        for i in range(n):
            for j in range(i+1, n):
                if abs(h[i] - h[j]) < 1e-8:
                    continue
                # Swap h[i] and h[j]
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h)
                delta = c5 - best_c5
                if delta < -best_delta:
                    best_delta = -delta
                    best_pair = (i, j, c5)
                h[i], h[j] = h[j], h[i]  # undo
        if best_pair is not None:
            i, j, val = best_pair
            h[i], h[j] = h[j], h[i]
            best_c5 = val
            improved = True
        else:
            break
    return h, best_c5
```

```python
import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    h_b = (h >= 0.5).astype(float)
    diff = target - int(h_b.sum())
    if diff > 0:
        zeros = np.where(h_b == 0)[0]
        order = zeros[np.argsort(-h[zeros])][:diff]
        h_b[order] = 1.0
    elif diff < 0:
        ones = np.where(h_b == 1)[0]
        order = ones[np.argsort(h[ones])][-diff:]
        h_b[order] = 0.0
    return h_b

def binary_local_search(h, max_rounds=200, time_limit=30.0):
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_val = best_c5
        best_move = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                c5 = compute_c5(h)
                if c5 < best_val - 1e-15:
                    best_val = c5
                    best_move = (i, j)
                h[i] = 1.0; h[j] = 0.0
        if best_move is None:
            break
        i, j = best_move
        h[i] = 0.0; h[j] = 1.0
        best_c5 = best_val
    return h, best_c5

def continuous_local_search(h, max_rounds=100, time_limit=30.0):
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        best_delta = 1e-12
        best_pair = None
        for i in range(n):
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-10:
                    continue
                h[i], h[j] = h[j], h[i]
                c5 = compute_c5(h)
                if c5 < best_c5 - best_delta:
                    best_delta = best_c5 - c5
                    best_pair = (i, j, c5)
                h[i], h[j] = h[j], h[i]
        if best_pair is None:
            break
        i, j, val = best_pair
        h[i], h[j] = h[j], h[i]
        best_c5 = val
    return h, best_c5

def grad_smooth(h, dx, T):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    cm = corr.max()
    shifted = T * (corr - cm)
    exps = np.exp(np.clip(shifted, -50, 50))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

def projected_grad_descend(h_init, time_budget, seed, T_init=80.0, lr=0.05):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.8
    T = T_init
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    while time.time() < deadline - 0.2:
        if no_improve > 500:
            h = project_feasible(best_h + rng.randn(n) * 0.05, target_sum)
            v[:] = 0.0
            T = T_init
            no_improve = 0
        elif no_improve > 200:
            lr = max(lr * 0.9, 0.002)
        g = grad_smooth(h, dx, T)
        v = beta * v - lr * g
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        T = min(T * 1.003, 500.0)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    n = len(best_overall_h)
    return best_overall_h, best_c5, n
```
