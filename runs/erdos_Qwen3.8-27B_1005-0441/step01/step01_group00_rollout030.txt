    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _lag_grad(h, m):
    """Gradient of overlap_m with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if m >= 0:
        # overlap_m = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        # d/dh[j] = (1-h[j-m]) * [j>=m] - h[j+m] * [j+m<n]
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    else:
        p = -m
        # overlap_{-p} = sum_{i=0}^{n-1-p} h[i]*(1-h[i+p])
        # d/dh[j] = (1-h[j+p])*[j<n-p] - h[j-p]*[j>=p]
        g[:n-p] += (1.0 - h[p:])

    return g

def _proj(h, n):

    # Phase 2: Basin hopping
    if best_h is not None and not dl(0.62):
        n = len(best_h)
        for _ in range(6):
            if dl(0.65):
                break
            hp = _proj(best_h + rng.randn(n) * 0.004, n)
            hr = _slsqp(hp, n, [0.02, 0.008, 0.003], [250, 350, 500])
            v = _c5(hr)
            if v < best_val:
                best_val = v
                best_h = hr.copy()

    # Phase 3: Subgradient
    if best_h is not None and not dl(0.72):
        n = len(best_h)
        hn, vn = _subgrad(best_h, n, t0 + budget_s * 0.80)
        if vn < best_val:
            best_val = vn
            best_h = hn

    # Phase 4: Coordinate descent (boundary check)
    if best_h is not None and not dl(0.82):
        h = best_h.copy()
        n = len(h)
        for sweep in range(80):
            if dl(0.86):
                break
            for j in range(n):
                h[j] = 0.0
                v0 = _c5(h)
                h[j] = 1.0
                v1 = _c5(h)
                h[j] = 0.5
                vh = _c5(h)
                if v0 <= v1 and v0 <= vh:
                    h[j] = 0.0
                elif v1 <= v0 and v1 <= vh:
                    h[j] = 1.0
                else:
                    h[j] = 0.5
                h = _proj(h, n)
            v = _c5(h)
            if v < best_val:
                best_val = v
                best_h = h.copy()

    # Phase 5: Final fine subgradient (single active lag)
    if best_h is not None and not dl(0.92):
        h = best_h.copy()
        n = len(h)
        lr = 0.04
        stall = 0
        for it in range(15000):
            if dl(0.95):
                break
            c = _corr(h)
            k_s = int(np.argmax(c))
            m = k_s - (n - 1)
            g = _lag_grad(h, m)
            h_new = _proj(h - lr * g * (2.0 / n), n)
            v = _c5(h_new)
            if v < best_val - 1e-14:
                best_val = v
                best_h = h_new.copy()
                h = h_new
                stall = 0
                if it % 300 == 0:
                    lr *= 0.93
            else:
                stall += 1
                lr *= 0.98
                if stall > 400 or lr < 1e-11:
                    break

    if best_h is None:
        best_h = np.full(100, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
