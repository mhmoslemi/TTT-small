import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        # Binary search for lambda
        lo, hi = -2.0, 2.0
        for _ in range(60):
            mid = (lo + hi) / 2
            h_try = np.clip(h + mid, 0.0, 1.0)
            s_try = np.sum(h_try)
            if s_try < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Final clip
    h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, n, top_k=5, tau=0.01):
    """Compute a subgradient of max_k c(k)*dx, using soft-weighted top lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx  # length 2n-1
    
    # Soft weighting
    c_max = np.max(c)
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    
    # Compute weighted subgradient
    # c(k) = sum_i h[i] * (1 - h[i+k]) * dx for lag k
    # d c(k) / d h[j] = (1-h[j+k])*dx - h[j-k]*dx (for linear corr)
    
    # The full correlation has lags from -(n-1) to (n-1)
    # corr index: lag = idx - (n-1)
    
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        idx = lag + (n - 1)
        w_lag = w[idx]
        if w_lag < 1e-10:
            continue
        # Contribution to grad from this lag
        for j in range(n):
            # h[j] appears in c(lag) as:
            # - as h[j] * (1-h[j+lag]) if 0 <= j+lag < n
            # - as -h[j-lag] * h[j] if 0 <= j-lag < n (from the (1-h) term)
            if 0 <= j + lag < n:
                grad[j] += w_lag * (1.0 - h[j + lag]) * dx
            if 0 <= j - lag < n:
                grad[j] -= w_lag * h[j - lag] * dx
    return grad

def projected_subgradient(h, n, steps=5000, lr_init=0.01, momentum=0.9, tau=0.01, top_taus=None):
    """Projected subgradient method with momentum and adaptive step size."""
    h = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    v = np.zeros(n)
    
    for it in range(steps):
        # Compute gradient
        g = subgradient(h, n, tau=tau)
        
        # Momentum update
        v = momentum * v + g
        h_new = h - lr_init * v
        
        # Project
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
        
        # Adapt step size (simple decay)
        lr_decay = 1.0 / (1.0 + it * 0.001)
        # Actually let me just use fixed lr with decay
        
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        lo, hi = -0.5, 0.5
        for _ in range(40):
            mid = (lo + hi) / 2.0
            s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s_mid < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    return h

def soft_grad(h, w, n, dx):
    u = 1.0 - h
    A = np.correlate(u, w, mode='full')[n-1:2*n-1]
    B = np.convolve(w, h, mode='full')[:n]
    return dx * (A - B)

def grad_optimize(h, n, max_time, tau_init=0.05, tau_final=0.001, lr=0.02, mom=0.9):
    t0 = time.time()
    best_h = h.copy()
    best_val = compute_c5(h)
    v = np.zeros(n)
    dx = 2.0 / n
    it = 0
    # Anneal: tau from tau_init to tau_final over max_time
    while time.time() - t0 < max_time:
        # Current tau (linear in it or time)
        frac = min(1.0, (time.time() - t0) / max_time)
        tau = tau_init * (tau_final / tau_init) ** frac
        
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        c_max = np.max(corr)
        w = np.exp((corr - c_max) / tau)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.full(2*n-1, 1.0/(2*n-1))
        else:
            w /= w_sum
        
        grad = soft_grad(h, w, n, dx)
        v = mom * v + grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, n)
        
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
        it += 1
        # Adapt lr: decrease slowly
        lr = max(0.001, 0.02 / (1.0 + it * 0.0005))
    
    return best_h, best_val

def slsqp_polish(h, n, max_time):
    t0 = time.time()
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)
    
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        m = np.max(corr)
        t = 0.005
        return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx
    
    def cons_eq(hv):
        return np.sum(hv) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': cons_eq, 'jac': lambda x: np.ones(n)},)
    
    for attempt in range(3):
        if time.time() - t0 > max_time - 5:
            break
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 200, 'ftol': 1e-10})
            h_new = project_feasible(res.x, n)
            val = compute_c5(h_new)
            if val < best_val:
                best_val = val
                best_h = h_new
        except:
            break
    
    return best_h, best_val

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Get initial
    if 'initial_h_values' in globals() and initial_h_values is not None:

        n0 = len(h0)
    else:

        h0 = np.full(n0, 0.5)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Gradient optimization at n0
    if remaining() > 200:
        h_opt, c5_opt = grad_optimize(h0.copy(), n0, max_time=min(150, remaining()-20),
                                       tau_init=0.05, tau_final=0.001, lr=0.02, mom=0.85)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Phase 2: SLSQP polish
    if remaining() > 60:
        h_pol, c5_pol = slsqp_polish(best_h.copy(), len(best_h), max_time=min(40, remaining()-10))
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
    
    # Phase 3: Try different n values
    for target_n in [100, 120, 200, 240]:
        if remaining() < 80:
            break
        if target_n == len(best_h):
            continue
        h_up = upsample(best_h, target_n) if target_n > len(best_h) else upsample(best_h, target_n)
        h_opt, c5_opt = grad_optimize(h_up, target_n, max_time=min(80, remaining()-30),
                                       tau_init=0.03, tau_final=0.001, lr=0.015, mom=0.85)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if remaining() > 30:
            h_pol, c5_pol = slsqp_polish(best_h.copy(), target_n, max_time=min(25, remaining()-15))
            if c5_pol < best_c5:
                best_h, best_c5 = h_pol, c5_pol
    
    # Phase 4: Random restarts
    n_cur = len(best_h)
    restart = 0
    while remaining() > 40 and restart < 30:
        h_pert = best_h + 0.05 * np.random.randn(n_cur)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(25):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        lo, hi = -1.0, 1.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s_mid < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    return h

def soft_gradient(h, w, n, dx):
    u = 1.0 - h
    A = np.correlate(u, w, mode='full')[n-1:2*n-1]
    B = np.convolve(w, h, mode='full')[:n]
    return dx * (A - B)

def grad_optimize(h, n, max_time, tau_init=0.05, tau_final=0.0005, lr_init=0.02, mom=0.85):
    t0 = time.time()
    best_h = h.copy()
    best_val = compute_c5(h)
    v = np.zeros(n)
    dx = 2.0 / n
    it = 0
    while time.time() - t0 < max_time - 1:
        frac = min(1.0, (time.time() - t0) / max_time)
        tau = tau_init * (tau_final / tau_init) ** (frac ** 0.7)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        c_max = np.max(corr)
        scaled = (corr - c_max) / tau
        scaled -= np.max(scaled)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = w / w_sum
        grad = soft_gradient(h, w, n, dx)
        v = mom * v + grad
        lr = lr_init / (1.0 + it * 0.001)
        h_new = h - lr * v
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        it += 1
    return best_h, best_val

def slsqp_polish(h, n, max_time):
    t0 = time.time()
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)

    def obj_fn(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        m = np.max(corr)
        t = 0.003
        return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx

    def cons_fn(x):
        return np.sum(x) - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': cons_fn, 'jac': cons_jac}
    for attempt in range(4):
        if time.time() - t0 > max_time - 3:
            break
        try:
            h_start = best_h.copy()
            if attempt > 0:
                h_start = h_start + 0.02 * np.random.randn(n)
                h_start = project_feasible(h_start, n)
            res = minimize(obj_fn, h_start, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 300, 'ftol': 1e-12})
            h_new = project_feasible(res.x, n)
            val = compute_c5(h_new)
            if val < best_val:
                best_val = val
                best_h = h_new
        except Exception:
            break
    return best_h, best_val

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def spectral_init(n, seed):
    np.random.seed(seed)
    g = np.zeros(n)
    freqs = np.fft.fftfreq(n)
    F = np.fft.fft(g)
    F_real = np.random.randn(n) + 1j * np.random.randn(n)
    F_real[0] = 0
    h = 0.5 + np.real(np.fft.ifft(F_real)) / np.max(np.abs(np.fft.ifft(F_real))) * 0.4
    return project_feasible(h, n)

def binary_init(n, seed):
    np.random.seed(seed)
    h = np.zeros(n)
    idx = np.random.permutation(n)[:n // 2]
    h[idx] = 1.0
    return h

def two_pulse_init(n, seed, gap_frac=0.3):
    np.random.seed(seed)
    h = np.zeros(n)
    half = n // 4
    gap = int(n * gap_frac)
    pos1 = np.random.randint(0, max(1, n - 2 * half - gap))
    pos2 = (pos1 + half + gap) % n
    for i in range(half):
        h[(pos1 + i) % n] = 1.0
    for i in range(half):
        h[(pos2 + i) % n] = 1.0
    # Fill remaining to get n/2 ones
    need = n // 2 - 2 * half
    if need > 0:
        remaining = np.where(h == 0)[0]
        extra = np.random.choice(remaining, size=min(need, len(remaining)), replace=False)
        h[extra] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def _rem():
        return budget_s - (time.time() - t0)

    # Get initial construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    else:
        n0 = 160
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradient optimization at initial n
    if _rem() > 200:
        h_opt, c5_opt = grad_optimize(h0.copy(), n0, max_time=min(160, _rem() - 20),
                                      tau_init=0.08, tau_final=0.0003, lr_init=0.025, mom=0.82)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: SLSQP polish
    if _rem() > 50:
        h_pol, c5_pol = slsqp_polish(best_h.copy(), len(best_h), max_time=min(40, _rem() - 10))
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    # Phase 3: Try alternative initializations
    alts = [
        spectral_init(n0, seed + 1),
        binary_init(n0, seed + 2),
        two_pulse_init(n0, seed + 3, 0.25),
        two_pulse_init(n0, seed + 4, 0.4),
    ]
    for i, h_alt in enumerate(alts):
        if _rem() < 60:
            break
        h_opt, c5_opt = grad_optimize(h_alt, n0, max_time=min(50, _rem() - 20),
                                      tau_init=0.04, tau_final=0.001, lr_init=0.015, mom=0.88)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Try different n values
    for target_n in [120, 200, 240, 300]:
        if _rem() < 70:
            break
        if target_n == len(best_h):
            continue
        h_up = upsample_h(best_h, target_n)
        h_opt, c5_opt = grad_optimize(h_up, target_n, max_time=min(70, _rem() - 30),
                                      tau_init=0.04, tau_final=0.0005, lr_init=0.015, mom=0.85)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if _rem() > 30:
            h_pol, c5_pol = slsqp_polish(best_h.copy(), target_n, max_time=min(25, _rem() - 15))
            if c5_pol < best_c5:
                best_h, best_c5 = h_pol, c5_pol

    # Phase 5: Random restarts at best resolution
    n_cur = len(best_h)
    restart = 0
    while _rem() > 35 and restart < 40:
        scale = 0.03 + 0.06 * (restart % 5) / 4.0
        h_pert = best_h + scale * np.random.randn(n_cur)
        h_pert = project_feasible(h_pert, n_cur)
        max_t = min(25, _rem() - 15)
        if max_t < 10:
            break
        h_opt, c5_opt = grad_optimize(h_pert, n_cur, max_time=max_t,
                                      tau_init=0.015, tau_final=0.0005, lr_init=0.01, mom=0.92)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        restart += 1

    # Phase 6: Final polish
    if _rem() > 15:
        h_pol, c5_pol = slsqp_polish(best_h.copy(), len(best_h), max_time=max(10, _rem() - 5))
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    return best_h, best_c5, len(best_h)
