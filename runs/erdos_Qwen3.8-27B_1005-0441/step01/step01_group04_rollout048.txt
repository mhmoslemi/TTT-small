def _fix_sum(h, n):
    """Fix sum to n/2, clip to [0,1]."""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.minimum(h, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.maximum(h, 0.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            i = int(np.argmin(h))
        else:
            i = int(np.argmax(h))
        h[i] += diff
        h = np.clip(h, 0, 1)
    return h

        g[lag:] -= h[:n-lag]
    else:
        ll = -lag
        g[:n-ll] -= h[ll:]
        g[ll:] += (1.0 - h[:n-ll])
    g *= dx
    return g

def _cd_pass(h, n, order=None):
    h = h.copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    if order is None:
        order = list(range(n))
    
    for i in order:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        # Compute slopes for all lags
        lags = np.arange(ncorr) - (n - 1)  # actual lag values
        # For coordinate i, slope at lag k:
        # s_k = dx * [(1-h[i+k]) if 0<=i+k<n else 0 - h[i-k] if 0<=i-k<n else 0]
        
        # Vectorize:
        i_plus = i + lags  # positions for the "first" factor
        i_minus = i - lags  # positions for the "second" factor
        valid_plus = (i_plus >= 0) & (i_plus < n)
        valid_minus = (i_minus >= 0) & (i_minus < n)
        
        s = np.zeros(ncorr)
        s[valid_plus] += (1.0 - h[i_plus[valid_plus]])
        s[valid_minus] -= h[i_minus[valid_minus]]
        s *= dx
        
        active = np.abs(s) > 1e-14
        if not np.any(active):
            continue
        sa = s[active]
        ia = corr[active] - h[i] * sa
        na = len(sa)
        
        # 1D minimax: min_x max_k (ia[k] + sa[k]*x) for x in [0,1]
        f0 = np.max(ia)
        f1 = np.max(ia + sa)
        bx = 0.0
        bf = min(f0, f1)
        if f1 < f0:
            bx = 1.0
        
        # Vectorized pairwise intersection
        if na > 2:
            pi, pj = np.triu_indices(na, k=1)
            ds = sa[pi] - sa[pj]
            valid = np.abs(ds) > 1e-14

                    mi = np.argmin(fv)
                    if fv[mi] < bf:
                        bf = fv[mi]
                        bx = xv[mi]
        
        h[i] = bx
    
    return h

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break

        else:

            break
        h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def _cd_pass(h, n, order=None):

                        bx = xv[mi]

def _subgrad_step(h, beta, step):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(corr)
    w = np.exp(np.clip(beta * (corr - cm), -400, 400))
    w /= np.sum(w)
    ncorr = len(corr)
    lags = np.arange(ncorr) - (n - 1)
    g = np.zeros(n)
    for k in range(ncorr):
        if w[k] < 1e-12:
            continue
        lag = lags[k]
        if lag >= 0:
            g[:n-lag] += w[k] * dx * (1.0 - h[lag:])
            g[lag:] -= w[k] * dx * h[:n-lag]
        else:
            ll = -lag
            g[:n-ll] -= w[k] * dx * h[ll:]
            g[ll:] += w[k] * dx * (1.0 - h[:n-ll])
    g -= np.mean(g)
    h_new = np.clip(h - step * g, 0.0, 1.0)
    return _fix_sum(h_new, n)
