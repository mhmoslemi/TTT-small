    best_h = h.copy()
    best_c5 = compute_c5(h)

        for k_star in top_idx:
            m = k_star - (n - 1)
            for i in range(n):
                j1 = i - m
                if 0 <= j1 < n:
                    grad[i] += (1.0 - h[j1])
                j2 = i + m
                if 0 <= j2 < n:
                    grad[i] -= h[j2]
        grad /= topk
        h = h - lr * grad
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr = max(lr * 0.9995, 0.002)
    return best_h, best_c5

def smooth_obj(h, n, p):
    corr = compute_corr(h) * (2.0 / n)
    return float(np.sum(corr**p) ** (1.0 / p))

def constraint_fun(x):
    return x.sum() - len(x) / 2.0

def constraint_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, t0, deadline, p_start=20):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    for p in [p_start, p_start * 4, p_start * 16]:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h, args=(n, p), method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def full_pipeline(h, n, rng, t0, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # Binary round + local search
    h_bin = round_binary(h, n)
    h_b, c5_b = binary_swap_search(h_bin, n, 2000, rng, t0,
                                   min(deadline, t0 + 8))
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
    # Subgradient
    h_sg, c5_sg = subgradient_optimize(best_h.copy(), n, t0,
                                       min(deadline, t0 + 30), n_iter=15000, topk=3)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()
    # SLSQP
    h_sl, c5_sl = slsqp_optimize(best_h.copy(), n, t0, deadline, p_start=20)
    if c5_sl < best_c5:
        best_c5 = c5_sl
        best_h = h_sl.copy()
    # One more subgradient round
    h_sg2, c5_sg2 = subgradient_optimize(best_h.copy(), n, t0,
                                         min(deadline, t0 + 20), n_iter=10000, topk=5)
    if c5_sg2 < best_c5:
        best_c5 = c5_sg2
        best_h = h_sg2.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.full(160, 0.5)

    n = len(h_start)
    h_start = project_feasible(h_start, n)

    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)

    # Main pipeline on initial
    deadline1 = t0 + budget_s * 0.30
    h1, c5_1 = full_pipeline(h_start.copy(), n, rng, t0, deadline1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Restart with perturbation
    for rnd in range(20):
        deadline = t0 + budget_s * 0.88
        if time.time() > deadline:
            break
        if rnd < 4:
            h_pert = best_h + rng.randn(n) * 0.03
        elif rnd < 8:
            h_pert = best_h + rng.randn(n) * 0.07
        elif rnd < 12:
            h_pert = best_h + (rng.rand(n) - 0.5) * 0.15
        else:
            block = max(1, n // rng.randint(4, 10))
            h_pert = np.zeros(n)
            for b in range(0, n, block):
                h_pert[b:b+block] = 1.0 if rng.rand() > 0.5 else 0.0
            h_pert += rng.randn(n) * 0.1
        h_pert = project_feasible(h_pert, n)
        dl = min(deadline, t0 + budget_s * 0.06)
        h_r, c5_r = full_pipeline(h_pert, n, rng, t0, dl)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Final polish
    if time.time() < t0 + budget_s * 0.93:
        h_f, c5_f = slsqp_optimize(best_h.copy(), n, t0,
                                   t0 + budget_s * 0.96, p_start=40)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    try:
        val = evaluate_erdos_solution(best_h)
        if val < final_c5:
            final_c5 = val
    except Exception:
        pass

    return (best_h, final_c5, n)
