                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = _fix_sum(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def _rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n_fallback = 160
        h0 = np.full(n_fallback, 0.5)

    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Fine-grained SLSQP temperature annealing
    temps = [0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
    for temp in temps:
        if _rem() < 40:
            break
        for it in range(3):
            if _rem() < 15:
                break
            try:
                h_new = _slsqp_round(best_h, n, temp, 400, target)
                v = compute_c5(h_new)
                if v < best_c5 - 1e-13:
                    best_h = h_new
                    best_c5 = v
            except Exception:
                break

    # Phase 2: Local refinement via pairwise swaps
    if _rem() > 30:
        h_new, v = _local_refine(best_h, 8000, rng, n)
        if v < best_c5:
            best_h = h_new
            best_c5 = v

    # Phase 3: Perturb and re-optimize (multiple restarts)
    restart_scales = [0.08, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005]
    for scale in restart_scales:
        if _rem() < 50:
            break
        h_pert = _perturb(best_h, scale, rng)
        h_pert = _fix_sum(h_pert, target)
        for temp in [0.001, 0.0003]:
            if _rem() < 20:
                break
            try:
                h_new = _slsqp_round(h_pert, n, temp, 300, target)
                v = compute_c5(h_new)
                if v < best_c5 - 1e-13:
                    best_h = h_new
                    best_c5 = v
                    h_pert = h_new.copy()
            except Exception:
                pass

    # Phase 4: Additional local refinement
    if _rem() > 20:
        h_new, v = _local_refine(best_h, 6000, rng, n)
        if v < best_c5:
            best_h = h_new
            best_c5 = v

    # Phase 5: Final deep SLSQP polish
    if _rem() > 25:
        for temp in [1e-05, 5e-06]:
            if _rem() < 10:
                break
            try:
                h_new = _slsqp_round(best_h, n, temp, 500, target)
                v = compute_c5(h_new)
                if v < best_c5 - 1e-14:
                    best_h = h_new
                    best_c5 = v
            except Exception:
                pass

    # Phase 6: Very fine local search
    if _rem() > 10:
        h_new, v = _local_refine(best_h, 4000, rng, n)
        if v < best_c5:
            best_h = h_new
            best_c5 = v

    # Final fix
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _fix_sum(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
