import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += d / mask.sum()
        else:
            h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_lse(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def obj_pen(h, alpha, pen):
    n = len(h)
    base = obj_lse(h, alpha)
    return base + pen * (np.sum(h) - n * 0.5) ** 2

def corrg_k(h, k, n):
    """Return d corr[k]/d h for np.correlate(h, 1-h, 'full')[k]"""
    grad = np.zeros(n)
    for i in range(n):
        j1 = i + k - n + 1
        j2 = i - k + n - 1
        v = 0.0
        if 0 <= j1 < n:
            v += (1.0 - h[j1])
        if 0 <= j2 < n:
            v -= h[j2]
        grad[i] = v
    return grad

def subgrad_polish(h, n, max_iters, t0, budget_s, frac):
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    lr = 0.01
    for it in range(max_iters):
        if time.time() - t0 > budget_s * frac:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        g = corrg_k(h, k_star, n) * dx
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break
    return h, best_c5

def make_sinusoidal(n, freq, amp):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + amp * np.sin(np.pi * freq * x)
    return project_feasible(np.clip(h, 0, 1), n)

def make_two_block(n):
    h = np.zeros(n)
    b = max(1, n // 4)
    h[:b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = make_two_block(160)
    n = len(h0)
    h_best = project_feasible(h0, n)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with penalty (fast, good initial progress)
    for alpha, pen in [(30.0, 2000.0), (100.0, 5000.0), (300.0, 10000.0)]:
        if time.time() - t0 > budget_s * 0.20:
            break
        res = minimize(obj_pen, h_best.copy(), args=(alpha, pen),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 1500, 'ftol': 1e-16, 'gtol': 1e-10})
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 2: SLSQP with equality constraint, increasing alpha
    def sum_eq(x):
        return np.sum(x) - n * 0.5
    def sum_jac(x):
        return np.ones(len(x))
    constraint = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]

    for alpha in [200.0, 600.0, 1500.0, 3000.0]:
        if time.time() - t0 > budget_s * 0.45:
            break
        res = minimize(obj_lse, h_best.copy(), args=(alpha,),
                       method='SLSQP', bounds=bounds, constraints=constraint,
                       options={'maxiter': 3000, 'ftol': 1e-15, 'maxls': 50})
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 3: Subgradient polish on the true max
    if time.time() - t0 < budget_s * 0.60:
        h_sg, c5_sg = subgrad_polish(h_best, n, 2000, t0, budget_s, 0.60)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 4: SLSQP from diverse starting points
    if time.time() - t0 < budget_s * 0.75:
        starts = []
        for freq in [2, 3, 4, 5]:
            starts.append(make_sinusoidal(n, freq, 0.4))
        starts.append(make_two_block(n))
        for _ in range(4):
            p = h_best + (0.03 + 0.03 * rng.random()) * rng.standard_normal(n)
            starts.append(project_feasible(p, n))

        for sp in starts:
            if time.time() - t0 > budget_s * 0.74:
                break
            for alpha in [400.0, 1000.0]:
                if time.time() - t0 > budget_s * 0.74:
                    break
                res = minimize(obj_lse, sp.copy(), args=(alpha,),
                               method='SLSQP', bounds=bounds, constraints=constraint,
                               options={'maxiter': 1500, 'ftol': 1e-14})
                hc = project_feasible(res.x, n)
                c5c = compute_c5(hc)
                if c5c < c5_best:
                    c5_best = c5c
                    h_best = hc.copy()

    # Phase 5: Subgradient polish again
    if time.time() - t0 < budget_s * 0.85:
        h_sg, c5_sg = subgrad_polish(h_best, n, 3000, t0, budget_s, 0.85)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    # Phase 6: Try at n-40 resolution (fewer variables, potentially better local opt)
    if time.time() - t0 < budget_s - 100 and n > 80:
        n_small = n - 40
        x_old = np.arange(n, dtype=float)
        x_new = np.linspace(0.0, n - 1, n_small)
        h_s = np.interp(x_new, x_old, h_best)
        h_s = project_feasible(h_s, n_small)
        bounds_s = [(0.0, 1.0)] * n_small
        def sum_eq_s(x):
            return np.sum(x) - n_small * 0.5
        def sum_jac_s(x):
            return np.ones(len(x))
        constraint_s = [{'type': 'eq', 'fun': sum_eq_s, 'jac': sum_jac_s}]
        for alpha in [300.0, 1000.0]:
            if time.time() - t0 > budget_s - 60:
                break
            res = minimize(obj_lse, h_s.copy(), args=(alpha,),
                           method='SLSQP', bounds=bounds_s, constraints=constraint_s,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            hc = project_feasible(res.x, n_small)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 7: Final SLSQP + subgradient
    if time.time() - t0 < budget_s - 30:
        n_final = len(h_best)
        if n_final != n:
            def _sq(x):
                return np.sum(x) - n_final * 0.5
            def _sj(x):
                return np.ones(len(x))
            constraint_f = [{'type': 'eq', 'fun': _sq, 'jac': _sj}]
            bounds_f = [(0.0, 1.0)] * n_final
        else:
            constraint_f = constraint
            bounds_f = bounds
        res = minimize(obj_lse, h_best.copy(), args=(5000.0,),
                       method='SLSQP', bounds=bounds_f, constraints=constraint_f,
                       options={'maxiter': 2000, 'ftol': 1e-15})
        hc = project_feasible(res.x, n_final)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()
        h_sg, c5_sg = subgrad_polish(h_best, n_final, 1500, t0, budget_s, 0.98)
        if c5_sg < c5_best:
            c5_best = c5_sg
            h_best = h_sg

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
