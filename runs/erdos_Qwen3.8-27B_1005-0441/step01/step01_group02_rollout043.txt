import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _smooth_obj(h, beta):
    corr = _corr_vals(h)
    mc = corr.max()
    s = np.sum(np.exp(beta * (corr - mc)))
    return (mc + np.log(s) / beta) * (2.0 / len(h))

def _smooth_grad(h, beta):
    corr = _corr_vals(h)
    n = len(h)
    mc = corr.max()
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2*n-1):
        if w[m] < 1e-15:
            continue
        k = m - (n-1)
        wk = w[m]
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wk * (1.0 - h[:n-kk])
            grad[:n-kk] -= wk * h[kk:]
    return grad * (2.0/n)

def _project(h, target):
    """Project h onto {x: 0≤x≤1, sum(x)=target}"""
    n = len(h)
    # Bisection on lambda
    lo = h.min() - target/n - 1.0
    hi = h.max() - target/n + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Fix any tiny residual
    diff = target - p.sum()
    if abs(diff) > 1e-12:
        free = (p > 1e-10) & (p < 1.0 - 1e-10)
        if free.sum() > 0:
            p[free] += diff / free.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def _pg_solve(h0, beta, n_iter, t_max):
    """Projected gradient with backtracking line search."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _smooth_obj(h, beta)
    alpha = 1.0
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > t_max:
            break
        g = _smooth_grad(h, beta)
        # Normalized gradient
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        # Step
        step = alpha / gn  # or alpha directly
        h_new = _project(h - step * g, target)
        val_new = _smooth_obj(h_new, beta)
        if val_new < best_val - 1e-15:
            h = h_new
            best_h = h.copy()
            best_val = val_new
            alpha = min(alpha * 1.5, 10.0)
        else:
            alpha *= 0.5
            if alpha < 1e-8:
                break
    return best_h, best_val

    mc = corr.max()
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * 2.0 / len(h))

def _slsqp(h0, beta, target, t_limit):
    n = len(h0)
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac, 'args': (target,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_s_obj, h0, args=(beta,), jac=_s_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-13})
        return _project(res.x, target)
    except Exception:
        return _project(h0.copy(), target)

def _pg(h0, beta, target, n_iter, t_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_v = _smooth_val(h, beta)
    alpha = 0.8
    t0 = time.time()
    for _ in range(n_iter):
        if time.time() - t0 > t_limit:
            break
        g = _smooth_grad(h, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = _project(h - (alpha / gn) * g, target)
        v = _smooth_val(h_new, beta)
        if v < best_v - 1e-14:
            h = h_new
            best_h = h.copy()
            best_v = v
            alpha = min(alpha * 1.4, 8.0)
        else:
            alpha *= 0.55
            if alpha < 1e-10:
                break
    return best_h, compute_c5(best_h)

def _to_bin(h):
    n = len(h)
    k = n // 2
    idx = np.argsort(h)[::-1][:k]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _bin_local(h, t_limit):
    h = h.copy()
    t0 = time.time()
    improved = True
    while improved and time.time() - t0 < t_limit:
        improved = False
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        c_cur = compute_c5(h)
        best_d = 1e-13
        best_p = None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                d = c_cur - c
                if d > best_d:
                    best_d = d
                    best_p = (i, j)
        if best_p is not None:
            h[best_p[0]], h[best_p[1]] = 0.0, 1.0
            improved = True
    return h, compute_c5(h)

def _bin_local2(h, t_limit):
    """2-swap local search: move two 1s to two 0 positions."""
    h = h.copy()
    t0 = time.time()
    improved = True
    while improved and time.time() - t0 < t_limit:
        improved = False
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        c_cur = compute_c5(h)
        best_d = 1e-13
        best_quad = None
        ni, nj = len(ones), len(zeros)
        for a in range(ni):
            for b in range(a + 1, ni):
                for c in range(nj):
                    for d in range(c + 1, nj):
                        i1, i2 = ones[a], ones[b]
                        j1, j2 = zeros[c], zeros[d]
                        h[i1] = h[i2] = 0.0
                        h[j1] = h[j2] = 1.0
                        cv = compute_c5(h)
                        h[i1] = h[i2] = 1.0
                        h[j1] = h[j2] = 0.0
                        d = c_cur - cv
                        if d > best_d:
                            best_d = d
                            best_quad = (i1, i2, j1, j2)
        if best_quad is not None:
            i1, i2, j1, j2 = best_quad
            h[i1] = h[i2] = 0.0
            h[j1] = h[j2] = 1.0
            improved = True
    return h, compute_c5(h)

def _nm_obj(x, target):
    return compute_c5(_project(x, target))

def _nm_polish(h0, target, t_limit, maxiter):
    try:
        res = minimize(_nm_obj, h0, args=(target,), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
        return _project(res.x, target)
    except Exception:
        return _project(h0.copy(), target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 70
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0

    n = len(h_init)
    target = n / 2.0
    h = _project(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing beta
    for beta in [8, 25, 60, 150, 400]:
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 20:
            break
        h = _slsqp(h, beta, target, min(55, t_rem - 3))
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 2: Projected gradient
    for beta in [200, 500, 1500]:
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 15:
            break
        h, c = _pg(h, beta, target, 500, min(45, t_rem - 3))
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 3: Binary local search (single swaps)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 25:
        hb = _to_bin(best_h)
        c_b = compute_c5(hb)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = hb.copy()
        hb, c_b = _bin_local(hb, min(200, t_rem - 10))
        if c_b < best_c5:
            best_c5 = c_b
            best_h = hb.copy()

    # Phase 4: 2-swap local search
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        hb, c_b = _bin_local2(best_h, min(180, t_rem - 10))
        if c_b < best_c5:
            best_c5 = c_b
            best_h = hb.copy()
        # Follow up with single swaps
        hb, c_b = _bin_local(hb, min(80, budget_s - (time.time() - t_start) - 10))
        if c_b < best_c5:
            best_c5 = c_b
            best_h = hb.copy()

    # Phase 5: Continuous refinement from best binary
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 25:
        h = _slsqp(best_h, 100, target, min(50, t_rem - 10))
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        t_rem = budget_s - (time.time() - t_start)
        if t_rem > 20:
            h, c = _pg(h, 500, target, 300, min(35, t_rem - 5))
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()

    # Phase 6: Nelder-Mead polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 15:
        h = _nm_polish(best_h, target, t_rem - 3, min(3000, int(4 * (t_rem - 3))))
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    # Phase 7: Random perturbation restarts
    t_rem = budget_s - (time.time() - t_start)
    n_restarts = min(int(t_rem / 80), 4) if t_rem > 80 else 0
    for _ in range(n_restarts):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 70:
            break
        hb = _to_bin(best_h)
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        n_swaps = rng.randint(2, 5)
        for _ in range(n_swaps):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            hb[i], hb[j] = hb[j], hb[i]
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
        hb, c_b = _bin_local(hb, min(50, t_rem - 35))
        if c_b < best_c5:
            best_c5 = c_b
            best_h = hb.copy()
        t_rem = budget_s - (time.time() - t_start)
        if t_rem > 30:
            hc = _slsqp(hb, 80, target, min(25, t_rem - 10))
            cc = compute_c5(hc)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()

    # Final polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 10:
        h = _nm_polish(best_h, target, t_rem - 2, 2000)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()

    best_h = _project(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, n)
