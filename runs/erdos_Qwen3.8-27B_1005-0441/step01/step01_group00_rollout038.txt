import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _smoobj(h, eps, n):
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * (2.0 / n)

            g[:n - m] -= wk * h[m:]
        else:

                best_t = t

            h[q] -= best_t

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _smoobj(h, eps, n):
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * (2.0 / n)

def _smograd(h, eps, n):
