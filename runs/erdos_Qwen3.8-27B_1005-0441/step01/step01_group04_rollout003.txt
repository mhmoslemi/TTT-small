        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

        if diff > 0:
            # Add to entries with most room
            room = 1.0 - h
            mask = room > 1e-10

            sub = min(-diff, np.sum(room[mask]))

from scipy.optimize import minimize

def _subgrad_all_lags(h, n):
    """Compute subgradients for all lags. Returns array of shape (2n-1, n)."""
    dx = 2.0 / n
    k_center = n - 1
    grads = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - k_center
        g = np.zeros(n)
        # d/dh[j] = dx * [(1-h[j+k]) * 1_{0<=j+k<n} - h[j-k] * 1_{0<=j-k<n}]
        jp = np.arange(n) + k
        vp = (jp >= 0) & (jp < n)
        g[vp] += (1.0 - h[jp[vp]])
        jm = np.arange(n) - k
        vm = (jm >= 0) & (jm < n)
        g[vm] -= h[jm[vm]]
        grads[m] = dx * g
    return grads

def _subgrad_at_lags(h, lags, n):
    """Compute average subgradient over given lags (indices into full corr)."""
    dx = 2.0 / n
    k_center = n - 1
    g = np.zeros(n)
    for m in lags:
        k = m - k_center
        jp = np.arange(n) + k
        vp = (jp >= 0) & (jp < n)
        g[vp] += (1.0 - h[jp[vp]])
        jm = np.arange(n) - k
        vm = (jm >= 0) & (jm < n)
        g[vm] -= h[jm[vm]]
    return dx * g / len(lags)

def _softmax_obj_h(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def _softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    weights = np.exp(alpha * shifted)
    weights /= weights.sum()
    # Gradient of softmax-approximated max
    # d/dh[j] of cmax + log(sum exp(alpha*(corr_m - cmax)))/alpha
    # = (1 - sum_m weights_m * 1_{m=argmax}) * d(cmax)/dh[j] 
    #   + sum_m weights_m * d(corr_m)/dh[j] (approximately, ignoring cmax in denominator)
    # Simplification: just use weighted average of subgradients
    top5 = np.argsort(corr)[-5:]
    # Use weights only for top lags
    w = weights[top5]
    w /= w.sum()
    k_center = n - 1
    g = np.zeros(n)
    for idx, m in enumerate(top5):
        k = m - k_center
        jp = np.arange(n) + k
        vp = (jp >= 0) & (jp < n)
        g[vp] += w[idx] * (1.0 - h[jp[vp]])
        jm = np.arange(n) - k
        vm = (jm >= 0) & (jm < n)
        g[vm] -= w[idx] * h[jm[vm]]
    return dx * g

    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _softmax_obj_h(x, alpha)
    
    def eq(x):
        return np.sum(x) - n / 2.0
    
    try:

                       constraints=[{'type': 'eq', 'fun': eq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.success or res.fun < _softmax_obj_h(h0, alpha):
            return _project_sum(res.x, n)
    except:
        pass
    return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_sum(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_avg(h, lags, n):
    dx = 2.0 / n
    kc = n - 1
    g = np.zeros(n)
    for m in lags:
        k = m - kc
        jp = np.arange(n) + k
        vp = (jp >= 0) & (jp < n)
        g[vp] += (1.0 - h[jp[vp]])
        jm = np.arange(n) - k
        vm = (jm >= 0) & (jm < n)
        g[vm] -= h[jm[vm]]
    return dx * g / len(lags)

def _softmax_obj(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(corr)
    return cm + np.log(np.sum(np.exp(alpha * (corr - cm)))) / alpha

        if res.fun < _softmax_obj(h0, alpha):
            return _project_sum(res.x, n)
    except Exception:
        pass
    return h0.copy()
