import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            n_add = min(n, max(1, int(abs(diff)) + 1))
            idx = np.argsort(-room)[:n_add]
            per = diff / len(idx)
            for i in idx:
                a = min(per, room[i])
                h[i] += a
                diff -= a
        else:
            idx = np.argsort(h)[:max(1, int(-diff) + 1)]
            per = -diff / len(idx)
            for i in idx:
                s = min(per, h[i])
                h[i] -= s
                diff += s
    return h

def _softmax_obj_factory(alpha):
    def obj(h, *args):
        n = len(h)
        dx = 2.0 / n

        cmax = np.max(corr)
        shifted = np.clip((corr - cmax) * alpha, -60, 0)
        return cmax + np.log1p(np.sum(np.exp(shifted))) / alpha
    return obj

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):

                h[i] += a
                diff -= a
