def best_swap(h, n, overlaps):
    """Find the best single swap to reduce max overlap."""
    current_max = np.max(overlaps)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    best_val = current_max
    best_pair = None
    
    for i in ones:
        for j in zeros:
            # Compute max of (overlaps + change) 
            # change(k) = -(1-h[i+k]) - h[i-k] + (1-h[j+k]) + h[j-k]
            # (with out-of-range = 0)
            max_new = 0.0
            for k in range(-(n-1), n):
                delta = 0.0
                ik = i + k
                if 0 <= ik < n:
                    delta -= (1 - h[ik])
                ik2 = i - k
                if 0 <= ik2 < n:
                    delta -= h[ik2]
                jk = j + k
                if 0 <= jk < n:
                    delta += (1 - h[jk])
                jk2 = j - k
                if 0 <= jk2 < n:
                    delta += h[jk2]
                val = overlaps[n-1+k] + delta
                if val > max_new:
                    max_new = val
            if max_new < best_val:
                best_val = max_new
                best_pair = (i, j)
    
    return best_pair, best_val

import time

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def project_feasible(h, n):
    """Project h onto {x: 0≤x[i]≤1, sum(x)=n/2} using bisection on threshold."""
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix tiny numerical issues
    diff = target - np.sum(hp)
    if abs(diff) > 1e-13:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def overlap_grad(h, n):
    """Gradient of the overlap vector w.r.t. h.
    overlap[k] = Σ_i h[i] * (1-h[i+k]) * dx
    d(overlap[k])/d(h[j]) = (1-h[j-k])*dx (if j-k valid) + (-h[j+k])*dx (if j+k valid)
    """
    dx = 2.0 / n
    grad = np.zeros((n, 2*n-1))  # grad[j, n-1+k] = d(overlap[k])/d(h[j])
    for j in range(n):
        for k in range(-(n-1), n):
            g = 0.0
            # Term where j is the "h[i]" factor: (1-h[j+k])
            jk = j + k
            if 0 <= jk < n:

            # Term where j is the "(1-h[i+k])" factor: -h[j-k]
            jk2 = j - k
            if 0 <= jk2 < n:
                g += (-h[jk2])
            grad[j, n-1+k] = g * dx
    return grad

                jk2 = j - lag

Actually, let me think about this differently. The key issue with the previous code was that it only looked at ONE lag at a time (the argmax). The soft-max approach looks at ALL lags simultaneously, weighted by how close they are to the max. This should give a much better gradient signal.

Let me also add a "pair flip" step: for the worst overlap, find the pair (i, j) where moving mass from i to j (or vice versa) most reduces the worst overlap.

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-13:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def compute_grad(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(-(n-1), n):
            wk = w[n-1+k]
            if wk < 1e-13:
                continue
            jk = j + k
            if 0 <= jk < n:
                s += wk * (1.0 - h[jk])
            jk2 = j - k
            if 0 <= jk2 < n:
                s += wk * (-h[jk2])
        grad[j] = dx * s
    return grad

def optimize_softmax(h_init, n, deadline, rng, tau_init=0.04):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    lr = 1.0
    tau = tau_init
    stalls = 0
    no_improve = 0
    
    while time.time() < deadline:
        ov = get_overlaps(h, n)
        max_ov = np.max(ov)
        
        # Top-k softmax for focus
        top_k = 5
        idx = np.argsort(ov)[-top_k:]
        w = np.zeros(2*n-1)
        top_vals = ov[idx]
        max_tv = np.max(top_vals)
        exps = np.exp((top_vals - max_tv) / tau)
        w[idx] = exps / np.sum(exps)
        
        grad = compute_grad(h, n, w)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = get_c5(h_new, n)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stalls = 0
            no_improve = 0
            lr = min(5.0, lr * 1.01)
        else:
            stalls += 1
            no_improve += 1
        
        h = h_new
        
        if stalls > 40:
            h = best_h + rng.normal(0, 0.08, n)
            h = project_feasible(h, n)
            lr = max(0.1, lr * 0.5)
            stalls = 0
        
        if no_improve > 500:
            # Bigger restart
            h = best_h + rng.normal(0, 0.2, n)
            h = project_feasible(h, n)
            lr = 1.5
            no_improve = 0
            tau = max(0.01, tau * 0.8)
    
    return best_h, best_c5
