        if int(h.sum()) != n // 2:

            h = np.zeros(n)
            h[::2] = 1.0
    return h

def to_binary(h_frac, n):
    target = n // 2
    rank = np.argsort(h_frac)[::-1]
    h = np.zeros(n)
    h[rank[:target]] = 1.0
    return h
