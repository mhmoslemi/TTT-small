= i+(-i), i+(-i+1), ..., i+(n-1-i) = 0, 1, ..., n-1-i
So it's (1-h)[0 : n-i].

And w indices: d + n - 1 for d from -i to n-1-i
= -i+n-1, -i+n, ..., n-1-i+n-1 = n-1-i, n-i, ..., 2n-2-i

So:

Wait, let me recheck the w index range for the first term:
d from -i to n-1-i, w index = d + n - 1
Start: -i + n - 1 = n-1-i
End: (n-1-i) + n - 1 = 2n-2-i
Number of values: (n-1-i) - (-i) + 1 = n - i
w indices: n-1-i, n-i, ..., 2n-2-i (that's n-i values)

For the second term:
d from i-n+1 to i, w index = d + n - 1
Start: i-n+1+n-1 = i
End: i+n-1
Number of values: i-(i-n+1)+1 = n
w indices: i, i+1, ..., i+n-1 (that's n values)

So:
grad[i] = dot(w[n-1-i : 2n-1-i], (1-h)[0:n-i]) - dot(w[i : i+n], h[::-1])

Wait, in numpy, w[a:b] gives elements a, a+1, ..., b-1. So:
First term: w[n-1-i : 2n-1-i] has elements at indices n-1-i, ..., 2n-2-i (that's n-i elements). ✓
Second term: w[i : i+n] has elements at indices i, ..., i+n-1 (that's n elements). ✓

And (1-h)[0:n-i] has n-i elements: (1-h)[0], (1-h)[1], ..., (1-h)[n-i-1]. ✓
h[::-1] has n elements. ✓

So the gradient can be computed as:
```python
def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    omh = 1.0 - h
    rev_h = h[::-1]
    grad = np.zeros(n)
    for i in range(n):
        # First term: dot of w[n-1-i : 2n-1-i] with omh[0 : n-i]
        grad[i] += np.dot(w[n-1-i : 2*n-1-i], omh[0 : n-i])
        # Second term: dot of w[i : i+n] with rev_h
        grad[i] -= np.dot(w[i : i+n], rev_h)
    return grad * dx
```

This is O(n²) but with small constant (just dot products). For n=70, it's 70 dot products of average length ~70, so ~5000 multiply-adds. Very fast.

    corr = np.correlate(h, 1.0 - h, mode="full")

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    omh = 1.0 - h
    rev_h = h[::-1]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] += np.dot(w[n-1-i:2*n-1-i], omh[0:n-i])
        grad[i] -= np.dot(w[i:i+n], rev_h)
    return grad * dx

    n = len(h0)
    target = n / 2.0
    def obj(x):
        return smooth_max_val(x, beta)
    def jac(x):
        return smooth_max_grad(x, beta)
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_h(res.x)
    except Exception:
        return project_h(h0.copy())

def to_binary(h):
    """Convert to binary with exactly n/2 ones."""
    n = len(h)
    target = n // 2
    # Threshold at 0.5, adjust if needed
    h_bin = (h >= 0.5).astype(float)
    cnt = int(h_bin.sum())
    if cnt > target:
        # Need to flip some 1s to 0s; prefer those closest to 0.5 from above
        ones = np.where(h_bin == 1.0)[0]
        # Sort by h value (ascending - closest to threshold first)
        order = np.argsort(h[ones])
        for p in ones[order[:cnt-target]]:
            h_bin[p] = 0.0
    elif cnt < target:
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros])
        for p in zeros[order[:target-cnt]]:
            h_bin[p] = 1.0
    return h_bin

def compute_base_deltas(h, n):
    base = np.zeros((n, 2*n-1))
    for p in range(n):
        for d in range(-(n-1), n):
            k = d + n - 1
            val = 0.0
            if 0 <= p + d < n:
                val -= (1.0 - h[p + d])
            if 0 <= p - d < n:
                val += h[p - d]
            base[p, k] = val
    return base

def binary_swap_search(h_bin, max_time=60.0):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    t0 = time.time()
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_max = np.max(corr)
    
    while time.time() - t0 < max_time:
        base = compute_base_deltas(h, n)
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        sweep_improved = False
        
        for i in ones:
            for j in zeros:
                # delta_corr = base[i] - base[j]
                new_max = np.max(corr + base[i] - base[j])
                if new_max < best_max - 1e-12:
                    # Apply swap
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    best_max = np.max(corr)
                    sweep_improved = True
                    # Recompute base after swap
                    base = compute_base_deltas(h, n)
                    ones = list(np.where(h == 1.0)[0])
                    zeros = list(np.where(h == 0.0)[0])
                    break  # restart inner loop with new state
            if sweep_improved:
                break
        
        if not sweep_improved:
            break
    
    return h

def gradient_descent(h0, beta, lr=0.01, max_iter=2000, t_limit=30.0):
    n = len(h0)
    h = h0.copy()
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > t_limit:
            break
        g = smooth_max_grad(h, beta)
        # Projected gradient step
        h_new = h - lr * g
        h_new = project_h.h  # project
        h_new = project_h(h_new)
        c_old = smooth_max_val(h, beta)
        c_new = smooth_max_val(h_new, beta)
        if c_new < c_old:
            h = h_new
            # Decrease lr slightly
            lr *= 0.999
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def smooth_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    omh = 1.0 - h
    rev_h = h[::-1]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] += np.dot(w[n - 1 - i:2 * n - 1 - i], omh[0:n - i])
        grad[i] -= np.dot(w[i:i + n], rev_h)
    return grad * dx

def slsqp_phase(h0, beta, maxiter=200):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return smooth_max_val(x, beta)
    def jac(x):
        return smooth_max_grad(x, beta)
    def cf(x):
        return x.sum() - target
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_h(res.x)
    except Exception:
        return project_h(h0.copy())

def to_binary(h):
    n = len(h)
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    cnt = int(h_bin.sum())
    if cnt > target:
        ones = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones])
        for p in ones[order[:cnt - target]]:
            h_bin[p] = 0.0
    elif cnt < target:
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros])
        for p in zeros[order[:target - cnt]]:
            h_bin[p] = 1.0
    return h_bin

def compute_base_deltas(h, n):
    base = np.zeros((n, 2 * n - 1))
    for p in range(n):
        for d in range(-(n - 1), n):
            k = d + n - 1
            val = 0.0
            if 0 <= p + d < n:
                val -= (1.0 - h[p + d])
            if 0 <= p - d < n:
                val += h[p - d]
            base[p, k] = val
    return base

def binary_swap_search(h_bin, max_time=30.0):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    t0 = time.time()
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_max = float(np.max(corr))
    while time.time() - t0 < max_time:
        base = compute_base_deltas(h, n)
        improved = False
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        for idx_i in range(len(ones)):
            i = ones[idx_i]
            for j in zeros:
                nm = np.max(corr + base[i] - base[j])
                if nm < best_max - 1e-12:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    best_max = float(np.max(corr))
                    improved = True
                    base = compute_base_deltas(h, n)
                    ones = list(np.where(h == 1.0)[0])
                    zeros = list(np.where(h == 0.0)[0])
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_max * dx

def continuous_polish(h0, t_budget=40.0):
    n = len(h0)
    h = project_h(h0.copy())
    best_c = compute_c5(h)
    hb = h.copy()
    t0 = time.time()
    for beta in [20.0, 40.0, 80.0, 150.0, 300.0]:
        if time.time() - t0 > t_budget - 5:
            break
        h = slsqp_phase(h, beta, maxiter=120)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            hb = h.copy()
    if time.time() - t0 < t_budget - 3:
        try:
            def nm_obj(x):
                return compute_c5(project_h(x))
            nm_res = minimize(nm_obj, hb.copy(), method='Nelder-Mead',
                             options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
            h_nm = project_h(nm_res.x)
            c_nm = compute_c5(h_nm)
            if c_nm < best_c:
                best_c = c_nm
                hb = h_nm
        except Exception:
            pass
    return hb, best_c

def fine_tune(h0, t_budget=20.0):
    n = len(h0)
    h = project_h(h0.copy())
    best_c = compute_c5(h)
    t0 = time.time()
    steps = [0.01, 0.005, 0.002, 0.001]
    for step in steps:
        if time.time() - t0 > t_budget:
            break
        for _ in range(3):
            for j in range(n):
                if time.time() - t0 > t_budget:
                    break
                for sign in [1, -1]:
                    try_val = h[j] + sign * step
                    if try_val < 0 or try_val > 1:
                        continue
                    k = (j + 1) % n
                    while k == j:
                        k = (k + 1) % n
                    comp = -sign * step
                    try_k = h[k] + comp
                    if try_k < 0 or try_k > 1:
                        continue
                    h_try = h.copy()
                    h_try[j] = try_val
                    h_try[k] = try_k
                    c_try = compute_c5(h_try)
                    if c_try < best_c - 1e-13:
                        h = h_try
                        best_c = c_try
                        break
    return project_h(h), best_c
