                    b = m2
                else:
                    a = m1

    # Initialize: top n/2 of h_init
    h = np.zeros(n)
    top = np.argsort(h_init)[-n//2:]
    h[top] = 1.0

        c = corr

        w = np.exp(c / T)
        w /= np.sum(w)

        # g[i] = sum_m w[m] * h[i + m - (n-1)]
        # = correlate(h, w) with appropriate indexing
        # np.correlate(h, w, 'full')[k] = sum_i h[i] * w[i + k - (n-1)]...

            m_max = min(2*n-2, n-1+i)
            if m_min <= m_max:

            T *= 0.8
            # Don't break immediately, try with lower T
            if T < 0.01:
                break
        else:
            h = new_h
            T *= 0.99

def _slsq_polish(h, max_iter=200):
    n = len(h)
    dx = 2.0 / n
    alpha = 100.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = np.max(corr)
        shifted = corr - cmax
        return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success or res.fun < _eval_c5(h):
            return res.x
    except:
        pass
    return h.copy()

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h

def _coord_descent(h, n_sweeps=200, seed=0):
