def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _softmax_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    F = raw * dx
    Fmax = np.max(F)
    sh = F - Fmax
    es = np.exp(alpha * sh)
    W = es / np.sum(es)
    obj = Fmax + np.log(np.sum(es)) / alpha
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for idx in range(len(F)):
        w = W[idx] * dx
        if w < 1e-14:
            continue
        l = lags[idx]
        lo1, hi1 = max(0, l), min(n, n + l)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1-l:hi1-l])
        lo2, hi2 = max(0, -l), min(n, n - l)
        if hi2 > lo2:
            grad[lo2:hi2] += w * (-h[lo2+l:hi2+l])
    return obj, grad

def _pgd_optimize(h, alphas, t0, budget_s, min_time=20):

def _lbbg_obj(x, n, alpha, mu):
