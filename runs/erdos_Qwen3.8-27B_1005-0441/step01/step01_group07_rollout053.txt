def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

            no_improve = 0

def bal_binary(h_init, budget, seed):
    """Simulated annealing with swaps"""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    tgt = n // 2
    h = (h_init >= 0.5).astype(float)
    d = tgt - int(h.sum())
    if d > 0:
        z = np.where(h==0)[0]; rng.shuffle(z)
        h[z[:d]] = 1
    elif d < 0:
        o = np.where(h==1)[0]; rng.shuffle(o)
        h[o[:d]] = 0
    bh, bc = h.copy(), compute_c5(h)
    cc = bc
    T, T_min, cool = 0.015, 0.001, 0.997
    t0 = time.time()
    while time.time() < t0 + budget - 0.2:
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)<1 or len(zeros)<1: break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0, 1
        nc = compute_c5(h)
        if nc - cc < 0 or rng.random() < math.exp(-(nc-cc)/T):
            cc = nc
            if cc < bc: bh, bc = h.copy(), cc
        else:
            h[i], h[j] = 1, 0
        # Multi-swap
        if rng.random() < 0.1 and len(ones)>=2 and len(zeros)>=2:
            ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
            i1,i2 = ones[rng.randint(len(ones))], ones[rng.randint(len(ones))]
            j1,j2 = zeros[rng.randint(len(zeros))], zeros[rng.randint(len(zeros))]
            if i1!=i2 and j1!=j2:
                bak = h[[i1,i2,j1,j2]].copy()
                h[i1],h[i2],h[j1],h[j2] = 0,0,1,1
                nc2 = compute_c5(h)
                if nc2-cc < 0 or rng.random() < math.exp(-(nc2-cc)/T):
                    cc = nc2
                    if cc < bc: bh, bc = h.copy(), cc
                else:
                    h[i1],h[i2],h[j1],h[j2] = bak
        T = max(T*cool, T_min)
    return bh, bc

Same pattern! And the sum of errors is 0.

OK so the gradient has a systematic error, but the error sums to 0. For the optimization, this means the effective gradient is (correct gradient) + (error vector that sums to 0). The projection absorbs the mean, but not the non-constant part.

However, this error is relatively small (±1) compared to the actual gradient values (which are weighted sums of correlation contributions). And the optimization is stochastic anyway, so this small bias might not matter much in practice.

            if j >= lag and j - lag <= n-1-lag:

            if j + L <= n-1:
