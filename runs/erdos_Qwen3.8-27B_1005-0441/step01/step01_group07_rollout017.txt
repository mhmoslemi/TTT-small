            h = project_feasible(best_h + rng.randn(n) * 0.03, target_sum)
            v[:] = 0.0
            lr = lr_init
            tau = tau_start
            no_improve = 0
        elif no_improve > 200:
            lr = max(lr * 0.92, 0.002)

        _, grad = compute_smooth_c5_and_grad(h, tau)
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
            lr = min(lr * 1.02, lr_init)
        h = h_new
    return best_h, best_c5

def scipy_optimize(h_init, time_budget, seed):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_feasible(h_init.copy(), target_sum)
    t0 = time.time()
    deadline = t0 + time_budget
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    for tau in [15.0, 30.0, 60.0, 120.0, 250.0, 500.0]:
        if time.time() > deadline - 5:
            break
        tries = max(1, int((deadline - time.time()) / 6.0))
        for t in range(min(tries, 3)):
            if time.time() > deadline - 2:
                break
            def obj(x):
                h = np.clip(x, 0.0, 1.0)
                c, _ = compute_smooth_c5_and_grad(h, tau)
                c += 200.0 * (h.sum() - target_sum) ** 2
                return c

            h_start = project_feasible(h0 + 0.01 * np.random.RandomState(seed + t).randn(n), target_sum)
            try:
                res = minimize(obj, h_start, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 500, 'maxfun': 2000, 'gtol': 1e-8})
                h_opt = project_feasible(res.x, target_sum)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
    return best_h, best_c5

def local_polish(h_init, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    step = 0.02
    while time.time() < deadline - 0.3:
        idx = rng.randint(n)
        delta = rng.uniform(-step, step)
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            step = min(step * 1.01, 0.05)
        else:
            step = max(step * 0.98, 0.005)
    return best_h, best_c5

def binary_local_search(h_frac, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    target_sum = int(n // 2)
    h = (h_frac >= 0.5).astype(float)
    diff = target_sum - int(h.sum())
    if diff > 0:
        zeros = np.where(h == 0)[0]
        for i in zeros[:diff]:
            h[i] = 1.0
    elif diff < 0:
        ones = np.where(h == 1)[0]
        for i in ones[:(-diff)]:
            h[i] = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    while time.time() < deadline - 0.3:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def make_structured_init(n, kind=0):
    target = n / 2.0
    if kind == 0:
        h = np.full(n, 0.5)
    elif kind == 1:
        h = np.zeros(n)
        block = max(1, n // 14)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif kind == 2:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3.0 * np.arange(n) / n)
    else:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * 5.0 * np.arange(n) / n + 1.0)
    return project_feasible(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    best_h = None
    best_c5 = float('inf')
    best_n = 70

    def update(h, c5, n_pts):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_pts

    # Use initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = np.full(70, 0.5)
    n = len(h_init)
    h_init = project_feasible(h_init, n / 2.0)
    update(h_init, compute_c5(h_init), n)

    # Phase 1: scipy L-BFGS-B with smooth max (most impactful)
    t_rem = deadline - time.time()
    if t_rem > 30:
        h_s, c5_s = scipy_optimize(h_init, min(t_rem * 0.35, 380), seed)
        update(h_s, c5_s, n)
        h_init = best_h  # update starting point

    # Phase 2: gradient descent with smooth max, multiple configs
    t_rem = deadline - time.time()
    if t_rem > 40:
        for cfg, t_alloc in [(0, 0.25), (1, 0.2), (2, 0.15)]:
            if time.time() > deadline - 20:
                break
            if cfg == 0:
                h_g, c5_g = grad_descent(best_h, min(t_rem * t_alloc, 150), seed + 10,
                                         tau_start=15.0, tau_max=400.0, lr_init=0.05)
            elif cfg == 1:
                h_g, c5_g = grad_descent(best_h, min(t_rem * t_alloc, 100), seed + 20,
                                         tau_start=8.0, tau_max=200.0, lr_init=0.08)
            else:
                h_g, c5_g = grad_descent(best_h, min(t_rem * t_alloc, 80), seed + 30,
                                         tau_start=5.0, tau_max=150.0, lr_init=0.1)
            update(h_g, c5_g, n)
            h_init = best_h

    # Phase 3: more scipy passes from current best with higher tau
    t_rem = deadline - time.time()
    if t_rem > 20:
        h_s2, c5_s2 = scipy_optimize(best_h, min(t_rem * 0.3, 120), seed + 100)
        update(h_s2, c5_s2, n)

    # Phase 4: local polish (coordinate-level)
    t_rem = deadline - time.time()
    if t_rem > 20:
        h_p, c5_p = local_polish(best_h, min(t_rem * 0.3, 80), seed + 200)
        update(h_p, c5_p, n)

    # Phase 5: binary local search
    t_rem = deadline - time.time()
    if t_rem > 20:
        hb1, c5b1 = binary_local_search(best_h, min(t_rem * 0.25, 50), seed + 300)
        update(hb1, c5b1, n)
        # Also from a perturbed fractional
        h_frac2 = project_feasible(best_h + 0.08 * np.random.RandomState(seed + 350).randn(n), n / 2.0)
        h_frac2, c5_f2 = grad_descent(h_frac2, min(t_rem * 0.15, 40), seed + 360,
                                       tau_start=5.0, tau_max=100.0, lr_init=0.06)
        update(h_frac2, c5_f2, n)
        hb2, c5b2 = binary_local_search(best_h, min(t_rem * 0.15, 40), seed + 400)
        update(hb2, c5b2, n)

    # Phase 6: random restarts from diverse initializations
    for trial in range(6):
        t_rem = deadline - time.time() - 10
        if t_rem < 25:
            break
        rng_t = np.random.RandomState(seed + 500 + trial * 17)
        h_try = project_feasible(best_h + rng_t.randn(n) * 0.06, n / 2.0)
        t_alloc = min(t_rem / (7 - trial), 60)
        h_opt, c5_opt = grad_descent(h_try, t_alloc, seed + 2000 + trial,
                                      tau_start=10.0, tau_max=300.0, lr_init=0.07)
        update(h_opt, c5_opt, n)

    # Phase 7: try different resolutions
    for n_alt in [64, 80, 60]:
        t_rem = deadline - time.time() - 5
        if t_rem < 30:
            break
        h_alt = project_feasible(np.interp(np.linspace(0, n - 1, n_alt),
                                           np.linspace(0, n - 1, n), best_h), n_alt / 2.0)
        h_alt, c5_alt = grad_descent(h_alt, min(t_rem * 0.3, 60), seed + 3000 + n_alt,
                                      tau_start=10.0, tau_max=200.0, lr_init=0.06)
        update(h_alt, c5_alt, n_alt)

    # Phase 8: final tight polish
    t_rem = deadline - time.time() - 1
    if t_rem > 8:
        h_final, c5_final = grad_descent(best_h, min(t_rem - 1, 40), seed + 9999,
                                          tau_start=100.0, tau_max=500.0, lr_init=0.008)
        update(h_final, c5_final, best_n)
    t_rem = deadline - time.time() - 1
    if t_rem > 5:
        h_pol, c5_pol = local_polish(best_h, min(t_rem - 1, 20), seed + 8888)
        update(h_pol, c5_pol, best_n)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
