import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0-h, mode="full")) * dx)

def get_all_corr(h):
    return np.correlate(h, 1.0-h, mode="full")

def project_box_simplex(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def smooth_max_obj(h, beta, dx):
    corr = get_all_corr(h)
    mc = corr.max()
    return (mc + np.log(np.sum(np.exp(beta*(corr-mc))))/beta) * dx

def smooth_max_grad(h, beta, dx):
    corr = get_all_corr(h)
    mc = corr.max()
    w = np.exp(beta*(corr-mc))
    ws = w.sum()
    w /= ws
    n = len(h)
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-16:
            continue
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        g[lo:hi+1] += w[k] * (1.0 - 2.0*h[k-hi:k-lo+1][::-1])
    return g * dx

def optimize_smooth(h0, beta, target, maxiter=200):
    """SLSQP on smooth max objective."""
    n = len(h0)
    dx = 2.0/n
    
    def obj(x):
        return smooth_max_obj(x, beta, dx)
    def jac(x):
        return smooth_max_grad(x, beta, dx)
    def con(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    
    res = minimize(obj, h0, jac=jac, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':con,'jac':con_jac}],
                   options={'maxiter':maxiter,'ftol':1e-14})
    return project_box_simplex(res.x, target)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    n = len(h_init) if h_init is not None else 70
    target = n / 2.0

    if h_init is None:
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0

    h_init = project_bs(h_init, target)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Smooth max annealing
    t_rem = budget_s - (time.time() - t_start)
    betas = [15.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    for beta in betas:
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 45:
            break
        h_new = slsqp_solve(best_h, beta, target, maxiter=250)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Active set line search
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 40:
        h_as, c_as = active_set_ls(best_h, target, top_m=10, max_iter=400,
                                    t_budget=min(70, t_rem - 10), rng=rng)
        if c_as < best_c5:
            best_c5 = c_as
            best_h = h_as

    # Phase 3: Very high beta SLSQP
    t_rem = budget_s - (time.time() - t_start)
    for beta in [800.0, 1500.0]:
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 35:
            break
        h_new = slsqp_solve(best_h, beta, target, maxiter=200)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new
        # Follow with active set
        t_rem = budget_s - (time.time() - t_start)
        if t_rem > 20:
            h_as, c_as = active_set_ls(best_h, target, top_m=5, max_iter=200,
                                        t_budget=min(25, t_rem - 5), rng=rng)
            if c_as < best_c5:
                best_c5 = c_as
                best_h = h_as

    # Phase 4: Structured restarts
    t_rem = budget_s - (time.time() - t_start)
    n_restarts = min(int(t_rem / 55), 5) if t_rem > 55 else 0
    for i in range(n_restarts):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 50:
            break
        scales = [0.06, 0.10, 0.15, 0.20, 0.30]
        h_pert = project_bs(best_h + scales[i % len(scales)] * rng.randn(n), target)
        h_new = slsqp_solve(h_pert, 200.0, target, maxiter=150)
        c_new = compute_c5(h_new)
        t_rem = budget_s - (time.time() - t_start)
        if t_rem > 15:
            h_as, c_as = active_set_ls(h_new, target, top_m=8, max_iter=200,
                                        t_budget=min(25, t_rem - 5), rng=rng)
            if c_as < c_new:
                c_new = c_as
                h_new = h_as
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 5: Pair coordinate descent
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 20:
        h_pc, c_pc = pair_coord_descent(best_h, target, max_rounds=30,
                                         t_budget=min(50, t_rem - 5), rng=rng)
        if c_pc < best_c5:
            best_c5 = c_pc
            best_h = h_pc

    # Phase 6: Nelder-Mead polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 15:
        h_nm, c_nm = nm_polish(best_h, target, t_budget=min(30, t_rem - 5))
        if c_nm < best_c5:
            best_c5 = c_nm
            best_h = h_nm

    # Phase 7: Final active set + coordinate
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 15:
        h_as, c_as = active_set_ls(best_h, target, top_m=3, max_iter=150,
                                    t_budget=min(20, t_rem - 5), rng=rng)
        if c_as < best_c5:
            best_c5 = c_as
            best_h = h_as
        h_cd, c_cd = pair_coord_descent(best_h, target, max_rounds=20,
                                         t_budget=min(15, t_rem - 3), rng=rng)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    best_h = project_bs(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
