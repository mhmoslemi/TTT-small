import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            mask = h < 1 if diff > 0 else h > 0
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for _ in range(n):
            if diff > 0:
                idx = np.argmax(1.0 - h)
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                idx = np.argmin(h)
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _softmax_obj(h, alpha):
    corr = _corr(h)
    cmax = corr.max()
    return cmax + np.log(np.sum(np.exp(alpha * (corr - cmax)))) / alpha

def _lbfgs_optimize(h0, alpha, max_iter=300, ftol=1e-14):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return _softmax_obj(np.clip(x, 0, 1), alpha)
    try:
        res = minimize(f, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10, 'maxcor': 30})
        return np.clip(res.x, 0, 1)
    except:
        return h0.copy()

def _coord_descent(h, n_sweeps=10):
    n = len(h)
    best_c5 = _eval_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            # Try values 0, 0.25, 0.5, 0.75, 1 and current
            candidates = [0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0, h[i]]
            best_val = h[i]

                improved = True
            best_c5 = min(best_c5, _eval_c5(h))
        if not improved:

def _subgradient_optimize(h, n_iter=5000, t0=time.time(), budget=30):
    n = len(h)
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    x = h.astype(float).copy()
    x_prev = x.copy()
    for t in range(1, n_iter):
        if time.time() - t0 > budget:
            break
        # Compute corr and find argmax
        corr = _corr(x)
        k_star = np.argmax(corr)
        # Compute gradient of corr[k_star] w.r.t. x
        u = n - 1 - k_star
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        grad = np.zeros(n)
        # First term: (1-h[u+j]) for j in [lo, hi] where 0<=u+j<n
        for j in range(lo, hi + 1):

            if 0 <= idx1 < n:
                grad[j] += (1.0 - x[idx1])
            idx2 = j - u  # for the second term: -h[j-u] where lo<=j-u<=hi
            if 0 <= idx2 < n and lo <= idx2 <= hi:
                grad[j] -= x[idx2]
        # Step size
