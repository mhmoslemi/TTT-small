**Minimax duality approach**:
At the optimum, there exists a probability distribution q over k such that the gradient of sum_k q_k corr[k] is zero (KKT conditions). We can try to find such a q iteratively.

Algorithm:
1. Start with uniform q
2. Compute g = sum_k q_k grad(corr[k])
3. Update h in direction -g (with projection)
4. Update q to put more weight on the max corr[k]
5. Repeat

This is like a game where one player chooses h and the other chooses k.

def _minimax_iter(h, n, iterations):

        # Use softmax with temperature

    # q has length 2n-1, indexed by p=0,...,2n-2
    # lag_p = p - (n-1)
    nc = 2*n - 1
    lags = np.arange(nc) - (n-1)  # from -(n-1) to n-1
    grad = np.zeros(n)
    
    for l_idx, l in enumerate(lags):
        if q[l_idx] < 1e-10: continue
        w = q[l_idx]
        if l >= 0:
            # (1-h[j-l]) for j >= l
            first = w * (1.0 - h[:n-l])  # h[j-l] for j=l,...,n-1 → h[0],...,h[n-1-l]
            # -h[j+l] for j <= n-1-l
            second = -w * h[l:n]  # h[j+l] for j=0,...,n-1-l → h[l],...,h[n-1]
            grad[l:] += first
            grad[:n-l] += second
        else:
            m = -l
            # (1-h[j+m]) for j+m < n, i.e., j < n-m
            first = w * (1.0 - h[m:n])  # h[j+m] for j=0,...,n-1-m → h[m],...,h[n-1]
            # -h[j-m] for j >= m
            second = -w * h[:n-m]  # h[j-m] for j=m,...,n-1 → h[0],...,h[n-1-m]
            grad[:n-m] += first
            grad[m:] += second
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _soft_obj(h, size, temp):
    dx = 2.0 / size
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * dx

def _proj(h, size):
    h = np.clip(h, 0.0, 1.0)
    tgt = size / 2.0
    for _ in range(40):
        s = np.sum(h)
        d = s - tgt
        if abs(d) < 1e-13:
            break
        if d > 0:
            msk = h > 1e-12
            c = np.sum(msk)
            if c == 0:
                break
            h[msk] = np.maximum(h[msk] - d / c, 0.0)
        else:
            msk = h < 1.0 - 1e-12
            c = np.sum(msk)
            if c == 0:
                break
            h[msk] = np.minimum(h[msk] - d / c, 1.0)
    return np.clip(h, 0.0, 1.0)

def _slsqp(h, size, temp, maxiter, ftol):
    bounds = [(0.0, 1.0)] * size
    def obj(x):
        return _soft_obj(x, size, temp)
    def ceq(x):
        return np.sum(x) - size / 2.0
    def ceqj(x):
        return np.ones(size)

                       constraints=[{'type': 'eq', 'fun': ceq, 'jac': ceqj}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _proj(res.x, size)

        return h.copy()
