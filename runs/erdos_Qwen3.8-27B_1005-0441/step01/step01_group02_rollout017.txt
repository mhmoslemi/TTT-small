import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target):
    """Project x onto {h: 0<=h<=1, sum(h)=target} using bisection."""
    # Find t such that sum(clip(x-t, 0, 1)) = target
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    h = np.clip(x - t, 0, 1)
    # Fix any residual
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute among non-bound variables
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0, 1)
    return h

def smooth_obj(h, beta, lam, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    val = (mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx
    val += lam * (h.sum() - target) ** 2
    return val

def smooth_grad(h, beta, lam, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        d = (n - 1) - i
        if d >= 0:
            idx = np.arange(n - d)
            grad[idx] += wi * (1.0 - h[idx + d])
            grad[idx + d] -= wi * h[idx]
        else:
            d2 = -d
            idx = np.arange(n - d2)
            grad[idx + d2] += wi * (1.0 - h[idx])
            grad[idx] -= wi * h[idx + d2]
    grad *= dx
    grad += 2 * lam * (h.sum() - target)
    return grad

def lbfgs_phase(h, beta, lam, target, maxiter=500):
    n = len(h)
    bounds = [(0, 1)] * n
    res = minimize(smooth_obj, h, jac=lambda x: smooth_grad(x, beta, lam, target),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return project_feasible(res.x, target)

def subgradient_step(h, alpha, target):
    """One subgradient step on actual C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    d = (n - 1) - k_star
    grad = np.zeros(n)
    if d >= 0:
        idx = np.arange(n - d)
        grad[idx] += (1.0 - h[idx + d])
        grad[idx + d] -= h[idx]
    else:
        d2 = -d
        idx = np.arange(n - d2)
        grad[idx + d2] += (1.0 - h[idx])
        grad[idx] -= h[idx + d2]
    grad *= dx
    h_new = h - alpha * (grad - grad.mean())  # maintain sum constraint
    return project_feasible(h_new, target)

    n = len(h)
    best = compute_c5(h)
    for sw in range(sweeps):
        imp = False
        for j in range(n):
            for delta in [0.008, -0.008, 0.015, -0.015, 0.004, -0.004, 0.02, -0.02]:
                kt = (j + n // 3 + sw) % n
                if kt == j:
                    kt = (j + 1) % n
                ht = h.copy()
                ht[j] += delta
                ht[kt] -= delta
                if 0 <= ht[j] <= 1 and 0 <= ht[kt] <= 1:
                    c = compute_c5(ht)
                    if c < best - 1e-14:
                        h = ht
                        best = c
                        imp = True
                        break
        if not imp:
            break
        h = project_feasible(h, target)
        best = compute_c5(h)
    return h, best

def multi_start_optimize(h0, target, t_budget, rng):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t_chk = time.time()
    # Anneal with L-BFGS-B
    for b in [3, 8, 16, 32, 64, 128, 256, 512]:
        if time.time() - t_chk > t_budget - 15:
            break
        h = lbfgs_opt(best_h, b, 0.003, target, mi=400)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    # SLSQP at intermediate beta
    if time.time() - t_chk < t_budget - 10:
        h = slsqp_opt(best_h, 60, target, mi=200)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    # Subgradient
    if time.time() - t_chk < t_budget - 8:
        h = subgrad_step(best_h.copy(), target, 500)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    # NM polish
    if time.time() - t_chk < t_budget - 5:
        h = nm_polish(best_h.copy(), target, 2000)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    # Coordinate descent
    if time.time() - t_chk < t_budget - 2:
        h, c = coord_desc(best_h.copy(), target, 150)
        if c < best_c5:
            best_c5 = c
            best_h = h
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 70
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0

    n = len(h0)
    target = n / 2.0
    h_init = project_feasible(h0.copy(), target)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Main optimization from initial (400s budget)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 80:
        h1, c1 = multi_start_optimize(h_init, target, min(380, t_rem), rng)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1

    # Phase 2: Perturbed restarts (varied magnitude)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 60:
        for mag in [0.08, 0.12, 0.2, 0.3]:
            t_rem = budget_s - (time.time() - t_start)
            if t_rem < 50:
                break
            h_p = project_feasible(best_h + mag * rng.randn(n), target)
            hp, cp = multi_start_optimize(h_p, target, min(45, t_rem - 5), rng)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp

    # Phase 3: Structured diverse starts
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 50:
        # Sinusoidal
        x = np.linspace(0, 2, n)
        h_sin = np.clip(0.5 + 0.5 * np.cos(2 * np.pi * 3 * x / 2), 0, 1)
        h_sin = project_feasible(h_sin, target)
        hs, cs = multi_start_optimize(h_sin, target, min(40, t_rem - 5), rng)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs

    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 50:
        # Alternating blocks
        h_blk = np.zeros(n)
        h_blk[::3] = 1.5
        h_blk = project_feasible(h_blk, target)
        hb, cb = multi_start_optimize(h_blk, target, min(40, t_rem - 5), rng)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb

    # Phase 4: Binary snap + re-optimize
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 40:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, target)
        # Ensure exactly n/2 ones
        if h_bin.sum() > target:
            order = np.argsort(h_bin - 0.5)[::-1]
            for idx in order:
                if h_bin.sum() <= target:
                    break
                h_bin[idx] = 0.0
        else:
            order = np.argsort(h_bin - 0.5)
            for idx in order:
                if h_bin.sum() >= target:
                    break
                h_bin[idx] = 1.0
        hbs, cbs = multi_start_optimize(h_bin, target, min(35, t_rem - 5), rng)
        if cbs < best_c5:
            best_c5 = cbs
            best_h = hbs

    # Phase 5: Final intensive polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        # L-BFGS at very high beta
        h = lbfgs_opt(best_h, 1000, 0.001, target, mi=600)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    if t_rem > 20:
        h = nm_polish(best_h.copy(), target, 4000)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    if t_rem > 10:
        h, c = coord_desc(best_h.copy(), target, 300)
        if c < best_c5:
            best_c5 = c
            best_h = h

    # Phase 6: Tiny perturbations + local search
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 10:
        for _ in range(3):
            if time.time() - t_start > budget_s - 3:
                break
            h_try = project_feasible(best_h + 0.01 * rng.randn(n), target)
            h_try = nm_polish(h_try, target, 500)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try

    best_h = project_feasible(best_h, target)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
