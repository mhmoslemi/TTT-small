    # Negative lags: corr[n-1-L] += h[i-L]-1 for L=1,...,i

    #   h[j] contributes to corr[n-1-L] iff j >= L (i.e., L <= j)

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def frac_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def binary_optimize(h_bin, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    cur_max = float(np.correlate(h, 1.0 - h, mode='full').max())
    best_max = cur_max
    best_h = h.copy()
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    pert_count = 0

    while time.time() < deadline - 1.0:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if no_improve > 200 + pert_count * 100:
            n_flip = rng.randint(4, min(15, min(len(ones), len(zeros)) + 1))
            oi = rng.choice(ones, size=n_flip, replace=False)
            oz = rng.choice(zeros, size=n_flip, replace=False)
            h[oi] = 0.0
            h[oz] = 1.0
            cur_max = float(np.correlate(h, 1.0 - h, mode='full').max())
            no_improve = 0
            pert_count += 1
            continue

        best_nm = cur_max + 1e-9
        best_i = best_j = -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nm = float(np.correlate(h, 1.0 - h, mode='full').max())
                h[i] = 1.0
                h[j] = 0.0
                if nm < best_nm:
                    best_nm = nm
                    best_i, best_j = i, j

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            cur_max = best_nm
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                pert_count = 0
            no_improve = 0
        else:
            found = False
            attempts = 80
            if len(ones) >= 2 and len(zeros) >= 2:
                for _ in range(attempts):
                    o2 = rng.choice(ones, 2, replace=False)
                    z2 = rng.choice(zeros, 2, replace=False)
                    h[o2[0]] = 0.0; h[o2[1]] = 0.0
                    h[z2[0]] = 1.0; h[z2[1]] = 1.0
                    nm = float(np.correlate(h, 1.0 - h, mode='full').max())
                    h[o2[0]] = 1.0; h[o2[1]] = 1.0
                    h[z2[0]] = 0.0; h[z2[1]] = 0.0
                    if nm < best_max:
                        h[o2[0]] = 0.0; h[o2[1]] = 0.0
                        h[z2[0]] = 1.0; h[z2[1]] = 1.0
                        cur_max = nm
                        best_max = nm
                        best_h = h.copy()
                        pert_count = 0
                        no_improve = 0
                        found = True
                        break
            if not found:
                no_improve += 1

    return best_h, best_max * dx

def structured_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    golden = 0.618033988749895
    vals = np.array([(i * golden) % 1.0 for i in range(n)])
    idx = np.argsort(vals)[::-1]
    h[idx[:target]] = 1.0
    return h

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    h_dst = project_feasible(h_dst, n_dst / 2.0)
    return h_dst

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
    else:
        h_init = np.full(70, 0.5)
    n0 = len(h_init)
    h_bin0 = frac_to_binary(h_init, n0)

    best_h = h_bin0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive binary optimization on initial n
    t_rem = deadline - time.time() - 200
    t_alloc = min(t_rem, 600)
    if t_alloc > 10:
        hb, cb = binary_optimize(h_bin0, t_alloc, seed)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    # Phase 2: Binary optimize from a perturbed version
    t_rem = deadline - time.time() - 180
    if t_rem > 15:
        rng_p = np.random.RandomState(seed + 100)
        h_pert = best_h.copy()
        ones_p = np.where(h_pert == 1.0)[0]
        zeros_p = np.where(h_pert == 0.0)[0]
        n_fl = min(8, len(ones_p), len(zeros_p))
        fi = rng_p.choice(ones_p, n_fl, replace=False)
        fz = rng_p.choice(zeros_p, n_fl, replace=False)
        h_pert[fi] = 0.0
        h_pert[fz] = 1.0
        t_alloc2 = min(t_rem, 200)
        hb2, cb2 = binary_optimize(h_pert, t_alloc2, seed + 200)
        if cb2 < best_c5:
            best_h, best_c5 = hb2, cb2

    # Phase 3: Try other n values
    for trial_n in [72, 64, 80, 74, 68]:
        t_rem = deadline - time.time() - 30
        if t_rem < 40:
            break
        n_try = trial_n
        h_try = interpolate_h(best_h, len(best_h), n_try)
        h_try_bin = frac_to_binary(h_try, n_try)
        t_alloc3 = min(t_rem, 120)
        t_rem_after = t_alloc3
        hb3, cb3 = binary_optimize(h_try_bin, t_alloc3, seed + 500 + trial_n)
        if cb3 < best_c5:
            best_h, best_c5 = hb3, cb3
            # Also try structured init for this n
            t_rem = deadline - time.time() - 20
            if t_rem > 30:
                h_struct = structured_init(n_try, seed + 900 + trial_n)
                t_alloc4 = min(t_rem, 80)
                hb4, cb4 = binary_optimize(h_struct, t_alloc4, seed + 1100 + trial_n)
                if cb4 < best_c5:
                    best_h, best_c5 = hb4, cb4
        else:
            # Also try structured init even if interpolation didn't help
            t_rem = deadline - time.time() - 20
            if t_rem > 30:
                h_struct = structured_init(n_try, seed + 700 + trial_n)
                t_alloc4 = min(t_rem, 60)
                hb4, cb4 = binary_optimize(h_struct, t_alloc4, seed + 1300 + trial_n)
                if cb4 < best_c5:
                    best_h, best_c5 = hb4, cb4

    # Phase 4: Final polish on best solution
    t_rem = deadline - time.time() - 2
    if t_rem > 10:
        hb_final, cb_final = binary_optimize(best_h, min(t_rem, 60), seed + 7777)
        if cb_final < best_c5:
            best_h, best_c5 = hb_final, cb_final

    final_c5 = compute_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
