    if elapsed() < budget_s - 250:
        ho = _opt_sm(best_h, 800)
        ho = _fix_sum(ho, n)
        c = _eval_c5(ho)
        if c < best_c5:
            best_c5, best_h = c, ho

    # Phase 2: SLSQP softmax with high alpha
    _SM_ALPHA = 500
    if elapsed() < budget_s - 200:
        ho = _opt_sm(best_h, 1000)
        ho = _fix_sum(ho, n)
        c = _eval_c5(ho)
        if c < best_c5:
            best_c5, best_h = c, ho

    # Phase 3: SLSQP minimax (direct max reduction)
    if elapsed() < budget_s - 180:
        ho = _opt_mm(best_h, 1500)
        ho = _fix_sum(ho, n)
        c = _eval_c5(ho)
        if c < best_c5:
            best_c5, best_h = c, ho

    # Phase 4: Softmax with very high alpha
    _SM_ALPHA = 1000
    if elapsed() < budget_s - 150:
        ho = _opt_sm(best_h, 800)
        ho = _fix_sum(ho, n)
        c = _eval_c5(ho)
        if c < best_c5:
            best_c5, best_h = c, ho

    # Phase 5: Another minimax round
    if elapsed() < budget_s - 120:
        ho = _opt_mm(best_h, 2000)
        ho = _fix_sum(ho, n)
        c = _eval_c5(ho)
        if c < best_c5:
            best_c5, best_h = c, ho

    # Phase 6: Perturbation + re-optimize cycle
    for scale in [0.03, 0.015, 0.008, 0.004]:
        for trial in range(4):
            if elapsed() > budget_s - 80:
                break
            hp = _perturb(best_h, rng, scale)
            _SM_ALPHA = 500
            ho = _opt_sm(hp, 400)
            ho = _fix_sum(ho, n)
            c = _eval_c5(ho)
            if c < best_c5:
                best_c5, best_h = c, ho
            if elapsed() > budget_s - 70:
                break
            ho2 = _opt_mm(best_h, 600)
            ho2 = _fix_sum(ho2, n)
            c2 = _eval_c5(ho2)
            if c2 < best_c5:
                best_c5, best_h = c2, ho2

    # Phase 7: Try neighboring n values
    for n_try in [n - 4, n + 4, n - 8, n + 8]:
        if n_try < 40 or n_try > 200:
            continue
        if elapsed() > budget_s - 60:
            break
        h_interp = _interp_h(best_h, n_try)
        _SM_ALPHA = 500
        ho = _opt_sm(h_interp, 500)
        ho = _fix_sum(ho, n_try)
        c = _eval_c5(ho)
        if c < best_c5:

            if c2 < best_c5:

    # Phase 8: Final polish with highest alpha + minimax
    _SM_ALPHA = 2000
    if elapsed() < budget_s - 30:
        ho = _opt_sm(best_h, 600)
        ho = _fix_sum(ho, len(ho))
        c = _eval_c5(ho)

        ho = _fix_sum(ho, len(ho))

        if c < best_c5:
            best_c5, best_h = c, ho

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
