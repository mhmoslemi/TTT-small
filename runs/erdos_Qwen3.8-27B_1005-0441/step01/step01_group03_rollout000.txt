    return v

def optimize_gd(h, n, deadline, rng):
    """Gradient descent with momentum on smoothed max."""
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    tau = 0.005
    lr = 0.4
    beta = 0.7
    momentum = np.zeros(n)
    stall = 0
    it = 0
    
    while time.time() < deadline:
        ovals = get_overlaps(h, n)
        c_cur = float(np.max(ovals))

            best_c5 = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        max_o = np.max(ovals)
        weights = np.exp((ovals - max_o) / tau)
        weights /= np.sum(weights)
        
        g = compute_gradient(h, n, weights)
        momentum = beta * momentum + (1 - beta) * g
        h_new = h - lr * momentum
        h_new = project_feasible(h_new, n)
        
        c_new = get_c5(h_new, n)
        if c_new < c_cur:
            h = h_new
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            h_new = h - lr * momentum
            h_new = project_feasible(h_new, n)
            h = h_new
        
        tau = max(tau * 0.9997, 0.0003)
        
        if stall > 300:
            h = best_h + rng.randn(n) * 0.08
            h = project_feasible(h, n)
            momentum = np.zeros(n)
            lr = 0.4
            tau = 0.005
            stall = 0

    best_h = h.copy()

    stall = 0
    it = 0

        v = fw_vertex(g, n)

            h = project_feasible(h, n)

def coord_descent(h, n, deadline, rng):

O(ℓ) = dx * Σ_j h[j] * (1 - h[j+ℓ]) where both indices are valid

For the derivative w.r.t. h[i]:
- h[i] appears in the term where j = i: h[i] * (1 - h[i+ℓ]) → derivative (1-h[i+ℓ])*dx if 0 ≤ i+ℓ < n

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-10:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_overlaps(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def compute_gradient(h, n, weights):
    dx = 2.0 / n
    g = np.zeros(n)

        w = weights[m]
