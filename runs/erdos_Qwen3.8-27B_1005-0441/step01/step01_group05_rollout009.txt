import time

_G_N = 160
_G_TEMP = 0.005

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (0.5 * n) / s
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    t = max(_G_TEMP, 1e-10)
    sm = m + t * np.log(np.sum(np.exp((corr - m) / t)))
    return sm * (2.0 / _G_N)

def _eqfun(x):
    return np.sum(x) - 0.5 * _G_N

def _eqjac(x):
    return np.ones(_G_N)

def _slsqp_opt(h_init, max_iters=3, max_time=60):
    global _G_N, _G_TEMP
    n = len(h_init)
    _G_N = n
    best = _project(h_init.copy())
    best_val = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _eqfun, 'jac': _eqjac},)
    t0 = time.time()
    
    temps = [0.05, 0.02, 0.005, 0.001, 0.0003]
    for temp in temps:
        if time.time() - t0 > max_time - 5:
            break
        _G_TEMP = temp
        h = best.copy()
        for _ in range(max_iters):
            if time.time() - t0 > max_time - 2:
                break

                             constraints=cons, options={'maxiter': 200, 'ftol': 1e-11})
                h_new = _project(res.x)
                v = compute_c5(h_new)
                if v < best_val - 1e-12:
                    best = h_new
                    best_val = v
                h = h_new

                break

    n = len(h_init)
    h = _project(h_init.copy())
    best = h.copy()
    best_val = compute_c5(h)
    t0 = time.time()
    step = c / (n * 0.5)  # initial step size

        if time.time() - t0 > max_time:
            break
        # Compute gradient numerically
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        t = 0.001
        weights = np.exp((corr - m) / t)
        weights /= weights.sum()

        # corr[k] = sum_i h[i] * (1-h)[i+k-(n-1)]

                continue

            best_val = v_new
            step = min(step * 1.05, 10.0 / n)
        else:
            step *= 0.7
        
        h = h_new

        v = 1.0 - h

        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
        n_bh += 1
        rem = budget_s - (time.time() - t0)

    # Phase 4: Basin hopping with projected gradient (faster, more iterations)
    rem = budget_s - (time.time() - t0)
    n_pg = 0
    while rem > 35 and n_pg < 40:
        if n_pg % 5 == 4:
            h_p = _block_perturb(best_h, rng)
        else:
            scale = 0.03 + 0.09 * (n_pg % 7) / 6.0
            h_p = _project(best_h + scale * rng.randn(n))
        mt = min(12, rem - 15)
        if mt < 6:
            break
        h_o, c5_o = _pg_step(h_p, 250, 3.0 / n, mt, 0.002)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
        n_pg += 1
        rem = budget_s - (time.time() - t0)

    # Phase 5: Try alternative n values
    rem = budget_s - (time.time() - t0)
    for target_n in [120, 200, 240]:
        rem = budget_s - (time.time() - t0)
        if rem < 55:
            break
        if target_n == n:
            continue
        h_up = _upsample(best_h, target_n)
        h_o, c5_o = _slsqp(h_up, [0.05, 0.01, 0.003, 0.0005], min(45, rem - 20), 2)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
            n = target_n

    # Phase 6: Fine basin hopping at best resolution
    rem = budget_s - (time.time() - t0)
    n_fine = 0
    while rem > 30 and n_fine < 20:
        scale = 0.02 + 0.05 * (n_fine % 5) / 4.0
        h_p = _project(best_h + scale * rng.randn(n))
        mt = min(15, rem - 15)
        if mt < 8:
            break
        h_o, c5_o = _pg_step(h_p, 300, 2.0 / n, mt, 0.001)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o
        n_fine += 1
        rem = budget_s - (time.time() - t0)

    # Phase 7: Final SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        h_o, c5_o = _slsqp(best_h.copy(), [0.003, 0.0005, 0.0001], max(10, rem - 5), 3)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 8: Ultra-fine final
    rem = budget_s - (time.time() - t0)
    if rem > 8:
        h_o, c5_o = _slsqp(best_h.copy(), [0.0001, 0.00003], max(5, rem - 2), 2)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    best_h = _project(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
