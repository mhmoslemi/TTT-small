import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def soft_max(corr, beta, top_k=None):
    """Smooth max of correlation values using log-sum-exp."""
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c = corr[idx]
    else:
        c = corr
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    return (m + np.log(np.sum(np.exp(shifted))) / beta)

def lbfgs_obj(x, n, beta, top_k, mu):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    val = soft_max(corr, beta, top_k) * dx
    sum_penalty = mu * (np.sum(h) - n/2.0) ** 2
    return val + sum_penalty

def optimize_lbfgs(h_start, n, t0, budget_s, max_time_frac=0.3):
    """Optimize using L-BFGS-B with penalty method."""
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    bounds = [(0.0, 1.0)] * n
    max_time = budget_s * max_time_frac
    
    mu_schedule = [1e3, 1e5, 1e7, 1e8]
    beta_schedule = [5, 20, 100, 500, 2000, 10000]
    top_k_schedule = [None, 20, 10, 5, 5, 3]
    
    for mu in mu_schedule:
        if time.time() - t0 > t0 + max_time:
            break
        for beta, tk in zip(beta_schedule, top_k_schedule):
            if time.time() - t0 > t0 + max_time:
                break
            try:
                res = minimize(lbfgs_obj, h, args=(n, beta, tk, mu),
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    h = best_h.copy()
            except Exception:
                pass
    
    return best_h, best_c5

Actually for a single pair swap, the change in corr_k is:
Δcorr_k = δ[(1-h[i+k]) - h[i-k]] + (-δ)[(1-h[j+k]) - h[j-k]]
= δ[(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]
= δ[A_i - A_j]

where A_m = (1-h[m+k]) - h[m-k] for each position m and shift k.

So corr_k(δ) = corr_k + δ(A_i - A_j)_k

The max over k of (corr_k + δ(A_i - A_j)_k) is a convex piecewise linear function of δ.
Minimizing it over δ ∈ [0, min(h[j], 1-h[i])] (for moving from j to i) or δ ∈ [-(min(h[i], 1-h[j])), 0] (for moving from i to j).

This can be solved by checking all breakpoints, which is O(n) per pair. Total: O(n³) per full pass. For n=320, that's about 33 million operations per pass. Doable but might be slow in Python.

Let me simplify: for each pair, just try δ = ±ε for small ε (like 0.01, 0.05), compute the new C₅, keep if better. This is O(n²) pairs × O(n²) correlation = O(n⁴) which is too slow.

Better: for each index i, try adjusting it (and one other index) by a small amount. Or just do gradient-based steps with the subgradient.

OK let me just go with a clean implementation:

1. SLSQP optimization (improved from before)
2. Subgradient descent polish (fast, O(n*M) per step)
3. A few random restarts

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999999
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def soft_max_val(corr, beta, top_k):
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c = corr[idx]
    else:
        c = corr
    m = c.max()
    s = beta * (c - m)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / beta

def slsqp_obj(x, n, beta, top_k):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return soft_max_val(corr, beta, top_k) * dx

def slsqp_jac(x, n, beta, top_k):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c = corr[idx]
    else:
        idx = np.arange(len(corr))
        c = corr
    m = c.max()
    exps = np.exp(beta * (c - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for j, ii in enumerate(idx):
        k = ii - (n - 1)
        wk = w[j] * beta
        # ∂corr_ii/∂h[i] = (1-h[i+k]) - h[i-k] for valid indices
        if k >= 0:
            for i in range(n - k):
                grad[i] += wk * (1.0 - x[i + k])
            for i in range(k, n):
                grad[i] -= wk * x[i - k]
        else:
            kk = -k
            for i in range(kk, n):
                grad[i] += wk * (1.0 - x[i - kk])
            for i in range(0, n - kk):
                grad[i] -= wk * x[i + kk]
    return grad * dx

def constraint_fun(x):
    return x.sum() - len(x) / 2.0

def constraint_jac(x):
    return np.ones(len(x))

def make_bounds(n):
    return [(0.0, 1.0)] * n

def optimize_slsqp(h_start, n, t0, budget_s, time_frac=0.35):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_end = t0 + budget_s * time_frac
    bounds = make_bounds(n)
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    phases = [(3, 30, 80), (10, 10, 40), (30, 5, 30), (100, 5, 25), (500, 3, 20), (2000, 3, 15)]
    for beta, tk, iters in phases:
        if time.time() > t_end:
            break
        try:
            res = minimize(slsqp_obj, h, args=(n, beta, tk), method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': iters, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
        except Exception:
            pass
    return best_h, best_c5

def subgradient_polish(h, n, t0, budget_s, time_frac=0.15, top_k=8):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_end = t0 + budget_s * time_frac
    dx = 2.0 / n
    lr = 0.02
    n_steps = 0
    while time.time() < t_end and n_steps < 5000:
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Soft subgradient
        beta = 200
        if top_k < len(corr):
            idx = np.argpartition(corr, -top_k)[-top_k:]
            c = corr[idx]
        else:
            idx = np.arange(len(corr))
            c = corr
        m = c.max()
        exps = np.exp(beta * (c - m))
        w = exps / exps.sum()
        grad = np.zeros(n)
        for j, ii in enumerate(idx):
            k = ii - (n - 1)
            wc = w[j]
            if k >= 0:
                for i in range(n - k):
                    grad[i] += wc * (1.0 - h[i + k])
                for i in range(k, n):
                    grad[i] -= wc * h[i - k]
            else:
                kk = -k
                for i in range(kk, n):
                    grad[i] += wc * (1.0 - h[i - kk])
                for i in range(0, n - kk):
                    grad[i] -= wc * h[i + kk]
        # Step with decreasing learning rate
        step = lr * (2.0 / (2.0 + n_steps * 0.01))
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            n_steps += 1
        else:
            lr *= 0.95
    return best_h, best_c5
