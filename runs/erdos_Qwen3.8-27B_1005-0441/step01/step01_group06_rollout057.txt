c[k] = sum over j≠i of h[j]*(1-h[j+k-n+1]) + h[i]*(1-h[i+k-n+1]) + (1-h[i])*h[i-k+n-1]
     = const[k] + h[i]*(1-h[i+k-n+1]) + h[i-k+n-1] - h[i]*h[i-k+n-1]

This is quadratic in h[i]! So for each coordinate, we can solve a small quadratic (or rather, find the h[i] that minimizes the max of several quadratics).

    c = np.correlate(h, 1.0 - h, mode="full")

    shifted = beta * (c - m)
    shifted = np.minimum(shifted, 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

    dx = 2.0 / n

    c_max = c.max()
    # Soft weights
    weights = np.exp(-(c[top_indices] - c) / 0.01)
    weights = weights / weights.sum()
    # Actually, just use the top lag gradient
    k_star = top_indices[0]
    # Gradient of c[k_star] w.r.t. h[j]
    grad = np.zeros(n)
    offset = n - 1 - k_star  # i = j - offset means h[j] appears as h[i] in term i

    # d/dh[j]: appears as h[j] in first factor when i=j: (1-h[j+k-n+1])
    #          appears as (1-h[j]) in second factor when i+k-n+1=j, i.e., i=j-k+n-1: h[j-k+n-1]
    for j in range(n):
        g = 0.0
        # First term: h[j] * (1-h[j+k_star-n+1])

            g += (1.0 - h[idx1])

        idx2 = j - k_star + (n - 1)
        if 0 <= idx2 < n:
            g += (-1.0) * h[idx2]
        grad[j] = g
    # Project gradient to satisfy sum constraint
    grad -= np.mean(grad)
    h_new = h - lr * grad
    return project_feasible(h_new, n)
