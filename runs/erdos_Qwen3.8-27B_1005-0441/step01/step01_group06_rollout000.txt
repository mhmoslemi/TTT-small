def project_feasible(h, n):

        if abs(diff) < 1e-13:

            cnt = np.sum(mask)
            if cnt == 0:
                break

            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:

            cnt = np.sum(mask)

def coord_descent_step(h, m):
    """Find optimal h[m] given all other coordinates fixed."""
    n = len(h)
    K = 2 * n - 1  # number of correlation lags
    
    # Compute alpha_lag: coefficient of h[m] in corr[lag]
    # corr[lag_index] = sum_i h[i]*(1-h[i+lag]) where lag_index = lag + n - 1
    # h[m] appears as first factor when 0 <= m < n and 0 <= m+lag < n
    # h[m] appears as second factor (in 1-h[m]) when 0 <= m-lag < n
    
    # For lag in [-m, n-m-1]: h[m] is first factor, coeff = 1-h[m+lag]
    # For lag in [m-n+1, m]: h[m] is second factor, coeff = -h[m-lag]
    
    alpha = np.zeros(K)
    # First factor contributions
    for lag in range(-m, n - m):
        # lag ranges from -m to n-m-1
        # m+lag ranges from 0 to n-1
        # index in corr array: lag + n - 1
        idx = lag + n - 1
        col = m + lag
        alpha[idx] += (1.0 - h[col])  # h[m] * (1-h[m+lag])
    
    # Second factor contributions
    for lag in range(m - n + 1, m + 1):
        # lag ranges from m-n+1 to m
        # m-lag ranges from 0 to n-1
        idx = lag + n - 1
        row = m - lag
        alpha[idx] += -h[row]  # h[m-lag] * (1-h[m]) = -h[m-lag]*h[m]
    
    # Current correlation
    corr = compute_corrs(h)
    
    # beta_lag = corr[lag] - alpha[lag] * h[m]
    beta = corr - alpha * h[m]
    
    # Find t* = argmin_{t in [0,1]} max_k (alpha[k]*t + beta[k])
    # Use bisection on subgradient
    # g(t) = max_k (alpha[k]*t + beta[k])
    
    vals_0 = alpha * 0 + beta  # = beta
    vals_1 = alpha * 1 + beta  # = alpha + beta
    
    # Check if minimum is at boundary
    # At t=0: subgradient = set of alpha[k] for k achieving max at t=0
    max_at_0 = np.max(vals_0)
    active_0 = alpha[vals_0 >= max_at_0 - 1e-15]
    if len(active_0) > 0 and np.min(active_0) >= 0:
        return 0.0  # all active lines have non-negative slope at t=0
    
    max_at_1 = np.max(vals_1)
    active_1 = alpha[vals_1 >= max_at_1 - 1e-15]
    if len(active_1) > 0 and np.max(active_1) <= 0:
        return 1.0  # all active lines have non-positive slope at t=1
    
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = alpha * mid + beta
        max_val = np.max(vals)
        active = alpha[vals >= max_val - 1e-15]
        if len(active) == 0:
            break
        if np.min(active) >= 0:
            hi = mid  # slope >= 0, minimum is to the left
        elif np.max(active) <= 0:
            lo = mid  # slope <= 0, minimum is to the right
        else:
            # 0 is in subgradient, we're at the minimum
            break
    return mid

Let me also think about the bisection more carefully. The subgradient of g(t) = max_k (a_k t + b_k) at t is:
∂g(t) = {a_k : k ∈ argmax (a_k t + b_k)}

If 0 ∈ ∂g(t), then t is a minimizer.
If all a_k in the subgradient are > 0, then g is increasing at t, so minimum is to the left.
If all a_k in the subgradient are < 0, then g is decreasing at t, so minimum is to the right.

The bisection:
- lo, hi = 0, 1
- mid = (lo+hi)/2
- If subgradient at mid contains 0: done
- If all elements of subgradient > 0: hi = mid
- If all elements of subgradient < 0: lo = mid

This converges in O(log(1/ε)) iterations. With ε = 1e-15, that's about 50 iterations.

But actually, since g is piecewise linear with at most K breakpoints, the bisection will find the exact minimum in at most O(log K) iterations (since each iteration eliminates at least one segment). With K=639, that's at most 10 iterations. Let me use 30 to be safe.

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

        if abs(diff) < 1e-14:
            break
        if diff > 0:

            cnt = np.sum(mask)
            if cnt == 0:

                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                h[:] = 0.5
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def coord_min_1d(alpha, beta):
    K = len(alpha)
    vals_0 = beta
    max0 = np.max(vals_0)
    act0 = alpha[vals_0 >= max0 - 1e-13]
    if len(act0) > 0 and np.min(act0) >= 0:
        return 0.0
    vals_1 = alpha + beta
    max1 = np.max(vals_1)
    act1 = alpha[vals_1 >= max1 - 1e-13]
    if len(act1) > 0 and np.max(act1) <= 0:
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = alpha * mid + beta
        mx = np.max(vals)
        act = alpha[vals >= mx - 1e-13]
        if len(act) == 0:
            break
        if np.min(act) >= 0:
            hi = mid
        elif np.max(act) <= 0:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, max_sweeps, deadline):
    best = h.copy()
    best_c5 = compute_c5(h)
    K = 2 * n - 1
    
    for sw in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        for m in range(n):
            if time.time() > deadline:

def slsqp_polish(h, n, deadline):
    dx = 2.0 / n
    K = 2 * n - 1
    best = h.copy()
    best_c5 = compute_c5(h)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        c = corr * dx
        m = c.max()
        # Smooth max
        beta = 200
        shifted = beta * (c - m)
        shifted -= shifted.max()
        return m + np.log(np.sum(np.exp(shifted))) / beta

        return np.sum(x) - n / 2.0

                return m + np.log(np.sum(np.exp(shifted))) / bb
