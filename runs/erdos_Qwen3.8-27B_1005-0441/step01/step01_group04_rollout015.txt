def _sin_h(n, freq, amp=0.35):
    x = np.linspace(0, 2 * np.pi * freq, n, endpoint=False)
    h = 0.5 + amp * np.sin(x)
    return _fix_sum(h, n)

def _block_h(n):
    h = np.array([(0.9 if (i % 3) < 2 else 0.1) for i in range(n)])
    return _fix_sum(h, n)

def _random_h(n, rng):
    h = rng.random(n)
    return _fix_sum(h, n)

def _perturb(h, rng, scale):
    return _fix_sum(h + rng.normal(0, scale, len(h)), len(h))

def _slsqp_smooth(h0, alpha, t_limit, max_iter=200):
    n = len(h0)
    dx = 2.0 / n

    def obj(x):
        if time.time() > t_limit:
            raise Exception("timeout")
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = np.max(c)
        sh = c - cmax
        return cmax + np.log(np.sum(np.exp(alpha * sh))) / alpha

    def eq(x):
        return float(np.sum(x) - n / 2.0)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': eq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-11})
        if res.x is not None:
            return _fix_sum(res.x, n)
    except Exception:
        pass
    return h0.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    def rem():
        return budget_s - (time.time() - t0)

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None:
        h_init = _fix_sum(h_init, len(h_init))
        best_c5 = _eval_c5(h_init)
        best_h = h_init.copy()
    else:
        best_h = _block_h(80)
        best_c5 = _eval_c5(best_h)

    # Phase 1: Multi-alpha gradient descent from best
    if rem() > 120:
        for a_s, a_e in [(30, 150), (100, 400), (200, 800)]:
            if rem() < 80:
                break
            tn = time.time() + rem() * 0.25
            h_new, c5_new = _grad_descent(best_h, a_s, a_e, 0.015, 400, tn)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 2: Subgradient (sharp max)
    if rem() > 60:
        tn = time.time() + rem() * 0.3
        h_new, c5_new = _subgrad_descent(best_h, 600, tn)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: SLSQP polish
    if rem() > 60:
        tn = time.time() + rem() * 0.35
        for alpha in [100, 300]:
            if rem() < 30:
                break
            h_new = _slsqp_smooth(best_h, alpha, tn, 150)
            h_new = _fix_sum(h_new, len(h_new))
            c5_new = _eval_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Different grid sizes
    for n in [48, 56, 64, 96]:
        if rem() < 80:
            break
        starts = [_random_h(n, rng), _block_h(n), _sin_h(n, 1, 0.4), _sin_h(n, 2, 0.3)]
        for h0 in starts:
            if rem() < 40:
                break
            tn = time.time() + rem() * 0.12
            h_new, c5_new = _grad_descent(h0, 50, 300, 0.02, 200, tn)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            if rem() > 30:
                tn2 = time.time() + rem() * 0.08
                h_new2, c5_new2 = _subgrad_descent(h_new, 200, tn2)
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2

    # Phase 5: Additional gradient + subgrad cycles on best
    if rem() > 50:
        for _ in range(4):
            if rem() < 20:
                break
            tn = time.time() + rem() * 0.2
            h_new, c5_new = _grad_descent(best_h, 200, 1000, 0.008, 300, tn)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            if rem() > 20:
                tn2 = time.time() + rem() * 0.15
                h_new2, c5_new2 = _subgrad_descent(best_h, 300, tn2)
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2

    # Phase 6: Perturbation restarts
    if rem() > 40:
        for scale in [0.03, 0.05, 0.01]:
            if rem() < 20:
                break
            h_pert = _perturb(best_h, rng, scale)
            tn = time.time() + rem() * 0.25
            h_new, c5_new = _grad_descent(h_pert, 100, 500, 0.01, 300, tn)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            if rem() > 15:
                h_new2, c5_new2 = _subgrad_descent(h_new, 200, time.time() + rem() * 0.15)
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2

    # Final polish
    if rem() > 20:
        tn = time.time() + rem() * 0.6
        h_new, c5_new = _subgrad_descent(best_h, 500, tn)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
