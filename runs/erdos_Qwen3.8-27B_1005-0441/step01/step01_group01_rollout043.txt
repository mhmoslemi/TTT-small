            options={'maxiter': 1500, 'ftol': 1e-14}
        )
        h_new = project_feasible(res.x, n)
        c5_new, _ = compute_c5(h_new)
        tl3 = min(t0 + budget_s * 0.92, time.time() + 30)
        h_new, c5_new = local_search(h_new, n, max_rounds=60, time_limit=tl3)
        if c5_new < c5_best:
            c5_best = c5_new
            h = h_new.copy()

    # Phase 6: Final high-alpha SLSQP polish
    if time.time() - t0 < budget_s * 0.95:
        for alpha in [3000, 5000]:
            if time.time() - t0 > budget_s * 0.97:
                break
            res = minimize(
                obj_lse, h.copy(), args=(alpha,),
                method='SLSQP', bounds=bounds, constraints=constraint,
                options={'maxiter': 2000, 'ftol': 1e-14}
            )
            h_new = project_feasible(res.x, n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h = h_new.copy()

    # Final local search
    if time.time() - t0 < budget_s - 5:
        h, c5_best = local_search(h, n, max_rounds=100, time_limit=t0 + budget_s - 3)

    h = project_feasible(h, n)
    c5_best, _ = compute_c5(h)
    return h, c5_best, n
