import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h):
    h = np.clip(h.copy(), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not mask.any():
            break
        h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
    return h

def smooth_obj(h, beta):
    corr = get_corr(h)
    mc = corr.max()
    return mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta

def smooth_grad(h, beta):
    n = len(h)
    corr = get_corr(h)
    mc = corr.max()
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        # Term A: d/dh[j] where h[j] = h[i+d], i=j-d
        a_lo = max(0, i_lo + d)
        a_hi = min(n, i_hi + d)
        if a_lo < a_hi:
            grad[a_lo:a_hi] += wi * (1.0 - h[a_lo - d:a_hi - d])
        # Term B: d/dh[j] where h[j] in (1-h[i]), i=j
        b_lo = max(0, i_lo)
        b_hi = min(n, i_hi)
        if b_lo < b_hi:
            grad[b_lo:b_hi] -= wi * h[b_lo + d:b_hi + d]
    return grad

def slsqp_optimize(h0, beta, maxiter=200):
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        return smooth_obj(x, beta)
    
    def jac(x):
        return smooth_grad(x, beta)
    
    def cons_fun(x):
        return x.sum() - target
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h0, jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x)

def projected_subgradient(h0, beta, n_iter=500, t_budget=30):
    h = h0.copy()
    t0 = time.time()
    alpha = 0.01
    best = h.copy()
    best_val = compute_c5(h)
    
    for it in range(n_iter):
        if time.time() - t0 > t_budget:
            break
        g = smooth_grad(h, beta)
        # Backtracking line search
        f0 = smooth_obj(h, beta)
        step = alpha
        for _ in range(20):
            h_new = project_feasible(h - step * g)
            f_new = smooth_obj(h_new, beta)
            if f_new < f0:
                break
            step *= 0.5
        h = h_new
        c = compute_c5(h)
        if c < best_val:
            best_val = c
            best = h.copy()
        alpha *= 0.999
    return best, best_val

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    for _ in range(5000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not mask.any():
            break
        h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
    return h

def _smooth_obj(h, beta):
    corr = get_corr(h)
    mc = corr.max()
    return float(mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta)

def _smooth_grad(h, beta):
    n = len(h)
    corr = get_corr(h)
    mc = corr.max()
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-14:
            continue
        d = k - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        a_lo = max(0, i_lo + d)
        a_hi = min(n, i_hi + d)
        if a_lo < a_hi:
            grad[a_lo:a_hi] += wi * (1.0 - h[a_lo - d:a_hi - d])
        b_lo = max(0, i_lo)
        b_hi = min(n, i_hi)
        if b_lo < b_hi:
            grad[b_lo:b_hi] -= wi * h[b_lo + d:b_hi + d]
    return grad

    return x.sum() - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))
