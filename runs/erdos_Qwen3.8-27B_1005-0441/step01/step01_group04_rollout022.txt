    grad_c = np.zeros(n)
    for m in range(2*n-1):
        w = weights[m]
        if w < 1e-15: continue

        for i in range(n):

            j2 = i + lag
            if 0 <= j2 < n:

    grad_c *= dx
    
    grad = grad_c + 2*lam*sum_viol*np.ones(n)
    return f, grad

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 0)
            if room.sum() < 1e-15:
                break
            h = np.minimum(h + diff * room / room.sum(), 1.0)
        else:
            mass = np.maximum(h, 0)
            if mass.sum() < 1e-15:
                break
            h = np.maximum(h - (-diff) * mass / mass.sum(), 0.0)
    return h

def _lse_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = _corr_vals(h)
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    # Numerical stability
    shifted = np.clip(shifted, -500, 500)
    ex = np.exp(shifted)
    lse = c_max + np.log(np.sum(ex)) / alpha
    weights = ex / np.sum(ex)
    
    # Gradient of LSE w.r.t. h
    grad_c = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            # (1-h[i-lag]) valid for i in [lag, n-1]
            grad_c[lag:] += w * (1.0 - h[:n - lag])
            # -h[i+lag] valid for i in [0, n-1-lag]
            grad_c[:n - lag] -= w * h[lag:]
        else:
            nl = -lag
            # (1-h[i+nl]) valid for i in [0, n-1-nl]
            grad_c[:n - nl] += w * (1.0 - h[nl:])
            # -h[i-nl] valid for i in [nl, n-1]
            grad_c[nl:] -= w * h[:n - nl]
    grad_c *= dx
    return lse, grad_c

def _pen_obj(h, alpha, lam):
    lse, _ = _lse_and_grad(h, alpha)
    sv = np.sum(h) - len(h) / 2.0
    return lse + lam * sv * sv

def _pen_grad(h, alpha, lam):
    lse, grad = _lse_and_grad(h, alpha)
    sv = np.sum(h) - len(h) / 2.0
    grad = grad + 2.0 * lam * sv * np.ones(len(h))
    f = lse + lam * sv * sv
    return f, grad

def _optimize_lse(h0, alpha, lam, max_iter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda h: _pen_obj(h, alpha, lam),
                   h0, method='L-BFGS-B',
                   jac=lambda h: _pen_grad(h, alpha, lam),
                   bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
    return res.x

def _slsqp_active(h0, top_k=8, max_iter=300):
    """SLSQP with only top-K active constraints + equality."""
    n = len(h0)
    dx = 2.0 / n
    c = _corr_vals(h0)
    top_idx = np.argsort(c)[-top_k:]
    t0 = c[np.max(top_idx)]
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        cv = _corr_vals(h)
        return cv[top_idx] - t  # Should be <= 0
    
    def eq(x):
        return np.sum(x[:-1]) - n / 2.0
    
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    try:
        res = minimize(obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': lambda x: -( _corr_vals(x[:-1])[top_idx] - x[-1])},
                                    {'type': 'eq', 'fun': eq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        return res.x[:-1]
    except Exception:
        return h0.copy()

def _run_lse_sequence(h0, alphas, lams, max_iter_each=200):
    n = len(h0)
    h = h0.copy()
    for alpha in alphas:
        for lam in lams:
            h = _optimize_lse(h, alpha, lam, max_iter_each)
            h = _fix_sum(h, n)
    return h

def _perturb(h, rng, scale):
    return _fix_sum(h + rng.normal(0, scale, len(h)), len(h))

def _periodic_h(n, period_vals):

    if len(h) < n:
        h = np.concatenate([h, period_vals[:n - len(h)]])
    return _fix_sum(h, n)
