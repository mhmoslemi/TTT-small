```python
def project_simplex(h, S):
    """Project h onto {0<=h<=1, sum(h)=S} using bisection on lambda."""
    n = len(h)
    lo, hi = -1.0, 1.0
    # After projection: p_i = clip(h_i - lam, 0, 1)

        if s > S:
```

```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, S):
    """Euclidean projection onto {0<=x<=1, sum(x)=S} via bisection."""
    n = len(h)
    # If already close, return
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - S) < 1e-15:
        return h
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def soft_val(h, beta):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    s = beta * (corr - mc)
    s = np.clip(s, -500, 500)
    return (mc + np.log(np.sum(np.exp(s))) / beta) * dx

def soft_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    mc = np.max(corr)
    s = np.clip(beta * (corr - mc), -500, 500)
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-16:
            continue
        d = idx - (n - 1)
        if d >= 0:
            hi = n - d
            if hi > 0:
                grad[:hi] += wi * omh[d:]
                if d > 0:
                    grad[d:] -= wi * h[:hi]
        else:
            dp = -d
            hi = n - dp
            if hi > 0:
                if dp > 0:
                    grad[dp:] += wi * omh[:hi]
                grad[:hi] -= wi * h[dp:]
    return grad * dx

    n = len(h0)
    S = n / 2.0
    
    def obj(x):
        return soft_val(x, beta)

        return soft_grad(x, beta)

        return x.sum() - S
    
    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n

        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(res.x, S)
    except Exception:
        return project_box_sum(h0.copy(), S)

def nm_polish(h0, S, maxiter=500):
    n = len(h0)

        return compute_c5(project_box_sum(x, S))
    
    try:

                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
        return project_box_sum(res.x, S)
    except Exception:
        return h0.copy()

def coord_descent(h0, S, max_rounds=50):
    n = len(h0)
    h = h0.copy()

    for r in range(max_rounds):

                    h = h_try.copy()
                    best_c = c_try
                    improved = True
                    break
        if not improved:
```

```python
def sg(x, b):
    n = len(x)
    d = 2.0 / n
    o = 1.0 - x
    c = np.correlate(x, o, mode="full")
    m = c.max()
    t = np.clip(b * (c - m), -500, 500)
    w = np.exp(t)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        dd = i - (n - 1)
        if dd >= 0:
            hi = n - dd
            g[dd:] += wi * o[:hi]
            g[:hi] -= wi * x[dd:dd+hi]
        else:
            dp = -dd
            hi = n - dp
            g[:hi] += wi * o[dp:]
            g[dp:] -= wi * x[:hi]
    return g * d
```
