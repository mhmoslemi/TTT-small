    h = h.copy()
    best_h = h.copy()

    t0 = time.time()
    deadline = t0 + time_budget
    sweep = 0
    while time.time() < deadline - 1:
        for p in range(n):
            # Optimize h[p] over [0,1]
            hp = h[p]
            h[p] = 0.0
            S0 = np.correlate(h, 1-h, mode='full') * (2.0/n)
            h[p] = 1.0
            S1 = np.correlate(h, 1-h, mode='full') * (2.0/n)
            h[p] = hp

            A = S0  # value at t=0
            # Minimize max(A + B*t) over t in [0,1]
            t_opt = minimize_pwl(A, B)
            h[p] = t_opt

        h = project_feasible(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        sweep += 1
    return best_h, best_c5

f(t) = max_k (A_k + B_k * t) is convex.
f'(t) (subgradient) = {B_k : k ∈ argmax}

We want the smallest t where f is non-decreasing, i.e., min of subgradient ≥ 0.
Equivalently, the largest t where f is non-increasing, i.e., max of subgradient ≤ 0.

If these coincide, that's the minimum.

Bisection:
- lo = 0, hi = 1
- At mid: compute active = argmax(A + B*mid)
  - If B[active].min() ≥ 0: f is non-decreasing at mid → optimum is ≤ mid → hi = mid
  - If B[active].max() ≤ 0: f is non-increasing at mid → optimum is ≥ mid → lo = mid
  - Else: 0 is in subgradient → mid is optimum (or in the minimum region)

After bisection, t* = (lo+hi)/2 is the answer.

import numpy as np
import time

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def minimize_pwl(A, B):
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = A + B * mid
        m = vals.max()
        active = vals >= m - 1e-15
        if B[active].min() >= 0:
            hi = mid
        elif B[active].max() <= 0:
            lo = mid
        else:
            hi = mid
            break
    t = (lo + hi) / 2.0
    return min(max(t, 0.0), 1.0)

def coordinate_descent(h, time_budget, seed):
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    sweep = 0
    no_improve = 0
    while time.time() < deadline - 0.5:
        for p in range(n):
            hp = h[p]
            h[p] = 0.0
            S0 = np.correlate(h, 1.0 - h, mode='full')
            h[p] = 1.0
            S1 = np.correlate(h, 1.0 - h, mode='full')
            h[p] = hp
            B = S1 - S0
            A = S0
            h[p] = minimize_pwl(A, B)
        h = project_feasible(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break
        sweep += 1
    return best_h, best_c5

def gradient_descent(h_init, time_budget, seed, tau_init=30.0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.7
    lr = 0.05
    tau = tau_init
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    while time.time() < deadline - 0.3:
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full')
        mc = corr.max()
        shifted = np.clip(tau * (corr - mc), -300, 300)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-14:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n-lag] -= w * h[lag:]
            else:
                L = -lag
                grad[:n-L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n-L]
        grad *= dx
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 0:
            tau = min(tau * 1.01, 500.0)
        if no_improve > 200:
            lr = max(lr * 0.8, 0.002)
            if no_improve > 500:
                h = project_feasible(best_h + rng.randn(n) * 0.03, target_sum)
                v[:] = 0
                lr = 0.05
                tau = 30.0
                no_improve = 0
    return best_h, best_c5

def simulated_annealing(h_init, time_budget, seed):

    n = len(h_init)
    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()

    c5_current = best_c5
    T = 0.05
    t0 = time.time()
    deadline = t0 + time_budget
    iters_per_s = 5000  # rough estimate
    total_iters = int((deadline - t0) * iters_per_s)
    for it in range(total_iters):
        if time.time() > deadline - 1:
            break
        # Pick a random coordinate and perturb it
        p = rng.randint(n)
        delta = rng.uniform(-0.1, 0.1)
        h[p] = min(max(h[p] + delta, 0.0), 1.0)
        h = project_feasible(h, target_sum)
        c5_new = compute_c5(h)
        delta_c = c5_new - c5_current
        if delta_c < 0 or rng.rand() < np.exp(-delta_c / T):
            c5_current = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h = project_feasible(best_h if delta_c > 0.01 else h, target_sum)  # revert
            # Actually, revert properly
            h[p] = min(max(h[p] - delta, 0.0), 1.0)
            h = project_feasible(h, target_sum)
        # Cool
        progress = it / total_iters
        T = 0.05 * (1 - progress) + 0.001
    return best_h, best_c5
