def _repair(h, n):

        s = np.sum(h)
        if abs(s - target) < 1e-6:
            break
    return h

    n = len(h)
    noise = rng.standard_normal(n)
    h_new = h + sigma * noise
    return _repair(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair(h, n):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(40):
        s = np.sum(h)
        if s < 1e-15:
            return np.full(n, 0.5)
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(np.sum(h) - target) < 1e-9:
            break
    return h

def _slsqp_refine(h, n, max_time=10):
    best = h.copy()
    best_val = compute_c5(best)
    dx = 2.0 / n
    t0 = time.time()
    for temp in [0.01, 0.003, 0.001, 0.0004]:
        if time.time() - t0 > max_time - 1:
            break
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            m = np.max(corr)
            t = max(temp, 1e-12)
            return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx
        def eq(x):
            return np.sum(x) - n / 2.0
        def eq_j(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq, 'jac': eq_j}]
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 250, 'ftol': 1e-12})
            h_new = _repair(res.x, n)
            v = compute_c5(h_new)
            if v < best_val - 1e-12:
                best, best_val = h_new, v
        except Exception:
            pass
    return best, best_val

def _nm_refine(h, n, max_time=8):
    best = h.copy()
    best_val = compute_c5(best)
    t0 = time.time()
    def obj(x):
        x_r = _repair(x, n)
        return compute_c5(x_r)
    try:
        res = minimize(obj, best, method='Nelder-Mead',
                       options={'xatol': 1e-7, 'fatol': 1e-11, 'maxfev': 6000,
                                'maxiter': 3000})
        h_new = _repair(res.x, n)
        v = compute_c5(h_new)
        if v < best_val - 1e-12:
            best, best_val = h_new, v
    except Exception:
        pass
    return best, best_val

def _adapt_sigma(rank_frac, sigma, rng):
    s = sigma * (1.0 + 0.12 * (0.45 - rank_frac) + 0.03 * rng.standard_normal())
    return float(np.clip(s, 0.003, 0.22))

def _diverse_init(rng, n, base_h, k):
    r = rng.random()
    if r < 0.25:
        h = base_h + 0.04 * rng.standard_normal(n)
    elif r < 0.45:
        h = np.zeros(n)
        h[rng.integers(0, n, n // 2)] = 1.0
    elif r < 0.60:
        x = np.linspace(0, 2, n)
        f1, f2 = rng.uniform(1, 5), rng.uniform(3, 9)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * f1 * x) + 0.15 * np.cos(2 * np.pi * f2 * x)
    elif r < 0.75:
        gap = int(n * rng.uniform(0.1, 0.45))
        half = (n - gap) // 2
        pos = rng.integers(0, max(1, n - half - gap))
        h = np.zeros(n)
        h[pos:pos + half] = 1.0
        p2 = (pos + half + gap) % n
        for i in range(half):
            h[(p2 + i) % n] = max(h[(p2 + i) % n], 1.0)
    else:
        h = 0.5 + 0.35 * rng.standard_normal(n)
    return _repair(h, n)
