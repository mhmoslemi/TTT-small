import numpy as np
import time
import math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def gradient_at_h(h, dx, top_m, tau):
    """Compute subgradient of max_k corr[k] at h."""
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    if tau > 0:
        mc = corr.max()
        shifted = np.clip(tau * (corr - mc), -500, 500)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
    else:
        idx = np.argsort(corr)[::-1][:top_m]
        weights = np.zeros(len(corr))
        weights[idx] = 1.0 / top_m
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx

def grad_optimize(h_init, time_budget, seed, top_m=3, tau=0.0, lr_init=0.1):
    """Projected gradient descent with momentum."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.7
    lr = lr_init
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    
    while time.time() < deadline - 0.2:
        grad = gradient_at_h(h, dx, top_m, tau)
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

                    v[:] = 0.0
                    lr = lr_init
                    no_improve = 0
        h = h_new

def to_binary(h_frac):
    """Convert fractional to binary, maintaining sum constraint."""
    n = len(h_frac)
    target = n // 2
    h = (h_frac >= 0.5).astype(float)
    current_sum = int(h.sum())
    if current_sum > target:
        # Need to flip (current_sum - target) ones to zeros
        # Choose the smallest h_frac among the ones
        ones = np.where(h == 1.0)[0]
        order = ones[np.argsort(h_frac[ones])]
        for idx in order[:current_sum - target]:
            h[idx] = 0.0
    elif current_sum < target:
        zeros = np.where(h == 0.0)[0]
        order = zeros[np.argsort(-h_frac[zeros])]
        for idx in order[:target - current_sum]:
            h[idx] = 1.0
    return h

def exhaustive_2opt(h_bin, time_budget, max_passes=200):
    """Try all possible 1<->0 swaps, apply best, repeat."""
    n = len(h_bin)
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()

    improved = True
    passes = 0

        passes += 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improve = 0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if best_c5 - c5 > best_improve:
                    best_improve = best_c5 - c5
                    best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            best_c5 -= best_improve
            improved = True
    
    # Keep best
    # Actually track the best throughout
    return h, best_c5

import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def subgrad(h, dx, top_m, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if tau > 0:
        mc = corr.max()
        sh = np.clip(tau * (corr - mc), -400, 400)
        wts = np.exp(sh)
        wts /= wts.sum()
    else:
        idx = np.argsort(corr)[-top_m:]
        wts = np.zeros(len(corr))
        wts[idx] = 1.0 / top_m
    g = np.zeros(n)
    for k in range(len(corr)):
        w = wts[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[lag:] += w * (1.0 - h[:n - lag])
            g[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            g[:n - L] += w * (1.0 - h[L:])
            g[L:] -= w * h[:n - L]
    return g * dx

def grad_desc(h0, t_budget, seed, top_m=3, tau=0, lr0=0.12, beta=0.75):
    rng = np.random.RandomState(seed)
    n = len(h0)

    dx = 2.0 / n
    h = project_feasible(h0.copy(), ts)
    bh, bc = h.copy(), compute_c5(h)
    v = np.zeros(n)
    lr = lr0
    t0 = time.time()
    dl = t0 + t_budget
    ni = 0
    while time.time() < dl - 0.2:
        g = subgrad(h, dx, top_m, tau)
        v = beta * v - lr * g
        h = project_feasible(h + v, ts)

        else:
            ni += 1
            if ni > 80:
                lr *= 0.92
            if ni > 350:
                h = project_feasible(bh + rng.randn(n) * 0.04, ts)
                v[:], lr, ni = 0.0, lr0, 0
    return bh, bc

def frac_to_bin(hf):
    n = len(hf)
    t = n // 2
    h = (hf >= 0.5).astype(float)
    s = int(h.sum())
    if s > t:
        oi = np.where(h == 1.0)[0]
        h[oi[np.argsort(hf[oi])][:s - t]] = 0.0
    elif s < t:
        zi = np.where(h == 0.0)[0]
        h[zi[np.argsort(-hf[zi])][:t - s]] = 1.0
    return h

def bin_2opt(hb, t_budget, max_rounds=500):
    n = len(hb)
    h = hb.copy()
    bc = compute_c5(h)
    t0 = time.time()
    dl = t0 + t_budget
    for _ in range(max_rounds):
        if time.time() >= dl - 0.1:
            break
        ones, zeros = np.where(h == 1.0)[0], np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_d, best_ij = 1e-12, None
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h)
                h[i], h[j] = 1.0, 0.0
                d = bc - c
                if d > best_d:
                    best_d, best_ij = d, (i, j)
        if best_ij is None:
            break
        h[best_ij[0]], h[best_ij[1]] = 0.0, 1.0
        bc -= best_d
    return h, bc

def iterated_2opt(hb, t_budget, seed):
    rng = np.random.RandomState(seed)
