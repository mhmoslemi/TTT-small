import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        ht = np.clip(h + mid, 0.0, 1.0)
        if np.sum(ht) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = len(mask)
        if cnt > 0 and abs(resid / cnt) < 0.5:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def top_k_gradient(h, n_active=5, alpha=300.0):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    vals = corr_raw * dx
    total_lags = len(vals)
    if n_active >= total_lags:
        idx = np.arange(total_lags)
    else:
        part = np.partition(vals, -n_active)[-n_active:]
        thresh = part[0]
        idx = np.where(vals >= thresh - 1e-12)[0]
    m = np.max(vals[idx])
    shifted = alpha * (vals[idx] - m)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for ii, wt in zip(idx, w):
        k = ii - (n - 1)
        if k >= 0:
            j1_lo, j1_hi = 0, n - 1 - k
            j2_lo, j2_hi = k, n - 1
        else:
            j1_lo, j1_hi = -k, n - 1
            j2_lo, j2_hi = 0, n - 1 + k
        if j1_hi >= j1_lo:
            js = np.arange(j1_lo, j1_hi + 1)
            grad[js] += wt * dx * (1.0 - h[js + k])
        if j2_hi >= j2_lo:
            js = np.arange(j2_lo, j2_hi + 1)
            grad[js] -= wt * dx * h[js - k]
    return grad

def smooth_obj(h, alpha=200.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -400.0, 400.0)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_eq(x, target):
    return np.sum(x) - target

def _sum_eq_jac(x):
    return np.ones(len(x))

def make_two_block(n):
    h = np.zeros(n)
    b = n // 4
    h[:b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def make_three_block(n):
    h = np.zeros(n)
    b = n // 6
    h[:b] = 1.0
    h[n//2 - b:n//2 + b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def make_sin_mod(n, phase=0.5):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + dx if False else np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * phase * x / 2.0 + np.pi/4)
    return project_feasible(h, n)

def subgrad_optimize(h, n, t0, budget_s, frac, n_active, alpha_sm, max_iters):
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best_local = c5_cur
    h_best_local = h.copy()
    momentum = np.zeros(n)
    mu = 0.85
    lr = 0.005
    prev_h = h.copy()
    prev_g = top_k_gradient(h, n_active, alpha_sm)
    t_deadline = t0 + budget_s * frac
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best_local - 1e-8:
            c5_best_local = c5_new
            h_best_local = h_new.copy()
        dh = h_new - h
        dg = top_k_gradient(h_new, n_active, alpha_sm) - prev_g
        s = np.dot(dh, dg)
        y = np.dot(dg, dg)
        if y > 1e-14:
            lr = np.clip(np.dot(dh, dh) / max(s, 1e-12) if s > 0 else lr, 1e-5, 0.02)
        prev_h = h_new
        prev_g = top_k_gradient(h_new, n_active, alpha_sm)
        momentum = mu * momentum - lr * prev_g
        h = h_new
        if it % 1000 == 999:
            lr *= 0.9
    return h_best_local, c5_best_local

def slsqp_polish(h, n, alpha, maxiter=600):
    bounds = [(0.0, 1.0)] * n
    target = n * 0.5
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
    def obj(x):
        return smooth_obj(x, alpha)
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=constraint, options={'maxiter': maxiter, 'ftol': 1e-13})
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if res.success or c5c < compute_c5(h):
            return hc, c5c
    except:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = make_two_block(160)
    n = len(h0)
    h_best = project_feasible(h0, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Subgradient with BB step size (main search engine)
    t_sub_end = t0 + budget_s * 0.35
    for na, al in [(3, 200), (5, 300), (8, 500), (5, 800), (3, 1000)]:
        if time.time() > t_sub_end:
            break
        h_sub, c5_sub = subgrad_optimize(h_best, n, t0, budget_s, 0.35, na, al, 20000)
        if c5_sub < c5_best:
            c5_best = c5_sub
            h_best = h_sub.copy()

    # Phase 2: SLSQP refinement at current n
    t_slsqp_end = t0 + budget_s * 0.55
    for al in [100.0, 300.0, 800.0, 2000.0]:
        if time.time() > t_slsqp_end:
            break
        h_best, c5_try = slsqp_polish(h_best, n, al, maxiter=400)
        if c5_try < c5_best:
            c5_best = c5_try

    # Phase 3: Multi-restart from structured initializations
    t_restart_end = t0 + budget_s * 0.70
    starts = [make_two_block(n), make_three_block(n), make_sin_mod(n, 0.75), make_sin_mod(n, 1.0)]
    for sp in starts:
        if time.time() > t_restart_end:
            break
        h_sub, c5_sub = subgrad_optimize(sp, n, t0, budget_s, 0.70, 5, 400, 8000)
        if c5_sub < c5_best:
            c5_best = c5_sub
            h_best = h_sub.copy()
        h_try, c5_try = slsqp_polish(h_best, n, 400.0, maxiter=500)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try.copy()

    # Phase 4: Random perturbation restarts
    t_pert_end = t0 + budget_s * 0.80
    while time.time() < t_pert_end:
        pert = 0.03 + 0.05 * rng.random()
        hp = h_best + pert * rng.standard_normal(n)
        hp = project_feasible(hp, n)
        h_sub, c5_sub = subgrad_optimize(hp, n, t0, budget_s, 0.80, 4, 500, 5000)
        if c5_sub < c5_best:
            c5_best = c5_sub
            h_best = h_sub.copy()

    # Phase 5: Upscale to finer grid
    if time.time() < t0 + budget_s * 0.90:
        n2 = min(n + 40, 240)
        if n2 > n:
            x_old = np.linspace(0.0, 2.0, n)
            x_new = np.linspace(0.0, 2.0, n2)
            h_up = np.interp(x_new, x_old, h_best)
            h_up = project_feasible(h_up, n2)
            c5_up = compute_c5(h_up)
            if c5_up < c5_best + 0.002:
                h_sub2, c5_sub2 = subgrad_optimize(h_up, n2, t0, budget_s, 0.90, 5, 500, 10000)
                if c5_sub2 < c5_best:
                    c5_best = c5_sub2
                    h_best = h_sub2.copy()
                    n = n2
                h_pol, c5_pol = slsqp_polish(h_best, n, 500.0, maxiter=400)
                if c5_pol < c5_best:
                    c5_best = c5_pol
                    h_best = h_pol.copy()

    # Final: one more SLSQP polish
    if time.time() < t0 + budget_s - 10:
        h_pol, c5_pol = slsqp_polish(h_best, n, 1000.0, maxiter=300)
        if c5_pol < c5_best:
            c5_best = c5_pol
            h_best = h_pol.copy()

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
