    c = np.correlate(h, 1.0 - h, mode="full")
    T = 2*n - 1  # length of c
    
    for sweep in range(n_sweeps):
        if order == 'random' and rng is not None:

        else:
            idx_order = range(n)

            hi = h[i]

            A = np.zeros(T)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.99999999
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def _soft_obj(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def _sum_jacobian(x):
    return np.ones(len(x))

def coord_descent(h, n, n_sweeps=100, rng=None, project_every=10):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode="full")
    T = 2 * n - 1
    phi = 0.618033988749895

    for sweep in range(n_sweeps):
        if rng is not None:
            idx_order = rng.permutation(n)
        else:
            idx_order = np.arange(n)

        for i in idx_order:
            hi = h[i]
            A = np.zeros(T)

            lo1 = max(0, n - 1 - i)
            hi1 = min(T - 1, 2 * n - 2 - i)
            if lo1 <= hi1:
                p1 = np.arange(lo1, hi1 + 1) + i - n + 1
                A[lo1:hi1 + 1] += 1.0 - h[p1]

            lo2 = max(0, i)
            hi2 = min(T - 1, i + n - 1)
            if lo2 <= hi2:
                p2 = i - np.arange(lo2, hi2 + 1) + n - 1
                A[lo2:hi2 + 1] += -h[p2]

            B = c - A * hi
            a_g, b_g = 0.0, 1.0
            c_g = b_g - phi * (b_g - a_g)
            d_g = a_g + phi * (b_g - a_gs) if False else a_g + phi * (b_g - a_g)
            fc = np.max(A * c_g + B)
            fd = np.max(A * d_g + B)

            for _ in range(45):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - phi * (b_g - a_g)
                    fc = np.max(A * c_g + B)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + phi * (b_g - a_g)
                    fd = np.max(A * d_g + B)

            v_opt = (a_g + b_g) / 2.0
            if abs(v_opt - hi) > 1e-14:
                nz = A != 0
                c[nz] += A[nz] * (v_opt - hi)
                h[i] = v_opt

        if project_every > 0 and (sweep + 1) % project_every == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode="full")

    return h

def swap_saanneal(h, n, n_iter=5000, rng=None, t0=None, budget_frac=0.2):
    t_start = time.time()
    h = h.copy()
    h_bin = (h > 0.5).astype(float)
    need = n / 2.0 - np.sum(h_bin)
    if need > 0:
        order = np.argsort(h)[::-1]
        for idx in range(int(need)):
            h_bin[order[idx]] = 1.0
    elif need < 0:
        order = np.argsort(h)
        for idx in range(int(-need)):
            h_bin[order[idx]] = 0.0
    h_bin = project_feasible(h_bin, n)
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    T = 2 * n - 1
    cur_c = np.correlate(h_bin, 1.0 - h_bin, mode="full")
    cur_max = cur_c.max()
    temp = 0.05
    cool = 0.9995

    for it in range(n_iter):
        if t0 is not None and (time.time() - t0) > budget_frac:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        # swap h[i]: 1->0, h[j]: 0->1
        # delta_c[m] = A_i[m]*(0-1) + A_j[m]*(1-0) = -A_i[m] + A_j[m]
        # A_i[m]: coefficient of h[i] in c[m]
        A_i = np.zeros(T)
        lo1 = max(0, n - 1 - i)
        hi1 = min(T - 1, 2 * n - 2 - i)
        if lo1 <= hi1:
            A_i[lo1:hi1+1] += 1.0 - h_bin[i + np.arange(lo1,hi1+1) - n + 1]
        lo2 = max(0, i)
        hi2 = min(T - 1, i + n - 1)
        if lo2 <= hi2:
            A_i[lo2:hi2+1] += -h_bin[i - np.arange(lo2,hi2+1) + n - 1]

        A_j = np.zeros(T)
        lo1 = max(0, n - 1 - j)
        hi1 = min(T - 1, 2 * n - 2 - j)
        if lo1 <= hi1:
            A_j[lo1:hi1+1] += 1.0 - h_bin[j + np.arange(lo1,hi1+1) - n + 1]
        lo2 = max(0, j)
        hi2 = min(T - 1, j + n - 1)
        if lo2 <= hi2:
            A_j[lo2:hi2+1] += -h_bin[j - np.arange(lo2,hi2+1) + n - 1]

        delta = -A_i + A_j
        new_c = cur_c + delta
        new_max = new_c.max()
        dx = 2.0 / n
        if new_max * dx <= best_c5 or rng.rand() < np.exp(-(new_max - cur_max) / (temp * n / 2.0)):
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            cur_c = new_c
            cur_max = new_max
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            if new_max * dx < best_c5:
                best_c5 = new_max * dx
                best = h_bin.copy()
        temp *= cool

    return best, best_c5
