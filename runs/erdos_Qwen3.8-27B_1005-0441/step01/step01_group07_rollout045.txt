def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def make_binary(h, target):
    n = len(h)
    order = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[order[:target]] = 1.0
    return hb

def compute_grad(h, weights, dx):
    """Corrected gradient of weighted sum of overlaps"""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        L = k - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:])

            M = -L
            grad[M:] += w * (1.0 - h[:n - M])
            grad[:n - M] -= w * h[M:]
    return grad * dx

def smooth_weights(corr, tau):
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    exps = np.exp(shifted)
    return exps / exps.sum()
