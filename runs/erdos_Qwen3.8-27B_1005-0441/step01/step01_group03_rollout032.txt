def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def to_binary(h, n):
    n_ones = n // 2
    h = np.clip(h, 0.0, 1.0)
    idx = np.argsort(h)[::-1][:n_ones]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def local_search(h, n, deadline):
    current_c5 = compute_c5(h, n)
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 1e-12
        best_i = -1
        best_j = -1
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h, n)
                h[i] = 1.0
                h[j] = 0.0
                d = current_c5 - c
                if d > best_delta:
                    best_delta = d
                    best_i = i
                    best_j = j
        if best_i < 0:

        h[best_i] = 0.0
        h[best_j] = 1.0
