import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def continuous_optimize(h_init, time_budget, seed, lr_init=0.06):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.85
    lr = lr_init
    tau = 40.0
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    while time.time() < deadline - 0.5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = corr.max()
        shifted = np.clip(tau * (corr - mc), -500, 500)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-12:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                g = np.zeros(n)
                g[:n - lag] += (1.0 - h[lag:])
                g[lag:] -= h[:n - lag]
                grad += w * g
            else:
                L = -lag
                g = np.zeros(n)
                g[L:] += (1.0 - h[:n - L])
                g[:n - L] -= h[L:]
                grad += w * g
        grad *= dx
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        tau = min(tau * 1.005, 300.0)
        if no_improve > 200:
            lr = max(lr * 0.8, 0.004)
        if no_improve > 500:
            h = project_feasible(best_h + rng.randn(n) * 0.03, target_sum)
            v[:] = 0.0
            lr = lr_init
            tau = 40.0
            no_improve = 0
    return best_h, best_c5

def to_binary(h_frac, target, rng):
    n = len(h_frac)
    h = (h_frac >= 0.5).astype(float)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        to_rm = rng.choice(ones, len(ones) - target, replace=False)
        h[to_rm] = 0.0
    elif len(ones) < target:
        to_add = rng.choice(zeros, target - len(ones), replace=False)
        h[to_add] = 1.0
    return h

def binary_sa(h_start, time_budget, seed, max_flip=4):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    target = n // 2
    h = to_binary(h_start.copy(), target, rng)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    T = 0.008
    T_min = 0.00005
    cooling = 0.9998
    it = 0
    while time.time() < deadline - 0.3:
        it += 1
        if it % 30 == 0:
            T = max(T * cooling, T_min)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        nflip = rng.randint(1, max_flip + 1)
        if nflip > min(len(ones), len(zeros)):
            nflip = min(len(ones), len(zeros))
        f_ones = rng.choice(ones, nflip, replace=False)
        f_zeros = rng.choice(zeros, nflip, replace=False)
        h[f_ones] = 0.0
        h[f_zeros] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        if delta < 0 or (T > T_min and rng.rand() < math.exp(min(-delta / T, 50.0))):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[f_ones] = 1.0
            h[f_zeros] = 0.0
    return best_h, best_c5

        ones = np.where(h == 1.0)[0]

                else:

def targeted_greedy(h_start, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    target = n // 2
    h = to_binary(h_start.copy(), target, rng)

        corr = np.correlate(h, 1.0 - h, mode='full')

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        improved = False
        for tl in top_lags:

            # Identify positions contributing to this lag
            if lag >= 0:

            else:
                L = -lag
                contrib_ones = np.arange(max(0, L), min(n, n - L + 1))
            c_ones = contrib_ones[contrib_ones >= 0 & (contrib_ones < n)]
            c_ones = c_ones[(h[c_ones] == 1.0)]
            if len(c_ones) == 0 or len(zeros) == 0:
                continue
            for _ in range(10):
                i = c_ones[rng.randint(len(c_ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
        if not improved:
            nflip = rng.randint(2, 5)
            nflip = min(nflip, min(len(ones), len(zeros)))
            f_ones = rng.choice(ones, nflip, replace=False)
            f_zeros = rng.choice(zeros, nflip, replace=False)
            h[f_ones] = 0.0
