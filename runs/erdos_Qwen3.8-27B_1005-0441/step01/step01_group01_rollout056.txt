```python
def compute_subgrad(h, n, top_k=10):
    one_mh = 1.0 - h
    corr = np.correlate(h, one_mh, mode='full')
    m = np.max(corr)
    
    # Get top k lags
    top_indices = np.argsort(corr)[-top_k:]
    weights = np.zeros(2*n-1)
    for idx in top_indices:
        weights[idx] = 1.0 / top_k
    
    # For each active lag, compute contribution to gradient
    grad = np.zeros(n)
    for idx in top_indices:
        k = idx - (n - 1)
        w = weights[idx]
        # Term 1: (1-h)[j-k] for valid j-k
        # j-k in [0, n-1] => j in [k, k+n-1] intersect [0, n-1]
        j_lo = max(0, k)
        j_hi = min(n, k + n)
        # (1-h)[j-k] for j in [j_lo, j_hi)
        grad[j_lo:j_hi] += w * one_mh[j_lo - k : j_hi - k]
        
        # Term 2: -h[j+k] for valid j+k  
        # j+k in [0, n-1] => j in [-k, n-1-k] intersect [0, n-1]
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        grad[j_lo2:j_hi2] -= w * h[j_lo2 + k : j_hi2 + k]
    
    return grad, m
```

                h_new = project_feasible(h_new, n)
                c5_new2 = compute_c5(h_new)
                if c5_new2 < c5_best:
                    c5_best = c5_new2
                    h = h_new
                else:
                    break
    return h, c5_best

def _coord_descent(h, n, deadline):
    c5_best = compute_c5(h)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_v = h[i]
            for trial in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(trial - old_v) < 1e-12:
                    continue
                h[i] = trial
                h[i] = np.clip(h[i], 0.0, 1.0)
                h = project_feasible(h, n)
                c5_try = compute_c5(h)
                if c5_try < c5_best:
                    c5_best = c5_try
                    improved = True
                    break
            h[i] = old_v
            h = project_feasible(h, n)
    return h, c5_best

def _upscale(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, float(n_old - 1), n_new)
    h_up = np.interp(x_new, x_old, h)
    return project_feasible(h_up, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h_best = _make_pattern(80, 0)
    n = len(h_best)
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B with smooth objective (60% of budget)
    dl1 = t0 + budget_s * 0.55
    h_best, c5_best = _lbfgs_round(h_best, n, bounds, t0, dl1)
    h_best = project_feasible(h_best, n)

    # Phase 2: Subgradient on true max (15% of budget)
    dl2 = t0 + budget_s * 0.70
    h_best, c5_best = _subgrad_round(h_best, n, t0, dl2)
    h_best = project_feasible(h_best, n)

    # Phase 3: Multi-start from different patterns (10% of budget)
    dl3 = t0 + budget_s * 0.80
    for kind in range(7):
        if time.time() > dl3:
            break
        h_try = _make_pattern(n, kind)
        h_try = h_try + 0.05 * rng.standard_normal(n)
        h_try = project_feasible(h_try, n)
        h_try, c5_try = _lbfgs_round(h_try, n, bounds, t0, dl3)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try.copy()

    # Phase 4: Perturbation restarts (5%)
    dl4 = t0 + budget_s * 0.85
    attempts = 0
    while time.time() < dl4 and attempts < 20:
        attempts += 1
        pert = 0.03 + 0.05 * rng.random()
        h_pert = h_best + pert * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)
        h_pert, c5_pert = _lbfgs_round(h_pert, n, bounds, t0, dl4)
        if c5_pert < c5_best:
            c5_best = c5_pert
            h_best = h_pert.copy()

    # Phase 5: Coordinate descent polish (5%)
    dl5 = t0 + budget_s * 0.90
    h_best, c5_best = _coord_descent(h_best, n, dl5)

    # Phase 6: Upscale to finer grid (10% remaining)
    if time.time() < t0 + budget_s * 0.95:
        n2 = min(n * 2, 320)
        if n2 > n:
            h_up = _upscale(h_best, n2)
            bounds2 = [(0.0, 1.0)] * n2
            deadline_up = t0 + budget_s * 0.99
            h_up, c5_up = _lbfgs_round(h_up, n2, bounds2, t0, deadline_up)
            if c5_up < c5_best:
                c5_best = c5_up
                h_best = h_up.copy()
                n = n2
            # Subgradient on fine grid
            if time.time() < t0 + budget_s * 0.995:
                h_best, c5_best2 = _subgrad_round(h_best, n, t0, t0 + budget_s * 0.995)
                if c5_best2 < c5_best:
                    c5_best = c5_best2

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
