    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
        n0 = len(h_init)
    else:
        h_init = make_start(70, seed)
        n0 = 70

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    def try_update(h_new, c5_new):
        nonlocal best_h, best_c5
        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new

    # Phase 1: binary SA from initial
    if time.time() < deadline - 100:
        hb, cb = binary_sa(h_init, min(60, budget_s * 0.12), seed)
        try_update(hb, cb)

    # Phase 2: smooth gradient from best
    if time.time() < deadline - 80:
        h1, c1 = opt_smooth(best_h, min(200, budget_s * 0.25), seed + 1)
        try_update(h1, c1)

    # Phase 3: top-m refinement
    if time.time() < deadline - 60:
        h2, c2 = opt_topm(best_h, min(200, budget_s * 0.25), seed + 2, m=2, lr0=0.02)
        try_update(h2, c2)

    # Phase 4: binary SA again from fractional best
    if time.time() < deadline - 50:
        hb2, cb2 = binary_sa(best_h, min(80, budget_s * 0.15), seed + 3)
        try_update(hb2, cb2)

    # Phase 5: smooth from binary
    if time.time() < deadline - 40:
        h3, c3 = opt_smooth(best_h, min(150, budget_s * 0.2), seed + 4, tau_end=2000.0)
        try_update(h3, c3)

    # Phase 6: top-1 very fine
    if time.time() < deadline - 20:
        h4, c4 = opt_topm(best_h, min(100, budget_s * 0.15), seed + 5, m=1, lr0=0.012)
        try_update(h4, c4)

    # Phase 7: random restarts
    for r in range(6):
        if time.time() > deadline - 20:
            break
        rng_r = np.random.RandomState(seed + 1000 + r * 97)
        h_try = best_h + rng_r.randn(n0) * 0.06
        h_try = project_feasible(h_try, n0 / 2.0)
        t_alloc = max(10, min(80, (deadline - time.time() - 10) / (7 - r)))
        hr, cr = opt_smooth(h_try, t_alloc, seed + 5000 + r, tau_end=500.0)
        try_update(hr, cr)
        if time.time() > deadline - 15:
            break
        hr2, cr2 = opt_topm(hr, max(10, t_alloc * 0.4), seed + 6000 + r, m=2, lr0=0.015)
        try_update(hr2, cr2)

    # Phase 8: try other n values quickly
    for nn in [68, 72, 74]:
        if time.time() > deadline - 30:
            break
        t_alloc = max(10, min(60, (deadline - time.time() - 5) / 2))
        h_nn = make_start(nn, seed + nn)
        hb_nn, cb_nn = binary_sa(h_nn, max(5, t_alloc * 0.4), seed + nn * 7)
        if t_alloc > 15 and time.time() < deadline - 5:
            h_nn2, c_nn2 = opt_smooth(hb_nn, t_alloc * 0.5, seed + nn * 13, tau_end=500)
            if c_nn2 < cb_nn:
                hb_nn, cb_nn = h_nn2, c_nn2
        if cb_nn < best_c5:
            best_h, best_c5 = hb_nn.copy(), cb_nn
            n0 = nn

    # Final polish
    if time.time() < deadline - 3:
        hp, cp = opt_topm(best_h, min(20, deadline - time.time() - 2), seed + 99999, m=1, lr0=0.005)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    best_c5 = compute_c5(best_h)
    n_final = len(best_h)
    # Ensure exact sum constraint
    target = n_final / 2.0
    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_final
