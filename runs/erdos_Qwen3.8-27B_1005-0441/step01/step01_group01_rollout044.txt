```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def corr_and_grad(h, n):
    """Compute correlation vector and gradient at max lag."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    k_max = int(np.argmax(corr))
    lag = k_max - (n - 1)
    # Gradient at this lag
    g = np.zeros(n)
    if lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    elif lag < 0:
        k = -lag
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        g[:] = 1.0 - 2.0 * h
    return corr, lag, g, k_max

def soft_gradient(h, n, temperature):
    """Compute soft gradient using all lags weighted by exp(f_k/T)."""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    # Temperature scaling
    c = corr - np.max(corr)
    weights = np.exp(c / temperature)
    w_sum = np.sum(weights)
    g = np.zeros(n)
    for k_full in range(len(corr)):
        w = weights[k_full]
        if w < 1e-10:
            continue
        lag = k_full - (n - 1)
        wg = w / w_sum
        if lag > 0:
            g[:n-lag] += wg * (1.0 - h[lag:])
            g[lag:] -= wg * h[:n-lag]
        elif lag < 0:
            k = -lag
            g[k:] += wg * (1.0 - h[:n-k])
            g[:n-k] -= wg * h[k:]
        else:
            g[:] += wg * (1.0 - 2.0 * h)
    return np.max(corr) * (2.0/n), g
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_lag(h, lag, n):
    """Gradient of f_lag(h) = sum h[i]*(1-h[i+lag]) w.r.t. h."""
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        k = -lag
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    return g

def soft_grad(h, n, temp):
    """Soft gradient: weighted combination of all lag gradients."""
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode='full')
    c = corr - np.max(corr)
    w = np.exp(c / temp)
    w /= np.sum(w)
    g = np.zeros(n)
    for k_full in range(2*n-1):
        if w[k_full] < 1e-10:
            continue
        lag = k_full - (n-1)
        g += w[k_full] * grad_lag(h, lag, n)
    return np.max(corr) * (2.0/n), g

def projected_subgradient(h, n, temp, steps, t0, budget_frac):
    """Run projected subgradient for given steps."""
    deadline = t0 + budget_frac * 1100  # will be overridden
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    momentum = 0.9
    v = np.zeros(n)
    eta = 0.1
    
    for s in range(steps):
        c5_cur, g = soft_grad(h, n, temp)
        # Nesterov-style momentum
        v = momentum * v - eta * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        # Adaptive step size
        if c5_new > c5_cur:
            eta *= 0.9
        else:
            eta = min(eta * 1.01, 0.5)
        if s % 50 == 0:
            temp *= 0.8  # Decrease temperature (sharpen the max)
    return best_h, best_c5
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        k = -lag
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    return g

def make_structured(n, mode, rng):
    if mode == 0:
        h = np.zeros(n)
        b = n // 4
        h[:b] = 1.0
        h[-b:] = 1.0
    elif mode == 1:
        h = np.zeros(n)
        b = n // 6
        h[:b] = 1.0
        h[n//2 - b:n//2 + b] = 1.0
        h[-b:] = 1.0
    elif mode == 2:
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    else:
        h = 0.5 + 0.3 * rng.standard_normal(n)
    return project_feasible(h, n)

def optimize_core(h, n, steps, t0, deadline, rng, seed_offset=0):
    rng2 = np.random.default_rng(1000 + seed_offset)
    eta = 0.04
    mom = 0.88
    v = np.zeros(n)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stall = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        one_h = 1.0 - h
        corr = np.correlate(h, one_h, mode='full')
        c5_cur = float(np.max(corr) * (2.0 / n))
        k_max = int(np.argmax(corr))
        max_val = corr[k_max]
        # Use top lags within 2% of max for stable gradient
        n_top = 5
        thresh = max_val - 0.02
        top_idx = np.where(corr >= thresh)[0][:n_top]
        if len(top_idx) == 0:
            top_idx = np.array([k_max])
        g = np.zeros(n)
        for kk in top_idx:
            lag = kk - (n - 1)
            g += grad_lag(h, lag, n)
        g /= len(top_idx)
        # Nesterov momentum
        v = mom * v - eta * g
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new)
        h = h_new
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            eta = min(eta * 1.003, 0.15)
        else:
            stall += 1
            if stall > 300:
                eta *= 0.9
                if eta < 0.002:
                    h = project_feasible(h + 0.03 * rng2.standard_normal(n), n)
                    v = np.zeros(n)
                    eta = 0.04
                    stall = 0
        mom = 0.88 + 0.04 * np.sin(it * 0.005)
    return best_h, best_c5

def coordinate_descent(h, n, t0, deadline, rng):
    c5 = compute_c5(h)
    step = 0.02
    for cd_pass in range(10):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = h[i]
            for delta in [step, -step, step * 0.5, -step * 0.5]:
                nv = old + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                h_try = h.copy()
                h_try[i] = nv
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5 - 1e-9:
                    h = h_try
                    c5 = c5_try
                    improved = True
                    break
        if not improved:
            step *= 0.5
            if step < 0.001:
                break
    return h, c5

def upscale(h, n, n2):
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0.0, n - 1, n2)
    h_up = np.interp(x_new, x_old, h)
    return project_feasible(h_up, n2)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15

    # Get starting point
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = make_structured(160, 0, rng)
    n = len(h0)
    h_best = project_feasible(h0.copy(), n)
    c5_best = compute_c5(h_best)

    # Phase 1: Multi-start with structured init (25% budget)
    d1 = t0 + budget_s * 0.25
    starts = [h_best.copy()]
    for m in range(4):
        starts.append(make_structured(n, m, rng))
    # Add a random start
    starts.append(project_feasible(0.5 + 0.4 * rng.standard_normal(n), n))

    for si, sp in enumerate(starts):
        if time.time() > d1:
            break
        hh, cc = optimize_core(sp, n, 20000, t0, d1, rng, si)
        if cc < c5_best:
            c5_best = cc
            h_best = hh.copy()

    # Phase 2: Focused optimization from best (45% budget)
    d2 = t0 + budget_s * 0.70
    hh, cc = optimize_core(h_best, n, 80000, t0, d2, rng, 100)
    if cc < c5_best:

        h_best = hh.copy()

    # Phase 3: Try different n (15% budget)

    for n_alt in [n // 2, n * 3 // 2]:
        if time.time() > d3 or n_alt < 40 or n_alt > 400:
            continue

        hh, cc = optimize_core(h_up, n_alt, 30000, t0, d3, rng, 200)
        if cc < c5_best:
            c5_best = cc

    # Phase 4: Coordinate descent refinement (10% budget)

    h_best, c5_best = coordinate_descent(h_best, n, t0, d4, rng)
```
