    return float(np.max(corr) * dx)

def _corr_vals(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _proj_sum(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            s = np.sum(room)
            if s < 1e-14:
                break
            h = np.clip(h + d * room / s, 0.0, 1.0)
        else:
            s = np.sum(h)
            if s < 1e-14:
                break
            h = np.clip(h + d * h / s, 0.0, 1.0)
    return h

def _grad_shift(h, L):
    n = len(h)
    if L == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    g[L:] += 1.0 - h[:-L]
    g[:-L] -= h[L:]
    return g

def _subgrad_method(h0, n_iter, n_active=3, eta0=0.02, seed=0, time_limit=None, t_start=None):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_c5 = _eval_c5(h)
    best_h = h.copy()
    rng = np.random.default_rng(seed)
    for it in range(n_iter):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        c5_now = cmax * 2.0 / n
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        grad = np.zeros(n)
        for ki in top_idx:
            lag = ki - (n - 1)
            grad += _grad_shift(h, abs(lag))
        for attempt in range(3):
            eta = eta0 / (1.0 + 0.005 * it)
            h_try = h - eta * grad
            h_try = _proj_sum(h_try, target)
            c5_try = np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * 2.0 / n
            if c5_try <= c5_now * 1.002:
                h = h_try
                break
            eta *= 0.1
        else:
            h = _proj_sum(h + rng.normal(0, 0.003, n), target)
    return best_h, best_c5

def _slsqp_opt(h0, alpha=200, max_iter=300):
    n = len(h0)
    dx = 2.0 / n
    cmax_init = np.max(np.correlate(h0, 1.0 - h0, mode='full'))

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cm = np.max(corr)
        shifted = corr - cm
        return cm + np.log(np.sum(np.exp(alpha * shifted))) / alpha

    def eq_fun(h):
        return np.sum(h) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': eq_fun}],
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success or res.fun < obj(h0):
            return _proj_sum(res.x, n / 2.0)
    except Exception:
        pass
    return h0.copy()

def _mm_opt(h0, max_iter=300):
    n = len(h0)
    dx = 2.0 / n
    t0 = _eval_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    def mm_obj(x):
        return x[-1]

    def mm_ineq(x):
        hv = x[:-1]
        t = x[-1]
        c = np.correlate(hv, 1.0 - hv, mode='full') * dx
        return t - c

    def mm_eq(x):
        return np.sum(x[:-1]) - n / 2.0

    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = minimize(mm_obj, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': mm_ineq},
                                    {'type': 'eq', 'fun': mm_eq}],
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.success:
            return _proj_sum(res.x[:-1], n / 2.0)
    except Exception:
        pass
    return h0.copy()

def _make_start(n, rng, style):
    if style == 0:
        h = rng.random(n)
    elif style == 1:
        h = np.array([0.9 if (i % 4) < 2 else 0.1 for i in range(n)])
    elif style == 2:
        x = np.linspace(0, 4 * np.pi, n, endpoint=False)
        h = 0.5 + 0.4 * np.sin(x)
    elif style == 3:
        half = n // 2
        h = np.concatenate([np.ones(half), np.zeros(n - half)])
        # Add smoothing
        k = 3
        kernel = np.ones(k) / k
        h = np.convolve(h, kernel, mode='same')
    else:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n) / n * (style + 1))
    return _proj_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    elapsed = lambda: time.time() - t0
    time_left = lambda: budget_s - elapsed()
    rng = np.random.default_rng(seed)

    best_c5 = np.inf
    best_h = None

    # Start from initial_h_values if available
    if 'initial_h_values' in dir() or True:
        try:
            h_init = np.array(initial_h_values, dtype=float)
            n_init = len(h_init)
            h_init = _proj_sum(h_init, n_init / 2.0)
            c5 = _eval_c5(h_init)
            best_c5 = c5
            best_h = h_init.copy()

            # Subgradient refinement on initial
            if time_left() > 100:
                h_sg, c5_sg = _subgrad_method(h_init, 8000, 3, 0.015, seed,
                                             time_limit=budget_s - 200, t_start=t0)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()

            # SLSQP polish with various alphas
            for alpha in [80, 150, 300, 600]:
                if time_left() < 60:
                    break
                h_new = _slsqp_opt(best_h, alpha, 350)
                c5 = _eval_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new

            # Minimax polish
            if time_left() > 80:
                h_new = _mm_opt(best_h, 400)
                c5 = _eval_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new
        except NameError:
            pass

    # Try other sizes with structured starts
    sizes_to_try = [48, 56, 64, 80]
    for n in sizes_to_try:
        if time_left() < 50:
            break
        for style in range(5):
            if time_left() < 30:
                break
            h0 = _make_start(n, rng, style)
            # Quick subgradient
            h_sg, c5_sg = _subgrad_method(h0, 3000, 3, 0.02, rng.integers(1000),
                                         time_limit=budget_s - 60, t_start=t0)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg
            # SLSQP polish
            if time_left() > 40:
                h_new = _slsqp_opt(h_sg, 200, 300)
                c5 = _eval_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new

    # Perturbation + re-optimize on best
    n_best = len(best_h)
    for trial in range(20):
        if time_left() < 30:
            break
        for scale in [0.01, 0.02, 0.035]:
            if time_left() < 20:
                break
            h_p = _proj_sum(best_h + rng.normal(0, scale, n_best), n_best / 2.0)
            h_sg, c5_sg = _subgrad_method(h_p, 2000, 3, 0.01, rng.integers(10000),
                                         time_limit=budget_s - 50, t_start=t0)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg
            if time_left() > 40:
                h_new = _slsqp_opt(h_sg, 300, 250)
                c5 = _eval_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new

    # Final polish
    if best_h is not None and time_left() > 30:
        h_fin = _slsqp_opt(best_h, 400, 500)
        c5 = _eval_c5(h_fin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fin
        if time_left() > 20:
            h_fin2 = _mm_opt(best_h, 350)
            c5 = _eval_c5(h_fin2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_fin2

    # One more subgradient pass on the very best
    if best_h is not None and time_left() > 40:
        h_sg, c5_sg = _subgrad_method(best_h, 5000, 5, 0.008, 999,
                                     time_limit=budget_s - 10, t_start=t0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    if best_h is None:
        best_h = _proj_sum(0.5 + 0.4 * np.sin(np.arange(80) * 2 * np.pi / 40), 40.0)
        best_c5 = _eval_c5(best_h)

    best_h = _proj_sum(best_h, len(best_h) / 2.0)
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
