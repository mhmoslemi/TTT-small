                    best_val = v

def _optimize_coord_hessian(h, i, n):
    """Optimize one coordinate using the kink-point method."""
    # Compute slope and intercept for each lag
    # corr_k(t) = slope_k * t + intercept_k (for k ≠ n-1)
    # corr_{n-1}(t) = intercept_{n-1} + t - t²
    
    h0 = h.copy()
    h0[i] = 0.0
    h1 = h.copy()
    h1[i] = 1.0
    c0 = _corr(h0)
    c1 = _corr(h1)
    
    slope = c1 - c0
    intercept = c0
    
    # For k = n-1 (lag 0), the actual function is intercept + t - t²
    # slope[n-1] should be 0 (since h=0 and h=1 both give 0 contribution)
    # Let's verify: c0[n-1] = sum h[j](1-h[j]) with h[i]=0
    # c1[n-1] = sum h[j](1-h[j]) with h[i]=1
    # Both have h[i] term = 0, so c0[n-1] = c1[n-1] = sum_{j≠i} h[j](1-h[j])
    # So slope[n-1] = 0. Good.
    
    # For k ≠ n-1: corr_k(t) = slope[k]*t + intercept[k]
    # For k = n-1: corr_k(t) = intercept[k] + t - t²
    
    # Find all kink points in [0, 1]
    candidates = [0.0, 1.0]
    
    # Kinks between two linear functions k and j (both ≠ n-1)
    for k in range(2*n-1):
        if k == n-1:
            continue
        for j in range(k+1, 2*n-1):
            if j == n-1:
                continue
            denom = slope[k] - slope[j]
            if abs(denom) < 1e-15:
                continue
            t = (intercept[j] - intercept[k]) / (slope[k] - slope[j])
            if 0.0 <= t <= 1.0:
                candidates.append(t)
    
    # Kinks between linear k and quadratic (n-1):
    # slope[k]*t + intercept[k] = intercept[n-1] + t - t²
    # t² + (slope[k]-1)*t + (intercept[k] - intercept[n-1]) = 0
    for k in range(2*n-1):
        if k == n-1:
            continue
        a = 1.0
        b = slope[k] - 1.0
        c = intercept[k] - intercept[n-1]
        disc = b*b - 4*a*c
        if disc < 0:
            continue
        sqrt_disc = np.sqrt(disc)
        for t_candidate in [(-b + sqrt_disc)/2, (-b - sqrt_disc)/2]:
            if 0.0 <= t_candidate <= 1.0:
                candidates.append(t_candidate)
    
    # Evaluate objective at all candidates
    best_v = np.inf
    best_t = 0.5
    for t in candidates:
        h_try = h.copy()
        h_try[i] = t
        v = np.max(_corr(h_try))
        if v < best_v:
            best_v = v
            best_t = t
    
    h[i] = best_t
    h = _proj(h, n)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = np.clip(c - cmax, -40.0 * eps, 0.0)
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = np.clip(c - cmax, -40.0 * eps, 0.0)
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g * dx

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _coord_opt(h, i, n):
    """Exact 1D optimization of coordinate i via kink-point method."""
    h[i] = 0.0
    c0 = _corr(h)
    h[i] = 1.0
    c1 = _corr(h)
    h[i] = h[i]  # restore not needed, we work with slopes
    slope = c1 - c0
    intercept = c0
    # Save original h[i]
    # Build candidate set
    cands = [0.0, 1.0]
    nn = 2 * n - 1
    mid = n - 1
    for k in range(nn):
        if k == mid:
            continue
        for j in range(k + 1, nn):
            if j == mid:
                continue
            denom = slope[k] - slope[j]
            if abs(denom) < 1e-15:
                continue
            t = (intercept[j] - intercept[k]) / denom
            if -0.001 <= t <= 1.001:
                cands.append(max(0.0, min(1.0, t)))
    for k in range(nn):
        if k == mid:
            continue
        bq = slope[k] - 1.0
        cq = intercept[k] - intercept[mid]
        disc = bq * bq - 4.0 * cq
        if disc >= 0.0:
            sq = np.sqrt(disc)
            for tr in [(-bq + sq) / 2.0, (-bq - sq) / 2.0]:
                if -0.001 <= tr <= 1.001:
                    cands.append(max(0.0, min(1.0, tr)))
    arr = np.array(cands)
    # Vectorized evaluation

    vals[:, mid] = intercept[mid] + arr - arr * arr
    max_vals = vals.max(axis=1)
    idx = int(np.argmin(max_vals))
    best_t = arr[idx]
    h[i] = best_t
    h = _proj(h, n)
    return h
