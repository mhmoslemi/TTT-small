    """Project to exact binary with n_ones ones"""

    b = np.zeros(n)
    b[idx] = 1.0
    return b

def greedy_swap(h, max_iters=500):
    n = len(h)
    best_c5 = compute_c5(h)
    for _ in range(max_iters):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_improve = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0

        j = zeros[rng.randint(len(zeros))]

import numpy as np
import time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_binary(h, n_ones):
    n = len(h)
    idx = np.argsort(h)[::-1][:n_ones]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def greedy_swap(h, max_iters=500, time_limit=15):
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    for _ in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_improve = 0.0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                improve = best_c5 - c
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        best_c5 = compute_c5(h)
    return h, best_c5

def simulated_anneal(h, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T_start = 0.04
    T_end = 0.002
    t0 = time.time()
    stuck = 0
    while time.time() - t0 < time_budget:
        if stuck > 2000:
            T = T_start * 0.5
            stuck = 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if rng.random() < 0.15 and len(ones) >= 2 and len(zeros) >= 2:
            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            h[i1] = h[i2] = 0.0
            h[j1] = h[j2] = 1.0
            c_new = compute_c5(h)
            delta = c_new - c5
            if delta > 0:
                progress = (time.time() - t0) / time_budget
                T = T_start * (T_end / T_start) ** min(progress, 1.0)
                if not (delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-8))):
                    h[i1] = h[i2] = 1.0
                    h[j1] = h[j2] = 0.0
                continue
            else:
                c5 = c_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h.copy()
                    stuck = 0
                continue
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - c5
        if delta > 0:

                h[i] = 1.0

            stuck = 0

def iterated_local_search(h, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_budget:
        n_flip = rng.randint(2, 5)
        h_cur = best_h.copy()
        for _ in range(n_flip):

            j = zeros[rng.randint(len(zeros))]

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    return best_h, best_c5

def structured_init(n, n_ones, seed, pattern):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    if pattern == 0:
        positions = np.arange(n_ones) * (n // n_ones)
        h[positions] = 1.0
    elif pattern == 1:
        step = n // n_ones
        offset = rng.randint(step)
        positions = np.arange(offset, n, step)[:n_ones]
        h[positions] = 1.0
    elif pattern == 2:
        positions = rng.choice(n, n_ones, replace=False)
        h[positions] = 1.0
    elif pattern == 3:
        phase = np.arange(n) * (2.0 * math.pi * n_ones / n)
        positions = np.arange(n)[np.sin(phase) > np.sin(phase).mean()]
        if len(positions) > n_ones:
            positions = rng.choice(positions, n_ones, replace=False)
        elif len(positions) < n_ones:
            remaining = np.setdiff1d(np.arange(n), positions)
            extra = rng.choice(remaining, n_ones - len(positions), replace=False)
            positions = np.concatenate([positions, extra])
        h[positions] = 1.0
    else:
        positions = rng.choice(n, n_ones, replace=False)
        h[positions] = 1.0
    if h.sum() != n_ones:
        h = np.zeros(n)
        h[rng.choice(n, n_ones, replace=False)] = 1.0
    return h
