    lo, hi = -2.0, 2.0

def gradient_step(h, dx, tau=50.0):

    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    exps = np.exp(shifted)
    weights = exps / exps.sum()

        h_try, c5_try = optimize_frac(h_try, 25, seed + n_try * 13, lr=0.05, tau=35.0)
        rng_bt = np.random.RandomState(seed + n_try * 7)
        h_bt = make_binary(h_try, n_try // 2, rng_bt)
        h_st, c5_st = simulated_anneal(h_bt, 30, seed + n_try * 31 + 4000, T_start=0.18)
        h_et, c5_et = exhaustive_single_swap(h_st, 10, max_rounds=400)
        if c5_et < best_c5:
            best_h = h_et
            best_c5 = c5_et
            n = n_try

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
