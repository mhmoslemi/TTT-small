import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        if not np.any(mask):
            break
        adjust = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + adjust, 0.0, 1.0)
    return h

def to_binary(h, rng, n):
    """Convert to binary with exact n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    s = int(np.sum(h_bin))
    if s > target:
        # Need to remove (s - target) ones
        ones_idx = np.where(h_bin == 1)[0]
        rng.shuffle(ones_idx)
        for i in range(s - target):
            # Prefer to remove those with highest h value (closest to 0.5 means less "committed")
            h_bin[ones_idx[i]] = 0
    elif s < target:
        zeros_idx = np.where(h_bin == 0)[0]
        rng.shuffle(zeros_idx)
        for i in range(target - s):
            h_bin[zeros_idx[i]] = 1
    return h_bin

            if len(ones) == 0 or len(zeros) == 0:

            j = zeros[rng.randint(len(zeros))]
            curr_h[i] = 0
            curr_h[j] = 1
            new_c5 = compute_c5(curr_h)

            curr_h[ii1] = curr_h[ii2] = 0

        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

    scores = h.copy()

    curr = h_init.copy()
    curr_c5 = best_c5

        curr[j] = 1

            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr.copy()
                stall = 0
            else:
                stall += 1
        else:
            if rng.random() < np.exp(-(new_c5 - curr_c5) / temp):
                curr_c5 = new_c5
                stall += 1
            else:
                curr[i] = 1
                curr[j] = 0

        temp *= 0.99998
        if temp < 0.0005:
            temp = 0.0005

                z = np.where(curr == 0)[0]

                b = z[rng.randint(len(z))]
                curr[a] = 0
                curr[b] = 1
            curr_c5 = compute_c5(curr)

    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def obj(x, beta):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        shifted = beta * (c - m)
        shifted -= shifted.max()
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5
