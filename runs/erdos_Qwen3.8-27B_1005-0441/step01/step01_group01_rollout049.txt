def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        # Add uniformly where possible
        free_up = h < 1.0
        free_dn = h > 0.0
        if d > 0:
            mask = free_up
        else:
            mask = free_dn
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    # Final correction
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def corr_gradient(h, k_star, n):
    """Compute gradient of corr[k_star] w.r.t. h."""
    dx = 2.0 / n
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        m = n - shift
        grad[shift:shift+m] += (1.0 - h[0:m])
        grad[0:m] += -h[shift:shift+m]
    else:
        m = n + shift
        grad[0:m] += (1.0 - h[-shift:])
        grad[-shift:] += -h[0:m]
    return grad * dx

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def corr_lag_gradient(h, shift, n):
    """Gradient of corr at a specific shift w.r.t. h."""
    grad = np.zeros(n)
    if shift >= 0:
        m = n - shift
        if m <= 0:
            return grad
        grad[shift:shift+m] += (1.0 - h[:m])
        grad[:m] -= h[shift:shift+m]
    else:
        m = n + shift
        if m <= 0:
            return grad
        grad[:m] += (1.0 - h[-shift:])
        grad[-shift:] -= h[:m]
    return grad

def multi_subgradient(h, n, frac=0.9):
    """Average subgradient over top fractional lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    threshold = frac * c_max
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        grad += corr_lag_gradient(h, shift, n)
    grad /= max(len(active), 1)
    return grad * dx, c_max * dx

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def smooth_obj_jac(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        shift = k - (n - 1)
        grad += weights[k] * corr_lag_gradient(h, shift, n)
    return grad * dx

def simulated_anneal_binary(n, t0, budget, rng, max_iter=50000):
    """Simulated annealing on binary patterns."""
    h = np.zeros(n)
    ones = np.arange(n)
    rng.shuffle(ones)
    h[ones[:n//2]] = 1.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = 0.05
    cool = 0.999
    for it in range(max_iter):
        if time.time() - t0 > budget:
            break
        # Propose: swap a 1 and a 0
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T *= cool
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() > 0:
            h[mask] += resid / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    w = exp_shifted / total
    obj = (m + np.log(total) / alpha) * dx
    # Gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        A = max(0, k - n + 1)
        B = min(k, n - 1)
        lo = k - B
        hi = k - A
        grad[A:B+1] += w[k] * (1.0 - 2.0 * h[lo:hi+1][::-1])
    grad *= dx
    return obj, grad

def smooth_obj_wrapper(h, alpha):
    return smooth_obj_and_grad(h, alpha)[0]

def smooth_jac_wrapper(h, alpha):
    return smooth_obj_and_grad(h, alpha)[1]

def make_block_pattern(n, block_frac=0.25):
    h = np.zeros(n)
    b = max(1, int(n * block_frac))
    h[:b] = 1.0
    h[n - b:] = 1.0
    return project_feasible(h, n)

def make_fourier_init(n, freqs=None, amps=None):
    x = np.linspace(0, 2, n, endpoint=False)
    if freqs is None:
        freqs = [1, 2, 3]
        amps = [0.3, 0.2, 0.1]
    h = np.full(n, 0.5)
    for f, a in zip(freqs, amps):
        h += a * np.cos(2 * np.pi * f * x / 2.0)
    return project_feasible(h, n)

def subgrad_descent(h, n, budget, step_init=1.0, frac=0.9, max_iter=10000):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > budget:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5 = c_max * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        thresh = frac * c_max
        active = np.where(corr >= thresh)[0]
        grad = np.zeros(n)
        for k in active:
            A = max(0, k - n + 1)
            B = min(k, n - 1)
            lo = k - B
            hi = k - A
            grad[A:B+1] += (1.0 - 2.0 * h[lo:hi+1][::-1])
        grad *= dx / max(len(active), 1)
        step = step_init / (1.0 + 0.05 * it)
        h = project_feasible(h - step * grad, n)
    return best_h, best_c5

    rng = np.random.default_rng(seed)
