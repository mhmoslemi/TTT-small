This makes coordinate descent quite tractable! For each coordinate:

import numpy as np
from scipy.optimize import minimize
import time


def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _corr_vec(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h


def _softmax_obj(h, alpha):
    corr = _corr_vec(h)
    cmax = np.max(corr)
    shifted = corr - cmax
    return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha


def _sm_wrapper(x, alpha):
    return _softmax_obj(x, alpha)


def _eq_sum(h):
    return float(np.sum(h) - len(h) / 2.0)


def _slsqp_opt(h0, alpha, max_iter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_wrapper, h0, args=(alpha,), method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _eq_sum}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.success or res.fun < _softmax_obj(h0, alpha):
            return res.x
    except Exception:
        pass
    return h0.copy()


def _block_h(n, perm=None):
    if perm is None:
        perm = np.tile([0, 0, 1], n // 3 + 1)[:n]
    h = np.where(perm == 0, 0.82, 0.24).astype(float)
    return _fix_sum(h, n)


def _sin_h(n, freq=1):
    x = np.linspace(0, 2.0 * np.pi * freq, n, endpoint=False)
    h = 0.5 + 0.30 * np.sin(x)
    return _fix_sum(h, n)


def _coord_descent(h, max_sweeps=30, time_limit=None, t_start=None):
    n = len(h)
    h = h.copy()
    best_c5 = _eval_c5(h)
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            if time_limit is not None and time.time() - t_start > time_limit:
                break
            h[i] = 0.0
            c0 = _corr_vec(h)
            h[i] = 1.0
            c1 = _corr_vec(h)
            d = c1 - c0
            ts = np.linspace(0.0, 1.0, 41)
            fvals = np.max(c0[None, :] + ts[:, None] * d[None, :], axis=1)
            t_best = ts[int(np.argmin(fvals))]
            if 0.0 < t_best < 1.0:
                idx = int(np.argmin(fvals))
                lo = max(0.0, ts[max(0, idx - 1)])
                hi = min(1.0, ts[min(len(ts) - 1, idx + 1)])
                ts2 = np.linspace(lo, hi, 31)
                f2 = np.max(c0[None, :] + ts2[:, None] * d[None, :], axis=1)
                t_best = ts2[int(np.argmin(f2))]
            h[i] = t_best
            h = _project(h, n)
            c5 = _eval_c5(h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
        if not improved:
            break
    return h
