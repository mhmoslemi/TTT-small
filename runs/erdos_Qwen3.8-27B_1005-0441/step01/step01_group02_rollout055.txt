        smooth_max = mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta
        return smooth_max * dx
    
    def jac(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        w = np.exp(beta * (corr - mc))
        w /= w.sum()
        # grad of corr[k] w.r.t. x[i]
        # corr[k] = sum_j x[j] * (1-x)[j-k+n-1]
        # d corr[k] / d x[i] = (1-x[i-k+n-1])*I(valid) - x[i+k-1... ]
        # Let me use the formula: for shift d = k-(n-1):
        # corr at shift d = sum_{valid} x[i]*(1-x[i+d])
        # d/dx[i] = (1-x[i+d])*I(i+d valid) - x[i-d]*I(i-d valid)
        g = np.zeros(n)
        for i in range(n):
            # i as the x[i] factor: contributes (1-x[i+d]) for each valid d where i+d in [0,n-1]
            # but we need to sum w[k] * d corr[k] / d x[i]
            # corr[k] for k = 0..2n-2, shift d = k-(n-1)
            # d corr/dx[i] = (1-x[i+d])*I(0<=i+d<=n-1) - x[i-d]*I(0<=i-d<=n-1) for shift d
            # But which k? For each d from -(n-1) to n-1, k = d+n-1
            # Sum over all valid d:
            # = sum_d w[d+n-1] * [(1-x[i+d])*I(0<=i+d<=n-1) - x[i-d]*I(0<=i-d<=n-1)]
            val = 0.0
            # i+d in [0,n-1]: d in [-i, n-1-i]
            # i-d in [0,n-1]: d in [i-n+1, i]
            for d in range(-i, n-i):
                if i + d < n:
                    val += w[d + n - 1] * (1.0 - x[i + d])
            for d in range(i - n + 1, i + 1):
                if i - d >= 0:
                    val -= w[d + n - 1] * x[i - d]
            g[i] = val * dx
        return g
