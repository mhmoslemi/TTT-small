import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += min(diff, tot) * room / tot
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            vals = np.where(mask, h, 0.0)
            tot = vals.sum()
            if tot < 1e-15:
                break
            h -= min(-diff, tot) * vals / tot
    return np.clip(h, 0.0, 1.0)

def _softmax_obj(x, alpha, n):
    h = x
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def _eq_sum(h):
    return float(h.sum() - len(h) / 2.0)

def _slsqp_refine(h0, alpha, max_iter=300, ftol=1e-10):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def obj(h):
        return _softmax_obj(h, alpha, n)
    try:
        res = minimize(obj, h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _eq_sum}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < np.inf:
            return res.x
    except Exception:
        pass
    return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(800):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            room = np.where(mask, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += min(diff, tot) * room / tot
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            vals = np.where(mask, h, 0.0)
            tot = vals.sum()
            if tot < 1e-15:
                break
            h -= min(-diff, tot) * vals / tot
    return np.clip(h, 0.0, 1.0)

def _softmax_obj_factory(n, alpha):
    def obj(h):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr)
        shifted = corr - cmax
        return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    return obj

def _eq_sum(h):
    return float(h.sum() - len(h) / 2.0)

def _slsqp_refine(h0, alpha, max_iter=300, ftol=1e-11):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    obj = _softmax_obj_factory(n, alpha)
    try:
        res = minimize(obj, h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _eq_sum}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < np.inf:
            return res.x
    except Exception:
        pass
    return h0.copy()

def _pso_optimize(h_init, n, n_part, max_iter, deadline, rng):
    v_hi, v_lo = 0.12, 0.01
    c1, c2 = 1.5, 2.0

    P = np.zeros((n_part, n))
    V = np.zeros((n_part, n))
    PB = np.zeros((n_part, n))
    PB_c = np.full(n_part, np.inf)

    P[0] = h_init.copy()
    for i in range(1, n_part):
        r = rng.random()
        if r < 0.25:
            freq = rng.integers(2, 10)
            x = np.arange(n)
            ph = rng.random() * 6.2832
            P[i] = 0.5 + rng.uniform(0.2, 0.42) * np.sin(2 * np.pi * freq * x / n + ph)
        elif r < 0.55:
            P[i] = h_init + rng.normal(0, 0.07, n)
        elif r < 0.75:
            per = rng.integers(3, 11)
            fr = rng.uniform(0.25, 0.7)
            P[i] = np.array([1.0 if (j % per) < int(fr * per) else 0.0 for j in range(n)])
        else:
            P[i] = rng.random(n)
        P[i] = _fix_sum(P[i], n)

    V = rng.uniform(-v_hi, v_hi, (n_part, n))
    best_h = h_init.copy()
    best_c = _eval_c5(best_h)
    stagnant = 0

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        inertia = 0.92 - 0.55 * (it / max_iter)
        vmax = v_hi - (v_hi - v_lo) * (it / max_iter)
        improved = False

        for i in range(n_part):
            c = _eval_c5(P[i])
            if c < PB_c[i]:
                PB_c[i] = c
                PB[i] = P[i].copy()
            if c < best_c:
                best_c = c
                best_h = P[i].copy()
                improved = True

        if not improved:
            stagnant += 1
        else:
            stagnant = 0

        if stagnant > 60:
            nrestart = max(3, n_part // 4)
            for _ in range(nrestart):
                idx = rng.integers(1, n_part)
                scale = 0.04 + 0.03 * min(stagnant / 40.0, 2.0)
                P[idx] = best_h + rng.normal(0, scale, n)
                P[idx] = _fix_sum(P[idx], n)
                V[idx] = rng.uniform(-0.04, 0.04, n)
            stagnant = 0

        gb = best_h
        for i in range(n_part):
            r1 = rng.random(n)
            r2 = rng.random(n)
            V[i] = (inertia * V[i]
                    + c1 * r1 * (PB[i] - P[i])
                    + c2 * r2 * (gb - P[i]))
            np.clip(V[i], -vmax, vmax, out=V[i])
            P[i] = P[i] + V[i]
            P[i] = _fix_sum(P[i], n)

    return best_h, best_c

def _resample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    hr = np.interp(x_new, x_old, np.concatenate([h, [h[0]]]))
    return _fix_sum(hr, n_new)

def _coord_descent(h, n, steps=50, deadline=None):
    best_c = _eval_c5(h)
    delta = 0.02
    for _ in range(steps):
        if deadline and time.time() >= deadline:
            break
        improved = False
        for i in range(n):
            if deadline and time.time() >= deadline:
                break
            for sign in (1, -1):
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + sign * delta, 0.0, 1.0)
                h_try = _fix_sum(h_try, n)
                c = _eval_c5(h_try)
                if c < best_c - 1e-14:
                    h = h_try
                    best_c = c
                    improved = True
                    break
        if not improved and delta > 0.002:
            delta *= 0.7
    return h, best_c
