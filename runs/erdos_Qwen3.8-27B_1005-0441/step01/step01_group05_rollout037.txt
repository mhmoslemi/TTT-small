import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = float(n) / 2.0
    s = np.sum(h)
    if abs(s - t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > t:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    d = t - np.sum(h)
    if abs(d) > 1e-13:
        m = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(m):
            h[m] += d / np.sum(m)
            h = np.clip(h, 0.0, 1.0)
    return h

def _fun(x, temp, n, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mx = np.max(corr)
    e = np.exp((corr - mx) / temp)
    Z = np.sum(e)
    w = e / Z
    f = mx + temp * np.log(Z)
    v = 1.0 - x[::-1]
    t1 = np.correlate(w, v, mode='valid')
    t2 = np.correlate(w, x, mode='valid')[::-1]
    g = dx * (t1 - t2)
    sd = np.sum(x) - float(n) / 2.0
    f += pen * sd * sd
    g = g + 2.0 * pen * sd
    return f, g

    best = _project(h0.copy(), n)

                v = compute_c5(x)

def _pert(h, sc, seed):
    np.random.seed(seed)
    return _project(h + sc * np.random.randn(len(h)), len(h))

def _two_block(n, gap_frac):
    h = np.zeros(n)
    half = n // 4
    g = int(n * gap_frac)
    h[:half] = 1.0
    start2 = n // 2 + g
    end2 = min(n, start2 + half)
    h[start2:end2] = 1.0
    return _project(h, n)

def _periodic(n, period):
    h = np.zeros(n)
    for i in range(n):
        if i % period < period // 2:

    return _project(h, n)

    h_new = np.interp(x_new, x_old, h)
    return _project(h_new, n_new)

        h_full[free_idx] = xf
        return _fun(h_full, temp, n, pen)

    best_x = x0
    best_v = compute_c5(h_s)
    for temp in [0.01, 0.002, 0.0005]:
        if time.time() - t0 > max_t - 1:
            break
        try:
            r = minimize(_sub_fun, best_x, args=(temp, n, 5000.0),
                        method='L-BFGS-B', jac=True, bounds=bnd_f,
                        options={'maxiter': 100, 'gtol': 1e-9})
            h_full = h_s.copy()
            h_full[free_idx] = np.clip(r.x, 0.0, 1.0)
            h_full = _project(h_full, n)
            v = compute_c5(h_full)
            if v < best_v:
                best_v = v
                h_s = h_full.copy()
                best_x = h_full[free_idx].copy()
        except Exception:
            pass
    return h_s, best_v

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 160
        h0 = np.full(n, 0.5)
    h0 = _project(h0, n)
    best_h, best_c5 = h0.copy(), compute_c5(h0)

    # Phase 1: Optimize initial construction
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        h, c = _opt(h0, n, t0, min(220, rem - 30), seed)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: Structured initializations
    inits = [_two_block(n, 0.08), _two_block(n, 0.12),
             _two_block(n, 0.2), _periodic(n, 4), _periodic(n, 6)]
    for i, hi in enumerate(inits):
        rem = budget_s - (time.time() - t0)
        if rem < 45:
            break
        h, c = _opt(hi, n, t0, min(35, rem - 15), seed + 400 + i)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: Random perturbations from best
    rem = budget_s - (time.time() - t0)
    for r in range(35):
        if rem < 30:
            break
        sc = 0.015 + 0.055 * ((r % 7) / 6.0)
        hp = _pert(best_h, sc, seed + 800 + r)
        mt = min(22, rem - 10)
        if mt < 10:
            break
        h, c = _opt(hp, n, t0, mt, seed + 100 + r)
        if c < best_c5:
            best_h, best_c5 = h, c
        rem = budget_s - (time.time() - t0)

    # Phase 4: Snap-and-refine
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        h, c = _snap_and_refine(best_h, n, t0, min(30, rem - 5), seed + 5000)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 5: Final polish at very low temperature
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        h, c = _opt(best_h, n, t0, rem - 3, seed + 9999)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = _project(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
