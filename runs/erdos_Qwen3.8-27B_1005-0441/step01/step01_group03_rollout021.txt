    g = np.zeros(n)
    # h[j] * (1 - h[j+lag]) for j where j+lag in range
    # h[j-lag] * (1 - h[j]) for j where j-lag in range

        for j in range(n - ll):

            g[j + ll] += 1.0 - h[j]

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] += diff / cnt
            np.clip(hp, 0, 1, out=hp)
    return hp

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, d, n):
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[d:] += 1.0 - h[:n - d]
        g[:n - d] -= h[d:]
    else:
        l = -d
        g[:n - l] += 1.0 - h[l:]
        g[l:] -= h[:n - l]
    return g

def optimize_h(h, n, deadline, rng, top_k=8, t_init=3.0):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    lr = 0.4
    t = t_init
    stalled = 0
    n_corr = 2 * n - 1

    while time.time() < deadline:
        ov = get_overlaps(h, n)
        c_cur = float(np.max(ov))

        kk = min(top_k, n_corr)
        top_idx = np.argsort(ov)[::-1][:kk]
        ov_top = ov[top_idx]
        w = np.exp((ov_top - ov_top[0]) / t)
        w /= w.sum()

        g = np.zeros(n)
        for i in range(kk):
            d = int(top_idx[i]) - (n - 1)
            g += w[i] * grad_lag(h, d, n)

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn

        improved = False
        for factor in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.03]:
            step = lr * factor
            h_new = project_feasible(h - step * g, n)
            c_new = get_c5(h_new, n)
            if c_new < c_cur - 1e-13:
                h = h_new
                improved = True
                if c_new < best_c5 - 1e-12:
                    best_c5 = c_new
                    best_h = h.copy()
                    stalled = 0
                break

        if not improved:
            stalled += 1
            if stalled % 100 == 0:
                lr *= 0.7
                t *= 1.3
            if stalled > 400:
                sigma = 0.02 + 0.02 * (stalled % 4) * 0.25
                h = best_h + rng.randn(n) * sigma
                h = project_feasible(h, n)
                lr = 0.4
                t = t_init
                stalled = 0
        else:
            if stalled > 30:
                lr = min(lr * 1.05, 1.5)
            stalled = max(0, stalled - 3)

        t = max(0.8, t * 0.9995)

    return best_h, best_c5

def smooth_perturb(h, n, rng, alpha):
    h2 = h.copy()
    half = (alpha - 1) // 2
    for i in range(half, n - half):
        h2[i] += alpha * 0.1 * (h[i - half] - 2 * h[i] + h[i + half])
    return project_feasible(h2, n)
