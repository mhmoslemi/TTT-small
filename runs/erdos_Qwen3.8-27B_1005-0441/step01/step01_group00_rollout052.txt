import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += d / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def _lag_grad(h, n, k):
    """Gradient of correlation at lag k w.r.t. h, scaled by dx"""
    dx = 2.0 / n
    m = k - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        p = -m
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g * dx

def _smooth_grad(h, n, eps):
    """Gradient of smooth max (log-sum-exp) of c5"""
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    w = np.exp((c - cmax) / eps)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += w[k] * (1.0 - h[:n - m])
            g[:n - m] -= w[k] * h[m:]
        else:
            p = -m
            g[:n - p] += w[k] * (1.0 - h[p:])
            g[p:] -= w[k] * h[:n - p]
    return g * dx

def _fista(h0, n, eps, max_iter, time_limit, t0):
    """FISTA for smooth max of c5"""
    h = h0.copy()
    y = h0.copy()
    t_aa = 1.0
    best = _c5(h)
    best_h = h.copy()
    lr = 0.003  # 1/L where L is Lipschitz constant estimate
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        g = _smooth_grad(y, n, eps)
        h_new = _proj(y - lr * g, n)
        t_new = (1 + np.sqrt(1 + 4 * t_aa * t_aa)) / 2
        y = h_new + (t_aa - 1) / t_new * (h_new - h)
        h = h_new
        t_aa = t_new
        f = _c5(h)
        if f < best:
            best = f
            best_h = h.copy()
    return best_h, best

def _subgrad_refine(h, n, max_iter, time_limit, t0):

    h = h.copy()
    best = _c5(h)
    best_h = h.copy()

    def eval_swap(h, i, j, delta):

        c = _corr(h)

        m_k = int(np.argmax(c))

                dc_j -= h[j - m]

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += d / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def _lag_grad(h, n, k):
    dx = 2.0 / n
    m = k - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        p = -m
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g * dx

def _smooth_grad(h, n, eps):
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    w = np.exp((c - cmax) / eps)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += w[k] * (1.0 - h[:n - m])
            g[:n - m] -= w[k] * h[m:]
        else:
            p = -m
            g[:n - p] += w[k] * (1.0 - h[p:])
            g[p:] -= w[k] * h[:n - p]
    return g * dx

def _fista(h0, n, eps, max_iter, time_limit, t0):
    h = h0.copy()
    y = h0.copy()
    t_aa = 1.0
    best = _c5(h)
    best_h = h.copy()
    lr = eps / (2.0 * n)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        g = _smooth_grad(y, n, eps)
        h_new = _proj(y - lr * g, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_aa * t_aa)) / 2.0
        y = h_new + (t_aa - 1.0) / t_new * (h_new - h)
        h = h_new
        t_aa = t_new
        f = _c5(h)
        if f < best:
            best = f
            best_h = h.copy()
    return best_h, best

def _subgrad_refine(h, n, max_iter, time_limit, t0):
    h = h.copy()
    best = _c5(h)
    best_h = h.copy()
    lr = 0.004
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        c = _corr(h)
        cmax = c.max()
        thresh = cmax * 0.97
        active = np.where(c >= thresh)[0]
        if len(active) > 12:
            active = active[np.argsort(-c[active])[:12]]
        g = np.zeros(n)
        for k in active:
            g += _lag_grad(h, n, k)
        g /= len(active)
        h_new = _proj(h - lr * g, n)
        v = _c5(h_new)
        if v < best - 1e-14:
            best = v

            h = h_new
            lr = min(lr * 1.005, 0.01)
        else:

            if lr < 1e-9:

    return best_h, best
