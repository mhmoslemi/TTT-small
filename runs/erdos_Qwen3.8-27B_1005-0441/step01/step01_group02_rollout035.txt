        hi2 = min(n, d + n)

import numpy as np
from scipy.optimize import minimize
import time

_g_tgt = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_wf(h, target):
    n = len(h)
    h = np.clip(h, -1.0, 2.0)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(h - t, 0.0, 1.0)

def sm_grad(h, beta, top_k=0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mc = float(corr.max())
    e = np.exp(beta * (corr - mc))
    es = e.sum()
    val = mc + np.log(es) / beta
    w = e / es
    if 0 < top_k < len(w):
        idx = np.argsort(w)[::-1][:top_k]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        w = w2 / w2.sum()
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-13:
            continue
        d = (n - 1) - i
        lo1, hi1 = max(0, -d), min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1+d:hi1+d])
        lo2, hi2 = max(0, d), min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2-d:hi2-d]
    return val, grad * dx

def pg_run(h0, betas, iters=250, tk=5, mom=0.88, tl=60.0):
    n = len(h0)
    tgt = n / 2.0
    h = project_wf(h0.astype(float), tgt)
    bv = compute_c5(h)
    bh = h.copy()
    v = np.zeros(n)
    t0 = time.time()
    for beta in betas:
        alpha = 2.5 / (beta + 1.0)
        for _ in range(iters):
            if time.time() - t0 > tl:
                return bh, bv
            val, g = sm_grad(h, beta, tk)
            v = mom * v - alpha * g
            h2 = project_wf(h + v, tgt)
            c2 = compute_c5(h2)
            if c2 < bv:
                bv = c2
                bh = h2.copy()
            if c2 <= val + 2e-12:
                h = h2
            else:
                found = False
                for f in [0.5, 0.25, 0.1, 0.05]:
                    h3 = project_wf(h + f * v, tgt)
                    if compute_c5(h3) <= val + 2e-12:
                        h = h3
                        found = True
                        break
                if not found:
                    h = project_wf(h + 0.02 * v, tgt)
    return bh, bv

def _nm_obj(x):
    n = len(x)
    h = project_wf(x, _g_tgt)
    return compute_c5(h)

def _coord_nudge(h, n, budget_end):
