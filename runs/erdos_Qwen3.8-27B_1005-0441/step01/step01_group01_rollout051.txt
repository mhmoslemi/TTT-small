```python
    #   contribution to T1 from this lag: w[k_idx] * (1 - h[i-k]) for i in [k, k+n-1] (clamped)

        wk = w[k_idx]
        # (1-h[i-k]) term: valid for i in [max(0,k), min(n, n+k)]
        i_start = max(0, k)
        i_end = min(n, n+k)
        if i_start < i_end:
            grad[i_start:i_end] += wk * (1.0 - h[i_start-k:i_end-k])
        # h[i+k] term: valid for i in [max(0,-k), min(n, n-k)]
        i_start2 = max(0, -k)
        i_end2 = min(n, n-k)
        if i_start2 < i_end2:
            grad[i_start2:i_end2] -= wk * h[i_start2+k:i_end2+k]
```

```python
def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()

        s = np.sum(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        order = np.argsort(h)
        mid = order[max(0, n//4):max(1, 3*n//4)]
        h[mid] += resid / len(mid)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_and_grad(x, alpha, pen):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = alpha * (corr - m)
    sh = np.clip(sh, -500.0, 500.0)
    e = np.exp(sh)
    se = np.sum(e)
    w = e / se
    obj_val = (m + np.log(se) / alpha) * dx
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wk = w[k_idx]
        i1 = max(0, k)
        i2 = min(n, n + k)
        if i1 < i2:
            grad[i1:i2] += wk * (1.0 - x[i1-k:i2-k])
        i3 = max(0, -k)
        i4 = min(n, n - k)
        if i3 < i4:
            grad[i3:i4] -= wk * x[i3+k:i4+k]
    grad *= dx
    
    if pen > 0:
        s = np.sum(x) - n * 0.5
        obj_val += pen * s * s
        grad += 2.0 * pen * s * np.ones(n)
    
    return obj_val, grad

def run_lbfgs(h0, alpha, pen, maxiter=500, ftol=1e-14):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return obj_and_grad(x, alpha, pen)[0]
    
    def g(x):
        return obj_and_grad(x, alpha, pen)[1]
    
    res = minimize(f, h0.copy(), jac=g, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol,
                                           'gtol': 1e-8, 'maxls': 50})
    return res.x

        b = n // 4
        h[:b] = 1.0
        h[-b:] = 1.0

        h = np.zeros(n)
        b = n // 6
        h[:b] = 1.0
        h[n//2-b:n//2+b] = 1.0
        h[-b:] = 1.0

    elif type == 'noise':
        h = np.random.default_rng(0).random(n)
    else:
```

```python
        i2 = min(n, n + k)
        if i1 < i2:
            grad_h[i1:i2] += wk * (1.0 - h[i1-k:i2-k])
        i3 = max(0, -k)
        i4 = min(n, n - k)
        if i3 < i4:
            grad_h[i3:i4] -= wk * h[i3+k:i4+k]
    grad_h *= dx
    
    # Chain rule: h[i] = z[i] for i < nh, h[i] = 1-z[i-nh] for i >= nh
    # dz[i] appears in h[i] (derivative 1) and h[i+nh] (derivative -1)
    grad_z = np.zeros(nh)
    grad_z += grad_h[:nh]  # from h[i] = z[i]
    grad_z -= grad_h[nh:]  # from h[i+nh] = 1-z[i], ∂h[i+nh]/∂z[i] = -1
    
    return obj_val, grad_z
```

This is elegant! The sum constraint ∫h = 1 is automatically satisfied because sum(h) = sum(z) + sum(1-z) = n/2.

And the constraint 0≤h≤1 becomes 0≤z≤1.

The optimization is over n/2 = 80 variables (for n=160) with simple box constraints. L-BFGS-B is perfect for this!

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
