    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 0

    n_list = [42, 44, 46, 48, 50, 52, 54, 56, 58, 60]
    if n0 > 0:
        n_list = [n0] + [x for x in n_list if x != n0]

    total_time = budget_s
    # Time allocation: generous to primary, less to secondary
    if n0 > 0:
        times = [total_time * 0.40]
    else:
        times = []
    remaining = total_time - sum(times) - 100  # reserve 100s for final refinement
    per_remaining = remaining / max(len(n_list) - (1 if n0 > 0 else 0), 1)
    for _ in range(len(n_list) - (1 if n0 > 0 else 0)):
        times.append(per_remaining)

    for idx, n in enumerate(n_list):
        if elapsed() > total_time - 100:
            break
        alloc = times[idx] if idx < len(times) else 60
        deadline = min(t0 + elapsed() + alloc, t0 + total_time - 80)
        if time.time() >= deadline:
            deadline = time.time() + max(alloc, 30)

        if n == n0 and h0 is not None:
            starts = [make_binary(h0, n)]
            alloc2 = deadline - time.time()
            h_opt, c5_opt = swap_search(starts[0].copy(), n, deadline, rng)
            if c5_opt < best_c5:
                best_c5, best_h, best_n = c5_opt, h_opt.copy(), n
            # Also try other starts for this n if time permits
            if time.time() < deadline - 30:
                for alt in [random_binary(n, rng), two_block_binary(n, rng)]:
                    if time.time() > deadline - 10:
                        break
                    h_opt2, c5_opt2 = swap_search(alt.copy(), n, deadline, rng)
                    if c5_opt2 < best_c5:
                        best_c5, best_h, best_n = c5_opt2, h_opt2.copy(), n
        else:
            starts = [random_binary(n, rng), block_binary(n, rng), two_block_binary(n, rng)]
            per_start = (deadline - time.time()) / max(len(starts), 1)
            for start in starts:
                if time.time() > deadline - 15:
                    break
                    sd = min(deadline, time.time() + per_start)
                    h_opt, c5_opt = swap_search(start.copy(), n, sd, rng)
                    if c5_opt < best_c5:
                        best_c5, best_h, best_n = c5_opt, h_opt.copy(), n

    # Final polish on best solution
    if best_h is not None and elapsed() < total_time - 20:
        fin_deadline = t0 + total_time - 5
        h_fin, c5_fin = swap_search(best_h.copy(), best_n, fin_deadline, rng, iters_per_perturb=25)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin.copy()

    # Validate
    if best_h is not None:
        dx = 2.0 / best_n
        s = np.sum(best_h)
        target = best_n / 2.0
        if abs(s - target) > 0.01:
            best_h = make_binary(best_h, best_n)
        best_c5 = get_c5(best_h, best_n)

    return best_h, float(best_c5), best_n
