    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode="full")
    m = corr.max()

    shifted -= shifted.max()
    w = np.exp(shifted) / np.exp(shifted).sum()
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]

        # grad contribution from (1-h)[j] part: -w_k * h[i]
        grad[j_indices] -= wk * h[i_indices]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        step = diff / n
        h = np.clip(h + step, 0.0, 1.0)
    # Exact fix by adjusting individual elements
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(-h)  # most room first
        else:
            idx = np.argsort(h)   # least value first
        for i in idx:
            if diff > 0:
                adj = min(1.0 - h[i], diff)
            else:
                adj = max(h[i] + diff, 0.0) - h[i]  # negative
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def soft_max_val(corr, beta):
    m = corr.max()
    s = beta * (corr - m)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / beta

def gradient_smooth(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = beta * (corr - m)
    s -= s.max()
    w = np.exp(s) / np.exp(s).sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k - 1)
        if lo > hi:
            continue
        i_arr = np.arange(lo, hi + 1)
        j_arr = i_arr - k + n - 1
        grad[i_arr] += wk * (1.0 - h[j_arr])
        grad[j_arr] -= wk * h[i_arr]
    return grad * dx

    h = h_init.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    
    penalty = 100.0
    for beta in [beta_start, beta_start*10, beta_start*100]:
        if time.time() > deadline:
            break

            corr = np.correlate(x, 1.0-x, mode="full")
            val = soft_max_val(corr, beta) * (2.0/n)
            s = x.sum() - n/2.0
            val += penalty * s * s
            return val
        
        def obj_grad(x):
            g = gradient_smooth(x, n, beta)
            s = x.sum() - n/2.0
            g += 2.0 * penalty * s * np.ones(n)
            return g

            c5 = compute_c5(h_new)
            if c5 < best_c5:

                best = h_new.copy()
            h = h_new

                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best = h_new.copy()
        except:
            pass
    
    return best, best_c5
