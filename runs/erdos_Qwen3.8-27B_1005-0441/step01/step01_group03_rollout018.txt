import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def grad_smooth(h, n, temp):
    ovs = get_overlaps(h, n)
    m = np.max(ovs)
    exps = np.exp((ovs - m) / temp)
    w = exps / np.sum(exps)
    dx = 2.0 / n
    g = np.zeros(n)
    for k_idx in range(len(w)):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        L = k_idx - (n - 1)
        if L >= 0:
            if L < n:
                g[L:] += wk * dx * (1.0 - h[:n - L])
                g[:n - L] -= wk * dx * h[L:]
        else:
            M = -L
            if M < n:
                g[:n - M] += wk * dx * (1.0 - h[M:])
                g[M:] -= wk * dx * h[:n - M]
    return g

def grad_exact_max(h, n):
    ovs = get_overlaps(h, n)
    k_max = int(np.argmax(ovs))
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:] += 1.0 - h[:n - lag]
            g[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            g[:n - m] += 1.0 - h[m:]
            g[m:] -= h[:n - m]
    return g, k_max

def optimize_smoothed(h, n, deadline, rng, t_start=0.25, t_end=0.005):
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    it = 0
    lr = 0.08
    stalled = 0
    total_steps = 8000
    while time.time() < deadline:
        it += 1
        frac = min(it / total_steps, 1.0)
        temp = t_start * (t_end / t_start) ** frac
        g = grad_smooth(h, n, temp)
        for _ in range(15):
            h_new = project_feasible(h - lr * g, n)
            c_new = get_c5(h_new, n)
            if c_new <= best_c5 + 0.0001:
                break
            lr *= 0.5
        else:
            h_new = project_feasible(h - lr * g, n)
            c_new = get_c5(h_new, n)
        if c_new < best_c5 - 1e-11:
            best_c5 = c_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        h = h_new
        if stalled > 200:
            h = best_h + rng.randn(n) * 0.08 * (stalled / 500.0 + 0.5)
            h = project_feasible(h, n)
            lr = 0.08
            stalled = 0
        if it % 1000 == 0 and stalled > 400:
            lr = 0.08
            h = best_h + rng.randn(n) * 0.15
            h = project_feasible(h, n)
            stalled = 0
    return best_h, best_c5

def polish_exact(h, n, deadline, rng):
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    dx = 2.0 / n
    lr = 0.25
    stalled = 0
    it = 0
    while time.time() < deadline:
        it += 1
        g, k_max = grad_exact_max(h, n)
        h_new = project_feasible(h - lr * g * dx, n)
        c_new = get_c5(h_new, n)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        h = h_new
        if stalled > 300:
            # Try top-5 lags
            ovs = get_overlaps(h, n)
            top = np.argsort(ovs)[-5:]
            improved = False
            for kk in top:
                lag2 = int(kk)
                g2 = np.zeros(n)
                L = lag2 - (n - 1)
                if L >= 0 and L < n:
                    g2[L:] += 1.0 - h[:n - L]
                    g2[:n - L] -= h[L:]
                elif L < 0:
                    M = -L
                    if M < n:
                        g2[:n - M] += 1.0 - h[M:]
                        g2[M:] -= h[:n - M]
                h2 = project_feasible(h - lr * 0.3 * g2 * dx, n)
                c2 = get_c5(h2, n)
                if c2 < best_c5 - 1e-12:
                    best_c5 = c2
                    best_h = h2.copy()
                    h = h2
                    improved = True
            if not improved:
                lr *= 0.6
                stalled = 0
                if lr < 0.003:
                    h = best_h + rng.randn(n) * 0.1
                    h = project_feasible(h, n)
                    lr = 0.25
        elif stalled > 100:
            lr *= 0.85
    return best_h, best_c5

def try_block_start(n, deadline, rng, offset_frac=0.0):
    h = np.zeros(n)
    half = n // 2
    shift = int(offset_frac * n)
    start = max(0, min(shift, n - half))
    h[start:start + half] = 1.0
    if n % 2 == 1:
        idx = start + half
        if idx < n:
            h[idx] = 0.5
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    # Phase 1: Optimize from initial construction (50% budget)
    if h0 is not None:

        h_start = project_feasible(h0.copy(), n0)
        c_start = get_c5(h_start, n0)
        best_h, best_c5, best_n = h_start, c_start, n0

        d1 = t0 + budget_s * 0.30

    elif best_h is None:
        n_default = 46
        h_start = try_block_start(n_default, t0, rng)
        best_h, best_c5, best_n = h_start, get_c5(h_start, n_default), n_default

    # Phase 2: Try other n values (35% budget)
    n_candidates = [40, 44, 48, 50, 38, 52, 42, 56, 46, 36]
    n_candidates = [x for x in n_candidates if x != best_n]
    remaining_frac = 0.35
    per_try = remaining_frac * budget_s / max(len(n_candidates), 1)

    for n_try in n_candidates:
        if time.time() > t0 + budget_s - 30:
            break
        d_try = min(time.time() + per_try, t0 + budget_s - 30)
        if d_try - time.time() < 5:
            continue
        h_struct = try_block_start(n_try, d_try, rng)
        bh, bc5 = optimize_smoothed(h_struct, n_try, d_try, rng)
        if bc5 < best_c5 - 1e-10:

        bh, bc5 = polish_exact(best_h, best_n, d_final, rng)

    # Final validation
    h_final = project_feasible(best_h, best_n)
    c5_final = get_c5(h_final, best_n)
    dx_f = 2.0 / best_n

    return h_final, float(c5_final), best_n
