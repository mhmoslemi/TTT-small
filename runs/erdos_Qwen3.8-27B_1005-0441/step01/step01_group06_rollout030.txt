    c_full = np.correlate(h, 1.0 - h, mode="full")

    w = np.ones(n_corr) / n_corr

        c_full = np.correlate(h, 1.0 - h, mode="full")

def make_shift_matrix(n):
    """Create the f matrix: f[i,k] = h[i+s_k]+h[i-s_k] (computed with h later)."""
    s_range = np.arange(-(n-1), n)
    i_idx = np.arange(n)[:, None]
    pos1 = i_idx + s_range[None, :]
    pos2 = i_idx - s_range[None, :]
    valid1 = (pos1 >= 0) & (pos1 < n)
    valid2 = (pos2 >= 0) & (pos2 < n)
    return pos1, pos2, valid1, valid2

def binary_local_search(h_bin, n, max_sweeps=5, t0=None, budget_s=None):
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    pos1, pos2, valid1, valid2 = make_shift_matrix(n)
    
    for sweep in range(max_sweeps):
        if t0 is not None and time.time() - t0 > budget_s * 0.85:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute f for all positions
        f = np.where(valid1, h[np.clip(pos1, 0, n-1)], 0.0) + \
            np.where(valid2, h[np.clip(pos2, 0, n-1)], 0.0)  # (n, 2n-1)
        
        c_full = np.correlate(h, 1.0 - h, mode="full")
        best_delta = 0.0
        best_a, best_b = -1, -1
        
        for ai in ones:
            base = c_full + f[ai, :]
            diffs = base[None, :] - f[zeros, :]
            new_maxes = diffs.max(axis=1)
            bi = np.argmin(new_maxes)
            if new_maxes[bi] < best_c5 / (2.0/n) - 1e-10 - best_delta:
                best_delta = best_c5 / (2.0/n) - new_maxes[bi]
                best_a, best_b = ai, zeros[bi]
        
        if best_a < 0:
            break
        
        h[best_a] = 0.0
        h[best_b] = 1.0
        best_c5 = compute_c5(h)
    
    return h, best_c5
