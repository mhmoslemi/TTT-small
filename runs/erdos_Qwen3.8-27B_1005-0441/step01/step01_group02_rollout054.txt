def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    """Clip to [0,1] and adjust sum to n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Add (target-s)/n to all, then clip
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    # Fine adjustment: find positions not at boundary and adjust
    s = h.sum()
    if abs(s - target) > 1e-10:
        diff = target - s
        # Distribute among interior points
        interior = np.where((h > 1e-6) & (h < 1 - 1e-6))[0]
        if len(interior) > 0:
            h[interior] += diff / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, beta):
    """Smooth approximation of max using log-sum-exp."""
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta

def local_search(h, n, max_iters=1000, t_limit=10.0):
    """Coordinate descent with adaptive step sizes."""
    h = project_to_feasible(h, n).copy()
    best_c = compute_c5(h)
    t0 = time.time()
    step = 0.05
    for it in range(max_iters):
        if time.time() - t0 > t_limit:
            break
        improved = False
        # Try each coordinate
        for i in range(n):
            for delta in [step, -step, 2*step, -2*step]:
                h_try = h.copy()
                h_try[i] += delta
                if h_try[i] < -0.5 or h_try[i] > 1.5:
                    continue
                h_try = project_to_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-14:
                    h = h_try
                    best_c = c_try
                    improved = True
                    break
        if not improved:
            step *= 0.8
            if step < 1e-4:
                break
        else:
            step = min(0.05, step * 1.1)
    return h, best_c

def stochastic_search(h, n, t_limit=30.0, seed=42):
    """Simulated annealing-style stochastic search."""

    best_h = h.copy()
    temp = 0.01

        idx = rng.choice(n, n_pert, replace=False)

            h = h_try
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()

    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx


def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-13:
            continue
        d = i - (n - 1)
        jl = max(0, -d)
        jh = min(n - 1, n - 1 - d)
        for j in range(jl, jh + 1):
            jd = j + d
            grad[jd] += wi * (1.0 - h[j])
            grad[j] -= wi * h[jd]
    return grad * dx


def lbfgsb_run(h0, n, beta, maxiter=1500, pen=5000.0):
    target = n / 2.0

    def objf(x):
        h = np.clip(x, 0, 1)
        return lse_obj(h, beta) + pen * (h.sum() - target) ** 2

    def gradf(x):
        h = np.clip(x, 0, 1)
        return lse_grad(h, beta) + 2.0 * pen * (h.sum() - target) / n

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objf, h0, jac=gradf, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project(res.x, n)
    except Exception:
        return project(h0, n)


def sa_search(h0, n, t_limit, seed=42):
    rng = np.random.RandomState(seed)
    h = project(h0.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    temp = 0.005
    t0 = time.time()
    while time.time() - t0 < t_limit and temp > 1e-5:
        npert = rng.randint(2, 9)
        idx = rng.choice(n, npert, replace=False)
        h_try = h.copy()
        h_try[idx] += rng.uniform(-0.35, 0.35, npert)
        h_try = project(h_try, n)
        c_try = compute_c5(h_try)
        d = c_try - cur_c
        if d < 0 or rng.random() < np.exp(-d / temp):
            h = h_try
            cur_c = c_try
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
        temp *= 0.9998
    return best_h, best_c


def coord_descent(h0, n, t_limit):
    h = project(h0.copy(), n)
    best_c = compute_c5(h)
    t0 = time.time()
    step = 0.03
    stag = 0
    while time.time() - t0 < t_limit:
        improved = False
        for i in range(n):
            if time.time() - t0 > t_limit:
                break
            for d in (step, -step):
                h_try = h.copy()
                h_try[i] += d
                if h_try[i] < -0.5 or h_try[i] > 1.5:

                h_try = project(h_try, n)

                    best_c = c_try

            step *= 0.6
            if step < 1e-5:

        else:
            stag = 0

def shift_explore(h0, n):

        if c_s < best_c - 1e-12:
