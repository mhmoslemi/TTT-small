import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def soft_objective(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def cfun(x):
    return np.sum(x) - len(x) / 2.0

def cjac(x):
    return np.ones(len(x))

def slsqp_polish(h, n, t0, budget_s, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    best = h.copy()
    best_c5 = compute_c5(h)
    betas = [50, 200, 1000, 5000]
    for beta in betas:
        if time.time() - t0 > budget_s * 0.92:
            break
        try:
            mi = maxiter if n <= 200 else max(50, maxiter // 2)
            res = minimize(soft_objective, best, args=(n, beta), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': mi, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best = h_new
                    best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def coordinate_descent(h, n, n_passes=20, time_limit=120, seed=0):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_c5 = compute_c5(h)
    for p in range(n_passes):
        if time.time() - t0 > time_limit:
            break
        order = rng.permutation(n)
        for m in order:
            hm = h[m]
            base = np.correlate(h, 1.0 - h, mode="full")
            s_vals = np.arange(-(n - 1), n)
            c = np.zeros(2 * n - 1)
            idx1 = m - s_vals
            v1 = (idx1 >= 0) & (idx1 < n)
            c[v1] += 1.0 - h[idx1[v1]]
            idx2 = m + s_vals
            v2 = (idx2 >= 0) & (idx2 < n)
            c[v2] -= h[idx2[v2]]
            n_g = 50
            u_lo, u_hi = -hm, 1.0 - hm
            u_vals = np.linspace(u_lo, u_hi, n_g)
            vals = base[np.newaxis, :] + u_vals[:, np.newaxis] * c[np.newaxis, :]
            vals[n_g // 2, n - 1] -= 0.0  # skip quadratic correction
            mx = vals.max(axis=1)
            bi = int(np.argmin(mx))
            new_hm = hm + u_vals[bi]
            lo = max(0, bi - 1)
            hi = min(n_g - 1, bi + 1)
            u_fine = np.linspace(u_lo + (u_hi - u_lo) * lo / (n_g - 1),
                                 u_lo + (u_hi - u_lo) * hi / (n_g - 1), 30)
            if len(u_fine) > 1 and u_fine[-1] > u_fine[0]:
                vals_f = base[np.newaxis, :] + (u_fine - hm)[:, np.newaxis] * c[np.newaxis, :]
                mx_f = vals_f.max(axis=1)
                new_hm = hm + u_fine[np.argmin(mx_f)]
            new_hm = float(np.clip(new_hm, 0.0, 1.0))
            if abs(new_hm - hm) > 1e-10:
                h[m] = new_hm
        h = project_feasible(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
    return h, best_c5

def binary_round(h, n):
    h_b = (h > 0.5).astype(float)
    need = int(n / 2) - int(np.sum(h_b))
    if need > 0:
        fracs = np.where(h_b < 1.0)[0]
        order = np.argsort(-(h[fracs]))[:need]
        h_b[fracs[order]] = 1.0
    elif need < 0:
        fracs = np.where(h_b > 0.0)[0]
        order = np.argsort(h[fracs])[:(-need)]
        h_b[fracs[order]] = 0.0
    return project_feasible(h_b, n)
