import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def greedy_swap_descent(h, max_rounds=500):
    n = len(h)
    best_c = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_swap = None
        best_new_c = best_c - 1e-14
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if c < best_new_c:
                    best_new_c = c
                    best_swap = (i, j)
        if best_swap is None:
            break
        h[best_swap[0]] = 0.0
        h[best_swap[1]] = 1.0
        best_c = best_new_c
    return h, best_c

def simulated_annealing(h, t0, t_end, n_iter):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    for it in range(n_iter):
        T = t0 * (t_end / t0) ** (it / n_iter)
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1.0
            h[j] = 0.0
            ones.remove(j)
            zeros.remove(i)
            ones.append(i)
            zeros.append(j)
    return best_h, best_c

def gradient_refine(h, t_budget=30):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.01, 0.99).copy()
    # Adjust sum
    s = h.sum()
    if abs(s - target) > 1e-10:
        h = h + (target - s) / n
        h = np.clip(h, 0.01, 0.99)
    
    t0 = time.time()
    best = compute_c5(h)
    hb = h.copy()
    
    beta_vals = [30, 60, 120, 240]
    for beta in beta_vals:
        if time.time() - t0 > t_budget - 5:
            break
        def obj(x):
            corr = compute_corr(x)
            mc = np.max(corr)
            return (mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * (2.0/n)
        
        def obj_grad(x):
            corr = compute_corr(x)
            mc = np.max(corr)
            w = np.exp(beta * (corr - mc))
            ws = w.sum()
            w /= ws
            grad = np.zeros(n)
            # grad[i] = sum_k w[k] * d/dx[i] corr[k]
            # corr[k] = sum_j x[j] * (1-x[j+k-n+1]) (for appropriate ranges)
            # This is complex; use numerical gradient for simplicity with small n
            eps = 1e-5
            for i in range(n):
                x[i] += eps
                cp = compute_corr(x)
                x[i] -= 2*eps
                cm = compute_corr(x)
                x[i] += eps
                grad[i] = (np.sum(w * (cp - cm)) / (2*eps)) * (2.0/n)
            return grad
        
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-10})
            h_new = np.clip(res.x, 0, 1)
            # Fix sum
            s = h_new.sum()
            h_new += (target - s) / n
            h_new = np.clip(h_new, 0, 1)
            c = compute_c5(h_new)
            if c < best:
                best = c
                hb = h_new
                h = h_new
        except:
            pass
    return hb, best

        for _ in range(n_flips):

def smooth_value_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = beta * (corr - mc)
    w = np.exp(shifted)
    ws = w.sum()
    value = (mc + np.log(ws) / beta) * dx
    w /= ws
    
    grad = np.zeros(n)
    nk = len(corr)  # 2n-1
    for k in range(nk):
        if w[k] < 1e-15:
            continue
        shift = k - n + 1
        # ∂c[k]/∂h[i] = (1-h[i+shift]) * [0≤i+shift<n] - h[i-shift] * [0≤i-shift<n]
        # First term: i ranges 0 to n-1-shift (if shift>=0) or 0 to n-1 (if shift<0, i+shift>=0 means i>=-shift)
        # Second term: i-shift ranges 0 to n-1, so i ranges shift to n-1+shift (if shift>=0) or 0 to n-1-shift (if shift<0)
        
        # First term: (1-h[i+shift]) for valid i
        if shift >= 0:
            # i+shift in [0, n-1] => i in [0, n-1-shift]
            grad[:n-shift] += w[k] * (1.0 - h[shift:])
        else:
            # i+shift in [0, n-1] => i in [-shift, n-1]
            grad[-shift:] += w[k] * (1.0 - h[:n+shift])
        
        # Second term: -h[i-shift] for valid i
        if shift >= 0:
            # i-shift in [0, n-1] => i in [shift, n-1]
            grad[shift:] -= w[k] * h[:n-shift]
        else:
            # i-shift in [0, n-1] => i in [0, n-1+shift]
            grad[:n+shift] -= w[k] * h[-shift:]
    
    grad *= dx
    return value, grad

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def to_binary(h):

    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def smooth_vg(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    s = beta * (corr - mc)
    w = np.exp(s)
    ws = w.sum()
    val = (mc + np.log(ws) / beta) * dx
    w /= ws
    grad = np.zeros(n)

        wk = w[k]
        if wk < 1e-15:

def grad_refine(h, t_budget=60):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    t0 = time.time()
    best = compute_c5(h)
    hb = h.copy()
    for beta in [20, 50, 100, 200, 400]:
        if time.time() - t0 > t_budget - 5:
            break
        def obj(x):
            return smooth_vg(x, beta)[0]
        def objg(x):
            return smooth_vg(x, beta)[1]
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:

                          bounds=bounds, constraints=cons,

            h_new = np.clip(h_new, 0, 1)
            c = compute_c5(h_new)
            if c < best:
                best = c
                hb = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return hb, best
