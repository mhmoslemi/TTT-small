def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def to_binary(h, n):
    # Project to binary: pick top n/2 values
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def binary_c5(h, n):
    # Fast computation for binary h
    dx = 2.0 / n
    ones = np.where(h == 1.0)[0]
    n_ones = len(ones)
    # For each positive lag k, count pairs (i, i+k) where h[i]=1, h[i+k]=0

    for k in range(1, n):
        overlap_pos = np.sum(arr[:n-k] * (1 - arr[k:]))
        overlap_neg = np.sum(arr[k:] * (1 - arr[:n-k]))
        ov = max(overlap_pos, overlap_neg) * dx
        if ov > max_ov:

    return max_ov

def swap_local_search(h, n, deadline, max_iter=100000):
    dx = 2.0 / n
    h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    best_c5 = get_c5(h, n)
    
    improved = True
    it = 0
    while improved and time.time() < deadline and it < max_iter:
        improved = False
        best_swap = None
        best_new_c5 = best_c5
        
        arr = (h == 1.0).astype(int)
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # Swap h[i] and h[j]
                h[i], h[j] = 0.0, 1.0
                arr[i], arr[j] = 0, 1
                
                # Compute new max overlap

                for k in range(1, n):
                    ov = max(np.sum(arr[:n-k]*(1-arr[k:])), np.sum(arr[k:]*(1-arr[:n-k]))) * dx
                    if ov > max_ov:
                        max_ov = ov
                
                if max_ov < best_new_c5 - 1e-12:
                    best_new_c5 = max_ov
                    best_swap = (i, j)
                
                # Undo swap

            # Update ones and zeros

            zeros = np.where(h == 0.0)[0]

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def bin_to_b(h, n):
    t = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n, dtype=np.uint8)
    b[idx[:t]] = 1
    return b

def bmax_ov(a, n, thresh=None):
    best = 0
    for k in range(1, n):
        m = n - k
        ca = np.count_nonzero(a[:m])
        cb = np.count_nonzero(a[k:])
        cc = np.count_nonzero(a[:m] & a[k:])
        ov = ca - cc
        ov2 = cb - cc
        if ov2 > ov:
            ov = ov2
        if ov > best:
            best = ov
            if thresh is not None and best > thresh:
                return best
    return best

def local_search(a, n, t_end, rng, pe=20, mp=60):
    dx = 2.0 / n
    a = a.copy()
    assert int(np.sum(a)) == n // 2
    i1 = np.where(a == 1)[0]
    i0 = np.where(a == 0)[0]
    bo = bmax_ov(a, n)
    bc = bo * dx
    gi = 0
    pc = 0
    while time.time() < t_end and pc < mp:
        bs = None
        bno = bo
        for ii in i1:
            for jj in i0:
                a[ii] = 0; a[jj] = 1
                no = bmax_ov(a, n, bno - 1)
                a[ii] = 1; a[jj] = 0
                if no < bno:
                    bno = no
                    bs = (ii, jj)
        if bs is not None:
            a[bs[0]], a[bs[1]] = 0, 1
            bo = bno
            bc = bo * dx
            gi = 0
            i1 = np.where(a == 1)[0]
            i0 = np.where(a == 0)[0]
        else:
            gi += 1
            if gi >= pe:
                ns = rng.randint(1, 4)
                for _ in range(ns):
                    ii = i1[rng.randint(len(i1))]
                    jj = i0[rng.randint(len(i0))]
                    a[ii], a[jj] = 0, 1
                bo = bmax_ov(a, n)
                bc = bo * dx
                i1 = np.where(a == 1)[0]
                i0 = np.where(a == 0)[0]
                gi = 0
                pc += 1
    return a, bc

def anneal(a, n, t_end, rng):
    dx = 2.0 / n
    a = a.copy()
    i1 = np.where(a == 1)[0]
    i0 = np.where(a == 0)[0]
    co = bmax_ov(a, n)
    cc = co * dx
    ba = a.copy()
    bc = cc
    T = 0.15
    while time.time() < t_end:
        ii = i1[rng.randint(len(i1))]
        jj = i0[rng.randint(len(i0))]
        a[ii], a[jj] = 0, 1
        no = bmax_ov(a, n)
        d = (no - co) * dx
        if d < 0 or rng.random() < np.exp(-d / T):
            co = no; cc = co * dx
            if cc < bc:
                bc = cc
                ba = a.copy()
            i1 = np.where(a == 1)[0]
            i0 = np.where(a == 0)[0]
        else:
            a[ii], a[jj] = 1, 0
        T *= 0.9999
        if T < 0.005:
            T = 0.005
    return ba, bc

    # Phase 3: Other n values (25% time)

```python
def bmax_ov(a, n, thresh):
    """Compute max overlap count for binary uint8 array a of length n.
    If thresh is not None, return early (with value > thresh) if exceeded.
    Returns the max overlap count (integer)."""
    best = 0
    for k in range(1, n):
        m = n - k
        # ov_k+ = #{i: 0<=i<m, a[i]=1, a[i+k]=0}
        #        = (# of 1s in a[:m]) - (# of 1s in both a[:m] and a[k:])
        # ov_k- = #{i: 0<=i<m, a[i+k]=1, a[i]=0}
        #        = (# of 1s in a[k:]) - (# of 1s in both)
        ca = np.count_nonzero(a[:m])
        cb = np.count_nonzero(a[k:])
        cc = np.count_nonzero(a[:m] & a[k:])
        ov = ca - cc
        if cb - cc > ov:
            ov = cb - cc
        if ov > best:
            best = ov
            if thresh is not None and best > thresh:
                return best
    return best
```

- `a[:m] & a[k:]` gives 1 where both are 1, 0 otherwise. `np.count_nonzero` counts the 1s.
- `ca - cc` = number of 1s in a[:m] that are NOT 1s in a[k:] = #{i: a[i]=1, a[i+k]=0}. Correct.
- `cb - cc` = #{i: a[i+k]=1, a[i]=0}. Correct.
