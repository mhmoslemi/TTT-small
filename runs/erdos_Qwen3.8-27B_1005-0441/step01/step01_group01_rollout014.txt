import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def subgradient_step(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    s = k_max - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        grad[0:n-s] += (1.0 - h[s:n]) * dx
        grad[s:n] += (-h[0:n-s]) * dx
    else:
        ss = -s
        grad[ss:n] += (1.0 - h[0:n-ss]) * dx
        grad[0:n-ss] += (-h[ss:n]) * dx
    return -grad, s, corr[k_max] * dx

def obj_smooth(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400.0, 400.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    return sm * dx

def _sum_eq(x, target):
    return np.sum(x) - target

def _sum_eq_jac(x, target):
    return np.ones(len(x))

def make_two_block(n):
    h = np.zeros(n)
    block = max(n // 4, 2)
    h[:block] = 1.0
    h[-block:] = 1.0
    return project_feasible(h, n)

def make_cosine(n):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.3 * np.cos(np.pi * x)
    return project_feasible(h, n)

def make_three_block(n):
    h = np.zeros(n)
    b = max(n // 6, 2)
    h[:b] = 1.0
    h[n//2 - b:n//2 + b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def subgradient_anneal(h, n, t_start, t_end, base_lr, smoothing_interval):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    it = 0
    while time.time() < t_end:
        neg_grad, s, c5_val = subgradient_step(h, n)
        lr = base_lr * (0.5 + 0.5 * max(0, (t_end - time.time()) / (t_end - t_start)))
        h = h + lr * neg_grad
        h = project_feasible(h, n)
        it += 1
        if it % smoothing_interval == 0:
            sigma = 2.0 + 3.0 * max(0, (t_end - time.time()) / (t_end - t_start))
            h = gaussian_filter1d(h, sigma, mode='wrap')
            h = project_feasible(h, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-7:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                h = best_h + 0.03 * np.random.default_rng(it).standard_normal(n)
                h = project_feasible(h, n)
                no_improve = 0
    return best_h, best_c5

def slsqp_polish(h, n, alpha, maxiter, t_limit):
    bounds = [(0.0, 1.0)] * n
    target = n * 0.5
    constraint = [{'type': 'eq', 'fun': lambda x: _sum_eq(x, target),
                   'jac': lambda x: _sum_eq_jac(x, target)}]
    res = minimize(obj_smooth, h.copy(), args=(alpha,),
                   method='SLSQP', bounds=bounds, constraints=constraint,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if time.time() > t_limit:
        return project_feasible(h, n), compute_c5(h)
    hc = project_feasible(res.x, n)
    c5c = compute_c5(hc)
    return hc, c5c

def upscale(h_old, n_new):
    x_old = np.arange(len(h_old), dtype=float)
    x_new = np.linspace(0.0, float(len(h_old) - 1), n_new)
    h_up = np.interp(x_new, x_old, h_old)
    return project_feasible(h_up, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_end = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = make_two_block(160)
    n = len(h0)
    h_best = project_feasible(h0.copy(), n)
    c5_best = compute_c5(h_best)

    # Phase 1: Subgradient annealing on initial grid
    t_phase1_end = t0 + (budget_s * 0.35)
    h_best, c5_best = subgradient_anneal(h_best, n, t0, t_phase1_end, base_lr=0.08, smoothing_interval=50)

    # Phase 2: SLSQP polish
    if time.time() < t_end - 60:
        for alpha in [100.0, 300.0]:
            if time.time() > t_end - 60:
                break
            hc, c5c = slsqp_polish(h_best, n, alpha, maxiter=600, t_limit=t_end - 60)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc

    # Phase 3: Try multiple structural starts and pick best
    starts = [h_best.copy(), make_two_block(n), make_cosine(n), make_three_block(n)]
    for sp in starts:
        if time.time() > t_end - 120:
            break
        t_sub = time.time()
        t_sub_end = min(t0 + (t_sub - t0) + 60, t_end - 120)
        hc, c5c = subgradient_anneal(sp, n, t_sub, t_sub_end, base_lr=0.06, smoothing_interval=40)
        if time.time() < t_end - 90:
            hc, c5c = slsqp_polish(hc, n, 200.0, maxiter=400, t_limit=t_end - 90)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc

    # Phase 4: Upscale to finer grid
    if time.time() < t_end - 80:
        n2 = min(n * 2, 480)
        if n2 > n:
            h_up = upscale(h_best, n2)
            n = n2
            t_sub = time.time()
            t_sub_end = min(t_sub + 120, t_end - 60)
            h_best, c5_best = subgradient_anneal(h_up, n, t_sub, t_sub_end, base_lr=0.05, smoothing_interval=30)
            if c5_best > compute_c5(project_feasible(h_up, n)):
                h_best = project_feasible(h_up, n)
                c5_best = compute_c5(h_best)
            if time.time() < t_end - 40:
                for alpha in [150.0, 400.0]:
                    if time.time() > t_end - 40:
                        break
                    hc, c5c = slsqp_polish(h_best, n, alpha, maxiter=500, t_limit=t_end - 40)
                    if c5c < c5_best:
                        c5_best = c5c
                        h_best = hc

    # Phase 5: Final subgradient refinement with small steps
    if time.time() < t_end - 20:
        t_sub = time.time()
        t_sub_end = t_end - 20
        h_best, c5_best = subgradient_anneal(h_best, n, t_sub, t_sub_end, base_lr=0.03, smoothing_interval=80)

    # Phase 6: Final SLSQP tight polish
    if time.time() < t_end - 10:
        hc, c5c = slsqp_polish(h_best, n, 500.0, maxiter=300, t_limit=t_end - 10)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
