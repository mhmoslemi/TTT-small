import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def grad_lag(h, L, n):
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
    else:
        M = -L
        if M < n:
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    return g

def optimize(h, n, deadline, rng, top_k=8, lr0=0.35, tau0=8.0, tau1=50.0):
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    lr = lr0
    momentum = np.zeros(n)
    beta = 0.85
    stalled = 0
    t_start = time.time()
    t_span = max(1.0, deadline - t_start)
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / t_span)
        tau = tau0 + (tau1 - tau0) * frac
        ov = get_overlaps(h, n)
        ov[n - 1] = -999.0
        idx = np.argsort(ov)[::-1][:top_k]
        ov_top = ov[idx]
        m = np.max(ov_top)
        exps = np.exp(tau * (ov_top - m))
        w = exps / np.sum(exps)
        g = np.zeros(n)
        for ii in range(top_k):
            lag = int(idx[ii]) - (n - 1)
            g += w[ii] * grad_lag(h, lag, n)
        momentum = beta * momentum - lr * g
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        c_new = get_c5(h_new, n)
        if c_new < best_c5 - 1e-14:
            best_c5 = c_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        h = h_new
        if stalled > 50:
            lr *= 0.7
            momentum *= 0.3
            stalled = 0
            if lr < 0.003:
                h = best_h + rng.randn(n) * 0.07
                h = project_feasible(h, n)
                lr = lr0
                momentum = np.zeros(n)
        elif stalled > 15:
            lr *= 0.93
    return best_h, best_c5

def local_search(h, n, rng, deadline):
    best_h = h.copy()
    best_c5 = get_c5(best_h, n)
    deltas = np.array([-0.2, -0.1, -0.05, 0.05, 0.1, 0.2])
    while time.time() < deadline:
        for _ in range(50):
            if time.time() >= deadline:
                break
            j = rng.randint(n)
            d = deltas[rng.randint(len(deltas))]
            h_t = best_h.copy()
            h_t[j] = np.clip(h_t[j] + d, 0.0, 1.0)
            h_t = project_feasible(h_t, n)
            c = get_c5(h_t, n)
            if c < best_c5 - 1e-14:
                best_c5 = c
                best_h = h_t
    return best_h, best_c5
