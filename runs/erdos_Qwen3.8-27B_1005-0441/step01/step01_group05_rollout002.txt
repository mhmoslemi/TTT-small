def pgd_step(h, dx, step_size, active_eps=0.01):
    n = len(h)
    o_h = 1.0 - h
    corr = np.correlate(h, o_h, mode="full")
    m = np.max(corr)
    # Active set: indices where corr[k] > m - active_eps
    active = corr > (m - active_eps)
    n_active = np.sum(active)
    
    # Compute gradient as average over active set
    # ∂corr[k]/∂h[j] = (1-h[j+shift_k]) - h[j-shift_k]
    # where shift_k = k - (n-1)
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)  # length 2n-1
    
    for k in range(2*n-1):
        if not active[k]:
            continue
        s = shifts[k]
        # (1-h[j+s]) for valid j+s
        valid1 = (np.arange(n) + s >= 0) & (np.arange(n) + s < n)
        # h[j-s] for valid j-s
        valid2 = (np.arange(n) - s >= 0) & (np.arange(n) - s < n)
        
        g1 = np.zeros(n)
        g1[valid1] = 1.0 - h[np.arange(n)[valid1] + s]
        
        g2 = np.zeros(n)
        g2[valid2] = h[np.arange(n)[valid2] - s]
        
        grad += (g1 - g2)
    
    grad /= (n_active * dx)  # normalize
    
    # Projected gradient step
    h_new = h - step_size * grad
    # Project to [0,1]
    h_new = np.clip(h_new, 0.0, 1.0)
    # Enforce sum constraint
    s = np.sum(h_new)
    target = n / 2.0
    h_new = h_new + (target - s) / n
    h_new = np.clip(h_new, 0.0, 1.0)
    
    return h_new, np.max(corr) * dx

            shifts = np.arange(2*n-1) - (n-1)
            c_plus = np.zeros(2*n-1)
            c_minus = np.zeros(2*n-1)

            valid_p = (i + shifts >= 0) & (i + shifts < n)
            c_plus[valid_p] = 1.0 - h[i + shifts[valid_p]]

from scipy.optimize import minimize
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def coord_descent_pass(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    for i in range(n):
        old = h[i]
        
        # Compute slopes for all k
        # slope[k] = 1 - h[i+shift_k] (if valid) - h[i-shift_k] (if valid)
        ip = i + shifts  # i + shift for each k
        im = i - shifts  # i - shift for each k
        
        valid_p = (ip >= 0) & (ip < n)
        valid_m = (im >= 0) & (im < n)
        
        slopes = np.zeros(2 * n - 1)
        slopes[valid_p] += 1.0
        slopes[valid_p] -= h[ip[valid_p]]
        slopes[valid_m] -= h[im[valid_m]]
        
        # Ternary search for optimal t in [0,1]
        lo, hi = 0.0, 1.0
        for _ in range(50):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            v1 = np.max(corr + (m1 - old) * slopes)
            v2 = np.max(corr + (m2 - old) * slopes)
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        t_star = (lo + hi) / 2.0
        
        # Update
        h[i] = t_star
        corr += (t_star - old) * slopes
    
    return h

def coord_descent(h, n_sweeps=100, time_limit=60):
    n = len(h)
    t0 = time.time()
    best_c5 = compute_c5(h)
    h_best = h.copy()
    
    stale = 0
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        h = coord_descent_pass(h)
        
        # Project to sum constraint
        target = n / 2.0
        err = target - np.sum(h)
        if abs(err) > 1e-10:
            h += err / n
            h = np.clip(h, 0.0, 1.0)
            # Re-project if needed (clip might change sum)
            for _ in range(10):
                err2 = target - np.sum(h)
                if abs(err2) < 1e-12:
                    break
                h += err2 / n
                h = np.clip(h, 0.0, 1.0)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            h_best = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 5:
                break
    
    return h_best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def coord_descent_pass(h, order=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    if order is None:
        order = range(n)
    
    for i in order:
        old = h[i]
        ip = i + shifts
        im = i - shifts
        valid_p = (ip >= 0) & (ip < n)
        valid_m = (im >= 0) & (im < n)
        
        slopes = np.zeros(2 * n - 1)
        slopes[valid_p] = 1.0 - h[ip[valid_p]]
        slopes[valid_m] -= h[im[valid_m]]
        
        lo, hi = 0.0, 1.0
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            v1 = np.max(corr + (m1 - old) * slopes)
            v2 = np.max(corr + (m2 - old) * slopes)
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        t_star = (lo + hi) / 2.0
        
        h[i] = t_star
        corr += (t_star - old) * slopes
    
    return h

def project_sum(h):
    n = len(h)
    target = n / 2.0
    for _ in range(50):
        err = target - np.sum(h)
        if abs(err) < 1e-14:
            break
        h = h + err / n
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n_sweeps=200, time_limit=120, rnd_order=False, seed=0):
    n = len(h)
    t0 = time.time()
    best_c5 = compute_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(seed)
    
    stale = 0
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        if rnd_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        h = coord_descent_pass(h, order)
        h = project_sum(h)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            h_best = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 10:
                break
    
    return h_best, best_c5

def slsqp_polish(h, time_limit=30):
    n = len(h)
    dx = 2.0 / n
    t0 = time.time()
    
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        m = np.max(corr)
        t = max(0.001, 1e-4)
        val = (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx
        # Penalty for sum constraint
        s = np.sum(hv) - n / 2.0
        val += 1e4 * s * s
        return val
    
    def obj_grad(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        m = np.max(corr)
        t = max(0.001, 1e-4)
        weights = np.exp((corr - m) / t)
        weights /= np.sum(weights)
        shifts = np.arange(2 * n - 1) - (n - 1)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-10:
                continue
            s = shifts[k]
            ip = np.arange(n) + s
            im = np.arange(n) - s
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            grad[vp] += w * (1.0 - hv[ip[vp]])
            grad[vm] -= w * hv[im[vm]]
        grad *= dx
        # Add penalty gradient
        grad += 2 * 1e4 * (np.sum(hv) - n / 2.0) / n
        return grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 200, 'gtol': 1e-10})
        h_new = project_sum(np.clip(res.x, 0.0, 1.0))
        if time.time() - t0 < time_limit:
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h):
                return h_new
    except:
        pass
    return h

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old  # cell centers
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = project_sum(h_new)
    return h_new

def make_binary_init(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.permutation(n)[:n // 2]
    h[idx] = 1.0

    h_p = h + scale * rng.randn(len(h))
    return project_sum(np.clip(h_p, 0.0, 1.0))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h):
    n = len(h)
    target = float(n) / 2.0
    h = h.copy()
    for _ in range(100):
        err = target - np.sum(h)
        if abs(err) < 1e-14:
            break
        h += err / n
        h = np.clip(h, 0.0, 1.0)
    return h

def cd_pass(h, order):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifts = np.arange(2 * n - 1) - (n - 1)
    for i in order:
        old = h[i]
        ip = i + shifts
        im = i - shifts
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        slopes = np.zeros(2 * n - 1)
        slopes[vp] = 1.0 - h[ip[vp]]
        slopes[vm] -= h[im[vm]]
        lo, hi = 0.0, 1.0
        for _ in range(45):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            v1 = np.max(corr + (m1 - old) * slopes)
            v2 = np.max(corr + (m2 - old) * slopes)
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        h[i] = (lo + hi) / 2.0
        corr += (h[i] - old) * slopes
    return h

def coord_descent(h, n_sweeps=500, time_limit=120, seed=0):
    n = len(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    h_best = h.copy()
    stale = 0
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        if sweep % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)
        h = cd_pass(h, order)
        h = project_sum(h)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            h_best = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 15:
                break
    return h_best, best_c5

def lbfgs_polish(h, time_limit=30):
    n = len(h)
    dx = 2.0 / n
    t0 = time.time()
    shifts = np.arange(2 * n - 1) - (n - 1)

    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        m = np.max(corr)
        t = 0.005
        val = (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx
        s = np.sum(hv) - n / 2.0
        val += 5e3 * s * s
        return val

    def obj_grad(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        m = np.max(corr)
        t = 0.005
        w = np.exp((corr - m) / t)
        w /= np.sum(w)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            s = shifts[k]
            idx = np.arange(n)
            ip = idx + s
            im = idx - s
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            grad[vp] += w[k] * (1.0 - hv[ip[vp]])
            grad[vm] -= w[k] * hv[im[vm]]
        grad *= dx
        grad += 2.0 * 5e3 * (np.sum(hv) - n / 2.0) / n
        return grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 300, 'gtol': 1e-12})
        h_new = project_sum(np.clip(res.x, 0.0, 1.0))
        if compute_c5(h_new) < compute_c5(h) and time.time() - t0 < time_limit:
            return h_new
    except Exception:
        pass
    return h

def resample(h, n_new):
    n = len(h)
    x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_sum(np.clip(h_new, 0.0, 1.0))

def make_perturb(h, scale, seed=0):
    rng = np.random.RandomState(seed)
    return project_sum(np.clip(h + scale * rng.randn(len(h)), 0.0, 1.0))

def make_structured_init(n, kind, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    if kind == 0:
        h = np.full(n, 0.5)
    elif kind == 1:
        h = 0.5 + 0.5 * np.cos(np.pi * x)
    elif kind == 2:
        h = np.zeros(n)
        half = n // 4
        gap = n // 10
        p1 = rng.randint(0, n - 2 * half - gap)
        h[p1:p1 + half] = 1.0
        p2 = (p1 + half + gap) % n
        for j in range(half):
            h[(p2 + j) % n] = 1.0
    elif kind == 3:
        h = np.zeros(n)
        idx = rng.permutation(n)[:n // 2]
        h[idx] = 1.0
    else:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    return project_sum(np.clip(h, 0.0, 1.0))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(160, 0.5)
    best_h = project_sum(best_h)
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)

    # Phase 1: CD on initial
    if rem() > 50:
        h_cd, c5_cd = coord_descent(best_h.copy(), 500, min(50, rem() - 10), seed)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 2: Try structured inits at n0
    for kind in range(5):
        if rem() < 40:
            break
        h_init = make_structured_init(n0, kind, seed + kind * 100)
        h_cd, c5_cd = coord_descent(h_init, 300, min(30, rem() - 10), seed + 500 + kind)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 3: Try different n
    for tn in [80, 100, 120, 200, 256]:
        if rem() < 50 or tn == n0:
            continue
        h_init = resample(best_h, tn)
        h_cd, c5_cd = coord_descent(h_init, 400, min(40, rem() - 15), seed + tn)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 4: Basin hopping
    n_cur = len(best_h)
    rng = np.random.RandomState(seed + 12345)
    for r in range(60):
        if rem() < 25:
            break
        scale = 0.03 + 0.08 * (r % 7) / 6.0
        h_pert = make_perturb(best_h, scale, seed + 5000 + r)
        mt = min(18, rem() - 8)
        if mt < 8:
            break
        h_cd, c5_cd = coord_descent(h_pert, 150, mt, seed + 6000 + r)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 5: L-BFGS polish + CD
    if rem() > 25:
        h_pol = lbfgs_polish(best_h.copy(), min(30, rem() - 10))
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
        if rem() > 15:
            h_cd, c5_cd = coord_descent(best_h.copy(), 200, min(15, rem() - 5), seed + 99999)
            if c5_cd < best_c5:
                best_h, best_c5 = h_cd, c5_cd

    # Final feasibility
    best_h = project_sum(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
