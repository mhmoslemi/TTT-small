def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    h = (hf >= 0.5).astype(float)
    d = tgt - int(h.sum())
    if d > 0:
        for i in np.argsort(-hf)[::1]:
            if d <= 0: break
            if h[i] == 0: h[i] = 1.0; d -= 1
    elif d < 0:
        for i in np.argsort(hf)[::1]:
            if d >= 0: break
            if h[i] == 1: h[i] = 0.0; d += 1
    return h

        Z = np.where(h == 0)[0]

                c = compute_c5(h)

        if bs is None: break
        h[bs[0]], h[bs[1]] = 0.0, 1.0

                gr[:n - L] += w * (1.0 - h[L:])
                gr[L:] -= w * h[:n - L]
        v = 0.8 * v - lr * gr
        h = project_feasible(h + v, ts)
    return h

def run(seed=42, budget_s=1100, **kw):
    t0 = time.time()
    dl = t0 + budget_s
    bh, bc5 = None, 1e9

    h0 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)

    n = 70
    tgt = 35
    if h0 is not None:
        n = len(h0)
        tgt = n // 2

    # Continuous gradient descent
    hf = h0.copy() if h0 is not None else np.full(n, 0.5)
    hf = project_feasible(hf, float(tgt))
    hf = cont_grad(hf, float(tgt), 150, dl - 80)
    c = compute_c5(hf)
    if c < bc5: bc5, bh = c, hf.copy()

    # Binary local search from continuous result
    hb = to_binary(hf, tgt)
    hb, cb = ls_binary(hb, 400)
    if cb < bc5: bc5, bh = cb, hb.copy()

    # Binary local search from initial values
    if h0 is not None:
        hb0 = to_binary(h0, tgt)
        hb0, cb0 = ls_binary(hb0, 300)
        if cb0 < bc5: bc5, bh = cb0, hb0.copy()

    # Kick-and-local-search (random restarts around best)
    rng = np.random.RandomState(seed)
    for k in range(40):
        tr = dl - time.time()
        if tr < 5: break
        hk = bh.copy()
        nk = rng.randint(3, 8)
        for _ in range(nk):
            O = np.where(hk == 1)[0]
            Z = np.where(hk == 0)[0]
            if len(O) == 0 or len(Z) == 0: break
            hk[O[rng.randint(len(O))]], hk[Z[rng.randint(len(Z))]] = 0.0, 1.0
        hk, ck = ls_binary(hk, 250)
        if ck < bc5: bc5, bh = ck, hk

    # Simulated annealing (multiple runs)
    for s in [seed + 10, seed + 77, seed + 155]:
        tr = dl - time.time()
        if tr < 15: break
        d1 = min(tr * 0.2, 80)
        sh, sc = sa_binary(bh, d1, 0.003, 0.0001, s)
        if sc < bc5: bc5, bh = sc, sh
        hb2 = bh.copy()
        hb2, cb2 = ls_binary(hb2, 300)
        if cb2 < bc5: bc5, bh = cb2, hb2

    # GA on binary space
    tr = dl - time.time()
    if tr > 25:
        rng2 = np.random.RandomState(seed + 500)
        ps = 50
        pop = [bh.copy() for _ in range(6)]
        for i in range(ps - 6):
            if i < 14 and np.allclose(bh, bh.round()):
                hp = bh.copy()
                O = np.where(hp == 1)[0]
                Z = np.where(hp == 0)[0]
                for _ in range(rng2.randint(3, 10)):
                    hp[O[rng2.randint(len(O))]], hp[Z[rng2.randint(len(Z))]] = 0.0, 1.0
                pop.append(hp)
            else:
                hr = np.zeros(n)
                hr[rng2.choice(n, tgt, replace=False)] = 1.0
                pop.append(hr)
        gh, gc = ga_search(pop, tgt, seed + 900, dl - 10)
        if gc < bc5: bc5, bh = gc, gh
        bh, cb = ls_binary(bh.copy(), 400)
        if cb < bc5: bc5, bh = cb, bh

    # More kick-and-local-search after GA
    rng3 = np.random.RandomState(seed + 2000)
    for k in range(20):
        tr = dl - time.time()
        if tr < 4: break
        hk = bh.copy()
        nk = rng3.randint(2, 6)
        for _ in range(nk):
            O = np.where(hk == 1)[0]
            Z = np.where(hk == 0)[0]
            if len(O) == 0 or len(Z) == 0: break
            hk[O[rng3.randint(len(O))]], hk[Z[rng3.randint(len(Z))]] = 0.0, 1.0
        hk, ck = ls_binary(hk, 200)
        if ck < bc5: bc5, bh = ck, hk

    # Final local search polish
    tr = dl - time.time()
    if tr > 2 and bh is not None:
        if np.allclose(bh, bh.round(), atol=0.01):
            bh = bh.round()
            bh, cb = ls_binary(bh, min(500, max(1, int(tr * 20))))
            if cb < bc5: bc5 = cb

    if bh is None:
        bh = np.full(70, 0.5)
    bc5 = compute_c5(bh)
    return bh, float(bc5), len(bh)
