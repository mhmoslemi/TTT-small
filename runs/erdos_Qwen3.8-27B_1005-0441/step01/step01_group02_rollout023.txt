import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h[free] = np.clip(h[free], 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for j in range(n):
            if 0.0 < h[j] < 1.0 and 0.0 < h[j] + diff < 1.0:
                h[j] += diff
                break
    return h

def _slsqp(h0, beta, maxiter=150):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    nm1 = n - 1
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = corr.max()
        return (mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta) * dx
    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = corr.max()
        w = np.exp(beta * (corr - mc))
        ws = w.sum()
        if ws < 1e-300:
            return np.zeros(n)
        w /= ws
        g = np.zeros(n)
        for i in range(2 * n - 1):
            wi = w[i]
            if wi < 1e-14:
                continue
            d = i - nm1
            if d >= 0:
                g[:n-d] += wi * (1.0 - x[d:])
                g[d:] -= wi * x[:n-d]
            else:
                dd = -d
                g[dd:] += wi * (1.0 - x[:n-dd])
                g[:n-dd] -= wi * x[dd:]
        return g * dx
    def _cf(x):
        return x.sum() - target
    def _cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(_obj, h0, jac=_grad, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return _project(r.x)
    except Exception:
        return _project(h0.copy())

def _nm_polish(h0, t_limit, niter=15000):
    def _obj(x):
        return compute_c5(_project(x))
    def _cb(x):
        if time.time() > t_limit:
            raise ValueError("timeout")
    try:
        r = minimize(_obj, h0.copy(), method='Nelder-Mead',
                     options={'maxiter': niter, 'xatol': 1e-12, 'fatol': 1e-14},
                     callback=_cb)
        h = _project(r.x)
        return h, compute_c5(h)
    except Exception:
        return _project(h0), compute_c5(_project(h0))

def _pairwise_cd(h, n_sweeps=4, t_budget=80):
    n = len(h)
    h = h.copy()
    c = compute_c5(h)
    t0 = time.time()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(n_sweeps):
        if time.time() - t0 > t_budget:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() - t0 > t_budget:
                    break
                ho_i, ho_j = h[i], h[j]
                s = ho_i + ho_j
                tlo = max(0.0, s - 1.0)
                thi = min(1.0, s)
                if thi - tlo < 1e-10:
                    continue
                a, b = tlo, thi
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                h[i], h[j] = c1, s - c1
                f1 = compute_c5(h)
                h[i], h[j] = c2, s - c2
                f2 = compute_c5(h)
                for _ in range(14):
                    if f1 < f2:
                        b, c2, f2 = c2, c1, f1
                        c1 = b - gr * (b - a)
                        h[i], h[j] = c1, s - c1
                        f1 = compute_c5(h)
                    else:
                        a, c1, f1 = c1, c2, f2
                        c2 = a + gr * (b - a)
                        h[i], h[j] = c2, s - c2
                        f2 = compute_c5(h)
                bm = (a + b) / 2.0
                h[i], h[j] = bm, s - bm
                cf = compute_c5(h)
                if cf < c - 1e-13:
                    c = cf
                    improved = True
                else:
                    h[i], h[j] = ho_i, ho_j
        if not improved:
            break
    h = _project(h)
    return h, compute_c5(h)

def _beam_search(h_best, c_best, rng, t_budget, pop_size=8):
    t0 = time.time()
    n = len(h_best)
    pop = [(h_best.copy(), c_best)]
    for k in range(1, pop_size):
        h = _project(h_best + 0.12 * rng.randn(n))
        pop.append((h, compute_c5(h)))
    h_rand = np.zeros(n)
    h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    pop.append((h_rand, compute_c5(h_rand)))
    pop = pop[:pop_size]
    for gen in range(60):
        if time.time() - t0 > t_budget:
            break
        pop.sort(key=lambda x: x[1])
        if pop[0][1] < c_best:
            c_best = pop[0][1]
            h_best = pop[0][0].copy()
        new_cands = []
        top_k = min(4, len(pop))
        for i in range(top_k):
            hi = pop[i][0]
            for scale in [0.03, 0.1, 0.25]:
                hn = _project(hi + scale * rng.randn(n))
                new_cands.append((hn, compute_c5(hn)))
        pop.extend(new_cands)
        pop.sort(key=lambda x: x[1])
        pop = pop[:pop_size]
        if gen % 5 == 4 and time.time() - t0 < t_budget - 20:
            hr = _slsqp(pop[0][0], beta=80, maxiter=100)
            cr = compute_c5(hr)
            pop[0] = (hr, cr)
            if cr < c_best:
                c_best = cr
                h_best = hr.copy()
    return h_best, c_best

def _try_rounding(h, c):
    n = len(h)
    for threshold in [0.02, 0.005, 0.001]:
        h_r = h.copy()
        changed = False
        for j in range(n):
            if h_r[j] < threshold:
                h_r[j] = 0.0
                changed = True
            elif 1.0 - h_r[j] < threshold:
                h_r[j] = 1.0
                changed = True
        if not changed:
            continue
        h_r = _project(h_r)
        c_r = compute_c5(h_r)
        if c_r < c:
            c = c_r
            h = h_r
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.zeros(70)
        h_init[:35] = 1.0
    n = len(h_init)
    best_h = _project(h_init)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing beta
    for beta in [25, 60, 150, 300]:
        if time.time() - t0 > budget_s - 500:
            break
        h = _slsqp(best_h, beta, maxiter=200)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h

    # Phase 2: NM polish
    if time.time() - t0 < budget_s - 350:
        best_h, best_c5 = _nm_polish(best_h, time.time() + 120)

    # Phase 3: Pairwise CD
    if time.time() - t0 < budget_s - 250:
        best_h, best_c5 = _pairwise_cd(best_h, n_sweeps=3, t_budget=60)

    # Phase 4: Beam search (global exploration)
    t_rem = budget_s - (time.time() - t0) - 120
    if t_rem > 80:
        best_h, best_c5 = _beam_search(best_h, best_c5, rng, t_budget=t_rem)

    # Phase 5: NM + CD + rounding
    if time.time() - t0 < budget_s - 60:
        best_h, best_c5 = _nm_polish(best_h, time.time() + 80)
        best_h, best_c5 = _pairwise_cd(best_h, n_sweeps=3, t_budget=40)
        best_h, best_c5 = _try_rounding(best_h, best_c5)

    # Phase 6: Second beam from best
    t_rem = budget_s - (time.time() - t0) - 60
    if t_rem > 60:
        best_h, best_c5 = _beam_search(best_h, best_c5, rng, t_budget=t_rem)

    # Phase 7: Final polish
    if time.time() - t0 < budget_s - 15:
        best_h, best_c5 = _nm_polish(best_h, time.time() + 40)
        best_h, best_c5 = _pairwise_cd(best_h, n_sweeps=3, t_budget=20)
        best_h, best_c5 = _try_rounding(best_h, best_c5)

    # Phase 8: Last tiny NM
    if time.time() - t0 < budget_s - 5:
        best_h, best_c5 = _nm_polish(best_h, budget_s - 3)
        best_h, best_c5 = _pairwise_cd(best_h, n_sweeps=2, t_budget=5)

    best_h = _project(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, n)
