import numpy as np
from scipy.optimize import minimize
import time
import math

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _to_binary(h, n, rng):
    t = n // 2
    idx = np.argsort(-h, kind='stable')[:t]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _eq_c(x):
    return float(np.sum(x) - len(x) / 2.0)

class _PNorm:
    def __init__(self, p):
        self.p = p
    def __call__(self, x):
        c = _corr_arr(x)
        p = self.p
        lc = np.log(c + 1e-300)
        m = float(np.max(lc))
        val = m + np.log(np.sum(np.exp(p * (lc - m)))) / p
        return float(np.exp(val))

def _bsearch_full(h, rng, rounds, tmax, ksize, kicks):
    n = len(h)
    ts = time.time()
    h = h.copy()
    c5 = _eval_c5(h)
    bc5 = c5
    bh = h.copy()
    for kick in range(kicks + 1):
        if time.time() - ts > tmax:
            break
        if kick > 0:
            for _ in range(ksize):
                oi = np.where(h > 0.5)[0]
                zi = np.where(h < 0.5)[0]
                if len(oi) == 0 or len(zi) == 0:
                    break
                a = int(rng.integers(len(oi)))
                b = int(rng.integers(len(zi)))
                h[oi[a]], h[zi[b]] = h[zi[b]], h[oi[a]]
            c5 = _eval_c5(h)
        for _ in range(rounds):
            if time.time() - ts > tmax:
                break
            oi = np.where(h > 0.5)[0]
            zi = np.where(h < 0.5)[0]
            if len(oi) == 0 or len(zi) == 0:
                break
            bd = 0.0
            bnc = c5
            bij = None
            for i in oi:
                for j in zi:
                    h[i], h[j] = h[j], h[i]
                    nc = _eval_c5(h)
                    d = nc - c5
                    h[i], h[j] = h[j], h[i]
                    if d < bd - 1e-15:
                        bd = d
                        bnc = nc
                        bij = (i, j)
            if bij is None:
                break
            i, j = bij
            h[i], h[j] = h[j], h[i]
            c5 = bnc
            if c5 < bc5:
                bc5 = c5
                bh = h.copy()
    return bh, bc5

    n = len(h0)
    try:
        res = minimize(_PNorm(p), h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _eq_c}],
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': iters, 'ftol': 1e-14})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0.copy()

import numpy as np
from scipy.optimize import minimize
import time
import math

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _to_binary(h, n, rng):
    t = n // 2
    idx = np.argsort(-h, kind='stable')[:t]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _eq_c(x):
    return float(np.sum(x) - len(x) / 2.0)

class _PNorm:
    def __init__(self, p):
        self.p = p
    def __call__(self, x):
        c = _corr_arr(x)
        p = self.p
        lc = np.log(c + 1e-300)
        m = float(np.max(lc))
        val = m + np.log(np.sum(np.exp(p * (lc - m)))) / p
        return float(np.exp(val))

def _bsearch_full(h, rng, rounds, tmax, ksize, kicks):
    n = len(h)
    ts = time.time()
    h = h.copy()
    c5 = _eval_c5(h)
    bc5 = c5
    bh = h.copy()
    for kick in range(kicks + 1):
        if time.time() - ts > tmax:
            break
        if kick > 0:
            for _ in range(ksize):
                oi = np.where(h > 0.5)[0]
                zi = np.where(h < 0.5)[0]
                if len(oi) == 0 or len(zi) == 0:
                    break
                a = int(rng.integers(len(oi)))
                b = int(rng.integers(len(zi)))
                h[oi[a]], h[zi[b]] = h[zi[b]], h[oi[a]]
            c5 = _eval_c5(h)
        for _ in range(rounds):
            if time.time() - ts > tmax:
                break
            oi = np.where(h > 0.5)[0]
            zi = np.where(h < 0.5)[0]
            if len(oi) == 0 or len(zi) == 0:
                break
            bd = 0.0
            bnc = c5
            bij = None
            for i in oi:
                for j in zi:
                    h[i], h[j] = h[j], h[i]
                    nc = _eval_c5(h)
                    d = nc - c5
                    h[i], h[j] = h[j], h[i]
                    if d < bd - 1e-15:
                        bd = d
                        bnc = nc
                        bij = (i, j)
            if bij is None:
                break
            i, j = bij
            h[i], h[j] = h[j], h[i]
            c5 = bnc
            if c5 < bc5:
                bc5 = c5
                bh = h.copy()
    return bh, bc5

    n = len(h0)
    try:
        res = minimize(_PNorm(p), h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _eq_c}],
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': iters, 'ftol': 1e-14})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0.copy()
