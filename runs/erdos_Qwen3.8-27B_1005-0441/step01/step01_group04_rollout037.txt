    c = corr * dx
    cmax = np.max(c)
    shifted = c - cmax
    return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def _softmax_grad(h, alpha):

    c = corr_raw * dx

    w = np.exp(alpha * shifted)
    w /= np.sum(w)

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _proj(p, n):
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

def _fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    return _proj(h, n)

def _softmax_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def _softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    w = np.exp(alpha * shifted)
    w /= np.sum(w)
    m = 2 * n - 1
    grad = np.zeros(n)
    for j in range(n):
        gj = 0.0
        for k in range(m):
            s_k = k - (n - 1)
            g = 0.0
            if 0 <= j - s_k < n:
                g += (1.0 - h[j - s_k])
            if 0 <= j + s_k < n:
                g += (-h[j + s_k])
            gj += w[k] * dx * g
        grad[j] = gj
    return grad

def _subgrad_optimize(h0, n, n_iter, c_lr):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    velocity = np.zeros(n)
    mu = 0.7
    for k in range(n_iter):

        shifted = c - cmax
        w = np.exp(50.0 * shifted)
        w /= np.sum(w)
        grad = np.zeros(n)
        for j in range(n):
            gj = 0.0
            m = 2 * n - 1
            for kk in range(m):
                s_k = kk - (n - 1)
                g = 0.0
                if 0 <= j - s_k < n:
                    g += (1.0 - h[j - s_k])
                if 0 <= j + s_k < n:
                    g += (-h[j + s_k])
                gj += w[kk] * g
            grad[j] = gj * (2.0 / n)
        lr = c_lr / np.sqrt(k + 1)
        velocity = mu * velocity + (1 - mu) * grad
        h_new = _proj(h - lr * velocity, n)
        c = _eval_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _coord_descent(h0, n, max_rounds=100):
    h = h0.copy()
    for r in range(max_rounds):
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr_raw))
        s = k_star - (n - 1)
        changed = False
        for j in range(n):
            coeff = 0.0
            if 0 <= j - s < n:
                coeff += (1.0 - h[j - s])
            if 0 <= j + s < n:
                coeff += (-h[j + s])
            if coeff > 1e-10:
                new_val = 0.0
            elif coeff < -1e-10:
                new_val = 1.0
            else:
                continue
            if abs(new_val - h[j]) > 1e-10:
                h[j] = new_val
                changed = True
        if not changed:
            break
        h = _proj(h, n)
    return h

def _slsqp_optimize(h0, n, alpha, max_iter):
    def fun(h):
        return _softmax_obj(h, alpha)
    def cons_eq(h):
        return np.sum(h) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': cons_eq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return _fix_sum(res.x, n)
    except Exception:
        return h0.copy()

def _binary_init(n, rng):
    pos = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    pos[ones] = 1.0
    pos += rng.normal(0, 0.03, n)
    return _fix_sum(pos, n)

def _structured_init(n, freq):
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / n + 0.3)
    return _fix_sum(h, n)

def _block_init(n, p):
    h = np.tile(np.array([0.9] * p + [0.1] * (n - p)), n // (n - p) + 1)[:n]
    return _fix_sum(h, n)
