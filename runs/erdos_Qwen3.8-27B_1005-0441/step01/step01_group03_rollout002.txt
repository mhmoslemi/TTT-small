        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-12:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_overlaps(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def compute_grad(h, n, ov, tau):
    w = np.exp(tau * (ov - ov.max()))
    w /= w.sum()
    nh = n - 1
    g = np.zeros(n)
    omh = 1.0 - h
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-6:
            continue
        L = m - nh
        if L == 0:
            g += wm * (omh - h)
        elif L > 0:
            g[L:] += wm * omh[:n - L]
            g[:n - L] -= wm * h[L:]
        else:
            L2 = -L
            g[:n - L2] += wm * omh[L2:]
            g[L2:] -= wm * h[:n - L2]
    return g

def optimize(h0, n, deadline, rng):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    lr = 0.3
    tau = 80.0
    stalled = 0
    it = 0
    cur_c5 = best_c5
    while time.time() < deadline:
        ov = get_overlaps(h, n)
        cur_c5 = float(np.max(ov))
        g = compute_grad(h, n, ov, tau)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        accepted = False
        for _ in range(20):
            h_new = project_feasible(h - lr * g, n)
            c_new = get_c5(h_new, n)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new.copy()
                h = h_new
                cur_c5 = c_new
                accepted = True
                stalled = 0
                break
            if c_new < cur_c5 - 1e-10:
                h = h_new
                cur_c5 = c_new
                accepted = True
                break
            lr *= 0.75
        if not accepted:
            stalled += 1
            h = best_h.copy()
            cur_c5 = best_c5
        it += 1
        tau = min(500.0, tau * 1.002)
        if stalled > 150:
            scale = 0.08 + 0.12 * min(1.0, stalled / 400.0)
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, n)
            cur_c5 = get_c5(h, n)
            lr = 0.3
            tau = max(50.0, tau * 0.5)
            stalled = 0
    for _ in range(800):
        if time.time() > deadline:
            break
        h_try = best_h + rng.randn(n) * 0.008
        h_try = project_feasible(h_try, n)
        c_try = get_c5(h_try, n)
        if c_try < best_c5 - 1e-12:
            best_c5 = c_try
            best_h = h_try.copy()
    return best_h, best_c5

def make_structured(n):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    if n % 2 == 1:
        h[half] = 0.5
    return project_feasible(h, n)

def make_sinusoidal(n):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.5 * np.sin(np.pi * x)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is not None:
        n0 = len(h0)
        dl1 = t0 + budget_s * 0.50
        bh, bc5 = optimize(h0, n0, dl1, rng)
        best_h, best_c5, best_n = bh, bc5, n0

    n_candidates = [48, 50, 52, 44, 54, 56, 42, 60]
    if h0 is not None:
        n_candidates = [x for x in n_candidates if x != len(h0)]

    frac2 = 0.42
    rem_start = budget_s - (budget_s - frac2 * budget_s)
    per_try = (budget_s * frac2) / len(n_candidates)
    for n_try in n_candidates:
        if time.time() > t0 + budget_s - 30:
            break
        dl_try = min(time.time() + per_try, t0 + budget_s - 30)
        if time.time() >= dl_try:
            continue
        starts = [make_structured(n_try), make_sinusoidal(n_try),
                  project_feasible(rng.rand(n_try), n_try)]
        for h_start in starts:
            if time.time() > dl_try:
                break
            bh, bc5 = optimize(h_start, n_try, dl_try, rng)
            if bc5 < best_c5:
                best_h, best_c5, best_n = bh, bc5, n_try

    if best_h is not None and time.time() < t0 + budget_s - 8:
        bh, bc5 = optimize(best_h, best_n, t0 + budget_s - 3, rng)
        if bc5 < best_c5:
            best_h, best_c5 = bh, bc5

    h_final = project_feasible(best_h, best_n)
    c5_final = get_c5(h_final, best_n)
    h_final = np.clip(h_final, 0.0, 1.0)
    return h_final, float(c5_final), best_n
