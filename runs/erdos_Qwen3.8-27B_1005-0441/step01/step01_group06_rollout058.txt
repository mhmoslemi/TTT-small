import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.999999
        else:
            mask = h > 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, max_sweeps=200, t0=0, budget_s=0, time_frac=0.5):
    h = h.copy()
    best_c5 = compute_c5(h)
    nc = 81
    x_c = np.linspace(0.0, 1.0, nc)
    x2_c = x_c ** 2
    for sweep in range(max_sweeps):
        if time.time() - t0 > budget_s * time_frac:
            break
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = old
            j_all = np.arange(2 * n - 1)
            p_idx = i + n - 1 - j_all
            q_idx = i - n + 1 + j_all
            slope = np.zeros(2 * n - 1)
            mask_p = (p_idx >= 0) & (p_idx < n) & (p_idx != i)
            if mask_p.any():
                slope[mask_p] += (1.0 - h[p_idx[mask_p]])
            mask_q = (q_idx >= 0) & (q_idx < n) & (q_idx != i)
            if mask_q.any():
                slope[mask_q] += (-h[q_idx[mask_q]])
            slope[n - 1] += 1.0
            quad = np.zeros(2 * n - 1)
            quad[n - 1] += -1.0
            C_all = c0[None, :] + slope[None, :] * x_c[:, None] + quad[None, :] * x2_c[:, None]
            f_vals = C_all.max(axis=1)
            h[i] = x_c[f_vals.argmin()]
        h = project_feasible(h, n)
        if (sweep + 1) % 10 == 0:
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
    return h, best_c5

def soft_obj(x, n, beta):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full")
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def eq_jac(x):
    return np.ones(len(x))

def slsqp_polish(h, n, t0, budget_s, time_frac=0.3):
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    betas = [10, 50, 200, 1000, 5000]
    for beta in betas:
        if time.time() - t0 > budget_s * time_frac:
            break
        try:
            mi = 100 if n <= 200 else 60
            res = minimize(soft_obj, best, args=(n, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best = h_new.copy()
                    best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def make_block_h(n, rng, nblocks):
    h = np.zeros(n)
    bsz = n // nblocks
    for b in range(nblocks):
        start = b * bsz
        end = start + bsz if b < nblocks - 1 else n
        val = rng.rand()
        h[start:end] = val
    return project_feasible(h, n)
