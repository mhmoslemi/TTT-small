```python
def solve_lp(h_star, n):
    from scipy.optimize import linprog
    dx = 2.0 / n
    num_lags = 2 * n - 1
    corr = np.correlate(h_star, 1.0 - h_star, mode='full')
    
    # Compute all gradients
    # grad[k][m] = (1-h[m+lag])*1{valid} - h[m-lag]*1{valid}, lag = k-(n-1)
    A_rows = []  # constraint: g_k^T (h - h_star) + c_k ≤ t → g_k^T h - t ≤ c_k - g_k^T h_star
    for k in range(num_lags):
        lag = k - (n - 1)
        g = np.zeros(n)
        # (1-h[m+lag]) for m+lag in [0,n)
        valid_plus = np.abs(np.arange(n) + lag) < n  # m+lag in [0,n)
        # m+lag >= 0 and m+lag < n
        valid_plus = (np.arange(n) + lag >= 0) & (np.arange(n) + lag < n)
        # m-lag in [0,n)
        valid_minus = (np.arange(n) - lag >= 0) & (np.arange(n) - lag < n)
        g[valid_plus] = 1.0 - h_star[np.arange(n)[valid_plus] + lag]
        g[valid_minus] -= h_star[np.arange(n)[valid_minus] - lag]
        
        # c_k - g^T h_star is the constant term
        c_k = corr[k]
        # Constraint: g^T h - t ≤ c_k - g^T h_star
        # In standard form: [g, -1] dot [h, t] ≤ c_k - g^T h_star
        A_rows.append(np.concatenate([g, [-1.0]]))
    
    A = np.array(A_rows)
    rhs = np.array([corr[k] - A[k, :-1] @ h_star for k in range(num_lags)])
    
    # Equality: sum h = n/2
    A_eq = np.concatenate([np.ones(n), [0.0]])
    b_eq = n * 0.5
    
    # Objective: minimize t → c_obj = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A, b_ub=rhs, A_eq=[A_eq], b_eq=[b_eq],
                  bounds=bounds, method='highs')
    return res
```
