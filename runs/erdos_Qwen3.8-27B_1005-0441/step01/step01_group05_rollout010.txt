    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()

            step = sigmas[p1] * rng.standard_normal(n)
            pop[i] = project_onto_feasible(pop[p1] + step, n)
            sigmas[i] = sigmas[p1] * (0.95 + 0.1 * rng.random())

                scale = 0.1 * rng.random()
                pop[i] = project_onto_feasible(h_best + scale * rng.standard_normal(n), n)
                sigmas[i] = scale
            
            pop[survivors[i - n_survive]] = project_onto_feasible(
                pop[survivors[i]] if False else pop[order[i - n_survive]] + 
                sigmas[order[i-n_survive]] * 0.3 * rng.standard_normal(n), n)
        
        # Evaluate new population
        for i in range(n_survive, pop_size):
            fitness[i] = compute_c5(pop[i])
        
        new_best = np.argmin(fitness)
        if fitness[new_best] < best_val:
            best_val = fitness[new_best]
            best_h = pop[new_best].copy()
    
    return best_h, best_val

def slsqp_polish(h, n, t_max, temps):
    """SLSQP local optimization with log-sum-exp."""
    dx = 2.0 / n
    targets = float(n) / 2.0
    best = h.copy()
    best_val = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    
    temp_global = [0.01]
    
    def obj(h_flat):
        t = temp_global[0]
        corr = np.correlate(h_flat, 1.0 - h_flat, mode="full") * dx
        m = np.max(corr)
        return m + t * np.log(np.sum(np.exp((corr - m) / t)))
    
    def cons_eq(h_flat):
        return np.sum(h_flat) - targets
    
    def cons_eq_jac(h_flat):
        return np.ones(n)
    
    cons = ({'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac},)
    t0 = time.time()
    
    for temp in temps:
        if time.time() - t0 > t_max:
            break
        temp_global[0] = temp
        h_cur = best.copy()
        for it in range(8):
            if time.time() - t0 > t_max:
                break
            try:
                res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                             constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_onto_feasible(res.x, n)
                v = compute_c5(h_new)
                if v < best_val - 1e-14:
                    best = h_new
                    best_val = v
                h_cur = h_new
            except Exception:
                break
    
    return best, best_val

_sg_temp = 0.01

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = float(n) / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) < 1e-14:
        return h
    if s > target:
        lo, hi = 0.0, 1.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            hc = np.maximum(0.0, h - mid)
            if float(np.sum(hc)) > target:
                lo = mid
            else:
                hi = mid
        h = np.maximum(0.0, h - (lo + hi) * 0.5)
    else:
        lo, hi = 0.0, 1.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            hc = np.minimum(1.0, h + mid)
            if float(np.sum(hc)) < target:
                lo = mid
            else:
                hi = mid
        h = np.minimum(1.0, h + (lo + hi) * 0.5)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _subgrad_dir(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    g = np.zeros(n)
    if 0 <= k_star < n:
        g[k_star:] += 1.0
        g[k_star:] -= h[:n - k_star]
        g[:n - k_star] -= h[k_star:]
    elif k_star >= n:
        k2 = k_star - (n - 1)
        lo = k2
        hi = n
        g[lo:hi] += 1.0 - h[:hi - lo]
        g[lo:hi] -= h[:hi - lo]
    else:
        k2 = n - 1 + k_star
        g[:n - k2] += 1.0 - h[k2:]
        g[:n - k2] -= h[k2:]
    g *= dx
    return g
