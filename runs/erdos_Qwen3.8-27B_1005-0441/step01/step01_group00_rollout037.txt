            for d in deltas:
                h2 = best.copy()
                h2[j] = np.clip(h2[j] + d, 0.0, 1.0)
                diff = n * 0.5 - h2.sum()
                if abs(diff) > 1e-12:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    free = mask & (h2 > 1e-8) & (h2 < 1.0 - 1e-8)
                    nf = free.sum()
                    if nf == 0:
                        continue
                    h2[free] += diff / nf
                    h2 = np.clip(h2, 0, 1)
                v = _c5(h2)
                if v < best_v - 1e-13:
                    best = h2
                    best_v = v
                    improved = True
                    break
        if not improved:
            break
    return best, best_v


def _try_quantize(h, n):
    """Try snapping to fewer levels and see if C5 improves."""
    best = h.copy()
    best_v = _c5(h)
    levels_sets = [
        np.array([0.0, 0.5, 1.0]),
        np.array([0.0, 0.25, 0.5, 0.75, 1.0]),
        np.array([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]),
    ]
    for levels in levels_sets:
        h_q = np.array([levels[np.argmin(np.abs(levels - x))] for x in h])
        h_q = _proj(h_q, n)
        v = _c5(h_q)
        if v < best_v:
            best = h_q
            best_v = v
    return best, best_v


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = np.inf

    def _remaining():
        return max(0.0, budget_s - (time.time() - t0))

    starts = []
    try:
        hinit = np.array(initial_h_values, dtype=float).copy()
        starts.append((_proj(hinit, len(hinit))))
    except Exception:
        pass

    n_vals = [70, 100]
    for nv in n_vals:
        t = np.arange(nv) / nv
        for freq in [1, 2, 3]:
            for amp in [0.15, 0.30, 0.40]:
                hh = 0.5 + amp * np.sin(2.0 * np.pi * freq * t)
                starts.append(_proj(hh, nv))
        for _ in range(2):
            hh = 0.2 + 0.3 * rng.rand(nv)
            starts.append(_proj(hh, nv))

    unique_starts = [starts[0]]
    for s in starts[1:]:
        if len(s) > 120:
            continue
        dup = False
        for u in unique_starts:
            if len(u) == len(s) and np.linalg.norm(s - u) < 0.03:
                dup = True
                break
        if not dup:
            unique_starts.append(s)
        if len(unique_starts) >= 15:
            break
    starts = unique_starts

    eps_sched = [0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.001]

    for idx, h0 in enumerate(starts):
        if _remaining() < 20:
            break
        n = len(h0)
        cur = h0.copy()
        cur_v = _c5(cur)
        for eps in eps_sched:
            if _remaining() < 12:
                break
            if eps >= 0.04:
                mi = 250
                ft = 1e-12
            elif eps >= 0.005:
                mi = 500
                ft = 1e-13
            else:
                mi = 800
                ft = 1e-14
            h_new = _slsqp_step(cur, n, eps, mi, ft)
            v_new = _c5(h_new)
            if v_new < cur_v:
                cur = h_new
                cur_v = v_new
            else:
                break
        if _remaining() > 40:
            cur, cur_v = _pg_polish(cur, n, 150)
        if _remaining() > 20:
            cur, cur_v = _coor_polish(cur, n, 20)
        if cur_v < best_val:
            best_val = cur_v
            best_h = cur.copy()

    if best_h is not None and _remaining() > 15:
        n = len(best_h)
        for eps in [0.0008, 0.0004]:
            if _remaining() < 10:
                break
            h_new = _slsqp_step(best_h, n, eps, 1000, 1e-15)
            v_new = _c5(h_new)
            if v_new < best_val:
                best_val = v_new
                best_h = h_new
        if _remaining() > 25:
            best_h, best_val = _pg_polish(best_h, n, 400)
        if _remaining() > 10:
            best_h, best_val = _coor_polish(best_h, n, 40)
        if _remaining() > 5:
            best_h, best_val = _try_quantize(best_h, n)
            if best_val > _c5(best_h):
                best_val = _c5(best_h)

    if best_h is None:
        n = 70
        best_h = _proj(np.full(n, 0.5), n)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
