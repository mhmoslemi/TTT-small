import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    return c5, corr * dx

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-15:
                break
            h += diff * room / room.sum()
        else:
            if h.sum() < 1e-15:
                break
            h *= min(1.0, target / h.sum())
        h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            for i in np.argsort(1.0 - h)[-10:]:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            for i in np.argsort(h)[-10:]:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return h

def _lse_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    return m + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def _topk_sum(h, k=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.sum(np.sort(corr)[-k:]))

    target = n / 2.0
    for _ in range(200):

        if abs(diff) < 1e-15:

                diff -= a

                h[i] -= s
                diff += s

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-15:
                break
            h += diff * room / room.sum()
        else:
            if h.sum() < 1e-15:
                break
            h *= min(1.0, target / max(h.sum(), 1e-15))
        h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            for i in np.argsort(1.0 - h)[-8:]:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
                if abs(diff) < 1e-14:
                    break
        else:
            for i in np.argsort(h)[-8:]:
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
                if abs(diff) < 1e-14:
                    break
    return h

class _LSE:
    def __init__(self, alpha):
        self.alpha = alpha
    def __call__(self, h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        s = corr - m
        return m + np.log(np.sum(np.exp(self.alpha * s))) / self.alpha

class _TopK:
    def __init__(self, k):
        self.k = k
    def __call__(self, h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argpartition(corr, -self.k)[-self.k:]
        return float(np.sum(corr[idx]))

class _SoftMax:
    def __init__(self, beta):
        self.beta = beta
    def __call__(self, h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        s = corr - m
        return m + np.log(np.sum(np.exp(self.beta * s))) / self.beta

def _eq_c(h):
    return float(np.sum(h) - len(h) / 2.0)

def _slsqp(h0, obj, n, max_iter=300, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_c}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        return np.asarray(res.x, dtype=float)
    except Exception:
        return h0.copy()

class _Coord1D:
    def __init__(self, h, j, n, k):
        self.h = h
        self.j = j
        self.n = n
        self.k = k
    def __call__(self, v):
        h2 = self.h.copy()
        h2[self.j] = v
        h2 = _fix_sum(h2, self.n)
        obj = _TopK(self.k)
        return obj(h2)

def _coord_polish(h, n, max_rounds=30, k=3):
    h = h.copy()
    cur = _eval_c5(h)
    for _ in range(max_rounds):
        best_j = -1
        best_v = None
        best_f = cur
        for j in range(n):
            o = _Coord1D(h, j, n, k)
            res = minimize_scalar(o, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-8})
            if res.fun < best_f - 1e-12:
                best_f = res.fun
                best_j = j
                best_v = res.x
        if best_j < 0:
            break
        h[best_j] = best_v
        h = _fix_sum(h, n)
        cur = _eval_c5(h)
    return h

def _perturb(h, rng, scale):
    h2 = h + rng.normal(0, scale, len(h))
    return _fix_sum(h2, len(h))

def _make_sin(n, rng):
    freq = rng.uniform(0.8, 2.5)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x + phase)
    return _fix_sum(h, n)

def _make_blocks(n, rng):
    nblk = int(rng.integers(3, 8))
    edges = np.sort(rng.uniform(0, n, nblk - 1)).astype(int)
    h = np.zeros(n)
    for b in range(nblk):
        lo = edges[b - 1] if b > 0 else 0
        hi = edges[b] if b < nblk - 1 else n
        h[lo:hi] = rng.uniform(0.15, 0.85)
    return _fix_sum(h, n)

    return budget_s - (time.time() - t0)
