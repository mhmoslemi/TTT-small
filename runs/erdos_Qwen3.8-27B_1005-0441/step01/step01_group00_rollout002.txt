def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0.0, 1.0)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(h, eps, n):
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smooth_grad(h, eps, n):
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    s = np.exp((c - cmax) / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum

            g[m:] += wk * (1.0 - h[:n - m])

        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g * dx

        except Exception:

        v = _c5(h_new)
