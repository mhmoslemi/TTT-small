def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        # Adjust only elements not at bounds
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) == 0:
            h += d / n
        else:
            h[mask] += d / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    # Final correction
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += resid / np.sum(mask)
            np.clip(h, 0.0, 1.0, out=h)
    return h

import numpy as np
from scipy.optimize import minimize
import time

_GLOBAL_N = 160
_GLOBAL_TARGET = 80.0

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) == 0:
            h += d / n
        else:
            h[mask] += d / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_gradients(h, w):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = n - 1 - k
        if s >= 0:
            i_lo, i_hi = 0, n - s
        else:
            i_lo, i_hi = -s, n
        if i_lo >= i_hi:
            continue
        grad[i_lo:i_hi] += wk * (1.0 - h[i_lo + s:i_hi + s])
        seg = h[i_lo:i_hi]
        if s != 0:
            grad[i_lo + s:i_hi + s] -= wk * seg
        else:
            grad[i_lo:i_hi] -= wk * seg
    return grad

def _sum_eq(x):
    return np.sum(x) - _GLOBAL_TARGET

def _sum_eq_jac(x):
    return np.ones(len(x))

def obj_smooth(x, alpha):
    n = _GLOBAL_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -400.0, 400.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    return sm * dx

def run(seed=42, budget_s=1100, **kwargs):
    global _GLOBAL_N, _GLOBAL_TARGET
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n_init = 160
        h0 = np.zeros(n_init)
        h0[:n_init // 4] = 1.0
        h0[-n_init // 4:] = 1.0
        h0 = project_feasible(h0, n_init)

    n = len(h0)
    _GLOBAL_N = n
    _GLOBAL_TARGET = n * 0.5
    h_best = project_feasible(h0, n)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]

    # Phase 1: SLSQP with increasing alpha for smooth max
    for alpha in [50.0, 200.0, 800.0]:
        if time.time() - t0 > budget_s * 0.22:
            break
        res = minimize(
            obj_smooth, h_best.copy(), args=(alpha,),
            method='SLSQP', bounds=bounds, constraints=constraint,
            options={'maxiter': 2500, 'ftol': 1e-14}
        )
        hc = project_feasible(res.x, n)
        c5c = compute_c5(hc)
        if c5c < c5_best:
            c5_best = c5c
            h_best = hc.copy()

    # Phase 2: Momentum subgradient descent
    if time.time() - t0 < budget_s * 0.52:
        alpha_sg = 400.0
        beta = 0.85
        v = np.zeros(n)
        base_lr = 0.04
        stagnation = 0
        for step in range(4000):
            if time.time() - t0 > budget_s * 0.52:
                break
            lr = base_lr / (1.0 + 0.0005 * step)
            corr = np.correlate(h_best, 1.0 - h_best, mode='full')
            m = np.max(corr)
            shifted = np.clip(alpha_sg * (corr - m), -500.0, 500.0)
            w = np.exp(shifted)
            sw = np.sum(w)
            if sw < 1e-30:
                break
            w /= sw
            g = compute_gradients(h_best, w)
            v = beta * v + (1.0 - beta) * g
            h_new = h_best - lr * v
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-10:
                c5_best = c5_new
                h_best = h_new.copy()

                stagnation += 1
            if stagnation > 800 and step > 600:

                        best_c5 = c5_try
                        best_val = cv
                if best_val != old_val:
                    h_best[i] = best_val
                    h_best = project_feasible(h_best, n)
                    c5_best = compute_c5(h_best)
                    improved = True
            if not improved:
                break

    # Phase 4: Multi-start with structured and random initializations
    if time.time() - t0 < budget_s * 0.88:
        x = np.linspace(0, 2, n)
        starts = []
        for freq in [1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0]:

                    starts.append(project_feasible(hs, n))
        for _ in range(8):
            hr = 0.5 + 0.45 * rng.standard_normal(n)
            starts.append(project_feasible(hr, n))
        # Two-block and three-block starts
        for block_frac in [0.2, 0.25, 0.3]:
            hb = np.zeros(n)
            b = int(n * block_frac / 2)
            hb[:b] = 1.0
            hb[-b:] = 1.0
            starts.append(project_feasible(hb, n))
        h3 = np.zeros(n)
        b3 = int(n / 6)
        h3[:b3] = 1.0
        h3[n // 2 - b3:n // 2 + b3] = 1.0
        h3[-b3:] = 1.0
        starts.append(project_feasible(h3, n))

        for sp in starts:
            if time.time() - t0 > budget_s * 0.88:
                break
            c5_sp = compute_c5(sp)
            if c5_sp < c5_best:
                c5_best = c5_sp
                h_best = sp.copy()
                continue
            res = minimize(
                obj_smooth, sp.copy(), args=(300.0,),
                method='SLSQP', bounds=bounds, constraints=constraint,
                options={'maxiter': 600, 'ftol': 1e-14}
            )
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

    # Phase 5: Final high-alpha SLSQP polish + subgradient
    if time.time() - t0 < budget_s - 25:
        for alpha_final in [1500.0, 5000.0, 15000.0]:
            if time.time() - t0 > budget_s - 25:
                break
            res = minimize(
                obj_smooth, h_best.copy(), args=(alpha_final,),
                method='SLSQP', bounds=bounds, constraints=constraint,
                options={'maxiter': 4000, 'ftol': 1e-15}
            )
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()

        # Final subgradient polish with high alpha
        alpha_sg2 = 800.0
        for _ in range(1500):
            if time.time() - t0 > budget_s - 25:
                break
            corr = np.correlate(h_best, 1.0 - h_best, mode='full')

            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new

    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
