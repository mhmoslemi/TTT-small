def smooth_max_gradient(h, n, temperature):
    """Compute gradient of smooth max of overlaps."""
    dx = 2.0 / n
    # Compute all overlaps
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    ov_centered = ov - np.max(ov)  # for numerical stability
    w = np.exp(ov_centered / temperature)
    w /= w.sum()
    # w has length 2n-1, indexed from lag -(n-1) to n-1
    # w[m] corresponds to lag m - (n-1)
    
    # Gradient: for each j, sum over L of w[L] * d_ov_L/dh_j
    grad = np.zeros(n)
    for m in range(2*n-1):

            jp = j + L
            jm = j - L
            grad[j] += wm * dx * ((1 - h[jp]) if 0 <= jp < n else 0 - (h[jm] if 0 <= jm < n else 0))

import numpy as np
import time

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}."""
    tgt = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > tgt:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    # Fix small numerical errors
    diff = tgt - np.sum(hp)
    if abs(diff) > 1e-10:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def smooth_max_grad(h, n, temp):
    """Gradient of t*log(sum(exp(ov_k/t))) w.r.t. h."""
    dx = 2.0 / n
    ov = compute_overlaps(h, n)
    ov_shift = ov - np.max(ov)
    w = np.exp(ov_shift / temp)
    w /= w.sum()
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        L = m - (n - 1)
        wm = w[m]
        if wm < 1e-12:
            continue
        if L >= 0:
            grad[:n - L] += wm * dx * (1.0 - h[L:n])
            grad[L:] -= wm * dx * h[:n - L]
        else:
            Lp = -L
            grad[Lp:] += wm * dx * (1.0 - h[:n - Lp])
            grad[:n - Lp] -= wm * dx * h[Lp:n]
    return grad, float(np.max(ov))

def optimize_nesterov(h_init, n, deadline, rng, verbose=False):
    """Nesterov-accelerated gradient descent on smooth max of overlaps."""
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), n)
    y = h.copy()  # Nesterov momentum variable
    beta = 0.9
    lr = 0.3
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    
    temp = 0.3  # initial temperature for smooth max
    temp_min = 0.02
    temp_decay = 0.995
    
    it = 0
    stall_count = 0
    stuck_iters = 0
    
    while time.time() < deadline:
        # Nesterov step
        y = project_feasible(h + beta * (y - h) if it > 0 else h.copy(), n)
        
        grad, c_sm = smooth_max_grad(y, n, temp)
        
        # Line search / step
        h_new = project_feasible(y - lr * grad, n)
        c_new = get_c5(h_new, n)
        
        if c_new < best_c5 - 1e-13:
            best_c5 = c_new
            best_h = h_new.copy()
            stalled = 0
            beta = min(beta * 1.01, 0.95)
        else:
            stalled += 1
            if stalled > 50:
                lr *= 0.8
                stalled = 0
                if lr < 0.005:
                    # Restart with perturbation
                    h = best_h + rng.randn(n) * 0.2
                    h = project_feasible(h, n)
                    y = h.copy()
                    lr = 0.3
                    beta = 0.9
                    temp = max(temp, 0.1)
            elif stalled > 20:
                lr *= 0.95
        
        h = h_new
        it += 1
        temp = max(temp_min, temp * temp_decay)
        
        if it % 500 == 0 and verbose:
            print(f"  it={it} c5={best_c5:.6f} lr={lr:.4f} temp={temp:.3f}")
    
    return best_h, best_c5
