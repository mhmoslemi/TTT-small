import numpy as np
import time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def to_binary(h_frac, n):
    target = n // 2
    h = np.zeros(n)
    # Top n//2 values become 1
    idx = np.argsort(-h_frac)[:target]
    h[idx] = 1.0
    return h

def local_search_1swap(h, time_budget, seed, max_no_improve=10000):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    no_improve = 0
    c5_count = 0
    
    while time.time() - t0 < time_budget:
        if no_improve > max_no_improve:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
        c5_count += 1
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    
    return best_h, best_c5

        j = zeros[rng.randint(len(zeros))]

        if c5_ls < best_c5 - 1e-15:
            best_h = h_ls
            best_c5 = c5_ls
    
    return best_h, best_c5

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def local_search_exhaustive(h, time_budget, max_rounds=500):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_budget:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 0
        best_swap = None
        
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                c5 = compute_c5(h)
                if c5 < best_c5 - best_delta - 1e-15:
                    best_delta = best_c5 - c5
                    best_swap = (int(i), int(j))
            h[i] = 1.0
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        best_c5 -= best_delta
        best_h = h.copy()
    
    return best_h, best_c5
