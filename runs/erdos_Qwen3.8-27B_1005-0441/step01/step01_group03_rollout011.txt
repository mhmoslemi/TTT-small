import numpy as np

def project_feasible(h, n):
    """Project h onto {x : 0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_overlaps(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr  # indices 0 to 2n-2, lag k corresponds to corr[k-(n-1)]

def gradient_for_lag(h, lag, n):
    """Gradient of overlap at given lag w.r.t. h"""
    g = np.zeros(n)
    # overlap(lag) = sum_{i} h[i] * (1-h[i+lag]) for valid i
    # d/dh[j]: appears as h[j] in term (j, j+lag) contributing (1-h[j+lag])
    #          appears as h[j] in term (j-lag, j) contributing -h[j-lag]... wait
    # Actually: overlap = sum_i h[i] * (1 - h[i+lag]) for i where both i and i+lag in [0,n)
    # = sum_i h[i] - sum_i h[i]*h[i+lag]
    # d/dh[j] = 1 - sum over i where i=j: (1-h[i+lag]) -- no wait
    # d/dh[j] of sum_i h[i]*(1-h[i+lag]):
    #   term where i=j: (1-h[j+lag]) if j+lag valid
    #   term where i+lag=j i.e. i=j-lag: -h[j-lag] if j-lag valid
    # So g[j] = (1-h[j+lag]) if j+lag<n else 0, minus h[j-lag] if j-lag>=0 else 0
    for j in range(n):

            g[j] += 1.0 - h[j + lag]
        if 0 <= j - lag < n:
            g[j] -= h[j - lag]
    return g

def optimize_smooth_max(h, n, deadline, rng, beta=50.0):
    """Minimize smooth max of overlaps using Adam + projection"""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.1
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Precompute lag indices
    n_lags = 2 * n - 1
    
    while time.time() < deadline:
        t += 1
        ov = np.correlate(h, 1.0 - h, mode='full') * dx  # shape (2n-1,)
        
        # Softmax weights for lags
        ov_norm = ov - np.max(ov)  # numerical stability
        w = np.exp(beta * ov_norm)
        w /= np.sum(w)
        
        # Gradient: for each lag, compute gradient and weight
        g = np.zeros(n)
        for k in range(n_lags):
            if w[k] < 1e-12:
                continue
            lag = k - (n - 1)
            # For lag L: g[j] += (1-h[j+L]) if j+L<n, -h[j-L] if j-L>=0
            if lag >= 0:
                # j from 0 to n-1-lag: g[j] += (1-h[j+lag])
                g[:n-lag] += w[k] * (1.0 - h[lag:])
                # j from lag to n-1: g[j] -= h[j-lag]
                g[lag:] -= w[k] * h[:n-lag]
            else:
                L = -lag
                # j from L to n-1: g[j] += (1-h[j-L])
                g[L:] += w[k] * (1.0 - h[:n-L])
                # j from 0 to n-1-L: g[j] -= h[j+L]
                g[:n-L] -= w[k] * h[L:]
        
        # Adam update
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = project_feasible(h_new, n)
        
        c_new = get_c5(h_new, n)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Decay beta over time (start smooth, end sharp)
        beta = max(5.0, 50.0 - t * 0.01)
    
    return best_h, best_c5

import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def optimize_adam(h, n, deadline, rng, beta0=30.0, lr0=0.3):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    stalled = 0
    
    while time.time() < deadline:
        t += 1
        ov = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Adaptive beta: start higher for smoother, end lower to focus
        beta = max(5.0, beta0 - t * 0.005)
        
        ov_shift = ov - np.max(ov)
        w = np.exp(beta * ov_shift)
        w /= np.sum(w)
        
        # Gradient
        g = np.zeros(n)
        for k in range(n_lags):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                g[:n - lag] += w[k] * (1.0 - h[lag:n])
                g[lag:n] -= w[k] * h[:n - lag]
            else:
                L = -lag
                g[L:n] += w[k] * (1.0 - h[:n - L])
                g[0:n - L] -= w[k] * h[L:n]

        v = b2 * v + (1.0 - b2) * g * g

            best_h = h_new.copy()
            stalled = 0
        else:

            stalled = 0

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def compute_gradient_smooth(h, n, beta):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    ov_shift = ov - np.max(ov)
    w = np.exp(beta * ov_shift)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros(n_lags)
        w[np.argmax(ov)] = 1.0
    else:
        w /= sw
    g = np.zeros(n)
    for k in range(n_lags):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n - lag] += w[k] * (1.0 - h[lag:n])
            g[lag:n] -= w[k] * h[:n - lag]
        else:
            L = -lag
            g[L:n] += w[k] * (1.0 - h[:n - L])
            g[:n - L] -= w[k] * h[L:n]
    return g * dx, ov

    v = np.zeros(n)
    t = 0
    lr = lr0
    b1, b2, eps = 0.82, 0.999, 1e-8
    stalled = 0
    while time.time() < deadline:
        t += 1
        beta = max(4.0, beta0 - t * 0.003)

def optimize_max_lag(h, n, deadline, rng, lr0=0.06):

    best_h = h.copy()
    best_c5 = get_c5(h, n)
    lr = lr0
    stalled = 0
    while time.time() < deadline:
