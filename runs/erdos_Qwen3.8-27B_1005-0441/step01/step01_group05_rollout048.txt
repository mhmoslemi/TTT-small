import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _normalize(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if s > 1e-12:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _project_onto_simplex_box(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} using water-filling"""
    target = n / 2.0
    # Binary search for threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fine-tune to get exact sum
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute the difference among non-bound elements
        mask = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def _smooth_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if temp < 1e-10:
        return m
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
    nm = np.sum(mask)
    if nm > 0 and abs(diff) > 1e-12:
        hp[mask] += diff / nm
        hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_val_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if tau < 1e-9:
        return m, None  # no gradient for hard max
    
    exp_v = np.exp((corr - m) / tau)
    p = exp_v / np.sum(exp_v)
    val = m + tau * np.log(np.sum(exp_v))
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        # First term: (1-h[i-s]) for valid i
        i1s = max(0, s)
        i1e = min(n, n + s)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    mask = (hp > 1e-10) & (hp < 1.0 - 1e-10)
    nm = np.sum(mask)
    if nm > 0 and abs(diff) > 1e-12:
        hp[mask] += diff / nm
        hp = np.clip(hp, 0.0, 1.0)
    return hp

def _smooth_val_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if tau < 1e-10:

        if i2s < i2e:

    return v + pw * d * d, g + 2.0 * pw * d
