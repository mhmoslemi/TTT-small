def project_feasible(h, n):

    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:

                    grad[j] += (1.0 - h[j-s]) * dx
                for j in range(0, n-s):

                    grad[j] += (1.0 - h[j-s_abs]) * dx
                for j in range(0, n-s_abs):
                    grad[j] += (-h[j+s_abs]) * dx

        for _ in range(20):

            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5:
                h = h_trial
                c5 = c5_trial
                last_improve = time.time()
                step *= 1.05
                break
            else:
                step *= 0.7
        
        if step < 1e-8:
            break
    
    return h, c5
