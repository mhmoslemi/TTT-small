def _smooth_val(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    dx = 2.0 / n
    mc = corr.max()
    b = _global_beta
    if b < 1e-3:
        return float(mc * dx)
    terms = np.exp(b * (corr - mc))
    return float((mc + np.log(terms.sum()) / b) * dx)

    n = len(h)

                g[lo1:hi1] += (1.0 - h[lo1 + d:hi1 + d])
            lo2 = max(0, d)
            hi2 = min(n, d + n)
            if lo2 < hi2:
                g[lo2:hi2] -= h[lo2 - d:hi2 - d]
        g = g - g.mean()
        h -= lr * g * dx
        h = _project(h)
        c = compute_c5(h)
        if c < best:
            best = c
        lr *= 0.9985
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 70
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0

    n = len(h_init)
    best_h = _project(h_init.copy())
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with adaptive beta
    betas = [5, 12, 25, 50, 100, 200]
    for i, b in enumerate(betas):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 35:
            break
        mi = 250 if i < 3 else 350
        h_new = _slsqp_phase(best_h, b, maxiter=mi)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Nelder-Mead polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 50:
        h_nm = _nm_polish(best_h, maxiter=min(800, int(t_rem * 2)))
        c_nm = compute_c5(h_nm)
        if c_nm < best_c5:
            best_c5 = c_nm
            best_h = h_nm

    # Phase 3: Pair Brent optimization (structured sweeps)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 40:
        h_p, c_p = _pair_optimize(best_h, t_budget=min(100, t_rem - 10), n_sweeps=500)
        if c_p < best_c5:
            best_c5 = c_p
            best_h = h_p

    # Phase 4: Coordinate descent
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        h_cd, c_cd = _coord_descend(best_h, n_passes=6, t_budget=min(50, t_rem - 5))
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 5: Subgradient descent (multi-lag)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 25:
        h_sg, c_sg = _subgrad_descent(best_h, n_iters=1000, t_budget=min(25, t_rem - 5))
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg

    # Phase 6: More pair optimization after subgradient
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        h_p2, c_p2 = _pair_optimize(best_h, t_budget=min(60, t_rem - 5), n_sweeps=300)
        if c_p2 < best_c5:
            best_c5 = c_p2
            best_h = h_p2

    # Phase 7: Perturbed restart
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 100:
        h_pert = _project(best_h + 0.06 * rng.randn(n))
        h_pert = _slsqp_phase(h_pert, 40, maxiter=200)
        c_pert = compute_c5(h_pert)
        if c_pert < best_c5:
            best_c5 = c_pert
            best_h = h_pert
        h_pert2 = _nm_polish(best_h, maxiter=400)
        c_pert2 = compute_c5(h_pert2)
        if c_pert2 < best_c5:
            best_c5 = c_pert2
            best_h = h_pert2
        h_pert3, c_pert3 = _pair_optimize(best_h, t_budget=30, n_sweeps=200)
        if c_pert3 < best_c5:
            best_c5 = c_pert3
            best_h = h_pert3

    # Phase 8: Second restart with different perturbation
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 80:
        h_pert = _project(best_h + 0.1 * rng.randn(n))
        h_pert = _slsqp_phase(h_pert, 50, maxiter=200)
        h_pert = _slsqp_phase(h_pert, 150, maxiter=200)
        c_pert = compute_c5(h_pert)
        if c_pert < best_c5:
            best_c5 = c_pert
            best_h = h_pert
        h_pert2 = _nm_polish(best_h, maxiter=400)
        c_pert2 = compute_c5(h_pert2)
        if c_pert2 < best_c5:
            best_c5 = c_pert2
            best_h = h_pert2

    # Phase 9: Final intensive pair + coord + NM
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 60:
        h_fin, c_fin = _pair_optimize(best_h, t_budget=min(50, t_rem // 2), n_sweeps=400)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin
        h_cd2, c_cd2 = _coord_descend(best_h, n_passes=8, t_budget=min(20, t_rem // 3))
        if c_cd2 < best_c5:
            best_c5 = c_cd2
            best_h = h_cd2
        t_rem = budget_s - (time.time() - t_start)
        if t_rem > 15:
            h_nm2 = _nm_polish(best_h, maxiter=min(500, int(t_rem * 2)))
            c_nm2 = compute_c5(h_nm2)
            if c_nm2 < best_c5:
                best_c5 = c_nm2
                best_h = h_nm2

    # Final projection
    best_h = _project(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
