def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""

    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:

        h += d / n
        np.clip(h, 0.0, 1.0, out=h)

from scipy.optimize import minimize, linprog

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += resid / cnt
    np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def lag_grad(h, n, ell):
    g = np.zeros(n)
    if ell >= 0:
        g[:n - ell] += 1.0 - h[ell:]
        g[ell:] -= h[:n - ell]
    else:
        m = -ell
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    return g

def softmax_grad(h, n, top_m, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = corr.max()
    idx = np.argsort(corr)[-top_m:]
    w = np.exp((corr[idx] - mv) * alpha)
    w /= w.sum()
    grad = np.zeros(n)
    for k, wk in zip(idx, w):
        grad += wk * lag_grad(h, n, k - (n - 1))
    return grad, mv, corr
