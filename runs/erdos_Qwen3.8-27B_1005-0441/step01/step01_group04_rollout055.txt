        cmin = np.min(corr)
        spread = cmax - cmin
        if spread < 1e-12:
            break
        frac = 0.2 + 0.1 * (it / max(1, n_iters))
        threshold = cmax - frac * spread
        active = np.where(corr >= threshold)[0] - (n - 1)
        grad = _subgrad(h_cur, active)
        alpha = alpha0 * (0.95 ** (it / 200))
        h_new = h_cur - alpha * grad
        h_new = _project_box_sum(h_new, n / 2.0)
        c_new = _eval_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_local = h_new.copy()
        h_cur = h_new
    return best_local, best_c

def _coord_search(h, n, sweeps=3):
    best = h.copy()
    best_c = _eval_c5(best)
    deltas = [0.002, -0.002, 0.008, -0.008, 0.02, -0.02, 0.05, -0.05]
    for _ in range(sweeps):
        improved = False
        for i in range(n):
            for d in deltas:
                h_try = best.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                diff = n / 2.0 - np.sum(h_try)
                if abs(diff) > 0.05:
                    continue
                h_try = _fix_sum(h_try, n)
                c = _eval_c5(h_try)
                if c < best_c - 1e-15:
                    best_c = c
                    best = h_try
                    improved = True
                    break
        if not improved:
            break
    return best, best_c

def _random_h(n, rng):
    h = rng.random(n)
    return _fix_sum(h, n)

def _block_h(n, p_hi=0.7, p_lo=0.1, period=3):
    pattern = [p_hi] * (period - 1) + [p_lo]
    h = np.array([pattern[i % period] for i in range(n)])
    return _fix_sum(h, n)

def _sin_h(n, freq=1.0, amp=0.3):
    x = np.linspace(0, 2 * np.pi * freq, n, endpoint=False)
    h = 0.5 + amp * np.sin(x)
    return _fix_sum(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None:
        h_init = _fix_sum(h_init, len(h_init))
        c5 = _eval_c5(h_init)
        best_c5 = c5
        best_h = h_init.copy()

    n_main = len(best_h) if best_h is not None else 80

    # Phase 1: PSG to explore neighborhood
    if best_h is not None and remaining() > 300:
        h_psg, c_psg = _psg_loop(best_h, 2000, 0.015)
        if c_psg < best_c5:
            best_c5 = c_psg
            best_h = h_psg

    # Phase 2: SLSQP softmax with increasing alpha
    if best_h is not None and remaining() > 250:
        for alpha in [80, 150, 300, 600, 1200]:
            if remaining() < 35:
                break
            h_opt = _slsqp_sm(best_h, alpha, 600)
            h_opt = _fix_sum(h_opt, n_main)
            c = _eval_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt

    # Phase 3: Active-set minimax SLSQP
    if best_h is not None and remaining() > 150:
        for top_k in [6, 12, 25]:
            if remaining() < 30:
                break
            corr = _corr_vals(best_h)
            top_idx = np.argsort(corr)[-top_k:]
            h_opt = _minimax_active(best_h, top_idx, 500)
            h_opt = _fix_sum(h_opt, n_main)
            c = _eval_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt

    # Phase 4: Refine with PSG from improved point
    if best_h is not None and remaining() > 100:
        h_psg2, c_psg2 = _psg_loop(best_h, 1500, 0.008)
        if c_psg2 < best_c5:
            best_c5 = c_psg2
            best_h = h_psg2

    # Phase 5: More SLSQP rounds
    if best_h is not None and remaining() > 80:
        for alpha in [400, 800, 2000]:
            if remaining() < 25:
                break
            h_opt = _slsqp_sm(best_h, alpha, 400)
            h_opt = _fix_sum(h_opt, n_main)
            c = _eval_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt

    # Phase 6: Active set again
    if best_h is not None and remaining() > 50:
        corr = _corr_vals(best_h)
        top_idx = np.argsort(corr)[-10:]
        h_opt = _minimax_active(best_h, top_idx, 400)
        h_opt = _fix_sum(h_opt, n_main)
        c = _eval_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt

    # Phase 7: Coordinate search polish
    if best_h is not None and remaining() > 30:
        best_h, best_c5 = _coord_search(best_h, n_main, sweeps=4)

    # Phase 8: Try other n values
    for n_try in [47, 50, 60, 100]:
        if remaining() < 25:
            break
        starts = [_random_h(n_try, rng), _block_h(n_try), _sin_h(n_try, 1.0), _sin_h(n_try, 2.0)]
        for h0 in starts:
            if remaining() < 15:
                break
            h_opt = _slsqp_sm(h0, 200, 250)
            h_opt = _fix_sum(h_opt, n_try)
            c = _eval_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt
                n_main = n_try
            corr = _corr_vals(h_opt)
            top_idx = np.argsort(corr)[-8:]
            h_opt2 = _minimax_active(h_opt, top_idx, 200)
            h_opt2 = _fix_sum(h_opt2, n_try)
            c2 = _eval_c5(h_opt2)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h_opt2
                n_main = n_try

    # Phase 9: Final polish
    if best_h is not None and remaining() > 20:
        h_final = _slsqp_sm(best_h, 1000, 400)
        h_final = _fix_sum(h_final, len(h_final))
        c = _eval_c5(h_final)
        if c < best_c5:
            best_c5 = c
            best_h = h_final
        best_h, best_c5 = _coord_search(best_h, len(best_h), sweeps=2)

    if best_h is None:
        best_h = np.full(80, 0.5)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
