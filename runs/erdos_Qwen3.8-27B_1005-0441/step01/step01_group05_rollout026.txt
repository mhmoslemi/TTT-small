def _upsample(h, n_new):

    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_p) > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        free = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            h_p[free] += diff / nf
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def compute_subgrad(h, n, active_lags):
    grad = np.zeros(n)
    for k_idx in active_lags:
        k = k_idx - (n - 1)
        if k >= 0:
            grad[k:] += 1.0 - h[:n - k]
            grad[:n - k] -= h[k:]
        else:
            kp = -k
            grad[:n - kp] += 1.0 - h[kp:]
            grad[kp:] -= h[:n - kp]
    grad /= len(active_lags)
    return grad

def subgrad_descent(h_init, n, lr0, max_iters, t0, time_limit):

    best_h = h.copy()
    best_val = compute_c5(h)
    lr = lr0
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mv = corr.max()
        active = np.where(corr >= mv - 1e-4)[0]
        grad = compute_subgrad(h, n, active)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new)

            best_val = val
            stall = 0
            lr = min(lr * 1.03, 0.05)
        else:
            stall += 1
            lr *= 0.92
            if stall > 200:
                break
        h = h_new
    return best_h, best_val

    h = h_init.copy()

            break

            c = np.correlate(x, 1.0 - x, mode="full")
            m = c.max()
            return (m + temp * np.log(np.sum(np.exp((c - m) / temp)))) * dx
        def eq_con(x):
            return np.sum(x) - n / 2.0
        bounds = [(0.0, 1.0)] * n
