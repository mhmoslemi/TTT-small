import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smoothed_max_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    l = len(corr)
    m = np.max(corr)
    shifted = (corr - m) / temp
    ms = np.max(shifted)
    exps = np.exp(shifted - ms)
    s = np.sum(exps)
    w = exps / s
    s_max = m + temp * (ms + np.log(s))
    obj = s_max * dx
    grad = np.zeros(n)
    shifts = np.arange(l) - (n - 1)
    for i in range(l):
        wi = w[i]
        if wi < 1e-13:
            continue
        wi_dx = wi * dx
        s_i = int(shifts[i])
        if s_i >= 0:
            grad[s_i:] += wi_dx * (1.0 - h[:n - s_i])
        else:
            grad[:n + s_i] += wi_dx * (1.0 - h[-s_i:])
        if s_i >= 0:
            grad[:n - s_i] -= wi_dx * h[s_i:]
        else:
            grad[-s_i:] -= wi_dx * h[:n + s_i]
    return obj, grad

def project_feasible(h, n):
    target = float(n) / 2.0
    lo, hi = h.min() - 1.0, h.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def pgd_optimize(h_init, temps, time_budget, t0):
    n = len(h_init)
    h = h_init.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    for temp in temps:
        if time.time() - t0 > time_budget:
            break
        obj, grad = smoothed_max_grad(h, temp)
        step = 1.0
        momentum = np.zeros(n)
        mu = 0.9
        iters = min(300, max(50, int(1200 / len(temps))))
        for it in range(iters):
            if time.time() - t0 > time_budget:
                break
            momentum = mu * momentum - step * grad
            grad_n = np.linalg.norm(momentum)
            if grad_n < 1e-12:
                break
            direction = momentum / grad_n
            alpha = step
            found = False
            for _ in range(40):
                h_try = project_feasible(h + alpha * direction, n)
                obj_try, _ = smoothed_max_grad(h_try, temp)
                if obj_try <= obj + 1e-4 * alpha * np.dot(grad, direction) - 1e-12:
                    found = True
                    break
                alpha *= 0.5
            if not found:
                break
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
            h = h_try
            obj, grad = smoothed_max_grad(h, temp)
            step = min(step * 1.1, 10.0)
    return best_h, best_c5

def binary_local_search(h_start, n, t0, budget):
    h_bin = (h_start >= 0.5).astype(float)
    ones = np.sum(h_bin)
    target = n // 2
    if ones > target:
        idx = np.where(h_bin == 1.0)[0]
        h_bin[np.sort(idx)[-int(ones - target):]] = 0.0
    elif ones < target:
        idx = np.where(h_bin == 0.0)[0]
        h_bin[np.sort(idx)[:int(target - ones)]] = 1.0
    best_h, best_c5 = h_bin.copy(), compute_c5(h_bin)
    n_tries = 200
    for _ in range(n_tries):
        if time.time() - t0 > budget:
            break
        for trial in range(5):
            i = np.random.randint(n)
            j_arr = np.random.randint(n)
            if h_bin[i] == 1.0 and h_bin[j_arr] == 0.0:
                h试行 = h_bin.copy()
                h试行[i] = 0.0
                h试行[j_arr] = 1.0
                c5_t = compute_c5(h试行)
                if c5_t < best_c5 - 1e-12:
                    best_h, best_c5 = h试行.copy(), c5_t
                    h_bin = h试行
                    break
    return best_h, best_c5

def perturb_h(h, scale, seed):
    rng = np.random.RandomState(seed)
    h_p = h + scale * rng.randn(len(h))
    return project_feasible(h_p, len(h))

def downsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(160, 0.5)
    n = len(best_h)
    best_c5 = compute_c5(best_h)

    temps_main = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]

    # Phase 1: PGD from start
    if budget_s - (time.time() - t0) > 60:
        h1, c1 = pgd_optimize(best_h.copy(), temps_main, time.time() - t0 + min(200, budget_s - 80), t0)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Binary local search + relaxation
    if budget_s - (time.time() - t0) > 50:
        h_bin, c5_bin = binary_local_search(best_h, n, t0, time.time() - t0 + 60)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
        h2, c2 = pgd_optimize(h_bin.copy(), [0.05, 0.02, 0.005, 0.001, 0.0003],
                              time.time() - t0 + 80, t0)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Perturbed restarts
    for r in range(8):
        if budget_s - (time.time() - t0) < 30:
            break
        scale = 0.05 + 0.03 * r
        h_p = perturb_h(best_h, scale, seed + 500 + r)
        temps_r = [0.03, 0.01, 0.003, 0.001, 0.0003]
        h3, c3 = pgd_optimize(h_p, temps_r, time.time() - t0 + min(40, budget_s - 20), t0)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Try smaller n (downsample)
    for n_try in [80, 100, 120]:
        if budget_s - (time.time() - t0) < 40:
            break
        if n_try == n:
            continue
        h_ds = downsample(best_h, n_try)
        temps_ds = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0003]
        h4, c4 = pgd_optimize(h_ds, temps_ds, time.time() - t0 + min(50, budget_s - 20), t0)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: Final polish with very low temperature
    if budget_s - (time.time() - t0) > 20:
        temps_final = [0.002, 0.0008, 0.0003, 0.0001, 0.00005]
        h5, c5 = pgd_optimize(best_h.copy(), temps_final, time.time() - t0 + max(15, budget_s - 5), t0)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
