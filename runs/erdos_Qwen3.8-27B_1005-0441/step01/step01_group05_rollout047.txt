    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_obj(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = (corr - m) / tau
    sh -= np.max(sh)
    e = np.exp(sh)
    s = np.sum(e)

        excess = int(np.ceil(s - target))
        idx = np.where(h == 1.0)[0]
        remove = rng.choice(idx, min(excess, len(idx)), replace=False)
        h[remove] = 0.0
    elif s < target:
        deficit = int(np.ceil(target - s))
        idx = np.where(h == 0.0)[0]
        if len(idx) > 0:
            add = rng.choice(idx, min(deficit, len(idx)), replace=False)
            h[add] = 1.0
    return h

def _sinusoid_init(n, freq, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.45 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    h = h * (float(n) / 2.0) / max(s, 1e-12)
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(160, 0.5)
    n = len(h0)
    h0 = np.clip(h0, 0.0, 1.0)
    s0 = np.sum(h0)
    h0 = h0 * (float(n) / 2.0) / max(s0, 1e-12)
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Main optimization at initial n with fine schedule
    rem = budget_s - (time.time() - t0)
    if rem > 180:
        tau_list = [0.04, 0.015, 0.006, 0.002, 0.0008, 0.0003, 0.0001, 0.00004]
        h_opt, c5_opt = _optimize(best_h.copy(), n, tau_list, 5, t0, min(280, rem - 50))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Alternative initializations
    rem = budget_s - (time.time() - t0)
    inits = [_block_init(n, 4, rng), _block_init(n, 8, rng),
             _sinusoid_init(n, 3, rng), _perturb_h(best_h, 0.12, rng)]
    for i, h_init in enumerate(inits):
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        tau_list = [0.02, 0.006, 0.0015, 0.0004, 0.0001]
        h_opt, c5_opt = _optimize(h_init.copy(), n, tau_list, 4, t0, min(55, rem - 20))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Upsample to higher resolution
    rem = budget_s - (time.time() - t0)
    n2 = 200
    if rem > 100 and len(best_h) != n2:
        h_up = _upsample(best_h, n2)
        tau_list = [0.03, 0.01, 0.004, 0.001, 0.0003, 0.0001, 0.00004]
        h_opt, c5_opt = _optimize(h_up, n2, tau_list, 5, t0, min(160, rem - 40))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Small-perturbation restarts at current best n
    n_cur = len(best_h)
    restart = 0
    while True:
        rem = budget_s - (time.time() - t0)
        if rem < 25 or restart >= 25:
            break
        scale = 0.015 + 0.025 * (restart % 6) / 5.0
        h_pert = _perturb_h(best_h, scale, rng)
        tau_list = [0.004, 0.001, 0.0003, 0.0001, 0.00003]
        max_t = min(30, rem - 10)
        if max_t < 12:
            break
        h_opt, c5_opt = _optimize(h_pert, n_cur, tau_list, 4, t0, max_t)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        restart += 1

    # Phase 5: Final ultra-fine polish
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        n_final = len(best_h)
        tau_list = [0.00008, 0.00003, 0.00001]
        h_opt, c5_opt = _optimize(best_h.copy(), n_final, tau_list, 6, t0, max(10, rem - 3))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final feasibility check
    best_h = np.clip(best_h, 0.0, 1.0)
    s_final = np.sum(best_h)
    target_final = float(len(best_h)) / 2.0
    if abs(s_final - target_final) > 1e-6:
        best_h = best_h * target_final / s_final
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
