import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_binary(h, n):
    """Project to binary with exactly n//2 ones."""
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    cnt = int(h_bin.sum())
    if cnt > target:
        ones = np.where(h_bin == 1)[0]
        # Flip lowest-valued ones
        flip_idx = ones[np.argsort(h[ones])[:cnt - target]]
        h_bin[flip_idx] = 0
    elif cnt < target:
        zeros = np.where(h_bin == 0)[0]
        flip_idx = zeros[np.argsort(-h[zeros])[:target - cnt]]
        h_bin[flip_idx] = 1
    return h_bin

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def soft_objective(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = corr.max()
    shifted = beta * (corr - m)
    shifted -= shifted.max()
    return (m + np.log(np.sum(np.exp(shifted))) / beta)

def binary_hill_climb(h, n, max_sweeps=500, time_limit=30, seed=42):
    """Local search on binary h with single flips."""
    rng = np.random.RandomState(seed)
    target = n // 2
    c5_val, _ = compute_c5(h)
    t_start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in rng.permutation(n):
            old_val = h[i]
            h[i] = 1.0 - h[i]
            new_c5, _ = compute_c5(h)
            if new_c5 < c5_val - 1e-14:
                c5_val = new_c5
                improved = True
            else:
                h[i] = old_val
        # Re-project to maintain sum
        h = project_binary(h, n)
        c5_val, _ = compute_c5(h)
        if not improved:
            break
    return h, c5_val

def binary_annealing(h, n, t_start, t_end, n_steps=5000, seed=42):
    """Simulated annealing on binary h."""
    rng = np.random.RandomState(seed)
    target = n // 2
    c5_val, _ = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_val
    
    for step in range(n_steps):
        if time.time() > t_end:
            break
        frac = step / n_steps
        temp = 0.05 * (1 - frac) ** 2
        
        # Pick a random flip
        i = rng.randint(n)
        h[i] = 1.0 - h[i]
        new_c5, _ = compute_c5(h)
        delta = new_c5 - c5_val
        
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            c5_val = new_c5
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        else:
            h[i] = 1.0 - h[i]
        
        # Periodic re-projection
        if step % 100 == 0:
            h = project_binary(h, n)
            c5_val, _ = compute_c5(h)
    
    h = project_binary(h, n)
    c5_val, _ = compute_c5(h)
    if c5_val < best_c5:

    return best_h, best_c5

def slsqp_optimize(h, n, t0, budget_s, max_frac=0.5):
    """SLSQP optimization with soft objective."""
    deadline = t0 + budget_s * max_frac
    
    def obj(x):
        return soft_objective(x, n, 200)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    
    betas = [5, 20, 100, 500, 2000, 10000]
    for beta in betas:
        if time.time() > deadline:
            break
        def obj_beta(x, b=beta):
            return soft_objective(x, n, b)
        try:
            res = minimize(obj_beta, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):

                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def flip_effect(h, p, n):
    """Compute change in full correlation when h[p] flips by delta=1-2*h[p]."""
    delta = 1.0 - 2.0 * h[p]  # +1 if flipping 0→1, -1 if 1→0
    lag_offset = n - 1
    effect = np.zeros(2 * n - 1)
    
    for i in range(2 * n - 1):
        lag = i - lag_offset
        j1 = p - lag  # h[p] appears in h[j1 + lag]
        j2 = p  # h[p] appears in (1-h[j2])
        
        if 0 <= j1 < n:
            effect[i] += delta * (1.0 - h[j1])
        if 0 <= p + lag < n:
            effect[i] -= delta * h[p + lag]
    
    return effect

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_binary(h, n):
    target = n // 2
    h_b = (h > 0.5).astype(float)
    cnt = int(h_b.sum())
    if cnt > target:
        ones = np.where(h_b == 1)[0]
        h_b[ones[np.argsort(h[ones])[:cnt - target]]] = 0
    elif cnt < target:
        zeros = np.where(h_b == 0)[0]
        h_b[zeros[np.argsort(-h[zeros])[:target - cnt]]] = 1
    return h_b

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            m = h < 1.0 - 1e-10
            c = m.sum()
            if c == 0:
                break
            h[m] = np.minimum(h[m] + diff / c, 1.0)
        else:
            m = h > 1e-10
            c = m.sum()
            if c == 0:
                break
            h[m] = np.maximum(h[m] + diff / c, 0.0)
    return np.clip(h, 0.0, 1.0)

def precompute_effects(h, n):
    effects = np.zeros((n, 2 * n - 1))
    lags = np.arange(-(n - 1), n)
    for p in range(n):
        d = 1.0 - 2.0 * h[p]
        j1 = p - lags
        j2 = p + lags
        m1 = (j1 >= 0) & (j1 < n)
        m2 = (j2 >= 0) & (j2 < n)
        effects[p, m1] += d * (1.0 - h[j1[m1]])
        effects[p, m2] -= d * h[j2[m2]]
    return effects

def one_flip_steepest(h, n, max_iters=2000, time_limit=40, seed=42):
    rng = np.random.RandomState(seed)
    c5_v = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    effects = precompute_effects(h, n)
    t0 = time.time()
    stuck = 0
    for _ in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        best_c5 = c5_v - 1e-14
        best_p = -1
        for p in range(n):
            nc = corr + effects[p]
            mv = nc.max() * dx
            if mv < best_c5:
                best_c5 = mv
                best_p = p
        if best_p >= 0:
            h[best_p] = 1.0 - h[best_p]
            c5_v = compute_c5(h)
            corr = np.correlate(h, 1.0 - h, mode="full")
            effects = precompute_effects(h, n)
            stuck = 0
        else:
            stuck += 1
            if stuck > 15:
                break
            for _ in range(8):
                p = rng.randint(n)
                h[p] = 1.0 - h[p]
            h = project_binary(h, n)
            c5_v = compute_c5(h)
            corr = np.correlate(h, 1.0 - h, mode="full")
            effects = precompute_effects(h, n)
    return h, c5_v

def two_swap_search(h, n, max_sweeps=300, time_limit=50, seed=42):
    rng = np.random.RandomState(seed)
    c5_v = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    effects = precompute_effects(h, n)
    t0 = time.time()
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for p in ones:
            base_p = corr + effects[p]
            cands = base_p[None, :] + effects[zeros, :]
            # Add cross terms: at lag q-p add 1, at lag p-q add -1
            qps = zeros - p + (n - 1)
            pqps = p - zeros + (n - 1)
            valid_q = (qps >= 0) & (qps < 2 * n - 1)
            valid_p = (pqps >= 0) & (pqps < 2 * n - 1)
            cands[valid_q, qps[valid_q]] += 1.0
            cands[valid_p, pqps[valid_p]] -= 1.0
            max_vals = cands.max(axis=1) * dx
            bi = int(np.argmin(max_vals))
            if max_vals[bi] < c5_v - 1e-14:
                q = zeros[bi]
                h[p] = 0.0
                h[q] = 1.0
                c5_v = compute_c5(h)
                corr = np.correlate(h, 1.0 - h, mode="full")
                effects = precompute_effects(h, n)
                improved = True
                break
        if not improved:
            break
    return h, c5_v

def anneal_binary(h, n, time_limit=60, seed=42, n_steps=20000):
    rng = np.random.RandomState(seed)
    c5_v = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_v
    t0 = time.time()
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        frac = step / n_steps
        temp = 0.03 * (1.0 - frac) ** 2
        i = rng.randint(n)
        h[i] = 1.0 - h[i]
        nc5 = compute_c5(h)
        d = nc5 - c5_v
        if d < 0 or rng.random() < np.exp(-d / max(temp, 1e-12)):
            c5_v = nc5
            if c5_v < best_c5:
                best_c5 = c5_v
                best_h = h.copy()
        else:
            h[i] = 1.0 - h[i]
        if step % 200 == 0:
            h = project_binary(h, n)
            c5_v = compute_c5(h)
    h = project_binary(h, n)
    cv = compute_c5(h)
    if cv < best_c5:
        best_c5, best_h = cv, h.copy()
    return best_h, best_c5

def slsqp_refine(h, n, deadline, max_iter=150):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        sh = 50.0 * (corr - m)
        sh -= sh.max()
        return (m + np.log(np.sum(np.exp(sh))) / 50.0) * (2.0 / n)
    def cf(x):
        return np.sum(x) - n / 2.0
    def cj(x):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def random_binary_init(n, rng):
    h = (rng.random(n) > 0.5).astype(float)
    return project_binary(h, n)

def make_structured_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    best_h, best_c5 = None, np.inf
    for freq in [1, 2, 3, 5, 8]:
        for ph in [0, 0.3, 0.7, 1.2]:
            h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2 + ph)
            h = project_feasible(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    for _ in range(5):
        h = random_binary_init(n, rng)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    if c5_b < best_c5:

    if time.time() < deadline - 200:

        if c5_s < best_c5:

    # Phase 2: 2-swap local search
    if time.time() < deadline - 150:
        h_sw, c5_sw = two_swap_search(best_h, n, max_sweeps=500, time_limit=60, seed=seed + 10)
        if c5_sw < best_c5:
            best_c5, best_h = c5_sw, h_sw

    # Phase 3: Another 1-flip pass
    if time.time() < deadline - 120:
        h_s2, c5_s2 = one_flip_steepest(best_h, n, max_iters=5000, time_limit=60, seed=seed + 20)

            if c5_a2 < best_c5:
