import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_obj_pen(x, alpha, lam):
    n = len(x)
    base = smooth_obj(x, alpha)
    pen = lam * (np.sum(x) - n * 0.5) ** 2
    return base + pen

def sum_eq(x):
    return np.sum(x) - len(x) * 0.5

def sum_eq_jac(x):
    return np.ones(len(x))

def subgradient_refine(h, n, n_steps=300, lr0=0.02, decay=0.96):
    hsg = h.copy()
    lr = lr0
    for step in range(n_steps):
        corr = np.correlate(hsg, 1.0 - hsg, mode='full')
        m = np.max(corr)
        alpha_sg = 100.0
        weights = np.exp(alpha_sg * (corr - m))
        weights /= np.sum(weights)
        top_k = np.argsort(weights)[-5:]
        grad = np.zeros(n)
        for k_idx in top_k:
            if weights[k_idx] < 1e-8:
                continue
            shift = k_idx - (n - 1)
            if shift >= 0:
                for i in range(n - shift):
                    grad[i + shift] += (1.0 - hsg[i])
                    grad[i] += (-hsg[i + shift])
            else:
                for i in range(-shift, n):
                    grad[i + shift] += (1.0 - hsg[i])
                    grad[i] += (-hsg[i + shift])
            grad *= weights[k_idx]
            grad = np.zeros(n)
            for i in range(n - shift) if shift >= 0 else range(-shift, n):
                grad[i + shift] += (1.0 - hsg[i])
                grad[i] += (-hsg[i + shift])
            grad *= weights[k_idx]
        hsg = hsg - lr * grad
        hsg = project_feasible(hsg, n)
        lr *= decay
    return hsg

    h[-block:] = 1.0

    h = np.zeros(n)

    h[:b] = 1.0
    h[n // 2 - b:n // 2 + b] = 1.0
    h[-b:] = 1.0

    w = max(1, int(seg * width_frac))

        start = t * seg
        h[start:start + w] = 1.0
