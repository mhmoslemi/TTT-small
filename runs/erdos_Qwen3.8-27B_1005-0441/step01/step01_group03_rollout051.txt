def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def subgradient(h, n, n_lags=3):
    """Subgradient using top n_lags overlaps"""
    dx = 2.0 / n
    ov = get_overlaps(h, n)
    idx = np.argsort(ov)[-n_lags:]
    g = np.zeros(n)

                g[j] += 1.0 - h[j + k]
            if 0 <= j - k < n:
                g[j] -= h[j - k]
    return g * dx / len(idx)

    """Change in all overlaps when h[i]:1->0, h[j]:0->1 (binary swap)"""

            val += 1.0 - h[j + k]
        # -h[j-k]
        if 0 <= j - k < n:
            val -= h[j - k]

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = float(np.sum(mask))
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_c5(h, n):
    return float(np.max(get_overlaps(h, n)))

def subgradient(h, n, n_top=4):
    dx = 2.0 / n
    ov = get_overlaps(h, n)
    idx = np.argsort(ov)[-n_top:]
    g = np.zeros(n)
    for ii in idx:
        k = int(ii) - (n - 1)
        j = np.arange(n)
        pk = j + k
        mk = j - k
        mask_p = (pk >= 0) & (pk < n)
        mask_m = (mk >= 0) & (mk < n)
        g[mask_p] += (1.0 - h[pk[mask_p]])
        g[mask_m] -= h[mk[mask_m]]
    return g * dx / len(idx)

def make_binary(h, n, rng):
    h_b = (h > 0.5).astype(float)
    s = np.sum(h_b)
    target = n // 2
    if s > target:
        ones = np.where(h_b == 1)[0]
        rng.shuffle(ones)

            h_b[ones[idx]] = 0
    elif s < target:
        zeros = np.where(h_b == 0)[0]
        rng.shuffle(zeros)
        for idx in range(int(target - s)):
            h_b[zeros[idx]] = 1
    return h_b

def swap_delta(h, i, j, n):
    dx = 2.0 / n
    k_range = np.arange(-(n - 1), n, dtype=int)
    delta = np.zeros(2 * n - 1)
    m1 = (i - k_range >= 0) & (i - k_range < n)
    delta[m1] += h[i - k_range[m1]]
    m2 = (i + k_range >= 0) & (i + k_range < n)
    delta[m2] += h[i + k_range[m2]]
    m3 = (j - k_range >= 0) & (j - k_range < n)
    delta[m3] -= h[j - k_range[m3]]
    m4 = (j + k_range >= 0) & (j + k_range < n)
    delta[m4] -= h[j + k_range[m4]]
    return delta * dx

def local_search_bin(h, n, deadline, rng):
