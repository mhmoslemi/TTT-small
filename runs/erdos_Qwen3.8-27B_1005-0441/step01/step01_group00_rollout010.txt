    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g * dx

def _proj(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _eq_fun(x, n):
    return x.sum() - n * 0.5

def _eq_jac(x, n):
    return np.ones(n)

def _ls_slsqp(h0, n, eps, maxit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_smoobj, h0.copy(), args=(eps,),
                       jac=_smograd, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-14})
        return _proj(res.x, n)
    except Exception:
        return h0.copy()

def _upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = 2.0 * np.arange(n_old) / n_old
    x_new = 2.0 * np.arange(n_new) / n_new
    h_up = np.interp(x_new, x_old, h_old)
    return _proj(h_up, n_new)

def _slug(h, n, lr, K):
    c = _corr(h)
    top_k = np.argsort(c)[-K:]
    g = np.zeros(n)
    for ks in top_k:
        m = ks - (n - 1)
        if m >= 0:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
        else:
            p = -m
            g[:n - p] += (1.0 - h[p:])
            g[p:] -= h[:n - p]
    g *= (2.0 / n) / K
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = np.inf

    def ok(frac):
        return time.time() - t0 < budget_s * frac

    # Build start list
    h_starts = []
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        h0 = _proj(h0, len(h0))
        h_starts.append(h0)
    except Exception:
        pass

    for n in [70, 100]:
        for a in [0.35, 0.5]:
            for freq in [2, 3]:
                hh = 0.5 + a * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
                h_starts.append(_proj(hh, n))

    # Phase 1: Multi-eps SLSQP from each start
    eps_schedule = [(0.4, 150), (0.15, 250), (0.06, 300), (0.025, 350), (0.01, 400)]
    for h0 in h_starts:
        if not ok(0.60):
            break
        n = len(h0)
        h_cur = h0.copy()
        for eps, miter in eps_schedule:
            if not ok(0.62):
                break
            h_new = _ls_slsqp(h_cur, n, eps, miter)
            v = _c5(h_new)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()
            h_cur = h_new

    # Phase 2: Nesterov-accelerated multi-lag subgradient
    if best_h is not None and ok(0.78):
        n = len(best_h)
        h = best_h.copy()
        v_mom = np.zeros(n)
        lr = 0.012
        momentum = 0.85
        for it in range(10000):
            if not ok(0.80):
                break
            K = 6
            g = _slug(h, n, lr, K)
            v_mom = momentum * v_mom - lr * g
            h_new = _proj(h + v_mom, n)
            v = _c5(h_new)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()
                h = h_new
            else:
                v_mom *= 0.5
                lr *= 0.996
                if lr < 1e-8:
                    lr = 0.008
                    v_mom = np.zeros(n)
                    h = _proj(h + 0.005 * rng.randn(n), n)

    # Phase 3: Upscale to higher resolution and re-optimize
    if best_h is not None and ok(0.88):
        n_old = len(best_h)
        for n_new in [100, 140]:
            if not ok(0.90):
                break
            if n_new <= n_old:
                continue
            h_up = _upsample(best_h, n_new)
            h_cur = h_up
            for eps, miter in [(0.08, 250), (0.03, 350), (0.012, 400), (0.005, 500)]:
                if not ok(0.92):
                    break
                h_new = _ls_slsqp(h_cur, n_new, eps, miter)
                v = _c5(h_new)
                if v < best_val:
                    best_val = v
                    best_h = h_new.copy()
                h_cur = h_new

    # Phase 4: Fine subgradient polish on final best
    if best_h is not None and ok(0.95):
        n = len(best_h)
        h = best_h.copy()
        lr = 0.004
        for it in range(8000):
            if not ok(0.97):
                break
            c = _corr(h)
            k_star = int(np.argmax(c))
            m = k_star - (n - 1)
            g = np.zeros(n)
            if m >= 0:
                g[m:] += (1.0 - h[:n - m])
                g[:n - m] -= h[m:]
            else:
                p = -m
                g[:n - p] += (1.0 - h[p:])
                g[p:] -= h[:n - p]
            g *= 2.0 / n
            h_new = _proj(h - lr * g, n)
            v = _c5(h_new)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()
                h = h_new
            else:
                lr *= 0.995
                if lr < 1e-10:
                    break

    # Phase 5: Final tiny-eps SLSQP
    if best_h is not None and ok(0.985):
        n = len(best_h)
        h_fin = _ls_slsqp(best_h, n, 0.003, 600)
        if _c5(h_fin) < best_val:
            best_val = _c5(h_fin)
            best_h = h_fin

    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
