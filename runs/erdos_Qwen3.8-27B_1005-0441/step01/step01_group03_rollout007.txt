import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        hp = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(hp)
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def get_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_gradient(h, p, n):
    g = np.zeros(n)
    for idx in range(2*n - 1):
        k = idx - (n - 1)
        w = p[idx]
        if w < 1e-15:
            continue
        if k >= 0:
            g[:n-k] += w * (1.0 - h[k:])
            g[k:] -= w * h[:n-k]
        else:
            kp = -k
            g[kp:] += w * (1.0 - h[:n-kp])
            g[:n-kp] -= w * h[kp:]
    return g

def optimize_soft(h, n, deadline, rng, tau_start=5.0, tau_end=50.0):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    lr = 0.3
    tau = tau_start
    momentum = np.zeros(n)
    beta = 0.85
    stalled = 0
    it = 0
    total_iters = 0
    
    while time.time() < deadline:
        ov = get_overlaps(h, n)
        max_ov = np.max(ov)
        exp_ov = np.exp(tau * (ov - max_ov))
        p = exp_ov / np.sum(exp_ov)
        
        g = soft_gradient(h, p, n)
        
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = h - lr * dx * momentum
        h_new = project_feasible(h_new, n)
        
        c_new = get_c5(h_new, n)
        if c_new < best_c5 - 1e-14:
            best_c5 = c_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        
        h = h_new
        it += 1
        total_iters += 1
        
        # Animate tau: start smooth, end sharp
        if it < 5000:
            tau = tau_start + (tau_end - tau_start) * (it / 5000.0)
        
        if stalled > 300:
            lr *= 0.7
            if lr < 0.005:
                h = best_h + rng.randn(n) * 0.08
                h = project_feasible(h, n)
                lr = 0.3
                momentum[:] = 0.0
                tau = tau_start
                stalled = 0
            else:
                stalled = 0
    
    return best_h, best_c5

def coordinate_descent_refine(h, n, deadline, rng):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    improved = True
    max_rounds = 200
    
    while improved and time.time() < deadline:
        improved = False
        for _ in range(max_rounds):
            if time.time() > deadline:
                break
            ov = get_overlaps(h, n)
            k_max = int(np.argmax(ov))
            lag = k_max - (n - 1)
            
            # Try perturbing each coordinate in small neighborhood
            for j in range(n):
                for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                    h_try = h.copy()
                    h_try[j] = np.clip(h[j] + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, n)
                    c_try = get_c5(h_try, n)
                    if c_try < best_c5 - 1e-14:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        h = h_try
                        improved = True
                        break
    
    return best_h, best_c5

def make_structured_init(n, mode='block'):
    h = np.zeros(n)
    if mode == 'block':
        h[:n//2] = 1.0
    elif mode == 'uniform':
        h[:] = 0.5
    elif mode == 'sin':
        for i in range(n):
            h[i] = 0.5 + 0.3 * np.cos(2.0 * np.pi * i / n)
    elif mode == 'stripe':
        h[::2] = 1.0
        h[1::2] = 0.0
    h = project_feasible(h, n)
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)

        for li in top_idx:
            lag = int(li) - (n - 1)
            for j in range(n):
                if time.time() > deadline:
                    return best_h, best_c5
                for d in deltas:
                    h_try = h.copy()
                    h_try[j] = np.clip(h[j] + d, 0.0, 1.0)
                    h_try = project_feasible(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c5 - 1e-14:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        h = h_try
                        improved = True
                        break
        for _ in range(200):
            if time.time() > deadline:
                return best_h, best_c5
            j = rng.randint(n)
            d = rng.uniform(-0.08, 0.08)
            h_try = h.copy()
            h_try[j] = np.clip(h[j] + d, 0.0, 1.0)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c5 - 1e-14:
                best_c5 = c_try
                best_h = h_try.copy()
                h = h_try
                improved = True
        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0
    return best_h, best_c5

def make_init(n, mode, rng):
    h = np.zeros(n)
    if mode == 'block':
        h[:n // 2] = 1.0
    elif mode == 'uniform':
        h[:] = 0.5
    elif mode == 'sin':
        for i in range(n):
            h[i] = 0.5 + 0.35 * np.cos(2.0 * np.pi * i / n)
    elif mode == 'stripe':
        h[::2] = 1.0
    elif mode == 'random':
        h[:] = rng.rand(n)
    return project_feasible(h, n)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    # Phase 1: Soft-gradient optimization on primary n (40% time)
    if h0 is not None:
        n0 = len(h0)
        dl1 = t0 + budget_s * 0.40
        bh, bc5 = optimize_soft(h0.copy(), n0, dl1, rng)
        best_h, best_c5, best_n = bh, bc5, n0

    # Phase 2: Coordinate refinement on best so far (30% time)
    if best_h is not None and time.time() < t0 + budget_s * 0.72:
        dl2 = t0 + budget_s * 0.72
        bh, bc5 = coord_refine(best_h.copy(), best_n, dl2, rng)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh

    # Phase 3: Try other n values (20% time)
    n_cands = [40, 42, 44, 48, 50, 54, 56, 60]
    if h0 is not None:
        n_cands = [x for x in n_cands if x != len(h0)]
    t_p3_start = time.time()
    t_p3_end = t0 + budget_s * 0.92
    per_try = max(5.0, (t_p3_end - t_p3_start) / max(1, len(n_cands)))

    for nc in n_cands:
        if time.time() > t_p3_end - 3:
            break
        inits = []
        if h0 is not None:
            inits.append(interp_h(h0, nc))
        inits.append(make_init(nc, 'block', rng))
        inits.append(make_init(nc, 'uniform', rng))
        dl_per = per_try / max(1, len(inits))
        for ini in inits:
            if time.time() > t_p3_end - 3:
                break
            dl = min(time.time() + dl_per * 0.85, t_p3_end - 3)
            bh, bc5 = optimize_soft(ini, nc, dl, rng)
            if bc5 < best_c5:
                best_c5 = bc5
                best_h = bh
                best_n = nc

    # Phase 4: Final refinement (8% time)
    if best_h is not None and time.time() < t0 + budget_s - 8:
        dl4 = t0 + budget_s - 5
        bh, bc5 = coord_refine(best_h.copy(), best_n, dl4, rng)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh

    # Phase 5: Very short final soft opt
    if best_h is not None and time.time() < t0 + budget_s - 3:
        bh, bc5 = optimize_soft(best_h.copy(), best_n, t0 + budget_s - 2, rng,
                                tau_start=30.0, tau_mult=2.0, anneal_it=500)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh

    h_final = project_feasible(best_h, best_n)
    c5_final = compute_c5(h_final, best_n)
    assert abs(np.sum(h_final) * 2.0 / best_n - 1.0) < 1e-8
    return h_final, float(c5_final), best_n
