import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def soft_max(c, beta):
    m = c.max()
    sh = beta * (c - m)
    sh -= sh.max()
    return m + np.log(np.sum(np.exp(sh))) / beta

def slsqp_obj(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return soft_max(c, beta) * dx

def slsqp_eq(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_eq_jac(x):
    return np.ones(len(x))

def to_binary(h, n):
    idx = np.argsort(h)[::-1][:n // 2]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def binary_sa(h_bin, t0, budget_s, frac=0.85, n_rounds=200000, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = corr.max()
    ones_list = [i for i in range(n) if h[i] > 0.5]
    zeros_list = [i for i in range(n) if h[i] < 0.5]
    T_start, T_end = 0.2, 0.00005
    best_h = h.copy()
    best_max = cur_max
    for it in range(n_rounds):
        if time.time() - t0 > budget_s * frac:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_rounds))
        ri = rng.randint(len(ones_list))
        rj = rng.randint(len(zeros_list))
        i, j = ones_list[ri], zeros_list[rj]
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = corr.max()
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_max = new_max
            ones_list[ri] = ones_list[-1]
            ones_list.pop()
            zeros_list[rj] = zeros_list[-1]
            zeros_list.pop()
            ones_list.append(j)
            zeros_list.append(i)
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_max * dx

def greedy_binary(h_bin, t0, budget_s, frac=0.85, max_rounds=200, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    ones_list = [i for i in range(n) if h[i] > 0.5]
    zeros_list = [i for i in range(n) if h[i] < 0.5]
    stuck = 0
    for rnd in range(max_rounds):
        if time.time() - t0 > budget_s * frac:
            break
        if stuck > 30:
            break
        best_delta = 0
        best_i, best_j = -1, -1
        np.random.seed(rng.randint(2**31))
        # Sample subset if too large
        if len(ones_list) * len(zeros_list) > 5000:
            sample_i = rng.choice(len(ones_list), size=60, replace=False)
            sample_j = rng.choice(len(zeros_list), size=60, replace=False)
        else:
            sample_i = range(len(ones_list))
            sample_j = range(len(zeros_list))
        for si in sample_i:
            for sj in sample_j:
                i, j = ones_list[si], zeros_list[sj]
                h[i] = 0.0
                h[j] = 1.0
                nm = np.correlate(h, 1.0 - h, mode="full").max()
                h[i] = 1.0
                h[j] = 0.0
                d = cur_max - nm
                if d > best_delta:
                    best_delta = d
                    best_i, best_j, best_si, best_sj = i, j, si, sj
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            cur_max -= best_delta
            ones_list[best_si] = ones_list[-1]
            ones_list.pop()
            zeros_list[best_sj] = zeros_list[-1]
            zeros_list.pop()
            ones_list.append(best_j)
            zeros_list.append(best_i)
            stuck = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stuck += 1
    return best_h, best_max * dx
