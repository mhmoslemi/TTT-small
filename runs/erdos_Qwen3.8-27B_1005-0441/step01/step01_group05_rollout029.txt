import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_max(vals, temp):
    m = np.max(vals)
    return m + temp * np.log(np.sum(np.exp((vals - m) / temp)))

def _project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_mid = np.clip(h - mid, 0.0, 1.0)
        sm = np.sum(h_mid)
        if sm > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _coord_descend(h, temp, lam, n_sweeps, t0, budget):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    target = float(n) / 2.0
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    sum_h = np.sum(h)
    lags = np.arange(-(n - 1), n)
    corr_len = 2 * n - 1

    for sweep in range(n_sweeps):
        elapsed = time.time() - t0
        if budget - elapsed < 5:
            return h
        for i in range(n):
            idx_p = i + lags
            idx_m = i - lags
            g = np.zeros(corr_len)
            vp = (idx_p >= 0) & (idx_p < n)
            vm = (idx_m >= 0) & (idx_m < n)
            g[vp] += (1.0 - h[idx_p[vp]])
            g[vm] -= h[idx_m[vm]]

            lo_d = -h[i]
            hi_d = 1.0 - h[i]
            if hi_d - lo_d < 1e-14:
                continue
            for _ in range(32):
                m1 = lo_d + (hi_d - lo_d) / 3.0
                m2 = hi_d - (hi_d - lo_d) / 3.0
                nc1 = corr_full + m1 * g
                nc2 = corr_full + m2 * g
                f1 = _smooth_max(nc1, temp) * dx + lam * (sum_h + m1 - target) ** 2
                f2 = _smooth_max(nc2, temp) * dx + lam * (sum_h + m2 - target) ** 2
                if f1 < f2:
                    hi_d = m2
                else:
                    lo_d = m1
            delta = (lo_d + hi_d) / 2.0
            if abs(delta) > 1e-14:
                h[i] += delta
                corr_full += delta * g
                sum_h += delta
        h = _project_sum(h, target)
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        sum_h = np.sum(h)
    return h

def _perturb(h, scale, rng):
    h_p = h + scale * rng.standard_normal(len(h))
    return _project_sum(h_p, float(len(h)) / 2.0)

def _shift_perturb(h, shift_amt, rng):
    n = len(h)
    shifted = np.roll(h, shift_amt)
    h_p = 0.7 * h + 0.3 * shifted + 0.01 * rng.standard_normal(n)
    return _project_sum(h_p, float(n) / 2.0)

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _project_sum(h_new, float(n_new) / 2.0)

def _binary_init(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.integers(0, n, size=n_ones)
    h[idx] = 1.0
    return h

def _block_init(n, n_blocks, rng):
    h = np.zeros(n)
    per_block = (n // 2) // n_blocks

        start = (b * n // n_blocks) + rng.integers(-3, 4)
        for j in range(per_block):
            h[(start + j) % n] = 1.0
    return h
