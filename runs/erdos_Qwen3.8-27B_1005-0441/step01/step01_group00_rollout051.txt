        h1 = h.copy(); h1[i] = 1.0
        c0 = _corr_full(h0)
        c1 = _corr_full(h1)
        a = c1 - c0
        b = c0.copy()
        a[n-1] = 1.0
        ts = np.linspace(0, 1, n_grid)
        vals = a[:, None] * ts[None, :] + b[:, None]
        vals[n-1, :] = -ts**2 + ts + b[n-1]
        obj = vals.max(axis=0)
        h[i] = ts[np.argmin(obj)]
    return h

def _smooth_obj(h, eps):
    n = len(h)
    dx = 2.0/n
    c = _corr_full(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(s.sum())) * dx

def _smooth_grad(h, eps):
    n = len(h)
    dx = 2.0/n
    c = _corr_full(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n-1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n-m])
            g[:n-m] -= wk * h[m:]
        else:
            p = -m
            g[:n-p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n-p]
    return g * dx

def _eq_fun(x, n):
    return x.sum() - n*0.5

def _eq_jac(x, n):
    return np.ones(n)

import numpy as np

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(_corr_full(h).max() * 2.0 / n)

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _coord_sweep(h, n, n_grid):
    for i in range(n):
        h0 = h.copy(); h0[i] = 0.0
        h1 = h.copy(); h1[i] = 1.0
        c0 = _corr_full(h0)
        c1 = _corr_full(h1)
        a = c1 - c0
        b = c0.copy()
        a[n - 1] = 1.0
        ts = np.linspace(0, 1, n_grid)
        vals = a[:, None] * ts[None, :] + b[:, None]
        vals[n - 1, :] = -ts ** 2 + ts + b[n - 1]
        obj = vals.max(axis=0)
        h[i] = ts[np.argmin(obj)]
    return h

                          jac=_smooth_grad, method='SLSQP',

            hr = _proj(res.x, n)
            v = _c5(hr)
            if v < best_v:

            pass
    return best, best_v
