    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def soft_max_val(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    x_max = corr.max()
    shifted = tau * (corr - x_max)
    return x_max + (1.0 / tau) * np.log(np.sum(np.exp(shifted)))

def soft_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    x_max = corr.max()
    shifted = tau * (corr - x_max)
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L >= 0:
            idx1 = np.arange(L, n)
            if len(idx1) > 0:
                grad[idx1] += w * (1.0 - h[idx1 - L])
            idx2 = np.arange(0, n - L)
            if len(idx2) > 0:
                grad[idx2] -= w * h[idx2]
        else:
            M = -L
            idx1 = np.arange(0, n - M)
            if len(idx1) > 0:
                grad[idx1] += w * (1.0 - h[idx1 + M])
            idx2 = np.arange(M, n)
            if len(idx2) > 0:
                grad[idx2] -= w * h[idx2]
    return grad * dx

def slsqp_optimize(h0, tau, time_budget, seed):
    n = len(h0)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                   'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return soft_max_val(x, tau)
    
    def grad(x):
        return soft_max_grad(x, tau)
    
    t0 = time.time()
    try:
        result = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-12})
    except:
        return h0.copy(), compute_c5(h0)
    
    h_opt = project_feasible(result.x, target_sum)
    return h_opt, compute_c5(h_opt)

def gradient_descent(h0, time_budget, seed, use_smooth=True, tau_init=50.0):
    rng = np.random.RandomState(seed)
    n = len(h0)
    target_sum = n / 2.0
    h = project_feasible(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.85
    lr = 0.06
    tau = tau_init
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    
    while time.time() < deadline - 0.3:
        if no_improve > 300:
            h = project_feasible(best_h + rng.randn(n) * 0.03, target_sum)
            v[:] = 0.0
            lr = 0.06
            no_improve = 0
        elif no_improve > 100:
            lr = max(lr * 0.9, 0.002)
        
        if use_smooth:
            g = soft_max_grad(h, tau)
            tau = min(tau * 1.005, 500.0)
        else:
            # Top-5 gradient
            n2 = len(h)
            dx = 2.0 / n2
            corr = np.correlate(h, 1.0 - h, 'full')
            top_idx = np.argsort(corr)[::-1][:5]
            w = np.zeros(len(corr))
            w[top_idx] = 0.2
            g = np.zeros(n2)
            for k in range(len(corr)):
                wi = w[k]
                if wi < 1e-15:
                    continue
                L = k - (n2 - 1)
                if L >= 0:
                    idx1 = np.arange(L, n2)
                    if len(idx1) > 0:
                        g[idx1] += wi * (1.0 - h[idx1 - L])
                    idx2 = np.arange(0, n2 - L)
                    if len(idx2) > 0:
                        g[idx2] -= wi * h[idx2]
                else:
                    M = -L
                    idx1 = np.arange(0, n2 - M)
                    if len(idx1) > 0:
                        g[idx1] += wi * (1.0 - h[idx1 + M])
                    idx2 = np.arange(M, n2)
                    if len(idx2) > 0:
                        g[idx2] -= wi * h[idx2]
            g *= dx
        
        v = beta * v - lr * g
        h = project_feasible(h + v, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            no_improve = 0
            if no_improve == 0:
                lr = min(lr * 1.01, 0.12)
        else:
            no_improve += 1
    
    return best_h, best_c5

def golden_init(n, seed):

    positions = np.floor(np.arange(target) * phi + offsets[0] * n).astype(int) % n
    h = np.zeros(n)
    h[positions] = 1.0
    return h

    n = len(h0)
    target_sum = n / 2.0

        corr = compute_corr(x) * (2.0 / n)
        x_max = corr.max()
        if tau < 1e6:
            return x_max + (1.0 / tau) * np.log(np.sum(np.exp(tau * (corr - x_max))))
        return x_max
    
    def gradient(x):
        corr = compute_corr(x) * (2.0 / n)
        x_max = corr.max()
        shifted = tau * (corr - x_max)
        weights = np.exp(np.clip(shifted, -50, 50))
        weights /= weights.sum()
        return compute_grad_weighted(x, weights, 2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                   'jac': lambda x: np.ones(n)}]

                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-14})
        h_opt = project_feasible(result.x, target_sum)
        return h_opt, compute_c5(h_opt)

import numpy as np
import time
import math
from scipy.optimize import minimize

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def corr_and_weights(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = tau * (corr - corr.max())
    exps = np.exp(np.clip(shifted, -500, 500))
    weights = exps / exps.sum()
    return corr, weights

def grad_from_weights(h, weights, dx):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L >= 0:
            a = np.arange(L, n)
            g[a] += w * (1.0 - h[a - L])
            b = np.arange(0, n - L)
            g[b] -= w * h[b]
        else:
            M = -L
            a = np.arange(0, n - M)
            g[a] += w * (1.0 - h[a + M])
            b = np.arange(M, n)
            g[b] -= w * h[b]
    return g * dx

def ngd_smooth(h0, tb, seed, t0v=60.0, t1v=600.0):
    rng = np.random.RandomState(seed)
    n = len(h0)
    ts = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h0.copy(), ts)
    bh, bc = h.copy(), compute_c5(h)
    v = np.zeros(n)
    gamma, lr, tau = 0.88, 0.05, t0v
    t0 = time.time()
    dl = t0 + tb
    ni = 0
    while time.time() < dl - 0.2:
        if ni > 400:
            h = project_feasible(bh + rng.randn(n) * 0.04, ts)
            v[:] = 0.0; lr = 0.05; tau = min(tau * 0.4, 100.0); ni = 0
        elif ni > 150:
            lr = max(lr * 0.9, 0.003)
        _, w = corr_and_weights(h, tau)
        g = grad_from_weights(h, w, dx)
        v = gamma * v - lr * g
        h = project_feasible(h + v, ts)
        c = compute_c5(h)
        if c < bc:
            bh, bc, ni = h.copy(), c, 0
        else:
            ni += 1
        tau = min(tau * 1.004, t1v)
    return bh, bc

def ngd_topk(h0, tb, seed, k=4):

    n = len(h0)
    ts = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h0.copy(), ts)

    v = np.zeros(n)

    while time.time() < dl - 0.2:

def binary_sa(hf, tb, seed):
    rng = np.random.RandomState(seed)
    n = len(hf)
    tgt = n // 2
    h = (hf >= 0.5).astype(float)
    d = tgt - int(h.sum())
    if d > 0:
        z = np.where(h == 0)[0]; rng.shuffle(z); h[z[:d]] = 1
    elif d < 0:
        o = np.where(h == 1)[0]; rng.shuffle(o); h[o[:d]] = 0
    bh, bc, cc = h.copy(), compute_c5(h), compute_c5(h)
    T = 0.04
    t0 = time.time(); dl = t0 + tb
    while time.time() < dl - 0.3:
        o = np.where(h == 1)[0]; z = np.where(h == 0)[0]
        if not o.size or not z.size: break
        i, j = o[rng.randint(o.size)], z[rng.randint(z.size)]
        h[i], h[j] = 0, 1
        nc = compute_c5(h)
        if nc <= cc or (T > 0.002 and rng.rand() < math.exp(-(nc - cc) / T)):
            cc = nc
            if nc < bc: bh, bc = h.copy(), nc
        else:
            h[i], h[j] = 1, 0
        T = max(T * 0.99995, 0.002)
    return bh, bc

def slsqp(h0, tau, tb):
    n = len(h0)
    ts = n / 2.0
    def f(x):
        c = np.correlate(x, 1-x, 'full') * (2.0/n)
        m = c.max()
        return m + np.logaddexp.reduce(tau*(c-m))/tau if tau < 1e5 else m
    def gf(x):
        c = np.correlate(x, 1-x, 'full')
        s = tau*(c - c.max())
        w = np.exp(np.clip(s,-500,500)); w /= w.sum()
        return grad_from_weights(x, w, 2.0/n)
    try:
        r = minimize(f, h0.copy(), jac=gf, method='SLSQP',
                     bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':lambda x:x.sum()-ts,'jac':lambda x:np.ones(n)}],
                     options={'maxiter':200,'ftol':1e-14})
        return project_feasible(r.x, ts), compute_c5(project_feasible(r.x, ts))
    except Exception:
        return h0.copy(), compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time(); dl = t0 + budget_s
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)
    n = len(h0)
    h0 = project_feasible(h0, n/2.0)
    bh, bc = h0.copy(), compute_c5(h0)
    
    # P1: SLSQP
    for tau in [80.0, 250.0]:
        rm = dl - time.time() - 100
        if rm > 30:
            h,c = slsqp(bh, tau, min(rm/2, 100))
            if c < bc: bh, bc = h, c
    
    # P2: Nesterov smooth
    rm = dl - time.time() - 90
    if rm > 30:
        h,c = ngd_smooth(bh, min(rm, 220), seed, 50.0, 400.0)
        if c < bc: bh, bc = h, c
    
    # P3: Top-k
    rm = dl - time.time() - 70
    if rm > 25:
        h,c = ngd_topk(bh, min(rm, 130), seed+60, k=3)
        if c < bc: bh, bc = h, c
    
    # P4: Binary SA
    rm = dl - time.time() - 50
