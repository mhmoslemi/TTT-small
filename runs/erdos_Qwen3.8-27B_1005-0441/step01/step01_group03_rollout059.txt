def compute_grad_all(h, n):
    """Compute gradient of each overlap_k w.r.t. h. Returns array of shape (2n-1, n)."""
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        if s > 0:
            grads[k, :n-s] += (1.0 - h[s:])
            grads[k, s:] += (-h[:n-s])
        elif s < 0:
            t = -s
            grads[k, t:] += (1.0 - h[:n-t])
            grads[k, :n-t] += (-h[t:])
        else:
            grads[k] = (1.0 - 2.0*h)
    grads *= dx
    return grads

= dx * 1 - dx * h[j+s] - dx * h[j-s]

So the gradient is: dx * (1_{valid} - h[j+s] - h[j-s])

This is affine in h (not linear). But for the purpose of computing w^T * grad_h(ov_k) for all k, I can use:

grad_total[j] = dx * sum_k w_k * [1_{j valid for k} - h[j+s_k] - h[j-s_k]]

The first term: dx * sum_k w_k * 1_{j valid for k} = dx * sum of weights of lags where j is in range.
The second term: -dx * sum_k w_k * h[j+s_k] - dx * sum_k w_k * h[j-s_k]

The second terms can be computed as correlations!
sum_k w_k * h[j+s_k] = correlation of h with w (shifted)
sum_k w_k * h[j-s_k] = another correlation

Let me think... Let W be the weight vector (length 2n-1) with W[k] = w_k.
Then sum_k w_k * h[j + (k-(n-1))] for valid j+k-(n-1):
This is like (W * h)[j] where * is the appropriate convolution.

Actually: sum_k W[k] * h[j + k - (n-1)] = (W ⊛ h)[j] where ⊛ is cross-correlation.

def to_binary(h, n):
    target = n // 2
    # Round all to nearest 0 or 1
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly target ones
    n_ones = int(np.sum(h_bin))
    while n_ones > target:
        # Flip the one with smallest h value (closest to 0.5 from above)
        idx = np.where(h_bin > 0.5)[0]
        j = idx[np.argmin(h[idx])]
        h_bin[j] = 0
        n_ones -= 1
    while n_ones < target:
        idx = np.where(h_bin <= 0.5)[0]
        j = idx[np.argmax(h[idx])]
        h_bin[j] = 1
        n_ones += 1
    return h_bin

def bin_local_search(h, n, deadline, rng):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    no_improve = 0
    perturb_interval = 100
    
    while time.time() < deadline:
        ov = get_overlaps(h, n)
        c5 = float(np.max(ov))
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > perturb_interval:
            # Perturb: swap m random pairs
            m = max(2, n // 8)
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            for _ in range(m):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i], h[j] = 0.0, 1.0
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
            no_improve = 0
            perturb_interval = int(perturb_interval * 1.3)
            continue
        
        # Find worst lag
        k_max = int(np.argmax(ov))
        s = k_max - (n - 1)
        
        # Generate candidate swaps
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        
        # For the worst lag, find contributing 1-positions
        cands = []
        if s > 0:
            for i in range(n - s):
                if h[i] > 0.5 and h[i+s] <= 0.5:
                    cands.append(i)
        elif s < 0:
            t = -s
            for i in range(t, n):
                if h[i] > 0.5 and h[i-t] <= 0.5:
                    cands.append(i)
        
        if not cands:
            cands = list(ones)
        
        # Try swaps: for each candidate 1-position, swap with a few 0-positions
        best_swap = None
        best_swap_c5 = c5
        
        # Limit number of candidates to try
        if len(cands) > 20:
            cands = list(rng.choice(cands, size=20, replace=False))
        
        n_zeros_try = min(8, len(zeros))
        z_sample = rng.choice(zeros, size=n_zeros_try, replace=False)
        
        for i in cands:
            for j in z_sample:
                if i == j:
                    continue
                h[i], h[j] = h[j], h[i]
                nc = get_c5(h, n)
                h[i], h[j] = h[j], h[i]
                if nc < best_swap_c5 - 1e-12:
                    best_swap_c5 = nc
                    best_swap = (i, j)
        
        if best_swap:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
        else:
            no_improve += 5
    
    return best_h, best_c5

def cont_optimize(h, n, deadline, rng, alpha=100.0):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = get_c5(h, n)
    lr = 0.05
    vel = np.zeros(n)
    beta = 0.9
    
    while time.time() < deadline:
        ov = get_overlaps(h, n)
        c5 = float(np.max(ov))
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        
        # Softmax weights
        m = np.max(ov)
        e = np.exp(ov - m)
        w = e / np.sum(e)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-15:
                continue
            s = k - (n-1)
            if s > 0:
                grad[:n-s] += w[k] * (1.0 - h[s:])
                grad[s:] += w[k] * (-h[:n-s])
            elif s < 0:
                t = -s
                grad[t:] += w[k] * (1.0 - h[:n-t])
                grad[:n-t] += w[k] * (-h[t:])
            else:
                grad += w[k] * (1.0 - 2.0*h)
        grad *= dx
        
        # Nesterov / momentum
        vel = beta * vel + grad
        h_new = h - lr * vel
        h_new = project_feasible(h_new, n)
