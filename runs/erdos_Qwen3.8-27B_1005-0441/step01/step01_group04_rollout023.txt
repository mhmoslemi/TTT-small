    except Exception:
        return h0.copy()

def _coord_descent(h0, n_iter, t_end):
    n = len(h0)
    h = h0.astype(float).copy()
    best_c5 = _eval_c5(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            best_v = h[i]
            best_c = best_c5
            for dv in [-0.1, -0.05, -0.01, 0.01, 0.05, 0.1]:
                nv = h[i] + dv
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                h2 = _fix_sum(h, n)
                c = _eval_c5(h2)
                if c < best_c:
                    best_c = c
                    best_v = nv
            if best_v != h[i]:
                h = _fix_sum(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def _random_h(n, rng):
    h = rng.random(n)
    return _fix_sum(h, n)

def _block_h(n):
    pat = np.array([0.9, 0.9, 0.1])
    h = np.tile(pat, n // 3 + 1)[:n]
    return _fix_sum(h, n)

def _sin_h(n, freq):
    x = np.linspace(0.0, 2.0 * np.pi * freq, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(x)
    return _fix_sum(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None and len(h_init) > 0:
        h_init = _fix_sum(h_init, len(h_init))
        c5 = _eval_c5(h_init)
        best_c5 = c5
        best_h = h_init.copy()
    else:
        best_h = _fix_sum(np.full(80, 0.5), 80)
        best_c5 = _eval_c5(best_h)

    n = len(best_h)
    t_phase1 = t0 + budget_s * 0.45
    t_phase2 = t0 + budget_s * 0.75
    t_phase3 = t0 + budget_s * 0.92

    # Phase 1: Subgradient optimization with parameter sweep
    for alpha in [2.0, 4.0, 8.0, 15.0, 30.0, 60.0]:
        for top_k in [3, 5, 10, 20]:
            if time.time() > t_phase1:
                break
            h_sgd, c5_sgd = _sgd_optimize(best_h, 4000, alpha, top_k, 0.004, t_phase1)
            if c5_sgd < best_c5:
                best_c5 = c5_sgd
                best_h = h_sgd.copy()

    # Phase 2: SLSQP polish with increasing alpha
    for alpha in [40, 80, 150, 300, 600]:
        if time.time() > t_phase2:
            break
        h_polish = _slsqp_polish(best_h, alpha, 500)
        c5 = _eval_c5(h_polish)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_polish

    # Coordinate descent polish
    if time.time() < t_phase3:
        h_cd, c5_cd = _coord_descent(best_h, 50, t_phase3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 3: Multi-start exploration with other n and perturbations
    for n_try in [n, 40, 50, 60, 100, 120, 150]:
        if time.time() > budget_s - 60:
            break
        if n_try > 200:
            continue
        for start_fn in ['rand', 'block', 'sin1', 'sin2']:
            if time.time() > budget_s - 40:
                break
            if start_fn == 'rand':
                s = _random_h(n_try, rng)
            elif start_fn == 'block':
                s = _block_h(n_try)
            elif start_fn == 'sin1':
                s = _sin_h(n_try, 1)
            else:
                s = _sin_h(n_try, 2)
            h_sgd, c5_sgd = _sgd_optimize(s, 2000, 5.0, 5, 0.004, budget_s - 40)
            h_polish = _slsqp_polish(h_sgd, 100, 300)
            c5 = _eval_c5(h_polish)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_polish

    # Perturbation-based local search on best
    for _ in range(10):
        if time.time() > budget_s - 20:
            break
        h_pert = _fix_sum(best_h + rng.normal(0, 0.02, len(best_h)), len(best_h))
        h_sgd, c5_sgd = _sgd_optimize(h_pert, 2000, 8.0, 4, 0.003, budget_s - 15)
        h_polish = _slsqp_polish(h_sgd, 200, 300)
        c5 = _eval_c5(h_polish)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_polish

    # Final SLSQP + coordinate descent
    if time.time() < budget_s - 10:
        h_polish = _slsqp_polish(best_h, 500, 400)
        c5 = _eval_c5(h_polish)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_polish
        h_cd, c5_cd = _coord_descent(best_h, 30, budget_s - 2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
