    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.99999999
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_penalty(h, n, beta, mu):
    dx = 2.0 / n
    c = get_corr(h)
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    logsumexp_term = np.log(np.sum(np.exp(shifted))) / beta
    penalty = mu * (np.sum(h) - n / 2.0) ** 2
    return (m + logsumexp_term) * dx + penalty

def lfgs_optimize(h, n, beta, mu, maxiter=300):
    res = minimize(smooth_obj_penalty, h, args=(n, beta, mu), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return project_feasible(h, n)

def slsqp_optimize(h, n, beta, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    def obj(x):
        return smooth_obj_penalty(x, n, beta, 0.0)
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return project_feasible(h, n)

        for i in range(nblocks):
            h[i*block_len:(i+1)*block_len] = 1.0 if rng.rand() > 0.5 else 0.0
        h[-n % block_len:] = 0
    elif style == 3:  # Smooth wave

        h = 0.5 + 0.4 * np.sin(np.pi * freq * x) * np.cos(np.pi * freq * 0.3 * x)
        h = np.clip(h, 0, 1)
