import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _feasible_project(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} approximately."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        # Distribute correction among non-boundary elements
        d = (target - s) / len(h)
        h = h + d
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / t
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    return (m + t * np.log(np.sum(np.exp(shifted)))) * dx

def _slsqp(h0, n, t, max_iter=300):
    target = n / 2.0
    h0 = _feasible_project(h0, target)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)},)
    try:
        res = minimize(lambda x: _smooth_obj(x, n, t), h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h_new = _feasible_project(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def _nm(h0, n, max_iter=2000):
    target = n / 2.0
    dx = 2.0 / n
    h0 = _feasible_project(h0, target)
    def obj(x):
        x = _feasible_project(x, target)
        return compute_c5(x)
    try:
        res = minimize(obj, h0, method='Nelder-Mead',
                       options={'maxiter': max_iter, 'xatol': 1e-7, 'fatol': 1e-10})
        h_new = _feasible_project(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def _binary_init(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def _block_init(n, rng):
    h = np.zeros(n)
    half = n // 4
    gap = rng.randint(2, n//4)
    s1 = rng.randint(0, max(1, n - 2*half - gap))
    for i in range(half):
        h[(s1+i) % n] = 1.0
        h[(s1+half+gap+i) % n] = 1.0
    # Fill remaining 1s
    remaining = n//2 - 2*half
    if remaining > 0:
        free = np.where(h == 0)[0]
        extra = rng.choice(free, remaining, replace=False)
        h[extra] = 1.0
    return h

def _sin_init(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5
    n_terms = rng.randint(2, 6)
    for _ in range(n_terms):
        h += rng.uniform(0.05, 0.2) * np.sin(np.pi * rng.randint(1, 12) * x + rng.uniform(0, 6.28))
    h = np.clip(h, 0.0, 1.0)
    return _feasible_project(h, n/2.0)

def _upsample(h, n_new):

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = np.sum(h)
        d = (target - s) / len(h)
        if abs(d) < 1e-12:
            break
        h = h + d
        h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(h, n, t):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    z = (corr - m) / max(t, 1e-12)
    z = np.clip(z, -100, 0)
    return (m + t * np.log(np.sum(np.exp(z)))) * (2.0 / n)

def _slsqp(h, n, t, iters=400):
    tgt = n / 2.0

    tgt = n / 2.0
    best = h.copy()

            h_t = best.copy()
            h_t[idx] = x
            h_t = _proj(h_t, tgt)
            return _sobj(h_t, n, 0.001)
        try:

            if c5 < best_c5 - 1e-12:

                best_c5 = c5
        except:
            pass
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(40):
        s = np.sum(h)
        d = (target - s) / len(h)
        if abs(d) < 1e-13:
            break
        h = h + d
        h = np.clip(h, 0.0, 1.0)
    return h

def _sobj(h, n, t):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    z = (corr - m) / max(t, 1e-12)
    z = np.clip(z, -200, 0)
    return (m + t * np.log(np.sum(np.exp(z)))) * (2.0 / n)

def _slsqp(h, n, t, iters=400):
    tgt = n / 2.0
    h = _proj(h, tgt)
    bnd = [(0.0, 1.0)] * n
    cn = ({'type': 'eq', 'fun': lambda x: np.sum(x) - tgt, 'jac': lambda x: np.ones(n)},)
    try:
        r = minimize(lambda x: _sobj(x, n, t), h, method='SLSQP', bounds=bnd,
                     constraints=cn, options={'maxiter': iters, 'ftol': 1e-13})
        hp = _proj(r.x, tgt)
        return hp, compute_c5(hp)
    except Exception:
        return h, compute_c5(h)
