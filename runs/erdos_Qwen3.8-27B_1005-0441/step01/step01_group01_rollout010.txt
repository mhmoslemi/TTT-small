import numpy as np
from scipy.optimize import minimize
import time

_SUM_TARGET = 0.0

def _sum_eq(x):
    return np.sum(x) - _SUM_TARGET

def _sum_eq_jac(x):
    return np.ones(len(x))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_smooth(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    return sm * dx

def make_two_block(n):
    h = np.zeros(n)
    b = n // 4
    h[:b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def make_three_block(n):
    h = np.zeros(n)
    b = n // 6
    h[:b] = 1.0
    h[n // 2 - b:n // 2 + b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def _f1d(t, A, B):
    return np.max(A * t + B)

def coord_solve_1d(A, B):
    """Minimize max_k(A_k*t + B_k) over t in [0,1] via golden section."""
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = _f1d(c, A, B)
    fd = _f1d(d, A, B)
    for _ in range(40):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = _f1d(c, A, B)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = _f1d(d, A, B)
    return (a + b) / 2.0

def coord_descent(h, n, max_sweeps, t0, budget_frac):
    """Cyclic coordinate descent for min max_k c_k with projection onto sum constraint."""
    dx = 2.0 / n
    h = h.copy()
    k_idx = np.arange(2 * n - 1)
    limit = t0 + budget_frac
    for sweep in range(max_sweeps):
        if time.time() > limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        for j in range(n):
            if time.time() > limit:
                break
            idx1 = j + n - 1 - k_idx
            idx2 = j + k_idx - n + 1
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            A = dx - np.where(valid1, h[np.clip(idx1, 0, n - 1)], 0.0)
            A = A - np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
            B = corr - A * h[j]
            t_opt = coord_solve_1d(A, B)
            h[j] = t_opt
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)
    return h

def slsqp_optimize(h, n, alpha, maxiter, t0, limit):
    """SLSQP with log-sum-exp smooth max."""
    global _SUM_TARGET
    _SUM_TARGET = n * 0.5
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    if time.time() > limit:
        return h
    res = minimize(
        obj_smooth, h.copy(), args=(alpha,),
        method='SLSQP', bounds=bounds, constraints=constraint,
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    if time.time() > limit:
        return h
    return project_feasible(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _SUM_TARGET
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = make_two_block(160)
    n = len(h_best)
    _SUM_TARGET = n * 0.5
    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)

    # Phase 1: Coordinate descent (main workhorse) - 55% budget
    limit_cd = t0 + budget_s * 0.55
    h_cd = coord_descent(h_best, n, max_sweeps=200, t0=t0, budget_frac=limit_cd)
    c5_cd = compute_c5(h_cd)
    if c5_cd < c5_best:
        c5_best = c5_cd
        h_best = h_cd.copy()

    # Phase 2: SLSQP polish on coord descent result - 15% budget
    limit_slsqp = t0 + budget_s * 0.70
    for alpha in [80.0, 200.0, 400.0]:
        if time.time() > limit_slsqp:
            break
        h_pol = slsqp_optimize(h_best, n, alpha, 1500, t0, limit_slsqp)
        c5_pol = compute_c5(h_pol)
        if c5_pol < c5_best:
            c5_best = c5_pol
            h_best = h_pol.copy()

    # Phase 3: Try alternative structures with coord descent + SLSQP
    limit_phase3 = t0 + budget_s * 0.85
    starts = [make_two_block(n), make_three_block(n)]
    for sp in starts:
        if time.time() > limit_phase3:
            break
        h_s = coord_descent(sp, n, max_sweeps=80, t0=t0, budget_frac=limit_phase3)
        c5_s = compute_c5(h_s)
        if c5_s < c5_best:
            c5_best = c5_s
            h_best = h_s.copy()
        if time.time() < limit_phase3:
            h_s2 = slsqp_optimize(h_best, n, 200.0, 800, t0, limit_phase3)
            c5_s2 = compute_c5(h_s2)
            if c5_s2 < c5_best:
                c5_best = c5_s2
                h_best = h_s2.copy()

    # Phase 4: Perturbation + coord descent + SLSQP
    limit_phase4 = t0 + budget_s * 0.95
    while time.time() < limit_phase4:
        pert = 0.03 + 0.03 * rng.random()
        h_p = h_best + pert * rng.standard_normal(n)
        h_p = project_feasible(h_p, n)
        h_p = coord_descent(h_p, n, max_sweeps=30, t0=t0, budget_frac=limit_phase4)
        c5_p = compute_c5(h_p)
        if c5_p < c5_best:
            c5_best = c5_p
            h_best = h_p.copy()
        if time.time() < limit_phase4:
            h_p2 = slsqp_optimize(h_best, n, 300.0, 500, t0, limit_phase4)
            c5_p2 = compute_c5(h_p2)
            if c5_p2 < c5_best:
                c5_best = c5_p2
                h_best = h_p2.copy()

    # Phase 5: Try coarser grid, upscale, polish
    if time.time() < t0 + budget_s - 20:
        n_c = max(40, n // 4)
        h_c = np.interp(np.arange(n_c), np.arange(0, float(n - 1)), h_best)
        h_c = project_feasible(h_c, n_c)
        h_c = coord_descent(h_c, n_c, max_sweeps=100, t0=t0, budget_frac=t0 + budget_s - 20)
        h_up = np.interp(np.arange(n), np.arange(n_c), h_c)
        h_up = project_feasible(h_up, n)
        h_up = slsqp_optimize(h_up, n, 200.0, 500, t0, t0 + budget_s - 10)
        c5_up = compute_c5(h_up)
        if c5_up < c5_best:
            c5_best = c5_up
            h_best = h_up.copy()

    h_best = project_feasible(h_best, len(h_best))
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
