import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h

def _to_binary(h, n):
    # Convert to binary with exactly n/2 ones using threshold + fix
    h = np.sort(h)
    # We want the top n/2 to be 1
    # But we need to be careful about ties
    threshold = h[n//2 - 1]
    binary = (h >= threshold).astype(float)
    # Fix count
    n_ones = int(binary.sum())
    if n_ones != n//2:
        # Re-sort original
        idx = np.argsort(h)[::-1]
        binary = np.zeros(n)
        binary[idx[:n//2]] = 1.0
    return binary

def _binary_local_search(h_bin, max_iter=500, timeout=None, t0=None):
    n = len(h_bin)
    dx = 2.0 / n
    best_c5 = _eval_c5(h_bin)
    h_cur = h_bin.copy()
    
    ones = np.where(h_cur == 1.0)[0]
    zeros = np.where(h_cur == 0.0)[0]
    
    for it in range(max_iter):
        if timeout is not None and time.time() - t0 > timeout:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        cmax = np.max(corr)
        
        best_delta = 1e-10
        best_i = -1
        best_j = -1
        
        for i in ones:
            for j in zeros:
                # Try swap
                h_try = h_cur.copy()
                h_try[i] = 0.0
                h_try[j] = 1.0
                c_try = _eval_c5(h_try)
                delta = c_try - cmax
                if delta < -best_delta:
                    best_delta = -delta
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h_cur[best_i] = 0.0
            h_cur[best_j] = 1.0
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
            best_c5 = _eval_c5(h_cur)
        else:
            break
    
    return h_cur, best_c5

from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h

def _softmax_obj(h, alpha):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shift = corr - cmax
    return cmax + np.log(np.sum(np.exp(alpha * shift))) / alpha

def _softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shift = corr - cmax
    weights = np.exp(alpha * shift) / np.sum(np.exp(alpha * shift))
    # grad of corr[m] w.r.t. h[j]
    # corr[m] = dx * sum_k h[k+s]*(1-h[k]) where s = m-(n-1)
    # d/dh[j]: dx * [(1-h[j-s]) * 1_{valid} - h[j+s] * 1_{valid}]
    grad = np.zeros(n)
    for m in range(2*n-1):
        w = weights[m]
        if w < 1e-10:
            continue
        s = m - (n - 1)
        # (1-h[j-s]) for j where 0<=j-s<n => s<=j<n+s
        if s >= 0:
            grad[s:n] += w * (1.0 - h[0:n-s])
        else:
            grad[0:n+s] += w * (1.0 - h[-s:n])
        # -h[j+s] for j where 0<=j+s<n => -s<=j<n-s
        if s >= 0:
            grad[0:n-s] -= w * h[s:n]
        else:
            grad[-s:n] -= w * h[0:n+s]
    grad *= dx
    return grad

def _eq_sum(h):
    return float(np.sum(h) - len(h) / 2.0)

def _optimize_slsqp_softmax(h0, alpha, maxiter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    h0 = _fix_sum(h0, n)
    try:
        from scipy.optimize import minimize
        res = minimize(
            lambda h: _softmax_obj(h, alpha),
            h0,
            jac=lambda h: _softmax_grad(h, alpha),
            method='SLSQP',
            constraints=[{'type': 'eq', 'fun': _eq_sum}],
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return _fix_sum(res.x, n)
    except Exception:
        return h0.copy()

        cmax = np.max(corr)
        
        best_delta = 1e-10
        best_i = -1
        best_j = -1
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        for ii in range(n_ones):
            i = ones[ii]
            for jj in range(n_zeros):
                j = zeros[jj]
                h_try = h_cur.copy()
                h_try[i] = 0.0
                h_try[j] = 1.0
                c_try = _eval_c5(h_try)
                if c_try < cmax - best_delta:
                    best_delta = cmax - c_try
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h_cur[best_i] = 0.0
            h_cur[best_j] = 1.0
            new_c5 = _eval_c5(h_cur)
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            break
    
    return h_cur, best_c5

def _subgradient_step(h, lr, active_tol=0.005):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    active = np.where(corr >= cmax - active_tol)[0]
    
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        if s >= 0:
            grad[s:n] += (1.0 - h[0:n-s])
            grad[0:n-s] -= h[s:n]
        else:
            grad[0:n+s] += (1.0 - h[-s:n])
            grad[-s:n] -= h[0:n+s]
    grad *= dx / len(active)
    
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _fix_sum(h_new, n)
    return h_new

def _random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def _structured_h(n, rng):
    """Create structured starting points"""
    x = np.linspace(0, 2*n, n, endpoint=False)
    patterns = []
    # Sine
    patterns.append(0.5 + 0.4*np.sin(2*np.pi*x/(2*n)))
    # Two frequencies
    patterns.append(0.5 + 0.3*np.sin(2*np.pi*x/(2*n)) + 0.2*np.sin(4*np.pi*x/(2*n)))
    # Block
    block = np.zeros(n)
    blen = max(1, n//8)
    for b in range(0, n, blen*2):
        block[b:b+blen] = 1.0
    patterns.append(block)
    # Random
    patterns.append(rng.random(n))
    
    p = rng.integers(len(patterns))
    h = patterns[p]
    h = np.clip(h, 0.0, 1.0)
    return _fix_sum(h, n)
