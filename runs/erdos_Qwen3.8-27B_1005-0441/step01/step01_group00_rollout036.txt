- Find δ* that minimizes max_k (c[k] + α_k * δ)
- The optimal δ* is where two (or more) of the lines c[k] + α_k * δ are equal and maximal

The 1D problem min_δ max_k (a_k + b_k * δ) can be solved by:
- Binary search on the value t: for a given t, check if there exists δ such that a_k + b_k * δ <= t for all k

- And also 0 <= δ <= min(h[i], 1-h[j])

import numpy as np

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _proj(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(150):
        d = n * 0.5 - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            h = np.clip(h + d / n, 0, 1)
            break
        h[free] += d / nf
        h = np.clip(h, 0, 1)
    return h

def _ograd_single(h, k, n):
    m = k - (n - 1)
    g = np.zeros(n)
    if m >= 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        p = -m
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g

def _smax_val_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    sh = np.clip(c - cmax, -200, 0)
    s = np.exp(sh / eps)
    ssum = s.sum()
    w = s / ssum
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-13:
            g += w[k] * _ograd_single(h, k, n)
    return (cmax + eps * np.log(ssum)) * dx, g * dx

def _grad_pair(h, i, j, n, top_lags):
    alpha = np.zeros(len(top_lags))
    for idx, k in enumerate(top_lags):
        m = k - (n - 1)
        di = dj = 0.0
        if m >= 0:
            if i >= m:
                di += (1.0 - h[i - m])
            if i <= n - 1 - m:
                di -= h[i + m]
            if j >= m:
                dj += (1.0 - h[j - m])
            if j <= n - 1 - m:
                dj -= h[j + m]
        else:

            if i >= p:
                di += (1.0 - h[i - p])
            if i <= n - 1 - p:
                di -= h[i + p]
            if j >= p:
                dj += (1.0 - h[j - p])
            if j <= n - 1 - p:
                dj -= h[j + p]
        alpha[idx] = di - dj
    return alpha

def _opt_delta(c_top, alpha, dmax):
    if dmax <= 1e-12:
        return 0.0
    candidates = [0.0, dmax]
    nl = len(c_top)
    for a in range(nl):
        for b in range(a + 1, nl):
            da = alpha[a] - alpha[b]
            if abs(da) < 1e-14:
                continue
            d = (c_top[b] - c_top[a]) / da
            if 0.0 <= d <= dmax:
                candidates.append(d)
    best_d = 0.0
    best_v = np.inf
    for d in candidates:
        v = np.max(c_top + alpha * d)
        if v < best_v:
            best_v = v
            best_d = d
    return best_d
