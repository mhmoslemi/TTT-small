def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _make_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _make_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    return h_b

def _max_corr(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def _local_search(h, t_budget=60):
    n = len(h)
    dx = 2.0 / n
    h = _make_binary(h.copy())
    best_val = _max_corr(h)
    best_h = h.copy()
    t0 = time.time()
    stall = 0
    while time.time() - t0 < t_budget - 1:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_new_val = best_val
        best_pair = None
        for i in ones:
            if time.time() - t0 > t_budget - 1:
                break
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                v = _max_corr(h)
                h[j] = 0.0
                if v < best_new_val - 1e-10:
                    best_new_val = v
                    best_pair = (i, j)
            h[i] = 1.0
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_val = best_new_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3:
                break
    return best_h, best_val * dx

def _sim_anneal(h, t_budget=120, rng=None):
    n = len(h)
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(42)
    h = _make_binary(h.copy())
    curr_val = _max_corr(h)
    best_val = curr_val
    best_h = h.copy()
    t0 = time.time()
    T0 = 4.0
    Tend = 0.05
    last_improve = t0
    while time.time() - t0 < t_budget - 1:
        elapsed = time.time() - t0
        T = T0 * (Tend / T0) ** (elapsed / max(t_budget, 1))
        if time.time() - last_improve > 8:
            T = max(T, 2.0)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        old_val = curr_val
        h[i] = 0.0
        h[j] = 1.0
        curr_val = _max_corr(h)
        if curr_val <= old_val:
            if curr_val < best_val - 1e-10:
                best_val = curr_val
                best_h = h.copy()
                last_improve = time.time()
        else:
            delta = curr_val - old_val
            if rng.random() < np.exp(-delta / T):
                pass
            else:
                h[i] = 1.0
                h[j] = 0.0
                curr_val = old_val
    return best_h, best_val * dx

def _perturb(h, rng, n_flips=3):
    h = h.copy()
    for _ in range(n_flips):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i != j:
            h[i] = 0.0
            h[j] = 1.0
    return h

def _optimize_n(n, h0, t_budget, rng):
    h = _make_binary(h0.copy())
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    frac_ls1 = 0.25
    frac_sa1 = 0.35
    frac_ls2 = 0.15
    frac_sa2 = 0.25
    t1 = t_budget * frac_ls1
    t2 = t_budget * frac_sa1
    t3 = t_budget * frac_ls2
    t4 = t_budget * frac_sa2
    h, c = _local_search(h, t_budget=t1)
    if c < best_c:
        best_c = c
        best_h = h
    elapsed = time.time() - t0
    h2, c2 = _sim_anneal(h, t_budget=min(t2, t_budget - elapsed - 2), rng=rng)
    if c2 < best_c:
        best_c = c2
        best_h = h2
    elapsed = time.time() - t0
    h3, c3 = _local_search(best_h, t_budget=min(t3, t_budget - elapsed - 2))
    if c3 < best_c:
        best_c = c3
        best_h = h3
    elapsed = time.time() - t0
    h4, c4 = _sim_anneal(best_h, t_budget=min(t4, t_budget - elapsed - 2), rng=rng)
    if c4 < best_c:
        best_c = c4
        best_h = h4
    elapsed = time.time() - t0
    remaining = t_budget - elapsed - 1
    if remaining > 20:
        h5, c5 = _local_search(best_h, t_budget=remaining)
        if c5 < best_c:
            best_c = c5
            best_h = h5
    return best_h, best_c
