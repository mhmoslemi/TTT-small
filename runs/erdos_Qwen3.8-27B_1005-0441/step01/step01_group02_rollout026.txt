def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(10000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = (target - s) / n
        h = np.clip(h + diff, 0.0, 1.0)
    return h

def smooth_max(corr, beta):
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta

def smooth_max_grad_wrt_h(h, beta):
    """Gradient of smooth_max(corr * dx, beta) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-12:
            continue
        d = i - (n - 1)  # actual lag
        if d >= 0:
            m = n - d
            grad[d:n] += wi * (1.0 - h[0:m])
            grad[0:m] -= wi * h[d:n]

            m = n - d2
            grad[d2:n] += wi * (1.0 - h[0:m])
            grad[0:m] -= wi * h[d2:n]

    w = np.exp(3.0 * w)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        d = ki - (n - 1)
        if d >= 0:
            m = n - d
            grad[d:n] += wi * (1.0 - h[0:m])
            grad[0:m] -= wi * h[d:n]
        else:
            d2 = -d
            m = n - d2
            grad[d2:n] += wi * (1.0 - h[0:m])
            grad[0:m] -= wi * h[d2:n]
    h_new = project_feasible(h - step_size * grad)
    return h_new

def subgradient_opt(h0, n_iter, t_budget, top_m=5):
    h = project_feasible(h0.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    step = 0.03
    for it in range(n_iter):
        if time.time() - t0 > t_budget:
            break
        h = subgradient_step(h, top_m, step)
        c = compute_c5(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        step = max(step * 0.995, 0.002)
    return best_h, best_c

def make_structured_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    n_ones = n // 2
    pattern = rng.choice(n, n_ones, replace=False)
    pattern = np.sort(pattern)
    h[pattern] = 1.0
    return project_feasible(h)

def sine_init(n):
    h = np.zeros(n)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * x / 2.0)
    h = np.clip(h, 0, 1)
    h = project_feasible(h)
    return h

def optimize_start(h0, t_budget):
    h = project_feasible(h0.copy())
    best_c = compute_c5(h)
    best_h = h.copy()
    t_phase1 = min(0.5 * t_budget, 200)
    for beta in [20, 50, 100, 200, 400]:
        rem = t_budget - (time.time() - (time.time() - t_budget))
        hp, cp = proj_grad_smooth(best_h, beta, 300, t_phase1)
        if cp < best_c:
            best_c = cp
            best_h = hp
    rem = t_budget - (time.time() - (time.time() - t_budget))
    hs, cs = subgradient_opt(best_h, 500, min(60, rem))
    if cs < best_c:
        best_c = cs
        best_h = hs
    rem = t_budget - (time.time() - (time.time() - t_budget))
    hn, cn = swap_local_search(best_h, 30, rem)
    if cn < best_c:
        best_c = cn
        best_h = hn
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 70

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    n_values = [70]
    if h_init is not None:
        n_orig = len(h_init)
        if n_orig not in n_values:
            n_values.insert(0, n_orig)
    for extra_n in [50, 60, 80, 100]:
        if extra_n not in n_values:
            n_values.append(extra_n)

    for n_try in n_values:
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 40:
            break
        t_alloc = min(0.25 * budget_s, 250)
        if n_try == (len(h_init) if h_init is not None else 70):
            t_alloc = min(0.4 * budget_s, 400)
        starts = []
        if h_init is not None and n_try == len(h_init):
            starts.append(h_init.copy())
        starts.append(sine_init(n_try))
        for si in range(3):
            starts.append(make_structured_init(n_try, seed + si * 137 + n_try))
        for si, h0 in enumerate(starts):
            t_rem = budget_s - (time.time() - t_start)
            if t_rem < 20:
                break
            t_phase = min(t_alloc / max(len(starts), 1), t_rem - 5)
            if t_phase < 10:
                continue
            try:
                hp, cp = optimize_start(h0, t_phase)
            except Exception:
                continue
            if cp < best_c5:
                best_c5 = cp
                best_h = hp
                best_n = n_try

    t_rem = budget_s - (time.time() - t_start)
    if best_h is not None and t_rem > 30:
        hn, cn = swap_local_search(best_h, 50, min(80, t_rem - 5))
        if cn < best_c5:
            best_c5 = cn
            best_h = hn

    t_rem = budget_s - (time.time() - t_start)
    if best_h is not None and t_rem > 20:
        for beta in [500, 1000, 2000]:
            t_rem = budget_s - (time.time() - t_start)
            if t_rem < 10:
                break
            hp, cp = proj_grad_smooth(best_h, beta, 200, min(30, t_rem - 3))
            if cp < best_c5:
                best_c5 = cp
                best_h = hp
            hn, cn = swap_local_search(best_h, 20, min(20, t_rem - 3))
            if cn < best_c5:
                best_c5 = cn
                best_h = hn

    if best_h is None:
        n = 70
        best_h = project_feasible(sine_init(n))
        best_n = n
        best_c5 = compute_c5(best_h)

    best_h = project_feasible(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, best_n)
