import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            d = min(diff, 1.0 - h[i])
            if d < 1e-16: break
            h[i] += d
        else:
            i = int(np.argmin(h))
            d = min(-diff, h[i])
            if d < 1e-16: break
            h[i] -= d
    return h

def _obj_grad(z, alpha, mu):
    n = len(z)
    h = 1.0 / (1.0 + np.exp(-np.clip(z, -50, 50)))
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    
    cmax = corr.max()
    if alpha > 1e4:
        w = np.full(2*n-1, 0.0)
        w[cmax.argmax()] = 1.0
        obj = cmax
    else:
        sh = np.clip(corr - cmax, -50.0/alpha, 0.0)
        e = np.exp(alpha * sh)
        w = e / e.sum()
        obj = cmax + np.log(e.sum()) / alpha
    
    # Gradient of corr w.r.t. h
    grad_h = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = m - n + 1  # shift
        # First term: (1-h[j+s]) for valid j
        lo1 = max(0, -s)
        hi1 = min(n - 1, n - 1 - s)
        if lo1 <= hi1:
            grad_h[lo1:hi1+1] += wm * (1.0 - h[lo1+s:hi1+1+s])
        # Second term: -h[j-s] for valid j
        lo2 = max(0, s)
        hi2 = min(n - 1, n - 1 + s)
        if lo2 <= hi2:
            grad_h[lo2:hi2+1] -= wm * h[lo2-s:hi2+1-s]
    grad_h *= dx
    
    # Penalty gradient
    penalty = mu * (h.sum() - n / 2.0) ** 2
    obj += penalty
    grad_h += 2.0 * mu * (h.sum() - n / 2.0)
    
    # Chain rule through sigmoid
    sz = h * (1.0 - h)
    grad_z = sz * grad_h
    
    return obj, grad_z

def _lbgfs(h0, alpha, mu, maxiter=300):
    n = len(h0)
    eps = 1e-10
    hc = np.clip(h0, eps, 1.0 - eps)
    z0 = np.log(hc / (1.0 - hc))
    
    try:
        res = minimize(_obj_grad, z0, args=(alpha, mu),
                       method='L-BFGS-B', jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_opt = 1.0 / (1.0 + np.exp(-np.clip(res.x, -50, 50)))
        return _fix_sum(h_opt, n)
    except Exception:
        return _fix_sum(h0.copy(), n)

def _run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None
    
    try:

    except Exception:
        h_init = None
    
    if h_init is not None:
        n = len(h_init)
        h_init = _fix_sum(h_init, n)
    else:
        n = 80
        h_init = np.full(n, 0.5)
    
    best_h = h_init.copy()
    best_c5 = _eval_c5(best_h)
    n = len(best_h)
    
    # Phase 1: Progressive alpha refinement
    alphas = [15, 30, 60, 100, 200, 400, 800, 1500]
    for a in alphas:
        if time.time() - t0 > budget_s - 60:
            break
        for mu in [1.0, 10.0, 50.0]:
            if time.time() - t0 > budget_s - 30:
                break
            h_opt = _lbgfs(best_h, a, mu, 250)
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                n = len(best_h)
    
    # Phase 2: Perturbation restarts
    for trial in range(40):
        if time.time() - t0 > budget_s - 40:

        for a in [60, 150, 400]:

            c5 = _eval_c5(h_opt)
            if c5 < best_c5:

    # Phase 3: Different resolutions
    for n2 in [40, 48, 56, 64, 96, 120, 160]:

        x_old = np.linspace(0, 2, n, endpoint=False)

        h_interp = np.interp(x_new, x_old, best_h, period=2)
        h0 = _fix_sum(h_interp, n2)
        for a in [40, 100, 300]:

                v = h[j] + dv
                if 0.0 <= v <= 1.0:
                    old = h[j]
                    h[j] = v
                    c = _eval_c5(h)
                    if c < best_c - 1e-12:
                        best_c = c
                        changed = True
                    else:
                        h[j] = old
        if not changed:
            break
    return _fix_sum(h, n), best_c

def _run(seed=42, budget_s=1100, **kw):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    bc, bh = np.inf, None
    try:
        hi = np.array(initial_h_values, dtype=float)
    except Exception:
        hi = None
    if hi is not None and len(hi) > 0:
        bh = _fix_sum(hi, len(hi))
    else:
        bh = np.full(80, 0.5)
    bc = _eval_c5(bh)
    n = len(bh)

    alphas = [15, 30, 50, 80, 120, 200, 350, 600, 1000, 2000]
    for a in alphas:
        if time.time() - t0 > budget_s - 80:
            break
        for mu in [1.0, 10.0, 50.0]:
            if time.time() - t0 > budget_s - 40:
                break
            ho = _opt(bh, a, mu, 250)
            c = _eval_c5(ho)
            if c < bc:
                bc, bh, n = c, ho, len(ho)

    for trial in range(60):
        if time.time() - t0 > budget_s - 50:
            break
        s = 0.003 + 0.03 * rng.random()
        hp = _fix_sum(bh + rng.normal(0, s, n), n)
        for a in [50, 120, 300]:
            if time.time() - t0 > budget_s - 25:
                break
            ho = _opt(hp, a, 50.0, 200)
            c = _eval_c5(ho)
            if c < bc:
                bc, bh, n = c, ho, len(ho)

    for n2 in [40, 48, 56, 64, 96, 100, 120, 160]:
        if time.time() - t0 > budget_s - 100:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h0 = _fix_sum(np.interp(xn, xo, bh, period=2.0), n2)
        for a in [40, 100, 300]:
            if time.time() - t0 > budget_s - 50:
                break
            ho = _opt(h0, a, 100.0, 250)
            c = _eval_c5(ho)
            if c < bc:
                bc, bh, n = c, ho, len(ho)

    if time.time() - t0 < budget_s - 30:
        bh, bc = _coord_refine(bh, n, rounds=5)
        bc = _eval_c5(bh)

    for a in [3000, 5000, 8000]:
        if time.time() - t0 > budget_s - 15:
            break
        ho = _opt(bh, a, 500.0, 300)
        c = _eval_c5(ho)
        if c < bc:
            bc, bh = c, ho

    if time.time() - t0 < budget_s - 10:
        bh, bc = _coord_refine(bh, len(bh), rounds=3)
        bc = _eval_c5(bh)

    bh = _fix_sum(bh, len(bh))
    bc = _eval_c5(bh)
    return bh, bc, len(bh)
