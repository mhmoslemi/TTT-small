    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):

        proj = np.clip(h - mid, 0.0, 1.0)

        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t = 0
    tau = 150.0
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    
    while time.time() < deadline - 0.5:
        t += 1
        lr = lr0 / (1 + t * 0.01)

        omh = 1.0 - h

import numpy as np
import time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -50, 50)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx

def adam_opt(h_init, time_budget, seed, lr0=0.05, tau0=200.0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(n)
    vv = np.zeros(n)
    b1, b2 = 0.8, 0.999
    t = 0
    tau = tau0
    t0 = time.time()
    deadline = t0 + time_budget
    no_imp = 0
    
    while time.time() < deadline - 0.3:
        t += 1
        lr = lr0 / (1.0 + 0.005 * t)
        
        grad = compute_soft_grad(h, tau)
        m = b1 * m + (1 - b1) * grad
        vv = b2 * vv + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        vv_hat = vv / (1 - b2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(vv_hat) + 1e-8)
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_imp = 0
        else:
            no_imp += 1
        
        h = h_new
        tau = min(tau * 1.003, 1000.0)
        
        if no_imp > 600:
            h = project_feasible(best_h + rng.randn(n) * 0.05, target_sum)
            m[:] = 0.0
            vv[:] = 0.0
            t = 0
            tau = tau0
            no_imp = 0
    
    return best_h, best_c5

def binary_ls2(h_frac, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    target = n // 2
    h = (h_frac >= 0.5).astype(float)
    diff = target - int(h.sum())
    if diff > 0:
        z = np.where(h == 0)[0]
        rng.shuffle(z)
        h[z[:diff]] = 1.0
    elif diff < 0:
        o = np.where(h == 1)[0]
        rng.shuffle(o)
        h[o[:(-diff)]] = 0.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    
    while time.time() < deadline - 0.3:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
            # 2-swap
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = ones[rng.randint(len(ones))], ones[rng.randint(len(ones))]
                j1, j2 = zeros[rng.randint(len(zeros))], zeros[rng.randint(len(zeros))]
                if i1 != i2 and j1 != j2:
                    h[i1], h[i2] = 0.0, 0.0
                    h[j1], h[j2] = 1.0, 1.0
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        best_h = h.copy()
                    else:
                        h[i1], h[i2] = 1.0, 1.0
                        h[j1], h[j2] = 0.0, 0.0
    
    return best_h, best_c5

        h[z[:diff]] = 1.0
    elif diff < 0:
        o = np.where(h == 1)[0]
        rng.shuffle(o)
        h[o[:(-diff)]] = 0.0

    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    while time.time() < deadline - 0.3:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
            if use_2swap and len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if i1 != i2 and j1 != j2:
                    h[i1], h[i2] = 0.0, 0.0
                    h[j1], h[j2] = 1.0, 1.0
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-14:
                        best_c5 = c5
                        best_h = h.copy()
                    else:
                        h[i1], h[i2] = 1.0, 1.0
                        h[j1], h[j2] = 0.0, 0.0
    return best_h, best_c5

def simulated_anneal(h_frac, time_budget, seed, T0=0.005, T1=0.00005):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    target = n // 2
    h = (h_frac >= 0.5).astype(float)
    diff = target - int(h.sum())
    if diff > 0:
        z = np.where(h == 0)[0]
        rng.shuffle(z)
        h[z[:diff]] = 1.0
    elif diff < 0:
        o = np.where(h == 1)[0]
        rng.shuffle(o)
        h[o[:(-diff)]] = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    t0 = time.time()
    deadline = t0 + time_budget
    n_iters = int((deadline - t0) * 60000)
    it = 0
    while time.time() < deadline - 0.5:
        it += 1
        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))
        if T < T1:
            T = T1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.rand() < 0.55:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.rand() < math.exp(-delta / max(T, 1e-12)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if i1 != i2 and j1 != j2:
                    h[i1], h[i2] = 0.0, 0.0
                    h[j1], h[j2] = 1.0, 1.0
                    new_c5 = compute_c5(h)
                    delta = new_c5 - cur_c5
                    if delta < 0 or rng.rand() < math.exp(-delta / max(T, 1e-12)):
                        cur_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                    else:
                        h[i1], h[i2] = 1.0, 1.0
                        h[j1], h[j2] = 0.0, 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    best_h = None
    best_c5 = float('inf')
    best_n = 70

    # Determine n values to try, prioritizing initial
    n_list = [70, 80, 64, 72, 90, 100]
    h_via_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_via_init = np.asarray(initial_h_values, dtype=float)
        n_init = len(h_via_init)
        if n_init not in n_list:
            n_list.insert(0, n_init)
        else:
            n_list.remove(n_init)
            n_list.insert(0, n_init)

    # Main optimization on primary n
    n_primary = n_list[0]
    if h_via_init is not None and n_primary == len(h_via_init):
        h_start = project_feasible(h_via_init.copy(), n_primary / 2.0)
    else:
        rng_s = np.random.RandomState(seed)
        h_start = project_feasible(np.full(n_primary, 0.5) + rng_s.randn(n_primary) * 0.1, n_primary / 2.0)

    # Phase 1: Adam on primary n
    t_a = min(250, deadline - time.time() - 300)
    if t_a > 20:
        h1, c5_1 = adam_opt(h_start, t_a, seed, lr0=0.06, tau0=200.0)
    else:
        h1, c5_1 = h_start, compute_c5(h_start)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c5_1, n_primary

    # Phase 2: Binary local search on primary
    t_b = min(80, deadline - time.time() - 200)
    if t_b > 15:
        hb, c5b = binary_ls(h1, t_b, seed + 1000, use_2swap=True)
        if c5b < best_c5:
            best_h, best_c5, best_n = hb.copy(), c5b, n_primary

    # Phase 3: Simulated annealing on primary (key improvement)
    t_sa = min(350, deadline - time.time() - 60)
    if t_sa > 30:
        hs, c5s = simulated_anneal(hb if 'hb' in dir() else h1, t_sa, seed + 2000, T0=0.004, T1=0.00003)
        if c5s < best_c5:
            best_h, best_c5, best_n = hs.copy(), c5s, n_primary
        # Second SA run with different params
        t_sa2 = min(150, deadline - time.time() - 30)
        if t_sa2 > 20:
            hs2, c5s2 = simulated_anneal(best_h, t_sa2, seed + 3000, T0=0.002, T1=0.00002)
            if c5s2 < best_c5:
                best_h, best_c5, best_n = hs2.copy(), c5s2, n_primary

    # Phase 4: Try secondary n values
    for n2 in n_list[1:4]:
        t_rem = deadline - time.time() - 20
        if t_rem < 50:
            break
        t_alloc = min(100, t_rem - 15)
        rng_s2 = np.random.RandomState(seed + n2 * 13)
        # Try a structured start: based on golden ratio
        golden = 0.618033988749895
        h_struct = np.array([1.0 if (i * golden) % 1.0 < 0.5 else 0.0 for i in range(n2)])
        h_struct = project_feasible(h_struct + rng_s2.randn(n2) * 0.05, n2 / 2.0)
        t_a2 = max(20, int(t_alloc * 0.5))
        h_opt2, c5_2 = adam_opt(h_struct, t_a2, seed + n2, lr0=0.05, tau0=150.0)
        t_b2 = max(10, int(t_alloc * 0.25))
        h_bin2, c5_bin2 = binary_ls(h_opt2, t_b2, seed + n2 + 500, use_2swap=True)
        t_s2 = max(10, int(t_alloc * 0.25))
        h_sa2, c5_sa2 = simulated_anneal(h_bin2, t_s2, seed + n2 + 900, T0=0.003, T1=0.00004)
        cands = [(h_opt2, c5_2), (h_bin2, c5_bin2), (h_sa2, c5_sa2)]
        for hc, cc in cands:
            if cc < best_c5:
                best_h, best_c5, best_n = hc.copy(), cc, n2

    # Phase 5: Final polish
    t_rem = deadline - time.time() - 2
    if t_rem > 10 and best_h is not None:
        h_p, c5_p = adam_opt(best_h, min(t_rem // 2, 30), seed + 77777, lr0=0.015, tau0=600.0)
        if c5_p < best_c5:
            best_h, best_c5 = h_p.copy(), c5_p
        t_b3 = deadline - time.time() - 1
        if t_b3 > 5:
            h_b3, c5_b3 = binary_ls(best_h, min(t_b3, 15), seed + 88888, use_2swap=True)
            if c5_b3 < best_c5:
                best_h, best_c5, best_n = h_b3.copy(), c5_b3, best_n

    best_c5 = compute_c5(best_h)
    n_final = len(best_h)
    return best_h, best_c5, n_final
