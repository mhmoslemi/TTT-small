from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary pattern with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Need to remove (n_ones - target) ones
        # Set positions closest to 0.5 first
        order = np.argsort(h_bin)  # zeros first, then ones
        # Among ones, set those with smallest h value to 0
        ones_idx = np.where(h_bin == 1)[0]
        order_ones = ones_idx[np.argsort(h[ones_idx])]
        for i in range(n_ones - target):
            h_bin[order_ones[i]] = 0.0
    elif n_ones < target:
        # Need to add (target - n_ones) ones
        zeros_idx = np.where(h_bin == 0)[0]
        order_zeros = zeros_idx[np.argsort(h[zeros_idx])[::-1]]
        for i in range(target - n_ones):
            h_bin[order_zeros[i]] = 1.0
    return h_bin

def binary_swap_search(h, n, t0, budget_s, max_iters=50000):
    """Local search by swapping 0↔1 pairs."""
    h = h.copy()
    c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_h = h.copy()
    best_c5 = c5
    
    rng = np.random.RandomState(42)
    T = 0.005
    T_min = 1e-7
    cool = 0.9997
    
    for it in range(max_iters):
        if time.time() - t0 > budget_s * 0.85:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick random swap
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c5 = new_c5
            # Update sets
            idx_in_ones = np.where(ones == a)[0]
            if len(idx_in_ones) > 0:
                ones = np.delete(ones, idx_in_ones[0])
            zeros = np.append(zeros, a)
            
            idx_in_zeros = np.where(zeros == b)[0]
            if len(idx_in_zeros) > 0:
                zeros = np.delete(zeros, idx_in_zeros[0])
            ones = np.append(ones, b)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
        
        T = max(T_min, T * cool)
    
    return best_h, best_c5

Let me also try: **Coordinate descent on the continuous problem**
- At each step, pick coordinate j
- Optimize h[j] (given all other h[i] fixed) to minimize C₅
- Since C₅ = max_k c_k, and c_k is linear in h[j] (when other variables fixed), the optimal h[j] is determined by which k is active

We want to minimize max_k (const_k + s_k·h[j]) over h[j] ∈ [0,1] subject to sum constraint.

Without the sum constraint, the optimal h[j] is:
- If all s_k < 0: h[j] = 1 (all c_k decrease)
- If all s_k > 0: h[j] = 0 (all c_k increase, so minimize by setting to 0)
- If mixed: h[j] = value that equalizes the max

This is a 1D problem: minimize max_k (a_k + b_k·x) over x ∈ [0,1], which can be solved analytically.

This coordinate descent is VERY fast per iteration (O(n) to compute all slopes, O(n log n) to solve the 1D subproblem).

def coord_descent_step(h, j, n):
    """Optimize h[j] given all other coordinates fixed."""
    dx = 2.0 / n
    # Compute c_k for all k, and the slope of c_k w.r.t. h[j]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # corr has 2n-1 entries, indexed from -(n-1) to (n-1)
    # corr[0] corresponds to lag -(n-1), corr[n-1] to lag 0, corr[2n-2] to lag n-1
    
    # Slope of c_k w.r.t. h[j]:
    # c_k = sum_i h[i]*(1-h[i+k])*dx
    # h[j] appears as h[i] when i=j (if j+k < n): contributes (1-h[j+k])*dx
    # h[j] appears as h[i+k] when i+k=j, i=j-k (if j-k >= 0): contributes -h[j-k]*dx
    
    slope = np.zeros(2*n-1)
    # For lag k (CORRELATION lag), h[j] contributes:
    # As first arg at position i=j: valid if 0 <= j < n and 0 <= j+k < n
    #   => contributes (1-h[j+k])*dx to c_k
    # As second arg at position i+k=j, i=j-k: valid if 0 <= j-k < n and 0 <= j < n
    #   => contributes -h[j-k]*dx to c_k
    
    # But we need to map to the correlate output index.
    # np.correlate(h, 1-h, mode="full")[idx] corresponds to lag (idx - (n-1))
    # For lag k: idx = k + (n-1)
    
    h_orig = h[j]
    # Slope for each lag k (from -(n-1) to n-1):
    # h[j] as first element of h at position j:
    #   appears in sum when i=j, so we need j+k in [0,n-1], i.e., k in [-j, n-1-j]
    #   contribution: (1-h[j+k])*dx
    # h[j] as second element (1-h) at position j, meaning h[i+k]=h[j] where i=j-k:
    #   appears when i=j-k in [0,n-1], i.e., k in [j-n+1, j]
    #   contribution: -h[j-k]*dx (because d(1-h[j])/dh[j] = -1, times h[i]=h[j-k])
    
    # Let's compute the slope array directly:
    # For each lag k in [-(n-1), n-1]:
    k_values = np.arange(-(n-1), n)
    slope = np.zeros(len(k_values))
    
    # h[j] as h[i] (first factor): valid when 0<=j<n (always) and 0<=j+k<n
    mask1 = (j + k_values >= 0) & (j + k_values < n)
    slope[mask1] += (1.0 - h[j + k_values[mask1]]) * dx
    
    # h[j] as h[i+k] (second factor, inside 1-h): valid when 0<=j-k<n and 0<=j<n
    mask2 = (j - k_values >= 0) & (j - k_values < n)
    slope[mask2] -= h[j - k_values[mask2]] * dx
    
    # Now minimize max_k (corr[k] + slope[k]*x) for x in [0,1]
    # where corr[k] is the current value (computed with old h[j])
    # and the change is slope[k] * (x - h_orig)
    # So: minimize max_k (corr[k] + slope[k]*(x - h_orig))
    
    base = corr + slope * (0 - h_orig)  # value at x=0
    # c_k(x) = base_k + slope_k * x
    
    # Find x in [0,1] that minimizes max_k (base_k + slope_k * x)
    # This is: minimize t s.t. base_k + slope_k * x <= t for all k, 0<=x<=1
    # Optimal x: find where the max is minimized
    
    best_x = h_orig  # default: don't change
    best_val = np.max(corr)
    
    # Try x=0, x=1, and the intersection points
    candidates = [0.0, 1.0, h_orig]
    
    # For pairs of lines (base_i + slope_i*x) and (base_j + slope_j*x):
    # They intersect at x = (base_j - base_i)/(slope_i - slope_j)
    # Only add intersections in (0,1)
    # But this is O(n^2) which is too expensive for n=320 (100K pairs)
    # Instead, use a simpler approach: ternary search or just evaluate at a few points
    
    # Actually, the function max_k(a_k + b_k*x) is convex in x (max of linear functions)
    # So we can use bisection on the derivative (subgradient)
    # Or just use the fact that the minimum is at a point where two lines cross
    
    # Simpler: evaluate at x where the current max is achieved, and at boundaries
    # The subgradient at x is {slope_k : c_k(x) = max}
    # At optimum, 0 is in the subgradient (or x is at boundary)

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def coord_descent_fast(h, n, t0, time_budget):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    best_h = h.copy()
    best_c5 = corr.max()
    k_vals = np.arange(-(n - 1), n)
    for p in range(1000):
        if time.time() - t0 > time_budget:
            break
        for j in range(n):
            slope = np.zeros(2 * n - 1)
            slope[n - 1 - j: 2 * n - 1 - j] += (1.0 - h) * dx
            slope[j: j + n] -= h[::-1] * dx
            base = corr - slope * h[j]
            lo, hi = 0.0, 1.0
            for _ in range(18):
                mid = (lo + hi) / 2.0
                vals = base + slope * mid
                m_val = vals.max()
                active = vals >= m_val - 1e-12
                if not np.any(active):
                    active[np.argmax(vals)] = True
                sg = slope[active].mean()
                if sg > 1e-10:
                    lo = mid
                elif sg < -1e-10:
                    hi = mid
                else:
                    lo = hi = mid
                    break
            x_opt = (lo + hi) / 2.0
            if abs(x_opt - h[j]) > 1e-12:
                corr += slope * (x_opt - h[j])
                h[j] = x_opt
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        c5_now = corr.max()
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def binary_sa_fast(h_bin, n, t0, time_budget, seed=42):
    dx = 2.0 / n
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c5 = float(corr.max())
    best_h = h.copy()
    best_c5 = c5
    rng = np.random.RandomState(seed)
    ones_pos = np.where(h == 1)[0].copy()
    zeros_pos = np.where(h == 0)[0].copy()
    n_half = n // 2
    T = 0.004
    T_min = 1e-9
    n_iter = 0
    max_iter = 500000
    while n_iter < max_iter and time.time() - t0 < time_budget:
        ia = rng.randint(n_half)
        ib = rng.randint(n_half)
        a = ones_pos[ia]
        b = zeros_pos[ib]
        delta_corr = np.zeros(2 * n - 1)
        s1 = n - 1 - a
        delta_corr[s1: s1 + n] -= (1.0 - h) * dx
        s2 = a
        delta_corr[s2: s2 + n] += h[::-1] * dx
        s3 = n - 1 - b
        delta_corr[s3: s3 + n] += (1.0 - h) * dx
        s4 = b
        delta_corr[s4: s4 + n] -= h[::-1] * dx
        new_c5 = float((corr + delta_corr).max())
        delta_c5 = new_c5 - c5
        if delta_c5 < 0 or (delta_c5 > 0 and rng.rand() < np.exp(-delta_c5 / T)):
            h[a] = 0.0
            h[b] = 1.0
            corr += delta_corr
            c5 = new_c5
            ones_pos[ia] = b
            zeros_pos[ib] = a
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        n_iter += 1
        if n_iter % 200 == 0:
            T = max(T_min, T * 0.9997)
    return best_h, best_c5

def greedy_binary(h_bin, n, t0, time_budget):
    dx = 2.0 / n
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    best_c5 = float(corr.max())
    best_h = h.copy()
    n_half = n // 2
    for phase in range(200):
        if time.time() - t0 > time_budget:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        best_delta = 0.0
        best_swap = None
        for ia in range(n_half):
            a = ones_idx[ia]
            for ib in range(n_half):
                b = zeros_idx[ib]
                dc = np.zeros(2 * n - 1)
                s1 = n - 1 - a
                dc[s1: s1 + n] -= (1.0 - h) * dx
                s2 = a
                dc[s2: s2 + n] += h[::-1] * dx
                s3 = n - 1 - b
                dc[s3: s3 + n] += (1.0 - h) * dx
                s4 = b
                dc[s4: s4 + n] -= h[::-1] * dx
                new_m = float((corr + dc).max())
                d = new_m - float(corr.max())
                if d < best_delta - 1e-15:
                    best_delta = d
                    best_swap = (ia, ib)
        if best_swap is None:
            break
        ia, ib = best_swap
        a = ones_idx[ia]
        b = zeros_idx[ib]
        dc = np.zeros(2 * n - 1)
        s1 = n - 1 - a
        dc[s1: s1 + n] -= (1.0 - h) * dx
        s2 = a
        dc[s2: s2 + n] += h[::-1] * dx
        s3 = n - 1 - b
        dc[s3: s3 + n] += (1.0 - h) * dx
        s4 = b
        dc[s4: s4 + n] -= h[::-1] * dx
        h[a] = 0.0
        h[b] = 1.0
        corr += dc
        c5_now = float(corr.max())
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def soft_obj(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def constraint_fun(x):
    return np.sum(x) - len(x) / 2.0

def constraint_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, t0, time_budget):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    bounds = [(0.0, 1.0)] * n
    for beta in [10, 50, 200, 1000]:
        if time.time() - t0 > time_budget:
            break
        try:
            res = minimize(soft_obj, best_h, args=(n, beta), method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n0 = 320
    best_h = None
    best_c5 = np.inf
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n0 = len(initial_h_values)
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.full(n0, 0.5)
    h_start = project_feasible(h_start, n0)
    # Phase 1: Continuous coordinate descent
    t_phase1 = t0 + 120
    h_cd, c5_cd = coord_descent_fast(h_start, n0, t0, t_phase1)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd
    # Phase 2: SLSQP from CD result
    if time.time() - t0 < budget_s * 0.25:
        h_s, c5_s = slsqp_optimize(h_cd, n0, t0, t0 + 100)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    # Phase 3: Binary + SA
    h_bin = round_to_binary(best_h, n0)
    t_phase3 = min(t0 + budget_s * 0.55, time.time() + 400)
    h_sa, c5_sa = binary_sa_fast(h_bin, n0, t0, t_phase3, seed=seed)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa
    # Phase 4: Greedy local search on binary
    if time.time() - t0 < budget_s * 0.8:
        h_g, c5_g = greedy_binary(best_h, n0, t0, t0 + budget_s * 0.8)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g
    # Phase 5: More SA runs with different seeds
    for sa_seed in [7, 99, 2024, 31415]:
        if time.time() - t0 > budget_s * 0.9:
            break
        h_sa2, c5_sa2 = binary_sa_fast(best_h, n0, t0, t0 + 120, seed=sa_seed)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2
    # Phase 6: Greedy again
    if time.time() - t0 < budget_s * 0.92:
        h_g2, c5_g2 = greedy_binary(best_h, n0, t0, t0 + 80)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2
    # Phase 7: Try SLSQP on the best (continuous relaxation of binary)
    if time.time() - t0 < budget_s * 0.95:
        h_fine = best_h + rng.randn(n0) * 0.05
        h_fine = project_feasible(h_fine, n0)
        h_s2, c5_s2 = slsqp_optimize(h_fine, n0, t0, t0 + 60)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2
    # Final
    best_h = project_feasible(best_h, n0)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    try:
        val = evaluate_erdos_solution(best_h)
        if val < final_c5:
            final_c5 = val
    except Exception:
        pass
    return (best_h, final_c5, n0)
