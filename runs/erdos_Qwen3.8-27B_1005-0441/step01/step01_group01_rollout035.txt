```python
    n = len(h)
    dx = 2.0 / n
    # Compute all shifts
    # c[s] for s = -(n-1), ..., n-1, stored as c[m] where m = s + (n-1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr has length 2n-1, indexed m = 0, ..., 2n-2
    # s = m - (n-1)
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    weights /= np.sum(weights)
```

```python
    # For each coordinate i, find optimal h[i]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        if resid > 0:
            order = np.argsort(-h)  # most room to increase first
            for j in order:
                adj = min(resid, 1.0 - h[j])
                if adj <= 0:
                    break
                h[j] += adj
                resid -= adj
                if abs(resid) < 1e-15:
                    break
        else:
            order = np.argsort(h)  # least to decrease first
            for j in order:
                adj = max(resid, -h[j])
                if adj >= 0:
                    break
                h[j] += adj
                resid -= adj
                if abs(resid) < 1e-15:
                    break
    return h

def smooth_obj(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    pen = lam * (np.sum(h) - n * 0.5) ** 2
    return sm * dx + pen

def smooth_grad(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-16:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
        else:
            grad[:n+s] += w * (1.0 - h[-s:])
            grad[-s:] -= w * h[:n+s]
    grad *= dx
    grad += 2.0 * lam * (np.sum(h) - n * 0.5)
    return grad

def coord_descent(h, n, n_sweeps=5):
    s_all = np.arange(-(n-1), n)
    for _ in range(n_sweeps):
        c = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            idx1 = i - s_all
            mask1 = (idx1 >= 0) & (idx1 < n)
            term1 = np.zeros(2*n-1)
            term1[mask1] = 1.0 - h[idx1[mask1]]
            idx2 = i + s_all
            mask2 = (idx2 >= 0) & (idx2 < n)
            term2 = np.zeros(2*n-1)
            term2[mask2] = h[idx2[mask2]]
            delta = term1 - term2
            c0 = c - h[i] * delta
            v0 = np.max(c0)
            v1 = np.max(c0 + delta)
            best_t = 0.0
            best_v = v0
            if v1 < best_v:
                best_t = 1.0
                best_v = v1
            ts = np.linspace(0, 1, 11)
            for t in ts[1:-1]:
                v = np.max(c0 + t * delta)
                if v < best_v:
                    best_v = v
                    best_t = t
            lo = max(0.0, best_t - 0.1)
            hi = min(1.0, best_t + 0.1)
            for t in np.linspace(lo, hi, 21):
                v = np.max(c0 + t * delta)
                if v < best_v:
                    best_v = v
                    best_t = t
            h[i] = best_t
            c = c0 + best_t * delta
        h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(160)
        h0[:40] = 1.0
        h0[-40:] = 1.0
        h0 = project_feasible(h0, 160)
    
    n = len(h0)
    h_best = project_feasible(h0.copy(), n)
    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n
    
    # Phase 1: L-BFGS-B with smooth max, alpha annealing
    alphas = [20.0, 50.0, 100.0, 200.0, 400.0]
    for alpha in alphas:
        if time.time() - t0 > budget_s * 0.4:
            break
        lam = 5.0
        res = minimize(
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        if resid > 0:
            order = np.argsort(-h)
        else:
            order = np.argsort(h)
        for j in order:
            if resid > 0:
                adj = min(resid, 1.0 - h[j])
            else:
                adj = max(resid, -h[j])
            if abs(adj) < 1e-16:
                continue
            h[j] += adj
            resid -= adj
            if abs(resid) < 1e-15:
                break
    return h

def smooth_obj(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    pen = lam * (np.sum(h) - n * 0.5) ** 2
    return sm * dx + pen

def smooth_grad(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-16:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            grad[:n + s] += w * (1.0 - h[-s:])
            grad[-s:] -= w * h[:n + s]
    grad *= dx
    grad += 2.0 * lam * (np.sum(h) - n * 0.5)
    return grad

def coord_descent(h, n, n_sweeps=5):
    s_all = np.arange(-(n - 1), n)
    for _ in range(n_sweeps):
        c = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            idx1 = i - s_all
            mask1 = (idx1 >= 0) & (idx1 < n)
            term1 = np.zeros(2 * n - 1)
            term1[mask1] = 1.0 - h[idx1[mask1]]
            idx2 = i + s_all
            mask2 = (idx2 >= 0) & (idx2 < n)
            term2 = np.zeros(2 * n - 1)
            term2[mask2] = h[idx2[mask2]]
            delta = term1 - term2
            c0 = c - h[i] * delta
            best_t = 0.0
            best_v = np.max(c0)
            v1 = np.max(c0 + delta)
            if v1 < best_v:
                best_t = 1.0
                best_v = v1
            for t in np.linspace(0.0, 1.0, 13)[1:-1]:
                v = np.max(c0 + t * delta)
                if v < best_v:
                    best_v = v
                    best_t = t
            lo = max(0.0, best_t - 0.08)
            hi = min(1.0, best_t + 0.08)
            for t in np.linspace(lo, hi, 33):
                v = np.max(c0 + t * delta)
                if v < best_v:
                    best_v = v
                    best_t = t
            h[i] = best_t
            c = c0 + best_t * delta
        h = project_feasible(h, n)
    return h

                c5_best = c5c
                h_best = hc.copy()

    # Phase 5: Final polish - heavy coord descent + one more L-BFGS
    if time.time() - t0 < budget_s - 15:
        h_cd = coord_descent(h_best.copy(), n, n_sweeps=15)
        c5cd = compute_c5(h_cd)
        if c5cd < c5_best:
            c5_best = c5cd
            h_best = h_cd.copy()
        if time.time() - t0 < budget_s - 5:
            res = minimize(smooth_obj, h_best.copy(), jac=smooth_grad,
                           args=(800.0, 10.0), method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-14})
            hc = project_feasible(res.x, n)
            c5c = compute_c5(hc)
            if c5c < c5_best:
                c5_best = c5c
                h_best = hc.copy()
            h_cd = coord_descent(h_best.copy(), n, n_sweeps=8)
            c5cd = compute_c5(h_cd)
            if c5cd < c5_best:
                c5_best = c5cd
                h_best = h_cd.copy()

    h_best = project_feasible(h_best, n)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
```
