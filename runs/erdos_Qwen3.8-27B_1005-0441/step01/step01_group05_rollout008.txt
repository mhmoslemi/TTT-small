```python
def _softmax_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = max(temp, 1e-8)
    m = np.max(corr)
    return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx

def _project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} approximately."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Simple scaling
    if s > 0:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    # If still infeasible, adjust
    s = np.sum(h)
    diff = target - s
    if diff > 0:
        # Need to increase some entries
        room = 1.0 - h
        idx = np.argsort(-room)
        acc = 0.0
        for i in range(n):
            add = min(diff - acc, room[idx[i]], 0.1)
            if add <= 0:

def _block_init(n, seed):
    """Two blocks of 1s with a gap."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    half = n_ones // 2
    gap = rng.randint(n // 8, n // 3)
    pos = rng.randint(0, max(1, n - 2 * half - gap))
    h[pos:pos + half] = 1.0
    pos2 = (pos + half + gap) % n
    if pos2 + half <= n:
        h[pos2:pos2 + half] = 1.0
    else:

    # Fix sum
    cnt = np.sum(h)
    if cnt != n_ones:

def _gauss_init(n, seed, n_pulses=4, width=0.15):

def _optimize_lbfgs(h_init, n, temp_start, n_iters, time_limit):
    """L-BFGS-B optimization with softmax objective."""
    t0 = time.time()
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _softmax_obj(x, n, temp_start)

        return _generate_gradient(x, n, temp_start)
    
    best = h.copy()
    best_val = compute_c5(best)
    
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,

            h_new = _project_feasible(res.x, n)
            v = compute_c5(h_new)
            if v < best_val:
                best = h_new.copy()
                best_val = v
            h = h_new
        except Exception:
```

```python
def _generate_gradient(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = max(temp, 1e-8)
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / t)
    probs = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    # corr[k] = sum_{i: valid} h[i+d] * (1-h[i]) where d = k-(n-1)
    # d corr[k] / d h[j]:
    #   From h[j] in the first factor (j = i+d): (1-h[j-d]) if 0 <= j-d < n
    #   From h[j] in the (1-h[i]) factor (j = i): -h[j+d] if 0 <= j+d < n
    
    # For efficiency, vectorize over k:
    for d in range(-(n-1), n):
        k = d + (n - 1)
        w = probs[k] * dx
        if w < 1e-15:
            continue
        # Range of i: max(0,-d) <= i <= min(n-1, n-1-d)
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx_i = np.arange(i_lo, i_hi + 1)
        idx_j = idx_i + d  # j = i + d
        
        # First term: grad[j] += w * (1 - h[i])
        np.add.at(grad, idx_j, w * (1.0 - h[idx_i]))
        # Second term: grad[i] += w * (-h[j])
        np.add.at(grad, idx_i, -w * h[idx_j])
    
    return grad
```

```python
    max_val = np.max(corr) * dx
    
    # Find active lags (within tolerance of max)
    active = corr >= np.max(corr) - 1e-3
    
    # Average gradient over active lags
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        k = d + (n-1)
        if not active[k]:
            continue
        i_lo = max(0, -d)
        i_hi = min(n-1, n-1-d)
        if i_lo > i_hi:
            continue
        idx_i = np.arange(i_lo, i_hi+1)
        idx_j = idx_i + d
        grad[idx_j] += (1.0 - h[idx_i]) * dx
        grad[idx_i] -= h[idx_j] * dx
    
    # Gradient descent step
    h_new = h - lr * grad
    h_new = _project_feasible(h_new, n)
    return h_new, max_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    if s > 1e-12:
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if diff > 1e-12:
        room = 1.0 - h
        idx = np.argsort(-room)
        acc = 0.0
        for i in idx:
            if acc >= diff:
                break
            add = min(diff - acc, room[i])
            h[i] += add
            acc += add
    elif diff < -1e-12:
        room = h.copy()
        idx = np.argsort(-room)
        acc = 0.0
        for i in idx:
            if acc >= -diff:
                break
            sub = min(-diff - acc, room[i])
            h[i] -= sub
            acc += sub
    return h

def _corr_gradient(h, n, weights):
    """Gradient of sum_k weights[k]*corr[k] w.r.t. h."""
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        k = d + (n - 1)
        w = weights[k]
        if abs(w) < 1e-16:
            continue
        i_lo = max(0, -d)
        i_hi = min(n - 1, n - 1 - d)
        if i_lo > i_hi:
            continue
        idx_i = np.arange(i_lo, i_hi + 1)
        idx_j = idx_i + d
        np.add.at(grad, idx_j, w * (1.0 - h[idx_i]))
        np.add.at(grad, idx_i, -w * h[idx_j])
    return grad

def _softmax_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = max(temp, 1e-8)
    m = np.max(corr)
    return (m + t * np.log(np.sum(np.exp((corr - m) / t)))) * dx

def _softmax_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = max(temp, 1e-8)
    m = np.max(corr)
    e = np.exp((corr - m) / t)
    weights = e / np.sum(e) * dx
    return _corr_gradient(h, n, weights)

def _active_grad(h, n, tol=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(corr)
    active = corr >= cm - tol
    weights = np.zeros(2 * n - 1)
    weights[active] = dx / np.sum(active)
    return _corr_gradient(h, n, weights)

def _optimize(h_init, n, temps, time_limit, max_iter_per_temp=40):
    t0 = time.time()
    h = h_init.copy()
    best = h.copy()
    best_val = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    for temp in temps:
        if time.time() - t0 > time_limit:
            break
        def obj(x):
            return _softmax_obj(x, n, temp)
        def jac(x):
            return _softmax_grad(x, n, temp)
        for it in range(max_iter_per_temp):
            if time.time() - t0 > time_limit:
                break
            try:
                res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': 60, 'ftol': 1e-15, 'gtol': 1e-12})
                h_new = _project_feasible(res.x, n)
                v = compute_c5(h_new)
                if v < best_val - 1e-12:
                    best = h_new.copy()
                    best_val = v
                h = h_new
            except Exception:
                break
    # Active-set polish
    if time.time() - t0 < time_limit - 5:
        for _ in range(20):
            if time.time() - t0 > time_limit - 2:
                break
            g = _active_grad(h, n, tol=0.005)
            for lr in [0.05, 0.02, 0.01, 0.005]:
                h_try = _project_feasible(h - lr * g, n)
                if compute_c5(h_try) < best_val:
                    h = h_try
                    best = h.copy()
                    best_val = compute_c5(best)
                    break
    return best, best_val

def _binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.permutation(n)[:n // 2]] = 1.0
    return h

def _two_block_init(n, seed, gap_frac=0.25):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    half = n // 4
    gap = max(1, int(n * gap_frac))
    pos = rng.randint(0, max(1, n - 2 * half - gap - 1))
    h[pos:pos + half] = 1.0
    p2 = pos + half + gap
    if p2 + half <= n:
        h[p2:p2 + half] = 1.0
    else:
        h[p2:] = 1.0
        h[:p2 + half - n] = 1.0
    return _project_feasible(h, n)

def _multi_pulse_init(n, seed, npulses=4, width=0.1):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for c in rng.uniform(0.1, 1.9, npulses):
        h += np.exp(-((x - c) / width) ** 2)
    h = np.clip(h, 0, 1)
    return _project_feasible(h, n)

    t_start = time.time()
```
