            old = h[i]
            for delta in [0.02, -0.02, 0.01, -0.01, 0.05, -0.05]:
                h[i] = np.clip(old + delta, 0, 1)
                h = _proj(h, n)
                v = _c5(h)
                if v < best_val - 1e-14:
                    best_val = v
                    improved = True
                    break
            h[i] = old
            h = _proj(h, n)
    return h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        h0 = _proj(h0, n0)
    except Exception:
        n0 = 70
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_val = _c5(h0)

    # Phase 1: SLSQP on original grid, decreasing epsilon
    eps_schedule = [0.06, 0.03, 0.015, 0.008, 0.004, 0.002]
    for i, eps in enumerate(eps_schedule):
        tf = 0.15 + 0.08 * i
        if time.time() - t0 > budget_s * tf:
            break
        h_res = _slsqp(best_h, eps, 600, t0, budget_s, tf + 0.08)
        v = _c5(h_res)
        if v < best_val:
            best_val = v
            best_h = h_res

    # Phase 2: Upscale to finer grids
    for n_new in [140, 200]:
        if time.time() - t0 > budget_s * 0.55:
            break
        h_up = _upscale(best_h, n_new)
        for eps in [0.025, 0.01, 0.005, 0.003]:
            if time.time() - t0 > budget_s * 0.70:
                break
            h_res = _slsqp(h_up, eps, 400, t0, budget_s, 0.72)
            v = _c5(h_res)
            if v < best_val:
                best_val = v
                best_h = h_res
            h_up = h_res

    # Phase 3: Try perturbations around best
    if time.time() - t0 < budget_s * 0.78:
        n_best = len(best_h)
        for attempt in range(8):
            if time.time() - t0 > budget_s * 0.78:
                break
            h_pert = best_h + rng.randn(n_best) * 0.03
            h_pert = _proj(h_pert, n_best)
            h_res = _slsqp(h_pert, 0.005, 300, t0, budget_s, 0.80)
            v = _c5(h_res)
            if v < best_val:
                best_val = v
                best_h = h_res

    # Phase 4: Subgradient on best
    if time.time() - t0 < budget_s * 0.82:
        h_sg, v_sg = _subgrad(best_h, t0, budget_s, 0.92)
        if v_sg < best_val:
            best_val = v_sg
            best_h = h_sg

    # Phase 5: Coordinate descent polish
    if time.time() - t0 < budget_s * 0.90:
        h_cd, v_cd = _coord_descent(best_h, t0, budget_s, 0.95)
        if v_cd < best_val:
            best_val = v_cd
            best_h = h_cd

    # Phase 6: Final SLSQP polish with tiny epsilon
    if time.time() - t0 < budget_s * 0.94:
        for eps in [0.0015, 0.001]:
            if time.time() - t0 > budget_s * 0.96:
                break
            h_res = _slsqp(best_h, eps, 400, t0, budget_s, 0.97)
            v = _c5(h_res)
            if v < best_val:
                best_val = v
                best_h = h_res

    # Phase 7: Final subgradient sweep
    if time.time() - t0 < budget_s * 0.97:
        h_sg2, v_sg2 = _subgrad(best_h, t0, budget_s, 0.995)
        if v_sg2 < best_val:
            best_val = v_sg2
            best_h = h_sg2

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
