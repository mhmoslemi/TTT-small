import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _lse_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    return sm * dx

def _pen_obj(x, alpha, pen):
    n = len(x)
    return _lse_obj(x, alpha) + pen * (np.sum(x) - n * 0.5) ** 2

def _topk_obj(x, top_idx, alpha, weight):
    n = len(x)
    dx = 2.0 / n
    base = _lse_obj(x, alpha)
    corr = np.correlate(x, 1.0 - x, mode='full')
    tv = corr[top_idx]
    penalty = weight * np.sum((tv - np.min(tv)) ** 2) * dx
    return base + penalty

def _sum_eq(x):
    return np.sum(x) - len(x) * 0.5

def _sum_jac(x):
    return np.ones(len(x))

def make_oscillatory(n, period, amp=0.5):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + amp * np.cos(2.0 * np.pi * x / period)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def make_blocks(n, nb, frac):
    h = np.zeros(n)
    bs = max(1, int(n * frac / nb))
    for i in range(nb):
        st = int(i * n / nb)
        h[st:st+bs] = 1.0
    return project_feasible(h, n)

def make_two_block(n):
    h = np.zeros(n)
    b = n // 4
    h[:b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def subgradient_step(h, top_k=5, step=0.01):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for m in idx:
        shift = m - (n - 1)
        g = np.zeros(n)
        if shift > 0:
            g[:n-shift] += (1.0 - h[shift:])
            g[shift:] += -h[:n-shift]
        elif shift < 0:
            g[-shift:] += (1.0 - h[:n+shift])
            g[:n+shift] += -h[-shift:]
        else:
            g += 1.0 - 2.0 * h
        grad += g
    grad /= top_k
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-12:
        h = h - step * grad / gnorm
    return h, corr[idx]

    c5_best = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        h += d / n
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _lse_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    sm = m + np.log(np.sum(np.exp(shifted))) / alpha
    return sm * dx

def _pen_obj(x, alpha, pen):
    n = len(x)
    return _lse_obj(x, alpha) + pen * (np.sum(x) - n * 0.5) ** 2

def _topk_obj(x, top_idx, alpha, weight):
    n = len(x)
    dx = 2.0 / n
    base = _lse_obj(x, alpha)
    corr = np.correlate(x, 1.0 - x, mode='full')
    tv = corr[top_idx]
    penalty = weight * np.sum((tv - np.min(tv)) ** 2) * dx
    return base + penalty

    return np.sum(x) - len(x) * 0.5

    return np.ones(len(x))

def make_blocks(n, nb, frac):
    h = np.zeros(n)
    bs = max(1, int(n * frac / nb))
    for i in range(nb):
        st = int(i * n / nb)
        h[st:st + bs] = 1.0
    return project_feasible(h, n)

def make_two_block(n):
    h = np.zeros(n)
    b = n // 4
    h[:b] = 1.0
    h[-b:] = 1.0
    return project_feasible(h, n)

def subgradient_step(h, top_k=5, step=0.01):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for m in idx:
        shift = m - (n - 1)
        g = np.zeros(n)
        if shift > 0:
            g[:n - shift] += (1.0 - h[shift:])
            g[shift:] += -h[:n - shift]
        elif shift < 0:
            g[-shift:] += (1.0 - h[:n + shift])
            g[:n + shift] += -h[-shift:]
        else:
            g += 1.0 - 2.0 * h
        grad += g
    grad /= top_k
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-12:
        h = h - step * grad / gnorm
    return h, corr[idx]
