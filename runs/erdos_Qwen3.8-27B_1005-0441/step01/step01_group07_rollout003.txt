def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_corr(h, dx, tau, top_m):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

        sh = np.clip(tau * (corr - corr.max()), -100, 100)
        e = np.exp(sh)

        else:

            g[L:] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[L:]
    return g * dx

    t3 = min(180, int(t_rem * 0.2))
    if t3 > 15:
        h3, c53 = bin_sa(best_h, t3, seed + 100)
        if c53 < best_c5:
            best_h, best_c5 = h3, c53

    # Phase 4: try n=100 via interpolation
    t_rem = deadline - time.time()
    if t_rem > 250:
        n4 = 100
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n4)
        h_int = np.interp(x_new, x_old, best_h)
        h_int = project_feasible(h_int, n4 / 2.0)
        t4a = min(35, int(t_rem * 0.1))
        if t4a > 8:
            h4f, _ = opt_frac(h_int, t4a, seed + 200)
            h4b = to_bin(h4f, n4)
            t4b = min(200, int(t_rem * 0.35))
            if t4b > 20:
                h4, c54 = bin_greedy(h4b, t4b, seed + 300)
                if c54 < best_c5:
                    best_h, best_c5 = h4, c54

    # Phase 5: try n=140 if time permits
    t_rem = deadline - time.time()
    if t_rem > 200:
        n5 = 140
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n5)
        h_int5 = np.interp(x_new, x_old, best_h)
        h_int5 = project_feasible(h_int5, n5 / 2.0)
        t5a = min(30, int(t_rem * 0.1))
        if t5a > 8:
            h5f, _ = opt_frac(h_int5, t5a, seed + 400)
            h5b = to_bin(h5f, n5)
            t5b = min(140, int(t_rem * 0.35))
            if t5b > 15:
                h5, c55 = bin_greedy(h5b, t5b, seed + 500)
                if c55 < best_c5:
                    best_h, best_c5 = h5, c55

    # Phase 6: final polish greedy
    t_rem = deadline - time.time()
    if t_rem > 25:
        nf = len(best_h)
        hf = to_bin(best_h, nf)
        hp, cpp = bin_greedy(hf, min(t_rem - 2, 100), seed + 999)
        if cpp < best_c5:
            best_h, best_c5 = hp, cpp

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
