def project_binary(h, n):
    """Project h to binary with exactly n/2 ones."""
    target = n // 2
    # Sort indices by value, keep top n/2 as 1
    if target == 0:
        return np.zeros(n)
    if target == n:
        return np.ones(n)
    # Use threshold: values above median threshold become 1
    threshold = np.sort(h)[n - target]
    h_bin = (h >= threshold).astype(float)
    # If we have too many 1s (ties at threshold), fix
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        # Demote some
        idx_ones = np.where(h_bin == 1)[0]
        # Sort by h value descending, keep top target
        order = np.argsort(-h[idx_ones])
        for i in range(n_ones - target):
            h_bin[idx_ones[order[i]]] = 0
    elif n_ones < target:
        idx_zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx_zeros])
        for i in range(target - n_ones):
            h_bin[idx_zeros[order[i]]] = 1
    return h_bin

def c5_of(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_to_binary(h, n):
    target = n // 2
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def make_initial(n, rng):
    """Create various initial binary patterns."""
    h = np.zeros(n)
    half = n // 2
    # Random arrangement of half ones
    pos = rng.permutation(n)[:half]
    h[pos] = 1.0
    return h

def make_block(n):
    """Centered block of ones."""
    h = np.zeros(n)
    half = n // 2
    start = (n - half) // 2
    h[start:start+half] = 1.0
    return h

def make_alternating(n):
    """Alternating pattern."""
    h = np.zeros(n)
    h[::2] = 1.0

    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]

            h[idx[i]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        for i in range(target - ones):
            h[idx[i]] = 1.0
    return h

            h[i] = 0.0
            h[j] = 1.0
            
            new_c5 = c5_of(h, n)
            delta = new_c5 - curr_c5

                curr_c5 = new_c5

                zeros_list[zeros_list.index(j)] = i

                h[j] = 0.0

import numpy as np
import time

def c5_of(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_to_binary(h, n):
    target = n // 2
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def make_block(n):
    h = np.zeros(n)
    half = n // 2
    start = (n - half) // 2
    h[start:start + half] = 1.0
    return h

def make_alternating(n):
    h = np.zeros(n)
    h[::2] = 1.0
    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]
        for i in range(ones - target):
            h[idx[i]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        for i in range(target - ones):
            h[idx[i]] = 1.0
    return h

def make_random(n, rng):
    h = np.zeros(n)
    half = n // 2
    pos = rng.permutation(n)[:half]
    h[pos] = 1.0
    return h

def make_spacing(n, rng):
    h = np.zeros(n)
    target = n // 2
    golden = 0.6180339887498948
    positions = sorted(set(int((i * golden * n) % n) for i in range(target)))
    while len(positions) < target:
        p = int(rng.randint(n))
        if p not in positions:
            positions.append(p)
            positions.sort()
    h[positions[:target]] = 1.0
    return h

def run_sa(h, n, deadline, rng):
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_of(h, n)
    curr_c5 = best_c5

    T_init = 0.025
    T_min = 1e-9
    alpha = 0.99975
    T = T_init
    no_improve = 0
    stag_limit = 4000
    it = 0

    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])

    while time.time() < deadline:
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = c5_of(h, n)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            oi = ones_list.index(i)
            zi = zeros_list.index(j)
            ones_list[oi] = j
            zeros_list[zi] = i
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

        it += 1
        if it % 5 == 0:
            T = max(T * alpha, T_min)

        if no_improve >= stag_limit:
            h = best_h.copy()
            n_pert = max(2, n // 6)
            ol = list(np.where(h == 1)[0])
            zl = list(np.where(h == 0)[0])
            for _ in range(n_pert):
                ii = ol[rng.randint(len(ol))]
                jj = zl[rng.randint(len(zl))]
                if ii != jj:
                    h[ii] = 0.0
                    h[jj] = 1.0
                    ol[ol.index(ii)] = jj
                    zl[zl.index(jj)] = ii
            ones_list = list(np.where(h == 1)[0])
            zeros_list = list(np.where(h == 0)[0])
            curr_c5 = c5_of(h, n)
            no_improve = 0
            T = max(T_init * 0.6, 0.005)

        if T <= T_min:
            h = best_h.copy()
            n_pert = max(2, n // 4)
            ol = list(np.where(h == 1)[0])
            zl = list(np.where(h == 0)[0])
            for _ in range(n_pert):
                ii = ol[rng.randint(len(ol))]
                jj = zl[rng.randint(len(zl))]
                if ii != jj:
                    h[ii] = 0.0
                    h[jj] = 1.0
                    ol[ol.index(ii)] = jj
                    zl[zl.index(jj)] = ii
            ones_list = list(np.where(h == 1)[0])
            zeros_list = list(np.where(h == 0)[0])
            curr_c5 = c5_of(h, n)
            no_improve = 0
            T = T_init * 0.4

    return best_h, best_c5

    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:

    except Exception:
        h0 = None

    def elapsed():
        return time.time() - t0

    # Build task list: (n, fraction, start_type)
    tasks = []
    if h0 is not None:
        n0 = len(h0)
        tasks.append((n0, 0.30, 'initial'))
        tasks.append((n0, 0.12, 'random'))
    else:
        tasks.append((46, 0.20, 'block'))
        tasks.append((46, 0.15, 'random'))

    other_ns = [40, 44, 48, 50, 52, 54, 56, 60, 64, 70, 72, 80, 88, 96, 108, 120, 140, 160]
    if h0 is not None:
        other_ns = [x for x in other_ns if x != len(h0)]
    else:
        other_ns = [x for x in other_ns if x != 46]

    remaining_frac = 1.0 - sum(f for _, f, _ in tasks)
    per = remaining_frac / len(other_ns)
    init_types = ['block', 'alternating', 'random', 'spacing']
    for k, nn in enumerate(other_ns):
        tasks.append((nn, per, init_types[k % len(init_types)]))

    # Shuffle non-initial tasks
    rng.shuffle(tasks[2:])

    for n_try, frac, start_type in tasks:
        if elapsed() > budget_s - 12:
            break
        alloc = max(3.0, min(budget_s * frac, (budget_s - 10 - elapsed()) * 0.85))
        deadline = time.time() + alloc

        if start_type == 'initial' and h0 is not None:
            h_start = project_to_binary(h0, n_try)
        elif start_type == 'block':
            h_start = make_block(n_try)
        elif start_type == 'alternating':
            h_start = make_alternating(n_try)
        elif start_type == 'spacing':
            h_start = make_spacing(n_try, rng)
        else:
            h_start = make_random(n_try, rng)

        bh, bc5 = run_sa(h_start, n_try, deadline, rng)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh
            best_n = n_try

    # Final refinement on best
    if best_h is not None and elapsed() < budget_s - 8:
        deadline = t0 + budget_s - 4
        bh, bc5 = run_sa(best_h, best_n, deadline, rng)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh
            best_n = len(bh)

    if best_h is None:
        best_n = 46
        best_h = make_block(best_n)
        best_c5 = c5_of(best_h, best_n)

    h_final = best_h.copy()
    n_final = best_n
    # Verify binary and constraint

    assert np.all((h_final == 0) | (h_final == 1))
    c5_final = c5_of(h_final, n_final)

    return h_final, float(c5_final), n_final
