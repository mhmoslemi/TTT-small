```python
def lp_refl(h, n, top_m=20, step_size=0.1):
    """LP-based reflection step."""
    import cvxpy as cp
    
    current_c = np.correlate(h, 1.0 - h, mode="full")
    t_current = current_c.max()
    
    # Get top M shifts
    idx = np.argsort(current_c)[::-1][:top_m]
    
    # Compute gradient of c[k] wrt h for top M shifts
    # c[k] = sum_j h[j] * (1 - h[j-k+(n-1)])
    # dc[k]/dh[i] = (1-h[i-k+(n-1)]) * valid1 - h[i+k-(n-1)] * valid2
    
    h_new = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_new >= 0, h_new <= 1, cp.sum(h_new) == n/2.0]
    
    for k_idx in idx:
        k = int(k_idx)
        # Linearize c[k] at current h
        c_k_linear = current_c[k]
        for i in range(n):
            # Partial of c[k] wrt h[i]
            # h[i] as first factor: i-k+(n-1) must be valid
            j2 = i - k + (n-1)  # index for second factor
            if 0 <= j2 < n:
                c_k_linear += (1 - h[j2]) * (h_new[i] - h[i])
            # h[i] as second factor (in 1-h): i+k-(n-1) must be valid
            j1 = i + k - (n-1)  # index for first factor
            if 0 <= j1 < n:
                c_k_linear -= h[j1] * (h_new[i] - h[i])
        constraints.append(cp.sum(c_k_linear) <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if h_new.value is not None:
        # Take a step towards LP solution
        h_step = h + step_size * (h_new.value - h)
        h_step = project_feasible(h_step, n)
```

```python
def compute_jacobian(h, n):
    """Compute J[k, i] = dc[k]/dh[i] for all k, i."""
    J = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        offset = k - (n-1)
        for i in range(n):
            # h[i] as first factor: partner is i - offset
            j2 = i - offset
            if 0 <= j2 < n:
                J[k, i] += (1 - h[j2])
            # h[i] in (1-h) as second factor: partner is i + offset  
            j1 = i + offset
            if 0 <= j1 < n:
                J[k, i] -= h[j1]
    return J
```

```python
def compute_jacobian(h, n):
    J = np.zeros((2*n-1, n))
    offset = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # For each k (row of J), offset[k] = k - (n-1)
    # J[k, i] = (1-h[i-offset[k]]) * valid1 - h[i+offset[k]] * valid2
    
    # i ranges over columns: shape (1, n)
    i_idx = np.arange(n)  # shape (n,)
    
    # j2 = i - offset: shape (2n-1, n)
    j2 = i_idx[None, :] - offset[:, None]
    valid1 = (j2 >= 0) & (j2 < n)
    # Clip j2 for safe indexing
    j2_clipped = np.clip(j2, 0, n-1)
    J += (1 - h[j2_clipped]) * valid1
    
    # j1 = i + offset: shape (2n-1, n)
    j1 = i_idx[None, :] + offset[:, None]
    valid2 = (j1 >= 0) & (j1 < n)
    j1_clipped = np.clip(j1, 0, n-1)
    J -= h[j1_clipped] * valid2
    
    return J
```

This is O(n²) which for n=320 is about 100K operations. Fast.

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def compute_jacobian(h, n):
    """J[k,i] = dc[k]/dh[i] where c = correlate(h, 1-h, full)."""
    i_idx = np.arange(n)
    offset = np.arange(2 * n - 1) - (n - 1)
    j2 = i_idx[None, :] - offset[:, None]
    valid1 = (j2 >= 0) & (j2 < n)
    j2c = np.clip(j2, 0, n - 1)
    J = (1.0 - h[j2c]) * valid1
    j1 = i_idx[None, :] + offset[:, None]
    valid2 = (j1 >= 0) & (j1 < n)
    j1c = np.clip(j1, 0, n - 1)
    J -= h[j1c] * valid2
    return J

def smooth_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    return m + np.log(np.sum(np.exp(shifted))) / beta

def sum_func(x):
    return np.sum(x) - len(x) / 2.0

def sum_jac(x):
    return np.ones(len(x))

def make_bounds(n):
    return [(0.0, 1.0)] * n

def lp_refine(h, n, step_size, use_cvxpy=True):
    """One LP-based refinement step. Returns (h_new, c5_new)."""
    c = np.correlate(h, 1.0 - h, mode="full")
    J = compute_jacobian(h, n)
    
    if use_cvxpy:
        try:
            import cvxpy as cp
            h_new_var = cp.Variable(n)
            t = cp.Variable()
            delta = h_new_var - h
            constraints = [h_new_var >= 0, h_new_var <= 1,
                           cp.sum(h_new_var) == n / 2.0,
                           c + J @ delta <= t]
            prob = cp.Problem(cp.Minimize(t), constraints)
            prob.solve(solver=cp.GLOP)
            if h_new_var.value is None:
                prob.solve(solver=cp.CBC)
            if h_new_var.value is not None:
                direction = h_new_var.value - h
                h_new = h + step_size * direction
                h_new = project_feasible(h_new, n)
                return h_new, compute_c5(h_new)
        except Exception:
            pass
    
    # Fallback: greedy subgradient step
    t_current = c.max()
    active = np.where(c > t_current * 0.95)[0]
    grad = J[active, :].mean(axis=0)
    h_new = h - 0.01 * step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new)

def optimize_smooth(h_start, n, beta, time_limit):
    h = project_feasible(h_start, n)
    best = h.copy()

        res = minimize(smooth_c5, h, args=(beta,), method='L-BFGS-B',
                      bounds=make_bounds(n),

        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
    except Exception:
        pass
    return best, best_c5

                no_progress = 0
                step_size = 0.5
            if no_progress > 500:
                break
    
    return best_h, best_c5
