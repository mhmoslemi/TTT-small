```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    logsum = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    obj = logsum * dx
    exp_t = np.exp((corr - m) / temp)
    w = exp_t / exp_t.sum()
    
    grad = np.zeros(2*n-1)  # temporary, we'll map to h indices
    # Actually, let's compute grad_h directly
    grad_h = np.zeros(n)
    for l in range(-(n-1), n):
        wl = w[l + n - 1]
        if wl < 1e-15:
            continue
        if l >= 0:
            grad_h[l:n] += wl * (1.0 - h[0:n-l])
            grad_h[0:n-l] -= wl * h[l:n]
        else:
            al = -l
            grad_h[0:n-al] += wl * (1.0 - h[al:n])
            grad_h[al:n] -= wl * h[0:n-al]
    return obj, grad_h * dx

def project_constraint(h, target):
    """Project h onto {x in [0,1] : sum(x) = target}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Simple: add/subtract diff/n, then re-clip and iterate
    for _ in range(20):
        h = np.clip(h + diff / len(h), 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
    return h

def projected_grad_descent(h0, n, max_time, temps=None, lr_scale=1.0):
    t0 = time.time()
    target = n / 2.0
    if temps is None:
        temps = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0005, 0.0002]
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    
    for temp in temps:
        if time.time() - t0 > max_time - 5:
            break
        h = best_h.copy()
        base_lr = lr_scale / (n * temp + 1.0)
        # Adaptive learning rate
        for it in range(300):
            if time.time() - t0 > max_time - 3:
                break
            obj, g = obj_grad(h, n, temp)
            # Gradient descent step
            h_new = h - base_lr * g
            h_new = project_constraint(h_new, target)
            # Check improvement
            v = compute_c5(h_new)
            if v < best_val:
                best_h = h_new.copy()
                best_val = v
            h = h_new
        # After each temp, evaluate
        v = compute_c5(h)
        if v < best_val:
            best_h = h.copy()
            best_val = v
    
    return best_h, best_val

def binary_local_search(h_bin, n, max_time, max_iters=50000):
    """Local search on binary h: swap 1s and 0s to reduce max corr."""
    t0 = time.time()
    h = h_bin.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_val = np.max(corr) * dx
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() - t0 > max_time - 2:
            break
        # Try a random swap
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        # Compute new corr
        h[a], h[b] = 0.0, 1.0
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_val = np.max(new_corr) * dx
        h[a], h[b] = 1.0, 0.0
        
        if new_val < best_val - 1e-12:
            h[a], h[b] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            best_val = new_val
        elif rng.random() < np.exp(-50.0 * (new_val - best_val)):
            # Accept worse (annealing)
            h[a], h[b] = 0.0, 1.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    
    return h, best_val
```

Wait, the binary local search as written is O(n²) per iteration (computing full correlation). For n=160, that's 25600 per eval, times 50000 iters = 1.3 billion ops. Too slow in Python.

I need to use the analytical update for the binary case. For binary h, if I flip h[a] from 1 to 0 and h[b] from 0 to 1:

Δcorr[k] = (change from flipping a) + (change from flipping b)

For flipping h[a] from 1 to 0:
- corr[k] = sum_i h[i+l]*(1-h[i]), l=k-(n-1)
- The term where h[a] appears as h[i+l] (i+a = a → i = a-l... wait, i+l = a → i = a-l):
  - Old contribution: 1 * (1-h[a-l]) = (1-h[a-l])
  - New contribution: 0 * (1-h[a-l]) = 0
  - Change: -(1-h[a-l]) = -1+h[a-l] (= -1 if h[a-l]=0, 0 if h[a-l]=1)
  - Valid if a-l is in valid i range for lag l.
- The term where h[a] appears in (1-h[i]) (i = a):
  - Old: h[a+l] * (1-1) = 0
  - New: h[a+l] * (1-0) = h[a+l]
  - Change: +h[a+l]
  - Valid if a is in valid i range AND a+l in [0,n-1].

For flipping h[b] from 0 to 1:
- The term where h[b] appears as h[i+l] (i+l = b → i = b-l):
  - Old: 0 * (1-h[b-l]) = 0
  - New: 1 * (1-h[b-l]) = (1-h[b-l])
  - Change: +(1-h[b-l])
  - Valid if b-l in valid i range.
- The term where h[b] appears in (1-h[i]) (i = b):
  - Old: h[b+l] * (1-0) = h[b+l]
  - New: h[b+l] * (1-1) = 0
  - Change: -h[b+l]
  - Valid if b in valid i range AND b+l in [0,n-1].

So Δcorr[k] (for swapping a→0, b→1) = 
  -(1-h[a-l]) * 1[a-l valid as i for lag l] + h[a+l] * 1[a valid as i and a+l in [0,n-1]]
  + (1-h[b-l]) * 1[b-l valid as i for lag l] - h[b+l] * 1[b valid as i and b+l in [0,n-1]]

For binary h, h[a+l] is 0 or 1. This is all just lookups.

For each lag l, the validity conditions are:
- "x valid as i for lag l": 
  - l >= 0: 0 <= x <= n-1-l
  - l < 0: -l <= x <= n-1
- "x+l in [0, n-1]": 0 <= x+l <= n-1

So the full Δcorr for a swap can be computed in O(n) (one for each lag). With n=160, that's 319 lags × O(1) per lag = 319 ops per swap candidate.

If I try all 80×80 = 6400 swaps per iteration, that's 6400 × 319 ≈ 2M ops per iteration. In numpy, I can vectorize over lags, making it fast.

Let me precompute the validity masks and then for each candidate (a,b), compute Δcorr vectorized:
