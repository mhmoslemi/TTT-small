```python
def _direct_gradient(h, weights):
    """Compute gradient directly. weights has length 2n-1."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in range(2 * n - 1):

        if w < 1e-15:

        for j in range(n):
            # d corr[s] / dh[j] = (1-h[j+k]) - h[j-k] (with 0 for out of range)
            term = 0.0
            if 0 <= j + k < n:
                term += 1.0 - h[j + k]
            if 0 <= j - k < n:
                term -= h[j - k]
            grad[j] += w * term
    return grad * dx
```

```python
def _direct_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-15:
            continue
        k = s - (n - 1)
        # (1-h[j+k]) for valid j: j+k in [0,n-1] → j in [max(0,-k), min(n-1, n-1-k)]
        # h[j-k] for valid j: j-k in [0,n-1] → j in [k, n-1+k] → j in [max(0,k), min(n-1, n-1+k)]
        j_pos = np.arange(max(0, -k), min(n, n - k))
        if len(j_pos) > 0:
            grad[j_pos] += w * (1.0 - h[j_pos + k])
        j_neg = np.arange(max(0, k), min(n, n + k))
        if len(j_neg) > 0:
            grad[j_neg] -= w * h[j_neg - k]
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        clipped = np.clip(h - mid, 0.0, 1.0)
        if np.sum(clipped) > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        mask = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def _slsqp_optimize(h_init, n, max_time, seed):
    """SLSQP optimization with soft-max objective"""
    t_start = time.time()
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    target_sum =(n / 2.0
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        # adaptive temperature
        t = max(0.001, 0.05 * np.exp(-0.01 * (time.time() - t_start)))
        shifted = np.clip((corr - m) / t, -50, 0)
        return (m + t * np.log(np.sum(np.exp(shifted)))) * dx
    
    def cons_eq(h):
        return np.sum(h) - target_sum
    
    def cons_eq_jac(h):
        return np.ones(n)
    
    cons = ({'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac},)
    
    best = h_init.copy()
    best_val = compute_c5(best)
    
    temps = [0.05, 0.01, 0.003, 0.001, 0.0003]
    for temp in temps:
        if time.time() - t_start > max_time - 5:
            break
        h = best.copy()
        for it in range(3):
            if time.time() - t_start > max_time - 2:
                break
            try:
                def obj_t(h):
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    m = np.max(corr)
                    t = max(temp, 1e-9)
                    shifted = np.clip((corr - m) / t, -50, 0)
                    return (m + t * np.log(np.sum(np.exp(shifted)))) * dx
                res = minimize(obj_t, h, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
                h_new = _project_feasible(res.x, target_sum)
                v = compute_c5(h_new)
                if v < best_val - 1e-12:
                    best = h_new.copy()
                    best_val = v
                h = h_new.copy()
            except Exception:
                break
    return best, best_val

def _projected_gradient_optimize(h_init, n, max_time, seed, K=10):
    """Custom projected gradient method"""
    t_start = time.time()
    np.random.seed(seed)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_init.copy()
    best = h.copy()
    best_val = compute_c5(best)
    
    alpha = 0.05  # initial step size
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    it = 0
    while time.time() - t_start < max_time:
        # Compute gradient using top-K lags
        K_eff = min(K, len(corr))
        idx = np.argpartition(corr, -K_eff)[-K_eff:]
        weights = np.zeros(len(corr))
        weights[idx] = 1.0 / K_eff
        
        dx_step = 2.0 / n
        grad = np.zeros(n)
        for s in idx:
            k = int(s) - (n - 1)
            j_pos = np.arange(max(0, -k), min(n, n - k))
            if len(j_pos) > 0:
                grad[j_pos] += (1.0 - h[j_pos + k]) / K_eff
            j_neg = np.arange(max(0, k), min(n, n + k))
            if len(j_neg) > 0:
                grad[j_neg] -= h[j_neg - k] / K_eff
        grad *= dx_step
        
        # Line search
        best_alpha = alpha
        best_obj = best_val
        for trial_alpha in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1, alpha * 2.0]:
            h_new = h - trial_alpha * grad
            h_new = _project_feasible(h_new, target_sum)
            v = compute_c5(h_new)
            if v < best_obj:
                best_obj = v
                best_alpha = trial_alpha
        
        h = h - best_alpha * grad
        h = _project_feasible(h, target_sum)
        
        v = compute_c5(h)
        if v < best_val:
            best = h.copy()
            best_val = v
            alpha = min(alpha * 1.2, 0.5)
        else:
            alpha *= 0.7
        
        # Recompute correlation periodically
        it += 1
        if it % 10 == 0:
            corr = np.correlate(h, 1.0 - h, mode="full")
            # Also try increasing K
            K = min(int(10 + it * 0.1), 30)
    
    return best, best_val

def _local_search(h, n, max_time, seed):
    """Try local modifications to individual h[j]"""
    t_start = time.time()
    np.random.seed(seed)
    target_sum = n / 2.0
    best = h.copy()
    best_val = compute_c5(best)
    
    dx = 2.0 / n
    improved = True
    while improved and time.time() - t_start < max_time:
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() - t_start > max_time - 1:
                break
            old_val = best[j]
            # Try several new values
            candidates = []
            candidates.append(old_val + 0.01)
            candidates.append(old_val - 0.01)
            candidates.append(old_val + 0.05)
            candidates.append(old_val - 0.05)
            candidates.append(min(1.0, old_val + 0.1))
            candidates.append(max(0.0, old_val - 0.1))
            candidates.append(0.0)
            candidates.append(1.0)
            candidates.append(0.5)
            
            for new_val in candidates:
                if new_val < -1e-6 or new_val > 1 + 1e-6:
                    continue
                new_val = float(np.clip(new_val, 0.0, 1.0))
                if abs(new_val - old_val) < 1e-8:
                    continue
                best[j] = new_val
                # Need to maintain sum constraint
                # Adjust one other element
                cur_sum = np.sum(best)
                diff = target_sum - cur_sum
                if abs(diff) < 1e-10:
                    pass
                else:
                    # Find the element with the most room to adjust
                    room = np.where(diff > 0, 1.0 - best, best)
                    if room[j] > 0.01:
                        room[j] = 0  # don't adjust the same element
                    if np.sum(room) < abs(diff):
                        best[j] = old_val
                        continue
                    # Add diff to the element with the most room
                    target_j = int(np.argmax(room))
                    adj = diff / room[target_j] if room[target_j] > 1e-8 else 0
                    best[target_j] += adj
                    if best[target_j] < 0 or best[target_j] > 1:
                        best[j] = old_val
                        continue
                
                v = compute_c5(best)
                if v < best_val - 1e-12:
                    best_val = v
                    improved = True
                    # Don't reset j to old_val; keep the improvement
                else:
                    best[j] = old_val
                    # Re-fix sum
                    best = _project_feasible(best, target_sum)
        
        if improved:
            best = _project_feasible(best, target_sum)
    
    return best, best_val

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float) * 2.0 / n_old
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = _project_feasible(h_new, n_new / 2.0)
    return h_new

def _binary_init(n, seed):
    np.random.seed(seed)
    h = np.zeros(n)
    n_ones = n // 2
    idx = np.random.permutation(n)[:n_ones]
    h[idx] = 1.0
    return h

def _two_pulse_init(n, gap_ratio, seed):
    np.random.seed(seed)
    h = np.zeros(n)
    n_ones = n // 2
    half = n_ones // 2
    gap = int(n * gap_ratio)
    pos1 = np.random.randint(0, max(1, n - 2 * half - gap))
    pos2 = (pos1 + half + gap) % n
    for i in range(half):
        h[(pos1 + i) % n] = 1.0
        h[(pos2 + i) % n] = 1.0
    return h

def _perturb(h, scale, seed):
    np.random.seed(seed)
    n = len(h)
    h_p = h + scale * np.random.randn(n)
    return _project_feasible(h_p, n / 2.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 160
_g_temp = 0.01

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        mask = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def _obj_func(h):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    t = max(_g_temp, 1e-9)
    shifted = np.clip((corr - m) / t, -50, 0)
    return (m + t * np.log(np.sum(np.exp(shifted)))) * dx

def _cons_eq(h):
    return np.sum(h) - _g_n / 2.0

def _cons_eq_jac(h):
    return np.ones(_g_n)

def _slsqp(h_init, n, max_time, seed):
    global _g_n, _g_temp
    t_start = time.time()
    _g_n = n
    target = n / 2.0
    best = h_init.copy()
    best_val = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac},)
    temps = [0.05, 0.01, 0.003, 0.001, 0.0003, 0.0001]
    for temp in temps:
        if time.time() - t_start > max_time - 4:
            break
        _g_temp = temp
        h = best.copy()
        for it in range(4):
            if time.time() - t_start > max_time - 2:
                break
            try:
                res = minimize(_obj_func, h, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
                h_new = _project_feasible(res.x, target)
                v = compute_c5(h_new)
                if v < best_val - 1e-13:
                    best, best_val = h_new.copy(), v
                h = h_new.copy()
            except Exception:
                break
    return best, best_val

def _proj_grad(h_init, n, max_time, seed, K0=8):
    t_start = time.time()
    np.random.seed(seed)
    dx = 2.0 / n
    target = n / 2.0
    h = h_init.copy()
    best, best_val = h.copy(), compute_c5(h)
    alpha = 0.08
    it = 0
    while time.time() - t_start < max_time:
        corr = np.correlate(h, 1.0 - h, mode="full")
        K = min(K0 + it // 20, 30)
        idx = np.argpartition(corr, -K)[-K:]
        grad = np.zeros(n)
        for s in idx:
            k = int(s) - (n - 1)
            j_pos = np.arange(max(0, -k), min(n, n - k))
            if len(j_pos) > 0:
                grad[j_pos] += (1.0 - h[j_pos + k]) / K
            j_neg = np.arange(max(0, k), min(n, n + k))
            if len(j_neg) > 0:
                grad[j_neg] -= h[j_neg - k] / K
        grad *= dx
        best_a, best_v = alpha, best_val
        for fa in [alpha, alpha * 0.5, alpha * 0.2, alpha * 2.0, alpha * 0.05]:
            h_new = _project_feasible(h - fa * grad, target)
            v = compute_c5(h_new)
            if v < best_v:
                best_v, best_a = v, fa
        h = _project_feasible(h - best_a * grad, target)
        v = compute_c5(h)
        if v < best_val:
            best, best_val = h.copy(), v
            alpha = min(alpha * 1.15, 0.6)
        else:
            alpha *= 0.65
        it += 1
    return best, best_val

def _local_search(h, n, max_time, seed):
    t_start = time.time()
    np.random.seed(seed)
    target = n / 2.0
    best = h.copy()
    best_val = compute_c5(best)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.2, -0.2]
    while time.time() - t_start < max_time:
        improved = False
        for j in np.random.permutation(n):
            if time.time() - t_start > max_time - 1:
                break
            saved = best.copy()
            old_j = best[j]
            for d in deltas:
                nv = float(np.clip(old_j + d, 0.0, 1.0))
                if abs(nv - old_j) < 1e-8:
                    continue
                cand = saved.copy()
                cand[j] = nv
                diff = target - np.sum(cand)
                if abs(diff) > 1e-12:
                    room = np.minimum(1.0 - cand, cand) if diff < 0 else np.minimum(1.0 - cand, np.ones(n))
                    room[j] = 0.0
                    if np.sum(room) < abs(diff) * 0.99:
                        continue
                    k2 = int(np.argmax(room))
                    cand[k2] += diff / max(room[k2], 1e-12)
                    cand = _project_feasible(cand, target)
                v = compute_c5(cand)
                if v < best_val - 1e-12:
                    best, best_val = cand, v
                    improved = True
                    break
        if not improved:
            break
    best = _project_feasible(best, target)
    return best, compute_c5(best)

def _upsample(h, nn):
    no = len(h)
    xo = np.arange(no, dtype=float) * 2.0 / no
    xn = np.arange(nn, dtype=float) * 2.0 / nn
    return _project_feasible(np.interp(xn, xo, h), nn / 2.0)

def _binary_init(n, seed):
    np.random.seed(seed)
    h = np.zeros(n)
    h[np.random.permutation(n)[:n // 2]] = 1.0
    return h

def _two_pulse(n, gap, seed):
    np.random.seed(seed)
    h = np.zeros(n)
    half = n // 4
    g = int(n * gap)
    p1 = np.random.randint(0, max(1, n - 2 * half - g))
    p2 = (p1 + half + g) % n
    for i in range(half):
        h[(p1 + i) % n] = 1.0
        h[(p2 + i) % n] = 1.0
    return _project_feasible(h, n / 2.0)

def _perturb(h, s, seed):
    np.random.seed(seed)
    return _project_feasible(h + s * np.random.randn(len(h)), len(h) / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def _rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 160
        h0 = np.full(n, 0.5)

    best_h, best_c5 = h0.copy(), compute_c5(h0)

    if _rem() > 130:
        h, c = _slsqp(h0.copy(), n, min(160, _rem() - 50), seed)
        if c < best_c5:
            best_h, best_c5 = h, c

    if _rem() > 80:
        h, c = _proj_grad(best_h.copy(), n, min(80, _rem() - 30), seed + 1, K0=6)
        if c < best_c5:
            best_h, best_c5 = h, c

    if _rem() > 60:
        h, c = _local_search(best_h.copy(), n, min(70, _rem() - 20), seed + 2)
        if c < best_c5:
            best_h, best_c5 = h, c

    inits = [
        _binary_init(n, seed + 100),
        _two_pulse(n, 0.25, seed + 200),
        _two_pulse(n, 0.4, seed + 300),
        _perturb(best_h, 0.12, seed + 400),
        _perturb(best_h, 0.25, seed + 500),
    ]
    for i, hi in enumerate(inits):
        if _rem() < 50:
            break
        h, c = _slsqp(hi.copy(), n, min(45, _rem() - 25), seed + 1000 + i)
        if c < best_c5:
            best_h, best_c5 = h, c
        if _rem() > 30:
            h2, c2 = _local_search((h if c < best_c5 else best_h).copy(),
                                   n, min(20, _rem() - 10), seed + 5000 + i)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2

    for tn in [80, 100, 120, 140, 200, 240]:
        if _rem() < 70:
            break
        if tn == n:
            continue
        hx = _upsample(best_h, tn)
        h, c = _slsqp(hx, tn, min(50, _rem() - 35), seed + tn)
        if c < best_c5:
            best_h, best_c5 = h, c
        if _rem() > 25 and c < best_c5 + 0.002:
            h2, c2 = _local_search(h.copy(), tn, min(20, _rem() - 10), seed + 2000 + tn)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2

    if _rem() > 30:
        nc = len(best_h)
        h, c = _slsqp(best_h.copy(), nc, min(35, _rem() - 10), seed + 9999)
        if c < best_c5:
            best_h, best_c5 = h, c
        if _rem() > 12:
            h2, c2 = _local_search(best_h.copy(), nc, max(8, _rem() - 3), seed + 9998)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2

    best_h = _project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
