```python
    n_full = len(x) + 1  # we're optimizing n-1 variables, last is determined
    # Reconstruct h
    s = np.sum(x)
    h_last = (n_full / 2.0) - s
    h = np.concatenate([x, [h_last]])
    h = np.clip(h, 0, 1)
    # Recompute with clipped values
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)

    obj = cmax + np.log(np.sum(np.exp(alpha * corr_shift))) / alpha
```

```python
def _project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    # Bisection on the Lagrange multiplier for the sum constraint
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune to get exact sum
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        # Distribute the remaining difference
        mask = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if mask.any():
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj
```

```python
def _softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    exp_shifted = np.exp(alpha * (corr - cmax))
    p = exp_shifted / exp_shifted.sum()  # softmax probabilities
    
    grad = np.zeros(n)
    # For each output element m of the correlate,
    # C[m] = sum_i h[i] * (1-h[i+m-(n-1)])
    # h[j] appears in C[m] when:
    #   1) i = j (as the h factor): contributes (1-h[j+m-(n-1)])
    #   2) i+m-(n-1) = j, i.e., i = j-m+(n-1) (as the (1-h) factor): contributes -h[i]
    
    for m in range(2*n - 1):
        shift = m - (n - 1)  # this is the "lag"
        # h[j] as first factor: C[m] contains h[j] * (1-h[j+shift]) for valid j+shift
        # h[j] as second factor (via 1-h): C[m] contains h[j-shift] * (1-h[j]) for valid j-shift
        for j in range(n):
            # Contribution 1: h[j] in first position
            j2 = j + shift
            if 0 <= j2 < n:
                grad[j] += p[m] * dx * (1.0 - h[j2])
            # Contribution 2: h[j] in second position (via 1-h[j])
            j2 = j - shift
            if 0 <= j2 < n:
                grad[j] -= p[m] * dx * h[j2]
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _softmax_obj_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    max_shift = np.max(shifted)
    exp_vals = np.exp(shifted - max_shift)
    return cmax + (max_shift + np.log(np.sum(exp_vals))) / alpha

def _softmax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    max_s = np.max(shifted)
    exp_vals = np.exp(shifted - max_s)
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-16:
            continue
        shift = m - (n - 1)
        pm = p[m] * dx
        if shift >= 0:
            k = n - shift
            if k > 0:
                grad[:k] += pm * (1.0 - h[shift:])
                grad[shift:] -= pm * h[:k]
        else:
            s = -shift
            k = n - s
            if k > 0:
                grad[s:] += pm * (1.0 - h[:k])
                grad[:k] -= pm * h[s:]
    return grad

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-13:
        mask = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def _fix_sum(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h

def _proj_grad_descent(h, alpha, max_iter, time_limit, t0):
    n = len(h)
    h = h.copy()
    best_f = _softmax_obj_val(h, alpha)
    lr = 0.01
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        g = _softmax_grad(h, alpha)
        h_new = h - lr * g
        h_new = _project(h_new, n)
        f_new = _softmax_obj_val(h_new, alpha)
        if f_new < best_f:
            best_f = f_new
            h = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return _fix_sum(h, n)

class _SO:
    def __init__(self, alpha):
        self.alpha = alpha
    def __call__(self, x):
        return _softmax_obj_val(x, self.alpha)

def _sq(x):
    return float(np.sum(x) - len(x) / 2.0)

def _sqsq(x):
    return float(np.sum(x) - len(x) / 2.0)

def _mmz(x):
    return x[-1]

def _mmi(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def _solsqp(h0, alpha, max_iter, t0, time_limit):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_SO(alpha), h0, method='SLSQP',
                       constraints=[{'type': 'eq', 'fun': _sq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        return res.x
    except Exception:
        return h0.copy()

def _solsqp_mm(h0, max_iter, t0, time_limit):
    n = len(h0)
    t_init = _eval_c5(h0)
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    try:
        res = minimize(_mmz, x0, method='SLSQP',
                       constraints=[{'type': 'ineq', 'fun': _mmi},
                                    {'type': 'eq', 'fun': _sqsq}],
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        return res.x[:-1]
    except Exception:
        return h0.copy()

def _gen_starts(n, rng):
    starts = []
    starts.append(np.full(n, 0.5))
    base = np.array([0.0, 1.0])
    for period in [2, 3, 4, 5, 7, 8]:
        pat = base[np.arange(n) % period]
        if np.sum(pat) < n / 2.0:
            pat = 1.0 - pat
        starts.append(_fix_sum(pat, n))
    for freq in [1, 2, 3, 5]:
        x = np.linspace(0, 2 * np.pi * freq, n, endpoint=False)
        starts.append(_fix_sum(0.5 + 0.4 * np.sin(x), n))
    gr = (1 + np.sqrt(5)) / 2
    for k in range(3):
        pos = np.floor(np.arange(n) * gr + k * 0.1).astype(int) % n
        b = np.zeros(n)
        b[np.sort(pos[:n // 2])] = 1.0
        starts.append(_fix_sum(b, n))
    for _ in range(4):
        starts.append(_fix_sum(rng.random(n), n))
    return starts

def _improve(h, t0, time_limit, rng, verbose=False):
    n = len(h)
    best = _eval_c5(h)
    bh = h.copy()
    for alpha in [20, 50, 100, 200, 400, 800]:
        if time.time() - t0 > time_limit:
            break
        h2 = _proj_grad_descent(bh, alpha, 500, time_limit, t0)
        c = _eval_c5(h2)
        if c < best:
            best = c
            bh = h2.copy()
    if time.time() - t0 < time_limit - 30:
        h3 = _solsqp(bh, 200, 300, t0, time_limit)
        h3 = _fix_sum(h3, n)
        c = _eval_c5(h3)
        if c < best:
            best = c
            bh = h3.copy()
    if time.time() - t0 < time_limit - 20:
        h4 = _solsqp_mm(bh, 200, t0, time_limit)
        h4 = _fix_sum(h4, n)
        c = _eval_c5(h4)
        if c < best:
            best = c
            bh = h4.copy()
    return bh, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = None

    n_list = [47, 49, 51, 53, 55, 57, 59, 61, 63, 65, 67, 69, 71, 73, 75, 77, 79]

    if h_init is not None:
        h_init = _fix_sum(h_init, len(h_init))
        c0 = _eval_c5(h_init)
        best_c5 = c0
        best_h = h_init.copy()
        n_init = len(h_init)
        h_init, best_c5 = _improve(h_init, t0, budget_s * 0.35, rng)
        best_h = h_init.copy()

    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 60
    if remaining > 30:
        per_n = remaining / len(n_list)
        for n in n_list:
            if time.time() - t0 > budget_s - 60:
                break
            tl = min(time.time() + per_n * 1.5, budget_s - 40)
            starts = _gen_starts(n, rng)
            for h0 in starts:
                if time.time() - t0 > tl:
                    break
                h_opt = _proj_grad_descent(h0, 50, 300, tl, t0)
                c = _eval_c5(h_opt)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_opt.copy()
            if best_c5 < 0.382 and time.time() - t0 < budget_s - 50:
                tl2 = min(time.time() + 20, budget_s - 40)
                h_opt, c_new = _improve(_fix_sum(best_h, len(best_h)) if len(best_h) == n else best_h, t0, tl2, rng)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_opt

    if best_h is not None and time.time() - t0 < budget_s - 40:
        for _ in range(50):
            if time.time() - t0 > budget_s - 30:
                break
            h_pert = _fix_sum(best_h + rng.normal(0, 0.02, len(best_h)), len(best_h))
            h_pert = _proj_grad_descent(h_pert, 200, 200, budget_s - 30, t0)
            c = _eval_c5(h_pert)
            if c < best_c5:
                best_c5 = c
                best_h = h_pert.copy()

    if best_h is not None and time.time() - t0 < budget_s - 20:
        h_fin = _solsqp_mm(best_h, 400, t0, budget_s - 20)
        h_fin = _fix_sum(h_fin, len(h_fin))
        c = _eval_c5(h_fin)
        if c < best_c5:
            best_c5 = c
            best_h = h_fin

    if best_h is None:
        best_h = np.full(50, 0.5)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
```
