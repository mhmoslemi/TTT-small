import numpy as np
import time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max()
    return float(max_val * dx)

def compute_c5_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, corr.max() * dx

def project_binary_init(h_frac, n):
    """Convert fractional to binary maintaining sum = n/2"""
    target = n // 2
    h = (h_frac >= 0.5).astype(float)
    current = int(h.sum())
    if current < target:
        # Need more 1s
        zeros = np.where(h == 0)[0]
        # Pick zeros with highest h_frac values
        order = np.argsort(h_frac[zeros])[::-1]
        for i in range(target - current):
            h[zeros[order[i]]] = 1.0
    elif current > target:
        # Need fewer 1s
        ones = np.where(h == 1)[0]
        order = np.argsort(h_frac[ones])
        for i in range(current - target):
            h[ones[order[i]]] = 0.0
    return h

def binary_sa(h_init, deadline, seed, n_smooth=20):

    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5, corr_best = compute_c5_lags(h)

    zeros = np.where(h == 0.0)[0]
    
    t_start = time.time()
    # Initial temperature
    T = 0.01
    T_min = 0.0001
    no_improve = 0
    max_no_improve = 500
    
    step = 0
    while time.time() < deadline - 0.5:
        # Adaptive temperature
        if no_improve > max_no_improve:
            T = 0.01
            no_improve = 0
            max_no_improve = 800
        step += 1
        T = max(T * 0.9997, T_min)
        
        # Choose move: single swap or double swap
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        move_type = rng.random()
        if move_type < 0.7:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5, new_corr = compute_c5_lags(h)
            delta = new_c5 - best_c5
            if new_c5 < best_c5 or rng.random() < math.exp(-delta / T):
                if new_c5 < best_c5:

                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:

                h[j] = 0.0
                no_improve += 1
        else:
            # Double swap
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), size=2, replace=False)
                j1, j2 = rng.choice(len(zeros), size=2, replace=False)
                i1, i2 = ones[i1], ones[i2]
                j1, j2 = zeros[j1], zeros[j2]
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                new_c5, _ = compute_c5_lags(h)
                delta = new_c5 - best_c5

                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
                    no_improve += 1
    
    return best_h, best_c5
