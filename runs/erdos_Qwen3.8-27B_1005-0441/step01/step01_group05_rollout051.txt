def project_feasible(h, n):
    target = float(n) / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Iterate: add/subtract to meet target
    for _ in range(10):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase some values
            room = 1.0 - h
            mask = room > 1e-6
            if not np.any(mask):
                break
            add = min(diff, room[mask].sum()) * 0.5
            h[mask] += add / mask.sum()
        else:
            # Need to decrease some values
            room = h
            mask = room > 1e-6
            if not np.any(mask):
                break
            sub = min(-diff, room[mask].sum()) * 0.5
            h[mask] -= sub / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    return h

    h_p = h + scale * rng.randn(n)
    return project_feasible(h_p, n)

def structured_perturb(h, n, seed):
    rng = np.random.RandomState(seed)
    h_p = h.copy()
    # Move mass in blocks
    block = max(3, n // 20)
    for _ in range(3):
        i = rng.randint(0, n - block)
        j = rng.randint(0, n - block)
        amount = rng.uniform(0.05, 0.15)
        h_p[i:i+block] += amount
        h_p[j:j+block] -= amount
    return project_feasible(h_p, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    remaining = budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(80, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Fast L-BFGS-B optimization at n0
    remaining = budget_s - (time.time() - t0)
    if remaining > 150:
        t_min = min(250, remaining - 80)
        h_opt, c5_opt = optimize_lbfgsb(h0, n0, t_min, seed)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 2: SLSQP with decreasing temperature
    remaining = budget_s - (time.time() - t0)
    if remaining > 100:
        t_min = min(150, remaining - 50)
        h_opt, c5_opt = optimize_slsqp(best_h, n0, t_min,
                                        [0.02, 0.005, 0.001, 0.0003, 0.0001])
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 3: Try different resolutions
    for target_n in [80, 120, 200]:
        remaining = budget_s - (time.time() - t0)
        if target_n == n0 or remaining < 70:
            continue
        h_r = resample(best_h, target_n)
        h_opt, c5_opt = optimize_lbfgsb(h_r, target_n, min(70, remaining - 40), seed + target_n)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 4: Many short restarts (L-BFGS-B is fast)
    restart = 0
    while True:
        remaining = budget_s - (time.time() - t0)
        if remaining < 35 or restart >= 100:
            break
        if restart % 3 == 0:
            h_pert = make_perturbed(best_h, 0.04 + 0.08 * (restart % 7) / 6.0, seed + 9000 + restart)
        else:
            h_pert = structured_perturb(best_h, len(best_h), seed + 9000 + restart)
        max_t = min(18, remaining - 20)
        if max_t < 8:
            break
        h_opt, c5_opt = optimize_lbfgsb(h_pert, len(best_h), max_t, seed + restart)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        restart += 1

    # Phase 5: Final SLSQP polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        h_opt, c5_opt = optimize_slsqp(best_h, len(best_h), min(50, remaining - 10),
                                        [0.001, 0.0003, 0.0001, 0.00005])
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Final L-BFGS-B with very high p
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        h_opt, c5_opt = optimize_lbfgsb(best_h, len(best_h), max(10, remaining - 5), seed + 99999)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    return best_h, best_c5, len(best_h)
