    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()

    v = np.zeros(n)
    beta = 0.8
    lr_cur = lr
    tau = tau_start
    t0 = time.time()

        corr = np.correlate(h, omh, mode='full')

        exps = np.exp(shifted)

        grad = np.zeros(n)

import numpy as np
import time
import math

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum} via bisection on offset."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.correlate(h, 1.0 - h, mode='full').max() * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_smooth(h, dx, tau):
    """Gradient of smooth max of correlations."""
    n = len(h)
    corr = get_corr(h)
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500.0, 500.0)
    exps = np.exp(shifted)
    exps_sum = exps.sum()
    if exps_sum < 1e-30:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exps / exps_sum
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # corr = sum_{i=lag}^{n-1} h[i]*(1-h[i-lag])
            # d/dh[j]: if j >= lag: (1-h[j-lag]); if j < n-lag: -h[j+lag]
            n_terms = n - lag
            grad[lag:] += w * (1.0 - h[:n_terms])
            grad[:n_terms] -= w * h[lag:]
        else:
            L = -lag
            n_terms = n - L
            # corr = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
            # d/dh[j]: if j < n-L: (1-h[j+L]); if j >= L: -h[j-L]
            grad[:n_terms] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n_terms]
    return grad * dx

    """Adam-like smooth max minimization."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(n)  # momentum
    v = np.zeros(n)  # variance (not used, just momentum)
    beta = 0.8
    t0 = time.time()
    deadline = t0 + budget
    no_imp = 0
    
    while time.time() < deadline - 0.5:
        g = compute_grad_smooth(h, dx, tau)
        m = beta * m + (1 - beta) * g
        h_new = project_feasible(h - lr * m, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:

        h = h_new
        if no_imp > 50:
            lr = max(lr * 0.95, 0.001)
        tau = min(tau * 1.002, 500.0)
    return best_h, best_c5

            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
                no_imp = 0
            else:
                # Accept with some probability based on elapsed time
                frac_elapsed = (time.time() - t0) / max(budget, 1)
                accept_prob = (1 - frac_elapsed) * 0.1
                if rng.random() > accept_prob:
                    h[i] = 1.0
                    h[j] = 0.0
                no_imp += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)
    
    n = 70
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        h_init = np.full(n, 0.5)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Phase 1: Smooth max gradient (300s)
    t_rem = deadline - time.time() - 300
    if t_rem > 30:
        h1, c1 = smooth_optimize(h_init, min(t_rem, 300), seed, tau=50, lr=0.04)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    
    # Phase 2: Second smooth pass with higher tau (150s)
    t_rem = deadline - time.time() - 250
    if t_rem > 20:
        h2, c2 = smooth_optimize(best_h, min(t_rem, 150), seed + 100, tau=100, lr=0.02)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
    
    # Phase 3: Binary greedy from fractional best (200s)
    t_rem = deadline - time.time() - 150
    if t_rem > 20:
        hb = round_to_binary(best_h, n)
        h3, c3 = binary_local_search(hb, min(t_rem, 200), seed + 200, mode='greedy')
        if c3 < best_c5:
            best_h, best_c5 = h3, c3
