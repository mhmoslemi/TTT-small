            v = compute_c5(h_new)
            if v < best_v:

                best_v = v
        except Exception:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    target = float(n) / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(40):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(free) > 0:
            h[free] += (target - s) / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def _grad_c5(h, temp, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    if temp < 1e-11:
        idx = np.argmax(corr)
        weights = np.zeros(len(corr))
        weights[idx] = 1.0
    else:
        weights = np.exp((corr - m) / temp)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            idx = np.argmax(corr)
            weights = np.zeros(len(corr))
            weights[idx] = 1.0
        else:
            weights /= wsum
    grad = np.zeros(n)
    nh = n - 1
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < 1e-14:
            continue
        s = j - nh
        if s >= 0:
            grad[s:] += wj * (1.0 - h[:n - s])
            grad[:n - s] -= wj * h[s:]
        else:
            t = -s
            grad[:n - t] += wj * (1.0 - h[t:])
            grad[t:] -= wj * h[:n - t]
    return grad * dx

def _smooth_obj(x, temp, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    if temp < 1e-11:
        return m * 2.0 / n
    return (m + temp * np.log(np.sum(np.exp((corr - m) / temp)))) * 2.0 / n

def _pgd(h, lr, temp, n, max_ls=30):
    g = _grad_c5(h, temp, n)
    f0 = _smooth_obj(h, temp, n)
    step = lr
    for _ in range(max_ls):
        h_new = _project(h - step * g, n)
        f_new = _smooth_obj(h_new, temp, n)
        if f_new <= f0 + 1e-14:
            return h_new, step
        step *= 0.5
    return h, 0.0

def _pgd_anneal(h, n, t0, max_time, schedules):

    best_v = compute_c5(h)
    for temp, n_iter, lr_init in schedules:
        if time.time() - t0 > max_time:
            break
        h = best_h.copy()
        lr = lr_init
        stagnant = 0
        for it in range(n_iter):
            if time.time() - t0 > max_time:
                break
            h_new, step_used = _pgd(h, lr, temp, n)
            if step_used > 0:
                v_new = compute_c5(h_new)
                if v_new < best_v - 1e-13:

                    best_v = v_new
                h = h_new
                stagnant = 0
                lr = min(lr * 1.01, 0.05)
            else:
                stagnant += 1
                lr *= 0.6
                if lr < 1e-7:
                    break
                if stagnant > 20:
                    break
    return best_h, best_v
