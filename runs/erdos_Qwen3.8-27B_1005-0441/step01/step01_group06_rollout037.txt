    m_amad = np.zeros(n)
    v_amad = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while time.time() < t_end:
        t += 1
        # Update beta
        beta = min(5.0 * (1.01 ** t), 10000.0)
        
        # Compute correlations
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        
        # Log-sum-exp
        max_c = corr.max()
        log_sum = max_c + np.log(np.sum(np.exp(corr - max_c))) / beta
        weights = np.exp(np.minimum(beta * (corr - max_c), 50))
        weights /= weights.sum()

        grad = np.zeros(n)

        # ∂corr[m]/∂h[i] = [i-m>=0](1-h[i-m]) - [i+m<n]h[i+m]

                if i + m < n:

def compute_grad(h, w):
    n = len(h)
    lags = np.arange(-(n-1), n)  # -(n-1), ..., n-1
    # w should be indexed by position in lags
    # w has length 2n-1
    
    grad = np.zeros(n)
    for i in range(n):
        # First term: sum over m ≤ i of w[m] * (1-h[i-m])
        m_range = lags[lags <= i]  # all lags ≤ i
        k_vals = i - m_range  # i - m, should be in [0, n-1]
        valid = (k_vals >= 0) & (k_vals < n)
        w_idx = m_range + (n-1)  # index into w array
        grad[i] += np.sum(w[w_idx[valid]] * (1.0 - h[k_vals[valid]]))
        
        # Second term: sum over m < n-i of w[m] * h[i+m]
        m_range2 = lags[lags < (n-i)]
        k_vals2 = i + m_range2
        valid2 = (k_vals2 >= 0) & (k_vals2 < n)
        w_idx2 = m_range2 + (n-1)
        grad[i] -= np.sum(w[w_idx2[valid2]] * h[k_vals2[valid2]])
    
    return grad

def adam_optimize(h0, t0, t_end, initial_beta=5.0, final_beta=5000.0):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    lags = np.arange(-(n-1), n)
    
    m_amad = np.zeros(n)
    v_amad = np.zeros(n)
    t = 0
    beta = initial_beta
    lr = 0.05
    
    while time.time() < t_end:
        t += 1
        beta = initial_beta * (final_beta / initial_beta) ** (t / 10000)
        beta = min(beta, final_beta)
        
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_c = corr.max()
        exp_vals = np.exp(np.minimum(beta * (corr - max_c), 50))
        w = exp_vals / exp_vals.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for i in range(n):
            # First term: sum over m ≤ i
            mask1 = lags <= i
            k1 = i - lags[mask1]
            valid1 = (k1 >= 0) & (k1 < n)
            w1 = w[mask1][valid1]
            h1 = h[k1[valid1]]
            grad[i] += np.sum(w1 * (1.0 - h1))
            
            # Second term: sum over m < n-i
            mask2 = lags < (n - i)
            k2 = i + lags[mask2]
            valid2 = (k2 >= 0) & (k2 < n)
            w2 = w[mask2][valid2]
            h2 = h[k2[valid2]]
            grad[i] -= np.sum(w2 * h2)
        
        # Project gradient to sum=0
        grad -= grad.mean()
        
        # ADAM
        lr_t = lr * np.sqrt(1 - 0.999**t) / (1 - 0.9**t)
        m_amad = 0.9 * m_amad + 0.1 * grad
        v_amad = 0.99 * v_amad + 0.01 * grad**2
        h = h - lr_t * m_amad / (np.sqrt(v_amad) + 1e-8)
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.minimum(h, 1.0)
        else:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.maximum(h, 0.0)
    return h

def make_binary(h, n, rng):
    b = (h > 0.5).astype(float)
    target = n // 2
    cur = int(b.sum())
    if cur < target:
        z = np.where(b == 0)[0]
        idx = rng.choice(z, target - cur, replace=False)
        b[idx] = 1.0
    elif cur > target:
        o = np.where(b == 1)[0]
        idx = rng.choice(o, cur - target, replace=False)
        b[idx] = 0.0
    return b

def sa_search(h0, t0, t_end, rng, T_init=0.12, cooling=0.99995):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T_init
    n_up = n // 2
    while time.time() < t_end:
        if T < 1e-8:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 <= best_c5 or rng.rand() < np.exp(-(new_c5 - best_c5) / T):
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling
    return best_h, best_c5

def sa_reheat(h0, t0, t_end, rng, n_reheats=3):
    h_curr = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    seg_time = (t_end - t0) / (n_reheats + 1)
    for r in range(n_reheats + 1):
        t_seg = t0 + (r + 1) * seg_time
        T_init = 0.12 / (r + 1)
        h_r, c5_r = sa_search(h_curr, t0, t_seg, rng, T_init=T_init)
        if c5_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c5_r
        h_curr = h_r
        if time.time() > t_end:
            break
    return best_h, best_c5

def adam_optimize(h0, t0, t_end, rng, beta_start=3.0, beta_end=8000.0, lr=0.08):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lags = np.arange(-(n - 1), n).astype(float)
    m_amad = np.zeros(n)
    v_amad = np.zeros(n)
    t = 0
    no_improve = 0
    while time.time() < t_end:
        t += 1
        beta = beta_start * (beta_end / beta_start) ** (min(t, 20000) / 20000.0)
        beta = min(beta, beta_end)
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        max_c = corr.max()
        scaled = np.minimum(beta * (corr - max_c), 40.0)
        exp_v = np.exp(scaled - scaled.max())
        w = exp_v / exp_v.sum()
        grad = np.zeros(n)
        for i in range(n):
            mask1 = lags <= i
            k1 = i - lags[mask1]
            valid1 = (k1 >= 0) & (k1 < n)
            wi = w[mask1][valid1]
            grad[i] = np.sum(wi * (1.0 - h[k1.astype(int)[valid1]]))
            mask2 = lags < (n - i)
            k2 = i + lags[mask2]
            valid2 = (k2 >= 0) & (k2 < n)
            wi2 = w[mask2][valid2]
            grad[i] -= np.sum(wi2 * h[k2.astype(int)[valid2]])
        grad -= grad.mean()
        lr_t = lr * np.sqrt(1.0 - 0.999**t) / (1.0 - 0.9**t)
        m_amad = 0.9 * m_amad + 0.1 * grad
        v_amad = 0.99 * v_amad + 0.01 * grad ** 2
        h_new = h - lr_t * m_amad / (np.sqrt(v_amad) + 1e-8)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 3000:
            h = best_h + rng.randn(n) * 0.02
            h = project_feasible(h, n)
            m_amad = np.zeros(n)
            v_amad = np.zeros(n)
            no_improve = 0
    return best_h, best_c5

        n_f = len(best_h)

            mc = c.max()
            sh = np.minimum(beta * (c - mc), 40)
