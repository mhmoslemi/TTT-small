        if wi < 1e-14:

        else:

            m = n - l
            if m > 0:
                grad[l:] += wi * (1.0 - h[:m])
            if l > 0 and m > 0:
                grad[:m] -= wi * h[l:]
    return grad

                       options={'maxiter': maxiter, 'ftol': 1e-14})

            g[l:] += (1.0 - h[:m])
        if l > 0 and m > 0:
            g[:m] -= h[l:]
