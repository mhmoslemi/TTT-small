    n = len(h)
    # Compute gradient of each c_k w.r.t. h[j]
    # c_k for correlation index k (0 to 2n-2), shift s = k - (n-1)
    g = np.zeros(2*n - 1)

        s = m - (n - 1)
        if s >= 0:
            # gradient contribution: (1-h[j-s])*1[s<=j<=n-1] - h[j]*1[j<=n-1-s]

            if s <= j <= n-1:
                grad += (1 - h[j-s])
            if 0 <= j <= n-1-s:
                grad -= h[j]

def compute_c5_correlation(h):
    """Compute correlation array and C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    return c, float(np.max(c))

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_simplex(h, target_sum):
    """Project h to {0 <= h <= 1, sum(h) = target_sum}."""
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        d = target_sum - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    # Final fix
    resid = target_sum - np.sum(h)
    if abs(resid) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

import numpy as np
from scipy.optimize import minimize
import time


def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c, float(np.max(c))


def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(30):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        h += d / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    resid = target - np.sum(h)
    if abs(resid) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += resid / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_g(h, j):
    n = len(h)
    g = np.zeros(2 * n - 1)
    s_range = np.arange(0, n)
    m_idx = n - 1 + s_range
    mask1 = s_range <= j
    if np.any(mask1):
        g[m_idx[mask1]] += 1.0 - h[j - s_range[mask1]]
    mask2 = s_range <= n - 1 - j
    if np.any(mask2):
        g[m_idx[mask2]] -= h[j + s_range[mask2]]
    t_range = np.arange(1, n)
    m_idx2 = n - 1 - t_range
    mask3 = t_range <= n - 1 - j
    if np.any(mask3):
        g[m_idx2[mask3]] += 1.0 - h[j + t_range[mask3]]
    mask4 = t_range <= j
    if np.any(mask4):
        g[m_idx2[mask4]] -= h[j - t_range[mask4]]
    return g


def coord_descent(h, timeout, t0):
    n = len(h)
    dx = 2.0 / n
    target = n * 0.5
    h = h.copy()
    c, cmax = c5_and_corr(h)
    best_c = cmax
    best_h = h.copy()
    no_improve = 0
    order = np.arange(n)
    for scan in range(100000):
        if time.time() - t0 > timeout:
            break
        np.random.shuffle(order)
        old_cmax = cmax
        for j in order:
            g = compute_g(h, j)
            lo, hi = -h[j], 1.0 - h[j]
            for _ in range(45):
                mid = 0.5 * (lo + hi)
                val = c + dx * g * mid
                kmax = int(np.argmax(val))
                sg = dx * g[kmax]
                if sg > 1e-16:
                    hi = mid
                elif sg < -1e-16:
                    lo = mid
                else:
                    break
            delta = 0.5 * (lo + hi)
            if abs(delta) > 1e-14:
                h[j] += delta
                h = project_sum(h, target)
                c, cmax = c5_and_corr(h)
        if cmax < best_c - 1e-15:
            best_c = cmax
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            break
    return best_h, best_c


def make_two_block(n):
    h = np.zeros(n)
    block = n // 4
    h[:block] = 1.0
    h[-block:] = 1.0
    return project_sum(h, n * 0.5)


def make_three_block(n):
    h = np.zeros(n)
    b = n // 6
    h[:b] = 1.0
    h[n // 2 - b:n // 2 + b] = 1.0
    h[-b:] = 1.0
    return project_sum(h, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

        pen = 50000.0

            m = np.max(corr)
            shifted = alpha * (corr - m)
            shifted = np.clip(shifted, -600.0, 600.0)
            base = (m + np.log(np.sum(np.exp(shifted))) / alpha) * dxx
            p = pen * (np.sum(x) - nn * 0.5) ** 2
            return base + p

        bounds = [(0.0, 1.0)] * n

        h_try = project_sum(h_try, try_n * 0.5)
        sub_deadline = min(time.time() + (budget_s - 50 - (time.time() - t0)) / 3.0,

            h_best = h_final

    h_best = project_sum(h_best, len(h_best) * 0.5)
    c5_best = c5_value(h_best)
    return h_best, c5_best, len(h_best)
