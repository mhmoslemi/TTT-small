    try:
        res = minimize(obj, project(h0), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project(res.x)
    except Exception:
        return project(h0.copy())

def smooth_slsqp(h0, beta, maxiter=200):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return smooth_val_grad(x, beta)[0]
    def jac(x):
        return smooth_val_grad(x, beta)[1]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, project(h0), jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project(res.x)
    except Exception:
        return project(h0.copy())

def simulated_annealing(h0, t_max, rng, T_start=0.08, T_end=0.0005):
    n = len(h0)
    h = h0.copy()
    cur_e = compute_c5(h)
    best = cur_e
    best_h = h.copy()
    t0 = time.time()
    n_total = max(1, int(t_max * 8))
    cooling = (T_end / T_start) ** (1.0 / n_total)
    T = T_start
    for step in range(n_total):
        if time.time() - t0 > t_max:
            break
        mt = rng.randint(3)
        if mt == 0:
            i, j = rng.choice(n, 2, replace=False)
            d_hi = min(1.0 - h[i], h[j])
            d_lo = max(-h[i], h[j] - 1.0)
            if d_hi <= d_lo + 1e-9:
                T *= cooling
                continue
            d = d_lo + rng.random() * (d_hi - d_lo)
            h[i] += d
            h[j] -= d
            new_e = compute_c5(h)
            de = new_e - cur_e
            if de < 0 or rng.random() < np.exp(-de / max(T, 1e-9)):
                cur_e = new_e
                if new_e < best:
                    best = new_e
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
        elif mt == 1:
            _, grad = smooth_val_grad(h, 40)
            h_new = project(h - 0.02 * grad)
            new_e = compute_c5(h_new)
            de = new_e - cur_e
            if de < 0 or rng.random() < np.exp(-de / max(T, 1e-9)):
                h = h_new
                cur_e = new_e
                if new_e < best:
                    best = new_e
                    best_h = h.copy()
        else:
            i, j = rng.choice(n, 2, replace=False)
            d = 0.03 * rng.randn()
            d = max(-h[i] + 1e-8, min(1 - h[i], d))
            d = max(-h[j] - (1 - 1e-8) + 1e-8, min(h[j] - 1e-8, -d))
            if abs(d) < 1e-9:
                continue
            h[i] += d
            h[j] -= d
            new_e = compute_c5(h)
            de = new_e - cur_e
            if de < 0 or rng.random() < np.exp(-de / max(T, 1e-9)):
                cur_e = new_e
                if new_e < best:
                    best = new_e
                    best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d
        T *= cooling
    return best_h, best

def coord_descent(h0, t_max, rng):
    n = len(h0)
    h = h0.copy()
    best = compute_c5(h)
    t0 = time.time()
    n_steps = int((t_max - (time.time() - t0)) * 3)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
    steps = 0
    while steps < n_steps and time.time() - t0 < t_max:
        j = rng.randint(n)
        k = (j + n // 3 + rng.randint(0, 5)) % n
        if k == j:
            k = (j + 1) % n
        d = deltas[rng.randint(len(deltas))]
        h_try = h.copy()
        h_try[j] += d
        h_try[k] -= d
        if 0 <= h_try[j] <= 1 and 0 <= h_try[k] <= 1:
            c_try = compute_c5(h_try)
            if c_try < best - 1e-14:
                best = c_try
                h = h_try
        steps += 1
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        n = 70
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0

    n = len(h_init)
    h_init = project(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP + gradient descent from initial (150s)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 100:
        h1 = slsqp_polish(best_h, maxiter=300, ftol=1e-14)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
        t_rem = budget_s - (time.time() - t_start)
        if t_rem > 60:
            h1b, c1b = grad_desc(best_h, min(80, t_rem - 10), betas=[20, 50, 100, 200])
            if c1b < best_c5:
                best_c5, best_h = c1b, h1b

    # Phase 2: Simulated annealing (400s)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 100:
        sa_time = min(350, t_rem - 50)
        h2, c2 = simulated_annealing(best_h, sa_time, rng)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: Multi-start local optimization (250s)
    t_rem = budget_s - (time.time() - t_start)
    n_restarts = min(int(t_rem / 50), 5)
    for i in range(n_restarts):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 50:
            break
        h_p = best_h.copy()
        ptype = i % 3
        if ptype == 0:
            h_p += 0.05 * rng.randn(n)
        elif ptype == 1:
            L = rng.randint(8, 20)
            s1 = rng.randint(0, n - L)
            s2 = (s1 + L + rng.randint(0, n - 2 * L)) % n
            h_p[s1:s1+L], h_p[s2:s2+L] = h_p[s2:s2+L].copy(), h_p[s1:s1+L].copy()
        else:
            h_p += 0.12 * rng.randn(n)
        h_p = project(h_p)
        t_alloc = min(45, t_rem - 5)
        # SLSQP polish
        hk = slsqp_polish(h_p, maxiter=200)
        ck = compute_c5(hk)
        if ck < best_c5:
            best_c5, best_h = ck, hk
        # Gradient descent follow-up
        t_rem2 = budget_s - (time.time() - t_start)
        if t_rem2 > 20:
            hk2, ck2 = grad_desc(hk, min(20, t_rem2 - 5), betas=[50, 150, 300])
            if ck2 < best_c5:
                best_c5, best_h = ck2, hk2

    # Phase 4: Smooth SLSQP for high-precision (120s)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 60:
        h4 = smooth_slsqp(best_h, beta=200, maxiter=150)
        c4 = compute_c5(h4)
        if c4 < best_c5:
            best_c5, best_h = c4, h4
        t_rem = budget_s - (time.time() - t_start)
        if t_rem > 40:
            h4b = slsqp_polish(best_h, maxiter=200, ftol=1e-15)
            c4b = compute_c5(h4b)
            if c4b < best_c5:
                best_c5, best_h = c4b, h4b

    # Phase 5: Coordinate descent micro-tuning (remaining)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 20:
        h5, c5 = coord_descent(best_h, min(t_rem - 5, 120), rng)
        if c5 < best_c5:
            best_c5, best_h = c5, h5

    # Final check
    best_h = project(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n)
