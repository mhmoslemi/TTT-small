            h_cur = _slsqp(h_cur, n, eps_schedule[ei], iters_per_eps[ei], deadline)
            v_cur = _c5(h_cur)

        # Subgradient polish
        h_cur, v_cur = _subgrad_refine(h_cur, n, 800, t0, deadline)

        # Fine SLSQP
        if v_cur < best_val + 0.005 and time.time() < deadline:
            for eps_f in [0.003, 0.001]:
                if time.time() > deadline:
                    break
                h_cur = _slsqp(h_cur, n, eps_f, 1000, deadline)
                v_cur = _c5(h_cur)

        if v_cur < best_val:
            best_val = v_cur
            best_h = h_cur.copy()

    # Final intensive polish
    if best_h is not None and time.time() < deadline:
        n = len(best_h)
        for eps_f in [0.004, 0.002, 0.001, 0.0005]:
            if time.time() > deadline:
                break
            best_h = _slsqp(best_h, n, eps_f, 1500, deadline)
            best_val = _c5(best_h)
            if best_val < 0.38085:
                break
        best_h, best_val = _subgrad_refine(best_h, n, 3000, t0, deadline)

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
