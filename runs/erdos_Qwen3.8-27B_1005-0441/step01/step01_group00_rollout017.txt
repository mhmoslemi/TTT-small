import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smooth_obj(h, eps, rho):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    val = (cmax + eps * np.log(s.sum())) * dx
    sdev = h.sum() - 0.5 * n
    val += 0.5 * rho * sdev * sdev
    return val

def _smooth_grad(h, eps, rho):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = c.max()
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = s.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-15:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    g *= dx
    sdev = h.sum() - 0.5 * n
    g += rho * sdev
    return g

def _proj(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)

        h = np.clip(h, 0.0, 1.0)
