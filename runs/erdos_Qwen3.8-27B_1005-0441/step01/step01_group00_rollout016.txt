def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = np.max(c)
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(np.sum(s))) * dx

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = np.max(c)
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = np.sum(s)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g * dx

def _proj(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += d / n
            h = np.clip(h, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

    m = b1 * m + (1 - b1) * g
    v = b2 * v + (1 - b2) * g * g

    h_new = h - lr * mhat / (np.sqrt(vhat) + 1e-8)

    return best_h, best_val

                # Actually for m<0: overlap = sum h[i]*(1-h[i+m]) = sum h[i]*(1-h[i-p])
                # Pairs: h[i]=1, h[i-p]=0
                # So i-p is the 0, i is the 1

                    j = i - p
                    if h[j] == 0 and i >= p:  # wait, h[i]=1, h[i-p]=0
                        # i-p is 0, i is 1

            # Let me redo: for m < 0, p = -m
            # overlap = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
            # Pairs: h[i]=1, h[i-p]=0
            bad_pairs = []
            for i in ones:
                j = i - p
                if j >= 0 and h[j] == 0:
                    bad_pairs.append(i)
        
        if len(bad_pairs) == 0:
            break
            
        # Try swapping: pick i from bad_pairs, swap h[i] (1) with h[j] (0)
        # where j is NOT the partner in the pair
        best_gain = 0
        best_move = None
        for i in bad_pairs[:10]:  # limit to first 10 for speed
            if m >= 0:
                partner = i + m
            else:
                partner = i - (-m)  # i + m = i - p
            
            for j in zeros:
                if j == partner:
                    continue
                # Tentative swap
                h[i], h[j] = h[j], h[i]
                c_new = _corr(h)
                new_max = np.max(c_new)
                h[i], h[j] = h[j], h[i]
                
                if new_max < best_val - 1e-10:
                    gain = best_val - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (i, j, new_max, c_new.copy())
        
        if best_move is None:
            break
        
        i, j, new_max, c_new = best_move
        h[i], h[j] = h[j], h[i]
        c = c_new
        best_val = new_max
    
    return h, best_val * 2.0 / n

import numpy as np
from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _smoobj(h, eps):
    n = len(h)
    c = _corr(h)
    cmax = np.max(c)
    shifted = c - cmax
    s = np.exp(shifted / eps)
    return (cmax + eps * np.log(np.sum(s))) * 2.0 / n

def _smograd(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    cmax = np.max(c)
    shifted = c - cmax
    s = np.exp(shifted / eps)
    ssum = np.sum(s)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = s[k] / ssum
        if wk < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= wk * h[m:]
        else:
            p = -m
            g[:n - p] += wk * (1.0 - h[p:])
            g[p:] -= wk * h[:n - p]
    return g * dx

def _proj(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(600):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += d / n
            h = np.clip(h, 0, 1)
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0, 1)
    return h

def _adam_optimize(h0, eps, n_steps, lr, t_limit):

        g = _smograd(h, eps)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1 - b1**t)
        vhat = v / (1 - b2**t)
        h = _proj(h - lr * mhat / (np.sqrt(vhat) + 1e-8), n)
        val = _c5(h)
        if val < best_val:

            best_h = h.copy()

    return x.sum() - 0.5 * n

    return np.ones(n)
