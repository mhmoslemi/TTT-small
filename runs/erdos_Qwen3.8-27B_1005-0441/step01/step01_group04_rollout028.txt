def _eval_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _corr_vals(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _soft_obj(h, alpha):
    corr = _corr_vals(h)
    cmax = np.max(corr)
    scaled = alpha * (corr - cmax)
    scaled = np.clip(scaled, -500, 500)
    return cmax + np.log(np.sum(np.exp(scaled))) / alpha

def _fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            idx = np.where(room > 1e-13)[0]
            if len(idx) == 0:
                break
            per = diff / len(idx)
            h[idx] += np.minimum(per, room[idx])
        else:
            idx = np.where(h > 1e-13)[0]
            if len(idx) == 0:
                break
            per = -diff / len(idx)
            h[idx] -= np.minimum(per, h[idx])
    return h
