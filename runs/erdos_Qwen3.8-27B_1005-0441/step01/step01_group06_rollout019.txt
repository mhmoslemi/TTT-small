            if c == 0:
                break
            h[m] = np.maximum(0.0, h[m] + diff / c)
    return np.clip(h, 0.0, 1.0)

def soft_max_fn(h, n, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = c.max()
    val = (m + np.log(np.sum(np.exp(beta * (c - m)))) / beta) * (2.0 / n)
    val += lam * (h.sum() - 0.5 * n) ** 2
    return val

def wcorr_grad(h, w):
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if abs(wk) > 1e-15:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
    for k in range(1, n):
        wk = w[n - 1 - k]
        if abs(wk) > 1e-15:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
    return grad

def proj_subgrad(h, n, max_iter, t0, deadline):
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(c.max() * 2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        beta = 30.0 + 15.0 * it
        cs = c - c.max()
        w = np.exp(beta * cs)
        ws = w.sum()
        if ws < 1e-30:
            break
        w /= ws
        g = wcorr_grad(h, w)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        lr = 0.8 / (1.0 + 0.05 * it)
        improved = False
        for scale in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_new = project_feasible(h - scale * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-12:
                h = h_new
                improved = True
                break
        if not improved and it > 50:
            break
    return best, best_c5

def lfbgbopt(h, n, t0, deadline):
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for beta in [10, 50, 200, 1000, 5000]:
        if time.time() > deadline:
            break
        lam = max(0.1, 10.0 / beta)
        maxiter = 150 if n <= 200 else 80
        try:
            res = minimize(soft_max_fn, h, args=(n, beta, lam),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best, best_c5

def try_binary(h_in, n):
    h_bin = (h_in > 0.5).astype(float)
    target = n / 2.0
    diff = target - h_bin.sum()
    if diff > 0:
        order = np.argsort(h_in)[::-1]
        for i in range(min(int(np.ceil(diff)), n)):
            h_bin[order[i]] = 1.0
    elif diff < 0:
        order = np.argsort(h_in)
        for i in range(min(int(np.ceil(-diff)), n)):
            h_bin[order[i]] = 0.0
    return project_feasible(h_bin, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 160
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    c5_0 = compute_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_0

    # Phase 1: L-BFGS-B main optimization
    dl1 = t0 + budget_s * 0.22
    h_opt, c5_opt = lfbgbopt(h0.copy(), n0, t0, dl1)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: Projected subgradient
    dl2 = t0 + budget_s * 0.38
    h_opt, c5_opt = proj_subgrad(best_h.copy(), len(best_h), 800, t0, dl2)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 3: Binary rounding + optimize
    if time.time() - t0 < budget_s * 0.60:
        nb = len(best_h)
        h_bin = try_binary(best_h, nb)
        dl3 = t0 + budget_s * 0.55
        h_opt, c5_opt = lfbgbopt(h_bin, nb, t0, dl3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        dl3b = t0 + budget_s * 0.62
        h_opt, c5_opt = proj_subgrad(h_bin, nb, 400, t0, dl3b)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Different resolutions
    nf = len(best_h)
    for scale in [0.5, 1.0 if nf != n0 else 1.5, 2.0, 1.5, 0.75]:
        if time.time() - t0 > budget_s * 0.78:
            break
        nt = int(round(nf * scale))
        if nt < 40 or nt > 800 or nt == nf:
            continue
        x_old = np.linspace(0, 2, nf, endpoint=False)
        x_new = np.linspace(0, 2, nt, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, nt)
        dlr = t0 + budget_s * 0.76
        h_opt, c5_opt = lfbgbopt(h_up, nt, t0, dlr)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Targeted local optimization at best resolution
    nf = len(best_h)
    dl5 = t0 + budget_s * 0.85
    h_opt, c5_opt = lfbgbopt(best_h.copy(), nf, t0, dl5)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    dl5b = t0 + budget_s * 0.92
    h_opt, c5_opt = proj_subgrad(best_h.copy(), nf, 600, t0, dl5b)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 6: Small random perturbations around best
    for rep in range(8):
        if time.time() - t0 > budget_s * 0.94:
            break
        sigma = 0.01 + 0.02 * rep / 8.0
        h_pert = best_h + rng.randn(nf) * sigma
        h_pert = project_feasible(h_pert, nf)
        dlp = t0 + budget_s * 0.93
        h_opt, c5_opt = lfbgbopt(h_pert, nf, t0, dlp)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final validation
    nf = len(best_h)
    best_h = project_feasible(best_h, nf)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-9:
        final_c5 = best_c5
    try:
        val = evaluate_erdos_solution(best_h)
        if val < final_c5:
            final_c5 = val
    except Exception:
        pass
    return (best_h, final_c5, nf)
