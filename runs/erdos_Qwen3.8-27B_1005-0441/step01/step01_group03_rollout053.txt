    lo, hi = -2.0, 2.0
    for _ in range(70):

        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:

        else:

        cur, cur_c = local_search_bin(best, n, max(0.05, tl))
        if cur_c < best_c:
            best_c = cur_c
            best = cur
        if r < n_restarts - 1:
            ones = np.where(best == 1.0)[0]
            zeros = np.where(best == 0.0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            n_swap = max(2, n // 8)
            for _ in range(n_swap):
                oi = int(np.random.choice(ones))
                zi = int(np.random.choice(zeros))
                best[oi] = 0.0
                best[zi] = 1.0
                ones = np.where(best == 1.0)[0]
                zeros = np.where(best == 0.0)[0]
    best, best_c = local_search_bin(best, n, max(0, time_limit - (time.time() - t0)))
    return best, best_c

def make_spaced(n, offset=0):
    h = np.zeros(n)
    half = n // 2
    if half < 1:
        return h
    positions = sorted(set((offset + i * (n + 1) // (half + 1)) % n for i in range(half)))
    for p in positions[:half]:
        h[p] = 1.0
    if int(np.sum(h)) < half:
        for idx, p in enumerate(positions[half:]):
            if h[p] < 1:
                h[p] = 1.0
                if int(np.sum(h)) == half:
                    break
    return h

def make_random_bin(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.permutation(n)[:half]
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_h = None
    best_c5 = 1.0
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    # Phase 1: Optimize initial construction (35% budget)
    if h0 is not None:
        n0 = len(h0)
        deadline1 = t0 + budget_s * 0.20
        h_cont = project_feasible(np.clip(h0, 0, 1), n0)
        bh, bc5 = optimize_adam_lse(h_cont, n0, deadline1, beta0=50.0, lr0=0.04)
        best_h, best_c5, best_n = bh, bc5, n0
        # Binary local search from rounded
        sorted_h = np.sort(bh)[::-1]
        hb = np.where(bh >= sorted_h[n0 // 2 - 1], 1.0, 0.0)
        if int(np.sum(hb)) != n0 // 2:
            hb = np.zeros(n0)
            hb[np.argsort(bh)[::-1][:n0 // 2]] = 1.0
        deadline1b = t0 + budget_s * 0.35
        hb_best, hb_c5 = iterated_ls(hb, n0, max(0.1, deadline1b - time.time()), n_restarts=15)
        if hb_c5 < best_c5:
            best_h, best_c5, best_n = hb_best, hb_c5, n0

    # Phase 2: Try other n values (45% budget)
    n_candidates = [34, 38, 40, 42, 44, 46, 48, 50, 52, 54, 56, 60, 64, 72, 80, 90, 100, 120]
    if h0 is not None:
        n_candidates = [x for x in n_candidates if x != len(h0)]
    deadline2 = t0 + budget_s * 0.82
    per_n = max(0.5, (deadline2 - time.time()) / len(n_candidates))

    for n_try in n_candidates:
        if time.time() > deadline2:
            break
        rem = deadline2 - time.time()
        if rem < 3:
            break
        alloc = min(per_n, rem)
        # Try multiple binary initials
        for init_type in range(3):
            if time.time() > deadline2:
                break
            if init_type == 0:
                h_bin = make_spaced(n_try, offset=rng.randint(0, n_try))
            elif init_type == 1:
                h_bin = make_random_bin(n_try, rng)
            else:
                h_bin = np.zeros(n_try)
                half = n_try // 2
                h_bin[:half] = 1.0
            # Binary local search
            dl_b = time.time() + alloc * 0.35
            hb_best, hb_c5 = iterated_ls(h_bin, n_try, max(0.1, dl_b - time.time()), n_restarts=8)
            if hb_c5 < best_c5:
                best_h, best_c5, best_n = hb_best, hb_c5, n_try
            # Continuous from rounded
            if time.time() < deadline2 - 2:
                h_cont_try = hb_best.astype(float)
                dl_c = time.time() + alloc * 0.3
                bh, bc5 = optimize_adam_lse(h_cont_try, n_try, min(dl_c, deadline2), beta0=40.0, lr0=0.03)
                if bc5 < best_c5:
                    best_h, best_c5, best_n = bh, bc5, n_try

    # Phase 3: Final refinement (18% budget)
    if best_h is not None and time.time() < t0 + budget_s - 5:
        deadline3 = t0 + budget_s - 2
        # Continuous polish
        dl_cont = time.time() + budget_s * 0.08
        if time.time() < deadline3 - 5:
            bh, bc5 = optimize_adam_lse(project_feasible(best_h, best_n), best_n,
                                        min(dl_cont, deadline3 - 5), beta0=100.0, lr0=0.02)
            if bc5 < best_c5:
                best_h, best_c5 = bh, bc5
        # Binary polish
        if int(np.sum(best_h)) != best_n // 2 or float(np.min(best_h)) > 0 or float(np.max(best_h)) < 1:
            hb3 = np.zeros(best_n)
            hb3[np.argsort(best_h)[::-1][:best_n // 2]] = 1.0
        else:
            hb3 = best_h.copy()
        dl_bin = deadline3 - 2
        hb3b, hb3c = iterated_ls(hb3, best_n, max(0.1, dl_bin - time.time()), n_restarts=30)
        if hb3c < best_c5:
            best_h, best_c5 = hb3b, hb3c

    # Final validation
    h_final = project_feasible(best_h, best_n)
    c5_final = get_c5(h_final, best_n)
    return h_final, float(c5_final), best_n
