    lo, hi = -2.0, 2.0
    for _ in range(80):

        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx), corr

def smooth_correlations(corr, tau):
    mc = corr.max()
    shifted = np.clip(tau * (corr - mc), -500, 500)
    exps = np.exp(shifted)
    return exps / exps.sum()

def compute_grad(h, dx, tau, top_m=0):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    if tau > 0:
        weights = smooth_correlations(corr, tau)
    else:
        idx = np.argsort(corr)[::-1][:top_m]

            grad[L:] -= w * h[:n - L]
    return grad * dx

def gran_descent(h_init, time_budget, seed, lr_init=0.05, tau_init=40.0, tau_max=500.0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    v = np.zeros(n)
    beta = 0.85
    lr = lr_init
    tau = tau_init
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    while time.time() < deadline - 0.2:
        grad = compute_grad(h, dx, tau)
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        tau = min(tau * 1.005, tau_max)
        if no_improve > 200:
            h = project_feasible(best_h + rng.randn(n) * 0.03, target_sum)
            v[:] = 0.0
            lr = lr_init
            tau = tau_init
            no_improve = 0
        elif no_improve > 80:
            lr = max(lr * 0.8, 0.002)
    return best_h, best_c5

def binary_local_search(h_init, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_ones = n // 2
    h = (h_init >= 0.5).astype(float)
    # Fix the number of ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) < target_ones:
        for _ in range(target_ones - len(ones)):
            h[zeros[0]] = 1.0
            zeros = zeros[1:]
    elif len(ones) > target_ones:
        for _ in range(len(ones) - target_ones):
            h[ones[0]] = 0.0
            ones = ones[1:]
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    while time.time() < deadline - 0.3:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5, _ = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def systematic_binary_search(h_init, time_budget, seed):
    """More systematic: try all pairs in a structured way"""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_ones = n // 2
    h = (h_init >= 0.5).astype(float)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) < target_ones:
        for idx in zeros[:target_ones - len(ones)]:
            h[idx] = 1.0
    elif len(ones) > target_ones:
        for idx in ones[:len(ones) - target_ones]:
            h[idx] = 0.0
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    # Multiple passes: in each pass, try all (i,j) pairs cyclically
    pass_num = 0
    while time.time() < deadline - 0.3:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Shuffle the order for diversity
        ones_shuf = ones[rng.permutation(len(ones))]
        zeros_shuf = zeros[rng.permutation(len(zeros))]
        improved = False
        for i in ones_shuf:
            for j in zeros_shuf:
                if time.time() > deadline - 0.3:
                    break
                h[i] = 0.0
                h[j] = 1.0
                c5, _ = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if time.time() > deadline - 0.3:
                break
        pass_num += 1
        if not improved and pass_num > 3:
            # Random perturbation
            ii = ones[rng.randint(len(ones))]
            jj = zeros[rng.randint(len(zeros))]
            h[ii] = 0.0
            h[jj] = 1.0
            c5, _ = compute_c5(h)
            if c5 > best_c5 + 0.01:
                h[ii] = 1.0
                h[jj] = 0.0
    return best_h, best_c5

import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx), corr

def get_weights(corr, tau, top_m):
    if tau > 0:
        mc = corr.max()
        shifted = np.clip(tau * (corr - mc), -500, 500)
        exps = np.exp(shifted)
        s = exps.sum()
        if s < 1e-300:
            idx = np.argmax(corr)
            w = np.zeros(len(corr))
            w[idx] = 1.0
            return w
        return exps / s
    else:
        idx = np.argsort(corr)[::-1][:top_m]
        w = np.zeros(len(corr))
        w[idx] = 1.0 / top_m
        return w

def compute_grad(h, dx, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            if n - L > 0:
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
    return grad * dx

import numpy as np

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx), corr

def get_weights(corr, tau, top_m):
    n_c = len(corr)
    if tau > 0:
        mc = corr.max()
        shifted = np.clip(tau * (corr - mc), -500, 500)
        exps = np.exp(shifted)
        s = exps.sum()
        if s < 1e-300:
            w = np.zeros(n_c)
            w[np.argmax(corr)] = 1.0
            return w
        return exps / s
    else:
        idx = np.argsort(corr)[::-1][:top_m]
        w = np.zeros(n_c)
        w[idx] = 1.0 / top_m
        return w

def compute_grad(h, dx, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            if n - L > 0:
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
    return grad * dx

        if it % 40 == 0:

        else:
            weights = get_weights(corr, tau, 0)
        grad = compute_grad(h, dx, weights)
        gnorm = np.linalg.norm(grad)

        v = beta * v - lr * grad / max(gnorm, 1e-10) * gnorm
