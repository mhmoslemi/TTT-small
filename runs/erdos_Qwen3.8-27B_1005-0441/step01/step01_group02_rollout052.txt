        if abs(s - target) < 1e-14:

    mc = np.max(corr)
    e = np.exp(beta * (corr - mc))
    return float((mc + np.log(e.sum()) / beta) * dx)

def _smooth_grad(h, beta):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    return h

def _logsumexp_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    e = np.exp(beta * (corr - mc))
    return float((mc + np.log(e.sum()) / beta) * dx)

def _logsumexp_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-14:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    return grad * dx

def _slsqp(h0, beta, maxiter=250):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return _logsumexp_val(x, beta)
    def jac(x):
        return _logsumexp_grad(x, beta)
    def con_fun(x):
        return [x.sum() - target]
    def con_jac(x):
        return np.ones((1, n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _project(res.x)
    except Exception:
        return _project(h0.copy())

                    h[zi] = 0.0
            # Continue even after a change (don't restart)
    return h, best_c5

def _pair_cd(h0, t_budget=20, steps=None):
    """Pairwise coordinate descent."""
    n = len(h0)
    h = _project(h0.copy())
    best_c5 = compute_c5(h)
    t0 = time.time()
    if steps is None:
        steps = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]
    outer = 0
    while time.time() - t0 < t_budget and outer < 200:
        outer += 1
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for s in steps:
                    h_try = h.copy()
                    h_try[i] += s
                    h_try[j] -= s
                    if 0.0 <= h_try[i] <= 1.0 and 0.0 <= h_try[j] <= 1.0:
                        c = compute_c5(h_try)
                        if c < best_c5 - 1e-13:
                            h = h_try
                            best_c5 = c
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def _top_k_smooth(h, k, beta):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            if n - d > 0:
                grad[0:n - d] += wi * (1.0 - h[d:n])
                grad[d:n] -= wi * h[0:n - d]
        else:
            dp = -d
            if n - dp > 0:
                grad[dp:n] += wi * (1.0 - h[0:n - dp])
                grad[0:n - dp] -= wi * h[dp:n]
    return grad * dx

def _slsqp(h0, beta, maxiter=200):
    n = len(h0)
    target = n / 2.0
    def obj(x):
        return _smooth_val(x, beta)
    def jac(x):
        return _smooth_grad(x, beta)
    def con_fun(x):
        return [x.sum() - target]
    def con_jac(x):
        return np.ones((1, n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _project(res.x)
    except Exception:
        return _project(h0.copy())

def _active_slsqp(h0, beta, top_k=5, maxiter=150):
    n = len(h0)
    target = n / 2.0
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    active = np.argsort(corr0)[::-1][:top_k]
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        topk = corr[active]
        mc = topk[0]
        e = np.exp(beta * (topk - mc))
        return float((mc + np.log(e.sum()) / beta) * (2.0 / n))
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        topk = corr[active]
        mc = topk[0]
        w = np.exp(beta * (topk - mc))
        ws = w.sum()
        if ws < 1e-300:
            return np.zeros(n)
        w /= ws
        grad = np.zeros(n)
        for ii in range(top_k):
            wi = w[ii]
            if wi < 1e-14:
                continue
            k = active[ii]
            d = k - (n - 1)
            if d >= 0:
                if n - d > 0:
                    grad[0:n-d] += wi * (1.0 - x[d:n])
                    grad[d:n] -= wi * x[0:n-d]
            else:
                dp = -d
                if n - dp > 0:
                    grad[dp:n] += wi * (1.0 - x[0:n-dp])
                    grad[0:n-dp] -= wi * x[dp:n]
        return grad * (2.0 / n)
    def con_fun(x):
        return [x.sum() - target]
    def con_jac(x):
        return np.ones((1, n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _project(res.x)
    except Exception:
        return _project(h0.copy())

def _binary_hill_climb(h0, t_budget=30):
    n = len(h0)
    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    best_c5 = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < t_budget:
        changed = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                c = compute_c5(h)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    changed = True
                else:
                    h[oi] = 1.0
                    h[zi] = 0.0
        if not changed:
            break
    return h, best_c5

def _pair_cd(h0, t_budget=25, step_list=None):
    n = len(h0)
    h = _project(h0.copy())
    best_c5 = compute_c5(h)
    t0 = time.time()
    if step_list is None:
        step_list = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]
    while time.time() - t0 < t_budget:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for s in step_list:
                    h[i] += s
                    h[j] -= s
                    if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                        c = compute_c5(h)
                        if c < best_c5 - 1e-14:
                            best_c5 = c
                            improved = True
                        else:
                            h[i] -= s
                            h[j] += s
                    else:
                        h[i] -= s
                        h[j] += s
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    if h_init is None:
        n = 70
        h_init = np.zeros(n)
        h_init[:n // 2] = 1.0
    else:
        h_init = h_init.copy()
    n = len(h_init)
    h_init = _project(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary hill climb
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 40:
        hb, cb = _binary_hill_climb(h_init, t_budget=min(35, t_rem - 5))
        h_pol = hb + 0.005 * rng.randn(n)
        h_pol = _project(h_pol)
        h_pol = _slsqp(h_pol, beta=80, maxiter=200)
        cp = compute_c5(h_pol)
        if cp < best_c5:
            best_c5 = cp
            best_h = h_pol
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()

    # Phase 2: Progressive full smooth-max SLSQP
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 80:
        h = best_h.copy()
        for beta in [30, 100, 300, 800, 2000]:
            if time.time() - t_start > budget_s - 40:
                break
            h = _slsqp(h, beta=beta, maxiter=250)
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h

    # Phase 3: Active-lag SLSQP (focused on top 5 lags)
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 60:
        for beta in [200, 600, 1500]:
            if time.time() - t_start > budget_s - 30:
                break
            h = _active_slsqp(best_h, beta=beta, top_k=5, maxiter=200)
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h

    # Phase 4: Pairwise coordinate descent
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 30:
        hc, cc = _pair_cd(best_h, t_budget=min(25, t_rem - 5))
        if cc < best_c5:
            best_c5 = cc
            best_h = hc

    # Phase 5: Diverse restarts
    t_rem = budget_s - (time.time() - t_start)
    n_restart = min(int(t_rem / 55), 6) if t_rem > 55 else 0
    for i in range(n_restart):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 50:
            break
        if i == 0:
            h_p = _project(best_h + 0.04 * rng.randn(n))
        elif i == 1:
            idx2 = np.argsort(best_h)[::-1]
            h_p = np.zeros(n)
            h_p[idx2[:n // 2]] = 1.0
            o = np.where(h_p == 1)[0]
            z = np.where(h_p == 0)[0]
            h_p[o[rng.randint(len(o))]], h_p[z[rng.randint(len(z))]] = 0.0, 1.0
            h_p = _project(h_p)
        elif i == 2:
            h_p = np.zeros(n)
            h_p[rng.choice(n, n // 2, replace=False)] = 1.0
        else:
            h_p = _project(best_h + 0.08 * rng.randn(n))
        h_p = _slsqp(h_p, beta=200, maxiter=200)
        h_p = _active_slsqp(h_p, beta=500, top_k=5, maxiter=150)
        c = compute_c5(h_p)
        if c < best_c5:
            best_c5 = c
            best_h = h_p

    # Phase 6: Final high-beta polish
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 25:
        h = _slsqp(best_h, beta=5000, maxiter=300)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h
        h = _active_slsqp(best_h, beta=3000, top_k=3, maxiter=200)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h

    # Phase 7: Fine coordinate descent
    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 15:
        hc, cc = _pair_cd(best_h, t_budget=min(12, t_rem - 3),
                          step_list=[0.003, -0.003, 0.005, -0.005, 0.01, -0.01])
        if cc < best_c5:
            best_c5 = cc
            best_h = hc

    best_h = _project(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, n)
