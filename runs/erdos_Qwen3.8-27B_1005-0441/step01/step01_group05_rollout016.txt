        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)

def compute_grad(h, w):
    """Compute gradient of sum_o w[o]*r[o] with respect to h.
    w is the weight vector of length 2n-1, indexed by offset o-(n-1).
    Returns gradient (without dx factor, caller supplies it)."""
    n = len(h)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        o = idx - (n - 1)
        wi = w[idx]
        if abs(wi) < 1e-15:
            continue
        # Term 1: (1-h[j+o]) for valid j
        jlo = max(0, -o)
        jhi = min(n - 1, n - 1 - o)
        if jlo <= jhi:
            grad[jlo:jhi + 1] += wi * (1.0 - h[jlo + o:jhi + o + 1])
        # Term 2: -h[j-o] for valid j
        jlo2 = max(0, o)
        jhi2 = min(n - 1, n - 1 + o)
        if jlo2 <= jhi2:
            grad[jlo2:jhi2 + 1] += -wi * h[jlo2 - o:jhi2 - o + 1]
    return grad

    n = len(h_init)
    dx = 2.0 / n
    target_sum = float(n) / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    # Temperature schedule (pieces)
    temps = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.0001, 0.00003]
    n_seg = n_max_steps // len(temps)
    
    for step in range(n_max_steps):
        if time.time() - t0 > max_time - 3:
            break
        
        seg = min(step // n_seg, len(temps) - 1)
        t_temp = temps[seg]
        
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        c_max = np.max(corr)
        exp_v = np.exp((corr - c_max) / t_temp)
        sw = np.sum(exp_v)
        w = exp_v / sw
        
        grad = compute_grad(h, w) * dx
        
        # Learning rate decay
        lr = lr_base / (1.0 + 0.3 * step / 5000)
        
        t_s = step + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_s)
        v_hat = v / (1.0 - beta2 ** t_s)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target_sum)

            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0

            h = h + amp * np.sin(2 * np.pi * k * np.arange(n) / n + ph)
            h = project_feasible(h, target_sum)
            m[:] = 0

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        h_mid = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_mid) > target_sum:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        mask = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_grad(h, w):
    n = len(h)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        o = idx - (n - 1)
        wi = w[idx]
        if abs(wi) < 1e-15:
            continue
        jlo = max(0, -o)
        jhi = min(n - 1, n - 1 - o)
        if jlo <= jhi:
            grad[jlo:jhi + 1] += wi * (1.0 - h[jlo + o:jhi + o + 1])
        jlo2 = max(0, o)
        jhi2 = min(n - 1, n - 1 + o)
        if jlo2 <= jhi2:
            grad[jlo2:jhi2 + 1] += -wi * h[jlo2 - o:jhi2 - o + 1]
    return grad

def adam_optimize(h_init, t0, max_time, lr_base=0.01, n_max_steps=40000, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    dx = 2.0 / n

    v = np.zeros(n)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    temps = [0.05, 0.02, 0.008, 0.003, 0.001, 0.0004, 0.00015, 0.00005]

    for step in range(n_max_steps):
        if time.time() - t0 > max_time - 3:

        c_max = np.max(corr)
        exp_v = np.exp((corr - c_max) / t_temp)
        w = exp_v / np.sum(exp_v)
        grad = compute_grad(h, w) * dx
        lr = lr_base / (1.0 + 0.4 * step / 6000.0)
        t_s = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad

def upsample_h(h_old, n_new):
    x_old = np.arange(len(h_old), dtype=float) * 2.0 / len(h_old)
    x_new = np.arange(n_new, dtype=float) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    target = float(n_new) / 2.0
    return project_feasible(h_new, target)

def init_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)[:n // 2]
    h[idx] = 1.0
    return h

def init_two_bump(n, rng, gap):
    x = np.linspace(0, 2, n, endpoint=False)
    c1 = rng.uniform(0.3, 0.6)
    c2 = min(1.7, c1 + gap)
    w = rng.uniform(0.2, 0.4)
    h = 0.6 * np.exp(-((x - c1) ** 2) / (2 * w * w)) + 0.6 * np.exp(-((x - c2) ** 2) / (2 * w * w))
    h = h / max(np.sum(h), 1e-12) * n / 2.0
    return project_feasible(np.clip(h, 0, 1), float(n) / 2.0)

def init_fourier_perturb(h_base, n, rng, scale=0.08):
    if len(h_base) != n:
        h_base = upsample_h(h_base, n)
    x = np.arange(n)
    pert = np.zeros(n)
    for _ in range(4):
        k = rng.randint(1, n // 3)
        ph = rng.uniform(0, 2.0 * np.pi)
        pert += np.sin(2.0 * np.pi * k * x / n + ph)
    pert *= scale / 2.0
    return project_feasible(h_base + pert, float(n) / 2.0)
