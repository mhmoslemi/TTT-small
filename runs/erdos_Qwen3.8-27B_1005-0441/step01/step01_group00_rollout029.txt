def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s * 0.97
    rng = np.random.RandomState(seed)
    best_h = None
    best_val = np.inf

    # Primary: n from initial_h_values
    n0 = None
    h0 = None
    try:
        h0 = _proj(np.array(initial_h_values, dtype=float), len(initial_h_values))
        n0 = len(h0)
    except Exception:
        pass

    if h0 is not None:
        best_h = h0.copy()
        best_val = _c5(best_h)
        best_h, v = _optimize_n(h0, n0, t_end)
        if v < best_val:
            best_val = v
            best_h = best_h.copy()
    else:
        n0 = 70
        h0 = _proj(np.full(n0, 0.5), n0)
        best_h = h0.copy()
        best_val = _c5(best_h)

    # Try n=100
    if time.time() < t_end - 80:
        n1 = 100
        h1 = _proj(0.5 + 0.35 * np.sin(2.0 * np.pi * 2.0 * np.arange(n1) / n1), n1)
        h1b, v1 = _optimize_n(h1, n1, min(t_end - 30, t0 + 250))
        if v1 < best_val:
            best_val = v1
            best_h = h1b

    # Try random structure at n0
    if time.time() < t_end - 60:
        n2 = n0
        h2 = _proj(rng.rand(n2), n2)
        h2b, v2 = _optimize_n(h2, n2, t_end - 10)
        if v2 < best_val:
            best_val = v2
            best_h = h2b

    # Final polish on best
    if best_h is not None and time.time() < t_end:
        nf = len(best_h)
        best_h, _ = _pair_sweep(best_h.copy(), t_end)
        if _c5(best_h) > best_val:
            best_h = best_h.copy()
            best_val = _c5(best_h)

    best_h = _proj(best_h, len(best_h))
    best_val = _c5(best_h)
    return (best_h, best_val, len(best_h))
