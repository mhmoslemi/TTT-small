import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def make_bounds(n):
    return [(0.0, 1.0)] * n

def sg_fun(x, n, beta, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    sh = beta * (corr - m)
    sh -= sh.max()
    e = np.exp(sh)
    p = e / e.sum()
    obj = (m + np.log(e.sum()) / beta) * dx
    grad = np.zeros(n)
    n_c = len(corr)
    for k in range(n_c):
        wk = p[k]
        if wk < 1e-11:
            continue
        shift = k - (n - 1)
        if shift > 0:
            grad[:n - shift] += wk * (1.0 - x[shift:])
            grad[shift:] -= wk * x[:n - shift]
        elif shift < 0:
            s = -shift
            grad[s:] += wk * (1.0 - x[:n - s])
            grad[:n - s] -= wk * x[s:]
        else:
            grad += wk * (1.0 - 2.0 * x)
    grad *= dx
    if pen > 0:
        s = np.sum(x) - n / 2.0
        obj += 0.5 * pen * s * s
        grad += pen * s
    return obj, grad

def do_lbfgs(h, n, t0, budget_s, frac=0.2):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = make_bounds(n)
    betas = [3, 10, 50, 200, 1000, 5000, 20000, 80000]
    for beta in betas:
        if time.time() - t0 > budget_s * (1 - frac):
            break
        try:
            res = minimize(sg_fun, h, args=(n, beta, 1.0), method='L-BFGS-B',
                          jac=True, bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            h = h_new
        except Exception:
            pass
    # Final high-beta polish
    if time.time() - t0 < budget_s * (1 - frac * 0.3):
        for beta in [50000, 200000]:
            try:
                res = minimize(sg_fun, best_h, args=(n, beta, 0.0), method='L-BFGS-B',
                              jac=True, bounds=bounds,
                              options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass
    return best_h, best_c5

def sgs_step(h, n, t_idx, use_nesterov=True):
    """Projected subgradient step targeting the max lag."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    if shift > 0:
        grad[:n - shift] += (1.0 - h[shift:])
        grad[shift:] -= h[:n - shift]
    elif shift < 0:
        s = -shift
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        grad[:] = 1.0 - 2.0 * h
    grad *= dx
    lr = 4.0 / (np.sqrt(t_idx + 1) + 1.0)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, float(corr[k_star] * dx)

def sgs_optimize(h0, n, t0, budget_s, frac=0.15, n_iter=2000):

        c5_h = compute_c5(h)

            best_c5 = c5_h
            best_h = h.copy()
        y_prev = h
    return best_h, best_c5
