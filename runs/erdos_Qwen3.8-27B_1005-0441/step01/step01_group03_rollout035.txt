            improved = True
    return best_h, best_c5

def sa_opt(h0, n, deadline, rng):

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        mask = (hp > 1e-13) & (hp < 1.0 - 1e-13)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] = np.clip(hp[mask] + diff / cnt, 0.0, 1.0)
    return hp

def c5_of(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def to_binary(h, n):
    idx = np.argsort(-h)
    r = np.zeros(n)
    r[idx[:n // 2]] = 1.0
    return r

def lse_grad(h, n, beta):
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode='full') * dx
    s = ov - ov.max()
    e = np.exp(beta * s)
    w = e / e.sum()
    g = np.zeros(n)
    for m in range(len(ov)):
        if w[m] < 1e-10:
            continue
        k = m - (n - 1)
        gs = np.ones(n)
        if k > 0:
            gs[:n - k] -= h[k:]
            gs[k:] -= h[:n - k]
        elif k < 0:
            kk = -k
            gs[kk:] -= h[:n - kk]
            gs[:n - kk] -= h[kk:]
        g += w[m] * gs
    return g * dx

def cont_opt(h, n, deadline, rng):
    best_h = h.copy()
    best_c5 = c5_of(h, n)
    y = h.copy()
    lr = 0.1
    mom = 0.8
    it = 0
    while time.time() < deadline:
        beta = 8.0 + 60.0 * min(1.0, it / 2500.0)
        g = lse_grad(y, n, beta)
        h_new = project_feasible(y - lr * g, n)
        y = project_feasible(h_new + mom * (h_new - h), n)
        h = h_new
        c5 = c5_of(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        it += 1
        if it % 600 == 0:
            lr *= 0.88
            mom *= 0.99
            if lr < 0.004:
                h = project_feasible(best_h + rng.randn(n) * 0.1, n)
                y = h.copy()
                lr, mom = 0.1, 0.8
    return best_h, best_c5

def hill_climb(h, n, deadline, max_passes=200):
    h = h.copy()
    cur = c5_of(h, n)
    best_h = h.copy()
    best_c5 = cur
    passes = 0
    while passes < max_passes and time.time() < deadline:
        passes += 1
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_swap = None
        best_nc = cur
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                nc = c5_of(h, n)
                h[i], h[j] = h[j], h[i]
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            cur = best_nc
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            improved = True
        if not improved:
            break
    return best_h, best_c5

def sa_opt(h0, n, deadline, rng, t_start=0.04, t_end=0.0003):
    h = h0.copy()
    cur = c5_of(h, n)
    best_h = h.copy()
    best_c5 = cur
    T = t_start
    while time.time() < deadline:
        T = max(t_end, T * 0.9996)
        for _ in range(400):
            if time.time() > deadline:
                break
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            nc = c5_of(h, n)
            d = nc - cur
            if d < 0 or rng.rand() < np.exp(-d / T):
                cur = nc
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def ills(h0, n, deadline, rng, n_restarts=5):
    """Iterated Local Search: SA + hill climb + perturb"""
    best_h = h0.copy()
    best_c5 = c5_of(h0, n)
    per_restart = (deadline - time.time()) / max(1, n_restarts)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        if r > 0:
            h_start = best_h.copy()
            ones = np.where(h_start == 1.0)[0]
            zeros = np.where(h_start == 0.0)[0]
            n_pert = 3 + rng.randint(3)
            for _ in range(n_pert):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_start[i], h_start[j] = h_start[j], h_start[i]
        else:
            h_start = h0.copy()
        dl = min(time.time() + per_restart * 0.7, deadline)
        bh, bc5 = sa_opt(h_start, n, dl, rng)
        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh
        dl2 = min(time.time() + per_restart * 0.3, deadline)
        hc, hc5 = hill_climb(bh, n, dl2, max_passes=100)
        if hc5 < best_c5:
            best_c5 = hc5
            best_h = hc
    return best_h, best_c5

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    starts = []
    hb_blk = np.zeros(n_use)

    hb_alt = np.zeros(n_use)
    hb_alt[0::2] = 1.0
    starts.append(to_binary(hb_alt, n_use))
