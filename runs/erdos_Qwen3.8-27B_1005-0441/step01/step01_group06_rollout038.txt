def targeted_binary_search(h, n, t0, time_limit, rng):
    """Targeted binary search focusing on the active (max) shift."""
    h = h.copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    max_val = float(corr.max())
    c5 = max_val * dx
    
    no_improve = 0
    patience = 5000
    
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Find the active shift
        max_idx = np.argmax(corr)
        s_star = max_idx - (n - 1)  # the actual shift
        
        # Strategy 1: Targeted - swap bits at positions contributing to max
        # Strategy 2: Random
        best_delta = 0.0
        best_i, best_j = -1, -1
        
        n_tries = 50
        for t in range(n_tries):
            if t < n_tries // 2 and s_star != 0:
                # Targeted: pick a 1 and a 0 near the contributing positions
                # c[n-1+s] = sum_i h[i]*(1-h[i+s])
                # Contributing pairs: i where h[i]=1 and h[i+s]=0
                if s_star > 0:
                    valid_i = np.where((h == 1))[0]
                    valid_i = valid_i[valid_i + s_star < n]
                    valid_i = valid_i[h[valid_i + s_star] == 0]
                else:
                    valid_i = np.where((h == 1))[0]
                    valid_i = valid_i[valid_i - s_star >= 0]
                    valid_i = valid_i[h[valid_i - s_star] == 0]
                
                if len(valid_i) > 0:
                    i = valid_i[rng.randint(len(valid_i))]
                else:
                    i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                # Random
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            
            if i == j:
                continue
            
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_corr(h)
            delta = float(c_new.max()) - max_val
            h[i] = 1.0
            h[j] = 0.0
            
            if delta < best_delta:
                best_delta = delta
                best_i = i
                best_j = j
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = compute_corr(h)
            max_val = float(corr.max())
            c5 = max_val * dx
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break
    
    return h, c5
