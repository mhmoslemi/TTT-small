import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smax_val(h, alpha, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    sh = np.clip(alpha * (corr - cmax), -500.0, 500.0)
    return (cmax + np.log(np.exp(sh).sum()) / alpha) * dx

def smax_grad(h, alpha, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    sh = np.clip(alpha * (corr - cmax), -500.0, 500.0)
    w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        i_lo = max(0, s)
        i_hi = min(n, s + n)
        if i_hi > i_lo:
            idx = np.arange(i_lo, i_hi)
            grad[idx] += wk * (1.0 - h[idx - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            idx2 = np.arange(i_lo2, i_hi2)
            grad[idx2] += -wk * h[idx2 + s]
    return grad * dx
