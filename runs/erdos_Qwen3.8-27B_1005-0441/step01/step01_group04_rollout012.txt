import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute the correction
    if diff > 0:
        # Add to elements with most room
        room = 1.0 - h
        order = np.argsort(-room)
        remaining = diff
        for i in order:
            add = min(remaining, room[i])
            h[i] += add
            remaining -= add
            if remaining < 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            sub = min(remaining, h[i])
            h[i] -= sub
            remaining -= sub
            if remaining < 1e-14:
                break
    return h

def _softmax_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    shifted -= np.max(shifted)
    p = np.exp(shifted)
    p_sum = np.sum(p)
    p /= p_sum
    obj = cmax + np.log(p_sum) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-14:
            continue
        s = m - n + 1
        if s >= 0:
            grad[0:n-s] += pm * (1.0 - h[s:n])
            grad[s:n] -= pm * h[0:n-s]
        else:
            s2 = -s
            grad[s2:n] += pm * (1.0 - h[0:n-s2])
            grad[0:n-s2] -= pm * h[s2:n]
    
    return obj, grad

def _project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    return _fix_sum(h, n)

def _grad_descent(h0, alpha, n_steps=200, verbose=False):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    obj, grad = _softmax_obj_and_grad(h, alpha)
    
    for it in range(n_steps):
        # Backtracking line search
        gamma = 0.1
        for _ in range(50):
            h_new = _project_feasible(h - gamma * grad, n)
            obj_new, _ = _softmax_obj_and_grad(h_new, alpha)
            if obj_new < obj - 1e-10 * abs(obj):
                break
            gamma *= 0.5
        else:
            break
            
        h = h_new
        obj, grad = _softmax_obj_and_grad(h, alpha)
        
        c5 = _eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def _slsqp_opt(h0, alpha, max_iter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj_fn(h):
        return _softmax_obj_and_grad(h, alpha)[0]
    
    def obj_grad(h):
        return _softmax_obj_and_grad(h, alpha)
    
    def obj_grad_combined(h):
        obj, grad = _softmax_obj_and_grad(h, alpha)
        return obj, grad
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0}]
    
    try:
        res = minimize(obj_grad_combined, h0, jac=True, method='SLSQP',
                      constraints=cons, bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-10})
        return res.x
    except Exception:
        return h0.copy()

4. Normalize weights

import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _fix_sum(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            # Add to the element with most room
            i = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[i])
            if add < 1e-15:
                break
            h[i] += add
        else:
            i = int(np.argmin(h))
            sub = min(-diff, h[i])
            if sub < 1e-15:
                break
            h[i] -= sub
    return h

def _corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _softgrad(h, alpha):
    """Compute softmax objective and gradient."""
    n = len(h)
    corr = _corr_vec(h)
    cmax = np.max(corr)
    a = alpha * (corr - cmax)
    a = a - np.max(a)
    e = np.exp(a)
    se = np.sum(e)
    p = e / se
    obj = cmax + np.log(se) / alpha
    
    # Gradient
    grad = np.zeros(n)
    # For each correlation index m, contribution to grad from that term
    # corr[m] = sum_i h[i]*(1-h[i+s]) where s = m-n+1
    # d/dh[j] = (1-h[j+s])*I[0<=j+s<n] - h[j-s]*I[0<=j-s<n]
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-13:
            continue
        s = m - n + 1
        if s >= 0:
            # j in [0, n-1-s]: contrib (1-h[j+s])
            grad[:n-s] += pm * (1.0 - h[s:])
            # j in [s, n-1]: contrib -h[j-s]
            grad[s:] -= pm * h[:n-s]
        else:
            s2 = -s
            # j in [s2, n-1]: contrib (1-h[j-s2])
            grad[s2:] += pm * (1.0 - h[:n-s2])
            # j in [0, n-1-s2]: contrib -h[j+s2]
            grad[:n-s2] -= pm * h[s2:]
    
    return obj, grad

def _gd_step(h, alpha, n_iter=100):
    """Gradient descent with backtracking line search."""
    n = len(h)
    h = h.copy()
    obj, grad = _softgrad(h, alpha)
    best_h = h.copy()
    best_obj = obj
    
    for it in range(n_iter):
        # Gradient step with backtracking
        gamma = max(0.5, 1.0 / (1 + 0.1 * it))
        improved = False
        for _ in range(40):
            h_new = _fix_sum(h - gamma * grad, n)
            obj_new, grad_new = _softgrad(h_new, alpha)
            if obj_new < obj - 1e-12:
                improved = True
                h = h_new
                obj = obj_new
                grad = grad_new
                break
            gamma *= 0.5
        if not improved:
            break
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
    
    return best_h

def _slsqp(h0, alpha, maxiter=300):
    """SLSQP optimization of softmax objective."""
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def fun(h):
        return _softgrad(h, alpha)[0]
    
    def jac(h):
        return _softgrad(h, alpha)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x
    except:
        return h0.copy()

def _minimax_slsqp(h0, maxiter=300, n_active=30):
    """Minimax SLSQP: minimize t s.t. t >= corr_k for active k."""
    n = len(h0)
    corr = _corr_vec(h0)
    t0 = np.max(corr)
    # Pick top n_active constraints
    top_idx = np.argsort(corr)[-n_active:]
    
    def fun(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = _corr_vec(h)
        return t - c[top_idx]
    
    def eq(x):
        return np.sum(x[:-1]) - n / 2.0
    
    x0 = np.append(h0, t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}]
    
    try:
        res = minimize(fun, x0, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x[:-1], res.x[-1]
    except:
        return h0.copy(), t0
