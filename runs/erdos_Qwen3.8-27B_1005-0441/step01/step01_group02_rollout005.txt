import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    shifted = np.clip(beta * (corr - mc), -700, 700)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
        val = mc * dx
    else:
        w /= ws
        val = (mc + np.log(ws) / beta) * dx
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-15:
            continue
        d = (n - 1) - i
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, d + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= wi * h[lo2 - d:hi2 - d]
    grad *= dx
    return val, grad

def project(h, target=None):

            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    return h

    n = len(h0)

    t0 = time.time()
    max_calls_out = [10000]

    def obj_pen(x):
        if time.time() - t0 > t_budget - 2:
            return 99999.0
        v, g = smooth_obj_grad(x, beta)
        pen = lam * (x.sum() - target) ** 2
        gp = g + 2.0 * lam * (x.sum() - target)
        return v + pen, gp

    try:
        res = minimize(lambda x: obj_pen(x)[0], h0,
                      jac=lambda x: obj_pen(x)[1],
                      method='L-BFGS-B', bounds=[(0, 1)] * n,
                      options={'maxiter': 8000, 'ftol': 1e-15,
