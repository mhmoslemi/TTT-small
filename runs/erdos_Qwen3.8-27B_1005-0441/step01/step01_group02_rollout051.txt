from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def to_binary(h, rng):
    n = len(h)
    target = n // 2
    h_noisy = h + 1e-8 * rng.randn(n)
    idx = np.argsort(h_noisy)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(800):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        h = np.clip(h + (target - s) / n, 0.0, 1.0)
    return h

def sa_binary(h0, rng, t_budget=30):
    n = len(h0)
    current = h0.copy()
    current_c5 = compute_c5(current)
    best_c5 = current_c5
    best = current.copy()
    t0 = time.time()
    T = 0.015
    T_min = 2e-5
    cool = 0.99988
    it = 0
    while time.time() - t0 < t_budget:
        T = max(T_min, T * cool)
        it += 1
        if it % 200 == 0:
            ones = np.flatnonzero(current == 1)
            zeros = np.flatnonzero(current == 0)
            if len(ones) == 0 or len(zeros) == 0:
                break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        current[i] = 0.0
        current[j] = 1.0
        new_c5 = compute_c5(current)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = current.copy()
                T *= 0.97
        else:
            current[i] = 1.0
            current[j] = 0.0
    return best, best_c5

def gradient_polish(h0, t_budget=25):
    n = len(h0)
    h = h0.copy().astype(float)
    t0 = time.time()
    def obj(x):
        return compute_c5(x)
    def con_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': con_fun}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-15})
        h = project(res.x)
    except Exception:
        pass
    rem = t_budget - (time.time() - t0)
    if rem > 5:
        try:
            res2 = minimize(obj, h.copy(), method='Nelder-Mead',
                          options={'maxiter': 1000, 'xatol': 1e-11, 'fatol': 1e-13})
            h = project(res2.x)
        except Exception:
            pass
    return h, compute_c5(h)

def make_pattern(n, kind, rng):
    h = np.zeros(n)
    half = n // 2
    if kind == 0:
        h[:half] = 1.0
    elif kind == 1:
        h[::2] = 1.0
        if h.sum() != half:
            idx = np.flatnonzero(h == 1)
            if h.sum() > half:
                h[idx[half]] = 0
            else:
                idx0 = np.flatnonzero(h == 0)
                h[idx0[0]] = 1
    elif kind == 2:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        pos = list(set(int((i * phi) % n) for i in range(half)))
        if len(pos) < half:
            rest = list(set(range(n)) - set(pos))
            pos += rest[:half - len(pos)]
        h[pos[:half]] = 1.0
    elif kind == 3:
        h[rng.choice(n, half, replace=False)] = 1.0
    elif kind == 4:
        q = n // 4
        h[:q] = 1.0
        h[n//2:n//2+q] = 1.0
    elif kind == 5:
        q = n // 3
        h[:q] = 1.0
        h[n//3 + q - 1:] = 1.0
    return h

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    starts = [h_init, to_binary(h_init, rng)]
    for k in range(6):
        starts.append(make_pattern(n, k, rng))
    for _ in range(4):
        starts.append(make_pattern(n, 3, rng))
    starts.sort(key=lambda x: compute_c5(x))

    for si in range(min(len(starts), 7)):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 18:
            break
        alloc = min(50, t_rem - 5) if si == 0 else min(25, t_rem - 5)
        h_sa, c_sa = sa_binary(starts[si], rng, t_budget=alloc)
        if c_sa < best_c5:
            best_c5 = c_sa
            best_h = h_sa

    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 20:
        h_pol, c_pol = gradient_polish(best_h, t_budget=min(22, t_rem - 3))
        if c_pol < best_c5:
            best_c5 = c_pol
            best_h = h_pol

    t_rem = budget_s - (time.time() - t_start)
    h_bin_best = to_binary(best_h, rng)
    n_extra = min(int(t_rem / 35), 5)
    for i in range(n_extra):
        t_rem = budget_s - (time.time() - t_start)
        if t_rem < 35:
            break
        h_pert = h_bin_best.copy()
        ns = rng.randint(2, 9)
        for _ in range(ns):
            ones = np.flatnonzero(h_pert == 1)
            zeros = np.flatnonzero(h_pert == 0)
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_pert[a], h_pert[b] = h_pert[b], h_pert[a]
        h_sa2, c_sa2 = sa_binary(h_pert, rng, t_budget=min(22, t_rem - 12))
        if t_rem > 18:
            h_p2, c_p2 = gradient_polish(h_sa2, t_budget=min(12, t_rem - 5))
            if c_p2 < c_sa2:
                c_sa2 = c_p2
                h_sa2 = h_p2
        if c_sa2 < best_c5:
            best_c5 = c_sa2
            best_h = h_sa2
            h_bin_best = to_binary(best_h, rng)

    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 20:
        h_fin, c_fin = gradient_polish(best_h, t_budget=min(t_rem - 3, 45))
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin

    t_rem = budget_s - (time.time() - t_start)
    if t_rem > 8:
        h_now = best_h.copy()
        for _ in range(50):
            improved = False
            for j in range(n):
                for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.015, -0.015]:
                    h_try = h_now.copy()
                    h_try[j] += delta
                    if h_try[j] < -0.02 or h_try[j] > 1.02:
                        continue
                    h_try = project(h_try)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5 - 1e-15:
                        best_c5 = c_try
                        best_h = h_try
                        h_now = h_try
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    best_h = project(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return (best_h, best_c5, n)
