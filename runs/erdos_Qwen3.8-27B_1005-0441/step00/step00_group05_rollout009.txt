import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def soft_max_weights(corr_vals, T):
    m = np.max(corr_vals)
    exps = np.exp((corr_vals - m) / T)
    return exps / np.sum(exps)

def grad_corr(h, j, n):
    s = j - (n - 1)
    grad = np.zeros(n)
    # First term: (1-h[m+s]) for m in [max(0,-s), min(n-1, n-1-s)]
    lo1 = max(0, -s)
    hi1 = min(n - 1, n - 1 - s)
    if lo1 <= hi1:
        grad[lo1:hi1+1] += (1.0 - h[lo1+s:hi1+s+1])
    # Second term: -h[m-s] for m in [max(s,0), min(n-1, n-1+s)]
    lo2 = max(s, 0)
    hi2 = min(n - 1, n - 1 + s)
    if lo2 <= hi2:
        grad[lo2:hi2+1] -= h[lo2-s:hi2-s+1]
    return grad

def project_sum(h, target):
    """Project h onto {x in [0,1]^n : sum(x) = target}"""
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda: find lambda s.t. sum(clip(h + lambda, 0, 1)) = target
    def f(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0)) - target
    if f(0) == 0:
        return h
    # f is increasing in lam

        if f(mid) > 0:
            hi = mid
        else:
            lo = mid

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

def grad_corr(h, j, n):
    """Gradient of the j-th correlation value w.r.t. h."""
    s = j - (n - 1)
    grad = np.zeros(n)
    # Term 1: (1-h[m+s]) for m in [max(0,-s), min(n-1, n-1-s)]
    lo1 = max(0, -s)
    hi1 = min(n - 1, n - 1 - s)
    if lo1 <= hi1:
        grad[lo1:hi1+1] += (1.0 - h[lo1+s:hi1+s+1])
    # Term 2: -h[m-s] for m in [max(s,0), min(n-1, n-1+s)]
    lo2 = max(0, s)
    hi2 = min(n - 1, n - 1 + s)
    if lo2 <= hi2:
        grad[lo2:hi2+1] -= h[lo2-s:hi2-s+1]
    return grad

def project_sum(h, target):
    """Project h onto {x in [0,1]^n : sum(x) = target} via bisection."""
    h = np.clip(h, 0.0, 1.0).copy()
    diff = h.sum() - target

    s = j - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -s)
    hi1 = min(n - 1, n - 1 - s)
    if lo1 <= hi1:
        grad[lo1:hi1 + 1] += (1.0 - h[lo1 + s:hi1 + s + 1])
