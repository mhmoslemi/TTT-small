    t0 = time.time()

    # Get initial solution
    ihv = globals().get('initial_h_values')
    if ihv is not None:
        best_h = np.array(ihv, dtype=float)
    else:
        best_h = binary_init(70, rng)
    n = len(best_h)
    best_c5 = eval_c5(best_h)

    # Phase 1: Binary local search (30% budget)
    t1 = min(budget_s * 0.3, budget_s - 100)
    if t0 + t1 < t0 + budget_s - 100:
        h_b, c5_b = binary_search(best_h, rng, t1)
        if c5_b < best_c5:
            best_h = h_b.copy()
            best_c5 = c5_b
        # Second pass from slightly different start
        h_b2 = fix_binary_sum(rng.rand(n), rng, n)
        if time.time() - t0 < budget_s - 150:
            t1b = min(budget_s * 0.15, budget_s - (time.time() - t0) - 150)
            if t1b > 5:
                h_b2, c5_b2 = binary_search(h_b2, rng, t1b)
                if c5_b2 < best_c5:
                    best_h = h_b2.copy()
                    best_c5 = c5_b2

    # Phase 2: Continuous SA (30% budget)
    elapsed = time.time() - t0
    t2 = min(budget_s * 0.30, budget_s - elapsed - 80)
    if t2 > 10:
        h_sa, c5_sa = sa_continuous(best_h, rng, t2, t_start=0.04)
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa

    # Phase 3: Fine-tuning SA (15% budget)
    elapsed = time.time() - t0
    t3 = min(budget_s * 0.15, budget_s - elapsed - 30)
    if t3 > 5:
        h_ft, c5_ft = sa_continuous(best_h, rng, t3, t_start=0.003)
        if c5_ft < best_c5:
            best_h = h_ft.copy()
            best_c5 = c5_ft

    # Phase 4: Try other n values with remaining time
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 20
    n_candidates = [98, 120, 160, 200, 300, 500]
    for n_try in n_candidates:
        if time.time() - t0 > budget_s - 80:
            break
        if n_try % 2 != 0:
            continue
        t_sub = min(remaining * 0.35, 90)
        if t_sub < 15:
            break
        remaining -= 100
        # Try multiple random starts
        best_sub_h = None
        best_sub_c5 = 1.0
        n_starts = min(3, max(1, int(t_sub / 25)))
        for s in range(n_starts):
            h_t = binary_init(n_try, rng)
            t_sub1 = t_sub / (n_starts + 1)
            h_t, c5_t = binary_search(h_t, rng, t_sub1)
            if c5_t < best_sub_c5:
                best_sub_c5 = c5_t
                best_sub_h = h_t.copy()
            if time.time() - t0 > budget_s - 80:
                break
        if best_sub_h is not None and best_sub_c5 < best_c5:
            # SA refinement
            t_sa = min(30, budget_s - (time.time() - t0) - 40)
            if t_sa > 5:
                h_t, c5_t = sa_continuous(best_sub_h, rng, t_sa, t_start=0.02)
                if c5_t < best_sub_c5:
                    best_sub_c5 = c5_t
                    best_sub_h = h_t
            if best_sub_c5 < best_c5:
                best_h = best_sub_h.copy()
                best_c5 = best_sub_c5
                n = n_try

    # Final validation and projection
    best_h = project(best_h, len(best_h))
    best_c5 = eval_c5(best_h)
    n_final = len(best_h)

    return best_h, best_c5, n_final
