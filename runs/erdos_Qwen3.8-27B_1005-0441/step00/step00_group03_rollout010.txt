import numpy as np
from scipy.optimize import minimize
import time

_N = 70
_TARGET = 35.0
_DX = 2.0 / 70
_ALPHA = 40.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, alpha):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha

def obj_fn(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr * dx, alpha)

def normalize(x, target):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if s > 1e-15:
        x = x * target / s
    return np.clip(x, 0.0, 1.0)

def _objective(x):
    return obj_fn(x, _ALPHA, _DX)

def _constraint(x):
    return float(np.sum(x) - _TARGET)

def _do_slsqp(h_start, bounds, cons, maxiter, ftol):
    try:
        res = minimize(_objective, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return normalize(res.x, _TARGET)
    except Exception:
        pass
    return None

def _local_search_pair(h, target, n, max_iter, time_limit):
    best_c5 = compute_c5(h)
    t_start = time.time()
    deltas = [0.01, 0.02, 0.05, 0.1]
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        if time.time() > time_limit:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = np.random.choice(deltas) * np.random.choice([1.0, -1.0])
        old_i, old_j = h[i], h[j]
        new_i = np.clip(old_i + delta, 0.0, 1.0)
        new_j = old_j - (new_i - old_i)
        new_j = np.clip(new_j, 0.0, 1.0)
        if abs(new_i + new_j - old_i - old_j) > 1e-12:
            continue
        h[i], h[j] = new_i, new_j
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
        else:
            h[i], h[j] = old_i, old_j
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _N, _TARGET, _DX, _ALPHA
    global initial_h_values
    np.random.seed(seed)
    t0 = time.time()

    n = 70
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0

    _N = n
    _TARGET = n / 2.0
    _DX = 2.0 / n
    _ALPHA = 40.0

    h0 = normalize(h0, _TARGET)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _constraint}]

    # Phase 1: SLSQP with increasing alpha for sharper max approximation
    for alpha in [20.0, 40.0, 80.0, 150.0, 300.0]:
        if time.time() - t0 > budget_s - 150:
            break
        _ALPHA = alpha
        h_try = _do_slsqp(best_h, bounds, cons, 3000, 1e-14)
        if h_try is not None:
            c = compute_c5(h_try)
            if c < best_c5:
                best_c5 = c
                best_h = h_try.copy()

    # Phase 2: Local pair-swap search
    remaining = budget_s - (time.time() - t0) - 80
    if remaining > 30:
        max_iter = min(200000, int(remaining / 0.002))
        best_h, best_c5 = _local_search_pair(best_h, _TARGET, n, max_iter, remaining)

    # Phase 3: Random restarts with SLSQP
    remaining = budget_s - (time.time() - t0) - 30
    if remaining > 60:
        n_restarts = min(15, int(remaining / 40))
        for r in range(n_restarts):
            if time.time() - t0 > budget_s - 30:
                break
            h_rand = np.random.rand(n)
            h_rand = normalize(h_rand, _TARGET)
            _ALPHA = 60.0
            h_try = _do_slsqp(h_rand, bounds, cons, 800, 1e-12)
            if h_try is not None:
                c = compute_c5(h_try)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_try.copy()

    # Phase 4: Fine-tune with high alpha SLSQP on best
    if time.time() - t0 < budget_s - 15:
        _ALPHA = 500.0
        h_try = _do_slsqp(best_h, bounds, cons, 2000, 1e-15)
        if h_try is not None:
            c = compute_c5(h_try)
            if c < best_c5:
                best_c5 = c
                best_h = h_try.copy()

    # Final polish: very small perturbations
    if time.time() - t0 < budget_s - 5:
        small_deltas = [0.005, 0.002, 0.001, 0.0005]
        for _ in range(5000):
            if time.time() - t0 > budget_s - 3:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:

            old_i, old_j = best_h[i], best_h[j]
            new_i = np.clip(old_i + delta, 0.0, 1.0)
            new_j = old_j - (new_i - old_i)
            if new_j < 0.0 or new_j > 1.0:
                continue
            best_h[i], best_h[j] = new_i, new_j
            c = compute_c5(best_h)
            if c < best_c5:
                best_c5 = c
            else:
                best_h[i], best_h[j] = old_i, old_j

    return best_h, best_c5, n
