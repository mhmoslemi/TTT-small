def slp_iteration(h_t, n):

    target_sum = n / 2.0
    
    h = cp.Variable(n)
    C = cp.Variable()
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
    ]
    
    for k in range(2*n - 1):
        m = k - (n - 1)
        S_lo = max(0, m)
        S_hi = min(n-1, n-1+m)
        
        # Linearized c[k] at h_t
        linear = cp.sum([h[i]*(1 - h_t[i-m]) - h_t[i]*h[i-m] for i in range(S_lo, S_hi+1)])
        const = np.sum(h_t[S_lo:S_hi+1] * h_t[S_lo-m:S_hi+1-m])

    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLOP)  # or CBC, or ECOS
    
    if h.value is not None and C.value is not None:
        return np.array(h.value).flatten(), C.value
    return h_t, np.inf

    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def slp_step(h_t, n, max_iter=50):
    """One SLP: linearize at h_t, solve LP"""
    dx = 2.0 / n
    target_sum = n / 2.0
    
    h = cp.Variable(n)
    C = cp.Variable()
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
    ]
    
    n_corr = 2 * n - 1
    for k in range(n_corr):
        m = k - (n - 1)
        S_lo = max(0, m)
        S_hi = min(n - 1, n - 1 + m)
        
        # Linearization of h[i]*h[i-m] at h_t:
        # h[i]*h_t[i-m] + h_t[i]*h[i-m] - h_t[i]*h_t[i-m]
        # So -h[i]*h[i-m] ≈ -h[i]*h_t[i-m] - h_t[i]*h[i-m] + h_t[i]*h_t[i-m]
        
        # c[k] = sum h[i] - sum h[i]*h[i-m]
        # ≈ sum h[i] - sum h[i]*h_t[i-m] - sum h_t[i]*h[i-m] + sum h_t[i]*h_t[i-m]
        
        # Linear part: sum h[i]*(1 - h_t[i-m]) - sum h_t[i]*h[i-m]
        # = sum (1-h_t[i-m])*h[i] - sum h_t[i]*h[i-m]

        coeff = np.zeros(n)
        for i in range(S_lo, S_hi + 1):
            coeff[i] += (1 - h_t[i - m])
            if 0 <= i - m < n:
                coeff[i - m] += (-h_t[i])

        h_new += diff / n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))


def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    free = (h > 1e-8) & (h < 1 - 1e-8)
    if free.sum() > 0:
        h[free] += diff / free.sum()
    else:
        h += diff / n
    h = np.clip(h, 0, 1)
    return h


def slp_solve(h_t, n, solver=None):

        h_opt, c_opt = slp_optimize(h_r, best_n, max_iters=35, time_limit=min(120, time_left() - 40))
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt
        attempt += 1

    # Phase 4: Final polish
    if time_left() > 30:
        h_f, c_f = slp_optimize(best_h, best_n, max_iters=25, time_limit=max(10, time_left() - 5))
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
        h_ls, c_ls = local_search(best_h, best_n, max_evals=200)
        if c_ls < best_c5:
            best_c5, best_h = c_ls, h_ls

    return best_h, best_c5, best_n
