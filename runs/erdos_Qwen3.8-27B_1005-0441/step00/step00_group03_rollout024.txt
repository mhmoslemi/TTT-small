import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    complement = 1.0 - h
    corr = np.correlate(h, complement, mode="full")
    return np.max(corr) * dx

def compute_all_f(h):
    n = len(h)
    dx = 2.0 / n
    complement = 1.0 - h
    corr = np.correlate(h, complement, mode="full")
    return corr * dx  # all lag values (not just max)

def subgradient(h, k):
    """Subgradient of f_k at h. k is the lag in [-n+1, n-1]."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # f_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j]: (1-h[j-k])*[j>=k] - h[j+k]*[j<=n-1-k]
        if k < n:
            # (1-h[j-k]) for j in [k, n-1]
            grad[k:] += (1 - h[:n-k])
            # -h[j+k] for j in [0, n-1-k]
            grad[:n-k] -= h[k:]
    else:
        kk = -k
        # f_k = sum_{i=kk}^{n-1} h[i-kk]*(1-h[i])
        # d/dh[j]: (1-h[j+kk])*[j<=n-1-kk] - h[j-kk]*[j>=kk]

            # (1-h[j+kk]) for j in [0, n-1-kk]
            grad[:n-kk] += (1 - h[kk:])
            # -h[j-kk] for j in [kk, n-1]
            grad[kk:] -= h[:n-kk]
    return grad

def project_constraint(v, target_sum):
    """Project v onto {0<=x<=1, sum(x)=target_sum}."""
    n = len(v)
    # Binary search on lambda
    lo, hi = -1.0, 2.0  # lambda range
    target = target_sum
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0, 1)

def projected_gradient(h, step_size, n_iter):
    """Run projected gradient descent to minimize max_k f_k(h)."""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iter):
        # Compute all f_k and find max
        f_all = compute_f_all(h)
        k_star = np.argmax(f_all) - (n-1)  # convert index to lag
        c_max = f_all[np.argmax(f_all)]
        
        if c_max < best_c5:
            best_c5 = c_max
            best_h = h.copy()
        
        # Subgradient
        g = subgradient(h, k_star)
        
        # Step
        h_new = h - step_size * g
        
        # Project
        h_new = project_constraint(h_new, target_sum)
        
        h = h_new
    
    return best_h, best_c5

def compute_f_all(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

So f_k as a function of h[j] is:
- Case 1 (j ≤ n-1-k and j ≥ k): h[j] - h[j]*h[j+k] + h[j-k] - h[j-k]*h[j]
  = h[j]*(1 - h[j+k] - h[j-k]) + h[j-k]
  This is LINEAR in h[j]!

- Case 2 (only j ≤ n-1-k): h[j]*(1-h[j+k]) + constant → linear in h[j]
- Case 3 (only j ≥ k): h[j-k]*(1-h[j]) + constant → linear in h[j]

Wait, that's interesting! f_k is LINEAR in h[j] for any fixed j! Because each h[j] appears in at most one product term (as the first or second factor), and the products are h[j]*h[j+k] or h[j-k]*h[j], both of which are linear in h[j].

So for coordinate descent:
- Fix all h[i] for i ≠ j
- f_k becomes linear in h[j] for each k
- min max_k f_k over h[j] ∈ [0,1] is a 1D convex optimization (min of max of linear functions)

The optimal h[j] is where the two largest slopes (at h[j]) cross, or at a boundary.

This makes coordinate descent much more efficient!

For each k, f_k(h) = a_k * h[j] + b_k where:
- a_k = coefficient of h[j] in f_k
- b_k = constant term

We want to find h[j] ∈ [0,1] that minimizes max_k (a_k * h[j] + b_k).

This is a Chebyshev approximation problem in 1D. The optimal solution is:
- If all a_k ≥ 0: minimize at h[j] = 0
- If all a_k ≤ 0: minimize at h[j] = 1
- Otherwise: find where the max is minimized, which is where two lines cross

Actually, the problem "minimize max_i (a_i x + b_i) over x ∈ [0,1]" can be solved by:
1. Compute the value at x=0: max_i b_i
2. Compute the value at x=1: max_i (a_i + b_i)
3. Find the crossing point if the minimizing line changes

More precisely, the solution is at the point where the maximum is minimized. This happens either at a boundary or where two of the lines (a_i x + b_i) achieve the maximum and cross.

A simple approach: binary search on the value t, checking if there exists x ∈ [0,1] with a_i x + b_i ≤ t for all i. This is equivalent to:
- For all i with a_i > 0: x ≤ (t - b_i) / a_i
- For all i with a_i < 0: x ≥ (t - b_i) / a_i
- For all i with a_i = 0: b_i ≤ t

So: max(0, max_{a_i<0} (t-b_i)/a_i) ≤ min(1, min_{a_i>0} (t-b_i)/a_i)

This gives a binary search on t.

import numpy as np
import time

# Helper functions (top-level)

def _compute_f_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def _compute_c5(h):
    return np.max(_compute_f_all(h))

def _subgrad_f(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[k:] += 1.0 - h[:m]
            g[:m] -= h[k:]
    else:
        kk = -k
        m = n - kk
        if m > 0:
            g[:m] += 1.0 - h[kk:]
            g[kk:] -= h[:m]
    return g

def _project(v, target):
    n = len(v)
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _optimize(h0, n_iter, step0, decay, top_k=3):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    h = _project(h, target)
    best = h.copy()
    best_c5 = _compute_c5(h)
    step = step0
    
    for it in range(n_iter):
        f_all = _compute_f_all(h)
        # Get top-k lags
        top_idx = np.argpartition(f_all, -top_k)[-top_k:]
        top_lags = top_idx - (n - 1)
        
        # Weighted subgradient
        g = np.zeros(n)
        for lag in top_lags:
            g += _subgrad_f(h, lag)
        g /= top_k
        
        c_max = float(np.max(f_all))
        if c_max < best_c5:
            best_c5 = c_max
            best = h.copy()
        
        h = h - step * g
        h = _project(h, target)
        step *= decay
    
    return best, best_c5

def _coord_descent(h0, n_passes):
    """Coordinate descent: try setting each h[j] to optimal 0/1/boundary."""
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    h = _project(h, target)
    best_c5 = _compute_c5(h)
    
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            # Try h[j] = 0 and h[j] = 1 (with projection for sum constraint)
            for val in [0.0, 1.0]:
                h_trial = h.copy()
                h_trial[j] = val
                h_trial = _project(h_trial, target)
                c5_trial = _compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-10:
                    h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

            best_h, best_c5, best_n = h_opt, c5_opt, n0
        
        # Fine-tune
        if remaining() > 10:
            n_iter2 = int(remaining() * 0.1)
            h_opt2, c5_opt2 = _optimize(best_h, n_iter2, 0.05, 0.9999)
            if c5_opt2 < best_c5:
                best_c5, best_h = c5_opt2, h_opt2
    except Exception:
        pass
