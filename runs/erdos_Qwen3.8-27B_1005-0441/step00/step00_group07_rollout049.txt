```python
    dx = 2.0 / n

    return np.max(corr) * dx

def project(h, n):
    """Project onto [0,1]^n with sum = n/2"""
    h = np.clip(h, 0, 1)
    diff = np.sum(h) - n / 2
    if diff > 0:
        # Need to decrease sum by diff
        # Reduce from largest values first
        order = np.argsort(-h)
        remaining = diff
        for i in order:
            reduce = min(h[i], remaining)
            h[i] -= reduce
            remaining -= reduce
            if remaining <= 0:
                break
    elif diff < 0:
        # Need to increase sum by -diff
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            increase = min(1 - h[i], remaining)
            h[i] += increase
            remaining -= increase
            if remaining <= 0:
                break
    return h
```

```python
    return float(np.max(corr) * dx)

    dx = 2.0 / n

def project_to_simplex(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target_sum
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h > 0)[0]
        step = diff / len(idx)
        h[idx] = np.maximum(0, h[idx] - step)
        # iterative fix
        for _ in range(10):
            diff = np.sum(h) - target_sum
            if abs(diff) < 1e-14:
                break
            idx = np.where(h > 0)[0]
            if len(idx) == 0:
                break
            h[idx] = np.maximum(0, h[idx] - diff / len(idx))
    else:
        idx = np.where(h < 1)[0]
        step = -diff / len(idx)
        h[idx] = np.minimum(1, h[idx] + step)
        for _ in range(10):

                break

            if len(idx) == 0:

    ones = np.where(h == 1)[0].copy()
```

```python
def grad_lag(h, k, n):
    """Gradient of correlation at lag k with respect to h.
    k is the actual lag (can be negative)."""
    grad = np.zeros(n)
    if k >= 0:
        # c_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j]: from h[j] appearing as h[i+k] (i=j-k): (1-h[j-k]) if valid
        #          from h[j] appearing as h[i] (i=j): -h[j+k] if valid
        if k < n:
            # First term: j = i+k, i from 0 to n-1-k, so j from k to n-1
            j1 = np.arange(k, n)
            grad[j1] += (1 - h[j1 - k])
            # Second term: j = i, i from 0 to n-1-k, so j from 0 to n-1-k
            j2 = np.arange(0, n - k)
            grad[j2] += (-1) * h[j2 + k]
    else:
        j_pos = -k
        # c_{-j_pos} = sum_{i=0}^{n-1-j_pos} h[i]*(1-h[i+j_pos])
        # d/dh[j]: from h[j] appearing as h[i] (i=j): (1-h[j+j_pos]) if valid
        #          from h[j] appearing as h[i+j_pos] (i=j-j_pos): -h[j-j_pos] if valid
        if j_pos < n:
            j1 = np.arange(0, n - j_pos)
            grad[j1] += (1 - h[j1 + j_pos])
            j2 = np.arange(j_pos, n)
            grad[j2] += (-1) * h[j2 - j_pos]
    return grad
```
