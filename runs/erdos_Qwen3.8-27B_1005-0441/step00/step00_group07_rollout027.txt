from scipy.optimize import minimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, temp=1.0):
    """Log-sum-exp approximation of max"""
    obj = np.asarray(obj)
    m = np.max(obj)
    if m > 50/temp:  # numerical stability
        return m
    return m + (1.0/temp) * np.log(np.sum(np.exp(temp * (obj - m))))

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(corr, temp)

        else:
            hi = mid

    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection."""
    h = np.array(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def active_gradient(h):
    """Subgradient of C5 at h (gradient of the active correlation term)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_best = np.argmax(corr)
    shift = k_best - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        js = j - shift
        if 0 <= js < n:
            grad[j] += (1.0 - h[js])
        jp = j + shift
        if 0 <= jp < n:
            grad[j] -= h[jp]
    return grad * dx

def subgradient_descent(h, target, max_iter, t0=0.5, decay=0.98):
    """Subgradient descent with projection."""
    n = len(h)
    h = project(h, target)
    best = h.copy()
    best_c5 = compute_c5(h)
    t = t0
    for it in range(max_iter):
        g = active_gradient(h)
        h_new = h - t * g
        h_new = project(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best = h_new.copy()
            best_c5 = c5_new
        h = h_new
        t *= decay
    return best, best_c5

def slsqp_optimize(h, target, temp, maxiter=200):
    """SLSQP with log-sum-exp smoothing."""
    n = len(h)
    dx = 2.0 / n
    def obj(z):
        corr = np.correlate(z, 1.0 - z, mode="full") * dx
        m = np.max(corr)
        shifted = corr - m
        scaled = temp * shifted
        max_s = np.max(scaled)
        if max_s > 50:
            return m + max_s / temp
        return float(m + (1.0 / temp) * np.log(np.sum(np.exp(scaled))))
    def obj_grad(z):
        corr = np.correlate(z, 1.0 - z, mode="full") * dx
        m = np.max(corr)
        shifted = corr - m
        scaled = temp * shifted
        max_s = np.max(scaled)
        if max_s > 50:
            e = np.zeros(len(corr))
            e[int(np.argmax(corr))] = 1.0
            return _corr_grad(z, np.argmax(corr)) * dx
        w = np.exp(scaled - max_s)
        w /= np.sum(w)
        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] < 1e-12:
                continue
            shift = k_idx - (n - 1)
            for j in range(n):
                js = j - shift
                if 0 <= js < n:
                    grad[j] += w[k_idx] * (1.0 - h[js])
                jp = j + shift
                if 0 <= jp < n:
                    grad[j] -= w[k_idx] * h[jp]
        return grad * dx
    try:
        res = minimize(obj, h.copy(), jac=obj_grad, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = project(res.x, target)
        return h_out, compute_c5(h_out)

        return h, compute_c5(h)
