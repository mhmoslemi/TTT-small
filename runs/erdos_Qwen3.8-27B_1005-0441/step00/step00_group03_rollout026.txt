import numpy as np
import time, math

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def compute_grads(h, dx, top_m, tau):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    if tau > 0:
        mc = corr.max()
        shifted = np.clip(tau * (corr - mc), -500, 500)
        exps = np.exp(shifted)
        weights = exps / exps.sum()
    else:
        idx = np.argsort(corr)[::-1][:top_m]
        weights = np.zeros(len(corr))
        weights[idx] = 1.0 / top_m
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            L = -lag
            grad[:n - L] += w * (1.0 - h[L:])
            grad[L:] -= w * h[:n - L]
    return grad * dx

def optimize(h_init, time_budget, seed, lr_init=0.08, use_smooth=True):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.8
    lr = lr_init
    tau = 60.0 if use_smooth else 0
    top_m = 5
    t0 = time.time()
    deadline = t0 + time_budget
    no_improve = 0
    while time.time() < deadline - 0.3:
        if no_improve > 400:
            h = project_feasible(best_h + rng.randn(n) * 0.04, target_sum)
            v[:] = 0.0
            lr = lr_init
            tau = 60.0 if use_smooth else 0
            no_improve = 0
        elif no_improve > 150:
            lr = max(lr * 0.85, 0.003)
        grad = compute_grads(h, dx, top_m, tau)
        v = beta * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if use_smooth:
            tau = min(tau * 1.004, 600.0)
    return best_h, best_c5

def binary_local_search(h_frac, time_budget, seed):
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    target_sum = int(n // 2)
    h = (h_frac >= 0.5).astype(float)
    for _ in range(target_sum - int(h.sum())):
        idx = np.argmin(h)
        h[idx] = 1.0
    excess = int(h.sum()) - target_sum
    for _ in range(excess):
        idx = np.argmax(h)
        h[idx] = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    deadline = t0 + time_budget
    while time.time() < deadline - 0.5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    n = 70
    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: smooth gradient descent
    t_rem = deadline - time.time() - 120
    if t_rem > 20:

        if c5_1 < best_c5:

    # Phase 2: top-m gradient from best
    t_rem = deadline - time.time() - 100

    # Phase 3: binary local search from fractional best
    t_rem = deadline - time.time() - 100

            best_h, best_c5 = h_frac2, c5_f2
        hb2, c5b2 = binary_local_search(h_frac2, min(t_rem - 160, 40), seed + 84)
        if c5b2 < best_c5:
            best_h, best_c5 = hb2, c5b2

    # Phase 4: random restarts
    for trial in range(5):
        t_rem = deadline - time.time() - 15
        if t_rem < 25:
            break
        rng_t = np.random.RandomState(seed + 700 + trial * 31)
        h_try = best_h + rng_t.randn(n) * 0.05
        h_try = project_feasible(h_try, target_sum)
        t_alloc = min(t_rem / (6 - trial), 70)

    # Phase 5: final polish
