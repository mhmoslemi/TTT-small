def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def swap_mutate(h, n_swaps, max_delta, rng):
    n = len(h)
    out = h.copy()
    for _ in range(n_swaps):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if out[i] >= out[j]:
            d = min(max_delta, out[i] - out[j])
            if d > 1e-15:
                out[i] -= d
                out[j] += d
        else:
            d = min(max_delta, out[j] - out[i])
            if d > 1e-15:
                out[j] -= d
                out[i] += d
    return out

def fix_sum(h, n):
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        for i in order:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= 0:
                break
    else:
        order = np.argsort(h)[::-1]
        for i in order:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= 0:
                break
    return np.clip(h, 0, 1)
