            continue
        h = h_new
        lr = min(lr * 1.02, 0.05)

    return best_h, best_c5

def perturb(h, mag, rng):
    n = len(h)
    S = n * 0.5
    hp = h + mag * rng.standard_normal(n)
    return project_feasible(hp, S)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(95) * 0.5

    n = len(h0)
    S = n * 0.5
    h0 = project_feasible(h0, S)

    best_h = np.copy(h0)
    best_c5 = np.max(compute_overlaps(h0))

    # Phase 1: main optimization from initial
    alloc = (time.time() - t0)
    remain = budget_s - alloc - 30
    if remain > 30:
        alloc1 = remain * 0.35
        h1, c1 = optimize_inner(h0, alloc1, best_c5)
        if c1 < best_c5:
            best_c5 = c1
            best_h = np.copy(h1)

    # Phase 2: restarts with perturbations
    remain = budget_s - (time.time() - t0) - 20
    if remain > 30:
        mags = [0.03, 0.06, 0.1, 0.15, 0.2, 0.08, 0.12]
        per = remain / len(mags)
        for i, mag in enumerate(mags):
            remain = budget_s - (time.time() - t0) - 15
            if remain < 15:
                break
            alloc_i = min(per * 1.2, remain)
            if alloc_i < 10:
                alloc_i = remain
            hp = perturb(best_h, mag, rng)
            hc, cc = optimize_inner(hp, alloc_i, best_c5)
            if cc < best_c5:
                best_c5 = cc
                best_h = np.copy(hc)

    # Phase 3: fine-tune with very high alpha (near-exact max)
    remain = budget_s - (time.time() - t0) - 5
    if remain > 15:
        alloc3 = min(remain, 60)
        hf, cf = optimize_inner(best_h, alloc3, best_c5)
        if cf < best_c5:
            best_c5 = cf
            best_h = np.copy(hf)

    # Final projection and evaluation
    best_h = project_feasible(best_h, S)
    final_c5 = np.max(compute_overlaps(best_h))
    if 'evaluate_erdos_solution' in globals():
        try:
            eval_c5 = evaluate_erdos_solution(best_h, 2.0 / n)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
        except Exception:
            pass

    # Safety: if final eval is worse, use the tracked best
    if final_c5 > best_c5 + 1e-6:
        final_c5 = best_c5

    return best_h, final_c5, n
