                if c5n < best_c5 - 1e-15:
                    best_h = hn
                    best_c5 = c5n
        except Exception:
            pass

        attempts += 1

    # Final polish with more iterations
    if elapsed() < budget_s - 10:
        try:
            res = do_slsqp(best_h, maxit=3000, ftol=1e-14)
            hn = fix_sum(np.clip(res.x, 0, 1), target_sum)
            c5n = c5_objective(hn, n_points)
            if c5n < best_c5:
                best_h, best_c5 = hn, c5n
        except Exception:
            pass

    # Verify sum
    best_h = fix_sum(best_h, target_sum)
    best_c5 = c5_objective(best_h, n_points)

    return best_h, best_c5, n_points
