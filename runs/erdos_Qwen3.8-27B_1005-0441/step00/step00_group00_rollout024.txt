    return m + np.log(np.sum(np.exp(tau * (corr - m)))) / tau

        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + s:hi1 + s])
        lo2, hi2 = max(0, s), min(N, N + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - s:hi2 - s]
    return grad

def optimize_phase(h, N, ts, tau, alpha, max_iters, t0, budget_s):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if budget_s - (time.time() - t0) < 12.0:
            break
        grad = smooth_grad(h, N, tau)
        h_new = project_box_sum(h - alpha * grad, ts)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            stal
l += 1
            if stall % 20 == 0:
                alpha *= 0.7
            h = project_box_sum(h - alpha * grad, ts)
            if stall > 2000:
                break
    return best_h, best_c5

def try_n_value(h_best, n_best, n_try, t0, budget_s, rng):
    if budget_s - (time.time() - t0) < 50:
        return None, None, 0
    ts = n_try / 2.0
    x_old = np.linspace(0, 2, n_best)
    x_new = np.linspace(0, 2, n_try)
    h = np.interp(x_new, x_old, h_best)
    h = project_box_sum(h, ts)
    best_c5 = compute_c5(h)
    for tau in [5.0, 20.0, 80.0, 200.0]:
        if budget_s - (time.time() - t0) < 15:
            break
        alpha = 0.4 / (tau ** 0.5)
        h, c5 = optimize_phase(h, n_try, ts, tau, alpha, 3000, t0, budget_s)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5, n_try

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    ts = n / 2.0
    h = project_box_sum(h0, ts)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Smoothed gradient with increasing tau
    taus = [2.0, 5.0, 10.0, 25.0, 50.0, 100.0, 200.0, 500.0]
    for tau in taus:
        if budget_s - (time.time() - t0) < 25:
            break
        alpha = 0.5 / (tau ** 0.7)
        h, c5 = optimize_phase(best_h, n, ts, tau, alpha, 4000, t0, budget_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Random restarts with short optimization
    for r in range(15):
        if budget_s - (time.time() - t0) < 30:
            break
        scale = 0.05 + 0.03 * r
        h = best_h + scale * rng.randn(n)
        h = project_box_sum(h, ts)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        for tau in [20.0, 100.0]:
            if budget_s - (time.time() - t0) < 15:
                break
            alpha = 0.3 / (tau ** 0.5)
            h, c5 = optimize_phase(h, n, ts, tau, alpha, 2000, t0, budget_s)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 3: Try different n values
    for n_try in [46, 50, 60, 80, 100, 120]:
        h_try, c5_try, _ = try_n_value(best_h, n, n_try, t0, budget_s, rng)
        if h_try is not None and c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            n = n_try

    # Phase 4: Final polish on best solution
    if budget_s - (time.time() - t0) > 30:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        for tau in [100.0, 300.0, 500.0]:
            if budget_s - (time.time() - t0) < 15:
                break
            alpha = 0.2 / (tau ** 0.5)
            best_h, c5 = optimize_phase(best_h, n_final, ts_final, tau, alpha, 5000, t0, budget_s)
            if c5 < best_c5:
                best_c5 = c5

    # Verify final solution
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
