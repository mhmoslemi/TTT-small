import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = h.astype(float).copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def eq_con(h, target):
    return np.sum(h) - target

def s1d(d, h, j, target):
    h_try = h.copy()
    h_try[j] = h[j] + d
    h_try = np.clip(h_try, 0.0, 1.0)
    h_try = fix_sum(h_try, target)
    return compute_c5(h_try)

def coord_polish(h, target):
    n = len(h)
    c5_cur = compute_c5(h)
    for _ in range(3):
        improved = False
        for j in range(n):
            d_lo = -h[j]
            d_hi = 1.0 - h[j]
            if d_hi - d_lo < 1e-10:
                continue
            try:

                                      method='bounded', options={'maxiter': 30, 'xatol': 1e-10})
                h_new = h.copy()
                h_new[j] += res.x
                h_new = fix_sum(np.clip(h_new, 0.0, 1.0), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h
