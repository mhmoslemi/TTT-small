```python
import numpy as np
from scipy.optimize import minimize
import time

# Module-level state for optimizer (set by run)
_gN = 70
_gDX = 2.0 / 70
_gTARGET = 35.0
_gNCORR = 139
_gLAGS = np.arange(139) - 69
_gA = None  # precomputed index matrix for term1
_gB = None  # precomputed index matrix for term2
_gVA = None
_gVB = None

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _build_index_matrices(n):
    ncorr = 2 * n - 1
    lags = np.arange(ncorr) - (n - 1)
    A = np.full((n, ncorr), 0, dtype=np.int64)
    B = np.full((n, ncorr), 0, dtype=np.int64)
    VA = np.zeros((n, ncorr), dtype=bool)
    VB = np.zeros((n, ncorr), dtype=bool)
    for j in range(n):
        for k in range(ncorr):
            jm = j - lags[k]
            jp = j + lags[k]
            if 0 <= jm < n:
                A[j, k] = jm
                VA[j, k] = True
            if 0 <= jp < n:
                B[j, k] = jp
                VB[j, k] = True
    return A, B, VA, VB

def _smooth_obj(h, alpha, mu):
    n = _gN
    ncorr = _gNCORR
    dx = _gDX
    target = _gTARGET
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = alpha * (corr - m)
    s -= s.max()
    exp_s = np.exp(s)
    sv = np.sum(h) - target
    obj = m + (s.max() + np.log(exp_s.sum())) / alpha + mu * sv * sv
    return obj

def _smooth_grad(h, alpha, mu):
    n = _gN
    ncorr = _gNCORR
    dx = _gDX
    target = _gTARGET
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = alpha * (corr - m)
    s -= s.max()
    exp_s = np.exp(s)
    w = exp_s / exp_s.sum()
    sv = np.sum(h) - target
    A, B, VA, VB = _gA, _gB, _gVA, _gVB
    w_row = w.reshape(1, ncorr)
    term1 = np.sum(VA * w_row * (1.0 - h[A]), axis=1)
    term2 = np.sum(VB * w_row * h[B], axis=1)
    grad = dx * (term1 - term2) + 2.0 * mu * sv
    return grad

def _run_lbfgs(h0, alpha, mu, maxiter, start_time, budget_s):
    n = _gN
    bounds = [(0.0, 1.0)] * n
    def _obj(x):
        return _smooth_obj(x, alpha, mu)
    def _grad(x):
        return _smooth_grad(x, alpha, mu)
    res = minimize(_obj, h0, jac=_grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                            'disp': False})
    h_new = np.clip(res.x, 0.0, 1.0)
    return h_new

def _local_search_binary(h_init, dx, target, max_iters, start_time, budget_s):
    n = len(h_init)
    h = np.round(h_init).clip(0, 1)
    # Fix sum
    while h.sum() > target:
        idx = np.random.choice(np.where(h > 0)[0])
        h[idx] -= 1.0
    while h.sum() < target:
        idx = np.random.choice(np.where(h < 1)[0])
        h[idx] += 1.0
    best_c5 = compute_c5(h, dx)
    stale = 0
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 10:
            break
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        i = one_idx[np.random.randint(len(one_idx))]
        j = zero_idx[np.random.randint(len(zero_idx))]
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            stale = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            stale += 1
        if stale > 500:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _gN, _gDX, _gTARGET, _gNCORR, _gLAGS, _gA, _gB, _gVA, _gVB
    start_time = time.time()
    np.random.seed(seed)

    # Get initial values
    try:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h_cur = np.random.uniform(0, 1, 70)
        h_cur *= 35.0 / h_cur.sum()
        h_cur = np.clip(h_cur, 0, 1)

    n = len(h_cur)
    _gN = n
    _gDX = 2.0 / n
    _gTARGET = n / 2.0
    _gNCORR = 2 * n - 1
    _gLAGS = np.arange(_gNCORR) - (n - 1)
    _gA, _gB, _gVA, _gVB = _build_index_matrices(n)
    dx = _gDX
    target = _gTARGET

    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h, dx)

    # Multi-start L-BFGS-B with increasing smoothing
    starts = [h_cur.copy()]
    for _ in range(2):
        hr = np.random.uniform(0, 1, n)
        hr *= target / hr.sum()
        hr = np.clip(hr, 0, 1)
        starts.append(hr)

    elapsed = time.time() - start_time
    for h0 in starts:
        for alpha in [50, 100, 200, 500, 1000]:
            if time.time() - start_time > budget_s - 60:
                break
            mu = 5.0
            maxiter = max(50, int(60.0 * (budget_s - elapsed) / 300.0))
            try:
                h_new = _run_lbfgs(h0, alpha, mu, maxiter, start_time, budget_s)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h0 = h_new.copy()
            except Exception:
                pass
        # Also try without penalty (fix sum by projection)
        try:
            h_fix = h0.copy()
            h_fix -= (h_fix.sum() - target) / n
            h_fix = np.clip(h_fix, 0, 1)
            c5_fix = compute_c5(h_fix, dx)
            if c5_fix < best_c5:
                best_c5 = c5_fix
                best_h = h_fix.copy()
        except Exception:
            pass

    # Phase 2: Binary local search from the best continuous solution
    if time.time() - start_time < budget_s - 20:
        h_bin, c5_bin = _local_search_binary(best_h, dx, target, 20000, start_time, budget_s)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3: Fine-tune continuous solution with small steps around best
    if time.time() - start_time < budget_s - 10:
        h_fine = best_h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s - 10:
                break
            j = np.random.randint(n)
            delta = np.random.uniform(-0.1, 0.1)
            h_trial = h_fine.copy()
            h_trial[j] = np.clip(h_trial[j] + delta, 0, 1)
            # Fix sum
            h_trial -= (h_trial.sum() - target) / n
            h_trial = np.clip(h_trial, 0, 1)
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h_fine = h_trial.copy()

    return best_h, best_c5, n
```
