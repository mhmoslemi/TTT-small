def gradient_for_lags(h, lags, n):

    grad = np.zeros(n)
    for k in lags:

        shift = k - (n - 1)
        # c_{shift} = sum_n h[n+shift] * (1-h[n])
        # d/dh_j: (1-h[j-shift]) - h[j+shift] (with bounds)

            grad[j] += val
    return grad

import time

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_smooth(h, alpha, n):
    """Gradient of (1/alpha)*log(sum(exp(alpha*corr_k)))"""
    corr = np.correlate(h, 1 - h, mode="full")
    cm = corr - np.max(corr)
    weights = np.exp(alpha * cm)
    sw = weights.sum()
    if sw < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= sw
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        # For shift s: c_s = sum_n h[n+s]*(1-h[n])
        # d/dh[j] = (1-h[j-s])*[0<=j-s<n] - h[j+s]*[0<=j+s<n]
        j_range = np.arange(n)
        # Term 1: j-s in [0,n-1]
        valid1 = (j_range - s >= 0) & (j_range - s < n)
        idx1 = j_range[valid1] - s
        grad[valid1] += w * (1.0 - h[idx1])
        # Term 2: j+s in [0,n-1]
        valid2 = (j_range + s >= 0) & (j_range + s < n)
        idx2 = j_range[valid2] + s
        grad[valid2] -= w * h[idx2]
    return grad

    target = n / 2.0
    h = project_feasible(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_C5(h)
    
    alpha = 20.0  # smoothness parameter

        # Actually, for minimization, we want the smooth max to be a good proxy
        # Higher alpha = closer to true max
        grad = gradient_smooth(h, alpha, n)
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_C5(h_new)
        
        if c5_new < best_c5 - 1e-9:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            patience = 0
            alpha = min(alpha * 1.05, 200.0)  # sharpen slightly

            lr *= 0.7

                lr = 0.02
                alpha = 20.0
                patience = 0
            if lr < 1e-5:

    return best_h, best_c5

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)

import numpy as np
import time

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_smooth(h, alpha, n):
    corr = np.correlate(h, 1 - h, mode="full")
    cm = corr - np.max(corr)
    weights = np.exp(alpha * cm)
    sw = weights.sum()
    if sw < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= sw
    grad = np.zeros(n)
    jr = np.arange(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        v1 = (jr - s >= 0) & (jr - s < n)
        if np.any(v1):
            grad[v1] += w * (1.0 - h[jr[v1] - s])
        v2 = (jr + s >= 0) & (jr + s < n)
        if np.any(v2):
            grad[v2] -= w * h[jr[v2] + s]
    return grad

def optimize_h(h0, n, deadline, rng):
    target = n / 2.0
    h = project_feasible(h0.astype(float), target)
    best_h = h.copy()
    best_c5 = compute_C5(h)
    alpha = 15.0
    lr = 0.03
    patience = 0
    best_lr = lr
    while time.time() < deadline:
        grad = gradient_smooth(h, alpha, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_C5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            patience = 0
            best_lr = lr
            lr = min(lr * 1.02, 0.06)
            alpha = min(alpha * 1.03, 300.0)
        else:
            patience += 1
            lr *= 0.75
            if patience > 25:
                h = best_h + rng.normal(0, 0.03, n)
                h = project_feasible(h, target)
                lr = best_lr
                alpha = 15.0
                patience = 0
            if lr < 1e-5 and alpha > 200 and patience > 60:
                break
    return best_h, best_c5

def _make_initial(n, rng):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    nf = rng.randint(2, 5)
    for _ in range(nf):
        freq = rng.randint(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.08, 0.25)
        h += amp * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, n / 2.0)

def _interp(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    return project_feasible(np.interp(xn, xo, h_old), n_new / 2.0)

    # Phase 3: fine-tune best
    if time.time() < deadline - 5:

    return (best_h, final_c5, int(n))
