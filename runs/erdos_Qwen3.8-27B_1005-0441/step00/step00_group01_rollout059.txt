    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    # log-sum-exp for numerical stability
    m = np.max(overlaps)
    return m + np.log(np.sum(np.exp(alpha * (overlaps - m)))) / alpha

def c5_softmax_grad(h, alpha):

    m = np.max(overlaps)

    hi = float(np.max(v)) + 1.5

            lo = mid
        else:
            hi = mid

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n

def project_feasible(v, n):
    S = n / 2.0
    lo = float(np.min(v)) - 1.5
    hi = float(np.max(v)) + 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n


def _obj(h):
    n = _g_n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ov = corr * (2.0 / n)
    m = float(np.max(ov))
    val = m + float(np.log(np.sum(np.exp(_g_alpha * (ov - m)))) / _g_alpha)
    val += _g_mu * (float(np.sum(h)) - n / 2.0) ** 2

    dx = 2.0 / n

    ov = corr * dx
    m = float(np.max(ov))

    g = np.zeros(n)
