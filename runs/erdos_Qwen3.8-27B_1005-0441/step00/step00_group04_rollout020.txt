        if s > target:
            lo = mid
        else:

    dx = 2.0 / n

import numpy as np
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_constraint(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_for_lag(h, n, k):
    """Subgradient for a specific lag k (actual shift, not index)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = dx * (1.0 - 2.0 * h)
        else:
            m = n - k
            grad[:m] += dx * (1.0 - h[k:])
            grad[k:] -= dx * h[:m]
    else:
        kp = -k
        m = n - kp
        grad[kp:] += dx * (1.0 - h[:m])
        grad[:m] -= dx * h[kp:]
    return grad

def optimize_subgradient(h0, n, start_time, time_limit, max_iter=50000):
    h = h0.astype(float).copy()
    h = project_constraint(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    vel = np.zeros(n)
    
    lr = 0.1
    mom = 0.85
    
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        
        overlaps = compute_overlaps(h)
        k_star = int(np.argmax(overlaps)) - (n - 1)
        grad = grad_for_lag(h, n, k_star)
        
        vel = mom * vel - lr * grad
        h_new = h + vel
        h_new = project_constraint(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Decrease step size
        lr *= 0.99995
        if lr < 0.001:
            lr = 0.001
    
    return best_h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    n = n_new
    h_new = project_constraint(h_new, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init = np.full(42, 0.5)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    n_work = 200
    h_work = upsample(h_init, n_work)
    
    # Generate several starting points
    x = np.linspace(0, 2, n_work, endpoint=False)

        h_c = project_constraint(h_c, n_work)
        starts.append(h_c)

    h_r = h_work + rng.randn(n_work) * 0.15
    h_r = np.clip(h_r, 0, 1)
    h_r = project_constraint(h_r, n_work)
    starts.append(h_r)
    
    # Optimize each starting point
    per_start_time = (budget_s - 30) / len(starts)
    for i, sp in enumerate(starts):
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 30
        if remaining < 10:
            break
        
        t_alloc = min(per_start_time, remaining)
        h_opt, c5_opt = optimize_subgradient(sp, n_work, start_time, elapsed + t_alloc)
        if c5_opt < best_c5:

            best_h = h_opt.copy()

import numpy as np
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_constraint(h, n):
    target = n / 2.0
    lo, hi = float(np.min(h)) - 1.0, float(np.max(h)) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_for_lag(h, n, k):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return grad
        grad[:m] += dx * (1.0 - h[k:k+m])
        if k > 0:
            grad[k:] -= dx * h[:m]
        else:
            grad[:] -= dx * h[:]
    else:
        kp = -k
        m = n - kp
        if m <= 0:
            return grad
        grad[kp:kp+m] += dx * (1.0 - h[:m])
        if kp > 0:
            grad[:m] -= dx * h[kp:kp+m]
    return grad

def smooth_grad(h, n, beta):
    overlaps = compute_overlaps(h)
    ov_max = np.max(overlaps)
    exps = np.exp(beta * (overlaps - ov_max))
    w = exps / np.sum(exps)
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-8:
            continue
        k = idx - (n - 1)
        if k >= 0:
            m = n - k
            if m <= 0:
                continue
            grad[:m] += wi * dx * (1.0 - h[k:k+m])
            if k > 0:
                grad[k:] -= wi * dx * h[:m]
            else:
                grad[:] -= wi * dx * h[:]
        else:
            kp = -k
            m = n - kp
            if m <= 0:
                continue
            grad[kp:kp+m] += wi * dx * (1.0 - h[:m])
            if kp > 0:
                grad[:m] -= wi * dx * h[kp:kp+m]
    return grad

def optimize_subgradient(h0, n, time_limit_s):
    h = h0.astype(float).copy()
    h = project_constraint(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    vel = np.zeros(n)
    lr = 0.08
    mom = 0.8
    beta = 15.0
    t_end = time.time() + time_limit_s
    patience = 0

    while time.time() < t_end:
        grad = smooth_grad(h, n, beta)
        vel = mom * vel - lr * grad
        h_new = project_constraint(h + vel, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h = h_new
        lr *= 0.99996
        if lr < 0.003:
            lr = 0.003
        beta = min(40.0, beta * 1.001)
        if patience > 5000 and lr <= 0.004:
            break
    return best_h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_constraint(h_new, n_new)

def try_round(h, n):
    """Try rounding to 0-1 while maintaining sum=n/2."""
    target = n // 2
    if n % 2 == 1:
        idx = np.argsort(h)[::-1]
        h_r = np.zeros(n)
        h_r[idx[:target]] = 1.0
        rem = target * 2 - n
        h_r[idx[target]] = rem
        h_r = project_constraint(h_r, n)
    else:
        idx = np.argsort(h)[::-1]
        h_r = np.zeros(n)
        h_r[idx[:target]] = 1.0
        h_r = project_constraint(h_r, n)
    return h_r

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init = np.full(42, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    n_work = 200
    h_work = upsample(h_init, n_work)
    x = np.linspace(0, 2, n_work, endpoint=False)

    starts = [h_work.copy()]
    for freq, amp in [(1.0, 0.2), (2.0, 0.15), (3.0, 0.1)]:
        h_c = np.clip(h_work + amp * np.cos(np.pi * freq * x), 0, 1)
        starts.append(project_constraint(h_c, n_work))
    h_r = h_work + rng.randn(n_work) * 0.12
    starts.append(project_constraint(np.clip(h_r, 0, 1), n_work))

    per_time = (budget_s - 40) / (len(starts) + 1)

    for i, sp in enumerate(starts):
        elapsed = time.time() - t_start
        rem = budget_s - elapsed - 40
        if rem < 15:
            break
        alloc = min(per_time, rem)
        h_opt, c5_opt = optimize_subgradient(sp, n_work, alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Try rounding the best solution
    elapsed = time.time() - t_start
    if elapsed < budget_s - 20:
        h_rnd = try_round(best_h, n_work)
        c5_rnd = compute_c5(h_rnd)
        if c5_rnd < best_c5:
            best_c5 = c5_rnd
            best_h = h_rnd.copy()
        # Refine rounded solution
        rem = budget_s - elapsed - 20
        if rem > 20:
            h_opt2, c5_opt2 = optimize_subgradient(h_rnd, n_work, rem * 0.5)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()

    # Higher resolution refinement
    elapsed = time.time() - t_start
    if elapsed < budget_s - 50:
        n_high = 350
        h_high = upsample(best_h, n_high)
        rem = budget_s - elapsed - 20
        h_opt3, c5_opt3 = optimize_subgradient(h_high, n_high, rem)
        if c5_opt3 < best_c5:
            best_c5 = c5_opt3
            best_h = h_opt3.copy()

    # Final cleanup
    n = len(best_h)
    best_h = project_constraint(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
