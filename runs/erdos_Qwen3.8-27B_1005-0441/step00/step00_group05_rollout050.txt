def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def _grad(h, n, weights=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if weights is None:
        max_val = np.max(corr)
        weights = np.where(corr >= max_val - 1e-4, 1.0, 0.0)
        if weights.sum() == 0:
            weights[int(np.argmax(corr))] = 1.0
        weights = weights / weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        w = weights[k]
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            ln = -lag
            grad[:n - ln] += w * (1.0 - h[ln:])
            grad[ln:] -= w * h[:n - ln]
    return grad * dx
