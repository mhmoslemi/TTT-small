import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_set(h, S):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgrad(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_val = np.max(corr) * dx
    idx_max = np.argmax(corr)
    k = idx_max - (n - 1)
    grad = np.zeros(n)
    a = max(0, k)
    b = min(n - 1, n - 1 + k)
    if a <= b:
        grad[a:b + 1] += (1.0 - h[a - k:b - k + 1])

    b2 = min(n - 1, n - 1 - k)
    if a2 <= b2:
        grad[a2:b2 + 1] -= h[a2 + k:b2 + k + 1]
    return c5_val, grad * dx
