    overlaps = compute_overlaps(h)
    max_val = float(np.max(overlaps))
    active = np.where(overlaps >= max_val - 1e-12)[0]

        v2 = (jp >= 0) & (jp < n)
        g[v2] -= h[jp[v2]]

import numpy as np
import time


def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, dx):
    return float(np.max(compute_overlaps(h))) * dx


def compute_subgradient(h, dx):
    n = len(h)
    ov = compute_overlaps(h)
    mx = float(np.max(ov))
    act = np.where(ov >= mx - 1e-12)[0]
    j = np.arange(n)
    grad = np.zeros(n)
    for ki in act:
        s = int(ki) - (n - 1)
        g = np.zeros(n)
        jm = j - s
        v1 = (jm >= 0) & (jm < n)
        g[v1] += 1.0 - h[jm[v1]]
        jp = j + s
        v2 = (jp >= 0) & (jp < n)
        g[v2] -= h[jp[v2]]
        grad += g
    grad *= dx / float(len(act))
    return grad, mx * dx


def project(h, ts):
    hm = max(1.0, float(np.max(np.abs(h))))
    lo, hi = -hm - 2.0, hm + 2.0
    for _ in range(200):

        v = np.sum(np.clip(h + mid, 0.0, 1.0))
        if abs(v - ts) < 1e-14:
            break
        if v < ts:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    r = np.clip(h + t, 0.0, 1.0)
    d = ts - np.sum(r)
    if abs(d) > 1e-14:
        m = (r > 1e-12) & (r < 1.0 - 1e-12)
        c = int(np.sum(m))
        if c > 0:
            r[m] += d / c
            r = np.clip(r, 0.0, 1.0)
    return r
