import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_simplex(v, target):
    """Project v onto {x: sum(x)=target, 0<=x_i<=1} via bisection on offset."""
    lo, hi = v.min() - target - 1.0, v.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix tiny numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_min_weights(ac, temp):
    m = np.min(ac)
    e = np.exp(-(ac - m) / temp)
    return e / e.sum()

def compute_grad_smooth_min(h, w, dx, n):
    """Gradient of smooth-min autocorrelation w.r.t. h."""
    grad = np.zeros(n)
    n_ac = len(w)
    for k_off in range(n_ac):
        wi = w[k_off]
        if wi < 1e-12:
            continue
        lag = k_off - (n - 1)
        if lag == 0:
            grad += wi * 2.0 * dx * h
        elif lag > 0:
            grad[lag:] += wi * dx * h[:-lag]
            grad[:-lag] += wi * dx * h[lag:]
        else:
            la = -lag
            grad[:-la] += wi * dx * h[la:]
            grad[la:] += wi * dx * h[:-la]
    return grad

def optimize_h(h0, n, dx, budget_s, rng, max_iter=8000):
    """Projected gradient ascent on min-autocorrelation. Returns best (h, c5)."""
    start = time.time()
    h = np.copy(h0)
    best_h = np.copy(h)
    best_c5 = compute_c5(h, n)
    lr = 0.005
    temp = 0.05
    no_imp = 0
    best_no_imp_reset = 0

    for it in range(max_iter):
        if time.time() - start > budget_s:
            break
        ac = np.correlate(h, h, mode='full') * dx
        min_ac = np.min(ac)
        c5 = 1.0 - min_ac
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = np.copy(h)
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 300:
            temp *= 0.6
            lr *= 0.6
            no_imp = 0
            if temp < 1e-7:
                break
        w = smooth_min_weights(ac, temp)
        grad = compute_grad_smooth_min(h, w, dx, n)
        # Line search
        accepted = False
        for trial in range(10):
            step = lr / (2.0 ** trial)
            h_new = project_simplex(h + step * grad, n * 0.5)
            ac_new = np.correlate(h_new, h_new, mode='full') * dx
            if np.min(ac_new) > min_ac - 1e-12:
                h = h_new
                accepted = True
                break
        if not accepted:
            lr *= 0.3
            if lr < 1e-10:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 95

    best_h = np.copy(h_init)
    best_c5 = compute_c5(best_h, n)
    # Build diverse starting points

    # Perturbed

    starts.append(project_simplex(np.clip(h_init + rng.randn(n) * 0.1, 0, 1), n * 0.5))
    # Binary
    hb = (rng.rand(n) < 0.5).astype(float)
    starts.append(project_simplex(hb, n * 0.5))
    # Smooth sinusoidal
    t = np.linspace(0, 2.0, n, endpoint=False)
    for freq in [2, 3, 5]:
        hh = 0.5 + 0.35 * np.sin(np.pi * freq * t + rng.rand() * 2 * np.pi)
        starts.append(project_simplex(np.clip(hh, 0, 1), n * 0.5))
    # Random
    starts.append(project_simplex(rng.rand(n), n * 0.5))
    # Staircase
    for ns in [3, 4, 5, 7]:
        staircase = np.repeat(rng.rand(ns), n // ns)[:n].astype(float)
        starts.append(project_simplex(staircase, n * 0.5))
    # Optimize each start
    remaining = budget_s - (time.time() - t0) - 5
    per_start = max(remaining / len(starts), 3.0)
    for si, h0 in enumerate(starts):
        if time.time() - t0 > budget_s - 5:

        alloc = min(per_start, budget_s - (time.time() - t0) - 3)

        if c5_trial < best_c5:

            best_h = np.copy(h_trial)
    # Polish from best with tight temperature
    remaining = budget_s - (time.time() - t0) - 3

        if c5_pol < best_c5:

            best_h = np.copy(h_pol)
    # Final cleanup

    best_h = project_simplex(best_h, n * 0.5)
    best_c5 = compute_c5(best_h, n)
    return (best_h, best_c5, n)
