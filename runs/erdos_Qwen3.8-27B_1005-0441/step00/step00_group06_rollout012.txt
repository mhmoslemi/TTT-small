import numpy as np
import time

def project_feasible(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}."""
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    # Fix rounding
    diff = target_sum - np.sum(h)
    h += diff / n
    h = np.clip(h, 0, 1)
    return h

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def compute_gradient(h, beta=1.0):
    """Gradient of smooth max of overlaps."""
    n = len(h)
    dx = 2.0 / n
    overlaps_raw = np.correlate(h, 1 - h, mode="full")
    
    max_ov = np.max(overlaps_raw)
    exp_vals = np.exp(beta * (overlaps_raw - max_ov))
    weights = exp_vals / np.sum(exp_vals)
    weights *= dx
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            s = shift
            if s < n:
                grad[:n-s] += w * (1 - h[s:])
                grad[s:] -= w * h[:n-s]
        else:
            s = -shift
            if s < n:
                grad[s:] += w * (1 - h[:n-s])
                grad[:n-s] -= w * h[s:]
    
    return grad

def projected_gradient_descent(h0, max_iter=5000, beta=100.0, lr=None):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_c5 = compute_overlaps(h).max()
    best_h = h.copy()
    
    if lr is None:
        lr = 2.0 / (n / 2.0)  # rough estimate
    
    for it in range(max_iter):
        grad = compute_gradient(h, beta)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        c5 = compute_overlaps(h_new).max()
        if c5 < best_c5:

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(v, target_sum):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def smooth_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(ov)
    exps = np.exp(beta * (ov - m))
    s = exps.sum()
    obj = (m + np.log(s) / beta) * dx
    w = exps / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            s2 = shift
            if s2 < n:
                grad[:n - s2] += wk * (1.0 - h[s2:])
                grad[s2:] -= wk * h[:n - s2]
        else:
            s2 = -shift
            if s2 < n:
                grad[s2:] += wk * (1.0 - h[:n - s2])
                grad[:n - s2] -= wk * h[s2:]
    return obj, grad * dx

def pgd_optimize(h0, max_iter=3000, beta=100.0, lr0=0.08, time_limit=100.0):
    n = len(h0)
    target_sum = n / 2.0
    h = project_feasible(h0.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    start = time.time()
    no_improve = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        _, grad = smooth_max_obj_and_grad(h, beta)
        h_new = project_feasible(h - lr * grad, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
            lr = min(lr * 1.03, 0.25)
        else:
            no_improve += 1
            if no_improve > 50:
                lr = max(lr * 0.9, 0.005)
                no_improve = 0
        h = h_new
    return best_h, best_c5

def slsqp_optimize(h0, beta=80.0, time_limit=120.0):
    n = len(h0)
    target_sum = n / 2.0

    def obj_fn(h):
        val, _ = smooth_max_obj_and_grad(h, beta)
        return val

    def grad_fn(h):
        _, g = smooth_max_obj_and_grad(h, beta)
        return g

    def cons_fun(h):
        return np.sum(h) - target_sum

    def cons_jac(h):
        return np.ones(n)

    try:
        result = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': 600, 'ftol': 1e-12, 'disp': False})
        h_opt = project_feasible(result.x.copy(), target_sum)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt
    except Exception:
        return h0, compute_c5(h0)
