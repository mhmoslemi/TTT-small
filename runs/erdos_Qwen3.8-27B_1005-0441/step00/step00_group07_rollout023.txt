from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        m = n - shift
        for j in range(n):
            if 0 <= j - shift < m:
                grad[j] += (1.0 - h[j - shift])
            if 0 <= j < m:
                grad[j] += (-h[j + shift])
    else:
        s = -shift
        m = n - s
        for j in range(n):
            if 0 <= j < m:
                grad[j] += (1.0 - h[j + s])
            if s <= j < n:
                grad[j] += (-h[j - s])
    return grad * dx

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        h = h - diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    
    def obj(h):
        return compute_c5(h)
    
    def obj_grad(h):
        return c5_gradient(h)
    
    def cons_fun(h):
        return np.sum(h) - n / 2.0
    
    def cons_jac(h):
        return np.ones(n) / n
    
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    
    result = optimize.minimize(
        obj, h, jac=obj_grad, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    h_out = project_h(result.x, n)
    return h_out, compute_c5(h_out)
