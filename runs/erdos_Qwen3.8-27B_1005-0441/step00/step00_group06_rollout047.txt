import numpy as np

def project_box_sum(h, S):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def get_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(get_correlation(h)) * dx)

def get_smooth_gradient(h, beta):
    n = len(h)
    corr = get_correlation(h)
    M = np.max(corr)
    exp_vals = np.exp(beta * (corr - M))
    sw = np.sum(exp_vals)
    weights = exp_vals / sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            l = -s
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]
    return grad

def optimize_smooth(h0, n, max_iters, beta, step0, time_limit):
