    corr = np.correlate(h, 1.0 - h, mode="full")

from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_func(h, dx):
    return compute_c5(h, dx)

def c5_grad_only(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    k = k_max - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        kk = -k
        grad[:n - kk] += (1.0 - h[kk:])
        grad[kk:] -= h[:n - kk]
    return grad * dx

def c5_smooth_obj(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -50, 50)
    return m + tau * np.log(np.sum(np.exp(shifted)))

def c5_smooth_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k]) * weights[idx]
            grad[:n - k] -= h[k:] * weights[idx]
        else:
            kk = -k
            grad[:n - kk] += (1.0 - h[kk:]) * weights[idx]
            grad[kk:] -= h[:n - kk] * weights[idx]
    return grad * dx

    s = np.sum(h)

    h = h * (target_sum / s)

    dx_final = 2.0 / n_pts
    c5_final = compute_c5(best_h, dx_final)

    # Final validation using evaluate_erdos_solution if available
    try:
        eval_result = evaluate_erdos_solution(best_h)
        if eval_result is not None:
            c5_final = eval_result
    except Exception:
        pass

    return best_h, float(c5_final), n_pts
