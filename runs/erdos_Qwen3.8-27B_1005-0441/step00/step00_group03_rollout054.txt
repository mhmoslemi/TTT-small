from scipy import optimize
from scipy.ndimage import correlate  # or just use np.correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def enforce_constraints(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        h = np.clip(h + (target - s) / n, 0.0, 1.0)

    h = enforce_constraints(x, n)
    return compute_c5(h)

    for i in range(n_restarts):
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.45:
            break
        scale = 0.01 + 0.08 * (i % 5) / 5.0
        if i % 4 == 0:
            h_rand = best_h + scale * rng.standard_normal(n)
        elif i % 4 == 1:
            h_rand = best_h.copy()
            mask = rng.random(n) < 0.15
            h_rand[mask] = rng.random(int(np.sum(mask)) + 1)
        elif i % 4 == 2:
            h_rand = best_h.copy()
            idx = rng.choice(n, size=min(10, n), replace=False)
            h_rand[idx] = np.clip(h_rand[idx] + 0.2 * rng.standard_normal(len(idx)), 0, 1)
        else:
            # Swap two segments to create a different structure
            h_rand = best_h.copy()
            a = int(rng.integers(0, n - 10))
            b = a + int(rng.integers(5, n - a - 5))
            seg = h_rand[a:b].copy()
            h_rand[a:b] = h_rand[b:b + (b - a)].copy() if b + (b - a) <= n else h_rand[a:b]
            h_rand[a:b] = np.clip(seg + 0.05 * rng.standard_normal(b - a), 0, 1)
        h_rand = enforce_constraints(h_rand, n)
        h_trial = nm_polish(h_rand, n, 30000, t0, 0.45)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial

    # Phase 3: Differential Evolution (global)
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.75:
        bounds = [(0.0, 1.0)] * n
        time_left = budget_s * 0.75 - elapsed
        maxiter_de = max(500, int(time_left / 3.0))
        try:
            de_res = optimize.differential_evolution(
                lambda x: objective_fn(x, n), bounds, seed=seed,
                maxiter=min(maxiter_de, 3000), popsize=20,
                tol=1e-10, mutation=(0.4, 1.0), recombination=0.8,
                polish=False, init='sohal'
            )
            h_de = enforce_constraints(de_res.x, n)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de
        except Exception:
            pass

    # Phase 4: Second DE with different params if time allows
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.88:
        bounds = [(0.0, 1.0)] * n
        maxiter_de2 = max(300, int((budget_s * 0.88 - elapsed) / 4.0))
        try:
            de_res2 = optimize.differential_evolution(
                lambda x: objective_fn(x, n), bounds, seed=seed + 1,
                maxiter=min(maxiter_de2, 2000), popsize=15,
                tol=1e-9, mutation=(0.2, 0.9), recombination=0.7,
                polish=False, init='latinhypercube'
            )
            h_de2 = enforce_constraints(de_res2.x, n)
            c5_de2 = compute_c5(h_de2)
            if c5_de2 < best_c5:
                best_c5 = c5_de2
                best_h = h_de2
        except Exception:
            pass

    # Phase 5: Final NM polish
    if time.time() - t0 < budget_s * 0.95:
        best_h = nm_polish(best_h, n, 100000, t0, 0.97)
        best_c5 = compute_c5(best_h)

    # Phase 6: Very fine NM
    if time.time() - t0 < budget_s * 0.98:
        try:
            res = optimize.minimize(
                lambda x: objective_fn(x, n), best_h, method='Nelder-Mead',
                options={'maxiter': 50000, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True}
            )
            h_fine = enforce_constraints(res.x, n)
            c5_fine = compute_c5(h_fine)
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_fine
        except Exception:
            pass

    best_h = enforce_constraints(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
