    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * 2.0 / n)

def lag_grad(h, j, n):
    dx = 2.0 / n
    sh = j - n + 1
    idx = np.arange(n)
    g = np.zeros(n)
    v1 = (idx + sh >= 0) & (idx + sh < n)
    v2 = (idx - sh >= 0) & (idx - sh < n)
    if v1.any():
        g[idx[v1]] += 1.0 - h[idx[v1] - sh]
    if v2.any():
        g[idx[v2]] -= h[idx[v2] + sh]

    n = len(h)
    vals = np.correlate(h, 1.0 - h, "full") * (2.0 / n)
    cv = float(vals.max())
    k = min(top_k, len(vals))
    top = np.argsort(vals)[-k:]
    g = np.zeros(n)
    for j in top:
        g += lag_grad(h, int(j), n)
    return g / k, cv

        if v2.any():

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0).copy()
    tgt = 0.5 * n
    for _ in range(500):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        cnt = int(m.sum())
        if cnt == 0:
            break
        h[m] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h


def c5_calc(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * 2.0 / n)


def update_best(best, h):
    c = c5_calc(h)
    if c < best[1]:
        best[0] = h.copy()
        best[1] = c
        best[2] = len(h)
    return c


def lag_grad(h, j, n):
    dx = 2.0 / n
    sh = j - n + 1
    idx = np.arange(n)
    g = np.zeros(n)
    v1 = (idx + sh >= 0) & (idx + sh < n)
    v2 = (idx - sh >= 0) & (idx - sh < n)
    if v1.any():
        g[idx[v1]] += 1.0 - h[idx[v1] - sh]
    if v2.any():
        g[idx[v2]] -= h[idx[v2] + sh]
    return g * dx


def subgrad_c5(h, top_k=5):
    n = len(h)
    vals = np.correlate(h, 1.0 - h, "full") * (2.0 / n)
    cv = float(vals.max())
    k = min(top_k, len(vals))
    top = np.argsort(vals)[-k:]
    g = np.zeros(n)
    for jj in top:
        g += lag_grad(h, int(jj), n)
    return g / k, cv


def softmax_cg(x, alpha):
    n = len(x)
    h = np.clip(x, 1e-7, 1.0 - 1e-7)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, "full") * dx
    a = min(alpha, 600.0 / (float(vals.max()) + 1e-12))
    sh = vals - vals.max()
    e = np.exp(a * sh)
    s = e.sum()
    obj = vals.max() + np.log(s) / a
    w = e / s
    idx = np.arange(n)
    g = np.zeros(n)
    for jj in range(len(vals)):
        wj = w[jj]
        if wj < 1e-9:
            continue
        sh2 = jj - n + 1
        v1 = (idx + sh2 >= 0) & (idx + sh2 < n)
        v2 = (idx - sh2 >= 0) & (idx - sh2 < n)
        if v1.any():
            g[idx[v1]] += wj * (1.0 - h[idx[v1] - sh2])
        if v2.any():
            g[idx[v2]] -= wj * h[idx[v2] + sh2]
    g *= dx
    return obj, g

    return softmax_cg(x, 30.0)[0]

    return softmax_cg(x, 30.0)[1]

    return softmax_cg(x, 120.0)[1]

    for n in [32, 45, 90, 180, 360]:

    pairs = [(sm_obj30, sm_grad30), (sm_obj60, sm_grad60), (sm_obj120, sm_grad120)]

            hr = project_feasible(res.x, n)
            update_best(best, hr)
            if c5_calc(hr) < c5_calc(h):
                h = hr
        except Exception:

    return h
