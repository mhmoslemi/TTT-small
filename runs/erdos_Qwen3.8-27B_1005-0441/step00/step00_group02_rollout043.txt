import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n / 2.0
    idx = np.argsort(-h)
    result = np.zeros(n)
    if abs(target - round(target)) < 1e-9:
        result[idx[:int(target)]] = 1.0
    else:
        lo = int(target)
        result[idx[:lo]] = 1.0
        result[idx[lo]] = target - lo
    return result

def ls_swap(h, time_limit):
    n = len(h)
    c5 = compute_c5(h)
    t0 = time.time()
    stuck_rounds = 0
    while time.time() - t0 < time_limit:
        best_c5 = c5
        best_pair = None
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-12:
                    continue
                h[i], h[j] = h[j], h[i]
                c = compute_c5(h)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_pair = (i, j)
                h[i], h[j] = h[j], h[i]
        if best_pair is None:
            stuck_rounds += 1
            if stuck_rounds > 2:
                break
            continue
        stuck_rounds = 0
        i, j = best_pair
        h[i], h[j] = h[j], h[i]
        c5 = best_c5
    return h, c5

def sa_continuous(h, time_limit, rng):
    n = len(h)
    c5 = compute_c5(h)
    best = h.copy()
    best_c5 = c5
    temp = 0.015
    min_temp = 1e-7
    t0 = time.time()
    patience = 0
    while time.time() - t0 < time_limit:
        n_moves = rng.randint(1, 4)
        accepted_any = False
        for _ in range(n_moves):
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(-0.4, 0.4)
            if 0 <= h[i] + delta <= 1 and 0 <= h[j] - delta <= 1:
                h[i] += delta
                h[j] -= delta
                c = compute_c5(h)
                d = c - c5
                if d < 0 or rng.random() < np.exp(-d / (temp if temp > min_temp else min_temp)):
                    c5 = c
                    accepted_any = True
                    if c5 < best_c5:
                        best_c5 = c5
                        best = h.copy()
                        patience = 0
                else:
                    h[i] -= delta
                    h[j] += delta
        if accepted_any:
            patience = 0
        else:
            patience += 1
        temp *= 0.99997
        if patience > 5000:
            temp = min_temp
            patience = 0
        if temp < min_temp:
            temp = min_temp
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start) - 15

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(45, 0.5)

    n = len(h_init)
    target = n / 2.0
    h_init = make_feasible(h_init, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary permutation local search from initial
    h_bin = to_binary(h_init, n).copy()
    t_alloc = min(150, max(30, remaining() * 0.2))
    h_bin, c5_bin = ls_swap(h_bin, t_alloc)
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin

    # Phase 2: Continuous SA from best
    if remaining() > 90:
        h_sa = best_h.copy()
        t_alloc = min(350, max(80, remaining() * 0.35))
        h_sa, c5_sa = sa_continuous(h_sa, t_alloc, rng)
        if c5_sa < best_c5:
            best_h = h_sa.copy()
            best_c5 = c5_sa

    # Phase 3: Binary local search from SA result
    if remaining() > 60:
        h_bin2 = to_binary(h_sa, n).copy()
        t_alloc = min(120, max(20, remaining() * 0.2))
        h_bin2, c5_bin2 = ls_swap(h_bin2, t_alloc)
        if c5_bin2 < best_c5:
            best_h = h_bin2.copy()
            best_c5 = c5_bin2

    # Phase 4: Random restarts with different seeds and sizes
    if remaining() > 40:
        sizes_to_try = [n]
        for ds in [5, 10, 15, 20]:
            nt = n + ds
            if nt % 2 == 0 and nt <= 200:
                sizes_to_try.append(nt)
        n_restarts = min(6, int(remaining() / 30))
        t_per = max(5, int(remaining() / (n_restarts + 1)))
        for r in range(n_restarts):
            if remaining() < t_per + 5:
                break
            n_try = sizes_to_try[r % len(sizes_to_try)]
            sr = np.random.RandomState(seed + 31 * (r + 1))
            h_r = sr.random(n_try)
            h_r = to_binary(h_r, n_try)
            h_r, c5_r = ls_swap(h_r, t_per)
            if c5_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c5_r

    # Phase 5: Final polish SA on best
    if remaining() > 20:
        h_polish = best_h.copy()
        t_alloc = min(60, remaining() - 10)
        if t_alloc > 5:
            rng2 = np.random.RandomState(seed + 999)
            h_polish, c5_polish = sa_continuous(h_polish, t_alloc, rng2)
            if c5_polish < best_c5:
                best_h = h_polish.copy()
                best_c5 = c5_polish

    # Final: ensure feasibility and verify
    n_final = len(best_h)
    best_h = make_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_final)
