If f is a sum of many independent random frequencies (structured like a Fourier series with random phases), the autocorrelation at non-zero lags will be small. With enough frequencies, it can be made very close to 0.

But we're limited to h ∈ [0,1] (f ∈ [-1/2, 1/2]) and the function must be a step function (piecewise constant).

For a step function with n points, we have n values f[i] ∈ [-1/2, 1/2] with sum f[i] = 0.

The autocorrelation at lag k: r[k] = ∑ f[i]f[i+k] / n (approximately, for the integral).
We want min_k r[k] to be as large as possible (close to 0 or positive).

This is like finding a "causal PR sequence" or "Barker sequence" type problem, but continuous.

For the discrete case, if we think of f as a sequence with values in [-1/2, 1/2] and mean 0, we want the autocorrelation to be maximally flat (close to 0 for all non-zero lags).

This is related to "low autocorrelation sequences" but in the continuous-valued case.

One approach: use a "Chirp" or "Costas" type sequence, or a "m-sequence" (maximum length sequence) mapped to continuous values.

def subgradient(h, corr, dx, n):
    """Compute subgradient of max(corr) with respect to h."""
    # corr has 2n-1 elements, indexed 0 to 2n-2
    # corr[i] corresponds to lag l = i - (n-1)
    # c[l] = sum_j h[j] * (1-h)[j+l] for valid j
    # d(c[l])/d(h[p]) = (1-h[p-l]) if valid - h[p+l] if valid
    # (where "valid" means the index is in [0, n-1])
    
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-6)[0]
    
    grad = np.zeros(n)
    for idx in active:
        l = idx - (n - 1)  # lag
        # d(c[l])/d(h[p]) for p = 0, ..., n-1
        # term 1: h[p] appears as h[j] with j=p, contributing (1-h)[p+l] if 0 <= p+l < n
        # term 2: h[p] appears as (1-h)[j+l] with j+l=p, contributing -h[j] = -h[p-l] if 0 <= p-l < n
        for p in range(n):
            g = 0.0
            if 0 <= p + l < n:
                g += 1 - h[p + l]
            if 0 <= p - l < n:
                g -= h[p - l]
            grad[p] += g
    grad /= len(active)
    return grad

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_corr(h) * dx))

def project_simplex(v, target):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        p = np.clip(v - mid, 0.0, 1.0)
        s = float(p.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-13:
            break
    p = np.clip(v - mid, 0.0, 1.0)
    diff = target - float(p.sum())
    if abs(diff) > 1e-12:
        free = (p > 1e-10) & (p < 1.0 - 1e-10)
        nf = int(free.sum())
        if nf > 0:
            p[free] += diff / nf
            p = np.clip(p, 0.0, 1.0)
    return p

def lag_grad(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n - s] += 1.0 - h[s:]
        if s > 0:
            g[s:] -= h[:n - s]
    else:

        if m < n:
            g[m:] += 1.0 - h[:n - m]
        if m > 0:
            g[:n - m] -= h[m:]
    return g

    def update(res_h, res_c5, res_n):
        nonlocal best_c5, best_h, best_n
        if res_c5 < best_c5:
            best_c5 = res_c5
            best_h = res_h.copy()
            best_n = res_n

    # Phase 1: Smooth gradient from initial (30% budget)
    h1, c5_1 = solve_smooth(h_init, n, dx, 20000, t0, t0 + budget_s * 0.30, alpha=40)
    update(h1, c5_1, n)

    # Phase 2: Subgradient refinement (10%)
    h2, c5_2 = solve_subgrad(best_h, n, dx, 50000, t0, t0 + budget_s * 0.40)
    update(h2, c5_2, n)

    # Phase 3: Random restarts (25%)
    for trial in range(6):
        t_start = time.time()
        if t_start - t0 > budget_s * 0.65:
            break
        t_alloc = min(t0 + budget_s * 0.65, t_start + budget_s * 0.06)
        # Binary random start
        hr = make_binary_random(rng, n)
        hr += rng.random(n) * 0.3 - 0.15
        hr = project_simplex(np.clip(hr, 0.0, 1.0), n / 2)
        h3, c5_3 = solve_subgrad(hr, n, dx, 15000, t0, t_alloc)
        update(h3, c5_3, n)
        # Polish best
        if time.time() - t0 < budget_s * 0.65:
            h4, c5_4 = solve_smooth(h3, n, dx, 5000, t0, t_alloc, alpha=60)
            update(h4, c5_4, n)

    # Phase 4: Higher resolution (20%)
    for n2 in [200, 300]:
        if time.time() - t0 > budget_s * 0.88:
            break
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_simplex(np.clip(h_up, 0.0, 1.0), n2 / 2)
        h5, c5_5 = solve_subgrad(h_up, n2, dx2, 20000, t0, t0 + budget_s * 0.88)
        update(h5, c5_5, n2)
        if time.time() - t0 < budget_s * 0.88:
            h6, c5_6 = solve_smooth(h5, n2, dx2, 5000, t0, t0 + budget_s * 0.88, alpha=50)
            update(h6, c5_6, n2)

    # Phase 5: Final polish with high alpha
    if time.time() - t0 < budget_s * 0.95:
        dx_f = 2.0 / best_n
        h7, c5_7 = solve_smooth(best_h, best_n, dx_f, 10000, t0, t0 + budget_s * 0.98, alpha=100)
        update(h7, c5_7, best_n)

    # Ensure constraint satisfaction
    best_h = project_simplex(np.clip(best_h, 0.0, 1.0), best_n / 2)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c5, best_n
