```python
def grad_overlap(h, shift, n):
    """Gradient of overlap at given shift index with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    # h[j+shift] * (1-h[j]) terms
    # For j where 0 <= j < n and 0 <= j+shift < n:
    #   contribution to grad[j] from (1-h[j]): -(h[j+shift])
    #   contribution to grad[j+shift] from h[j+shift]: (1-h[j])
    
    if shift >= 0:
        valid = np.arange(n - shift)  # j where j+shift < n
    else:
        valid = np.arange(-shift, n)  # j where j+shift >= 0
    
    # For each valid j:
    # grad[j] -= dx * h[j+shift]
    # grad[j+shift] += dx * (1-h[j])
    
    j_idx = valid
    jk_idx = valid + shift
    grad[j_idx] -= dx * h[jk_idx]
    grad[jk_idx] += dx * (1 - h[j_idx])
    
    return grad
```

This is O(n) per shift.

import numpy as np
import time
from scipy import optimize

def project_feasible(h, n):
    """Project h onto {0 <= h_i <= 1, sum(h) = n/2} via bisection on lambda."""
    h = np.asarray(h, dtype=float)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        adj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(adj) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    res = np.clip(h - lam, 0.0, 1.0)
    # Fine-tune for exact sum
    diff = np.sum(res) - target
    if abs(diff) > 1e-14:
        mask = (res > 1e-10) & (res < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            res[mask] -= diff / cnt
            res = np.clip(res, 0.0, 1.0)
    return res

def grad_overlap(h, shift, n):
    """Gradient of overlap at shift (in index units) wrt h, times dx."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if shift >= 0:
        valid = np.arange(n - shift)
    else:
        valid = np.arange(-shift, n)
    if len(valid) == 0:
        return grad
    j_idx = valid
    jk_idx = valid + shift
    grad[j_idx] -= dx * h[jk_idx]
    grad[jk_idx] += dx * (1.0 - h[j_idx])
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(overlaps), overlaps

def subgradient_optimize(h, budget_s, n, K=5):
    """Projected subgradient descent with adaptive step size."""
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    h = h.copy()
    lr = 0.08
    no_improve = 0
    
    while time.time() - t_start < budget_s - 5:
        c5, overlaps = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Top-K weighted gradient
        top_idx = np.argsort(overlaps)[-K:]
        w = np.exp(overlaps[top_idx] - overlaps[top_idx].max())
        w /= w.sum()
        grad = np.zeros(n)
        for ki in range(K):
            shift = top_idx[ki] - (n - 1)
            grad += w[ki] * grad_overlap(h, shift, n)
        
        if no_improve == 40:
            lr = max(lr * 0.7, 0.005)
        elif no_improve == 120:
            # Perturb and restart
            h = best_h + 0.03 * np.random.randn(n)
            h = project_feasible(h, n)
            lr = 0.08
            no_improve = 0
            continue
        
        h = project_feasible(h - lr * grad, n)
    
    return best_h, best_c5

def scipy_polish(h, n, time_left):
    """Local polish using smooth max approximation."""
    if time_left < 30:
        return h, compute_c5(h)[0]
    dx = 2.0 / n
    
    def objective(x, T=50.0):
        ovl = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(ovl)
        # Smooth max via log-sum-exp
        vals = np.exp(T * (ovl - m))
        return m + np.log(np.sum(vals)) / T
    
    def jac(x, T=50.0):
        ovl = np.correlate(x, 1.0 - x, mode="full") * dx
        m = np.max(ovl)
        vals = np.exp(T * (ovl - m))
        w = T * vals / np.sum(vals)
        grad = np.zeros(n)
        for ki in range(len(ovl)):
            shift = ki - (n - 1)
            grad += w[ki] * grad_overlap(x, shift, n)
        return grad
    
    res = optimize.minimize(
        objective, h, jac=lambda x: jac(x), method='L-BFGS-B',
        bounds=[(0, 1)] * n,
        options={'maxiter': 500, 'ftol': 1e-14}
    )
    h_out = project_feasible(res.x, n)
    c5_out = compute_c5(h_out)[0]
    return h_out, c5_out
