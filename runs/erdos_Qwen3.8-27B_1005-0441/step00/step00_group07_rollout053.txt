            c5_best = c5_cd

    # Phase 3: Random restarts with re-optimization
    if time.time() - t0 < budget_s * 0.85:
        for _r in range(8):
            if time.time() - t0 > budget_s * 0.85:
                break
            scale = 0.03 + 0.03 * _r
            h_r = _feasible(h_best + rng.randn(n) * scale, n)
            _temp = 0.01
            h_r = _slsqp_opt(h_r, n, 200, 1e-10)
            c5_r = _dot_c5(h_r)
            if c5_r < c5_best - 1e-10:
                h_best = h_r
                c5_best = c5_r
                _temp = 0.003
                h_best = _slsqp_opt(h_best, n, 200, 1e-11)
                c5_best = _dot_c5(h_best)

    # Phase 4: Upsample to 2n and optimize
    if time.time() - t0 < budget_s * 0.94:
        n_up = min(2 * n, 320)
        x_s = np.arange(n, dtype=float)
        x_d = np.linspace(0, n - 1, n_up)
        h_up = np.interp(x_d, x_s, h_best)
        h_up = _feasible(h_up, n_up)
        c5_up = _dot_c5(h_up)
        if c5_up < c5_best + 0.08:
            for temp in [0.05, 0.01, 0.003]:
                if time.time() - t0 > budget_s * 0.96:
                    break
                _temp = temp
                h_up = _slsqp_opt(h_up, n_up, 200, 1e-10)
                c5_up = _dot_c5(h_up)
            if c5_up < c5_best - 1e-9:
                h_best = h_up
                c5_best = c5_up

    # Final: ensure feasibility and compute
    h_best = _feasible(h_best, len(h_best))
    c5_final = _dot_c5(h_best)
    return h_best, c5_final, len(h_best)
