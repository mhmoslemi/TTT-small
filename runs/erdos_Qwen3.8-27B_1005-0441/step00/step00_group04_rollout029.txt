    O_max = np.max(O)
    log_sum = O_max + eps * np.log(np.sum(np.exp((O - O_max) / eps)))

import time
from scipy.optimize import minimize

def compute_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps(h, dx) * dx))

def proj_box_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - h_p.sum()
    if abs(d) > 1e-13:
        mask = (h_p > 1e-13) & (h_p < 1.0 - 1e-13)
        if mask.any():
            h_p[mask] += d / mask.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def lse_val(h, dx, eps):
    O = compute_overlaps(h, dx) * dx
    m = O.max()
    return m + eps * np.log(np.sum(np.exp((O - m) / eps)))

def lse_grad(h, dx, eps):
    n = len(h)
    O = compute_overlaps(h, dx) * dx
    m = O.max()
    w = np.exp((O - m) / eps)
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(len(O)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[0:n-k] -= wi * h[k:n]
            g[k:n] += wi * (1.0 - h[0:n-k])
        else:
            kp = -k
            g[0:n-kp] += wi * (1.0 - h[kp:n])
            g[kp:n] -= wi * h[0:n-kp]
    return g

    eps_schedule = [0.05, 0.02, 0.01, 0.005]

            return lse_val(x, dx, eps) + pen
        def fgrad(x):
            pen = 1000.0 * (x.sum() - target)
            return lse_grad(x, dx, eps) + pen * np.ones_like(x)
        res = minimize(f, h, jac=fgrad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * len(h),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
        h = proj_box_simplex(np.clip(res.x, 0.0, 1.0), target)
    return h

import numpy as np
import time
from scipy.optimize import minimize

def compute_overlaps_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlaps_raw(h) * dx))

def proj_box_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - h_p.sum()
    if abs(d) > 1e-13:
        mask = (h_p > 1e-13) & (h_p < 1.0 - 1e-13)
        if mask.any():
            h_p[mask] += d / mask.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def lse_val(h, dx, eps):
    O = compute_overlaps_raw(h) * dx
    m = O.max()
    return m + eps * np.log(np.sum(np.exp((O - m) / eps)))

def lse_grad(h, dx, eps):
    n = len(h)
    O = compute_overlaps_raw(h) * dx
    m = O.max()
    w = np.exp((O - m) / eps)
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(len(O)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[0:n-k] -= wi * h[k:n]
            g[k:n] += wi * (1.0 - h[0:n-k])
        else:
            kp = -k
            g[0:n-kp] += wi * (1.0 - h[kp:n])
            g[kp:n] -= wi * h[0:n-kp]
    return g * dx

def optimize_lse(h, dx, target, max_iter=300):
    eps_schedule = [0.05, 0.02, 0.008, 0.004]
    pen_w = 5000.0
    for eps in eps_schedule:

            return lse_val(x, dx, eps) + pen_w * (x.sum() - target) ** 2
        def fgrad(x):
            return lse_grad(x, dx, eps) + 2.0 * pen_w * (x.sum() - target) * np.ones_like(x)
        res = minimize(f, h, jac=fgrad, method='L-BFGS-B',
