            if trial > 0:
                start = project_feasible(start + rng.normal(0, 0.03, n_f), n_f)
            try:
                res = minimize(compute_c5, start, method='SLSQP',
                              bounds=[(0, 1)] * n_f,
                              constraints=[{'type': 'eq',
                                            'fun': lambda x: x.sum() - n_f / 2.0,
                                            'jac': lambda x: np.ones(n_f)}],
                              options={'maxiter': 2000, 'ftol': 1e-16})
                hr = project_feasible(res.x, n_f)
                c = compute_c5(hr)
                if c < best_c5:
                    best_c5 = c
                    best_h = hr.copy()
            except Exception:
                pass

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    c_final = compute_c5(best_h)
    if c_final < best_c5:
        best_c5 = c_final

    return (best_h, best_c5, len(best_h))
