    v_t = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = np.inf
    dx = 2.0 / n

    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-3:]
        cg = np.zeros(n)
        for si in top_idx:
            k = si - (n - 1)
            cg += grad_shift(h, k, n)
        cg /= len(top_idx)

        t += 1
        lr = base_lr / (1.0 + 0.05 * t)
        m_t = b1 * m_t + (1 - b1) * cg
        v_t = b2 * v_t + (1 - b2) * cg * cg
        m_hat = m_t / (1 - b1 ** t)
        v_hat = v_t / (1 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_constraint(h_new, target_sum)
        if not np.all(np.isfinite(h_new)):
            break
        h = h_new

        if it % 100 == 0:
            c5_now = corr.max() if it == 0 else np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()

    c5_final = compute_c5(h)[0]
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.ones(70) * 0.5

    n = len(h0)
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_constraint(h0, target_sum)

    best_h, best_c5 = h0.copy(), compute_c5(h0)[0]
    rng = np.random.RandomState(seed)

    # Phase 1: Direct optimization from parent
    bh, bc5 = adam_optimize(h0, n, target_sum, 8000, 0.02, seed)
    if bc5 < best_c5:
        best_h, best_c5 = bh, bc5

    # Phase 2: Perturbation restarts
    n_restarts = 12
    for r in range(n_restarts):
        if elapsed() > budget_s - 80:
            break
        # Create perturbation
        scale = 0.05 + 0.1 * rng.rand()
        noise = rng.normal(0, scale, size=n)
        hp = h0 + noise
        # Also try small random flip patterns
        if r % 3 == 2:
            flip_mask = rng.rand(n) < 0.15
            hp[flip_mask] = 1.0 - h0[flip_mask] * 0.5
        hp = project_to_constraint(np.clip(hp, 0, 1), target_sum)

        lr = [0.01, 0.02, 0.03, 0.005][r % 4]
        iters = 6000 if elapsed() < budget_s * 0.5 else 3000
        bh, bc5 = adam_optimize(hp, n, target_sum, iters, lr, seed + 100 + r)
        if bc5 < best_c5:
            best_h, best_c5 = bh, bc5

    # Phase 3: Fine-tune best solution
    if elapsed() < budget_s - 30:
        bh, bc5 = adam_optimize(best_h, n, target_sum, 5000, 0.003, seed + 500)
        if bc5 < best_c5:
            best_h, best_c5 = bh, bc5
        # Even finer
        if elapsed() < budget_s - 15:
            bh, bc5 = adam_optimize(best_h, n, target_sum, 5000, 0.001, seed + 600)
            if bc5 < best_c5:
                best_h, best_c5 = bh, bc5

    # Phase 4: Try slightly different n via interpolation
    if elapsed() < budget_s - 40:
        for n_try in [60, 80, 100]:
            if elapsed() > budget_s - 30:
                break
            x_orig = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_orig, best_h)
            ts = n_try / 2.0
            h_interp = project_to_constraint(np.clip(h_interp, 0, 1), ts)
            bh, bc5 = adam_optimize(h_interp, n_try, ts, 4000, 0.01, seed + 800)
            if bc5 < best_c5:
                best_h, best_c5 = bh, bc5

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_constraint(best_h, len(best_h) / 2.0)
    best_c5, _ = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
