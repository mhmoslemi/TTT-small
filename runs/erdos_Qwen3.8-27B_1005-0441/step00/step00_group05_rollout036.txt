def _slsqp_optimize(h_start, maxiter=1500, ftol=1e-11):
    n = len(h_start)
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': _constraint_sum}]
    try:
        res = minimize(_objective, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < _objective(h_start):
            return res.x
    except Exception:
        pass
    return h_start

def _perturb_search(h_best, n_iter, rng):
    n = len(h_best)
    h = h_best.copy()
    c5 = _objective(h)
    scale = 0.3
    for _ in range(n_iter):
        if scale < 0.01:
            break
        h_try = h.copy()
        n_flip = max(1, n // 10)
        idx = rng.choice(n, size=n_flip, replace=False)
        h_try[idx] += rng.uniform(-scale, scale, size=n_flip)
        h_try = _fix_sum(h_try)
        c5_try = _objective(h_try)
        if c5_try < c5:
            h = h_try
            c5 = c5_try
            scale *= 0.95
        else:
            scale *= 0.85
    return h, c5

def _binary_search(h_best, rng, n_iter):
    n = len(h_best)
    target = n // 2
    h = np.zeros(n)
    ones = set(rng.choice(n, size=target, replace=False).tolist())
    for i in ones:
        h[i] = 1.0
    c5 = _objective(h)
    for _ in range(n_iter):
        h_try = h.copy()
        i = rng.randint(n)
        j = rng.randint(n)
        if h_try[i] == 1.0 and h_try[j] == 0.0:
            h_try[i] = 0.0
            h_try[j] = 1.0
            c5_try = _objective(h_try)
            if c5_try < c5:
                h = h_try
                c5 = c5_try
            else:
                h_try[i] = 1.0
                h_try[j] = 0.0
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    # Get initial values
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h0 is None:
        h0 = np.zeros(70)
        h0[:35] = 1.0

    n = len(h0)
    h0 = _fix_sum(h0)
    best_h = h0.copy()
    best_c5 = _objective(h0)

    # Phase 1: SLSQP from initial
    if time.time() - t_start < budget_s - 60:
        h_opt = _slsqp_optimize(h0, maxiter=3000, ftol=1e-12)
        c5_opt = _objective(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Phase 2: Perturbation local search from best
    if time.time() - t_start < budget_s - 60:
        h_p, c5_p = _perturb_search(best_h, 500, rng)
        if c5_p < best_c5:
            best_h = h_p
            best_c5 = c5_p
        # SLSQP refine
        h_opt = _slsqp_optimize(best_h, maxiter=2000, ftol=1e-12)
        c5_opt = _objective(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Phase 3: Try different n values
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    n_candidates = [50, 60, 70, 80, 90, 100, 120]
    per_n = max(2, int(remaining / (len(n_candidates) * 45)))

    for n_try in n_candidates:
        if time.time() - t_start > budget_s - 30:
            break
        if n_try == n:
            continue
        for r in range(per_n):
            if time.time() - t_start > budget_s - 30:
                break
            if r == 0:
                h_start = np.zeros(n_try)
                h_start[:n_try // 2] = 1.0
            elif r == 1:
                h_start = np.random.RandomState(seed + n_try + r).rand(n_try)
            else:
                h_start = rng.rand(n_try)
            h_start = _fix_sum(h_start)

            h_opt = _slsqp_optimize(h_start, maxiter=800, ftol=1e-10)
            c5_opt = _objective(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

    # Phase 4: Binary search for the best n
    n_best = len(best_h)
    if time.time() - t_start < budget_s - 30:
        h_bin, c5_bin = _binary_search(best_h, rng, n_best * 50)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin

    # Phase 5: Final SLSQP polish
    if time.time() - t_start < budget_s - 5:
        h_final = _slsqp_optimize(best_h, maxiter=3000, ftol=1e-13)
        c5_final = _objective(h_final)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _fix_sum(best_h)
    best_c5 = _objective(best_h)

    # Try evaluate_erdos_solution if available
    try:
        val = evaluate_erdos_solution(best_h)
        if val < best_c5:
            best_c5 = val
    except Exception:
        pass

    return (best_h, best_c5, len(best_h))
