        if jlo <= jhi:
            idx = np.arange(jlo, jhi + 1) + lag
            g[jlo:jhi + 1] -= wm * h[idx] * dx
    return g

def optimize_loop(h, n, tgt, deadline, rng, lr, temp):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    while time.time() < deadline:
        dx = 2.0 / n
        cv = np.correlate(h, 1.0 - h, mode="full") * dx
        cur = np.max(cv)
        if cur < best_c5 - 1e-13:
            best_c5 = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 40:
            lr = max(lr * 0.8, 3e-5)
            temp = min(temp * 1.35, 0.08)
            h = h + rng.randn(n) * max(lr, 0.004)
            h = np.clip(h, 0.0, 1.0)
            h = project_sum(h, tgt)
            stall = 0
            if lr <= 4e-5 and temp > 0.07:
                break
        g = grad_soft(h, cv, temp, dx)
        alpha = lr
        improved = False
        for _ in range(7):
            h_new = h - alpha * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_sum(h_new, tgt)
            if compute_c5(h_new) < cur - 1e-14:
                h = h_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            h = h + rng.randn(n) * 0.002
            h = np.clip(h, 0.0, 1.0)
            h = project_sum(h, tgt)
    return best_h, best_c5, lr, temp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(45, 0.5)
    n = len(h_init)
    tgt = n / 2.0
    h_init = project_sum(h_init, tgt)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: gradient descent from initial
    deadline1 = t0 + budget_s * 0.4
    best_h, best_c5, lr, temp = optimize_loop(h_init, n, tgt, deadline1, rng, 0.03, 0.006)

    # Phase 2: random restarts blended with best
    deadline2 = t0 + budget_s * 0.7
    attempt = 0
    while time.time() < deadline2 and attempt < 30:
        h_r = 0.4 * best_h + 0.6 * rng.rand(n)
        h_r = project_sum(h_r, tgt)
        dl = min(time.time() + 25, deadline2)
        h_opt, c5_opt, lr2, temp2 = optimize_loop(h_r, n, tgt, dl, rng, 0.02, 0.01)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            lr, temp = lr2, temp2
        attempt += 1

    # Phase 3: higher resolution
    for mult in [2, 3]:
        n2 = n * mult
        if n2 > 400 or time.time() > t0 + budget_s * 0.85:
            break
        h_up = np.interp(
            np.linspace(0, 2, n2, endpoint=False),
            np.linspace(0, 2, n, endpoint=False),
            best_h
        )
        h_up = project_sum(h_up, n2 / 2.0)
        deadline3 = t0 + budget_s * 0.92
        h_opt, c5_opt, _, _ = optimize_loop(h_up, n2, n2 / 2.0, deadline3, rng, 0.02, 0.005)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n = n2

    # Phase 4: final SLSQP polish
    if time.time() < t0 + budget_s - 5:
        try:
            from scipy.optimize import minimize
            n = len(best_h)
            tgt2 = n / 2.0
            dx2 = 2.0 / n
            t2 = 0.004

            def sobj(x):
                cv = np.correlate(x, 1.0 - x, mode="full") * dx2
                m = np.max(cv)
                return m + t2 * np.log(np.sum(np.exp((cv - m) / t2)))

            def sgrad(x):
                cv = np.correlate(x, 1.0 - x, mode="full") * dx2
                return grad_soft(x, cv, t2, dx2)

            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt2,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(sobj, best_h, jac=sgrad, method='SLSQP',
                           bounds=[(0, 1)] * n, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-12})
            if res.fun < best_c5 + 0.01:
                x = project_sum(np.clip(res.x, 0, 1), tgt2)
                cx = compute_c5(x)
                if cx < best_c5:
                    best_h, best_c5 = x, cx
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
