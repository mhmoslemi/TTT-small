    h_t = h.copy()
    n = len(h)
    start = int(rng.integers(0, n))
    length = int(rng.integers(2, max(3, n // 4)))
    end = min(start + length, n)
    h_t[start:end] += rng.normal(0, scale / np.sqrt(length), end - start)
    return h_t

def local_descent(h, target, max_rounds=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n = len(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            old = best_h[i]
            for step in (0.003, -0.003, 0.007, -0.007, 0.014, -0.014, 0.001, -0.001):
                h_t = best_h.copy()
                h_t[i] = old + step
                h_t = repair_sum(h_t, target)
                c5_t = compute_c5(h_t)
                if c5_t < best_c5 - 1e-12:
                    best_h = h_t
                    best_c5 = c5_t
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Initialize from initial_h_values if available
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h = np.full(42, 0.5)

    n = len(h)
    target = n / 2.0
    h = repair_sum(h, target)

    # Best tracking
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Also try a few random restarts to find good starting points
    for trial in range(5):
        if time.time() - t0 > 10:
            break
        h_r = np.clip(rng.normal(0.5, 0.25, n), 0, 1)
        h_r = repair_sum(h_r, target)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_h = h_r
            best_c5 = c5_r

    # === Phase 1: Simulated Annealing ===
    T = 0.004
    T_floor = 1e-9
    cool_rate = 0.99995
    stuck = 0
    stuck_limit = 200
    phase1_deadline = t0 + budget_s * 0.75

    while time.time() < phase1_deadline:
        h_t = best_h.copy()
        r = rng.random()
        scale = 0.01 + 0.03 * min(1.0, T / 0.004)
        if r < 0.4:
            h_t = perturb_gauss(best_h, rng, max(2, n // 8), scale)
        elif r < 0.7:
            h_t = perturb_block(best_h, rng, scale)
        else:
            h_t = perturb_swap(best_h, rng)

        h_t = repair_sum(h_t, target)
        c5_t = compute_c5(h_t)
        delta = c5_t - best_c5

        if delta < -1e-12:
            best_h = h_t
            best_c5 = c5_t
            stuck = 0
        else:
            stuck += 1
            if T > T_floor and rng.random() < np.exp(-max(delta, 0) / T):
                best_h = h_t
                best_c5 = c5_t

        T = max(T * cool_rate, T_floor)

        if stuck >= stuck_limit:
            stuck = 0
            T = 0.006 + 0.003 * rng.random()
            stuck_limit = min(int(stuck_limit * 1.4), 5000)

    # === Phase 2: Local Descent (coordinate) ===
    phase2_deadline = t0 + budget_s * 0.92
    while time.time() < phase2_deadline:
        best_h, best_c5 = local_descent(best_h, target, max_rounds=10)
        # Small kick to escape local minimum
        if time.time() < phase2_deadline - 5:
            h_kick = best_h + rng.normal(0, 0.005, n)
            h_kick = repair_sum(h_kick, target)
            c5_kick = compute_c5(h_kick)
            if c5_kick < best_c5:
                best_h = h_kick
                best_c5 = c5_kick
            else:
                break

    # === Phase 3: Final fine-tune ===
    final_deadline = t0 + budget_s - 3
    while time.time() < final_deadline:
        improved = False
        for i in range(n):
            old = best_h[i]
            for step in (0.0005, -0.0005, 0.001, -0.001, 0.002, -0.002):
                h_t = best_h.copy()
                h_t[i] = old + step
                h_t = repair_sum(h_t, target)
                c5_t = compute_c5(h_t)
                if c5_t < best_c5 - 1e-12:
                    best_h = h_t
                    best_c5 = c5_t
                    improved = True
                    break
        if not improved:
            break

    best_h = repair_sum(best_h, target)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
