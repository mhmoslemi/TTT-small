  That's o1h[0:n-off] added to grad[off:n]

- m+off < n → m < n-off
  -h[m+off] is indexed as -h[off:m+off] ... wait
  -h[m+off] for m = 0, 1, ..., n-off-1
  That's -h[off:n] added to grad[0:n-off]

So for off ≥ 0:
  grad[off:n] += o1h[0:n-off]
  grad[0:n-off] += -h[off:n]

For off < 0, let a = -off > 0:
- m-off = m+a ≥ 0 (always), m+a < n → m < n-a
  (1-h[m+a]) for m = 0, 1, ..., n-a-1
  That's o1h[a:n] added to grad[0:n-a]

- m+off = m-a ≥ 0 → m ≥ a, m-a < n (always)
  -h[m-a] for m = a, a+1, ..., n-1
  That's -h[0:n-a] added to grad[a:n]

So for off < 0 (a = -off):
  grad[0:n-a] += o1h[a:n]
  grad[a:n] += -h[0:n-a]

```python
def compute_c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    o1h = 1.0 - h
    corr = np.correlate(h, o1h, mode='full')
    c5 = float(np.max(corr) * dx)
    max_val = float(np.max(corr))
    tol = max_val * 1e-6 + 1e-12
    active = np.where(np.abs(corr - max_val) < tol)[0]
    grad = np.zeros(n)
    for mk in active:
        off = mk - (n - 1)
        if off >= 0:
            grad[off:] += o1h[:n - off]
            grad[:n - off] -= h[off:]
        else:
            a = -off
            grad[:n - a] += o1h[a:]
            grad[a:] -= h[:n - a]
    grad /= len(active) if len(active) > 0 else 1
    return c5, grad * dx
```

This is O(n * |active|) per call, and np.correlate is O(n²) for n points (since it computes 2n-1 correlations, each O(n)). So total is O(n²) per iteration.

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    o1h = 1.0 - h
    corr = np.correlate(h, o1h, mode='full')
    c5 = float(np.max(corr) * dx)
    max_val = float(np.max(corr))
    tol = max_val * 1e-6 + 1e-12
    active = np.where(np.abs(corr - max_val) < tol)[0]
    grad = np.zeros(n)
    for mk in active:
        off = mk - (n - 1)
        if off >= 0:
            grad[off:] += o1h[:n - off]
            grad[:n - off] -= h[off:]
        else:
            a = -off
            grad[:n - a] += o1h[a:]
            grad[a:] -= h[:n - a]
    if len(active) > 0:
        grad /= len(active)
    return c5, grad * dx


def project_feasible(v, n):
    target = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def pg_optimize(h, n, t0, tlim, max_it=5000):
    lr = 0.3
    prev_c5 = compute_c5(h)
    for it in range(max_it):
        if time.time() - t0 > tlim:
            break
        c5, grad = compute_c5_subgrad(h)
        accepted = False
        for _ in range(20):
            h_new = project_feasible(h - lr * grad, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                accepted = True
                if it > 0 and abs(prev_c5 - c5) < 1e-11:
                    if lr < 0.02:
                        return h, c5_new
                prev_c5 = c5
                lr = min(lr * 1.15, 5.0)
                break
            lr *= 0.5
        if not accepted:
            if lr < 1e-8:
                break
    return h, compute_c5(h)


def lbfgs_obj(x, n):
    hc = np.clip(x, 0.0, 1.0)
    c, g = compute_c5_subgrad(hc)
    pen = 200.0 * (x.sum() - 0.5 * n)
    gp = g + (200.0 / n) * np.ones(n)
    return c + pen, gp


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    except Exception:
        n = 50
        h = np.full(n, 0.5)
    h = project_feasible(h, n)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = n

    # Phase 1: Main projected gradient optimization
    tlim1 = t0 + budget_s * 0.30
    h, _ = pg_optimize(h, n, t0, tlim1, max_it=5000)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n

    # L-BFGS-B refinement
    if time.time() - t0 < t0 + budget_s * 0.45:
        try:
            res = minimize(lbfgs_obj, h.copy(), args=(n,), jac=True,
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 400, 'fatol': 1e-14})
            h_ref = project_feasible(res.x, n)
        except Exception:
            h_ref = h
        tlim2 = t0 + budget_s * 0.50
        h_ref, _ = pg_optimize(h_ref, n, t0, tlim2, max_it=3000)
        c5r = compute_c5(h_ref)
        if c5r < best_c5:
            best_c5, best_h, best_n = c5r, h_ref.copy(), n

    # Phase 2: Perturbation restarts
    n_restarts = 10
    tlim2_end = t0 + budget_s * 0.80
    for r in range(n_restarts):
        if time.time() - t0 > tlim2_end:
            break
        eps = 0.015 * (1.0 + 0.3 * r)
        h_p = best_h + rng.normal(0, eps, n)
        h_p = project_feasible(h_p, n)
        tlim_r = min(t0 + budget_s * 0.82, time.time() + (tlim2_end - t0) / max(1, n_restarts - r) + 60)
        h_p, _ = pg_optimize(h_p, n, t0, tlim_r, max_it=3000)
        c5p = compute_c5(h_p)
        if c5p < best_c5:
            best_c5, best_h, best_n = c5p, h_p.copy(), n

    # Phase 3: Try larger n via interpolation
    tlim3 = t0 + budget_s * 0.95
    for n_new in [2 * best_n, 4 * best_n]:
        if time.time() - t0 > tlim3 - 30 or n_new > 512:
            break
        xo = np.linspace(0.0, 2.0, best_n)
        xn = np.linspace(0.0, 2.0, n_new)
        h_new = np.interp(xn, xo, best_h)
        h_new = project_feasible(h_new, n_new)
        h_new, _ = pg_optimize(h_new, n_new, t0, tlim3, max_it=4000)
        c5n = compute_c5(h_new)
        if c5n < best_c5:
            best_c5, best_h, best_n = c5n, h_new.copy(), n_new

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
```
