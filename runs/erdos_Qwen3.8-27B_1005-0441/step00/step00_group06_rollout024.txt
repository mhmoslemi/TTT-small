This is O(n^2) per computation. After each change to h, I need to update C. But since I'm only changing one element at a time, I can update C incrementally.

Actually, for simplicity, let me just recompute C after each improvement. The full pass structure is:
1. Compute S = correlate(h, 1-h)
2. Compute C = contrib_matrix(h)
3. For each pair (i,j): find best delta, apply if improving
4. After improvement, recompute S and C
5. Repeat

Let me estimate: step 3 is O(n^2 * 4 * n_lags) = O(n^3) ≈ 1.7M operations per pass. In pure Python with numpy vectorization of the inner loop, maybe 0.1-1s per pass. With 900s budget, that's 900-9000 passes. Should be enough.

import time

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
        rem = diff
        for idx in order:
            a = min(rem, 1.0 - h[idx])
            h[idx] += a
            rem -= a
            if rem < 1e-14:
                break
    else:
        order = np.argsort(h)
        rem = -diff
        for idx in order:
            a = min(rem, h[idx])
            h[idx] -= a
            rem -= a
            if rem < 1e-14:
                break
    return h

def _contrib_matrix(h):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    n_lags = 2 * n - 1
    C = np.zeros((n, n_lags))
    for i in range(n):
        ikm = i - lags
        ik = i + lags
        m1 = (ikm >= 0) & (ikm < n)
        m2 = (ik >= 0) & (ik < n)
        C[i, m1] = 1.0 - h[ikm[m1]]
        C[i, m2] -= h[ik[m2]]
    return C

    """Find d in [0, dmax] minimizing max(S + d*b)."""
    if dmax < 1e-12:
        return 0.0
    cur_max = np.max(S)
    best_d = 0.0
    best_val = cur_max
    # Try a few candidate deltas
    candidates = np.array([0.25, 0.5, 0.75, 1.0]) * dmax
    for d in candidates:
        v = np.max(S + d * b)
        if v < best_val - 1e-12:
            best_val = v
            best_d = d
    return best_d
