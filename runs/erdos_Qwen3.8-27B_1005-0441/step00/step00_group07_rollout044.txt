def project_onto_feasible(h, n):

    h = np.clip(h, 0, 1)
    # Rescale to sum = n/2
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if current_sum > 0 and abs(current_sum - target_sum) > 1e-10:
        # Simple rescaling might push out of [0,1], so use a more careful approach
        # Binary search for the right scale
        lo, hi = 0.0, max(1.0, target_sum / current_sum * 1.1)

            if np.sum(h_trial) < target_sum:

For mode="full", the output has length len(a) + len(v) - 1 = 2n-1.
The output index j corresponds to lag k = j - (len(v) - 1) = j - (n-1).

So c[j] = sum_n h[n+k] * (1-h[n]) where k = j - (n-1) and the sum is over valid n (0 <= n < n, 0 <= n+k < n).

For k >= 0: valid n are 0 <= n <= n-1-k

For k < 0, k = -m where m > 0: valid n are m <= n <= n-1

Now, the gradient of c[j] with respect to h[p]:

For k >= 0:
c[j] = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    top_indices = np.argsort(corr)[-top_k:]
    weights = np.exp(alpha * corr[top_indices])

    return g * dx, corr[top_indices].max() * dx

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64).copy()
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = 0.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h * mid, 0, 1)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h * (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h)
    if diff > 1e-10:
        mask = h < 1.0
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
    elif diff < -1e-10:
        mask = h > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def grad_at_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n and k > 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] += -h[k:]
        elif k == 0:
            g += 1.0 - 2.0 * h
    else:
        m = -k
        if m < n and m > 0:
            g[:n - m] += (1.0 - h[m:])
            g[m:] += -h[:n - m]
    return g

def optimize_smooth(h_init, t_limit, t_start, alpha=5.0, base_step=1.0):
    n = len(h_init)
    h = h_init.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    step = base_step
    no_improve = 0
    h_mom = h.copy()
    beta = 0.7

    for it in range(50000):
        if time.time() - t_start > t_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        cmax = np.max(corr)
        c5 = cmax * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000 and step < 0.005:
            break
        tol = max(1e-10, 0.02 * cmax)
        active = np.where(corr >= cmax - tol)[0]
        top_k = min(len(active), 15)
        idx_sorted = active[np.argsort(corr[active])[-top_k:]]
        lags = (idx_sorted - (n - 1)).astype(int)
        w = np.exp(alpha * (corr[idx_sorted] - cmax))
        w /= np.sum(w)
        g = np.zeros(n)
        for i in range(len(lags)):
            g += w[i] * grad_at_lag(h, lags[i])
        g *= dx
        h_new = h_mom - step * g
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5 - 1e-14:
            h_mom = h_new + beta * (h_new - h)
            h = h_new
            step = min(step * 1.08, 30.0)
        else:
            h_mom = h + beta * (h - h_mom)
            step *= 0.82
            if step < 1e-9:
                step = 0.5
                no_improve = 0
                h_mom = h.copy()
    return best_h, best_c5
