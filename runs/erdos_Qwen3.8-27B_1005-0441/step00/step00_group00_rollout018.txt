                best_c5 = c_val
                best_h = h.copy()
            grad = compute_grad(h, lag, dx)
            v = mu * v + (1.0 - mu) * grad
            h_new = project(h - lr * v, n)
            if compute_c5(h_new) <= c_val + 1e-12:
                h = h_new
            else:
                lr *= 0.85
                if lr < 1e-8:
                    break

    try_update(best_h)
    return best_h, best_c5, len(best_h)
