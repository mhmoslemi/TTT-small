def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, dx

def project_h(h, target_sum):
    """Project h to [0,1] with sum = target_sum"""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute evenly among non-saturated elements
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if np.sum(mask) == 0:
        return np.clip(h, 0, 1)
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)

def gradient_max_corr(h):
    """Compute subgradient of max_k corr[k] w.r.t. h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

    # Use top fraction of lags for smoother gradient

    # Normalize gradient
    norm = np.linalg.norm(grad)

import numpy as np

import time

def compute_c5(h):
    """Compute C5 for given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):  # iterate to convergence
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        n_mask = mask.sum()
        if n_mask == 0:
            break
        h[mask] += diff / n_mask
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_c5(h):
    """Fast C5 computation for binary h."""
    n = len(h)
    m = int(np.sum(h))
    if m == 0 or m == n:
        return 1.0
    # Autocorrelation
    N = 2 * n
    H = np.fft.fft(h.astype(float), N)
    ac = np.fft.ifft(H * np.conj(H)).real[:n-1]
    # ac[0] = m, ac[s] = autocorrelation at lag s
    min_ac = np.min(ac[1:])  # exclude lag 0
    return (m - min_ac) * (2.0 / n)

def exhaustive_search(n, budget_s=60):
    """Exhaustive search over binary sequences with n/2 ones."""
    m = n // 2
    best_c5 = 1.0
    best_h = None

        c5 = (m - min_ac) * (2.0 / n)

        count += 1

def gradient_step(h, top_k=5):
    """Compute gradient of max correlation (using top-k lags)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Use top-k lags weighted by their values
    sorted_idx = np.argsort(corr)[::-1]
    top_indices = sorted_idx[:top_k]
    weights = corr[top_indices]
    w_sum = weights.sum()
    if w_sum < 1e-10:
        return np.zeros(n)
    weights = weights / w_sum
    
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        shift = (n - 1) - k
        for j in range(n):
            val = 0.0
            if 0 <= j + shift < n:
                val += (1.0 - h[j + shift])
            if 0 <= j - shift < n:
                val -= h[j - shift]
            grad[j] += w * val
    
    return grad

def gradient_descent(h0, n_iter=2000, budget_s=100):
    """Project gradient descent for C5 minimization."""
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.05

            break
        
        c5 = compute_c5(h)

def local_swap_search(h, max_iters=5000, budget_s=30):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_grad(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    if cmax < 1e-15:
        return np.zeros(n), cmax * dx
    w = np.exp(beta * (corr - cmax))
    w /= w.sum()
    # Compute gradient of weighted sum of corr components
    grad = np.zeros(n)
    max_k = int(np.ceil(len(corr) * 0.05))  # only top 5% for speed? No, use all with weights
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        s = (n - 1) - k
        wk = w[k]
        # Vectorized gradient for this shift
        if s >= 0:
            # term1[j] = 1-h[j+s] for j in [0, n-s)
            # term2[j] = h[j-s] for j in [s, n)
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n-s]
        else:
            s2 = -s
            # term1[j] = 1-h[j-s2] for j in [s2, n)
            # term2[j] = h[j+s2] for j in [0, n-s2)
            grad[s2:] += wk * (1.0 - h[:n-s2])
            grad[:n-s2] -= wk * h[s2:]
    return grad * dx, cmax * dx

            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:

        grad, _ = smooth_max_grad(h, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:

        improved = False
        for _ in range(10):
            h_new = project_h(h - step * grad / gn, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5:

        else:

            if step < 1e-8:

    return best_h, best_c5

One more optimization idea: for the swap search, instead of random swaps, I can be smarter:
- Compute corr for current h
- Find the worst lag k*
- For that lag, find pairs (i, j) where h[i] and h[j+k*] are both high (contributing to high overlap)

def directed_swap(h, budget_s=30, seed=0):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    
    while time.time() - start < budget_s:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s = (n-1) - k_star
        
        # Find the pair that most contributes to corr[k_star]
        # corr[k_star] = sum_i h[i]*(1-h[i+s]) (for valid i)
        # We want to find i where h[i] is high and h[i+s] is low
        # Swapping h[i] with h[i+s] would help if h[i] > h[i+s]
        
        # Get all contributing pairs
        if s >= 0:
            idxs = range(0, n-s)
            pairs = [(i, i+s) for i in idxs if h[i] > h[i+s] + 0.1]
        else:
            s2 = -s
            idxs = range(0, n-s2)
            pairs = [(i+s2, i) for i in idxs if h[i+s2] > h[i] + 0.1]
        
        if not pairs:
            break
        
        # Try each pair (or a random sample)
        if len(pairs) > 50:
            pairs = [pairs[rng.randint(len(pairs))] for _ in range(50)]
        
        improved = False
        for (i, j) in pairs:
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                improved = True
                break
            else:
                h[i], h[j] = h[j], h[i]
        
        if not improved:
            break
    
    return h, best_c5

import numpy as np
import math
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_grad(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    if cmax < 1e-15:
        return np.zeros(n)
    w = np.exp(beta * (corr - cmax))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = (n - 1) - k
        if s >= 0:
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]
        else:
            s2 = -s
            grad[s2:] += wk * (1.0 - h[:n - s2])
            grad[:n - s2] -= wk * h[s2:]
    return grad * dx

def gradient_descent(h0, budget_s=60, beta_init=200.0):
    n = len(h0)
    target = n / 2.0
    h = project_h(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.02
    start = time.time()
    stall = 0
    it = 0
    while time.time() - start < budget_s and it < 50000:
        it += 1
        beta = max(15.0, beta_init * math.exp(-it / 400.0))
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 400:
            break
        grad = smooth_grad(h, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        improved = False
        for _ in range(12):
            h_new = project_h(h - step * grad / gn, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                improved = True
                break
            step *= 0.55
        if improved:
            step = min(step * 1.15, 0.5)
        else:
            step *= 0.4
            if step < 1e-9:
                break
    return best_h, best_c5

def directed_swap(h, budget_s=30, seed=0):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    stall = 0
    while time.time() - start < budget_s:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s = (n - 1) - k_star
        pairs = []
        if s >= 0:
            for i in range(n - s):
                if h[i] > h[i + s] + 0.05:
                    pairs.append((i, i + s))
        else:
            s2 = -s
            for i in range(n - s2):
                if h[i + s2] > h[i] + 0.05:
                    pairs.append((i + s2, i))
        if not pairs:
            stall += 1
        else:
            if len(pairs) > 80:
                picks = rng.choice(len(pairs), 80, replace=False)
                pairs = [pairs[p] for p in picks]
            improved = False
            for (i, j) in pairs:
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    improved = True
                    break
                h[i], h[j] = h[j], h[i]
            if improved:
                stall = 0
            else:
                stall += 1
        if stall > 50:
            break
    return h, best_c5

def random_swap(h, budget_s=30, seed=0):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    while time.time() - start < budget_s:
        i, j = rng.randint(0, n, 2)
        if i == j or abs(h[i] - h[j]) < 1e-9:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
        else:
            h[i], h[j] = h[j], h[i]
    return h, best_c5

def make_block(n, period=8):
    h = np.zeros(n)
    h[np.arange(n) % period < period // 2] = 1.0
    return h

def make_random_binary(n, rng):
    h = np.zeros(n)
    m = n // 2
    idx = rng.choice(n, m, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        h[idx[0]] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0

    def remaining():
        return max(0.0, budget_s - (time.time() - t0))

    starts = []

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)

        starts.append((1.0 - h0.copy(), 1))
        starts.append((h0[::-1].copy(), 2))
        for dn in [1, 2, -1, 3, -2]:

            if nn >= 8:
                hr = np.interp(np.linspace(0, 1, nn), np.linspace(0, 1, n0), h0)
                starts.append((hr, 10 + dn))

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_final
