        res = minimize(obj_fn, h0, method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        h_opt = project_sum(res.x, n)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(project_sum(h0, n))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    h_ref = None
    n_ref = 0
    try:
        h_ref = np.array(initial_h_values, dtype=float)
        n_ref = len(h_ref)
        c5r = compute_c5(h_ref)
        print(f"Ref: n={n_ref}, c5={c5r:.6f}")
        best_h, best_c5, best_n = h_ref.copy(), c5r, n_ref
    except Exception:
        pass

    ns = set()
    if n_ref > 0:
        ns.add(n_ref)
    ns.update([80, 160, 320])
    ns = sorted(ns)

    for n in ns:
        if time.time() > deadline - 45:
            break
        starts = []
        if h_ref is not None and n == n_ref:
            starts.append(h_ref.copy())
        if h_ref is not None:
            starts.append(make_start(n, seed, h_ref))
        starts.append(make_start(n, seed + 1))
        starts.append(make_start(n, seed + 77))

        for ci, h0 in enumerate(starts):
            if time.time() > deadline - 20:
                break
            c5s = compute_c5(h0)
            print(f"n={n} s{ci} c5={c5s:.5f}", end=" ", flush=True)
            h_cur = h0
            for t_val in [5.0, 15.0, 40.0, 100.0]:
                if time.time() > deadline - 10:
                    break
                h_cur, c5_cur = proj_grad_opt(h_cur, n, t_val, 400, deadline)
            # perturbation restarts
            for rest in range(4):
                if time.time() > deadline - 10:
                    break
                rng_p = np.random.RandomState(seed * 13 + n + ci * 31 + rest)
                h_p = h_cur + 0.03 * rng_p.randn(n)
                h_cur, c5_cur = proj_grad_opt(h_p, n, 30.0, 200, deadline)
            # SLSQP polish
            if time.time() < deadline - 30:
                h_sl, c5_sl = slsqp_opt(h_cur, n, 50.0, deadline)
                if c5_sl < c5_cur:
                    h_cur, c5_cur = h_sl, c5_sl
            h_cur = project_sum(h_cur, n)
            c5_cur = compute_c5(h_cur)
            print(f"-> {c5_cur:.6f}", flush=True)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n
                print(f"  ** IMPROVED to {best_c5:.6f}")

    best_h = project_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)
    print(f"FINAL: n={best_n} c5={best_c5:.6f}")
    return best_h, best_c5, best_n
