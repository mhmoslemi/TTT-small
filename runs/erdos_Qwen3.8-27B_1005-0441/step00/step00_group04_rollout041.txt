def _project_feasible(h, n):
    target = n * 0.5
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _c5_raw(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_val(h):
    return float(_c5_raw(h).max())

def _smooth_obj(h, s):
    corr = _c5_raw(h)
    m = corr.max()
    if m < 1e-15:
        return m
    return m + np.log(np.exp(s * (corr - m)).sum()) / s

def _smooth_grad(h, s):
    n = len(h)
    dx = 2.0 / n
    corr = _c5_raw(h)
    m = corr.max()
    ev = np.exp(s * (corr - m))
    w = ev / ev.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        shift = k - (n - 1)
        wk = w[k] * dx
        if shift >= 0:
            grad[:n - shift] += wk * (1.0 - h[shift:])
            grad[shift:] -= wk * h[:n - shift]
        else:
            s2 = -shift
            grad[s2:] += wk * (1.0 - h[:n - s2])
            grad[:n - s2] -= wk * h[s2:]
    return grad

def _subgrad_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = _c5_raw(h)
    k_max = int(np.argmax(corr))
    shift = k_max - (n - 1)
    sg = np.zeros(n)
    if shift >= 0:
        sg[:n - shift] += dx * (1.0 - h[shift:])
        sg[shift:] -= dx * h[:n - shift]
    else:
        s2 = -shift
        sg[s2:] += dx * (1.0 - h[:n - s2])
        sg[:n - s2] -= dx * h[s2:]
    return sg

    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return np.interp(x_new, x_old, h)

    obj0 = _smooth_obj(h, s)

        if time.time() > deadline:
            break
        g = _smooth_grad(h, s)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        h_new = _project_feasible(h - lr * g, n)
        obj_new = _smooth_obj(h_new, s)
        if obj_new < obj0 - 1e-14:
            h = h_new
            obj0 = obj_new
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
    return h
