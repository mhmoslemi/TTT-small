def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c

def make_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)[:n//2]
    h[idx] = 1.0
    return h

def evaluate_c5_direct(h):
    """Direct evaluation using the specified formula."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def sa_binary(h, n, deadline, rng):
    """Simulated annealing for binary h."""
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    cur_h = h.copy()
    cur_c5 = best_c5
    temp = 0.005
    stall = 0
    
    while time.time() < deadline - 3:
        on = np.where(cur_h > 0.5)[0]
        off = np.where(cur_h < 0.5)[0]
        if len(on) == 0 or len(off) == 0:

        i = on[rng.integers(len(on))]
        j = off[rng.integers(len(off))]
