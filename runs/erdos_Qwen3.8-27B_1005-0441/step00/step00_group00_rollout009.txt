from scipy.optimize import minimize
import time

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(200):
        t = (lo + hi) / 2.0
        tmp = np.clip(h - t, 0, 1)
        if tmp.sum() > target:
            lo = t
        else:
            hi = t
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def softmax_c5_grad(h, beta):
    """Computes smooth gradient of softmax of c_s*dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # softmax weights
    shifted = c * dx - np.max(c * dx)
    weights = np.exp(beta * shifted)
    weights /= weights.sum()
    # gradient = sum_s weights[s] * grad_s * dx
    grad = np.zeros(n)
    n1 = n - 1  # center of full correlation
    for s_idx in range(len(c)):
        if weights[s_idx] < 1e-10:
            continue
        s = s_idx - n1  # actual shift
        if s >= 0:
            grad[s:] += weights[s_idx] * (1.0 - h[:n-s])
            grad[:n-s] -= weights[s_idx] * h[s:]
        else:
            ss = -s
            grad[:n-ss] += weights[s_idx] * (1.0 - h[ss:])
            grad[ss:] -= weights[s_idx] * h[:n-ss]
    return grad * dx

    """Projected gradient descent with softmax objective"""
    rng = np.random.RandomState(seed)
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init

    no_improve = 0

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(200):
        t = (lo + hi) / 2.0
        tmp = np.clip(h - t, 0.0, 1.0)
        if tmp.sum() > target:
            lo = t
        else:
            hi = t
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def softmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    vals = c * dx
    m = np.max(vals)
    e = np.exp(beta * (vals - m))
    w = e / e.sum()
    grad = np.zeros(n)
    n1 = n - 1
    for idx in range(len(c)):
        if w[idx] < 1e-8:
            continue
        s = idx - n1
        if s >= 0:
            grad[s:] += w[idx] * (1.0 - h[:n - s])
            grad[:n - s] -= w[idx] * h[s:]
        else:
            ss = -s
            grad[:n - ss] += w[idx] * (1.0 - h[ss:])
            grad[ss:] -= w[idx] * h[:n - ss]
    return grad * dx

def active_grad(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        s = idx - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[s:] = 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
        else:
            ss = -s
            g[:n - ss] = 1.0 - h[ss:]
            g[ss:] -= h[:n - ss]
        grad += g
    return grad / top_k * dx

    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = 0.85
    vel = np.zeros(n)
    no_improve = 0
    start = time.time()
    lr_init = lr
    while time.time() - start < max_time:
        if no_improve % 40 == 0 and no_improve > 0:
            beta_eff = min(200.0, beta + no_improve * 0.5)
        else:
            beta_eff = beta
        grad = softmax_grad(h, beta_eff)
        vel = momentum * vel - lr * grad
        new_h = project_feasible(h + vel, n)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5 - 1e-12:
            best_h = new_h.copy()
            best_c5 = new_c5
            h = new_h

        else:

            if no_improve % 30 == 0:

                best_c5 = compute_c5(h)
                no_improve = 0
                lr = lr_init
                vel = np.zeros(n)

def scipy_refine(h_init, n, max_time):
    def obj(h):
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        return m + np.log(np.sum(np.exp(c - m))) / 80.0
    def obj_grad(h):
        return softmax_grad(h, 80.0)
    equality = {'type': 'eq', 'fun': lambda h: h.sum() - n / 2.0,
                'jac': lambda h: np.ones(n)}
