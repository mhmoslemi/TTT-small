from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx, corr * dx

def project_to_constraint(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}"""
    n = len(h)
    h = np.clip(h, 0, 1)
    # Bisection on lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    lo, hi = -2, 2
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo+hi)/2, 0, 1)
    # Fine-tune to get exact sum
    diff = target_sum - np.sum(h_proj)
    # Distribute the difference
    if abs(diff) > 1e-12:
        # Add/subtract from elements that aren't at bounds
        free = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if np.any(free):
            n_free = np.sum(free)
            h_proj[free] += diff / n_free
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_overlaps(h):
    """Compute all overlap values (without dx factor)."""
    return np.correlate(h, 1.0 - h, mode="full")

def logsumexp_obj(h, T):
    """Smooth approximation to max of overlaps."""
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h) * dx
    m = np.max(overlaps)
    return (m + np.log(np.sum(np.exp(T * (overlaps - m)))) / T)

def logsumexp_grad(h, T):
    """Gradient of smooth approximation."""
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h) * dx
    m = np.max(overlaps)
    exp_arg = T * (overlaps - m)
    w = np.exp(exp_arg - np.max(exp_arg))
    w /= np.sum(w)
    
    # w has length 2n-1, indexed by n-1+k for lag k
    # For each j, grad[j] = dx * sum_k w[k] * [(1-h[j+k])*I - h[j-k]*I]
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)  # k values
    for idx, k in enumerate(lags):
        if w[n-1+k] > 1e-15:  # skip negligible weights
            # (1-h[j+k]) for j+k in [0,n)
            valid_j_plus = (k >= 0)  # j+k >= 0 always if j>=0
            # j+k < n: j < n-k, so j in [0, n-k-1] if k>=0, or j in [0, n+k-1] if k<0
            # Actually: 0 <= j+k < n means: max(0,-k) <= j <= min(n-1, n-1-k)
            j_start = max(0, -k)
            j_end = min(n-1, n-1-k)
            if j_start <= j_end:
                grad[j_start:j_end+1] += w[n-1+k] * (1 - h[j_start+k:j_end+k+1])
            
            # -h[j-k] for j-k in [0,n)
            # 0 <= j-k < n means: max(0,k) <= j <= min(n-1, n-1+k)
            j_start2 = max(0, k)
            j_end2 = min(n-1, n-1+k)
            if j_start2 <= j_end2:
                grad[j_start2:j_end2+1] -= w[n-1+k] * h[j_start2-k:j_end2-k+1]
    
    return dx * grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlap_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(overlaps)
    exp_vals = np.exp(T * (overlaps - m))
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for idx, k in enumerate(lags):
        wk = w[n-1+k]
        if wk < 1e-15:
            continue
        # Term 1: (1-h[j-k]) for 0<=j-k<n, i.e., j in [k, n-1+k] ∩ [0,n-1]
        j_start = max(0, k)
        j_end = min(n-1, n-1+k)
        if j_start <= j_end:
            grad[j_start:j_end+1] += wk * (1.0 - h[j_start-k:j_end-k+1])
        # Term 2: -h[j+k] for 0<=j+k<n, i.e., j in [-k, n-1-k] ∩ [0,n-1]
        j_start2 = max(0, -k)
        j_end2 = min(n-1, n-1-k)
        if j_start2 <= j_end2:
            grad[j_start2:j_end2+1] -= wk * h[j_start2+k:j_end2+k+1]
    
    return dx * grad

    cur = compute_c5(h)
    best_c5, best_h = cur, h.copy()
    for T in T_sched:
        stall = 0
        for it in range(max_iter):
            if time.time() > deadline:
                return best_h, best_c5
            g = overlap_grad(h, T)
            ng = np.linalg.norm(g)
            if ng < 1e-14:
                break
            lr = 0.5 / (ng + 1e-12)
            ok = False
            for _ in range(50):
                hn = proj(h - lr * g, s)
                cn = compute_c5(hn)
                if cn < cur - 1e-14:
                    ok = True
                    break
                lr *= 0.5
            if ok:
                h, cur = hn, cn
                stall = 0
                if cn < best_c5:
                    best_c5, best_h = cn, h.copy()
            else:
                stall += 1
                if stall > 50:
                    break
    return best_h, best_c5

def alt_init(n, pattern, seed):
    h = np.zeros(n)
    half = n // 2
    if pattern == 0:
        for i in range(n):
            h[i] = 1.0 if i % 2 == 0 else 0.0
        if n % 2 == 1:
            h[n - 1] = 0.0
        h = proj(h, n / 2.0)
    elif pattern == 1:
        rng = np.random.RandomState(seed)
        idx = rng.permutation(n)[:half]
        h[idx] = 1.0
        if n % 2 == 1:
            e = rng.choice(n)
            if h[e] < 1.0:
                h[e] += 0.5
            else:
                h[e] = 0.5
                h[idx[0]] = 0.5
        h = np.clip(h, 0, 1)
        h = proj(h, n / 2.0)
    else:
        rng = np.random.RandomState(seed)
        blocks = [3, 2]
        i = 0
        bi = 0
        while i < n:
            bl = blocks[bi % 2]
            h[i:i + bl] = 1.0 if bi % 2 == 0 else 0.0
            i += bl
            bi += 1
        h = proj(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.95
    np.random.seed(seed)
    best_h, best_c5, best_n = None, np.inf, 0
    has_init = 'initial_h_values' in dir() and initial_h_values is not None

    if has_init:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = proj(h0, n0 / 2.0)
        c5_0 = compute_c5(h0)
        best_h, best_c5, best_n = h0.copy(), c5_0, n0
        h, c5 = local_opt(h0, [5.0, 15.0, 50.0, 200.0, 800.0], 400, min(deadline, t0 + 250))
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, len(h)

    n_cands = [95] if has_init else [89, 93, 95, 97, 99]
    for n in n_cands:
        if time.time() > deadline - 40:
            break
        for p in range(3):
            if time.time() > deadline - 20:
                break
            h_s = alt_init(n, p, seed + n * 7 + p)
            h, c5 = local_opt(h_s, [3.0, 10.0, 40.0, 150.0, 600.0], 250,
                              min(deadline - 20, time.time() + 120))
            if c5 < best_c5:
                best_h, best_c5, best_n = h.copy(), c5, n

    if best_h is not None and time.time() < deadline - 15:
        h, c5 = local_opt(best_h, [200.0, 800.0, 3000.0, 10000.0], 500, deadline)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    if best_h is None:
        n = 95
        best_h = np.zeros(n)
        for i in range(n):
            best_h[i] = 1.0 if i % 2 == 0 else 0.0
        best_h = proj(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
