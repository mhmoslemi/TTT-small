import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_c5_all(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def project_h(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-13:
            break
    return h


def grad_c_k(h, n, lag):
    grad = np.zeros(n)
    i_lo = max(0, lag)
    i_hi = min(n, n + lag)
    for i in range(i_lo, i_hi):
        j = i - lag
        grad[i] += (1.0 - h[j])
        grad[j] -= h[i]
    return grad


def sum_fun(x):
    return x.sum() - len(x) / 2.0


def obj_func(x):
    return compute_c5(x, len(x))


def obj_and_subgrad(x):
    n = len(x)
    dx = 2.0 / n

    c5 = float(np.max(cvals))
    k_max = int(np.argmax(cvals))
    lag = k_max - (n - 1)
    grad = grad_c_k(x, n, lag) * dx
    return c5, grad
