    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def ccp_one_step(h0, n):
    h = cp.Variable(n)
    t = cp.Variable()
    cons = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    
    for j in range(2 * n - 1):
        k = j - (n - 1)
        i_min = max(0, -k)
        i_max = min(n - 1, n - 1 - k)
        if i_min > i_max:

                tl3 = min(20.0, budget_s - (time.time() - t_start) - 2.0)
                if tl3 > 3.0:
                    h3, c53 = refine_gradient(best_h, len(best_h), time_limit=tl3)
                    if c53 < best_c5:
                        best_c5 = c53
                        best_h = h3.copy()

    # Final validation
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    sf = np.sum(best_h)
    if sf > 1e-10:
        best_h *= (n_final / 2.0) / sf
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_final)
