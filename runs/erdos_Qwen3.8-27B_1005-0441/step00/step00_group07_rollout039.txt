def soft_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    n_corr = len(corr)  # 2n-1
    
    # log-sum-exp
    m = np.max(corr)
    exp_corr = np.exp(beta * (corr - m))
    sum_exp = np.sum(exp_corr)
    weights = exp_corr / sum_exp
    obj = (m + np.log(sum_exp) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n_corr):
        shift = j - n + 1
        w = weights[j]
        if w < 1e-15:
            continue
        if shift >= 0:

            grad[shift:] += w * (1.0 - h[:n-shift])

            grad[:n-t] += w * (1.0 - h[t:])
            # -h[m-t] for m in [t, n-1]
            grad[t:] -= w * h[:n-t]
    
    grad *= dx
    return obj, grad

def soft_obj_and_grad(h, beta, penalty=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)

    sum_exp = np.sum(exp_corr)

        w = weights[j]
        if w < 1e-15:
            continue
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:n - shift]
        else:
            t = -shift

            grad[:n - t] -= w * h[t:]

        pass

    return (best_h, best_c5, len(best_h))

def run(seed=42, budget_s=1100, **kwargs):
    return fm_restart(seed=seed, budget_s=budget_s, **kwargs)
