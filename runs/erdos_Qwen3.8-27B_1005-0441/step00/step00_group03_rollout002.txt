    if m == 0:
        return 0.0
    return m + np.log(np.mean(np.exp((vals - m) * alpha))) / alpha

def make_feasible(h, target_sum):

    s = h.sum()
    if s > 0:

    h = np.clip(h, 0, 1)

                else:
                    h_ls[i], h_ls[j] = h_ls[j], h_ls[i]
                    no_improve += 1
            # Perturb and retry
            severity += 1
            n_flip = min(severity * 2, n // 5)
            if n_flip > 0:
                o = np.where(h_ls == 1)[0]
                z = np.where(h_ls == 0)[0]
                flips_o = rng.choice(o, size=min(n_flip, len(o)), replace=False)
                flips_z = rng.choice(z, size=min(n_flip, len(z)), replace=False)
                h_ls[flips_o] = 0.0
                h_ls[flips_z] = 1.0
            if time.time() > t_end - 5:
                break

    # Multi-start from best: perturb and re-optimize
    if best_h is not None and time.time() < t_end - 20:
        n = len(best_h)
        for restart in range(3):
            if time.time() > t_end - 10:
                break
            h_pert = best_h + rng.randn(n) * 0.1
            h_pert = project_feasible(h_pert, n / 2.0)
            alphas = [200, 600, 1500]
            h_opt, c5_opt = do_optimize(h_pert, n, alphas, t_end - 10)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    if best_h is None:
        n = 70
        best_h = np.zeros(n)
        best_h[:n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
