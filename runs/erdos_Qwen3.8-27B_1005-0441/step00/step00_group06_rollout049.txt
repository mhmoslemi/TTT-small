def project_onto_constraints(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):

        if abs(excess) < 1e-10:

        # Bisection on lambda: find lambda such that sum(clip(h+lambda, 0, 1)) = target_sum
        # If excess > 0, we need to decrease: h_new = clip(h - tau, 0, 1) for some tau > 0
        # If excess < 0, we need to increase: h_new = clip(h + tau, 0, 1) for some tau > 0
        lo, hi = 0.0, 2.0

    m = len(h)
    g = np.ones(m)
    if delta >= 0:
        if m - delta > 0:
            g[:m-delta] -= h[delta:]
        if delta > 0:
            g[delta:] -= h[:m-delta]
    else:

        if d > 0:
            g[d:] -= h[:m-d]
        if m - d > 0:

    return (cmax + eps * np.log(np.sum(np.exp((c - cmax) / eps)))) * dx

def smooth_grad(h, n, eps):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full")
    cmax = c.max()
    w = np.exp((c - cmax) / eps)
    w /= w.sum()
    m = n
    grad = np.zeros(m)
    for k in range(len(c)):
        if w[k] < 1e-10:
            continue
        grad += w[k] * grad_lag(h, k - (m - 1))
    return grad * dx

def sum_constraint(h, target_sum):
    return np.sum(h) - target_sum

def sum_constraint_jac(h, target_sum):
    return np.ones_like(h)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def exact_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * (2.0 / n))
