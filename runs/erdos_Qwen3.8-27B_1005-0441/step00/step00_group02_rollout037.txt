                break
            r = np.random.rand(n)
            r = project_h(r, n)
            try_start(r, 0.92)

    # Final polish
    t_polish_end = t0 + budget_s - 3
    if time.time() < t_polish_end:
        b_h, b_C = polish_single_lag(best_h, n, t_polish_end)
        if b_C < best_C:
            best_h = b_h.copy()
            best_C = b_C

    # Verify and return
    best_C = np.max(compute_overlaps(best_h, n))
    best_h = project_h(best_h, n)
    best_C = np.max(compute_overlaps(best_h, n))
    return best_h, best_C, n
