def project_feasible(h, n):

    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(800):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        i = n // 2
        h[i] = min(1.0, max(0.0, h[i] + diff))
    return h


def grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        g[:m] += 1.0 - h[s:]
        g[s:] -= h[:m]
    else:
        a = -s
        m = n - a
        g[a:] += 1.0 - h[:m]
        g[:m] -= h[a:]
    return g


def optimize_core(h, n, deadline):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.06
    tau = 10.0
    stall = 0
    max_stall = 400
    ncands = 2 * n - 1
    top_k = min(9, ncands)

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        ov = corr * (2.0 / n)
        idx_top = np.argpartition(ov, -top_k)[-top_k:]
        max_ov = ov[idx_top].max()
        w = np.exp(tau * (ov[idx_top] - max_ov))
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(top_k)
            w[0] = 1.0
        else:
            w /= wsum

        g = np.zeros(n)
        for i in range(top_k):
            if w[i] > 1e-7:
                s = idx_top[i] - (n - 1)
                g += w[i] * grad_shift(h, s)

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)

        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
            step = min(step * 1.03, 0.25)
            tau = min(tau * 1.015, 100.0)
        else:
            stall += 1
            step *= 0.96
            if stall >= max_stall:
                h = best_h.copy()
                step = 0.04
                tau = max(tau * 0.4, 3.0)
                stall = 0
                if tau < 2.0:
                    break
    return best_h, best_c5


def upsample(h, n_new):
    n = len(h)
    if n_new == n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)


def random_within_feasible(n, seed_offset, rng):
    h = rng.rand(n)
    h = project_feasible(h, n)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Initialize from parent construction
    n0 = len(initial_h_values)
    h_init = project_feasible(np.array(initial_h_values, dtype=float), n0)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Optimize at original resolution
    deadline = t0 + budget_s * 0.12
    h_opt, c5 = optimize_core(h_init.copy(), n0, deadline)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_opt

    # Phase 2: Multi-resolution optimization
    resolutions = [n0 * 2, n0 * 3, n0 * 4, n0 * 6, n0 * 8]
    fractions = [0.18, 0.18, 0.18, 0.15, 0.12]

    for i, n in enumerate(resolutions):
        if n > 600:
            break
        elapsed = time.time() - t0
        remaining = budget_s - elapsed
        if remaining < budget_s * 0.1:
            break
        # Use upsampled best + perturbation
        h = upsample(best_h, n)
        eps = 0.008
        h = project_feasible(h + rng.randn(n) * eps, n)

        alloc = remaining * fractions[i] if i < len(fractions) else remaining * 0.15
        deadline = time.time() + alloc
        h_opt, c5 = optimize_core(h, n, deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 3: Random restarts at moderate resolution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 20:
        n_re = n0 * 3
        n_re = min(n_re, 200)
        for restart in range(3):
            if time.time() - t0 > budget_s * 0.88:
                break
            h = random_within_feasible(n_re, restart, rng)
            # Blend with best
            if len(best_h) != n_re:
                h_best_up = upsample(best_h, n_re)
                h = 0.5 * h + 0.5 * h_best_up
                h = project_feasible(h, n_re)
            alloc = remaining * 0.15
            deadline = time.time() + alloc
            h_opt, c5 = optimize_core(h, n_re, deadline)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    # Phase 4: Final polish at best resolution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 15:
        n_best = len(best_h)
        deadline = t0 + budget_s * 0.97
        h_opt, c5 = optimize_core(best_h.copy(), n_best, deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Final constraint enforcement
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
