import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def compute_linearized(h_star, n):
    dx = 2.0 / n
    ns = 2 * n - 1
    A = np.zeros((ns, n))
    b = np.zeros(ns)
    for m in range(ns):
        s = m - (n - 1)
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        for i in range(i_lo, i_hi):
            A[m, i] += (1.0 - h_star[i - s]) * dx
            A[m, i - s] -= h_star[i] * dx
            b[m] += h_star[i] * h_star[i - s] * dx
    return A, b

def solve_lp(A, b, n):
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ub = -b
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return np.clip(res.x[:n], 0.0, 1.0), float(res.x[-1])
    return None, None

def enforce_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:

            adj = min(abs(diff), abs(1.0 - h[i]) if diff > 0 else abs(h[i]))
            if diff > 0:
                h[i] += min(adj, diff)
            else:
                h[i] -= min(adj, -diff)
            diff = target - h.sum()
            if abs(diff) < 1e-12:

    return np.clip(h, 0.0, 1.0)

        h_new = enforce_sum(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best = h_new.copy()
        if abs(c5_new - prev_c5) < 1e-9 and it > 3:
            break
        prev_c5 = c5_new
        h = h_new
    return best, best_c5
