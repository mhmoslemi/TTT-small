    err = target_sum - h.sum()
    if err > 0:
        room = 1 - h

        else:
            h += err / len(h)

import numpy as np
import time

def _c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx, c

def _grad_lag(h, k):
    n = len(h)
    shift = k - n + 1
    g = np.zeros(n)
    lo = max(0, -shift)
    hi = min(n - 1, n - 1 - shift)
    for i in range(lo, hi + 1):
        g[i] += 1.0 - h[i + shift]
        g[i + shift] -= h[i]
    return g

def _proj(h, ts):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(30):
        e = ts - h.sum()
        if abs(e) < 1e-13:
            break
        if e > 0:
            r = 1.0 - h
            m = r > 1e-12
            if m.any() and e < r[m].sum() * (1 - 1e-12):
                h[m] += e * r[m] / r[m].sum()
            else:
                h += e / len(h)
        else:
            m = h > 1e-12
            if m.any() and -e < h[m].sum() * (1 - 1e-12):
                h[m] += e * h[m] / h[m].sum()
            else:
                h += e / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _softmax_grad(h, c, t):
    n = len(h)
    dx = 2.0 / n
    exp_c = np.exp(t * c * dx)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = exp_c / s
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-6:
            g += w[k] * _grad_lag(h, k)
    return g * dx

def _opt_subgrad(h0, deadline, rng):
    n = len(h0)
    ts = n / 2.0
    h = h0.copy()
    bh, bv = h.copy(), _c5_corr(h)[0]
    v = np.zeros(n)
    lr, beta = 0.4, 0.65
    stuck = 0
    t_smooth = 3.0  # soft-max temperature

    while time.time() < deadline:
        val, c = _c5_corr(h)
        if val < bv - 1e-10:
            bv, bh, stuck = val, h.copy(), 0
        else:
            stuck += 1

        if stuck > 400:
            h = _proj(h + 0.12 * rng.randn(n), ts)
            v[:] = 0
            lr = 0.3
            t_smooth = rng.uniform(1.0, 5.0)
            stuck = 0

        # Use soft-max gradient for smoother optimization
        if stuck > 100:
            g = _softmax_grad(h, c, t_smooth)
            t_smooth *= 0.98
        else:
            k = np.argmax(c)
            thresh = c[k] * 0.9995
            active = c >= thresh
            g = np.zeros(n)
            for kk in np.where(active)[0]:
                g += _grad_lag(h, kk)
            g /= max(active.sum(), 1.0)
            g *= (2.0 / n)

        v = beta * v - lr * g
        h_new = _proj(h + v, ts)
        nv, _ = _c5_corr(h_new)
        if nv <= val + 1e-9:
            h = h_new
        else:
            lr *= 0.9
            if lr < 0.003:
                lr = 0.35
                v[:] = 0
        lr *= 0.9998
        if lr < 0.003:
            lr = 0.003

    return bh, bv

def _round_and_local(h0, deadline, rng):
    n = len(h0)
    ts = n / 2.0
    # Round to near-binary: sort and assign
    sorted_h = np.sort(h0)
    # Create binary-like pattern
    sorted_idx = np.argsort(h0)
    h = np.zeros(n)
    n_ones = int(ts)  # e.g., 22 for n=45
    n_zeros = n - n_ones - 1
    h[sorted_idx[:n_zeros]] = 0.0
    h[sorted_idx[-n_ones:]] = 1.0
    # Middle element gets fractional value
    mid = sorted_idx[n_zeros]
    h[mid] = ts - n_ones  # e.g., 0.5 for n=45
    h = _proj(h, ts)
    bh, bv = _c5_corr(h)[0], h.copy()
    bv, bh = _c5_corr(h)

    # Local search: try swaps
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h > 0.9)[0]
        zeros = np.where(h < 0.1)[0]
        best_gain = 1e-9
        best_swap = None
        for i in ones:
            for j in zeros:
                h_test = h.copy()
                h_test[i], h_test[j] = h_test[j], h_test[i]
                nv, _ = _c5_corr(h_test)
                gain = bv - nv
                if gain > best_gain:
                    best_gain = gain
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            h = _proj(h, ts)
            bv, _ = _c5_corr(h)
            bh = h.copy()
            improved = True
        # Also try adjusting the fractional element
        frac_idx = np.where((h > 0.05) & (h < 0.95))[0]
        for fi in frac_idx:
            for delta in [0.1, 0.2, 0.3, -0.1, -0.2, -0.3]:
                h_test = h.copy()
                h_test[fi] += delta
                h_test = _proj(h_test, ts)
                nv, _ = _c5_corr(h_test)
                if nv < bv - 1e-9:
                    h = h_test
                    bv = nv
                    bh = h.copy()
                    improved = True
                    break
    return bh, bv

def _opt_from(h0, deadline, rng):
    """Combined optimization: subgradient + rounding + more subgradient."""
    h_opt = h0.copy()
    n = len(h0)
    # Phase 1: subgradient
    d1 = min(time.time() + 120, deadline - 30)
    h_opt, v1 = _opt_subgrad(h0, d1, rng)
    # Phase 2: round and local search
    d2 = min(time.time() + 60, deadline - 10)
    h_r, v2 = _round_and_local(h_opt, d2, rng)
    # Phase 3: more subgradient from best
    h_final, v3 = _opt_subgrad(h_r if v2 < v1 else h_opt, deadline, rng)
    best_v = min(v1, v2, v3)
    if best_v == v1:
        return h_opt, v1
    elif best_v == v2:
        return h_r, v2
    else:
        return h_final, v3
