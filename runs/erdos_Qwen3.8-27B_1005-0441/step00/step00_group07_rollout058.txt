def project_to_polytope(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    h = h.copy()
    for _ in range(50):
        # Project onto sum constraint
        h = h + (target_sum - h.sum()) / len(h)
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        # Check if sum constraint is satisfied
        if abs(h.sum() - target_sum) < 1e-10:
            break
    # Final adjustment for exact sum
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference
        if diff > 0:

```python
import numpy as np
import time

def _c5_val(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _fix_sum(h, target):
    """Adjust h so that sum(h) == target exactly, keeping h in [0,1]"""
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    # Find elements to adjust
    if diff > 0:
        # Need to increase some elements
        candidates = np.where(h < 1.0 - 1e-15)[0]
    else:
        # Need to decrease some elements
        candidates = np.where(h > 1e-15)[0]
    if len(candidates) == 0:
        return h
    # Adjust the most flexible element
    # Spread the adjustment
    per = diff / len(candidates)
    for i in candidates:
        adj = np.clip(per, -(h[i]), 1.0 - h[i])
        h[i] += adj
    # Final fixup (should be tiny)
    rem = target - h.sum()
    if abs(rem) > 1e-15:
        for i in range(len(h)):
            adj = np.clip(rem, -(h[i]), 1.0 - h[i])
            h[i] += adj
            rem -= adj
            if abs(rem) < 1e-15:
                break
    return h

def _project(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} using bisection on offset"""
    n = len(h)
    lo = min(0.0, (target - h.sum()) / n) - 1.0
    hi = max(1.0 - h.max(), (target - h.sum()) / n) + 1.0
    # Actually simpler: bisection
    lo = -h.max()
    hi = 1.0 - h.min()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _grad_c5(h, dx, t):
    """Compute gradient of smoothed C5 = (1/t)log(sum(exp(t*g_k)))"""
    n = len(h)
    ones = 1.0 - h
    corr = np.correlate(h, ones, mode='full').astype(float) * dx
    v = np.max(corr)
    w = np.exp(t * (corr - v))
    w /= w.sum()
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-17:
            continue
        tau = m - (n - 1)
        i0 = max(0, -tau)
        i1 = min(n, n - tau)
        if i0 >= i1:
            continue
        ww = w[m] * dx
        idx = np.arange(i0, i1)
        grad[idx] += ww * ones[idx + tau]
        grad[idx + tau] -= ww * h[idx]
    return grad
```

Let me also think about coordinate descent as an alternative:

For each coordinate j, find the best value of h[j] that minimizes C₅, given all other coordinates fixed. This is a 1D optimization problem for each j.

Actually, for a given j, C₅ as a function of h[j] is the max of:
- The overlap values that involve h[j]
- Other overlap values (which don't change)

The values that change with h[j] are those where j appears in the correlation sum. These are the shifts τ where j is a valid index for h[i+τ] or h[i].

For shift τ: h[j] appears as h[i+τ] when i = j-τ (if valid), contributing h[j]*(1-h[j-τ]) to g_τ.
And h[j] appears as h[i] when i = j (always valid), contributing h[j+τ]*(1-h[j]) to g_τ.

So g_τ as a function of x = h[j]:
- First role: x * (1-h[j-τ]) = x * c₁ (linear in x)
- Second role: (h[j+τ]) * (1-x) = h[j+τ] - x*h[j+τ] (linear in x)

Total contribution to g_τ from h[j]:
x*(1-h[j-τ]) + h[j+τ]*(1-x) [only for valid indices]
= x*(1-h[j-τ] - h[j+τ]) + h[j+τ]

So each g_τ is linear in h[j]! That means C₅(h[j]) = max(linear functions of h[j]), which is a piecewise linear convex function in h[j].

The minimum of a convex piecewise linear function over [0,1] occurs at a breakpoint or endpoint. The breakpoints are where two different shifts achieve the max simultaneously.

This means coordinate descent should work well here!

For coordinate descent:
- For each j, compute the current C₅ as a function of h[j]
- Find the minimizer (which will be at 0, 1, or a breakpoint)
- Update h[j]
- Repeat

The breakpoints occur where g_τ₁(h[j]) = g_τ₂(h[j]) for some pair of shifts. Since each g_τ is linear in h[j]:
g_τ(h[j]) = a_τ * h[j] + b_τ

Breakpoints: h[j] = (b_τ₂ - b_τ₁)/(a_τ₁ - a_τ₂) for pairs where a_τ₁ ≠ a_τ₂.

This is a 1D problem with at most O(n) breakpoints. We can find the minimum by evaluating at all breakpoints and endpoints.

import time

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(h, s):
    n = len(h)
    if abs(h.sum() - s) < 1e-16:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -h.max() - 1, 1.0 - h.min() + 1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > s:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def _fix(h, s):
    d = s - h.sum()
    if abs(d) < 1e-14:
        return h
    idx = np.argsort(-h) if d > 0 else np.argsort(h)
    for i in idx:
        a = min(d, 1.0 - h[i]) if d > 0 else max(d, -h[i])
        h[i] += a
        d -= a
        if abs(d) < 1e-14:
            break
    return h

def _grad(h, dx, t):
    n = len(h)
    o = np.correlate(h, 1.0 - h, mode='full').astype(float) * dx
    mx = o.max()
    w = np.exp(t * (o - mx))
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-16:
            continue
        tau = m - n + 1
        i0, i1 = max(0, -tau), min(n, n - tau)
        if i0 >= i1:
            continue
        ww = w[m] * dx
        ix = np.arange(i0, i1)
        g[ix] += ww * (1.0 - h[ix + tau])
        g[ix + tau] -= ww * h[ix]
    return g

def _pgd(h0, n, dx, rng, t_start=20.0, t_end=200.0, max_time=60.0, t0=0.0):
    h = _proj(h0.copy(), n * 0.5)
    _fix(h, n * 0.5)
    best, best_v = h.copy(), _c5(h, dx)
    t = t_start
    lr = 1.0
    stuck = 0
    while time.time() - t0 < max_time:
        g = _grad(h, dx, t)
        for _ in range(40):
            nh = _proj(h - lr * g, n * 0.5)
            _fix(nh, n * 0.5)
            nv = _c5(nh, dx)
            if nv < best_v - 1e-13:
                h = nh
                if nv < best_v:
                    best_v, best = nv, h.copy()
                    stuck = 0
                break
            lr *= 0.5

            h = _proj(h, n * 0.5)
            _fix(h, n * 0.5)
            lr = 1.0
            stuck = 0

def _binary_ls(h_bin, dx, n, t0, max_time=30.0):
