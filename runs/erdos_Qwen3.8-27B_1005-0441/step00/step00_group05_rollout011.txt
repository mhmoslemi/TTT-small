def compute_c5(h):

def adam_optimize(h0, lr=0.02, time_limit=30):
    n = len(h0)
    h = h0.astype(float).copy()
    target_sum = n / 2.0
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        t += 1
        corr = np.correlate(h, 1-h, mode='full')
        T = np.max(corr)
        k_star = np.argmax(corr)
        L = k_star - n + 1
        
        # Gradient of c(L) w.r.t. h
        grad = np.zeros(n)
        if L >= 0:
            grad[L:] += (1 - h[:n-L])
            grad[:n-L] -= h[L:]
        else:
            m_ = -L
            grad[:n-m_] += (1 - h[m_:])
            grad[m_:] -= h[:n-m_]
        
        # Sum constraint penalty
        grad += 0.5 * (np.sum(h) - target_sum) / n
        
        # ADAM
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0 and abs(s - target_sum) > 1e-6:
            # Proportional scaling
            h = h * (target_sum / s)
            h = np.clip(h, 0, 1)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Decay learning rate
        lr *= 0.9999
    
    return best_h, best_c5
