def project_to_constraint(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    target = n / 2.0
    
    # Bisection on lambda
    # We want sum(clip(h - lam, 0, 1)) = target
    # f(lam) = sum(clip(h - lam, 0, 1)) is decreasing in lam
    
    lo, hi = -1.0, 2.0  # lambda range
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

    n = len(h)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def c5_gradient_smooth(h, temperature):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_corr = np.max(corr)
    exps = np.exp((corr - max_corr) * temperature)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    threshold = np.max(weights) * 0.005
    active = np.where(weights > threshold)[0]
    for k in active:
        s = k - 1
        w = weights[k]
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            s2 = -s
            grad[:n - s2] += w * (1.0 - h[s2:])
            grad[s2:] -= w * h[:n - s2]
    grad *= dx
    return grad
