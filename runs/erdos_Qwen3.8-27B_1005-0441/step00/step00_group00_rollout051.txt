    zeros = np.where(h < 0.5)[0]

    T_min = 1e-5
    cooling = 0.9999
    
    for it in range(max_iters):
        if not time_fn():
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Compute Δc(s) for all s
        # Δc(s) = (1-h[i+s])*1_{valid} + h[i-s]*1_{valid} - (1-h[j+s])*1_{valid} - h[j-s]*1_{valid}
        # s ranges from -(n-1) to (n-1), indexed 0 to 2n-2 in corr
        dc = np.zeros(2*n-1)
        
        # Contribution from h[i]: 0->1
        # (1-h[i+s]) for s in [-i, n-1-i]
        lo_s = max(-i, -(n-1))
        hi_s = min(n-1-i, n-1)
        for s in range(lo_s, hi_s+1):
            dc[s + n - 1] += 1.0 - h[i+s]
        # h[i-s] for s in [i-n+1, i]
        lo_s = max(i-n+1, -(n-1))
        hi_s = min(i, n-1)
        for s in range(lo_s, hi_s+1):
            dc[s + n - 1] += h[i-s]
        
        # Contribution from h[j]: 1->0
        lo_s = max(-j, -(n-1))
        hi_s = min(n-1-j, n-1)
        for s in range(lo_s, hi_s+1):
            dc[s + n - 1] -= 1.0 - h[j+s]
        lo_s = max(j-n+1, -(n-1))
        hi_s = min(j, n-1)
        for s in range(lo_s, hi_s+1):
            dc[s + n - 1] -= h[j-s]
        
        # Apply
        new_corr = corr + dc
        new_max = float(np.max(new_corr)) * dx
        delta = new_max - max_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            max_c5 = new_max
            # Update ones and zeros
            ones = ones[ones != i]
            ones = np.append(ones, j)
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h.copy()
        else:
            # Reject
            pass
        
        T = max(T * cooling, T_min)
    
    return best_h, best_c5
