```python
import numpy as np
from scipy.optimize import minimize
import math
import time

def project_to_sum(h, n):
    """Project h to satisfy 0<=h[i]<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Redistribute
    if diff > 0:
        # Need to add more: add to elements not at 1
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        add_per = diff / len(idx)
        h[idx] = np.clip(h[idx] + add_per, 0, 1)
    else:
        # Need to remove: remove from elements not at 0
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        rem_per = -diff / len(idx)
        h[idx] = np.clip(h[idx] - rem_per, 0, 1)
    # Fine adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1.0 - h)[:max(1, int(abs(diff)))]
        else:
            idx = np.argsort(h)[:max(1, int(abs(-diff)))]
        h[idx] = np.clip(h[idx] + np.sign(diff) * abs(diff) / len(idx), 0, 1)
    return h

def compute_c5(h):
    """Compute C5 = max of correlate(h, 1-h) * dx."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx, c

def compute_gradient(h, m_star):
    """Compute gradient of c[m_star] w.r.t. h."""
    n = len(h)
    d = m_star - (n - 1)
    grad = np.zeros(n)
    # Term 1: (1-h[j-d]) for valid j-d
    if d >= 0:
        # j-d >= 0 means j >= d, j-d < n means j < n+d (always true since j<n)
        for j in range(d, n):
            grad[j] += (1.0 - h[j - d])
        # j+d < n means j < n-d
        if d < n:
            for j in range(0, n - d):
                grad[j] -= h[j + d]
    else:
        # d < 0, let d_neg = -d
        d_neg = -d
        # j-d = j+d_neg >= 0 always, j+d_neg < n means j < n-d_neg
        for j in range(0, n - d_neg):
            grad[j] += (1.0 - h[j + d_neg])
        # j+d = j-d_neg >= 0 means j >= d_neg
        for j in range(d_neg, n):
            grad[j] -= h[j - d_neg]
    return grad
```

```python
def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        if not m.any():
            break
        h[m] += d / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx, c

def grad_shift(h, d):
    n = len(h)
    g = np.zeros(n)
    if 0 <= d < n:
        g[d:] += 1.0 - h[:n - d]
        g[:n - d] -= h[d:]
    elif -n < d <= -1:
        dd = -d
        g[:n - dd] += 1.0 - h[dd:]
        g[dd:] -= h[:n - dd]
    return g
```
