```python
def projected_gradient_step(h, k_star, lr):
    n = len(h)
    dx = 2.0 / n
    # Compute gradient for shift k_star
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j - k_star < n:
            grad[j] += (1 - h[j - k_star])
        if 0 <= j + k_star < n:
            grad[j] -= h[j + k_star]
    # Projected gradient descent step
    h_new = h - lr * dx * grad
    # Project onto {0<=h<=1, sum h = n/2}
    h_new = np.clip(h_new, 0, 1)
    # Fix sum
    diff = n/2.0 - np.sum(h_new)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to increase sum; increase elements with most room
            room = 1.0 - h_new
            # Proportional adjustment
            total_room = np.sum(room)
            if total_room > 1e-12:
                h_new += diff * room / total_room
        else:
            # Need to decrease sum; decrease elements with most room above 0
            room = h_new
            total_room = np.sum(room)
            if total_room > 1e-12:
                h_new += diff * room / total_room
        h_new = np.clip(h_new, 0, 1)
        # Final fix
        diff2 = n/2.0 - np.sum(h_new)
        if abs(diff2) > 1e-12:
            mask = (h_new > 1e-8) & (h_new < 1.0 - 1e-8)
            if np.any(mask):
                h_new[mask] += diff2 / np.sum(mask)
    return h_new
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_onto_simplex_sum(h, target_sum):
    """Project h onto {v : 0<=v_i<=1, sum(v_i)=target_sum}"""
    n = len(h)
    v = h.copy()
    # Bisection on threshold
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        v_clipped = np.clip(h - mid, 0, 1)
        s = np.sum(v_clipped)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    v_clipped = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix residual

        if np.sum(mask) > 0:
```
