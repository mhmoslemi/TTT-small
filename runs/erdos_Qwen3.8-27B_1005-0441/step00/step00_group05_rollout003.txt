            h[mask] = np.maximum(h[mask] - step, 0.0)
        else:
            mask = h < 1.0 - 1e-7
            cnt = mask.sum()
            if cnt == 0:
                break
            step = -diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_weighted(h, n, weights):
    """Subgradient of sum_k weights[k]*corr[k] w.r.t. h. weights has length 2n-1."""
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        m = k - n + 1
        wd = w * dx
        # Term 1: (1-h[j+m]) for valid j
        j0 = max(0, -m)
        j1 = min(n, n - m)
        if j0 < j1:
            g[j0:j1] += wd * (1.0 - h[j0 + m:j1 + m])
        # Term 2: -h[j-m] for valid j
        j0 = max(0, m)
        j1 = min(n, n + m)
        if j0 < j1:
            g[j0:j1] -= wd * h[j0 - m:j1 - m]
    return g

def optimize_subgradient(h, n, max_time, rng, lr_init=0.02, temp=5.0):
    """Iterative subgradient descent with softmax weighting."""
    best_h = h.copy()
    best_c5 = compute_overlaps(h, n).max()
    lr = lr_init
    stuck = 0
    t0 = time.time()
    
    while time.time() - t0 < max_time:
        corr = compute_overlaps(h, n)
        c5 = corr.max()
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Softmax weights over correlations
        temp_eff = max(temp, 0.01)
        shifted = (corr - c5) / temp_eff
        w = np.exp(shifted)
        w /= w.sum()
        
        g = subgradient_weighted(h, n, w)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_overlaps(h_new, n).max()
        if c5_new < c5:
            h = h_new
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.85
            stuck += 1
        
        if stuck > 100:
            lr = max(lr * 0.5, 1e-5)
            temp = max(temp * 0.8, 0.01)
            stuck = 0
    
    return best_h, best_c5

def try_perturbation(h, n, rng, n_trials=30):
    """Try random perturbations and local optimization."""
    best_h = h.copy()
    best_c5 = compute_overlaps(h, n).max()
    
    for _ in range(n_trials):
        h_p = h.copy()
        n_pert = max(2, n // 7)
        idx = rng.choice(n, size=n_pert, replace=False)
        h_p[idx] += rng.uniform(-0.15, 0.15, size=n_pert)
        h_p = project_feasible(h_p, n)
        
        # Quick local search
        lr = 0.03
        for _ in range(30):
            corr = compute_overlaps(h_p, n)
            c5 = corr.max()
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h_p.copy()
            # Focus on top 5 correlations
            top_idx = np.argsort(corr)[-5:]
            w = np.zeros(2 * n - 1)
            w[top_idx] = 1.0 / len(top_idx)
            g = subgradient_weighted(h_p, n, w)
            h_p = h_p - lr * g
            h_p = project_feasible(h_p, n)
            lr *= 0.95
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    else:
        n = 70
        h = np.full(n, 0.5)
    
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_overlaps(h, n).max()
    
    elapsed = time.time() - start_time
    t1 = min(400, budget_s - 60)
    
    # Phase 1: Subgradient descent from initial
    if t1 > 30:
        best_h, best_c5 = optimize_subgradient(h, n, t1, rng, lr_init=0.015, temp=8.0)
    
    elapsed = time.time() - start_time
    t2 = min(300, budget_s - 40)
    
    # Phase 2: More refined subgradient with lower temperature
    if t2 > 20:
        h2 = best_h.copy()
        h2, c5_2 = optimize_subgradient(h2, n, t2, rng, lr_init=0.008, temp=2.0)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
    
    elapsed = time.time() - start_time
    t3 = min(200, budget_s - 20)
    
    # Phase 3: Perturbation-based exploration
    if t3 > 20:
        n_trials = max(5, int(t3 / 8))
        best_h, best_c5 = try_perturbation(best_h, n, rng, n_trials=n_trials)
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Phase 4: Final polish - very fine subgradient
    if remaining > 30:
        h4 = best_h.copy()
        h4, c5_4 = optimize_subgradient(h4, n, remaining - 5, rng, lr_init=0.003, temp=0.5)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
    
    # Final: ensure feasibility and evaluate
    best_h = project_feasible(best_h, n)
    final_c5 = compute_overlaps(best_h, n).max()
    
    return best_h, final_c5, n
