def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h = np.clip(h + d / n, 0, 1)
    return h

def subgradient_vec(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1 - h[:n-s])
        g[:n-s] -= h[s:]
    else:
        sp = -s
        g[:n-sp] += (1 - h[sp:])
        g[sp:] -= h[:n-sp]
    return g

def optimize_subgrad(h, n, time_limit):
    start = time.time()
    dx = 2.0 / n
    best = h.copy()
    best_corr_max = np.max(compute_corr(h))
    alpha = 0.5
    no_improve = 0
    
    while time.time() - start < time_limit:
        corr = compute_corr(h)
        curr_max = np.max(corr)
        
        if curr_max < best_corr_max - 1e-10:
            best_corr_max = curr_max
            best = h.copy()
            no_improve = 0
            alpha = min(alpha * 1.05, 1.0)
        else:
            no_improve += 1
        
        if no_improve > 500:
            alpha *= 0.5
            no_improve = 0
            if alpha < 1e-10:
                break

        top_k = 5
        idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for k_idx in idx:
            s = k_idx - (n - 1)
            g += subgradient_vec(h, s, n)
        g /= top_k
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Adaptive step

            g[:n - s] -= h[s:]
    else:
