import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_C(h):
    n = len(h)
    return float(np.max(raw_corr(h)) * 2.0 / n)

def soft_C_and_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = raw_corr(h)
    cmax = float(np.max(corr))
    scaled = (corr - cmax) / eps
    # Clamp for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exps = np.exp(scaled)
    log_sum = cmax + eps * (np.max(scaled) + np.log(np.sum(np.exp(scaled - np.max(scaled)))))
    C_val = dx * log_sum
    weights = exps / np.sum(exps)
    offset = n - 1
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-10:
            continue
        lag = m - offset
        wscale = w * dx
        if lag == 0:
            grad += wscale * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[:n - lag] += wscale * (1.0 - h[lag:n])
            grad[lag:] -= wscale * h[:n - lag]
        elif -n < lag < 0:
            l = -lag
            grad[l:] += wscale * (1.0 - h[:n - l])
            grad[:n - l] -= wscale * h[l:n]
    return C_val, grad

def local_pair_search(h, n, rng, n_iters, temp=0.01):
    C_curr = compute_C(h)
    for _ in range(n_iters):
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            continue
        delta = float(rng.uniform(-0.05, 0.05))
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i], h[j] = hi, hj
        C_new = compute_C(h)
        if C_new < C_curr or (temp > 1e-6 and rng.random() < np.exp(-(C_new - C_curr) / temp)):
            C_curr = C_new
        else:
            h[i], h[j] = hi - delta, hj + delta

import numpy as np
import time

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_C(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * 2.0 / n)

def soft_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(np.max(corr))
    scaled = np.clip((corr - cmax) / eps, -400.0, 400.0)
    sm = np.max(scaled)
    exps = np.exp(scaled - sm)
    sw = float(np.sum(exps))
    weights = exps / sw
    offset = n - 1
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-10:
            continue
        lag = m - offset
        ws = w * dx
        if lag == 0:
            grad += ws * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[:n - lag] += ws * (1.0 - h[lag:n])
            grad[lag:] -= ws * h[:n - lag]
        else:
            l = -lag
            grad[l:] += ws * (1.0 - h[:n - l])
            grad[:n - l] -= ws * h[l:n]
    return grad
