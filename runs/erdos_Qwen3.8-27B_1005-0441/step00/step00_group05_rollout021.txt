def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def weighted_gradient(h, corr, weights, n):
    """Compute gradient of sum_k weights[k]*corr[k] w.r.t h"""
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        shift = k - (n - 1)
        w = weights[k]
        # (1-h[i+shift]) for valid
        valid_p = (idx + shift >= 0) & (idx + shift < n)
        grad[valid_p] += w * (1.0 - h[idx[valid_p] + shift])
        # -h[i-shift] for valid
        valid_m = (idx - shift >= 0) & (idx - shift < n)
        grad[valid_m] -= w * h[idx[valid_m] - shift]
    return grad
