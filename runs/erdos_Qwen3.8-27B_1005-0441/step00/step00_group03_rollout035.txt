    """Flip a 1 to 0 and a 0 to 1, maintaining the sum."""
    n = len(h)

    h_new = h.copy()
    h_new[ones[i]] = 0.0
    h_new[zeros[j]] = 1.0

def local_search(h, dx, rng, max_sweeps=50, time_limit=None):

    c5 = eval_c5(h, dx)
