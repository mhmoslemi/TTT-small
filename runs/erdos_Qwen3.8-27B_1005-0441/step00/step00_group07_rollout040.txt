import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def enforce(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n * 0.5
    for _ in range(2000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-15:
                break
            w = room / room.sum()
            h += diff * w
        else:
            avail = h
            if avail.sum() < 1e-15:
                break
            w = avail / avail.sum()
            h += diff * w
    return np.clip(h, 0.0, 1.0)

def c5_and_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cv = corr * dx
    ks = int(np.argmax(cv))
    c5 = cv[ks]
    d = (n - 1) - ks
    g = np.zeros(n)
    if d >= 0:
        g[:n-d] += (1.0 - h[d:n])
        g[d:] += (-h[:n-d])
    else:
        L = -d
        g[L:] += (1.0 - h[:n-L])
        g[:n-L] += (-h[L:n])
    return c5, g * dx

def upsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    f = interp1d(x_old, h, kind='linear', bounds_error=False, fill_value='extrapolate')
    return np.clip(f(x_new), 0, 1)

def obj_full(x, n):
    return compute_c5(enforce(x, n))

def obj_pattern(x, p, reps, tsum, nn):
    hp = np.clip(x, 0, 1)
    s = hp.sum()
    hp = hp + (tsum - s) / p
    hp = np.clip(hp, 0, 1)
    hf = np.tile(hp, reps)
    hf = enforce(hf, nn)
    return compute_c5(hf)
