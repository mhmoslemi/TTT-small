    c = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if s > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    vmax = np.max(c)
    return float(vmax + np.log(np.sum(np.exp(alpha * (c - vmax)))) / alpha)

def smooth_max_grad(h, n, alpha):

    c = np.correlate(h, 1.0 - h, mode="full") * dx
    vmax = np.max(c)
    p = np.exp(alpha * (c - vmax))

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if s > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    vmax = np.max(c)
    e = np.exp(alpha * (c - vmax))
    return float(vmax + np.log(np.sum(e)) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    vmax = np.max(c)
    p = np.exp(alpha * (c - vmax))
    psum = np.sum(p)
    if psum < 1e-300:
        return np.zeros(n)
    p = p / psum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        sz = k - (n - 1)
        pkdx = p[k] * dx
        for j in range(n):
            d = 0.0
            isz = j - sz
            if 0 <= isz < n:
                d += (1.0 - h[isz])
            jsz = j + sz
            if 0 <= jsz < n:
                d -= h[jsz]
            if d != 0.0:
                grad[j] += pkdx * d
    return grad

def slsqp_optimize(h, n, alpha, maxiter=800, ftol=1e-14):
    target = n / 2.0
    def obj(x):
        return smooth_max_val(np.clip(x, 0, 1), n, alpha)
    def jac(x):
        return smooth_max_grad(np.clip(x, 0, 1), n, alpha)
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def local_refine(h, n, rng, n_iter=5000, scale=0.02):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(n_iter):
        h_try = best + rng.normal(0, scale, n)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best = h_try
            best_c5 = c5_try
            scale = max(0.005, scale * 0.98)
        else:
            scale = max(0.005, scale * 0.995)
    return best, best_c5

def coord_descent(h, n, max_sweeps=30):
    best = h.copy()
    best_c5 = compute_c5(best)
    target = n / 2.0
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            h_try = best.copy()
            for val in np.linspace(0.0, 1.0, 15):
                h_try[j] = val
                s = np.sum(h_try)
                if abs(s - target) > 1e-12:
                    others = np.ones(n, dtype=bool)
                    others[j] = False
                    adj = (target - s) / np.sum(others)
                    h_try[others] = np.clip(h_try[others] + adj, 0, 1)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best = h_try.copy()
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0, 1)
    return project_feasible(h_new, n_new)

def downsample(h, n_old, n_new):
    if n_old <= n_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
