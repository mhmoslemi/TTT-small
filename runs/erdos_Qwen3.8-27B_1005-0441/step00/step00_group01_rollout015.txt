def project_constraint(h, target):

        else:

import numpy as np
import time

def project_constraint(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} via bisection on shift."""
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_c5(h):
    """Compute C5 bound = max_k overlap_k * dx."""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def smooth_grad(h, t):
    """Gradient of (1/t)*log(sum(exp(t*ov_k))) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(ov)
    exps = np.exp(t * (ov - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        wk = w[idx]
        if k >= 0:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
        else:
            m2 = -k
            grad[m2:] += wk * (1.0 - h[:n - m2])
            grad[:n - m2] -= wk * h[m2:]
    return grad * dx

def run_opt(h_init, n_pts, t_lo, t_hi, max_iters, seed, deadline):
    """Smoothed gradient descent with annealed temperature."""
    rng = np.random.RandomState(seed)
    target = n_pts * 0.5
    h = project_constraint(h_init.copy(), target)
    c5 = get_c5(h)
    best_h, best_c5 = h.copy(), c5
    lr = 0.8
    stuck = 0
    for it in range(max_iters):
        if time.time() > deadline - 3:
            break
        frac = min(float(it) / max(float(max_iters), 1.0), 1.0)
        t_cur = t_lo + (t_hi - t_lo) * frac * frac
        g = smooth_grad(h, t_cur)
        h_new = project_constraint(h - lr * g, target)
        c5_new = get_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            h, c5 = h_new, c5_new
            best_h, best_c5 = h.copy(), c5
            stuck = 0
        elif c5_new < c5:
            h, c5 = h_new, c5_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 300:
                lr *= 0.5
                stuck = 0
                if lr < 1e-9:
                    break
            if stuck > 1500 and rng.random() < 0.2:
                h = project_constraint(h + rng.randn(n_pts) * 0.08, target)
                c5 = get_c5(h)
                stuck = 0
    return best_h, best_c5

def binary_local_search(h, deadline, seed):
    """Local search: swap pairs to reduce C5, pushing toward binary."""
    rng = np.random.RandomState(seed)
    n = len(h)
    target = n * 0.5
    h = project_constraint(h.copy(), target)
    c5 = get_c5(h)
    for _ in range(500000):
        if time.time() > deadline - 3:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([-0.1, -0.05, 0.05, 0.1])
        trial = h.copy()
        trial[i] += d
        trial[j] -= d
        trial = project_constraint(trial, target)
        c5t = get_c5(trial)
        if c5t < c5 - 1e-14:
            h, c5 = trial, c5t
    return h, c5
