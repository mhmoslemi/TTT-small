from scipy import optimize
import time

def compute_c5_vec(h):
    """Compute C5 value for a given h vector."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def objective_fn(h):
    """Objective: max correlation * dx (to be minimized)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def grad_objective(h):
    """Gradient of the max correlation objective (for the active constraint)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    # Gradient of corr[k_max] with respect to h
    # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)]) for valid i
    # d(corr[k])/d(h[j]) = sum over valid i: if i==j then (1-h[i+k-(n-1)]), if i+k-(n-1)==j then -h[i]
    grad = np.zeros(n)

        j = i + offset

            grad[i] += (1.0 - h[j])
            grad[j] += (-h[i])

        res = optimize.minimize(obj_smooth, h_init, method='L-BFGS-B',
                                bounds=bounds,
                                options={'maxiter': 5000, 'gtol': 1e-12})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h_init.copy()

def perturb(h, rng):
    n = len(h)
    sigma = rng.uniform(0.01, 0.15)
    h_new = h + rng.normal(0.0, sigma, n)
    h_new = np.clip(h_new, 0.0, 1.0)
    s = np.sum(h_new)
    if s > 0.01:
        h_new = h_new * (0.5 * n) / s
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1e9

    candidates = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        candidates.append(np.array(initial_h_values, dtype=float))
    candidates.append(np.full(47, 0.5))

    for h0 in candidates:
        c0 = eval_c5(h0)
        if c0 < best_c5:
            best_h, best_c5 = h0.copy(), c0
        if budget_s - (time.time() - t0) < 180:
            break
        h_opt = do_optimize(h0)
        c_opt = eval_c5(h_opt)
        if c_opt < best_c5:
            best_h, best_c5 = h_opt.copy(), c_opt

    for _ in range(30000):
        if budget_s - (time.time() - t0) < 180:
            break
        h_init = perturb(best_h, rng)
        h_opt = do_optimize(h_init)
        c_opt = eval_c5(h_opt)
        if c_opt < best_c5:
            best_h, best_c5 = h_opt.copy(), c_opt

    n = len(best_h)
    target = 0.5 * n
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        best_h = best_h + diff / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = eval_c5(best_h)
    return best_h, final_c5, n
