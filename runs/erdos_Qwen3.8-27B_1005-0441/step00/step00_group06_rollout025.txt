import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project(h, target_sum):
    n = len(h)
    h = h.copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) > 1e-10:
        idx = int(np.argmax(h)) if diff < 0 else int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_gradient(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    c_shift = c - c.max()
    w = np.exp(alpha * c_shift)
    ws = w.sum()
    if ws < 1e-15:
        w = w / w.max() if w.max() > 0 else np.ones(len(c)) / len(c)
    else:
        w = w / ws

    grad = np.zeros(n)
    for m in range(len(c)):
        wm = w[m]
        if wm < 1e-12:
            continue
        j = m - (n - 1)
        lo1 = max(0, j)
        hi1 = min(n - 1, j + n - 1)
        if lo1 <= hi1:
            ks = np.arange(lo1, hi1 + 1)
            grad[lo1:hi1 + 1] += wm * (1.0 - h[ks - j])
        lo2 = max(0, -j)
        hi2 = min(n - 1, n - 1 - j)
        if lo2 <= hi2:
            ks = np.arange(lo2, hi2 + 1)
            grad[lo2:hi2 + 1] -= wm * h[ks + j]
    return grad

def gd_optimize(h, dx, target_sum, budget_s, max_iters=30000,
                alpha0=5.0, lr0=0.003, seed=0):
    rng = np.random.default_rng(seed)
    n = len(h)
    h = h.copy().astype(float)
    h = project(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = alpha0
    lr = lr0
    patience = 0
    t_start = time.time()

    for it in range(max_iters):
        if time.time() - t_start > budget_s:
            break
        grad = soft_gradient(h, alpha)
        h_new = h - lr * grad
        h_new = project(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            h = h_new
            patience += 1

        if patience > 1500:
            lr *= 0.65
            alpha *= 1.9
            patience = 0

        if patience > 4000:
            h += 0.015 * rng.standard_normal(n)
            h = project(h, target_sum)
            patience = 0
            lr = lr0
            alpha = alpha0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - t0

        return budget_s - elapsed()

        final_c5 = best_c5

    return best_h, final_c5, len(best_h)
