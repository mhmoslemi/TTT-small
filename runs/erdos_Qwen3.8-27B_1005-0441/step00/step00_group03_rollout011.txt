                    best_h = h.copy()

    # Final polish: one more high-precision pass on the best solution
    elapsed = time.time() - t0
    if elapsed < budget_s - 15:
        n = len(best_h)
        h = best_h.copy()
        for beta in [500.0, 2000.0, 8000.0]:
            _b = beta
            _p = 200.0
            res = minimize(_fun, h, jac=_grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-12})
            h = np.clip(res.x, 0.0, 1.0)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5, len(best_h)
