import numpy as np
from scipy import optimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, t):
    x = np.asarray(x, dtype=float)
    m = float(x.max())
    if t < 1e-15:
        return m
    return m + t * float(np.log(np.sum(np.exp((x - m) / t))))

def project_feasible(x, n):
    x = np.clip(x.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        d = target - float(x.sum())
        if abs(d) < 1e-14:
            break
        mask = (x < 1.0) if d > 0 else (x > 0.0)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        x[mask] = np.clip(x[mask] + d / cnt, 0.0, 1.0)
    return x

def make_obj(n_pts, temp):
    dx = 2.0 / n_pts
    def obj_fn(x):
        xp = project_feasible(x, n_pts)
        c = np.correlate(xp, 1.0 - xp, mode="full")
        return smooth_max(c * dx, temp)
    return obj_fn

def local_search(h, n_pts, rng, max_iter=8000, stale_limit=1500):
    target = n_pts / 2.0
    h_loc = h.copy()
    lc5 = eval_c5(h_loc)
    stale = 0
    best = lc5
    for _ in range(max_iter):
        i, j = rng.choice(n_pts, 2, replace=False)
        d = rng.uniform(0.005, 0.15) * (1.0 if rng.rand() < 0.5 else -1.0)
        h_try = h_loc.copy()
        old_i, old_j = h_try[i], h_try[j]
        h_try[i] = np.clip(old_i + d, 0.0, 1.0)
        h_try[j] = np.clip(old_j - (h_try[i] - old_i), 0.0, 1.0)
        if abs(h_try.sum() - target) < 1e-10:
            c5t = eval_c5(h_try)
            if c5t < lc5 - 1e-14:
                h_loc = h_try
                lc5 = c5t
                stale = 0
                if lc5 < best:
                    best = lc5
            else:
                stale += 1
        else:
            stale += 1
        if stale > stale_limit:
            break
    return h_loc, best

def binary_local_search(h, n_pts, rng, max_iter=6000, stale_limit=2000):
    target = n_pts / 2.0
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    if n_ones != n_pts // 2:
        idx_ones = np.where(h_bin > 0.5)[0]
        idx_zeros = np.where(h_bin < 0.5)[0]
        if n_ones > n_pts // 2:
            for k in range(n_ones - n_pts // 2):
                h_bin[idx_ones[np.random.randint(len(idx_ones))]] = 0
        elif n_ones < n_pts // 2:
            for k in range(n_pts // 2 - n_ones):
                h_bin[idx_zeros[np.random.randint(len(idx_zeros))]] = 1
    h_loc = h_bin.copy()
    lc5 = eval_c5(h_loc)
    best = lc5
    stale = 0
    ones_idx = set(np.where(h_loc > 0.5)[0].tolist())
    zeros_idx = set(np.where(h_loc < 0.5)[0].tolist())
    for _ in range(max_iter):
        if not ones_idx or not zeros_idx:
            break
        i = list(ones_idx)[rng.randint(len(ones_idx))]
        j = list(zeros_idx)[rng.randint(len(zeros_idx))]
        h_try = h_loc.copy()
        h_try[i], h_try[j] = 0.0, 1.0

            lc5 = c5t
            stale = 0
            if lc5 < best:
                best = lc5
        else:
            stale += 1
        if stale > stale_limit:
            break
    return h_loc, best

                h_new = project_feasible(res.x, n_pts)

                if c5 < eval_c5(h) - 1e-12:
