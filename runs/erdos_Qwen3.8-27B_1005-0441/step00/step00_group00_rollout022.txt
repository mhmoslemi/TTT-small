            best_c5 = c5
            h = h_new
            no_imp = 0
            alpha = min(alpha * 1.003, 60.0)
            lr = max(lr * 0.9995, 0.05)
        else:
            no_imp += 1
            if no_imp % 200 == 0:
                lr *= 0.6
            if no_imp > 800:
                h = project_feasible(best_h + rng.randn(n) * 0.03, n * 0.5)
                lr = max(0.3, lr)
                alpha = 8.0
                no_imp = 0

    # Phase 2: Subgradient refinement (30% budget)
    lr2 = 0.8
    no_imp2 = 0
    phase2_end = t0 + budget_s - 5.0

    while time.time() < phase2_end:
        cor = np.correlate(h, 1.0 - h, mode='full')
        max_idx = cor.argmax()
        k = max_idx - (n - 1)
        grad = grad_sub(h, k)
        h_new, c5 = try_step(h, grad, lr2, n)
        if c5 < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5
            h = h_new
            no_imp2 = 0
            lr2 = max(lr2 * 0.999, 0.01)
        else:
            no_imp2 += 1
            if no_imp2 > 300:
                lr2 *= 0.5
                no_imp2 = 0
                if lr2 < 0.005:
                    h = project_feasible(best_h + rng.randn(n) * 0.01, n * 0.5)
                    lr2 = 0.2
                else:
                    h = best_h.copy()

    # Phase 3: Final polish with small steps
    lr3 = 0.05
    no_imp3 = 0
    while time.time() < t0 + budget_s - 2.0:
        cor = np.correlate(h, 1.0 - h, mode='full')
        max_idx = cor.argmax()
        k = max_idx - (n - 1)
        grad = grad_sub(h, k)
        h_new, c5 = try_step(h, grad, lr3, n)
        if c5 < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5
            h = h_new
            no_imp3 = 0
        else:
            no_imp3 += 1
            if no_imp3 > 500:
                lr3 *= 0.5
                no_imp3 = 0
                if lr3 < 0.001:
                    break

    best_h = project_feasible(best_h, n * 0.5)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
