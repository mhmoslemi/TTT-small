import numpy as np
from scipy.optimize import minimize
import time

def smooth_max(x, beta):
    mx = np.max(x)
    if mx > 500/beta:  # prevent overflow
        return mx + np.log(np.mean(np.exp(beta*(x-mx))))/beta
    return np.log(np.sum(np.exp(beta*x)))/beta

def c5_value(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0-h, mode='full') * dx
    return np.max(c)

def objective(h, n, beta):
    h = np.clip(h, 0, 1)
    dx = 2.0 / n
    c = np.correlate(h, 1.0-h, mode='full') * dx
    mx = np.max(c)
    # Smooth max approximation
    c_shifted = c - mx
    return mx + np.log(np.mean(np.exp(beta * c_shifted))) / beta

def run(seed=42, budget_s=1100, **kwargs):

            h_try = make_feasible(h_try, n0)

        h_opt, c5_opt = optimize_h(h_try, n0, beta=80, maxiter=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Binary rounding + re-optimization
    if time.time() - t0 < budget_s - 20:
        h_bin = (best_h > 0.5).astype(float)
        need_ones = n0 // 2
        cur_ones = int(np.sum(h_bin > 0.5))
        if cur_ones < need_ones:
            order = np.argsort(0.5 - best_h)
            for i in order:
                if cur_ones >= need_ones:
                    break
                h_bin[i] = 1.0
                cur_ones += 1
        elif cur_ones > need_ones:
            order = np.argsort(0.5 + best_h)
            for i in order:
                if cur_ones <= need_ones:
                    break
                h_bin[i] = 0.0
                cur_ones -= 1
        h_bin = make_feasible(h_bin, n0)
        h_opt, c5_opt = optimize_h(h_bin, n0, beta=100, maxiter=400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Final high-precision polish
    if time.time() - t0 > 2:
        for beta in [200, 500]:
            h_opt, c5_opt = optimize_h(best_h, n0, beta=beta, maxiter=300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    final_c5 = c5_from_h(best_h)
    return (best_h, final_c5, len(best_h))
