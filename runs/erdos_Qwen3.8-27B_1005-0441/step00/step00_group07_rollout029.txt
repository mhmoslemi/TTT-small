            if res.fun < smooth_obj(h, beta):
                return normalize_h(res.x, len(h))
        except Exception:
            pass
        return h

    # Phase 1: Optimize at initial resolution with beta annealing
    n = len(best_h)
    h = best_h.copy()
    for beta in [15.0, 40.0, 100.0, 250.0, 600.0, 1500.0]:
        if time_left() < 20:
            break
        h = do_slsqp(h, beta, 400)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Random restarts at this resolution
    for i in range(8):
        if time_left() < 15:
            break
        sigma = 0.04 * (1.0 + i * 0.3)
        noise = rng.normal(0, sigma, size=n)
        h_try = normalize_h(best_h + noise, n)
        h_try = do_slsqp(h_try, 200.0, 300)
        h_try = do_slsqp(h_try, 600.0, 200)
        c5 = compute_c5(h_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 3: Multi-resolution refinement
    for target_n in [120, 160, 200, 240, 320, 400]:
        if time_left() < 80:
            break
        h = normalize_h(upsample_h(best_h, target_n), target_n)
        c5_init = compute_c5(h)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h.copy()

        for beta in [50.0, 150.0, 400.0, 1000.0, 2500.0]:
            if time_left() < 25:
                break
            h = do_slsqp(h, beta, 350)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Random restarts at this resolution
        for i in range(4):
            if time_left() < 20:
                break
            sigma = 0.02 * (1.0 + i * 0.5)
            noise = rng.normal(0, sigma, size=target_n)
            h_try = normalize_h(best_h + noise, target_n)
            h_try = do_slsqp(h_try, 200.0, 250)
            h_try = do_slsqp(h_try, 800.0, 200)
            c5 = compute_c5(h_try)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()

    # Phase 4: Final polish - alternate high-beta SLSQP
    if time_left() > 30:
        h = best_h.copy()
        for beta in [500.0, 1000.0, 2000.0, 5000.0]:
            if time_left() < 15:
                break
            h = do_slsqp(h, beta, 300)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Final normalization for exact constraint
    n_final = len(best_h)
    best_h = np.clip(best_h.astype(float), 0.0, 1.0)
    target = n_final / 2.0
    diff = target - np.sum(best_h)
    best_h += diff / n_final
    best_h = np.clip(best_h, 0.0, 1.0)

    # Exact sum fix via iterative projection
    for _ in range(20):
        best_h = np.clip(best_h, 0.0, 1.0)
        diff = target - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        best_h += diff / n_final

    best_c5 = compute_c5(best_h)
    n_final = len(best_h)

    return (best_h, best_c5, n_final)
