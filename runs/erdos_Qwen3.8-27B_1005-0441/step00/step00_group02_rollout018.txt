import numpy as np
import time


def grad_single_lag(s, h, n):
    if s >= 0:
        g = np.zeros(n)
        g[:n - s] = 1.0 - h[s:]
        g[s:] -= h[:n - s]
        return g
    else:
        s2 = -s
        g = np.zeros(n)
        g[s2:] = 1.0 - h[:n - s2]
        g[:n - s2] -= h[s2:]
        return g


def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -1.5, 1.5
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h + lam, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        free = (result > 1e-8) & (result < 1.0 - 1e-8)
        nf = np.sum(free)
        if nf > 0:
            result[free] += diff / nf
    return np.clip(result, 0.0, 1.0)


def optimize_h(h, n, dx, target_sum, budget_fn):
    """Optimize h with projected gradient + momentum, return best (h, C)."""
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    C = c.max()
    best_h = h.copy()
    best_C = C

    lr = 0.05
    mu = 0.8
    v = np.zeros(n)
    no_improve = 0

    for it in range(60000):
        if budget_fn():
            break

        c = np.correlate(h, 1.0 - h, mode="full") * dx
        C = c.max()

        if C < best_C:
            best_C = C
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            lr *= 0.5
            v *= 0.3
            no_improve = 0
            if lr < 1e-9:
                break

        top_k = 3
        top_idx = np.argsort(c)[::-1][:top_k]

        g = np.zeros(n)
        wsum = 0.0
        for rank, idx in enumerate(top_idx):
            w = 1.0 / (rank + 1.0)
            wsum += w
            s = (n - 1) - idx
            g += w * grad_single_lag(s, h, n)
        g /= wsum

        v = mu * v + lr * g
        h_new = project_feasible(h - v, target_sum)

        c_new = np.correlate(h_new, 1.0 - h_new, mode="full") * dx
        if c_new.max() <= C * 1.3 + 0.02:
            h = h_new
        else:
            v *= 0.3

    # Fine-tune phase
    for it in range(20000):
        if budget_fn():
            break
        lr_s = 5e-5 / (1.0 + 0.002 * it)
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        C = c.max()
        if C < best_C:
            best_C = C
            best_h = h.copy()

        worst_idx = int(np.argmax(c))
        s = (n - 1) - worst_idx
        g = grad_single_lag(s, h, n)
        h = project_feasible(h - lr_s * g, target_sum)

    return best_h, best_C

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def timeout():
        return (time.time() - t0) > (budget_s - 5)

    best_h = None
    best_C = np.inf
    best_n = 0

    # Get initial construction if available
    h_init = None

        if initial_h_values is not None:

    except NameError:

    return best_h, best_C, best_n
