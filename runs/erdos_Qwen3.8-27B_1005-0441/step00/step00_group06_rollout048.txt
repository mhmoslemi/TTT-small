def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx


def _grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        m = n - s
        if m > 0:
            g[s:] += 1.0 - h[:m]
            g[:m] -= h[s:]
    else:
        s2 = -s
        m = n - s2
        if m > 0:
            g[:m] += 1.0 - h[s2:]
            g[s2:] -= h[:m]
    return g

    return m + tau * np.log(np.mean(np.exp((ov - m) / tau)))


def _soft_max_grad(h, tau):
