import numpy as np

import time

def project_feasible(h, n):
    t = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        m = (lo + hi) / 2.0
        if np.clip(h + m, 0.0, 1.0).sum() > t:
            lo = m
        else:
            hi = m
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def c5_of(h, n):
    return float(np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n))

def smooth_obj(x, n, eps, pen):
    hv = np.clip(x, 0.0, 1.0)
    ov = np.correlate(hv, 1.0 - hv, mode="full") * (2.0 / n)
    m = ov.max()
    s = np.clip((ov - m) / eps, -50.0, 0.0)
    return m + eps * np.log(np.sum(np.exp(s))) + pen * (hv.sum() - n * 0.5) ** 2
