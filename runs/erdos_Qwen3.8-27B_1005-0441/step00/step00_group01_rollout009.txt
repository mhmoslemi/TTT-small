        for step in [0.05, 0.02, 0.01, 0.005, 0.002]:
            max_iters = 3000
            for _ in range(max_iters):
                if time.time() - t0 > budget_s * 0.85:
                    break
                i = rng.randint(n)
                improved = False
                for d in [step, -step]:
                    h_new = best_h.copy()
                    h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                    h_new = normalize_h(h_new, n)
                    c5_new = obj_c5(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        improved = True
                        break
                if not improved and rng.random() < 0.3:
                    continue

    # Phase 5: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.95:
        res = minimize(obj_c5, best_h, args=(dx,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 1000, 'ftol': 1e-13})
        if res.fun < best_c5:
            best_h = normalize_h(res.x, n)
            best_c5 = obj_c5(best_h, dx)

    # Ensure final sum constraint is tight
    best_h = normalize_h(best_h, n)
    best_c5 = obj_c5(best_h, dx)

    return best_h, best_c5, n
