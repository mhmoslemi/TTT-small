def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    """Project h onto {0<=h<=1, sum=h_target}"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    # Iterative projection
    for _ in range(50):
        diff = h.sum() - target
        if abs(diff) < 1e-12:
            break
        # Distribute evenly
        h -= diff / n
        h = np.clip(h, 0, 1)
    # Final adjustment
    remaining = target - h.sum()
    if abs(remaining) > 1e-10:
        # Find a coordinate we can adjust
        if remaining > 0:
            idx = np.argsort(-h)[:max(1, int(abs(remaining)))]
        else:
            idx = np.argsort(h)[:max(1, int(abs(-remaining)))]
        for i in idx:
            adj = min(remaining, 1 - h[i]) if remaining > 0 else max(remaining, -h[i])
            h[i] += adj
            remaining -= adj
            if abs(remaining) < 1e-10:
                break
    return h

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum(h, target):
    for _ in range(1000):
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] -= diff / mask.sum()
        else:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            h[mask] += (-diff) / mask.sum()
        np.clip(h, 0, 1, out=h)
    return h

def subgradient(h, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    active = np.where(corr >= max_val - 1e-12)[0]
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                grad[i] += (1.0 - h[j])
            j2 = i - shift
            if 0 <= j2 < n:
                grad[i] -= h[j2]
    grad /= len(active)
    grad *= dx
    # Project gradient to maintain sum constraint
    grad = grad - grad.mean()
    return grad
