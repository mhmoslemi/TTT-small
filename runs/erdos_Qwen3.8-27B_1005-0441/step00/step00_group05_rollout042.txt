import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def grad_single_shift(h, shift):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j - shift < n:
            grad[j] += 1.0 - h[j - shift]
        if 0 <= j + shift < n:
            grad[j] -= h[j + shift]
    return grad

def proj_feasible(h, target_sum, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

        top = np.argsort(c)[-4:]
        grad = np.zeros(n)
