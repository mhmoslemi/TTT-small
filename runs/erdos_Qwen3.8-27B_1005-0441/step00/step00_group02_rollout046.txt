def evaluate_c5(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            excess = s - target
            mask = h > 1e-15
            if mask.sum() == 0:

            h = np.clip(h, 0.0, 1.0)

            mask = h < 1.0 - 1e-15

    """Compute f_k for all lags k"""
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return corr * dx

            grad[k:] -= h[:n-k]
    else:

        if n - kk > 0:
            grad[kk:] += (1.0 - h[:n-kk])
            grad[:n-kk] -= h[kk:]
    return grad

import numpy as np
import time

def evaluate_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            adj = min(diff / cnt, (1.0 - h[mask]).max())
            h[mask] += adj
        else:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            adj = min((-diff) / cnt, h[mask].max())
            h[mask] -= adj
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += 1.0 - h[lag:]
            g[lag:] -= h[:m]
    else:
        kk = -lag
        m = n - kk
        if m > 0:
            g[kk:] += 1.0 - h[:m]
            g[:m] -= h[kk:]
    return g

def coord_descent_pass(h, n, dx):
    best_c5_old = np.max(np.correlate(h, 1-h, mode='full')) * dx
    improved = True
    while improved:
        improved = False
        for j in range(n):
            old_val = h[j]
            # Compute a_k and b_k for all lags
            n_lags = 2 * n - 1
            a = np.zeros(n_lags)
            b = np.zeros(n_lags)
            for m_idx in range(n_lags):
                lag = m_idx - (n - 1)
                # f_lag = sum_i h[i]*(1-h[i+lag])
                # w.r.t. h[j]: if j in range, contributes (1-h[j+lag]) as slope
                # and if j-lag in range, contributes -h[j-lag] as slope
                slope = 0.0
                const = 0.0
                lo = max(0, -lag)
                hi = min(n, n - lag)
                for i in range(lo, hi):
                    if i == j:
                        if 0 <= i + lag < n:
                            slope += (1.0 - h[i + lag])
                    else:
                        if 0 <= i + lag < n and (i + lag) == j:
                            slope += -h[i]
                        else:
                            const += h[i] * (1.0 - h[i + lag]) if 0 <= i + lag < n else 0.0
                a[m_idx] = slope
                b[m_idx] = const
            # Find min of max_k (a_k * t + b_k) for t in [0,1]
            # Evaluate at t=0, t=1, and breakpoints
            candidates = [0.0, 1.0]
            # Add breakpoints from top lags
            f_at_0 = a * 0.0 + b
            top_lags = np.argsort(f_at_0)[-20:]
            for m1 in top_lags:
                for m2 in top_lags:
                    if m1 == m2:
                        continue
                    da = a[m1] - a[m2]
                    if abs(da) > 1e-12:
                        tb = (b[m2] - b[m1]) / da
                        if 0.0 < tb < 1.0:
                            candidates.append(tb)
            best_t = old_val
            best_val = np.max(a * old_val + b) * dx
            for t in candidates:
                t = max(0.0, min(1.0, t))
                val = np.max(a * t + b) * dx
                if val < best_val - 1e-12:
                    best_val = val
                    best_t = t
            if abs(best_t - old_val) > 1e-10:
                h[j] = best_t
                h = project_feasible(h, n)
                improved = True
    return h

def optimize_h(h0, n, budget, rng):
    dx = 2.0 / n
    h = project_feasible(h0.copy(), n)
    f = np.correlate(h, 1 - h, mode='full') * dx
    best_c5 = f.max()
    best_h = h.copy()
    beta = 15.0
    lr = 0.01
    momentum = np.zeros(n)
    beta_mom = 0.9
    n_iter = 0
    stalls = 0
    t0 = time.time()
    while time.time() - t0 < budget:
        n_iter += 1
        f = np.correlate(h, 1 - h, mode='full') * dx
        c5 = f.max()
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stalls = 0
        else:
            stalls += 1
        mf = f.max()
        w = np.exp(beta * (f - mf))
        sw = w.sum()
        if sw < 1e-30:
            break
        w /= sw
        thresh = max(w.max() * 0.005, 0.005)
        active = np.where(w > thresh)[0]
        grad = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            grad += w[idx] * grad_lag(h, lag)
        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        step = lr * momentum
        # Backtracking line search
        accepted = False
        for _ls in range(15):
            h_new = project_feasible(h - step, n)
            f_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
            if f_new.max() <= c5 + lr * np.max(np.abs(momentum)) * 0.1:
                h = h_new
                accepted = True
                break
            step *= 0.5
        if not accepted:
            lr *= 0.7
            if lr < 1e-6:
                break
            continue
        lr = min(lr * 1.02, 0.05)
        # Adaptive beta
        n_active = len(active)
        if n_active < 5:
            beta *= 0.8
        elif n_active > 50:
            beta *= 1.2
        beta = np.clip(beta, 1.0, 200.0)
        # Periodic coordinate descent polish
        if n_iter % 500 == 0 and stalls > 50:
            h_cd = h.copy()
            t_cd = time.time()
            h_cd = coord_descent_pass(h_cd, n, dx)
            f_cd = np.correlate(h_cd, 1 - h_cd, mode='full') * dx
            if f_cd.max() < c5:
                h = h_cd
                c5 = f_cd.max()
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            stalls = 0
            if time.time() - t0 > budget:
                break
        # Random perturbation if stalled
        if stalls > 500 and n_iter % 200 == 0:
            h = project_feasible(h + rng.randn(n) * 0.02, n)
            stalls = 0
    # Final coordinate descent
    if time.time() - t0 < budget - 1:
        h_final = coord_descent_pass(best_h.copy(), n, dx)
        f_final = np.correlate(h_final, 1 - h_final, mode='full') * dx
        if f_final.max() < best_c5:
            best_h = h_final
            best_c5 = f_final.max()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    t_elapsed = time.time() - t_global

    # Phase 3: try slightly different n
