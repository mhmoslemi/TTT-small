    n = len(h)

        grad[s:s+m] -= h[:m]
    elif s < 0 and -s < n:
        m = n + s  # n - |s|
        grad[-s:] += (1.0 - h[:m])
        grad[:m] -= h[-s:]

def grad_softmax(h, temperature):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_shift(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if abs(s) >= n:
        return grad
    if s >= 0:
        m = n - s
        grad[:m] += (1.0 - h[s:s + m])
        grad[s:s + m] -= h[:m]
    else:
        sh = -s
        m = n - sh
        grad[sh:] += (1.0 - h[:m])
        grad[:m] -= h[sh:]
    return grad * dx

def grad_softmax(h, temperature):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mval = np.max(corr)
    weights = np.exp((corr - mval) / temperature)
    wsum = np.sum(weights)
    if wsum > 0:
        weights /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-7:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += weights[k] * (1.0 - h[s:s + m])
            grad[s:s + m] -= weights[k] * h[:m]
        else:
            sh = -s
            m = n - sh
            grad[sh:] += weights[k] * (1.0 - h[:m])
            grad[:m] -= weights[k] * h[sh:]
    return grad * dx, corr

def optimize(h, max_time, T_init=0.01):
    start = time.time()
    n = len(h)
    lr = 0.08
    mom = 0.25
    vel = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    use_soft = T_init > 1e-6
    T = T_init

    while time.time() - start < max_time:
        if use_soft:
            grad, corr = grad_softmax(h, T)
        else:
            corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
            k_star = np.argmax(corr)
            s = k_star - (n - 1)
            grad = grad_shift(h, s)

        current_c5 = np.max(corr)
        if current_c5 < best_c5 - 1e-13:
            best_c5 = current_c5
            best = h.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 25:
            lr = max(lr * 0.72, 1e-10)
            stuck = 0
            if use_soft:
                T = max(T * 0.9, 1e-6)
            if lr < 1e-10:
                break

        vel = mom * vel - lr * grad
        h = h + vel
        h = project(h, n)

    return best, best_c5

def binary_project(h, n, rng):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def interpolate_to(h_from, n_to):
    n_from = len(h_from)
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False) + 1.0 / n_from
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False) + 1.0 / n_to
    return np.interp(x_new, x_old, h_from)
