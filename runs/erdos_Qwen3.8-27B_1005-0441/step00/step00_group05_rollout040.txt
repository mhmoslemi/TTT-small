import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_valid_binary(h, n):
    """Ensure h is binary with exactly n//2 ones"""
    h = np.round(h).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        idx = np.random.choice(ones, n_ones - target, replace=False)
        h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        idx = np.random.choice(zeros, target - n_ones, replace=False)
        h[idx] = 1
    return h

def local_search(h, deadline):
    n = len(h)
    current_c5 = compute_c5(h)
    while time.time() < deadline - 0.1:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones:
            if time.time() >= deadline - 0.1:
                break
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h)
                if new_c5 < current_c5:
                    current_c5 = new_c5
                    improved = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    np.random.shuffle(ones)
                    np.random.shuffle(zeros)
                    break
                else:
                    h[i] = 1
                    h[j] = 0
        if not improved:
            break
    return h, current_c5

def simulated_annealing(h, deadline, T_init=0.003):
    n = len(h)
    current_c5 = compute_c5(h)
    T = T_init
    T_min = 1e-5
    
    while time.time() < deadline - 0.1:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(np.random.choice(ones))
        j = int(np.random.choice(zeros))
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or (T > T_min and np.random.random() < np.exp(-delta / T)):
            current_c5 = new_c5
        else:
            h[i] = 1
            h[j] = 0
        
        T = max(T_min, T * 0.9995)
    
    return h, current_c5

def random_binary(n):
    h = np.random.random(n)
    threshold = np.sort(h)[n // 2 - 1] if n // 2 > 0 else 1.0
    h = (h >= threshold).astype(float)
    # Fix count

            h[int(np.where(h == 1)[0][0])] = 0
        else:
            h[int(np.where(h == 0)[0][0])] = 1
