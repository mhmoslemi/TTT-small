            idx = np.arange(j2_lo + lag, j2_hi + lag)
            grad_h[j2_lo:j2_hi] += w * (-h[idx])

import numpy as np
import time
from scipy.optimize import minimize

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_val(h):
    n = len(h)
    return full_corr(h).max() * (2.0 / n)

def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break
        h[m] -= diff / c
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, t_s):
    c = full_corr(h)
    m = c.max()
    return m + np.log(np.sum(np.exp(t_s * (c - m)))) / t_s

def smooth_grad(h, t_s):
    c = full_corr(h)
    m = c.max()
    shifted = t_s * (c - m)
    se = np.exp(shifted)
    w = se / se.sum()
    n = len(h)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wi = w[k_idx]
        if wi < 1e-12:
            continue
        lag = k_idx - (n - 1)
        j1_lo, j1_hi = max(0, lag), min(n, lag + n)
        if j1_lo < j1_hi:
            grad[j1_lo:j1_hi] += wi * (1.0 - h[j1_lo - lag:j1_hi - lag])
        j2_lo, j2_hi = max(0, -lag), min(n, n - lag)
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] += wi * (-h[j2_lo + lag:j2_hi + lag])
    return grad
