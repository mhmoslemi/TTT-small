import numpy as np
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def make_binary(n, rng):
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

def thue_morse(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def sa_optimize(h, n, dx, rng, time_limit, temp0=0.3):
    h = h.copy()
    end = time.time() + time_limit
    tc = compute_c5(h, dx)
    bc, bh = tc, h.copy()
    temp = temp0
    nin = 0
    step = 0
    
    while time.time() < end:
        # pick swap
        while True:
            i, j = rng.integers(n, 2)
            if h[i] != h[j]:
                break
        h[i], h[j] = h[j], h[i]
        nc = compute_c5(h, dx)
        step += 1
        if nc <= tc or (temp > 1e-6 and rng.random() < np.exp((tc-nc)/temp)):
            tc = nc
            if nc < bc:
                bc, bh = nc, h.copy()
                nin = 0
            else:
                nin += 1
        else:
            h[i], h[j] = h[j], h[i]
            nin += 1
        if step % 5000 == 0:
            temp = max(1e-6, temp * 0.998)
        if nin > 200000:
            break
    return bh, bc

def polish(h, n, dx):
    """Try all single swaps, greedily improve"""
    h = h.copy()
    c = compute_c5(h, dx)
    improved = True
    while improved:
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                nc = compute_c5(h, dx)
                if nc < c:
                    c = nc
                    improved = True
                else:
                    h[i], h[j] = 1.0, 0.0
    return h, c

        n0 = len(h0)

        d = tgt - h0.sum()

            h0 = np.clip(h0, 0, 1)
        c0 = compute_c5(h0, dx0)

        if time.time() - t0 > budget_s * 0.85:

        # time per n
        tn = tl / max(1, len(ns) - [x for x in ns if x <= n].index(n))
        # restarts
        nr = max(1, int(tn / 1.0))

            rt = min(4.0, (budget_s*0.85 - (time.time()-t0)) / max(1, nr-r))

            else:

            h, c = polish(h, n, dx)
            
            if c < best_c:

            elif r == 0 and n == 72:
                # Periodic blocking: 2 ones, 2 zeros pattern
                h = np.zeros(n)
                blk = np.array([1.0, 1.0, 0.0, 0.0])
                reps = n // 4
                h[:reps * 4] = np.tile(blk, reps)
            else:
                h = make_binary(n, rng)

            dx = 2.0 / n
            h, c = sa_optimize(h, n, dx, rng, rt, temp0=0.3)
            h, c = polish(h, n, dx)

            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), n

    # Final refinement on the best solution
    if best_h is not None and best_n % 2 == 0:
        ft = budget_s - (time.time() - t0) - 5.0
        if ft > 3.0 and np.all((best_h == 0.0) | (best_h == 1.0)):
            dx = 2.0 / best_n
            best_h, best_c = sa_optimize(best_h, best_n, dx, rng, ft * 0.7, temp0=0.08)
            best_h, best_c = polish(best_h, best_n, dx)

    # Final check and normalization
    h = np.clip(best_h, 0.0, 1.0).astype(np.float64)
    nf = len(h)
    dxf = 2.0 / nf
    tgt = nf / 2.0
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        if d > 0:
            idx = np.where(h < 0.999)[0]
        else:
            idx = np.where(h > 0.001)[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + d, 0.0, 1.0)
    # Rounding for binary solutions
    if best_n % 2 == 0:
        h = np.where(h > 0.5, 1.0, 0.0).astype(np.float64)

    cf = compute_c5(h, dxf)
    return h, cf, nf
