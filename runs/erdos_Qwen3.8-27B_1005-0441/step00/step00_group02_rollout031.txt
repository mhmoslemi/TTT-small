def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Reduce: take from largest first
        order = np.argsort(h)[::-1]

        remaining = diff
        for i in order:

            h[i] -= take
            remaining -= take

                        h = project_feasible(h + rng.randn(n_try) * 0.03, n_try)
                        lr = 0.05
                        stall = 0
                        continue
                thresh = c5 - 0.005
                active = np.where(ov >= thresh)[0]
                sg = np.zeros(n_try)
                for idx in active:
                    k = idx - (n_try - 1)
                    sg += subgrad_shift(h, k, n_try)
                sg /= len(active)
                h_new = project_feasible(h - lr * sg, n_try)
                if compute_overlaps(h_new).max() <= c5 + 1e-7:
                    h = h_new

    # Phase 4: Final fine-tuning on best solution
    if still_going(0.98):
        h = best_h.copy()
        n = len(h)
        for it in range(100000):
            if not still_going(0.99):
                break
            ov = compute_overlaps(h)
            c5 = ov.max()
            worst = ov.argmax()
            k = worst - (n - 1)
            sg = subgrad_shift(h, k, n)
            improved = False
            for lr_t in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                h_new = project_feasible(h - lr_t * sg, n)
                if compute_overlaps(h_new).max() < c5 - 1e-10:
                    h = h_new
                    improved = True
                    if compute_overlaps(h).max() < best_c5:
                        best_c5 = compute_overlaps(h).max()
                        best_h = h.copy()
                    break
            if not improved:
                # Try a small random perturbation
                if it % 500 == 0:
                    h = project_feasible(h + rng.randn(n) * 0.005, n)

    best_h = project_feasible(best_h, n_best)
    best_c5 = compute_overlaps(best_h).max()
    return best_h, best_c5, n_best
