def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_polytope(v, s):
    n = len(v)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        if np.sum(h) > s:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_c5_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    C = corr_raw * dx
    Cmax = np.max(C)
    exp_C = np.exp((C - Cmax) / temp)
    w = exp_C / np.sum(exp_C)
    S = Cmax + temp * np.log(np.sum(exp_C))
    
    # u_k = n-1-k for k=0..2n-2
    u = np.arange(n - 1, -(n), -1)  # [n-1, n-2, ..., 0, -1, ..., -(n-1)]
