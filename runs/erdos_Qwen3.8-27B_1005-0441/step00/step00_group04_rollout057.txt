def project_feasible(v, n):

    lo, hi = -3.0, 3.0
    for _ in range(80):
        lam = (lo + hi) / 2.0
        h = np.clip(v + lam, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, k, n):
    s = k - (n - 1)
    grad = np.zeros(n)
    j_min = max(0, -s)
    j_max = min(n, n - s) - 1
    if j_min > j_max:
        return grad
    js = np.arange(j_min, j_max + 1)
    grad[js + s] += (1.0 - h[js])
    grad[js] -= h[js + s]
    return grad

def evaluate_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def smooth_gradient(h, n, alpha, c_scaled):
    cmax = np.max(c_scaled)
    shifted = alpha * (c_scaled - cmax)
    max_shifted = np.max(shifted)
    weights = np.exp(shifted - max_shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        k_max = int(np.argmax(c_scaled))
        return grad_at_lag(h, k_max, n)
    weights /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] > 1e-10:
            grad += weights[k] * grad_at_lag(h, k, n)
    return grad

def optimize_start(h0, n, t_end, rng, best_c5, best_h, best_n):
    h = h0.copy()
    h = project_feasible(h, n)
    c5 = evaluate_c5(h, n)
    if c5 < best_c5:
        best_c5, best_h, best_n = c5, h.copy(), n

    momentum = np.zeros(n)
    beta = 0.6
    step = 0.1
    alpha = 15.0
    stall = 0
    base_c5 = best_c5

    while time.time() < t_end:
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        c_scaled = c_raw * (2.0 / n)

        grad = smooth_gradient(h, n, alpha, c_scaled)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-12:
            break

        momentum = beta * momentum - step * grad / (norm_grad + 1e-8)
        h = project_feasible(h + momentum, n)

        c5 = evaluate_c5(h, n)
        if c5 < best_c5 - 1e-10:
            best_c5, best_h, best_n = c5, h.copy(), n
            stall = 0
            alpha = min(alpha * 1.03, 300.0)
        else:
            stall += 1

        if stall > 300:
            step *= 0.8
            stall = 0
            if step < 0.005:
                h = best_h.copy()
                perturb = rng.uniform(-0.05, 0.05, n)
                h += perturb
                h = project_feasible(h, n)
                step = 0.1
                alpha = 15.0
                momentum = np.zeros(n)
                stall = 0

    return best_h, best_c5, best_n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def budget_remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Use initial construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        t_end = t0 + budget_s * 0.35
        best_h, best_c5, best_n = optimize_start(h0, n0, t_end, rng, best_c5, best_h, best_n)

    # Phase 2: Try various n and structured starts
    for n_try in [42, 60, 84, 100, 120, 150, 200]:
        if budget_remaining() < 50:
            break
        for trial in range(3):
            if budget_remaining() < 40:
                break
            x = np.linspace(0.0, 2.0, n_try, endpoint=False)
            freq = 1.0 + trial * 0.8 + 0.3 * n_try / 100.0
            h0 = 0.5 + 0.3 * np.cos(2.0 * np.pi * freq * x)
            h0 += 0.15 * np.sin(4.0 * np.pi * freq * x)
            h0 += rng.uniform(-0.1, 0.1, n_try)
            h0 = np.clip(h0, 0.0, 1.0)
            h0 = project_feasible(h0, n_try)
            alloc = min(budget_remaining() - 10, 35.0)
            if alloc < 5:
                break
            t_end = time.time() + alloc
            best_h, best_c5, best_n = optimize_start(h0, n_try, t_end, rng, best_c5, best_h, best_n)

    # Phase 3: Random starts with larger n
    for n_try in [200, 300, 500]:
        if budget_remaining() < 40:
            break
        h0 = rng.uniform(0.0, 1.0, n_try)
        h0 = project_feasible(h0, n_try)
        alloc = min(budget_remaining() - 10, 30.0)
        if alloc < 5:
            break
        t_end = time.time() + alloc
        best_h, best_c5, best_n = optimize_start(h0, n_try, t_end, rng, best_c5, best_h, best_n)

    # Phase 4: Final refinement on best solution
    if best_h is not None and budget_remaining() > 20:
        t_end = t0 + budget_s - 1.0
        best_h, best_c5, best_n = optimize_start(best_h, best_n, t_end, rng, best_c5, best_h, best_n)

    if best_h is None:
        n = 50
        best_h = project_feasible(np.ones(n) * 0.5, n)
        best_c5 = evaluate_c5(best_h, n)
        best_n = n

    best_h = project_feasible(best_h, best_n)
    best_c5 = evaluate_c5(best_h, best_n)
    return best_h, best_c5, best_n
