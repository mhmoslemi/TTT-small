import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(comp))

def compute_c5_and_lag(h):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full") * dx
    max_idx = int(np.argmax(comp))
    return float(comp[max_idx]), max_idx

def subgradient_c5(h, max_idx):
    n = len(h)
    k = max_idx - (n - 1)  # actual lag
    grad = np.zeros(n)
    j_lo = max(0, -k)
    j_hi = min(n - 1, n - 1 - k)
    for j in range(j_lo, j_hi + 1):
        i1 = j + k  # index in h for h_{n+k}
        i2 = j       # index in h for h_n
        if 0 <= i1 < n:
            grad[i1] += (1.0 - h[i2])
        if 0 <= i2 < n:
            grad[i2] -= h[i1]
    return grad

def project_feasible(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum}."""
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    proj = np.clip(h - mid, 0.0, 1.0)
    # Fix rounding
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        # Adjust the element closest to 0.5
        idx = np.argmin(np.abs(proj - 0.5))
        proj[idx] += diff
        proj[idx] = np.clip(proj[idx], 0.0, 1.0)
    return proj

def optimize_local(h, target_sum, max_iters=5000, initial_lr=0.01):
    """Subgradient + projection optimization."""
    n = len(h)
    h = h.copy()
    h = project_feasible(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        c5, max_idx = compute_c5_and_lag(h)

        h = h_new

def coordinate_descent(h, target_sum, max_passes=100):

    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(comp))

def compute_c5_and_lag(h):
    n = len(h)
    dx = 2.0 / n

    return float(comp[max_idx]), max_idx

            grad[i] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            grad[i] -= h[i + k]

import time
import scipy.optimize

    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(comp))

def soft_c5(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full") * dx
    # Log-sum-exp for smooth max
    max_val = np.max(comp)
    return max_val + np.log(np.mean(np.exp(comp - max_val))) / beta

def soft_c5_grad(h, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = np.max(comp)
    weights = np.exp(comp - max_val) / beta
    weights = weights / weights.sum()
    # Gradient is weighted sum of subgradients
    grad = np.zeros(n)
    for idx in range(len(comp)):
        if weights[idx] < 1e-10:
            continue
        k = idx - (n - 1)
        for i in range(n):
            if 0 <= i - k < n:
                grad[i] += weights[idx] * (1.0 - h[i - k])
            if 0 <= i + k < n:
                grad[i] -= weights[idx] * h[i + k]
    return grad * dx

def subgradient_c5(h, max_idx):
    n = len(h)
    k = max_idx - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i - k < n:
            grad[i] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            grad[i] -= h[i + k]
    return grad * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(proj - 0.5))
        for idx in order:
            if diff > 0 and proj[idx] < 1.0:
                add = min(diff, 1.0 - proj[idx])
                proj[idx] += add
                diff -= add
            elif diff < 0 and proj[idx] > 0.0:
                sub = min(-diff, proj[idx])
                proj[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return proj

import numpy as np
import time
import scipy.optimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(comp))

def compute_c5_and_lag(h):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full") * dx
    max_idx = int(np.argmax(comp))
    return float(comp[max_idx]), max_idx

def subgradient_c5(h, max_idx):
    n = len(h)
    k = max_idx - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i - k < n:
            grad[i] += (1.0 - h[i - k])
        if 0 <= i + k < n:
            grad[i] -= h[i + k]
    return grad * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(proj - 0.5))
        for idx in order:
            if diff > 0 and proj[idx] < 1.0:
                add = min(diff, 1.0 - proj[idx])
                proj[idx] += add
                diff -= add
            elif diff < 0 and proj[idx] > 0.0:
                sub = min(-diff, proj[idx])
                proj[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return proj

def subgradient_optimize(h, target_sum, max_iters=15000, base_lr=0.03):
    n = len(h)
    h = project_feasible(h.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    momentum = np.zeros(n)
    beta_mom = 0.8
    for it in range(max_iters):
        c5, max_idx = compute_c5_and_lag(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        grad = subgradient_c5(h, max_idx)
        lr = base_lr / (1.0 + 0.005 * it)
        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        h_new = h - lr * momentum
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 0.05:
            h_new = h
        h = h_new
    return best_h, best_c5

def scipy_polish(h, target_sum, max_it=200):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        comp = np.correlate(x, 1.0 - x, mode="full") * dx
        return float(np.max(comp))
    def obj_grad(x):
        comp = np.correlate(x, 1.0 - x, mode="full") * dx
        mi = int(np.argmax(comp))
        k = mi - (n - 1)
        g = np.zeros(n)
        for i in range(n):
            if 0 <= i - k < n:
                g[i] += (1.0 - x[i - k])
            if 0 <= i + k < n:
                g[i] -= x[i + k]
        return g * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum, 'jac': lambda x: np.ones(n)}]
    try:
        res = scipy.optimize.minimize(obj, h.copy(), jac=obj_grad, method='L-BFGS-B',
                                       bounds=bounds, constraints=cons,
                                       options={'maxiter': max_it, 'ftol': 1e-12, 'gtol': 1e-8})
        hx = project_feasible(res.x, target_sum)
        return hx, compute_c5(hx)
    except Exception:
        return h.copy(), compute_c5(h)

def random_feasible(n, rng):
    h = rng.random(n)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        init_n = len(init_h)
        init_h = project_feasible(init_h, init_n / 2.0)
        best_h = init_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = init_n
    else:
        init_n = 100
        best_h = np.full(init_n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = init_n

    # Phase 1: optimize initial
    rem = budget_s - (time.time() - t0) - 20
    n_it1 = max(5000, int(rem * 40))
    h1, c1 = subgradient_optimize(best_h, best_n / 2.0, max_iters=n_it1)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1, c1, best_n
    # Polish with scipy
    h2, c2 = scipy_polish(best_h, best_n / 2.0, max_it=300)
    if c2 < best_c5:
        best_h, best_c5, best_n = h2, c2, best_n

    # Phase 2: try multiple n values
    n_candidates = [40, 46, 48, 50, 52, 56, 60, 64, 72, 78, 80, 84, 96, 100, 104, 112, 120, 128]
    for n in n_candidates:
        if time.time() - t0 > budget_s - 40:
            break
        target = n / 2.0
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, target)

        for start_idx in range(3):
            if time.time() - t0 > budget_s - 25:
                break
            if start_idx == 0:
                h_s = h_interp.copy()
            else:
                h_s = random_feasible(n, rng)
            rem_local = budget_s - 25 - (time.time() - t0)
            n_it = max(2000, int(rem_local * 25 / 3))
            n_it = min(n_it, 20000)
            hr, cr = subgradient_optimize(h_s, target, max_iters=n_it, base_lr=0.02)
            if cr < best_c5:
                hr, cr2 = scipy_polish(hr, target, max_it=150)
                if cr2 < best_c5:
                    best_h, best_c5, best_n = hr, cr2, n
                else:
                    best_h, best_c5, best_n = hr, cr, n

    # Phase 3: final polish on best
    if time.time() - t0 < budget_s - 10:
        hp, cp = scipy_polish(best_h, best_n / 2.0, max_it=500)
        if cp < best_c5:
            best_h, best_c5, best_n = hp, cp, best_n
        # Additional subgradient rounds
        hr, cr = subgradient_optimize(best_h, best_n / 2.0, max_iters=5000, base_lr=0.01)
        if cr < best_c5:
            best_h, best_c5, best_n = hr, cr, best_n

                if cp2 < best_c5:
                    best_h, best_c5, best_n = hp2, cp2, best_n

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
