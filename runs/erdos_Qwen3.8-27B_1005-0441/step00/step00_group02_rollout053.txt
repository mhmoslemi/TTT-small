import numpy as np
import time

def compute_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def evaluate_C(h):
    return np.max(compute_corr(h))

def pair_optimize(h, i, j):
    """Optimize transfer between h[i] and h[j] to minimize max correlation."""
    hi = h[i]
    hj = h[j]
    emin = -hj
    emax = 1.0 - hi
    if emin >= emax - 1e-15:
        return h
    d = 1e-8
    ht = h.copy()
    ht[i] = hi + d
    ht[j] = hj - d
    ct = compute_corr(ht)
    c0 = compute_corr(h)
    sl = (ct - c0) / d
    ic = c0
    m = len(sl)
    S = sl[:, None] - sl[None, :]
    I = ic[None, :] - ic[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):
        bp = I / S
    mask = (np.abs(S) > 1e-14) & (bp >= emin) & (bp <= emax) & np.isfinite(bp)
    cands = bp[mask]
    if len(cands) > 0:
        cands = np.unique(cands)
    allc = np.concatenate([cands, np.array([emin, emax, 0.0])])
    allc = np.unique(allc)
    allc = np.clip(allc, emin, emax)
    vals = np.max(ic[None, :] + sl[None, :] * allc[:, None], axis=1)
    bi = np.argmin(vals)
    eps = allc[bi]
    if abs(eps) > 1e-12:
        h[i] = hi + eps
        h[j] = hj - eps
    return h

def project_sum(h, target):
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > 1e-15:
        h = h * (target / s)
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        flat = (h > 1e-6) & (h < 1.0 - 1e-6)
        if flat.any():
            h[flat] += diff / flat.sum()
        elif len(h) > 0:
            h[np.argmax(h)] += diff
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = 0.5 * np.ones(45)

    n = len(h0)
    tgt = n / 2.0
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]

    h = project_sum(h0, tgt)
    best_h = h.copy()
    best_C = evaluate_C(best_h)

    # Main loop
    rem_thresh = 20
    while budget_s - (time.time() - t0) > rem_thresh:
        # Full sweep in forward order
        hc = h.copy()
        for pi, pj in pairs:
            if budget_s - (time.time() - t0) < rem_thresh:
                break
            pair_optimize(hc, pi, pj)
        Cc = evaluate_C(hc)
        if Cc < best_C:
            best_C = Cc
            best_h = hc.copy()
            h = hc.copy()
            # Continue sweeping from the new best
            continue
        # No improvement: perturb and try again
        scale = 0.01 + 0.05 * max(0.0, 1.0 - (budget_s - (time.time() - t0)) / budget_s)
        h = best_h + rng.normal(0, scale, n)
        h = project_sum(h, tgt)
        # Partial sweep
        n_partial = min(300, len(pairs))
        idxs = rng.choice(len(pairs), size=n_partial, replace=False)
        for idx in idxs:
            if budget_s - (time.time() - t0) < rem_thresh:
                break
            pi, pj = pairs[int(idx)]
            pair_optimize(h, pi, pj)

    # Final polish
    if budget_s - (time.time() - t0) > 5:
        hf = best_h.copy()
        for _ in range(20):
            for pi, pj in pairs:
                if budget_s - (time.time() - t0) < 5:
                    break
                pair_optimize(hf, pi, pj)
        Cf = evaluate_C(hf)
        if Cf < best_C:
            best_C = Cf
            best_h = hf.copy()

    best_h = project_sum(best_h, tgt)
    best_C = evaluate_C(best_h)
    return best_h, best_C, n
