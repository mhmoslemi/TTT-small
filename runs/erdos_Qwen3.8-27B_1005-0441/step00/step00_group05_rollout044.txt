        s = h.sum()
        if abs(s - target) < 1e-12:
            return h
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target_sum, rng):
    hb = (h >= 0.5).astype(float)
    diff = int(round(target_sum - hb.sum()))
    if diff > 0:
        z = np.where(hb == 0)[0]
        hb[z[rng.randint(len(z), size=min(diff, len(z)))]] = 1
    elif diff < 0:
        o = np.where(hb == 1)[0]
        hb[o[rng.randint(len(o), size=min(-diff, len(o)))]] = 0
    return hb

def greedy_binary_ls(h_start, target_sum, rng, tleft_fn, max_seconds=60):
    n = len(h_start)
    h = h_start.copy()
    c5 = compute_c5(h)
    t_end = time.time() + max_seconds
    while time.time() < t_end and tleft_fn() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        for i in range(len(ones)):
            if time.time() > t_end or tleft_fn() < 3:
                break
            for j in range(len(zeros)):
                oi, zj = ones[i], zeros[j]
                h[oi] = 0
                h[zj] = 1
                c5_t = compute_c5(h)
                if c5_t < c5 - 1e-13:
                    c5 = c5_t
                    improved = True
                    break
                h[oi] = 1
                h[zj] = 0
            if improved:
                break
        if not improved:
            break
    return h, c5

def binary_sa(h_start, target_sum, rng, tleft_fn, max_seconds=300):
    n = len(h_start)
    h = h_start.copy()
    c5 = compute_c5(h)
    T = 0.05
    T_min = 0.002
    t_end = time.time() + max_seconds
    it = 0
    while time.time() < t_end and tleft_fn() > 10:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        oi = ones[rng.randint(len(ones))]
        zj = zeros[rng.randint(len(zeros))]
        h[oi] = 0
        h[zj] = 1
        c5_t = compute_c5(h)
        d = c5_t - c5
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            c5 = c5_t
        else:
            h[oi] = 1
            h[zj] = 0
        if it % 2000 == 0:
            T = max(T * 0.98, T_min)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def tleft():
        return budget_s - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target_sum = n / 2.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Greedy binary local search from initial
    if tleft() > 30:
        hb = make_binary(h0, target_sum, rng)
        hb, c5_b = greedy_binary_ls(hb, target_sum, rng, tleft, 120)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = hb.copy()

    # Phase 2: Binary simulated annealing
    if tleft() > 60:
        h_start2 = make_binary(best_h, target_sum, rng)
        hb2, c5_b2 = binary_sa(h_start2, target_sum, rng, tleft, 250)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = hb2.copy()
        # Re-run greedy after SA
        if tleft() > 30:
            hb2g, c5_b2g = greedy_binary_ls(hb2, target_sum, rng, tleft, 60)
            if c5_b2g < best_c5:
                best_c5 = c5_b2g
                best_h = hb2g.copy()

    # Phase 3: Continuous SLSQP optimization
    if tleft() > 80:
        alpha = 80.0
        def obj_c(hv):
            cv = get_c5_vals(hv)
            m = cv.max()
            return m + np.log(np.sum(np.exp(alpha * (cv - m)))) / alpha

        def jac_c(hv):
            nn = len(hv)
            dxl = 2.0 / nn
            cv = get_c5_vals(hv)
            m = cv.max()
            ev = np.exp(alpha * (cv - m))
            w = ev / ev.sum()
            g = np.zeros(nn)
            for p in range(nn):
                klo = max(0, nn - 1 - p)
                khi = min(2 * nn - 2, nn - 1 + p)
                if klo <= khi:
                    idx = np.arange(klo, khi + 1) - (nn - 1) + p
                    g[p] += np.dot(w[klo:khi + 1], 1.0 - hv[idx])
                klo2 = max(0, p)
                khi2 = min(2 * nn - 2, p + nn - 1)
                if klo2 <= khi2:
                    idx2 = p + (nn - 1) - np.arange(klo2, khi2 + 1)
                    g[p] -= np.dot(w[klo2:khi2 + 1], hv[idx2])
            g *= dxl
            return g

        try:
            cons = [{'type': 'eq', 'fun': lambda hv: hv.sum() - target_sum,
                     'jac': lambda hv: np.ones(n)}]
            bnds = [(0.0, 1.0)] * n
            res = minimize(obj_c, best_h.copy(), method='SLSQP',
                          bounds=bnds, constraints=cons, jac=jac_c,
                          options={'maxiter': 200, 'ftol': 1e-10})
            hn = project_sum(res.x, target_sum)
            c5n = compute_c5(hn)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn.copy()
        except Exception:
            pass

    # Phase 4: Continuous simulated annealing
    if tleft() > 80:
        h_sa = best_h.copy()
        c5_sa = compute_c5(h_sa)
        T = 0.02
        step = 0.12
        it = 0
        t_end = time.time() + min(200, tleft() - 20)
        while time.time() < t_end and tleft() > 10:
            it += 1
            mask = rng.random(n) < 0.5
            h_t = h_sa + rng.randn(n) * step * mask
            h_t = project_sum(h_t, target_sum)
            c5_t = compute_c5(h_t)
            d = c5_t - c5_sa
            if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
                h_sa, c5_sa = h_t, c5_t
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
            if it % 3000 == 0:
                T = max(T * 0.97, 0.001)
                step = max(step * 0.95, 0.02)

    # Phase 5: Final greedy binary local search
    if tleft() > 20:
        hb5 = make_binary(best_h, target_sum, rng)
        hb5, c5_b5 = greedy_binary_ls(hb5, target_sum, rng, tleft, 60)
        if c5_b5 < best_c5:
            best_c5 = c5_b5
            best_h = hb5.copy()

    # Phase 6: Nelder-Mead polish
    if tleft() > 15:
        def obj_nm(hv):
            return compute_c5(hv) + 50.0 * (hv.sum() - target_sum) ** 2
        try:
            res = minimize(obj_nm, best_h.copy(), method='Nelder-Mead',
                          options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-12})
            hn = project_sum(res.x, target_sum)
            c5n = compute_c5(hn)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hn.copy()
        except Exception:
            pass

    # Phase 7: Random restart binary search
    if tleft() > 60:
        for restart in range(3):
            if tleft() < 20:
                break
            h_r = best_h.copy()
            flip_n = max(3, n // 15)
            flip_idx = rng.choice(n, size=flip_n, replace=False)
            for idx in flip_idx:
                h_r[idx] = 1.0 - h_r[idx]
            h_r = project_sum(h_r, target_sum)
            h_rb = make_binary(h_r, target_sum, rng)
            h_rb, c5_rb = greedy_binary_ls(h_rb, target_sum, rng, tleft, 30)
            if c5_rb < best_c5:
                best_c5 = c5_rb
                best_h = h_rb.copy()

    return best_h, best_c5, n
