import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[L:] += 1.0 - h[:n - L]
            g[:n - L] -= h[L:]
    else:
        p = -L
        if p < n:
            g[:n - p] += 1.0 - h[p:]
            g[p:] -= h[:n - p]
    return g

def proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        m = (lo + hi) * 0.5
        if np.sum(np.clip(h - m, 0.0, 1.0)) > tgt:
            lo = m
        else:
            hi = m
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def interp_to(h, n):
    no = len(h)
    if no == n:
        return h.astype(float).copy()
    xo = np.linspace(0.0, 2.0, no, endpoint=False) + 1.0 / no
    xn = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    return np.interp(xn, xo, h)

def optimize_h(h, tgt, deadline, rng):

    hc = h.astype(float).copy()
    bh = hc.copy()

            noi += 1
