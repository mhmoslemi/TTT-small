import numpy as np
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(-(1.0 - h))
    else:
        order = np.argsort(h)
    remaining = diff
    for idx in order:
        if remaining > 0:
            add = min(1.0 - h[idx], remaining)
            h[idx] += add
            remaining -= add
        elif remaining < 0:
            sub = min(h[idx], -remaining)
            h[idx] -= sub
            remaining += sub
        else:
            break
    return np.clip(h, 0.0, 1.0)


def optimize_local(h, n, max_time, rng):
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    start = time.time()

        perm = rng.permutation(n)

            if time.time() - start > max_time:
                break
            old_val = h[i]
            for target in [0.0, 1.0]:
                delta = target - old_val
                if abs(delta) < 1e-10:
                    continue
                if delta > 0:
                    cands = np.where(h > 1e-8)[0]
                    cands = cands[cands != i]
                else:
                    cands = np.where(h < 1.0 - 1e-8)[0]
                    cands = cands[cands != i]
                if len(cands) == 0:
                    continue
                try_n = min(8, len(cands))
                partners = cands[rng.choice(len(cands), try_n, replace=False)]
                found_better = False
                for j in partners:
                    h_new = h.copy()
                    h_new[i] = target
                    h_new[j] = h[j] - delta
                    if h_new[j] < -1e-10 or h_new[j] > 1.0 + 1e-10:
                        continue
                    h_new = np.clip(h_new, 0.0, 1.0)
                    sf = n / 2.0 - h_new.sum()

                    c5_t = compute_c5(h_new)

                        h = h_new
