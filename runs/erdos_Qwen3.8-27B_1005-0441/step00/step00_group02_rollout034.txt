    return float(np.max(c)), c

def project_constrained(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0

        mid = (lo + hi) / 2.0

        if proj.sum() > target:
            lo = mid
        else:

import numpy as np
from scipy.optimize import minimize as spmin
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c

def project_constrained(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def lse_obj_grad(h, n, eps):
    dx = 2.0 / n
    F = np.correlate(h, 1.0 - h, mode='full') * dx
    Fm = F.max()
    eF = np.exp(eps * (F - Fm))
    Z = eF.sum()
    p = eF / Z
    obj = Fm + np.log(Z) / eps
    grad = np.zeros(n)
    # lag 0
    grad += p[n-1] * (1.0 - 2.0 * h)
    # positive and negative lags
    for kp in range(1, n):
        wp = p[n-1+kp]
        wn = p[n-1-kp]
        # lag +kp
        grad[kp:] += wp * (1.0 - h[:n-kp])
        grad[:n-kp] -= wp * h[kp:]
        # lag -kp
        grad[:n-kp] += wn * (1.0 - h[kp:])
        grad[kp:] -= wn * h[:n-kp]
    return obj, grad

def grad_descent(h, n, budget_s, t0, eps_init=10.0, eps_mult=1.6, max_eps=200.0,
                 iters_per_round=8000, alpha_init=0.04, alpha_min=0.0005,
                 decay=0.9998):
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    eps = eps_init
    for rnd in range(12):
        if time.time() - t0 > budget_s - 40:
            break
        h = best_h.copy()
        alpha = alpha_init
        for it in range(iters_per_round):
            if time.time() - t0 > budget_s - 40:
                break
            obj, grad = lse_obj_grad(h, n, eps)
            h = h - alpha * grad
            h = project_constrained(h, n)
            c5, _ = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            alpha = max(alpha * decay, alpha_min)
        eps = min(eps * eps_mult, max_eps)
    return best_h, best_c5
