def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
            c = m.sum()
            if c == 0:
                break
            h[m] = np.minimum(h[m] + diff / c, 1.0)
        else:
            m = h > 0.0
            c = m.sum()
            if c == 0:
                break
            h[m] = np.maximum(h[m] + diff / c, 0.0)
    return h

def make_binary_init(n, rng):
    h = np.zeros(n)
    t = n / 2.0
    n_ones = int(t)
    frac = t - n_ones

    h[idx[:n_ones]] = 1.0
    if frac > 1e-12:

    return h
