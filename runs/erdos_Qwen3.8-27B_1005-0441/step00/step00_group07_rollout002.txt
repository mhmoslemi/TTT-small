For the max objective, if the max is at index m* (corresponding to k* = m* - (n-1)):
d(max)/dh[j] = dx * d(result[m*])/dh[j] = dx * [(1-h[j-k*]) * I[0<=j-k*<n] - h[j+k*] * I[0<=j+k*<n]]

For the smooth max with temperature T:

where w_m = exp(T*(result[m]*dx - max)) / sum exp(T*(result[m']*dx - max))

This is O(n * (number of significant lags)) per gradient evaluation.

    return mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    mc = np.max(corr)
    ew = np.exp(T * (corr - mc))
    w = ew / np.sum(ew)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-8:
            continue
        k = m - (n - 1)
        wm = dx * w[m]
        if k >= 0:
            grad[k:] += wm * (1.0 - h[:n-k])
            grad[:n-k] += -wm * h[k:]
        else:
            kk = -k
            grad[:n-kk] += wm * (1.0 - h[kk:])
            grad[kk:] += -wm * h[:n-kk]
    return grad

def cons_fun(h):
    return np.sum(h) - len(h) / 2.0

def cons_jac(h):
    return np.ones(len(h))

def optimize_stage(h_init, T_list, max_iter, t0, budget_s):
    n = len(h_init)
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    for T in T_list:
        if time.time() - t0 > budget_s * 0.85:
            break
        if n > 400 and T > 200:
            break  # avoid slow optimization at high n with high T
        res = minimize(smooth_obj_val, h, args=(T,), jac=smooth_obj_grad,
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h = np.clip(res.x, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Initial values
    iv = globals().get('initial_h_values', None)

    # Continue annealing

    if c5 < c5_best:
        c5_best, h_best = c5, h.copy()
    
    # Phase 2: Try higher resolution

        x_old = np.linspace(0, 2, n_best)
        for n_new in [160, 320]:
            if time.time() - t0 > budget_s * 0.7:

            h_interp = np.interp(x_new, x_old, h_best)
            h = optimize_stage(h_interp, [10, 50, 150], 200, t0, budget_s)
            c5 = evaluate_c5(h)
            if c5 < c5_best:
                c5_best, h_best, n_best = c5, h.copy(), n_new
    
    # Phase 3: Random restarts
    n_restarts = 10
    for r in range(n_restarts):
        if time.time() - t0 > budget_s * 0.88:
            break
        scale = 0.02 + 0.03 * r / max(n_restarts, 1)
        h_rand = h_best + scale * np.random.randn(n_best)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand += (n_best / 2.0 - np.sum(h_rand)) / n_best
        h_rand = np.clip(h_rand, 0, 1)
        h = optimize_stage(h_rand, [10, 50, 100], 150, t0, budget_s)
        c5 = evaluate_c5(h)
        if c5 < c5_best:
            c5_best, h_best = c5, h.copy()
    
    # Phase 4: Try to snap to near-binary and re-optimize
    if time.time() - t0 < budget_s * 0.92:
        for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
            h_bin = (h_best > thresh).astype(float)

                # Re-optimize from binary

                c5 = evaluate_c5(h)
                if c5 < c5_best:
                    c5_best, h_best = c5, h.copy()
    
    return h_best, c5_best, n_best
