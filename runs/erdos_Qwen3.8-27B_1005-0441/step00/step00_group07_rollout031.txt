def project_constraint(x, target_sum):
    lo = x.min() - target_sum - 1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_objective(x, T):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    tc = T * (c - c.max())
    return c.max() + np.log(np.sum(np.exp(tc))) / T

def smooth_gradient(x, T):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    tc = T * (c - c.max())
    e = np.exp(tc)
    p = e / e.sum()

    best_h = project_constraint(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
