def adam_step(grad, m, v, t, lr, beta1, beta2, eps):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    return lr * m_hat / (np.sqrt(v_hat) + eps)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_gradient(h, n, T):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    exp_c = np.exp((c - max_c) / T)
    p = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = p[m]
        if w < 1e-12:
            continue
        lag = m - (n - 1)

            if n_valid > 0:
                grad[lag:n] += w * (1.0 - h[0:n_valid])
                grad[0:n_valid] -= w * h[lag:n]
        else:

            if n_valid > 0:
                grad[0:n_valid] += w * (1.0 - h[al:n])
                grad[al:n] -= w * h[0:n_valid]
    
    grad *= dx
    return grad

def project(h, n):

        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        h -= diff / n
    return np.clip(h, 0.0, 1.0)

        else:

def project(h, n):

        d = h.sum() - target
        if abs(d) < 1e-13:
            break
        h -= d / n
    h = np.clip(h, 0.0, 1.0)

    d = h.sum() - target

        if d > 0:
            idx = np.argmax(h)
            h[idx] -= d
        else:
            idx = np.argmin(h)
            h[idx] -= d
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def compute_gradient(h, n, T):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = c.max()
    e = np.exp((c - max_c) / T)
    p = e / e.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = p[m]
        if w < 1e-13:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            nv = n - lag
            if nv > 0:
                grad[lag:] += w * (1.0 - h[:nv])
                grad[:nv] -= w * h[lag:]
        else:
            al = -lag
            nv = n - al
            if nv > 0:
                grad[:nv] += w * (1.0 - h[al:])
                grad[al:] -= w * h[:nv]
    grad *= dx
    return grad

def project(h, n):
    target = n * 0.5
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        d = h.sum() - target
        if abs(d) < 1e-13:
            break
        h = h - d / n
    h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    d = h.sum() - target
    if abs(d) > 1e-15:
        if d > 0:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] - d)
        else:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] - d)
    return h

def adam_optimize(h_init, n, max_iters, T_start, T_end, lr, seed=0, patience=800):
    h = h_init.astype(float).copy()
    h = project(h, n)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    last_imp = 0
    rng = np.random.RandomState(seed)
    
    for t in range(1, max_iters + 1):
        frac = min(1.0, t / max_iters)
        T = T_start * (T_end / T_start) ** frac
        grad = compute_gradient(h, n, T)
        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        m_hat = m_buf / (1.0 - b1 ** t)
        v_hat = v_buf / (1.0 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-9:
            best_c = c
            best_h = h_new.copy()
            last_imp = t
        h = h_new
        if t - last_imp > patience and t > patience:
            h = project(h + 0.02 * rng.randn(n), n)
            m_buf[:] = 0.0
            v_buf[:] = 0.0
            last_imp = t
    return best_h, best_c
