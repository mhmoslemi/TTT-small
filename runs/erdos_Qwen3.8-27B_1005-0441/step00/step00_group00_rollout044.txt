    d = t - h.sum()
    if abs(d) > 1e-12:
        ix = np.where((h < 1.0) if d > 0 else (h > 0.0))[0]
        if ix.size:
            h[ix] += d / ix.size
            h = np.clip(h, 0.0, 1.0)
    return h

def _rh(n, rng):
    return _norm(rng.uniform(0.05, 0.95, n))

def _sh(n, k, ph):
    return _norm(0.5 + 0.35 * np.sin(2.0 * np.pi * k * np.arange(n) / n + ph))

def _bh(n, nb, cnt):
    h = np.zeros(n)
    w = max(1, n // nb)
    for j in range(min(cnt, nb)):
        h[j*w:j*w+w] = 1.0
    return _norm(h)

def _per(n, p, q):
    h = np.zeros(n)
    xv = np.arange(n)
    h[xv % p < q] = 1.0
    return _norm(h)

def _slv(h0, mi):
    global _G
    n = len(h0)
    _G = n
    return minimize(_c5_obj, h0, method='SLSQP',
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                    options={'maxiter': mi, 'ftol': 1e-12})

def _bin_rnd(h):
    n = len(h)
    tg = n // 2
    if tg == 0:
        return h.copy()
    ix = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[ix[:tg]] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values, _G
    rng = np.random.RandomState(seed)
    t0 = time.time()

    n0 = 45
    if initial_h_values is not None:
        h0 = _norm(np.array(initial_h_values, dtype=float))
        n0 = len(h0)
    else:
        h0 = np.full(n0, 0.5)
    _G = n0
    bh, bc = h0.copy(), _c5_obj(h0)

    # Phase 1: SLSQP at n0 from diverse starts
    ss = [h0]
    for _ in range(50):
        ss.append(_rh(n0, rng))
    for k in (1, 2, 3, 4, 5):
        for p in (0.0, 1.2, 2.5, 4.0):
            ss.append(_sh(n0, k, p))
    for nb in range(2, 12):
        for ct in range(1, nb):
            ss.append(_bh(n0, nb, ct))
    for pv in (2, 3, 4, 5, 6, 7, 8, 10, 12, 15):
        for q in range(1, pv):
            ss.append(_per(n0, pv, q))

    for s in ss:
        if time.time() - t0 > budget_s - 300:
            break
        r = _slv(s, 400)
        if r.x is not None:
            _G = n0
            ho = _norm(r.x)
            co = _c5_obj(ho)
            if co < bc:
                bc, bh = co, ho.copy()
            if co - bc < 0.02:
                hb = _bin_rnd(r.x)
                _G = n0
                cb = _c5_obj(hb)
                if cb < bc:
                    bc, bh = cb, hb.copy()

    # Phase 2: DE at n0
    if time.time() - t0 < budget_s - 300:
        _G = n0
        dr = differential_evolution(_pen_obj, [(0.0, 1.0)] * n0,
                                    seed=rng.randint(2**31), maxiter=80,
                                    popsize=25, tol=1e-8, polish=False)
        if dr.x is not None:
            ho = _norm(dr.x)
            _G = n0
            co = _c5_obj(ho)
            if co < bc:
                bc, bh = co, ho.copy()
            r = _slv(ho, 500)
            if r.x is not None:
                ho2 = _norm(r.x)
                _G = n0
                co2 = _c5_obj(ho2)
                if co2 < bc:
                    bc, bh = co2, ho2.copy()

    # Phase 3: multi-resolution search
    for n2 in (48, 64, 80, 96, 128, 160, 200):
        if time.time() - t0 > budget_s - 80:
            break
        xo = np.linspace(0.0, 2.0, len(bh), endpoint=False)
        xn = np.linspace(0.0, 2.0, n2, endpoint=False)
        hi = _norm(np.interp(xn, xo, bh))
        rs = [hi]
        for _ in range(12):
            rs.append(_rh(n2, rng))
        for k in (1, 2, 3):
            rs.append(_sh(n2, k, rng.uniform(0.0, 6.0)))
        for pv in (3, 4, 5, 7, 10, 13):
            for q in range(1, pv):
                rs.append(_per(n2, pv, q))

        for s in rs:
            if time.time() - t0 > budget_s - 30:
                break
            r = _slv(s, 300)
            if r.x is not None:
                _G = n2
                ho = _norm(r.x)
                co = _c5_obj(ho)
                if co < bc:
                    bc, bh = co, ho.copy()

        if n2 <= 96 and time.time() - t0 < budget_s - 100:
            _G = n2
            dr2 = differential_evolution(_pen_obj, [(0.0, 1.0)] * n2,
                                         seed=rng.randint(2**31), maxiter=40,
                                         popsize=20, tol=1e-8, polish=False)
            if dr2.x is not None:
                ho = _norm(dr2.x)
                co = _c5_obj(ho)
                if co < bc:
                    bc, bh = co, ho.copy()
                r = _slv(ho, 400)
                if r.x is not None:
                    ho2 = _norm(r.x)
                    _G = n2
                    co2 = _c5_obj(ho2)
                    if co2 < bc:
                        bc, bh = co2, ho2.copy()

    # Phase 4: final polish at best n
    if time.time() - t0 < budget_s - 10:
        r = _slv(bh, 1000)
        if r.x is not None:
            _G = len(bh)
            ho = _norm(r.x)
            co = _c5_obj(ho)
            if co < bc:
                bc, bh = co, ho.copy()

    bh = _norm(bh)
    _G = len(bh)
    bc = _c5_obj(bh)
    return (bh, bc, len(bh))
