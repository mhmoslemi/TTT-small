import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def gradient_at_shift(h, n, shift):
    """Gradient of overlap(shift) w.r.t. h. shift is the actual lag, not correlation index."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        js = j + shift
        jm = j - shift
        if 0 <= js < n:
            g += (1.0 - h[js])
        if 0 <= jm < n:
            g -= h[jm]
        grad[j] = g * dx
    return grad

def optimize_from(h0, n, budget_s, start_time):
    h = h0.copy().astype(float)
    h = project_feasible(h, n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    alpha = 0.1
    plateau = 0
    max_plateau = 50
    
    while time.time() - start_time < budget_s - 5:

        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode="full")
        overlaps = corr * dx
        
        # Find worst shift
        k_max = np.argmax(overlaps)
        shift = k_max - (n - 1)
        
        # Gradient
        grad = gradient_at_shift(h, n, shift)
        
        # Also add small gradient from second worst for robustness
        k_sorted = np.argsort(overlaps)[::-1]

            grad2 = gradient_at_shift(h, n, shift2)
            grad = 0.7 * grad + 0.3 * grad2

import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def gradient_shift(h, n, s):
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        js = j + s
        if 0 <= js < n:
            g += 1.0 - h[js]
        jm = j - s
        if 0 <= jm < n:
            g -= h[jm]
        grad[j] = g * dx
    return grad

def smooth_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ov = corr * dx
    mx = np.max(ov)
    ew = np.exp(beta * (ov - mx))
    sw = np.sum(ew)
    w = ew / sw
    grad = np.zeros(n)
    sig = w > 1e-10
    for m in np.where(sig)[0]:
        s = m - (n - 1)
        grad += w[m] * gradient_shift(h, n, s)
    return grad, mx + np.log(sw) / beta

            lo = mid
        else:

def grad_shift(h, n, s):

    g = np.zeros(n)
    for j in range(n):
        v = 0.0
        a = j + s
        if 0 <= a < n:

        b = j - s
        if 0 <= b < n:
