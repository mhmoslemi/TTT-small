def enforce_constraints(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        h = np.zeros_like(h)
        h[:int(target_sum)] = 1.0
        return h
    h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)

    diff = target_sum - np.sum(h)

        # Spread the correction
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

        diff = target_sum - np.sum(h)

                diff -= adj

                h[idx] -= adj
                diff += adj
            idx += 1

import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def enforce(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = float(h.sum())
    if s < 1e-12:
        h = np.zeros_like(h)
        h[:max(1, int(round(target)))] = 1.0
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    d = target - float(h.sum())
    if abs(d) > 1e-12:
        n = len(h)
        h += d / n
        h = np.clip(h, 0.0, 1.0)
        d = target - float(h.sum())
        for i in range(n):
            if abs(d) < 1e-12:
                break
            if d > 0 and h[i] < 1.0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            elif d < 0 and h[i] > 0.0:
                a = min(-d, h[i]); h[i] -= a; d += a
    return h

def upsample(h, new_n):
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, new_n)
    return enforce(np.interp(x_new, x_old, h), new_n / 2.0)

def sa_optimize(h0, deadline, rng, T0=1.0, T1=0.0005):
    h = h0.astype(float).copy()
    n = len(h)
    target = n / 2.0
    cur = compute_c5(h)
    best = h.copy()
    best_c = cur
    it = 0
    total_est = 200000
    stall = 0
    stall_limit = 8000

    while time.time() < deadline:
        T = T0 * (T1 / T0) ** (it / max(1, total_est))
        it += 1

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        eps = max(0.005, T * rng.uniform(0.05, 1.0))
        hi, hj = h[i], h[j]

        if rng.random() < 0.5:
            d = min(eps, 1.0 - hi, hj)
            if d < 1e-10:
                continue
            h[i] = hi + d
            h[j] = hj - d
        else:
            d = min(eps, 1.0 - hj, hi)
            if d < 1e-10:
                continue
            h[i] = hi - d
            h[j] = hj + d

        new_c = compute_c5(h)

        if new_c <= cur or rng.random() < math.exp(-(new_c - cur) / max(T, 1e-10)):
            cur = new_c
            if cur < best_c:
                best_c = cur
                best = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = hi, hj
            stall += 1

        if stall > stall_limit and time.time() < deadline - 5:
            h = best.copy()
            cur = best_c
            h += rng.normal(0, 0.03, n)
            h = enforce(h, target)
            cur = compute_c5(h)
            stall = 0
            stall_limit = int(stall_limit * 1.5)

    return best, best_c

def random_start(n, rng):
    x = np.linspace(0, 2, n)
    freq = rng.uniform(1, 4)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + rng.uniform(0.2, 0.5) * np.sin(2 * np.pi * freq * x + phase)
    h += rng.normal(0, 0.1, n)
    return enforce(h, n / 2.0)

def block_start(n, rng):
    h = np.zeros(n)
    num_blocks = int(rng.integers(2, n // 10 + 2))
    fill = 0.5
    for b in range(num_blocks):
        start = b * n // num_blocks
        length = max(1, int(n / num_blocks * fill))
        h[start:start + length] = 1.0
    h += rng.normal(0, 0.05, n)
    return enforce(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in dir() or True:
        try:
            h0 = np.array(initial_h_values, dtype=float)
            n0 = len(h0)
            h0 = enforce(h0, n0 / 2.0)
        except NameError:
            n0 = 70
            h0 = random_start(n0, rng)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # Phase 1: SA from initial (25% budget)
    ph1_end = t0 + budget_s * 0.22
    h, c = sa_optimize(h0, ph1_end, rng, T0=1.0, T1=0.001)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 2: Multi-start at various n (25% budget)
    ph2_end = t0 + budget_s * 0.47
    n_options = [len(best_h), 50, 100, 140, 200]
    n_options = sorted(set(n_options))
    per_n = max(1, len(n_options))
    time_per_n = (ph2_end - time.time()) / per_n

    for n_try in n_options:
        if time.time() > ph2_end - 1:
            break
        alloc_end = min(ph2_end, time.time() + max(3, time_per_n))

        h_try = upsample(best_h, n_try)
        h, c = sa_optimize(h_try, alloc_end, rng, T0=0.5, T1=0.0005)
        if c < best_c:
            best_c, best_h = c, h.copy()

        if time.time() > ph2_end - 1:
            break
        # Also try a random start at this n
        h_rand = random_start(n_try, rng)
        h, c = sa_optimize(h_rand, min(ph2_end, time.time() + max(2, time_per_n * 0.4)), rng, T0=1.0, T1=0.001)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Block-structured start (5% budget)
    ph3_end = t0 + budget_s * 0.55
    for _ in range(3):
        if time.time() > ph3_end - 1:
            break
        n_try = len(best_h)
        h_blk = block_start(n_try, rng)
        alloc_end = min(ph3_end, time.time() + 8)
        h, c = sa_optimize(h_blk, alloc_end, rng, T0=0.8, T1=0.001)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: Final refinement on best (remaining ~40% budget)
    if time.time() < deadline - 10:
        refined_deadline = deadline - 5
        # Multiple rounds with decreasing T
        h_cur = best_h.copy()
        for T0_r in [0.15, 0.05, 0.015, 0.005, 0.001]:
            round_end = min(refined_deadline, time.time() + max(3, (refined_deadline - time.time()) * 0.25))
            if time.time() > deadline - 8:
                break
            h_cur, c = sa_optimize(h_cur, round_end, rng, T0=T0_r, T1=T0_r * 0.01)
            if c < best_c:
                best_c, best_h = c, h.copy()
                h_cur = best_h.copy()

    # Final enforce
    best_h = enforce(best_h, len(best_h) / 2.0)
    final_c = compute_c5(best_h)
    best_c = min(best_c, final_c)

    return (best_h, best_c, len(best_h))
