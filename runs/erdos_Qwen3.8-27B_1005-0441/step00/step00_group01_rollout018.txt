import numpy as np
from scipy import optimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max_obj(h, dx, ls):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = (corr - m) * ls
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / ls

def eq_con(h, n_ones):
    return np.sum(h) - n_ones

def do_slsqp(h_start, n, ls, maxiter, best, rng, t0, bs):
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con, 'args': (n / 2.0,)}]
    try:
        res = optimize.minimize(smooth_max_obj, h_start, args=(dx, ls),
                                method='SLSQP', bounds=bnds, constraints=cons,
                                options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = np.clip(res.x, 0, 1)
        s = np.sum(h_opt)
        if s > 1e-10:
            h_opt = np.clip(h_opt * (n / 2.0 / s), 0, 1)
        c5 = compute_c5(h_opt, dx)
        if c5 < best['c5']:
            best['h'], best['c5'], best['n'] = h_opt.copy(), c5, n
    except:
        pass

def local_search(h, n, n_iters, step, best, rng, t0, bs):
    dx = 2.0 / n
    h = h.copy()
    n_ones = n / 2.0
    cur = compute_c5(h, dx)
    for _ in range(n_iters):
        if time.time() - t0 > bs * 0.86:
            break
        i = rng.randint(n)
        d = rng.uniform(-step, step)
        old = h[i]
        h[i] = np.clip(old + d, 0, 1)
        s = np.sum(h)
        if s > 1e-10:
            h = np.clip(h * (n_ones / s), 0, 1)
        c5 = compute_c5(h, dx)
        if c5 > cur + 1e-15:
            h[i] = old
            h = np.clip(h * (n_ones / np.sum(h)), 0, 1)
        else:

            if c5 < best['c5']:

def binary_round(h, n):

    h_b[idx[n_ones:]] = 1.0

    return h_b
