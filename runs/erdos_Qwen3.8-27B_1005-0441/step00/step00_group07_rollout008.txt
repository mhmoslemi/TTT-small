    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if h_p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    c5 = corr[max_idx] * dx
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if 0 <= j - lag < n:
            g += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            g -= h[j + lag]
        grad[j] = g * dx
    return c5, grad

def make_constraints(target_sum):
    def con_fun(x):
        return x.sum() - target_sum
    def con_jac(x):
        return np.ones_like(x)
    return [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(80, 0.5)

    n = len(h0)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = make_constraints(target_sum)

    h0 = project_to_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP from initial
    try:
        res = minimize(obj_and_grad, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints, jac=True,
                       options={'maxiter': 400, 'ftol': 1e-10})
        if res.x is not None:
            hc = project_to_feasible(res.x, target_sum)
            c5c = compute_c5(hc)
            if c5c < best_c5:
                best_h, best_c5 = hc, c5c
    except Exception:
        pass

    # Phase 2: Multi-restart optimization
    noise_levels = [0.02, 0.05, 0.1, 0.15, 0.25, 0.4]
    restart_count = 0

    while True:
        elapsed = time.time() - t0
        time_per_iter = 4.0
        if elapsed + time_per_iter > budget_s * 0.93:
            break
        if restart_count > 500:
            break

        # Choose perturbation
        r = restart_count
        level = noise_levels[max(0, min(len(noise_levels)-1, r // 10))]

        mode = r % 4
        if mode == 0:
            h_start = best_h + rng.randn(n) * level
        elif mode == 1:
            freq = rng.uniform(1.0, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            h_start = best_h + level * np.sin(2*np.pi*freq*np.arange(n)/n + phase)
        elif mode == 2:
            # Block perturbation
            block = max(2, n // 10)
            idx = rng.randint(0, n - block)
            h_start = best_h.copy()
            h_start[idx:idx+block] += rng.choice([-1, 1]) * level * 2
        else:
            # Random binary-ish start
            h_start = rng.rand(n)
            h_start = np.where(h_start > 0.5, 1.0, 0.0)

        h_start = project_to_feasible(h_start, target_sum)

        try:
            res = minimize(obj_and_grad, h_start, method='SLSQP', bounds=bounds,
                           constraints=constraints, jac=True,
                           options={'maxiter': 300, 'ftol': 1e-10})
            if res.x is not None:
                hc = project_to_feasible(res.x, target_sum)
                c5c = compute_c5(hc)
                if c5c < best_c5:
                    best_h, best_c5 = hc.copy(), c5c
        except Exception:
            pass

        restart_count += 1

    # Phase 3: Final polishing with small steps
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.85:
        for _ in range(5):
            if time.time() - t0 > budget_s * 0.95:
                break
            h_start = best_h + rng.randn(n) * 0.01
            h_start = project_to_feasible(h_start, target_sum)
            try:
                res = minimize(obj_and_grad, h_start, method='SLSQP', bounds=bounds,
                               constraints=constraints, jac=True,
                               options={'maxiter': 300, 'ftol': 1e-11})
                if res.x is not None:
                    hc = project_to_feasible(res.x, target_sum)
                    c5c = compute_c5(hc)
                    if c5c < best_c5:
                        best_h, best_c5 = hc.copy(), c5c
            except Exception:
                pass

    best_h = project_to_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
