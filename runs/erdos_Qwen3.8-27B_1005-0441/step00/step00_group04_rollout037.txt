            if c5_new < cur_c5:
                accept = True
            else:
                delta = c5_new - cur_c5
                accept = delta < 20.0 * T and rng.random() < np.exp(-delta / T)

            if accept:
                cur_h, cur_c5 = h_new, c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

            T = max(T * alpha, T_min)

            if stall > 4000:
                stall = 0
                T = float(rng.uniform(0.001, 0.005))
                cur_h = best_h.copy()
                n_r = int(rng.integers(3, 15))
                for _ in range(n_r):
                    cur_h = do_transfer(cur_h, rng, n, 1)
                cur_c5 = compute_c5(cur_h)

            # Periodic local refine
            if stall > 20000 and time.time() - t0 > 60:
                best_h, best_c5 = local_refine(best_h, rng, n, 300)
                cur_h = best_h.copy()
                cur_c5 = best_c5
                stall = 0
                T = 0.001

        else:
            # Local refine phase near end
            best_h, best_c5 = local_refine(best_h, rng, n, 200)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
