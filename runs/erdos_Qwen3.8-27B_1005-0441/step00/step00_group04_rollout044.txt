import numpy as np
import time

def compute_overlap(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def compute_c5(h, n):
    return float(np.max(compute_overlap(h, n)))

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0).copy()
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            m = hp < 1.0
        else:
            m = hp > 0.0
        if m.any():
            hp[m] += diff / m.sum()
    return np.clip(hp, 0.0, 1.0)

def grad_one(h, n, ki):
    lag = ki - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        for j in range(n - lag):
            g[j + lag] += 1.0 - h[j]
            g[j] -= h[j + lag]
    else:
        for j in range(-lag, n):
            g[j] += 1.0 - h[j - lag]
            g[j - lag] -= h[j]
    return g
