                    continue
                h_try = project(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-10:
                    h_cur = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h_cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Get initial construction
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(80, 0.5)
    n0 = len(h0)
    h0 = project(h0, n0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Build task list: (n, start_h)
    tasks = []
    resolutions = sorted(set([n0, 80, 100, 120, 160, 200, 320]))
    for n in resolutions:
        if n > 500:
            continue
        # Interpolated from best
        xo = np.arange(len(best_h)) * (2.0 / len(best_h))
        xn = np.arange(n) * (2.0 / n)
        hi = project(np.interp(xn, xo, best_h), n)
        tasks.append((n, hi))
        # Random starts
        for _ in range(3):
            hr = np.random.uniform(0.3, 0.7, n)
            tasks.append((n, project(hr, n)))
        # Symmetric random
        half_n = n // 2
        hs = np.random.uniform(0.3, 0.7, half_n + 1)
        if n % 2 == 0:
            hs = np.concatenate([hs[:half_n], hs[:half_n][::-1]])
        else:
            hs = np.concatenate([hs[1:], hs])
        tasks.append((n, project(hs, n)))
        # Triangular: high in middle
        tri = 1.0 - 2.0 * np.abs(np.arange(n) - n / 2.0) / n
        tri = 0.2 + 0.6 * tri / tri.max()
        tasks.append((n, project(tri, n)))

    lam = 100.0
    temps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    for (n, h_start) in tasks:
        if elapsed() > budget_s * 0.88:
            break
        c0 = compute_c5(h_start)
        if c0 < best_c5:
            best_c5, best_h = c0, h_start.copy()

        for t in temps:
            if elapsed() > budget_s * 0.90:
                break
            remaining = budget_s * 0.93 - elapsed()
            if remaining < 3:
                break
            time_alloc = min(remaining * 0.2, 45.0)
            max_iter = max(80, int(time_alloc * 3))
            h_opt = optimize_lg(h_start, n, t, lam, max_iter)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()

        # Local search on best at this resolution
        if elapsed() < budget_s * 0.90:
            h_ls, c5_ls = local_search(best_h if len(best_h) == n else h_start, n, steps=100)
            if c5_ls < best_c5:
                best_c5, best_h = c5_ls, h_ls

    # Final polish: gradient step on best solution
    if elapsed() < budget_s * 0.95:
        n_best = len(best_h)
        for t in [0.001, 0.0005, 0.0002]:
            if elapsed() > budget_s * 0.96:
                break
            h_pol = optimize_lg(best_h, n_best, t, 200.0, 200)
            c_pol = compute_c5(h_pol)
            if c_pol < best_c5:
                best_c5, best_h = c_pol, h_pol

    # Final local search
    if elapsed() < budget_s * 0.97:
        h_ls, c5_ls = local_search(best_h, len(best_h), steps=300)
        if c5_ls < best_c5:
            best_c5, best_h = c5_ls, h_ls

    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
