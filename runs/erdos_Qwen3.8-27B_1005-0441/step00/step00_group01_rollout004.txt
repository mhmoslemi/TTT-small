import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def grad_max_overlap(h, top_k=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    max_val = float(np.max(corr))
    k = min(top_k, len(corr))
    top_indices = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for k_idx in top_indices:
        lag = int(k_idx) - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            m = n - lag
            if m > 0:
                g[0:m] += (1.0 - h[lag:lag + m])
                g[lag:lag + m] -= h[0:m]
        else:
            ll = -lag
            m = n - ll
            if m > 0:
                g[ll:ll + m] += (1.0 - h[0:m])
                g[0:m] -= h[ll:ll + m]
        grad += dx * g
    grad /= float(k)
    return grad, max_val

def gradient_descent(h0, target_sum, max_iters=5000, deadline=None, top_k=3):
    h = h0.copy().astype(float)
    c5 = compute_c5(h)
    step = 0.03
    best_c5 = c5
    best_h = h.copy()
    stalled = 0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        grad, _ = grad_max_overlap(h, top_k=top_k)
        accepted = False
        for ls in range(50):
            h_new = h - step * grad
            h_new = project_to_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new <= c5 + 1e-14:
                accepted = True
                break
            step *= 0.5
        if accepted:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            step = min(step * 1.08, 0.3)
            stalled = 0
        else:
            stalled += 1
            step *= 0.5
            if step < 1e-9:
                break
        if stalled > 800:
            break
    return best_h, best_c5

def coordinate_refine(h, target_sum, rounds=30, deadline=None):
    n = len(h)
    c5 = compute_c5(h)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]
    for rnd in range(rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            old = h[i]
            for d in deltas:
                h[i] = old + d
                h_ptr = project_to_feasible(h, target_sum)
                c5_try = compute_c5(h_ptr)
                if c5_try < c5 - 1e-12:
                    h = h_ptr.copy()
                    c5 = c5_try
                    improved = True
                    break
            h[i] = old if not improved else h[i]
            if improved:
                old = h[i]
        if not improved:
            break
    return h, c5

def try_binary_round(h, target_sum, deadline=None):
    n = len(h)
    c5 = compute_c5(h)
    order = np.argsort(h)[::-1]
    nb = int(round(target_sum))
    hb = np.zeros(n)
    for idx in order[:nb]:
        hb[idx] = 1.0
    hb = project_to_feasible(hb, target_sum)
    c5b = compute_c5(hb)
    if c5b < c5:
        return hb, c5b
    return h, c5
