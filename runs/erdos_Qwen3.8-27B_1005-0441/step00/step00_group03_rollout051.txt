import numpy as np
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for i in order:
        if diff > 0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        else:
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
        if abs(diff) < 1e-15:
            break
    return h
