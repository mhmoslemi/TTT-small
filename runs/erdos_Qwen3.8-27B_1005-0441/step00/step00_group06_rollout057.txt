def solve_lp(n, time_limit=60):
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2.0]
    
    for k in range(-(n-1), n):
        i_min = max(0, -k)
        i_max = min(n-1, n-1-k)
        n_terms = i_max - i_min + 1
        y = cp.Variable(n_terms)
        sum_hi = 0
        sum_yi = 0
        for idx in range(n_terms):
            i = i_min + idx
            j = i + k
            sum_hi += h[i]
            constraints.append(y[idx] <= h[i])
            constraints.append(y[idx] <= h[j])
            constraints.append(y[idx] >= h[i] + h[j] - 1)
