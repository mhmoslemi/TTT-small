    dx = 2.0/n

    # Smoothed max
    max_corr = np.max(corr)
    shifted = (corr - max_corr) / T  # for numerical stability
    weights = np.exp(shifted)
    weights /= weights.sum()
    obj = max_corr + T * np.log(weights.sum())

import time

def compute_all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    cmax = corr.max()

import numpy as np
import time

def compute_all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode="full") * dx

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smoothed_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    cmax = corr.max()
    shifted = (corr - cmax) / T
    exp_s = np.exp(shifted)
    wsum = exp_s.sum()
    weights = exp_s / wsum
    obj = cmax + T * np.log(wsum)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:

            grad[:n - s] -= h[s:] * wd
        else:
            grad[:n + s] += (1.0 - h[-s:]) * wd
            grad[-s:] -= h[:n + s] * wd
    return obj, grad
