import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Binary search for additive offset
    lo, hi = -1.0, 1.0

        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:

        else:
