import numpy as np
import time

def compute_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def compute_correlations(h):
    """Compute all correlation values"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def subgradient(h, m_star):
    """Compute subgradient of c[m_star] w.r.t. h"""
    n = len(h)
    offset = m_star - n + 1
    grad = np.zeros(n)
    for j in range(n):
        # Term 1: (1 - h[j + offset]) if j + offset in [0, n)
        val1 = 0.0
        idx1 = j + offset
        if 0 <= idx1 < n:
            val1 = 1.0 - h[idx1]
        # Term 2: -h[j - offset] if j - offset in [0, n)
        val2 = 0.0
        idx2 = j - offset
        if 0 <= idx2 < n:
            val2 = -h[idx2]
        grad[j] = val1 + val2
    return grad

def optimize_h(h_init, max_time):
    """Optimize h to minimize C5"""
    h = h_init.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    
    # Ensure constraint
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2.0) / current_sum
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    alpha = 0.01
    no_improve = 0
    
    while time.time() - start_time < max_time:
        correlations = compute_correlations(h)
        # Find top few shifts
        top_k = min(5, len(correlations))
        top_indices = np.argsort(correlations)[-top_k:]
        m_star = top_indices[-1]  # The maximum
        
        # Compute weighted subgradient (weight by how close to max)
        thresh = correlations[m_star] - 0.01
        weights = np.zeros(len(correlations))
        for m in top_indices:
            w = max(0, correlations[m] - thresh + 0.01)
            weights[m] = w + 0.1
        
        grad = np.zeros(n)
        for m_idx in range(len(top_indices)):
            m = top_indices[m_idx]
            if weights[m] > 0:
                offset = m - n + 1
                for j in range(n):
                    idx1 = j + offset
                    if 0 <= idx1 < n:
                        grad[j] += weights[m] * (1.0 - h[idx1])
                    idx2 = j - offset
                    if 0 <= idx2 < n:
                        grad[j] -= weights[m] * h[idx2]
        
        # Center gradient to maintain sum constraint
        grad -= np.mean(grad)
        
        # Update
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        s = np.sum(h_new)
        if s > 0:
            h_new = h_new * (n / 2.0) / s
            h_new = np.clip(h_new, 0, 1)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
            h = h_new
        else:
            h = h_new  # Still update, might converge
            no_improve += 1
        
        if no_improve > 200:
            alpha *= 0.7
            no_improve = 0
        
        if alpha < 1e-8:
            break
    
    return best_h, best_c5

import time

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def vec_subgrad_topk(h, c_vals, top_k):
    n = len(h)
    idx = np.argsort(c_vals)[-top_k:]
    grad = np.zeros(n)
    for m in idx:
        offset = m - n + 1
        if offset >= 0:
            grad[:n-offset] += 1.0 - h[offset:]
            grad[offset:] -= h[:n-offset]
        else:
            o = -offset
            grad[o:] += 1.0 - h[:n-o]
            grad[:n-o] -= h[o:]
    return grad

def renormalize(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

        grad = vec_subgrad_topk(h, c_vals, top_k)
        grad -= np.mean(grad)

        if c1 < best_c5:

    # Phase 2: perturbations of best (20%)
    t2 = budget_s * 0.65
    rem = t2 - (time.time() - t0)
    rng = np.random.RandomState(seed + 7)
    if rem > 20:
        for attempt in range(5):
            if rem < 15:
                break
            noise_scale = 0.02 + 0.01 * attempt
            h_alt = best_h + rng.normal(0, noise_scale, len(best_h))
            h_alt = np.clip(h_alt, 0, 1)
            h_alt = renormalize(h_alt, len(best_h) / 2.0)
            t_sub = min(rem * 0.35, 70)
            h_a, c_a = optimize_subgrad(h_alt, t_sub)
            if c_a < best_c5:
                best_h, best_c5 = h_a, c_a
            rem = t2 - (time.time() - t0)

    # Phase 3: different n values (15%)
    t3 = budget_s * 0.82
    rem = t3 - (time.time() - t0)
    if rem > 25:
        for n_alt in [100, 140]:
            if rem < 20:
                break
            h_alt = make_initial(n_alt, seed + 200 + n_alt)
            t_sub = min(rem * 0.45, 90)
            h_a, c_a = optimize_subgrad(h_alt, t_sub)
            if c_a < best_c5:
                best_h, best_c5 = h_a, c_a
            rem = t3 - (time.time() - t0)

    # Phase 4: final refinement (13%)
    rem = budget_s * 0.95 - (time.time() - t0)
    if rem > 5:
        h_f, c_f = optimize_subgrad(best_h, rem)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f

    best_c5 = compute_c5_fast(best_h)
    return best_h, best_c5, len(best_h)
