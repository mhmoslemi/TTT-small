            ss = max(ss * 0.98, 0.005)
    return best_h, best_c5


def zero_one_refine(h, time_end):
    n = len(h)
    h_zo = np.round(h)
    h_zo = project_normalize(h_zo, n)
    best_h = h_zo.copy()
    best_c5 = compute_c5(best_h)
    ones = np.where(h_zo > 0.5)[0].tolist()
    zeros = np.where(h_zo < 0.5)[0].tolist()
    for _ in range(200000):
        if time.time() > time_end:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[int(np.random.randint(len(ones)))]
        j = zeros[int(np.random.randint(len(zeros)))]
        h_try = best_h.copy()
        h_try[i] = 0.0
        h_try[j] = 1.0
        h_try = project_normalize(h_try, n)
        c5 = compute_c5(h_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try
            ones = np.where(h_try > 0.5)[0].tolist()
            zeros = np.where(h_try < 0.5)[0].tolist()
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None or len(h0) == 0:
        n0 = 100
        h0 = np.zeros(n0)
        h0[::2] = 1.0
        h0 = project_normalize(h0, n0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: PGD on initial resolution
    t1 = min(t0 + budget_s * 0.30, t0 + budget_s - 60)
    h1, c1 = pgd_optimize(best_h, 0.08, 3, t1, temp=0.03)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: PGD with tighter focus
    t2 = min(t0 + budget_s * 0.45, t0 + budget_s - 40)
    h2, c2 = pgd_optimize(best_h, 0.03, 5, t2, temp=0.02)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2

    # Phase 3: Zero-one refinement at current n
    t3 = min(t0 + budget_s * 0.55, t0 + budget_s - 30)
    h3, c3 = zero_one_refine(best_h, t3)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3

    # Phase 4: Interpolate to higher resolution and optimize
    for target_n in [140, 200]:
        if time.time() > t0 + budget_s * 0.82:
            break
        old_n = len(best_h)
        if old_n >= target_n:
            continue
        old_x = np.linspace(0, 2, old_n, endpoint=False)
        new_x = np.linspace(0, 2, target_n, endpoint=False)
        h_interp = np.interp(new_x, old_x, best_h)
        h_interp = project_normalize(h_interp, target_n)
        t_exp = min(t0 + budget_s * 0.92, t0 + budget_s - 20)
        h4, c4 = pgd_optimize(h_interp, 0.04, 3, t_exp, temp=0.03)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4

    # Phase 5: Final local search
    t5 = t0 + budget_s * 0.97
    h5, c5 = local_search(best_h, t5, ss_init=0.03)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h5

    best_h = project_normalize(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
