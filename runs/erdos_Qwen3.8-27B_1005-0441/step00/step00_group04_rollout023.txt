def smooth_max_and_grad(h, beta):

    n = len(h)
    max_o = np.max(overlaps)
    shifted = beta * (overlaps - max_o)
    exps = np.exp(shifted)
    total = np.sum(exps)
    weights = exps / total
    obj = max_o + np.log(total) / beta
    
    grad = np.zeros(n)
    # Lag range: -(n-1) to (n-1), index in overlaps: 0 to 2n-2
    # Index m corresponds to lag = m - (n-1)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        lag = m_idx - (n - 1)
        if lag >= 0:
            k = lag
            if k == 0:
                grad[:] += w * (1.0 - 2.0 * h)
            else:
                mm = n - k
                grad[k:] += w * (1.0 - h[:mm])
                grad[:mm] -= w * h[k:]
        else:
            k = -lag
            mm = n - k
            grad[:mm] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:mm]
    return obj, grad
