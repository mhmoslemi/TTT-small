import numpy as np
from scipy.optimize import minimize
import time

_g = {'temp': 0.01, 'dx': 0.04, 'n': 47}

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def project_constrained(x, target_sum):
    n = len(x)
    for _ in range(n + 2):
        x = np.clip(x, 0.0, 1.0)
        diff = target_sum - np.sum(x)
        if abs(diff) < 1e-14:
            return np.clip(x, 0.0, 1.0)
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            return np.clip(x, 0.0, 1.0)
        x[mask] += diff / cnt
    return np.clip(x, 0.0, 1.0)

def _sobj(x):
    t, dx, n = _g['temp'], _g['dx'], _g['n']
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = np.max(c)
    return (cm + t * np.log(np.sum(np.exp((c - cm) / t)))) * dx

def _sjac(x):
    t, dx, n = _g['temp'], _g['dx'], _g['n']
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = np.max(c)
    w = np.exp((c - cm) / t)
    w /= np.sum(w)
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        p = shifts[m]
        dp = np.zeros(n)
        if p >= 0:
            dp[p:] += (1.0 - x[:n - p])
            if n - p > 0:
                dp[:n - p] -= x[p:]
        else:
            pp = -p
            dp[:n - pp] += (1.0 - x[pp:])
            if n - pp > 0:
                dp[pp:] -= x[:n - pp]
        grad += w[m] * dp
    return grad * dx

def _eq_fun(x):
    return x.sum() - _g['n'] / 2.0

def _eq_jac(x):
    return np.ones(_g['n'])

def do_optimize(h_init, n, t0, budget_s, max_temps=7):
    dx = 2.0 / n
    _g['n'] = n
    _g['dx'] = dx
    h = h_init.copy()
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    if max_temps < len(temps):
        temps = temps[:max_temps]
    bounds = [(0.0, 1.0)] * n
    constr = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    for temp in temps:
        if time.time() - t0 > budget_s - 5:
            break
        _g['temp'] = temp
        res = minimize(_sobj, h, jac=_sjac, method='SLSQP',
                      bounds=bounds, constraints=constr,
                      options={'maxiter': 150, 'ftol': 1e-10})
        h = project_constrained(res.x, n / 2.0)
    c = compute_c5(h, dx)
    return h, c
