        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_grad_smooth(h, dx, smooth_t):
    """Compute smoothed gradient of max_k corr_k * dx"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    w = np.exp((corr - cmax) / smooth_t)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    ones_h = 1.0 - h
    
    for k_idx in range(len(corr)):
        wi = w[k_idx]
        if wi < 1e-7:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            # corr = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
            if k == 0:
                grad += wi * (ones_h - h)
            else:
                m = n - 1 - k
                grad[k:n] += wi * ones_h[0:m+1]
                grad[0:m+1] -= wi * h[k:n]
        else:
            s = -k
            # corr = sum_{n=s}^{n-1} h[n-s]*(1-h[n])
            m = n - s
            grad[0:m] += wi * ones_h[s:n]
            grad[s:n] -= wi * h[0:m]
    
    return grad * dx

def optimize_h(h0, dx, target_sum, max_iter, lr0, t0, time_limit):
    """Gradient descent with softmax smoothing and annealing"""
    h = project_h(np.copy(h0), target_sum)
    best_h = np.copy(h)
    best_c5 = compute_c5(h, dx)
    lr = lr0
    smooth_t = t0
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        grad = compute_grad_smooth(h, dx, smooth_t)
        h_new = h - lr * grad
        h_new = project_h(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = np.copy(h_new)
            h = h_new
            no_improve = 0
        else:
            h = h_new  # still follow gradient for exploration
            no_improve += 1
        
        if no_improve == 100:
            lr *= 0.6
            no_improve = 0
        elif no_improve == 250:
            lr *= 0.5
            smooth_t = max(0.005, smooth_t * 0.6)
            no_improve = 0
            h = np.copy(best_h)
        elif no_improve >= 500:
            break
        
        if it % 200 == 199 and smooth_t > 0.01:
            smooth_t = max(0.005, smooth_t * 0.85)
    
    return best_h, best_c5

def coordinate_local_search(h, dx, target_sum, time_limit, max_rounds=200):
    """Coordinate descent to polish the solution"""
    n = len(h)
    best_h = np.copy(h)
    best_c5 = compute_c5(h, dx)
    step = 0.01
    
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        for j in range(n):
            for direction in [1, -1]:
                h_trial = np.copy(best_h)
                h_trial[j] += direction * step
                h_trial = project_h(h_trial, target_sum)
                c5_trial = compute_c5(h_trial, dx)
                if c5_trial < best_c5 - 1e-10:
                    best_h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
        if not improved:
            if step < 1e-4:
                break
            step *= 0.7
        # Also try larger random perturbations occasionally
        if rnd % 20 == 0:
            idx = np.random.randint(n)
            h_trial = np.copy(best_h)
            h_trial[idx] += np.random.choice([-1, 1]) * np.random.uniform(0.05, 0.2)
            h_trial = project_h(h_trial, target_sum)
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    time_limit = start_time + budget_s - 15
    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        n = 95
        h_init = np.full(n, 0.5)
    
    dx = 2.0 / n
    target_sum = n / 2.0
    
    best_h = project_h(np.copy(h_init), target_sum)
    best_c5 = compute_c5(best_h, dx)
    
    # Phase 1: Gradient descent from initial construction
    h1, c5_1 = optimize_h(best_h, dx, target_sum,
                          max_iter=3000, lr0=0.003, t0=0.1,
                          time_limit=time_limit)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    
    # Phase 2: Coordinate local search to polish
    if time.time() < time_limit - 60:
        h2, c5_2 = coordinate_local_search(best_h, dx, target_sum,
                                           time_limit=min(time_limit, start_time + budget_s - 40),
                                           max_rounds=100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
    
    # Phase 3: Random restarts if time permits
    if time.time() < time_limit - 120:
        for restart in range(20):
            if time.time() > time_limit:
                break
            # Generate diverse random starts
            h_r = rng.uniform(0, 1, n)
            # Add some structure: half-ones
            if rng.random() < 0.5:
                half = n // 2
                idx = rng.permutation(n)
                h_r = np.zeros(n)
                h_r[idx[:half]] = 1.0
                # Add noise
                h_r += rng.normal(0, 0.1, n)
            h_r = project_h(h_r, target_sum)
            
            # Short gradient optimization
            h_r_opt, c5_r = optimize_h(h_r, dx, target_sum,
                                       max_iter=800, lr0=0.005, t0=0.15,
                                       time_limit=min(time_limit, time_limit - 30))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r_opt
                # Polish
                if time.time() < time_limit - 60:
                    hp, cp = coordinate_local_search(best_h, dx, target_sum,
                                                     time_limit=time_limit, max_rounds=50)
                    if cp < best_c5:
                        best_c5 = cp
                        best_h = hp
    
    # Final polish
    if time.time() < time_limit:
        h_final, c5_final = coordinate_local_search(best_h, dx, target_sum,
                                                    time_limit=time_limit, max_rounds=50)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Ensure exact constraint satisfaction
    best_h = project_h(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    
    return best_h, final_c5, n
