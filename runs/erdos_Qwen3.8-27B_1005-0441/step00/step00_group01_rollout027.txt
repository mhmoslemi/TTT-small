                break

        else:
            m = h > 1e-15
            if not m.any():
                break
            h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _sobj(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    return float(m + np.log(np.sum(np.exp(c - m))) / beta)

def _sobj_wrap(h, dx, beta):
    return _sobj(h, dx, beta)

def _sobj_grad(h, dx, beta):
    eps = 1e-7
    g = np.zeros(_G['n'])
    for i in range(_G['n']):
        h[i] += eps
        f1 = _sobj(h, dx, beta)
        h[i] -= 2.0 * eps
        f2 = _sobj(h, dx, beta)
        h[i] += eps
        g[i] = (f1 - f2) / (2.0 * eps)
    return g

def _seqlen(h):
    return np.sum(h) - _G['ts']

def _seqjac(h):
    return np.ones(_G['n'])

def _coord_sweep(h, dx, ts, time_limit, max_sweeps=300):
    n = _G['n']
    t_end = time.time() + time_limit
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                break
            j = (i + 1) % n
            c = ts - (np.sum(h) - h[i] - h[j])
            lo = max(0.0, c - 1.0)
            hi = min(1.0, c)
            if lo >= hi:
                continue
            bv = _c5(h, dx)
            old_i, old_j = h[i], h[j]
            best_t = old_i
            for t in np.linspace(lo, hi, 20):
                h[i] = t
                h[j] = c - t
                if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                    v = _c5(h, dx)
                    if v < bv - 1e-14:
                        bv = v
                        best_t = t
                        imp = True
            h[i], h[j] = best_t, c - best_t
        if not imp:
            break
    return h

def _interp_h(h_src, n_src, n_tgt):
    if n_src == n_tgt:
        return h_src.copy()
    idx = (np.arange(n_tgt) * n_src) / n_tgt
    h_new = np.zeros(n_tgt)
    for i in range(n_tgt):
        lo = int(idx[i])
        hi = min(lo + 1, n_src - 1)
        frac = idx[i] - lo
        h_new[i] = (1.0 - frac) * h_src[lo] + frac * h_src[hi]
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    global _G
    t0 = time.time()
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    ts0 = n0 / 2.0
    _G['n'] = n0
    _G['ts'] = ts0
    _G['dx'] = dx0

    h_best = _project(h0, ts0)
    c5_best = _c5(h_best, dx0)
    n_best = n0

    bounds = [(0.0, 1.0)] * n0
    constr = [{'type': 'eq', 'fun': _seqlen, 'jac': _seqjac}]

    def _elapsed():
        return time.time() - t0

    def _ok(frac):
        return _elapsed() < budget_s * frac

    # Phase 1: Continuation SLSQP on original n
    for beta in [5, 15, 40, 100, 250, 600, 1500]:
        if not _ok(0.25):
            break
        res = minimize(_sobj_wrap, h_best, jac=_sobj_grad,
                      args=(dx0, beta), method='SLSQP',
                      bounds=bounds, constraints=constr,
                      options={'maxiter': 1200, 'ftol': 1e-14})
        h_new = _project(res.x, ts0)
        c5_new = _c5(h_new, dx0)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    # Phase 2: Coordinate descent polish
    if _ok(0.35):
        h_best = _coord_sweep(h_best, dx0, ts0, budget_s * 0.12)
        c5_best = _c5(h_best, dx0)

    # Phase 3: Multi-start on original n
    if _ok(0.45):
        remaining = budget_s * 0.65 - _elapsed()
        nstarts = max(5, int(remaining / 4.0))
        for i in range(nstarts):
            if not _ok(0.65):
                break
            scale = 0.1 * (1.0 + 0.3 * (i % 5))
            h_s = _project(h_best + scale * np.random.randn(n0), ts0)
            res = minimize(_sobj_wrap, h_s, args=(dx0, 100), method='SLSQP',
                         bounds=bounds, constraints=constr,
                         options={'maxiter': 600, 'ftol': 1e-14})
            h_new = _project(res.x, ts0)
            h_cd = _coord_sweep(h_new, dx0, ts0, 6.0, max_sweeps=100)
            c5_cd = _c5(h_cd, dx0)
            if c5_cd < c5_best:
                h_best, c5_best = h_cd, c5_cd

    # Phase 4: Try nearby n values (48, 49, 50, 51)
    if _ok(0.7):
        for n_try in [n0 + 1, n0 + 2, n0 - 1, n0 + 3]:
            if not _ok(0.9) or n_try < 10 or n_try > 100:
                continue
            n_t = n_try
            dx_t = 2.0 / n_t
            ts_t = n_t / 2.0
            _G['n'] = n_t
            _G['ts'] = ts_t
            _G['dx'] = dx_t
            h_interp = _interp_h(h_best, n_best, n_t)
            h_try = _project(h_interp, ts_t)
            bounds_t = [(0.0, 1.0)] * n_t
            constr_t = [{'type': 'eq', 'fun': _seqlen, 'jac': _seqjac}]
            c5_try = _c5(h_try, dx_t)
            best_t = h_try
            for beta in [40, 200]:
                if not _ok(0.9):
                    break
                res = minimize(_sobj_wrap, best_t, args=(dx_t, beta), method='SLSQP',
                             bounds=bounds_t, constraints=constr_t,
                             options={'maxiter': 800, 'ftol': 1e-14})
                h_new = _project(res.x, ts_t)
                h_new = _coord_sweep(h_new, dx_t, ts_t, 8.0, max_sweeps=80)
                c5_new = _c5(h_new, dx_t)
                if c5_new < c5_try:
                    best_t, c5_try = h_new, c5_new
            if c5_try < c5_best:
                h_best, c5_best, n_best = best_t, c5_try, n_t
            # Restore globals for original n
            _G['n'] = n0
            _G['ts'] = ts0
            _G['dx'] = dx0

    # Phase 5: Final coordinate descent
    if _ok(0.97):
        _G['n'] = n_best
        _G['ts'] = n_best / 2.0
        _G['dx'] = 2.0 / n_best
        h_best = _coord_sweep(h_best, 2.0 / n_best, n_best / 2.0,
                             max(1.0, budget_s - _elapsed() - 3), max_sweeps=500)
        c5_best = _c5(h_best, 2.0 / n_best)

    h_best = _project(h_best, n_best / 2.0)
    c5_best = _c5(h_best, 2.0 / n_best)
    return h_best, c5_best, n_best
