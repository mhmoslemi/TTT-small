    comp = 1 - h
    corr = np.correlate(h, comp, mode="full")
    return np.max(corr) * dx

def compute_all_overlaps(h, dx):

    corr = np.correlate(h, comp, mode="full")

def project_h(h, n):
    h = np.clip(h, 0, 1)
    # Adjust sum to n/2
    diff = n/2 - np.sum(h)
    if diff > 0:
        # Add to elements below 1
        room = 1 - h
        if room.sum() > 0:
            h += diff * room / room.sum()
        else:
            h += diff / n
    else:
        # Subtract from elements above 0
        space = h
        if space.sum() > 0:
            h += diff * space / space.sum()
        else:
            h += diff / n
    return np.clip(h, 0, 1)

import time

    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return corr * dx

    return np.max(compute_all_overlaps(h, dx))

def smooth_max(overlaps, T):
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    return max_val + np.log(np.sum(np.exp(T * shifted))) / T

    ovl = compute_all_overlaps(xc, dx)
    m = ovl.max()
    w = np.exp(T * (ovl - m))
    w /= w.sum()
    return grad_weighted_overlaps(xc, dx, w)

def projected_gradient_opt(h, dx, n, iters, T, t0, budget_s, rng):
    for it in range(iters):
        if time.time() - t0 > budget_s - 30:
            break
        h = np.clip(h, 0, 1)
        ovl = compute_all_overlaps(h, dx)
        c5_cur = ovl.max()
        w = np.exp(T * (ovl - c5_cur))
        w /= w.sum()
        g = -grad_weighted_overlaps(h, dx, w)
        lr = 2.0 / (30.0 + 0.3 * it)
        c5_old = compute_c5(h, dx)
        improved = False
        for _ in range(20):
            h_new = project_feasible(h + lr * g, n)
            if compute_c5(h_new, dx) < c5_old + 1e-14:
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            h = project_feasible(h + 0.003 * rng.randn(n), n)
    return h

def upsample_h(h_small, n_large):
    n_small = len(h_small)
    idx = np.linspace(0, n_small - 1, n_large)
    lo = np.floor(idx).astype(int)
    hi = np.minimum(lo + 1, n_small - 1)
    frac = idx - lo
    return h_small[lo] * (1 - frac) + h_small[hi] * frac

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    h_initial = None
    try:
        h_initial = np.array(initial_h_values, dtype=float)
    except:
        pass

    configs = []
    if h_initial is not None and len(h_initial) == 45:
        configs.append((45, h_initial.copy()))

    n_list = [45, 55, 60, 75, 80, 100, 120, 150]
    for n in n_list:
        if time.time() - t0 > budget_s - 90:
            break
        h_sm = np.full(n, 0.5) + 0.12 * rng.randn(n)
        configs.append((n, h_sm))
        h_rnd = rng.uniform(0, 1, n)
        configs.append((n, h_rnd))
        if n % 2 == 0:
            h_alt = np.zeros(n)
            h_alt[0:n//2] = 1.0
            configs.append((n, h_alt + 0.05 * rng.randn(n)))

    for n, h0 in configs:
        if time.time() - t0 > budget_s - 60:
            break
        dx = 2.0 / n
        h0 = project_feasible(h0, n)
        c5 = compute_c5(h0, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()

        iters = 400 if n <= 60 else 250 if n <= 100 else 150
        h_opt = projected_gradient_opt(h0, dx, n, iters, 40.0, t0, budget_s, rng)
        c5 = compute_c5(h_opt, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

        if n % 2 == 0 and time.time() - t0 < budget_s - 60:
            h_bin = np.zeros(n)
            idx_top = np.argsort(-h_opt)[:n // 2]
            h_bin[idx_top] = 1.0
            c5b = compute_c5(h_bin, dx)
            if c5b < best_c5:
                best_c5 = c5b
                best_h = h_bin.copy()

    # Upsample best to larger n and re-optimize
    if best_h is not None and time.time() - t0 < budget_s - 90:
        n_cur = len(best_h)
        for n_up in [100, 150, 200, 300]:
            if n_up <= n_cur or time.time() - t0 > budget_s - 60:
                continue
            h_up = upsample_h(best_h, n_up)
            h_up = project_feasible(h_up, n_up)
            dx_up = 2.0 / n_up
            c5_up = compute_c5(h_up, dx_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
            iters = 150 if n_up <= 150 else 100
            h_up = projected_gradient_opt(h_up, dx_up, n_up, iters, 50.0, t0, budget_s, rng)
            c5_up = compute_c5(h_up, dx_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()

    # Scipy L-BFGS-B polish
    if best_h is not None:
        n = len(best_h)
        dx = 2.0 / n
        for T in [40.0, 80.0, 150.0, 300.0]:
            if time.time() - t0 > budget_s - 25:
                break
            try:
                res = minimize(scipy_smooth_obj, best_h.copy(), args=(dx, T),
                               jac=scipy_smooth_grad,
                               method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 800, 'gtol': 1e-12})
                h_res = project_feasible(res.x, n)
                c5r = compute_c5(h_res, dx)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = h_res.copy()
            except:
                pass

    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / n)

    return (best_h, best_c5, len(best_h))
