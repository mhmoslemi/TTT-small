import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def project_h(h, target_sum):
    n = len(h)
    h = np.asarray(h, dtype=float)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)


def smooth_obj(x, dx, t):
    omx = 1.0 - x
    corr = np.correlate(x, omx, mode="full") * dx
    return logsumexp(t * corr) / t


def smooth_grad(x, dx, t):
    omx = 1.0 - x
    corr = np.correlate(x, omx, mode="full") * dx
    ex = np.exp(t * corr - logsumexp(t * corr))
    n = len(x)
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        w = ex[k] * dx
        if w < 1e-14:
            continue
        L = k - (n - 1)
        jp = idx + L
        jn = idx - L
        mask_p = (jp >= 0) & (jp < n)
        mask_n = (jn >= 0) & (jn < n)
        contrib = np.zeros(n)
        contrib[mask_n] += 1.0 - x[jn[mask_n]]
        contrib[mask_p] -= x[jp[mask_p]]
        grad += w * contrib
    return grad


def obj_func(x, dx, t):
    return smooth_obj(x, dx, t)


def grad_func(x, dx, t):
    return smooth_grad(x, dx, t)


def optimize_from_h(h, dx, t, target_sum, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj_func, h.copy(), method='L-BFGS-B',
        jac=grad_func, args=(dx, t),
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9}
    )
    h_new = project_h(res.x, target_sum)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    # Get initial construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(47, 0.5)

    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    # Ensure initial feasibility
    h = project_h(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Anneal temperature from low to high
    temps = [10, 25, 50, 100, 200, 400, 800]
    for t_val in temps:
        el = time.time() - t_start
        if el > budget_s - 60:
            break
        h_cur = optimize_from_h(best_h, dx, t_val, target_sum, 300)
        c5_cur = compute_c5(h_cur, dx)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Phase 2: Random restarts with perturbations
    el = time.time() - t_start
    n_restarts = min(30, int((budget_s - el - 30) / 15))
    for _ in range(n_restarts):
        el = time.time() - t_start
        if el > budget_s - 20:
            break

        # Random perturbation
        scale = 0.03 + 0.07 * rng.random()
        pert = rng.standard_normal(n) * scale
        h_try = project_h(best_h + pert, target_sum)

        # Antisymmetric perturbation for diversity
        if rng.random() < 0.3:
            h_try = best_h.copy()
            for _ in range(5):
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                d = rng.uniform(0.02, 0.1)
                h_try[i1] += d
                h_try[i2] -= d
            h_try = project_h(h_try, target_sum)

        t_val = 100 + 200 * rng.random()
        h_cur = optimize_from_h(h_try, dx, t_val, target_sum, 150)
        c5_cur = compute_c5(h_cur, dx)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Phase 3: Fine-tune with very high temperature
    el = time.time() - t_start
    if el < budget_s - 30:
        h_cur = optimize_from_h(best_h, dx, 2000, target_sum, 200)
        c5_cur = compute_c5(h_cur, dx)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Final projection and verification
    best_h = project_h(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Verify constraint satisfaction
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)
    assert abs(np.sum(best_h) - target_sum) < 1e-6

    return (best_h, best_c5, n)
