import numpy as np
import time

def _project(hv, target_sum):
    hv = np.clip(hv, 0.0, 1.0)
    diff = target_sum - hv.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            free = np.where(hv < 1.0 - 1e-10)[0]
        else:
            free = np.where(hv > 1e-10)[0]
        if len(free) > 0:
            step = diff / len(free)
            hv[free] = np.clip(hv[free] + step, 0.0, 1.0)
    # Nudge to hit exact target
    diff = target_sum - hv.sum()
    if abs(diff) > 1e-15:
        idx = np.argmax(np.where(diff > 0, 1.0 - hv, hv))
        hv[idx] = np.clip(hv[idx] + diff, 0.0, 1.0)
    return hv

def _eval(hv, dx):
    return float(np.max(np.correlate(hv, 1.0 - hv, mode='full')) * dx)

def _soft_grad(hv, n, p):
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    cm = np.max(corr)
    w = np.exp(p * (corr - cm))
    s = w.sum()
    if s < 1e-30:
        return np.zeros(n), 0.0
    w /= s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-8:
            continue
        k = i - (n - 1)
        if k == 0:
            grad += wi * (1.0 - 2.0 * hv)
        elif k > 0:
            # term1[j] = 1-hv[j-k] for j in [k, n-1]
            # term2[j] = hv[j+k] for j in [0, n-1-k]
            j1 = np.arange(k, n)
            grad[j1] += wi * (1.0 - hv[j1 - k])
            j2 = np.arange(0, n - k)
            grad[j2] -= wi * hv[j2 + k]
        else:
            m = -k
            # term1[j] = 1-hv[j+m] for j in [0, n-1-m]
            # term2[j] = hv[j-m] for j in [m, n-1]
            j1 = np.arange(0, n - m)
            grad[j1] += wi * (1.0 - hv[j1 + m])
            j2 = np.arange(m, n)
            grad[j2] -= wi * hv[j2 - m]
    grad -= grad.mean()
    return grad, cm

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

        n = len(h)

    h = _project(h, tgt)

    best_c5 = _eval(best_h, dx)
    
    # Phase 1: Soft-max gradient descent
    p = 20
    alpha = 0.05
    for it in range(200000):
        if time.time() > deadline:

        grad, _ = _soft_grad(best_h, n, p)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:

        c5 = _eval(trial, dx)
        if c5 < best_c5:

            best_c5 = c5
            alpha = max(0.005, alpha * 0.9995)
        else:
            alpha *= 0.95
            if alpha < 0.002:
                best_h = _project(best_h + np.random.randn(n) * 0.01, tgt)
                best_c5 = _eval(best_h, dx)
                alpha = 0.02
        if it % 5000 == 4999:

    # Phase 2: Local search - pair moves
    step = 0.03
    stuck = 0
    for it in range(2000000):
        if time.time() > deadline:

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:

        d = step * (1 if np.random.rand() < 0.5 else -1)
        trial = best_h.copy()
        trial[i] += d
        trial[j] -= d
        if min(trial[i], trial[j]) < -1e-10 or max(trial[i], trial[j]) > 1 + 1e-10:

        trial = _project(trial, tgt)
        c5 = _eval(trial, dx)
        if c5 < best_c5:

import numpy as np
import time

def _project(hv, tgt):
    hv = np.clip(hv, 0.0, 1.0)
    d = tgt - hv.sum()
    if abs(d) > 1e-14:
        if d > 0:
            fr = np.where(hv < 1.0 - 1e-9)[0]
        else:
            fr = np.where(hv > 1e-9)[0]
        if len(fr) > 0:
            hv[fr] += d / len(fr)
            hv = np.clip(hv, 0.0, 1.0)
    d = tgt - hv.sum()
    if abs(d) > 1e-15:
        idx = int(np.argmax(np.where(d > 0, 1.0 - hv, hv)))
        hv[idx] = float(np.clip(hv[idx] + d, 0.0, 1.0))
    return hv

def _eval(hv, dx):
    return float(np.max(np.correlate(hv, 1.0 - hv, mode='full')) * dx)

def _grad_soft(hv, n, p):
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    cm = np.max(corr)
    w = np.exp(p * (corr - cm))
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n), cm
    w /= ws
    grad = np.zeros(n)

        wi = w[i]
        if wi < 1e-9:
            continue
        k = i - (n - 1)
        if k == 0:
            grad += wi * (1.0 - 2.0 * hv)

            j1 = np.arange(k, n)
            grad[j1] += wi * (1.0 - hv[j1 - k])

            grad[j1] += wi * (1.0 - hv[j1 + m])
            j2 = np.arange(m, n)
            grad[j2] -= wi * hv[j2 - m]

    return grad, cm
