def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = 1.0

    # Get initial values
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h0 is not None:
        n0 = len(h0)
        h0 = project_feasible(h0, n0 / 2.0)
        c5_0 = compute_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_0

    # N values to try
    n_list = [45, 60, 80, 100, 120]
    if h0 is not None:
        n_list.insert(0, len(h0))
    n_list = list(dict.fromkeys(n_list))

    time_limit = t0 + budget_s * 0.92

    for n in n_list:
        if time.time() - t0 > time_limit:
            break
        target_sum = n / 2.0

        # Build starting points
        starts = []
        if h0 is not None and len(h0) == n:
            starts.append(h0.copy())
        # Random starts
        for s in range(2):
            h_r = rng.uniform(0.1, 0.9, n)
            starts.append(project_feasible(h_r, target_sum))
        # Structured start: smooth sigmoid
        x = np.linspace(0, 1, n)
        h_sig = 0.5 + 0.4 * np.sin(np.pi * x)
        starts.append(project_feasible(h_sig, target_sum))

        for h_init in starts:
            if time.time() - t0 > time_limit:
                break
            h_opt, c5_opt = optimize_single(
                h_init, target_sum, t0, time_limit, rng, best_c5
            )
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Final validation
    if best_h is not None:
        n = len(best_h)
        best_h = project_feasible(best_h, n / 2.0)
        try:
            final_c5 = evaluate_erdos_solution(best_h)
        except Exception:
            final_c5 = compute_c5(best_h)
        return (best_h, final_c5, n)
    else:
        h = np.full(45, 0.5)
        return (h, 0.5, 45)
