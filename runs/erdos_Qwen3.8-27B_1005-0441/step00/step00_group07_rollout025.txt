    """Local search on binary sequences."""
    n = len(h)
    h = h.copy()
    best_c5 = c5_value(h)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    it = 0
    while it < max_iters:
        if time_limit and time.time() - start_time > time_limit:
            break
        improved = False
        # Try all pair swaps
        # For efficiency, sort by some criterion
        for i_idx in range(len(ones)):
            for j_idx in range(len(zeros)):
                pos1 = ones[i_idx]
                pos0 = zeros[j_idx]
                # Swap
                h[pos1] = 0
                h[pos0] = 1

    return h, best_c5
