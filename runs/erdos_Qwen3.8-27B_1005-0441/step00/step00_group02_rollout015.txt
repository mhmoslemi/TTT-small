        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: random restarts with perturbations
    for _ in range(30):
        if time.time() - t0 > budget_s * 0.90:
            break
        n_cur = len(best_h)
        h_r = best_h + rng.randn(n_cur) * 0.04
        h_r = project_feasible(h_r, n_cur)
        h_opt, c5_opt = optimize_one(h_r, n_cur, 1500, t0, budget_s, 0.92)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: try bisection-style structured starts
    for _ in range(5):
        if time.time() - t0 > budget_s * 0.93:
            break
        n_s = min(len(best_h) + 20, 250)
        # Alternating blocks
        h_s = np.zeros(n_s)
        period = max(4, n_s // 10)
        for i in range(n_s):
            h_s[i] = 1.0 if (i % (2 * period)) < period else 0.0
        h_s += rng.randn(n_s) * 0.1
        h_s = project_feasible(h_s, n_s)
        h_opt, c5_opt = optimize_one(h_s, n_s, 2000, t0, budget_s, 0.95)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final polish at best size
    if time.time() - t0 < budget_s * 0.96:
        h_opt, c5_opt = optimize_one(best_h.copy(), len(best_h), 2000, t0, budget_s, 0.99)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    return best_h, best_c5, len(best_h)
