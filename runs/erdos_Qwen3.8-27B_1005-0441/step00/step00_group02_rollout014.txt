                lr = 0.05
                s = max(10.0, s * 0.5)
                stall = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = np.full(45, 0.5)
    n_init = len(h_init)
    deadline = start_time + budget_s

    best_h = h_init.copy()
    best_c5 = np.max(compute_corr(best_h, n_init))

    # Phase 1: optimize initial
    dl1 = min(deadline, start_time + budget_s * 0.30)
    hh, cc = optimize_h(h_init, n_init, dl1)
    if cc < best_c5:
        best_c5, best_h = cc, hh

    # Phase 2: random restarts with same n
    dl2 = min(deadline, start_time + budget_s * 0.60)
    nrestarts = max(3, int((dl2 - time.time()) / 2.0))
    for r in range(nrestarts):
        if time.time() > dl2 - 2:
            break
        sub_dl = min(dl2, time.time() + 2.0)
        hr = np.random.rand(n_init)
        hr = project_feasible(hr, n_init / 2.0)
        # Mix: 50% binary-ish
        if r % 2 == 1:
            order = np.random.permutation(n_init)
            hr = np.zeros(n_init)
            half = n_init // 2
            hr[order[:half]] = 1.0
            rm = hr.copy()
            rm = project_feasible(rm, n_init / 2.0)
            hr = 0.5 * hr + 0.5 * rm
        hr = project_feasible(hr, n_init / 2.0)
        hh, cc = optimize_h(hr, n_init, sub_dl)
        if cc < best_c5:
            best_c5, best_h = cc, hh

    # Phase 3: try other n
    dl3 = deadline - 10
    for n_try in [23, 30, 44, 46, 50, 55, 60, 80, 100, 150, 200, 300]:
        if time.time() > dl3:
            break
        sub_budget = min(6.0, dl3 - time.time() - 3)
        if sub_budget < 1.5:
            break
        hr = np.random.rand(n_try)
        hr = project_feasible(hr, n_try / 2.0)
        # Also try a structured init
        hc = 0.5 + 0.4 * np.cos(2 * np.pi * np.arange(n_try) / n_try * 3)
        hc = project_feasible(hc, n_try / 2.0)
        for init in [hr, hc]:
            if time.time() > dl3:
                break
            hh, cc = optimize_h(init, n_try, min(dl3, time.time() + sub_budget / 2))
            if cc < best_c5:
                best_c5, best_h = cc, hh

    # Final polish on best
    if time.time() < deadline - 5:
        n_f = len(best_h)
        dh = deadline - 5
        hh, cc = optimize_h(best_h, n_f, dh)
        if cc < best_c5:
            best_c5, best_h = cc, hh

    n_final = len(best_h)
    final_c5 = np.max(compute_corr(best_h, n_final))
    best_h = project_feasible(best_h, n_final / 2.0)
    final_c5 = np.max(compute_corr(best_h, n_final))
    return best_h, final_c5, n_final
