            idx2 = np.arange(n - kp)
            grad[idx1] += w * (1.0 - h[idx2])
            grad[idx2] -= w * h[idx1]
    return val, grad

import numpy as np
import scipy.optimize as opt
import time

def _lags(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    return float(np.max(_lags(h))) * (2.0 / len(h))

def _feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        d = (target - np.sum(h)) / len(h)
        h += d
        h = np.clip(h, 0.0, 1.0)
        if abs(np.sum(h) - target) < 1e-14:
            break
    return h

def _smooth_max(h, beta):
    l = _lags(h)
    m = np.max(l)
    e = np.exp(beta * (l - m))
    return m + np.log(np.sum(e)) / beta

def _smooth_max_grad(h, beta):
    l = _lags(h)
    n = len(h)
    m = np.max(l)
    e = np.exp(beta * (l - m))
    p = e / np.sum(e)
    val = m + np.log(np.sum(e)) / beta
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = p[k_idx]
        if w < 1e-8:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            idx1 = np.arange(n - k)
            idx2 = np.arange(k, n)
            grad[idx1] += w * (1.0 - h[idx2])
            grad[idx2] -= w * h[idx1]
        else:
            kp = -k
            idx1 = np.arange(kp, n)
            idx2 = np.arange(n - kp)
            grad[idx1] += w * (1.0 - h[idx2])
            grad[idx2] -= w * h[idx1]
    return val, grad

def _opt(h0, n, beta, max_iter):
    target = n / 2.0

    def fg(x):
        v, g = _smooth_max_grad(x, beta)
        return v, g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = opt.minimize(fg, h0, jac=True, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None:
            return _feasible(res.x, target)
    except Exception:
        pass
    return h0

def _interpolate_h(h_src, n_dst):
    xp = np.linspace(0.0, 1.0, len(h_src))
    fp = np.linspace(0.0, 1.0, n_dst)
    return np.interp(fp, xp, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    T0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)

    bh = h0.copy()
    bc5 = _c5(bh)
    bn = bh.shape[0]

    def rem():
        return budget_s - (time.time() - T0)

    # Phase 1: optimize from initial
    if rem() > 80:
        h = _opt(bh, bn, 60.0, 600)
        c = _c5(h)
        if c < bc5:
            bh, bc5 = h, c
        # Second pass with higher beta for sharper max
        if rem() > 80:
            h2 = _opt(bh, bn, 120.0, 400)
            c2 = _c5(h2)
            if c2 < bc5:
                bh, bc5 = h2, c2

    # Phase 2: try different n values
    for nt in [bn, 47, 67, 95, 143, 191, 287]:
        if rem() < 100:
            break
        if nt == bn:
            hs = bh.copy()
        elif nt < bn:
            hs = _interpolate_h(bh, nt)
        else:
            hs = _interpolate_h(bh, nt)
        hs = _feasible(hs, nt / 2.0)
        h = _opt(hs, nt, 60.0, 400)
        c = _c5(h)
        if c < bc5:
            bh, bc5, bn = h, c, nt
        # Refine
        if rem() > 80:
            h2 = _opt(bh, bn, 120.0, 300)
            c2 = _c5(h2)
            if c2 < bc5:
                bh, bc5 = h2, c2

    # Phase 3: random restarts
    while rem() > 60:
        hr = np.random.rand(bn)
        hr = _feasible(hr, bn / 2.0)
        h = _opt(hr, bn, 60.0, 200)
        c = _c5(h)
        if c < bc5:
            bh, bc5 = h, c

    # Phase 4: structured starts
    for i in range(5):
        if rem() < 50:
            break
        hs = 0.5 + 0.3 * np.cos(2.0 * np.pi * (i + 1) * np.arange(bn) / bn)
        hs = _feasible(hs, bn / 2.0)
        h = _opt(hs, bn, 60.0, 300)
        c = _c5(h)
        if c < bc5:
            bh, bc5 = h, c

    # Phase 5: final polish
    if rem() > 30:
        h = _opt(bh, bn, 100.0, 500)
        c = _c5(h)
        if c < bc5:
            bh, bc5 = h, c

    bh = _feasible(bh, bn / 2.0)
    bc5 = _c5(bh)
    return (bh, bc5, bn)
