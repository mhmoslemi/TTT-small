                    h_best = h_try
                    no_improve = 0
                else:
                    break

    return h_best, best_c5

def crossover(p1, p2, rng):
    n = len(p1)
    if rng.random() < 0.5:
        cut = rng.integers(1, n - 1)
        c1 = np.concatenate([p1[:cut], p2[cut:]])
        c2 = np.concatenate([p2[:cut], p1[cut:]])
    else:
        mask = rng.random(n) < 0.5
        c1 = np.where(mask, p1, p2)
        c2 = np.where(mask, p2, p1)
    return enforce_constraints(c1, n), enforce_constraints(c2, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    h_init = enforce_constraints(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Initial local optimization
    best_h, best_c5 = local_optimize(best_h, rng, max_evals=3000, temp=0.001)

    # Genetic algorithm
    pop_size = 50
    population = [best_h.copy()]
    scores = [best_c5]

    for i in range(pop_size - 1):
        scale = 0.1 + 0.3 * rng.random()
        h_rand = enforce_constraints(best_h + rng.normal(0, scale, n), n)
        population.append(h_rand)
        scores.append(compute_c5(h_rand))

    gen = 0
    stagnation = 0

    while time.time() < deadline - 90:
        # Sort population
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_c5 - 1e-14:
            best_c5 = scores[0]
            best_h = population[0].copy()
            stagnation = 0
        else:
            stagnation += 1

        # Local refine top candidates
        for idx in range(min(4, len(population))):
            if time.time() > deadline - 90:
                break
            h_ref, c5_ref = local_optimize(population[idx], rng, max_evals=800, temp=0.0005)
            population[idx] = h_ref
            scores[idx] = c5_ref
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                stagnation = 0

        # Selection: tournament-like, keep top half
        keep = max(pop_size // 2, 10)
        survivors = [population[i].copy() for i in range(keep)]

        # Generate offspring
        offspring = []
        n_off = pop_size - len(survivors)
        while len(offspring) < n_off:
            i1 = rng.integers(0, len(survivors))
            i2 = rng.integers(0, len(survivors))
            c1, c2 = crossover(survivors[i1], survivors[i2], rng)
            c1 = enforce_constraints(c1 + rng.normal(0, 0.06, n), n)
            c2 = enforce_constraints(c2 + rng.normal(0, 0.06, n), n)
            offspring.append(c1)
            if len(offspring) < n_off:
                offspring.append(c2)

        # Inject diversity
        n_div = min(4, len(offspring))
        for i in range(n_div):
            scale = 0.15 + 0.3 * rng.random()
            offspring[i] = enforce_constraints(best_h + rng.normal(0, scale, n), n)

        if stagnation > 8:
            for i in range(min(6, len(offspring))):
                offspring[i] = enforce_constraints(best_h + rng.normal(0, 0.4, n), n)

        population = survivors + offspring
        scores = [compute_c5(h) for h in population]
        gen += 1

    # Final refinement with annealing
    if time.time() < deadline - 15:
        remaining = max(5, deadline - time.time() - 15)
        h_ref, c5_ref = local_optimize(best_h, rng, max_evals=int(remaining * 400), temp=0.002)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    # Very final polish - greedy improvement
    if time.time() < deadline - 3:
        n_local = len(best_h)
        for _ in range(200):
            if time.time() > deadline - 3:
                break
            i = rng.integers(0, n_local)
            h_try = best_h.copy()
            h_try[i] += rng.uniform(-0.05, 0.05)
            h_try = enforce_constraints(h_try, n_local)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    return best_h, best_c5, n
