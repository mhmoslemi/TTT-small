import scipy.optimize as so

_global_target = 0.0
_global_eps = 0.01

def _sum_constraint(x):
    return np.sum(x) - _global_target

def _sum_constraint_jac(x):
    return np.ones(len(x))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    x = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(x)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        x = np.clip(x + diff / n, 0.0, 1.0)
    return x

def smooth_obj_fn(x, eps):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    if m < 1e-30:
        return m
    vals = corr / eps
    shifted = vals - np.max(vals)
    return m + eps * float(np.log(np.sum(np.exp(shifted))))

def penalty_obj(x):
    val = smooth_obj_fn(x, _global_eps)
    pen = 5000.0 * (np.sum(x) - _global_target) ** 2
    return val + pen

def get_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    shift = k_max - (n - 1)
    idx = np.arange(n)
    js = idx - shift
    jn = idx + shift
    term1 = np.ones(n)
    ms = (js >= 0) & (js < n)
    term1[ms] = 1.0 - h[js[ms]]
    term2 = np.zeros(n)
    mn = (jn >= 0) & (jn < n)
    term2[mn] = h[jn[mn]]
    return dx * (term1 - term2)

        h_copy[j] = best_v
        h_copy = project_feasible(h_copy, ts)
        best_c5 = compute_c5(h_copy)
