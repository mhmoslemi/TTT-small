        grad[valid2] -= w * h[i2[valid2]]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-13:
        frac = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = np.count_nonzero(frac)
        if cnt > 0:
            h[frac] += diff / cnt
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, corr, temperature, topk=15):
    n = len(h)
    t = temperature
    sh = t * corr
    sh -= np.max(sh)
    weights = np.exp(sh)
    total = np.sum(weights)
    weights /= total
    # Only use top-k weights
    idx_sorted = np.argsort(weights)[::-1]
    idx_top = idx_sorted[:topk]
    w_top = weights[idx_top]
    w_top /= np.sum(w_top)
    
    grad = np.zeros(n)
    j = np.arange(n)
    for k, idx in enumerate(idx_top):
        w = w_top[k]
        if w < 1e-10:
            continue
        s = idx - (n - 1)
        i1 = j - s
        v1 = (i1 >= 0) & (i1 < n)
        grad[v1] += w * (1.0 - h[i1[v1]])
        i2 = j + s
        v2 = (i2 >= 0) & (i2 < n)
        grad[v2] -= w * h[i2[v2]]
    return grad

def gradient_descent(h, dx, target_sum, t_limit, alpha_init=0.02, temp_init=30.0, max_iter=20000):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = alpha_init
    temp = temp_init
    no_imp = 0
    prev_best = best_c5
    
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_val = np.max(corr) * dx
        if c5_val < best_c5 - 1e-11:
            best_c5 = c5_val
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 250:
            alpha *= 0.6
            temp = max(5.0, temp * 0.85)
            no_imp = 0
            if alpha < 1e-7:
                break
        grad = smooth_grad(h, corr, temp)
        h = project(h - alpha * grad, target_sum)
    
    return best_h, best_c5
