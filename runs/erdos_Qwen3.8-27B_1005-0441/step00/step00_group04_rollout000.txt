import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max(), c

def project_simplex(h, target_sum):
    n = len(h)
    if n == 0:
        return h
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_at_max(h, active_frac=0.0):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    c_max = c.max()
    if active_frac > 0:
        threshold = c_max - active_frac * (c_max - c.min())
        active = c >= threshold
    else:
        active = np.zeros_like(c, dtype=bool)
        active[np.argmax(c)] = True
    grad = np.zeros(n)
    for idx in range(n):
        if not active[idx]:
            continue
        k = idx - (n - 1)
        for i in range(n):
            if 0 <= i - k < n:
                grad[i] += (1.0 - h[i - k])
            if 0 <= i + k < n:
                grad[i] -= h[i + k]
    count = active.sum()
    if count > 1:
        grad /= count
    return grad, c_max, c

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    padded = np.concatenate([h, [h[0]]])
    return np.interp(x_new, x_old, padded)

def gradient_optimize(h, target_sum, max_time, initial_lr=0.05, momentum=0.9):
    n = len(h)
    v = np.zeros(n)
    lr = initial_lr
    best = compute_c5(h)[0]
    best_h = h.copy()
    stall = 0
    start = time.time()
    while time.time() - start < max_time:
        grad, cmax, c = compute_grad_at_max(h, active_frac=0.02)
        v = momentum * v + lr * grad
        h_new = project_simplex(h - v, target_sum)
        new_c5 = compute_c5(h_new)[0]
        if new_c5 < best - 1e-9:
            best = new_c5
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1
            if stall > 40:
                lr *= 0.6
                v *= 0.5
                stall = 0
        if lr < 1e-6:
            break
    return best_h, best

def binary_local_search(h, target_sum, max_time, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    best = compute_c5(h)[0]
    best_h = h.copy()
    start = time.time()
    temp = 0.01
    while time.time() - start < max_time:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h_new = best_h.copy()
        move = min(h_new[i], 1.0 - h_new[j]) * rng.uniform(0.3, 1.0)

        h_new[j] += move
        h_new = project_simplex(h_new, target_sum)

        delta = new_c5 - best
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-8)):
            best = new_c5
            best_h = h_new
        temp *= 0.995
    return best_h, best
