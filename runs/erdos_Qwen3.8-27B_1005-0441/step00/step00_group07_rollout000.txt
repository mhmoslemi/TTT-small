                    break
                hcand = project_feasible(hcand, n0)
                hb, cb = binary_search(hcand, max_iter=100, time_limit=min(30, time_left() - 10), rng=rng)
                if cb < best_c5:
                    best_h, best_c5, best_n = hb.copy(), cb, n0
            # SLSQP on best so far
            if time_left() > 40:
                hc, vc = slsqp_refine(best_h, best_n, time_limit=min(40, time_left() - 10), beta=150.0)
                dx_s = 2.0 / best_n
                c5_s = compute_c5(hc, dx_s)
                if c5_s < best_c5:
                    best_h, best_c5, best_n = hc.copy(), c5_s, best_n

    # Phase 6: Final binary polish
    if time_left() > 20:
        hb, cb = binary_search(best_h, max_iter=200, time_limit=min(40, time_left() - 5), rng=rng)
        if cb < best_c5:
            best_h, best_c5, best_n = hb.copy(), cb, best_n

    # Ensure final feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
