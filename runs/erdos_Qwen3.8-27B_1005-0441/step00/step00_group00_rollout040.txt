import numpy as np
import time

def project_to_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        if not np.any(mask):
            return h
        room = (1.0 - h)[mask]
        tr = np.sum(room)
        if tr < 1e-14:
            return h
        frac = min(1.0, diff / tr)
        h[mask] += frac * room
    else:
        mask = h > 1e-12
        if not np.any(mask):
            return h
        room = h[mask]
        tr = np.sum(room)
        if tr < 1e-14:
            return h
        frac = min(1.0, -diff / tr)
        h[mask] -= frac * room
    h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += residual / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += residual / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def local_refine(h, target_sum, max_time):
    n = len(h)
    h = h.copy()

    t_end = time.time() + max_time
    # Coordinate descent with decreasing steps
    steps = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for ss in steps:
        if time.time() > t_end:
            break
        improved = True
        sweep = 0
        while improved and time.time() < t_end and sweep < 20:
            improved = False
            sweep += 1
            order = np.random.permutation(n)
            for i in order:
                for sign in (1, -1):
                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + sign * ss, 0.0, 1.0)
                    h_new = project_to_feasible(h_new, target_sum)
                    val = compute_c5(h_new)
                    if val < best_val - 1e-13:
                        h = h_new
                        best_val = val
                        improved = True
                        break
    # Random hill climbing
    while time.time() < t_end:
        batch = 50
        for _ in range(batch):
            i = np.random.randint(n)
            delta = np.random.uniform(-0.08, 0.08)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new = project_to_feasible(h_new, target_sum)
            val = compute_c5(h_new)
            if val < best_val - 1e-13:
                h = h_new
                best_val = val
    return h, best_val
