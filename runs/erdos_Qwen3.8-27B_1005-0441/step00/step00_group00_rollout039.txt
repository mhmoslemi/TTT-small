import numpy as np
import time

def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        excess = h.sum() - target_sum
        if abs(excess) < 1e-14:
            break
        if excess > 0:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            h[idx] = np.maximum(h[idx] - excess / len(idx), 0.0)
        else:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) == 0:
                break
            h[idx] = np.minimum(h[idx] + (-excess) / len(idx), 1.0)
    return np.clip(h, 0.0, 1.0)

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_c5(h):
    return get_overlaps(h).max()

def compute_grad_weighted(h, overlaps, top_count=5):
    n = len(h)
    idx_order = np.argsort(overlaps)[-top_count:]
    vals = overlaps[idx_order]
    w = np.exp(15.0 * (vals - vals.max()))
    w = w / w.sum()
    grad = np.zeros(n)
    for wi, k_full in zip(w, idx_order):
        if wi < 1e-10:
            continue
        lag = k_full - (n - 1)
        for m in range(n):
            if 0 <= m - lag < n:
                grad[m] += wi * (1.0 - h[m - lag])
            if 0 <= m + lag < n:
                grad[m] -= wi * h[m + lag]
    return grad

def local_search(h, target, rng, n_trials=500):
    n = len(h)
    c5_curr = get_c5(h)
    for _ in range(n_trials):
        i = rng.randint(n)
        h_t = h.copy()
        h_t[i] = np.clip(h_t[i] + rng.uniform(-0.15, 0.15), 0.0, 1.0)
        h_t = project_h(h_t, target)
        c5_t = get_c5(h_t)
        if c5_t < c5_curr - 1e-12:
            h = h_t
            c5_curr = c5_t
    return h, c5_curr
