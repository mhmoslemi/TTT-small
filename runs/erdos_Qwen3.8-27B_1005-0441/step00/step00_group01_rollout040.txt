    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(corr, beta)

def objective_direct(h, beta, n):
    return compute_c5(h)

def project_feasible(h, target_sum):
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
    return h

            options={'maxiter': maxiter, 'ftol': 1e-14}

        return h0
