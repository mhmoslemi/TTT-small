import numpy as np

def project_feasible(h, N):
    target = N / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def get_corr(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_corr_k(h, k, N):
    dx = 2.0 / N
    delta = k - (N - 1)
    grad = np.zeros(N)
    oh = 1.0 - h
    if delta >= 0:
        j1 = np.arange(delta, N)
        if len(j1) > 0:
            grad[j1] += oh[j1 - delta]
        j2 = np.arange(0, N - delta)
        if len(j2) > 0:
            grad[j2] -= h[j2 + delta]
    else:
        d = -delta
        j1 = np.arange(0, N - d)
        if len(j1) > 0:
            grad[j1] += oh[j1 + d]
        j2 = np.arange(d, N)
        if len(j2) > 0:
            grad[j2] -= h[j2 - d]
    return dx * grad

def weighted_grad(h, weights, N):
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        wk = weights[k]
        if wk > 1e-10:
            grad += wk * grad_corr_k(h, k, N)
    return grad
