    h = 0.5 + 0.35 * np.sin(x)
    return _fix_sum(h, n)

def _perturb(h, rng, scale):
    return _fix_sum(h + rng.normal(0, scale, len(h)), len(h))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_c5 = np.inf
    best_h = None

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None:
        h_init = _fix_sum(h_init, len(h_init))
        c5 = _eval_c5(h_init)
        best_c5 = c5
        best_h = h_init.copy()

        for alpha in [50, 100, 200, 400]:
            if time.time() - t0 > budget_s - 120:
                break
            h_opt = _optimize_sm(h_init, alpha, 400)
            h_opt = _fix_sum(h_opt, len(h_opt))
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

        if time.time() - t0 < budget_s - 120:
            h_opt = _optimize_mm(best_h, 500)
            h_opt = _fix_sum(h_opt, len(h_opt))
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    n_list = [47, 50, 53, 60, 67, 80, 100]
    for n in n_list:
        if time.time() - t0 > budget_s - 60:
            break
        starts = [_random_h(n, rng), _block_h(n), _sin_h(n, 1), _sin_h(n, 2)]
        for h0 in starts:
            if time.time() - t0 > budget_s - 30:
                break
            h_opt = _optimize_sm(h0, 100, 300)
            h_opt = _fix_sum(h_opt, n)
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            if time.time() - t0 > budget_s - 30:
                break
            h_opt2 = _optimize_sm(h_opt, 300, 300)
            h_opt2 = _fix_sum(h_opt2, n)
            c52 = _eval_c5(h_opt2)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h_opt2.copy()
            if time.time() - t0 > budget_s - 20:
                break
            h_opt3 = _optimize_mm(h_opt2, 300)
            h_opt3 = _fix_sum(h_opt3, n)
            c53 = _eval_c5(h_opt3)
            if c53 < best_c5:
                best_c5 = c53
                best_h = h_opt3.copy()

    if best_h is not None:
        for _ in range(30):
            if time.time() - t0 > budget_s - 20:
                break
            h_pert = _perturb(best_h, rng, 0.015)
            h_opt = _optimize_sm(h_pert, 200, 300)
            h_opt = _fix_sum(h_opt, len(h_opt))
            c5 = _eval_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    if best_h is not None and time.time() - t0 < budget_s - 15:
        h_final = _optimize_mm(best_h, 600)
        h_final = _fix_sum(h_final, len(h_final))
        c5 = _eval_c5(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final

    if best_h is None:
        best_h = np.full(47, 0.5)
        best_c5 = _eval_c5(best_h)

    best_h = _fix_sum(best_h, len(best_h))
    best_c5 = _eval_c5(best_h)
    return best_h, best_c5, len(best_h)
