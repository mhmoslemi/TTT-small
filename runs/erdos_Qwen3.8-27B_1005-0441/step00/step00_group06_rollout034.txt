def project_onto_set(h, n):

def compute_subgradient(h, n, top_k=3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:top_k]
    cmin = float(corr[active[-1]])
    w = corr[active] - cmin + 1.0
    w = w / np.sum(w)
    grad = np.zeros(n)
    for wi, j in zip(w, active):
        k = int(j) - (n - 1)
        lo1 = max(0, -k)
        hi1 = min(n - 1, n - 1 - k)
        if lo1 <= hi1:

        lo2 = max(0, k)
        hi2 = min(n - 1, n - 1 + k)
        if lo2 <= hi2:
