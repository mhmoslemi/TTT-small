            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 2: SLSQP with exact non-smooth objective
    def exact_obj(h):
        return compute_c5(h, dx)

    if elapsed() < budget_s * 0.50:
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
                'jac': lambda h: np.ones(n)}]
        try:
            res = minimize(exact_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 3: Coordinate descent (exact 1D minimization per coordinate)
    if elapsed() < budget_s * 0.72:
        for sweep in range(50):
            if elapsed() > budget_s * 0.72:
                break
            improved = False
            for i in range(n):
                if elapsed() > budget_s * 0.72:
                    break
                sum_others = float(np.sum(best_h)) - best_h[i]
                lo = max(0.0, n / 2.0 - sum_others)
                hi = min(1.0, n / 2.0 - sum_others)
                if hi - lo < 1e-12:
                    continue
                def obj_i(vi):
                    h_tmp = best_h.copy()
                    h_tmp[i] = vi
                    return compute_c5(h_tmp, dx)
                try:
                    r_i = minimize_scalar(obj_i, bounds=(lo, hi), method='bounded',
                                          options={'xatol': 1e-12})
                    if r_i.fun < best_c5 - 1e-14:
                        best_h[i] = r_i.x
                        best_c5 = r_i.fun
                        improved = True
                except Exception:
                    pass
            if not improved:
                break

    # Phase 4: Multi-start with varying perturbation sizes
    for r in range(25):
        if elapsed() > budget_s * 0.92:
            break
        sigma = 0.05 + 0.3 * rng.random()
        h_r = best_h + rng.normal(0, sigma, n)
        h_r = project_feasible(h_r, n)
        for beta in [50, 500]:
            if elapsed() > budget_s * 0.92:
                break
            def obj_rs(h, _b=beta):
                corr = np.correlate(h, 1.0 - h, mode="full") * dx
                return softmax_max(corr, _b) + 1000.0 * (np.sum(h) - n / 2.0) ** 2
            try:
                res = minimize(obj_rs, h_r, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-14})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 5: Quick coordinate polish on best
    if elapsed() < budget_s * 0.97:
        for i in range(n):
            if elapsed() > budget_s * 0.97:
                break
            sum_others = float(np.sum(best_h)) - best_h[i]
            lo = max(0.0, n / 2.0 - sum_others)
            hi = min(1.0, n / 2.0 - sum_others)
            if hi - lo < 1e-12:
                continue
            def obj_pol(vi):
                h_tmp = best_h.copy()
                h_tmp[i] = vi
                return compute_c5(h_tmp, dx)
            try:
                r_i = minimize_scalar(obj_pol, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-14})
                if r_i.fun < best_c5 - 1e-15:
                    best_h[i] = r_i.x
                    best_c5 = r_i.fun
            except Exception:
                pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
