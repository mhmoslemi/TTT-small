Valid range: max(0, -s) ≤ j ≤ min(n-1, n-1-s)

  output[k] = sum_j h[j] * (1 - h[j+s])

d/dh[m] sum_j h[j]*h[j+s]:

For the term h[j]*h[j+s] where j+s=m, i.e., j=m-s: contributes h[m-s] (if m-s valid)

∂f/∂h[m] = sum_k w[k] * (1 - h[m+s_k]*I - h[m-s_k]*I) * dx

where s_k = k - n + 1.

The "1" term: sum_k w[k] * 1 * dx = dx (since sum_k w[k] = 1)
So ∂f/∂h[m] = dx - dx * sum_k w[k] * (h[m+s_k]*I + h[m-s_k]*I)

The first part (dx) is constant w.r.t. m, so it doesn't affect the optimization direction (it's in the null space of the constraint sum(h)=n/2). We can drop it for the gradient step.

So effectively: ∂f/∂h[m] ∝ -sum_k w[k] * (h[m+s_k] + h[m-s_k]) (with validity)

This is like a weighted "mixture" of shifted versions of h. For the max (tau→0), only the argmax k matters, so the gradient is just -(h[m+s*] + h[m-s*]) (up to the constant dx).

This is a very clean formula! And it's easy to compute.

def analytic_gradient(h, w, dx, n):
    """Compute gradient of smooth max objective (up to constant offset)."""
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - n + 1
        wk = w[k]
        # Contribution: -wk * dx * (h[m+s] + h[m-s]) for all m
        if s >= 0:
            # h[m+s] valid for m in [0, n-1-s]
            grad[:n-s] -= wk * dx * h[s:]
            # h[m-s] valid for m in [s, n-1]
            grad[s:] -= wk * dx * h[:n-s]
        else:
            t = -s  # t > 0
            # h[m+s] = h[m-t] valid for m in [t, n-1]
            grad[t:] -= wk * dx * h[:n-t]
            # h[m-s] = h[m+t] valid for m in [0, n-1-t]
            grad[:n-t] -= wk * dx * h[t:]
    return grad

    c_s = c - c_max
    scaled = c_s / tau
    shifted = scaled - np.max(scaled)
    exps = np.exp(shifted)
    total = np.sum(exps)
    w = exps / total
    obj = c_max + tau * (np.log(total) + np.max(scaled))

                grad[s:] -= wk * dx * h[:n-s]
        else:

            if n - t > 0:
                grad[t:] -= wk * dx * h[:n-t]
                grad[:n-t] -= wk * dx * h[t:]

def analytic_gradient(h, w, dx, n):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - n + 1
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += wk * dx * (1.0 - h[s:])
                grad[s:] -= wk * dx * h[:ns]
        else:
            t = -s
            nt = n - t
            if nt > 0:
                grad[t:] += wk * dx * (1.0 - h[:nt])
                grad[:nt] -= wk * dx * h[t:]
    return grad

def optimize_run(h0, n, deadline, lr=0.1, tau0=0.05, tau1=0.001, max_iters=5000):
    h = project_feasible(h0.copy(), n)
    dx = 2.0 / n
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iters)
        tau = tau0 * (tau1 / tau0) ** frac
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        w = get_weights(c, tau)
        grad = analytic_gradient(h, w, dx, n)
        m_a = b1 * m_a + (1.0 - b1) * grad
        v_a = b2 * v_a + (1.0 - b2) * grad * grad
        m_hat = m_a / (1.0 - b1 ** t)
        v_hat = v_a / (1.0 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000 and tau < 0.01:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.default_rng(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.zeros(70)
        h_init[:35] = 1.0
    n0 = len(h_init)
    h_init = project_feasible(h_init, n0)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n0
    # Phase 1: optimize at n0
    for lr in [0.1, 0.2, 0.05]:
        if time.time() > deadline:
            break
        h_opt, c5_opt = optimize_run(best_h, n0, deadline, lr=lr, max_iters=6000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    # Phase 2: upsample
    for factor in [2, 3]:
        if time.time() > deadline:
            break
        n = n0 * factor
        if n > 300:
            continue
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n)
        h_opt, c5_opt = optimize_run(h_up, n, deadline, lr=0.08, max_iters=4000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n
    # Phase 3: random restarts
    attempts = 0
    while time.time() < deadline - 5 and attempts < 50:
        attempts += 1
        sigma = 0.02 + 0.04 * rng.random()
        h_rand = best_h + sigma * rng.standard_normal(best_n)
        h_rand = project_feasible(h_rand, best_n)
        h_opt, c5_opt = optimize_run(h_rand, best_n, deadline, lr=0.08, max_iters=600)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    # Phase 4: fine polish on best
    if time.time() < deadline - 5:
        h_opt, c5_opt = optimize_run(best_h, best_n, deadline, lr=0.03, tau0=0.01, tau1=0.0005, max_iters=3000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
