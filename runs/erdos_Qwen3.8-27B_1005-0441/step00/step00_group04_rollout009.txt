import numpy as np
import time

def project(v, S):
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def c5_val(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = int(np.argmax(corr))
    offset = k - (n - 1)
    idx = np.arange(n)
    t1 = idx + offset
    t2 = idx - offset
    m1 = (t1 >= 0) & (t1 < n)
    m2 = (t2 >= 0) & (t2 < n)
    grad = np.zeros(n)
    grad[m1] += (1.0 - h[t1[m1]])
    grad[m2] += -h[t2[m2]]
    return dx * grad, corr[k] * dx

def optimize_local(h, max_iter, t_limit, t0, base_step=0.08):
    n = len(h)
    step = base_step
    best_h = h.copy()
    best_v = c5_val(h)
    stale = 0
    for it in range(max_iter):
        if time.time() - t0 > t_limit:
            break
        grad, val = c5_subgrad(h)
        if val < best_v:
            best_v = val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        h_new = project(h - step * grad, n / 2)
        new_val = c5_val(h_new)
        if new_val < val + 1e-10:
            h = h_new
            step = min(step * 1.03, 0.5)
        else:
            step *= 0.8
            if step < 0.002:
                kick = np.random.randn(n) * (0.03 + 0.02 * stale * 0.1)
                h = project(h + kick, n / 2)
                step = base_step * 0.5
    return best_h, best_v

    x_n = np.linspace(0.0, 2.0, n)
    return np.interp(x_n, x_s, h_src)
