def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    n = len(x)
    # Bisection on lambda: h_i = clamp(x_i - lambda, 0, 1)
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):

        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target_sum:

        else:

    n = len(x)
    lo, hi = x.min() - 1, x.max() + 1

        s = h.sum()
        if s > target_sum:

        else:
            hi = mid

    if abs(diff) > 1e-9:

        if interior.sum() > 0:

def objective_and_grad(h, n, beta, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    soft = (m + np.log(np.sum(exps)) / beta)
    
    # Gradient of soft max
    weights = exps / np.sum(exps)  # sum to 1
    
    grad = np.zeros(n)

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum} via bisection on Lagrange mult."""
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_grad(h, n, beta, pen):
    """Soft-max objective and gradient with sum penalty."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    se = np.sum(e)

                no_improve += 1
            hp = hp_new
            if no_improve > 300:
                lr *= 0.85
                no_improve = 0
            if lr < 1e-9:
                break

    # Try re-optimizing from best with different beta
    if best_h is not None and time.time() - t0 < budget_s - 20:
        n = len(best_h)
        target = n * 0.5
        for beta in [200.0]:
            _spec['n'] = n
            _spec['beta'] = beta
            _spec['pen'] = 5000.0
            bounds = [(0.0, 1.0)] * n
            res = minimize(_obj_func, best_h.copy(), jac=_grad_func, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 80})
            hopt = _project(res.x, target)
            cv = _compute_c5(hopt)
            if np.isfinite(cv) and cv < best_c5:
                best_c5 = cv
                best_h = hopt.copy()

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_c5 = _compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
