import numpy as np
import time


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def c5_of(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def grad_soft(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        a, b = max(0, sh), min(n, sh + n)
        if b > a:
            grad[a:b] += wk * (1.0 - h[a - sh:b - sh])
        a, b = max(0, -sh), min(n, n - sh)
        if b > a:
            grad[a:b] -= wk * h[a + sh:b + sh]
    return grad * dx


def optimize(h0, n, t_limit, rng):
    dx = 2.0 / n
    ts = n / 2.0
    h = project_onto_constraint(h0.copy(), ts)
    best_h = h.copy()
    best_c5 = c5_of(h, n)
    step = 0.03
    beta = 20.0
    stall = 0

    while time.time() < t_limit:
        g = grad_soft(h, n, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        improved = False
        s_try = step
        for _ in range(25):
            hnew = project_onto_constraint(h - s_try * g, ts)
            c5n = c5_of(hnew, n)
            if c5n < best_c5 - 1e-14:
                h = hnew
                best_c5 = c5n
                best_h = h.copy()
                improved = True
                break
            s_try *= 0.5
        if improved:
            stall = 0
            beta = min(400.0, beta * 1.08)
            step = min(0.5, step * 1.15)
        else:
            stall += 1
            step *= 0.7
            if stall > 100:
                break
        if stall > 30 and rng.random() < 0.4:
            h = project_onto_constraint(h + rng.normal(0, 0.04, n), ts)
            stall = 0
            step = 0.03
    return best_h, best_c5


def make_binary_init(n, rng):
    h = np.zeros(n)
    npass = n // 2
    pos = rng.choice(n, size=npass, replace=False)
    h[pos] = 1.0
    rem = n / 2.0 - npass
    if rem > 0:
        h[npass] = rem
    return h

    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
