        best[j] -= delta
        c5 = _compute_c5(best)
        if c5 < best_c5:
            best_c5 = c5
        else:
            best[i], best[j] = si, sj
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = None
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    if h0 is None or len(h0) == 0:
        h0 = 0.5 * np.ones(80)

    n = len(h0)
    h0 = _project(h0, n)
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    # Main SLSQP optimization
    if time.time() - t0 < budget_s * 0.25:
        h_opt, c5 = _optimize(h0, n, 1500, 1e-13)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Local refinement
    if time.time() - t0 < budget_s * 0.35:
        best_h, best_c5 = _local_refine(best_h, n, 500)

    # Multi-start with perturbations
    for i in range(40):
        if time.time() - t0 > budget_s * 0.6:
            break
        scale = 0.08 * (1.0 - i / 60.0) + 0.005
        h_pert = best_h + rng.randn(n) * scale
        h_pert = _project(h_pert, n)
        h_opt, c5 = _optimize(h_pert, n, 400, 1e-11)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
            if time.time() - t0 < budget_s * 0.55:
                best_h, best_c5 = _local_refine(best_h, n, 300)

    # Try higher resolutions
    for n_hi in [120, 160, 240]:
        if time.time() - t0 > budget_s * 0.82:
            break
        old_n = len(best_h)
        x_new = np.linspace(0.0, 2.0 * (n_hi - 1) / n_hi, n_hi)
        x_old = np.linspace(0.0, 2.0 * (old_n - 1) / old_n, old_n)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _project(h_up, n_hi)
        h_opt, c5 = _optimize(h_up, n_hi, 600, 1e-11)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5

    # Final polish
    if time.time() - t0 < budget_s * 0.95:
        fn = len(best_h)
        h_opt, c5 = _optimize(best_h, fn, 3000, 1e-14)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        best_h, best_c5 = _local_refine(best_h, fn, 500)

    return (best_h, best_c5, len(best_h))
