import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_corr(h, alpha):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -60, 60)
    return m + np.log(np.sum(np.exp(s))) / alpha

def project_feasible(h, target):
    for _ in range(20):
        h = h + (target - h.sum()) / len(h)
        h = np.clip(h, 0.0, 1.0)
