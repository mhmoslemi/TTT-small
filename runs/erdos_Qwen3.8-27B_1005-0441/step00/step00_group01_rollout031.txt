import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def sum_con(vars, n):
    return float(np.sum(vars[:n]) - n / 2.0)

def sum_jac(vars, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def overlap_cons(vars, n, dx):
    h = vars[:n]
    t = vars[n]
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - overlaps

def overlap_jac(vars, n, dx):
    h = vars[:n]
    n_ov = 2 * n - 1
    jac = np.zeros((n_ov, n + 1))
    jac[:, n] = 1.0
    j_idx = np.arange(n)
    for k in range(n_ov):
        s = k - (n - 1)
        js = j_idx - s
        jsp = j_idx + s
        mask1 = (js >= 0) & (js < n)
        mask2 = (jsp >= 0) & (jsp < n)
        g = np.zeros(n)
        if mask1.any():
            g[mask1] = 1.0 - h[js[mask1]]
        if mask2.any():
            g[mask2] = g[mask2] - h[jsp[mask2]]
        jac[k, :n] = -dx * g
    return jac

def obj_fn(vars, n):
    return float(vars[n])

def grad_fn(vars, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def do_slsqp(h_start, n, dx, constraints, bounds, maxiter):
    c5_start = compute_c5(h_start, dx)
    x0 = np.concatenate([h_start, [c5_start + 0.005]])
    res = minimize(obj_fn, x0, method='SLSQP', jac=grad_fn, args=(n,),
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.x is not None and res.success:
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        c5_new = compute_c5(h_new, dx)
        return h_new, c5_new
    return None, None

    diff = n / 2.0 - np.sum(h)

        return h
    mask = (h > 0.005) & (h < 0.995)
    if mask.sum() > 0:
        h = h.copy()
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
