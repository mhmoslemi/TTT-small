        grad[:n - s] -= dx * h[s:]
    else:
        t = -s
        grad[:n - t] += dx * (1.0 - h[t:])
        grad[t:] -= dx * h[:n - t]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 45
        h = np.full(n, 0.5)
    h = project_feasible(h, n)

    best_h = h.copy()
    cor = compute_overlaps(h)
    best_c5 = float(np.max(cor))

    beta = 30.0
    lr = 0.008
    stall = 0
    iters_per_restart = 300

    while time.time() - t0 < budget_s - 3:
        cor = compute_overlaps(h)
        cur = float(np.max(cor))

        if cur < best_c5 - 1e-10:
            best_c5 = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > iters_per_restart:
            h = np.clip(h + rng.randn(n) * 0.06, 0, 1)
            h = project_feasible(h, n)
            lr = 0.008
            beta = 15.0 + rng.rand() * 30
            stall = 0
            iters_per_restart = 200 + int(rng.rand() * 200)
            continue

        w = np.exp(beta * (cor - cor.max()))
        w /= w.sum()

        g = np.zeros(n)
        for k in range(len(cor)):
            if w[k] > 1e-9:
                g += w[k] * grad_at_lag(h, k, n)

        improved = False
        a = lr
        for _ in range(8):
            h2 = project_feasible(h - a * g, n)
            c2 = float(np.max(compute_overlaps(h2)))
            if c2 < cur - 1e-12:
                h = h2
                improved = True
                break
            a *= 0.5

        if improved:
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.6
            if lr < 5e-6:
                h = np.clip(h + rng.randn(n) * 0.04, 0, 1)
                h = project_feasible(h, n)
                lr = 0.01
                beta = 15.0 + rng.rand() * 25

        if (stall % 50) == 0 and stall > 0:
            beta = min(beta * 1.15, 120.0)

    # Polish: a few small steps at the best solution
    h = best_h.copy()
    for _ in range(5000):
        if time.time() - t0 > budget_s - 1:
            break
        cor = compute_overlaps(h)
        cur = float(np.max(cor))
        w = np.exp(80.0 * (cor - cor.max()))
        w /= w.sum()
        g = np.zeros(n)
        for k in range(len(cor)):
            if w[k] > 1e-10:
                g += w[k] * grad_at_lag(h, k, n)
        h2 = project_feasible(h - 0.001 * g, n)
        c2 = float(np.max(compute_overlaps(h2)))
        if c2 < cur:
            h = h2
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = float(np.max(compute_overlaps(best_h)))
    return best_h, best_c5, n
