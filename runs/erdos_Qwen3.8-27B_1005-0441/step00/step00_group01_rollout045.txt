    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

    h = np.zeros(n)

            h[i] = 1.0

    target = n // 2

        ones = np.where(h == 1)[0]

        for idx in rng.choice(ones, excess, replace=False):
            h[idx] = 0
    elif cnt < target:
        zeros = np.where(h == 0)[0]

        for idx in rng.choice(zeros, deficit, replace=False):

            tlim = min(time.time() + max(2, time_per_n / n_starts - 1), deadline - 70)
            hb, cb = sa_binary(h_s, tlim, rng)
            update(hb, cb)

    # Phase 2: Focus on best, longer SA + targeted LS
    if best_h is not None:
        # Longer SA with multiple restarts
        n_restarts = 3
        t_per_restart = min(120, (deadline - time.time() - 60) / n_restarts)
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            if r == 0:
                h_start = best_h
            else:
                h_start = best_h.copy()
                mask = rng.random(len(h_start)) < 0.1
                # Randomly flip some bits
                flip_idx = np.where(mask)[0]
                for idx in flip_idx[:len(flip_idx)//2]:
                    if h_start[idx] > 0.5:
                        h_start[idx] = 0.0
                    else:
                        h_start[idx] = 1.0
            tlim = time.time() + t_per_restart
            hb, cb = sa_binary(h_start, min(tlim, deadline - 50),
                               rng, T0=0.02, Tmin=0.0005, cool=0.99995)
            update(hb, cb)

        # Targeted local search on best
        if time.time() < deadline - 40:
            tlim_ls = time.time() + min(80, (deadline - 40 - time.time()))
            n_iter = 0
            while time.time() < tlim_ls:
                h_ls, c5_ls = targeted_ls(best_h, 5000, rng)
                update(h_ls, c5_ls)
                n_iter += 1

        # Continuous refinement
        if time.time() < deadline - 15:
            tlim_c = time.time() + min(60, (deadline - 15 - time.time()))
            hb, cb = sa_continuous(best_h, tlim_c, rng)
            update(hb, cb)

    # Phase 3: Final polish - short SA from best
    if best_h is not None and time.time() < deadline - 5:
        tlim = deadline - 3
        hb, cb = sa_binary(best_h, tlim, rng, T0=0.008, Tmin=0.0003, cool=0.99997)
        update(hb, cb)

    # Fallback
    if best_h is None:
        n = 46
        best_h = make_binary_h(n, rng)
        best_c5 = compute_c5(best_h)

    # Enforce constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    target = len(best_h) / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(np.abs(best_h - 0.5))
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - best_h[idx])
                best_h[idx] += add
                diff -= add
            else:
                sub = min(-diff, best_h[idx])
                best_h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
