def grad_of_max_corr(h):
    """Compute subgradient of max(np.correlate(h, 1-h, mode='full') * dx) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n

    k_max = int(np.argmax(c))
    eps = 1e-7
    grad = np.zeros(n)
    for j in range(n):
        h_p = h.copy()
        h_p[j] = min(h_p[j] + eps, 1.0)
        actual_eps = h_p[j] - h[j]

            h_p[j] = 1.0
            actual_eps = 1.0 - h[j]
            if actual_eps < 1e-12:
                grad[j] = 0.0
                continue
        c_p = np.correlate(h_p, 1.0 - h_p, mode='full')
        grad[j] = (c_p[k_max] - c[k_max]) / actual_eps
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def normalize_h(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_max_corr(h, top_k=1):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    # Get top_k indices
    idx = np.argsort(c)[-top_k:]
    eps = 1e-7
    grad = np.zeros(n)
    for ki in idx:
        ki = int(ki)
        c_ki = c[ki]
        for j in range(n):
            h_p = h.copy()
            h_p[j] = min(h[j] + eps, 1.0)
            ae = h_p[j] - h[j]
            if ae < 1e-12:
                grad[j] += 0.0
                continue
            cp = np.correlate(h_p, 1.0 - h_p, mode='full')[ki]
            grad[j] += (cp - c_ki) / ae
    return grad / top_k * dx

def run(seed=42, budget_s=1100, **kwargs):
