import numpy as np
import time

def _project(h, n):
    """Project h to [0,1]^n with sum(h) = n/2 via water-filling."""
    target = n / 2.0
    h = h.copy()
    for _ in range(80):
        c = (target - h.sum()) / n
        if abs(c) < 1e-14:
            break
        h = np.clip(h + c, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def _corr_overlaps(h, n, dx):
    """Return all (2n-1) overlap values scaled by dx."""
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _lse_gradient(h, n, dx, beta):
    """Analytical gradient of LSE smoothed max overlap."""
    overlaps = _corr_overlaps(h, n, dx)
    s = overlaps - overlaps.max()
    w = np.exp(beta * s)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        l = k - (n - 1)
        if l == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif l > 0:
            p1 = np.zeros(n); p1[:n-l] = 1.0 - h[l:]
            p2 = np.zeros(n); p2[l:] = h[:n-l]
            grad += wk * (p1 - p2)
        else:
            lp = -l
            p1 = np.zeros(n); p1[lp:] = 1.0 - h[:n-lp]
            p2 = np.zeros(n); p2[:n-lp] = h[lp:]
            grad += wk * (p1 - p2)
    return grad

def _coord_step_pair(h, j, k, n, dx):
    """Optimize h[j], h[k] jointly maintaining h[j]+h[k] constant."""
    s = h[j] + h[k]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-14:
        return h
    # Evaluate two points to get linear coefficients
    h[j], h[k] = 0.0, s
    c0 = _corr_overlaps(h, n, dx)
    h[j], h[k] = s, 0.0
    c1 = _corr_overlaps(h, n, dx)
    h[j], h[k] = h[j], h[k]  # already s,0
    # Restore original
    h[j], h[k] = h[j], h[k]
    # c(x) where x=h[j], h[k]=s-x
    # c at x=0: c0, c at x=s: c1
    # c(x) = c0 + (c1-c0)/s * x  (linear in x)
    a = (c1 - c0) / s if s > 1e-14 else np.zeros(2*n-1)
    b = c0
    # Minimize max(a*x + b) for x in [lo, hi]
    best_x, best_val = lo, np.inf
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = np.max(a * m1 + b)
        v2 = np.max(a * m2 + b)
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    x_star = (lo + hi) / 2.0
    h[j] = x_star
    h[k] = s - x_star
    return h

        h = _project(h - lr * grad, n)
        c5 = np.max(_corr_overlaps(h, n, dx))
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 80:
            lr *= 0.5
            beta = min(beta * 1.3, 500.0)
            stuck = 0
            if lr < 1e-5:
                break
    return best_h, best_c5

def _run_coord(h, n, dx, t_end, seed):
    """Paired coordinate descent."""
    rng = np.random.RandomState(seed)
    h = _project(h.copy(), n)
    best_h = h.copy()
    best_c5 = np.max(_corr_overlaps(h, n, dx))
    idx = list(range(n))
    while time.time() < t_end:
        rng.shuffle(idx)
        for i in range(n // 2):
            j, k = idx[2 * i], idx[2 * i + 1]
            _coord_step_pair(h, j, k, n, dx)
        c5 = np.max(_corr_overlaps(h, n, dx))
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s * 0.92

    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    except Exception:
        n = 70
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0

    dx = 2.0 / n
    best_h = _project(h0.copy(), n)
    best_c5 = np.max(_corr_overlaps(best_h, n, dx))

    # Strategy 1: Gradient descent from initial
    h_g, c5_g = _run_grad(best_h.copy(), n, dx, t_end, seed)
    if c5_g < best_c5:
        best_c5, best_h = c5_g, h_g

    # Strategy 2: Coordinate descent
    h_c, c5_c = _run_coord(best_h.copy(), n, dx, t_end, seed + 1)
    if c5_c < best_c5:
        best_c5, best_h = c5_c, h_c

    # Strategy 3: Gradient from perturbed starts
    rng = np.random.RandomState(seed + 100)
    for attempt in range(3):
        if time.time() >= t_end:
            break
        h_pert = best_h + 0.05 * rng.randn(n)
        h_pert = _project(h_pert, n)
        h_g2, c5_g2 = _run_grad(h_pert, n, dx, t_end, seed + 200 + attempt)
        if c5_g2 < best_c5:
            best_c5, best_h = c5_g2, h_g2

    # Strategy 4: Coordinate from binary block starts
    for frac in [0.45, 0.55]:
        if time.time() >= t_end:
            break
        h_blk = np.zeros(n)
        block_len = int(n * frac)
        h_blk[:block_len] = 1.0
        h_blk = _project(h_blk, n)
        h_c2, c5_c2 = _run_coord(h_blk, n, dx, t_end, seed + 300)
        if c5_c2 < best_c5:
            best_c5, best_h = c5_c2, h_c2

    # Strategy 5: Final gradient polish
    h_final, c5_final = _run_grad(best_h.copy(), n, dx, t_end, seed + 999)
    if c5_final < best_c5:
        best_c5, best_h = c5_final, h_final

    # Ensure constraints
    best_h = _project(best_h, n)
    best_c5 = np.max(_corr_overlaps(best_h, n, dx))

    return best_h, best_c5, n
