import numpy as np
import time

def proj_sum(h, target_sum):
    N = len(h)
    lo = np.min(h) - target_sum - 1.0
    hi = np.max(h) + target_sum + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_c5(h, dx, temperature=50.0):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_vals = corr * dx
    c_max = np.max(c_vals)
    exp_vals = np.exp(temperature * (c_vals - c_max))
    p = exp_vals / exp_vals.sum()
    grad = np.zeros(N)
    for m in range(N):
        s = 0.0
        for k in range(2 * N - 1):
            pt = p[k]
            if pt < 1e-12:
                continue
            t = k - (N - 1)
            if 0 <= m + t < N:
                s += pt * (1.0 - h[m + t])
            if 0 <= m - t < N:
                s -= pt * h[m - t]
        grad[m] = s * dx
    return grad

def optimize_from(h0, dx, target_sum, max_time, rng, temperature=50.0):
    N = len(h0)
    h = proj_sum(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 0.1
    start = time.time()
    no_improve = 0
    patience = 200
    while time.time() - start < max_time:
        c5 = compute_c5(h, dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            scale = 0.02 + 0.02 * (no_improve % 50) / 50.0
            h = proj_sum(h + rng.normal(0, scale, N), target_sum)
            no_improve = 0
            lr = 0.08
            continue
        grad = gradient_c5(h, dx, temperature)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            h = proj_sum(h + rng.normal(0, 0.02, N), target_sum)
            continue
        step = lr
        improved = False
        for _ in range(12):
            h_new = proj_sum(h - step * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5 - 1e-13:
                h = h_new
                improved = True
                break
            step *= 0.5
        if improved:
            lr = min(step * 1.3, 0.5)
        else:
            lr *= 0.5
            if lr < 1e-7:
                h = proj_sum(h + rng.normal(0, 0.03, N), target_sum)
                lr = 0.05
    return best_h, best_c5
