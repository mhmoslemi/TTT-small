import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr))

def smooth_max_val(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def c5_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return smooth_max_val(corr, T)

def c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(T * (corr - m))
    p = exps / np.sum(exps)
    A = np.convolve(p, np.ones(n), mode='valid')
    B = np.correlate(p, h[::-1], mode='valid')
    C = np.correlate(p, h, mode='valid')[::-1]
    return dx * (A - B - C)

def project_box_hyperplane(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def interpolate_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.arange(n_src) * 2.0 / n_src
    x_new = np.arange(n_dst) * 2.0 / n_dst
    return np.interp(x_new, x_old, h_src)

def optimize_smooth(h, n, T, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        return c5_obj(x, T)
    def grabj(x):
        f = c5_obj(x, T)
        g = c5_grad(x, T)
        return f, g
    res = minimize(grabj, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return project_box_hyperplane(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining_frac():
        return 1.0 - elapsed() / budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    h_init = project_box_hyperplane(h_init, n0)

    best_h = h_init.copy()
    best_n = n0
    best_c5 = compute_c5(h_init)

    resolutions = sorted(set([n0, 50, 80, 100, 140, 200]))
    T_list = [8.0, 15.0, 30.0, 60.0, 120.0]

    for n in resolutions:
        if remaining_frac() < 0.25:
            break
        h_work = interpolate_h(h_init, n0, n)
        h_work = project_box_hyperplane(h_work, n)

        for T in T_list:
            if remaining_frac() < 0.2:
                break
            try:
                h_out = optimize_smooth(h_work, n, T, maxiter=300)
                c5_val = compute_c5(h_out)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_out.copy()
                    best_n = n
                h_work = h_out
            except Exception:
                pass

        # SLSQP with explicit constraint at moderate T
        if remaining_frac() > 0.2 and n <= 140:
            def slobj(x):
                return c5_obj(x, 40.0)
            try:
                res = minimize(slobj, h_work, method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=[{'type': 'eq',
                                             'fun': lambda x: np.sum(x) - n / 2.0}],
                               options={'maxiter': 200, 'ftol': 1e-10})
                h_out = project_box_hyperplane(res.x, n)
                c5_val = compute_c5(h_out)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_out.copy()
                    best_n = n
            except Exception:
                pass

    # Nelder-Mead refinement on raw max (non-smooth)
    if remaining_frac() > 0.15:
        n_nm = best_n
        for attempt in range(6):
            if remaining_frac() < 0.12:
                break
            h_pert = best_h + 0.005 * rng.randn(best_n) * (1 + attempt * 0.5)
            h_pert = project_box_hyperplane(h_pert, best_n)
            try:
                res = minimize(lambda x: compute_c5(x), h_pert,
                               method='Nelder-Mead',
                               options={'maxiter': 800, 'xatol': 1e-9, 'fatol': 1e-12})
                h_out = project_box_hyperplane(res.x, best_n)
                c5_val = compute_c5(h_out)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_out.copy()
                    best_n = best_n
            except Exception:
                pass

    # If we have time, try a different starting structure at best_n
    if remaining_frac() > 0.2:
        n_tm = max(best_n, 70)
        # Try a "block" pattern as alternative start
        h_alt = np.zeros(n_tm)
        block = n_tm // 4
        for start in range(0, n_tm, block * 2):
            h_alt[start:start + block] = 1.0
        h_alt = project_box_hyperplane(h_alt, n_tm)
        for T in [15.0, 40.0]:
            if remaining_frac() < 0.15:
                break
            try:
                h_out = optimize_smooth(h_alt, n_tm, T, maxiter=200)
                c5_val = compute_c5(h_out)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_out.copy()
                    best_n = n_tm
            except Exception:
                pass

    # Final projection and verification
    best_h = project_box_hyperplane(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
