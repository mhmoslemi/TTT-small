    weights /= np.sum(weights)

    # d(corr[k])/dh[j] = I(j in A_s) - I(j in A_s)*h[j+s] - I(j in B_s)*h[j-s]
    # where s = k - (n-1)

        if w < 1e-12:
            continue

        a_lo = max(0, -s)
        a_hi = min(n - 1, n - 1 - s)
        # j in B_s: 0 <= j < n and 0 <= j-s < n => max(0,s) <= j <= min(n-1, n-1+s)
        b_lo = max(0, s)
        b_hi = min(n - 1, n - 1 + s)
        
        if a_lo <= a_hi:
            grad[a_lo:a_hi+1] += w * 1.0
            grad[a_lo:a_hi+1] -= w * h[a_lo+s:a_hi+s+1]
        
        if b_lo <= b_hi:
            grad[b_lo:b_hi+1] -= w * h[b_lo-s:b_hi-s+1]
    
    return grad * dx

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_c5(h, dx):
    return float(np.max(compute_overlaps(h)) * dx)

def project_simplex(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gradient_smooth_max(h, tau):
    n = len(h)
    corr = compute_overlaps(h)
    m = np.max(corr)
    exp_corr = np.exp((corr - m) / tau)
    weights = exp_corr / np.sum(exp_corr)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        a_lo = max(0, -s)
        a_hi = min(n - 1, n - 1 - s)
        b_lo = max(0, s)
        b_hi = min(n - 1, n - 1 + s)
        
        if a_lo <= a_hi:
            grad[a_lo:a_hi+1] += w
            grad[a_lo:a_hi+1] -= w * h[a_lo+s:a_hi+s+1]
        if b_lo <= b_hi:
            grad[b_lo:b_hi+1] -= w * h[b_lo-s:b_hi-s+1]
    return grad

def binary_local_search(h_bin, dx, n_sweeps=10, seed=42):
    n = len(h_bin)
    target = n // 2
    rng = np.random.RandomState(seed)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    
    best_c5 = evaluate_c5(h_bin, dx)
    best_h = h_bin.copy()
    
    for sweep in range(n_sweeps):
        improved = False
        # Shuffle for diversification
        ones_shuffled = ones[:]
        zeros_shuffled = zeros[:]
        rng.shuffle(ones_shuffled)
        rng.shuffle(zeros_shuffled)

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_c5(h, dx):
    return float(np.max(compute_overlaps(h)) * dx)

def project_simplex(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gradient_smooth_max(h, tau):
    n = len(h)
    corr = compute_overlaps(h)
    m = np.max(corr)
    exp_c = np.exp((corr - m) / tau)
    wts = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = wts[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        a_lo = max(0, -s)
        a_hi = min(n - 1, n - 1 - s)
        b_lo = max(0, s)
        b_hi = min(n - 1, n - 1 + s)
        if a_lo <= a_hi:
            grad[a_lo:a_hi+1] += w
            grad[a_lo:a_hi+1] -= w * h[a_lo+s:a_hi+s+1]
        if b_lo <= b_hi:
            grad[b_lo:b_hi+1] -= w * h[b_lo-s:b_hi-s+1]
    return grad

def pgd_optimize(h, dx, n_iter, tau0=0.08, tau_min=0.005, lr_init=0.004):

    target = n / 2.0
    h = project_simplex(np.clip(h.copy(), 0, 1), target)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    lr = lr_init
    vel = np.zeros(n)
    mu = 0.9

    for it in range(n_iter):
        tau = max(tau_min, tau0 * (0.997 ** it))
        g = gradient_smooth_max(h, tau)
        vel = mu * vel - lr * g
        h_new = h + vel
        h_new = project_simplex(h_new, target)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Adaptive lr
        if c5_new > best_c5 + 1e-6:
            lr *= 0.85
            vel *= 0.5
        elif it % 30 == 29:
            lr = min(lr * 1.05, 0.02)
        h = h_new
        if lr < 1e-7:
            break
    return best_h, best_c5

def to_binary(h):
    n = len(h)
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def binary_local_search(h_bin, dx, n_sweeps=8, seed=42):
    n = len(h_bin)
    rng = np.random.RandomState(seed)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()
    best_c5 = evaluate_c5(h_bin, dx)
    best_h = h_bin.copy()

    for sweep in range(n_sweeps):
        improved = False
        ones_s = ones[:]
        zeros_s = zeros[:]
        rng.shuffle(ones_s)
        rng.shuffle(zeros_s)
        for j in zeros_s:
            for k in ones_s:
                if j == k:
                    continue
                h_bin[j] = 1.0
                h_bin[k] = 0.0
                c5 = evaluate_c5(h_bin, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h_bin.copy()
                    improved = True
                    ones.remove(k)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(k)
                    break
                h_bin[j] = 0.0
                h_bin[k] = 1.0
            if improved:
                improved = False
        if not improved:
            break
    return best_h, best_c5

def random_init(n, seed):
    rng = np.random.RandomState(seed)
    mode = seed % 4
    x = np.arange(n, dtype=float)
    if mode == 0:
        c = n * rng.uniform(0.3, 0.7)

        h = np.exp(-0.5 * ((x - c) / w) ** 2)
    elif mode == 1:
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n * rng.uniform(0.65, 0.85)
        w = n * rng.uniform(0.1, 0.2)
        h = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)
    elif mode == 2:

    else:
        h = rng.uniform(0.2, 0.8, n)
        h = np.convolve(h, np.ones(5) / 5, mode='same')
    h = np.clip(h, 0, 1)
    return project_simplex(h, n / 2.0)

    x_o = np.arange(n_orig) / n_orig
    x_n = np.arange(n_new) / n_new
    h_new = np.interp(x_n, x_o, h_orig)
    return project_simplex(np.clip(h_new, 0, 1), n_new / 2.0)
