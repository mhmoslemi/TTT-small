    configs = []
    if h_init is not None:
        configs.append(h_init.copy())
    
    # Structured starts for various n
    for n in [40, 50, 60, 70, 80, 100, 120]:
        if h_init is not None and n == n_init:
            continue
        # Block: 1 in middle
        h = np.zeros(n)
        h[n//4:3*n//4] = 1.0
        h = make_feasible(h, n)
        configs.append(h)
        # Alternating blocks
        h2 = np.zeros(n)
        block = max(1, n // 8)
        for start in range(0, n, 2 * block):
            h2[start:start+block] = 1.0
        h2 = make_feasible(h2, n)
        configs.append(h2)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    def anneal_run(h, n, T0, alpha, iters_per_T, max_rounds, budget_frac=0.75):
        nonlocal best_h, best_c5
        h = h.copy()
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        T = T0
        for rnd in range(max_rounds):
            if time_left() < budget_frac * 5:
                break
            for _ in range(iters_per_T):
                if time_left() < budget_frac * 5:
                    break
                h_try = anneal_perturb(h, n, rng)
                c5_try = compute_c5(h_try)
                d = c5_try - c5_cur
                if d < 0 or rng.random() < np.exp(-d / T):
                    h = h_try
                    c5_cur = c5_try
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
            T *= alpha
            if T < 0.001:
                break
        return h, c5_cur
    
    # Phase 1: Anneal from each config
    for idx, h_cfg in enumerate(configs):
        if time_left() < 10:
            break
        n_cfg = len(h_cfg)
        # Longer runs for smaller n
        iters = 20 if n_cfg <= 80 else 10
        max_r = int(800 * min(1.0, 80.0 / n_cfg))
        T0 = 0.03 + idx * 0.01  # Vary starting temperature
        alpha = 0.99
        anneal_run(h_cfg, n_cfg, T0, alpha, iters, max_r)
    
    # Phase 2: Restart and refine from best found
    if best_h is not None and time_left() > 20:
        n_best = len(best_h)
        for restart in range(5):
            if time_left() < 15:
                break
            h_r = best_h.copy()
            # Perturb from best
            for _ in range(20):
                i = int(rng.integers(n_best))
                j = int(rng.integers(n_best))
                if i != j:
                    h_r[i], h_r[j] = h_r[j], h_r[i]
            h_r = make_feasible(h_r, n_best)
            anneal_run(h_r, n_best, 0.02, 0.992, 15, 400)
    
    # Phase 3: Coordinate descent polish on best
    if best_h is not None and time_left() > 10:
        h = best_h.copy()
        n = len(h)
        c5_cur = compute_c5(h)
        deltas = [0.05, -0.05, 0.02, -0.02, 0.1, -0.1, 0.01, -0.01]
        for sweep in range(100):
            if time_left() < 5:
                break
            improved = False
            order = rng.permutation(n)
            for i in order:
                if time_left() < 5:
                    break
                orig = h[i]
                for d in deltas:
                    h[i] = orig + d
                    h = make_feasible(h, n)
                    c5_try = compute_c5(h)
                    if c5_try < c5_cur - 1e-12:
                        c5_cur = c5_try
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
                    h[i] = orig
                    h = make_feasible(h, n)
                h[i] = orig
                h = make_feasible(h, n)
            if not improved:
                break
    
    # Final feasibility check
    if best_h is not None:
        n = len(best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        s = best_h.sum()
        target = n / 2.0
        if abs(s - target) > 1e-10:
            best_h = make_feasible(best_h, n)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
