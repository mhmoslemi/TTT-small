```python
import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_max_overlap(h, temperature=50.0):
    n = len(h)
    dx = 2.0 / n
    f_vals = np.correlate(h, 1 - h, mode="full")
    
    # Softmax weights for smooth max
    max_f = np.max(f_vals)
    exp_args = (f_vals - max_f) * temperature
    w = np.exp(exp_args)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.zeros_like(f_vals)
        w[np.argmax(f_vals)] = 1.0
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-8:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += w[k] * (1 - h[s:])
            grad[s:] -= w[k] * h[:n-s]
        else:
            a = -s
            grad[a:] += w[k] * (1 - h[:n-a])
            grad[:n-a] -= w[k] * h[a:]
    
    return grad * dx

def project_onto_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""
    h = np.clip(h, 0, 1)
    # Adjust sum
    current = h.sum()
    if current != target_sum:
        # Simple: add uniform adjustment, then clip
        diff = target_sum - current
        h += diff / len(h)
        h = np.clip(h, 0, 1)
        # If still not matching, adjust non-saturated values
        for _ in range(10):
            diff = target_sum - h.sum()
            if abs(diff) < 1e-10:
                break
            mask = (h > 1e-8) & (h < 1 - 1e-8)
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def optimize_h(h0, budget_s, n_target=None):
    """Optimize h to minimize C5."""
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy().astype(float)
    h = project_onto_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Adam optimizer
    lr = 0.01
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start_time = time.time()
    
    # Adaptive temperature
    temp = 20.0
    
    while time.time() - start_time < budget_s:
        t += 1
        grad = gradient_max_overlap(h, temperature=temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h = project_onto_feasible(h_new, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            # Can try to increase temperature (focus more on max)
            temp = min(temp * 1.05, 200.0)
        else:
            # Slight decrease in temperature
            temp = max(temp * 0.99, 5.0)
        
        # Periodic restart with small perturbation
        if t % 500 == 0 and t > 1000:
            if compute_c5(h) > best_c5 * 1.01:
                h = best_h + 0.01 * np.random.randn(n)
                h = project_onto_feasible(h, target_sum)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = 0.005
    
    return best_h, best_c5
```

```python
        h_opt, c5_opt = optimize_adam(pert, alloc_r, lr=0.005, temp_start=30.0, seed=seed + 10 + r)
        if c5_opt < best_c5_overall:
            best_overall, best_c5_overall = h_opt, c5_opt
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        n_final = len(best_overall)
        h_opt, c5_opt = optimize_adam(best_overall, remaining - 5, lr=0.001, temp_start=80.0, seed=seed + 99)
        if c5_opt < best_c5_overall:
            best_overall, best_c5_overall = h_opt, c5_opt
    final_n = len(best_overall)
    best_overall = project_feasible(best_overall, final_n / 2.0)
    best_c5_overall = compute_c5(best_overall)
    return best_overall, best_c5_overall, final_n
```
