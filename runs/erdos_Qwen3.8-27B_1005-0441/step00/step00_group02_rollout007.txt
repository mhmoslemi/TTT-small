    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    return float(np.max(corr) * dx)

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, s):
    n = len(h)
    g = np.zeros(n)
    if 0 <= s < n:
        idx = np.arange(n - s)
        g[idx] += (1.0 - h[idx + s])
        g[idx + s] += -h[idx]
    elif -n < s < 0:
        t = -s
        idx = np.arange(t, n)
        g[idx] += (1.0 - h[idx - t])
        g[idx - t] += -h[idx]
    return g

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h
