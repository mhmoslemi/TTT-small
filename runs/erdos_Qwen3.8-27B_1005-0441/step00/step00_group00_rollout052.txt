import numpy as np
from scipy.optimize import minimize
import time

_target_sum = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(overlaps)
    return m + (1.0 / tau) * np.log(np.sum(np.exp(tau * (overlaps - m))))

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(overlaps)
    exps = np.exp(tau * (overlaps - m))
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-10:
            continue
        shift = s - (n - 1)
        if shift >= 0:
            a = n - shift
            grad[:a] += w * (1.0 - h[shift:]) * dx
            grad[shift:] += w * (-h[:a]) * dx
        else:
            s2 = -shift
            a = n - s2
            grad[s2:] += w * (1.0 - h[:a]) * dx
            grad[:a] += w * (-h[s2:]) * dx
    return grad

def sum_constraint(h):
    return np.sum(h) - _target_sum

def sum_constraint_jac(h):
    return np.ones(len(h))

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt == 0:
            mask = np.ones(len(h), dtype=bool)
            cnt = len(h)
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def hill_climb(h, target, n_iter=500, rng=np.random.RandomState(0)):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    for _ in range(n_iter):
        k = rng.randint(1, 4)
        idxs = rng.choice(n, size=k, replace=False)
        h_try = best.copy()
        for idx in idxs:
            h_try[idx] += rng.uniform(-0.05, 0.05)
        h_try = project_feasible(h_try, target)
        val = compute_c5(h_try)
        if val < best_val - 1e-10:
            best = h_try
            best_val = val
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    global _target_sum
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h_init = np.array(globals()['initial_h_values'], dtype=float)
    n = len(h_init)
    target = n / 2.0
    _target_sum = target

    best_h = project_feasible(h_init, target)
    best_c5 = compute_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]

    # Phase 1: Progressive SLSQP from initial
    for tau in [5, 15, 30, 50, 100, 200, 500]:
        if time.time() - t0 > budget_s - 180:
            break
        try:
            res = minimize(smooth_max_obj, best_h, args=(tau,),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         jac=smooth_max_grad,
                         options={'maxiter': 600, 'ftol': 1e-14})
            if res.x is not None:
                c5 = compute_c5(res.x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = project_feasible(res.x, target)
        except Exception:
            pass

    # Phase 2: Hill climb on current best
    if time.time() - t0 < budget_s - 120:
        best_h, best_c5 = hill_climb(best_h, target, n_iter=300, rng=rng)

    # Phase 3: Random restarts
    n_restarts = max(5, int((budget_s - 60 - (time.time() - t0)) / 15))
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 60:
            break
        strategy = r % 4
        if strategy == 0:
            h_r = best_h + rng.randn(n) * 0.03
        elif strategy == 1:
            h_r = best_h + rng.randn(n) * 0.1
        elif strategy == 2:
            h_r = (best_h + np.roll(best_h, 1)) / 2.0 + rng.randn(n) * 0.05
        else:
            h_r = np.clip(best_h * rng.uniform(0.3, 1.7, size=n), 0, 1)
        h_r = project_feasible(h_r, target)
        init_c5 = compute_c5(h_r)

        for tau in [100, 500]:
            if time.time() - t0 > budget_s - 60:
                break
            try:
                res = minimize(smooth_max_obj, h_r, args=(tau,),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             jac=smooth_max_grad,
                             options={'maxiter': 400, 'ftol': 1e-14})
                if res.x is not None:
                    c5 = compute_c5(res.x)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = project_feasible(res.x, target)
            except Exception:
                pass

        # Quick hill climb on improved solution
        if compute_c5(best_h) < init_c5 + 1e-8 and time.time() - t0 < budget_s - 60:
            best_h, best_c5 = hill_climb(best_h, target, n_iter=100, rng=rng)

    # Phase 4: Try upsampled n
    if time.time() - t0 < budget_s - 200:
        for n_new in [90, 180]:
            if time.time() - t0 > budget_s - 100:
                break
            x_old = (np.arange(n) + 0.5) * (2.0 / n)
            x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_feasible(h_up, n_new / 2.0)
            _target_sum = n_new / 2.0
            bounds_new = [(0.0, 1.0)] * n_new
            cons_new = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]

            for tau in [30, 100, 500]:
                if time.time() - t0 > budget_s - 80:
                    break
                try:
                    res = minimize(smooth_max_obj, h_up, args=(tau,),
                                 method='SLSQP', bounds=bounds_new, constraints=cons_new,
                                 jac=smooth_max_grad,
                                 options={'maxiter': 500, 'ftol': 1e-14})
                    if res.x is not None:
                        c5 = compute_c5(res.x)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = project_feasible(res.x, n_new / 2.0)
                            n = n_new
                            _target_sum = n / 2.0
                except Exception:
                    pass

    # Final polish
    if time.time() - t0 < budget_s - 10:
        _target_sum = len(best_h) / 2.0
        try:
            bounds_f = [(0.0, 1.0)] * len(best_h)
            cons_f = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
            res = minimize(smooth_max_obj, best_h, args=(1000,),
                         method='SLSQP', bounds=bounds_f, constraints=cons_f,
                         jac=smooth_max_grad,
                         options={'maxiter': 500, 'ftol': 1e-15})
            if res.x is not None:
                c5 = compute_c5(res.x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = res.x
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
