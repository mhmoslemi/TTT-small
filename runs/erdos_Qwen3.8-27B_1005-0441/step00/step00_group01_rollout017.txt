def run(seed=42, budget_s=1100, **kwargs):
    global _glob_n
    np.random.seed(seed)
    t0 = time.time()
    bh = None
    bc = np.inf
    bn = 0

    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    except NameError:
        h0 = make_sin(47, 1, 0.0, 0.3)
    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    c0 = c5_value(h0)
    bh, bc, bn = h0.copy(), c0, n0

    # Phase 1: PGD at initial resolution
    tl = budget_s * 0.2 - (time.time() - t0)
    if tl > 5:
        ho, co = pgd(h0, 5000, 0.05, 300, 20, tl)
        if co < bc:
            bh, bc, bn = ho, co, len(ho)

    # Phase 2: Upsample and optimize at higher n
    for nt in [64, 96, 128, 192, 256]:
        if time.time() - t0 > budget_s * 0.65:
            break
        ht = project_feasible(upsample_to(bh, nt), nt)
        ta = (budget_s * 0.65 - (time.time() - t0)) / 5.0
        if ta < 3:
            continue
        ho, co = pgd(ht, 3000, 0.03, 100, 5, min(ta, 120.0))
        if co < bc:
            bh, bc, bn = ho, co, nt

    # Phase 3: Try alternative initial patterns
    if time.time() - t0 < budget_s * 0.55:
        pats = [make_sin(bn, 1, 0.0, 0.25), make_sin(bn, 2, 1.0, 0.2),
                make_sin(bn, 1, 2.0, 0.35), make_bin(bn, seed + 1), make_bin(bn, seed + 2)]
        for p in pats:
            if time.time() - t0 > budget_s * 0.7:
                break
            ta = budget_s * 0.7 - (time.time() - t0)
            if ta < 3:
                break
            ho, co = pgd(p, 2000, 0.03, 50, 5, min(ta * 0.25, 30.0))
            if co < bc:
                bh, bc, bn = ho, co, len(ho)

    # Phase 4: Fine polish with smaller lr
    tl = budget_s - (time.time() - t0) - 3.0
    if tl > 10 and bh is not None:
        ho, co = pgd(bh, 5000, 0.01, 50, 3, tl * 0.4)
        if co < bc:
            bh, bc = ho, co

    # Phase 5: SLSQP polish
    tl2 = budget_s - (time.time() - t0) - 2.0
    if tl2 > 10 and bh is not None:
        _glob_n = len(bh)
        bnds = [(0.0, 1.0)] * len(bh)
        cstr = [{'type': 'eq', 'fun': _cnst}]
        try:
            res = optimize.minimize(c5_value, bh, method='SLSQP',
                                    bounds=bnds, constraints=cstr,
                                    options={'maxiter': 500, 'ftol': 1e-14})
            hs = project_feasible(res.x, len(bh))
            cs = c5_value(hs)
            if cs < bc:
                bh, bc, bn = hs, cs, len(hs)
        except Exception:
            pass

    # Final verification
    bh = project_feasible(bh, bn)
    bc = c5_value(bh)
    return bh, bc, bn
