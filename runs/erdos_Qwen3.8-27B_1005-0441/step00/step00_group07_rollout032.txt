import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    s = idx - (n - 1)
    idx_arr = np.arange(n)
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        ip = idx_arr + s
        vp = (ip >= 0) & (ip < n)
        grad[vp] += 1.0 - h[ip[vp]]
        im = idx_arr - s
        vm = (im >= 0) & (im < n)
        grad[vm] -= h[im[vm]]
    return grad * dx

def project(h, S):
    lo, hi = -3.0, 3.0
    for _ in range(120):
        mid = 0.5 * (lo + hi)
        t = np.clip(h - mid, 0.0, 1.0)
        if t.sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgrad_descent(h, S, duration, rng):
    h = h.copy()
    t0 = time.time()
    best = compute_c5(h)
    best_h = h.copy()
    gamma = 0.02
    for it in range(500000):
        if time.time() - t0 > duration:
            break
        g = c5_subgrad(h)
        alpha = gamma / (1.0 + 0.005 * it)
        h = h - alpha * g
        h = project(h, S)
        c = compute_c5(h)
        if c < best:
            best = c
            best_h = h.copy()
    return best_h, best

def coord_descent(h, S, duration, rng):
    h = h.copy()
    n = len(h)
    t0 = time.time()
    best_c = compute_c5(h)
    no_improve = 0
    for it in range(5000000):
        if time.time() - t0 > duration:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        scale = 0.04 + 0.06 * (1.0 - min(1.0, it / 200000.0))
        d = float(rng.uniform(-1.0, 1.0)) * scale
        hi_v = h[i] + d
        hj_v = h[j] - d
        if hi_v < 0.0 or hi_v > 1.0 or hj_v < 0.0 or hj_v > 1.0:
            continue
        h[i] = hi_v
        h[j] = hj_v
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            no_improve = 0
        else:
            h[i] -= d
            h[j] += d
            no_improve += 1
            if no_improve > 50000:
                scale *= 0.9
                no_improve = 0
    return h, best_c

def full_hill_climb(h, S, duration, rng):
    h = h.copy()
    n = len(h)
    t0 = time.time()
    best_c = compute_c5(h)
    for it in range(50000000):
        if time.time() - t0 > duration:
            break
        i = int(rng.integers(0, n))
        # Try to adjust h[i] and compensate at another index
        j = int(rng.integers(0, n))
        if i == j:

        d = float(rng.uniform(-0.03, 0.03))
        hi_v = h[i] + d
        hj_v = h[j] - d
        if hi_v < 0.0 or hi_v > 1.0 or hj_v < 0.0 or hj_v > 1.0:

        h[i] = hi_v
        h[j] = hj_v
        c = compute_c5(h)
        if c < best_c:
            best_c = c
        else:
            h[i] -= d
            h[j] += d
    return h, best_c
