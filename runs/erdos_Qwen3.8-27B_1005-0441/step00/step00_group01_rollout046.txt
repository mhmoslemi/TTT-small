import scipy.optimize

def eval_c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx

    return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha

            break

    return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha

def project_h(h, n):
    target = n / 2.0

            break
