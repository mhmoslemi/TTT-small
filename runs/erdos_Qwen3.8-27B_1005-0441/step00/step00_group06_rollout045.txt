    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_correlate(h)) * dx)

def overlap_grad(h, k_idx):
    n = len(h)
    s = k_idx - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    v1 = (j + s >= 0) & (j + s < n)
    g[v1] += 1.0 - h[j[v1] + s]
    v2 = (j - s >= 0) & (j - s < n)
    g[v2] -= h[j[v2] - s]
    return g

def project_box_sum(h, ts):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def lse_obj(x, dx, beta):
    v = compute_correlate(x) * dx
    m = v.max()
    return m + np.log(np.sum(np.exp(beta * (v - m)))) / beta

def lse_grad(x, dx, beta):
    n = len(x)
    v = compute_correlate(x) * dx
    m = v.max()
    w = np.exp(beta * (v - m))
    w /= w.sum()
    idx = np.where(w > 1e-12)[0]
    g = np.zeros(n)
    for k in idx:
        g += w[k] * overlap_grad(x, k)
    return g * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    _ihv = globals().get('initial_h_values')
    if _ihv is not None:
        h0 = np.array(_ihv, dtype=float)
    else:
        h0 = np.full(95, 0.5)

    n = len(h0)
    dx = 2.0 / n
    ts = 0.5 * n
    h0 = project_box_sum(h0, ts)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Subgradient descent (broad)
    h = best_h.copy()
    lr = 0.04
    for it in range(10000):
        if rem() < 250:
            break
        vals = compute_correlate(h) * dx
        mi = int(np.argmax(vals))
        c5 = vals[mi]
        if c5 < best_c5 - 1e-10:
            best_c5, best_h = c5, h.copy()
        g = overlap_grad(h, mi) * dx
        h2 = project_box_sum(h - lr * g, ts)
        c52 = compute_c5(h2, dx)
        if c52 <= c5 + 1e-8:
            h = h2
        else:
            lr = max(1e-5, lr * 0.85)

    # Phase 2: Smooth LSE + SLSQP (precise)
    if rem() > 120:
        for beta in [12, 40, 100, 250]:
            if rem() < 120:
                break
            try:
                res = minimize(
                    lse_obj, best_h.copy(), method='SLSQP',
                    args=(dx, beta), jac=lambda x: lse_grad(x, dx, beta),
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                    options={'maxiter': 600, 'ftol': 1e-13}
                )
                ht = project_box_sum(np.clip(res.x, 0.0, 1.0), ts)
                c5t = compute_c5(ht, dx)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
            except Exception:
                pass

    # Phase 3: Second SLSQP round with higher beta for polish
    if rem() > 80:
        for beta in [200, 500]:
            if rem() < 80:
                break
            try:
                res = minimize(
                    lse_obj, best_h.copy(), method='SLSQP',
                    args=(dx, beta), jac=lambda x: lse_grad(x, dx, beta),
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                    options={'maxiter': 800, 'ftol': 1e-14}
                )
                ht = project_box_sum(np.clip(res.x, 0.0, 1.0), ts)
                c5t = compute_c5(ht, dx)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
            except Exception:
                pass

    # Phase 4: Fine subgradient polish
    if rem() > 20:
        h = best_h.copy()
        lr = 0.004
        for it in range(8000):
            if rem() < 15:
                break
            vals = compute_correlate(h) * dx
            mi = int(np.argmax(vals))
            c5 = vals[mi]
            if c5 < best_c5 - 1e-11:
                best_c5, best_h = c5, h.copy()
            g = overlap_grad(h, mi) * dx
            h2 = project_box_sum(h - lr * g, ts)
            if compute_c5(h2, dx) <= c5 + 1e-10:
                h = h2
            else:
                lr = max(1e-7, lr * 0.92)

    return (best_h, best_c5, n)
