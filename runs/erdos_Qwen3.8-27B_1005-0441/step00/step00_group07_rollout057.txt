def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, n):
    # Project onto {0 <= h <= 1, sum(h) = n/2}
    h = np.clip(h, 0, 1)
    # Adjust to get sum = n/2
    target = n / 2
    # Binary search for the shift
    lo, hi = -1, 1
    for _ in range(60):
        mid = (lo + hi) / 2
        h_shifted = np.clip(h + mid, 0, 1)
        if np.sum(h_shifted) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    # Final adjustment
    diff = target - np.sum(h)
    # Distribute the difference to entries not at bounds
    mask = (h > 0) & (h < 1)
    if np.sum(mask) > 0:
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def subgradient_step(h, n, dx, alpha):
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)  # lag
    
    # Compute subgradient for this lag
    grad = np.zeros(n)
    if k >= 0:
        # corr_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])

        for j in range(k, n):
            grad[j] -= h[j - k]

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def corr_gradient(h, n, k):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    else:
        m = -k
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    return g

def smooth_gradient_descent(h, n, t_end, seed, T_init=80.0, T_max=800.0):
    rng = np.random.RandomState(seed)
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    T = T_init
    alpha = 2.0 / (T * 2.0 / n)
    stuck = 0
    max_stuck = 400

    while time.time() < t_end:
        corr = np.correlate(h, 1.0 - h, mode="full")
        scaled = corr * (2.0 / n)
        mx = np.max(scaled)
        w = np.exp(scaled - mx)
        sw = np.sum(w)
        active_idx = np.where(w > 0.02 * sw)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(w)])
        w_a = w[active_idx]
        w_a /= w_a.sum()
        grad = np.zeros(n)
        for ii in range(len(active_idx)):
            k = int(active_idx[ii] - (n - 1))
            grad += w_a[ii] * corr_gradient(h, n, k)
        grad *= (2.0 / n) / T

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)

        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            stuck = 0
            alpha = min(alpha * 1.03, 3.0)
        else:
            stuck += 1
            if stuck > max_stuck:
                alpha *= 0.9
                T = min(T * 1.5, T_max)
                alpha = min(2.0 / (T * 2.0 / n), 3.0)
                stuck = 0

        h = h_new

    return best_h, best_c5

def local_search(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    deltas = [0.02, 0.05, 0.1, 0.2, -0.02, -0.05, -0.1, -0.2]
    while time.time() < t_end:
        improved = False
        for i in range(n):
            for d in deltas:
                h_try = best_h.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try, n)
                if c5_try < best_c5 - 1e-10:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h
