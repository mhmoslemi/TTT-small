import numpy as np

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)
    # Fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(h)  # increase smallest first
            for i in order:
                add = min(1.0 - h[i], diff)
                if add > 0:
                    h[i] += add
                    diff -= add
                    if diff < 1e-12:
                        break
        else:
            order = np.argsort(-h)  # decrease largest first
            for i in order:

                    diff += rem

    cv = corr_full(h)
    c5 = cv.max() * 2.0 / n
    idx = np.argsort(cv)[::-1][:top_k]
    wts = cv[idx].copy()
    wts /= wts.sum()
    g = np.zeros(n)
    for w, j in zip(wts, idx):
        k = int(j) - (n - 1)
        if k >= 0:
            m = n - k
            g[:m] -= w * h[k:]
            g[k:] += w * (1.0 - h[:m])
        else:
            m = -k
            g[m:] -= w * h[:n-m]
            g[:n-m] += w * (1.0 - h[m:])
    return c5, g

def optimize_subgrad(h0, n, t_end, top_k=5):

    best_h, best_c5 = h.copy(), compute_c5(h)
    step = 0.01
    best_step = 0.01
    st = 0
    while time.time() < t_end:
        c5, g = subgrad_step(h, n, top_k)
        h_new = project_feasible(h - step * g, n)
        nc = compute_c5(h_new)
        if nc < c5:
            h = h_new
            st = 0
            step = min(step * 1.01, 0.1)
        else:

            st += 1
            if st > 200 or step < 1e-6:
                break

            best_h = h_new.copy()
            best_step = step
        elif st > 100:
            step = best_step
            st = 50
    return best_h, best_c5

def optimize_bb_local(h0, n, t_end):
