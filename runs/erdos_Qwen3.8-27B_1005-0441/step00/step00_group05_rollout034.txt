    h = np.clip(h, 0, 1)

import numpy as np
import math, time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    target = n * 0.5
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    free = (h > 1e-10) & (h < 1.0 - 1e-10)
    nf = int(np.sum(free))
    if nf > 0:
        h[free] += diff / nf
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if diff > 1e-10:
        for i in np.argsort(1.0 - h):
            a = min(diff, 1.0 - h[i])
            h[i] += a; diff -= a
            if diff < 1e-12: break
    elif diff < -1e-10:
        for i in np.argsort(h):
            s = min(-diff, h[i])
            h[i] -= s; diff += s
            if diff > -1e-12: break
    return h

def coord_descend(h, max_time, t0, n, cur_c5, best_h, best_c5, rng):
    """Coordinate descent: try setting each coordinate to 0, 0.25, 0.5, 0.75, 1"""
    for sweep in range(500):
        if time.time() - t0 > max_time: break
        improved = False
        for j in range(n):
            if time.time() - t0 > max_time: break
            h_save = h.copy()
            c_best = cur_c5
            h_best = h_save.copy()
            for val in (0.0, 0.25, 0.5, 0.75, 1.0):
                h_t = h_save.copy()
                h_t[j] = val
                h_t = project_h(h_t)
                c = compute_c5(h_t)
                if c < c_best:
                    c_best = c
                    h_best = h_t.copy()
            if c_best < cur_c5 - 1e-13:
                h = h_best
                cur_c5 = c_best
                improved = True
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
        if not improved:
            break
    return h, cur_c5, best_h, best_c5

def sa_optimize(h, n, cur_c5, best_h, best_c5, t0, t_end, rng):
    """Simulated annealing local search"""
    T = 0.004
    stall = 0
    while time.time() - t0 < t_end:
        j = rng.integers(n)
        d = rng.normal(0.0, 0.04)
        h_new = h.copy()
        h_new[j] += d
        h_new = project_h(h_new)
        c_new = compute_c5(h_new)
        if c_new <= cur_c5:
            h = h_new; cur_c5 = c_new; stall = 0
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            if -(c_new - cur_c5) / T > math.log(max(rng.random(), 1e-15)):
                h = h_new; cur_c5 = c_new
            stall += 1
            if stall > 300:
                T = min(T * 1.5, 0.008)
                stall = 0
        T = max(T * 0.99997, 1e-7)
    return h, cur_c5, best_h, best_c5
