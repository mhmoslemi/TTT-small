def compute_grad_max(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    max_val = c.max()
    
    # Use all lags within a fraction of max (for subgradient)
    threshold = max_val * 0.8  # top 20%
    active = c >= threshold
    n_active = active.sum()
    if n_active == 0:
        active[np.argmax(c)] = True
        n_active = 1

            if 0 <= m + lag < n:
                g += (1 - h[m + lag])

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def grad_max_overlap(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    max_val = c.max()
    if max_val <= 0:
        return np.zeros(n)
    threshold = max_val - 0.15 * max_val
    active = c >= threshold
    if not active.any():
        active[np.argmax(c)] = True
    n_act = active.sum()
    
    grad = np.zeros(n)
    for k_idx in np.where(active)[0]:
        lag = int(k_idx) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            # First term: m in [0, n-1-lag], add (1-h[m+lag])
            grad[0:n-lag] += (1.0 - h[lag:n])
            # Second term: m in [lag, n-1], subtract h[m-lag]
            grad[lag:n] -= h[0:n-lag]
        else:
            la = -lag
            # First term: m in [la, n-1], add (1-h[m-lag]) = (1-h[m+la])
            grad[la:n] += (1.0 - h[0:n-la])
            # Second term: m in [0, n-1-la], subtract h[m-lag] = h[m+la]
            grad[0:n-la] -= h[la:n]
    
    return grad / n_act

    # Log-sum-exp for numerical stability
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    weights = exp_c / exp_c.sum()
    smooth_val = c_max + np.log(exp_c.sum()) / alpha
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if weights[k_idx] < 1e-6:
            continue
        lag = k_idx - (n - 1)
        w = weights[k_idx]

        elif lag > 0:

            grad[lag:n] -= w * h[0:n-lag]
        else:
            la = -lag
            grad[la:n] += w * (1.0 - h[0:n-la])
            grad[0:n-la] -= w * h[la:n]
    
    return smooth_val, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def smooth_max_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    s = exp_c.sum()
    weights = exp_c / s
    val = c_max + np.log(s) / alpha
    grad = np.zeros(n)
    active = np.where(weights > 1e-7)[0]
    for k_idx in active:
        lag = int(k_idx) - (n - 1)
        w = weights[k_idx]
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[0:n-lag] += w * (1.0 - h[lag:n])
            grad[lag:n] -= w * h[0:n-lag]
        else:
            la = -lag
            grad[la:n] += w * (1.0 - h[0:n-la])
            grad[0:n-la] -= w * h[la:n]
    return val, grad

def coord_refine(h, n, deadline):
    best_c5 = eval_c5(h)
    steps = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.03, -0.03]
    for _ in range(50):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for d in steps:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                c5_new = eval_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5_new
                    improved = True
                    break
    return h, best_c5

def optimize_one(h0, n, deadline, rng):
    h = project_feasible(h0.copy(), n)
    best = h.copy()
    best_c5 = eval_c5(h)
    step = 0.04
    alpha = 12.0
    stall = 0
    while time.time() < deadline:
        c5_cur = eval_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best = h.copy()
            stall = 0
        else:
            stall += 1
        alpha = min(80.0, alpha * 1.003)
        _, g = smooth_max_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-11:
            break
        g /= gn
        improved = False
        for _ in range(10):
            h_new = project_feasible(h - step * g, n)
            c5_new = eval_c5(h_new)
            if c5_new < c5_cur - 1e-13:
                h = h_new
                improved = True
                break
            step *= 0.5
        if not improved:
            step = 0.04
            h = project_feasible(h + 0.02 * rng.randn(n), n)
            stall += 10
        else:
            step = min(0.08, step * 1.08)
        if stall > 300:
            h = project_feasible(h + 0.08 * rng.randn(n), n)
            step = 0.03
            stall = 0
    if time.time() < deadline:
        h, best_c5 = coord_refine(h, n, deadline)
        if best_c5 > eval_c5(best):
            pass
        if eval_c5(h) < best_c5:
            best_c5 = eval_c5(h)
            best = h.copy()
    return best, best_c5
