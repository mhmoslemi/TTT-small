    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def grad_at_lag(h, q):
    n = len(h)
    g = np.zeros(n)
    if q >= 0:
        g[:n - q] += 1.0 - h[q:]
        g[q:] -= h[:n - q]
    else:
        g[-q:] += 1.0 - h[:n + q]
        g[:n + q] -= h[-q:]
    return g


def optimize(h_init, end_time, seed):
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    lr = 0.04
    stag = 0
    total_iter_budget = max(10000, int((end_time - time.time()) / 0.0002))
    
    while time.time() < end_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        frac = min(1.0, total_iter_budget - 2000, 1.0)
        topk = 3 if stag < 200 else 1
        idx = np.argsort(corr)[-topk:]
        g = np.zeros(n)
        for i in range(topk):
            g += grad_at_lag(h, n - 1 - idx[i])
        g /= float(topk)
        
        h_new = project_onto_box_sum(h - lr * g, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stag = 0
        else:
            stag += 1
            if stag > 800:
                lr = max(lr * 0.6, 0.001)
                stag = 0
            if stag > 3000:
                h = best_h + rng.randn(n) * 0.05
                h = project_onto_box_sum(h, target)
                lr = 0.03
                stag = 0
        h = h_new
    
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    starts = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_box_sum(h0, n0 / 2.0)
        starts.append(h0)
    
    nw = 280
    x = np.linspace(0.0, 2.0, nw, endpoint=False)
    hw = 0.5 + 0.35 * np.sin(2.0 * np.pi * 3.0 * x / 2.0)
    hw = np.clip(hw, 0.0, 1.0)
    hw = project_onto_box_sum(hw, nw / 2.0)
    starts.append(hw)
    
    n_s = len(starts)
    phase1_dur = (budget_s - 100.0) * 0.55
    phase2_dur = (budget_s - 100.0) * 0.35
    
    for i, hs in enumerate(starts):
        seg_len = phase1_dur / n_s
        seg_end = t0 + (i + 1) * seg_len
        if time.time() >= seg_end:
            seg_end = time.time() + seg_len * 0.5
        h_opt, c5_opt = optimize(hs, seg_end, seed + i * 7)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = len(h_opt)
    
    if best_h is None:
        best_h = starts[-1].copy()
        best_c5 = compute_c5(best_h)
        best_n = len(best_h)
    
    phase2_start = t0 + phase1_dur
    if time.time() < phase2_start:
        time.sleep(max(0, phase2_start - time.time() - 2.0))
    
    p2_end = min(time.time() + phase2_dur, t0 + budget_s - 30.0)
    hh, cc = optimize(best_h, p2_end, seed + 200)
    if cc < best_c5:
        best_c5 = cc
        best_h = hh
    
    if time.time() < t0 + budget_s - 10.0:
        p3_end = t0 + budget_s - 10.0
        h3 = best_h + rng.randn(best_n) * 0.02
        h3 = project_onto_box_sum(h3, best_n / 2.0)
        hh3, cc3 = optimize(h3, p3_end, seed + 400)
        if cc3 < best_c5:
            best_c5 = cc3
            best_h = hh3
    
    return best_h, best_c5, best_n
