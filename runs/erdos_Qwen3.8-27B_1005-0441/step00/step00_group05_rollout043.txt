import numpy as np

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def eval_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def soft_gradient(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = np.max(c)
    w = np.exp(T * (c - cm))
    w /= w.sum()
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        ws = w[s]
        if ws < 1e-10:
            continue
        d = s - (n - 1)
        if d >= 0:
            cnt = n - d
            grad[:cnt] += ws * (1.0 - h[d:d + cnt])
            grad[d:d + cnt] -= ws * h[:cnt]
        else:
            d2 = -d
            cnt = n - d2
            grad[d2:d2 + cnt] += ws * (1.0 - h[:cnt])
            grad[:cnt] -= ws * h[d2:d2 + cnt]
    return grad, cm * (2.0 / n)
