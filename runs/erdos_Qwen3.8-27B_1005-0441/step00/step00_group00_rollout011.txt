    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if w < 1e-8:
            continue
        shift = k - (n - 1)

            # grad[0:n-shift] += w * (1 - h[shift:n]) * dx
            # grad[shift:n] += -w * h[0:n-shift] * dx
            m = n - shift
            grad[:m] += w * (1.0 - h[shift:]) * dx
            grad[shift:] += -w * h[:m] * dx

            grad[s:] += w * (1.0 - h[:m]) * dx
