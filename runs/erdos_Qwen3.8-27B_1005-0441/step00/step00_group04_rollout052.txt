def compute_smoothed_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    c_scaled = c * dx
    # Softmax weights
    # Shift to avoid overflow
    c_shifted = c_scaled - np.max(c_scaled)
    exp_c = np.exp(alpha * c_shifted)
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)  # shifts: -(n-1), ..., n-1
    
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        s = shifts[k]
        # dc_k/dh_j for all j
        # term 1: (1-h[j+s]) for valid j+s
        j1_max = n - s  # j+s < n => j < n-s
        j1_min = -s  # j+s >= 0 => j >= -s
        j1_min = max(0, j1_min)
        j1_max = min(n, j1_max)
        if j1_max > j1_min:
            grad[j1_min:j1_max] += w[k] * (1.0 - h[j1_min+s:j1_max+s])
        
        # term 2: -h[j-s] for valid j-s
        j2_min = s  # j-s >= 0 => j >= s
        j2_max = n + s  # j-s < n => j < n+s
        j2_min = max(0, j2_min)
        j2_max = min(n, j2_max)
        if j2_max > j2_min:
            grad[j2_min:j2_max] -= w[k] * h[j2_min-s:j2_max-s]
    
    return w, grad

    if time.time() - t0 < budget_s - 20:
        h0, c5v = optimize_phase(best_h, n, dx, ts, t0, budget_s, 1000.0, 0.0001, 40000)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h0.copy()

    # Binarize + refine
    if time.time() - t0 < budget_s - 20:
        h_bin, c5_bin, h_bo, c5_bo = try_binarize(best_h, n, ts, t0, budget_s)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if c5_bo < best_c5:
            best_c5 = c5_bo
            best_h = h_bo.copy()

    # Random restarts
    for trial in range(12):
        if time.time() - t0 > budget_s - 12:
            break
        scale = 0.03 + 0.02 * rng.rand()
        hp = best_h + rng.randn(n) * scale
        hp = project_feasible(hp, ts)
        hp, c5v = optimize_phase(hp, n, dx, ts, t0, budget_s, 30.0, 0.004, 12000)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hp.copy()
        # Also try binarized version
        if time.time() - t0 < budget_s - 12:
            hb = np.zeros(n)
            idx = np.argsort(hp)[::-1]
            hb[idx[:int(ts)]] = 1.0
            c5_tb = np.max(np.correlate(hb, 1.0 - hb, mode="full")) * dx
            if c5_tb < best_c5:
                best_c5 = c5_tb
                best_h = hb.copy()
            hb, c5_ro = optimize_phase(hb, n, dx, ts, t0, budget_s, 50.0, 0.002, 6000)
            if c5_ro < best_c5:
                best_c5 = c5_ro
                best_h = hb.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, ts)
    final_c5 = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx
    return (best_h, final_c5, n)
