    pos = np.empty((NP, n))
    vel = np.zeros((NP, n))
    pbest = np.empty((NP, n))
    pcb = np.full(NP, np.inf)
    for i in range(NP):
        pos[i] = h0 if i == 0 else h0 + rng.normal(0, 0.2, n)
        project_inplace(pos[i], tsum)
        pcb[i] = compute_c5(pos[i])
        pbest[i] = pos[i].copy()
    gi = int(pcb.argmin())
    gbest, gcb = pbest[gi].copy(), pcb[gi]
    best_h, best_c5 = gbest.copy(), gcb

    t_pso_end = t0 + budget_s * 0.5
    t_restart = t0
    while time.time() < t_pso_end:
        pr = min(1.0, (time.time() - t0) / max(1.0, t_pso_end - t0))
        w = WMAX - (WMAX - WMIN) * pr
        for i in range(NP):
            vel[i] = (w * vel[i] + C1 * rng.random(n) * (pbest[i] - pos[i])
                      + C2 * rng.random(n) * (gbest - pos[i]))
            np.clip(vel[i], -VMAX, VMAX, out=vel[i])
            pos[i] += vel[i]
            project_inplace(pos[i], tsum)
            c = compute_c5(pos[i])
            if c < pcb[i]:
                pcb[i] = c
                pbest[i] = pos[i].copy()
                if c < gcb:
                    gcb = c
                    gbest = pos[i].copy()
                    best_h, best_c5 = gbest.copy(), c
        if time.time() - t_restart > 20:
            t_restart = time.time()
            for i in pcb.argsort()[-NP // 5:]:
                pos[i] = rng.uniform(0, 1, n)
                project_inplace(pos[i], tsum)
                vel[i] = rng.normal(0, 0.05, n)
                pcb[i] = compute_c5(pos[i])
                pbest[i] = pos[i].copy()
            gn = int(pcb.argmin())
            if pcb[gn] < gcb:
                gcb = pcb[gn]
                gbest = pbest[gn].copy()
                best_h, best_c5 = gbest.copy(), gcb

    t_end = t0 + budget_s * 0.95
    while time.time() < t_end:
        frac = min(1.0, (time.time() - t0) / budget_s)
        scale = 0.01 + 0.09 * (1.0 - frac)
        hp = best_h + rng.normal(0, scale, n)
        project_inplace(hp, tsum)
        hr, cr = coord_descent(hp, tsum, 20)
        if cr < best_c5:
            best_h, best_c5 = hr, cr

    best_h, best_c5 = coord_descent(best_h, tsum, 10)
    project_inplace(best_h, tsum)
    return (best_h, compute_c5(best_h), n)
