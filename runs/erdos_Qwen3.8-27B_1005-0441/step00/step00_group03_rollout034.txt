def project(h, n_target):
    """Project h onto {0<=h<=1, sum(h)=n_target}."""
    h = np.clip(h, 0.0, 1.0)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, n_target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = n_target - h.sum()
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-12
            c = m.sum()
            if c == 0:
                break
            h[m] += d / c
        else:
            m = h > 1e-12
            c = m.sum()
            if c == 0:
                break
            h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def make_random_h(n, rng):
    h = np.full(n, 0.5)
    for _ in range(5):
        f = rng.integers(1, n // 2 + 1)
        p = rng.uniform(0, 2.0 * np.pi)
        a = rng.uniform(0.05, 0.25)
        h += a * np.sin(2.0 * np.pi * f * np.arange(n) / n + p)
    return project_h(h, n / 2.0)

def make_binary_h(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def sa_loop(h, n, budget, rng, t_start=0.01, t_end=0.0004):
    best = h.copy()
    best_c5 = compute_c5(h)
    cur = h.copy()
    cur_c5 = best_c5
    nt = n / 2.0
    t = t_start
    start_time = time.time()
    est_iters = max(2000, int(budget * 15000))
    alpha = (t_end / t_start) ** (1.0 / est_iters)
    n_swap = 3
    delta = 0.05

    while time.time() - start_time < budget:
        h2 = cur.copy()
        bad = False
        for _ in range(n_swap):
            i, j = rng.integers(0, n, 2)
            if i == j:
                j = (i + 1) % n
            d = rng.uniform(-delta, delta)
            h2[i] += d
            h2[j] -= d
            if h2[i] < 0.0 or h2[i] > 1.0 or h2[j] < 0.0 or h2[j] > 1.0:
                bad = True
                break
        if bad:
            h2 = project_h(h2, nt)
        c5 = compute_c5(h2)
        de = c5 - cur_c5
        if de < 0.0 or (t > t_end and rng.random() < np.exp(-de / t)):
            cur = h2
            cur_c5 = c5
            if c5 < best_c5:
                best = h2.copy()
                best_c5 = c5
        t *= alpha
        if t < t_end:
            t = t_start * 0.4
            rem = max(1.0, budget - (time.time() - start_time))
            n_rem = max(1000, int(rem * 15000))
            alpha = (t_end / t) ** (1.0 / n_rem)
    return best, best_c5

def bin_local_search(h, n, budget, rng):
    h = (h > 0.5).astype(float)
    cnt = int(h.sum())
    tgt = n // 2
    if cnt > tgt:
        oi = np.where(h == 1)[0]

def coord_descent(h, n, budget, rng):
    """Focus on the argmax lag to reduce it."""
    best = h.copy()
    best_c5 = compute_c5(h)
    cur = h.copy()
    nt = n / 2.0
    start_time = time.time()

    while time.time() - start_time < budget:
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        k_star = int(np.argmax(corr))
        # Try to reduce corr[k_star]
        # corr[k_star] = sum_i cur[i+k] * (1-cur[i]) where k = k_star - (n-1)
        k = k_star - (n - 1)
        # Gradient of corr[k_star] w.r.t. cur[m]:
        # d/dm [sum_i cur[i+k]*(1-cur[i])]
        # = (1-cur[m-k]) if m appears as i+k, i.e., i=m-k
        #   - cur[m+k] if m appears as i, i.e., m=i
        dx = 2.0 / n
        for _ in range(200):

                j = (i + 1) % n
            gi = 0.0
            gj = 0.0
            if 0 <= i - k < n:
                gi += (1.0 - cur[i - k])
            if 0 <= i + k < n:
                gi -= cur[i + k]
            if 0 <= j - k < n:
                gj += (1.0 - cur[j - k])
            if 0 <= j + k < n:
                gj -= cur[j + k]
            # Move i up, j down (or vice versa) to decrease corr[k_star]
            # We want to decrease gi*delta + (-gj)*(-delta) = gi*delta + gj*delta
            # Actually: delta corr = delta_gi * d_i + delta_gj * d_j

            # To decrease, if gi > gj: d_i = -eps, d_j = +eps

            h2[j] -= eps

                h2 = project_h(h2, nt)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                cur = h2
                best = h2.copy()
                best_c5 = c5
    return best, best_c5
