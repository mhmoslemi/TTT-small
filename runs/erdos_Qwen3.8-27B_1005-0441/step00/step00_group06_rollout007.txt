def irls_opt(h_start, n_pts, n_iter=50, beta_start=10, beta_end=200):
    h = h_start.copy()
    for it in range(n_iter):
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, n_iter-1))
        corr = np.correlate(h, 1.0 - h, mode='full')
        weights = np.exp(beta * (corr - corr.max()))
        weights /= weights.sum()
        
        # Minimize sum_k w_k * corr_k
        # corr_k is quadratic in h, so this is a QP
        # Use SLSQP or trust-constr
        def obj(h):
            c = np.correlate(h, 1.0 - h, mode='full')

                      constraints=cons, options={'maxiter': 100, 'ftol': 1e-12})
        h = np.clip(res.x, 0, 1)

  V = {i : 0<=i<n AND 0<=i+lag<n}
  For the second term δ_{i+lag,j}: we need i+lag=j, so i=j-lag. We need j-lag in V, i.e., 0<=j-lag<n AND 0<=j<n.
  
So: d(corr_k)/d(h[j]) = [j in V] - [j in V]*h[j+lag] - [j-lag in V]*h[j-lag]

Where [P] is 1 if P is true, 0 otherwise.

And j in V means max(0,-lag) <= j <= min(n-1, n-1-lag).

Let me optimize: instead of computing the full correlation, I can compute just the weighted sum directly:

```python
def weighted_corr(h, weights, n_pts):
    """Compute sum_k weights[k] * corr_k without computing full correlation."""
    # corr_k = sum_{i valid for lag k} h[i]*(1-h[i+lag_k]) where lag_k = k - n_pts + 1
    total = 0.0
    for k, w in enumerate(weights):
        if w < 1e-15:
            continue
        lag = k - n_pts + 1
        lo = max(0, -lag)
        hi = min(n_pts, n_pts - lag)
        total += w * np.sum(h[lo:hi] * (1.0 - h[lo+lag:hi+lag]))
    return total
```

This is O(n²) per evaluation (sum over k of O(n) work). Same as np.correlate but potentially faster due to skipping small weights.

Actually, for the IRLS approach, as β increases, the weights become more concentrated on the top few lags, so the computation gets faster.

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _wcorr(h, weights):
    n = len(h)
    total = 0.0
    for k, w in enumerate(weights):
        if w < 1e-14:
            continue
        lag = k - n + 1
        lo = max(0, -lag)
        hi = min(n, n - lag)
        total += w * np.dot(h[lo:hi], 1.0 - h[lo+lag:hi+lag])
    return total

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    h = h + (n / 2.0 - h.sum()) / n
    return np.clip(h, 0.0, 1.0)

def _slsqp_max(h0, n, maxiter=300):
    def obj(h):
        return float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bnds = list(zip(np.zeros(n), np.ones(n)))
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return _project(res.x, n)
    except:
        return h0.copy()

def _irls(h0, n, iters=30, bmax=100):
    h = h0.copy()
    for it in range(iters):
        beta = 5.0 * (bmax / 5.0) ** (it + 1) / iters
        corr = np.correlate(h, 1.0 - h, mode='full')
        w = np.exp(beta * (corr - corr.max()))
        w /= w.sum()
        def obj(h2):
            return _wcorr(h2, w)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        bnds = list(zip(np.zeros(n), np.ones(n)))
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': 80, 'ftol': 1e-12})
            h = _project(res.x, n)
        except:
            pass
    return h

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    # Adjust to get sum = n/2
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute evenly, then clip
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        # If still off, adjust individual entries
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            # Adjust entries that have room
            idx = np.argsort(h)  # start from entries closest to 0 or 1
            if diff > 0:
                for i in range(n):
                    room = 1.0 - h[i]
                    adj = min(room, diff)
                    h[i] += adj
                    diff -= adj
                    if abs(diff) < 1e-12:
                        break
            else:
                for i in range(n-1, -1, -1):
                    room = h[i]
                    adj = min(room, -diff)
                    h[i] -= adj
                    diff += adj
                    if abs(diff) < 1e-12:
                        break
    return h
