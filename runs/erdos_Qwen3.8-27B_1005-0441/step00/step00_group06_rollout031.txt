import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(logsumexp(alpha * c) / alpha)

def project(h, req):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (req / s)
    return np.clip(h, 0.0, 1.0)

def binarize(h, req):
    hb = (h > 0.5).astype(float)
    diff = req - np.sum(hb)
    if diff > 0:
        zeros = np.where(hb == 0)[0]
        full = int(diff)
        for i in range(min(full, len(zeros))):
            hb[zeros[i]] = 1.0
        if len(zeros) > full:
            hb[zeros[full]] = diff - full
    elif diff < 0:
        ones = np.where(hb == 1)[0]
        full = int(-diff)
        for i in range(min(full, len(ones))):
            hb[ones[i]] = 0.0
        if len(ones) > full:
            hb[ones[full]] = 1.0 - (-diff - full)
    return project(hb, req)

def local_search(h, max_rounds=200, time_limit=None, t0=None):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        for i in ones:
            for j in zeros:
                if h[i] == h[j]:
                    continue
                h[i], h[j] = h[j], h[i]
                c = compute_c5(h)
                if c < best - 1e-14:
                    best = c
                    improved = True
                    ones = list(np.where(h > 0.5)[0])
                    zeros = list(np.where(h < 0.5)[0])
                    break
                else:
                    h[i], h[j] = h[j], h[i]
            if improved:
                break
        if not improved:
            break
    return h, best

def simulated_annealing(h, n, req, max_rounds=5000, t0=None, time_limit=None):
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    T = 0.01
    T_min = 1e-8
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    for it in range(max_rounds):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        if not ones or not zeros:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h)
        if c_new < best or np.random.random() < np.exp(-(c_new - best) / max(T, T_min)):
            best = c_new
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= 0.999
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
    return best_h, best
