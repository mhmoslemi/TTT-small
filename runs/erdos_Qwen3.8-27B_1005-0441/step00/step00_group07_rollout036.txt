import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def compute_grad(h, n, T):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    w = np.exp(T * (c - c_max))
    w /= w.sum()
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for idx in range(n_lags):
        k = idx - (n - 1)
        wk = w[idx]
        if wk < 1e-15:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        else:
            d = -k
            grad[:n - d] += wk * (1.0 - h[d:])
            grad[d:] -= wk * h[:n - d]
    grad *= dx
    return c_max, grad

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_descent(h_init, n, time_limit, start_time, T_init=15, T_end=300, max_iters=20000):
    h = project_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.02
    no_improve = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        T_cur = T_init + (T_end - T_init) * min(1.0, it / (max_iters * 0.5))
        c_max, grad = compute_grad(h, n, T_cur)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.03, 0.08)
        else:
            no_improve += 1
            lr *= 0.92
            if no_improve > 300:
                lr = 0.005 + rng_global.random() * 0.01
                no_improve = 0
    return best_h, best_c5

rng_global = np.random.RandomState(42)

def coord_descent_polish(h_init, n, max_rounds=30, time_limit=60.0, start_time=0.0):
    h = project_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_trial = 16
    trials = np.linspace(0.0, 1.0, n_trial)
    for rd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for i in range(n):
            h_save = h[i]
            best_val = h_save
            best_obj = compute_c5(h)
            for t in trials:
                if abs(t - h_save) < 1e-4:
                    continue
                h[i] = t
                obj = compute_c5(h)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_val = t
            h[i] = best_val
            if best_val != h_save:
                improved = True
        h = project_feasible(h, n)
        cur = compute_c5(h)
        if cur < best_c5 - 1e-13:
            best_c5 = cur
            best_h = h.copy()
        if not improved and rd > 2:
            break
    return best_h, best_c5

def upscale(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new)

def make_random_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = rng.rand(n) * 0.6 + 0.2
    return project_feasible(h, n)

def make_sin_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    freq = rng.randint(1, 4)
    phase = rng.rand() * 2 * np.pi
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(freq * np.pi * x + phase)
    return project_feasible(h, n)
