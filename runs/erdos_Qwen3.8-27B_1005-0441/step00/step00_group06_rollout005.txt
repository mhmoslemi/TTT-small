import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def smooth_max_preserve(x, tau):
    m = np.max(x)
    shifted = (x - m) * tau
    return m + np.log(np.sum(np.exp(shifted))) / tau


def obj_func(h, n, tau, pen_w):
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))
    s = np.sum(h)
    return smooth_max_preserve(corr, tau) + pen_w * (s - 0.5 * n) ** 2


def fix_sum(h, target_sum):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def local_refine(h, n, target_sum, time_limit):
    tl = time.time()
    h = np.array(h, dtype=float)
    for sweep in range(150):
        if time.time() - tl > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - tl > time_limit:
                return h
            base_c5 = compute_c5(h)
            old = h[j]
            for d in [0.01, 0.005, 0.002, -0.002, -0.005, -0.01]:
                nv = old + d
                if nv < 0.0 or nv > 1.0:
                    continue
                h[j] = nv
                h = fix_sum(h, target_sum)
                nc5 = compute_c5(h)
                if nc5 < base_c5:
                    improved = True
                    base_c5 = nc5
                    old = nv
                else:
                    h[j] = old
                    h = fix_sum(h, target_sum)
                    break
        if not improved:
            break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()

        h0 = np.ones(95) * 0.5
    n = len(h0)
    target_sum = 0.5 * n

    h0 = fix_sum(h0, target_sum)
    best_h = np.array(h0, dtype=float)
    best_c5 = compute_c5(best_h)

    # Phase 1: Refine from initial with decreasing temperature
    for tau in [40.0, 15.0, 7.0]:
        if budget_s - (time.time() - t0) < 25:
            break
        try:
            r = minimize(obj_func, best_h, args=(n, tau, 1e7),
                        method='L-BFGS-B',
                        bounds=[(0.0, 1.0)] * n,
                        options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-10})
            ht = fix_sum(np.clip(r.x, 0, 1), target_sum)
            c5 = compute_c5(ht)
            if c5 < best_c5:
                best_h = ht
                best_c5 = c5
        except Exception:
            pass

    # Phase 2: Multistart optimization
    for r in range(100):
        if budget_s - (time.time() - t0) < 35:
            break
        if r == 0:
            h_start = np.array(best_h, dtype=float)
        elif r % 6 == 1:

            phase = rng.rand() * 2.0 * np.pi
