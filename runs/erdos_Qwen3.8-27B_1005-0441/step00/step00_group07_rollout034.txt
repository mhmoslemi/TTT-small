- If i = n+L_k for some valid n: contributes h[i]*(1-h[i-L_k])
- If both: contributes h[i]*(1-h[i]) ... wait no.

Actually for a given lag L:
g_L = ∑_n h[n+L]*(1-h[n])*dx

The terms involving h[i]:
- If i is in the "first" position (n+L = i, so n = i-L): contributes h[i]*(1-h[i-L])
- If i is in the "second" position (n = i): contributes h[i+L]*(1-h[i])
- If both (L=0): contributes h[i]*(1-h[i])

So for L ≠ 0:
If i-L is valid: g_L contains h[i]*(1-h[i-L])*dx
If i+L is valid: g_L contains h[i+L]*(1-h[i])*dx = -h[i+L]*h[i]*dx + h[i+L]*dx

So g_L = A_L*h[i] - B_L*h[i] + C_L where:
A_L = (1-h[i-L])*dx if i-L valid, else 0
B_L = h[i+L]*dx if i+L valid, else 0
C_L = (rest of the sum) * dx

So g_L is linear in h[i]! (For a fixed lag.) The max of linear functions is convex in h[i]. So coordinate descent on h[i] can find the optimal value by checking where the max of these linear functions is minimized.

The optimal h[i] for coordinate descent:
- g_k(h[i]) = (A_k - B_k) * h[i] + C_k for each k
- We want to minimize max_k g_k(h[i])
- h[i] ∈ [0,1] and subject to sum constraint

    n = len(x)
    target = n / 2.0
    lo, hi = x.min() - 2.0, x.max() + 2.0

        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def subgradient_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    L = max_idx - (n - 1)
    grad = np.zeros(n)
    if L >= 0:
        idx = np.arange(L, n)
        grad[idx] += (1.0 - h[idx - L])

        grad[idx] += (1.0 - h[idx - L])
        idx2 = np.arange(-L, n)
        grad[idx2] -= h[idx2 + L]

import numpy as np
import time

def project_feasible(x):
    n = len(x)
    target = n / 2.0
    lo, hi = x.min() - 2.0, x.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lag_gradient(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L >= 0:
        idx = np.arange(L, n)
        grad[idx] += (1.0 - h[idx - L])
        idx2 = np.arange(0, n - L)
        grad[idx2] -= h[idx2 + L]
    else:
        idx = np.arange(0, n + L)
        grad[idx] += (1.0 - h[idx - L])
        idx2 = np.arange(-L, n)
        grad[idx2] -= h[idx2 + L]
    return grad

def subgradient_c5(h, temp=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if temp <= 0:
        max_idx = int(np.argmax(corr))
        L = max_idx - (n - 1)
        return lag_gradient(h, L) * dx
    shifted = corr / temp
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(len(corr)):
        if w[i] < 1e-10:
            continue
        L = i - (n - 1)
        grad += w[i] * lag_gradient(h, L)
    return grad * dx

def local_search(h, n_trials, rng):
    n = len(h)
    h = h.copy()

        h_old = h.copy()
        k = rng.randint(1, max(2, n // 8))

        h = project_feasible(h)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
        else:
