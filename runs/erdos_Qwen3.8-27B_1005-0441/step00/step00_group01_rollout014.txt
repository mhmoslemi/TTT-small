    # Final: one more SLSQP polish
    if elapsed() < budget_s * 0.9:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h_polish = run_slsqp(best_h, dx_f, ts_f, 10000, maxiter=500)
        c5_polish = compute_c5(h_polish, dx_f)
        if c5_polish < best_c5:
            best_c5, best_h = c5_polish, h_polish

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, final_c5, len(best_h)
