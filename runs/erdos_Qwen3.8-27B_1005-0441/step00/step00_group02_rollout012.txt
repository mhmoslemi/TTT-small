    return corr.max(), corr

def project_feasible(h, n_points):
    # Project onto [0,1]^n with sum = n_points/2
    h = np.clip(h, 0, 1)
    target = n_points / 2
    diff = target - h.sum()
    if diff != 0:
        # Distribute the difference
        if diff > 0:
            # Need to increase, do so where h < 1
            mask = h < 1
            h[mask] += diff / mask.sum()
        else:
            # Need to decrease, do so where h > 0
            mask = h > 0
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    return h

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            h += diff2 / n
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = compute_corr(h)
    return float((corr * dx).max())

def logsumexp_gradient(h, corr, n, tau):
    shifted = corr - corr.max()
    w = np.exp(shifted / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones_like(corr) / len(corr)
    else:
        w /= w_sum
    g = np.zeros(n)
    for k_idx in range(len(corr)):
        s = abs(k_idx - (n - 1))
        wt = w[k_idx]
        if wt < 1e-10:
            continue
        g[s:n] += wt * (1.0 - h[:n-s])
        g[:n-s] -= wt * h[s:n]
    return g

def optimize_lse(h0, n, max_time, seed, tau_init=2.0, tau_min=0.1):
    np.random.seed(seed)
    h = project_feasible(h0.copy(), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = 0.08
    momentum = np.zeros(n)
    beta = 0.85
    patience = 0
    max_pat = 200
    t0 = time.time()
    step = 0
    tau = tau_init
    while time.time() - t0 < max_time:
        corr = compute_corr(h)
        c5 = float((corr * dx).max())
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        if patience > max_pat:
            lr *= 0.6
            tau = max(tau * 0.9, tau_min)
            momentum *= 0.3
            patience = 0
            if lr < 1e-7:
                break
        # Animate tau downward
        tau = max(tau_min, tau_init * (0.995 ** step))
        g = logsumexp_gradient(h, corr, n, tau)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm * np.sqrt(n)
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = h - lr * momentum
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            h = h_new

            h_try = project_feasible(h_try, n)
            if evaluate_c5(h_try) < best_c5:
                h = h_try
            else:
                patience += 5
        step += 1
    return best_h, best_c5
