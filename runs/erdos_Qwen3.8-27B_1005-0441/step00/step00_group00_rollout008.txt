    if s < 1e-300:

        w[c.argmax()] = 1.0
    else:
        w /= s
    return w

    max_c = corr.max()
    # log-sum-exp
    s = np.sum(np.exp(tau * (corr - max_c)))
    return max_c + np.log(s) / tau

    w = softmax_weights(corr, tau)
    g = weighted_grad(w, h, len(h))

    # Actually softmax_weights already gives the right weights for d/dh of smooth max
    return corr.max() + np.log(np.sum(np.exp(tau * (corr - corr.max())))) / tau, g

import numpy as np

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = diff / cnt
            h[mask] = np.minimum(h[mask] + per, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            per = -diff / cnt
            h[mask] = np.maximum(h[mask] - per, 0.0)
    return h

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def weighted_grad(weights, h, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        j_lo = max(0, n - 1 - k)
        j_hi = min(n - 1, 2 * n - 1 - k)
        if j_lo <= j_hi:
