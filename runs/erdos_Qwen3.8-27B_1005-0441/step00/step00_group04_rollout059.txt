    corr = np.correlate(h, 1 - h, mode="full") * dx
    k_max = np.argmax(corr)

def compute_subgradient(h, n, L):
    """Compute gradient of c_L (correlation at lag L) w.r.t. h."""
    grad = np.zeros(n)
    if L == 0:
        grad[:] = 1 - 2*h
    elif L > 0:
        # c_L = sum_{i=0}^{n-1-L} h[i+L]*(1-h[i])
        grad[L:] += (1 - h[:n-L])  # h[j-L] term for j>=L

        grad[L2:] -= h[:n-L2]  # -h[j-L2] for j>=L2

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - h.sum()
        if abs(diff) < 1e-13:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def to_binary(h, n):
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:n // 2]] = 1.0
    return b

def resample_linear(h, n_old, n_new):
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h)

def compute_subgrad(h, n, L):
    grad = np.zeros(n)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        grad[L:] += (1.0 - h[:n - L])
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += (1.0 - h[L2:])
        grad[L2:] -= h[:n - L2]
    return grad

def sa_search(h_b, n, t_limit, seed_rng):

            strib = 0

                c5_cur = c5_new
            else:
