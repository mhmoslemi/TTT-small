import numpy as np
import time
import math


def _eval_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx


def _feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n * 0.5 - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        idx = np.where(m)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h


def _perturb_pairs(h, n, rng, scale):
    hp = h.copy()
    npairs = rng.randint(1, 5)
    for _ in range(npairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-scale, scale)
        a, b = hp[i] + d, hp[j] - d
        if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
            hp[i], hp[j] = a, b
    return hp


def _perturb_swap(h, n, rng, scale):
    hp = h.copy()
    i, j = rng.choice(n, 2, replace=False)
    d = min(abs(hp[i] - hp[j]), scale)
    if hp[i] > hp[j]:
        hp[i] -= d
        hp[j] += d
    else:
        hp[i] += d
        hp[j] -= d
    return hp


def _perturb_wave(h, n, rng, scale):
    hp = h.copy()
    k = rng.randint(1, max(2, n // 8))
    amp = rng.uniform(-scale, scale)
    phase = rng.uniform(0.0, 2.0 * np.pi)
    x = np.arange(n, dtype=float)
    hp += amp * np.sin(2.0 * np.pi * k * x / n + phase)
    hp = _feasible(hp, n)
    return hp


def _sa_round(h, n, dx, rng, T, scale, t_limit):
    c5_curr = _eval_c5(h, dx)
    best_local = h.copy()
    best_local_c5 = c5_curr
    cool = 0.9996
    T_end = 5e-7
    while T > T_end and time.time() < t_limit:
        r = rng.rand()
        if r < 0.45:
            hp = _perturb_pairs(h, n, rng, scale)
        elif r < 0.65:
            hp = _perturb_swap(h, n, rng, scale)
        else:
            hp = _perturb_wave(h, n, rng, scale)
        c5_new = _eval_c5(hp, dx)
        if c5_new <= c5_curr:
            h = hp
            c5_curr = c5_new
            if c5_new < best_local_c5:
                best_local_c5 = c5_new
                best_local = hp.copy()
        else:
            dE = c5_new - c5_curr
            if dE < T:
                if rng.random() < math.exp(-dE / T):
                    h = hp
                    c5_curr = c5_new
        T *= cool
        scale = max(0.003, scale * 0.99995)
    return best_local, best_local_c5
