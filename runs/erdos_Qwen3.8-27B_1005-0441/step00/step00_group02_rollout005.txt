import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target):
    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def build_S(fixed, n):
    """S[m,j] = coefficient of var[j] in corr(m) = correlate(var, fixed, full)[m]"""
    n_corr = 2 * n - 1
    S = np.zeros((n_corr, n))
    for j in range(n):
        e = np.zeros(n)
        e[j] = 1.0
        S[:, j] = np.correlate(e, fixed, mode="full")
    return S

def solve_lp(S, n, dx):
    """min C s.t. dx*S@var - C <= 0, sum(var)=n/2, 0<=var<=1"""
    n_corr = S.shape[0]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack((dx * S, -np.ones((n_corr, 1))))
    b_ub = np.zeros(n_corr)
    A_eq = np.hstack((np.ones((1, n)), np.zeros(1)))
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return project_to_sum(res.x[:n], n / 2.0), float(res.x[-1])
    return None, np.inf

def alternating_lp(h, n, max_iter=60):
    """Alternate: fix h→solve y, fix y→solve h."""
    h = h.copy()
    dx = 2.0 / n
    for it in range(max_iter):
        # Fix h, solve for y
        S_y = build_S(h, n)
        y_sol, _ = solve_lp(S_y, n, dx)
        if y_sol is None:
            break
        h_new = 1.0 - y_sol
        h_new = project_to_sum(h_new, n / 2.0)
        c5_old = compute_c5(h)
        c5_new = compute_c5(h_new)
        if c5_new < c5_old - 1e-12:
            h = h_new
        else:
            h = project_to_sum(0.7 * h + 0.3 * h_new, n / 2.0)
    return h

def local_search(h, n, max_passes=5):
    """Pairwise coordinate swaps to polish."""
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(max_passes):
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                for delta in [0.02, 0.05, 0.1, -0.02, -0.05, -0.1]:
                    h_t = h.copy()
                    h_t[i] += delta
                    h_t[j] -= delta
                    if h_t[i] < -1e-12 or h_t[i] > 1 + 1e-12 or h_t[j] < -1e-12 or h_t[j] > 1 + 1e-12:
                        continue
                    h_t = np.clip(h_t, 0, 1)
                    c5_t = compute_c5(h_t)
                    if c5_t < c5 - 1e-11:
                        h = h_t
                        c5 = c5_t
                        improved = True
                        break
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(45, 0.5)

    n = len(h_best)
    h_best = project_to_sum(h_best, n / 2.0)
    c5_best = compute_c5(h_best)

    def elapsed():
        return time.time() - t0

    # Phase 1: Alternating LP
    if elapsed() < budget_s - 90:
        h_alt = alternating_lp(h_best, n, max_iter=50)
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_best:
            h_best, c5_best = h_alt, c5_alt

    # Phase 2: SLSQP with multiple restarts
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n

    for attempt in range(40):
        if elapsed() > budget_s - 40:
            break
        if attempt == 0:
            h_init = h_best.copy()
        elif attempt <= 8:
            h_init = h_best + np.random.randn(n) * 0.015
        elif attempt <= 20:
            h_init = h_best + np.random.randn(n) * 0.04
        else:
            h_init = np.random.uniform(0, 1, n)
        h_init = project_to_sum(h_init, n / 2.0)

        try:
            res = minimize(compute_c5, h_init, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 250, 'ftol': 1e-14})
            h_cand = project_to_sum(res.x, n / 2.0)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_best:
                h_best, c5_best = h_cand, c5_cand
        except Exception:
            pass

    # Phase 3: Local search polish
    if elapsed() < budget_s - 20:
        h_best = local_search(h_best, n, max_passes=4)
        c5_best = compute_c5(h_best)

    # Phase 4: Try different sizes
    for n2 in [max(20, n // 2), 3 * n]:
        if n2 >= n and n2 <= n and elapsed() > budget_s - 30:
            continue
        if n2 < 15 or elapsed() > budget_s - 30:
            continue
        if n2 == n:
            continue
        t_old = (np.arange(n) + 0.5) * (2.0 / n)
        t_new = (np.arange(n2) + 0.5) * (2.0 / n2)
        h2 = np.interp(t_new, t_old, h_best, left=h_best[0], right=h_best[-1])
        h2 = project_to_sum(h2, n2 / 2.0)
        c5_2 = compute_c5(h2)
        # Quick SLSQP on new size
        cons2 = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n2 / 2.0}]
        bnds2 = [(0.0, 1.0)] * n2
        if elapsed() < budget_s - 30:
            try:
                res2 = minimize(compute_c5, h2, method='SLSQP',
                               bounds=bnds2, constraints=cons2,
                               options={'maxiter': 150, 'ftol': 1e-13})
                h2c = project_to_sum(res2.x, n2 / 2.0)
                c5_2 = compute_c5(h2c)
                if c5_2 < c5_best:
                    h_best, c5_best, n = h2c, c5_2, n2
            except Exception:
                pass

    # Final polish
    h_best = project_to_sum(h_best, len(h_best) / 2.0)
    c5_best = compute_c5(h_best)

    return (h_best, c5_best, len(h_best))
