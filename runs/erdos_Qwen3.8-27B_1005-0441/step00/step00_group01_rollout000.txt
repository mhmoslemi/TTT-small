import numpy as np
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    deficit = target_sum - np.sum(h)
    for _ in range(1000):
        if abs(deficit) < 1e-14:
            break
        if deficit > 0:
            free = h < 1.0 - 1e-15
        else:
            free = h > 1e-15
        nfree = np.sum(free)
        if nfree == 0:
            break
        step = deficit / nfree
        h[free] += step
        h = np.clip(h, 0.0, 1.0)
        deficit = target_sum - np.sum(h)
    return h

def get_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_star = np.argmax(corr)
    s = m_star - (n - 1)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n + s] += 1.0 - h[-s:]
        g[-s:] -= h[:n + s]
    return g * dx

def run_subgrad(h0, max_iters, time_limit, alpha0=0.05, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(h0)
    target_sum = n / 2.0
    h = project_feasible(h0.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = alpha0
    vel = np.zeros(n)
    mu = 0.9
    no_improve = 0
    stall_count = 0
    t0 = time.time()
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        g = get_subgrad(h)
        vel = mu * vel - alpha * g
        h_new = h + vel
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            stall_count = 0
        else:
            no_improve += 1
            stall_count += 1
        h = h_new
        if no_improve > 30:
            alpha *= 0.9
            no_improve = 0
        if stall_count > 200:
            alpha = alpha0 * 0.5
            vel *= 0.1
            h = best_h + alpha * 0.3 * rng.standard_normal(n)
            h = project_feasible(h, target_sum)
            stall_count = 0
    return best_h, best_c5

def make_ramp(n):
    x = np.linspace(0, 2.0, n, endpoint=False) + 1.0 / n
    h = np.zeros(n)
    for i in range(n):
        if x[i] <= 1.0:
            h[i] = x[i]
        else:
            h[i] = 2.0 - x[i]
    # Normalize to sum = n/2
    h = h / np.sum(h) * (n / 2.0)
    return np.clip(h, 0, 1)

def make_sine(n):
    x = np.linspace(0, 2.0, n, endpoint=False) + 1.0 / n
    h = 0.5 + 0.5 * np.sin(np.pi * (x - 0.5))
    h = h / np.sum(h) * (n / 2.0)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()
    t_end = t_start + budget_s - 30  # leave 30s buffer
    best_h = None
    best_c5 = np.inf
    best_n = 0

    # Try multiple n values
    n_candidates = []
    if 'initial_h_values' in dir() and initial_h_values is not None:
        n0 = len(initial_h_values)
        n_candidates.append(n0)
    for nn in [46, 48, 50, 52, 44, 54, 40, 60, 64]:
        if nn not in n_candidates:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))

    # Build initial guesses
    starts = {}
    for nn in n_candidates:
        targets = nn / 2.0
        if nn == n0:
            h_init = project_feasible(np.array(initial_h_values, dtype=float), targets)
            starts[nn] = [h_init, make_ramp(nn), make_sine(nn)]
        else:
            starts[nn] = [make_ramp(nn), make_sine(nn)]
            # Add a "soft toggle"
            x = np.linspace(0, 2, nn, endpoint=False) + 1.0 / nn
            ht = np.clip(0.5 + 0.4 * np.sin(2 * np.pi * x), 0, 1)
            ht = project_feasible(ht, targets)
            starts[nn].append(ht)

    # Phase 1: Subgradient descent from all starts
    for nn in n_candidates:
        if time.time() > t_end:
            break
        for h_init in starts[nn]:
            if time.time() > t_end:
                break
            remaining = max(10, t_end - time.time())
            iters = min(20000, int(remaining * 200))
            h_opt, c5 = run_subgrad(h_init, iters, remaining * 0.9,
                                     alpha0=0.05, rng=rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = nn

    # Phase 2: Polish best solution with finer steps and random perturbations
    if best_h is not None:
        h_cur = best_h.copy()
        n = best_n
        target_sum = n / 2.0
        for trial in range(20):
            if time.time() > t_end:
                break
            remaining = max(5, t_end - time.time())
            # Add perturbation
            scale = max(0.01, 0.1 * (best_c5 - 0.38))
            h_pert = h_cur + scale * rng.standard_normal(n)
            h_pert = project_feasible(h_pert, target_sum)
            iters = min(5000, int(remaining * 100))
            h_opt, c5 = run_subgrad(h_pert, iters, remaining * 0.8,
                                     alpha0=0.02, rng=rng)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_opt.copy()
                h_cur = h_opt.copy()

        # Very fine local polish
        if time.time() < t_end - 10:
            remaining = max(5, t_end - time.time())
            iters = min(30000, int(remaining * 300))
            h_opt, c5 = run_subgrad(h_cur, iters, remaining * 0.9,
                                     alpha0=0.005, rng=rng)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 3: Try smoothed-max gradient for a few iterations
    if best_h is not None and time.time() < t_end - 10:
        h_cur = best_h.copy()
        n = best_n
        target_sum = n / 2.0
        T = 0.05
        alpha = 0.01
        for it in range(5000):
            if time.time() > t_end - 5:
                break
            dx = 2.0 / n
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            # soft-max weight
            exp_c = np.exp(corr / T - corr.max() / T)
            w = exp_c / exp_c.sum()
            # Gradient: sum_m w[m] * dcorr[m]/dh[j]
            g = np.zeros(n)
            for m in range(len(corr)):
                if w[m] < 1e-8:
                    continue
                s = m - (n - 1)
                if s == 0:
                    g[:] += w[m] * (1.0 - 2.0 * h_cur)
                elif s > 0:
                    g[s:] += w[m] * (1.0 - h_cur[:n - s])
                    g[:n - s] -= w[m] * h_cur[s:]
                else:
                    g[:n + s] += w[m] * (1.0 - h_cur[-s:])
                    g[-s:] -= w[m] * h_cur[:n + s]
            g *= dx
            h_cur = h_cur - alpha * g
            h_cur = project_feasible(h_cur, target_sum)
            c5_cur = compute_c5(h_cur)
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h_cur.copy()

    if best_h is None:
        # Fallback
        n = 48
        h_cur = make_ramp(n)
        best_c5 = compute_c5(h_cur)
        best_h = h_cur

    # Final verification
    n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
