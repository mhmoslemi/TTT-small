import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_valid_binary(h):
    """Ensure h is binary with exactly len(h)//2 ones."""
    n = len(h)
    h = np.where(h >= 0.5, 1.0, 0.0).astype(float)
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - n_ones]] = 1.0
    return h

def exhaustive_local_search(h, max_rounds=1000, deadline=None):
    """Try all swaps, accept best, repeat."""
    n = len(h)
    h = make_valid_binary(h)
    target = n // 2
    current_c5 = compute_c5(h)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for _ in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        
        best_delta = 0.0
        best_a = -1
        best_b = -1
        
        for a in ones:
            for b in zeros:
                h[a], h[b] = 0.0, 1.0
                nc5 = compute_c5(h)
                h[a], h[b] = 1.0, 0.0
                d = nc5 - current_c5
                if d < best_delta - 1e-15:
                    best_delta = d
                    best_a = a
                    best_b = b
        
        if best_a >= 0:
            h[best_a], h[best_b] = 0.0, 1.0
            current_c5 += best_delta
            ones = np.setdiff1d(ones, best_a)
            zeros = np.setdiff1d(zeros, best_b)
            ones = np.append(ones, best_b)
            zeros = np.append(zeros, best_a)
        else:
            break
    
    return h, current_c5

    """SA for binary swaps."""
    n = len(h)

    # Phase 4: Other n values
    if time.time() < deadline - 20:
        for nn in [40, 50, 60, 80, 90, 100, 120, 140]:
            if nn == n_opt or time.time() > deadline - 10:
                continue
            hn = np.zeros(nn)
            hn[rng.permutation(nn)[:nn // 2]] = 1.0
            hn1, cn1 = local_search(hn, max_rounds=300, deadline=deadline)
            hn2, cn2 = sa_search(hn1, n_iter=15000, rng=rng, deadline=deadline)
            hn3, cn3 = local_search(hn2, max_rounds=300, deadline=deadline)
            cmin = min(cn1, cn2, cn3)
            hmin = hn1 if cmin == cn1 else (hn2 if cmin == cn2 else hn3)
            if cmin < best_c5:
                best_c5, best_h = cmin, hmin.copy()

    # Phase 5: Final exhaustive polish
    if time.time() < deadline:
        hf, cf = local_search(best_h, max_rounds=3000, deadline=deadline)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    # Phase 6: fractional gradient step (quick)
    if time.time() < deadline - 5:
        nf = len(best_h)
        def obj(x):
            return compute_c5(x)
        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * nf,
                           constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - nf / 2.0}],
                           options={'maxiter': 200, 'ftol': 1e-12})
            if res.fun < best_c5:
                best_c5, best_h = float(res.fun), res.x.copy()
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
