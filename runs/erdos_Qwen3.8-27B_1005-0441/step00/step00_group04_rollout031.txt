```python
def project_onto_constraints(h, target_sum):
    """Project h onto {0<=h[i]<=1, sum(h)=target_sum} using bisection."""
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mu = (lo + hi) / 2.0
        h_p = np.clip(h - mu, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target_sum:
            lo = mu
        else:
            hi = mu
    mu = (lo + hi) / 2.0
    h_p = np.clip(h - mu, 0.0, 1.0)
    # Fine correction
    diff = target_sum - np.sum(h_p)
    if abs(diff) > 1e-12:
        free = (h_p > 1e-9) & (h_p < 1.0 - 1e-9)
        nf = np.sum(free)
        if nf > 0:
            adj = diff / nf
            h_p[free] = np.clip(h_p[free] + adj, 0.0, 1.0)
            # Second pass
            diff2 = target_sum - np.sum(h_p)
            free2 = ~np.isclose(h_p, 0.0, atol=1e-9) & ~np.isclose(h_p, 1.0, atol=1e-9)
            nf2 = np.sum(free2)
            if nf2 > 0:
                h_p[free2] = np.clip(h_p[free2] + diff2 / nf2, 0.0, 1.0)
    return h_p

    """Compute all overlap values (not just max)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def compute_grad_for_lags(h, lags, weights):
    """Compute weighted gradient sum for given lags"""
    n = len(h)
    n_half = n - 1
    grad = np.zeros(n)
    for idx, k_idx in enumerate(lags):
        k = k_idx - n_half
        w = weights[idx]
        # Part 1: (1-h[j-k]) for max(0,k) <= j < min(n, n+k)
        lo1 = max(0, int(k))
        hi1 = min(n, n + int(k))
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - int(k):hi1 - int(k)])
        # Part 2: -h[j+k] for max(0,-k) <= j < min(n, n-k)
        lo2 = max(0, -int(k))
        hi2 = min(n, n - int(k))
        if lo2 < hi2:
            grad[lo2:hi2] += w * (-h[lo2 + int(k):hi2 + int(k)])
    return grad

    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    h = project_onto_constraints(h, target_sum)
    
    best_h = h.copy()
    best_c5 = np.max(compute_c5_vec(h))
    
    velocity = np.zeros(n)
    momentum = 0.9
    step = base_step
    no_improve = 0
    
    t0 = time.time()
    for it in range(max_iters):

        corr = compute_c5_vec(h)
        c5 = np.max(corr)

            no_improve = 0

    """Upsample h from n_old to n_new points"""
```

```python
def compute_b(h, j, n):
    """Compute b_k for coordinate j: C_k = a_k + h[j]*b_k"""
    n_full = 2*n - 1
    b = np.zeros(n_full)
    n_half = n - 1
    for k_idx in range(n_full):
        k = k_idx - n_half
        val = 0.0
        # h[j] as h[i+k]: i = j-k, valid if 0<=j-k<n
        if 0 <= j - k < n:
            val += (1.0 - h[j - k])
        # h[j] as h[i]: i = j, valid if 0<=j+k<n
        if 0 <= j + k < n:
            val += (-h[j + k])
        b[k_idx] = val
    return b
```

This is O(n) per j. For a full sweep of n coordinates: O(n²). For n=200: 40k operations per sweep. At 1 ns per op (very optimistic in Python), that's 40μs per sweep. More realistically, with Python loop overhead, maybe 1ms per sweep. In 1000 seconds: 1 million sweeps. Way more than enough.

```python
import numpy as np
import time

def proj(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mu = (lo + hi) / 2.0
        if np.sum(np.clip(h - mu, 0.0, 1.0)) > target:
            lo = mu
        else:
            hi = mu
    mu = (lo + hi) / 2.0
    hp = np.clip(h - mu, 0.0, 1.0)
    d = target - np.sum(hp)
    if abs(d) > 1e-12:
        free = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        nf = np.sum(free)
        if nf > 0:
            hp[free] = np.clip(hp[free] + d / nf, 0.0, 1.0)
    return hp

def get_ov(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def compute_b(h, j, n):
    n_half = n - 1
    ks = np.arange(2 * n - 1) - n_half
    ik = j - ks
    valid1 = (ik >= 0) & (ik < n)
    jk = j + ks
    valid2 = (jk >= 0) & (jk < n)
    b = np.zeros(2 * n - 1)
    if np.any(valid1):
        b[valid1] = (1.0 - h[np.clip(ik[valid1], 0, n - 1)])
    if np.any(valid2):
        b[valid2] += (-h[np.clip(jk[valid2], 0, n - 1)])
    return b

def find_best_t(a, b):
    ts = np.linspace(0.0, 1.0, 60)
    vals = np.max(a[:, None] + b[:, None] * ts[None, :], axis=0)
    bi = np.argmin(vals)
    lo = ts[max(0, bi - 1)]
    hi = ts[min(len(ts) - 1, bi + 1)]
    ts2 = np.linspace(lo, hi, 80)
    vals2 = np.max(a[:, None] + b[:, None] * ts2[None, :], axis=0)
    return ts2[np.argmin(vals2)]

def coord_descent(h0, time_limit):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = proj(h0, target)
    ov = get_ov(h).copy()
    best_h = h.copy()
    best_c5 = np.max(ov) * dx

    t0 = time.time()
    stale = 0
    order = np.arange(n)

    while time.time() - t0 < time_limit:
        stale += 1
        if stale % 50 == 0:
            np.random.shuffle(order)
        improved = False
        for j in order:
            if time.time() - t0 > time_limit:
                break
            b = compute_b(h, j, n)
            hs = h[j]
            a = ov - hs * b
            best_t = find_best_t(a, b)
            delta = best_t - h[j]
            if abs(delta) > 1e-7:
                h[j] = best_t
                ov += delta * b
                improved = True

        c5 = np.max(ov) * dx
        if c5 < best_c5 - 1e-11:
            best_c5 = c5
            best_h = h.copy()

        if not improved:
            break
        if stale > 200:
            break

    return best_h, best_c5

def grad_descend(h0, time_limit, top_k=15, step0=0.05):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    h = proj(h0, target)
    best_h = h.copy()
    ov = get_ov(h)
    best_c5 = np.max(ov) * dx

    vel = np.zeros(n)
    mom = 0.85
    step = step0
    no_imp = 0
    t0 = time.time()

    while time.time() - t0 < time_limit:
        n_half = n - 1
        c5 = np.max(ov) * dx
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 300:
                step *= 0.9
                if no_imp > 1000:
                    h = proj(h + 0.05 * np.random.randn(n), target)
                    vel[:] = 0.0
                    step = step0
                    no_imp = 0
                    ov = get_ov(h)

        top_idx = np.argsort(ov)[-top_k:]
        top_vals = ov[top_idx]
        mx = np.max(top_vals)
        wts = np.exp((top_vals - mx) / 0.1)
        sw = np.sum(wts)
        if sw < 1e-30:
            wts = np.ones(top_k) / top_k
        else:
            wts = wts / sw * top_k

        grad = np.zeros(n)
        for i in range(top_k):
            k = top_idx[i] - n_half
            w = wts[i]
            lo1, hi1 = max(0, k), min(n, n + k)
            if lo1 < hi1:
                grad[lo1:hi1] += w * (1.0 - h[lo1 - k:hi1 - k])
            lo2, hi2 = max(0, -k), min(n, n - k)
            if lo2 < hi2:
                grad[lo2:hi2] += w * (-h[lo2 + k:hi2 + k])

        vel = mom * vel - step * grad
        h = proj(h + vel, target)
        ov = get_ov(h)

    c5 = np.max(ov) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, left=0.0, right=0.0)
    return proj(h_new, n_new / 2.0)

def make_binary(n, seed):
    rng = np.random.default_rng(seed)
    h = (rng.random(n) > 0.5).astype(float)
    return proj(h, n / 2.0)

def make_blocky(n, seed):
    rng = np.random.default_rng(seed)
    block_size = max(3, n // 20)
    h = np.zeros(n)
    for start in range(0, n, block_size * 2):
        end = min(start + block_size, n)
        h[start:end] = rng.uniform(0.7, 1.0)
    return proj(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_init = np.array(globals()['initial_h_values'], dtype=float)
    else:
        h_init = make_binary(42, seed)

    n_init = len(h_init)
    n_candidates = sorted(set([n_init, 42, 56, 64, 80, 100, 128, 160, 200]))
    n_candidates = [n for n in n_candidates if n <= 400]

    total_budget = budget_s - 15
    n_per = max(30, total_budget // len(n_candidates))

    for n in n_candidates:
        elapsed = time.time() - t_start
        remaining = budget_s - 15 - elapsed
        if remaining < 10:
            break
        alloc = min(n_per, remaining)

        starts = []
        if n == n_init:
            starts.append(h_init.copy())
        else:
            starts.append(upsample(h_init, n))
        if best_h is not None:
            if len(best_h) == n:
                starts.append(best_h.copy())
            else:
                starts.append(upsample(best_h, n))
        starts.append(make_binary(n, seed + n * 7))
        starts.append(make_blocky(n, seed + n * 13))

        for si, h_s in enumerate(starts):
            if time.time() - t_start > budget_s - 10:
                break
            tl = (budget_s - 10 - (time.time() - t_start)) / (len(starts) - si)

            # Alternate between methods
            if si % 2 == 0:
                h_r, c5_r = coord_descent(h_s, tl * 0.9)
            else:
                h_r, c5_r = grad_descend(h_s, tl * 0.9, top_k=min(20, n // 4), step0=0.03)

            # Refine with coord descent
            if time.time() - t_start < budget_s - 10:
                h_r, c5_r2 = coord_descent(h_r, min(30, budget_s - 10 - (time.time() - t_start)))
                c5_r = min(c5_r, c5_r2)

            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n

    if best_h is None:
        best_h = h_init
        best_c5 = np.max(get_ov(best_h)) * 2.0 / len(best_h)

    return best_h, best_c5, len(best_h)
```
