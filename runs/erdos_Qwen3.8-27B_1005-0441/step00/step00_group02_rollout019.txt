    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h, dx):
    return np.correlate(h, 1-h, mode="full") * dx

    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(500):

            break
        if diff > 0:

        else:
            mask = h > 1e-15

        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_gradient(h, n, dx, beta=30.0):
    overlaps = compute_all_overlaps(h, dx)
    m = np.max(overlaps)
    w = np.exp(beta * (overlaps - m))
    sw = np.sum(w)
    w /= sw
    off = n - 1
    grad = np.zeros(n)
    for k in range(len(overlaps)):

            continue

        # For this lag, compute gradient contribution
        # d(overlap_L)/dh[j] = (1-h[j-L]) if 0<=j-L<n  - h[j+L] if 0<=j+L<n
        if L >= 0:
            # First term: j from L to n-1, (1-h[j-L])

        h_bin = project_feasible(h_bin, n)
        ho, co = grad_descent(h_bin, n, dx, t_each, beta=50)
        if co < best_c:
            best_h, best_c, best_n = ho.copy(), co, n
        # Smooth init
        if remaining() < 60:
            break
        x = np.linspace(0, 2, n, endpoint=False)
        h_s = 0.5 + 0.45 * np.sin(2.5 * np.pi * x + rng.uniform(0, 6.28))
        h_s = project_feasible(h_s, n)
        t2 = min(40, remaining() * 0.15)
        if t2 < 5:
            break
        ho2, co2 = grad_descent(h_s, n, dx, t2, beta=35)
        if co2 < best_c:
            best_h, best_c, best_n = ho2.copy(), co2, n

    # Phase 3: Final polish with higher beta and local search
    if remaining() > 15 and best_h is not None:
        dx_b = 2.0 / best_n
        tl_final = min(60, remaining() * 0.4)
        ho, co = grad_descent(best_h, best_n, dx_b, tl_final, beta=100)
        if co < best_c:
            best_h, best_c = ho.copy(), co
        tl_ls = min(40, remaining() * 0.3)
        if tl_ls > 3:
            hs, cs = local_search(best_h, best_n, dx_b, tl_ls, rng)
            if cs < best_c:
                best_h, best_c = hs.copy(), cs

    # Fallback
    if best_h is None:
        n_fb = 45
        best_h = 0.5 * np.ones(n_fb)
        best_c = compute_c5(best_h, 2.0 / n_fb)
        best_n = n_fb

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c, best_n
