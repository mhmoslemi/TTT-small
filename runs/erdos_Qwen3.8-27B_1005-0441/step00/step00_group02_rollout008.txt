import numpy as np
from scipy.optimize import minimize
import time

def _corr_weights(corr, beta):
    """Compute softmax weights for correlations."""
    shifted = beta * corr - beta * np.max(corr)
    w = np.exp(shifted)
    return w / w.sum()

def _smooth_objective(h, beta):
    """Smooth max of correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def _smooth_gradient(h, beta):
    """Analytical gradient of smooth max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = _corr_weights(corr, beta)
    # w has length 2n-1, indexed by j=0..2n-2, corresponding to k=j-(n-1)
    grad = np.zeros(n)
    for j in range(2 * n - 1):

        # ∂corr[j]/∂h[i] = (1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n]

            ik_plus = i + k
            if 0 <= ik_plus < n:
                contrib -= h[ik_plus]
            grad[i] += w[j] * contrib

from scipy.optimize import minimize
import time

def _compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(_compute_corr(h)) * dx

def _smooth_obj(h, beta):
    corr = _compute_corr(h)
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def _smooth_grad(h, beta):
    n = len(h)
    corr = _compute_corr(h)
    shifted = beta * (corr - np.max(corr))
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-12:
            continue
        k = j - (n - 1)
        if k >= 0:
            grad[k:n] += w[j] * (1.0 - h[0:n - k])
            grad[0:n - k] += w[j] * (-h[k:n])
        else:
            kn = -k
            grad[0:n - kn] += w[j] * (1.0 - h[kn:n])
            grad[kn:n] += w[j] * (-h[0:n - kn])
    return grad

def _eq_fun(h):
    return np.sum(h) - _TARGET_SUM

def _eq_jac(h):
    return np.ones(len(h))

def _project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    h += (target_sum - np.sum(h)) / len(h)
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        # Adjust unclipped elements
        mask = (h > 0.01) & (h < 0.99)
        if np.any(mask):
            h[mask] += residual / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def _local_search(h, target_sum, max_iters=500):
    n = len(h)
    best = _c5_value(h)
    step = 0.05
    for it in range(max_iters):
        i = it % n
        h2 = h.copy()
        h2[i] = np.clip(h[i] + step, 0, 1)
        # Adjust another element to maintain sum
        j = (i + 1) % n
        diff = h[i] - h2[i]
        h2[j] = np.clip(h2[j] + diff, 0, 1)
        diff2 = h2[i] + h2[j] - h[i] - h[j]
        if abs(diff2) < 1e-10:
            c2 = _c5_value(h2)
            if c2 < best - 1e-10:
                h = h2
                best = c2
                step = min(step * 1.1, 0.3)
            else:
                h2 = h.copy()
                h2[i] = np.clip(h[i] - step, 0, 1)
                h2[j] = np.clip(h2[j] + (h[i] - h2[i]), 0, 1)
                if abs(h2[i] + h2[j] - h[i] - h[j]) < 1e-10:
                    c2 = _c5_value(h2)
                    if c2 < best - 1e-10:
                        h = h2
                        best = c2
                    else:
                        step *= 0.7
                        if step < 0.001:
                            break
        else:
            step *= 0.7
    return _project(h, target_sum), best
