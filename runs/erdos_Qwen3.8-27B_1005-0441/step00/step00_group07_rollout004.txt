    grad = np.zeros(n)
    
    one_minus_h = 1.0 - h
    
    for k in range(2*n - 1):
        wk = w[k]
        mk = shifts[k]
        if mk >= 0:
            # (1-h[i+mk]) valid for i in [0, n-1-mk]
            if n-1-mk >= 0:
                grad[:n-mk] += wk * one_minus_h[mk:]
            # -h[i-mk] valid for i in [mk, n-1]
            if mk <= n-1:
                grad[mk:] -= wk * h[:n-mk]
        else:

            if abs_mk <= n-1:
                grad[:abs_mk] -= wk * h[abs_mk:]
    
    return grad * dx
