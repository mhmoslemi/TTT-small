import numpy as np
from scipy.optimize import minimize
import time

def c5_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def sum_eq(h):
    return np.sum(h) - len(h) / 2.0

def ones_jac(h):
    return np.ones(len(h))

def project_feasible(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def projected_subgrad(h, n_steps=2000, alpha_init=0.01):
    """Projected subgradient method on the max term."""
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy())
    for it in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        # Gradient of corr[max_idx] w.r.t. h
        grad = np.zeros(n)
        k = max_idx - (n - 1)  # shift
        # corr[k] = sum_i h[i] * (1 - h[i-k]) for valid i
        # d/dh[j]: appears in sum as h[j]*(1-h[j-k]) + sum of (1-h[i])*h[j] where i-k=j
        # Actually let's compute numerically for reliability at this cost
        for j in range(n):
            hp = h.copy()
            hp[j] += 1e-7
            corr_p = np.correlate(hp, 1.0 - hp, mode="full")
            grad[j] = (corr_p[max_idx] - corr[max_idx]) / 1e-7
        alpha = alpha_init / (1.0 + 0.001 * it)
        h -= alpha * grad / dx
        h = project_feasible(h)
    return h
