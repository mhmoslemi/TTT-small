    target = n / 2.0

    if s > target:

        # Reduce from largest to smallest
        idx = np.argsort(h)[::-1]
        h_new = h.copy()
        remaining = excess
        for i in idx:
            if remaining <= 0:
                break
            reduce = min(h_new[i] - 0, remaining)  # can reduce to 0
            h_new[i] -= reduce
            remaining -= reduce
        return h_new
    else:

            increase = min(1 - h_new[i], remaining)
            h_new[i] += increase
            remaining += 0 # remaining -= increase
            remaining -= increase
        return h_new
