    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-10:
            break
        step = diff / len(h)
        h = np.clip(h - step, 0.0, 1.0)
    # Final adjustment
    diff = np.sum(h) - target_sum
    if abs(diff) > 1e-10:
        # Distribute to non-boundary elements
        mask = (h > 1e-8) & (h < 1 - 1e-8)
        if mask.sum() > 0:
            h[mask] += -diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx

    w = np.exp(alpha * (vals - mx))

import numpy as np
import time
from scipy.optimize import minimize

def _evaluate(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def _project(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_sm(h, alpha=40.0):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = np.max(vals)
    w = np.exp(alpha * (vals - mx))
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        # h[j] appears as h[n+k] at n=j-k: contrib (1-h[j-k])*dx, valid 0<=j-k<n
        a = max(0, k)
        b = min(n, n + k)
        if a < b:
            grad[a:b] += w[m] * dx * (1.0 - h[a - k:b - k])
        # h[j] appears as h[n] at n=j: contrib -h[j+k]*dx, valid 0<=j+k<n
        a = max(0, -k)
        b = min(n, n - k)
        if a < b:
            grad[a:b] -= w[m] * dx * h[a + k:b + k]
    return grad

def _proj_grad_opt(h0, max_iter=5000, alpha_init=40.0):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best = _evaluate(h)
    best_h = h.copy()
    lr = 0.005
    alpha = alpha_init
    for it in range(max_iter):
        g = _grad_sm(h, alpha)
        h_new = _project(h - lr * g, target)
        v = _evaluate(h_new)
        if v < best - 1e-12:
            best = v
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.3, 0.05)
            alpha = min(alpha * 1.05, 200.0)
        else:
            lr *= 0.75
            alpha = max(alpha * 0.95, 20.0)
            if lr < 1e-8:
                lr = 0.005
                alpha = alpha_init
            if lr < 1e-9:
                break
    return best_h, best

def _slsqp_opt(h0, n, max_iter=300):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        vals = np.correlate(x, 1.0 - x, mode="full") * dx
        mx = np.max(vals)
        s = (1.0 / 60.0) * np.log(np.sum(np.exp(60.0 * (vals - mx))))
        return mx + s
    def obj_grad(x):
        vals = np.correlate(x, 1.0 - x, mode="full") * dx
        mx = np.max(vals)
        w = np.exp(60.0 * (vals - mx))
        w /= w.sum()
        g = np.zeros(n)
        for m in range(2 * n - 1):
            if w[m] < 1e-12:
                continue
            k = m - (n - 1)
            a = max(0, k)
            b = min(n, n + k)
            if a < b:
                g[a:b] += w[m] * dx * (1.0 - x[a - k:b - k])
            a = max(0, -k)
            b = min(n, n - k)
            if a < b:
                g[a:b] -= w[m] * dx * x[a + k:b + k]
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bnds = [(0, 1)] * n
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None:
            hx = _project(res.x, target)
            cv = _evaluate(hx)
            return hx, cv
    except Exception:
        pass
    return h0, _evaluate(h0)

    h = rng.random(n)
    return _project(h, n / 2.0)

        end = min(start + max(1, n // nblocks), n)
        h[start:end] = rng.uniform(0, 1)
    return _project(h, n / 2.0)
