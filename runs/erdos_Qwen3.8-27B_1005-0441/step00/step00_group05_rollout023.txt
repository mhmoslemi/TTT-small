import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return m + np.log(np.mean(np.exp(shifted))) / alpha

def sum_con(h, target):
    return np.sum(h) - target

def proj(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo = -float(np.max(h)) - 0.2
    hi = float(np.max(1.0 - h)) + 0.2
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def try_optimize(h0, n, target, bounds, alpha, maxiter, tol):
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (target,)}]
    try:
        res = minimize(smooth_obj, h0, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol})
        if res.success or res.fun < smooth_obj(h0, alpha):
            return proj(res.x, target)
    except Exception:
        pass
    return None

def coord_descent(h, n, target, steps=50):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(steps):
        improved = False
        for j in range(n):
            for delta in [-0.01, 0.01, -0.005, 0.005, -0.02, 0.02]:
                h_new = h.copy()
                h_new[j] = np.clip(h_new[j] + delta, 0.0, 1.0)
                h_new = proj(h_new, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5 - 1e-9:
                    h = h_new
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

    bounds = [(0.0, 1.0)] * n

    # Phase 1: SLSQP with increasing smoothing and restarts
    for restart in range(20):

            break
        if restart == 0:

        else:
            scale = 0.05 + 0.03 * (restart % 5)
            h0 = best_h + scale * rng.randn(n)
            h0 = proj(h0, target)
        alpha = 80.0 + 15.0 * restart
        h_new = try_optimize(h0, n, target, bounds, alpha, 400, 1e-12)
        if h_new is not None:
