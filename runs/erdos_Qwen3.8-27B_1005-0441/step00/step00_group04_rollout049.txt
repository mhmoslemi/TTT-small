import numpy as np
import time
import math


def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx


def repair_to_sum(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h = np.where(mask, h + diff / float(cnt), h)
        h = np.clip(h, 0.0, 1.0)
    return h


def perturb_swap(h, n, sigma, rng):
    h_new = h.copy()
    nsw = int(rng.integers(1, 4))
    for _ in range(nsw):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (i + 1) % n
        delta = float(rng.normal(0.0, sigma))
        lo = max(-h_new[j], h_new[i] - 1.0)
        hi = min(h_new[i], 1.0 - h_new[j])
        if lo >= hi:
            continue
        delta = float(np.clip(delta, lo, hi))
        if abs(delta) < 1e-15:
            continue
        h_new[i] -= delta
        h_new[j] += delta
    return h_new


def block_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % 4) < 2 else 0.0
    return repair_to_sum(h, n)


def binary_round(h, n):
    hb = (h > 0.5).astype(float)
    s = int(hb.sum())
    target = n // 2
    if s != target:
        if s > target:
            idx = np.where(hb == 1)[0]
            order = np.argsort(h[idx])[:s - target]
            hb[idx[order]] = 0
        else:
            idx = np.where(hb == 0)[0]
            order = np.argsort(-h[idx])[:target - s]
            hb[idx[order]] = 1
    return repair_to_sum(hb, n)


def optimize_n(h0, n, dx, rng, time_limit):
    h = h0.copy()
    c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    T = 0.004
    sigma = 0.10
    stuck = 0

    while time.time() < time_limit:
        h_new = perturb_swap(h, n, sigma, rng)
        c5_new = compute_c5(h_new, dx)

        if c5_new <= best_c5:
            best_c5 = c5_new
            best_h = h_new
            h = h_new
            stuck = 0
        else:
            exp_arg = -(c5_new - best_c5) / max(T, 1e-12)
            if exp_arg > -700 and rng.random() < math.exp(exp_arg):
                h = h_new
            stuck += 1

        if stuck > 4000:
            h = best_h + rng.normal(0, 0.03, n)
            h = repair_to_sum(h, n)
            T = 0.004
            sigma = 0.10
            stuck = 0

        # periodic cooling

        if stuck % 5000 == 4999:

            cb = compute_c5(hb, dx)
            if cb < best_c5:
