import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(c), c

def gradient_max_ith(h, c_all, n, m):
    shift = m - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        grad[shift:] += (1.0 - h[:n - shift])
        grad[:n - shift] += (-h[shift:])
    else:
        sp = -shift
        grad[:n - sp] += (1.0 - h[sp:])
        grad[sp:] += (-h[:n - sp])
    return grad

def gradient_smooth(h, c_all, n, tau):
    shifted = c_all - np.max(c_all)
    p = np.exp(tau * shifted)

    grad = np.zeros(n)

            grad[sp:] += p[m] * (-h[:n - sp])
    return grad
