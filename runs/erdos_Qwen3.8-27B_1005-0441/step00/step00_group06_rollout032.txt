import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def solve_lp_n(n, time_limit=120):
    """LP relaxation using circular shifts (approximation)"""
    h_var = cp.Variable(n)
    t = cp.Variable()
    cons = [cp.sum(h_var) == n / 2, h_var >= 0, h_var <= 1, t >= 0]
    for k in range(1, n):  # k=0 gives 0 for binary
        idx = (np.arange(n) + k) % n
        z = cp.Variable(n)
        cons.extend([z >= 0, z <= h_var, z <= h_var[idx], z >= h_var + h_var[idx] - 1])
        cons.append(t >= cp.sum(h_var) - cp.sum(z))
    prob = cp.Problem(cp.Minimize(t), cons)
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_time=time_limit)
            if h_var.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                return np.clip(h_var.value, 0, 1)
        except Exception:
            continue
    return None

def simulated_annealing(h, dx, rng, max_iter=20000, T0=0.1, T_end=0.001):

    c5 = compute_c5(h, dx)
    T = T0
    cooling = (T_end / T0) ** (1.0 / max_iter)
    
    for it in range(max_iter):
        T *= cooling
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - c5

            c5 = new_c5
        else:
            h[i], h[j] = h[j], h[i]
    return h, c5

import time

def solve_lp_n(n, time_limit=120):

            prob.solve(solver=solver, max_time=time_limit)

    return None


def round_to_target(h, target):

    h_r = np.where(h > 0.5, 1.0, 0.0)
    current = h_r.sum()

        excess = current - target
        ones = np.where(h_r == 1.0)[0]
        h_r[ones[:int(np.floor(excess))]] = 0.0
        if excess - int(np.floor(excess)) > 1e-12:
            h_r[ones[int(np.floor(excess))]] = 1.0 - (excess - int(np.floor(excess)))
    elif current < target:
        deficit = target - current
        zeros = np.where(h_r == 0.0)[0]

        if deficit - int(np.floor(deficit)) > 1e-12:
            h_r[zeros[int(np.floor(deficit))]] = deficit - int(np.floor(deficit))

        else:
