def compute_correlations(h):
    """Return the full correlation of h with 1-h, scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def compute_c5(h):
    return np.max(compute_correlations(h))

def smoothed_obj(h, beta, penalty_coeff):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    # Log-sum-exp for max
    max_c = np.max(c)
    log_sum_exp = max_c + np.log(np.sum(np.exp(c - max_c))) / beta
    penalty = penalty_coeff * (np.sum(h) - n / 2.0) ** 2

    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)

            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_objective_function(n, target_sum, beta, penalty_coeff):

        h = np.clip(z, 0.0, 1.0)

        max_c = np.max(c)

        log_sum_exp = max_c + np.log(np.sum(exp_c)) / beta
        penalty = penalty_coeff * (np.sum(h) - target_sum) ** 2
        return log_sum_exp + penalty
    return obj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-15:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))

            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
