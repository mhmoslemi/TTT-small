    corr = np.correlate(h, 1 - h, mode="full")
    return corr * dx

def initial_h_construct(n):
    """Create a reasonably good initial binary h."""
    h = np.zeros(n)
    # Alternating pattern
    for i in range(n):

    # For odd n, n//2 ones and (n-n//2) zeros

    """Compute change in C5 if we swap h[i] and h[j]."""
    # h[i] should be 0, h[j] should be 1 (or vice versa)
    # New h: h[i]=1, h[j]=0 (or vice versa)

    h_new[i], h_new[j] = h_new[j], h_new[i]
    c5_new = compute_c5(h_new, dx)
    c5_old = compute_c5(h, dx)

        for i in range(int(np.ceil(diff))):
            if i < len(zeros_idx):
                h[zeros_idx[i]] = 1.0
        residual = diff - h.sum() + current_sum
        if abs(residual) > 1e-10 and len(zeros_idx) > 0:
            best = zeros_idx[np.argmin(np.abs(h[zeros_idx] - 0.5))]
            h[best] += residual
    elif diff < 0:
        ones_idx = np.where(h == 1)[0]
        for i in range(int(np.ceil(-diff))):
            if i < len(ones_idx):
                h[ones_idx[i]] = 0.0
        residual = diff - (h.sum() - current_sum)
        if abs(residual) > 1e-10 and len(ones_idx) > 0:
            best = ones_idx[np.argmin(np.abs(h[ones_idx] - 0.5))]
            h[best] += residual
    h = np.clip(h, 0, 1)
    return h

def local_search(h, dx, max_iters=5000, time_budget=120, seed=42):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h, dx)
    rng = np.random.RandomState(seed)
    t_s = time.time()
    stuck = 0

    for it in range(max_iters):
        if time.time() - t_s > time_budget:
            break

        zeros = np.where(h < 0.01)[0]
        ones = np.where(h > 0.99)[0]
        halves = np.where((h >= 0.01) & (h <= 0.99))[0]

        best_d = 1e-12
        best_move = None

        for zi in zeros:
            for oi in ones:
                h[zi] = 1.0
                h[oi] = 0.0
                c5n = compute_c5(h, dx)
                d = c5n - c5
                h[zi] = 0.0
                h[oi] = 1.0
                if d < -best_d:
                    best_d = -d
                    best_move = ('swap', zi, oi, 1.0, 0.0)

        for hi in halves:
            for oi in ones:
                h[hi] = 1.0
                h[oi] = 0.5
                c5n = compute_c5(h, dx)
                d = c5n - c5
                h[hi] = 0.5
                h[oi] = 1.0
                if d < -best_d:
                    best_d = -d
                    best_move = ('move', hi, oi, 1.0, 0.5)

        if best_move is None:
            stuck += 1
            if stuck < 30:
                if stuck % 3 == 0:
                    zi = rng.choice(zeros)
                    oi = rng.choice(ones)
                    h[zi] = 1.0
                    h[oi] = 0.0
                    c5 = compute_c5(h, dx)
                elif stuck % 3 == 1 and len(halves) > 0:
                    hi = halves[0]
                    oi = rng.choice(ones)
                    h[hi] = 1.0
                    h[oi] = h[hi]
                    h[hi] = 1.0
                    h[oi] = 0.5
                    c5 = compute_c5(h, dx)
            else:
                break
        else:
            kind, i1, i2, v1, v2 = best_move
            h[i1] = v1
            h[i2] = v2
            c5 -= best_d
            stuck = 0

    h = np.clip(h, 0, 1)
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:
        idx = np.where((h < 1) if diff > 0 else (h > 0))[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0, 1)
    c5 = compute_c5(h, dx)
    return h, c5

def continuous_refine(h, dx, time_budget=60, seed=42):
    n = len(h)
    h0 = h.copy()
    result = None

    def objective(x):
        hi = x[:n]
        t = x[n]
        corr = np.correlate(hi, 1.0 - hi, mode="full")
        c5 = np.max(corr) * dx
        return max(c5, t) + 0.01 * max(0, c5 - t)

    def constraint_sum(x):
        return np.sum(x[:n]) - n / 2.0

    cons = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    for k in range(2 * n - 1):
        pass  # Skip individual correlation constraints for speed; rely on pen

    bounds = [(0, 1)] * n + [(0, 0.5)]
    t_end = time.time() + time_budget

    try:
        res = minimize(objective, h0.tolist() + [compute_c5(h0, dx)],
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success or res.fun < compute_c5(h0, dx):
            hi = np.clip(res.x[:n], 0, 1)
            diff = n / 2.0 - hi.sum()
            if abs(diff) > 1e-8:
                idx = np.where((hi < 1) if diff > 0 else (hi > 0))[0]
                if len(idx) > 0:
                    hi[idx[0]] += diff
                    hi = np.clip(hi, 0, 1)
            c5_new = compute_c5(hi, dx)
            if c5_new < compute_c5(h0, dx) - 1e-10:
                result = (hi, c5_new)
    except Exception:
        pass

    if result is not None:
        return result
    return h0, compute_c5(h0, dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - 15 - elapsed()

    init_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)

    candidates = []
    if init_h is not None:
        candidates.append((len(init_h), init_h.copy(), 'init'))

    for n in [47, 49, 51, 55, 60, 80]:
        candidates.append((n, make_binary(n, seed=seed), 'rand'))
        candidates.append((n, make_thue_morse(n), 'tm'))

    for n, h_init, tag in candidates:
        if remaining() < 20:
            break
        dx = 2.0 / n
        h = h_init.copy()

        # Phase 1: local search
        tb1 = min(60, remaining() - 10)
        h, c5 = local_search(h, dx, max_iters=3000, time_budget=tb1, seed=seed)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_c5
            best_n = n

        # Phase 2: continuous refinement
        if remaining() > 30:
            tb2 = min(40, remaining() - 10)
            h2, c52 = continuous_refine(h, dx, time_budget=tb2, seed=seed)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2.copy()
                best_n = n

        # Phase 3: another local search from refined point
        if remaining() > 20:
            h3, c53 = local_search(h2, dx, max_iters=2000, time_budget=min(30, remaining() - 10), seed=seed + 7)
            if c53 < best_c5:
                best_c5 = c53
                best_h = h3.copy()
                best_n = n

    if best_h is None:
        n = 47
        h = make_binary(n, seed=seed)
        dx = 2.0 / n
        h, best_c5 = local_search(h, dx, max_iters=2000, time_budget=remaining(), seed=seed)
        best_h = h
        best_n = n

    return (best_h, best_c5, best_n)
