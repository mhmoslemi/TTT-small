import numpy as np
import time

def compute_c_vector(h):
    """Compute all overlap values for all shifts"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def compute_gradients_all(h):
    """Compute dC[k]/dh[j] for all k and j. Returns (2n-1 x n) matrix."""
    n = len(h)
    dx = 2.0 / n
    # C[k] = sum_i h[i] * (1-h[i+shift_k]) where shift_k = n-1-k
    # dC[k]/dh[j] = (1-h[j+shift_k]) * 1[j+shift_k valid] - h[j-shift_k] * 1[j-shift_k valid]
    
    num_k = 2 * n - 1
    grad = np.zeros((num_k, n))
    for k in range(num_k):
        shift = n - 1 - k
        for j in range(n):
            val = 0.0
            if 0 <= j + shift < n:
                val += (1.0 - h[j + shift])
            if 0 <= j - shift < n:
                val -= h[j - shift]
            grad[k, j] = val * dx
    return grad

def project_to_feasible(h, n):
    """Project h to [0,1] with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] = np.minimum(h[mask] + diff / np.sum(mask), 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] = np.maximum(h[mask] + diff / np.sum(mask), 0.0)
    return h

def soft_max_gradient(h, temperature=10.0):
    """Compute smoothed gradient using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    c = compute_c_vector(h)
    # Soft-max weights
    c_max = np.max(c)
    weights = np.exp(temperature * (c - c_max))
    weights /= np.sum(weights)
    
    # Gradient is weighted sum of individual gradients
    num_k = len(c)
    grad = np.zeros(n)
    for k in range(num_k):
        shift = n - 1 - k
        for j in range(n):
            val = 0.0
            if 0 <= j + shift < n:
                val += (1.0 - h[j + shift])
            if 0 <= j - shift < n:
                val -= h[j - shift]
            grad[j] += weights[k] * val * dx
    
    max_val = float(np.max(c))
    smooth_val = (1.0 / temperature) * (c_max + np.log(np.sum(weights)))
    return grad, max_val, smooth_val
