    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp

        b2 = min(n, n - lag)

import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(corr)
    c_shifted = corr - c_max
    exp_c = np.exp(tau * c_shifted)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    val = c_max + np.log(sum_exp) / tau
    
    n_corr = len(corr)
    grad = np.zeros(n)
    for k in range(n_corr):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        # Term 1: (1-h[j-lag]) for 0<=j-lag<n → j in [max(0,lag), min(n,n+lag))
        a = max(0, lag)
        b = min(n, n + lag)
        if b > a:
            idx = np.arange(a, b)
            grad[idx] += wk * (1.0 - h[idx - lag])
        # Term 2: -h[j+lag] for 0<=j+lag<n → j in [max(0,-lag), min(n,n-lag))
        a2 = max(0, -lag)
        b2 = min(n, n - lag)
        if b2 > a2:
            idx2 = np.arange(a2, b2)
            grad[idx2] -= wk * h[idx2 + lag]
    grad *= dx
    return val, grad

    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        if diff < 0:
            idx = int(np.argmax(h))
        else:
            idx = int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff, 0, 1))
    return h

        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(95, 0.5)
    n = len(h0)
    target = n / 2.0
    h0 = project_h(h0, target)

    bh = h0.copy()
    bc5 = compute_c5(bh)

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    # Phase 1: Multi-stage PGD with increasing tau
    h = h0.copy()
    tau_list = [15, 40, 100, 300, 800]
    for ti, tau in enumerate(tau_list):
        if rem() < 90:
            break
        alloc = min(150, (rem() - 60) / max(1, len(tau_list) - ti))
        h, c = pgd_optimize(h, tau, 0.01, 8000, target, alloc)
        if c < bc5:
            bc5 = c
            bh = h.copy()

    # Phase 2: Scipy L-BFGS-B with penalty for sum constraint
    if rem() > 70:
        pen = 5000.0
        def fobj(x):
            v = smooth_c5_and_grad(x, 400)[0]
            return v + pen * (np.sum(x) - target) ** 2
        def fgrad(x):
            v, g = smooth_c5_and_grad(x, 400)
            g = g.copy()
            g += 2.0 * pen * (np.sum(x) - target)
            return v + pen * (np.sum(x) - target) ** 2, g
        try:
            r = minimize(fobj, bh, jac=fgrad, method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 500, 'max_time': min(100, rem() - 25), 'gtol': 1e-7})
            ht = project_h(r.x, target)
            ct = compute_c5(ht)
            if ct < bc5:
                bc5 = ct
                bh = ht
        except Exception:
            pass

    # Phase 3: Binary snapping + local refinement
    if rem() > 20:
        h_bin = (bh > 0.5).astype(float)
        h_bin = project_h(h_bin, target)
        cb = compute_c5(h_bin)
        if cb < bc5:
            bc5 = cb
            bh = h_bin.copy()
        # Gradient polish from binary
        h2, c2 = pgd_optimize(bh, 500, 0.005, 3000, target, max(1, rem() - 10))
        if c2 < bc5:
            bc5 = c2
            bh = h2.copy()

    # Phase 4: Small random perturbations (final polish)
    if rem() > 5:
        for i in range(800):
            if rem() < 2:
                break
            s = 0.003 * (1.0 - i / 800.0) + 0.0005
            ht = project_h(bh + rng.normal(0, s, n), target)
            ct = compute_c5(ht)
            if ct < bc5:
                bc5 = ct
                bh = ht.copy()

    # Final feasibility check
    bh = project_h(bh, target)
    bc5 = compute_c5(bh)
    return bh, bc5, n
