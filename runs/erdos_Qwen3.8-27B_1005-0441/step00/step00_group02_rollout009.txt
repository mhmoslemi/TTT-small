import numpy as np
import time
from scipy.optimize import minimize

def project_onto_box_sum(h, S):
    """Project h onto {x: sum(x)=S, 0<=x_i<=1} via bisection on uniform shift."""
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    proj = np.clip(h - t, 0.0, 1.0)
    diff = S - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
    proj = np.clip(proj, 0.0, 1.0)
    # Final exact fix
    diff = S - proj.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(proj - 0.5))
        proj[idx] = np.clip(proj[idx] + diff, 0.0, 1.0)
    return proj

def compute_c5(h, dx):
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    return r.max(), r

def subgrad_L(h, L):
    """Subgradient of f_L(h) = sum_n h[n+L]*(1-h[n]) at h."""
    n = len(h)
    j = np.arange(n)
    g = np.zeros(n)
    m1 = (j - L >= 0) & (j - L < n)
    g[m1] += 1.0 - h[j[m1] - L]
    m2 = (j + L >= 0) & (j + L < n)
    g[m2] -= h[j[m2] + L]
    return g

def optimize_subgrad(h0, S, dx, max_time):

    h = project_onto_box_sum(h0, S)
    best_h = h.copy()
    best_c, _ = compute_c5(h, dx)
    alpha = 0.02
    no_improve = 0
    t0 = time.time()
    
    while time.time() - t0 < max_time:
        c, r = compute_c5(h, dx)
        idx_max = int(np.argmax(r))
        L = idx_max - (n - 1)
        
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            no_improve = 0

            no_improve += 1

        if g_norm < 1e-12:

        g = g / g_norm

def polish_lbfgs(h0, S, dx, max_time):
    """Polish using L-BFGS-B on log-sum-exp smoothed objective."""
    n = len(h0)
    t_smooth = 0.01  # smoothing parameter
    
    def objective(hv):
        h = np.clip(hv, 0, 1)
        r = np.correlate(h, 1-h, mode='full') * dx
        penalties = np.exp(r / t_smooth)
        obj = t_smooth * np.log(penalties.sum())
        obj += 1000.0 * (h.sum() - S) ** 2
        return obj
    
    def gradient(hv):
        h = np.clip(hv, 0, 1)
        r = np.correlate(h, 1-h, mode='full') * dx
        weights = np.exp(r / t_smooth)

            g[m2] -= w * h[j[m2] + L] * dx
        g += 2 * 1000.0 * (h.sum() - S)
        return g

                    h = h_p
                    best_c = c_p
                    best_h = h.copy()
                    no_improve = 0
                    alpha = 0.004
                    improved = True
                    break
            if not improved:
                no_improve = 0
                tau *= 0.93
                alpha = 0.005

        if no_improve % 2000 == 0 and no_improve > 0:
            tau *= 0.995
            if tau < 0.002:
                tau = 0.002

    return best_h, best_c


def polish_lbfgs(h0, S, dx, max_time):
    """Polish using L-BFGS-B on log-sum-exp smoothed objective."""
    n = len(h0)
    tau_s = 0.008

    def obj_func(hv):
        hv2 = np.clip(hv, 0.0, 1.0)
        r = np.correlate(hv2, 1.0 - hv2, mode='full') * dx
        obj = tau_s * np.log(np.sum(np.exp((r - r.max()) / tau_s))) + r.max()
        obj += 800.0 * (hv2.sum() - S) ** 2
        return obj

    def grad_func(hv):
        hv2 = np.clip(hv, 0.0, 1.0)
        r = np.correlate(hv2, 1.0 - hv2, mode='full') * dx
        shifted = (r - r.max()) / tau_s
        w = np.exp(shifted)
        w /= w.sum()
        j = np.arange(n)
        g = np.zeros(n)
        for idx in range(len(w)):
            if w[idx] < 1e-8:
                continue
            L = idx - (n - 1)
            m1 = (j - L >= 0) & (j - L < n)
            if m1.any():
                g[m1] += w[idx] * (1.0 - hv2[j[m1] - L]) * dx
            m2 = (j + L >= 0) & (j + L < n)
            if m2.any():
                g[m2] -= w[idx] * hv2[j[m2] + L] * dx
        g += 1600.0 * (hv2.sum() - S)
        return g

    try:
        res = minimize(obj_func, h0, jac=grad_func, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 2000, 'ftol': 1e-14, 'maxfun': 8000})
        h_pol = project_onto_box_sum(res.x, S)
        c_pol = evaluate_c5(h_pol)
        if c_pol < evaluate_c5(h0):
            return h_pol, c_pol
    except Exception:
        pass
    return h0, evaluate_c5(h0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Get initial values from pre-imported global
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init = np.full(45, 0.5)

    n = len(h_init)
    S = n / 2.0
    dx = 2.0 / n
    h_init = project_onto_box_sum(h_init, S)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Main subgradient optimization
    t1 = min(remaining() * 0.55, 500.0)
    if t1 > 5:
        h1, c1 = subgradient_optimize(h_init, S, dx, t1, seed=seed)
        if c1 < best_c:
            best_c = c1
            best_h = h1

    # Phase 2: L-BFGS polish
    t2 = min(remaining() * 0.15, 120.0)
    if t2 > 15:
        h2, c2 = polish_lbfgs(best_h, S, dx, t2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Multiple random restarts from current best
    phase3_budget = remaining() * 0.2
    if phase3_budget > 20:
        rng = np.random.RandomState(seed + 100)
        per_restart = phase3_budget / 8
        for i in range(8):
            if remaining() < 10:
                break
            scale = 0.05 + 0.02 * i
            h_r = best_h + scale * rng.randn(n)
            h_r = project_onto_box_sum(h_r, S)
            t_r = min(per_restart, remaining() * 0.4)
            if t_r < 2:
                break
            h_r, c_r = subgradient_optimize(h_r, S, dx, t_r, seed=seed + 200 + i)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r

    # Phase 4: Fine subgradient with hard max (single active constraint)
    time_left = remaining() - 5
    if time_left > 10:
        h = best_h.copy()
        alpha_f = 0.001
        for _ in range(int(time_left * 8000)):
            if time.time() - t0 > budget_s - 2:
                break
            r_raw = np.correlate(h, 1.0 - h, mode='full')
            idx_max = int(np.argmax(r_raw))
            g = hard_subgradient(h, idx_max, dx, n)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            g /= gn
            h_new = project_onto_box_sum(h - alpha_f * g, S)
            c_new = evaluate_c5(h_new)
            if c_new < best_c - 1e-11:
                best_c = c_new
                best_h = h_new
            if c_new <= evaluate_c5(h) + 1e-12:
                h = h_new
            else:
                alpha_f *= 0.99
                if alpha_f < 1e-6:
                    alpha_f = 5e-4

    # Phase 5: Final L-BFGS polish
    time_left = remaining() - 3
    if time_left > 10:
        h5, c5 = polish_lbfgs(best_h, S, dx, time_left)
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Ensure exact constraint satisfaction
    best_h = project_onto_box_sum(best_h, S)
    best_c = evaluate_c5(best_h)

    return best_h, best_c, n
