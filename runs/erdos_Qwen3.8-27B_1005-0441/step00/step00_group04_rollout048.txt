```python
    cur = h.copy()

    T = 0.02
    for it in range(n_iter):
        # Flip a pair
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        cur[i] = 0.0
        cur[j] = 1.0
        c5_new = evaluate_c5(cur)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):

            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
        else:
            # Revert
            cur[i] = 1.0
            cur[j] = 0.0
        T *= 0.9998
    return best, best_c5
```

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        if not m.any():
            break
        h[m] = np.clip(h[m] + d / m.sum(), 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.arange(old_n) * 2.0 / old_n
    new_x = np.arange(new_n) * 2.0 / new_n
    new_h = np.interp(new_x, old_x, h)
    return make_feasible(new_h, new_n)

def sa_continuous(h, n, rng, t_start, t_end, n_iter, sig0, sigmin):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    cur = h.copy()
    cur_c5 = best_c5
    sigma = sig0
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = t_start * (t_end / t_start) ** frac
        r = rng.random()
        if r < 0.4:
            h2 = cur.copy()
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                j = (j + 1) % n
            d = float(rng.normal(0, sigma))
            h2[i] += d
            h2[j] -= d
            h_new = make_feasible(h2, n)
        elif r < 0.75:
            h2 = cur.copy()
            k = int(min(3, n))
            idx = rng.choice(n, size=k, replace=False)
            vals = rng.normal(0, sigma, size=k)
            vals -= vals.mean()
            h2[idx] += vals
            h_new = make_feasible(h2, n)
        else:
            h2 = cur.copy()
            k = max(2, int(n * 0.08))
            idx = rng.choice(n, size=k, replace=False)
            vals = rng.normal(0, sigma, size=k)
            vals -= vals.mean()
            h2[idx] += vals
            h_new = make_feasible(h2, n)
        c5_new = evaluate_c5(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-14)):
            cur = h_new
            cur_c5 = c5_new
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
        sigma = max(sigma * 0.9995, sigmin)
    return best, best_c5

def sa_binary(h, n, rng, n_iter):
    h = np.where(h > 0.5, 1.0, 0.0)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    target_ones = n // 2
    while len(ones) > target_ones:
        idx = int(rng.integers(len(ones)))
        h[ones[idx]] = 0.0
        ones = np.delete(ones, idx)
        zeros = np.append(zeros, int(rng.integers(n)))
    while len(ones) < target_ones:
        zidx = int(rng.integers(len(zeros)))
        h[zeros[zidx]] = 1.0
        zeros = np.delete(zeros, zidx)
        ones = np.append(ones, int(rng.integers(n)))
    best = h.copy()
    best_c5 = evaluate_c5(best)
    cur = h.copy()
    cur_c5 = best_c5
    T = 0.015
    no_improve = 0
    for it in range(n_iter):
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[int(rng.integers(len(ones)))])
        j = int(zeros[int(rng.integers(len(zeros)))])
        cur[i] = 0.0
        cur[j] = 1.0
        c5_new = evaluate_c5(cur)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-14)):
            cur_c5 = c5_new
            no_improve = 0
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
        else:
            cur[i] = 1.0
            cur[j] = 0.0
            no_improve += 1
        T *= 0.9997
        if no_improve > 800:
            no_improve = 0
            T = min(T * 3.0, 0.02)
    return best, best_c5

def hybrid_optimize(h, n, rng, max_iter):
    """Alternate between continuous SA and binary local search."""
    best = h.copy()
    best_c5 = evaluate_c5(best)
    cur = h.copy()
    for phase in range(max_iter):
        # Continuous SA (short)
        h_c, c5_c = sa_continuous(cur, n, rng, 0.01, 0.0005, 1500, 0.1, 0.005)
        if c5_c < best_c5:
            best = h_c.copy()
            best_c5 = c5_c
            cur = h_c.copy()
        # Binary local search
        h_b = np.where(cur > 0.5, 1.0, 0.0)
        h_b, c5_b = sa_binary(h_b, n, rng, 2000)
        if c5_b < best_c5:
            best = h_b.copy()
            best_c5 = c5_b
            cur = h_b.copy()
        # Try continuous from best binary
        h_c2, c5_c2 = sa_continuous(best + 0.01 * rng.standard_normal(n), n, rng, 0.005, 0.0002, 800, 0.05, 0.002)
        if c5_c2 < best_c5:
            best = h_c2.copy()
            best_c5 = c5_c2
            cur = h_c2.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    global initial_h_values
    if 'initial_h_values' in dir() and initial_h_values is not None and len(initial_h_values) > 0:
        h0 = np.array(list(initial_h_values), dtype=float)
    else:
        h0 = np.zeros(42)
        h0[::2] = 1.0

    n0 = len(h0)
    h0 = make_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Hybrid optimization at original resolution
    best_h, best_c5 = hybrid_optimize(best_h, n0, rng, 12)
    elapsed = time.time() - t0

    # Phase 2: Try upsampling
    for mult in [2, 3, 4, 5, 6, 8]:
        tn = n0 * mult
        if tn > 500 or time.time() - t0 > budget_s * 0.65:
            break
        h_up = upsample_h(best_h, tn)
        h_opt, c5_opt = hybrid_optimize(h_up, tn, rng, 5)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Phase 3: Random restarts at n0
    n_restarts = 8
    elapsed = time.time() - t0
    while n_restarts > 0 and time.time() - t0 < budget_s * 0.9:
        h_r = rng.random(n0)
        h_r = make_feasible(h_r, n0)
        # Quick SA + binary
        h_r, c5_r = sa_continuous(h_r, n0, rng, 0.05, 0.0003, 2000, 0.2, 0.01)
        h_rb = np.where(h_r > 0.5, 1.0, 0.0)
        h_rb, c5_rb = sa_binary(h_rb, n0, rng, 3000)
        c5_best_r = min(c5_r, c5_rb)
        h_best_r = h_r if c5_r <= c5_rb else h_rb
        if c5_best_r < best_c5:
            best_h = h_best_r.copy()
            best_c5 = c5_best_r
        n_restarts -= 1

    # Phase 4: Final polish with longer binary search on best
    if time.time() - t0 < budget_s * 0.95:
        h_polish = np.where(best_h > 0.5, 1.0, 0.0)
        h_polish, c5_polish = sa_binary(h_polish, len(h_polish), rng, 5000)
        if c5_polish < best_c5:
            best_h = h_polish.copy()
            best_c5 = c5_polish

    # Final: ensure feasibility and recompute
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
```
