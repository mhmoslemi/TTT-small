import numpy as np
import time
from scipy.optimize import minimize

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def project(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room < 1e-12:
                break
            h += diff * room / total_room
        else:
            room = h.copy()
            total_room = room.sum()
            if total_room < 1e-12:
                break
            h += diff * room / total_room
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, k, n):
    """Subgradient of overlap at lag k w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(n):
        j1 = m + k - (n - 1)
        if 0 <= j1 < n:
            grad[m] += (1.0 - h[j1])
        j2 = m - (n - 1) + k
        if 0 <= j2 < n:
            grad[m] -= h[j2]
    return grad * dx

def optimize_h(h0, n, max_iters, time_limit, seed):
    """Optimize h using subgradient method."""
    rng = np.random.default_rng(seed)
    h = project(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    lr = 0.005
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time_limit is not None and it % 100 == 0:
            pass  # time check done outside
        
        overlaps = compute_overlaps(h)
        c5 = float(np.max(overlaps))
        
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        
        # Active constraints: top 3 lags
        top_idx = np.argsort(overlaps)[-3:]
        grad = np.zeros(n)
        for k in top_idx:
            grad += subgrad_for_lag(h, k, n)
        grad /= len(top_idx)
        
        # Momentum update
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project(h_new, n)
        
        if compute_c5(h_new) <= c5 + 1e-10:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
        
        if it % 200 == 0 and lr > 1e-6:
            lr *= 0.8
    
    return best_h, best_c5

        h_f, c5_f = optimize_from(h_f, nf, 5000, seed + 9999, deadline)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        h_f2, c5_f2 = nearby_local_search(best_h, nf, seed + 12345, min(deadline, time.time() + 5))
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_f2
    
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
