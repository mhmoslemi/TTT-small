import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0 - 1e-14
            if not m.any():
                break
            cap = 1.0 - h[m]
            s = min(d / max(cap.sum(), 1e-30), 1.0)
            h[m] = np.minimum(1.0, h[m] + cap * s)
        else:
            m = h > 1e-14
            if not m.any():
                break
            exc = h[m]
            s = min((-d) / max(exc.sum(), 1e-30), 1.0)
            h[m] = np.maximum(0.0, h[m] - exc * s)
    return h


def soft_obj(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    vmax = vals.max()
    shifted = np.clip(tau * (vals - vmax), -400, 400)
    return float(vmax + (1.0 / tau) * np.log(np.sum(np.exp(shifted))))

    sm = soft_obj(h, n, tau)
    viol = (h.sum() - n * 0.5) ** 2
    return sm + pen * viol
