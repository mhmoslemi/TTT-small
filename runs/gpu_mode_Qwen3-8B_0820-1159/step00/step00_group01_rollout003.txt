import torch
import triton
import triton.language as tl

# Module docstring: Implements the TriMul operation for protein structure prediction
# using Triton kernels for high performance. This kernel handles the forward pass
# of the TriMul module, including layer normalization, projections, gates, and
# the core tensor multiplication operation. The kernel is designed for CUDA L40S
# hardware with Ada Lovelace architecture.

@triton.jit
def trimul_kernel(
    input_ptr, mask_ptr, norm_weight_ptr, norm_bias_ptr,
    left_proj_weight_ptr, right_proj_weight_ptr,
    left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
    to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
    output_ptr,
    B: tl.constexpr, N: tl.constexpr, C: tl.constexpr, H: tl.constexpr,
    BLOCK_B: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Get program ID and compute the position in the B dimension
    pid_b = tl.program_id(0)
    # Get position in the N dimension (sequence length)
    pid_n = tl.program_id(1)
    # Compute the position in the C dimension (channel dimension)
    pid_c = tl.program_id(2)

    # Offset for current B dimension
    offset_b = pid_b * BLOCK_B + tl.arange(0, BLOCK_B)
    # Offset for current sequence dimension (N)
    offset_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Offset for current C dimension
    offset_c = pid_c * BLOCK_C + tl.arange(0, BLOCK_C)

    # Initialize output
    output = tl.zeros((BLOCK_B, BLOCK_N, BLOCK_C), dtype=tl.float32)

    # Load normalization parameters
    norm_weight = tl.load(norm_weight_ptr + offset_c, mask=offset_c < C, other=0.0)
    norm_bias = tl.load(norm_bias_ptr + offset_c, mask=offset_c < C, other=0.0)

    # Load projection weights
    left_proj_weight = tl.load(left_proj_weight_ptr + offset_c[:, None] * H + tl.arange(0, H), 
                               mask=(offset_c[:, None] < C) & (tl.arange(0, H) < H), other=0.0)
    right_proj_weight = tl.load(right_proj_weight_ptr + offset_c[:, None] * H + tl.arange(0, H), 
                                mask=(offset_c[:, None] < C) & (tl.arange(0, H) < H), other=0.0)

    # Load gate weights
    left_gate_weight = tl.load(left_gate_weight_ptr + offset_c[:, None] * H + tl.arange(0, H), 
                               mask=(offset_c[:, None] < C) & (tl.arange(0, H) < H), other=0.0)
    right_gate_weight = tl.load(right_gate_weight_ptr + offset_c[:, None] * H + tl.arange(0, H), 
                                mask=(offset_c[:, None] < C) & (tl.arange(0, H) < H), other=0.0)
    out_gate_weight = tl.load(out_gate_weight_ptr + offset_c[:, None] * H + tl.arange(0, H), 
                              mask=(offset_c[:, None] < C) & (tl.arange(0, H) < H), other=0.0)

    # Load normalization weights for the intermediate layer
    to_out_norm_weight = tl.load(to_out_norm_weight_ptr + offset_c, mask=offset_c < H, other=0.0)
    to_out_norm_bias = tl.load(to_out_norm_bias_ptr + offset_c, mask=offset_c < H, other=0.0)

    # Load output weights
    to_out_weight = tl.load(to_out_weight_ptr + offset_c[:, None] * H + tl.arange(0, H), 
                            mask=(offset_c[:, None] < C) & (tl.arange(0, H) < H), other=0.0)

    # Initialize intermediate tensors
    left = tl.zeros((BLOCK_B, BLOCK_N, BLOCK_H), dtype=tl.float32)
    right = tl.zeros((BLOCK_B, BLOCK_N, BLOCK_H), dtype=tl.float32)
    left_gate = tl.zeros((BLOCK_B, BLOCK_N, BLOCK_H), dtype=tl.float32)
    right_gate = tl.zeros((BLOCK_B, BLOCK_N, BLOCK_H), dtype=tl.float32)
    out_gate = tl.zeros((BLOCK_B, BLOCK_N, BLOCK_H), dtype=tl.float32)
    intermediate = tl.zeros((BLOCK_B, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Load input tensor
    input = tl.load(input_ptr + offset_b[:, None, None] * N * C + offset_n[None, :, None] * C + offset_c[None, None, :], 
                    mask=(offset_b[:, None, None] < B) & (offset_n[None, :, None] < N) & (offset_c[None, None, :] < C), other=0.0)

    # Normalize input
    input = (input - tl.mean(input, axis=(1, 2))) * norm_weight + norm_bias

    # Project to hidden dimension
    left = tl.dot(input, left_proj_weight)
    right = tl.dot(input, right_proj_weight)

    # Load mask and apply it to left and right
    mask = tl.load(mask_ptr + offset_b[:, None, None] * N * N + offset_n[None, :, None] * N + tl.arange(0, N), 
                   mask=(offset_b[:, None, None] < B) & (offset_n[None, :, None] < N) & (tl.arange(0, N) < N), other=0.0)
    mask = mask.to(tl.float32)
    left = left * mask
    right = right * mask

    # Apply gates
    left_gate = tl.dot(input, left_gate_weight).sigmoid()
    right_gate = tl.dot(input, right_gate_weight).sigmoid()
    out_gate = tl.dot(input, out_gate_weight).sigmoid()

    # Apply gate activations
    left = left * left_gate
    right = right * right_gate

    # Compute the core multiplication
    for k in range(0, N, BLOCK_K):
        k_mask = (k + tl.arange(0, BLOCK_K)) < N
        a = tl.load(input_ptr + offset_b[:, None, None] * N * C + offset_n[None, :, None] * C + offset_c[None, None, :], 
                    mask=(offset_b[:, None, None] < B) & (offset_n[None, :, None] < N) & (offset_c[None, None, :] < C), other=0.0)
        a = (a - tl.mean(a, axis=(1, 2))) * norm_weight + norm_bias
        a = tl.dot(a, left_proj_weight)
        a = a * mask
        a = a * tl.dot(input, left_gate_weight).sigmoid()
        b = tl.load(input_ptr + offset_b[:, None, None] * N * C + offset_n[None, :, None] * C + offset_c[None, None, :], 
                    mask=(offset_b[:, None, None] < B) & (offset_n[None, :, None] < N) & (offset_c[None, None, :] < C), other=0.0)
        b = (b - tl.mean(b, axis=(1, 2))) * norm_weight + norm_bias
        b = tl.dot(b, right_proj_weight)
        b = b * mask
        b = b * tl.dot(input, right_gate_weight).sigmoid()
        intermediate = intermediate + tl.dot(a, b)

    # Normalize intermediate result
    intermediate = (intermediate - tl.mean(intermediate, axis=(1, 2))) * to_out_norm_weight + to_out_norm_bias

    # Apply output gate
    intermediate = intermediate * out_gate

    # Project back to original dimension
    output = tl.dot(intermediate, to_out_weight)

    # Save the output
    tl.store(output_ptr + offset_b[:, None, None] * N * C + offset_n[None, :, None] * C + offset_c[None, None, :], 
             output, mask=(offset_b[:, None, None] < B) & (offset_n[None, :, None] < N) & (offset_c[None, None, :] < C))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, C, H = config.get("bs", 1), config["seqlen"], config["dim"], config["hidden_dim"]

    # Cast to float32
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.to(torch.float32)

    # Initialize output
    output = torch.empty((B, N, N, C), device=input_tensor.device, dtype=torch.float32)

    # Triton kernel parameters
    BLOCK_B = 128
    BLOCK_N = 128
    BLOCK_C = 128
    BLOCK_H = 128
    BLOCK_K = 128

    # Prepare pointers
    input_ptr = input_tensor.data_ptr()
    mask_ptr = mask.data_ptr()
    norm_weight = weights["norm.weight"].contiguous().data_ptr()
    norm_bias = weights["norm.bias"].contiguous().data_ptr()
    left_proj_weight = weights["left_proj.weight"].contiguous().data_ptr()
    right_proj_weight = weights["right_proj.weight"].contiguous().data_ptr()
    left_gate_weight = weights["left_gate.weight"].contiguous().data_ptr()
    right_gate_weight = weights["right_gate.weight"].contiguous().data_ptr()
    out_gate_weight = weights["out_gate.weight"].contiguous().data_ptr()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous().data_ptr()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous().data_ptr()
    to_out_weight = weights["to_out.weight"].contiguous().data_ptr()
    output_ptr = output.data_ptr()

    # Grid dimensions
    grid_x = triton.cdiv(B, BLOCK_B)
    grid_y = triton.cdiv(N, BLOCK_N)
    grid_z = triton.cdiv(C, BLOCK_C)

    # Launch the Triton kernel
    trimul_kernel[grid_x, grid_y, grid_z](
        input_ptr, mask_ptr, norm_weight, norm_bias,
        left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_norm_bias, to_out_weight,
        output_ptr,
        B, N, C, H,
        BLOCK_B, BLOCK_N, BLOCK_C, BLOCK_H, BLOCK_K
    )

    return output
