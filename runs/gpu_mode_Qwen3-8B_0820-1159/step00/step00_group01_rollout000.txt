import torch
import triton
import triton.language as tl
import math

# Define the kernel for the TriMul operation
@triton.jit
def trimal_kernel(
    input_ptr, mask_ptr, weights_norm_weight_ptr, weights_norm_bias_ptr,
    weights_left_proj_ptr, weights_right_proj_ptr, weights_left_gate_ptr, weights_right_gate_ptr,
    weights_out_gate_ptr, weights_to_out_norm_weight_ptr, weights_to_out_norm_bias_ptr,
    weights_to_out_ptr,
    batch_size, seq_len, dim, hidden_dim,
    BLOCK_BATCH: tl.constexpr, BLOCK_SEQ: tl.constexpr, BLOCK_DIM: tl.constexpr,
    BLOCK_HIDDEN: tl.constexpr
):

    # Compute the program IDs
    pid_batch = tl.program_id(0)
    pid_seq_i = tl.program_id(1)
    pid_seq_j = tl.program_id(2)
    pid_dim = tl.program_id(3)

    # Compute offsets
    offs_batch = pid_batch * BLOCK_BATCH + tl.arange(0, BLOCK_BATCH)
    offs_seq_i = pid_seq_i * BLOCK_SEQ + tl.arange(0, BLOCK_SEQ)
    offs_seq_j = pid_seq_j * BLOCK_SEQ + tl.arange(0, BLOCK_SEQ)
    offs_dim = pid_dim * BLOCK_DIM + tl.arange(0, BLOCK_DIM)
    offs_hidden = tl.arange(0, BLOCK_HIDDEN)

    # Load weights for all operations
    # LayerNorm (norm)
    norm_weight = tl.load(weights_norm_weight_ptr + offs_dim, mask=offs_dim < dim, other=0.0)
    norm_bias = tl.load(weights_norm_bias_ptr + offs_dim, mask=offs_dim < dim, other=0.0)

    # Linear projections
    left_proj_weight = tl.load(weights_left_proj_ptr + offs_hidden[:, None] * hidden_dim + offs_dim[None, :], mask=(offs_hidden[:, None] < hidden_dim) & (offs_dim[None, :] < dim), other=0.0)
    right_proj_weight = tl.load(weights_right_proj_ptr + offs_hidden[:, None] * hidden_dim + offs_dim[None, :], mask=(offs_hidden[:, None] < hidden_dim) & (offs_dim[None, :] < dim), other=0.0)

    # Gates
    left_gate_weight = tl.load(weights_left_gate_ptr + offs_hidden[:, None] * hidden_dim + offs_dim[None, :], mask=(offs_hidden[:, None] < hidden_dim) & (offs_dim[None, :] < dim), other=0.0)
    right_gate_weight = tl.load(weights_right_gate_ptr + offs_hidden[:, None] * hidden_dim + offs_dim[None, :], mask=(offs_hidden[:, None] < hidden_dim) & (offs_dim[None, :] < dim), other=0.0)
    out_gate_weight = tl.load(weights_out_gate_ptr + offs_hidden[:, None] * hidden_dim + offs_dim[None, :], mask=(offs_hidden[:, None] < hidden_dim) & (offs_dim[None, :] < dim), other=0.0)

    # Normalization for out
    to_out_norm_weight = tl.load(weights_to_out_norm_weight_ptr + offs_hidden, mask=offs_hidden < hidden_dim, other=0.0)
    to_out_norm_bias = tl.load(weights_to_out_norm_bias_ptr + offs_hidden, mask=offs_hidden < hidden_dim, other=0.0)
    to_out_weight = tl.load(weights_to_out_ptr + offs_dim[:, None] * hidden_dim + offs_hidden[None, :], mask=(offs_dim[:, None] < dim) & (offs_hidden[None, :] < hidden_dim), other=0.0)

    # Load input
    input = tl.load(input_ptr + (offs_batch[:, None, None, None] * batch_size * seq_len * seq_len + offs_seq_i[:, None, None] * seq_len * seq_len + offs_seq_j[:, None] * seq_len + offs_dim[None, :]), mask=(offs_batch[:, None, None, None] < batch_size) & (offs_seq_i[:, None, None] < seq_len) & (offs_seq_j[:, None] < seq_len) & (offs_dim[None, :] < dim), other=0.0)
    input = input - norm_bias[None, None, None, offs_dim]
    input = input * norm_weight[None, None, None, offs_dim]

    # Compute projections
    left = tl.dot(input, left_proj_weight)
    right = tl.dot(input, right_proj_weight)

    # Load mask and apply it
    mask = tl.load(mask_ptr + (offs_batch[:, None, None] * batch_size * seq_len * seq_len + offs_seq_i[:, None] * seq_len * seq_len + offs_seq_j[:, None] * seq_len), mask=(offs_batch[:, None, None] < batch_size) & (offs_seq_i[:, None] < seq_len) & (offs_seq_j[:, None] < seq_len), other=0.0)
    mask = mask.unsqueeze(-1)
    left = left * mask
    right = right * mask

    # Compute gates
    # Left gate
    left_gate = tl.dot(input, left_gate_weight)
    left_gate = left_gate.sigmoid()
    left = left * left_gate

    # Right gate
    right_gate = tl.dot(input, right_gate_weight)
    right_gate = right_gate.sigmoid()
    right = right * right_gate

    # Compute out
    out = tl.dot(left, right.transpose(2, 1))
    out = out + to_out_norm_bias[None, None, None, offs_hidden] * to_out_norm_weight[None, None, None, offs_hidden]
    out = out * out_gate
    out = tl.dot(out, to_out_weight)

    # Store result
    tl.store(output_ptr + (offs_batch[:, None, None, None] * batch_size * seq_len * seq_len + offs_seq_i[:, None, None] * seq_len * seq_len + offs_seq_j[:, None] * seq_len + offs_dim[None, :]), out, mask=(offs_batch[:, None, None, None] < batch_size) & (offs_seq_i[:, None, None] < seq_len) & (offs_seq_j[:, None] < seq_len) & (offs_dim[None, :] < dim))


def custom_kernel(data):
    """
    Triton kernel implementation of the Triangle Multiplicative Update (TriMul) operator for BioML models.
    This implementation is optimized for performance on NVIDIA L40S devices.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Allocate output tensor
    output = torch.empty_like(input_tensor)

    # Compute grid for the kernel
    num_batch = triton.cdiv(batch_size, BLOCK_BATCH)
    num_seq = triton.cdiv(seq_len, BLOCK_SEQ)
    num_dim = triton.cdiv(dim, BLOCK_DIM)
    num_hidden = triton.cdiv(hidden_dim, BLOCK_HIDDEN)

    # Launch the kernel
    num_warps = 16
    kernel_config = triton.Config(
        {"BLOCK_BATCH": BLOCK_BATCH, "BLOCK_SEQ": BLOCK_SEQ, "BLOCK_DIM": BLOCK_DIM, "BLOCK_HIDDEN": BLOCK_HIDDEN},
        num_warps=num_warps
    )

    trimal_kernel[triton.cdiv(batch_size, kernel_config.block_batch) * triton.cdiv(seq_len, kernel_config.block_seq) * triton.cdiv(seq_len, kernel_config.block_seq) * triton.cdiv(dim, kernel_config.block_dim)](
        input_tensor, mask, weights["norm.weight"], weights["norm.bias"],
        weights["left_proj.weight"], weights["right_proj.weight"],
        weights["left_gate.weight"], weights["right_gate.weight"],
        weights["out_gate.weight"], weights["to_out_norm.weight"],
        weights["to_out_norm.bias"], weights["to_out.weight"],
        output,
        batch_size, seq_len, dim, hidden_dim,
        kernel_config.block_batch, kernel_config.block_seq, kernel_config.block_dim, kernel_config.block_hidden
    )

    return output
