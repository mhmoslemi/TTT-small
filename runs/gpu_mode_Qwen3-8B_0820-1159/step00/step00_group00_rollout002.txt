import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    input_ptr, mask_ptr, 
    left_proj_weight_ptr, right_proj_weight_ptr, left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
    to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
    norm_weight_ptr, norm_bias_ptr, 
    out_ptr,
    bs, seq_len, dim, hidden_dim,
    BLOCK_SEQ: tl.constexpr, BLOCK_DIM: tl.constexpr,
    has_mask: tl.constexpr
):
    # Thread ids
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets
    seq_offsets = pid_i * BLOCK_SEQ + tl.arange(0, BLOCK_SEQ)
    seq_offsets_j = pid_j * BLOCK_SEQ + tl.arange(0, BLOCK_SEQ)
    
    # Mask
    if has_mask:
        mask = tl.load(mask_ptr + pid_b * seq_len * seq_len + seq_offsets[:, None] * seq_len + seq_offsets_j[None, :], 
                      mask=(seq_offsets[:, None] < seq_len) & (seq_offsets_j[None, :] < seq_len), 
                      other=0.0)
    else:
        mask = tl.ones((BLOCK_SEQ, BLOCK_SEQ), dtype=tl.float32)

    # Load norm weights
    norm_weight = tl.load(norm_weight_ptr + tl.arange(0, dim), mask=tl.arange(0, dim) < dim, other=0.0)
    norm_bias = tl.load(norm_bias_ptr + tl.arange(0, dim), mask=tl.arange(0, dim) < dim, other=0.0)

    # Compute x_norm = (x - mean) / std
    x = tl.load(input_ptr + pid_b * seq_len * seq_len * dim + seq_offsets[:, None] * seq_len * dim + seq_offsets_j[None, :] * dim + tl.arange(0, dim),
                mask=(seq_offsets[:, None] < seq_len) & (seq_offsets_j[None, :] < seq_len) & (tl.arange(0, dim) < dim),
                other=0.0)
    x_norm = (x - tl.sum(x * mask, axis=(0, 1), keepdim=True) / tl.sum(mask, axis=(0, 1), keepdim=True)) / tl.sqrt(tl.sum((x - tl.sum(x * mask, axis=(0, 1), keepdim=True) / tl.sum(mask, axis=(0, 1), keepdim=True)) * (x - tl.sum(x * mask, axis=(0, 1), keepdim=True) / tl.sum(mask, axis=(0, 1), keepdim=True)) * mask, axis=(0, 1), keepdim=True) / tl.sum(mask, axis=(0, 1), keepdim=True) + 1e-6)
    x_norm = x_norm * norm_weight + norm_bias

    # Compute left_proj, right_proj
    left_proj_weight = tl.load(left_proj_weight_ptr + tl.arange(0, hidden_dim)[:, None] * dim + tl.arange(0, dim)[None, :], 
                               mask=(tl.arange(0, hidden_dim)[:, None] < hidden_dim) & (tl.arange(0, dim)[None, :] < dim), 
                               other=0.0)
    right_proj_weight = tl.load(right_proj_weight_ptr + tl.arange(0, hidden_dim)[:, None] * dim + tl.arange(0, dim)[None, :], 
                                mask=(tl.arange(0, hidden_dim)[:, None] < hidden_dim) & (tl.arange(0, dim)[None, :] < dim), 
                                other=0.0)

    left = tl.dot(x_norm, left_proj_weight)
    right = tl.dot(x_norm, right_proj_weight)

    # Apply mask
    left = left * mask
    right = right * mask

    # Compute gates
    left_gate_weight = tl.load(left_gate_weight_ptr + tl.arange(0, hidden_dim)[:, None] * dim + tl.arange(0, dim)[None, :], 
                               mask=(tl.arange(0, hidden_dim)[:, None] < hidden_dim) & (tl.arange(0, dim)[None, :] < dim), 
                               other=0.0)
    right_gate_weight = tl.load(right_gate_weight_ptr + tl.arange(0, hidden_dim)[:, None] * dim + tl.arange(0, dim)[None, :], 
                                mask=(tl.arange(0, hidden_dim)[:, None] < hidden_dim) & (tl.arange(0, dim)[None, :] < dim), 
                                other=0.0)
    out_gate_weight = tl.load(out_gate_weight_ptr + tl.arange(0, hidden_dim)[:, None] * dim + tl.arange(0, dim)[None, :], 
                              mask=(tl.arange(0, hidden_dim)[:, None] < hidden_dim) & (tl.arange(0, dim)[None, :] < dim), 
                              other=0.0)

    left_gate = tl.dot(x_norm, left_gate_weight).sigmoid()
    right_gate = tl.dot(x_norm, right_gate_weight).sigmoid()
    out_gate = tl.dot(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute out
    out = tl.dot(left, right)

    # Apply to_out_norm
    to_out_norm_weight = tl.load(to_out_norm_weight_ptr + tl.arange(0, hidden_dim), mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)
    to_out_norm_bias = tl.load(to_out_norm_bias_ptr + tl.arange(0, hidden_dim), mask=tl.arange(0, hidden_dim) < hidden_dim, other=0.0)

    out = (out - tl.sum(out * mask, axis=(0, 1), keepdim=True) / tl.sum(mask, axis=(0, 1), keepdim=True)) / tl.sqrt(tl.sum((out - tl.sum(out * mask, axis=(0, 1), keepdim=True) / tl.sum(mask, axis=(0, 1), keepdim=True)) * (out - tl.sum(out * mask, axis=(0, 1), keepdim=True) / tl.sum(mask, axis=(0, 1), keepdim=True)) * mask, axis=(0, 1), keepdim=True) / tl.sum(mask, axis=(0, 1), keepdim=True) + 1e-6)
    out = out * to_out_norm_weight + to_out_norm_bias

    # Apply out_gate
    out = out * out_gate

    # Apply to_out
    to_out_weight = tl.load(to_out_weight_ptr + tl.arange(0, dim)[:, None] * hidden_dim + tl.arange(0, hidden_dim)[None, :], 
                            mask=(tl.arange(0, dim)[:, None] < dim) & (tl.arange(0, hidden_dim)[None, :] < hidden_dim), 
                            other=0.0)

    out = tl.dot(out, to_out_weight)

    # Store output
    out_ptr_current = out_ptr + pid_b * seq_len * seq_len * dim + seq_offsets[:, None] * seq_len * dim + seq_offsets_j[None, :] * dim
    tl.store(out_ptr_current, out, mask=(seq_offsets[:, None] < seq_len) & (seq_offsets_j[None, :] < seq_len))


def custom_kernel(data):
    """
    Efficient Triton kernel implementation of the TriMul module.
    
    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [batch_size, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [batch_size, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Prepare input and mask
    input_ptr = input_tensor.data_ptr()
    mask_ptr = mask.data_ptr()
    has_mask = (mask.dtype != torch.float32)

    # Prepare weights
    norm_weight_ptr = weights["norm.weight"].data_ptr()
    norm_bias_ptr = weights["norm.bias"].data_ptr()

    left_proj_weight_ptr = weights["left_proj.weight"].data_ptr()
    right_proj_weight_ptr = weights["right_proj.weight"].data_ptr()
    left_gate_weight_ptr = weights["left_gate.weight"].data_ptr()
    right_gate_weight_ptr = weights["right_gate.weight"].data_ptr()
    out_gate_weight_ptr = weights["out_gate.weight"].data_ptr()

    to_out_norm_weight_ptr = weights["to_out_norm.weight"].data_ptr()
    to_out_norm_bias_ptr = weights["to_out_norm.bias"].data_ptr()
    to_out_weight_ptr = weights["to_out.weight"].data_ptr()

    # Output tensor
    output = torch.empty(bs, seq_len, seq_len, dim, device=input_tensor.device, dtype=input_tensor.dtype)

    # Launch Triton kernel
    BLOCK_SEQ = 64
    BLOCK_DIM = 128
    grid = (triton.cdiv(bs, 1), triton.cdiv(seq_len, BLOCK_SEQ), triton.cdiv(seq_len, BLOCK_SEQ))
    tri_mul_kernel[grid](
        input_ptr, mask_ptr, 
        left_proj_weight_ptr, right_proj_weight_ptr, left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
        to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
        norm_weight_ptr, norm_bias_ptr, 
        output.data_ptr(),
        bs, seq_len, dim, hidden_dim,
        BLOCK_SEQ=BLOCK_SEQ, BLOCK_DIM=BLOCK_DIM,
        has_mask=has_mask
    )

    return output
