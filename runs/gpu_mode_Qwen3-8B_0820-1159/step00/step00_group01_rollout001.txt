import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    
    batch_size = input_tensor.shape[0]
    seq_len = input_tensor.shape[1]
    _, _, _, _ = input_tensor.shape
    
    # Precompute masks
    mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]
    mask = mask.to(torch.float32)

    # Load weights into Triton-compatible format
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    
    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    
    # Norm layers
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    
    # Output projection
    to_out_weight = weights["to_out.weight"]
    
    # Convert weights to device pointers
    norm_weight_ptr = tl.make_zero_init(norm_weight)
    norm_bias_ptr = tl.make_zero_init(norm_bias)
    
    left_proj_weight_ptr = tl.make_zero_init(left_proj_weight)
    right_proj_weight_ptr = tl.make_zero_init(right_proj_weight)
    left_gate_weight_ptr = tl.make_zero_init(left_gate_weight)
    right_gate_weight_ptr = tl.make_zero_init(right_gate_weight)
    out_gate_weight_ptr = tl.make_zero_init(out_gate_weight)
    
    to_out_norm_weight_ptr = tl.make_zero_init(to_out_norm_weight)
    to_out_norm_bias_ptr = tl.make_zero_init(to_out_norm_bias)
    
    to_out_weight_ptr = tl.make_zero_init(to_out_weight)

    # Triton kernel definition
    @triton.jit
    def tri_mul_kernel(
        input_ptr, mask_ptr, 
        left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_norm_bias_ptr, to_out_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK_SEQ: tl.constexpr, BLOCK_DIM: tl.constexpr, BLOCK_HIDDEN: tl.constexpr
    ):
        # Determine the current position in the batch, sequence, and hidden dimension
        pid_batch = tl.program_id(0)
        pid_seq = tl.program_id(1)
        pid_dim = tl.program_id(2)
        
        # Compute offsets
        offset_batch = pid_batch * batch_size
        offset_seq = pid_seq * seq_len
        offset_dim = pid_dim * dim
        
        # Load input
        input = tl.load(input_ptr + offset_batch + offset_seq + tl.arange(0, seq_len)[:, None] * seq_len + tl.arange(0, seq_len)[None, :] * seq_len + tl.arange(0, dim), dtype=tl.float32)
        
        # Apply LayerNorm
        norm_weight = tl.load(norm_weight_ptr + tl.arange(0, dim), dtype=tl.float32)
        norm_bias = tl.load(norm_bias_ptr + tl.arange(0, dim), dtype=tl.float32)
        
        input = (input - tl.math.mean(input, axis=-1, keepdims=True)) * norm_weight + norm_bias
        input = tl.where(mask_ptr + offset_batch + offset_seq + tl.arange(0, seq_len)[:, None] * seq_len + tl.arange(0, seq_len)[None, :] * seq_len, input, 0.0)
        
        # Compute left and right projections
        left_proj = tl.dot(input, left_proj_ptr + tl.arange(0, dim)[:, None] * hidden_dim + tl.arange(0, hidden_dim)[None, :], axis=-1)
        right_proj = tl.dot(input, right_proj_ptr + tl.arange(0, dim)[:, None] * hidden_dim + tl.arange(0, hidden_dim)[None, :], axis=-1)
        
        left_gate = tl.dot(input, left_gate_ptr + tl.arange(0, dim)[:, None] * hidden_dim + tl.arange(0, hidden_dim)[None, :], axis=-1)
        right_gate = tl.dot(input, right_gate_ptr + tl.arange(0, dim)[:, None] * hidden_dim + tl.arange(0, hidden_dim)[None, :], axis=-1)
        out_gate = tl.dot(input, out_gate_ptr + tl.arange(0, dim)[:, None] * hidden_dim + tl.arange(0, hidden_dim)[None, :], axis=-1)
        
        left_gate = tl.sigmoid(left_gate)
        right_gate = tl.sigmoid(right_gate)
        out_gate = tl.sigmoid(out_gate)
        
        left = left_proj * left_gate
        right = right_proj * right_gate
        
        # Compute the matrix multiplication (out = einsum('... i k d, ... j k d -> ... i j d')
        # This is an O(N^3) operation that is not vectorized, so we will need to implement it explicitly
        
        # Output buffer
        out = tl.zeros((seq_len, dim), dtype=tl.float32)
        
        # Iterate over all k and accumulate the results
        for k in range(seq_len):
            # Load current left and right slices
            left_k = tl.load(left + offset_seq + k * seq_len + tl.arange(0, seq_len)[:, None] * seq_len + tl.arange(0, dim), dtype=tl.float32)
            right_k = tl.load(right + offset_seq + k * seq_len + tl.arange(0, seq_len)[:, None] * seq_len + tl.arange(0, dim), dtype=tl.float32)
            
            # Compute dot product
            temp = tl.dot(left_k, right_k, axis=-1)
            
            # Accumulate into the output
            out += temp
        
        # Apply LayerNorm
        to_out_norm_weight = tl.load(to_out_norm_weight_ptr + tl.arange(0, hidden_dim), dtype=tl.float32)
        to_out_norm_bias = tl.load(to_out_norm_bias_ptr + tl.arange(0, hidden_dim), dtype=tl.float32)
        
        out = (out - tl.math.mean(out, axis=-1, keepdims=True)) * to_out_norm_weight + to_out_norm_bias
        
        # Apply output gate
        out = out * out_gate
        
        # Apply final projection
        out = tl.dot(out, to_out_ptr + tl.arange(0, hidden_dim)[:, None] * dim + tl.arange(0, dim)[None, :], axis=-1)
        
        # Store the result
        tl.store(output_ptr + offset_batch + offset_seq + tl.arange(0, seq_len)[:, None] * seq_len + tl.arange(0, seq_len)[None, :] * seq_len + tl.arange(0, dim), out, mask=tl.arange(0, seq_len)[:, None] < seq_len & tl.arange(0, seq_len)[None, :] < seq_len)

    # Define kernel parameters
    BLOCK_SEQ = 64
    BLOCK_DIM = 64
    BLOCK_HIDDEN = 128

    # Launch kernel
    output = torch.empty((batch_size, seq_len, seq_len, dim), dtype=torch.float32, device=input_tensor.device)
    output_ptr = output.data_ptr()
    
    grid = (triton.cdiv(batch_size, 1), triton.cdiv(seq_len, BLOCK_SEQ), triton.cdiv(dim, BLOCK_DIM))
    tri_mul_kernel[grid](
        input_tensor.data_ptr(), mask.data_ptr(),
        left_proj_weight_ptr, right_proj_weight_ptr, left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
        to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK_SEQ=BLOCK_SEQ, BLOCK_DIM=BLOCK_DIM, BLOCK_HIDDEN=BLOCK_HIDDEN
    )

    return output
