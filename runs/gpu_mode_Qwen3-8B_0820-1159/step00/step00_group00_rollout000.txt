import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Convert mask to float32 and broadcast to channel dimension
    mask = mask.to(torch.float32).unsqueeze(-1)

    # Prepare weights for Triton kernel
    norm_weight = weights["norm.weight"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    # Output tensor
    output = torch.empty((batch_size, seq_len, seq_len, dim), device=input_tensor.device, dtype=torch.float32)

    # Launch Triton kernel
    kernel_launcher(output, input_tensor, mask, norm_weight, left_proj_weight, right_proj_weight,
                    left_gate_weight, right_gate_weight, out_gate_weight, to_out_norm_weight, to_out_norm_bias,
                    to_out_weight, batch_size, seq_len, dim, hidden_dim)

    return output

@triton.jit
def kernel_launcher(output_ptr, input_ptr, mask_ptr, norm_weight_ptr, left_proj_weight_ptr, right_proj_weight_ptr,
                    left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr, to_out_norm_weight_ptr,
                    to_out_norm_bias_ptr, to_out_weight_ptr, batch_size, seq_len, dim, hidden_dim,
                    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    local_batch = pid % batch_size
    local_seq_len = seq_len
    batch_idx = pid // batch_size

    # Prepare offsets for batch and sequence dimension
    batch_offsets = batch_idx * local_seq_len * local_seq_len * dim
    seq_offsets = tl.arange(0, local_seq_len)
    seq_offsets = seq_offsets[:, None, None] * local_seq_len * dim

    # Create thread indices for i, j, k
    i = tl.arange(0, local_seq_len)
    j = tl.arange(0, local_seq_len)
    k = tl.arange(0, local_seq_len)

    # Load input tensor
    input = tl.load(input_ptr + batch_offsets + seq_offsets + i[:, None, None] * dim + k[None, None, :], mask=(i < local_seq_len) & (k < local_seq_len), other=0.0)

    # Apply norm
    norm_weight = tl.load(norm_weight_ptr)
    norm_bias = tl.load(norm_weight_ptr + 1)
    input = input * norm_weight + norm_bias

    # Apply projections
    left_proj = tl.dot(input, left_proj_weight_ptr)
    right_proj = tl.dot(input, right_proj_weight_ptr)

    # Apply gates
    left_gate = tl.dot(input, left_gate_weight_ptr).sigmoid()
    right_gate = tl.dot(input, right_gate_weight_ptr).sigmoid()
    out_gate = tl.dot(input, out_gate_weight_ptr).sigmoid()

    # Apply masks
    mask = tl.load(mask_ptr + batch_offsets + seq_offsets + i[:, None, None] * dim, mask=(i < local_seq_len), other=0.0)
    left_proj = left_proj * mask
    right_proj = right_proj * mask

    # Apply gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # Compute output using einsum equivalent
    out = tl.zeros((local_seq_len, local_seq_len, dim), dtype=tl.float32)
    for k in range(0, seq_len, BLOCK_SIZE):
        k_mask = (k + tl.arange(0, BLOCK_SIZE)) < seq_len
        a = left_proj + (k + tl.arange(0, BLOCK_SIZE))[:, None, None] * seq_len
        b = right_proj + (k + tl.arange(0, BLOCK_SIZE))[:, None, None] * seq_len
        acc = tl.dot(a, b)
        out += acc

    # Apply to_out norm
    to_out_norm_weight = tl.load(to_out_norm_weight_ptr)
    to_out_norm_bias = tl.load(to_out_norm_bias_ptr)
    out = out * to_out_norm_weight + to_out_norm_bias

    # Apply out gate
    out = out * out_gate

    # Apply final projection
    out = tl.dot(out, to_out_weight_ptr)

    # Store output
    output_ptr_idx = output_ptr + batch_offsets + seq_offsets + i[:, None, None] * dim
    tl.store(output_ptr_idx, out, mask=(i < local_seq_len) & (j < local_seq_len))
