import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Prepare masks
    mask = mask.unsqueeze(-1).to(torch.float32)

    # Load weights and reshape as needed
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous().view(hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"].contiguous().view(hidden_dim, dim)
    left_gate_weight = weights["left_gate.weight"].contiguous().view(hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"].contiguous().view(hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"].contiguous().view(hidden_dim, dim)
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous().view(dim, hidden_dim)

    # Output tensor
    output = torch.empty((batch_size, seq_len, seq_len, dim), device=input_tensor.device, dtype=torch.float32)

    # Load input and apply normalization
    input_data = input_tensor.to(torch.float32)
    norm_mean = torch.sum(input_data * norm_weight, dim=-1, keepdim=True) + norm_bias
    norm_output = (input_data - norm_mean) * norm_weight

    # Apply projection and gates
    left_proj = torch.einsum('bikd,dh->bikh', norm_output, left_proj_weight)
    right_proj = torch.einsum('bikd,dh->bikh', norm_output, right_proj_weight)

    left_gate = torch.einsum('bikd,dh->bikh', norm_output, left_gate_weight)
    right_gate = torch.einsum('bikd,dh->bikh', norm_output, right_gate_weight)
    out_gate = torch.einsum('bikd,dh->bikh', norm_output, out_gate_weight)

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left_proj * mask * left_gate
    right = right_proj * mask * right_gate

    # Triton kernel for the main TriMul operation
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, output_ptr,
        batch_size, seq_len, dim, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_output_b, stride_output_i, stride_output_j, stride_output_d,
        BLOCK_SEQ: tl.constexpr, BLOCK_HIDDEN: tl.constexpr
    ):
        pid_batch = tl.program_id(0)
        pid_seq = tl.program_id(1)
        pid_block = tl.program_id(2)

        # Compute offsets
        offs_batch = pid_batch * batch_size + tl.arange(0, batch_size)
        offs_seq = pid_seq * BLOCK_SEQ + tl.arange(0, BLOCK_SEQ)
        offs_block = pid_block * BLOCK_HIDDEN + tl.arange(0, BLOCK_HIDDEN)

        # Compute block indices for the output
        off_i = offs_seq[:, None] + offs_block[None, :]
        off_j = tl.arange(0, seq_len)
        off_d = tl.arange(0, dim)

        # Load left and right data
        left = tl.load(left_ptr + (offs_batch[:, None, None] * stride_left_b) + (offs_seq[None, :, None] * stride_left_i) + (offs_block[None, None, :] * stride_left_k) + (off_d[None, None, :] * stride_left_d),
                        mask=(offs_seq[:, None] < seq_len) & (offs_block[None, :] < hidden_dim), other=0.0)

        right = tl.load(right_ptr + (offs_batch[:, None, None] * stride_right_b) + (offs_seq[None, :, None] * stride_right_i) + (offs_block[None, None, :] * stride_right_k) + (off_d[None, None, :] * stride_right_d),
                        mask=(offs_seq[:, None] < seq_len) & (offs_block[None, :] < hidden_dim), other=0.0)

        # Compute the product
        acc = tl.zeros((BLOCK_SEQ, BLOCK_HIDDEN, dim), dtype=tl.float32)
        for k in range(seq_len):
            a = left[k, :, :]
            b = right[k, :, :]
            acc += tl.dot(a, b)

        # Store the result
        output = tl.load(output_ptr + (offs_batch[:, None, None] * stride_output_b) + (offs_seq[None, :, None] * stride_output_i) + (off_j[None, None, :] * stride_output_j) + (off_d[None, None, :] * stride_output_d),
                        mask=(offs_seq[:, None] < seq_len) & (off_j[None, :] < seq_len) & (off_d[None, :] < dim), other=0.0)

        output = acc * out_gate
        tl.store(output_ptr + (offs_batch[:, None, None] * stride_output_b) + (offs_seq[None, :, None] * stride_output_i) + (off_j[None, None, :] * stride_output_j) + (off_d[None, None, :] * stride_output_d),
                 output, mask=(offs_seq[:, None] < seq_len) & (off_j[None, :] < seq_len) & (off_d[None, :] < dim))

    # Configure kernel launch
    grid = (batch_size, triton.cdiv(seq_len, 128), triton.cdiv(hidden_dim, 32))
    BLOCK_SEQ = 128
    BLOCK_HIDDEN = 32
    stride_left_b = input_tensor.stride(0)
    stride_left_i = input_tensor.stride(1)
    stride_left_k = input_tensor.stride(2)
    stride_left_d = input_tensor.stride(3)
    stride_right_b = input_tensor.stride(0)
    stride_right_i = input_tensor.stride(1)
    stride_right_k = input_tensor.stride(2)
    stride_right_d = input_tensor.stride(3)
    stride_output_b = output.stride(0)
    stride_output_i = output.stride(1)
    stride_output_j = output.stride(2)
    stride_output_d = output.stride(3)

    # Launch kernel
    tri_mul_kernel[grid](
        left_ptr=left.contiguous().data_ptr(),
        right_ptr=right.contiguous().data_ptr(),
        output_ptr=output.contiguous().data_ptr(),
        batch_size=batch_size,
        seq_len=seq_len,
        dim=dim,
        hidden_dim=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_d,
        stride_output_b=stride_output_b,
        stride_output_i=stride_output_i,
        stride_output_j=stride_output_j,
        stride_output_d=stride_output_d,
        BLOCK_SEQ=BLOCK_SEQ,
        BLOCK_HIDDEN=BLOCK_HIDDEN
    )

    # Apply normalization and return output
    to_out_norm_mean = torch.mean(output, dim=(1, 2), keepdim=True)
    to_out_norm_std = torch.std(output, dim=(1, 2), keepdim=True)
    output = (output - to_out_norm_mean) * to_out_norm_weight + to_out_norm_bias
    output = torch.einsum('bijkd,dk->bijkd', output, to_out_weight)

    return output
