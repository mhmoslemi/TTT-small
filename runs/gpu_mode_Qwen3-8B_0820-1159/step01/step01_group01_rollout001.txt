import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len = input_tensor.shape[0], input_tensor.shape[1]

    # Precompute the mask
    mask = mask.unsqueeze(-1).to(input_tensor.device)
    mask = mask.to(torch.float32)

    # Access the given weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Allocate the output tensor
    output = torch.empty((batch_size, seq_len, seq_len, dim), device=input_tensor.device, dtype=torch.float32)

    # Launch the Triton kernel
    kernel_launcher(
        input_tensor, mask, norm_weight, norm_bias, left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight, to_out_norm_weight, to_out_norm_bias, to_out_weight,
        output, batch_size, seq_len, dim, hidden_dim
    )
    return output

@triton.jit
def kernel_launcher(
    input_ptr, mask_ptr, norm_weight_ptr, norm_bias_ptr,
    left_proj_weight_ptr, right_proj_weight_ptr,
    left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
    to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
    output_ptr, batch_size, seq_len, dim, hidden_dim,
    BLOCK: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Parameters
    seq_len = tl.constexpr(seq_len)
    dim = tl.constexpr(dim)
    hidden_dim = tl.constexpr(hidden_dim)
    BLOCK_K = tl.constexpr(BLOCK_K)

    # Calculate the block index
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Initialize offsets
    m_offsets = pid_m * BLOCK + tl.arange(0, BLOCK)
    n_offsets = pid_n * BLOCK + tl.arange(0, BLOCK)
    k_offsets = tl.arange(0, BLOCK_K)

    # Load the norm weights and biases
    norm_weight = tl.load(norm_weight_ptr, root=0)
    norm_bias = tl.load(norm_bias_ptr, root=0)

    # Load the input tensor
    input = tl.load(input_ptr + m_offsets[:, None, None] * input_tensor.strides[0] + 
                   k_offsets[None, :, None] * input_tensor.strides[1] + 
                   n_offsets[None, None, :] * input_tensor.strides[2], 
                   mask=(m_offsets[:, None] < batch_size) & (k_offsets[None, :] < seq_len) & (n_offsets[None, :] < seq_len), 
                   other=0.0)

    # Apply LayerNorm
    input = input - tl.load(norm_weight_ptr + tl.arange(0, dim)) * input + tl.load(norm_bias_ptr + tl.arange(0, dim))

    # Project left and right
    left_proj_weight = tl.load(left_proj_weight_ptr, root=0)
    right_proj_weight = tl.load(right_proj_weight_ptr, root=0)

    left = tl.load(input_ptr + m_offsets[:, None, None] * input_tensor.strides[0] + 
                   k_offsets[None, :, None] * input_tensor.strides[1] + 
                   n_offsets[None, None, :] * input_tensor.strides[2], 
                   mask=(m_offsets[:, None] < batch_size) & (k_offsets[None, :] < seq_len) & (n_offsets[None, :] < seq_len), 
                   other=0.0)

    left = left @ left_proj_weight
    right = left @ right_proj_weight

    # Apply mask
    mask = tl.load(mask_ptr + m_offsets[:, None] * mask.strides[0] + 
                   n_offsets[None, :] * mask.strides[1], 
                   mask=(m_offsets[:, None] < batch_size) & (n_offsets[None, :] < seq_len), 
                   other=0.0)

    left = left * mask
    right = right * mask

    # Apply gates
    left_gate_weight = tl.load(left_gate_weight_ptr, root=0)
    right_gate_weight = tl.load(right_gate_weight_ptr, root=0)
    out_gate_weight = tl.load(out_gate_weight_ptr, root=0)

    left_gate = tl.load(input_ptr + m_offsets[:, None, None] * input_tensor.strides[0] + 
                        k_offsets[None, :, None] * input_tensor.strides[1] + 
                        n_offsets[None, None, :] * input_tensor.strides[2], 
                        mask=(m_offsets[:, None] < batch_size) & (k_offsets[None, :] < seq_len) & (n_offsets[None, :] < seq_len), 
                        other=0.0)

    left_gate = left_gate @ left_gate_weight
    left_gate = left_gate.sigmoid()

    right_gate = tl.load(input_ptr + m_offsets[:, None, None] * input_tensor.strides[0] + 
                         k_offsets[None, :, None] * input_tensor.strides[1] + 
                         n_offsets[None, None, :] * input_tensor.strides[2], 
                         mask=(m_offsets[:, None] < batch_size) & (k_offsets[None, :] < seq_len) & (n_offsets[None, :] < seq_len), 
                         other=0.0)

    right_gate = right_gate @ right_gate_weight
    right_gate = right_gate.sigmoid()

    out_gate = tl.load(input_ptr + m_offsets[:, None, None] * input_tensor.strides[0] + 
                       k_offsets[None, :, None] * input_tensor.strides[1] + 
                       n_offsets[None, None, :] * input_tensor.strides[2], 
                       mask=(m_offsets[:, None] < batch_size) & (k_offsets[None, :] < seq_len) & (n_offsets[None, :] < seq_len), 
                       other=0.0)

    out_gate = out_gate @ out_gate_weight
    out_gate = out_gate.sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute the einsum: left[i,k,d] * right[j,k,d] -> out[i,j,d]
    out = tl.zeros((BLOCK, BLOCK, hidden_dim), dtype=tl.float32)
    for k in range(0, seq_len, BLOCK_K):
        k_mask = (k + k_offsets) < seq_len
        left_k = tl.load(left_ptr + m_offsets[:, None] * left.strides[0] + 
                         k_offsets[None, :] * left.strides[1] + 
                         n_offsets[None, None, :] * left.strides[2], 
                         mask=(m_offsets[:, None] < batch_size) & (k_offsets[None, :] < seq_len) & (n_offsets[None, :] < seq_len), 
                         other=0.0)

        right_k = tl.load(right_ptr + m_offsets[:, None] * right.strides[0] + 
                          k_offsets[None, :] * right.strides[1] + 
                          n_offsets[None, None, :] * right.strides[2], 
                          mask=(m_offsets[:, None] < batch_size) & (k_offsets[None, :] < seq_len) & (n_offsets[None, :] < seq_len), 
                          other=0.0)

        out += tl.dot(left_k, right_k)

    # Apply LayerNorm to the output
    to_out_norm_weight = tl.load(to_out_norm_weight_ptr, root=0)
    to_out_norm_bias = tl.load(to_out_norm_bias_ptr, root=0)
    output = out @ to_out_norm_weight + to_out_norm_bias

    # Apply out_gate and to_out
    output = output * out_gate
    output = output @ to_out_weight

    # Store the output
    output = tl.load(output_ptr + m_offsets[:, None] * output.strides[0] + 
                     n_offsets[None, :] * output.strides[1] + 
                     k_offsets[None, None, :] * output.strides[2], 
                     mask=(m_offsets[:, None] < batch_size) & (n_offsets[None, :] < seq_len) & (k_offsets[None, None, :] < dim), 
                     other=0.0)
