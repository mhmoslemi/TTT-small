import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Load all weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Compute norm
    norm_weight = tl.load(norm_weight, None, 0)
    norm_bias = tl.load(norm_bias, None, 0)
    input_tensor = tl.load(input_tensor, None, 0)
    input_tensor = input_tensor * norm_weight + norm_bias

    # Compute left_proj and right_proj
    left_proj = tl.zeros((batch_size, seq_len, seq_len, hidden_dim), dtype=tl.float32)
    right_proj = tl.zeros((batch_size, seq_len, seq_len, hidden_dim), dtype=tl.float32)
    for b in range(batch_size):
        for i in range(seq_len):
            for k in range(seq_len):
                for d in range(hidden_dim):
                    left_proj[b, i, k, d] = tl.load(left_proj_weight + d * hidden_dim + k * dim, dtype=tl.float32) * input_tensor[b, i, k, d]
                    right_proj[b, k, i, d] = tl.load(right_proj_weight + d * hidden_dim + k * dim, dtype=tl.float32) * input_tensor[b, k, i, d]

    # Compute left_gate and right_gate
    left_gate = tl.zeros((batch_size, seq_len, seq_len, hidden_dim), dtype=tl.float32)
    right_gate = tl.zeros((batch_size, seq_len, seq_len, hidden_dim), dtype=tl.float32)
    for b in range(batch_size):
        for i in range(seq_len):
            for k in range(seq_len):
                for d in range(hidden_dim):
                    left_gate[b, i, k, d] = tl.sigmoid(tl.load(left_gate_weight + d * hidden_dim + k * dim, dtype=tl.float32) * input_tensor[b, i, k, d])
                    right_gate[b, k, i, d] = tl.sigmoid(tl.load(right_gate_weight + d * hidden_dim + k * dim, dtype=tl.float32) * input_tensor[b, k, i, d])

    # Apply gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # Apply mask
    mask = tl.load(mask, None, 0)
    mask = mask.unsqueeze(-1)
    left_proj = left_proj * mask
    right_proj = right_proj * mask

    # Compute output
    output = tl.zeros((batch_size, seq_len, seq_len, dim), dtype=tl.float32)
    for b in range(batch_size):
        for i in range(seq_len):
            for j in range(seq_len):
                for k in range(seq_len):
                    for d in range(dim):
                        output[b, i, j, d] += tl.dot(left_proj[b, i, k, :], right_proj[b, j, k, :])

    # Normalize and apply out_gate
    to_out_norm_weight = tl.load(to_out_norm_weight, None, 0)
    to_out_norm_bias = tl.load(to_out_norm_bias, None, 0)
    output = output * to_out_norm_weight + to_out_norm_bias
    out_gate = tl.zeros((batch_size, seq_len, seq_len, hidden_dim), dtype=tl.float32)
    for b in range(batch_size):
        for i in range(seq_len):
            for j in range(seq_len):
                for d in range(hidden_dim):
                    out_gate[b, i, j, d] = tl.sigmoid(tl.load(out_gate_weight + d * hidden_dim + i * seq_len * seq_len + j * seq_len, dtype=tl.float32) * input_tensor[b, i, j, d])
    output = output * out_gate

    # Apply to_out
    to_out_weight = tl.load(to_out_weight, None, 0)
    output = output @ to_out_weight

    return output
