import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # A more sophisticated initial approximation with multiple segments
    n_points = 200
    h = np.zeros(n_points)
    # Segment 1: [0, 0.25]
    h[:int(n_points * 0.25)] = 0.5
    # Segment 2: [0.25, 0.5]
    h[int(n_points * 0.25):int(n_points * 0.5)] = 1.0
    # Segment 3: [0.5, 0.75]
    h[int(n_points * 0.5):int(n_points * 0.75)] = 0.5
    # Segment 4: [0.75, 1.0]
    h[int(n_points * 0.75):int(n_points * 1.0)] = 0.0
    # Segment 5: [1.0, 1.25]
    h[int(n_points * 1.0):int(n_points * 1.25)] = 0.5
    # Segment 6: [1.25, 1.5]
    h[int(n_points * 1.25):int(n_points * 1.5)] = 1.0
    # Segment 7: [1.5, 1.75]
    h[int(n_points * 1.5):int(n_points * 1.75)] = 0.5
    # Segment 8: [1.75, 2.0]
    h[int(n_points * 1.75):] = 0.0
    # Normalize to meet the integral constraint
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a higher resolution initial approximation
    n_points = 100
    initial_h = initial_h_values()
    best_h = np.array(initial_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Adaptive refinement and local optimization
    def refine_h(h):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        def objective(x):
            h_values = x.copy()
            return evaluate_erdos_solution(h_values)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        opt_params = {
            'method': 'SLSQP',
            'bounds': bounds,
            'constraints': [{'type': 'eq', 'fun': constraint_sum}],
            'options': {'ftol': 1e-10, 'maxiter': 200}
        }
        result = optimize.minimize(
            fun=objective,
            x0=h,
            **opt_params
        )
        return result.x
    
    # Function to perform local optimization
    def local_optimize(x0):
        return refine_h(x0)
    
    # Adaptive perturbation and refinement strategy
    def global_optimize(x0):
        results = []
        for _ in range(3):
            # Apply a structured perturbation
            perturbation = np.zeros(n_points)
            perturbation[np.random.choice(n_points, size=20, replace=False)] = np.random.uniform(-0.1, 0.1, size=20)
            perturbed = x0 + perturbation
            perturbed = np.clip(perturbed, 0.0, 1.0)
            # Adjust to maintain the integral constraint
            perturbed = perturbed / np.sum(perturbed) * n_points / 2
            refined = local_optimize(perturbed)
            results.append(refined)
        
        evaluated = [(evaluate_erdos_solution(h), h) for h in results]
        best_eval, best_h = min(evaluated, key=lambda x: x[0])
        return best_h
    
    # Perform multiple refinements with early stopping
    for _ in range(5):
        best_h = global_optimize(best_h)
        best_c5 = evaluate_erdos_solution(best_h)
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Check if time is up
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
