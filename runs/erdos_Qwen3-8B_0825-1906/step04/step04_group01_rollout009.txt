import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    sum_h = np.sum(h) * 2.0 / n_points
    h = h * (1.0 / sum_h)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    n_points = 60
    best_c5 = 100.0
    best_h = initial_h_values()
    
    # Evaluate initial solution
    initial_c5 = evaluate_erdos_solution(best_h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
    
    # Hyperparameters
    num_initial_points = 20
    num_iterations = 50
    mutation_rate = 0.05
    elite_size = 10
    
    def evaluate(solution):
        return evaluate_erdos_solution(solution)
    
    # Initial population
    initial_population = [generate_random_h(n_points) for _ in range(num_initial_points)]
    initial_population_fitness = [evaluate(sol) for sol in initial_population]
    
    # Best solution tracking
    best_index = np.argmin(initial_population_fitness)
    best_h = initial_population[best_index]
    best_c5 = initial_population_fitness[best_index]
    
    # Optimization loop
    for iteration in range(num_iterations):
        # Create new population
        new_population = []
        # Add elites
        elite_indices = np.argsort(initial_population_fitness)[:elite_size]
        for idx in elite_indices:
            new_population.append(initial_population[idx])
        
        # Generate new solutions
        for _ in range(len(new_population), num_initial_points):
            # Select two parents
            parent1 = np.random.choice(elite_indices)
            parent2 = np.random.choice(elite_indices)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = initial_population[parent1][:crossover_point] + initial_population[parent2][crossover_point:]
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            
            # Normalize
            sum_child = np.sum(child) * 2.0 / n_points
            child = np.array(child) * (1.0 / sum_child)
            
            new_population.append(child.tolist())
        
        # Evaluate new population
        population_fitness = [evaluate(sol) for sol in new_population]
        
        # Update best solution
        current_best_index = np.argmin(population_fitness)
        current_best = new_population[current_best_index]
        current_best_fitness = population_fitness[current_best_index]
        
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = current_best
        
        # Update population
        initial_population = new_population
        initial_population_fitness = population_fitness
        
        # Early stopping
        if time.time() - start_time > budget_s - 1:
            break
    
    # Local optimization of the best solution found
    def objective(x):
        h_values = x.copy()
        h_values = np.clip(h_values, 0.0, 1.0)
        sum_h_current = np.sum(h_values) * 2.0 / n_points
        if np.abs(sum_h_current - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h_current
            h_values = h_values * scaling_factor
        return evaluate_erdos_solution(h_values)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = optimize.minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
