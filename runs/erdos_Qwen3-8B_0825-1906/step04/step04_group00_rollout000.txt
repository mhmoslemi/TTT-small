```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a refined and randomized initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)  # Keep it consistent for the rest of the run
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Particle Swarm Optimization for Global Search
    def particle_swarm_optimization(n_particles=30, n_iterations=50, inertia=0.5, cognitive=1.2, social=1.2):
        # Initialize particles
        particles = np.random.uniform(0, 1, (n_particles, n_points))
        particles = particles / np.sum(particles, axis=1, keepdims=True) * n_points / 2
        
        # Initialize velocities
        velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
        
        # Initialize personal bests
        personal_best = particles.copy()
        personal_best_values = np.array([evaluate_erdos_solution(p) for p in personal_best])
        
        # Initialize global best
        global_best = particles[0].copy()
        global_best_value = personal_best_values[0]
        
        for _ in range(n_iterations):
            # Update velocities and positions
            for i in range(n_particles):
                r1 = np.random.rand(n_points)
                r2 = np.random.rand(n_points)
                velocities[i] = inertia * velocities[i] + cognitive * r1 * (personal_best[i] - particles[i]) + social * r2 * (global_best - particles[i])
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0.0, 1.0)
                particles[i] = particles[i] / np.sum(particles[i]) * n_points / 2
            
            # Evaluate fitness
            current_values = np.array([evaluate_erdos_solution(p) for p in particles])
            
            # Update personal bests
            for i in range(n_particles):
                if current_values[i] < personal_best_values[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_values[i] = current_values[i]
            
            # Update global best
            best_idx = np.argmin(personal_best_values)
            if personal_best_values[best_idx] < global_best_value:
                global_best = personal_best[best_idx].copy()
                global_best_value = personal_best_values[best_idx]
            
            # Update best solution
            if global_best_value < best_c5:
                best_c5 = global_best_value
                best_h = global_best.copy()
        
        return best_h, best_c5
    
    # Run particle swarm optimization
    try:
        best_h, best_c5 = particle_swarm_optimization(n_particles=30, n_iterations=50, inertia=0.5, cognitive=1.2, social=1.2)
    except:
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        return result.x
    
    # Refine the best solution
    best_h = local_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perturb and refine
    for _ in range(3):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Run local refinement on the perturbed solution
        refined_h = local_refinement(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)
        
        # Update best solution
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
```
