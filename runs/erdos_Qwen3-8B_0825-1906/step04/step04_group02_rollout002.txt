import numpy as np
import scipy.optimize as optimize
import time
from functools import lru_cache
import random
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Return a higher-resolution initial approximation of the solution
    # This is a more refined and balanced initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    h = np.array(initial_h)
    
    # Define the objective function
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define the constraint function
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for variables
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # First optimization run with SLSQP, with tighter tolerances and more iterations
    result = optimize.minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    
    # Check if time is up
    if time.time() - start_time > budget_s - 1:
        h_values = result.x
        final_c5 = evaluate_erdos_solution(h_values)
        return (h_values.tolist(), final_c5, n_points)
    
    # Evaluate the initial solution
    h_values = result.x
    final_c5 = evaluate_erdos_solution(h_values)
    
    # Try a few more iterations with different perturbations and methods
    for _ in range(5):
        # Perturb the current solution slightly
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = h_values + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Use different optimization method for perturbed solution
        result_perturbed = optimize.minimize(
            fun=objective,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        # Check if this perturbed solution gives a better result
        current_c5 = evaluate_erdos_solution(result_perturbed.x)
        if current_c5 < final_c5:
            final_c5 = current_c5
            h_values = result_perturbed.x
    
    # Try a different initial point with a different structure (e.g., more balanced)
    # This prevents getting stuck in local optima
    new_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.2 <= x <= 0.8:
            new_h[i] = 1.0
    new_h = new_h / np.sum(new_h) * n_points / 2
    new_h = np.clip(new_h, 0.0, 1.0)
    
    # Optimize from this new initial point
    result_new = optimize.minimize(
        fun=objective,
        x0=new_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 500}
    )
    
    # Evaluate and compare
    current_c5 = evaluate_erdos_solution(result_new.x)
    if current_c5 < final_c5:
        final_c5 = current_c5
        h_values = result_new.x
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_values)
    return (h_values.tolist(), final_c5, n_points)
