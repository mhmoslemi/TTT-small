import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return the initial approximation of the solution
    # This is a placeholder - in practice, it would be loaded from a file or generated
    # For demonstration, we return a simple symmetric function
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h) * 2.0 / n_points
    h = h * (1.0 / sum_h)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    n_points = 120  # Double the number of samples for better resolution
    best_c5 = 100.0
    best_h = initial_h_values()
    
    # Evaluate initial solution
    initial_c5 = evaluate_erdos_solution(best_h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
    
    # Hybrid optimization approach: simulated annealing with local refinement
    # Parameters
    temp_initial = 1.0
    temp_final = 0.01
    cooling_rate = 0.995
    iterations = 1000
    num_samples = 50
    
    # Evaluate a solution
    def evaluate(solution):
        return evaluate_erdos_solution(solution)
    
    # Generate a random neighbor solution
    def generate_neighbor(solution):
        neighbor = solution.copy()
        idx = np.random.randint(0, len(solution))
        neighbor[idx] = np.random.uniform(0, 1)
        # Normalize to satisfy the integral constraint
        sum_neighbor = np.sum(neighbor) * 2.0 / n_points
        neighbor = neighbor * (1.0 / sum_neighbor)
        return neighbor
    
    # Initialize with the initial solution
    current_solution = best_h.copy()
    current_c5 = evaluate(current_solution)
    
    # Simulated annealing
    temp = temp_initial
    for _ in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
        for _ in range(num_samples):
            neighbor = generate_neighbor(current_solution)
            neighbor_c5 = evaluate(neighbor)
            delta = neighbor_c5 - current_c5
            if delta < 0 or np.random.random() < np.exp(-delta / temp):
                current_solution = neighbor
                current_c5 = neighbor_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_solution
        temp *= cooling_rate
    
    # Local optimization of the best solution found
    def objective(x):
        h_values = x.copy()
        # Ensure h_values are within [0, 1]
        h_values = np.clip(h_values, 0.0, 1.0)
        # Ensure the integral constraint is met
        sum_h_current = np.sum(h_values) * 2.0 / n_points
        if np.abs(sum_h_current - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h_current
            h_values = h_values * scaling_factor
        # Compute the overlap integral
        return evaluate_erdos_solution(h_values)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = optimize.minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
