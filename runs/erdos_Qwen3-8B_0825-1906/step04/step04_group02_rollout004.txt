```python
import numpy as np
import scipy.optimize as optimize
import time
from functools import lru_cache
import random
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Return a higher-resolution initial approximation of the solution
    # This is a more refined and balanced initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    
    # Try with different n_points
    n_points_candidates = [200, 300, 400, 500]
    best_c5 = float('inf')
    best_h = initial_h
    
    for n in n_points_candidates:
        if time.time() - start_time > budget_s - 1:
            break
        # Create initial h with n points
        h = np.zeros(n)
        for i in range(n):
            x = i * 2.0 / n
            if 0.25 <= x <= 0.75:
                h[i] = 1.0
        h = h / np.sum(h) * n / 2
        h = h.tolist()
        
        # Function to evaluate the objective function
        def objective(x):
            h_values = x.copy()
            return evaluate_erdos_solution(h_values)
        
        # Function to enforce the constraint
        def constraint_sum(x):
            return np.sum(x) - n / 2
        
        # Bounds for variables
        bounds = [(0.0, 1.0) for _ in range(n)]
        
        # First optimization run with SLSQP
        result = optimize.minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        
        # Evaluate the current solution
        current_c5 = evaluate_erdos_solution(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
        
        # Perturb the solution and try to improve
        for _ in range(3):
            if time.time() - start_time > budget_s - 1:
                break
            # Perturb the current solution slightly
            perturbation = np.random.uniform(-0.1, 0.1, size=n)
            perturbed_h = result.x + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h / np.sum(perturbed_h) * n / 2
            
            result_perturbed = optimize.minimize(
                fun=objective,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            current_c5_perturbed = evaluate_erdos_solution(result_perturbed.x)
            if current_c5_perturbed < best_c5:
                best_c5 = current_c5_perturbed
                best_h = result_perturbed.x
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
```
