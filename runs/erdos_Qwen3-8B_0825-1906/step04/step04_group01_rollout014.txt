import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return the initial approximation of the solution
    # This is a placeholder - in practice, it would be loaded from a file or generated
    # For demonstration, we return a simple symmetric function
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h) * 2.0 / n_points
    h = h * (1.0 / sum_h)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    n_points = 100  # Use more samples for better accuracy
    best_c5 = 1.0
    best_h = initial_h_values()
    
    # Evaluate initial solution
    initial_c5 = evaluate_erdos_solution(best_h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
    
    # Local optimization with gradient descent
    def objective(x):
        h_values = x.copy()
        # Ensure h_values are within [0, 1]
        h_values = np.clip(h_values, 0.0, 1.0)
        # Ensure the integral constraint is met
        sum_h_current = np.sum(h_values) * 2.0 / n_points
        if np.abs(sum_h_current - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h_current
            h_values = h_values * scaling_factor
        # Compute the overlap integral
        return evaluate_erdos_solution(h_values)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = np.array(best_h)
    
    # Optimization with constraints
    result = optimize.minimize(
        objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # If time remains, perform a more refined optimization
    if time.time() - start_time < budget_s - 1:
        # Gradient-based refinement
        result_refine = optimize.minimize(
            objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
            options={'maxiter': 50, 'ftol': 1e-10, 'disp': False}
        )
        best_h = result_refine.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    # If time remains, perform a small genetic algorithm refinement
    if time.time() - start_time < budget_s - 2:
        population_size = 10
        generations = 5
        mutation_rate = 0.05
        
        def fitness_function(h_values):
            return evaluate_erdos_solution(h_values)
        
        # Initialize population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * (1.0 / (np.sum(h) * 2.0 / n_points)) for h in population]
        population_fitness = [fitness_function(h) for h in population]
        
        for _ in range(generations):
            # Sort by fitness
            sorted_indices = np.argsort(population_fitness)
            sorted_population = [population[i] for i in sorted_indices]
            sorted_fitness = [population_fitness[i] for i in sorted_indices]
            
            # Select elites
            elites = sorted_population[:5]
            elite_fitness = sorted_fitness[:5]
            
            # Create new population
            new_population = elites.copy()
            for _ in range(population_size - 5):
                # Select two parents
                parent1 = elites[np.random.randint(5)]
                parent2 = elites[np.random.randint(5)]
                
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                
                # Mutation
                for i in range(n_points):
                    if np.random.random() < mutation_rate:
                        child[i] = np.random.uniform(0, 1)
                
                # Normalize to satisfy the integral constraint
                sum_child = np.sum(child) * 2.0 / n_points
                if np.abs(sum_child - 1.0) > 1e-6:
                    child = child * (1.0 / sum_child)
                
                new_population.append(child)
            
            population = new_population
            population_fitness = [fitness_function(h) for h in population]
            
            # Update best solution
            if np.min(population_fitness) < best_c5:
                best_c5 = np.min(population_fitness)
                best_h = population[np.argmin(population_fitness)]
        
        best_h = population[np.argmin(population_fitness)]
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
