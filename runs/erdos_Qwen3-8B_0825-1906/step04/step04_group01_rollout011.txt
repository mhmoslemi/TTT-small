import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return the initial approximation of the solution
    # This is a placeholder - in practice, it would be loaded from a file or generated
    # For demonstration, we return a simple symmetric function
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h) * 2.0 / n_points
    h = h * (1.0 / sum_h)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    
    # Let's try a larger n_points to get a better approximation
    n_points = 100
    
    # Generate initial solution from the initial_h_values
    initial_h = initial_h_values()
    # Interpolate to n_points samples
    initial_h = np.array(initial_h)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = initial_h.tolist()
    
    # Evaluate initial solution
    best_c5 = evaluate_erdos_solution(initial_h)
    best_h = initial_h
    
    # We'll try a gradient-based optimization approach with constraints
    def objective(x):
        # Ensure x is in [0, 1]
        x = np.clip(x, 0.0, 1.0)
        # Enforce integral constraint
        sum_h = np.sum(x) * 2.0 / n_points
        x = x * (1.0 / sum_h) if sum_h != 0 else x
        # Compute the overlap integral
        return evaluate_erdos_solution(x)
    
    # Bounds for the optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings with constraints
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}]
    
    # Use SLSQP method for constrained optimization
    result = optimize.minimize(
        objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Now perform some additional local optimization
    # We'll use a simple gradient descent with momentum
    learning_rate = 0.01
    momentum = 0.9
    num_iterations = 50
    
    for _ in range(num_iterations):
        # Compute objective and gradient
        x = best_h.copy()
        # Add small perturbation for numerical gradient
        perturb = np.random.normal(0, 1e-3, n_points)
        x_perturb = np.clip(x + perturb, 0.0, 1.0)
        objective_perturb = evaluate_erdos_solution(x_perturb)
        
        # Compute gradient
        grad = (objective_perturb - best_c5) / np.linalg.norm(perturb)
        
        # Update with momentum
        best_h = np.clip(best_h - learning_rate * grad, 0.0, 1.0)
        best_h = best_h * (1.0 / (np.sum(best_h) * 2.0 / n_points)) if np.sum(best_h) != 0 else best_h
        
        # Enforce integral constraint
        sum_h = np.sum(best_h) * 2.0 / n_points
        best_h = best_h * (1.0 / sum_h) if sum_h != 0 else best_h
        
        # Evaluate new solution
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
