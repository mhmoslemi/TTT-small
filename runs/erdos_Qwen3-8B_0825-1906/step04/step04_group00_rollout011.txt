import numpy as np
import scipy.optimize as optimize
import time
import random
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a refined and randomized initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)  # Keep it consistent for the rest of the run
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Define a more refined initial solution with random perturbations
    def refine_initial_solution(h0):
        h = h0.copy()
        for i in range(n_points):
            h[i] += np.random.uniform(-0.02, 0.02)
            h[i] = np.clip(h[i], 0.0, 1.0)
        h = h / np.sum(h) * n_points / 2
        return h
    
    refined_initial_h = refine_initial_solution(initial_h)
    best_h = refined_initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define a local optimization function
    def local_optimize(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        return result.x
    
    # Use gradient-based optimization on some initial points
    for _ in range(5):
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        h = local_optimize(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Run a more refined genetic algorithm
    def genetic_algorithm(n_gen=30, pop_size=30, mutation_rate=0.05, crossover_rate=0.8):
        population = np.random.uniform(0, 1, (pop_size, n_points))
        population = population / np.sum(population, axis=1, keepdims=True) * n_points / 2
        
        for gen in range(n_gen):
            fitness = np.array([evaluate_erdos_solution(ind) for ind in population])
            best_idx = np.argmin(fitness)
            best_fitness = fitness[best_idx]
            
            if best_fitness < best_c5:
                best_c5 = best_fitness
                best_h = population[best_idx].copy()
            
            # Select parents
            probabilities = fitness / np.sum(fitness)
            parents_idx = np.random.choice(range(pop_size), size=pop_size, p=probabilities)
            
            # Crossover
            offspring = []
            for i in range(0, pop_size, 2):
                if i + 1 < pop_size:
                    parent1 = population[parents_idx[i]]
                    parent2 = population[parents_idx[i + 1]]
                    child1 = parent1.copy()
                    child2 = parent2.copy()
                    for j in range(n_points):
                        if np.random.rand() < crossover_rate:
                            child1[j], child2[j] = parent2[j], parent1[j]
                    offspring.append(child1)
                    offspring.append(child2)
            
            # Mutation
            for i in range(pop_size):
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(n_points)
                    population[i][idx] = np.random.uniform(0, 1)
                    population[i] = population[i] / np.sum(population[i]) * n_points / 2
                
                if i < len(offspring):
                    offspring[i] = offspring[i] / np.sum(offspring[i]) * n_points / 2
            
            population = np.array(offspring)
        
        return best_h, best_c5
    
    best_h, best_c5 = genetic_algorithm(n_gen=30, pop_size=30, mutation_rate=0.05, crossover_rate=0.8)
    
    # Local refinement using SLSQP
    best_h = local_optimize(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Add perturbations and refine
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        refined_h = local_optimize(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)
        
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
