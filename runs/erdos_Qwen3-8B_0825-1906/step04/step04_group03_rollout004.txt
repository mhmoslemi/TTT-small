```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    n_points = 47
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h) * 2.0 / n_points
    h = h * (1.0 / sum_h)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    n_points = 80  # Keep 80 samples for balance between accuracy and speed
    
    # Generate initial guess with a more optimal construction
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        # Create a more optimized pattern: stepwise function with 20% overlap
        if 0.35 <= x <= 0.65:
            h_initial[i] = 1.0
        elif 0.25 <= x <= 0.45 or 0.75 <= x <= 0.95:
            h_initial[i] = 0.5
    h_initial = h_initial.tolist()
    
    best_c5 = evaluate_erdos_solution(h_initial)
    best_h = h_initial
    
    # Try random initializations to escape local optima
    for _ in range(20):
        h = generate_random_h(n_points)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Define the objective function for optimization
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        # Ensure the integral constraint is met
        sum_h = np.sum(h) * 2.0 / n_points
        if np.abs(sum_h - 1.0) > 1e-6:
            h = h * (1.0 / sum_h)
        return evaluate_erdos_solution(h)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a more robust optimization method with tighter constraints
    result = optimize.minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Early stop if time is running out
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Second optimization with perturbed initial guess
    perturbed_h = best_h + np.random.uniform(-0.05, 0.05, n_points)
    perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
    sum_perturbed = np.sum(perturbed_h) * 2.0 / n_points
    perturbed_h = perturbed_h * (1.0 / sum_perturbed)
    
    result = optimize.minimize(
        objective,
        x0=perturbed_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
```
