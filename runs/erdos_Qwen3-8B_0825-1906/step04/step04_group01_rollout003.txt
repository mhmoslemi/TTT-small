import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return the initial approximation of the solution
    # This is a placeholder - in practice, it would be loaded from a file or generated
    # For demonstration, we return a simple symmetric function
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h) * 2.0 / n_points
    h = h * (1.0 / sum_h)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    n_points = 60
    best_c5 = 100.0
    best_h = initial_h_values()
    
    # Evaluate initial solution
    initial_c5 = evaluate_erdos_solution(best_h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
    
    # Particle Swarm Optimization (PSO) for global search
    n_particles = 20
    max_iterations = 100
    inertia = 0.5
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = [generate_random_h(n_points) for _ in range(n_particles)]
    velocities = np.zeros((n_particles, n_points))
    personal_best_positions = particles.copy()
    personal_best_values = [evaluate_erdos_solution(p) for p in particles]
    global_best_position = best_h
    global_best_value = best_c5
    
    for iteration in range(max_iterations):
        for i in range(n_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (np.array(personal_best_positions[i]) - np.array(particles[i])) + social * r2 * (np.array(global_best_position) - np.array(particles[i]))
            
            # Update position
            particles[i] = np.array(particles[i]) + velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            
            # Normalize to satisfy the integral constraint
            sum_h_current = np.sum(particles[i]) * 2.0 / n_points
            if np.abs(sum_h_current - 1.0) > 1e-6:
                scaling_factor = 1.0 / sum_h_current
                particles[i] = particles[i] * scaling_factor
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if fitness < personal_best_values[i]:
                personal_best_positions[i] = particles[i].tolist()
                personal_best_values[i] = fitness
            
            # Update global best
            if fitness < global_best_value:
                global_best_position = particles[i].tolist()
                global_best_value = fitness
        
        # Update best solution
        if global_best_value < best_c5:
            best_c5 = global_best_value
            best_h = global_best_position
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Local optimization of the best solution found
    def objective(x):
        h_values = x.copy()
        # Ensure h_values are within [0, 1]
        h_values = np.clip(h_values, 0.0, 1.0)
        # Ensure the integral constraint is met
        sum_h_current = np.sum(h_values) * 2.0 / n_points
        if np.abs(sum_h_current - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h_current
            h_values = h_values * scaling_factor
        # Compute the overlap integral
        return evaluate_erdos_solution(h_values)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = optimize.minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
