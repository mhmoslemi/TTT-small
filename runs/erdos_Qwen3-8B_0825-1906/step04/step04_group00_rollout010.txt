import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a refined and randomized initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points_current = len(best_h)
    
    # Try smaller n_points values to speed things up and avoid local minima
    for n_points in [100, 150, 200, 250, 300]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial guess with uniform distribution
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        
        # Define objective function to minimize
        def objective(x):
            h_values = x.copy()
            return evaluate_erdos_solution(h_values)
        
        # Define constraint for total integral
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        # Use SLSQP for local optimization
        result = optimize.minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        
        # Update best solution
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x.copy()
            n_points_current = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points_current)
    
    return (best_h.tolist(), final_c5, n_points_current)
