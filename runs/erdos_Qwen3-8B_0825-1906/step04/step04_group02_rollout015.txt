```python
import numpy as np
import scipy.optimize as optimize
import time
from functools import lru_cache
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Return a higher-resolution initial approximation of the solution
    # This is a more refined and balanced initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    h = np.array(initial_h)
    
    # Function to evaluate the objective function
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Function to enforce the constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for variables
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # First optimization run with SLSQP
    result = optimize.minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 500}
    )
    
    # If time runs out, return the best solution found so far
    if time.time() - start_time > budget_s - 1:
        h_values = result.x
        final_c5 = evaluate_erdos_solution(h_values)
        return (h_values.tolist(), final_c5, n_points)
    
    # Evaluate the initial solution
    h_values = result.x
    final_c5 = evaluate_erdos_solution(h_values)
    
    # Try a few more iterations with slightly perturbed initializations
    for _ in range(5):
        # Perturb the current solution slightly
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = h_values + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        result_perturbed = optimize.minimize(
            fun=objective,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        
        if evaluate_erdos_solution(result_perturbed.x) < final_c5:
            final_c5 = evaluate_erdos_solution(result_perturbed.x)
            h_values = result_perturbed.x
    
    # Try adding more points to get better precision
    n_points_new = n_points * 2
    h_new = np.zeros(n_points_new)
    for i in range(n_points_new):
        x = i * 2.0 / n_points_new
        if 0.25 <= x <= 0.75:
            h_new[i] = 1.0
    h_new = h_new / np.sum(h_new) * n_points_new / 2
    
    result_new = optimize.minimize(
        fun=objective,
        x0=h_new,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 500}
    )
    
    h_values_new = result_new.x
    final_c5_new = evaluate_erdos_solution(h_values_new)
    
    if final_c5_new < final_c5:
        final_c5 = final_c5_new
        h_values = h_values_new
    
    # One more refinement with the best solution so far
    result_refined = optimize.minimize(
        fun=objective,
        x0=h_values,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 500}
    )
    
    h_values_refined = result_refined.x
    final_c5_refined = evaluate_erdos_solution(h_values_refined)
    
    if final_c5_refined < final_c5:
        final_c5 = final_c5_refined
        h_values = h_values_refined
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_values)
    return (h_values.tolist(), final_c5, n_points_new)
```
