import numpy as np
import scipy.optimize as optimize
import time
import random
import warnings

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    n_points = 80
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h) * 2.0 / n_points
    h = h * (1.0 / sum_h)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial construction
    n_points = 80
    best_c5 = 100.0
    best_h = initial_h_values()
    initial_c5 = evaluate_erdos_solution(best_h)
    best_c5 = initial_c5
    best_h = np.array(best_h)
    
    # Try random initializations
    for _ in range(20):
        h = generate_random_h(n_points)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Define the objective function
    def objective(x):
        h_values = x.copy()
        # Ensure values are within [0, 1]
        h_values = np.clip(h_values, 0.0, 1.0)
        # Normalize to meet the integral constraint
        sum_h = np.sum(h_values) * 2.0 / n_points
        if np.abs(sum_h - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h
            h_values = h_values * scaling_factor
        # Compute the overlap integral
        return evaluate_erdos_solution(h_values)
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}
    ]
    
    # Optimization using SLSQP
    result = optimize.minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Early stopping if time is up
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Try a second optimization with perturbed initial guess
    perturbed_h = best_h + np.random.uniform(-0.05, 0.05, n_points)
    perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
    sum_perturbed = np.sum(perturbed_h) * 2.0 / n_points
    perturbed_h = perturbed_h * (1.0 / sum_perturbed)
    
    result_perturbed = optimize.minimize(
        objective,
        x0=perturbed_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result_perturbed.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try a third optimization with different parameters
    result_third = optimize.minimize(
        objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result_third.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure we don't exceed the budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Final check and return the best solution
    return (best_h.tolist(), best_c5, n_points)
