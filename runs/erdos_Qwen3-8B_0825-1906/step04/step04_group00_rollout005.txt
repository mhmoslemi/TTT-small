import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Use a combination of local optimization and global search
    
    # Local refinement with SLSQP
    def local_refinement(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        return result.x
    
    # Run local refinement on initial solution
    best_h = local_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Use a more sophisticated global search with simulated annealing
    def simulated_annealing(n_iter=500, temp=100.0, cooling_rate=0.995):
        # Initialize with the best solution
        current_h = best_h.copy()
        current_c5 = best_c5
        
        for i in range(n_iter):
            # Generate a perturbed solution
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed_h = current_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
            perturbed_c5 = evaluate_erdos_solution(perturbed_h)
            
            # Calculate the difference
            delta = perturbed_c5 - current_c5
            
            # Accept the new solution with probability if it improves or with some probability otherwise
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                current_h = perturbed_h.copy()
                current_c5 = perturbed_c5
                
                # Update best solution
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            
            # Cool down
            temp *= cooling_rate
            
            # Check time budget
            if time.time() - start_time > budget_s - 1:
                return best_h, best_c5
    
        return best_h, best_c5
    
    # Run simulated annealing
    try:
        best_h, best_c5 = simulated_annealing(n_iter=500, temp=100.0, cooling_rate=0.995)
    except:
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Perturb and refine
    for _ in range(3):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Run local refinement on the perturbed solution
        refined_h = local_refinement(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)
        
        # Update best solution
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
