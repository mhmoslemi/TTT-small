import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return the initial approximation of the solution
    # This is a placeholder - in practice, it would be loaded from a file or generated
    # For demonstration, we return a simple symmetric function
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h) * 2.0 / n_points
    h = h * (1.0 / sum_h)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    
    # Use a slightly larger number of points for better resolution
    n_points = 80
    
    # Initialize with the initial construction
    h = initial_h_values()
    best_c5 = evaluate_erdos_solution(h)
    best_h = h
    
    # Define a function to compute the maximum overlap
    def compute_c5(h_values):
        return evaluate_erdos_solution(h_values)
    
    # Define a function to compute the integral constraint
    def constraint_integral(h_values):
        return np.sum(h_values) * 2.0 / n_points - 1.0
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Set up the optimization problem
    def objective(x):
        h_values = x.copy()
        # Ensure values are in [0, 1]
        h_values = np.clip(h_values, 0.0, 1.0)
        # Ensure the integral constraint is satisfied
        sum_h = np.sum(h_values) * 2.0 / n_points
        if np.abs(sum_h - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h
            h_values = h_values * scaling_factor
        return compute_c5(h_values)
    
    # Use SLSQP for optimization
    result = optimize.minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    # Update best_h and best_c5
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Use a simple greedy search to improve the solution
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        # Randomly perturb one element
        idx = np.random.randint(n_points)
        perturbation = np.random.uniform(-0.1, 0.1)
        new_h = best_h.copy()
        new_h[idx] += perturbation
        # Ensure within [0, 1]
        new_h = np.clip(new_h, 0.0, 1.0)
        # Normalize the integral constraint
        sum_h_new = np.sum(new_h) * 2.0 / n_points
        if np.abs(sum_h_new - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h_new
            new_h = new_h * scaling_factor
        # Check if this is better
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
    
    # Refine with a local optimization
    result = optimize.minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
