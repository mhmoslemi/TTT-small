import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a refined and randomized initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)  # Keep it consistent for the rest of the run
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Simulated Annealing for Global Search
    def simulated_annealing(temp_start=1.0, temp_end=0.1, cooling_rate=0.995, iterations=1000):
        current_h = np.random.uniform(0, 1, n_points)
        current_h = current_h / np.sum(current_h) * n_points / 2
        current_c5 = evaluate_erdos_solution(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        temperature = temp_start
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                return best_h, best_c5
            
            # Generate a neighbor
            neighbor_h = current_h.copy()
            idx = np.random.randint(n_points)
            neighbor_h[idx] = np.random.uniform(0, 1)
            neighbor_h = neighbor_h / np.sum(neighbor_h) * n_points / 2
            
            neighbor_c5 = evaluate_erdos_solution(neighbor_h)
            
            # Acceptance probability
            delta = neighbor_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
                current_h = neighbor_h
                current_c5 = neighbor_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
            # Cool down
            temperature *= cooling_rate
        
        return best_h, best_c5
    
    # Run simulated annealing
    try:
        best_h, best_c5 = simulated_annealing(temp_start=1.0, temp_end=0.1, cooling_rate=0.995, iterations=2000)
    except:
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        return result.x
    
    # Refine the best solution
    best_h = local_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perturb and refine
    for _ in range(3):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Run local refinement on the perturbed solution
        refined_h = local_refinement(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)
        
        # Update best solution
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
