def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_shifted = np.concatenate([h[-1:], h[:-1]])
    corr = signal.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values[:n_points], dtype=np.float64)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def objective(h):
        return evaluate_erdos_solution(h)
    
    def perturb(h):
        h = h.copy()
        idxs = np.random.choice(n_points, size=32, replace=False)
        for i in idxs:
            h[i] = np.clip(h[i] + np.random.uniform(-0.2, 0.2), 0, 1)
        return h
    
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h = perturb(best_h)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            print(f"Found better solution: C5 = {best_c5}")
    
    # Try a convex optimization approach
    h_var = cp.Variable(n_points, nonneg=True)
    constraints = [cp.sum(h_var) == n_points / 2]
    objective = cp.Maximize(cp.Max(signal.correlate(h_var.value, 1 - h_var.value, mode='full') * dx))
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.SCS, eps=1e-4)
        best_h = h_var.value
        best_c5 = evaluate_erdos_solution(best_h)
    except:
        pass
    
    return best_h, best_c5, n_points
