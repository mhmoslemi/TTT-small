import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Initial parameters
    n_points = 42  # Starting with the initial sample count
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)

    # Try increasing sample count if time allows
    start_time = time.time()
    for n_points in range(42, 100):
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 2 - 1  # Random initialization
        h = np.clip(h, 0, 1)
        h *= n_points / 2  # Adjust to meet integral constraint
        h = np.round(h).astype(float)  # Discretize

        # Check if we're under time limit
        if time.time() - start_time > budget_s - 1:
            break

        # Evaluate current solution
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            print(f"Found better solution with n_points={n_points}, C₅={best_c5:.6f}")

    # Final check with the best found so far
    # Use a more sophisticated optimization if time allows
    if time.time() - start_time < budget_s - 5:
        # Try to further refine the solution with a small convex optimization
        n_points = 100
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 2 - 1  # Random initialization
        h = np.clip(h, 0, 1)
        h *= n_points / 2  # Adjust to meet integral constraint
        h = np.round(h).astype(float)  # Discretize

        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            print(f"Found better solution with n_points={n_points}, C₅={best_c5:.6f}")

    return h, best_c5, n_points
