def run(seed=42, budget_s=60, **kwargs):

    # Define the number of points and initial guess

    # Function to evaluate the C5 value
    def evaluate_c5(h):
        h = np.array(h)
        # Ensure h has length n_points
        if len(h) != n_points:
            raise ValueError("h must have exactly {} points".format(n_points))
        # Compute the overlap integral using convolution
        # Using numpy.correlate with mode="full" gives all possible overlaps
        correlation = np.correlate(h, 1 - h, mode="full")
        # The overlap integral is the maximum of the correlation values multiplied by dx
        dx = 2.0 / n_points
        c5 = np.max(correlation) * dx
        return c5

    # Function to optimize h_values with a penalty for constraint violation
    def optimize_h_values(h_values, max_iterations=100, tolerance=1e-5):
        for _ in range(max_iterations):
            # Convert to numpy array
            h = np.array(h_values)
            # Compute the constraint violation
            constraint_violation = np.abs(np.sum(h) - n_points / 2)
            if constraint_violation < tolerance:
                break

            # Define the objective function to minimize (C5 with penalty)
            def objective_func(x):
                h = np.array(x)
                # Evaluate the C5 value
                c5 = evaluate_c5(h)
                # Penalize constraint violation
                penalty = 1e4 * constraint_violation
                return c5 + penalty

            # Use L-BFGS-B optimization
            result = optimize.minimize(
                objective_func,
                h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)]
            )
            h_values = result.x

            # Early stopping if the constraint is satisfied
            if np.abs(np.sum(h_values) - n_points / 2) < tolerance:
                break

        return h_values

    # Start with the initial guess
    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h)

    for n_points in [70, 100, 150, 200]:

            break

        if len(best_h) != n_points:

        optimized_h = optimize_h_values(best_h.copy())

        if current_c5 < best_c5:

            best_c5 = current_c5
