import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.roll(h, shift=1, axis=0)
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values(n_points):
    # Simple initial construction with uniform distribution
    return np.repeat([0.5], n_points)

# Strategy: Use a hybrid approach combining gradient descent with convex optimization
# We will start from the initial solution and refine it using a combination of local search
# and convex optimization to minimize the maximum overlap integral

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 256  # Start with 256 points for a balance between accuracy and speed

    # Function to evaluate the solution
    def evaluate(h):
        nonlocal best_c5
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Initial guess
    h_initial = initial_h_values(n_points)

    # Define the optimization problem
    def optimize_h(h_guess):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        # Objective: minimize the maximum overlap integral
        # We use a convex relaxation by minimizing the maximum over all shifted versions
        # We approximate this by maximizing the integral for each shift and taking the maximum
        shifted_h = cp.diag(np.roll(np.eye(n_points), shift=1, axis=0))
        overlap_int = cp.multiply(h, 1 - cp.matmul(shifted_h, h))
        overlap_int = cp.sum(overlap_int, axis=1)
        c5 = cp.Max(overlap_int)
        objective = cp.Minimize(c5)
        problem = cp.Problem(objective, constraints)
        result = problem.solve(solver=cp.SCS, eps=1e-4, verbose=False)
        if result is not None:
            return h.value
        return h_guess

    # Main optimization loop
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        h = optimize_h(h_initial)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h_initial = h
        c5 = evaluate(h_initial)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_initial.copy()

    # Refine using gradient descent
    def objective_func(h):
        return evaluate_erdos_solution(h)

    result = opt.minimize(objective_func, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'ftol': 1e-6, 'maxiter': 100})
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)

    # Final check to ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)

    return best_h, best_c5, n_points
