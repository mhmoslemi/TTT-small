```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import matplotlib.pyplot as plt

# Pre-imported functions
initial_h_values = np.random.rand(42) * 2 - 1  # Placeholder for initial values
initial_h_values = np.clip(initial_h_values, 0, 1)
initial_h_values = initial_h_values * (42 / 2) / np.sum(initial_h_values)
initial_h_values = np.round(initial_h_values, 5)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)

    def objective(h):
        nonlocal best_c5
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    def simulate(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    # Try smaller n_points
    for n_points_test in [42, 60, 84, 100, 120, 140, 160, 180, 200]:
        if time.time() - start_time > budget_s - 1:
            break
        print(f"Trying n_points = {n_points_test}")
        h_initial = np.random.rand(n_points_test) * 2 - 1
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points_test / 2) / np.sum(h_initial)
        h_initial = np.round(h_initial, 5)

        h_initial = h_initial.reshape(-1)
        res = minimize(simulate, h_initial, method='L-BFGS-B', bounds=[(0, 1)] * n_points_test)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
                n_points = n_points_test

        # Try a few local improvements
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
            h_test = best_h.copy()
            idx = np.random.randint(0, n_points_test)
            h_test[idx] = np.random.rand() * 2 - 1
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points_test / 2) / np.sum(h_test)
            c5_test = simulate(h_test)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()

    # Final optimization
    if time.time() - start_time < budget_s - 1:
        print("Performing final optimization")
        h_initial = best_h.copy()
        res = minimize(simulate, h_initial, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()

    return best_h, best_c5, n_points
```
