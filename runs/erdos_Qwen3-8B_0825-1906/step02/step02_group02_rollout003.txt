```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.concatenate([np.zeros(1), h[:-1]])
    overlap = np.correlate(h, 1 - h_shifted, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    return [0.5 for _ in range(42)]  # Initial guess with uniform distribution

def run(seed=42, budget_s=60, **kwargs):
    random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values()
    n_points = len(best_h)
    
    # Try varying n_points from 40 to 100, step by 10
    for n_points in range(40, 110, 10):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Adjust n_points to ensure it fits within the time budget
        # Define dx
        dx = 2.0 / n_points
        
        # Define the optimization variables
        h = cp.Variable(n_points, name='h')
        
        # Objective function: minimize the maximum overlap
        overlaps = []
        for k in range(1, n_points):
            h_shifted = cp.vstack([h[-k:], h[:-k]])
            overlap = cp.multiply(h, 1 - h_shifted)
            overlaps.append(overlap)
        
        # Calculate the maximum overlap
        max_overlap = cp.Max(overlaps)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2  # Ensure the integral is 1
        ]
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iter=1000)
            if prob.status == 'optimal':
                h_opt = np.round(h.value, 4)
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, len(best_h)
```
