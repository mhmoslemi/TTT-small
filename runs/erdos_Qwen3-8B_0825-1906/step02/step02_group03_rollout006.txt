import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = 100  # start with 100 points for faster computation
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        dx = 2.0 / n_points
        h_expanded = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        h_shifted = np.zeros((n_points + 1, n_points + 1))
        for i in range(n_points + 1):
            h_shifted[i] = h_expanded[i:i + n_points]
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)  # Returns the maximum overlap value

    # Function to compute the objective for optimization
    def objective(h_values):
        return evaluate_overlap(h_values)

    # Function to compute the constraint: total integral must be 1
    def constraint_total_integral(h_values):
        dx = 2.0 / n_points
        return np.sum(h_values) * dx - 1.0

    # Initial guess
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]  # Ensure it's of the correct length

    # Setup optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_total_integral})
    result = opt.minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons)
    best_h = result.x
    best_c5 = result.fun

    # Check if we can try with more points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        n_points = 200
        dx = 2.0 / n_points
        # Re-initialize with the best found so far, rescaled to the new number of points
        best_h_rescaled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        # Ensure the integral is exactly 1
        scale_factor = (n_points / 2) / np.sum(best_h_rescaled)
        best_h_rescaled *= scale_factor

        def evaluate_overlap_high_res(h_vals):
            h_expanded = np.pad(h_vals, (0, n_points), mode='constant', constant_values=0)
            h_shifted = np.zeros((n_points + 1, n_points + 1))
            for i in range(n_points + 1):
                h_shifted[i] = h_expanded[i:i + n_points]
            overlap = np.correlate(h_vals, 1 - h_vals, mode="full") * dx
            return np.max(overlap)

        def objective_high_res(h_values):
            return evaluate_overlap_high_res(h_values)

        def constraint_total_integral_high_res(h_values):
            dx = 2.0 / n_points
            return np.sum(h_values) * dx - 1.0

        bounds_high_res = [(0, 1) for _ in range(n_points)]
        cons_high_res = ({'type': 'eq', 'fun': constraint_total_integral_high_res})
        result_high_res = opt.minimize(objective_high_res, best_h_rescaled, method='SLSQP', bounds=bounds_high_res, constraints=cons_high_res)
        best_h = result_high_res.x
        best_c5 = result_high_res.fun

    # Final evaluation
    final_c5 = best_c5
    return (best_h, final_c5, n_points)
