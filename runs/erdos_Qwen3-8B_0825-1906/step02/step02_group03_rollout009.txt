```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can be increased if time allows
    dx = 2.0 / n_points
    max_iter = 100  # Maximum iterations
    eval_interval = 5  # Evaluate every 5 iterations
    
    # Initial guess
    h = np.zeros(n_points)
    h[:n_points // 2] = 1.0  # Start with a simple step function
    
    # Ensure the integral condition is satisfied
    total_integral = np.sum(h) * dx
    scaling_factor = 1.0 / total_integral
    h *= scaling_factor
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Optimization step
        x = cp.Variable(n_points)
        constraints = [
            0 <= x,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        objective = cp.Maximize(cp.sum(x * (1 - cp.sum(x[cp.reshape(cp.range(n_points), (n_points, 1)) - cp.reshape(cp.arange(n_points), (1, n_points)) + 1, axis=0) <= n_points - 1])) * dx)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4)
            if prob.status == 'OPTIMAL':
                h = x.value
        except:
            # If optimization fails, use a simple perturbation
            h += 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Random perturbation to escape local minima
        if np.random.rand() < 0.1:
            h += 0.05 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h)
    return best_h, best_c5, n_points
```
