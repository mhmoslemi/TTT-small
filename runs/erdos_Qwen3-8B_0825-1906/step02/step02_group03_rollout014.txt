def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Ensure h_values are in [0, 1]
    h_values = np.clip(h_values, 0, 1)
    # Compute the convolution
    corr = signal.correlate(h_values, 1 - h_values, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # Initial construction with 70 points
    n_points = 70

            h[i] = 1.0
        else:
            h[i] = 0.5
    # Normalize to satisfy the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    return h
