import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction (42 samples)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Number of samples
    dx = 2.0 / n_points
    max_iterations = 100  # Maximum number of iterations for optimization
    tolerance = 1e-4  # Stopping tolerance
    
    # Initialize h with a starting point
    h = np.random.rand(n_points)
    h = h * (n_points / 2)  # Scale to meet the integral constraint
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.concatenate([h, np.zeros(n_points)])
        result = correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5
    
    # Function to check integral constraint
    def check_integral(h):
        return np.abs(np.sum(h) * dx - 1)
    
    # Function to optimize h using gradient descent
    def optimize_h(h_initial):
        h = h_initial.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        t0 = time.time()
        for _ in range(max_iterations):
            # Compute overlap integral
            c5 = compute_c5(h)
            
            # Compute gradient using finite differences
            gradient = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-4
                c5_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-4
                c5_minus = compute_c5(h_minus)
                gradient[i] = (c5_plus - c5_minus) / (2 * 1e-4)
            
            # Project gradient to maintain integral constraint
            # Compute the gradient of the constraint function
            constraint_grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-4
                constraint_plus = check_integral(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-4
                constraint_minus = check_integral(h_minus)
                constraint_grad[i] = (constraint_plus - constraint_minus) / (2 * 1e-4)
            
            # Compute the gradient of the objective function (negative of gradient)
            # and project it to the tangent space of the constraint manifold
            gradient_projected = gradient - (np.dot(gradient, constraint_grad) / np.dot(constraint_grad, constraint_grad)) * constraint_grad
            
            # Update h
            h -= 0.1 * gradient_projected
            h = np.clip(h, 0, 1)
            
            # Scale h to maintain integral constraint
            h = h / np.sum(h) * (n_points / 2)
            
            # Update best solution
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            
            # Stop if time is up or convergence is achieved
            if time.time() - t0 > budget_s - 1:
                break
            if np.abs(current_c5 - best_c5) < tolerance:
                break
        
        return best_h, best_c5
    
    # Run the optimization
    h_optimized, c5_bound = optimize_h(h)
    
    # Return the best solution
    return h_optimized, c5_bound, n_points
