evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h_values.npy')  # Assume this is a numpy array of shape (n_points,)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    n_points = len(initial_h_values)
    h = np.copy(initial_h_values)
    
    # Optimize h to minimize max overlap
    for _ in range(100):  # Limit number of iterations to stay within time budget
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate current overlap
        c5 = evaluate_erdos_solution(h)
        if c5 <= 0.380876:
            break  # Achieved target if possible
        
        # Create a function to evaluate overlap for a given h
        def objective_func(x):
            h_new = np.copy(x)
            return evaluate_erdos_solution(h_new)
        
        # Use gradient descent with simulated annealing
        def gradient_descent_step(h):
            # Use a small step size and random perturbation
            h_perturbed = h + 0.01 * np.random.randn(len(h))
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Ensure the integral constraint is satisfied
            h_scaled = h_perturbed * (len(h) / 2.0) / np.sum(h_perturbed)
            return h_scaled
        
        h = gradient_descent_step(h)
    
    # One final evaluation
    c5 = evaluate_erdos_solution(h)
    return h, c5, len(h)
