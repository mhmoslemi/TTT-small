def run(seed=42, budget_s=60, **kwargs):

    start_time = time.time()
    
    # Initial construction
    n_points = 70
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h_values):
        n = len(h_values)
        dx = 2.0 / n
        h = np.array(h_values)
        shifted = np.zeros(2 * n - 1)
        for i in range(n):
            shifted[i:i+n] = h
        h_shifted = (1 - h) * np.roll(h, 1)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)

    def optimize_h_values(h_values, budget_s):
        n = len(h_values)
        dx = 2.0 / n
        best_h = h_values.copy()
        best_c5 = evaluate_overlap(best_h)
        best_iter = 0
        
        # Use a local search with random perturbations
        for iter in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
            # Randomly perturb the function
            perturbation = np.random.rand(n) - 0.5
            perturbation *= 0.15
            h_new = np.clip(h_values + perturbation, 0, 1)
            
            # Ensure the integral constraint is satisfied
            sum_h = np.sum(h_new) * dx
            if abs(sum_h - 1) > 1e-5:
                # Rescale if constraint is violated
                scale = 1 / (sum_h / 1)
                h_new *= scale
            
            # Evaluate the overlap
            c5 = evaluate_overlap(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
                best_iter = iter
        
        return best_h, best_c5
    
    # Initial optimization
    best_h, best_c5 = optimize_h_values(h, budget_s)

    for _ in range(3):

        sum_h = np.sum(h) * dx
        scale = 1 / (sum_h / 1)
        h *= scale
        h, c5 = optimize_h_values(h, budget_s)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    return best_h, best_c5, n_points
