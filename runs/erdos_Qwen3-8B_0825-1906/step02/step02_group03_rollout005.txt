import numpy as np

import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial solution from initial_h_values (n=70)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Adjust initial solution to satisfy the integral constraint
    if abs(sum_h - n_points / 2) > 1e-6:
        # Normalize to meet the constraint
        initial_h = initial_h * (n_points / 2) / sum_h
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Reshape to ensure it's 1D
        h = h.reshape(-1)
        # Compute the autocorrelation-like convolution
        corr = np.convolve(h, 1 - h, mode="full")
        # Only consider the relevant part of the convolution
        # The max overlap occurs between k=0 and k=2 (since the interval is [0,2])
        max_corr = max(corr[1:n_points + 1])
        # Multiply by dx to get the integral
        return max_corr * dx
    
    # Optimization: use gradient descent with adaptive step size
    def optimize_h(h_initial, max_iter=100, tol=1e-4):
        h = np.array(h_initial)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        start_time = time.time()
        
        for i in range(max_iter):
            # Compute the overlap integral
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Compute the gradient of c5 w.r.t h
            # This is approximate using finite differences
            grad = np.zeros_like(h)
            for idx in range(len(h)):
                # Small perturbation
                h_perturbed = h.copy()
                h_perturbed[idx] += 1e-5
                grad[idx] = (compute_c5(h_perturbed) - c5) / 1e-5
            
            # Update step
            step_size = 0.01
            h -= step_size * grad
            
            # Ensure h remains in [0,1]
            h = np.clip(h, 0, 1)
            
            # Check time budget
            elapsed = time.time() - start_time
            if elapsed > budget_s - 1:
                break
        
        return best_h, best_c5
    
    # Optimize from the initial solution
    optimized_h, c5_bound = optimize_h(initial_h)
    
    # Finalize the result
    n_points = len(optimized_h)
    return optimized_h, c5_bound, n_points
