def run(seed=42, budget_s=60, **kwargs):

    start_time = time.time()
    
    # Initial attempt with the pre-imported initial solution
    n_points = len(initial_h_values)

    # Function to compute the C5 value
    def compute_c5(h):
        dx = 2.0 / len(h)
        h = np.array(h)
        h_shifted = np.zeros(len(h) + len(h) - 1)
        h_shifted[:len(h)] = h
        h_shifted[len(h)-1:] = h
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Function to compute the C5 value and the loss for optimization
    def evaluate(h_values):
        dx = 2.0 / len(h_values)
        h = np.array(h_values)
        h_shifted = np.zeros(len(h) + len(h) - 1)
        h_shifted[:len(h)] = h
        h_shifted[len(h)-1:] = h
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Optimization function to minimize C5
    def objective_function(h_values):
        return evaluate(h_values)
    
    # Optimization using simulated annealing to find a better solution
    def optimize_h():
        nonlocal h_values
        max_iterations = 1000
        temp = 100.0
        cooling_rate = 0.995
        best_h = np.array(h_values)
        best_c5 = evaluate(best_h)
        
        for _ in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            # Perturb the current solution
            perturbation = np.random.normal(0, 0.05, size=len(h_values))
            h_candidate = best_h + perturbation
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Ensure the integral constraint is satisfied
            total_integral = np.sum(h_candidate) * (2.0 / len(h_candidate))
            if abs(total_integral - 1.0) > 1e-5:
                continue
            
            c5_candidate = evaluate(h_candidate)
            # Accept the candidate with a probability based on the temperature
            if c5_candidate < best_c5 or np.random.rand() < np.exp((best_c5 - c5_candidate) / temp):
                best_h = h_candidate
                best_c5 = c5_candidate
            temp *= cooling_rate
        
        h_values = best_h
        return best_c5
    
    # Local optimization using gradient-based method for fine-tuning
    def local_optimize():
        nonlocal h_values
        n = len(h_values)
        h = cp.Variable(n)
        dx = 2.0 / n
        
        # Objective function
        def objective(h):
            h_shifted = np.zeros(n + n - 1)
            h_shifted[:n] = h
            h_shifted[n-1:] = h
            correlation = np.correlate(h, 1 - h, mode="full")
            return cp.Max(correlation) * dx
        
        # Constraint: total integral equals 1
        constraint = cp.sum(h) * dx == 1
        
        # Solve the optimization problem
        prob = cp.Problem(cp.Minimize(objective(h)), [constraint])
        try:
            prob.solve(solver=cp.SCS)
            if prob.status == 'optimal':
                h_values = h.value
        except:
            pass
    
    # Run optimization
    best_c5 = optimize_h()
    local_optimize()
    
    # Final evaluation
    final_c5 = evaluate(h_values)
    
    # Ensure the result is within the time budget
    if time.time() - start_time > budget_s:
        return h_values, final_c5, len(h_values)
    
    return h_values, final_c5, len(h_values)
