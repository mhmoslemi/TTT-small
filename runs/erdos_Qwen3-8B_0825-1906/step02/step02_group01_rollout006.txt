import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt
from scipy.optimize import Bounds, minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return the initial approximation of the solution
    # This is a placeholder - in practice, it would be loaded from a file or generated
    # For demonstration, we return a simple symmetric function
    n_points = 47
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Start with a higher resolution to get a better initial guess
    n_points = 100
    initial_h = initial_h_values()
    initial_h = np.array(initial_h)
    
    # Adjust initial_h to fit the interval [0,2]
    # Extend to n_points
    initial_h_extended = np.zeros(n_points)
    for i in range(len(initial_h)):
        initial_h_extended[i * n_points // len(initial_h)] = initial_h[i]
    
    # Scale to satisfy the integral constraint
    sum_initial = np.sum(initial_h_extended) * (2.0 / n_points)
    initial_h_extended *= (1.0 / sum_initial)
    
    # Define the objective function
    def objective(x):
        h_values = x.copy()
        # Ensure h_values are within [0, 1]
        h_values = np.clip(h_values, 0.0, 1.0)
        # Ensure the integral constraint is met
        sum_h_current = np.sum(h_values) * (2.0 / n_points)
        if np.abs(sum_h_current - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h_current
            h_values = h_values * scaling_factor
        # Compute the overlap integral
        return evaluate_erdos_solution(h_values)
    
    # Set bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a more robust optimization method to escape local optima
    result = minimize(
        objective,
        x0=initial_h_extended,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 150, 'ftol': 1e-8, 'gtol': 1e-8, 'disp': False}
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Ensure the constraint is satisfied
    sum_h = np.sum(best_h) * (2.0 / n_points)
    if np.abs(sum_h - 1.0) > 1e-6:
        scaling_factor = 1.0 / sum_h
        best_h = best_h * scaling_factor
        best_c5 = evaluate_erdos_solution(best_h)
    
    # If the time budget is not exhausted, perform a local search
    if time.time() - start_time < budget_s - 5:
        # Define a local search with perturbations
        def perturbate(x, perturbation=0.05):
            x_perturbed = x.copy()
            for i in range(len(x_perturbed)):
                if np.random.rand() < 0.1:
                    x_perturbed[i] += np.random.uniform(-perturbation, perturbation)
                    x_perturbed[i] = np.clip(x_perturbed[i], 0.0, 1.0)
            return x_perturbed
        
        # Perform a few iterations of local search
        for _ in range(5):
            perturbed_h = perturbate(best_h)
            perturbed_c5 = evaluate_erdos_solution(perturbed_h)
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5
    
    # Ensure the constraint is satisfied again after local search
    sum_h = np.sum(best_h) * (2.0 / n_points)
    if np.abs(sum_h - 1.0) > 1e-6:
        scaling_factor = 1.0 / sum_h
        best_h = best_h * scaling_factor
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
