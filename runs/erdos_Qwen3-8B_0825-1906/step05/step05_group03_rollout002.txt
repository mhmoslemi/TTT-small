import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    n_points = 80
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    n_points = 80  # Start with a moderate number of points for balance between speed and accuracy
    
    best_c5 = 100.0
    best_h = initial_h_values()
    
    # Initial solution
    initial_c5 = evaluate_erdos_solution(best_h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
    
    # Try a few random initializations
    for _ in range(10):
        h = np.random.uniform(0, 1, n_points)
        # Normalize to satisfy the integral constraint
        sum_h = np.sum(h) * 2.0 / n_points
        h = h * (1.0 / sum_h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Optimization using Bayesian optimization with constraints
    def objective(x):
        h_values = x.copy()
        # Ensure h_values are within [0, 1]
        h_values = np.clip(h_values, 0.0, 1.0)
        # Ensure the integral constraint is met
        sum_h_current = np.sum(h_values) * 2.0 / n_points
        scaling_factor = 1.0 / sum_h_current
        h_values = h_values * scaling_factor
        # Compute the overlap integral
        return evaluate_erdos_solution(h_values)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Bayesian optimization with constraints
    from scipy.optimize import minimize
    from skopt import BayesSearchCV
    from skopt.space import Real, Categorical, Integer
    from skopt.utils import use_fixed_bounds
    
    # Define the search space
    search_space = [Real(0, 1, name=f'h_{i}') for i in range(n_points)]
    
    # Initialize Bayesian optimization
    opt = BayesSearchCV(
        optimizer=optimize.minimize,
        objective=objective,
        search_space=search_space,
        n_iter=50,
        n_jobs=-1,
        random_state=seed,
        constraint=lambda x: np.sum(x) * 2.0 / n_points - 1.0
    )
    
    # Initial guess with the best so far
    opt.fit(np.reshape(best_h, (1, -1)))
    
    # Get the best found solution
    best_h = opt.best_params_.values
    best_h = np.array(best_h).flatten()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Early stopping if we run out of time
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Try another Bayesian optimization run with perturbed initial guess
    perturbed_h = best_h + np.random.uniform(-0.05, 0.05, n_points)
    perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
    sum_perturbed = np.sum(perturbed_h) * 2.0 / n_points
    perturbed_h = perturbed_h * (1.0 / sum_perturbed)
    
    opt.fit(np.reshape(perturbed_h, (1, -1)))
    
    best_h = opt.best_params_.values
    best_h = np.array(best_h).flatten()
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
