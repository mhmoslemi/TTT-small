```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    n_points = 80
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    n_points = 100  # Increased to 100 for better resolution
    
    best_c5 = 1.0
    best_h = initial_h_values()
    
    # Initial solution
    initial_c5 = evaluate_erdos_solution(best_h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
    
    # Try a few random initializations
    for _ in range(10):
        h = np.random.uniform(0, 1, n_points)
        # Normalize to satisfy the integral constraint
        sum_h = np.sum(h) * 2.0 / n_points
        h = h * (1.0 / sum_h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Function to evaluate the objective and constraints
    def objective(x):
        h_values = x.copy()
        # Ensure h_values are within [0, 1]
        h_values = np.clip(h_values, 0.0, 1.0)
        # Ensure the integral constraint is met
        sum_h_current = np.sum(h_values) * 2.0 / n_points
        if np.abs(sum_h_current - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h_current
            h_values = h_values * scaling_factor
        # Compute the overlap integral
        return evaluate_erdos_solution(h_values)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization using Sequential Least Squares Programming with constraints
    result = optimize.minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Early stopping if we run out of time
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Try another optimization run with perturbed initial guess
    perturbed_h = best_h + np.random.uniform(-0.05, 0.05, n_points)
    perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
    sum_perturbed = np.sum(perturbed_h) * 2.0 / n_points
    perturbed_h = perturbed_h * (1.0 / sum_perturbed)
    result = optimize.minimize(
        objective,
        x0=perturbed_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
```
