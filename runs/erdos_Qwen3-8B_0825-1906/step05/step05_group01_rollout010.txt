import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Initial approximation based on a simple periodic pattern
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75 or 1.25 <= x <= 1.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a refined initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75 or 1.25 <= x <= 1.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Use a more advanced optimization strategy with genetic algorithm
    def genetic_optimization(population_size=50, generations=50, mutation_rate=0.1):
        # Initialize population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h / np.sum(h) * n_points / 2 for h in population]
        
        for generation in range(generations):
            # Evaluate fitness
            fitness = [evaluate_erdos_solution(h) for h in population]
            
            # Select top performers
            top_indices = np.argsort(fitness)[:int(population_size / 2)]
            selected = [population[i] for i in top_indices]
            
            # Create new population through crossover and mutation
            new_population = []
            for _ in range(population_size):
                # Crossover
                parent1, parent2 = random.sample(selected, 2)
                child = np.random.rand(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
                
                # Ensure integral constraint
                child = child / np.sum(child) * n_points / 2
                new_population.append(child)
            
            population = new_population
        
        # Find the best solution in the final population
        final_fitness = [evaluate_erdos_solution(h) for h in population]
        best_index = np.argmin(final_fitness)
        best_h = population[best_index]
        best_c5 = final_fitness[best_index]
        return best_h, best_c5
    
    # Run genetic optimization
    best_h, best_c5 = genetic_optimization()
    
    # Local refinement with SLSQP
    def local_refinement(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        return result.x
    
    best_h = local_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
