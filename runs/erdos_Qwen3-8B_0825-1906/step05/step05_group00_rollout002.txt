```python
import numpy as np
import scipy.optimize as optimize
import time
import random
import sys

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Return a higher-resolution initial approximation of the solution
    # This is a more refined and balanced initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    h = np.array(initial_h)
    
    # Function to evaluate the objective function
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Function to enforce the constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for variables
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to perform a single optimization step with constraints
    def optimize_with_constraints(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 100, 'disp': False}
        )
        return result.x
    
    # Evaluate initial solution
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed_time = time.time() - start_time
    
    # Start with the initial optimization
    current_h = optimize_with_constraints(best_h)
    current_c5 = evaluate_erdos_solution(current_h)
    if current_c5 < best_c5:
        best_h = current_h
        best_c5 = current_c5
    
    # Use a different initialization strategy for exploration
    for _ in range(10):  # Limit iterations based on time
        if elapsed_time > budget_s - 1:
            break
        
        # Generate a new random initialization
        new_h = np.random.uniform(0.0, 1.0, size=n_points)
        new_h = new_h / np.sum(new_h) * n_points / 2
        
        # Optimize from the new starting point
        new_h_opt = optimize_with_constraints(new_h)
        new_c5 = evaluate_erdos_solution(new_h_opt)
        
        # Update best solution if improved
        if new_c5 < best_c5:
            best_h = new_h_opt
            best_c5 = new_c5
        
        # Update current solution for next iteration
        current_h = new_h_opt
        elapsed_time = time.time() - start_time
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
```
