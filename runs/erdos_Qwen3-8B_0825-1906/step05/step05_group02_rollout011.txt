import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # A good starting point with a smooth transition
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)  # Keep it consistent for the rest of the run
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def evaluate_and_update(x):
        current_c5 = evaluate_erdos_solution(x)
        if current_c5 < best_c5:
            nonlocal best_c5
            best_c5 = current_c5
            best_h = x.copy()
        return current_c5
    
    # Gradient descent with momentum for global search
    def gradient_descent(n_iter=100, learning_rate=0.01, momentum=0.9, perturb_rate=0.1):
        h = np.array(initial_h)
        h = h / np.sum(h) * n_points / 2
        velocity = np.zeros(n_points)
        best_h = h.copy()
        best_c5 = evaluate_erdos_solution(best_h)
        
        for iter in range(n_iter):
            # Random perturbation
            if np.random.rand() < perturb_rate:
                h += np.random.uniform(-0.1, 0.1, size=n_points)
                h = np.clip(h, 0.0, 1.0)
                h = h / np.sum(h) * n_points / 2
            
            # Compute current objective and gradient
            current_c5 = evaluate_erdos_solution(h)
            
            # Estimate gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-6
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = h_plus / np.sum(h_plus) * n_points / 2
                grad[i] = (evaluate_erdos_solution(h_plus) - current_c5) / eps
            
            # Update velocity and h
            velocity = momentum * velocity - learning_rate * grad
            h += velocity
            h = np.clip(h, 0.0, 1.0)
            h = h / np.sum(h) * n_points / 2
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        return best_h, best_c5
    
    # Run gradient descent
    try:
        best_h, best_c5 = gradient_descent(n_iter=200, learning_rate=0.01, momentum=0.9, perturb_rate=0.1)
    except:
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=Bounds([0.0]*n_points, [1.0]*n_points),
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300, 'disp': False}
        )
        return result.x
    
    # Refine the best solution
    best_h = local_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perturb and refine
    for _ in range(5):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Run local refinement on the perturbed solution
        refined_h = local_refinement(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)
        
        # Update best solution
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Add a final perturbation and refinement to fine-tune
    perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
    perturbed_h = best_h + perturbation
    perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
    perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
    
    refined_h = local_refinement(perturbed_h)
    refined_c5 = evaluate_erdos_solution(refined_h)
    
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
