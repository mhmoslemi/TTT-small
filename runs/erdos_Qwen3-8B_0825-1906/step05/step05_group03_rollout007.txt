```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    n_points = 80
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    random.seed(seed)
    
    # Try different numbers of points
    n_points_list = [64, 80, 96, 112, 128]
    best_c5 = 1.0
    best_h = initial_h_values()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Initialize with a pattern to avoid local optima
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * 2.0 / n_points
            if 0.25 <= x <= 0.75:
                h[i] = 1.0
            elif 0.45 <= x <= 0.55:
                h[i] = 0.5
            elif 0.3 <= x <= 0.4 or 0.6 <= x <= 0.7:
                h[i] = 0.75
        h = h.tolist()
        
        # Normalize to satisfy the integral constraint
        sum_h = np.sum(h) * 2.0 / n_points
        h = np.array(h) * (1.0 / sum_h)
        h = h.tolist()
        
        def objective(x):
            h_values = x.copy()
            # Ensure h_values are within [0, 1]
            h_values = np.clip(h_values, 0.0, 1.0)
            # Ensure the integral constraint is met
            sum_h_current = np.sum(h_values) * 2.0 / n_points
            if np.abs(sum_h_current - 1.0) > 1e-6:
                scaling_factor = 1.0 / sum_h_current
                h_values = h_values * scaling_factor
            # Compute the overlap integral
            return evaluate_erdos_solution(h_values)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization with a perturbed initial guess
        result = optimize.minimize(
            objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) * 2.0 / n_points - 1.0}],
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        
        h_opt = result.x
        c5 = evaluate_erdos_solution(h_opt)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.tolist()
    
    return (best_h, best_c5, len(best_h))
```
