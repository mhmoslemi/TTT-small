```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial approximation
    initial_h = initial_h_values()
    best_h = np.array(initial_h)
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Optimizer parameters
    optimizer = optimize.minimize
    opt_params = {
        'method': 'L-BFGS-B',
        'bounds': bounds,
        'constraints': [{'type': 'eq', 'fun': constraint_sum}],
        'options': {'ftol': 1e-10, 'maxiter': 300, 'disp': False}
    }
    
    # Function to perform local optimization
    def local_optimize(x0):
        result = optimizer(
            fun=objective,
            x0=x0,
            **opt_params
        )
        return result.x
    
    # Function to perform global optimization with perturbation
    def global_optimize(x0):
        results = []
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed = x0 + perturbation
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed = perturbed / np.sum(perturbed) * n_points / 2
            refined = local_optimize(perturbed)
            results.append(refined)
        
        evaluated = [(evaluate_erdos_solution(h), h) for h in results]
        best_eval, best_h = min(evaluated, key=lambda x: x[0])
        return best_h
    
    # Perform multiple refinements
    for _ in range(5):
        best_h = global_optimize(best_h)
        best_c5 = evaluate_erdos_solution(best_h)
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Check if time is up
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
```
