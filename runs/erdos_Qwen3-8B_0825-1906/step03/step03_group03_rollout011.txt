import numpy as np
import scipy.optimize as optimize
import time
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return a higher-resolution initial approximation of the solution
    # This is a more refined and balanced initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    # Normalize to satisfy the integral constraint
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()

    # Try different numbers of points in the search
    n_points_list = [100, 150, 200, 250, 300]
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    for n_points in n_points_list:
        # Initialize with a balanced pattern
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            x = i * 2.0 / n_points
            if 0.25 <= x <= 0.75:
                h_initial[i] = 1.0
        h_initial = h_initial / np.sum(h_initial) * n_points / 2

        # Define the objective function to minimize
        def objective(x):
            h_values = x.copy()
            return evaluate_erdos_solution(h_values)

        # Define the constraint
        def constraint_sum(x):
            return np.sum(x) - n_points / 2

        # Set up bounds for variables
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Use a different optimization method and increase the number of iterations
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-9, 'maxiter': 500}
        )

        # Check if we have exceeded the time budget
        if time.time() - start_time > budget_s - 1:
            # If we have run out of time, return the best solution found so far
            h_values = result.x
            final_c5 = evaluate_erdos_solution(h_values)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_values
                best_n = n_points
            break

        # Evaluate the final solution
        h_values = result.x
        final_c5 = evaluate_erdos_solution(h_values)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_values
            best_n = n_points

    # Return the best solution found
    return (best_h.tolist(), best_c5, best_n)
