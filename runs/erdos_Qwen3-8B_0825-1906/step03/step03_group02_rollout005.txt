import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import differential_evolution, minimize
from scipy.optimize import Bounds

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return the initial approximation of the solution
    # This is a placeholder - in practice, it would be loaded from a file or generated
    # For demonstration, we return a simple symmetric function
    n_points = 50
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Use more points for better resolution

    # Initialize the h_values with a better starting guess
    h_initial = initial_h_values()
    h_initial = np.array(h_initial)
    sum_h = np.sum(h_initial) * (2.0 / n_points)
    if np.abs(sum_h - 1.0) > 1e-6:
        h_initial = h_initial * (1.0 / sum_h)
    h_initial = h_initial.tolist()
    
    # Define the objective function
    def objective(x):
        h_values = x.copy()
        h_values = np.clip(h_values, 0.0, 1.0)
        sum_h_current = np.sum(h_values) * (2.0 / n_points)
        if np.abs(sum_h_current - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h_current
            h_values = h_values * scaling_factor
        return evaluate_erdos_solution(h_values)
    
    # Define the constraint function
    def constraint(x):
        h_values = x.copy()
        h_values = np.clip(h_values, 0.0, 1.0)
        sum_h_current = np.sum(h_values) * (2.0 / n_points)
        return sum_h_current - 1.0
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint_func = {'type': 'eq', 'fun': constraint}
    
    # Use a combination of differential evolution and local optimization
    # First, use differential evolution to find a good starting point
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        seed=seed
    )
    
    # Use the result from differential evolution as the initial guess for local optimization
    initial_guess = result_de.x
    initial_guess = np.clip(initial_guess, 0.0, 1.0)
    sum_h_current = np.sum(initial_guess) * (2.0 / n_points)
    if np.abs(sum_h_current - 1.0) > 1e-6:
        initial_guess = initial_guess * (1.0 / sum_h_current)
    
    # Perform local optimization with L-BFGS-B
    result = minimize(
        objective,
        initial_guess,
        bounds=Bounds([0.0]*n_points, [1.0]*n_points),
        method='L-BFGS-B',
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure the best solution meets the constraint
    sum_h = np.sum(best_h) * (2.0 / n_points)
    if np.abs(sum_h - 1.0) > 1e-6:
        scaling_factor = 1.0 / sum_h
        best_h = best_h * scaling_factor
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
