import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import differential_evolution

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return the initial approximation of the solution
    # This is a placeholder - in practice, it would be loaded from a file or generated
    # For demonstration, we return a simple symmetric function
    n_points = 50
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 50  # Start with 50 points
    
    # Initialize the h_values with a better starting guess
    h_initial = initial_h_values()
    h_initial = np.array(h_initial)
    sum_h = np.sum(h_initial) * (2.0 / n_points)
    if np.abs(sum_h - 1.0) > 1e-6:
        h_initial = h_initial * (1.0 / sum_h)
    h_initial = h_initial.tolist()
    
    # Try different numbers of points to find the best balance between accuracy and speed
    for n_points in [50, 60, 70, 80, 90, 100]:
        if time.time() - start_time > budget_s - 0.5:
            break
        
        # Define the objective function
        def objective(x):
            h_values = x.copy()
            h_values = np.clip(h_values, 0.0, 1.0)
            sum_h_current = np.sum(h_values) * (2.0 / n_points)
            if np.abs(sum_h_current - 1.0) > 1e-6:
                scaling_factor = 1.0 / sum_h_current
                h_values = h_values * scaling_factor
            return evaluate_erdos_solution(h_values)
        
        # Use differential evolution for global optimization
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed
        )
        
        current_h = result.x
        current_c5 = result.fun
        
        # Ensure the current solution meets the constraint
        sum_h = np.sum(current_h) * (2.0 / n_points)
        if np.abs(sum_h - 1.0) > 1e-6:
            scaling_factor = 1.0 / sum_h
            current_h = current_h * scaling_factor
            current_c5 = evaluate_erdos_solution(current_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
