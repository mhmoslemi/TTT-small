import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return the initial approximation of the solution
    # This is a placeholder - in practice, it would be loaded from a file or generated
    # For demonstration, we return a simple symmetric function
    n_points = 47
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    return h.tolist()

def generate_random_h(n_points):
    h = np.random.uniform(0, 1, n_points)
    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h) * 2.0 / n_points
    h = h * (1.0 / sum_h)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    # Start timer
    start_time = time.time()
    random.seed(seed)
    # Use a higher resolution for better accuracy
    n_points = 60
    
    # Function to evaluate the overlap
    def evaluate_overlap(h_values):
        h = np.array(h_values)
        one_minus_h = 1 - h
        conv = np.correlate(h, one_minus_h, mode="full")
        max_overlap = np.max(conv)
        dx = 2.0 / n_points
        return max_overlap * dx
    
    # Function to compute the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) * 2.0 / n_points - 1.0

    # Function to clip and scale the h values to meet the integral constraint
    def clip_and_scale(h_values):
        h_values = np.clip(h_values, 0.0, 1.0)
        sum_h = np.sum(h_values) * 2.0 / n_points
        if sum_h != 0:
            scaling_factor = 1.0 / sum_h
            h_values *= scaling_factor
        return h_values
    
    # Initial solution
    best_h = initial_h_values()
    best_c5 = evaluate_overlap(best_h)
    
    # Try random initializations
    for _ in range(10):
        h = generate_random_h(n_points)
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Optimization using differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define the objective function
    def objective(x):
        h_values = clip_and_scale(x)
        return evaluate_overlap(h_values)
    
    # Use differential evolution for global optimization
    result = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        polish=True,
        vectorized=True
    )
    
    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Check elapsed time and return early if necessary
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Use a more robust optimization method for refinement
    def objective_with_constraint(x):
        h_values = clip_and_scale(x)
        return evaluate_overlap(h_values)
    
    result = optimize.minimize(
        objective_with_constraint,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': integral_constraint}],
        options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Final check for elapsed time
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
