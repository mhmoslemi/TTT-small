import numpy as np
import scipy.optimize as optimize
import time
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

def initial_h_values():
    # Return a higher-resolution initial approximation of the solution
    # This is a more refined and balanced initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    # Normalize to satisfy the integral constraint
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 200
    initial_h = initial_h_values()
    
    # Try different initializations to avoid local minima
    def try_initialization(h_start):
        h = np.array(h_start)
        def objective(x):
            h_values = x.copy()
            return evaluate_erdos_solution(h_values)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Use L-BFGS-B for faster convergence
        result = optimize.minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        return result.x
    
    # Try multiple initializations to avoid getting stuck in local minima
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try 3 different starting points
    for _ in range(3):
        # Random perturbation
        np.random.seed(seed + _)  # Ensure reproducibility
        perturbation = np.random.rand(n_points) * 0.1
        new_h = np.clip(initial_h + perturbation, 0, 1)
        new_h = new_h / np.sum(new_h) * n_points / 2  # Enforce integral constraint
        
        # Perform optimization
        optimized_h = try_initialization(new_h)
        current_c5 = evaluate_erdos_solution(optimized_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        # If we have run out of time, return the best solution found so far
        h_values = best_h
        final_c5 = best_c5
        return (h_values.tolist(), final_c5, n_points)
    
    # Evaluate the final solution
    h_values = best_h
    final_c5 = best_c5
    
    return (h_values.tolist(), final_c5, n_points)
