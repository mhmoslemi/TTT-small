import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)  # Keep it consistent for the rest of the run
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Define a function to update the best solution
    def update_best(current_h, current_c5):
        nonlocal best_h, best_c5
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Define a function to perturb the current solution
    def perturb_solution(h_values):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = h_values + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        return perturbed_h
    
    # Use a combination of local optimization and global search
    # Start with a local optimization on the initial solution
    result = optimize.minimize(
        fun=objective,
        x0=np.array(initial_h),
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    update_best(best_h, best_c5)
    
    # Perform a small-scale global search using random sampling
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        # Randomly perturb the current solution
        perturbed_h = perturb_solution(best_h)
        # Local refinement
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        refined_h = result.x
        refined_c5 = evaluate_erdos_solution(refined_h)
        update_best(refined_h, refined_c5)
    
    # Perform a more intensive local optimization
    result = optimize.minimize(
        fun=objective,
        x0=np.array(best_h),
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 300}
    )
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    update_best(best_h, best_c5)
    
    # Additional perturbations and refinements
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        perturbed_h = perturb_solution(best_h)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        refined_h = result.x
        refined_c5 = evaluate_erdos_solution(refined_h)
        update_best(refined_h, refined_c5)
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
