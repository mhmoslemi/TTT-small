import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)  # Keep it consistent for the rest of the run
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Local refinement with SLSQP
    def local_refinement(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        return result.x
    
    # Run local refinement on initial solution
    best_h = local_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try multiple starting points with random perturbations
    for _ in range(10):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Run local refinement on the perturbed solution
        refined_h = local_refinement(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)
        
        # Update best solution
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Try gradient descent for further refinement
    def gradient_descent(h_values, learning_rate=0.01, max_iter=100):
        for _ in range(max_iter):
            gradient = np.zeros_like(h_values)
            for k in range(n_points):
                # Compute the gradient of the objective function
                # This is a simplified approximation using finite differences
                h_plus = h_values.copy()
                h_plus[k] += 1e-5
                h_plus = h_plus / np.sum(h_plus) * n_points / 2
                obj_plus = evaluate_erdos_solution(h_plus)
                
                h_minus = h_values.copy()
                h_minus[k] -= 1e-5
                h_minus = h_minus / np.sum(h_minus) * n_points / 2
                obj_minus = evaluate_erdos_solution(h_minus)
                
                gradient[k] = (obj_plus - obj_minus) / (2 * 1e-5)
            
            # Update the solution
            h_values = h_values - learning_rate * gradient
            h_values = np.clip(h_values, 0.0, 1.0)
            h_values = h_values / np.sum(h_values) * n_points / 2
            
            # Early stopping if improvement is minimal
            if np.abs(evaluate_erdos_solution(h_values) - best_c5) < 1e-6:
                break
        
        return h_values
    
    # Run gradient descent
    best_h = gradient_descent(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
