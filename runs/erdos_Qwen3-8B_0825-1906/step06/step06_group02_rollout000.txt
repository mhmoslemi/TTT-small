import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a lower resolution to save time
    n_points = 100
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    n_points_current = len(best_h)
    
    # Function to evaluate and return the overlap integral
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Constraint: integral of h(x) over [0, 2] is 1
    def constraint_sum(x):
        return np.sum(x) - n_points_current / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points_current)]
    
    # Function to refine using SLSQP
    def local_refinement(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        return result.x
    
    # Step 1: Use a global optimizer to find a good initial solution
    # Use differential evolution (DE) for global search
    def differential_evolution(pop_size=20, max_iter=50):
        # Initialize population
        population = np.random.uniform(0.0, 1.0, (pop_size, n_points_current))
        population = population / np.sum(population, axis=1, keepdims=True) * n_points_current / 2
        best_individual = population[0].copy()
        best_fitness = objective(best_individual)
        
        for _ in range(max_iter):
            for i in range(pop_size):
                # Randomly select three distinct individuals
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = population[a] + np.random.uniform(-1, 1, n_points_current) * (population[b] - population[c])
                mutant = np.clip(mutant, 0.0, 1.0)
                mutant = mutant / np.sum(mutant) * n_points_current / 2
                mutant_fitness = objective(mutant)
                
                # Compare with current individual
                if mutant_fitness < objective(population[i]):
                    population[i] = mutant.copy()
                    if mutant_fitness < best_fitness:
                        best_individual = mutant.copy()
                        best_fitness = mutant_fitness
        return best_individual, best_fitness
    
    # Run DE to find a good initial solution
    try:
        best_h, best_c5 = differential_evolution(pop_size=20, max_iter=50)
    except:
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Step 2: Local refinement with SLSQP
    best_h = local_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Step 3: Perturb and refine multiple times
    for _ in range(5):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points_current)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points_current / 2
        
        # Run local refinement on the perturbed solution
        refined_h = local_refinement(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)
        
        # Update best solution
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Step 4: Try with a finer grid to see if we can get a better result
    if time.time() - start_time < budget_s - 5:
        n_points_new = 200
        # Initial approximation for finer grid
        h_new = np.zeros(n_points_new)
        for i in range(n_points_new):
            x = i * 2.0 / n_points_new
            if 0.25 <= x <= 0.75:
                h_new[i] = 1.0
        h_new = h_new / np.sum(h_new) * n_points_new / 2
        initial_h_new = h_new.tolist()
        
        # Run DE on the finer grid
        best_h_new, best_c5_new = differential_evolution(pop_size=20, max_iter=50)
        best_h_new = local_refinement(best_h_new)
        best_c5_new = evaluate_erdos_solution(best_h_new)
        
        # Perturb and refine
        for _ in range(5):
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points_new)
            perturbed_h_new = best_h_new + perturbation
            perturbed_h_new = np.clip(perturbed_h_new, 0.0, 1.0)
            perturbed_h_new = perturbed_h_new / np.sum(perturbed_h_new) * n_points_new / 2
            
            refined_h_new = local_refinement(perturbed_h_new)
            refined_c5_new = evaluate_erdos_solution(refined_h_new)
            
            if refined_c5_new < best_c5_new:
                best_c5_new = refined_c5_new
                best_h_new = refined_h_new
        
        # Update the best solution if the new one is better
        if best_c5_new < best_c5:
            best_c5 = best_c5_new
            best_h = best_h_new
            n_points_current = n_points_new
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points_current)
    
    return (best_h.tolist(), final_c5, n_points_current)
