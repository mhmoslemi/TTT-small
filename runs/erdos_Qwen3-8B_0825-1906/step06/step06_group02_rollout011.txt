import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)  # Keep it consistent for the rest of the run
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Genetic algorithm for global search
    def genetic_algorithm(pop_size=50, generations=100, mutation_rate=0.05):
        # Initialize population
        population = np.random.uniform(0.0, 1.0, (pop_size, n_points))
        population = population / np.sum(population, axis=1, keepdims=True) * n_points / 2
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = np.array([evaluate_erdos_solution(ind) for ind in population])
            
            # Select parents using tournament selection
            tournament_size = 5
            parents_indices = np.random.choice(pop_size, (pop_size, tournament_size), replace=False)
            parent_fitness = fitness[parents_indices]
            parent_indices = np.argmin(parent_fitness, axis=1)
            parents = population[parent_indices]
            
            # Crossover
            offspring = np.zeros((pop_size, n_points))
            for i in range(pop_size):
                if np.random.rand() < 0.8:
                    # Single-point crossover
                    crossover_point = np.random.randint(1, n_points - 1)
                    offspring[i, :crossover_point] = parents[i, :crossover_point]
                    offspring[i, crossover_point:] = np.random.choice(parents[np.random.randint(0, pop_size), crossover_point:], size=n_points - crossover_point)
                else:
                    # Clone
                    offspring[i] = parents[i].copy()
            
            # Mutation
            for i in range(pop_size):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i, j] = np.random.uniform(0.0, 1.0)
                        offspring[i, j] = offspring[i, j] / np.sum(offspring[i]) * n_points / 2
            
            # Evaluate offspring
            offspring_fitness = np.array([evaluate_erdos_solution(ind) for ind in offspring])
            
            # Select next generation
            combined = np.column_stack((population.flatten(), fitness))
            combined_offspring = np.column_stack((offspring.flatten(), offspring_fitness))
            combined = np.vstack((combined, combined_offspring))
            combined = combined[combined[:, 1].argsort()]
            population = combined[:pop_size, 0].reshape(pop_size, n_points)
        
        best_index = np.argmin(fitness)
        return population[best_index], fitness[best_index]
    
    # Run genetic algorithm
    try:
        best_h, best_c5 = genetic_algorithm(pop_size=50, generations=100, mutation_rate=0.05)
    except:
        best_h = initial_h
        best_c5 = evaluate_erdos_solution(best_h)
    
    # Local refinement with SLSQP
    def local_refinement(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        return result.x
    
    # Refine the best solution
    best_h = local_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perturb and refine
    for _ in range(5):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Run local refinement on the perturbed solution
        refined_h = local_refinement(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)
        
        # Update best solution
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
