```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Return a more refined and distributed initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.2 <= x <= 0.8:
            h[i] = 0.5
        elif 0.1 <= x <= 0.3 or 0.7 <= x <= 0.9:
            h[i] = 0.8
        elif 0.05 <= x <= 0.15 or 0.85 <= x <= 0.95:
            h[i] = 0.3
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Try different initializations to avoid local optima
    n_points_list = [100, 150, 200, 250]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 200
    
    for n_points in n_points_list:
        h = initial_h_values()
        h = np.array(h)
        h = np.resize(h, n_points)
        h = h / np.sum(h) * n_points / 2
        
        # Function to evaluate the objective function
        def objective(x):
            h_values = x.copy()
            return evaluate_erdos_solution(h_values)
        
        # Function to enforce the constraint
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        # Bounds for variables
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Function to perform a single optimization step with constraints
        def optimize_with_constraints(x0):
            result = optimize.minimize(
                fun=objective,
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
            )
            return result.x
        
        # Evaluate initial solution
        current_h = h
        current_c5 = evaluate_erdos_solution(current_h)
        
        # Perform a more systematic search with diverse perturbations
        for _ in range(30):  # Limit iterations based on time
            if time.time() - start_time > budget_s - 1:
                break
            
            # Generate a perturbed solution
            perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
            perturbed_h = current_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
            
            # Optimize from the perturbed starting point
            perturbed_h_opt = optimize_with_constraints(perturbed_h)
            perturbed_c5 = evaluate_erdos_solution(perturbed_h_opt)
            
            # Update best solution if improved
            if perturbed_c5 < current_c5:
                current_h = perturbed_h_opt
                current_c5 = perturbed_c5
            
            # Update current solution for next iteration
            current_h = perturbed_h_opt
        
        # Update the best solution overall
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n_points = n_points
    
    # Return the best solution found across all initializations
    return (best_h.tolist(), best_c5, best_n_points)
```
