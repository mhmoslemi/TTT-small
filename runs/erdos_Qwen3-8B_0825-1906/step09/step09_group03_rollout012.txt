```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Return a more refined and distributed initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.2 <= x <= 0.8:
            h[i] = 0.5
        elif 0.1 <= x <= 0.3 or 0.7 <= x <= 0.9:
            h[i] = 0.8
        elif 0.05 <= x <= 0.15 or 0.85 <= x <= 0.95:
            h[i] = 0.3
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    h = np.array(initial_h)
    
    # Define a function to evaluate the objective and constraints
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define the constraint for the total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Define bounds for variables
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define a function to perform optimization with constraints
    def optimize_with_constraints(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
        )
        return result.x
    
    # Evaluate the initial solution
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed_time = time.time() - start_time
    
    # Start the optimization process with a global search
    for _ in range(10):
        if elapsed_time > budget_s - 1:
            break
        
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Perform local optimization from this perturbed starting point
        perturbed_h_opt = optimize_with_constraints(perturbed_h)
        perturbed_c5 = evaluate_erdos_solution(perturbed_h_opt)
        
        # Update the best solution if this one is better
        if perturbed_c5 < best_c5:
            best_h = perturbed_h_opt
            best_c5 = perturbed_c5
        
        # Update the current solution for the next iteration
        h = perturbed_h_opt
        elapsed_time = time.time() - start_time
    
    # Perform a more refined local optimization with smaller perturbations
    for _ in range(15):
        if elapsed_time > budget_s - 1:
            break
        
        # Generate a smaller local perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Perform local optimization from this perturbed starting point
        perturbed_h_opt = optimize_with_constraints(perturbed_h)
        perturbed_c5 = evaluate_erdos_solution(perturbed_h_opt)
        
        # Update the best solution if this one is better
        if perturbed_c5 < best_c5:
            best_h = perturbed_h_opt
            best_c5 = perturbed_c5
        
        # Update the current solution for the next iteration
        h = perturbed_h_opt
        elapsed_time = time.time() - start_time
    
    # Add a final local optimization step
    for _ in range(5):
        if elapsed_time > budget_s - 1:
            break
        
        # Generate a very small perturbation
        perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
        perturbed_h = h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Perform local optimization from this perturbed starting point
        perturbed_h_opt = optimize_with_constraints(perturbed_h)
        perturbed_c5 = evaluate_erdos_solution(perturbed_h_opt)
        
        # Update the best solution if this one is better
        if perturbed_c5 < best_c5:
            best_h = perturbed_h_opt
            best_c5 = perturbed_c5
        
        # Update the current solution for the next iteration
        h = perturbed_h_opt
        elapsed_time = time.time() - start_time
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
```
