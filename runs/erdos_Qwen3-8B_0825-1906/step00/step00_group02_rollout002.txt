```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

def initial_h_values(n_points):
    # A simple initial guess that approximates an optimal distribution
    h = np.zeros(n_points)
    step = n_points // 4
    for i in range(step):
        h[i] = 1.0
    for i in range(step, 2 * step):
        h[i] = 0.5
    for i in range(2 * step, 3 * step):
        h[i] = 0.0
    for i in range(3 * step, n_points):
        h[i] = 0.5
    h = np.roll(h, -step)
    return h

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = signal.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Start with the initial n_points
    max_evals = 50  # Limit the number of evaluations to stay within budget
    best_h = initial_h_values(n_points)
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    start_time = time.time()

    for _ in range(max_evals):
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(best_h + perturbation, 0, 1)
        # Ensure the integral constraint is satisfied
        h_perturbed -= (np.sum(h_perturbed) - n_points / 2) / n_points
        # Evaluate the overlap
        c5 = evaluate_erdos_solution(h_perturbed, n_points)
        if c5 < best_c5:
            best_h = h_perturbed
            best_c5 = c5
        # Check if time is up
        if time.time() - start_time > budget_s:
            break

    # Try a larger n_points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.1:
        n_points = 100
        best_h = initial_h_values(n_points)
        best_c5 = evaluate_erdos_solution(best_h, n_points)
        start_time = time.time()
        for _ in range(20):
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(best_h + perturbation, 0, 1)
            h_perturbed -= (np.sum(h_perturbed) - n_points / 2) / n_points
            c5 = evaluate_erdos_solution(h_perturbed, n_points)
            if c5 < best_c5:
                best_h = h_perturbed
                best_c5 = c5
            if time.time() - start_time > remaining_time:
                break

    return best_h, best_c5, n_points
```
