```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different numbers of points to find the best balance between accuracy and speed
    n_points_list = [100, 200, 300, 400]
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Start with an initial guess
        h_initial = initial_h_values.copy()
        h_initial = np.interp(x, np.linspace(0, 2, len(h_initial)), h_initial)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define the optimization variables
        h = cp.Variable(n_points, bounds=(0, 1))
        constraints = [cp.sum(h) == n_points / 2]
        
        # Define the overlap integral for all shifts
        max_overlap = 0
        for k in range(1, n_points):
            # Compute (1 - h(x + k)) using periodic extension
            h_shifted = np.roll(h.value, k)
            overlap = cp.sum(h * (1 - h_shifted)) if h.value is not None else 0
            max_overlap = cp.Max(max_overlap, overlap)
        
        # Objective: minimize the maximum overlap
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        
        # Solve the optimization problem
        start_time = time.time()
        try:
            prob.solve(solver=cp.ECOS, max_iter=1000, verbose=False, time_limit=budget_s - (time.time() - start_time))
        except:
            pass
        
        if prob.status == 'optimal':
            h_opt = h.value
            c5 = max_overlap.value * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n_points = n_points
        
        # Early exit if time is up
        if time.time() - start_time >= budget_s:
            break
    
    # Evaluate the best solution found
    if best_h is not None:
        c5_bound = best_c5
        n_points = best_n_points
    else:
        # Fallback to initial guess if no optimization was performed
        h_initial = initial_h_values.copy()
        h_initial = np.interp(x, np.linspace(0, 2, len(h_initial)), h_initial)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        c5_bound = evaluate_erdos_solution(h_initial)
        n_points = len(h_initial)
    
    return best_h, c5_bound, n_points
```
