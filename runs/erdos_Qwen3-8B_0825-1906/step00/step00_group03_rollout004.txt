import numpy as np
import scipy.optimize as optimize
import time
import os
import sys
import math
import random
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial setup
    n_points = 100  # Start with 100 points
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points])  # Use initial values and then refine
    h = h / (dx * n_points) * (n_points / 2)  # Normalize to ensure sum(h) * dx = 1

    def compute_c5(h_values):
        h = np.array(h_values)
        h = np.concatenate((h, h[:2]))  # Extend for full convolution
        shifted = np.array([1 - h[k:] for k in range(len(h))])
        corr = np.correlate(h, shifted, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(x):
        h = np.array(x)
        return compute_c5(h)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    # Optimization setup
    x0 = h
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}]

    try:
        result = optimize.minimize(
            objective,
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
    except:
        best_h = h
        best_c5 = compute_c5(h)

    # Refinement: Try increasing n_points
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

        def objective(x):
            h = np.array(x)
            return compute_c5(h)

        def constraint_sum(x):
            return np.sum(x) - n_points / 2

        x0 = h
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}]

        try:
            result = optimize.minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
            )
            best_h = result.x
            best_c5 = compute_c5(best_h)
        except:
            best_h = h
            best_c5 = compute_c5(h)

    # Final check with n_points = 300
    if time.time() - start_time < budget_s - 5:
        n_points = 300
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)

        def objective(x):
            h = np.array(x)
            return compute_c5(h)

        def constraint_sum(x):
            return np.sum(x) - n_points / 2

        x0 = h
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}]

        try:
            result = optimize.minimize(
                objective,
                x0,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 30, 'ftol': 1e-8, 'eps': 1e-8}
            )
            best_h = result.x
            best_c5 = compute_c5(best_h)
        except:
            best_h = h
            best_c5 = compute_c5(h)

    return (best_h, best_c5, n_points)
