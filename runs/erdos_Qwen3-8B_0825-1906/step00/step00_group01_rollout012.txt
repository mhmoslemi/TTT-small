import numpy as np
import scipy.optimize as optimize

import time

# Pre-imported functions
# evaluate_erdos_solution is a function that takes h_values and returns C5
# initial_h_values is a starting point for h_values (n=47 samples)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial_h_values
    n_points = len(initial_h_values)
    h = np.copy(initial_h_values)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h_pad = np.concatenate([h, np.zeros_like(h)])
        result = np.correlate(h_pad, 1 - h, mode="full")
        return np.max(result) * (2.0 / n_points)
    
    # Function to compute the integral constraint
    def integral_constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization parameters
    max_iterations = 100
    learning_rate = 0.1
    tolerance = 1e-4
    improvement_threshold = 1e-5
    
    best_c5 = compute_c5(h)
    best_h = np.copy(h)
    
    for _ in range(max_iterations):
        # Evaluate current C5
        current_c5 = compute_c5(h)
        
        # If we've improved enough, break early
        if abs(current_c5 - best_c5) < improvement_threshold:
            break
        
        # If current C5 is better, update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
        
        # Perturb h slightly
        perturbation = np.random.rand(n_points) * 0.1 - 0.05
        h += perturbation
        h = np.clip(h, 0, 1)
        
        # Enforce integral constraint
        integral_diff = integral_constraint(h)
        h += integral_diff / n_points
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break
    
    # Try a simple optimization using scipy.optimize
    try:
        def objective(x):
            h = x.reshape(n_points)
            return compute_c5(h)
        
        # Constraints: integral constraint
        cons = ({'type': 'eq', 'fun': lambda x: integral_constraint(x.reshape(n_points))})
        
        # Initial guess
        x0 = h.reshape(n_points)
        
        # Bounds: each h[i] ∈ [0, 1]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run optimization
        result = optimize.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        
        # Update best solution
        h = result.x
        h = np.clip(h, 0, 1)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    except:
        pass
    
    # Final check: Ensure integral constraint is satisfied
    h = np.clip(best_h, 0, 1)
    integral_diff = integral_constraint(h)
    h += integral_diff / n_points
    h = np.clip(h, 0, 1)
    best_c5 = compute_c5(h)
    
    return (h.tolist(), best_c5, n_points)
