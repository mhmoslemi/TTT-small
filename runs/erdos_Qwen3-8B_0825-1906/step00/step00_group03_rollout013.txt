import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 200  # Reduced from 700 for faster computation
    dx = 2.0 / n_points
    
    # Start with the initial solution
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]  # Ensure it fits the n_points
    
    # Normalize to satisfy the integral constraint
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k:] = h[:len(h) - k]
        overlap = np.sum(h * (1 - shifted))
        return overlap
    
    # Optimization function
    def objective(h):
        # Normalize to ensure the integral constraint is satisfied
        h_norm = h / np.sum(h) * (n_points / 2)
        # Compute maximum overlap
        max_overlap = 0
        for k in range(1, len(h_norm) - 1):
            shifted = np.zeros_like(h_norm)
            shifted[k:] = h_norm[:len(h_norm) - k]
            overlap = np.sum(h_norm * (1 - shifted))
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # Optimize using a genetic algorithm
    result = optimize.differential_evolution(
        objective,
        bounds=[(0, 1) for _ in range(n_points)],
        strategy='best1bin',
        population_size=15,
        mutation=(0.5, 1),
        recombination=0.8,
        maxiter=100,
        tol=1e-6,
        disp=False,
        polish=True,
        polish_method=None,
        seed=seed
    )
    
    # Get the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure the integral constraint is satisfied
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Final evaluation to ensure correct C5 value
    h_values = best_h
    h_values = np.clip(h_values, 0, 1)  # Ensure values are in [0, 1]
    
    # Compute the actual C5 value
    max_overlap = 0
    for k in range(1, len(h_values) - 1):
        shifted = np.zeros_like(h_values)
        shifted[k:] = h_values[:len(h_values) - k]
        overlap = np.sum(h_values * (1 - shifted))
        max_overlap = max(max_overlap, overlap)
    
    # Final check for correctness
    if np.abs(np.sum(h_values) * dx - 1) > 1e-6:
        raise ValueError("Integral constraint not satisfied")
    
    # Return best solution
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (h_values, best_c5, n_points)
    
    return (h_values, best_c5, n_points)
