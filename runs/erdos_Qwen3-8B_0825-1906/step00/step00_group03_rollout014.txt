```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial h_values
    n_points = 100
    h = np.copy(initial_h_values[:n_points])
    
    # Normalize to ensure sum(h) == n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    
    # Define the overlap integral
    def compute_c5(h_vals):
        h = np.array(h_vals)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function to minimize
    def objective(x):
        return compute_c5(x)
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - (n_points / 2)
    
    # Optimization settings
    max_iterations = 100
    tolerance = 1e-6
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial trial with gradient descent
    x0 = h.copy()
    x = np.copy(x0)
    for _ in range(max_iterations):
        dx = 0.01
        grad = np.zeros_like(x)
        for i in range(n_points):
            x_plus = np.copy(x)
            x_plus[i] += dx
            x_minus = np.copy(x)
            x_minus[i] -= dx
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * dx)
        x -= grad * tolerance
        x = np.clip(x, 0, 1)
        if np.abs(objective(x) - objective(x0)) < tolerance:
            break
    
    # Local optimization using scipy
    res = opt.minimize(objective, x0=x, bounds=bounds, method='L-BFGS-B', tol=tolerance)
    best_h = res.x
    best_c5 = compute_c5(best_h)
    
    # Evaluate if we have time left
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0:
        # Try a more robust optimization using CVXPY
        h_cvx = cp.Variable(n_points)
        obj = cp.Minimize(compute_c5(h_cvx))
        constr = [cp.sum(h_cvx) == n_points / 2, h_cvx >= 0, h_cvx <= 1]
        prob = cp.Problem(obj, constr)
        try:
            prob.solve(solver=cp.SCS, eps=tolerance, time_limit=remaining)
            best_h = h_cvx.value
            best_c5 = compute_c5(best_h)
        except:
            pass
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
```
