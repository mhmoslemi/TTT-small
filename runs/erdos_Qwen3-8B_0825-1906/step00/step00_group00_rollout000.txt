import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Initial solution
    n_points = 45
    h_values = initial_h_values.copy()
    
    # Try a few different starting points
    start_points = [initial_h_values.copy()]
    for _ in range(3):
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_values = np.clip(initial_h_values + perturbation, 0, 1)
        start_points.append(h_values)
    
    best_c5 = float('inf')
    best_h = None
    
    # Try each starting point
    for h in start_points:
        # Ensure the integral is 1
        h = h.copy()
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = h * (n_points / 2) / sum_h
        
        # Evaluate initial C5
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Try to improve further with optimization
    for _ in range(3):
        # Create a slightly larger grid
        n_points = int(np.ceil(45 * np.sqrt(2)))
        h_values = np.random.uniform(0, 1, size=n_points)
        sum_h = np.sum(h_values)
        h_values = h_values * (n_points / 2) / sum_h
        
        # Evaluate initial C5
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
    
    # Run a local optimization
    def objective(h):
        h = h.copy()
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        return evaluate_erdos_solution(h)
    
    # Use a more efficient method for local optimization
    for _ in range(3):
        # Create a slightly larger grid
        n_points = int(np.ceil(45 * np.sqrt(2)))
        h_values = np.random.uniform(0, 1, size=n_points)
        sum_h = np.sum(h_values)
        h_values = h_values * (n_points / 2) / sum_h
        
        # Optimization
        result = optimize.minimize(objective, h_values, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])

        best_h = initial_h_values.copy()

    sum_h = np.sum(best_h)
    best_h = best_h * (n_points / 2) / sum_h
    
    # Clip values to [0, 1]
    best_h = np.clip(best_h, 0, 1)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution found
    return best_h, final_c5, len(best_h)
