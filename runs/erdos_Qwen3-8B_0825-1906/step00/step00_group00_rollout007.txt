```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial solution
    n_points = 45
    h_values = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h_values, n_points)
    best_h = h_values.copy()
    
    # Try increasing n_points to improve the bound
    for n_points in [45, 60, 80, 100, 120, 150, 180, 200]:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Resample to n_points
        dx = 2.0 / n_points
        h_resampled = np.zeros(n_points)
        
        # Use a simple resampling strategy: spread the original values evenly
        h_resampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_values)), h_values)
        h_resampled = np.clip(h_resampled, 0, 1)
        h_resampled = h_resampled * (n_points / 2)  # Ensure the integral is 1
        
        # Optimize the resampled h_values
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Compute the overlap integral
        def objective(h):
            h_vals = h.value
            h_vals = np.clip(h_vals, 0, 1)
            correlation = np.correlate(h_vals, 1 - h_vals, mode="full")
            c5 = np.max(correlation) * dx
            return c5
        
        # Use a local optimization method
        result = optimize.minimize(
            objective,
            h_resampled,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=lambda x: np.gradient(objective(x), x)
        )
        
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        c5 = evaluate_erdos_solution(h_opt, n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n_points
        
        print(f"n={n_points}, C5={c5:.6f}")
    
    return best_h, best_c5, best_n
```
