import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45  # Start with 45 points
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def compute_c5(h_values):
        h = np.array(h_values)
        h_pad = np.zeros(len(h) + len(h) - 1)
        h_pad[:len(h)] = h
        h_pad[len(h):] = h
        h_shifted = np.zeros(len(h_pad))
        h_shifted[len(h)-1:] = h
        overlap = np.correlate(h_pad, 1 - h_shifted, mode="full")
        return np.max(overlap) * dx
    
    def evaluate_solution(h_values):
        return compute_c5(h_values)
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate_solution(best_h)
    
    # Try different n_points up to 1000 within budget
    for n_points in [45, 60, 90, 120, 150, 180, 210, 240, 270, 300, 330, 360, 390, 420, 450, 480, 510, 540, 570, 600]:
        if time.time() - start_time > budget_s - 1:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial guess
        h = np.zeros(n_points)
        h[::3] = 1.0  # Initial piecewise constant guess
        h = h * (n_points / 2)  # Ensure integral is 1
        
        # Optimization: minimize the maximum overlap
        # Use convex optimization to find a better solution
        
        # Objective: minimize the maximum of integrals over shifted h
        # We use a convex relaxation by using a binary search on the maximum value
        # and minimizing the integral under the constraint that it's less than the bound
        
        # We'll run a convex optimization for a few iterations to find a good solution
        for _ in range(5):
            # Binary search on the maximum overlap
            low = 0.0
            high = 1.0
            for _ in range(30):
                mid = (low + high) / 2
                # Define variables
                h_var = cp.Variable(n_points, nonneg=True)
                constraints = [
                    cp.sum(h_var) == n_points / 2,  # Integral constraint
                    h_var <= 1.0,
                    h_var >= 0.0
                ]
                # Objective: minimize max integral over shifts
                # We'll use a convex upper bound for the max
                integrals = []
                for k in range(n_points):
                    h_shifted = np.roll(h_var.value, k) if _ == 0 else np.roll(h_var, k)
                    integral = cp.sum(h_var * (1 - h_shifted)) * dx
                    integrals.append(integral)
                obj = cp.Minimize(cp.Max(integrals))
                prob = cp.Problem(obj, constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-5)
                except:
                    break
                
                if prob.status != 'OPTIMAL':
                    break
                
                if prob.value < mid:
                    high = mid
                else:
                    low = mid
            
            # Update h
            h = h_var.value
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Try to fine-tune with local optimization
        for _ in range(5):
            # Gradient descent with adaptive step size
            step_size = 0.1
            h = best_h.copy()
            for _ in range(100):
                # Compute overlaps for all shifts
                overlap = np.zeros(n_points)
                for k in range(n_points):
                    h_shifted = np.roll(h, k)
                    integral = np.sum(h * (1 - h_shifted)) * dx
                    overlap[k] = integral
                max_overlap = np.max(overlap)
                
                # Compute gradient of max_overlap with respect to h
                # Using subgradient for max function
                grad = np.zeros_like(h)
                for k in range(n_points):
                    h_shifted = np.roll(h, k)
                    integral = np.sum(h * (1 - h_shifted)) * dx
                    if integral == max_overlap:
                        grad += (1 - h_shifted) * dx
                # Update h using gradient descent
                h -= step_size * grad
                h = np.clip(h, 0, 1)
            
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final check
    final_c5 = evaluate_solution(best_h)
    
    return best_h, final_c5, len(best_h)
