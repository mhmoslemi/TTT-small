import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points_init = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=np.float64)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / len(h)
        h_flat = h.flatten()
        conv = signal.correlate(h_flat, 1 - h_flat, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Optimization function that minimizes the max overlap
    def objective(x):
        h = x.reshape(-1)
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        dx = 2.0 / len(h)
        if np.abs(np.sum(h) * dx - 1) > 1e-6:
            return np.inf
        conv = signal.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    # Function to perform optimization
    def optimize_h(h_initial, n_points, max_iter=100, tolerance=1e-5):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h_initial,
            bounds=bounds,
            method='SLSQP',
            options={'maxiter': max_iter, 'ftol': tolerance, 'eps': 1e-8}
        )
        return result.x
    
    # Try different numbers of points, starting from the initial
    best_c5 = np.inf
    best_h = h_values.copy()
    
    # Try with the initial number of points
    h_initial = h_values.copy()
    h_opt = optimize_h(h_initial, len(h_initial))
    c5_val = compute_c5(h_opt)
    if c5_val < best_c5:
        best_c5 = c5_val
        best_h = h_opt
    
    # Try with a smaller number of points (e.g., 100)
    n_points_small = 100
    h_small_initial = np.zeros(n_points_small)
    h_small_initial[:n_points_small // 2] = 1.0
    h_small_initial = h_small_initial + np.random.normal(0, 0.1, n_points_small)
    h_small_initial = np.clip(h_small_initial, 0, 1)
    h_small_initial = h_small_initial * (n_points_small / 2) / np.sum(h_small_initial)
    h_opt_small = optimize_h(h_small_initial, n_points_small)
    c5_val_small = compute_c5(h_opt_small)
    if c5_val_small < best_c5:
        best_c5 = c5_val_small
        best_h = h_opt_small
    
    # Try with a middle number of points (e.g., 200)
    n_points_mid = 200
    h_mid_initial = np.zeros(n_points_mid)
    h_mid_initial[:n_points_mid // 2] = 1.0
    h_mid_initial = h_mid_initial + np.random.normal(0, 0.1, n_points_mid)
    h_mid_initial = np.clip(h_mid_initial, 0, 1)
    h_mid_initial = h_mid_initial * (n_points_mid / 2) / np.sum(h_mid_initial)
    h_opt_mid = optimize_h(h_mid_initial, n_points_mid)
    c5_val_mid = compute_c5(h_opt_mid)
    if c5_val_mid < best_c5:
        best_c5 = c5_val_mid
        best_h = h_opt_mid
    
    # Final check to ensure we are within the budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 5:
        return (best_h, best_c5, len(best_h))
    
    # Try with a larger number of points (e.g., 300)
    n_points_large = 300
    h_large_initial = np.zeros(n_points_large)
    h_large_initial[:n_points_large // 2] = 1.0
    h_large_initial = h_large_initial + np.random.normal(0, 0.1, n_points_large)
    h_large_initial = np.clip(h_large_initial, 0, 1)
    h_large_initial = h_large_initial * (n_points_large / 2) / np.sum(h_large_initial)
    h_opt_large = optimize_h(h_large_initial, n_points_large)
    c5_val_large = compute_c5(h_opt_large)
    if c5_val_large < best_c5:
        best_c5 = c5_val_large
        best_h = h_opt_large
    
    return (best_h, best_c5, len(best_h))
