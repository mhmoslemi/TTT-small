    def evaluate_C5(h):
        h = np.array(h)
        if np.any(h < 0) or np.any(h > 1):
            return float('inf')
        k_values = np.arange(-n_points + 1, n_points)
        max_overlap = 0.0
        for k in k_values:
            if k == 0:
                continue
            shifted = np.roll(h, k)
            overlap = np.dot(h, 1 - shifted)
            max_overlap = max(max_overlap, overlap)
        return max_overlap * dx
    
    # Optimization function to minimize C₅
    def optimize_C5(h):
        # Ensure h is in [0, 1] and has the correct integral
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) * dx - 1) > 1e-6:
            return float('inf')
        return evaluate_C5(h)
    
    # Start with initial guess
    best_h = initial_h.copy()
    best_c5 = evaluate_C5(best_h)
    
    # Local optimization using L-BFGS-B
    start_time = time.time()
    result = optimize.minimize(
        optimize_C5,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100}
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = evaluate_C5(best_h)

    remaining_time = budget_s - (time.time() - start_time)

        def fitness(h):
            return evaluate_C5(h)

        for _ in range(50):
            # Evaluate fitness
            fitness_scores = np.array([fitness(ind) for ind in population])

            selected = population[best_indices]
            
            # Crossover
            new_population = []

                    population[i] = best_h.copy()
            
            # Update best solution
            for ind in population:
                current_c5 = fitness(ind)
                if current_c5 < best_c5:
                    best_h = ind.copy()
                    best_c5 = current_c5
        
    return best_h, best_c5, n_points
