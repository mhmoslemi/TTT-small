import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
initial_h_values = np.array([0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define n_points and initial guess
    n_points = 100  # Start with 100 points for balance between performance and accuracy
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.resize(h, (n_points,))  # Reshape to n_points samples
    h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the constraint sum(h) == n_points / 2
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.resize(h, (n_points,))
        h = np.clip(h, 0, 1)  # Ensure values are in [0, 1]
        result = correlate(h, 1 - h, mode="full") * dx
        return np.max(result)
    
    # Optimization: Use local search with gradient descent
    def optimize_h(h_initial):
        h = h_initial.copy()
        learning_rate = 0.01
        max_iterations = 1000
        for _ in range(max_iterations):
            # Compute overlap
            c5 = evaluate_overlap(h)
            # Compute gradient numerically
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                c5_plus = evaluate_overlap(h_plus)
                grad[i] = (c5_plus - c5) / 1e-5
            # Update h
            h -= learning_rate * grad
            # Clip values to [0, 1] and normalize
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        return h, c5
    
    # Perform multiple local searches with perturbations
    best_h = h
    best_c5 = evaluate_overlap(h)
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        h_optimized, c5 = optimize_h(h_perturbed)
        if c5 < best_c5:
            best_h = h_optimized
            best_c5 = c5
    
    # Further refinement with convex optimization
    # Define variables
    h_vars = cp.Variable(n_points)
    # Objective: minimize the maximum overlap
    objective = cp.Minimize(cp.Max(cp.multiply(h_vars, 1 - h_vars)))
    # Constraint: sum(h) * dx == 1
    constraints = [cp.sum(h_vars) * dx == 1, 0 <= h_vars, h_vars <= 1]
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.MOSEK, verbose=False)
        if problem.status == 'OPTIMAL':
            h = h_vars.value
            best_c5 = evaluate_overlap(h)
    except:
        pass
    
    # Final check and early termination if budget is exceeded
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Optional: Increase n_points if time remains
    if elapsed < budget_s - 5:
        n_points = 150
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        best_h = h
        best_c5 = evaluate_overlap(h)
        for _ in range(5):
            perturb = np.random.normal(0, 0.05, n_points)
            h_perturbed = h + perturb
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            h_optimized, c5 = optimize_h(h_perturbed)
            if c5 < best_c5:
                best_h = h_optimized
                best_c5 = c5
    
    return (best_h, best_c5, n_points)
