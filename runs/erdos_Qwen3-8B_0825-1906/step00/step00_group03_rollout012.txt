import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    max_points = 1000  # Max points to consider, adjust based on time
    n_points = 100  # Start with n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess
    h_initial = initial_h_values.copy()
    if len(h_initial) != n_points:
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h_extended = np.concatenate([h, np.zeros(n_points)])
        correlation = np.correlate(h, 1 - h_extended, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Function to check constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization with bounds
    def optimize_h(n_points):
        dx = 2.0 / n_points
        h_bounds = [(0, 1) for _ in range(n_points)]
        h_initial = initial_h_values.copy()
        if len(h_initial) != n_points:
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        
        # Set up constraints and variables
        h = np.array(h_initial, dtype=float)
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x), 'jac': lambda x: np.ones_like(x)}]
        
        # Use L-BFGS-B for faster optimization
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=h_bounds,
            constraints=cons,
            jac=None,
            tol=1e-6,
            options={'maxiter': 1000, 'eps': 1e-6}
        )
        
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
            return best_h, best_c5
        else:
            # If optimization fails, try a simpler gradient descent
            h = np.array(h_initial, dtype=float)
            for _ in range(100):
                grad = np.zeros(n_points)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += 1e-5
                    h_minus = h.copy()
                    h_minus[i] -= 1e-5
                    grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * 1e-5)
                
                # Update h using gradient descent
                h -= 0.1 * grad
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum if h_sum != 0 else h
        
            best_c5 = compute_c5(h)
            return h, best_c5
    
    # Try increasing n_points gradually within budget
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for n_points in range(100, max_points + 1, 100):
        if time.time() - start_time > budget_s:
            break
        h, c5 = optimize_h(n_points)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            print(f"n_points = {n_points}, C5 = {c5}")
    
    # Final check with initial h
    h_initial_check, c5_initial = optimize_h(100)
    if c5_initial < best_c5:
        best_h = h_initial_check
        best_c5 = c5_initial
    
    return best_h.tolist(), best_c5, len(best_h)
