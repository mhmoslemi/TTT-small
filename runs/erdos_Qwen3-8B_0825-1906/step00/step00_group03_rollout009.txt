```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points for faster computation
    max_n_points = 200  # Upper limit to prevent excessive computation

    # Start with the initial solution
    h_initial = initial_h_values
    h_initial = np.array(h_initial)
    if len(h_initial) < n_points:
        h_initial = np.concatenate([h_initial, np.zeros(n_points - len(h_initial))])
    elif len(h_initial) > n_points:
        h_initial = h_initial[:n_points]
    h_initial = h_initial / np.sum(h_initial) * n_points / 2
    h_initial = np.clip(h_initial, 0, 1)
    
    # Initial evaluation
    c5_initial = evaluate_erdos_solution(h_initial)
    best_c5 = c5_initial
    best_h = h_initial.copy()

    # Try increasing n_points if time allows
    for n_points in range(100, max_n_points + 1, 20):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate a random initial guess
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)

        # Define the objective function to minimize the maximum overlap
        def objective(x):
            h = x.reshape(n_points)
            dx = 2.0 / n_points
            # Compute the circular convolution with 1 - h
            # We use the "full" mode to get the full convolution
            conv = signal.correlate(h, 1 - h, mode="full")
            # The overlap integral is the convolution multiplied by dx
            c5_val = np.max(conv) * dx
            return c5_val

        # Use a local optimization method to minimize c5
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        c5_val = objective(h_opt)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
        
        print(f"n_points={n_points}, best C5={best_c5:.6f}")

    # Final evaluation with the best solution found
    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_erdos_solution(best_h)
    return best_h, best_c5, len(best_h)
```
