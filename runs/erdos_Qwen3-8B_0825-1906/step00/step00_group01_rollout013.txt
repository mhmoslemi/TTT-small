def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points for balance between detail and speed
    
    # Initial guess: take the initial_h_values, rescale to match the sum constraint
    h_initial = np.array(initial_h_values[:n_points])
    total = np.sum(h_initial)
    h_initial = h_initial * (n_points / 2) / total if total != 0 else np.full(n_points, 0.5)
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))
        return c5

    # Optimization function using scipy.optimize
    def optimize_h():
        nonlocal n_points
        best_h = h_initial.copy()
        best_c5 = evaluate_overlap(best_h)
        
        # Try reducing n_points if the current solution is not improving
        for n in range(50, n_points + 1, 5):
            if time.time() > budget_s - 5:
                break
            h = np.random.uniform(0, 1, n)
            h = h * (n / 2) / np.sum(h) if np.sum(h) != 0 else np.full(n, 0.5)
            c5 = evaluate_overlap(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                n_points = n
        
        # Use a more refined optimization on the best found solution
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) != 0 else np.full(n_points, 0.5)
        best_c5 = evaluate_overlap(h)
        return h, best_c5

    # Run the optimization
    start_time = time.time()
    best_h, best_c5 = optimize_h()

    return best_h, best_c5, n_points
