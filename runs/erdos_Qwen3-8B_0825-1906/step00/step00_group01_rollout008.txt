import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shift = np.array([1 - h[i] for i in range(n_points)])
    corr = np.correlate(h, shift, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # Initial construction with 47 points
    return np.load('initial_h.npy')  # Placeholder for actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Try different numbers of points
    n_points_list = [50, 60, 70, 80, 90, 100]
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Define the grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial guess from the preloaded initial_h_values
        if n_points == len(initial_h_values()):
            h_initial = initial_h_values()
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Optimization variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Objective function
        # We want to minimize the maximum overlap
        # So we minimize the maximum of the correlation values
        corr = []
        for k in range(1, n_points):
            shifted = np.roll(h.value, k) if h.value is not None else np.roll(h, k)
            correlation = np.dot(h.value, 1 - shifted)
            corr.append(correlation)
        
        # Use a convex relaxation approach
        # Minimize the maximum of the correlations
        # Use a convex objective with a max constraint
        obj = cp.Minimize(cp.Max(corr))
        
        # Solve the problem
        prob = cp.Problem(obj, constraints)
        try:
            result = prob.solve(solver=cp.ECOS, verbose=False)
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = h.value
        except:
            pass
        
        # Evaluate the solution
        h_values = h.value
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values
    
    return best_h, best_c5, len(best_h)
