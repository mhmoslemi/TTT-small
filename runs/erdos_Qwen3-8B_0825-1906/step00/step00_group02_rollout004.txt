import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define initial guess
    n_points = 45
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        result = np.correlate(h_extended, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5
    
    # Function to evaluate the objective and constraints
    def objective_and_constraints(h):
        # Objective: minimize the maximum overlap integral
        c5 = compute_c5(h)
        return c5, []
    
    # Function to update h based on gradient descent
    def gradient_descent_step(h, learning_rate=0.01, num_steps=100):
        h = h.copy()
        for _ in range(num_steps):
            c5, _ = objective_and_constraints(h)
            # Compute gradient with finite differences
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                c5_plus, _ = objective_and_constraints(h_plus)
                grad[i] = (c5_plus - c5) / 1e-5
            # Update h
            h -= learning_rate * grad
            # Clip h to [0, 1]
            h = np.clip(h, 0, 1)
        return h
    
    # Function to optimize h using gradient descent
    def optimize_h(h_start, max_iterations=1000, learning_rate=0.01):
        h = h_start.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        for _ in range(max_iterations):
            h = gradient_descent_step(h, learning_rate=learning_rate)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Early stopping if no improvement
            if np.abs(current_c5 - best_c5) < 1e-6:
                break
        return best_h, best_c5
    
    # Perform gradient descent optimization
    start_time = time.time()
    h_optimized, c5_bound = optimize_h(h_initial, max_iterations=500, learning_rate=0.01)
    
    # Ensure the integral constraint is satisfied
    dx = 2.0 / len(h_optimized)
    integral = np.sum(h_optimized) * dx
    if np.abs(integral - 1.0) > 1e-6:
        # If the integral constraint is not met, adjust h
        h_optimized = h_optimized / integral * 1.0
    
    # Evaluate the best solution
    c5_bound = compute_c5(h_optimized)
    
    # Check if we have exceeded the budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return h_optimized, c5_bound, len(h_optimized)
    
    # Final check and return
    return h_optimized, c5_bound, len(h_optimized)
