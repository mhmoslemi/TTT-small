```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assuming the initial h is loaded here

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points; use initial h to determine size
    n_points_initial = len(initial_h_values)
    n_points = n_points_initial  # Start with same number of points as initial
    
    # Objective: minimize the maximum overlap integral
    # Constraints: sum(h) = n_points / 2 and 0 <= h[i] <= 1

    # Attempt to improve with more points
    for n_points in [n_points_initial, n_points_initial * 2, n_points_initial * 3]:
        if n_points > 2000:
            continue
        h = np.random.rand(n_points) * 2 - 1
        h = (h + 1) / 2  # Initial guess between 0 and 1
        h = h * (n_points / 2) / np.sum(h)  # Normalize to meet constraint

        # Define variables
        x = cp.Variable(n_points)

        # Objective: minimize the maximum overlap integral
        # Compute correlation with 1 - h for all shifts
        # We use a reduced form since full correlation is expensive
        # Only evaluate for a few relevant shifts (e.g., up to 100)
        shifts = np.arange(0, 100)
        overlaps = []
        for shift in shifts:
            if shift >= n_points:
                continue
            overlap = cp.sum(x * (1 - x[shift:]))
            overlaps.append(overlap)

        # Maximize the overlaps
        obj = cp.Minimize(cp.Max(overlaps))

        # Constraints: sum(x) == n_points / 2 and 0 <= x[i] <= 1
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]

        # Solve
        prob = cp.Problem(obj, constraints)
        try:
            result = prob.solve(solver=cp.SCIP, verbose=False, time_limit=budget_s - 1)
            if result is not None:
                h_values = x.value
                c5_bound = evaluate_erdos_solution(h_values)
                return (h_values, c5_bound, n_points)
        except:
            pass

    # If no improvement found, return initial h
    h_values = initial_h_values
    c5_bound = evaluate_erdos_solution(h_values)
    return (h_values, c5_bound, len(h_values))
```
