import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    # Define n_points
    n_points = 100  # Start with 100 points for balance between speed and accuracy

    # Initialize h_values with the initial construction
    h_values = np.copy(initial_h_values[:n_points])
    h_values = h_values / np.sum(h_values) * (n_points / 2)

    def objective(h):
        # Objective: minimize the maximum overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    def constraint_total_area(h):
        # Constraint: total area under h must be 1
        return np.sum(h) * (2.0 / n_points) - 1

    # Optimization: use Nelder-Mead with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start with the initial guess
    h_values = h_values.astype(float)

    start_time = time.time()
    result = optimize.minimize(
        fun=objective,
        x0=h_values,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'disp': False}
    )
    
    # Check if we have time left to refine
    if time.time() - start_time < budget_s - 5:
        # Try a small perturbation to avoid local optima
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = result.x + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        result = optimize.minimize(
            fun=objective,
            x0=perturbed_h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 50, 'disp': False}
        )
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(result.x)
    return (result.x, final_c5, n_points)
