import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import os
import random
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=np.float64)
    
    # Define parameters
    max_n_points = 1000  # Maximum number of points allowed
    min_n_points = 100   # Minimum number of points to use (for efficiency)
    dx = 2.0 / min_n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h = np.pad(h, (0, 1), mode='constant')  # Pad for full convolution
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5
    
    # Function to evaluate a candidate h
    def evaluate_candidate(h):
        if len(h) < min_n_points:
            return float('inf')
        total = np.sum(h) * dx
        if np.abs(total - 1.0) > 1e-6:
            return float('inf')
        return compute_c5(h)
    
    # Simulated annealing algorithm with adaptive search
    def simulated_annealing(start_h, budget):
        h = start_h.copy()
        best_h = h.copy()
        best_c5 = evaluate_candidate(h)
        temperature = 100.0
        cooling_rate = 0.995
        iter_count = 0
        max_iter = int(budget * 0.7)
        while temperature > 1e-5 and iter_count < max_iter:
            # Random perturbation
            perturbation = np.random.rand(len(h)) * 2 - 1
            perturbation *= 0.1 * temperature
            h_perturbed = np.clip(h + perturbation, 0.0, 1.0)
            
            # Evaluate
            c5_perturbed = evaluate_candidate(h_perturbed)
            
            # Acceptance probability
            if c5_perturbed < best_c5:
                h = h_perturbed.copy()
                best_h = h.copy()
                best_c5 = c5_perturbed
            else:
                delta = c5_perturbed - best_c5
                prob = np.exp(-delta / temperature)
                if np.random.rand() < prob:
                    h = h_perturbed.copy()
            
            temperature *= cooling_rate
            iter_count += 1
        
        return best_h, best_c5
    
    # Function to generate random candidate h
    def generate_random_candidate(n_points):
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        norm = np.sum(h) * 2.0 / n_points
        h = h / norm
        return h
    
    # Function to create a more intelligent starting point
    def generate_intelligent_candidate(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = np.clip(h, 0.0, 1.0)
        norm = np.sum(h) * 2.0 / n_points
        h = h / norm
        return h
    
    # Start with initial candidate
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = evaluate_candidate(h)
    best_n_points = len(h)
    
    # Try with different n_points
    for n_points in range(min_n_points, max_n_points + 1):
        if n_points == len(h):
            continue
        # Generate new candidate
        if n_points < len(h):
            h = h[:n_points]
        else:
            h = np.concatenate([h, np.zeros(n_points - len(h))])
        h = generate_intelligent_candidate(n_points)
        current_c5 = evaluate_candidate(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            best_n_points = n_points
            print(f"Found better solution with n_points={n_points}, C5={best_c5}")
    
    # Try simulated annealing on best candidate
    start_time = time.time()
    h_sa, c5_sa = simulated_annealing(best_h, budget_s - (time.time() - start_time))
    best_c5 = min(best_c5, c5_sa)
    best_h = h_sa.copy()
    
    # Try generating random candidates
    for _ in range(5):
        h_rand = generate_random_candidate(best_n_points)
        current_c5 = evaluate_candidate(h_rand)
        if current_c5 < best_c5:
            best_h = h_rand.copy()
            best_c5 = current_c5
    
    # Final checks
    if len(best_h) < min_n_points:
        best_h = np.concatenate([best_h, np.zeros(min_n_points - len(best_h))])
    best_c5 = evaluate_candidate(best_h)
    
    return best_h, best_c5, len(best_h)
