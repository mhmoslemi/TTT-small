import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=60, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h) * dx
    # Adjust to meet the constraint sum(h) * dx = 1
    if sum_h != 1:
        scaling = 1.0 / sum_h
        initial_h *= scaling
    
    # Function to compute C5 for a given h
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        dx = 2.0 / len(h)
        # Convolve h with (1 - h)
        convolution = np.correlate(h, 1 - h, mode="full")
        # The overlapping integral is the max of the convolution multiplied by dx
        c5 = np.max(convolution) * dx
        return c5
    
    # Function to evaluate and return the best solution found
    def optimize_h(initial_guess, max_iter=100, tol=1e-5, n_points=n_points):
        # Objective function for optimization: minimize C5
        def objective(h):
            return compute_c5(h)
        
        # Constraints: sum(h) * dx = 1
        def constraint(h):
            return np.sum(h) * dx - 1
        
        # Optimization
        result = opt.minimize(
            fun=objective,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            tol=tol,
            maxiter=max_iter
        )
        
        if result.success:
            best_h = result.x
        else:
            best_h = initial_guess
        
        return best_h, compute_c5(best_h)
    
    # Initial optimization
    best_h, best_c5 = optimize_h(initial_h)
    
    # Try some random perturbations to escape local optimum
    for _ in range(5):
        perturbed_h = best_h.copy()

        indices = np.random.choice(n_points, size=min(10, n_points), replace=False)
        perturbed_h[indices] += np.random.uniform(-0.1, 0.1, size=len(indices))
        perturbed_h = np.clip(perturbed_h, 0, 1)
        # Ensure the constraint is satisfied
        scaling = 1.0 / (np.sum(perturbed_h) * dx)
        perturbed_h *= scaling
        # Evaluate
        current_c5 = compute_c5(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5

    best_h, best_c5 = optimize_h(best_h)
