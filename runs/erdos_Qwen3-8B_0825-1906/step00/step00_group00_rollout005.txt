import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=60, **kwargs):
    # Initial setup
    np.random.seed(seed)
    n_points = 100  # Use 100 points for efficiency and reasonable accuracy
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # If initial_h_values is provided, use it; otherwise, start with a uniform distribution
    if "initial_h_values" in globals():
        initial_h_values = np.array(initial_h_values)
        if len(initial_h_values) != n_points:
            initial_h_values = np.linspace(0, 1, n_points)
    
    # Set up the initial h_values
    h = initial_h_values.copy()
    h = np.clip(h, 0.0, 1.0)
    h = h / (np.sum(h) * dx) * (n_points / 2)
    
    # Define the function to compute the overlap integral for a given h
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0.0, 1.0)
        h = h / (np.sum(h) * dx) * (n_points / 2)
        h = np.concatenate([h, np.zeros(n_points)])
        h = h[:n_points + n_points]
        h = h[:2 * n_points]
        h = h[:n_points + n_points]
        h = h[:2 * n_points]
        h = h[:n_points + n_points]
        h = h[:n_points + n_points]
        
        h = np.clip(h, 0.0, 1.0)
        h = h / (np.sum(h) * dx) * (n_points / 2)
        h = h[:2 * n_points]
        
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr * dx
        c5 = np.max(corr)
        return c5
    
    # Define the objective function for optimization
    def objective(x):
        return compute_c5(x)
    
    # Define the constraints
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Set up the bounds and constraints
    bounds = Bounds(0.0, 1.0)
    cons = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    
    # Track time and return best solution
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        print("Time budget exceeded")
        return h, compute_c5(h), n_points
    
    best_solution = result.x
    best_c5 = compute_c5(best_solution)
    
    return best_solution, best_c5, n_points
