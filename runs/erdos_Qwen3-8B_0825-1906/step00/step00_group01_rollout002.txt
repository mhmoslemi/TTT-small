```python
import numpy as np
import scipy.optimize as optimize
import time
import copy
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load("initial_h_values.npy")  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Start with the initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5

    # Try reducing the number of points by half
    n_points_candidate = max(20, n_points // 2)
    print(f"Trying n_points = {n_points_candidate}")
    h_candidate = np.zeros(n_points_candidate)
    h_candidate[:n_points_candidate // 2] = 1.0
    h_candidate[n_points_candidate // 2:] = 0.0
    h_candidate = (h_candidate * (n_points_candidate / 2)) / (np.sum(h_candidate))
    h_candidate = np.round(h_candidate, 4)
    c5_candidate = evaluate_erdos_solution(h_candidate)
    if c5_candidate < best_c5:
        best_h = h_candidate.copy()
        best_c5 = c5_candidate
        n_points = n_points_candidate

    # Try a random construction
    print("Trying random construction")
    h_random = np.random.uniform(0, 1, n_points)
    h_random = (h_random * (n_points / 2)) / (np.sum(h_random))
    h_random = np.round(h_random, 4)
    c5_random = evaluate_erdos_solution(h_random)
    if c5_random < best_c5:
        best_h = h_random.copy()
        best_c5 = c5_random

    # Try a grid-based construction
    print("Trying grid-based construction")
    h_grid = np.zeros(n_points)
    h_grid[::2] = 1.0
    h_grid[1::2] = 0.0
    h_grid = (h_grid * (n_points / 2)) / (np.sum(h_grid))
    h_grid = np.round(h_grid, 4)
    c5_grid = evaluate_erdos_solution(h_grid)
    if c5_grid < best_c5:
        best_h = h_grid.copy()
        best_c5 = c5_grid

    # Define optimization problem
    def optimize_h(n_points, h_initial):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        # Objective: minimize the maximum overlap
        overlap = cp.Maximize(cp.multiply(h, 1 - h[1:] - h[0]))
        prob = cp.Problem(overlap, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == 'optimal':
                return h.value
        except:
            pass
        return h_initial

    # Try optimization
    print("Trying optimization")
    h_opt = optimize_h(n_points, best_h)
    c5_opt = evaluate_erdos_solution(h_opt)
    if c5_opt < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5_opt

    # Try refinement
    print("Trying refinement")
    h_refined = copy.deepcopy(best_h)
    for _ in range(5):
        for i in range(n_points):
            h_refined[i] = np.clip(h_refined[i], 0, 1)
        c5_refined = evaluate_erdos_solution(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined

    # Final check
    print(f"Best C5: {best_c5}")
    return best_h, best_c5, n_points
```
