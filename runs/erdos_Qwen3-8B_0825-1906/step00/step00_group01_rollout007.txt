```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions and initial solution
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 47  # Starting with the initial solution
    max_n_points = 100  # Try up to 100 points for better resolution
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = None
    
    # Starting with the initial solution
    h_initial = np.array(initial_h_values) if n_points == len(initial_h_values) else np.random.rand(n_points)
    h_initial = h_initial / (n_points / 2)  # Normalize to satisfy the integral constraint
    
    # Start with initial solution
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try increasing the number of points if time allows
    for n_points in range(50, max_n_points + 1):
        # Compute dx
        dx = 2.0 / n_points
        # Define x values
        x = np.linspace(0, 2, n_points)
        
        # Define variables
        h = cp.Variable(n_points, nonneg=True, lb=0, ub=1)
        objective = cp.Minimize(0)  # We will minimize the maximum overlap
        
        # Define the overlap integral for each shift k
        # We use a sliding window approach to compute the overlap for all k
        max_overlap = cp.Constant(0)
        for k in range(1, n_points):
            # Shift h by k points
            h_shifted = np.roll(h, k)
            # Compute the overlap integral
            overlap = cp.sum(h * (1 - h_shifted)) * dx
            max_overlap = cp.max(max_overlap, overlap)
        
        # Formulate the problem
        problem = cp.Problem(objective, [max_overlap <= 0.380876, cp.sum(h) == n_points / 2])
        
        try:
            start_time = time.time()
            problem.solve(solver=cp.ECOS, eps=tolerance)
            elapsed_time = time.time() - start_time
            
            if elapsed_time > budget_s:
                break
            
            if problem.status == 'OPTIMAL':
                h_current = np.round(h.value, 5)
                c5_current = evaluate_erdos_solution(h_current)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h_current
        except:
            pass
        
        if elapsed_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
```
