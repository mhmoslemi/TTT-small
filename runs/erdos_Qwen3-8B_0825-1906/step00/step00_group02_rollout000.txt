    # Evaluate the initial solution
    def evaluate_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        h_extended = np.concatenate([h, np.zeros_like(h)])
        correlation = np.correlate(h_extended, 1 - h, mode='full')
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    c5_initial = evaluate_c5(initial_h_values)
    print(f"Initial C5: {c5_initial:.6f}")
    
    # Define a function to generate a random perturbation of h
    def perturb_h(h, perturbation_strength=0.1):
        h_perturbed = h.copy()
        indices = np.random.choice(len(h_perturbed), int(len(h_perturbed) * 0.1), replace=False)
        h_perturbed[indices] += np.random.uniform(-perturbation_strength, perturbation_strength, len(indices))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Adjust to maintain the integral constraint
        total = np.sum(h_perturbed)
        scale = (n_points_initial / 2) / total
        h_perturbed *= scale
        return h_perturbed
    
    # Define a function to evaluate a candidate h
    def objective(h):
        return evaluate_c5(h)
    
    # Optimization parameters
    n_points = n_points_initial
    max_iterations = 100
    perturbation_strength = 0.1
    best_h = initial_h_values.copy()
    best_c5 = c5_initial
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Perturb the current best solution
        h_candidate = perturb_h(best_h, perturbation_strength)
        c5_candidate = objective(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    if time.time() - start_time < budget_s - 5:
