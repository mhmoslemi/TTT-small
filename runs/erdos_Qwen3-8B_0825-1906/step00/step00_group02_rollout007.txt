```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 45
    h = initial_h_values.copy()
    
    # Define the overlap integral function
    def evaluate_overlap(h):
        dx = 2.0 / n_points
        h_shifted = np.zeros((n_points, n_points))
        for i in range(n_points):
            h_shifted[i, :] = h[(i + 1) % n_points:]
        overlap = np.sum(np.multiply(h, 1 - h_shifted), axis=1)
        return np.max(overlap) * dx
    
    # Define the objective function for optimization
    def objective(h):
        return evaluate_overlap(h)
    
    # Define constraints
    def constraints(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    opt_h = cp.Variable(n_points)
    objective_func = cp.Minimize(objective(opt_h))
    constraints_func = [cp.sum(opt_h) == n_points / 2, 0 <= opt_h, opt_h <= 1]
    
    # Solve the optimization problem
    start_time = time.time()
    prob = cp.Problem(objective_func, constraints_func)
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iter=1000)
    except:
        prob.solve(solver=cp.SCS, eps=1e-3, max_iter=500)
    
    if prob.status == 'optimal':
        h_opt = np.round(opt_h.value, 5)
    else:
        h_opt = h
    
    # Evaluate the overlap
    c5_bound = evaluate_overlap(h_opt)
    
    # Check if we can improve with a larger number of points
    n_points = min(100, int(budget_s * 10))  # Limit to 100 points for efficiency
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    
    # Optimize again with the new number of points
    opt_h = cp.Variable(n_points)
    objective_func = cp.Minimize(evaluate_overlap(opt_h))
    constraints_func = [cp.sum(opt_h) == n_points / 2, 0 <= opt_h, opt_h <= 1]
    
    start_time = time.time()
    prob = cp.Problem(objective_func, constraints_func)
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iter=1000)
    except:
        prob.solve(solver=cp.SCS, eps=1e-3, max_iter=500)
    
    if prob.status == 'optimal':
        h_opt = np.round(opt_h.value, 5)
    else:
        h_opt = h
    
    c5_bound = evaluate_overlap(h_opt)
    
    # Ensure we stay within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return h_opt, c5_bound, n_points
    
    return h_opt, c5_bound, n_points
```
