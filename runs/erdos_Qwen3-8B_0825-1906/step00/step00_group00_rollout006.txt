import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 128
    # Adjust n_points for speed and accuracy
    n_points = min(512, max(64, n_points))
    h = initial_h_values.copy()
    h = h[:n_points]
    # Ensure the sum is correct
    h = h * (n_points / 2) / np.sum(h)
    
    # Precompute the kernel
    n = len(h)
    kernel = 1 - h
    max_overlap = 0.0
    
    # Optimization: minimize the maximum overlap
    # We will use a non-convex optimization approach with scipy
    def objective(x):
        x = x.reshape(n)
        x = np.clip(x, 0, 1)
        overlap = np.correlate(x, kernel, mode="full")
        return np.max(overlap) * (2.0 / n)
    
    def constraint(x):
        x = x.reshape(n)
        return np.sum(x) - n / 2
    
    # Initial guess
    x0 = h.copy()
    
    # Optimization with bounds and constraints
    bounds = [(0, 1) for _ in range(n)]
    res = optimize.minimize(
        objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint}],
        options={"ftol": 1e-8, "maxiter": 100}
    )
    
    h = res.x
    h = np.clip(h, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    # Evaluate the result
    c5_bound = evaluate_erdos_solution(h)
    
    # Final evaluation
    if time.time() - start_time > budget_s - 1:
        return (h, c5_bound, n_points)
    
    # Try a few more iterations for refinement
    for _ in range(3):
        res = optimize.minimize(
            objective,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint}],
            options={"ftol": 1e-8, "maxiter": 50}
        )
        h = res.x
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
        c5_bound = evaluate_erdos_solution(h)
        
        if time.time() - start_time > budget_s - 1:
            break
    
    return (h, c5_bound, n_points)
