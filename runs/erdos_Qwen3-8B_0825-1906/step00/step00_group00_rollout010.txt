import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    n_initial = len(initial_h_values)
    h_initial = initial_h_values
    
    # Try different n_points to find the best balance
    n_points_list = [100, 200, 300, 400, 500, 600]
    
    best_c5 = float('inf')
    best_h = None
    best_n = n_initial
    
    for n_points in n_points_list:
        if time.time() > time.time() + budget_s:
            break
        
        dx = 2.0 / n_points
        
        # Create a grid of x values
        x = np.linspace(0, 2, n_points)
        
        # Define the initial guess from the initial construction, interpolated
        h_initial_interp = np.interp(x, np.linspace(0, 2, n_initial), h_initial)
        h_start = h_initial_interp.copy()
        
        # Define variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,  # ∫h dx = 1
            0 <= h,
            h <= 1
        ]
        
        # Objective: minimize the maximum overlap integral
        # We approximate the max over k by evaluating at discrete k values
        max_overlap = 0.0
        for k in range(1, n_points - 1):
            h_shift = np.roll(h.value, k) if h.value is not None else np.roll(h, k)
            overlap = np.sum(h.value * (1 - h_shift)) * dx
            max_overlap = max(max_overlap, overlap)
        
        def objective_func(h_val):
            max_overlap = 0.0
            for k in range(1, n_points - 1):
                h_shift = np.roll(h_val, k)
                overlap = np.sum(h_val * (1 - h_shift)) * dx
                max_overlap = max(max_overlap, overlap)
            return max_overlap
        
        # Optimization
        def optimize_h(h_start):
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2,
                0 <= h,
                h <= 1
            ]
            prob = cp.Problem(cp.Minimize(cp.max(objective_func(h))), constraints)
            try:
                prob.solve(solver=cp.SCS, max_iter=1000)
                if prob.status == 'optimal':
                    return h.value
            except:
                pass
            return h_start
        
        h_opt = optimize_h(h_start)
        
        # Evaluate the best solution
        max_overlap = 0.0
        for k in range(1, n_points - 1):
            h_shift = np.roll(h_opt, k)
            overlap = np.sum(h_opt * (1 - h_shift)) * dx
            max_overlap = max(max_overlap, overlap)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h_opt
            best_n = n_points
    
    return best_h, best_c5, best_n
