import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial H values
    n_points_initial = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points_initial
    sum_h = np.sum(h) * dx
    assert np.isclose(sum_h, 1.0), f"Initial sum is {sum_h}, should be 1.0"
    
    # Define the function to compute C5
    def compute_c5(h_values):
        h = np.array(h_values, dtype=np.float64)
        dx = 2.0 / len(h)
        shifted = np.zeros(len(h) * 2 - 1)
        for k in range(1, len(h) - 1):
            shifted[k:] += h[:len(h) - k] * (1 - h[k:])
        shifted = np.concatenate([h * (1 - h[1:]), shifted, h[:-1] * (1 - h)])
        overlap = np.max(shifted) * dx
        return overlap
    
    # Define the objective function for optimization
    def objective(x):
        h = x
        dx = 2.0 / len(h)
        sum_h = np.sum(h) * dx
        if not np.isclose(sum_h, 1.0):
            return float('inf')
        c5 = compute_c5(h)
        return c5
    
    # Use L-BFGS-B for optimization
    def optimize_h(h_initial, max_iter=100, tol=1e-6):
        h = h_initial.copy()
        dx = 2.0 / len(h)
        sum_h = np.sum(h) * dx
        if not np.isclose(sum_h, 1.0):
            h = h_initial.copy()
            h[0] = h[0] + (1.0 / len(h) - sum_h / dx)
        for _ in range(max_iter):
            # Compute the objective and gradient
            obj_val = objective(h)
            # Compute gradient (approximate using finite differences)
            grad = np.zeros_like(h)
            eps = 1e-6
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                obj_plus = objective(h_plus)
                h_minus = h.copy()
                h_minus[i] -= eps
                obj_minus = objective(h_minus)
                grad[i] = (obj_plus - obj_minus) / (2 * eps)
            # Perform gradient descent step
            step_size = 0.1
            h_new = h - step_size * grad
            # Enforce bounds
            h_new = np.clip(h_new, 0.0, 1.0)
            # Enforce sum constraint
            sum_h_new = np.sum(h_new) * dx
            if not np.isclose(sum_h_new, 1.0):
                h_new = h_new.copy()
                h_new[0] = h_new[0] + (1.0 / len(h_new) - sum_h_new / dx)
            # Check for convergence
            if np.abs(obj_val - objective(h_new)) < tol:
                break
            h = h_new
        return h
    
    # Try optimizing with different starting points
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Simple random perturbation approach
    h_perturbed = h.copy()
    h_perturbed += np.random.randn(len(h)) * 0.05
    h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
    sum_h_perturbed = np.sum(h_perturbed) * 2.0 / len(h_perturbed)
    delta = 1.0 - sum_h_perturbed
    h_perturbed[0] += delta
    h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
    h_perturbed = optimize_h(h_perturbed)
    c5_perturbed = compute_c5(h_perturbed)
    if c5_perturbed < best_c5:
        best_h = h_perturbed
        best_c5 = c5_perturbed
    
    # Try a few more iterations of optimization
    for _ in range(3):
        h_optimized = optimize_h(best_h)
        c5_optimized = compute_c5(h_optimized)
        if c5_optimized < best_c5:
            best_h = h_optimized
            best_c5 = c5_optimized
    
    # Return the best result
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h.tolist(), best_c5, len(best_h))
    
    # Try a few more random starting points
    for _ in range(2):
        h_start = np.random.rand(len(best_h))
        h_start = h_start / np.sum(h_start) * 1.0
        h_start = optimize_h(h_start)
        c5_start = compute_c5(h_start)
        if c5_start < best_c5:
            best_h = h_start
            best_c5 = c5_start
    
    # Final check
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h.tolist(), best_c5, len(best_h))
    
    return (best_h.tolist(), best_c5, len(best_h))
