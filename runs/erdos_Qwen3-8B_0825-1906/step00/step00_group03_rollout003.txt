import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h = np.pad(h, (0, 1), 'constant')  # Pad to make full convolution work
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Function to evaluate and store the best solution
    def evaluate_and_update(h):
        nonlocal h_values, c5_bound
        current_c5 = compute_c5(h)
        if current_c5 < c5_bound:
            h_values = h.copy()
            c5_bound = current_c5
    
    # Initial evaluation
    c5_bound = compute_c5(h_values)
    
    # Optimization: Use a local search with simulated annealing
    def objective(h):
        return compute_c5(h)
    
    # Simulated Annealing
    def simulated_annealing(h_initial, temp=100.0, cooling_rate=0.995, iterations=1000):
        h = h_initial.copy()
        current_energy = objective(h)
        best_h = h.copy()
        best_energy = current_energy
        
        for i in range(iterations):
            # Perturb the solution
            idx = np.random.randint(0, len(h))
            h[idx] += np.random.uniform(-0.1, 0.1)
            h = np.clip(h, 0, 1)
            
            # Ensure the integral constraint is satisfied
            if np.abs(np.sum(h) - len(h) / 2) > 1e-6:
                h = h_initial.copy()
                continue
            
            new_energy = objective(h)
            
            # Acceptance probability
            if new_energy < current_energy or np.random.rand() < np.exp((current_energy - new_energy) / temp):
                current_energy = new_energy
                h = h.copy()
            
            if new_energy < best_energy:
                best_energy = new_energy
                best_h = h.copy()
            
            temp *= cooling_rate
        
        return best_h, best_energy
    
    # Run simulated annealing
    best_h, _ = simulated_annealing(h_values)
    
    # Evaluate and update the best solution
    evaluate_and_update(best_h)
    
    # Try optimizing with CVXOPT
    def cvxopt_optimize():
        # Convert h_values to a variable
        n = len(h_values)
        h = cp.Variable(n)
        constraints = [
            cp.sum(h) == n / 2,
            0 <= h,
            h <= 1
        ]
        # Objective: minimize the maximum overlap
        # This is a bit tricky to implement directly, so we'll use a heuristic approach
        # instead of trying to minimize the max directly
        # Here, we just improve using local search and accept the best
        return best_h
    
    # Run CVXOPT optimization
    best_h = cvxopt_optimize()
    
    # Final evaluation
    evaluate_and_update(best_h)
    
    # Return the best solution
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h_values, c5_bound, n_points)
    
    # Try a few more local optimizations
    for _ in range(5):
        best_h, _ = simulated_annealing(best_h)
        evaluate_and_update(best_h)
    
    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < c5_bound:
        h_values = best_h.copy()
        c5_bound = final_c5
    
    return (h_values, c5_bound, n_points)
