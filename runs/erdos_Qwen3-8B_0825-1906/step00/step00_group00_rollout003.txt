```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h)):
            shifted[k:] = h[:-k]
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate the objective
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        return compute_c5(h)
    
    # Function to compute the constraints
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start from initial solution
    x0 = h_initial.copy()
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    # Check if we are under time budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (x0, compute_c5(x0), n_points)
    
    # Try some local perturbations
    for _ in range(5):
        # Perturb a few points
        idxs = np.random.choice(n_points, size=5, replace=False)
        for idx in idxs:
            x0[idx] += np.random.uniform(-0.1, 0.1)
            x0[idx] = np.clip(x0[idx], 0, 1)
        
        # Evaluate
        current_c5 = compute_c5(x0)
        if current_c5 < result.fun:
            result = optimize.minimize(
                objective,
                x0,
                bounds=bounds,
                method='L-BFGS-B',
                jac=None,
                options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
            )
        
        # Check time
        if time.time() - start_time >= budget_s:
            break
    
    # Try a convex optimization approach
    h = cp.Variable(n_points)
    objective = cp.Maximize(cp.sum(cp.multiply(h, 1 - h)))
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve()
        h_result = np.round(h.value, 6)
        c5_result = compute_c5(h_result)
        if c5_result < result.fun:
            result = optimize.minimize(
                objective,
                h_result,
                bounds=bounds,
                method='L-BFGS-B',
                jac=None,
                options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
            )
    except:
        pass
    
    # Final check
    final_h = np.clip(result.x, 0, 1)
    best_c5 = compute_c5(final_h)
    return (final_h, best_c5, n_points)
```
