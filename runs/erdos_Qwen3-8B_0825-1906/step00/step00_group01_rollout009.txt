```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as sla
import time
import cvxpy as cp
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        nonlocal best_c5
        if len(h) != n_points:
            h = np.resize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Start with the initial solution
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h *= (n_points / 2) / np.sum(h)  # Ensure the integral is 1
    c5 = evaluate_overlap(h)

    # Gradient descent optimization
    learning_rate = 0.01
    max_iterations = 1000
    for _ in range(max_iterations):
        h = np.clip(h + learning_rate * np.random.randn(n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure the integral is 1
        c5 = evaluate_overlap(h)
        if time.time() - start_time > budget_s:
            break

    # Simple genetic algorithm
    population_size = 20
    generations = 100
    for _ in range(generations):
        population = np.random.rand(population_size, n_points)
        population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
        fitness = np.array([evaluate_overlap(p) for p in population])
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        if time.time() - start_time > budget_s:
            break

    # Use CVXPY for optimization (slower but more precise)
    if time.time() - start_time < budget_s * 0.5:
        x = cp.Variable(n_points)
        constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
        obj = cp.Maximize(cp.max(cp.correlate(x, 1 - x, mode="full")) * (2.0 / n_points))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4)
            h = x.value
        except:
            h = best_h

    # Try increasing the number of points if time permits
    if time.time() - start_time < budget_s * 0.8:
        n_points = 200
        h = np.resize(h, n_points)
        h = h * (n_points / 2) / np.sum(h)
        learning_rate = 0.005
        for _ in range(500):
            h = np.clip(h + learning_rate * np.random.randn(n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_overlap(h)
            if time.time() - start_time > budget_s:
                break

    # Final evaluation
    h = np.resize(h, n_points)
    h = h * (n_points / 2) / np.sum(h)
    c5 = evaluate_overlap(h)

    return h, c5, n_points
```
