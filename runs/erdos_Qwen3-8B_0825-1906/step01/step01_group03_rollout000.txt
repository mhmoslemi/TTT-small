```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = 70
    h = np.array(initial_h_values)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Compute C5
        c5 = np.max(corr) * dx
        return c5
    
    # Define the constraints
    def constraints(h):
        return np.sum(h) - n_points / 2
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    res = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraints, 'jac': lambda h: np.zeros(len(h)) + 1},
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (h, 0.38092, n_points)
    
    # If the optimization wasn't successful, fallback to initial solution
    if not res.success:
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    else:
        best_h = res.x
        best_c5 = objective(best_h)
    
    # Try increasing the number of points if time allows
    if elapsed < budget_s - 5:
        n_points_new = min(150, n_points + 20)
        h_new = np.zeros(n_points_new)
        # Initialize with a uniform distribution
        h_new = np.random.uniform(0, 1, n_points_new)
        # Enforce the integral constraint
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)
        # Evaluate
        c5_new = objective(h_new)
        
        # If the new solution is better, keep it
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Final check for time
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
```
