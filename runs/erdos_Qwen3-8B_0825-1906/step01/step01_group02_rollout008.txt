    start_time = time.time()
    best_c5 = float('inf')

    max_evals = 100  # Limit evaluations to keep within budget

    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2)  # Ensure integral is 1

    # Helper function to compute C5 for a given h
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        h_extended = np.zeros(2 * n_points)
        h_extended[:n_points] = h
        h_extended[n_points:] = h
        overlap = correlate(h_extended, 1 - h, mode='full')
        overlap = overlap[n_points - 1:2 * n_points - 1]  # Extract relevant part
        c5 = np.max(overlap * dx)
        return c5

    def evaluate_and_update(h):
        nonlocal best_c5, best_h
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Initial evaluation
    current_h = initial_h.copy()
    current_c5 = evaluate_and_update(current_h)

    # Optimization using a combination of gradient descent and random sampling
    for _ in range(max_evals):
        # Perturb current_h
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_h = current_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2)  # Ensure integral is 1

        # Evaluate new_h
        new_c5 = evaluate_and_update(new_h)
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5

        # Check time budget
        if time.time() - start_time > budget_s - 1:
