```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start from initial_h_values
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # Try a few different starting points for diversity
    start_points = [h_initial]
    
    # Try small n_points first for speed
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Function to compute C5
        def compute_c5(h):
            h = np.clip(h, 0, 1)
            h = np.concatenate((h, np.zeros(n_points)))  # Pad for convolution
            conv = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(conv * dx)
            return c5
        
        # Optimization problem: minimize C5
        # Use a heuristic search with simulated annealing
        def objective(h):
            return compute_c5(h)
        
        # Use the initial solution as a starting point
        h_start = np.repeat(h_initial, n_points // n_points_initial)
        h_start = np.clip(h_start, 0, 1)
        h_start = np.pad(h_start, (0, n_points - len(h_start)), 'constant')
        
        # Run simulated annealing
        result = opt.optimize(
            objective,
            h_start,
            method='anneal',
            options={'maxiter': 1000, 'temperature': 1.0, 'T0': 10.0, 'Tmin': 0.1}
        )
        
        # Extract best solution
        h_best = result.x
        c5_bound = compute_c5(h_best)
        
        # Check if this is the best solution so far
        if c5_bound < 0.380876:
            # Return the best solution
            return (h_best, c5_bound, n_points)
    
    # If no better solution found, return the initial solution
    return (h_initial, evaluate_erdos_solution(h_initial), len(h_initial))
```
