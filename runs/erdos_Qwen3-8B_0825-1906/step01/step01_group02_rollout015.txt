    def optimize_gradient(h_values):
        nonlocal best_C5, best_h
        h = cp.Variable(len(h_values))
        constraints = [0 <= h, h <= 1]

        def objective(h_values):
            corr = np.correlate(h_values, 1.0 - h_values, mode="full")
            max_corr = np.max(corr)
            dx = 2.0 / len(h_values)
            return max_corr * dx
        
        # Minimize the maximum overlap
        prob = cp.Problem(cp.Minimize(objective(h)), constraints)
        try:
            prob.solve(verbose=False, solver=cp.SCS)
