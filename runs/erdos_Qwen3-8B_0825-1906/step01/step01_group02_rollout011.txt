```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
from scipy.signal import correlate

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from the provided starting point
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=np.float64)
    
    # Try with n_points = 100 (faster to optimize)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:] = 0.5  # Start with uniform distribution
    
    # Optimize the function using a local optimizer
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Ensure integral constraint
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return float('inf')
        # Compute overlap integral
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization with bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure integral constraint is satisfied
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        # Adjust to meet the constraint
        adjustment = (n_points / 2 - sum_h) / n_points
        best_h += adjustment
        best_h = np.clip(best_h, 0.0, 1.0)
    
    # Evaluate the result
    c5_bound = evaluate_erdos_solution(best_h, n_points, dx)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        print("Time budget exceeded, returning best so far.")
    
    return (best_h.tolist(), c5_bound, n_points)
```
