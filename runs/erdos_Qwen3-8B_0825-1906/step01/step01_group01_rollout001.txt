import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can be increased if time permits
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1  # Random initial guess in [-1, 1]
    h = np.clip(h, 0, 1)  # Ensure h[i] ∈ [0, 1]
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet ∫h(x) dx = 1

    # Function to compute C5
    def compute_c5(h):
        h = np.array(h, dtype=float)
        h = np.concatenate([h, h[:n_points//2]])  # Double the array for full convolution
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    # Optimization using gradient descent with line search
    best_h = h.copy()
    best_c5 = compute_c5(h)
    time_start = time.time()

    for _ in range(1000):  # Max iterations
        if time.time() - time_start > budget_s:
            break
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_i = h[i]
            h_test = h.copy()
            h_test[i] = h_i + 1e-4
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            c5_test = compute_c5(h_test)
            grad[i] = (c5_test - best_c5) / 1e-4

        # Gradient descent step
        h = h - 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5, n_points
