import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from scipy.optimize import Bounds

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points for better performance
    
    # Initial guess
    h_init = initial_h_values.copy()
    h_init = h_init[:n_points]  # Ensure it fits the n_points

    # Define the function to compute C5 for a given h
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:1]])  # Extend to handle full convolution
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    # Define the objective function for optimization
    def objective(x):
        nonlocal best_c5
        h = x.copy()
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Bounds for the optimization
    bounds = Bounds(0, 1)

    # Perform optimization using L-BFGS-B
    result = optimize.minimize(
        objective,
        h_init,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'disp': False}
    )

    # If we have time, try increasing the number of points
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        h_init = np.zeros(n_points)
        # Distribute the integral evenly
        h_init = np.repeat(0.5, n_points)
        h_init[:n_points // 2] = 0
        h_init[n_points // 2:] = 1
        h_init = np.clip(h_init, 0, 1)
        
        # Define the function to compute C5 for the new n_points
        def compute_c5_new(h):
            h = np.array(h)
            h = np.concatenate([h, h[:1]])  # Extend to handle full convolution
            corr = signal.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            return c5
        
        # Define the objective function for optimization
        def objective_new(x):
            nonlocal best_c5
            h = x.copy()
            h = np.clip(h, 0, 1)
            c5 = compute_c5_new(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            return c5
        
        # Bounds for the optimization
        bounds_new = Bounds(0, 1)
        
        # Perform optimization using L-BFGS-B
        result_new = optimize.minimize(
            objective_new,
            h_init,
            method='L-BFGS-B',
            bounds=bounds_new,
            options={'maxiter': 100, 'disp': False}
        )

    return best_h, best_c5, n_points
