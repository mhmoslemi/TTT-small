```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    
    # Calculate initial C5
    max_overlap = 0.0
    for k in range(1, n_points):
        overlap = np.correlate(h, 1 - h, mode="full")
        max_overlap = max(max_overlap, np.max(overlap))
    c5_bound = max_overlap * (2.0 / n_points)
    
    # Evaluate initial solution
    initial_c5 = evaluate_erdos_solution(h)
    print(f"Initial C5: {initial_c5}")
    
    # Define the function to minimize
    def objective(x):
        h_new = x.reshape(n_points)
        overlaps = np.correlate(h_new, 1 - h_new, mode="full")
        max_overlap = np.max(overlaps)
        return max_overlap * (2.0 / n_points)
    
    # Define constraints
    constraints = [
        cp.sum(h) == n_points / 2
    ]
    
    # Define variables
    h_var = cp.Variable(n_points, bounds=(0, 1))
    
    # Define problem
    problem = cp.Problem(cp.Minimize(objective(h_var)), constraints)
    
    # Run optimization
    start_time = time.time()
    try:
        result = problem.solve(solver=cp.ECOS, verbose=False)
        h_opt = h_var.value
        c5_bound = result
    except:
        h_opt = h
        c5_bound = initial_c5
    
    # Check elapsed time
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Further refinement if time allows
        # Use a local optimization method to refine the solution
        def objective_func(x):
            h_new = x.reshape(n_points)
            overlaps = np.correlate(h_new, 1 - h_new, mode="full")
            max_overlap = np.max(overlaps)
            return max_overlap * (2.0 / n_points)
        
        def constraint_func(x):
            return np.sum(x) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective_func,
            h_opt.flatten(),
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'disp': False, 'ftol': 1e-8}
        )
        h_opt = result.x.reshape(n_points)
        c5_bound = result.fun * (2.0 / n_points)
    
    # Return the optimized solution
    return (h_opt.tolist(), c5_bound, n_points)
```
