import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 samples for speed
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    initial_h_values = np.load('initial_h_values.npy')  # Load initial construction
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h

    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def minimize_c5(h):
        return objective(h)

    def optimize_h(h):
        result = opt.minimize(
            minimize_c5, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        return result.x

    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)
    elapsed = 0

    for _ in range(10):
        if elapsed >= budget_s - 1:
            break
        h = optimize_h(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        elapsed = time.time() - start_time

    if elapsed < budget_s:
        n_points = min(200, int(np.sqrt(budget_s * 100)))  # Increase points if time allows
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h

        def objective_large(h):
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        def minimize_c5_large(h):
            return objective_large(h)

        def optimize_h_large(h):
            result = opt.minimize(
                minimize_c5_large, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            return result.x

        best_h_large = h.copy()
        best_c5_large = objective_large(h)
        for _ in range(5):
            if elapsed >= budget_s - 1:
                break
            h = optimize_h_large(h)
            current_c5 = objective_large(h)
            if current_c5 < best_c5_large:
                best_h_large = h.copy()
                best_c5_large = current_c5
            elapsed = time.time() - start_time

        best_h = best_h_large
        best_c5 = best_c5_large

    return best_h, best_c5, n_points
