```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = initial_h * (n_points / 2) / sum_h

    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        n_points = len(h)
        dx = 2.0 / n_points
        # Ensure the sum is correct
        if not np.isclose(np.sum(h), n_points / 2):
            h = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        # Extract the relevant part
        corr = corr[n_points - 1 : 2 * n_points - 1]
        # Compute max overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to optimize
    def optimize_h():
        n_points = len(initial_h)
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        # Objective: minimize the max overlap
        # We use a trick to minimize the maximum value
        max_overlap = cp.Variable()
        constraints += [max_overlap >= cp.sum(h[i] * (1 - h[(i + k) % n_points])) * (2.0 / n_points) for k in range(1, n_points)]
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-4, max_iter=1000)
            if result is not None:
                return [float(h[i]) for i in range(n_points)]
        except:
            pass
        return initial_h

    # Perform a few iterations of optimization
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        h = optimize_h()
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
```
