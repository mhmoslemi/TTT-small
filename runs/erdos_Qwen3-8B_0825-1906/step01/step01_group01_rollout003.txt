def run(seed=42, budget_s=60, **kwargs):

    # Set up initial parameters
    n_points = 45  # Start with the initial sample size
    max_iterations = 100  # Limit the number of optimization steps
    
    # Precompute the spacing
    dx = 2.0 / n_points
    
    # Optimization function
    def optimize_h(h):
        # Convert to numpy array
        h = np.array(h)
        
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        
        return c5

    # Function to generate a random perturbation
    def perturb(h, magnitude=0.1):
        h_perturbed = h.copy()
        for i in range(len(h_perturbed)):
            h_perturbed[i] += np.random.uniform(-magnitude, magnitude)
            h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
        return h_perturbed

    # Function to perform a local search
    def local_search(h, iterations=20, magnitude=0.1):
        best_h = h.copy()
        best_c5 = optimize_h(best_h)
        
        for _ in range(iterations):
            h_perturbed = perturb(best_h, magnitude)
            c5 = optimize_h(h_perturbed)
            if c5 < best_c5:
                best_h = h_perturbed
                best_c5 = c5
        
        return best_h, best_c5

    # Function to perform a global search using a random restart
    def global_search(h, restarts=5, iterations=20, magnitude=0.1):
        best_h = h.copy()
        best_c5 = optimize_h(best_h)
        
        for _ in range(restarts):
            h_perturbed = perturb(h, magnitude)
            h_perturbed = local_search(h_perturbed, iterations, magnitude)[0]
            c5 = optimize_h(h_perturbed)
            if c5 < best_c5:
                best_h = h_perturbed
                best_c5 = c5
        
        return best_h, best_c5

    h = initial_h_values.copy()
