import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
# evaluate_erdos_solution is used to compute the C5 value
# initial_h_values is the starting point for the optimization (n=45 samples)

def run(seed=42, budget_s=60, **kwargs):
    # Start timing
    start_time = time.time()
    
    # Number of points
    n_points = 200  # We use 200 points for good resolution
    
    # Initialize h_values with initial_h_values scaled to n_points
    # Resample initial_h_values to n_points using linear interpolation
    initial_h = initial_h_values
    x_initial = np.linspace(0, 2, len(initial_h))
    x_new = np.linspace(0, 2, n_points)
    h_values = np.interp(x_new, x_initial, initial_h)
    
    # Normalize to ensure the integral equals 1
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    # Optimization setup
    # We use a numerical optimizer to minimize C5
    # We'll use the Nelder-Mead method for its robustness

    def objective(h_values):
        # Reshape h_values to be a function over [0, 2]
        h = np.reshape(h_values, (n_points,))
        # Compute the overlap integral using convolution
        corr = signal.correlate(h, 1 - h, mode="full")
        # Take the maximum over all lags
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    # Bounds for h_values
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Perform optimization
    result = optimize.minimize(
        objective,
        h_values,
        method="Nelder-Mead",
        bounds=bounds,
        options={"xatol": 1e-5, "fatol": 1e-5, "gtol": 1e-5}
    )

    # Extract the best solution found
    best_h_values = result.x
    best_c5 = result.fun

    # Ensure we stay within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        # If time is up, return the best found so far
        return (best_h_values, best_c5, n_points)

    # Perform a few more iterations to improve the solution if time allows
    for _ in range(3):
        result = optimize.minimize(
            objective,
            best_h_values,
            method="Nelder-Mead",
            bounds=bounds,
            options={"xatol": 1e-5, "fatol": 1e-5, "gtol": 1e-5}
        )
        best_h_values = result.x
        best_c5 = result.fun
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break

    return (best_h_values, best_c5, n_points)
