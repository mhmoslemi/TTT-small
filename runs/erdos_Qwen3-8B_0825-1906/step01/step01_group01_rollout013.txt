import numpy as np
import scipy.optimize as scio
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute the cross-correlation
    corr = correlate(h, 1 - h, mode="full")
    # Find maximum overlap
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # Dummy initial values
    return np.array([0.5] * 45)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points
    max_points = 150  # Max points to consider
    step = 50  # Step size for increasing points

    # Initial solution
    h = initial_h_values()
    n_points = len(h)
    if n_points < max_points:
        while n_points < max_points:
            # Try increasing the number of points
            new_n = n_points + step
            # Interpolate the current h to new_n points
            new_h = np.zeros(new_n)
            dx = 2.0 / new_n
            for i in range(n_points):
                x = i * dx
                new_h = new_h + h[i] * np.ones(new_n) * dx
            # Normalize to ensure integral is 1
            new_h = new_h / (np.sum(new_h) * dx)
            # Evaluate
            c5 = evaluate_erdos_solution(new_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = new_h
                n_points = new_n
            else:
                # If not improving, try decreasing points
                break
    else:
        # Start with max_points
        h = np.zeros(max_points)
        dx = 2.0 / max_points
        # Fill h with 0.5
        h = np.ones(max_points) * 0.5
        # Normalize to ensure integral is 1
        h = h / (np.sum(h) * dx)
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        best_c5 = c5
        best_h = h

    # Optimization loop
    for n_points in range(50, max_points + 1, 20):
        # Generate initial h for this n_points
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        # Fill h with 0.5
        h = np.ones(n_points) * 0.5
        # Normalize to ensure integral is 1
        h = h / (np.sum(h) * dx)
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

        # Run optimization
        max_iter = 100
        for _ in range(max_iter):
            # Convert h into a decision variable
            h_var = cp.Variable(n_points, bounds=(0, 1))
            # Constraint: integral of h is 1
            constraint = cp.sum(h_var) == n_points / 2
            # Objective: minimize the maximum overlap
            # We'll use a convex relaxation with an epsilon
            epsilon = 1e-4
            # Create an objective that captures the max overlap
            # For simplicity, we approximate the max by evaluating all lags
            # and minimizing the maximum of the overlaps
            # But for optimization, we'll use a heuristic
            # We'll use a weighted sum of overlaps
            overlaps = []
            for k in range(1, n_points):
                # Compute overlap for shift k
                # Shift h by k
                h_shift = np.roll(h_var.value, k)
                # Compute overlap
                overlap = np.sum(h_var.value * (1 - h_shift)) * dx
                overlaps.append(overlap)
            # Minimize the maximum overlap
            # As an approximation, we'll minimize the sum of overlaps
            # This is a heuristic to help find better solutions
            objective = cp.Minimize(cp.sum(overlaps))
            # Solve the optimization
            prob = cp.Problem(objective, [constraint])
            try:
                prob.solve(solver='ECOS', eps=1e-6, max_time=10)
                if prob.status == 'optimal':
                    h = h_var.value
                    # Evaluate
                    c5 = evaluate_erdos_solution(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h
                else:
                    break
            except:
                break

    # Final evaluation
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return best_h, best_c5, len(best_h)

    return best_h, best_c5, len(best_h)
