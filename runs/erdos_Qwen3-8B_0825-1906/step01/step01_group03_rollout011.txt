```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can increase if time allows

    # Try initial solution
    h = initial_h_values.copy()
    h = h[:n_points]
    c5 = evaluate_erdos_solution(h, n_points)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    # Try random perturbations
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        h = initial_h_values.copy()
        h = h[:n_points]
        h += np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Try optimizing with CVXPY
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        dx = 2.0 / n_points
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)

        # Objective is to minimize the maximum overlap
        # Construct all shifted versions of h
        overlap = []
        for k in range(1, n_points):
            shifted_h = np.roll(h.value, k)
            overlap.append(h * (1 - shifted_h))

        # Minimize max(overlap)
        obj = cp.Minimize(cp.Max(overlap))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver='ECOS', eps=1e-3)
            if prob.status == 'OPTIMAL':
                h_opt = h.value
                c5 = evaluate_erdos_solution(h_opt, n_points)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass

    # Final check for best solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h[:n_points]
    c5 = evaluate_erdos_solution(best_h, n_points)
    return best_h, c5, n_points
```
