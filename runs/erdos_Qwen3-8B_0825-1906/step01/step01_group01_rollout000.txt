    def compute_c5(h):
        h = np.array(h)

        return np.max(result) * dx
    
    # Function to optimize the function h with given number of points
    def optimize_h(n_points):
        dx = 2.0 / n_points
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]
        objective = cp.Minimize(cp.max([cp.max(cp.multiply(h[i], 1 - h[j + i]) for i in range(n_points) for j in range(n_points) if i + j < n_points)]))
        problem = cp.Problem(objective, constraints)
        result = problem.solve(solver=cp.ECOS, eps=1e-4, max_iter=1000)
        if result is not None:
            h_values = h.value
            return h_values
        else:
            return initial_h
    
    # Try with increasing number of points within budget
    start_time = time.time()
