import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions and variables
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 45  # Start with the initial n_points

    # Try increasing n_points for better resolution
    for n_points in [45, 60, 80, 100, 120, 150, 180, 200]:
        if time.time() - start_time > budget_s - 5:
            break
        dx = 2.0 / n_points
        # Create initial guess from the given initial_h_values, then interpolate
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        # Scale to satisfy the integral constraint
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        # Define optimization variables
        h = cp.Variable(n_points, bounds=(0, 1))
        # Constraint: integral of h over [0, 2] equals 1
        constraint = cp.sum(h) == n_points / 2
        # Objective: minimize the maximum overlap integral
        # Compute all shifts k (from 1 to n_points-1)
        overlaps = []
        for k in range(1, n_points):  # Only k in [1, n_points-1]
            h_shifted = cp.reshape(cp.roll(h, k), (n_points, 1))
            one_minus_h = cp.reshape(1 - h, (1, n_points))
            overlap = cp.sum(h_shifted * one_minus_h) * dx
            overlaps.append(overlap)
        # Max overlap
        max_overlap = cp.Max(overlaps)
        # Solve the optimization problem
        try:
            problem = cp.Problem(cp.Minimize(max_overlap), [constraint])
            result = problem.solve(solver=cp.CPLEX, eps=1e-5, verbose=False)
            if result is not None:
                h_values = h.value
                c5 = max_overlap.value
                # Update best solution
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
        except:
            pass
        # Early exit if we've exceeded the budget
        if time.time() - start_time > budget_s - 5:
            break

    # Return the best solution found
    return (best_h, best_c5, n_points)
