```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h_values = np.copy(initial_h_values)
    dx = 2.0 / n_points
    
    def calculate_c5(h):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h)):
            shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def optimize_h(h_values):
        # Initial optimization attempt
        result = opt.minimize_scalar(lambda h: calculate_c5(h), bounds=(0, 1), method='bounded', options={'xatol': 1e-6})
        if result.success:
            h_values[:] = result.x
        else:
            h_values[:] = initial_h_values
    
    def refine_h(h_values):
        # Perform a local refinement using gradient descent
        learning_rate = 0.01
        max_its = 100
        for _ in range(max_its):
            c5 = calculate_c5(h_values)
            grad = np.zeros_like(h_values)
            for i in range(len(h_values)):
                h_plus = h_values.copy()
                h_plus[i] += 1e-5
                c5_plus = calculate_c5(h_plus)
                grad[i] = (c5_plus - c5) / 1e-5
            h_values -= learning_rate * grad
            h_values = np.clip(h_values, 0, 1)
    
    # Initial evaluation
    best_c5 = calculate_c5(h_values)
    best_h = h_values.copy()
    
    # Try several refinements
    for _ in range(3):
        optimize_h(h_values)
        refine_h(h_values)
        current_c5 = calculate_c5(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = calculate_c5(best_h)
    return best_h, final_c5, len(best_h)
```
