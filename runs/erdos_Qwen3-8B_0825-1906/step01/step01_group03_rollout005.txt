import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial setup
    n_points = 70  # Start with the initial n_points
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Ensure sum(h) * dx = 1
    sum_h = np.sum(h)
    if abs(sum_h * dx - 1.0) > 1e-8:
        raise ValueError("Initial h does not satisfy the integral constraint")
    
    # Function to compute C5
    def compute_c5(h_values):
        h = h_values
        n = len(h)
        dx = 2.0 / n
        h_full = np.zeros(2 * n - 1)
        h_full[:n] = h
        corr = np.correlate(h_full, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Optimization using CVXPY (for exploration)
    def optimize_h(h_initial, n_points, max_iter=100, tolerance=1e-5):
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        # Objective: minimize max over all k of integral of h(x)(1 - h(x + k))
        # We use a convex relaxation by minimizing the maximum over k of the integral
        # Approximate the integral using the trapezoidal rule
        dx = 2.0 / n_points
        c5 = 0.0
        for k in range(1, n_points):
            if k >= n_points:
                break
            h_k = np.roll(h, k)
            integral = np.sum(h * (1 - h_k)) * dx
            c5 = max(c5, integral)
        objective = cp.Minimize(c5)
        problem = cp.Problem(objective, constraints)
        result = problem.solve(solver=cp.SCS, max_iter=max_iter, abstol=tolerance)
        if result is None:
            return h_initial
        return h.value
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    iterations = 0
    while time.time() - start_time < budget_s:
        # Try different n_points sizes to avoid local minima
        n_points_list = [70, 100, 140, 180]
        for n_points in n_points_list:
            if time.time() - start_time >= budget_s:
                break
            # Initialize h with a small random perturbation
            h = np.random.rand(n_points) * 2 - 1
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if abs(sum_h * 2.0 / n_points - 1.0) > 1e-5:
                h = h * (1.0 / (sum_h * 2.0 / n_points))
            # Use a simple gradient descent to improve the solution
            for _ in range(10):
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                # Perturb h slightly
                perturbation = np.random.rand(n_points) * 0.1 - 0.05
                h += perturbation
                h = np.clip(h, 0, 1)
                sum_h = np.sum(h)
                if abs(sum_h * 2.0 / n_points - 1.0) > 1e-5:
                    h = h * (1.0 / (sum_h * 2.0 / n_points))
            # Use CVXPY for more structured optimization
            h_opt = optimize_h(h, n_points)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
        
        # Early stopping if we've tried all n_points
        if n_points >= 180:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
