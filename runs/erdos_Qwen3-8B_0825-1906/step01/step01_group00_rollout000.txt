import numpy as np
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the convolution
    conv = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap corresponds to the maximum value in the convolution
    max_overlap = np.max(conv)
    # Multiply by dx to get the integral value
    return max_overlap * dx

    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):

    population_size = 50

            return 100.0  # Penalize for not meeting integral constraint
        return c5

    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8

        # Select parents
