```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Start with the initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = np.array([0.5] * n_points)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        shifted = np.zeros(2 * n - 1)
        for i in range(n):
            shifted[i:i + n] += h[i] * (1 - h[i + 1:n + 1])
        overlap = np.max(shifted * dx)
        return overlap
    
    # Function to evaluate the objective for optimization
    def objective_func(x):
        x = np.array(x)
        if np.any(x < 0) or np.any(x > 1):
            return float('inf')
        if not np.isclose(np.sum(x), n_points / 2):
            return float('inf')
        return compute_c5(x)
    
    # Optimization
    result = optimize.minimize(
        objective_func,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    best_n_points = len(best_h)
    
    # Check if we can further improve with a smaller n_points
    for n in range(50, best_n_points, -5):
        dx = 2.0 / n
        h = np.array([0.5] * n)
        # Use a simple heuristic to adjust h
        for i in range(n):
            h[i] = 0.5 + np.sin(np.pi * i / n) * 0.25
        if not np.isclose(np.sum(h), n / 2):
            h = np.array([0.5] * n)
        # Compute C5
        shifted = np.zeros(2 * n - 1)
        for i in range(n):
            shifted[i:i + n] += h[i] * (1 - h[i + 1:n + 1])
        overlap = np.max(shifted * dx)
        if overlap < best_c5:
            best_h = h
            best_c5 = overlap
            best_n_points = n
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    print(f"Final C5: {final_c5}, n_points: {len(best_h)}")
    
    # Ensure the time budget is respected
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        print("Time budget exceeded. Returning best solution found.")
    
    return best_h, final_c5, len(best_h)
```
