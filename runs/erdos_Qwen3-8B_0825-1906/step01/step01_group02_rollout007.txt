```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Start with the initial solution
    h_initial = initial_h_values
    n_points_initial = len(h_initial)
    dx_initial = 2.0 / n_points_initial
    c5_initial = evaluate_erdos_solution(h_initial, dx_initial)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = h_initial
        best_n_points = n_points_initial
    
    # Try reducing n_points to speed up computation
    for n_points in range(n_points_initial, 10, -1):
        if time.time() - start_time > budget_s - 1:
            break
        dx = 2.0 / n_points
        # Define the objective function and constraints
        def objective(h):
            # Compute the overlap integral
            h_flat = np.array(h).flatten()
            corr = np.correlate(h_flat, 1 - h_flat, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for h
        bounds = Bounds(0, 1)
        
        # Initial guess
        h_initial_reduced = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        
        # Optimization
        result = minimize(objective, h_initial_reduced, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if result.success:
            h_candidate = result.x
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n_points = n_points
        
        # Early stopping if budget is near
        if time.time() - start_time > budget_s - 2:
            break
    
    # Try increasing n_points for better precision
    for n_points in range(n_points_initial + 1, 100, 1):
        if time.time() - start_time > budget_s - 1:
            break
        dx = 2.0 / n_points
        # Define the objective function and constraints
        def objective(h):
            # Compute the overlap integral
            h_flat = np.array(h).flatten()
            corr = np.correlate(h_flat, 1 - h_flat, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for h
        bounds = Bounds(0, 1)
        
        # Initial guess
        h_initial_reduced = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        
        # Optimization
        result = minimize(objective, h_initial_reduced, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if result.success:
            h_candidate = result.x
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n_points = n_points
        
        # Early stopping if budget is near
        if time.time() - start_time > budget_s - 2:
            break
    
    # Final check with the best solution
    # Convert to list for output
    best_h_list = best_h.tolist()
    return best_h_list, best_c5, best_n_points
```
