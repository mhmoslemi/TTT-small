import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    if not np.isclose(sum_h, 1.0):
        h = h * (1.0 / sum_h) * dx
    
    # Function to compute C5
    def compute_c5(h_values):
        h = np.array(h_values)
        dx = 2.0 / len(h)
        h_ones = np.ones_like(h)
        overlap = np.correlate(h, h_ones - h, mode="full") * dx
        return np.max(overlap)
    
    # Function to compute the cost for optimization
    def compute_cost(h_values):
        return compute_c5(h_values)
    
    # Function to evaluate the solution
    def evaluate(h_values):
        return compute_c5(h_values)
    
    # Function to perform simple gradient descent
    def gradient_descent(h, learning_rate=0.01, steps=1000, tol=1e-5):
        for _ in range(steps):
            grad = np.zeros_like(h)
            dx = 2.0 / len(h)
            for i in range(len(h)):
                h_i = h[i]
                for k in range(len(h)):
                    if k == 0:
                        continue
                    overlap = np.correlate(h, 1 - h, mode="full") * dx
                    grad[i] += (overlap[k] - overlap[k - 1]) * dx
            grad = grad / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else grad
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            if np.abs(compute_c5(h) - compute_c5(h - learning_rate * grad)) < tol:
                break
        return h
    
    # Function to perform coordinate descent
    def coordinate_descent(h, steps=1000):
        for _ in range(steps):
            dx = 2.0 / len(h)
            for i in range(len(h)):
                h_i = h[i]
                h_test = h.copy()
                h_test[i] = 0.5
                if compute_c5(h_test) < compute_c5(h):
                    h = h_test
            return h
    
    # Function to perform simulated annealing
    def simulated_annealing(h, steps=1000, temp=1.0, cooling=0.99):
        for _ in range(steps):
            temp *= cooling
            h_test = h.copy()
            idx = np.random.randint(len(h))
            h_test[idx] = np.random.uniform(0, 1)
            cost = compute_c5(h_test)
            if np.random.random() < np.exp((compute_c5(h) - cost) / temp):
                h = h_test
        return h
    
    # Optimize with multiple strategies
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Perform gradient descent
    h = gradient_descent(h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5
    
    # Perform coordinate descent
    h = coordinate_descent(h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5
    
    # Perform simulated annealing
    h = simulated_annealing(h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5
    
    # Check if we have enough time for additional steps
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 1.0:
        # Perform a few more iterations of coordinate descent
        for _ in range(100):
            h = coordinate_descent(h)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    
    return best_h, best_c5, len(best_h)
