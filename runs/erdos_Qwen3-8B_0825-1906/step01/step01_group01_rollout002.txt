```python
import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Try different n_points values starting from the initial one
    n_points_list = [45, 60, 80, 100, 120, 140, 160, 180, 200]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        if time.time() > time.time() + budget_s:
            break
        
        # Generate initial guess based on the provided initial_h_values
        initial_h = initial_h_values
        if n_points != len(initial_h):
            # Resample to match n_points
            initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        
        # Ensure that the integral is exactly 1
        integral = np.sum(initial_h) * (2.0 / n_points)
        if np.abs(integral - 1.0) > 1e-6:
            # Adjust to make the integral exactly 1
            scale = 1.0 / integral
            initial_h = initial_h * scale
        
        # Define the objective function
        def objective(h):
            dx = 2.0 / n_points
            h = np.clip(h, 0, 1)
            shifted_h = np.roll(h, 1)  # Shift by 1 for the overlap
            overlap_integral = np.sum(h * (1 - shifted_h)) * dx
            return overlap_integral
        
        # Bounds for h: [0, 1]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B optimizer
        result = opt.minimize(
            objective,
            initial_h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'disp': False}
        )
        
        # Evaluate the result
        h_final = result.x
        dx = 2.0 / n_points
        shifted_h = np.roll(h_final, 1)
        overlap_integral = np.sum(h_final * (1 - shifted_h)) * dx
        c5 = overlap_integral
        
        # Update the best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final
            best_n = n_points
        
        print(f"n_points={n_points}, C5={c5:.6f}, best={best_c5:.6f}")
    
    # Use the best solution found
    dx = 2.0 / best_n
    shifted_h = np.roll(best_h, 1)
    c5_final = np.sum(best_h * (1 - shifted_h)) * dx
    
    return best_h, c5_final, best_n
```
