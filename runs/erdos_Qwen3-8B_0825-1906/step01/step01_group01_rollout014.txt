```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution from provided
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / n_points
        h_shifted = np.zeros((n_points, n_points))
        for i in range(n_points):
            h_shifted[i, :] = h[(i + 1) % n_points:]
        h_shifted = np.vstack([h_shifted, h_shifted[:, -1]])
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        return c5
    
    # Function to compute the objective and constraints for optimization
    def objective_func(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        return c5
    
    # Function to check constraints
    def constraint_func(x):
        h = x.copy()
        return np.sum(h) - n_points / 2
    
    # Initial guess
    x0 = h.copy()
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-4
    
    # Perform optimization
    best_h = x0
    best_c5 = compute_c5(best_h)
    
    for _ in range(max_iter):
        # Use a local optimization method
        x = cp.Variable(n_points)
        obj = cp.Minimize(objective_func(x))
        cons = [cp.sum(x) == n_points / 2]
        prob = cp.Problem(obj, cons)
        try:
            result = prob.solve(solver=cp.SCIP, verbose=False, time_limit=0.5)
        except:
            result = np.inf
        
        if result is not None and result < best_c5:
            best_h = x.value
            best_c5 = result
            print(f"New best C5: {best_c5}")
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 0.5:
            break
    
    # Evaluate the best solution
    final_c5 = compute_c5(best_h)
    print(f"Final C5: {final_c5}")
    
    return best_h.tolist(), final_c5, n_points
```
