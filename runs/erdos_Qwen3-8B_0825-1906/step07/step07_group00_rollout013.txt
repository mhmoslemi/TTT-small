import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with an initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try different n_points
    for n_points in [200, 250, 300, 350, 400]:
        # Reset best solution for this n_points
        best_h_for_n = initial_h.copy()
        best_c5_for_n = evaluate_erdos_solution(best_h_for_n)
        
        # Define bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        # Define objective function to minimize
        def objective(x):
            h_values = x.copy()
            return evaluate_erdos_solution(h_values)
        
        # Local refinement with SLSQP
        def local_refinement(h_values):
            result = optimize.minimize(
                fun=objective,
                x0=h_values,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'ftol': 1e-8, 'maxiter': 300}
            )
            return result.x
        
        # Use a more sophisticated optimization approach with multiple restarts
        def global_refinement(h_values):
            results = []
            for _ in range(3):
                perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
                perturbed_h = h_values + perturbation
                perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
                perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
                refined = local_refinement(perturbed_h)
                results.append(refined)
            
            # Evaluate all refined solutions
            evaluated = [(evaluate_erdos_solution(h), h) for h in results]
            best_eval, best_h = min(evaluated, key=lambda x: x[0])
            return best_h
        
        # Refine the best solution
        best_h_for_n = global_refinement(best_h_for_n)
        best_c5_for_n = evaluate_erdos_solution(best_h_for_n)
        
        # Perturb and refine with more sophisticated perturbation strategies
        for _ in range(5):
            # Perturb the current solution
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed_h = best_h_for_n + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
            
            # Run global refinement on the perturbed solution
            refined_h = global_refinement(perturbed_h)
            refined_c5 = evaluate_erdos_solution(refined_h)
            
            # Update best solution
            if refined_c5 < best_c5_for_n:
                best_c5_for_n = refined_c5
                best_h_for_n = refined_h
        
        # Update global best solution
        if best_c5_for_n < best_c5:
            best_c5 = best_c5_for_n
            best_h = best_h_for_n
        
        # Check if we have exceeded the time budget
        if time.time() - start_time > budget_s - 1:
            return (best_h.tolist(), best_c5, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
