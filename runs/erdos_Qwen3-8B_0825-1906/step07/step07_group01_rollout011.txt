import numpy as np
import scipy.optimize as optimize
import time
import random
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Higher-resolution initial approximation
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial approximation with improved structure
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.25 <= x <= 0.75:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)
    
    # Define objective and constraint functions
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define the optimization method with more efficient parameters
    def local_refinement(h_values):
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 300, 'disp': False}
        )
        return result.x
    
    # Refinement strategy with global search and simulated annealing
    def global_refinement(h_values):
        results = []
        # Random perturbations
        for _ in range(3):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed_h = h_values + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
            refined = local_refinement(perturbed_h)
            results.append((evaluate_erdos_solution(refined), refined))
        
        # Simulated annealing for further refinement
        def simulated_annealing(h_values, temp=1.0, cooling_rate=0.995, steps=100):
            current = h_values.copy()
            current_val = evaluate_erdos_solution(current)
            best = current_val
            best_state = current.copy()
            for _ in range(steps):
                perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                perturbed = current + perturbation
                perturbed = np.clip(perturbed, 0.0, 1.0)
                perturbed = perturbed / np.sum(perturbed) * n_points / 2
                perturbed_val = evaluate_erdos_solution(perturbed)
                delta = perturbed_val - current_val
                if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                    current = perturbed
                    current_val = perturbed_val
                if current_val < best:
                    best = current_val
                    best_state = current.copy()
                temp *= cooling_rate
            return best_state
        
        # Run simulated annealing on the best refined solution
        best_result = min(results, key=lambda x: x[0])
        refined_h = simulated_annealing(best_result[1], temp=1.0, steps=200)
        return refined_h
    
    # Initial refinement
    best_h = global_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Further refinement with multiple restarts and perturbations
    for _ in range(5):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Run refinement on the perturbed solution
        refined_h = global_refinement(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)
        
        # Update best solution
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Final evaluation and early stopping
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
