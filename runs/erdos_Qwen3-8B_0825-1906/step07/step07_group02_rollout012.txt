import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Return a more refined and distributed initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.2 <= x <= 0.8:
            h[i] = 0.5
        elif 0.1 <= x <= 0.3 or 0.7 <= x <= 0.9:
            h[i] = 0.8
        elif 0.05 <= x <= 0.15 or 0.85 <= x <= 0.95:
            h[i] = 0.3
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    h = np.array(initial_h)
    
    # Function to evaluate the objective function
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Function to enforce the constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for variables
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to perform a single optimization step with constraints
    def optimize_with_constraints(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
        )
        return result.x
    
    # Evaluate initial solution
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed_time = time.time() - start_time
    
    # Use a more sophisticated optimization strategy: simulated annealing
    # Define a more diverse initial population
    num_initial_populations = 5
    initial_populations = [np.random.uniform(0, 1, n_points) for _ in range(num_initial_populations)]
    initial_populations = [h / np.sum(h) * n_points / 2 for h in initial_populations]
    
    # Evaluate all initial populations
    best_pop = initial_populations[0]
    best_c5_pop = evaluate_erdos_solution(best_pop)
    for h_pop in initial_populations:
        h_pop = h_pop / np.sum(h_pop) * n_points / 2
        c5_pop = evaluate_erdos_solution(h_pop)
        if c5_pop < best_c5_pop:
            best_pop = h_pop
            best_c5_pop = c5_pop
    
    # Set the best initial solution as the starting point
    best_h = best_pop
    best_c5 = best_c5_pop
    
    # Simulated annealing parameters
    temp = 1.0
    temp_min = 1e-5
    temp_decay = 0.995
    
    # Perform simulated annealing
    h_current = best_h.copy()
    current_c5 = best_c5
    
    while temp > temp_min and elapsed_time < budget_s - 1:
        # Generate a random perturbation
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_candidate = h_current + perturbation
        h_candidate = np.clip(h_candidate, 0.0, 1.0)
        h_candidate = h_candidate / np.sum(h_candidate) * n_points / 2
        
        # Evaluate the candidate
        candidate_c5 = evaluate_erdos_solution(h_candidate)
        
        # Acceptance probability
        if candidate_c5 < current_c5:
            h_current = h_candidate
            current_c5 = candidate_c5
        else:
            # Use Boltzmann acceptance
            delta = candidate_c5 - current_c5
            prob = np.exp(-delta / temp)
            if np.random.rand() < prob:
                h_current = h_candidate
                current_c5 = candidate_c5
        
        # Update temperature
        temp *= temp_decay
        elapsed_time = time.time() - start_time
    
    # Update the best solution
    if current_c5 < best_c5:
        best_h = h_current
        best_c5 = current_c5
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
