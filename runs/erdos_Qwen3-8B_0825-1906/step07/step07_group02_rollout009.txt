```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Return a more refined and distributed initialization
    n_points = 200
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.2 <= x <= 0.8:
            h[i] = 0.5
        elif 0.1 <= x <= 0.3 or 0.7 <= x <= 0.9:
            h[i] = 0.8
        elif 0.05 <= x <= 0.15 or 0.85 <= x <= 0.95:
            h[i] = 0.3
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    h = np.array(initial_h)
    
    # Function to evaluate the objective function
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Function to enforce the constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for variables
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to perform a single optimization step with constraints
    def optimize_with_constraints(x0):
        result = optimize.minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-9, 'maxiter': 200, 'disp': False}
        )
        return result.x
    
    # Evaluate initial solution
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed_time = time.time() - start_time
    
    # Use a more systematic and diverse search
    for _ in range(10):  # Limit iterations based on time
        if elapsed_time > budget_s - 1:
            break
        
        # Generate perturbations in multiple ways
        perturbations = []
        # Random perturbation
        perturbations.append(np.random.uniform(-0.1, 0.1, size=n_points))
        # Small shift in one region
        shift_perturbation = np.zeros(n_points)
        shift_perturbation[50:100] = np.random.uniform(-0.2, 0.2, size=50)
        perturbations.append(shift_perturbation)
        # Large shift in one region
        shift_perturbation = np.zeros(n_points)
        shift_perturbation[50:100] = np.random.uniform(-0.5, 0.5, size=50)
        perturbations.append(shift_perturbation)
        
        for perturbation in perturbations:
            perturbed_h = h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
            
            # Optimize from the perturbed starting point
            perturbed_h_opt = optimize_with_constraints(perturbed_h)
            perturbed_c5 = evaluate_erdos_solution(perturbed_h_opt)
            
            # Update best solution if improved
            if perturbed_c5 < best_c5:
                best_h = perturbed_h_opt
                best_c5 = perturbed_c5
            
            # Update current solution for next iteration
            h = perturbed_h_opt
            elapsed_time = time.time() - start_time
    
    # Additional refinement using a small grid search
    for _ in range(5):  # Limit iterations based on time
        if elapsed_time > budget_s - 1:
            break
        
        # Create a small grid of perturbations
        grid_perturbations = []
        for i in range(5):
            for j in range(5):
                perturbation = np.zeros(n_points)
                perturbation[i*n_points//5 : (i+1)*n_points//5] = np.random.uniform(-0.1, 0.1, size=n_points//5)
                grid_perturbations.append(perturbation)
        
        for perturbation in grid_perturbations:
            perturbed_h = h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
            
            # Optimize from the perturbed starting point
            perturbed_h_opt = optimize_with_constraints(perturbed_h)
            perturbed_c5 = evaluate_erdos_solution(perturbed_h_opt)
            
            # Update best solution if improved
            if perturbed_c5 < best_c5:
                best_h = perturbed_h_opt
                best_c5 = perturbed_c5
            
            # Update current solution for next iteration
            h = perturbed_h_opt
            elapsed_time = time.time() - start_time
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
```
