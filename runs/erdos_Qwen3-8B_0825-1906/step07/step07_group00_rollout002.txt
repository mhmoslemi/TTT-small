import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    conv = np.correlate(h, one_minus_h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def initial_h_values():
    # Initial guess: periodic pattern with 3 segments
    n_points = 100
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.2 <= x <= 0.4:
            h[i] = 1.0
        elif 0.6 <= x <= 0.8:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with an initial approximation
    n_points = 100
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.2 <= x <= 0.4:
            h[i] = 1.0
        elif 0.6 <= x <= 0.8:
            h[i] = 1.0
    h = h / np.sum(h) * n_points / 2
    initial_h = h.tolist()
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)  # Keep it consistent for the rest of the run
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define objective function to minimize
    def objective(x):
        h_values = x.copy()
        return evaluate_erdos_solution(h_values)
    
    # Define constraint for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Refinement algorithm with more strategic perturbations
    def refine_solution(h_values):
        # Perform local optimization
        result = optimize.minimize(
            fun=objective,
            x0=h_values,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        refined_h = result.x
        
        # Try perturbed versions of the refined solution
        for _ in range(3):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed_h = refined_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
            result = optimize.minimize(
                fun=objective,
                x0=perturbed_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'ftol': 1e-8, 'maxiter': 200}
            )
            refined_h = result.x
        
        return refined_h
    
    # Use a more sophisticated optimization approach with multiple restarts
    def global_refinement(h_values):
        results = []
        for _ in range(5):
            # Add random noise to the current solution
            perturbation = np.random.uniform(-0.15, 0.15, size=n_points)
            perturbed_h = h_values + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
            
            # Refine the perturbed solution
            refined_h = refine_solution(perturbed_h)
            results.append(refined_h)
        
        # Evaluate all refined solutions
        evaluated = [(evaluate_erdos_solution(h), h) for h in results]
        best_eval, best_h = min(evaluated, key=lambda x: x[0])
        return best_h
    
    # Refine the best solution
    best_h = global_refinement(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perturb and refine with more sophisticated perturbation strategies
    for _ in range(5):
        # Perturb the current solution
        perturbation = np.random.uniform(-0.15, 0.15, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2
        
        # Run global refinement on the perturbed solution
        refined_h = global_refinement(perturbed_h)
        refined_c5 = evaluate_erdos_solution(refined_h)

        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Final evaluation
    final_c5 = best_c5
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
