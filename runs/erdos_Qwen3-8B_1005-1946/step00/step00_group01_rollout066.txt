import numpy as np
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def transfer_up(h):
    return np.repeat(h, 2)

def perturb_h(h, delta=0.01):
    new_h = h.copy()
    n = len(new_h)
    idx = np.random.randint(n)
    new_h[idx] += delta
    other_idx = np.random.randint(n)
    new_h[other_idx] -= delta
    new_h = np.clip(new_h, 0, 1)
    current_sum = np.sum(new_h)
    target_sum = n / 2
    if abs(current_sum - target_sum) > 1e-6:
        scale = target_sum / current_sum
        new_h = new_h * scale
        new_h = np.clip(new_h, 0, 1)
    return new_h
