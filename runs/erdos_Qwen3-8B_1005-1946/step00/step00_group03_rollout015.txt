def run(seed=42, budget_s=1100, **kwargs):

    # Check if initial_h_values is available
    initial_h = initial_h_values if 'initial_h_values' in locals() else None

    # Set n_points and dx
    n_points = 70
    dx = 2.0 / n_points

    # Initialize h
    if initial_h is not None:

        while np.sum(h_initial) != n_points // 2:
            diff = np.sum(h_initial) - n_points // 2
            if diff > 0:
                idx = np.where(h_initial == 1)[0]
                np.random.shuffle(idx)
                flip_idx = idx[:diff]
                h_initial[flip_idx] = 0
            else:
                idx = np.where(h_initial == 0)[0]
                np.random.shuffle(idx)
                flip_idx = idx[:abs(diff)]
                h_initial[flip_idx] = 1
    h_cont = h_initial.astype(float)

    # Objective function
    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')

    # Optimization with SLSQP
