        n_points = len(initial_h)
        if n_points in resolutions:
            h = initial_h.copy()
        else:
            h = generate_random_h(45)
    else:
        h = generate_random_h(45)

    for n in resolutions:
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            h = initial_h_values.copy()
        else:
            h = generate_random_h(n)
        candidates = [generate_random_h(n) for _ in range(5)] + [h]
        evaluated = []
        for candidate in candidates:
            c5 = evaluate_erdos_solution(candidate)
            evaluated.append((c5, candidate))
        best_candidate, best_c5_candidate = min(evaluated, key=lambda x: x[0])
        optimized_h = spsa_optimize(best_candidate, n, budget_s - time_used, max_iter=1000)
        c5 = evaluate_erdos_solution(optimized_h)
        if c5 < best_c5:
            best_h = optimized_h
            best_c5 = c5
        time_used += 500

    return best_h, best_c5, len(best_h)
