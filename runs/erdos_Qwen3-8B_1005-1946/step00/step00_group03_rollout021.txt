        return max_corr * dx
    
    for n in n_points_list:
        if n == 70 and 'initial_h_values' in globals():
            h = initial_h_values.copy()
        else:
            h = np.zeros(n, dtype=int)
            h[:n//2] = 1
            np.random.shuffle(h)
        current_c5 = compute_c5(h, n)
        best_h_for_n = h.copy()
        best_c5_for_n = current_c5
        T = 10.0
        cooling_rate = 0.95
        max_iterations = 1000
        iterations = 0
        while iterations < max_iterations and time.time() - start_time < budget_s:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, n)
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_c5 = new_c5
                if current_c5 < best_c5_for_n:
                    best_c5_for_n = current_c5
                    best_h_for_n = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
            T *= cooling_rate
            iterations += 1
        if best_c5_for_n < best_c5:
            best_c5 = best_c5_for_n
            best_h = best_h_for_n.copy()
            best_n = n
    return (best_h, best_c5, best_n)
