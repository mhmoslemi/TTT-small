import numpy as np
import time

def repair_solution(h, n):
    s = n / 2.0
    s0 = np.sum(h)
    delta = (s - s0) / n
    h_adj = h + delta
    h_clip = np.clip(h_adj, 0, 1)
    s_clip = np.sum(h_clip)
    residual = s - s_clip
    if abs(residual) < 1e-5:
        return h_clip
    if residual > 0:
        mask = h_clip < 1
        room = 1 - h_clip[mask]
        indices = np.argsort(room)[::-1]
        for i in indices:
            if residual <= 0:
                break
            add = min(room[i], residual)
            h_clip[i] += add
            residual -= add
    else:
        mask = h_clip > 0
        room = h_clip[mask]
        indices = np.argsort(room)
        for i in indices:
            if residual >= 0:
                break
            subtract = min(room[i], -residual)
            h_clip[i] -= subtract
            residual += subtract
    return h_clip

def objective(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = kwargs.get('initial_h_values')
    
    n_points_list = [70, 140, 280]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_points_list:
        if n == 70 and initial_h is not None:
            h = initial_h.copy()
        else:
            h = np.random.rand(n)
            h = repair_solution(h, n)
        
        h_repaired = repair_solution(h, n)
        c5_val = objective(h_repaired)
        best_c5 = c5_val
        best_h = h_repaired
        best_n = n
        
        T = 10.0
        cooling_rate = 0.95
        max_iter = 1000
        start_time = time.time()
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            indices = np.where(h == 0)[0]
            if len(indices) < 1:
                break
            i1 = np.random.choice(indices)
            indices1 = np.where(h == 1)[0]
            if len(indices1) < 1:
                break
            i2 = np.random.choice(indices1)
            
            h[i1], h[i2] = 1, 0
            
            new_c5 = objective(h)

                if np.random.random() < prob:
                    best_c5 = new_c5
                    best_h = h.copy()

            break
