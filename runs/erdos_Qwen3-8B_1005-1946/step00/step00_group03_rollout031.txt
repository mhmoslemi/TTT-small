import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points_list = [70, 140]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    time_per_n = {'70': 550, '140': 550}
    start_time = time.time()
    
    def repair_binary(h):
        diff = np.sum(h) - len(h) / 2
        if abs(diff) < 1e-6:
            return h
        for i in range(len(h)):
            if diff > 0:
                if h[i] == 0:
                    h[i] = 1
                    diff -= 1
            else:
                if h[i] == 1:
                    h[i] = 0
                    diff += 1
            if abs(diff) < 1e-6:
                break
        return h
    
    def objective_binary(h):
        h = np.array(h, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))
    
    def objective_cont(h):
        h = np.array(h, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))
    
    def grad_cont(h):
        h = np.array(h, dtype=float)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        dx = 2.0 / len(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            for k in range(len(h)):
                if max_idx - k >= 0 and max_idx - k < len(h):
                    grad[i] += (1 - h[i + k]) if (i + k < len(h)) else 0
                if max_idx + k < len(h):
                    grad[i] += h[i - k] if (i - k >= 0) else 0
        return grad * dx
    
    for n in n_points_list:
        if time.time() - start_time > budget_s:
            break
        if n == 70 and initial_h_values is not None:
            h_pop = np.array(initial_h_values, dtype=float)
        else:
            h_pop = np.random.rand(n)
            h_pop = np.where(np.random.rand(n) < 0.5, 1, 0)
            h_pop = repair_binary(h_pop)
        pop_size = 50
        pop = [h_pop.copy() for _ in range(pop_size)]
        for gen in range(1000):
            if time.time() - start_time > budget_s:
                break
            fitness = [objective_binary(h) for h in pop]
            elite_idx = np.argmin(fitness)
            best_h_gen = pop[elite_idx].copy()
            if fitness[elite_idx] < best_c5:
                best_c5 = fitness[elite_idx]
                best_h = best_h_gen.copy()
                best_n = n
            # Tournament selection
            tournament = np.random.choice(range(pop_size), size=3, replace=False)
            selected = tournament[np.argmin([fitness[i] for i in tournament])]
            # Crossover
            if np.random.rand() < 0.7:
                cross_idx = np.random.randint(1, n-1)
                child = np.concatenate([pop[selected][:cross_idx], pop[np.random.choice(range(pop_size), replace=False)][:n - cross_idx]])
                child = repair_binary(child)
                pop[selected] = child
            # Mutation
            if np.random.rand() < 0.01:
                mut_idx = np.random.randint(n)
                pop[selected][mut_idx] = 1 - pop[selected][mut_idx]
                pop[selected] = repair_binary(pop[selected])
        # Local optimization
        if time.time() - start_time > budget_s:
            continue
        h_cont = best_h.copy()
        h_cont = np.array(h_cont, dtype=float)
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}
        res = opt.minimize(objective_cont, h_cont, method='trust-constr', bounds=bounds, constraints=cons, jac=grad_cont, options={'maxiter': 100})
        if res.success:
            h_cont = res.x
            corr = np.correlate(h_cont, 1 - h_cont, mode='full')
            c5 = np.max(corr) * (2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cont.copy()
                best_n = n
    return (best_h, best_c5, best_n)
