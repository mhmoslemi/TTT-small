            if c5 < best_c5:
                best_h = h_adjusted.copy()
                best_c5 = c5
            elapsed = time.time() - start_time
            if elapsed > time_per_resolution:
                break
        total_time += elapsed
        if total_time >= budget_s:
            break

    return (best_h, best_c5, len(best_h))
