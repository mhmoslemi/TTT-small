import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / n)
    resolutions = [94, 188, 376, 752]
    elapsed = 0.0
    total_time = budget_s

    for res in resolutions:
        h = np.repeat(best_h, 2)
        time_limit = 100.0
        start_time = time.time()
        for _ in range(1000):
            idx = np.random.choice(len(h), 10, replace=False)
            h_perturbed = h.copy()
            h_perturbed[idx] += np.random.normal(0, 0.1, 10)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            s = np.sum(h_perturbed)
            h_perturbed *= (res / 2) / s
            dx = 2.0 / res
            corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturbed.copy()
            if time.time() - start_time > time_limit:
                break
        elapsed += time.time() - start_time
        if elapsed >= total_time:
            break
    return (best_h, best_c5, len(best_h))
