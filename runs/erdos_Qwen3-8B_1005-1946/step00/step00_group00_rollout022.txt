        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = repair(h, n_points)
    
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    
    resolutions = [45, 30, 20, 10]
    time_budgets = [600, 400, 100, 50]
    total_time = 0
    
    for i, (n, t_budget) in enumerate(zip(resolutions, time_budgets)):
        start_time = time.time()
        if i > 0:
            prev_n = resolutions[i-1]
            prev_h = best_h
            if prev_n > n:
                h = np.array([np.mean(prev_h[i*n//prev_n : (i+1)*n//prev_n]) for i in range(n)])
            else:
                x = np.linspace(0, 2, prev_n)
                x_new = np.linspace(0, 2, n)
                h = np.interp(x_new, x, prev_h)
            h = np.clip(h, 0, 1)
            h = repair(h, n)
        else:
            if 'initial_h_values' in globals():
                h = initial_h_values.copy()
            else:
                h = np.random.rand(n)
                h = h / np.sum(h) * (n / 2)
                h = np.clip(h, 0, 1)
                h = repair(h, n)
        
        def objective(x):
            x = np.clip(x, 0, 1)
            x = repair(x, n)
            return evaluate_erdos_solution(x)
        
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n, options={'maxiter': 500})
        h = result.x
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        
        elapsed_time = time.time() - start_time
        total_time += elapsed_time
        if total_time >= budget_s:
            break
    
    return (best_h, best_c5, len(best_h))

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) < 1e-5:
        return h
    scale = target_sum / current_sum
    h = h * scale
    h = np.clip(h, 0, 1)
    return h
