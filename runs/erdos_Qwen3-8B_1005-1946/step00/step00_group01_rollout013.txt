            if len(zeros) == 0 or len(ones) == 0:
                break
            i = np.random.choice(zeros)
            j = np.random.choice(ones)
            current_h[i], current_h[j] = 1, 0
            c5 = compute_C5(current_h)
            if c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = c5
            current_h[i], current_h[j] = 0, 1
    
    return (best_h, best_c5, len(best_h))
