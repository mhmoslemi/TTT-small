import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = kwargs.get('initial_h_values', None)
    n_points_list = [70, 140, 280]
    best_c5 = float('inf')
    best_h = None
    best_n = 70
    
    def repair(h, n):
        s = n / 2.0
        s0 = np.sum(h)
        delta = (s - s0) / n
        h_adj = h + delta
        h_clip = np.clip(h_adj, 0, 1)
        s_clip = np.sum(h_clip)
        residual = s - s_clip
        if abs(residual) < 1e-5:
            return h_clip
        if residual > 0:
            mask = h_clip < 1
            room = 1 - h_clip[mask]
            indices = np.argsort(room)[::-1]
            for i in indices:
                if residual <= 0:
                    break
                add = min(room[i], residual)
                h_clip[i] += add
                residual -= add
        else:
            mask = h_clip > 0
            room = h_clip[mask]
            indices = np.argsort(room)
            for i in indices:
                if residual >= 0:
                    break
                subtract = min(room[i], -residual)
                h_clip[i] -= subtract
                residual += subtract
        return h_clip

    def objective(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * (2.0 / len(h))

    for n in n_points_list:
        if n == 70 and initial_h is not None:
            h = initial_h.copy()
        else:
            h = np.zeros(n)
            h[:n//2] = 1.0
            np.random.shuffle(h)
        h = repair(h, n)
        c5_val = objective(h)
        best_c5, best_h, best_n = c5_val, h, n
        
        T = 10.0
        cooling_rate = 0.95
        max_iter = 1000
        start_time = time.time()
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            idx0 = np.where(h == 0)[0]
            idx1 = np.where(h == 1)[0]
            if not idx0.size or not idx1.size:
                break
            i1, i2 = np.random.choice(idx0), np.random.choice(idx1)
            h[i1], h[i2] = 1, 0
            h = repair(h, n)
            new_c5 = objective(h)
            if new_c5 < best_c5:
                best_c5, best_h, best_n = new_c5, h, n
            else:
                if np.random.random() < np.exp(-(new_c5 - best_c5) / T):
                    best_c5, best_h, best_n = new_c5, h, n
            T *= cooling_rate
        
        if best_c5 < 0.38085:
            break
    
    return best_h, best_c5, best_n
