def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        n_points = 45
        h = np.random.rand(n_points)
        target_sum = n_points / 2
        current_sum = np.sum(h)
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        h_initial = h
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def repair_h(h_vec):
        h = np.clip(h_vec, 0, 1)
        target_sum = n_points / 2
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-5:
            return h
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def objective(h_vec):
        h_repaired = repair_h(h_vec)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        sum_h = np.sum(h_repaired)
        if abs(sum_h - n_points / 2) > 1e-5:
            c5 = max(c5, 1e6)
        return c5

    bounds = [(0, 1)] * n_points
    start_time = time.time()
    result = minimize(objective, h_initial, method='CMA', bounds=bounds, options={'maxiter': 100})
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        h_best = result.x
        c5 = objective(h_best)
        return (h_best, c5, n_points)
    else:
        h_best = result.x
        c5 = objective(h_best)
        return (h_best, c5, n_points)
