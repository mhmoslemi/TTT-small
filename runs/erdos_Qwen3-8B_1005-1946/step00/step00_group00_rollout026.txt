            break
        allocated_time = time_budgets.get(n, 0)
        if allocated_time <= 0:
            continue
        
        candidates = []
        for _ in range(5):
            h = np.random.rand(n)
            h = h / np.sum(h) * (n / 2)
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            target_sum = n / 2
            scale = target_sum / current_sum
            h = h * scale
            h = np.clip(h, 0, 1)
            candidates.append(h)
        
        if best_h is not None and len(best_h) == n:
            candidates.append(best_h)
        
        evaluated = []
        for h in candidates:
            if np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n/2) < 1e-5:
                c5 = evaluate_erdos_solution(h)
                evaluated.append((c5, h))
        
        if evaluated:
            evaluated.sort()
            current_best_c5, current_best_h = evaluated[0]
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_h = current_best_h
            
            time_left = allocated_time - 100
            if time_left <= 0:
                continue
            
            h_spsa = current_best_h.copy()
            a = 0.1
            c = 0.1
            gamma = 0.9
            b = 0.5
            alpha = 0.1
            max_iter = int(time_left / 10)
            
            for _ in range(max_iter):
                delta = np.random.rand(n) * 2 - 1
                delta = delta * (1 - h_spsa) * h_spsa
                h_plus = h_spsa + c * delta
                h_minus = h_spsa - c * delta
                
                h_plus = np.clip(h_plus, 0, 1)
                h_plus_sum = np.sum(h_plus)
                target_sum = n / 2
                scale_plus = target_sum / h_plus_sum
                h_plus = h_plus * scale_plus
                h_plus = np.clip(h_plus, 0, 1)
                
                h_minus = np.clip(h_minus, 0, 1)
                h_minus_sum = np.sum(h_minus)
                scale_minus = target_sum / h_minus_sum
                h_minus = h_minus * scale_minus
                h_minus = np.clip(h_minus, 0, 1)
                
                f_plus = evaluate_erdos_solution(h_plus)
                f_minus = evaluate_erdos_solution(h_minus)
                
                grad = (f_plus - f_minus) / (2 * c) * delta
                h_spsa = h_spsa - alpha * grad
                h_spsa = np.clip(h_spsa, 0, 1)
                h_spsa_sum = np.sum(h_spsa)
                scale = target_sum / h_spsa_sum
                h_spsa = h_spsa * scale
                h_spsa = np.clip(h_spsa, 0, 1)
                
                if abs(f_plus - f_minus) < 1e-6:
                    break
                alpha *= gamma
            
            c5_spsa = evaluate_erdos_solution(h_spsa)
            if c5_spsa < current_best_c5:
                current_best_c5 = c5_spsa
                current_best_h = h_spsa
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_h = current_best_h
    
    return (best_h, best_c5, len(best_h))
