import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    best_h = None
    best_c5 = None
    best_n = 45
    n_list = [45, 90, 180, 360]
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))
    
    def grad_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_indices = np.where(corr == max_val)[0]
        lags = [idx - (len(h) - 1) for idx in max_indices]
        grad = np.zeros_like(h)
        for lag in lags:
            if lag >= 0:
                for j in range(len(h) - lag):
                    grad[j] += (1 - h[j + lag])
            else:
                k = -lag
                for j in range(k, len(h)):
                    grad[j] += (1 - h[j - k])
        if lags:
            grad /= len(lags)
        return grad * (2.0 / len(h))
    
    def constraint(h):
        return np.sum(h) - len(h)/2
    
    for n in n_list:
        start_time = time.time()
