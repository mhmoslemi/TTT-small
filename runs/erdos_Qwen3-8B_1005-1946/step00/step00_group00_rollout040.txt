import numpy as np
import scipy.optimize
import time

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    target = n_points / 2
    current = np.sum(h)
    if abs(current - target) < 1e-5:
        return h
    scale = target / current
    h = h * scale
    h = np.clip(h, 0, 1)
    if abs(np.sum(h) - target) < 1e-5:
        return h
    non_boundary = np.where((h > 1e-6) & (h < 1 - 1e-6))[0]
    if len(non_boundary) == 0:
        return h
    diff = target - np.sum(h)
    h[non_boundary] += diff / len(non_boundary)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
        h_initial = initial_h_values
    else:
        n_points = 45
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    bounds = [(0, 1)] * n_points
    def obj(h):
        h_r = repair(h, n_points)
        corr = np.correlate(h_r, 1 - h_r, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    start = time.time()
    res = scipy.optimize.minimize(obj, h_initial, method='CMA', bounds=bounds, options={'maxiter': 100})
    elapsed = time.time() - start

    best_h = res.x
    best_c5 = obj(best_h)
