        candidates = []
        for _ in range(5):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            scale = (n_points / 2) / total
            h = h * scale
            candidates.append(h)
        
        feasible_candidates = []
        for h_candidate in candidates:
            if np.all(h_candidate >= 0) and np.all(h_candidate <= 1):
                if abs(np.sum(h_candidate) - n_points / 2) < 1e-5:
                    feasible_candidates.append(h_candidate)
        
        if not feasible_candidates:
            continue
        
        best_candidate = feasible_candidates[0]
        best_c5_candidate = evaluate_erdos_solution(best_candidate)
        for h_candidate in feasible_candidates[1:]:
            c5 = evaluate_erdos_solution(h_candidate)
            if c5 < best_c5_candidate:
                best_candidate = h_candidate
                best_c5_candidate = c5
        
        spsa_time = alloc_time - 50
        if spsa_time <= 0:
            current_c5 = best_c5_candidate
            current_h = best_candidate
        else:
            h = best_candidate.copy()
            a = 0.1
            c = 0.1
            gamma = 0.9
            alpha = 0.5
            start_time = time.time()
            for _ in range(1000):
                delta = np.random.choice([-1, 1], size=n_points)
                h_plus = h + c * delta
                h_plus = np.clip(h_plus, 0, 1)
                h_plus_sum = np.sum(h_plus)
                h_plus = h_plus * (n_points / 2) / h_plus_sum
                h_minus = h - c * delta
                h_minus = np.clip(h_minus, 0, 1)
                h_minus_sum = np.sum(h_minus)
                h_minus = h_minus * (n_points / 2) / h_minus_sum
                f_plus = evaluate_erdos_solution(h_plus)
                f_minus = evaluate_erdos_solution(h_minus)
                grad = (f_plus - f_minus) / (2 * c) * delta
                h_new = h - alpha * grad
                h_new = np.clip(h_new, 0, 1)
                h_new_sum = np.sum(h_new)
                h_new = h_new * (n_points / 2) / h_new_sum
                f_new = evaluate_erdos_solution(h_new)
                if f_new < evaluate_erdos_solution(h):
                    h = h_new
                alpha *= gamma
                if time.time() - start_time > spsa_time:
                    break
            current_c5 = evaluate_erdos_solution(h)
            current_h = h
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n = n_points
        
        total_time_used += alloc_time
    
    return (best_h, best_c5, best_n)
