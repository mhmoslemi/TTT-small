    for gen in range(100):
        if time.time() - start > budget_s:
            break
        fit = [np.max(np.correlate(ind, 1 - ind, mode='full')) * (2.0 / n_points) for ind in pop]
        parents = []
        for _ in range(pop_size):
            t = np.random.choice(range(pop_size), 3, replace=False)
            best = t[np.argmin(fit[t])]
            parents.append(pop[best])
        children = []
        for i in range(0, pop_size, 2):
            p1, p2 = parents[i], parents[i+1]
            c = np.random.choice([p1, p2], size=n_points)
            s = np.sum(c)
            t = n_points / 2
            if s != 0:
                c = c * (t / s)
            c = np.clip(c, 0, 1)
            children.append(c)
        for i in range(len(children)):
            c = children[i]
            m = np.random.rand(n_points) < 0.1
            if np.sum(m) > 0:
                c[m] += np.random.normal(0, 0.05, np.sum(m))
            c = np.clip(c, 0, 1)
            s = np.sum(c)
            if s != 0:
                c = c * (t / s)
            children[i] = np.clip(c, 0, 1)
        pop = children
        for ind in pop:
            f = np.max(np.correlate(ind, 1 - ind, mode='full')) * (2.0 / n_points)
            if f < best_c5:
                best_c5 = f
                best_h = ind
    return (best_h, best_c5, n_points)
