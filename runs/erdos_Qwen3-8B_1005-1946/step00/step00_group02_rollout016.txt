            if best_h is not None:
                x_new = np.linspace(0, 2, n)
                x_old = np.linspace(0, 2, best_n)
                h0 = np.interp(x_new, x_old, best_h)
            else:
                h0 = np.random.dirichlet(np.ones(n), size=1)[0] * (n/2)
                h0 = np.clip(h0, 0, 1)

        h0 = np.clip(h0, 0, 1)
        current_sum = np.sum(h0)
        if abs(current_sum - n/2) > 1e-5:
            h0 = h0 * (n/2) / current_sum

        res = minimize(objective, h0, method='SLSQP', bounds=[(0,1)]*n, 
                       constraints={'type': 'eq', 'fun': constraint}, 
                       jac=grad_objective, options={'maxiter': 100, 'maxeval': 1000})
        if res.success:
            current_c5 = res.fun
            if best_c5 is None or current_c5 < best_c5:
                best_h = res.x
                best_c5 = current_c5
                best_n = n

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, best_n
