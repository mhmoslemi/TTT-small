        if res.success and res.fun < best_c5 or best_c5 is None:
            best_h = res.x
            best_c5 = res.fun
            best_n = n

        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    return best_h, best_c5, best_n
