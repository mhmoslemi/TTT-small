import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points

    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h):
        return compute_C5(h)

    def transfer_up(h, n_points):
        return np.repeat(h, 2)

    best_h = h.copy()
    best_c5 = compute_C5(h)
    start_time = time.time()

    def optimize(n_points, h_initial):
        nonlocal best_h, best_c5
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        return result.success

    for _ in range(4):
        if time.time() - start_time > budget_s:
            break
        if optimize(n_points, best_h):
            best_h = transfer_up(best_h, n_points)
