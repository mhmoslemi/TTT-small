    def repair(h, n_points):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) < 1e-5:
            return h
        scale = target_sum / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
        delta = (target_sum - np.sum(h)) / (n_points - np.sum(h == 1))
        h[h < 1] += delta
        h = np.clip(h, 0, 1)
        return h
    
    def objective(h):
        h = repair(h, n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        c5 = np.max(corr) * dx
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-5:
            return max(c5, 1e6)
        return c5
    
    bounds = [(0, 1)] * n_points
    result = minimize(objective, h_initial, method='differential_evolution', bounds=bounds, options={'maxiter': 100})
    best_h = result.x
    best_h_repaired = repair(best_h, n_points)
    corr = np.correlate(best_h_repaired, 1 - best_h_repaired, mode='full')
    dx = 2.0 / n_points
    best_c5 = np.max(corr) * dx
    return (best_h_repaired, best_c5, n_points)
