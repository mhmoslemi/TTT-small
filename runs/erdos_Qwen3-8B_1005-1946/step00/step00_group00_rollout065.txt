    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    h_final = np.clip(h_scaled, 0, 1)
    current_sum = np.sum(h_final)
    if abs(current_sum - target_sum) > 1e-5:
        diff = target_sum - current_sum
        h_final += diff / n_points
        h_final = np.clip(h_final, 0, 1)
    return h_final

def generate_initial(n_points):
    h = np.random.rand(n_points)
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    target_sum = n_points / 2
    scale = target_sum / current_sum
    h_scaled = h_clipped * scale
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n_points = 45
        initial_h = generate_initial(n_points)
    dx = 2.0 / n_points
    def objective(h):
        h_repaired = repair(h, n_points)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        current_sum = np.sum(h_repaired)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-5:
            return 1e6
        return c5
    bounds = [(0, 1)] * n_points
    start_time = time.time()
    def callback(x):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            raise KeyboardInterrupt
    try:
        result = optimize.minimize(objective, initial_h, method='CMA', bounds=bounds, options={'popsize': 45, 'maxiter': 100}, callback=callback)
    except KeyboardInterrupt:
        pass
    best_h = result.x
    c5 = objective(best_h)
    return (best_h, c5, n_points)
