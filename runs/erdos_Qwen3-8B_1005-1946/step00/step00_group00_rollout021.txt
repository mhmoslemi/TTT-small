        candidates = []
        for _ in range(num_candidates):
            h = np.random.rand(n_points)
            total = np.sum(h)
            h = h * (n_points / 2) / total
            candidates.append(h)
        for candidate in candidates:
            current_h = candidate.copy()
            current_c5 = evaluate_erdos_solution(current_h)
            temp = 1.0
            cooling_rate = 0.995
            max_iter = 1000
            iter_count = 0
            while iter_count < max_iter:
                current_time = time.time()
                if current_time - start_time > budget_s:
                    break
                delta = np.random.uniform(-0.1, 0.1, n_points)
                h_new = current_h + delta
                h_new = np.clip(h_new, 0, 1)
                h_new_sum = np.sum(h_new)
                if h_new_sum < n_points / 2 - 1e-5:
                    scale = (n_points / 2) / h_new_sum
                    h_new = h_new * scale
                elif h_new_sum > n_points / 2 + 1e-5:
                    scale = (n_points / 2) / h_new_sum
                    h_new = h_new * scale
                new_c5 = evaluate_erdos_solution(h_new)
                delta_energy = new_c5 - current_c5
                if delta_energy < 0:
                    accept = True
                else:
                    accept = np.random.rand() < np.exp(-delta_energy / temp)
                if accept:
                    current_h = h_new.copy()
                    current_c5 = new_c5
                temp *= cooling_rate
                iter_count += 1
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n_points = n_points
    return (best_h, best_c5, best_n_points)
