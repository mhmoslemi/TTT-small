        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def perturb_and_adjust(h, target_sum, std=0.1, num_perturb=10):
        h_perturbed = h.copy()
        indices = np.random.choice(len(h_perturbed), size=num_perturb, replace=False)
        h_perturbed[indices] += np.random.normal(0, std, size=num_perturb)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum == 0:
            return h_perturbed
        scale = target_sum / current_sum
        h_scaled = h_perturbed * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled
    
    for n in resolutions:
        if n == current_resolution:
            continue
        if n < current_resolution:
            pass
        else:
            current_h = np.repeat(current_h, 2)
            current_resolution = n
            target_sum = n / 2
            for _ in range(100):
                h_perturbed = perturb_and_adjust(current_h, target_sum)
                dx = 2.0 / n
                c5 = compute_C5(h_perturbed, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_perturbed.copy()
                elapsed = time.time() - time_start
                if elapsed >= budget_s:
                    break
            elapsed = time.time() - time_start
            if elapsed >= budget_s:
                break
    
    return (best_h, best_c5, len(best_h))
