            method='trust-constr',
            bounds=bounds,
            constraints=cons,
            options={'gtol': 1e-8, 'maxiter': 100}
        )
        h_opt = result.x
        c5_opt = compute_c5(h_opt, n_points)

        # Update best solution
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_points

    return (best_h, best_c5, best_n)
