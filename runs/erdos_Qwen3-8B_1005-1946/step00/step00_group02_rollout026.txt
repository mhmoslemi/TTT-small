def select_tournament(population, fitness, tournament_size=3):
    # Select tournament_size individuals
    indices = np.random.choice(len(population), tournament_size, replace=False)
    # Get their fitness
    tournament_fitness = [fitness[i] for i in indices]
    # Select the one with the lowest fitness (best)
    best_idx = indices[np.argmin(tournament_fitness)]
    return population[best_idx]

def crossover(parent1, parent2):
    # Uniform crossover with blend
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        # Blend between parent1 and parent2
        child[i] = (parent1[i] + parent2[i]) / 2
    return child
