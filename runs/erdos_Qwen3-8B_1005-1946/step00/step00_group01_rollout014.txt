            new_h = current_best.copy()
            new_h[i] = 1
            new_h[j] = 0
            new_c5 = compute_C5(new_h)
            if new_c5 < current_c5:
                current_c5 = new_c5
                current_best = new_h.copy()
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        
        elapsed_resolution = time.time() - start_time
        total_elapsed += elapsed_resolution
        if total_elapsed >= budget_s:
            break
    
    return (best_h, best_c5, len(best_h))
