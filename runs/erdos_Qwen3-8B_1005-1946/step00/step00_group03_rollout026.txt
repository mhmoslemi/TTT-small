        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    x0 = best_h.astype(float)
    def obj_cont(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(obj_cont, x0, method='trust-constr', bounds=bounds, constraints=cons)
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    return (best_h, best_c5, n_points)
