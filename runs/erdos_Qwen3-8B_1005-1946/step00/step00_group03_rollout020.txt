    best_c5 = float('inf')
    best_n = 0
    n_points_list = [70, 140, 280]
    time_per_n = {70: 550, 140: 400, 280: 150}
    
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    remaining_time = budget_s
    
    for n in n_points_list:
        if remaining_time <= 0:
            break
        allocated_time = time_per_n[n]
        n_start = time.time()
        
        if n == 70 and 'initial_h_values' in globals():
            initial_pop = [initial_h_values]
            for _ in range(49):
                h = np.random.randint(0, 2, n)
                repair_binary(h, n)
                initial_pop.append(h)
        else:
            initial_pop = [np.random.randint(0, 2, n) for _ in range(50)]
            for i in range(len(initial_pop)):
                repair_binary(initial_pop[i], n)
        
        population = initial_pop
        for gen in range(1000):
            if time.time() - n_start > allocated_time:
                break
            fitness = [compute_c5(h, n) for h in population]
            best_idx = np.argmin(fitness)
            current_c5 = fitness[best_idx]
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = population[best_idx].copy()
                best_n = n
            
            parents = []
            for _ in range(2):
                t = np.random.choice(len(population), 3, replace=False)
                winner = t[np.argmin(fitness[t])]
                parents.append(population[winner])
            
            if np.random.rand() < 0.7:
                p1, p2 = parents
                pt = np.random.randint(1, n)
                c1 = np.concatenate((p1[:pt], p2[pt:]))
                c2 = np.concatenate((p2[:pt], p1[pt:]))
                repair_binary(c1, n)
                repair_binary(c2, n)
                population = [c1, c2] + population[:-2]
            else:
                population = parents + population[:-2]
            
            for i in range(len(population)):
                if np.random.rand() < 0.01:
                    idx = np.random.randint(n)
                    population[i][idx] ^= 1
                    repair_binary(population[i], n)
        
        if best_h is not None and best_n == n:
            def obj(x):
                return compute_c5(x, n)
            bounds = [(0.0, 1.0) for _ in range(n)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}]
            res = scipy.optimize.minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons)
            if res.success:
                h_cont = res.x
                c5 = compute_c5(h_cont, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_cont.copy()
                    best_n = n
        
        remaining_time -= (time.time() - n_start)
    
    return (best_h, best_c5, best_n)
