            j = np.random.choice(zeros)
            h[i], h[j] = h[j], h[i]
            
            new_c5 = compute_c5(h, dx_list[n_points_list.index(n)])
            delta = new_c5 - current_c5
            
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_c5 = new_c5
                if current_c5 < best_c5_n:
                    best_c5_n = current_c5
                    best_h_n = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
            
            T *= cooling_rate
            iteration += 1
        
        if best_c5_n < best_c5:
            best_c5 = best_c5_n
            best_h = best_h_n.copy()
            best_n = n
    
    return (best_h.tolist(), best_c5, best_n)
