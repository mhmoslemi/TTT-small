import numpy as np
import time
import random

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    d = n_points / 2.0
    if abs(s - d) < 1e-5:
        return h
    h *= d / s
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - d) > 1e-5:
        diff = d - s
        n = int(len(h) * 0.1)
        idx = np.random.choice(len(h), n, replace=False)
        h[idx] += 0.01 * (diff / s) * h[idx]
        h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2):
    child = np.zeros_like(parent1)
    for i in range(len(parent1)):
        if random.random() < 0.5:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return child

def mutate(individual):
    n = len(individual)
    m = int(n * 0.1)
    idx = np.random.choice(n, m, replace=False)
    individual[idx] = np.random.rand(m)
    return individual

def tournament_selection(population, fitness, tsize=3):
    sel = np.random.choice(len(population), tsize, replace=False)
    best = sel[np.argmin([fitness[i] for i in sel])]
    return population[best]

def run_ga(n_points, time_limit, initial_pop=None):

        pop = [np.random.rand(n_points) for _ in range(50)]
    else:
        pop = initial_pop[:]

        pop = pop[:50]
    fit = [evaluate_erdos_solution(ind) for ind in pop]

    best_fit = fit[best_idx]
    best_ind = pop[best_idx]
    start = time.time()
    while time.time() - start < time_limit:
        new_pop = []
        for _ in range(50):
            p1 = tournament_selection(pop, fit)
            p2 = tournament_selection(pop, fit)
            child = crossover(p1, p2)
            if random.random() < 0.3:
                child = mutate(child)
            child = repair(child, n_points)
