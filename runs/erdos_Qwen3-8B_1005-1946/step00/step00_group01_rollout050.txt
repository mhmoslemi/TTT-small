    best_c5 = compute_C5(best_h)
    n_points = len(best_h)
    resolutions = [47, 94, 188, 376, 752]
    start_time = time.time()
    
    for n in resolutions:
        if n == 47:
            current_h = best_h.copy()
        else:
            current_h = np.repeat(current_h, 2)
        if len(current_h) != n:
            current_h = np.repeat(current_h, 2)
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
        for _ in range(1000):
            zeros = np.where(current_h == 0)[0]
            ones = np.where(current_h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            i = np.random.choice(zeros)
            j = np.random.choice(ones)
            current_h[i], current_h[j] = 1, 0
            c5 = compute_C5(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
            current_h[i], current_h[j] = 0, 1
            if time.time() - start_time > budget_s:
                break
        current_h = best_h.copy()
    
    sum_h = np.sum(best_h)
    dx = 2.0 / len(best_h)
    assert np.isclose(sum_h * dx, 1.0), "Integral constraint violated"
    return (best_h, best_c5, len(best_h))

def compute_C5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
