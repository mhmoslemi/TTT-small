import numpy as np
from scipy.optimize import minimize
import time

def repair(h, n_points):
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - n_points / 2) < 1e-5:
        return h
    scale = (n_points / 2) / total
    h *= scale
    return h

def objective(h, n_points):
    h_repaired = repair(h, n_points)
    corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    
    result = minimize(
        fun=lambda x: objective(x, n_points),
        x0=h_initial,
        method='CMA',
        bounds=bounds,
        options={
            'maxiter': 100,
            'popsize': 45,
            'seed': seed,
            'ftol': 1e-8,
            'gtol': 1e-8
        }
    )
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_initial, objective(h_initial, n_points), n_points)
    
    best_h = result.x
    best_h_repaired = repair(best_h, n_points)
    c5_bound = objective(best_h_repaired, n_points)
    
    return (best_h_repaired, c5_bound, n_points)
