    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()
    res = minimize(objective, h, method='trust-constr', bounds=bounds, constraints=cons, options={'maxiter': 100})
    elapsed = time.time() - start_time

    best_h = res.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
