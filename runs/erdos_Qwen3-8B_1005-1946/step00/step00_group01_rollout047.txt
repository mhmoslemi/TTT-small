    new_h = np.clip(new_h, 0.0, 1.0)
    current_sum = np.sum(new_h)
    target_sum = n / 2.0
    diff = current_sum - target_sum
    adjust_idx = np.random.randint(n)
    new_h[adjust_idx] -= diff
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    best_h = initial_h.copy()
    best_c5 = compute_C5(best_h)
    n_points = len(best_h)
    start_time = time.time()
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        new_h = perturb(best_h)
        new_c5 = compute_C5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    return best_h, best_c5, n_points
