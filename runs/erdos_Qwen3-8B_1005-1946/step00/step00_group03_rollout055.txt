        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * (2.0 / n)
    for n_points in n_points_list:
        if n_points == 70:
            h = initial_h_values.copy()
        else:
            indices = np.random.choice(n_points, size=n_points//2, replace=False)
            h = np.zeros(n_points, dtype=int)
            h[indices] = 1
        current_c5 = compute_c5(h, n_points)
        best_c5_for_n = current_c5
        best_h_for_n = h.copy()
        T = 10.0
        cooling_rate = 0.95
        start_time = time.time()
        while True:
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if not ones.size or not zeros.size:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h, n_points)
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_c5 = new_c5
                if current_c5 < best_c5_for_n:
                    best_c5_for_n = current_c5
                    best_h_for_n = h.copy()
            T *= cooling_rate
            if T < 1e-5:
                break
        if best_c5_for_n < best_c5:
            best_c5 = best_c5_for_n
            best_h = best_h_for_n.copy()
    return (best_h.astype(float), best_c5, len(best_h))
