import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points_list = [70, 140]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    time_remaining = budget_s

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / len(h)
        return np.max(corr) * dx

    def repair(h, n):
        s = np.sum(h)
        t = n / 2
        delta = s - t
        if delta == 0:
            return h
        h = h.copy()
        if delta > 0:
            ones = np.where(h == 1)[0]
            if len(ones) >= delta:
                h[ones[:delta]] = 0
        else:
            zeros = np.where(h == 0)[0]
            if len(zeros) >= -delta:
                h[zeros[: -delta]] = 1
        return h

    def generate_initial(n):
        h = np.zeros(n, dtype=int)
        h[:n//2] = 1
        return h

    def genetic_algorithm(n, initial=None):
        pop_size = 50
        gens = 1000
        tournament = 3
        cross = 0.7
        mut = 0.01
        pop = np.array([initial] if initial is not None else [generate_initial(n) for _ in range(pop_size)])
        best_h = pop[0]
        best_c5 = compute_c5(best_h)
        for _ in range(gens):
            fit = np.array([compute_c5(ind) for ind in pop])
            best_idx = np.argmin(fit)
            if fit[best_idx] < best_c5:
                best_c5 = fit[best_idx]
                best_h = pop[best_idx]
            parents = []
            for _ in range(pop_size // 2):
                idxs = np.random.choice(pop_size, tournament)
                best = idxs[np.argmin(fit[idxs])]
                parents.append(pop[best])
            new_pop = []
            for i in range(0, len(parents), 2):
                p1, p2 = parents[i], parents[i+1] if i+1 < len(parents) else parents[i]
                if np.random.rand() < cross:
                    pt = np.random.randint(1, n)
                    c1 = np.concatenate([p1[:pt], p2[pt:]])
                    c2 = np.concatenate([p2[:pt], p1[pt:]])
                else:
                    c1, c2 = p1.copy(), p2.copy()
                new_pop.extend([repair(c1, n), repair(c2, n)])
            for i in range(len(new_pop)):
                if np.random.rand() < mut:
                    idx = np.random.randint(n)
                    new_pop[i][idx] ^= 1
                new_pop[i] = repair(new_pop[i], n)
            pop = np.array(new_pop)
        return best_h, best_c5

    for n in n_points_list:

            dx = 2.0 / n
            return np.max(corr) * dx

        bounds = [(0.0, 1.0) for _ in range(n)]
