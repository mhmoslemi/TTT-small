import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points_list = [45, 90, 180, 360]
    best_h = initial_h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / len(h))
    
    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        k = np.argmax(corr)
        grad = np.zeros_like(h)
        n = len(h)
        for i in range(n):
            if k >= 0:
                j = i + k
                if j < n:
                    grad[i] += (1 - h[j])
                    grad[j] -= h[i]
            else:
                j = i + k
                if j >= 0:
                    grad[i] += (1 - h[j])
                    grad[j] -= h[i]
        return grad
    
    def optimize(n_points):
        if len(best_h) < n_points:
            h = np.repeat(best_h, 2)[:n_points]
        else:
            h = best_h.copy()
        sum_h = np.sum(h)
        target_sum = n_points / 2
        if abs(sum_h - target_sum) > 1e-6:
            scale = target_sum / sum_h
            h *= scale
        def objective(x):
            return compute_c5(x)
        def grad(x):
            return compute_gradient(x)
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            jac=grad,
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        return best_h, best_c5
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        best_h, best_c5 = optimize(n_points)
    
    return (best_h, best_c5, len(best_h))
