def run(seed=42, budget_s=1100, **kwargs):

    best_c5 = None

        corr = np.correlate(h, 1 - h, mode='full')

        return max_corr_val * (2.0 / n)

    def grad_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_indices = np.flatnonzero(corr == np.max(corr))
        lags = [idx - (n - 1) for idx in max_indices]
        grad = np.zeros_like(h)
        for lag in lags:
            if lag >= 0:

                    grad[j] += (1 - h[j + lag])
            else:
                k = -lag
                for j in range(k, n):
                    grad[j] += (1 - h[j - k])
        if lags:
            grad /= len(lags)
        return grad * (2.0 / n)

    def constraint(h):
        return np.sum(h) - n / 2

    n_list = [45, 90, 180, 360]
