import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = kwargs.get('initial_h_values', None)
    n_points_list = [70, 140, 280]
    best_c5 = float('inf')
    best_h = None
    best_n = 70
    
    for n in n_points_list:
        if n == 70 and initial_h is not None:
            h = initial_h.copy()
        else:
            h = np.zeros(n, dtype=int)
            indices = np.random.choice(n, size=n//2, replace=False)
            h[indices] = 1
        
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * (2.0 / len(h))
        
        c5_val = objective(h)
        best_c5 = c5_val
        best_h = h
        best_n = n
        
        T = 10.0
        cooling_rate = 0.95
        max_iter = 1000
        start_time = time.time()
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            
            i1 = np.random.choice(zeros)
            i2 = np.random.choice(ones)
            
            h[i1], h[i2] = 1, 0
            new_c5 = objective(h)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h
                best_n = n
            else:
                prob = np.exp(-(new_c5 - best_c5) / T)
                if np.random.random() < prob:
                    best_c5 = new_c5
                    best_h = h
                    best_n = n
            
            T *= cooling_rate
        
        if best_c5 < 0.38085:
            break
    
    return best_h, best_c5, best_n
