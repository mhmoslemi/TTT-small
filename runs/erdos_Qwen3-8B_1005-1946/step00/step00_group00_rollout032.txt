import numpy as np
from scipy.optimize import differential_evolution
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - n_points / 2) < 1e-5:
            return h
        scale = (n_points / 2) / s
        h = h * scale
        return h
    
    def objective(h):
        h = repair(h)
        s = np.sum(h)
        if abs(s - n_points / 2) > 1e-5:
            return 1e6
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    bounds = [(0.0, 1.0)] * n_points
    start_time = time()
    result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed, maxiter=1000)
    elapsed = time() - start_time
    if elapsed > budget_s:
        return h, objective(h), n_points
    h_best = result.x
    c5_bound = objective(h_best)
    return h_best, c5_bound, n_points
