            child1[np.random.randint(n_points)] ^= 1
        if np.random.random() < 0.01:
            child2[np.random.randint(n_points)] ^= 1
        child1 = repair(child1, n_ones)
        child2 = repair(child2, n_ones)
        c5_val1 = objective(child1)
        c5_val2 = objective(child2)
        if c5_val1 < best_local_c5:
            best_local_c5 = c5_val1
            local_best = child1.copy()
        if c5_val2 < best_local_c5:
            best_local_c5 = c5_val2
            local_best = child2.copy()
        population = [child1, child2] + population[:-2]

    if time.time() - start_time < budget_s:
        def constraint_sum(h):
            return np.sum(h) - n_ones
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1)] * n_points
        res = minimize(objective, local_best, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            best_h = res.x
            best_c5 = res.fun
        else:
            best_h = local_best
            best_c5 = best_local_c5

    return best_h, best_c5, n_points
