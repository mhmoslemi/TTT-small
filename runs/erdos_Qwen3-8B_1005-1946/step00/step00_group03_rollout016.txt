            corr = np.correlate(current_h, 1 - current_h, mode='full')
            new_c5 = np.max(corr) * (2.0 / n)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
                best_n = n
            else:
                prob = np.exp(-(new_c5 - best_c5) / T)
                if np.random.random() < prob:
                    best_c5 = new_c5
                    best_h = current_h.copy()
                    best_n = n
            
            T *= cooling_rate
        
        if best_c5 < 0.38085:
            break
    
    return best_h, best_c5, best_n
