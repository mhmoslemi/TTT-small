```python
import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        initial_h = h

    def repair(h, n_points):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) < 1e-5:
            return h_clipped
        scale = target_sum / current_sum
        return h_clipped * scale

    def objective(h):
        h_repaired = repair(h, len(h))
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        dx = 2.0 / len(h)
        max_corr = np.max(corr)
        return max_corr * dx

    bounds = [(0, 1) for _ in range(len(initial_h))]
    start_time = time.time()
    result = differential_evolution(objective, bounds, popsize=15, maxiter=100, tol=1e-5, seed=seed)
    elapsed_time = time.time() - start_time

    best_h = result.x
    best_h_repaired = repair(best_h, len(best_h))
    best_c5 = objective(best_h_repaired)

    n_points = len(best_h_repaired)
    return (best_h_repaired, best_c5, n_points)
```
