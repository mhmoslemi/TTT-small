        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, tol=1e-6)
        h = result.x
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 1e-12:
            h = h * (n_points / 2) / h_sum
        new_c5 = compute_c5(h, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        if np.abs(new_c5 - best_c5) < 1e-6:
            break
    
    return best_h, best_c5, n_points
