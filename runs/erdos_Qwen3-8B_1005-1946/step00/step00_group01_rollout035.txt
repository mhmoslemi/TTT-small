import numpy as np
import time

def transfer_up(h):
    return np.repeat(h, 2)

def compute_C5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 47
        h = np.random.rand(n_points)
        h = np.round(h * 2).astype(int)
        while np.sum(h) * 2 != n_points:
            h = np.random.rand(n_points)
            h = np.round(h * 2).astype(int)
    resolutions = [47, 94, 188, 376, 752]
    best_c5 = float('inf')
    best_h = h.copy()
    time_start = time.time()
    for n in resolutions:
        if n == 47:
            current_h = h.copy()
        else:
            current_h = transfer_up(best_h)
        time_remaining = budget_s - (time.time() - time_start)
        if time_remaining <= 0:
            break
        for _ in range(1000):
            zeros = np.where(current_h == 0)[0]
            ones = np.where(current_h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            i = np.random.choice(zeros)
            j = np.random.choice(ones)
            current_h[i], current_h[j] = 1, 0
            c5 = compute_C5(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
