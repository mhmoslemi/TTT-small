        current_h = transfer_up(current_h)
        current_n = res
        optimized_h = optimize_resolution(current_h, current_n)
        current_c5 = compute_C5(optimized_h, current_n)
        if current_c5 < best_c5:
            best_h = optimized_h.copy()
            best_c5 = current_c5
            best_n = current_n
        time_used = time.time() - start_time
        if time_used >= budget_s:
            break

    return (best_h, best_c5, best_n)
