    def generate_initial(n_points):
        if n_points == 45:
            return initial_h if initial_h is not None else np.random.rand(n_points)
        elif n_points % 45 == 0:
            base = initial_h if initial_h is not None else np.random.rand(45)
            expanded = np.repeat(base, 2)
            scale = (n_points / 2) / np.sum(expanded)
            return expanded * scale
        else:
            h = np.random.rand(n_points)
            scale = (n_points / 2) / np.sum(h)
            return h * scale
    
    def evaluate(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def local_search(h, n_points):
        def objective(x):
            h_new = x
            dx = 2.0 / n_points
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            return np.max(corr) * dx
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return result.x
    
    for n_points in n_points_list:
        if time_used >= budget_s:
            break
        if n_points == 45 and initial_h is None:
            h = np.random.rand(n_points)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            h = np.clip(h, 0, 1)
        else:
            h = generate_initial(n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            scale = (n_points / 2) / sum_h
            h *= scale
        h = np.clip(h, 0, 1)
        
        pop_size = 50
        generations = 100
        mutation_rate = 0.1
        population = [h.copy() for _ in range(pop_size)]
        
        for gen in range(generations):
            if time_used >= budget_s:
                break
            fitness = [evaluate(ind) for ind in population]
            parents = [population[np.argmin(fitness)] for _ in range(pop_size)]
            
            new_population = []
            for i in range(0, pop_size, 2):
                p1, p2 = parents[i], parents[i+1]
                child1 = (p1 + p2) / 2

                scale = (n_points / 2) / sum_h
                h = np.clip(h * scale, 0, 1)
                new_population[i] = h
            
            population = new_population
            current_best = min(population, key=evaluate)
            current_c5 = evaluate(current_best)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_best.copy()

            scale = (len(best_h) / 2) / sum_h
            best_h *= scale
        best_h = np.clip(best_h, 0, 1)
    
    return (best_h, best_c5, len(best_h))
