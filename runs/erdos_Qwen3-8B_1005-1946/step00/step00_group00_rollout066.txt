import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        n_points = len(initial_h)
    else:
        n_points = 45
        h = np.random.rand(n_points)

        h = h / np.sum(h) * target_sum
        initial_h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def repair(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        s = np.sum(h_clipped)
        if abs(s - target_sum) < 1e-5:
            return h_clipped
        scale = target_sum / s
        h_scaled = h_clipped * scale
        h_final = np.clip(h_scaled, 0, 1)
        s_final = np.sum(h_final)
        if abs(s_final - target_sum) < 1e-5:
            return h_final
        return h_final
    
    def objective(h_vec):
        h_repaired = repair(h_vec)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        c5 = np.max(corr) * dx
        s = np.sum(h_repaired)
        penalty = 1e6 * abs(s - target_sum)
        return max(c5, penalty)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(objective, bounds, popsize=15, maxiter=100, tol=1e-5, seed=seed)
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        best_h = result.x
        best_c5 = objective(best_h)
        h_repaired = repair(best_h)
        return (h_repaired, best_c5, n_points)
