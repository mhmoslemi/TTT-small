                    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                    new_population.extend([child1, child2])
                else:
                    new_population.append(parents[i])
            for i in range(len(new_population)):
                h = new_population[i]
                if np.random.rand() < mutation_rate:
                    idx = np.random.randint(n)
                    h[idx] = 1 - h[idx]
                    h = repair_binary(h, n)
                new_population[i] = h
            population = new_population
        best_idx = np.argmin([evaluate_c5(h, n, dx) for h in population])
        return population[best_idx]

    for n in n_points_list:
        dx = 2.0 / n
        if n == 70 and 'initial_h_values' in globals():
            initial_h = np.array(initial_h_values)
            if len(initial_h) != n:
                initial_h = np.random.rand(n)
                initial_h = np.round(initial_h).astype(int)
                while np.sum(initial_h) != n // 2:
                    initial_h[0] = 1 - initial_h[0]
        else:
            initial_h = np.random.rand(n)
            initial_h = np.round(initial_h).astype(int)
            while np.sum(initial_h) != n // 2:
                initial_h[0] = 1 - initial_h[0]
        best_h_binary = run_ga(n, initial_h)
        h_initial = best_h_binary.astype(float)
        def objective_cont(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        def constraint_sum(h):
            return np.sum(h) - n / 2
        bounds = [(0, 1) for _ in range(n)]
        result = opt.minimize(objective_cont, h_initial, method='trust-constr', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum})
        h_opt = result.x
        c5_opt = evaluate_c5(h_opt, n, dx)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n
    return (best_h, best_c5, best_n)
