import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals():
        h_initial = initial_h_values
        n_initial = len(h_initial)

    else:
        h_initial = np.random.rand(45)
        h_initial = np.clip(h_initial, 0, 1)
        s = np.sum(h_initial)
        scale = 45 / 2 / s
        h_initial = h_initial * scale
        h_initial = np.clip(h_initial, 0, 1)
        n_points_list = [45, 30, 20, 10]
    
    best_h = h_initial
    best_c5 = float('inf')
    elapsed = 0
    current_n = 45
    
    def repair(h, n):
        h = np.clip(h, 0, 1)

        scale = (n / 2) / s
        h = h * scale
        h = np.clip(h, 0, 1)
        return h
    
    for n in n_points_list:
        if n == current_n:

        x_current = np.linspace(0, 2, current_n)

        h_new = np.interp(x_new, x_current, best_h)
        h_new = repair(h_new, n)
        bounds = [(0, 1) for _ in range(n)]
