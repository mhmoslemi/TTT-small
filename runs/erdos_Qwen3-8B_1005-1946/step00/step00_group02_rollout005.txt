from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    best_h = None
    best_c5 = None
    best_n = 45
    elapsed_time = 0

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_val = np.max(corr)
        return max_corr_val * (2.0 / n)

    def grad_objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_indices = np.flatnonzero(corr == max_val)
        lags = [idx - (n - 1) for idx in max_indices]
        grad = np.zeros_like(h)
        for lag in lags:
            if lag >= 0:
                for j in range(n):
                    if j + lag < n:
                        grad[j] += (1 - h[j + lag])
                    if j - lag >= 0:
                        grad[j] += -h[j - lag]
            else:
                k = -lag
                for j in range(n):
                    if j + k < n:
                        grad[j] += (1 - h[j + k])
                    if j - k >= 0:
                        grad[j] += -h[j - k]
        if lags:
            grad /= len(lags)
        return grad * (2.0 / n)

    def constraint(h):
        return np.sum(h) - n / 2

    n_list = [45, 90, 180, 360]
    for n in n_list:
        start_time = time.time()
        if n == 45 and initial_h_values is not None:
            h0 = initial_h_values.copy()
        else:
            if best_h is not None:
                x = np.linspace(0, 2, best_n)
                x_new = np.linspace(0, 2, n)
                h0 = np.interp(x_new, x, best_h)
            else:
                h0 = np.random.dirichlet(np.ones(n), size=1)[0] * (n / 2)
                h0 = np.clip(h0, 0, 1)
        h0 = np.clip(h0, 0, 1)
        current_sum = np.sum(h0)
        if abs(current_sum - n / 2) > 1e-5:
            h0 = h0 * (n / 2) / current_sum
        res = minimize(objective, h0, method='SLSQP', bounds=[(0, 1)] * n,

        if res.success and (best_c5 is None or res.fun < best_c5):
