import numpy as np
import time

def transfer_up(h):
    return np.repeat(h, 2)

def compute_C5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    resolutions = [47, 94, 188, 376, 752]
    best_c5 = float('inf')
    best_h = h.copy()
    best_n = len(h)
    elapsed = 0.0
    start_time = time.time()
    
    for n in resolutions:
        h = transfer_up(h)
        current_n = n
        while elapsed < budget_s:
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if not zeros.size or not ones.size:
                break
            i, j = np.random.choice(zeros), np.random.choice(ones)
            h_new = h.copy()
            h_new[i], h_new[j] = 1, 0
            c5 = compute_C5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                best_n = current_n
    return (best_h, best_c5, best_n)
