                while np.sum(h) != n//2:
                    h = rs.choice([0, 1], size=n)
                population.append(h)
        else:
            population = [rs.choice([0, 1], size=n) for _ in range(50)]
            for i in range(len(population)):
                h = population[i]
                while np.sum(h) != n//2:
                    h = rs.choice([0, 1], size=n)
                population[i] = h

        def evaluate(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * (2.0 / n)

        best_h_current = population[0]
        best_c5_current = evaluate(best_h_current)

        for gen in range(100):
            parents = []
            for _ in range(2):
                tournament = rs.choice(population, size=3)
                winner = min(tournament, key=evaluate)
                parents.append(winner)

            if rs.rand() < 0.7:
                point = rs.randint(1, n-1)
                parent1, parent2 = parents
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
                child1 = repair_binary(child1, n)
                child2 = repair_binary(child2, n)
                parents = [child1, child2]
            else:
                parents = [parents[0], parents[1]]

            for i in range(2):
                h = parents[i]
                if rs.rand() < 0.01:
                    idx = rs.randint(n)
                    h[idx] = 1 - h[idx]
                    h = repair_binary(h, n)
                parents[i] = h

            population = parents + population[:len(population)-2]

            for h in population:
                c5 = evaluate(h)
                if c5 < best_c5_current:
                    best_c5_current = c5
                    best_h_current = h

        elapsed_time += time.time() - time_start
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
            best_n = n

    return (best_h, best_c5, best_n)
