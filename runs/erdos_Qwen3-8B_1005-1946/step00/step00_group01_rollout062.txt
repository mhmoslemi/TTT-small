import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # List of resolutions to try
    resolutions = [47, 94, 188, 376, 752]
    start_time = time.time()
    
    for res in resolutions:
        # Transfer current h to the new resolution
        if res > n_points:
            repeat_times = res // n_points
            h = np.repeat(h, repeat_times)
        n_points = res
        
        # Check current h
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Allocate time for this resolution
        res_time_start = time.time()
        while time.time() - res_time_start < 100 and time.time() - start_time < budget_s:
            # Perturb 10 random indices
            indices = np.random.choice(n_points, 10, replace=False)
            noise = np.random.normal(0, 0.1, 10)
            h_perturbed = h.copy()
            h_perturbed[indices] += noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum = np.sum(h_perturbed)
            if current_sum == 0:
                continue  # Avoid division by zero
            scale_factor = (n_points / 2) / current_sum
            h_perturbed *= scale_factor
            
            # Evaluate C5
            c5 = evaluate_erdos_solution(h_perturbed)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturbed.copy()
    
    return (best_h, best_c5, len(best_h))
