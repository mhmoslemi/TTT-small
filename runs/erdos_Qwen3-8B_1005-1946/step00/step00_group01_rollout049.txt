        # Define objective function
        def objective(h):
            dx = 2.0 / len(h)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx

        # Bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint(h):
            return np.sum(h) - n_points / 2
        cons = [{'type': 'eq', 'fun': constraint}]

        # Optimization
        result = minimize(objective, h_transfer, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if result.success:
            current_h = result.x
            current_c5 = result.fun
        else:
            current_h = h_transfer
            current_c5 = compute_C5(current_h, 2.0 / len(current_h))

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            best_n = n_points

    return (best_h, best_c5, best_n)
