import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points_list = [70, 140, 280]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    start_time = time.time()
    
    np.random.seed(seed)
    random.seed(seed)
    
    def objective(h, n_points):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    for n in n_points_list:
        if n == 70 and 'initial_h_values' in globals():
            h = np.array(initial_h_values, dtype=np.float64)
        else:
            h = np.zeros(n, dtype=np.int64)
            h[:n//2] = 1
            np.random.shuffle(h)
            h = h.astype(np.float64)
        
        current_c5 = objective(h, n)
        T = 10.0
        cooling_rate = 0.95
        max_iter = 1000
        iter_count = 0
        
        best_h_current = h.copy()
        best_c5_current = current_c5
        
        while iter_count < max_iter and time.time() - start_time < budget_s:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            h[i], h[j] = h[j], h[i]
            
            new_c5 = objective(h, n)
            
            if new_c5 < current_c5:
                current_c5 = new_c5
            else:
                delta = new_c5 - current_c5
                if np.random.rand() < np.exp(-delta / T):
                    current_c5 = new_c5
            
            if current_c5 < best_c5_current:
                best_c5_current = current_c5
                best_h_current = h.copy()
            
            T *= cooling_rate
            iter_count += 1
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
            best_n = n
    
    if best_h is None:
        best_h = np.array([0.5]*70)
    
    return (best_h, best_c5, best_n)
