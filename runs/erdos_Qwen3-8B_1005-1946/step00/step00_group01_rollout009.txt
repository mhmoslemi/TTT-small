import numpy as np
import numpy.random as npr
import time

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

            current_h = h.copy()

            current_h = transfer_up(best_h, factor=2)
        current_best = current_h.copy()
        current_best_c5 = compute_C5(current_best)

            elapsed = time.time() - start_time
