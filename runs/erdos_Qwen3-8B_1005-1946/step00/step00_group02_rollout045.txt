from scipy.optimize import minimize
import time

    if 'initial_h_values' in globals():
        h = initial_h_values
        n_points = len(h)

        n_points = 45

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    def optimize(n_points, h_initial):
        result = minimize(compute_c5, h_initial, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)

        return h_initial, compute_c5(h_initial)
    
    start_time = time.time()
