        while time.time() - start_time < allocated_time:
            new_pop = []
            for _ in range(len(population) // 2):
                t = np.random.choice(population, 3, replace=False)
                best_t = min(t, key=evaluate)
                new_pop.append(best_t)
            for i in range(len(new_pop) // 2):
                p1, p2 = new_pop[i], new_pop[-i-1]
                if np.random.rand() < 0.7:
                    pt = np.random.randint(1, n)
                    c1 = np.concatenate([p1[:pt], p2[pt:]])
                    c2 = np.concatenate([p2[:pt], p1[pt:]])
                    new_pop.append(c1)
                    new_pop.append(c2)
                else:
                    new_pop.append(p1)
                    new_pop.append(p2)
            for i in range(len(new_pop)):
                h = new_pop[i]
                if np.random.rand() < 0.01:
                    idx = np.random.randint(n)
                    h[idx] ^= 1
                    s = np.sum(h)
                    t = n // 2
                    if s != t:
                        if s < t:
                            z = np.where(h == 0)[0]
                            f = np.random.choice(z, size=t - s, replace=False)
                            h[f] = 1
                        else:
                            o = np.where(h == 1)[0]
                            f = np.random.choice(o, size=s - t, replace=False)
                            h[f] = 0
            for h in new_pop:
                c5 = evaluate(h)
                if c5 < best_c5_ga:
                    best_c5_ga = c5
                    best_h_ga = h
            population = new_pop

        h_continuous = best_h_ga.astype(float)
        def obj(x):
            h = x.reshape(n)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * (2.0 / n)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
        bounds = [(0, 1)] * n
        res = minimize(obj, h_continuous, method='trust-constr', bounds=bounds, constraints=cons, options={'gtol': 1e-8, 'maxiter': 100})
        if res.success:
            h_opt = res.x
            c5_opt = res.fun
        else:
            h_opt = h_continuous
            c5_opt = evaluate(h_continuous)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n
        time_remaining -= (time.time() - start_time)

    return best_h, best_c5, best_n
