import numpy as np
from scipy.optimize import minimize
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    n_points_list = [70, 140]
    total_time = 0.0

    def compute_c5(h):
        n = len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)

    def repair(ind, n):
        s = np.sum(ind)
        target = n // 2
        diff = s - target
        if diff != 0:
            if diff > 0:
                ones = np.where(ind == 1)[0]
                flip = np.random.choice(ones, diff, replace=False)
                ind[flip] = 0
            else:
                zeros = np.where(ind == 0)[0]
                flip = np.random.choice(zeros, -diff, replace=False)
                ind[flip] = 1
        return ind

    for n in n_points_list:
        if total_time >= budget_s:
            break
        if n == best_n:
            continue
        start_time = time.time()
        pop_size = 50
        generations = 100
        tournament_size = 3
        crossover_prob = 0.7
        mutation_prob = 0.01
        if n == 70:
            pop = [best_h.copy()]
        else:
            pop = []
            for _ in range(pop_size):
                vec = np.zeros(n, dtype=int)
                vec[:n//2] = 1
                np.random.shuffle(vec)
                pop.append(vec)
        for _ in range(generations):
            fit = [compute_c5(ind) for ind in pop]
            parents = []
            for _ in range(pop_size):
                idxs = np.random.choice(pop_size, tournament_size)
                best_idx = idxs[np.argmin(fit[idxs])]
                parents.append(pop[best_idx])
            new_pop = []
            for i in range(0, pop_size, 2):

                    c1 = np.concatenate([p1[:pt], p2[pt:]])
                    c2 = np.concatenate([p2[:pt], p1[pt:]])
                    new_pop.extend([c1, c2])
                else:
                    new_pop.extend([parents[i], parents[i+1]])
            for i in range(len(new_pop)):
