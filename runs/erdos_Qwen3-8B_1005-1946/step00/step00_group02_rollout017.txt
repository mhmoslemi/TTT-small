import random

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    best_h = None
    best_c5 = float('inf')
    n_points_list = [45, 90, 180, 360]
    elapsed = 0
    max_generations = 100

    def generate_random_individual(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        diff = (n_points / 2) - sum_h
        h += diff / n_points
        h = np.clip(h, 0, 1)
        return h

    def evaluate_fitness(h, n_points):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    for n_points in n_points_list:
        if elapsed >= budget_s:
            break
        population_size = 50
        population = []
        if initial_h_values is not None and n_points == 45:
            initial_h = initial_h_values
            population = [initial_h]
            for _ in range(population_size - 1):
                individual = generate_random_individual(n_points)
                population.append(individual)
        else:
            population = [generate_random_individual(n_points) for _ in range(population_size)]
        
        population_fitness = [evaluate_fitness(ind, n_points) for ind in population]
        best_idx = np.argmin(population_fitness)
        best_c5 = population_fitness[best_idx]
        best_h = population[best_idx].copy()

        for gen in range(max_generations):
            if elapsed >= budget_s:
                break
            parents = []
            for _ in range(population_size):
                tournament = np.random.choice(population_size, size=3, replace=False)
                best_tournament = tournament[np.argmin(population_fitness[tournament])]
                parents.append(population[best_tournament])
            
            offspring = []
            for i in range(0, population_size, 2):
                parent1 = parents[i]
                parent2 = parents[i+1] if i+1 < len(parents) else parents[0]
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                offspring.append(child1)
                offspring.append(child2)
            
            mutated_offspring = []
            for child in offspring:
                noise = np.random.normal(0, 0.05, child.shape)
                child += noise
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                target_sum = n_points / 2
                diff = target_sum - sum_child
                child += diff / n_points
                child = np.clip(child, 0, 1)
                mutated_offspring.append(child)
            
            population = mutated_offspring
            new_population_fitness = [evaluate_fitness(ind, n_points) for ind in population]
            sorted_indices = np.argsort(new_population_fitness)
            population = [population[i] for i in sorted_indices[:population_size//2]]
            population_fitness = new_population_fitness
            current_best_idx = np.argmin(population_fitness)
            if population_fitness[current_best_idx] < best_c5:
                best_c5 = population_fitness[current_best_idx]
                best_h = population[current_best_idx].copy()
            elapsed += 10

    return (best_h, best_c5, len(best_h))
