import numpy as np
import time
from scipy.optimize import minimize

def repair(ind, n_ones):
    s = np.sum(ind)
    if s == n_ones:
        return ind
    elif s > n_ones:
        excess = s - n_ones
        ones_indices = np.where(ind == 1)[0]
        if len(ones_indices) >= excess:
            ind[ones_indices[:excess]] = 0
        else:
            zeros_indices = np.where(ind == 0)[0]
            num_add = n_ones - (s - len(zeros_indices))
            if num_add > 0:
                ind[zeros_indices[:num_add]] = 1
            ind[ones_indices] = 0
    else:
        zeros_indices = np.where(ind == 0)[0]
        needed = n_ones - s
        if len(zeros_indices) >= needed:
            ind[zeros_indices[:needed]] = 1
        else:
            ones_indices = np.where(ind == 1)[0]
            num_remove = s - (n_ones - len(zeros_indices))
            if num_remove > 0:
                ind[ones_indices[:num_remove]] = 0
            ind[zeros_indices] = 1
    return ind

def objective(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = kwargs.get('initial_h_values')
    best_c5 = float('inf')
    best_h = None
    best_n = 70

    def local_optimize(h):
        def constraint_sum(h_vec):
            return np.sum(h_vec) - len(h_vec) // 2
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1)] * len(h)
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
        return res.x, res.fun

    if initial_h is not None:
        h = initial_h.copy()
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = len(h)
        return best_h, best_c5, best_n

    n_points = 140
    n_ones = n_points // 2
    population = []
    for _ in range(50):
        ind = np.zeros(n_points)
        indices = np.random.choice(range(n_points), n_ones, replace=False)
        ind[indices] = 1
        population.append(ind)

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        parents = []
        for _ in range(2):
            tournament = np.random.choice(range(50), 3)
            winner = tournament[np.argmin([objective(population[i]) for i in tournament])]
            parents.append(population[winner])
        idx = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parents[0][:idx], parents[1][idx:]])
        child2 = np.concatenate([parents[1][:idx], parents[0][idx:]])
        if np.random.random() < 0.01:
            child1[np.random.randint(n_points)] ^= 1
        if np.random.random() < 0.01:
            child2[np.random.randint(n_points)] ^= 1
        child1 = repair(child1, n_ones)
        child2 = repair(child2, n_ones)
        for c in [child1, child2]:
            c5 = objective(c)
            if c5 < best_c5:
                best_c5 = c5
                best_h = c.copy()
                best_n = n_points
        population = [child1, child2] + population[:-2]

    if time.time() - start_time < budget_s:
        h_opt, c5_opt = local_optimize(best_h)

            best_c5 = c5_opt
            best_h = h_opt

    return best_h, best_c5, best_n
