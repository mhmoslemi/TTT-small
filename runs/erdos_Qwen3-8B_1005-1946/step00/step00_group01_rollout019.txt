def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    n_points = len(initial_h_values)
    best_h = initial_h_values.copy()
    best_c5 = compute_C5(best_h)
    resolutions = [47, 94, 188, 376, 752]
    total_time = 0

    def compute_C5(h):
        n = len(h)
        dx = 2.0 / n
        h_1_minus_h = 1 - h
        corr = np.correlate(h, h_1_minus_h, mode='full')
        return np.max(corr) * dx

    for res in resolutions:
        if res != n_points:
            best_h = np.repeat(best_h, 2)
            n_points = res
            best_c5 = compute_C5(best_h)
        start_time = time.time()
        while time.time() - start_time < 100:
            h_candidate = best_h.copy()
            indices = np.random.choice(n_points, 10, replace=False)
            noise = np.random.normal(0, 0.1, 10)
            h_candidate[indices] += noise
            h_candidate = np.clip(h_candidate, 0, 1)
            current_sum = np.sum(h_candidate)
            scale = (n_points / 2) / current_sum
            h_candidate *= scale
            current_c5 = compute_C5(h_candidate)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_candidate
            if time.time() - start_time >= 100:
                break
        total_time += time.time() - start_time
        if total_time >= budget_s:
            break
    return (best_h, best_c5, len(best_h))
