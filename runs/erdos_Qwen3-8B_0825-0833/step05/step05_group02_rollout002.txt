import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Use a slightly larger number of points for better resolution
    n_points = 60
    best_h = None
    best_c5 = 1.0
    start_time = time.time()
    
    # Generate initial guess based on initial_h_values
    h_initial = initial_h_values()
    h_initial = h_initial[:n_points]  # Ensure it's the correct length
    
    # Generate a few variations to avoid local minima
    for variation in range(3):
        # Add small random perturbations to avoid getting stuck in local optima
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = h_initial + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Normalize the sum to ensure the constraint is satisfied
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Define the objective function
        def objective(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Define the constraint: sum(h) = n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Bounds for each h[i] (must be in [0, 1])
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        options = {
            'disp': False,
            'maxiter': 200,
            'ftol': 1e-8,
            'eps': 1e-8
        }
        
        # Perform optimization
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        
        # Extract the best solution found
        best_h_current = result.x
        best_c5_current = evaluate_erdos_solution(best_h_current)
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
    
    return best_h, best_c5, n_points
