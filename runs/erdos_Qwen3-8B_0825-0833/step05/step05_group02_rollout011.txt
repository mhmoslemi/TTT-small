import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from sklearn.decomposition import PCA
from scipy.signal import correlate
from scipy.fft import fft, ifft

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 60  # Start with 60 points, which is the initial construction
    
    # Initial guess based on the existing construction
    initial_h = initial_h_values()
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        n_points = len(h)
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    # Function to evaluate the objective for optimization
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        return evaluate_overlap(h)
    
    # Constraint: sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for each h[i] (must be in [0, 1])
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'disp': False,
        'maxiter': 200,
        'ftol': 1e-8,
        'eps': 1e-8
    }
    
    # Perform optimization
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options=options)
    
    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Try a few perturbations to avoid local minima
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, size=n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        perturbed_h = best_h + perturb
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        current_c5 = evaluate_overlap(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
