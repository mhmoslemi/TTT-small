```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = 1.0
    elapsed_time = 0
    start_time = time.time()
    
    # Start from the initial construction (n=90)
    n_points = 90
    h_initial = initial_h_values()
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Try a few different n_points around the initial value
    n_points_list = [80, 90, 100, 110, 120]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Create a more refined initial guess based on the previous solution
        if best_h is not None:
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        else:
            h_initial = np.random.rand(n_points)
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define the objective function
        def objective(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Define the constraint: sum(h) = n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Bounds for each h[i] (must be in [0, 1])
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        options = {
            'disp': False,
            'maxiter': 200,
            'ftol': 1e-8,
            'eps': 1e-8
        }
        
        # Perform optimization
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        
        # Extract the best solution found
        best_h_current = result.x
        best_c5_current = evaluate_erdos_solution(best_h_current)
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
        
        elapsed_time = time.time() - start_time
    
    return best_h, best_c5, len(best_h)
```
