```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction and explore around it
    initial_h = initial_h_values()
    n_points = len(initial_h)
    
    # Define the objective function
    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    # Define the constraint: sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Bounds for each h[i] (must be in [0, 1])
    bounds = Bounds(0.0, 1.0)
    
    # Optimization settings
    options = {
        'disp': False,
        'maxiter': 150,
        'ftol': 1e-8,
        'eps': 1e-8
    }
    
    # Perform optimization
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try some perturbations
    for perturbation in [0.01, 0.02, 0.03, 0.04]:
        perturbed_h = best_h + np.random.uniform(-perturbation, perturbation, size=n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        result = minimize(objective, perturbed_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        current_c5 = evaluate_erdos_solution(result.x)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, len(best_h)
```
