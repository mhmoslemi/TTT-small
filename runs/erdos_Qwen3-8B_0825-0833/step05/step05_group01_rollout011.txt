import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the known initial solution
    initial_h = initial_h_values()
    n_points = len(initial_h)
    
    # Function to evaluate the C5 bound
    def evaluate_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode="full") * dx
        return np.max(corr)
    
    # Function to optimize the function to minimize the max overlap
    def objective(x):
        return evaluate_c5(x)
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for each h[i] (must be in [0, 1])
    bounds = Bounds(0.0, 1.0)
    
    # Linear constraint for the sum
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    
    # Optimization settings
    options = {
        'disp': False,
        'maxiter': 200,
        'ftol': 1e-8,
        'eps': 1e-8
    }
    
    # Perform optimization starting from the initial guess
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=constraint, options=options)
    
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
