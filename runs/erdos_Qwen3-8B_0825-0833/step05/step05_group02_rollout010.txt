import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds, LinearConstraint

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction
    n_points = 60
    h_initial = initial_h_values()
    
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(h_initial)
    best_n_points = n_points
    
    start_time = time.time()
    
    # Try different number of points
    n_points_list = [60, 70, 80, 90, 100]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initial guess based on uniform distribution and constraints
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * n_points / 2
        h_initial = np.clip(h_initial, 0.0, 1.0)
        
        # Define the objective function
        def objective(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Define the constraint: sum(h) = n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Bounds for each h[i] (must be in [0, 1])
        bounds = Bounds([0.0]*n_points, [1.0]*n_points)
        cons = LinearConstraint(np.ones(n_points), [n_points/2], [n_points/2])
        
        # Optimization settings
        options = {
            'disp': False,
            'maxiter': 300,
            'ftol': 1e-9,
            'eps': 1e-9
        }
        
        # Perform optimization
        try:
            result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options=options)
            best_h_current = result.x
            best_c5_current = evaluate_erdos_solution(best_h_current)
            
            if best_c5_current < best_c5:
                best_c5 = best_c5_current
                best_h = best_h_current
                best_n_points = n_points
        except:
            pass
    
    return best_h, best_c5, best_n_points
