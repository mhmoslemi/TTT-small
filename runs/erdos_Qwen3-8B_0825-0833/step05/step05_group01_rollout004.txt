import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start from a good initial guess
    n_points = 90  # Use the initial construction with 90 samples
    h_initial = initial_h_values()
    
    # Define the objective function
    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    # Define the constraint: sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Bounds for each h[i] (must be in [0, 1])
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings with tighter tolerances
    options = {
        'disp': False,
        'maxiter': 300,
        'ftol': 1e-10,
        'eps': 1e-8
    }
    
    # Perform optimization
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try a few perturbations to escape local optima
    for _ in range(3):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Perform optimization on the perturbed initial guess
        result = minimize(objective, perturbed_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        
        # Update best solution if improvement is found
        current_c5 = evaluate_erdos_solution(result.x)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
    
    return best_h, best_c5, len(best_h)
