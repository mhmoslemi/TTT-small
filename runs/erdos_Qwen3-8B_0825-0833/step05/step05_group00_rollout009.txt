import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a smaller number of points for faster computation
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = initial_h_values / np.sum(initial_h_values) * (n_points / 2)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def objective(h_values):
        h = h_values
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint_integral(h_values):
        h = h_values
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx
        return integral - 1.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Store the best result
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)

    # Optimization parameters
    max_iter = 50
    ftol = 1e-8
    time_budget = budget_s
    start_time = time.time()

    # Try initial optimization
    result = optimize.minimize(
        objective,
        x0=best_h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        method='SLSQP',
        options={'ftol': ftol, 'maxiter': max_iter}
    )
    best_h = result.x
    best_c5 = result.fun

    # Try simulated annealing with local improvement
    for _ in range(20):
        if time.time() - start_time >= time_budget - 1:
            break
        # Generate a small perturbation
        perturbation = np.random.rand(n_points) * 0.1 - 0.05
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h) if np.sum(new_h) > 0 else new_h

        # Local optimization
        result = optimize.minimize(
            objective,
            x0=new_h,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_integral}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': max_iter}
        )
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5

    # Final refinement
    for _ in range(10):
        if time.time() - start_time >= time_budget - 1:
            break
        # Generate a small random perturbation
        perturbation = np.random.rand(n_points) * 0.05 - 0.025
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h) if np.sum(new_h) > 0 else new_h

        # Local optimization
        result = optimize.minimize(
            objective,
            x0=new_h,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_integral}],
            method='SLSQP',
            options={'ftol': ftol, 'maxiter': max_iter}
        )
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5

    # Ensure the integral constraint is approximately satisfied
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    # Final evaluation
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    return (best_h, best_c5, n_points)
