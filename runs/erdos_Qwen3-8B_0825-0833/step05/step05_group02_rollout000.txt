import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Strategy: Use a different optimization approach with a more structured initial guess
    # and consider different n_points values within the budget.
    
    # Define a function to generate an initial guess with a structured pattern
    def generate_initial_guess(n_points):
        # Start with a pattern of 0.5s and perturb
        base_h = np.array([0.5] * n_points)
        # Introduce small random perturbations
        perturbation = np.random.rand(n_points) * 0.2 - 0.1
        h_initial = np.clip(base_h + perturbation, 0.0, 1.0)
        # Normalize the sum to meet the constraint
        h_initial = h_initial / np.sum(h_initial) * n_points / 2
        return h_initial
    
    best_h = None
    best_c5 = 1.0
    best_n_points = 0
    
    start_time = time.time()
    
    # Try a range of n_points values
    n_points_list = [45, 50, 60, 70, 80, 90, 100, 110, 120, 130, 140, 150]
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Create an initial guess
        h_initial = generate_initial_guess(n_points)
        
        # Define the objective function
        def objective(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Define the constraint: sum(h) = n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Bounds for each h[i] (must be in [0, 1])
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        options = {
            'disp': False,
            'maxiter': 200,
            'ftol': 1e-8,
            'eps': 1e-8
        }
        
        # Perform optimization with SLSQP
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        
        # Extract the best solution found
        best_h_current = result.x
        best_c5_current = evaluate_erdos_solution(best_h_current)
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
