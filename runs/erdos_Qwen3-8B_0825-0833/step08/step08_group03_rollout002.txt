def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Best solution found so far
    best_h = None
    best_c5 = 1.0
    
    # We'll use a smaller n_points to keep within the time budget
    max_points = 100
    
    # Strategy:
    # 1. Start with a good initial guess (based on known construction)
    # 2. Use a genetic algorithm for global optimization
    # 3. Try different sizes and variations to avoid being stuck in local optimum
    
    # Define a function to create a valid initial guess
    def create_initial_guess(n_points):
        # Start from the initial guess and scale it
        initial = initial_h_values()
        if len(initial) > n_points:
            initial = initial[:n_points]
        else:
            initial = np.pad(initial, (0, n_points - len(initial)), 'constant')
        sum_h = np.sum(initial)
        scale = (n_points / 2) / sum_h
        return initial * scale
    
    # Define the objective function (minimize the max overlap)
    def objective(x, n_points):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    # Define the constraint function
    def constraint(x, n_points):
        return np.sum(x) - n_points / 2
    
    # Define bounds for each h[i] (must be in [0, 1])
    def get_bounds(n_points):
        return [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'disp': False,
        'maxiter': 200,
        'ftol': 1e-8,
        'eps': 1e-8
    }
    
    # Start timer
    start_time = time.time()
    
    # Try different n_points to find the best configuration
    for n_points in range(50, max_points + 1):
        if time.time() - start_time > budget_s:
            break
        
        # Define a genetic algorithm to find a good initial guess
        def genetic_algorithm(n_points):
            # Create a population
            population_size = 10
            population = [create_initial_guess(n_points) + np.random.uniform(-0.1, 0.1, n_points) for _ in range(population_size)]
            population = [np.clip(p, 0, 1) for p in population]
            
            # Evaluate the population
            def evaluate_individual(x):
                return evaluate_erdos_solution(x)
            
            # Find the best individual
            best_individual = min(population, key=evaluate_individual)
            return best_individual
        
        # Generate a good initial guess
        h_initial = genetic_algorithm(n_points)
        
        # Define the constraints
        cons = {'type': 'eq', 'fun': lambda x: constraint(x, n_points)}
        
        # Define bounds
        bounds = get_bounds(n_points)
        
        # Perform optimization
        result = minimize(objective, h_initial, args=(n_points,), method='SLSQP', 
                        bounds=bounds, constraints=cons, options=options)
        
        # Evaluate the result
        h_candidate = result.x
        c5_candidate = evaluate_erdos_solution(h_candidate)
        
        # Update best solution
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
    
    return best_h, best_c5, len(best_h)
