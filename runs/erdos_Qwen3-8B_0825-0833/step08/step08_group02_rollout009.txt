def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the best solution found so far
    best_h = None
    best_c5 = 1.0
    
    # Try different n_points to find the best configuration
    max_points = 100  # Limit to ensure we stay within the time budget
    
    # Optimization settings
    options = {
        'disp': False,
        'maxiter': 200,
        'ftol': 1e-8,
        'eps': 1e-8
    }
    
    # Start timer
    start_time = time.time()
    
    # Function to create an initial guess
    def create_initial_guess(n_points):
        initial = initial_h_values()
        if len(initial) > n_points:
            initial = initial[:n_points]
        else:
            initial = np.pad(initial, (0, n_points - len(initial)), 'constant')
        sum_h = np.sum(initial)
        scale = (n_points / 2) / sum_h
        return initial * scale
    
    # Function to evaluate the objective given h_values
    def objective(x, n_points):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    # Function to define the constraint
    def constraint(x, n_points):
        return np.sum(x) - n_points / 2
    
    # Bounds for each h[i] (must be in [0, 1])
    def get_bounds(n_points):
        return [(0.0, 1.0) for _ in range(n_points)]
    
    # Function to perform a simulated annealing optimization
    def simulated_annealing(n_points, initial_guess, max_iter=1000, temp=100, cooling_rate=0.995):
        h = initial_guess.copy()
        current_energy = objective(h, n_points)
        best_h = h.copy()
        best_energy = current_energy
        temp_start = temp
        temp = temp_start
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            # Generate a random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            new_h = np.clip(h + perturbation, 0.0, 1.0)
            new_energy = objective(new_h, n_points)
            
            # Acceptance probability
            if new_energy < current_energy or np.random.rand() < np.exp((current_energy - new_energy) / temp):
                h = new_h
                current_energy = new_energy
                
                # Update best solution
                if current_energy < best_energy:
                    best_energy = current_energy
                    best_h = h.copy()
            
            temp *= cooling_rate
        
        return best_h, best_energy
    
    # Try different n_points to find the best configuration
    for n_points in range(50, max_points + 1):
        if time.time() - start_time > budget_s:
            break
        
        # Try multiple initializations
        for _ in range(3):
            try:
                # Create an initial guess
                h_initial = create_initial_guess(n_points)
                
                # Perform simulated annealing optimization
                h_candidate, c5_candidate = simulated_annealing(n_points, h_initial)
                
                # Update best solution
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
                    
            except:
                pass
        
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
