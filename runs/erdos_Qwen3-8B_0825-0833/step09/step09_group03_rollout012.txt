def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    best_h = None
    best_c5 = 1.0
    time_start = time.time()
    
    # Try different n_points to find the best configuration
    for n_points in range(40, 130):  # Range of points to try
        if time.time() - time_start > budget_s:
            break
        
        # Attempt multiple initializations
        for _ in range(3):  # Try multiple initializations
            try:
                # Create an initial guess based on the provided one
                initial = initial_h_values()
                if len(initial) > n_points:
                    initial = initial[:n_points]
                else:
                    initial = np.pad(initial, (0, n_points - len(initial)), 'constant')
                sum_h = np.sum(initial)
                scale = (n_points / 2) / sum_h
                h_initial = initial * scale
                
                # Define the objective function (minimize the maximum overlap)
                def objective(x):
                    h = x.reshape(n_points)
                    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
                    return np.max(corr)
                
                # Define the constraint function
                def constraint(x):
                    return np.sum(x) - n_points / 2
                
                # Define bounds for each h[i] (must be in [0, 1])
                bounds = [(0.0, 1.0) for _ in range(n_points)]
                
                # Optimization settings
                options = {
                    'disp': False,
                    'maxiter': 100,
                    'ftol': 1e-8,
                    'eps': 1e-8
                }
                
                # Perform optimization
                result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds,

                h_candidate = result.x

                # Update best solution
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
            except:
                pass
        
        # Check time
        if time.time() - time_start > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
