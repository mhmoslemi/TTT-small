    return np.max(corr)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    # Start with the initial guess
    initial_h = initial_h_values()
    n_points_initial = len(initial_h)
    sum_h_initial = np.sum(initial_h)
    scale_initial = (n_points_initial / 2) / sum_h_initial
    scaled_initial_h = initial_h * scale_initial
    
    # Check the initial guess
    c5_candidate = evaluate_erdos_solution(scaled_initial_h)
    if c5_candidate < best_c5:
        best_c5 = c5_candidate
        best_h = scaled_initial_h
        best_n = n_points_initial
    
    # Try different n_points and starting points
    max_points = 120
    for n_points in range(60, max_points + 1):
        if time.time() - time.time() > budget_s:
            break
        
        # Create multiple initial guesses with random perturbations
        for _ in range(5):
            # Create an initial guess based on the scaled initial guess
            h_initial = np.copy(scaled_initial_h[:n_points])
            if len(h_initial) < n_points:
                h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')
            # Random perturbation
            h_initial += np.random.uniform(-0.05, 0.05, size=n_points)
            h_initial = np.clip(h_initial, 0, 1)
            
            # Ensure the sum constraint is satisfied
            sum_h = np.sum(h_initial)
            scale = (n_points / 2) / sum_h
            h_initial = h_initial * scale
            
            # Define the objective function (minimize the maximum overlap)
            def objective(x):
                h = x.reshape(n_points)
                corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
                return np.max(corr)
            
            # Define the constraint function
            def constraint(x):
                return np.sum(x) - n_points / 2
            
            # Define bounds for each h[i] (must be in [0, 1])
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            # Optimization settings
            options = {
                'disp': False,
                'maxiter': 200,
                'ftol': 1e-8,
                'eps': 1e-8
            }
            
            # Perform optimization
            try:
                result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options=options)
                h_candidate = result.x
                c5_candidate = evaluate_erdos_solution(h_candidate)
                
                # Update best solution
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
                    best_n = n_points
            except:
                pass
            
            # Check time
            if time.time() - time.time() > budget_s:
                break
    
    return best_h, best_c5, best_n
