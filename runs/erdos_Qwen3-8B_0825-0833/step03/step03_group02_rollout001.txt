import numpy as np
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import minimize
from numpy.random import choice, uniform

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Set up parameters
    n_points = 200  # Using 200 points for better accuracy
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = (initial_h_values + 1) / 2  # Shift and scale to [0, 1]
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def evaluate(h_values):
        # Ensure the integral constraint is satisfied
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        # Compute the overlap integral
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value

    def fitness(h_values):
        return evaluate(h_values)

    # Genetic algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 200
    time_budget = budget_s

    start_time = time.time()

    # Initialize population
    population = np.random.rand(population_size, n_points)
    population = (population + 1) / 2  # Shift and scale to [0, 1]
    population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
    population = np.clip(population, 0, 1)

    best_h = population[0]
    best_c5 = fitness(best_h)
    best_gen = 0

    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed >= time_budget:
            break
        # Evaluate fitness for all individuals
        fitness_values = np.array([fitness(h) for h in population])
        # Select the top half for breeding
        indices = np.argsort(fitness_values)
        selected = population[indices[:population_size // 2]]
        # Crossover
        offspring = []
        for i in range(population_size // 2):
            parent1 = selected[i]
            parent2 = selected[np.random.randint(len(selected))]
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:split], parent2[split:]))
            else:
                child = parent1.copy()
            # Mutate
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            offspring.append(child)
        # Replace the population
        population = np.array(offspring)
        # Update best solution
        for h in population:
            c5 = fitness(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                best_gen = gen

    # Local refinement with L-BFGS-B
    def objective(h_values):
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        # Compute the overlap integral
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value

    bounds = Bounds(0, 1)
    result = minimize(
        objective,
        x0=best_h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = result.fun
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    return (best_h, best_c5, n_points)
