import numpy as np
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Set up parameters
    n_points = 200  # Use more points for better accuracy
    dx = 2.0 / n_points
    # Initialize with a better starting point
    initial_h_values = np.zeros(n_points)
    # Create a step function that repeats every 0.5 interval
    for i in range(n_points):
        x = i * dx
        if x < 0.5:
            initial_h_values[i] = 1.0
        elif x < 1.0:
            initial_h_values[i] = 0.0
        elif x < 1.5:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    # Ensure the integral constraint is satisfied
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def evaluate(h_values):
        # Ensure the integral constraint is satisfied
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        # Compute the overlap integral
        correlation = correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value

    def objective(h_values):
        return evaluate(h_values)

    # Use differential evolution for global optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        callback=None,
        vectorized=True
    )

    best_h = result.x
    best_c5 = result.fun
    # Refinement with L-BFGS-B
    def objective_refined(h_values):
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        # Compute the overlap integral
        correlation = correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value

    bounds_refined = Bounds(0, 1)
    result_refined = minimize(
        objective_refined,
        x0=best_h,
        bounds=bounds_refined,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result_refined.x
    best_c5 = result_refined.fun
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx

    return (best_h, best_c5, n_points)
