import numpy as np
import scipy.signal as signal
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 80  # Use smaller samples for faster computation
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    best_h = None
    best_c5 = 1.0
    start_time = time.time()
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = initial_h_values()
        if len(h) > n_points:
            h = h[:n_points]
        else:
            h = np.pad(h, (0, n_points - len(h)), 'constant')
        # Ensure the integral constraint is satisfied
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        population.append(h)
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = []
        for h in population:
            c5_val = evaluate_erdos_solution(h)
            fitness.append(c5_val)
        
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(range(population_size), tournament_size, replace=False)
            best_index = tournament_indices[np.argmin([fitness[i] for i in tournament_indices])]
            parents.append(population[best_index])
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i + 1]
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
                offspring.extend([child1, child2])
            else:
                offspring.append(parents[i])
        
        # Apply mutation
        for i in range(len(offspring)):
            h = offspring[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.uniform(-0.1, 0.1)
                    h[j] = np.clip(h[j], 0, 1)
            offspring[i] = h
        
        # Enforce integral constraint
        for i in range(len(offspring)):
            h = offspring[i]
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            offspring[i] = h * scale
        
        # Replace population
        population = offspring
        
        # Track best solution
        for h in population:
            c5_val = evaluate_erdos_solution(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
    
    return best_h, best_c5, n_points
