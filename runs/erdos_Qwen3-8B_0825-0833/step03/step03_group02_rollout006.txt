import numpy as np
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import differential_evolution
from numpy.random import choice, uniform

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a refined initial guess
    n_points = 100  # Reduced from the previous 100 to maintain speed while allowing for good quality
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = (initial_h_values + 1) / 2  # Shift and scale to [0, 1]
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def evaluate(h_values):
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value
    
    # Optimization with differential evolution
    def objective(h_values):
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    
    result = differential_evolution(
        objective,
        bounds,
        popsize=15,
        maxiter=50,
        strategy='best1bin',
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Local refinement with L-BFGS-B
    def objective_local(h_values):
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value
    
    bounds_local = Bounds(0, 1)
    result_local = minimize(
        objective_local,
        x0=best_h,
        bounds=bounds_local,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result_local.x
    best_c5 = result_local.fun
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h, best_c5, n_points)
