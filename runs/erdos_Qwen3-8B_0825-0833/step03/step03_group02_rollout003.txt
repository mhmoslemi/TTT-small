import numpy as np
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import differential_evolution
from numpy.random import choice, uniform
from sklearn.cluster import KMeans

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Set up parameters
    n_points = 100  # Using 100 points for a good balance between speed and accuracy
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = (initial_h_values + 1) / 2  # Shift and scale to [0, 1]
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def evaluate(h_values):
        # Ensure the integral constraint is satisfied
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        # Compute the overlap integral
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value

    def fitness(h_values):
        return evaluate(h_values)

    # Differential evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    time_budget = budget_s
    start_time = time.time()

    # Initial population using initial_h_values
    initial_population = np.zeros((10, n_points))
    initial_population[0] = initial_h_values
    for i in range(1, 10):
        initial_population[i] = np.random.rand(n_points)
        initial_population[i] = (initial_population[i] + 1) / 2  # Shift and scale to [0, 1]
        initial_population[i] = initial_population[i] * (n_points / 2) / np.sum(initial_population[i])
        initial_population[i] = np.clip(initial_population[i], 0, 1)

    # Run differential evolution
    result = differential_evolution(
        func=fitness,
        bounds=bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=10,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=seed,
        polish=True,
        disp=False,
        vectorized=True,
        x0=initial_population
    )

    best_h = result.x
    best_c5 = result.fun

    # Refinement with L-BFGS-B
    def objective(h_values):
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        # Compute the overlap integral
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value

    bounds = Bounds(0, 1)
    result_refined = minimize(
        objective,
        x0=best_h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result_refined.x
    best_c5 = result_refined.fun
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    return (best_h, best_c5, n_points)
