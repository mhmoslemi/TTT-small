import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = 1.0
    n_points = 60  # Starting with 60 points for better resolution

    # Function to evaluate the current configuration
    def evaluate_config(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    # Function to perform the optimization
    def optimize_config(h_initial):
        n_points = len(h_initial)
        # Define objective function
        def objective(x):
            h = x.reshape(n_points)
            dx = 2.0 / n_points
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(corr)

        # Define constraint: sum(h) = n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2

        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Optimization settings
        options = {
            'disp': False,
            'maxiter': 100
        }

        # Perform optimization
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        return result.x

    # Generate initial configuration
    h_initial = initial_h_values()
    h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant') if len(h_initial) < n_points else h_initial[:n_points]
    sum_h = np.sum(h_initial)
    scale = (n_points / 2) / sum_h
    h_initial = h_initial * scale

    # Optimize the configuration
    optimized_h = optimize_config(h_initial)
    current_c5 = evaluate_config(optimized_h)
    best_h = optimized_h
    best_c5 = current_c5

    # Check if we have time to try different numbers of points
    if time.time() - start_time < budget_s - 5:
        n_points_list = [45, 60, 80, 100]
        for n_points in n_points_list:
            if time.time() - start_time > budget_s:
                break
            # Generate initial configuration
            h_initial = initial_h_values()
            h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant') if len(h_initial) < n_points else h_initial[:n_points]
            sum_h = np.sum(h_initial)
            scale = (n_points / 2) / sum_h
            h_initial = h_initial * scale

            # Optimize the configuration
            optimized_h = optimize_config(h_initial)
            current_c5 = evaluate_config(optimized_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h

    return best_h, best_c5, len(best_h)
