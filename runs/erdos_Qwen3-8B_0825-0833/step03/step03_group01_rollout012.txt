import scipy.optimize as optimize
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def run(seed=42, budget_s=60, **kwargs):

    # Try different n_points to find the best configuration
    n_points_list = [45, 60, 80, 100]
    best_h = None
    best_c5 = 1.0
    best_time = 0

    for n_points in n_points_list:

        def objective(x):
            h = x.reshape(n_points)
