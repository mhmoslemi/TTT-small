import random
import time
from deap import base, creator, tools, algorithms
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

    def evaluate(individual):

        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_normalized = h * scale
        # Evaluate the overlap integral
        corr = np.correlate(h_normalized, 1 - h_normalized, mode="full") * (2.0 / n_points)
        return np.max(corr),

    # Create the fitness function
