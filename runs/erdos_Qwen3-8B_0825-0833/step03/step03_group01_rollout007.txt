import numpy as np
import random
import time
from deap import base, creator, tools, algorithms

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 45  # Based on the initial configuration
    pop_size = 100
    generations = 100
    tournament_size = 3
    crossover_prob = 0.8
    mutation_prob = 0.1
    elite_size = 1
    
    # Convert initial to binary representation
    initial_h = initial_h_values()
    initial_binary = np.round(initial_h).astype(int)
    
    # Initialize DEAP
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("indexes", random.sample, range(n_points), n_points)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutFlipBit, indpb=0.01)
    toolbox.register("select", tools.selTournament, tournsize=tournament_size)
    toolbox.register("population", tools.initIterate, creator.Individual, 
                     lambda: [random.randint(0, 1) for _ in range(n_points)])
    
    toolbox.register("evaluate", lambda ind: evaluate_erdos_solution(ind * 1.0))
    toolbox.register("evaluate", lambda ind: evaluate_erdos_solution(ind * 1.0))
    
    # Initialize population
    population = toolbox.population(n=pop_size)
    population = [tools.uniform_cross(ind, initial_binary, 0.5) for ind in population]
    
    # Elitism
    best_individual = None
    best_fitness = float('inf')
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all individuals in the population
        offspring = algorithms.varAnd(population, toolbox, cxpb=crossover_prob, mutpb=mutation_prob)
        
        # Evaluate the offspring
        fits = [toolbox.evaluate(ind) for ind in offspring]
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = (fit,)
        
        # Select the next generation
        population = toolbox.select(offspring, len(population))
        
        # Elitism: keep the best individual from the current population
        if best_individual is None or population[0].fitness.values[0] < best_fitness:
            best_individual = population[0]
            best_fitness = best_individual.fitness.values[0]
    
    # Scale the best individual to [0, 1] and normalize sum
    best_h = best_individual * 1.0
    sum_h = np.sum(best_h)
    scale = (n_points / 2) / sum_h
    best_h = best_h * scale
    
    return best_h, best_fitness, n_points
