from numpy.random import choice, uniform

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Set up parameters
    n_points = 100  # Using 100 points for a good balance between speed and accuracy
    dx = 2.0 / n_points
    # Start from the initial construction
    initial_h_values = np.load('initial_h_values.npy')  # Load the initial construction

    def evaluate(h_values):
        # Ensure the integral constraint is satisfied
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        # Compute the overlap integral
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value

    def fitness(h_values):
        return evaluate(h_values)

    # Start with the initial construction
    best_h = initial_h_values.copy()
    best_c5 = evaluate(best_h)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    # Use a local search with gradient-based optimization
    def objective(h_values):
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        # Compute the overlap integral
        correlation = np.correlate(h_values, 1 - h_values, mode="full")
        c5_value = np.max(correlation) * dx
        return c5_value

    bounds = Bounds(0, 1)
    result = minimize(
        objective,

    )

    best_h = result.x

    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    return (best_h, best_c5, n_points)
