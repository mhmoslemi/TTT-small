def run(seed=42, budget_s=60, **kwargs):

    best_c5 = float('inf')
    best_h_values = None
    best_n_points = 0
    
    start_time = time.time()

        best_c5 = initial_c5

    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Construct initial guess with uniform spacing
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = np.zeros(n_points)
        h[::2] = 1  # Initial guess: alternate between 0 and 1
        h = h / (np.sum(h) * 2 / n_points) * (n_points / 2)  # Normalize to satisfy the integral constraint
        
        # Define objective function
        def objective(h):
            h = np.clip(h, 0, 1)
            shifted_h = np.roll(h, 1)
            overlap = np.correlate(h, 1 - shifted_h, mode="full")
            c5 = np.max(overlap * (2 / n_points))
            return c5
        
        # Define bounds
        bounds = Bounds(0, 1)
        
        # Use L-BFGS-B for optimization
        result = optimize.minimize(
            fun=objective,
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
        )
        
        # Evaluate the result
        h_result = result.x
        c5_result = objective(h_result)
        
        # Update best result
        if c5_result < best_c5:
            best_c5 = c5_result
            best_h_values = h_result
            best_n_points = n_points
    
    return best_h_values, best_c5, best_n_points
