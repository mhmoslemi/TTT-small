import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assume this is preloaded with 70 samples

def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # Start with 100 points for a balance between speed and accuracy
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5  # Initial guess with random values in [0, 0.5]
    h = h * 2.0 - 1.0  # Shift to [ -1.0, 1.0 ] and then scale to [0, 1.0]
    h = np.clip(h, 0.0, 1.0)
    h = h * (n_points / 2.0) / np.sum(h)  # Ensure the integral is 1

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        correlation = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5

    # Optimization function to minimize C5
    def objective(h):
        return compute_c5(h)

    # Constraint: integral of h is 1
    def constraint(h):
        return np.sum(h) * dx - 1.0

    # Optimization with constraints
    h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
    objective_func = cp.Minimize(objective(h_var))
    constraints = [constraint(h_var) == 0.0]
    prob = cp.Problem(objective_func, constraints)
    try:
        result = prob.solve(solver=cp.SCS, max_iter=100, time_limit=budget_s - 0.1)
        if result is not None:
            h_opt = h_var.value
        else:
            h_opt = h
    except:
        h_opt = h

    # Convert to numpy array and ensure it's valid
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = h_opt * (n_points / 2.0) / np.sum(h_opt)  # Normalize again to ensure integral is 1

    # Evaluate the result
    c5_bound = compute_c5(h_opt)
    return h_opt, c5_bound, n_points
