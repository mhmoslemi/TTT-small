import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Start with 45 points
    initial_h = initial_h_values()
    
    # Define the objective function for optimization
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    # Define constraints: h in [0, 1] and integral is exactly 1
    def constraint_integrate(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * 2.0 / n_points - 1.0
    
    def constraint_bounds(h_flat):
        h = h_flat.reshape(n_points)
        return np.all((h >= 0) & (h <= 1))
    
    # Initial guess
    h_initial = initial_h.copy()
    h_initial = h_initial.astype(float)
    
    # Use scipy.optimize to minimize the objective function
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[
            {'type': 'eq', 'fun': constraint_integrate},

        method='SLSQP',
