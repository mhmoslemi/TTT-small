```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assuming this file is pre-loaded with the initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points
    n_points_list = [100, 200, 300, 400, 500]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.random.permutation(h)
        h = h / np.sum(h) * (n_points / 2)
        
        # Define the objective function
        def objective(x):
            h = x.reshape(-1)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * (2.0 / n_points)
            return c5
        
        # Define constraints
        def constraints(x):
            h = x.reshape(-1)
            return [np.sum(h) - n_points / 2]
        
        # Set up optimization
        x = cp.Variable(n_points)
        obj = cp.Minimize(objective(x))
        cons = [constraints(x)]
        
        try:
            prob = cp.Problem(obj, cons)
            result = prob.solve(solver=cp.SCS, max_iter=100, verbose=False)
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = x.value
                best_n = n_points
        except:
            pass
    
    # Evaluate best solution
    if best_h is None:
        # Fallback to initial construction
        best_h = initial_h_values
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, best_n)
```
