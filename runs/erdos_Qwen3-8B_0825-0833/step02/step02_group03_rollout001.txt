def run(seed=42, budget_s=60, **kwargs):
    # Set up parameters
    n_points = 100  # Using 100 points for a good balance between speed and accuracy
    dx = 2.0 / n_points
    # Initial guess as the current construction
    initial_h_values = np.zeros(n_points)
    # Copy initial construction if available
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
        initial_h_values = np.clip(initial_h_values, 0, 1)
    else:
        # Random initial guess
        h = np.random.rand(n_points)
        h = h * 2 - 1  # Shift to range [-1, 1]
        h = (h + 1) / 2  # Shift and scale to [0, 1]
        h = h * (n_points / 2) / np.sum(h)  # Adjust for the integral constraint
        initial_h_values = h

    # Define objective function to minimize the maximum correlation
    def objective(h_values):
        h = h_values
        # Enforce bounds
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        h = h.reshape(-1, 1)
        # Compute all shifted versions of h
        overlap = []
        for shift in range(1, n_points):
            shifted_h = np.roll(h, shift=shift, axis=0)
            overlap.append(h * (1 - shifted_h))
        # Compute the maximum overlap
        max_overlap = np.max(np.hstack(overlap))
        return max_overlap

    # Define bounds for the optimization
    bounds = Bounds(0, 1)

    # Optimize
    start_time = time.time()
    # Use a more robust method with a better initial guess
    result = minimize(
        objective,
        x0=initial_h_values,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    # Ensure the integral constraint is satisfied
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    # Evaluate the final C5 value
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    # Return the best solution
    return (best_h, best_c5, n_points)
