import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points to find the best configuration
    n_points_list = [45, 60, 80, 100, 120]
    best_h = None
    best_c5 = 1.0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with the provided initial values or a random configuration
        h = initial_h_values()
        if len(h) > n_points:
            h = h[:n_points]
        else:
            h = np.pad(h, (0, n_points - len(h)), 'constant')
        
        # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        
        best_h_current = h.copy()
        best_c5_current = evaluate_erdos_solution(h)
        
        # Use a more efficient optimization method with constraints
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        def constraint_func(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        # Set up the optimization problem
        h_var = cp.Variable(n_points, bounds=(0, 1))
        objective = cp.Minimize(objective_func(h_var))
        constraint = cp.Constraint(constraint_func(h_var), lb=0, ub=0)
        problem = cp.Problem(objective, [constraint])
        
        # Solve the optimization problem
        try:
            result = problem.solve(solver=cp.OSQP, verbose=False)
            if result is not None:
                h_opt = h_var.value
                current_c5 = evaluate_erdos_solution(h_opt)
                if current_c5 < best_c5_current:
                    best_c5_current = current_c5
                    best_h_current = h_opt.copy()
        except:
            pass
        
        # Additional refinement with gradient descent
        learning_rate = 0.1
        max_iterations = 100
        grad_buffer = np.zeros(n_points)
        
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            
            # Compute the objective and gradient
            def objective(h_flat):
                h = h_flat.reshape(n_points)
                corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
                return np.max(corr)
            
            def gradient(h_flat):
                h = h_flat.reshape(n_points)
                corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
                c5_val = np.max(corr)
                grad = np.zeros(n_points)
                
                for k in range(n_points):
                    if corr[k] == c5_val:
                        for i in range(n_points):
                            grad[i] += (1 - np.roll(h, k))[i] * (2.0 / n_points)
                            grad[i] -= h[i] * (2.0 / n_points) * (np.roll(h, k)[i] == h[i])
                
                return grad.flatten()
            
            # Compute the gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                obj_plus = objective(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                obj_minus = objective(h_minus)
                grad[i] = (obj_plus - obj_minus) / (2 * 1e-6)
            
            # Update with momentum
            grad_buffer = 0.9 * grad_buffer + grad
            h -= learning_rate * grad_buffer
            
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0, 1)
            
            # Enforce the integral constraint
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = h * scale
            
            current_c5 = evaluate_erdos_solution(h)
            
            if current_c5 < best_c5_current:
                best_c5_current = current_c5
                best_h_current = h.copy()
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
    
    return best_h, best_c5, len(best_h)
