import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions and variables
# initial_h_values is provided, representing a starting point for the optimization

def run(seed=42, budget_s=60, **kwargs):
    # Seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points (can be adjusted based on time budget)
    n_points = 100  # Start with 100 points, can be increased if time allows
    
    # Function to evaluate the overlap integral (C_5)
    def compute_c5(h):
        # Convert h to a function for convolution
        h_func = np.interp(np.linspace(0, 2, n_points), np.arange(n_points), h)
        # Compute the convolution (overlap integral)
        conv = np.correlate(h_func, 1 - h_func, mode="full")
        # Scale by dx (2 / n_points)
        dx = 2.0 / n_points
        c5_values = conv * dx
        # Return the maximum value
        return np.max(c5_values)
    
    # Function to compute the overlap integral and return the max value
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return compute_c5(h)
    
    # Function to compute the constraint (integral of h over [0,2] equals 1)
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2  # Ensure sum(h) == n_points / 2
    
    # Initial guess based on provided initial_h_values
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess[:n_points]  # Ensure we don't exceed n_points
    
    # Optimization setup
    h_flat = cp.Variable(n_points)
    objective_func = cp.Minimize(objective(h_flat))
    constraint_func = cp.Constraint(constraint(h_flat), name="integral_constraint")
    
    # Define problem
    prob = cp.Problem(objective_func, [constraint_func])
    
    # Solve the optimization problem
    try:
        result = prob.solve(solver=cp.SCS, max_iters=1000, time_limit=budget_s - (time.time() - start_time))
        if result is not None:
            h_values = h_flat.value
        else:
            h_values = initial_guess
    except:
        h_values = initial_guess
    
    # Ensure h_values is within [0, 1]
    h_values = np.clip(h_values, 0, 1)
    
    # Compute the final C_5 value
    c5_bound = compute_c5(h_values)
    
    # Return the result
    return h_values, c5_bound, n_points
