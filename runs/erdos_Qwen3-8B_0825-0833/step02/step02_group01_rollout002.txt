import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use a finer grid for better resolution
    initial_h = initial_h_values()
    initial_h = np.pad(initial_h, (0, n_points - len(initial_h)), 'constant')  # Pad to n_points
    
    # Create optimization variables
    h = cp.Variable(n_points, bounds=(0, 1))
    
    # Objective: minimize the maximum overlap integral
    # We'll use a convex approximation by considering the maximum over all lags
    # Since the overlap integral is piecewise linear and convex, we can use a convex formulation
    
    # Define the overlap integral for each lag
    max_overlap = 0
    for k in range(1, n_points):
        overlap_k = cp.sum(h * (1 - h[k:])) * (2.0 / n_points)
        max_overlap = cp.Max(max_overlap, overlap_k)
    
    # Constraint: total area under the function is 1
    area_constraint = cp.sum(h) * (2.0 / n_points) == 1
    
    # Optimization problem
    prob = cp.Problem(cp.Minimize(max_overlap), [area_constraint])
    
    start_time = time.time()
