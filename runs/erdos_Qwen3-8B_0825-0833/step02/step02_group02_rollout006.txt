import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
import cvxpy as cp

# Pre-imported
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assume this is a pre-loaded initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start from initial solution or random
    n_points = 100  # Use 100 points for efficiency
    h = initial_h_values[:n_points].copy()
    
    # Normalize to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Define objective function: minimize the maximum overlap
    def objective(h_values):
        # Reshape to correct shape
        h = h_values.reshape((n_points,))
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Find the maximum value
        return np.max(corr) * (2.0 / n_points)
    
    # Define constraints
    def constraints(h_values):
        h = h_values.reshape((n_points,))
        return np.sum(h) - n_points / 2
    
    # Optimization: use gradient descent with bounds
    bounds = Bounds(0, 1)
    result = opt.minimize(
        fun=objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Early stopping: check for improvement
    if best_c5 < 0.3808:
        print(f"Found better solution with C₅ = {best_c5}")
        return (best_h, best_c5, n_points)
    
    # Fallback to random restarts if needed
    for _ in range(5):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    # Final check for time
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        print("Time budget reached, returning current best")
    
    return (best_h, best_c5, n_points)
