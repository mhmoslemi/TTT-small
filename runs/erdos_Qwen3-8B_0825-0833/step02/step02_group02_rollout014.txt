```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / n_points
        overlap = signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    # Initial C5 value
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Define optimization variables
    x = cp.Variable(n_points)
    constraints = [
        cp.sum(x) == n_points / 2,
        0 <= x,
        x <= 1
    ]
    
    # Objective function to minimize the maximum overlap integral
    def optimize_c5():
        nonlocal best_c5, best_h
        # For each shift k, compute the overlap integral
        # We'll use an approximate method due to computational constraints
        dx = 2.0 / n_points
        max_overlap = 0.0
        for k in range(n_points):
            # Compute h * (1 - h[(k+1):k+n_points+1])
            # We'll use a sliding window approach
            if k + n_points <= n_points:
                overlap = np.sum(h * (1 - h[k+1:k+n_points+1])) * dx
            else:
                overlap = np.sum(h * (1 - h[k+1:n_points])) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # Use a gradient-based approach with simulated annealing for optimization
    def objective(x_vec):
        nonlocal best_c5, best_h
        x_vec = np.clip(x_vec, 0, 1)
        c5 = optimize_c5()
        if c5 < best_c5:
            best_c5 = c5
            best_h = x_vec.copy()
        return c5
    
    # Perform optimization
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    # Check if we have a better solution
    if result.fun < best_c5:
        best_c5 = result.fun
        best_h = result.x.copy()
    
    # Ensure the solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # If time allows, perform a local refinement
        for _ in range(10):
            # Small perturbation
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            perturbed_h = best_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            
            c5 = compute_c5(perturbed_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = perturbed_h
    
    return best_h.tolist(), best_c5, n_points
```
