def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1 - h
    # Compute the cross-correlation
    corr = np.correlate(h, one_minus_h, mode="full")
    # The maximum overlap integral is the maximum of the full correlation
    # scaled by dx
    c5 = np.max(corr) * dx
    return c5
