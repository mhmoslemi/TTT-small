import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points to find the best configuration
    n_points_list = [45, 60, 80, 100]
    best_h = None
    best_c5 = 1.0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with the provided initial values or a random configuration
        h = initial_h_values()
        if len(h) > n_points:
            h = h[:n_points]
        else:
            h = np.pad(h, (0, n_points - len(h)), 'constant')
        
        # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        
        # Define the optimization problem
        x = cp.Variable(n_points)
        constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
        objective = cp.Minimize(cp.Max(cp.multiply(x, 1 - cp.roll(x, 1)) + cp.multiply(x, 1 - cp.roll(x, 2)) + 
                                       cp.multiply(x, 1 - cp.roll(x, 3)) + cp.multiply(x, 1 - cp.roll(x, 4)) + 
                                       cp.multiply(x, 1 - cp.roll(x, 5))))
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver='CVXOPT', verbose=False)
            if prob.status == 'optimal':
                h_opt = np.array(x.value)
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass
        
        # Perform additional local search
        h = np.array(h)
        best_h_current = h.copy()
        best_c5_current = evaluate_erdos_solution(h)
        
        learning_rate = 0.1
        momentum = 0.9
        max_iterations = 100
        grad_buffer = np.zeros(n_points)
        
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            
            # Compute the gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                obj_plus = evaluate_erdos_solution(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                obj_minus = evaluate_erdos_solution(h_minus)
                grad[i] = (obj_plus - obj_minus) / (2 * 1e-6)
            
            # Update with momentum
            grad_buffer = momentum * grad_buffer + grad
            h -= learning_rate * grad_buffer
            
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0, 1)
            
            # Enforce the integral constraint
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = h * scale
            
            current_c5 = evaluate_erdos_solution(h)
            
            if current_c5 < best_c5_current:
                best_c5_current = current_c5
                best_h_current = h.copy()
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
    
    return best_h, best_c5, len(best_h)
