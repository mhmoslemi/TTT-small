import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    shifted_h = np.zeros_like(h_values)
    for k in range(1, n_points):
        shifted_h[k:] = h_values[:-k]
    overlap = np.correlate(h_values, 1 - shifted_h, mode="full")
    c5 = np.max(overlap * dx)
    return c5

def initial_h_values(n_points):
    # Initial construction with n_points samples
    h = np.zeros(n_points)
    h[0::2] = 1.0
    h = h / np.sum(h) * n_points / 2
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    n_points = 128  # Start with 128 points for reasonable performance
    h_initial = initial_h_values(n_points)
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try to refine the solution with a local optimization
    # Convert h to a continuous function for optimization
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    
    # Define the objective to minimize the maximum overlap
    # We can't directly minimize the maximum, so we use a minimax formulation
    # Approximate the maximum by computing it numerically
    def objective_func(h_vals):
        h = h_vals
        dx = 2.0 / n_points
        shifted_h = np.zeros_like(h)
        for k in range(1, n_points):
            shifted_h[k:] = h[:-k]
        overlap = np.correlate(h, 1 - shifted_h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Use a local optimization to refine the solution
    result = optimize.minimize(
        objective_func,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Update the best solution
    if result.success:
        h_refined = result.x
        c5_refined = objective_func(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Try a few more refinements with different starting points
    for _ in range(3):
        # Perturb the current solution slightly

            objective_func,
            perturbed_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        if result.success:
            h_refined = result.x
            c5_refined = objective_func(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
