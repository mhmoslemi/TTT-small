import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points to find the best configuration
    n_points_list = [45, 60, 80, 100]
    best_h = None
    best_c5 = 1.0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with the provided initial values or a random configuration
        h = initial_h_values()
        if len(h) > n_points:
            h = h[:n_points]
        else:
            h = np.pad(h, (0, n_points - len(h)), 'constant')
        
        # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        
        best_h_current = h.copy()
        best_c5_current = evaluate_erdos_solution(h)
        
        # Perform optimization with a different strategy: simulated annealing
        max_iterations = 200
        temp = 1.0
        temp_decay = 0.995
        best_c5_current = evaluate_erdos_solution(h)
        
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            
            # Generate a random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = np.clip(h + perturbation, 0, 1)
            
            # Enforce the integral constraint
            sum_h_perturbed = np.sum(h_perturbed)
            if sum_h_perturbed != 0:
                scale_perturbed = (n_points / 2) / sum_h_perturbed
                h_perturbed = h_perturbed * scale_perturbed
            
            current_c5 = evaluate_erdos_solution(h_perturbed)
            
            # Accept the new solution with probability based on temperature
            delta_c5 = current_c5 - best_c5_current
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
                h = h_perturbed
                best_c5_current = current_c5
            
            temp *= temp_decay
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = h.copy()
    
    return best_h, best_c5, len(best_h)
