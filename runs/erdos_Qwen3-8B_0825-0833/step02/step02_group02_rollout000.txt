import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Choose number of points (less than 1k for speed)
    n_points = 256  # Fast enough for optimization within 60s
    
    # Initial guess
    h = np.copy(initial_h_values[:n_points])
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h = h[:n_points]
    h = np.clip(h, 0, 1)
    
    # Ensure the integral constraint
    total_area = np.sum(h) * (2.0 / n_points)
    if total_area < 1 - 1e-6:
        h = np.clip(h + (1 - total_area) / n_points, 0, 1)
    elif total_area > 1 + 1e-6:
        h = np.clip(h - (total_area - 1) / n_points, 0, 1)
    
    # Optimization objective
    def objective(h):
        # Compute the overlap integral for all shifts
        max_overlap = 0
        for k in range(1, n_points):
            overlap = correlate(h, 1 - h, mode="full")
            max_overlap = max(max_overlap, overlap[k])
        return max_overlap * (2.0 / n_points)
    
    # Optimization using gradient descent
    h = h.astype(float)
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    
    for _ in range(1000):
        # Gradient descent step
        h = h + 0.01 * np.random.randn(n_points)
        h = np.clip(h, 0, 1)
        total_area = np.sum(h) * (2.0 / n_points)
        if total_area < 1 - 1e-6:
            h = np.clip(h + (1 - total_area) / n_points, 0, 1)
        elif total_area > 1 + 1e-6:
            h = np.clip(h - (total_area - 1) / n_points, 0, 1)
        
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Check if we can do better with a convex optimization (CVXPY)
    # This is a simplified version for demonstration
    x = cp.Variable(n_points)
    constraints = [cp.sum(x) == n_points / 2, 0 <= x, x <= 1]
    objective = cp.Maximize(cp.Max(correlate(x.value, 1 - x.value, mode="full") * (2.0 / n_points)))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS)
        best_h = x.value
        best_c5 = prob.value
    except:
        pass
    
    return best_h, best_c5, n_points
