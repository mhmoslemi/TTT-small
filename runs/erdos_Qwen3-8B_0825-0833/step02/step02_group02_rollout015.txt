import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize


def run(seed=42, budget_s=60, **kwargs):
    # Initial setup and parameters
    initial_h_values = np.load("initial_h_values.npy")  # Assume this is a pre-loaded initial guess
    n_points_initial = len(initial_h_values)
    
    # Set up parameters
    max_n_points = 1000
    target_c5 = 0.380876
    
    # Function to compute the overlap integral as per the problem
    def compute_c5(h_values):
        h = np.concatenate([h_values, h_values[:1]])  # Extend to handle circularity in the integral
        result = np.correlate(h, 1 - h, mode="full")
        # Only consider the relevant part of the correlation
        result = result[1: n_points_initial + 1]
        integral = result * (2.0 / n_points_initial)
        return np.max(integral)
    
    # Function to convert h_values to a function for use in optimization
    def objective_function(h_values):
        return compute_c5(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points_initial / 2
    
    # Function to generate a random initial guess
    def random_initial_guess(n_points):
        h = np.random.rand(n_points)
        h = (h * 2.0 - 1.0) * 0.5 + 0.5  # Uniform distribution [0, 1]
        h = h * (n_points / 2.0) / np.sum(h)  # Normalize to meet the constraint
        return h
    
    # Function to perform a local optimization
    def local_optimize(n_points, initial_guess, seed):
        np.random.seed(seed)
        # Define bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        # Define constraints
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        # Perform optimization
        result = minimize(
            objective_function,
            initial_guess,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'ftol': 1e-9, 'maxiter': 1000}
        )
        return result.x if result.success else initial_guess
    
    # Function to perform global optimization using a grid of initial guesses
    def global_optimize(n_points, seed):
        np.random.seed(seed)
        best_h = None
        best_c5 = float('inf')
        for i in range(5):
            initial_guess = random_initial_guess(n_points)
            h = local_optimize(n_points, initial_guess, seed + i)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        return best_h, best_c5
    
    # Search for the best n_points
    best_h_values = None
    best_c5_bound = float('inf')
    best_n_points = 0
    
    start_time = time.time()
    for n_points in range(100, max_n_points + 1, 100):
        # Use initial guess based on the provided initial_h_values
        initial_guess = np.zeros(n_points)
        initial_guess[:len(initial_h_values)] = initial_h_values
        if len(initial_guess) > n_points:
            initial_guess = initial_guess[:n_points]
        elif len(initial_guess) < n_points:
            initial_guess = np.pad(initial_guess, (0, n_points - len(initial_guess)), 'constant')
        # Normalize to meet the constraint
        initial_guess = initial_guess * (n_points / 2.0) / np.sum(initial_guess)
        
        # Optimize with this n_points
        h_values, c5 = global_optimize(n_points, seed)
        if c5 < best_c5_bound:
            best_c5_bound = c5
            best_h_values = h_values
            best_n_points = n_points
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    # Return the best result
    return best_h_values, best_c5_bound, best_n_points
