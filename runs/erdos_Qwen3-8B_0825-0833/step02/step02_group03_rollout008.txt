import numpy as np
import time
import random
from deap import base, creator, tools, algorithms
import scipy.signal as signal

def run(seed=42, budget_s=60, **kwargs):
    # Setup parameters
    n_points = 100  # Choose a reasonable number of points for speed and accuracy
    dx = 2.0 / n_points
    initial_h_values = np.load("initial_h_values.npy")  # Assuming initial_h_values is provided
    h_initial = initial_h_values.copy()
    
    # Normalization to ensure the integral constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_initial = np.clip(h_initial, 0, 1)

    population_size = 50
    crossover_prob = 0.8
    mutation_prob = 0.1
    generations = 100
    max_time = budget_s  # Time budget in seconds

    # Define the evaluation function
    def evaluate(individual):
        h = np.clip(individual, 0, 1)
        # Enforce the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        return (max_overlap,)

    # Set up the genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.rand, n_points)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Setup the population and run the genetic algorithm
    population = toolbox.population(n=population_size)
    start_time = time.time()

    for gen in range(generations):
        # Check if time budget is exceeded
        if time.time() - start_time > max_time:
            break

        # Evaluate the population
        fitnesses = list(map(toolbox.evaluate, population))
        for ind, fit in zip(population, fitnesses):
            ind.fitness.values = fit
        
        # Select the next generation
        offspring = algorithms.varAnd(population, toolbox, cxpb=crossover_prob, mutpb=mutation_prob)
        offspring = list(map(toolbox.evaluate, offspring))
        for ind, fit in zip(offspring, offspring):
            ind.fitness.values = fit
        
        # Replace the population
        population = toolbox.select(population + offspring, k=population_size)

    # Find the best solution
    best_individual = min(population, key=lambda ind: ind.fitness.values[0])
    best_h = best_individual
    best_c5 = best_individual.fitness.values[0]
    
    # Ensure the solution adheres to the integral constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = np.max(signal.correlate(best_h, 1 - best_h, mode="full")) * dx

    return (best_h, best_c5, n_points)
