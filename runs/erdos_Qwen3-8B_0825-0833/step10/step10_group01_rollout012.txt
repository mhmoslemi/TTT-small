def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    best_h = None
    best_c5 = 1.0
    best_n_points = 0
    
    start_time = time.time()
    
    # Start from the known good initial construction
    initial_h = initial_h_values()
    initial_n_points = len(initial_h)
    
    # Try different n_points around the initial construction
    for n_points in range(int(0.9 * initial_n_points), int(1.2 * initial_n_points) + 1):
        if time.time() - start_time > budget_s:
            break
        
        # Create an initial guess
        if len(initial_h) > n_points:
            h_initial = initial_h[:n_points]
        else:
            h_initial = np.pad(initial_h, (0, n_points - len(initial_h)), 'constant')
        
        # Normalize the initial guess
        sum_h = np.sum(h_initial)
        scale = (n_points / 2) / sum_h
        h_initial = h_initial * scale
        
        # Perturb the initial guess to avoid local optima
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_initial += perturbation
        h_initial = np.clip(h_initial, 0.0, 1.0)
        
        # Define the objective function
        def objective(x):
            h = x
            corr = correlate(h, 1 - h, mode="full")
            corr = corr * (2.0 / n_points)
            return np.max(corr)
        
        # Define the constraint
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        # Bounds for each h[i] (must be in [0, 1])
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform optimization
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, 
                          constraints={'type': 'eq', 'fun': constraint}, 
                          options={'disp': False, 'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8})
        
        # Check if the optimization was successful
        if result.success:
            h_candidate = result.x
            c5_candidate = evaluate_erdos_solution(h_candidate)
            
            # Update best solution
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n_points = n_points
        
        # Also try a few random initializations to avoid local optima
        for _ in range(2):
            h_initial = np.random.uniform(0.0, 1.0, n_points)
            sum_h = np.sum(h_initial)
            scale = (n_points / 2) / sum_h
            h_initial = h_initial * scale
            
            result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, 
                              constraints={'type': 'eq', 'fun': constraint}, 
                              options={'disp': False, 'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8})
            
            if result.success:
                h_candidate = result.x
                c5_candidate = evaluate_erdos_solution(h_candidate)
                
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
                    best_n_points = n_points
        
        # Check time
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
