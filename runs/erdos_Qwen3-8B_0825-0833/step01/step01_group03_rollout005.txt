```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points (n_points), start with 100 for speed
    n_points = 100
    dx = 2.0 / n_points
    max_attempts = 10  # Number of attempts to find a better solution
    
    # Initial guess based on provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Ensure length matches n_points
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        h = h.reshape(-1, 1)
        shifted_h = np.roll(h, shift=1, axis=0)
        overlap = np.sum(h * (1 - shifted_h[:-1])) * dx
        return np.max(overlap)
    
    # Define optimization variables and objective
    def optimize_h(n_points, initial_guess=None):
        h = cp.Variable(n_points, nonneg=True, lb=0, ub=1)
        objective = cp.Minimize(0)  # Placeholder, we'll optimize using constraints

        # Constraint: integral of h(x) over [0, 2] equals 1
        constraint = [cp.sum(h) * dx == 1]

        # Solve using ECOS (for speed)
        prob = cp.Problem(objective, constraint)
        prob.solve(solver=cp.ECOS, max_iter=1000)
        if prob.status != 'OPTIMAL':
            print("Optimization failed")
            return initial_guess
        
        # Convert to numpy array
        h_values = h.value
        h_values = np.clip(h_values, 0, 1)
        return h_values
    
    best_c5 = float('inf')
    best_h = h_initial
    
    # Attempt multiple optimizations with different starting points
    for attempt in range(max_attempts):
        print(f"Attempt {attempt + 1}/{max_attempts}")
        
        # Start with a random perturbation of the initial guess
        h = initial_h_values.copy()
        h = h[:n_points]
        h += np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # Ensure the integral is 1
        
        # Evaluate the initial guess
        c5 = evaluate_overlap(h)
        print(f"Initial C₅: {c5}")
        
        # Run optimization
        start_time = time.time()
        h_opt = optimize_h(n_points, h)
        elapsed_time = time.time() - start_time
        print(f"Optimization time: {elapsed_time:.2f} seconds")
        
        # Evaluate the optimized h
        c5_candidate = evaluate_overlap(h_opt)
        print(f"Optimized C₅: {c5_candidate}")
        
        # Update best solution
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_opt
            print(f"New best C₅: {best_c5}")
        
        # If we've used up the budget, exit early
        if elapsed_time > budget_s:
            print("Time budget exhausted")
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_overlap(best_h)
    print(f"Final best C₅: {final_c5}")
    
    return best_h, final_c5, n_points
```
