        return c5

    def optimize_C5(h):
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Optimization using scipy
    def optimize_h(n_points, dx, budget_s):
        h = np.random.rand(n_points)

        # Ensure integral is 1
        total = np.sum(h) * dx

        start_time = time.time()
        result = opt.minimize(
            optimize_C5,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-6, 'gtol': 1e-6, 'maxiter': 100}
        )
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            return h
        return result.x

    # Try increasing n_points if time allows
    best_h = initial_h
