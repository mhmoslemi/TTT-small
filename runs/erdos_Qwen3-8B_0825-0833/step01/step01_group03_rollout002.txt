import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Predefined parameters
    max_n_points = 500  # Maximum number of points allowed
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    
    def objective(h_values):
        # Evaluate the overlap integral
        h = np.array(h_values)
        h_shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            h_shifted[k:] = h[:len(h) - k]
        overlap = np.sum(h * (1 - h_shifted)) * dx
        return overlap
    
    def evaluate_c5(h_values):
        # Compute the maximum overlap
        h = np.array(h_values)
        h_shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            h_shifted[k:] = h[:len(h) - k]
        overlaps = np.array([np.sum(h * (1 - h_shifted[k:])) * dx for k in range(len(h) - 1)])
        return np.max(overlaps)
    
    def evaluate_c5_fast(h_values):
        # Use numpy correlate for faster computation
        h = np.array(h_values)
        h_shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            h_shifted[k:] = h[:len(h) - k]
        corr = np.correlate(h, 1 - h_shifted, mode="full")
        return np.max(corr * dx)
    
    # Initial guess from provided function
    h_initial = initial_h_values
    h_initial = np.array(h_initial)
    
    # Check if initial guess has correct sum
    if np.sum(h_initial) * dx != 1:
        h_initial = np.array([1.0 / len(h_initial)] * len(h_initial))
    
    # Optimization parameters
    max_iterations = 100
    best_c5 = float('inf')
    best_h = h_initial.copy()
    elapsed_time = 0
    
    # Optimization loop
    for _ in range(max_iterations):
        start_time = time.time()
        
        # Convert to numpy array for optimization
        h = np.array(best_h)
        
        # Define objective function for optimization
        def objective_func(x):
            h_new = x.reshape((n_points,))
            # Ensure constraints
            h_new = np.clip(h_new, 0, 1)
            # Ensure integral constraint
            h_new = h_new + (1 - np.sum(h_new)) / n_points
            # Compute overlap integral for all shifts
            overlaps = np.zeros(len(h_new) - 1)
            for k in range(1, len(h_new) - 1):
                h_shifted = np.zeros_like(h_new)
                h_shifted[k:] = h_new[:len(h_new) - k]
                overlaps[k - 1] = np.sum(h_new * (1 - h_shifted)) * dx
            return np.max(overlaps)
        
        # Optimization using scipy
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)]
        )
        
        # Update best solution
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new + (1 - np.sum(h_new)) / n_points
            current_c5 = evaluate_c5_fast(h_new)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
        
        # Check time budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Return best result
    return best_h, best_c5, n_points
