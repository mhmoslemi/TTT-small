import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial guess
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Evaluate initial C5
    c5_initial = evaluate_erdos_solution(h)
    
    # Function to evaluate C5 for a given h
    def evaluate_c5(h_values):
        h = h_values.copy()
        return evaluate_erdos_solution(h)
    
    # Function to evaluate the objective for optimization
    def objective(h_values):
        return evaluate_c5(h_values)
    
    # Function to evaluate the constraints for optimization
    def constraints(h_values):
        # Constraint: sum(h_values) == n_points / 2
        return [np.sum(h_values) - n_points / 2]
    
    # Optimization with CVXPY
    # Create variables
    h_var = cp.Variable(n_points, name='h')
    
    # Objective: minimize the maximum overlap integral
    # To handle the max, we use a trick with auxiliary variables
    # We introduce auxiliary variables for the overlap integrals
    # and minimize the maximum among them
    
    n_overlap = 2 * n_points - 1  # Length of the full convolution result
    max_overlap = cp.Variable(n_overlap, name='max_overlap')
    
    # Constraint: overlap integrals must be less than or equal to max_overlap
    overlap_integrals = [0] * n_overlap
    for i in range(n_overlap):
        # Compute the overlap integral using h and 1 - h
        # The overlap at shift k is the integral of h(x)(1 - h(x + k))
        # For the full convolution, we need to be careful with the indices
        # We construct the overlap integrals manually
        
        # For the full convolution, the overlap integral for shift k is
        # sum(h[i] * (1 - h[(i + k) % n_points]) * dx for i in range(n_points))
        # However, we need to handle the case where i + k exceeds the array length
        # For simplicity, we will construct the overlap integral directly
        
        # We can approximate the integral using sum(h[i] * (1 - h[(i + k) % n_points]) * dx)
        # for each k
        overlap = 0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap += h[i] * (1 - h[j])
        overlap_integrals[i] = overlap * dx
    
    # We can't directly implement this in CVXPY without knowing k, so we use a trick
    # We will use a binary search to find the optimal h that minimizes the maximum overlap
    # This will be a nested optimization problem
    
    # We will use a numerical optimization approach with gradient descent
    # This approach is faster for small n_points
    
    # We will use a gradient-based optimization to minimize the maximum overlap
    # This is a constrained optimization problem
    
    # Define the objective function for gradient-based optimization
    def objective_f(h_values):
        # Compute the overlaps
        overlaps = []
        for k in range(2 * n_points - 1):
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_values[i] * (1 - h_values[j])
            overlaps.append(overlap * dx)
        return np.max(overlaps)
    
    # Define the constraint for the average value
    def constraint_f(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Use L-BFGS-B for optimization
    # Initial guess is the initial_h_values
    result = opt.minimize(objective_f, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': constraint_f})
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure we stay within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        # If time is running out, return the best solution so far
        return (best_h, best_c5, n_points)
    
    # After optimization, perform a local search to refine the solution
    # We will use a small perturbation to find better solutions
    def perturb_h(h, perturbation_factor=0.05):
        return np.clip(h + perturbation_factor * np.random.randn(n_points), 0, 1)
    
    # Perform a local search for a few iterations
    for _ in range(5):
        perturbed_h = perturb_h(best_h)
        perturbed_c5 = objective_f(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    return (best_h, best_c5, n_points)
