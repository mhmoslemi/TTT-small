import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal

import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Starting with 45 samples
    
    # Construct a better initial guess: alternating high and low values
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 1.0
        elif i % 3 == 1:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    # Gradient descent with momentum for faster convergence
    learning_rate = 0.01
    momentum = 0.9
    decay_rate = 0.999
    max_iter = 1000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s:
        h = best_h.copy()
        h = h.astype(float)
        
        # Compute the overlap integral for all shifts
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5_val = np.max(corr)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for k in range(n_points):
            if corr[k] == c5_val:
                # Gradient contribution from this shift
                for i in range(n_points):
                    grad[i] += (1 - np.roll(h, k))[i] * (2.0 / n_points)
                    grad[i] -= h[i] * (2.0 / n_points) * (np.roll(h, k)[i] == h[i])
        
        # Update using momentum
        momentum_term = momentum * grad
        h = h - learning_rate * momentum_term
        
        # Ensure h remains in [0, 1]
        h = np.clip(h, 0, 1)
        
        # Check constraints
        total = np.sum(h) * (2.0 / n_points)
        if abs(total - 1.0) > 1e-6:
            # Normalize if constraint is violated
            h = h / total * 1.0
        
        # Update best solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Update learning rate with decay
        learning_rate *= decay_rate
        iter_count += 1
    
    # Try with a coarser grid to avoid local minima
    n_points_new = 60
    h_new = np.zeros(n_points_new)
    for i in range(n_points_new):
        if i % 3 == 0:
            h_new[i] = 1.0
        elif i % 3 == 1:
            h_new[i] = 0.5
        else:
            h_new[i] = 0.0
    h_new = h_new / np.sum(h_new) * (n_points_new / 2)
    
    # Evaluate on new grid
    current_c5_new = evaluate_erdos_solution(h_new)
    if current_c5_new < best_c5:
        best_c5 = current_c5_new
        best_h = h_new.copy()
    elif n_points_new < n_points:
        # Use the coarser grid if it's better and has fewer points
        best_h = h_new.copy()
        best_c5 = current_c5_new
        n_points = n_points_new
    
    return best_h, best_c5, n_points
