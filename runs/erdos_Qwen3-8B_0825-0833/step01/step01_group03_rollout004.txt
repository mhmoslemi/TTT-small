import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from functools import lru_cache

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are available

# Constants
MAX_ITERATIONS = 20
MAX_TIME = 60
INITIAL_H = initial_h_values
n_points_initial = len(INITIAL_H)
dx_initial = 2.0 / n_points_initial
target_c5 = 0.380876

def compute_overlap_integral(h, dx):
    h = np.array(h)
    shifted_h = np.zeros_like(h)
    for k in range(1, len(h) - 1):
        shifted_h[k:] = h[:len(h) - k]
    overlap = np.correlate(h, 1 - shifted_h, mode="full")
    return np.max(overlap * dx)

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = INITIAL_H.copy()
    n_points = n_points_initial
    dx = 2.0 / n_points
    
    # Try decreasing n_points
    for n_points in range(n_points_initial - 1, 10, -1):
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h[:len(INITIAL_H)] = INITIAL_H[:n_points]
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.isclose(sum_h, n_points / 2):
            c5 = compute_overlap_integral(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Need to adjust to meet the sum constraint
            h = h / sum_h * (n_points / 2)
            c5 = compute_overlap_integral(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Now perform optimization with more iterations
    for _ in range(MAX_ITERATIONS):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        
        h = np.array(best_h)
        n_points = len(h)
        dx = 2.0 / n_points
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            h = h / sum_h * (n_points / 2)
        
        # Compute shifts and current overlap
        shifts = np.arange(1, n_points)
        overlaps = []
        for k in shifts:
            shifted_h = np.zeros_like(h)
            shifted_h[k:] = h[:n_points - k]
            overlap = np.dot(h, 1 - shifted_h)
            overlaps.append(overlap)
        current_c5 = np.max(overlaps) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Optimization using CVXPY
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        obj = cp.Minimize(cp.Max([cp.sum(x * (1 - x[k:])) * dx for k in range(1, n_points)]))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=100, time_limit=10)
        except:
            pass
        
        if prob.status == 'OPTIMAL':
            h_opt = np.clip(x.value, 0, 1)
            sum_h_opt = np.sum(h_opt)
            if not np.isclose(sum_h_opt, n_points / 2):
                h_opt = h_opt / sum_h_opt * (n_points / 2)
            c5_opt = compute_overlap_integral(h_opt, dx)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    return best_h, best_c5, len(best_h)
