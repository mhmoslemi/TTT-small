def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 500  # A balance between accuracy and speed
    dx = 2.0 / n_points
    h0 = initial_h_values()
    h0 = h0[:n_points]  # Ensure length matches

    # Initial guess
    h = np.array(h0, dtype=np.float64)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Define the objective function for optimization
    def objective_func(h_values):
        nonlocal best_c5
        h = np.array(h_values, dtype=np.float64)
        # Compute the constraint sum(h) == n_points / 2
        constraint_sum = np.sum(h) - n_points / 2
        # Compute the maximum overlap
        h_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(h_corr) * dx
        # Update the best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Optimization settings
    max_iterations = 50
    for _ in range(max_iterations):
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_candidate = np.clip(h + perturbation, 0.0, 1.0)
        c5_candidate = objective_func(h_candidate)
        # Update the solution
        h = h_candidate

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h.tolist(), final_c5, n_points
