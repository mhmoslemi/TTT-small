import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=np.float64)
    
    # Define parameters
    dx = 2.0 / n_points
    integral = np.sum(h_values) * dx
    if not np.isclose(integral, 1.0):
        h_values = h_values / integral * 1.0
    
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        dx = 2.0 / len(h)
        h = np.concatenate([h, np.zeros(1)])  # To handle full convolution
        shifted = 1 - np.roll(h, 1)
        correlate = np.correlate(h, shifted, mode="full")
        c5 = np.max(correlate) * dx
        return c5
    
    best_c5 = compute_c5(h_values)
    best_h = h_values.copy()
    
    # Optimization parameters
    max_iterations = 100
    step_size = 0.01
    tolerance = 1e-3
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb h_values
        perturbation = np.random.uniform(-step_size, step_size, size=len(h_values))
        h_perturbed = h_values + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        
        # Ensure integral constraint
        integral_perturbed = np.sum(h_perturbed) * dx
        scale_factor = 1.0 / integral_perturbed
        h_perturbed = h_perturbed * scale_factor
        
        c5_perturbed = compute_c5(h_perturbed)
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
    
    # Further refine using gradient descent-like approach
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute current c5
        current_c5 = compute_c5(best_h)
        
        # Compute shifted and correlate
        h = best_h.copy()
        dx = 2.0 / len(h)
        h = np.concatenate([h, np.zeros(1)])
        shifted = 1 - np.roll(h, 1)
        correlate = np.correlate(h, shifted, mode="full")
        
        # Find maximum overlap position
        max_overlap_pos = np.argmax(correlate)
        max_overlap_val = correlate[max_overlap_pos]
        
        # Compute gradient-like direction to reduce overlap
        h_shifted = np.roll(h, 1)
        overlap = correlate[max_overlap_pos]
        gradient = -np.gradient(h, dx)
        gradient_shifted = -np.gradient(h_shifted, dx)
        direction = gradient[max_overlap_pos] + gradient_shifted[max_overlap_pos]
        
        # Perturb h to reduce overlap
        perturbation = np.zeros_like(h)
        perturbation[max_overlap_pos] = -direction * 0.01
        perturbation = np.clip(perturbation, -0.1, 0.1)
        h_new = h + perturbation
        h_new = np.clip(h_new, 0.0, 1.0)
        
        # Adjust integral
        integral_new = np.sum(h_new) * dx
        scale_factor = 1.0 / integral_new
        h_new = h_new * scale_factor
        
        c5_new = compute_c5(h_new)
        
        if c5_new < current_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final check using CVXPY for a better optimization
    n = len(best_h)
    h_var = cp.Variable(n, bounds=(0.0, 1.0))
    dx = 2.0 / n
    objectives = []
    constraints = [cp.sum(h_var) * dx == 1.0]
    
    for k in range(1, n):
        shifted = 1 - cp.roll(h_var, k)
        overlap = cp.sum(h_var * shifted) * dx
        objectives.append(overlap)
    
    obj = cp.Minimize(cp.Max(objectives))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == 'OPTIMAL':
            best_h = h_var.value
            best_c5 = compute_c5(best_h)
    except:
        pass
    
    return best_h, best_c5, n
