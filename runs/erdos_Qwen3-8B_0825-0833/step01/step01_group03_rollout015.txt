import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
initial_h_values = np.random.rand(42) * 2 - 1  # Placeholder for actual initial values
initial_h_values = np.clip(initial_h_values, 0, 1)
initial_h_values = initial_h_values * (42 / 2)  # Normalize to sum to 21
initial_h_values = initial_h_values / (42 / 2)  # Ensure sum is 1 when multiplied by dx (2/42)

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros_like(h)
    for k in range(1, n_points):
        shifted_h[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap * dx)
    return c5

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Initial setup
    n_points = 42
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = h / (np.sum(h) * dx) * (n_points / 2)  # Normalize to sum to 1 * dx
    
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Parameters for optimization
    max_iterations = 100
    learning_rate = 0.1
    tolerance = 1e-5
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute overlap integral
        c5 = evaluate_erdos_solution(h)
        
        # If we have a better solution, update
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient descent step
        h = h.copy()
        h[h < 0] = 0
        h[h > 1] = 1
        
        # Update h to reduce maximum overlap
        for k in range(1, n_points):
            overlap = np.correlate(h, 1 - h, mode="full")[k]
            overlap = overlap * dx
            if overlap > best_c5:
                # Adjust h to reduce overlap at this shift
                for i in range(n_points):
                    if i + k < n_points:
                        h[i + k] = np.clip(h[i + k] - learning_rate, 0, 1)
                    else:
                        h[i] = np.clip(h[i] - learning_rate, 0, 1)
        
        # Ensure h is within [0, 1] and normalizes to 1*dx
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) * (n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
