    def compute_c5(h):
        h = np.array(h)

        h_diff = 1 - h
        corr = np.correlate(h_pad, h_diff, mode="full")
        dx = 2.0 / len(h)
        c5 = np.max(corr * dx)
        return c5

    def objective_and_constraints(h):
        h = np.array(h)
        obj = compute_c5(h)
        # Constraint: integral of h is 1
        constraint = np.sum(h) * (2.0 / len(h)) - 1.0
        return obj, constraint

    # Optimization loop
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Try a few different starting points
        for _ in range(3):
            # Perturb the current best h
            perturbation = np.random.normal(0, 0.05, size=len(best_h))
            h_perturbed = np.clip(best_h + perturbation, 0, 1)

            current_c5 = compute_c5(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed.copy()
