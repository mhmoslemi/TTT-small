import numpy as np
import scipy.optimize as optimize
import time

import random

# Pre-imported functions
# evaluate_erdos_solution and initial_h_values are assumed to be available

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess from provided construction
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values, dtype=np.float64)
    
    # Function to compute the overlap integral C_5
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        # Normalize h to ensure the integral is 1
        h_norm = h / (np.sum(h) / n_points)
        # Compute the convolution
        conv = np.correlate(h_norm, 1 - h_norm, mode="full")
        # Multiply by dx
        dx = 2.0 / n_points
        c5 = np.max(conv * dx)
        return c5
    
    # Function to optimize the function h
    def optimize_h(n_points, initial_guess, max_iter=100, tolerance=1e-5):
        h = np.copy(initial_guess)
        for _ in range(max_iter):
            # Convert h to a function we can optimize
            def objective(x):
                h_new = np.copy(x)
                # Normalize the function to ensure the integral is 1
                h_norm = h_new / (np.sum(h_new) / n_points)
                # Compute the overlap integral
                conv = np.correlate(h_norm, 1 - h_norm, mode="full")
                dx = 2.0 / n_points
                c5 = np.max(conv * dx)
                return c5
            
            # Use L-BFGS-B optimization
            result = optimize.minimize(
                objective,
                h,
                bounds=[(0, 1) for _ in range(n_points)],
                method='L-BFGS-B',
                tol=tolerance
            )
            h = result.x
            if result.success:
                print(f"Iteration {_ + 1}, C5 = {result.fun:.8f}")
            else:
                print("Optimization failed")
                break
        return h
    
    # Try different n_points in the range 100 to 500
    best_c5 = float('inf')
    best_h = h_initial
    best_n = n_points
    
    # Start timing
    start_time = time.time()
    
    # Try with the initial n_points
    print(f"Trying with n_points = {n_points}")
    h_opt = optimize_h(n_points, h_initial)
    c5_opt = compute_c5(h_opt)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt
        best_n = n_points
    
    # Try with different n_points

        h_opt = optimize_h(n_points, h_initial)
        c5_opt = compute_c5(h_opt)

            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_points
    
    # Return the best solution found
