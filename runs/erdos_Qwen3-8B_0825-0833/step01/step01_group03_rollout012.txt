        # We use a constrained optimization with CVXPY

        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [

        overlap_integrals = []

        max_overlap = cp.Max(overlap_integrals)
        objective = cp.Minimize(max_overlap)

        try:
            prob = cp.Problem(objective, constraints)
