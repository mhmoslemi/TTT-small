```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming it's available

def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1  # Initial random guess
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Adjust to meet the integral constraint

    # Define the overlap integral for all shifts
    def compute_c5(h):
        h_vec = np.array(h)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx

    # Optimization
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Try multiple initializations
    for _ in range(5):
        # Start with a slightly different initial guess
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Check time budget
        if time.time() - start_time > budget_s:
            break

    # Try a smaller number of points for faster evaluation
    n_points = 50
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Define a simple construction with alternating values
    h[:n_points//2] = 1
    h[n_points//2:] = 0
    h = h * (n_points / 2) / np.sum(h)
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_h = h.copy()
        best_c5 = current_c5

    # Final check
    if time.time() - start_time < budget_s:
        # Try one more optimization step
        x = cp.Variable(n_points)
        constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
        objective = cp.Maximize(cp.Max(np.correlate(x.value, 1 - x.value, mode="full")) * dx)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(max_iter=100, solver='SCIP')
            if prob.status == 'optimal':
                best_h = x.value
                best_c5 = evaluate_erdos_solution(best_h)
        except:
            pass

    return best_h, best_c5, n_points
```
