import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
initial_h_values = np.load('initial_h_values.npy')  # Load initial construction (n=45 samples)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 45  # Start with the initial construction
    
    # Function to compute overlap integral
    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
    
    # Function to optimize h using convex optimization
    def optimize_h(n_points, initial_h=None):
        dx = 2.0 / n_points
        h = cp.Variable(n_points)
        
        # Objective: minimize the maximum overlap integral
        overlap = cp.Maximize(cp.Max([cp.sum(cp.multiply(h[i], 1 - h[i + k])) for k in range(1, n_points)]))
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,  # Integral constraint
            0 <= h,
            h <= 1
        ]
        
        # Solve the optimization problem
        prob = cp.Problem(overlap, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-5, verbose=True)
            if prob.status != 'OPTIMAL':
                return None
            return h.value
        except:
            return None
    
    # Function to evaluate and improve the solution
    def evaluate_and_improve(n_points, initial_h=None):
        # Start with initial guess
        h = initial_h if initial_h is not None else np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy integral constraint
        
        # Optimize using convex optimization
        optimized_h = optimize_h(n_points, initial_h=h)
        if optimized_h is not None:
            h = optimized_h
        else:
            # If optimization fails, use a genetic algorithm
            def fitness_func(h):
                h = h.reshape(n_points)
                h = h * (n_points / 2) / np.sum(h)
                return compute_c5(h)
            
            bounds = [(0, 1) for _ in range(n_points)]
            h = opt.differential_evolution(fitness_func, bounds, strategy='best1bin', maxiter=100, popsize=15, tol=1e-4, mutation=(0.5, 1), recombination=0.9, seed=seed).x
        
        # Evaluate the solution
        c5 = compute_c5(h)
        return h, c5
    
    # Run the optimization
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try increasing the number of points gradually
    for n_points in range(45, 100, 5):
        # Start with initial guess from previous solution
        h, c5 = evaluate_and_improve(n_points, initial_h=best_h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_c5, len(best_h)
