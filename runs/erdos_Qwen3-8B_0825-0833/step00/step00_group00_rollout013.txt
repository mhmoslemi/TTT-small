import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros((n_points, n_points))
    for i in range(n_points):
        h_shifted[i, :] = h
    corr = np.zeros(n_points)
    for k in range(n_points):
        corr[k] = np.sum(h * (1 - h_shifted[k, :])) * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 45  # start with 45 points
    initial_h = initial_h_values()
    h = initial_h.copy()
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # Optimization strategy: use gradient descent with a custom loss function
    def objective_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.zeros(n_points)
        for k in range(n_points):
            corr[k] = np.sum(h * (1 - np.roll(h, k))) * (2.0 / n_points)
        return np.max(corr)
    
    def objective_func_with_gradient(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.zeros(n_points)
        for k in range(n_points):
            corr[k] = np.sum(h * (1 - np.roll(h, k))) * (2.0 / n_points)
        c5_val = np.max(corr)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for k in range(n_points):
            # Gradient from max(corr)
            if corr[k] == c5_val:
                for i in range(n_points):
                    grad[i] += (1 - np.roll(h, k))[i] * (2.0 / n_points)
                    grad[i] -= h[i] * (2.0 / n_points) * (np.roll(h, k)[i] == h[i])
        return c5_val, grad.flatten()
    
    # Gradient descent optimization
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Use approximate gradient
        objective = lambda x: objective_func(x)
        grad = lambda x: objective_func_with_gradient(x)[1]
        
        # Use L-BFGS-B optimizer
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            jac=grad,
            bounds=[(0, 1) for _ in range(n_points)]
        )
        
        h = result.x
        current_c5 = evaluate_erdos_solution(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Optionally, try a finer grid or different n_points
        # This is a heuristic to escape local minima
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
