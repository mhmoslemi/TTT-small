def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()

    # Define the initial h_values
    n_points = 100
    initial_h_values = np.array([0.5] * n_points)
    initial_h_values = np.random.permutation(initial_h_values)
    initial_h_values[::2] = 0.0
    initial_h_values[1::2] = 1.0

    # Optimize the function
    def objective(h_values):
        # Compute the overlap integral for all shifts
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * 2.0 / n_points

    def optimize_h_values(h_values):
        # Convert to CVXPY variables
        h = cp.Variable(n_points, non Negativity=True, bounded=(0, 1))
        # Constraint: sum(h) == n_points / 2
        constraint = [cp.sum(h) == n_points / 2]

        # Objective: minimize the maximum overlap
        corr = np.correlate(h.value, 1 - h.value, mode="full")
        max_corr = np.max(corr)
        objective = cp.Minimize(max_corr)

        prob = cp.Problem(objective, constraint)
        result = prob.solve(solver=cp.MOSEK, eps=1e-5)
        return h.value

    # Optimization loop
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points

    while time.time() - start_time < budget_s:
        # Generate a new candidate h_values
        candidate_h = best_h.copy()
        idx = np.random.randint(0, n_points)
        candidate_h[idx] = np.random.uniform(0, 1)

        # Compute the overlap integral for the candidate
        c5 = evaluate_erdos_solution(candidate_h)

        # Update the best solution
        if c5 < best_c5:
            best_h = candidate_h
            best_c5 = c5
            best_n_points = n_points

        # Try using CVXPY optimization
        optimized_h = optimize_h_values(best_h)
        optimized_c5 = evaluate_erdos_solution(optimized_h)

        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
            best_n_points = n_points

    return best_h, best_c5, best_n_points
