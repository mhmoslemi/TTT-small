import numpy as np

import time

from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.0, 0.0, 0.5, 0.0, 0.5, 0.5, 0.0, 0.5, 0.0, 0.0, 0.0, 0.5, 0.0, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h_values = initial_h_values[:n_points].copy()  # Start with initial values

    # Ensure sum(h) * dx = 1
    h_values = h_values / np.sum(h_values) * (n_points / 2)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        # Compute the overlap integrals
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def objective_2(h_flat):
        h = h_flat.reshape(n_points)
        # Compute the overlap integrals
        corr = np.correlate(h, 1 - h, mode="full")
        return corr

    def objective_3(h_flat):
        h = h_flat.reshape(n_points)
        # Compute the overlap integrals
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Initial evaluation
    best_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()

    # Iterative improvement
    for _ in range(10):  # 10 iterations
        start_time = time.time()
        # Use a local optimization method to refine h
        result = opt.minimize(
            fun=objective,
            x0=h_values,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        h_values = result.x
        # Ensure the integral constraint is satisfied
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        # Evaluate the solution
        current_c5 = evaluate_erdos_solution(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        if time.time() - start_time > budget_s:
            break

    # Try with different n_points for better performance
    n_points = 150
    dx = 2.0 / n_points
    h_values = best_h[:n_points].copy()
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    for _ in range(5):
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=h_values,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        h_values = result.x
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        current_c5 = evaluate_erdos_solution(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        if time.time() - start_time > budget_s:
            break

    return (best_h, best_c5, n_points)
