import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.0]*45)  # Placeholder for the actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial guess
    n_points = 100  # Start with 100 points, can increase if time permits
    h = np.random.rand(n_points) * 2 - 1  # Random initialization in [-1, 1], then scale to [0, 1]
    h = (h + 1) / 2
    h /= np.sum(h) * (2.0 / n_points)  # Normalize to ensure ∫ h(x) dx = 1
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed = 0
    
    # Define function to compute C5
    def compute_c5(h):
        return evaluate_erdos_solution(h)
    
    # Define function to compute the overlap integral for a given shift
    def overlap_integral(h, shift):
        x = np.linspace(0, 2, n_points)
        shifted_h = np.roll(h, shift)
        return np.sum(h * (1 - shifted_h)) * (2.0 / n_points)
    
    # Define function to compute max overlap
    def compute_max_overlap(h):
        return np.max([overlap_integral(h, k) for k in range(1, n_points)])
    
    # Use a simple gradient descent approach
    for _ in range(100):  # Max 100 iterations
        if elapsed >= budget_s - 1:
            break
        # Compute current C5 value
        current_c5 = compute_max_overlap(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        # Evaluate gradient or direction manually
        # Here we use a simple finite difference approach to approximate the direction
        direction = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 0.01
            h_plus /= np.sum(h_plus) * (2.0 / n_points)  # Maintain integral constraint
            c5_plus = compute_max_overlap(h_plus)
            h_minus = h.copy()
            h_minus[i] -= 0.01
            h_minus /= np.sum(h_minus) * (2.0 / n_points)  # Maintain integral constraint
            c5_minus = compute_max_overlap(h_minus)
            direction[i] = (c5_plus - c5_minus) / 0.02
        # Simple gradient descent step
        h -= 0.01 * direction
        # Ensure h stays within [0, 1] and normalize
        h = np.clip(h, 0, 1)
        h /= np.sum(h) * (2.0 / n_points)
        elapsed = time.time() - start_time
