```python
import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 70  # Start with the initial number of points

    # Try with different numbers of points, starting with the initial one
    for n in [70, 100, 150, 200, 250]:
        if time.time() - start_time > budget_s:
            break
        # Generate initial guess
        h_initial = copy.deepcopy(initial_h_values)
        h_initial = np.resize(h_initial, n)
        h_initial = np.clip(h_initial, 0, 1)
        # Ensure integral constraint is satisfied
        h_initial = h_initial / (np.sum(h_initial) / (n / 2))
        # Define optimization variables
        x = optimize.Bounds(0, 1, len(h_initial))
        # Objective function: minimize max correlation
        def objective(h):
            h = np.reshape(h, (n,))
            # Ensure integral constraint
            h = h / (np.sum(h) / (n / 2))
            # Compute the overlap integral for all k
            corr = np.correlate(h, 1 - h, mode="full")
            # We want to minimize the maximum of corr
            return np.max(corr) * (2.0 / n)
        # Use differential evolution to minimize the objective
        result = optimize.differential_evolution(
            objective,
            bounds=x,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            disp=False,
            x0=h_initial,
            polish=True
        )
        h_opt = result.x
        c5 = result.fun
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            n_points = n
        print(f"n={n}, best_c5={best_c5}, time={time.time()-start_time:.2f}s")
    
    # Evaluate the best solution found
    final_result = evaluate_erdos_solution(best_h, n_points)
    print(f"Final solution: C5 = {final_result}")
    return best_h, final_result, n_points
```
