import numpy as np

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points to use
    n_points = 100
    
    # Initialize the function values
    h = initial_h_values()
    if len(h) > n_points:
        h = h[:n_points]
    else:
        h = np.pad(h, (0, n_points - len(h)), 'constant')
    
    # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h = h * scale
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Set up optimization parameters
    max_iterations = 200
    learning_rate = 0.05
    momentum = 0.8
    random_perturbation = 0.02
    grad_buffer = np.zeros(n_points)
    
    # Perform a gradient descent with momentum and random perturbations
    for iteration in range(max_iterations):
        if time.time() > budget_s:
            break
        
        # Add random perturbation to escape local minima
        h += np.random.uniform(-random_perturbation, random_perturbation, size=n_points)
        h = np.clip(h, 0, 1)
        
        # Enforce the integral constraint
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = h * scale
        
        # Compute the objective
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5_val = np.max(corr)
        
        # Compute the gradient numerically
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * (2.0 / n_points)
            c5_val_plus = np.max(corr_plus)
            
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full") * (2.0 / n_points)
            c5_val_minus = np.max(corr_minus)
            
            grad[i] = (c5_val_plus - c5_val_minus) / (2 * 1e-6)
        
        # Update with momentum
        grad_buffer = momentum * grad_buffer + grad
        h -= learning_rate * grad_buffer
        
        # Ensure h stays within [0, 1]
        h = np.clip(h, 0, 1)
        
        # Enforce the integral constraint
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h = h * scale
        
        current_c5 = evaluate_erdos_solution(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
