import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # We will use a more sophisticated strategy with multiple restarts and adaptive parameter tuning
    # Starting with a small n_points and increasing if time allows
    n_points_list = [45, 60, 80, 100, 120, 140, 160]
    best_h = None
    best_c5 = 1.0
    max_attempts = 5  # More attempts for better exploration
    total_time = 0
    
    start_time = time.time()
    
    def evaluate_and_store(h_values):
        nonlocal best_c5, best_h
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate multiple initial guesses with random perturbations
        for _ in range(max_attempts):
            try:
                h_initial = initial_h_values()
                if len(h_initial) > n_points:
                    h_initial = h_initial[:n_points]
                else:
                    h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')
                
                # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
                sum_h = np.sum(h_initial)
                scale = (n_points / 2) / sum_h
                h_initial = h_initial * scale
                
                # Define the objective function
                def objective(x):
                    h = x.reshape(n_points)
                    corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
                    return np.max(corr)
                
                # Define the constraint: sum(h) = n_points / 2
                def constraint(x):
                    return np.sum(x) - n_points / 2
                
                cons = {'type': 'eq', 'fun': constraint}
                
                # Bounds for each h[i] (must be in [0, 1])
                bounds = [(0.0, 1.0) for _ in range(n_points)]
                
                # Optimization settings with adaptive parameters
                options = {
                    'disp': False,
                    'maxiter': 150,
                    'ftol': 1e-8,
                    'tol': 1e-8
                }
                
                # Perform optimization
                result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options=options)
                evaluate_and_store(result.x)
                
                # Add random noise to the solution for diversity
                noise = np.random.normal(0, 0.05, size=n_points)
                noisy_h = np.clip(result.x + noise, 0, 1)
                noisy_h = noisy_h * (n_points / 2) / np.sum(noisy_h)
                evaluate_and_store(noisy_h)
                
                # Early stopping if time is up
                if time.time() - start_time > budget_s:
                    break
            except:
                # Skip if an error occurs
                pass
    
    return best_h, best_c5, len(best_h)
