def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Configuration
    n_points = 80  # Choose a reasonable number of samples
    dx = 2.0 / n_points
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.1
    max_generations = 100
    time_budget = budget_s
    
    # Initialize the genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    # Initialize tools
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    # Fitness function
    def eval_func(individual):
        h = np.array(individual)
        # Ensure sum constraint
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_scaled = h * scale
        # Compute C5
        corr = np.correlate(h_scaled, 1 - h_scaled, mode="full") * dx
        return (np.max(corr),)

    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxSimulatedBinaryBounded, eta=3.0, crossover_rate=crossover_rate)
    toolbox.register("mutate", tools.mutPolynomialBounded, eta=3.0, indpb=mutation_rate)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Start the genetic algorithm
    pop = toolbox.population(n=population_size)
    start_time = time.time()
    best_h = None
    best_c5 = 1.0
    
    for gen in range(max_generations):
        if time.time() - start_time > time_budget:
            break
        
        # Evaluate the population
        offspring = algorithms.varAnd(pop, toolbox, cxpb=crossover_rate, mutpb=mutation_rate)
        fits = [toolbox.evaluate(ind) for ind in offspring]
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        
        # Select the next generation
        pop = toolbox.select(offspring, k=len(pop))
        
        # Update best solution
        for ind in pop:
            if ind.fitness.values[0] < best_c5:
                best_c5 = ind.fitness.values[0]
                best_h = np.array(ind)

    return best_h, best_c5, n_points
