```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start from the initial construction and try a few different point counts
    n_points_list = [45, 60, 80, 100, 120, 140, 160]
    best_h = None
    best_c5 = 1.0
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize from the initial construction or a random configuration
        h = initial_h_values()
        if len(h) > n_points:
            h = h[:n_points]
        else:
            h = np.pad(h, (0, n_points - len(h)), 'constant')
        
        # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        
        # Define the objective and constraint
        h_var = cp.Variable(n_points, lb=0, ub=1)
        objective = cp.Maximize(cp.Max(np.correlate(h_var, 1 - h_var, mode="full") * (2.0 / n_points)))
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Solve using CVXPY
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver='SCS', eps=1e-4, max_iters=500)
            if prob.status == 'optimal':
                h_opt = h_var.value
                c5_val = evaluate_erdos_solution(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt
        except:
            pass
        
        # Try a few random perturbations to escape local optima
        for _ in range(3):
            if time.time() - start_time > budget_s:
                break
            
            # Add small random noise to the current solution
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            sum_h = np.sum(h_perturbed)
            scale = (n_points / 2) / sum_h
            h_perturbed = h_perturbed * scale
            
            # Evaluate the solution
            c5_val = evaluate_erdos_solution(h_perturbed)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_perturbed
    
    return best_h, best_c5, len(best_h)
```
