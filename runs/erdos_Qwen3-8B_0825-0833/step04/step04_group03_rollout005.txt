import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points and budget
    n_points = 60  # Use a slightly higher number for better resolution
    max_attempts = 3
    best_h = None
    best_c5 = 1.0
    total_time = 0
    start_time = time.time()
    
    # Function to create a better initial guess with pattern-based construction
    def generate_initial_guess(n_points):
        h = np.zeros(n_points)
        # Use a pattern of 0.5 and 0.0 with some variation to reduce overlap
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
            elif i % 3 == 1:
                h[i] = 0.5
            else:
                h[i] = 0.0
        # Normalize to satisfy the integral constraint
        total = np.sum(h)
        h = h * (n_points / 2) / total
        return h
    
    # Generate initial guess
    h_initial = generate_initial_guess(n_points)
    
    # Define the objective function
    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(corr)
    
    # Define the constraint: sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Bounds for each h[i] (must be in [0, 1])
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    options = {
        'disp': False,
        'maxiter': 100,
        'ftol': 1e-8,
        'eps': 1e-8
    }
    
    # Perform the optimization
    for _ in range(max_attempts):
        # Use a different initial guess to avoid local optima
        if _ == 0:
            h_initial = generate_initial_guess(n_points)
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Optimize
        try:
            result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options=options)
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                print(f"Attempt {_+1}: Achieved C5 = {best_c5}")
                
            # Check time
            current_time = time.time()
            if current_time - start_time > budget_s:
                break
        except:
            pass
    
    return best_h, best_c5, n_points
