```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
from functools import partial

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define a list of n_points to try, prioritizing more resolution
    n_points_list = [45, 60, 80, 100, 120, 140, 160]
    best_h = None
    best_c5 = 1.0
    start_time = time.time()
    
    # Define objective function for optimization
    def objective(h_flat):
        h = h_flat.reshape(-1)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        return np.max(corr)
    
    # Define constraint function for optimization
    def constraint(h_flat):
        h = h_flat.reshape(-1)
        return np.sum(h) - len(h) / 2  # Enforces sum(h) = len(h)/2
    
    # Define a function to optimize with different initializations
    def optimize_solution(n_points, initial_guess=None):
        # Ensure initial guess is of correct length
        if initial_guess is None:
            initial_guess = initial_h_values()
            initial_guess = initial_guess[:n_points]
            if len(initial_guess) < n_points:
                initial_guess = np.pad(initial_guess, (0, n_points - len(initial_guess)), 'constant')
        
        # Scale initial guess to satisfy the integral constraint
        initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)
        
        # Create optimization variables
        h = cp.Variable(n_points)
        
        # Define objective and constraints
        obj = cp.Minimize(objective(h))
        cons = [constraint(h) == 0]
        
        # Set up and solve optimization problem
        prob = cp.Problem(obj, cons)
        
        # Try different solvers based on availability
        solvers = ['ECOS', 'SCS', 'CVXOPT', 'GUROBI', 'MOSEK']
        for solver in solvers:
            try:
                prob.solve(solver=solver, eps=1e-6, verbose=False)
                if prob.status == 'optimal':
                    h_opt = h.value
                    h_opt = np.clip(h_opt, 0, 1)
                    return h_opt
            except:
                continue
        
        # Fallback to a local optimization if exact solution not found
        h_local = initial_guess.copy()
        learning_rate = 0.1
        momentum = 0.9
        max_iterations = 100
        grad_buffer = np.zeros(n_points)
        random_perturbation = 0.01
        
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            
            # Add random perturbation to escape local minima
            h_local += np.random.uniform(-random_perturbation, random_perturbation, size=n_points)
            h_local = np.clip(h_local, 0, 1)
            
            # Enforce the integral constraint
            sum_h = np.sum(h_local)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h_local = h_local * scale
            
            # Compute the objective
            corr = np.correlate(h_local, 1 - h_local, mode="full") * (2.0 / n_points)
            c5_val = np.max(corr)
            
            # Compute the gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_local.copy()
                h_plus[i] += 1e-6
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * (2.0 / n_points)
                c5_val_plus = np.max(corr_plus)
                
                h_minus = h_local.copy()
                h_minus[i] -= 1e-6
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full") * (2.0 / n_points)
                c5_val_minus = np.max(corr_minus)
                
                grad[i] = (c5_val_plus - c5_val_minus) / (2 * 1e-6)
            
            # Update with momentum
            grad_buffer = momentum * grad_buffer + grad
            h_local -= learning_rate * grad_buffer
            
            # Ensure h stays within [0, 1]
            h_local = np.clip(h_local, 0, 1)
            
            # Enforce the integral constraint
            sum_h = np.sum(h_local)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h_local = h_local * scale
            
            current_c5 = evaluate_erdos_solution(h_local)
            
            if current_c5 < evaluate_erdos_solution(h_local):
                h_local = h_local.copy()
        
        return h_local
    
    # Try different n_points values and find the best solution
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with the provided initial values or a random configuration
        initial_guess = initial_h_values()
        if len(initial_guess) > n_points:
            initial_guess = initial_guess[:n_points]
        else:
            initial_guess = np.pad(initial_guess, (0, n_points - len(initial_guess)), 'constant')
        
        # Scale initial guess to satisfy the integral constraint
        initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)
        
        # Optimize solution
        h_opt = optimize_solution(n_points, initial_guess)
        
        # Evaluate the solution
        c5_val = evaluate_erdos_solution(h_opt)
        
        # Update best solution
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
    
    return best_h, best_c5, len(best_h)
```
