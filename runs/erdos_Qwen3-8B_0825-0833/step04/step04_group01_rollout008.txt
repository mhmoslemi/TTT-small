import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Start with the initial guess
    n_points = 500
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    h_initial = np.clip(h_initial, 0, 1)
    
    def objective(h_values):
        h = h_values
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint_integral(h_values):
        h = h_values
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx
        return integral - 1.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial optimization with the starting point
    result = optimize.minimize(
        objective,
        x0=h_initial,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = result.fun

    # Try local optimization with a different method
    result_local = optimize.minimize(
        objective,
        x0=best_h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 50}
    )

    best_h = result_local.x
    best_c5 = result_local.fun

    # Try a global search with simulated annealing to escape local minima
    def objective_sa(h_values):
        h = h_values
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    sa_result = optimize.minimize(
        objective_sa,
        x0=best_h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 50}
    )

    best_h = sa_result.x
    best_c5 = sa_result.fun

    # Ensure integral constraint is satisfied
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    # Final evaluation
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx

    # Early termination if time is up
    elapsed = time.time() - time.time()
    if elapsed > budget_s - 1:
        return (best_h, best_c5, n_points)

    return (best_h, best_c5, n_points)
