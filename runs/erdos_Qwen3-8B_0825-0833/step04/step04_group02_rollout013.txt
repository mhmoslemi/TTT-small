import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points_list = [45, 60, 80, 100]
    best_h = None
    best_c5 = 1.0
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initial guess based on the provided initial values or a random configuration
        h = initial_h_values()
        if len(h) > n_points:
            h = h[:n_points]
        else:
            h = np.pad(h, (0, n_points - len(h)), 'constant')
        
        # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        
        best_h_current = h.copy()
        best_c5_current = evaluate_erdos_solution(h)
        
        # Use a more sophisticated optimization method: particle swarm optimization
        # Define the objective function
        def objective(h_vec):
            h = h_vec.reshape(n_points)
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = h * scale
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Particle swarm optimization parameters
        num_particles = 20
        inertia_weight = 0.8
        cognitive_weight = 1.2
        social_weight = 1.2
        max_iterations = 50
        particles = np.random.uniform(0, 1, (num_particles, n_points))
        velocities = np.random.uniform(-0.1, 0.1, (num_particles, n_points))
        personal_best_positions = particles.copy()
        personal_best_values = np.array([objective(p) for p in particles])
        global_best_position = particles[np.argmin(personal_best_values)]
        global_best_value = np.min(personal_best_values)
        
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            
            for i in range(num_particles):
                # Update velocity
                r1 = np.random.rand(n_points)
                r2 = np.random.rand(n_points)
                velocities[i] = inertia_weight * velocities[i] + \
                                cognitive_weight * r1 * (personal_best_positions[i] - particles[i]) + \
                                social_weight * r2 * (global_best_position - particles[i])
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                
                # Evaluate new position
                current_value = objective(particles[i])
                
                # Update personal best
                if current_value < personal_best_values[i]:
                    personal_best_values[i] = current_value
                    personal_best_positions[i] = particles[i].copy()
            
            # Update global best
            current_global_best_value = np.min(personal_best_values)
            if current_global_best_value < global_best_value:
                global_best_value = current_global_best_value
                global_best_position = personal_best_positions[np.argmin(personal_best_values)]
        
        # After PSO, use the best position as the current best
        best_h_current = global_best_position.copy()
        best_c5_current = global_best_value
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
    
    return best_h, best_c5, len(best_h)
