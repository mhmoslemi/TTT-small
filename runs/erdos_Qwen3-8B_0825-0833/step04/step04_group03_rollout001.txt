import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points to find the best configuration
    n_points_list = [45, 60, 80, 100, 120]
    best_h = None
    best_c5 = 1.0
    max_attempts = 3
    total_time = 0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        for _ in range(max_attempts):
            # Create an initial guess based on periodic patterns
            h_initial = np.zeros(n_points)
            for i in range(n_points):
                h_initial[i] = 0.5 + 0.5 * np.cos(2 * np.pi * i / n_points)
            h_initial = np.clip(h_initial, 0.0, 1.0)
            
            # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
            sum_h = np.sum(h_initial)
            scale = (n_points / 2) / sum_h
            h_initial = h_initial * scale
            
            # Define the objective function
            def objective(x):
                h = x.reshape(n_points)
                corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
                return np.max(corr)
            
            # Define the constraint: sum(h) = n_points / 2
            def constraint(x):
                return np.sum(x) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint}
            
            # Bounds for each h[i] (must be in [0, 1])
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            # Optimization settings
            options = {
                'disp': False,
                'maxiter': 100
            }
            
            # Perform optimization
            try:
                result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options=options)
                best_h_current = result.x
                best_c5_current = evaluate_erdos_solution(best_h_current)
                
                if best_c5_current < best_c5:
                    best_c5 = best_c5_current
                    best_h = best_h_current
            except:
                pass
            
            # Check time
            current_time = time.time()
            if current_time - start_time > budget_s:
                break
    
    return best_h, best_c5, len(best_h)
