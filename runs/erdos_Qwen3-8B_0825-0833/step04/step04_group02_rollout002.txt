import numpy as np
import scipy.optimize as optimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points_list = [45, 60, 80, 100, 120]
    best_h = None
    best_c5 = 1.0
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate an initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = np.roll(h, n_points//4)
        h = h + np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h = h * scale
        
        best_h_current = h.copy()
        best_c5_current = evaluate_erdos_solution(h)
        
        # Optimization loop
        max_iterations = 100
        learning_rate = 0.1
        momentum = 0.9
        random_perturbation = 0.05
        grad_buffer = np.zeros(n_points)
        
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            
            # Add random perturbation to escape local minima
            h += np.random.uniform(-random_perturbation, random_perturbation, size=n_points)
            h = np.clip(h, 0, 1)
            
            # Enforce the integral constraint
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = h * scale
            
            # Compute the objective
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            c5_val = np.max(corr)
            
            # Compute the gradient
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * (2.0 / n_points)
                c5_val_plus = np.max(corr_plus)
                
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full") * (2.0 / n_points)
                c5_val_minus = np.max(corr_minus)
                
                grad[i] = (c5_val_plus - c5_val_minus) / (2 * 1e-6)
            
            # Update with momentum
            grad_buffer = momentum * grad_buffer + grad
            h -= learning_rate * grad_buffer
            
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0, 1)
            
            # Enforce the integral constraint
            sum_h = np.sum(h)
            if sum_h != 0:
                scale = (n_points / 2) / sum_h
                h = h * scale
            
            current_c5 = evaluate_erdos_solution(h)
            
            if current_c5 < best_c5_current:
                best_c5_current = current_c5
                best_h_current = h.copy()
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
    
    return best_h, best_c5, len(best_h)
