import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points to find the best configuration
    n_points_list = [45, 50, 60, 70, 80, 90, 100, 110, 120, 130, 140]
    best_h = None
    best_c5 = 1.0
    
    start_time = time.time()
    
    # Start with the initial solution
    initial_h = initial_h_values()
    n_points_initial = len(initial_h)
    if n_points_initial > 140:
        initial_h = initial_h[:140]
    else:
        initial_h = np.pad(initial_h, (0, 140 - len(initial_h)), 'constant')
    
    # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
    sum_h = np.sum(initial_h)
    scale = (140 / 2) / sum_h
    initial_h = initial_h * scale
    
    # Perform optimization for the initial n_points
    def objective(x):
        h = x.reshape(140)
        corr = np.correlate(h, 1 - h, mode="full") * (2.0 / 140)
        return np.max(corr)
    
    def constraint(x):
        return np.sum(x) - 140 / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0.0, 1.0) for _ in range(140)]
    
    options = {
        'disp': False,
        'maxiter': 200,
        'ftol': 1e-8,
        'eps': 1e-8
    }
    
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
    best_h_current = result.x
    best_c5_current = evaluate_erdos_solution(best_h_current)
    
    if best_c5_current < best_c5:
        best_c5 = best_c5_current
        best_h = best_h_current
    
    # Try other n_points
    for n_points in n_points_list[1:]:
        if time.time() - start_time > budget_s:
            break
        
        # Create a new initial guess based on the best solution so far
        if best_h is not None:
            h_initial = np.zeros(n_points)
            if n_points <= len(best_h):
                h_initial[:n_points] = best_h[:n_points]
            else:
                h_initial[:len(best_h)] = best_h
                h_initial[len(best_h):] = 0.5
        else:
            h_initial = initial_h_values()
            if len(h_initial) > n_points:
                h_initial = h_initial[:n_points]
            else:
                h_initial = np.pad(h_initial, (0, n_points - len(h_initial)), 'constant')
        
        # Ensure the integral constraint is satisfied (sum(h) = n_points / 2)
        sum_h = np.sum(h_initial)
        scale = (n_points / 2) / sum_h
        h_initial = h_initial * scale
        
        # Define the objective function
        def objective(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(corr)
        
        # Define the constraint: sum(h) = n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Bounds for each h[i] (must be in [0, 1])
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization settings
        options = {
            'disp': False,
            'maxiter': 200,
            'ftol': 1e-8,
            'eps': 1e-8
        }
        
        # Perform optimization
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        
        # Extract the best solution found
        best_h_current = result.x
        best_c5_current = evaluate_erdos_solution(best_h_current)
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
    
    return best_h, best_c5, len(best_h)
