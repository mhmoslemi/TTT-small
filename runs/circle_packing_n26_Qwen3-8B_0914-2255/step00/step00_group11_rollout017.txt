import cvxpy as cp

    radii = []

        centers.append([x, y])
        radii.append(0.05)

    # We'll assume a uniform radius for all circles and optimize their positions
    # to maximize the sum of radii

    # Optimization variables
    r = cp.Variable(1)
    c_x = cp.Variable((num_circles, 1))
    c_y = cp.Variable((num_circles, 1))

    # Objective: maximize the total radii
    objective = cp.Maximize(num_circles * r)

    # Constraints: each circle is within the unit square
    constraints = [
        cp.SOC(c_x[i] - r <= 1.0 for i in range(num_circles)),
        cp.SOC(r <= c_x[i] for i in range(num_circles)),
        cp.SOC(c_y[i] - r <= 1.0 for i in range(num_circles)),
        cp.SOC(r <= c_y[i] for i in range(num_circles)),
    ]

            dist = cp.sqrt((c_x[i] - c_x[j])**2 + (c_y[i] - c_y[j])**2)
            constraints.append(dist >= 2 * r)

    # Set up and solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS)

    # Extract results
    centers = np.vstack((c_x.value, c_y.value)).T
    radii = np.full(num_circles, r.value)

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
