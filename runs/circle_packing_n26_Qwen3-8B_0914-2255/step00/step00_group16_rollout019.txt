from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():

    def initial_hexagonal_packing(n_circles):
        rows = int(np.ceil(n_circles / 4))
        cols = 4
        centers = []
        for row in range(rows):
            for col in range(cols):
                if col >= n_circles - row * cols:
                    break
                x = col + 0.5 * (row % 2)
                y = row + 0.5
                centers.append([x, y])
        return np.array(centers)

    n_circles = 26
    initial_centers = initial_hexagonal_packing(n_circles)
    initial_radii = np.full(n_circles, 0.1)

        return -np.sum(radii)  # Maximize sum of radii by minimizing negative sum

    def constraints(radii):
        # Boundary constraints
        for i in range(n_circles):
            if initial_centers[i, 0] - radii[i] < 0 or initial_centers[i, 0] + radii[i] > 1:
                return False
            if initial_centers[i, 1] - radii[i] < 0 or initial_centers[i, 1] + radii[i] > 1:
                return False
        # Non-overlap constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = distance.euclidean(initial_centers[i], initial_centers[j])
                if dist < radii[i] + radii[j] - 1e-6:
                    return False
        return True

    # Optimization
    bounds = [(0, 1) for _ in range(n_circles)]
    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints={
            'type': 'ineq',
            'fun': constraints,
            'jac': None
        }

    centers = initial_centers
