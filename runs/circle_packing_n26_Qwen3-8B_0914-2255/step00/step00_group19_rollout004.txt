import numpy as np
import scipy.optimize as opt
import warnings

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def compute_overlap_penalty(centers, radii, epsilon=1e-8):
    n = centers.shape[0]
    penalty = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - epsilon:
                overlap = (radii[i] + radii[j] - dist) - epsilon
                penalty += overlap
    return penalty

def run_packing():
    # Initial grid placement: 5x5 grid with 1 at the center and 24 around it
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place circles in a grid
    for i in range(n):
        if i == 0:
            # Center circle
            centers[i] = [0.5, 0.5]
            radii[i] = 0.1
        else:
            row = (i - 1) // 5
            col = (i - 1) % 5
            x = 0.2 + col * 0.2
            y = 0.2 + row * 0.2
            centers[i] = [x, y]
            radii[i] = 0.1

    def objective_function(x):
        # x contains all centers and radii in flattened format
        x_flat = x.reshape(-1)
        centers_flat = x_flat[:n*2].reshape(n, 2)

                return 1e6  # Penalize out-of-bound centers

        # Compute overlap penalty
        penalty = compute_overlap_penalty(centers_flat, radii_flat)
        return penalty

    # Optimization parameters
    bounds = []
    for _ in range(n*2):
        bounds.append((0, 1))  # x and y coordinates must be in [0, 1]
    for _ in range(n):
        bounds.append((0, 1))  # Radii must be non-negative (but we'll start with positive values)

    # Initial guess
    x0 = np.zeros(n*3)
    for i in range(n):
        x0[i*2] = centers[i, 0]
        x0[i*2 + 1] = centers[i, 1]
        x0[n*2 + i] = radii[i]

    # Optimization
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        result = opt.minimize(
            objective_function,
            x0,
            bounds=bounds,
            method='L-BFGS-B',
