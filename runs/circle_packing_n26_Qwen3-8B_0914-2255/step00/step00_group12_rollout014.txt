import math
from scipy.optimize import minimize

def run_packing():

    # Hexagonal packing grid parameters
    radius = 0.1  # Initial guess
    centers = []

    layer_radii = [1, 2, 3, 4]
    layers = []
    for layer in layer_radii:
        layer_centers = []
        for angle in range(0, 360, 60):
            theta = math.radians(angle)
            x = math.cos(theta)
            y = math.sin(theta)
            layer_centers.append((x * layer, y * layer))
        layers.append(layer_centers)

    # Flatten into a single list of positions
    positions = []
    for layer in layers:
        for x, y in layer:
            positions.append((x, y))

    # Scale to unit square and center
    scale = 0.5  # Adjust this to fit better
    square = np.array([(0.5, 0.5)])  # Central position
    positions = np.array(positions) * scale + square

    # Add all positions
    centers = positions

    # Optimize radii
    def objective(radii):
        return -np.sum(radii)  # Maximize sum

    def constraints(radii):
        n = len(radii)
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.sqrt(dx * dx + dy * dy)
                dists[i, j] = dist
                dists[j, i] = dist

        # Ensure all circles are inside unit square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False

        # Ensure non-overlap
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False

        return True

    # Optimization
    initial_radii = np.random.uniform(0.05, 0.2, size=num_circles)
    result = minimize(
        objective, initial_radii,
        constraints=constraints,
        method='SLSQP',
        bounds=[(0, 0.3) for _ in range(num_circles)]
