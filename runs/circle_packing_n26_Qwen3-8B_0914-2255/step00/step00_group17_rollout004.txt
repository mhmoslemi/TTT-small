import math

def run_packing():

    # Parameters
    n = 26
    rows = [5, 4, 4, 3, 2, 2]
    row_height = 1.0 / (len(rows) + 1)  # Spacing between rows
    col_width = 1.0 / (max(rows) + 1)  # Spacing between columns (row-wise)
    
    centers = []
    for r_idx, r in enumerate(rows):
        for c_idx in range(r):
            x = (c_idx + 0.5) * col_width
            y = (r_idx + 0.5) * row_height
            centers.append([x, y])
    
    # Convert to numpy array
    centers = np.array(centers)
    
    # Radius assignment based on nearest neighbor and boundary constraints
    radii = np.zeros(n)
    
    for i in range(n):
        x, y = centers[i]
        first_neighbor = np.inf
        
        # Check vertical neighbors
        for j in range(n):
            if i == j:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = math.hypot(dx, dy)
            if dist < first_neighbor:
                first_neighbor = dist
        
        # Check horizontal bounds
        left_dist = x
        right_dist = 1 - x
        bottom_dist = y
        top_dist = 1 - y
        
        # Compute minimal spacing
        min_dist = min(first_neighbor, left_dist, right_dist, bottom_dist, top_dist)
        radii[i] = min_dist / 2
    
    # Build final result array
    return centers, radii, np.sum(radii)
