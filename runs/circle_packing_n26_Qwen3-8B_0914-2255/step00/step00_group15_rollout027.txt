import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Grid layout with 5 rows and 5 columns, but we will embed 26 circles
    # We will place them in a grid pattern, but allow some to adjust radii
    # to better utilize the space

    # Grid positions for the circles
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    centers = []
    for i in range(5):
        for j in range(5):
            if i * 5 + j < n:
                centers.append([grid_x[j], grid_y[i]])
    centers = np.array(centers)

    # Optimize radii: variables will be the radii for each circle
    radii = cp.Variable(n)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints:
    # Each circle must fit within the unit square
    constraints = []
    for i in range(n):
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-12)
        constraints.append(x + radii[i] <= 1 + 1e-12)
        constraints.append(y - radii[i] >= -1e-12)
        constraints.append(y + radii[i] <= 1 + 1e-12)

    # Each pair of circles must not overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)

    # Extract results
    centers_result = centers
    radii_result = radii.value
    sum_radii = np.sum(radii_result)

    return (centers_result, radii_result, sum_radii)
