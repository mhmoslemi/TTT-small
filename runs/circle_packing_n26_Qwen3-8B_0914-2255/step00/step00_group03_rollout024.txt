def run_packing():

    n = 26

    # Initial grid layout: 7 columns x 4 rows = 28 positions
    # We'll use 26 out of the 28
    grid = np.linspace(0, 1, 8)  # 8 points for x (0 to 1), 7 intervals
    row_positions = np.linspace(0, 1, 5)  # 5 points for y

    centers = []
    for i in range(7):
        for j in range(4):
            x = grid[i]
            y = row_positions[j]
            centers.append((x, y))
    centers = np.array(centers)

    # We'll exclude the last two points to make it 26
    centers = centers[:26]

    # Initial guesses for radii
    radii = np.full(n, 0.05)

    # Optimization parameters
    learning_rate = 0.01
    max_iterations = 10000
    penalty_coeff = 100.0

    for _ in range(max_iterations):

        # Objective function
        def objective(params):
            flat_params = params.reshape(-1)
            x = flat_params[:n * 2].reshape(n, 2)
            r = flat_params[n * 2:]

            # Calculate total radius
            total_radius = np.sum(r)

            # Penalty terms for boundary violation
            boundary_penalty = 0
            for i in range(n):
                x_i, y_i = x[i]
                r_i = r[i]
                if x_i - r_i < -1e-6 or x_i + r_i > 1.0 + 1e-6:
                    boundary_penalty += r_i
                if y_i - r_i < -1e-6 or y_i + r_i > 1.0 + 1e-6:
                    boundary_penalty += r_i

            # Penalty terms for overlap
            overlap_penalty = 0

                    dx = x[i, 0] - x[j, 0]
                    dy = x[i, 1] - x[j, 1]
                    dist = np.sqrt(dx * dx + dy * dy)
                    if dist < r[i] + r[j] - 1e-6:
                        overlap_penalty += (r[i] + r[j]) - dist

            penalty = boundary_penalty + overlap_penalty
            return -total_radius + penalty_coeff * penalty

        # Optimization using gradient descent
        params = np.concatenate([centers.reshape(-1), radii])
        params = params.astype(float)

        gradient = np.zeros_like(params)
        for i in range(n):
            for j in range(i + 1, n):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
