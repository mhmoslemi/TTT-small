import numpy as np
import cvxpy as cp

def run_packing():
    # Hexagonal grid constants
    max_cols = 5  # Number of columns
    max_rows = 5  # Number of rows
    spacing = 0.5  # Spacing between adjacent circles

    # Generate initial hexagonal grid
    centers = []
    radii = []

    for row in range(max_rows):
        for col in range(max_cols):
            x = col * spacing
            y = row * (spacing * np.sqrt(3) / 2)
            centers.append([x, y])
            # Initial radius based on spacing
            radii.append(spacing / 2)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define optimization variables
    n = len(centers)
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    # Boundary constraints
    for i in range(n):
        constraints.append(0 <= x[i, 0] - r[i])
        constraints.append(x[i, 0] + r[i] <= 1)
        constraints.append(0 <= x[i, 1] - r[i])
        constraints.append(x[i, 1] + r[i] <= 1)

    # Non-overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-8)

    # Optimization problem
    prob = cp.Problem(objective, constraints)

    # Solve the optimization
    prob.solve(solver=cp.SCS)

    # Extract results
    centers = x.value
    radii = r.value
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
