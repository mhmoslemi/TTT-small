import numpy as np
import scipy.optimize as opt
from typing import Tuple

def create_hexagonal_grid(num_circles: int) -> np.ndarray:
    """
    Create a hexagonal grid for circle centers with uniform spacing.
    """
    if num_circles == 0:
        return np.array([])
    
    # Estimate row and column counts for hexagonal grid
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    # Calculate spacing based on circle packing efficiency
    spacing = 1 / (rows + 1)
    offset = spacing / 2
    
    grid = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + (i % 2) * offset
            y = i * spacing
            if np.sqrt(x**2 + y**2) <= 1 + 1e-12:
                grid.append([x, y])
    
    # If more circles are needed, just add more in a hexagonal fashion
    while len(grid) < num_circles:
        for i in range(len(grid)):
            x, y = grid[i]
            # Add mirrored circle across x-axis
            grid.append([x, 1 - y])
    
    return np.array(grid)

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial estimate: create a hexagonal packing
    centers = create_hexagonal_grid(26)
    radii = np.full(26, 0.05)  # Initial radius estimate
    
    # Define the objective: maximize sum of radii
    def objective(x: np.ndarray) -> float:
        # x is flattened: [radii_0, radii_1, ..., radii_25]
        return -np.sum(x)  # we minimize negative sum to maximize sum

    # Define constraints:
    # 1. All centers are within [0,1]^2
    # 2. All radii are non-negative
    # 3. Pairwise distances between centers ≥ radii[i] + radii[j]
    
    # Optimize with bounds and constraints
    n = centers.shape[0]
    bounds = [(0, 1) for _ in range(n)]  # radii bounds

    # Create a list of constraints for pairwise circle distances
    constraints = []

    for i in range(n):
        for j in range(i + 1, n):
            # Constraint: distance between centers >= radii[i] + radii[j]
            def constraint_func(x):
                r_i = x[i]
                r_j = x[j]
                dist = np.linalg.norm(centers[i] - centers[j])
                return dist - r_i - r_j
            constraints.append({'type': 'ineq', 'fun': constraint_func})

    # Add boundary constraints
    for i in range(n):
        # Constraint: center[i][0] - radius[i] >= 0
        def constraint_left_x(x):
            return centers[i, 0] - x[i]
        constraints.append({'type': 'ineq', 'fun': constraint_left_x})

        # Constraint: center[i][0] + radius[i] <= 1
        def constraint_right_x(x):
            return 1 - centers[i, 0] - x[i]
        constraints.append({'type': 'ineq', 'fun': constraint_right_x})

        # Constraint: center[i][1] - radius[i] >= 0
        def constraint_left_y(x):
            return centers[i, 1] - x[i]
        constraints.append({'type': 'ineq', 'fun': constraint_left_y})

        # Constraint: center[i][1] + radius[i] <= 1
        def constraint_right_y(x):
            return 1 - centers[i, 1] - x[i]
        constraints.append({'type': 'ineq', 'fun': constraint_right_y})

    # Optimization: minimize -sum(radii) with constraints
    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract results
    optimized_radii = result.x
    sum_radii = np.sum(optimized_radii)
    return centers, optimized_radii, sum_radii
