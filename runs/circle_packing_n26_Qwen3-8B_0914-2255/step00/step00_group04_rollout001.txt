import cvxpy as cp

def run_packing():
    n = 26
    # Initial grid configuration: 5 rows and 5 columns with spacing between circles
    # We'll fix 4 corner circles and place the rest in a grid-like configuration

    # Define variables
    x = cp.Variable((n, 1))
    y = cp.Variable((n, 1))
    r = cp.Variable(n)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints: ensure circles stay within the unit square [0,1]^2
    constraints = []
    for i in range(n):
        constraints.append(x[i] - r[i] >= 0)
        constraints.append(x[i] + r[i] <= 1)
        constraints.append(y[i] - r[i] >= 0)
        constraints.append(y[i] + r[i] <= 1)

    # Non-overlapping constraint between any pair of circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i] - x[j])**2 + (y[i] - y[j])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-8)

    # Create and solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-5)

    # Extract solution
    centers = np.column_stack((x.value, y.value))
    radii = r.value
    sum_radii = np.sum(radii)

    # Here, we assume the solver guarantees correctness

    return (centers, radii, sum_radii)
