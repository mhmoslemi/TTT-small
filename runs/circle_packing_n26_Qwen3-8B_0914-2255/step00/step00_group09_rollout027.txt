from scipy.optimize import minimize

def run_packing():

    radii = np.zeros(num_circles)

    rows = 4
    cols = 6
    spacing = 1.0 / (rows + 1)  # Spacing between rows
    col_spacing = 1.0 / (cols + 1)

        for j in range(rows):
            x = i * col_spacing
            y = j * spacing
            centers[i * rows + j] = [x, y]

    radii = np.full(num_circles, spacing * 0.6)

    # 2. No overlap between circles
    def constraint_inside(radii):
        for i in range(num_circles):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return False
        return True

    def constraint_overlap(radii):
