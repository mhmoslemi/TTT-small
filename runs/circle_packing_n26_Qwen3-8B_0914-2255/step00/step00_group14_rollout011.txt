import numpy as np
import scipy.optimize as opt

def distance_squared(a, b):
    return np.sum((a - b) ** 2)

def run_packing():

    # 4 rows × 6 columns (14 circles), and 12 more arranged around the perimeter
    centers = []
    radii = []

    for i in range(4):

            x = j / 5
            y = i / 3
            centers.append([x, y])
            radii.append(0.05)  # Initial radius estimate

    # Add 12 more circles on the outer perimeter
    for i in range(12):
        angle = i * (2 * np.pi / 12)
        r = 0.1
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers.append([x, y])
        radii.append(r)

    # Initial guess for radii
    initial_radii = np.array(radii)

    # Objective function: Maximize sum of radii
    def objective(r):
        return -np.sum(r)  # Negative because we minimize

    # Constraints: Distance between centers is at least sum of radii
    def constraint(r):
        dists = np.zeros((len(centers), len(centers)))
        for i in range(len(centers)):

        return dists - (r.reshape(-1, 1) + r.reshape(1, -1)) + 1e-12

    bounds = [(0, 1) for _ in range(len(centers))]  # Radii must be non-negative

    result = opt.minimize(objective, initial_radii, bounds=bounds, constraints={"type": "ineq", "fun": constraint})

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
