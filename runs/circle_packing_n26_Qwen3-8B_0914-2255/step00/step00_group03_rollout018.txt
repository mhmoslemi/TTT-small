import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():

    n_cols = 5
    grid_x = np.linspace(0, 1, n_cols)
    grid_y = np.linspace(0, 1, n_rows)
    centers = np.array([[x, y] for y in grid_y for x in grid_x])
    
    # Initial guess for radii
    initial_radii = np.full(centers.shape[0], 0.05)

    def objective(radii):
        return -np.sum(radii)  # Negative because we use minimization

        return dist - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-8
    
    def constraint_func_wrap(radii):
        return constraint_func(radii).flatten()
    
    # Optimize using scipy
    bounds = [(0.0, 0.1) for _ in range(len(centers))]  # Radii cannot be negative or exceed 0.1
    result = optimize.minimize(

        constraints={"type": "ineq", "fun": constraint_func_wrap},

    )
