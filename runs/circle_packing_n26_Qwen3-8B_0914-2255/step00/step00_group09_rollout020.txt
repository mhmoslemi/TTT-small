import numpy as np
import scipy.optimize as opt

def run_packing():

    # Initial guesses for circle centers (hexagonal grid near the center)
    centers = np.zeros((n, 2))
    r = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = 0.3 + col * 0.25
        centers[i, 1] = 0.3 + row * 0.25
        r[i] = 0.05  # Initial small radius

    # Define optimization variables: positions and radii
    x = np.concatenate((centers.flatten(), r))

    def objective(x):
        return -np.sum(x[n*2:])  # Maximize sum of radii

    def constraint_boundary(x):
        pos = x[:n*2].reshape(n, 2)
        r = x[n*2:]
        for i in range(n):
            if pos[i, 0] - r[i] < -1e-8 or pos[i, 0] + r[i] > 1 + 1e-8:
                return -1
            if pos[i, 1] - r[i] < -1e-8 or pos[i, 1] + r[i] > 1 + 1e-8:
                return -1
        return 0

    def constraint_overlap(x):
        pos = x[:n*2].reshape(n, 2)
        r = x[n*2:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = pos[i, 0] - pos[j, 0]
                dy = pos[i, 1] - pos[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

        bounds.extend([(0, 1) for _ in range(2)])  # for positions
        bounds.extend([(0, 1) for _ in range(1)])  # for radii

    result = opt.minimize(
        objective,
        x,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    if not result.success:
        raise ValueError("Optimization failed to converge")

    optimal_x = result.x
    centers = optimal_x[:n*2].reshape(n, 2)
    radii = optimal_x[n*2:]
    sum_radii = np.sum(radii)
