from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import random

def run_packing():

    radii = np.zeros(n)

    rows = int(np.sqrt(n))

        return -np.sum(radii)

        return True

    bounds = [(0, 1.0) for _ in range(n)]

    initial_guess = np.full(n, 0.05)

    result = minimize(
        fun=objective,
        x0=initial_guess,

        method="SLSQP"
    )

    radii = result.x
