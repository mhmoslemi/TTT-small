import numpy as np
import scipy.optimize as optimize

def run_packing():

    num_circles = 26
    grid_radius = 0.08  # initial estimate for radius

    # Define initial grid positions
    centers = []
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = col / 4.5 + 0.1
        y = row / 4.5 + 0.1
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radii for all circles
    radii = np.full(num_circles, grid_radius)

    # Objective function: maximize total radius (minimize negative of sum)
    def objective(x):
        # x contains all the variables: [centers flattened, radii]
        x_centers = x[:2 * num_circles].reshape(num_circles, 2)
        x_radii = x[2 * num_circles:]
        # Objective: -sum(radii), as minimize
        return -np.sum(x_radii)

    # Constraints:
    # 1. Overlap between circles: distance between centers >= (radii[i] + radii[j])
    # 2. Boundary condition: x - r >= 0, x + r <= 1
    # 3. All variables should be non-negative (radii >= 0)

    # Define bounds for each variable
    # x[:2*num_circles] are x and y positions (each bounded between 0 and 1)
    # x[2*num_circles:] are radii (each bounded between 0 and 0.5)
    bounds = []
    for _ in range(num_circles):
        # x and y bounds
        bounds.append((0, 1))
        bounds.append((0, 1))
        # radius bounds
        bounds.append((0, 0.5))
    bounds = bounds[:2 * num_circles + num_circles]

    # Constraints for overlaps and boundaries
    def overlap_constraints(x):
        x_centers = x[:2 * num_circles].reshape(num_circles, 2)
        x_radii = x[2 * num_circles:]
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = x_centers[i, 0] - x_centers[j, 0]
                dy = x_centers[i, 1] - x_centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                constr = dist - (x_radii[i] + x_radii[j])
                constraints.append(constr)
        return constraints

    def boundary_constraints(x):
        x_centers = x[:2 * num_circles].reshape(num_circles, 2)
        x_radii = x[2 * num_circles:]
        constraints = []
        for i in range(num_circles):
            for j in range(2):
                # x - r >= 0
                x_val = x_centers[i, j]
                r = x_radii[i]
                constraints.append(x_val - r)
                # x + r <= 1
                constraints.append(1 - x_val - r)
        return constraints

    # Set up optimization problem
    # Initial guess: centers and radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Define constraints as a list for optimization
    cons = []
    cons.append({'type': 'ineq', 'fun': lambda x: overlap_constraints(x)})
    cons.append({'type': 'ineq', 'fun': lambda x: boundary_constraints(x)})

    # Run optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract centers and radii
    centers = result.x[:2 * num_circles].reshape(num_circles, 2)
    radii = result.x[2 * num_circles:]

    # Compute sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
