import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n = 26

    # Define grid spacing and initial placement
    grid_rows = 5
    grid_cols = 5
    grid_spacing = 1.0 / (grid_cols - 1)
    center_x = np.linspace(0, 1, grid_cols)
    center_y = np.linspace(0, 1, grid_rows)

    # Initialize centers and radii
    centers = []

        for j in range(grid_cols):
            x = center_x[j]
            y = center_y[i]
            centers.append([x, y])

    radii = np.full(n, 0.05)  # Initial small radius

    def distance(c1, c2):
        return np.sqrt((c1[0] - c2[0]) ** 2 + (c1[1] - c2[1]) ** 2)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: non-overlapping and boundaries
    constraints = []

    # Boundary constraint: each circle must be fully inside the unit square
    for i in range(n):
        def boundary_constraint(radii):

            return (x - radii[i] >= -1e-12) and (x + radii[i] <= 1 + 1e-12) and \
                   (y - radii[i] >= -1e-12) and (y + radii[i] <= 1 + 1e-12)
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: boundary_constraint(r)})

    # Non-overlapping constraint: for each pair of circles

            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: non_overlap_constraint(r)})

    bounds = [(0.0, 1.0) for _ in range(n)]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
