from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    num_circles = 26
    rows = 5
    cols = 5

    y_coords = np.linspace(0.1, 0.9, rows)
    centers = np.array([(x, y) for y in y_coords for x in x_coords])

    initial_radii = np.ones(num_circles) * 0.14

    # Constraints for QP:
    # - No circle outside the square
    # - No overlap between circles
    # - Radius ≥ 0

    def objective(radii):
        return -np.sum(radii)

    # Constraint: Each circle lies within the square
    def square_constraint(radii):
        return np.array([r for r in radii])

    # Constraint: No overlap between circles
    def no_overlap_constraint(radii):
        dists = np.zeros(num_circles)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                dists[i] += radii[i] + radii[j] - dist
                dists[j] += radii[i] + radii[j] - dist
        return dists

    # Bounds: All radii ≥ 0
    bounds = Bounds(np.zeros(num_circles), np.inf)

    # Solve via minimize with SLSQP
    result = minimize(
        fun=objective,
        x0=initial_radii,
        method='SLSQP',
        constraints=[
