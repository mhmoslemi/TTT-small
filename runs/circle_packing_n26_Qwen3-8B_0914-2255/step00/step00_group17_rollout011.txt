from scipy.optimize import minimize

def create_hexagonal_packing(n_circles):
    """
    Create a hexagonal packing of n_circles.
    Returns (x_centers, y_centers).
    """
    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))
    
    x = np.linspace(0, 1, cols + 1)
    y = np.linspace(0, 1, rows + 1)
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x_val = x[j] + (x[1] - x[0]) / 2 * (i % 2)
            y_val = y[i]
            centers.append((x_val, y_val))
    
    # Remove the last point if overfit
    if len(centers) > n_circles:
        centers.pop()
    
    return np.array(centers)

def scale_to_unit_square(centers):

    Scale the given centers to fit within the unit square.

    dy = y_max - y_min
    scale = min(1 / dx, 1 / dy)
