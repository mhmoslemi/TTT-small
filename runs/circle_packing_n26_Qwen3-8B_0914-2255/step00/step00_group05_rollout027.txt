from scipy.optimize import minimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal arrangement of 26 circles
    # We will use a hexagonal packing with varying rows to fit 26 circles

    # Create initial positions
    n = 26
    centers = []
    for row in range(6):
        for col in range(6 - row):
            x = col + 0.5 * row
            y = row
            centers.append([x, y])
    centers = np.array(centers)

    initial_radii = np.ones(n) * 0.05

    def objective(radii):
        return -np.sum(radii)  # Minimize -sum to maximize sum

    def constraint_func(radii):
        # Ensure all circles are within the unit square
        for i in range(n):
            x, y = centers[i]

                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return False
        # Ensure all circles are non-overlapping
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]

        initial_guess,
        constraints={"type": "constraint", "fun": constraint},

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):

            return False, f"Circle {i} has negative radius {radii[i]}"

            if dist < radii[i] + radii[j] - 1e-12:
