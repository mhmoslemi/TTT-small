import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    def hexagonal_packing(n):
        num_rows = int(np.ceil(np.sqrt(n)))
        num_cols = int(np.ceil(n / num_rows))
        x = np.linspace(0.5, 0.5, num_cols)
        y = np.linspace(0.5, 0.5, num_rows)
        max_radius = 0.25
        centers = []

            for j in range(num_cols):
                if i % 2 == 0:
                    x_pos = x[j]
                else:
                    x_pos = x[j] + 0.5 * (1 - 2 * x[j])  # Offset for odd rows
                centers.append([x_pos, y[i]])
        centers = np.array(centers)
        return centers, np.full(len(centers), max_radius)

    def objective(x, n):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return -np.sum(radii)

    def constraint(x, n):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=2))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    n = 26
    initial_centers, initial_radii = hexagonal_packing(n)
    initial_x = np.concatenate([initial_centers.flatten(), initial_radii])

    result = minimize(
        fun=objective,
        x0=initial_x,
        method='SLSQP',
        bounds=Bounds(bounds),
