import scipy.optimize as optimize
import cvxpy as cp

def run_packing():
    # Initial grid
    n = 26
    grid_x = np.linspace(0, 1, int(np.sqrt(n)) + 1)
    grid_y = np.linspace(0, 1, int(np.sqrt(n)) + 1)
    grid = np.array([(x, y) for y in grid_y for x in grid_x])
    centers = grid[:n]
    
    # Initial estimate of radii
    def initial_radius(x, y):
        nearest_neighbor = np.min(np.sqrt((centers - np.array([x, y])) ** 2), axis=1)
        return np.min(nearest_neighbor) / 2
    
    # Initial estimate of radii
    radii = np.array([initial_radius(x, y) for x, y in centers])
    
    # Objective: maximize sum of radii
    def objective(r):
        return -np.sum(r)  # Negative because we minimize
    
    # Constraints:
    # - Non-overlapping: dist between any two centers >= r_i + r_j
    # - Center within [0, 1]^2
    # - Radius >= 0
    
    # Define variables
    r = cp.Variable(n)
    
    # Constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            constraints.append(dist >= r[i] + r[j] - 1e-12)
        constraints.append(r[i] >= 0)
    
    # Optimization problem
    prob = cp.Problem(cp.Minimize(objective(r)), constraints)
    prob.solve()

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
