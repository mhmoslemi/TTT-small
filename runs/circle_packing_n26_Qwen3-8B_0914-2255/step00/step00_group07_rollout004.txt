import numpy as np
import cvxpy as cp

def run_packing():
    # Set up a hexagonal grid of 26 circles with equal radii
    # The grid can be generated with 8 rows and 4 columns (total circles = 8*4 - 3 = 29), but we'll adjust it for 26
    # I'll generate a hexagonal grid with 7 rows and 4 columns (total = 7*4 - 3 = 25), then add an extra circle in a corner

    # Generate a hexagonal grid with (nrows, ncols)
    def generate_hex_grid(nrows, ncols, side_length):
        grid = []
        for row in range(nrows):
            for col in range(ncols):
                x = col * (1 + side_length)
                y = row * (np.sqrt(3)/2) * (1 + side_length)
                grid.append((x, y))
        return np.array(grid)

    # Generate a hexagonal grid of 26 circles (adjusted to fit into [0,1]^2)
    side_length = 0.2
    grid = generate_hex_grid(7, 4, side_length)
    # Trim the grid to fit within the unit square and add one extra circle
    centers = []
    for x, y in grid:
        if 0 <= x <= 1 and 0 <= y <= 1:
            centers.append((x, y))
    if len(centers) < 26:
        # Add one more circle manually
        centers.append((0.7, 0.9))
    centers = np.array(centers)

    # Initial radii (equal for all)
    r = 0.1
    radii = np.full(len(centers), r)

    # Step 1: Optimize radii with fixed center positions
    # We will use a numerical optimization to maximize the sum of radii under the constraints

    # Define variables
    n = len(centers)
    r = cp.Variable(n)
    centers = cp.Parameter(n, 2)
    centers.value = centers
    constraints = []
    for i in range(n):
        if r[i] < 0:
            constraints.append(r[i] >= 0)
        if not (0 <= centers[i, 0] <= 1 and 0 <= centers[i, 1] <= 1):
            constraints.append(cp.Square(centers[i, 0]) <= 1)
            constraints.append(cp.Square(centers[i, 1]) <= 1)
    for i in range(n):
        for j in range(i+1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-6)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Solve the optimization
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.SCS)
    except:
        raise RuntimeError("Optimization failed")

    # Extract solution
    centers_final = centers.value
    radii_final = r.value
    sum_radii = np.sum(radii_final)

    # Validate the solution
    valid, msg = validate_packing(centers_final, radii_final)
    if not valid:
        raise RuntimeError(f"Invalid packing: {msg}")

    return centers_final, radii_final, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
