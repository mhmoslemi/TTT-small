import numpy as np
import cvxpy as cp

def run_packing():
    # Define initial hexagonal grid configuration
    num_rows = 6
    num_cols = 5
    grid = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j + 0.5 * i
            y = i
            grid.append((x, y))
    centers = np.array(grid)
    
    # Scale grid within [0, 1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    
    # Number of circles
    n = len(centers)
    
    # Define optimization variables
    radii = cp.Variable(n, nonneg=True)
    
    # Constraints
    constraints = []
    for i in range(n):
        # Circle i must be within the unit square
        x, y = centers[i]
        constraints.append(x - radii[i] >= -1e-12)
        constraints.append(x + radii[i] <= 1 + 1e-12)
        constraints.append(y - radii[i] >= -1e-12)
        constraints.append(y + radii[i] <= 1 + 1e-12)
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)
    
    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve()
    
    # Extract results
    centers_result = centers
    radii_result = radii.value
    sum_radii = np.sum(radii_result)
    
    return (centers_result, radii_result, sum_radii)
