from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Define initial guess: grid-based layout with a hexagonal pattern
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial guess: mostly circles of radius 0.04
    for i in range(n):
        r = 0.04
        radii[i] = r
        centers[i, 0] = i / 5
        centers[i, 1] = (i % 5) / 5

    # Optimization problem: maximize sum of radii
    # This is equivalent to minimizing the negative of sum of radii

    def objective(x):
        # x has shape (n * 2 + n) = (26*3) = 78 elements:
        # First 26*2 = 52: center coordinates (x, y) for each circle
        # Then next 26: radii for each circle
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]

        sum_radii = np.sum(radii_flat)
        return -sum_radii

    # Constraints: for each pair i, j < n: distance >= sum of radii
    def constraint_function(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii_flat[i] + radii_flat[j]))
        return np.array(constraints)

    # Nonlinear constraint: distance between centers >= sum of radii
    nonlinear_constraint = NonlinearConstraint(
        constraint_function,
        lb=np.full(n * (n - 1) // 2, 0),
        ub=np.full(n * (n - 1) // 2, np.inf)
    )

    # Bounds: center coordinates must be in [0,1], radii >= 0
    bounds = Bounds(
        lb=np.concatenate([np.zeros(2 * n), np.zeros(n)]),
        ub=np.concatenate([np.ones(2 * n), np.full(n, 1)])
    )

    x0 = np.zeros(3 * n)
    x0[:2 * n] = centers.reshape(-1)

    result = minimize(

        x0=x0,
        bounds=bounds,

    centers_final = result.x[:2 * n].reshape(n, 2)

    return (centers_final, radii_final, sum_radii)
