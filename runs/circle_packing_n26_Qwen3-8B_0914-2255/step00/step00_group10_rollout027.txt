import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial positions: grid + slightly staggered pattern
    positions = np.zeros((26, 2))
    radii = np.zeros(26)

    # Define grid points slightly offset and staggered
    grid = np.array([
        [0.1, 0.1],
        [0.2, 0.1],
        [0.3, 0.1],
        [0.4, 0.1],
        [0.5, 0.1],
        [0.6, 0.1],
        [0.7, 0.1],
        [0.8, 0.1],
        [0.9, 0.1],
        [0.1, 0.2],
        [0.2, 0.2],
        [0.3, 0.2],
        [0.4, 0.2],
        [0.5, 0.2],
        [0.6, 0.2],
        [0.7, 0.2],
        [0.8, 0.2],
        [0.9, 0.2],
        [0.1, 0.3],
        [0.2, 0.3],
        [0.3, 0.3],
        [0.4, 0.3],
        [0.5, 0.3],
        [0.6, 0.3],
        [0.7, 0.3],
    ])

    # Sort to have predictable grid first
    grid = grid[np.argsort(grid[:, 0])]
    positions[:25] = grid
    positions[25] = [0.05, 0.95]  # Place final circle in a corner

    # Initial radii
    radii = np.full(26, 0.05)

    # Optimization bounds
    bounds = []
    for i in range(26):
        bounds.append((0.0, 1.0))
    bounds = [(0.0, 1.0) for _ in range(26)]  # for radii

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint: distance between any two centers > r_i + r_j - 1e-8
    def constraint_func(x):
        # x has shape (52)
        centers = x[:52].reshape(26, 2)
        radii = x[52:]

        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1000  # Penalize overlapping circles
        return 0

    # Initial guess for positions (fixed)
    x0 = np.concatenate((positions.flatten(), radii))

    # Optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[{'fun': constraint_func}],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    centers = result.x[:52].reshape(26, 2)
    radii = result.x[52:]

    # Validate packing
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(f"Packing validation failed: {msg}")

    total_radii = np.sum(radii)
    return centers, radii, total_radii
