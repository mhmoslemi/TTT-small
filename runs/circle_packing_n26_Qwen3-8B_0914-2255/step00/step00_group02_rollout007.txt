from scipy.optimize import minimize

def run_packing():

    n = 26

    # Generate an initial hexagonal grid with spacing adjusted to fit within [0,1]×[0,1]
    # This is a simplified starting grid for demonstration
    grid = []
    for row in range(4):
        for col in range(6):
            x = col / 5.0
            y = row / 4.0
            if row % 2 == 0:
                x += 0.5 / 5.0
            grid.append((x, y))

    centers = np.array(grid)
    radii = np.full(n, 0.05)  # Initial guess for radii

    # Define the objective function to maximize sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum to maximize sum

    # Define the constraints
    constraints = []

    # Boundary constraint: circles must lie fully within the unit square
    def boundary_constraint(radii_vec):
        for i in range(n):
            if centers[i, 0] - radii_vec[i] < 0 or centers[i, 0] + radii_vec[i] > 1:
                return 1
            if centers[i, 1] - radii_vec[i] < 0 or centers[i, 1] + radii_vec[i] > 1:
                return 1
        return 0
    constraints.append({'type': 'ineq', 'fun': boundary_constraint})

    # Non-overlapping constraint: distance between centers must be at least sum of radii
    def overlap_constraint(radii_vec):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]

    # Optimize the radii to maximize the sum under constraints

        constraints=constraints
