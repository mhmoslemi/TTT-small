import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize

def run_packing():
    num_circles = 26

    # Adaptive initial placement with varying spacing and edge bias
    centers = []
    for row in range(6):
        for col in range(6):
            # Use a hexagonal grid with dynamic spacing
            x = col / (6 - 1)  # Normalize x to [0,1]
            y = row / (6 - 1)  # Normalize y to [0,1]
            centers.append([x, y])
    # Remove one circle for better packing
    centers.pop(0)
    centers = np.array(centers)

    # Initial radii estimate
    radii = np.full(num_circles, 0.08)

    # Objective function: maximize total radius (minimize negative of sum)
    def objective(x):
        x_centers = x[:2 * num_circles].reshape(num_circles, 2)
        x_radii = x[2 * num_circles:]
        return -np.sum(x_radii)

    # Bounds for each variable
    bounds = []
    for _ in range(num_circles):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 0.5))  # radius
    bounds = bounds[:2 * num_circles + num_circles]

    # Constraints:
    # 1. No overlap between circles
    def overlap_constraints(x):
        x_centers = x[:2 * num_circles].reshape(num_circles, 2)
        x_radii = x[2 * num_circles:]
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = x_centers[i, 0] - x_centers[j, 0]
                dy = x_centers[i, 1] - x_centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                constraints.append(dist - (x_radii[i] + x_radii[j]))
        return constraints

    # 2. Circles must stay within the square [0,1]×[0,1]
    def boundary_constraints(x):
        x_centers = x[:2 * num_circles].reshape(num_circles, 2)
        x_radii = x[2 * num_circles:]
        constraints = []
        for i in range(num_circles):
            for j in range(2):
                x_val = x_centers[i, j]
                r = x_radii[i]
                constraints.append(x_val - r)
                constraints.append(1 - x_val - r)
        return constraints

    # Initial guess: centers and radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Define constraints as a list for optimization
    cons = []
    cons.append({'type': 'ineq', 'fun': overlap_constraints})
    cons.append({'type': 'ineq', 'fun': boundary_constraints})

    # Run optimization with tighter tolerances and more iterations
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 1500, 'ftol': 1e-12, 'eps': 1e-8, 'iprint': 0}
    )

    # Extract centers and radii
    centers = result.x[:2 * num_circles].reshape(num_circles, 2)
    radii = result.x[2 * num_circles:]

    # Compute sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
