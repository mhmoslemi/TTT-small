import numpy as np
import cvxpy as cp

def run_packing():
    n = 26

    # Define variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints: circles must fit in square [0,1]x[0,1]
    constraints = []

    for i in range(n):
        # Circle i must be within the unit square
        constraints += [
            centers[i, 0] - radii[i] >= 0,
            centers[i, 0] + radii[i] <= 1,
            centers[i, 1] - radii[i] >= 0,
            centers[i, 1] + radii[i] <= 1,
        ]

    # Non-overlap constraint: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            # Compute distance between center i and center j
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints += [dist >= radii[i] + radii[j]]

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.OSQP, eps=1e-6)

    if not problem.status in ["OPTIMAL", "OPTIMAL_INFEASIBLE"]:
        raise ValueError("Optimization failed with status: %s" % problem.status)

    # Extract results
    final_centers = centers.value
    final_radii = radii.value

    sum_radii = np.sum(final_radii)
