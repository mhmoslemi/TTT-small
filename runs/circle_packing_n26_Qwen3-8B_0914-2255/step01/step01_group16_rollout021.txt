import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Define initial configuration: 3 rows of 8, 7, and 7 circles
    # This is a hybrid arrangement with some circles near the corners and some in the center
    # Initial guess for centers
    centers = np.array([
        [0.1, 0.1], [0.2, 0.1], [0.3, 0.1], [0.4, 0.1], [0.5, 0.1], [0.6, 0.1], [0.7, 0.1], [0.8, 0.1],
        [0.1, 0.5], [0.2, 0.5], [0.3, 0.5], [0.4, 0.5], [0.5, 0.5], [0.6, 0.5], [0.7, 0.5], [0.8, 0.5],
        [0.1, 0.9], [0.2, 0.9], [0.3, 0.9], [0.4, 0.9], [0.5, 0.9], [0.6, 0.9], [0.7, 0.9], [0.8, 0.9]
    ])
    
    # Initial guess for radii: based on spacing between centers
    radii = np.full(26, 0.05)  # Placeholder
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            radii[i] = min(radii[i], dist / 2)
            radii[j] = min(radii[j], dist / 2)
    
    # Define the objective function: maximize the sum of radii
    def objective(x):
        # x is a flat array: [r0, r1, ..., r25]
        radii = x.reshape(26)
        return -np.sum(radii)  # Negative for minimization

    # Define constraints
    def constraint1(x):
        # Radii must be non-negative
        radii = x.reshape(26)
        return radii >= 0

    def constraint2(x):
        # Each circle must be entirely inside the unit square
        radii = x.reshape(26)
        for i in range(26):
            x_i, y_i = centers[i]
            if x_i - radii[i] < -1e-12 or x_i + radii[i] > 1 + 1e-12:
                return -np.inf
            if y_i - radii[i] < -1e-12 or y_i + radii[i] > 1 + 1e-12:
                return -np.inf
        return 0

    def constraint3(x):
        # Circles must not overlap
        radii = x.reshape(26)
        for i, j in combinations(range(26), 2):
            x_i, y_i = centers[i]
            x_j, y_j = centers[j]
            dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            if dist < radii[i] + radii[j] - 1e-12:
                return -np.inf
        return 0

    # Setup optimization problem
    x0 = radii.flatten()
    constraints = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint3}
    ]

    result = minimize(
        fun=objective,
        x0=x0,
        method='SLSQP',
        bounds=[(0, 1)] * 26,
        constraints=constraints,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    if not result.success:
        raise ValueError("Optimization failed.")

    radii_opt = result.x
    centers_opt = centers
    sum_radii = np.sum(radii_opt)

    return (centers_opt, radii_opt, sum_radii)
