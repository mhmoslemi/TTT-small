import scipy.optimize as opt
from scipy.optimize import Bounds, NonlinearConstraint

def run_packing():

    centers = np.random.rand(n, 2) * 0.8 + 0.1  # Initial centers spread out with padding
    radii = np.random.rand(n) * 0.1  # Initial radii with small values

    def objective(radii):
        return -np.sum(radii)

    def boundary_constraint(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12:
                return -1
            if y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0

    def non_overlap_constraint(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    bounds = [(0, 1) for _ in range(n)]
    nonlinear_constraints = [
        NonlinearConstraint(boundary_constraint, 0, 0, jac=None),
        NonlinearConstraint(non_overlap_constraint, 0, 0, jac=None)
    ]

    result = opt.minimize(
        objective,
        x0=radii,
        bounds=bounds,
        constraints=nonlinear_constraints,
        method='SLSQP'
    )

    final_radii = result.x
    final_centers = centers

    if result.success:
        sum_radii = np.sum(final_radii)

        sum_radii = 0
