def create_hexagonal_grid(n, side_length):
    """Create a hexagonal grid with n circles, and return their centers."""
    grid = []
    for i in range(n):
        row = i // 3
        col = i % 3
        x = col * (side_length * np.sqrt(3)/2)
        y = row * (side_length * 1.5)
        grid.append((x, y))
    return np.array(grid)
