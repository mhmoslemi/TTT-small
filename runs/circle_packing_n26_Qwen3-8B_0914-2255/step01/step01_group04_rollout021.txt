import scipy.optimize as opt

        [0.95, 0.95],

    ]

        [0.05, 0.5],
        [0.5, 0.95],
        [0.95, 0.5]
    ]

    hex_centers = []
    r = 0.1  # Initial radius to compute angular spacing
    num_hex = n - len(corners) - len(edge_centers)

        max_rows = int(np.sqrt(num_hex / 2)) + 1
        num_cols = (num_hex + max_rows - 1) // max_rows
        
        # Generate hexagonal grid
        hex_centers = []
        for i in range(max_rows):
            for j in range(num_cols):
                x = 0.5 + (j - (num_cols - 1) / 2) * r * np.sqrt(3)
                y = 0.5 + i * r * 0.5
