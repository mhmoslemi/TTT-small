import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    n = 26

    # Initial hexagonal grid setup
    def hexagonal_grid(num_circles):
        centers = []
        for i in range(num_circles):
            row = i // 5
            col = i % 5
            x = 0.1 + col * 0.2
            y = 0.1 + row * 0.2
            centers.append((x, y))
        return np.array(centers)

    centers = hexagonal_grid(n)
    radii = np.full(n, 0.1)

    def objective(radii):
        return -np.sum(radii)  # Minimize the negative of the sum

    def constraint_func(radii):
        # Ensure non-overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        # Ensure circles are within the square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0

    bounds = [(0, 1) for _ in range(n)]

    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_func},
        method='SLSQP'
    )

    final_radii = result.x

    if result.success:

    return (final_centers, final_radii, sum_radii)
