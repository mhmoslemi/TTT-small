import numpy as np
import cvxpy as cp

def run_packing():
    # Define number of circles
    n = 26

    # Define variables for centers and radii
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary constraints (circles must be within [0,1]^2)
    for i in range(n):
        x, y = centers[i]
        constraints.append(x >= 0)
        constraints.append(x <= 1)
        constraints.append(y >= 0)
        constraints.append(y <= 1)

    # Non-overlapping constraints
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Non-negative radius constraint
    constraints.append(radii >= 0)

    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-5)

    # Extract results
    centers_np = centers.value
    radii_np = radii.value
    sum_radii = np.sum(radii_np)

    # Return the result
    return (centers_np, radii_np, sum_radii)
