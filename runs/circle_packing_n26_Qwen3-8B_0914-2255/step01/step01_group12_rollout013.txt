import numpy as np
import scipy.optimize as optimize

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():

    ])

    radii = np.full(26, 0.06)

    # Objective function: maximize sum of radii
    def objective(x):
        # x contains all variables: radii
        # We optimize with a negative sign to maximize
        return -np.sum(x)

    # Constraints
    constraints = []
    for i in range(26):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i]})  # Radii must be non-negative

    for i in range(26):
        for j in range(i + 1, 26):
            # Non-overlap constraint
            x_i, y_i = centers[i]
            x_j, y_j = centers[j]
            dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: dist - (x[i] + x[j])
            })

    # Bounds for radii
    bounds = [(0, 1)] * 26

    # Optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract optimized radii and centers
    radii_opt = result.x

    valid, message = validate_packing(centers, radii_opt)
    if not valid:
