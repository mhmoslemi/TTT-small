import scipy.optimize as opt

def run_packing():

    num_circles = 26
    centers = []
    for row in range(6):

                y = row * 0.25
                centers.append([x, y])

    centers.append([0.5, 0.5])

    centers = np.array(centers)
    radii = np.full(num_circles, 0.05)

        return -np.sum(radii)  # Maximize the sum of radii

        return np.sum(radii)  # Maximize the sum of radii
