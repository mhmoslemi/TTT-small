from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    grid = np.linspace(0, 1, 5)
    centers = []
    for i in range(5):
        for j in range(5):
            if i == 2 and j == 2:
                continue  # skip center circle
            centers.append([grid[i], grid[j]])
    centers = np.array(centers)

    n = len(centers)
    variables = np.concatenate([centers.flatten(), initial_radii])

    bounds = Bounds(
        lb=np.concatenate([np.zeros(2 * n), np.zeros(n)]),
        ub=np.concatenate([np.ones(2 * n), np.ones(n)])
    )

        radii_ = x[2 * n:]

        # Objective: maximize sum of radii
        return -np.sum(radii_)

    def constraints(x):
        # Extract centers and radii
        centers_ = x[:2 * n].reshape(n, 2)
        radii_ = x[2 * n:]

        # Enforce within square
        for i in range(n):
            x_i, y_i = centers_[i]
            r_i = radii_[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
