import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial hexagonal grid layout
    num_circles = 26
    grid_rows = int(np.sqrt(num_circles))
    grid_cols = num_circles // grid_rows + (1 if num_circles % grid_rows != 0 else 0)
    
    # Hexagonal packing: positions for centers
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // grid_cols
        col = i % grid_cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]
    
    # Normalize to unit square
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= np.sqrt(max_x**2 + max_y**2)
    
    # Initial radii based on distances to boundaries
    radii = np.zeros(num_circles)
    for i in range(num_circles):
        x, y = centers[i]
        radii[i] = min(x, 1 - x, y, 1 - y) * 0.95  # small buffer
    
    def objective(radii):
        return -np.sum(radii)  # maximize sum of radii
    
    def constraints(radii):
        # Ensure non-negative radii
        if any(r < 0 for r in radii):
            return np.inf
        
        # Ensure non-overlapping
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-9:
                    return np.inf
        return 0.0
    
    # Run optimization
    bounds = [(0, 1) for _ in range(num_circles)]
    result = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            opt.SLSQP(lambda r: constraints(r), np.zeros(num_circles), 0, 1e6)
        ],
        method='SLSQP',
        options={'disp': False}
    )

        if radii[i] < 0:

    return centers, radii, sum_radii
