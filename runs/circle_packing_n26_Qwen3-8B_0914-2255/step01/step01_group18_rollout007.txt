import scipy.optimize as opt

    centers_base = []
    radii_base = []

    r_center = 0.1

                y = y_pos
                if 0 <= x <= 1 and 0 <= y <= 1:
                    centers_base.append([x, y])
                    radii_base.append(r_center)

    # Add corner circles
    corners = [(0, 0), (1, 0), (1, 1), (0, 1)]

        centers_base.append([x + eps, y + eps])
        radii_base.append(r_corner)

    # Trying to fill gaps
    # Use local optimization to maximize sum of radii with constraints
    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(c, r):
        for i in range(len(c)):
            for j in range(i + 1, len(c)):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - eps:

        return True
