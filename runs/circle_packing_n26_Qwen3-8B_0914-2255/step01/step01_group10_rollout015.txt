```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import Bounds

def run_packing():
    # Define the number of circles
    N = 26

    # Define the number of initial clusters (for a hybrid approach)
    K = 4

    # Initial cluster centers (spread evenly across the square)
    cluster_centers = np.array([
        [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]
    ])

    # Initialize centers as a grid for the clusters
    cluster_radii = np.ones(K) * 0.1
    cluster_counts = [N // K] * K
    cluster_counts[-1] += N % K

    # Generate initial centers
    centers = []
    for i in range(K):
        x = np.linspace(cluster_centers[i, 0] - 0.2, cluster_centers[i, 0] + 0.2, cluster_counts[i])
        y = np.linspace(cluster_centers[i, 1] - 0.2, cluster_centers[i, 1] + 0.2, cluster_counts[i])
        for x_idx in range(len(x)):
            for y_idx in range(len(y)):
                centers.append([x[x_idx], y[y_idx]])

    centers = np.array(centers)
    radii = np.full(N, 0.1)

    def objective(radii):
        # Objective is to maximize the sum of radii
        return -np.sum(radii)

    def constraint_overlap(i, j, radii, centers):
        # Define constraint: distance between centers >= sum of radii
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - (radii[i] + radii[j])

    # Define bounds for radii
    radius_bounds = [(0, 1) for _ in range(N)]

    # Define constraints for non-overlapping circles
    constraints = []
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append({'type': 'ineq', 'fun': lambda radii, i=i, j=j: constraint_overlap(i, j, radii, centers)})

    # Define bounds and initial guess
    initial_guess = np.full(N, 0.1)

    # Optimize
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=Bounds(radius_bounds),
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract the result
    radii = result.x
    centers = centers

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)

    # Compute sum of radii
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
