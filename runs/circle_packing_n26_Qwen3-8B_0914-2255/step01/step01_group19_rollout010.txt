import scipy.optimize as optimize

    ])

    # Define the objective function to maximize sum of radii
    def objective(x, centers, radii):

    # Define the constraints
    def constraint_square(x, centers, radii):
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return -1
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:

        return 0

    def constraint_overlap(x, centers, radii):

                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
