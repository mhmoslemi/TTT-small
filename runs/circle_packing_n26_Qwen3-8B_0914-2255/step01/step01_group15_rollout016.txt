import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal packing for 26 circles
    rows = 5
    cols = 5
    radius_initial = 0.05  # Initial uniform radius

    # Generate starting centers in hexagonal pattern
    centers = []
    for r in range(rows):
        for c in range(cols):
            x = c + 0.5 * r
            y = r * np.sqrt(3) / 2
            if x < 1 and y < 1:
                centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(len(centers), radius_initial)

    # Optimization objective: minimize negative total area (i.e., maximize it)
    def objective(x):
        x_flat = x.reshape(-1, 2)
        r_flat = x[len(x_flat) // 2:]
        r_flat = r_flat.reshape(-1)
        return -np.sum(np.pi * r_flat ** 2)

    # Constraints: circles must be within the square
    def square_constraint(x):
        x_flat = x.reshape(-1, 2)
        r_flat = x[len(x_flat) // 2:]
        r_flat = r_flat.reshape(-1)
        for i in range(len(x_flat)):
            x_i, y_i = x_flat[i]
            r_i = r_flat[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8 or y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e-8:
                return -1
        return 0

    # Constraints: circles must not overlap
    def no_overlap_constraint(x):
        x_flat = x.reshape(-1, 2)
        r_flat = x[len(x_flat) // 2:]
        r_flat = r_flat.reshape(-1)
        for i in range(len(x_flat)):
            for j in range(i + 1, len(x_flat)):
                dx = x_flat[i, 0] - x_flat[j, 0]
                dy = x_flat[i, 1] - x_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r_flat[i] + r_flat[j] - 1e-8:
                    return -1
        return 0

    # Combine constraints
    cons = [
        {"type": "ineq", "fun": square_constraint},
        {"type": "ineq", "fun": no_overlap_constraint}
    ]

    # Define the variables: (x1, y1, x2, y2, ..., r1, r2, ..., r26)
    x0 = np.concatenate([centers.flatten(), radii])

    # Run optimization with bounds: x and y in [0, 1], r ≥ 0
    bounds = [(0, 1) for _ in range(len(centers) * 2)] + [(0, 1) for _ in range(len(radii))]

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons)

    # Extract final centers and radii
    final_centers = result.x[:len(centers) * 2].reshape(-1, 2)
    final_radii = result.x[len(centers) * 2:].reshape(-1)

    # Calculate total sum of radii
    total_sum = np.sum(final_radii)

    return final_centers, final_radii, total_sum
