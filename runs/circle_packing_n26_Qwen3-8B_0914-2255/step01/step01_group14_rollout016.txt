from scipy.optimize import minimize

def run_packing():

    def constraint_overlap(centers, radii, i, j):

        return dist - (radii[i] + radii[j])

    def constraint_boundary(centers, radii, i):
        x, y = centers[i]
        r = radii[i]
        return (x - r >= 0) + (x + r <= 1) + (y - r >= 0) + (y + r <= 1)

        y = 0.5 + r * np.sin(theta)
        grid.append([x, y])

    centers = np.array(grid)
    radii = np.full(n, 0.05)  # Initial guess for radii

    # Optimization bounds: radii must be positive, centers in [0,1]^2
    bounds = [(-np.inf, 1 - 1e-8) for _ in range(n * 2)]

    # Optimization constraints
    cons = []
    for i in range(n):

            cons.append({
