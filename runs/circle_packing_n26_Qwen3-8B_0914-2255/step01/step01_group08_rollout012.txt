- $ x_i $, $ y_i $: center coordinates of circle $ i $

Objective: maximize $ \sum_{i=1}^{26} r_i $

import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Initial grid layout: hexagonal packing, perturbed to fit square
    n = 26
    centers = []
    radii = []

    # Create 4 rows with 7, 6, 6, 7 circles for 26 total
    for row in range(4):
        for col in range(7 if row in (0, 3) else 6):
            x = col + 0.5 * (row % 2)
            y = row + 0.5
            centers.append([x, y])

    # Normalize centers to fit [0,1]^2
    centers = np.array(centers)
    center_min = centers.min(axis=0)
    center_max = centers.max(axis=0)
    scale = 1.0 / (center_max - center_min)
    centers = (centers - center_min) * scale

    # Initial guess for radii: uniform radius based on average spacing
    spacing = np.sqrt(np.sum((centers[1] - centers[0])**2))

    radii = [r0] * n

        for i in range(n):

    # Optimize with SLSQP

    centers = np.array(centers)

    sum_radii = np.sum(radii)
