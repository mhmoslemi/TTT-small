import numpy as np
import cvxpy as cp

def run_packing():
    # Number of circles
    n = 26
    
    # Initial grid-based layout with small spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = col * 0.2 + 0.1
        centers[i, 1] = row * 0.2 + 0.1
        
    # Define radius variable
    radii = cp.Variable(n)
    
    # Define constraints
    constraints = []
    for i in range(n):
        # Distance to boundaries
        dist_left = centers[i, 0] - radii[i]
        dist_right = 1 - centers[i, 0] - radii[i]
        dist_top = 1 - centers[i, 1] - radii[i]
        dist_bottom = centers[i, 1] - radii[i]
        constraints.append(dist_left >= -1e-8)
        constraints.append(dist_right >= -1e-8)
        constraints.append(dist_top >= -1e-8)
        constraints.append(dist_bottom >= -1e8)
    
    # Non-overlap constraints between circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)
    
    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS)
    
    # Extract results
    centers_np = centers.value
    radii_np = radii.value
    
    # Return as numpy arrays
    return (np.array(centers_np), np.array(radii_np), np.sum(radii_np))
