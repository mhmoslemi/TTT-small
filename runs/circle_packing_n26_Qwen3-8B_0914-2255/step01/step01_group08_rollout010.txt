import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial estimate of grid parameters
    num_rows = 5
    num_cols = 5
    side_length = 0.25
    radius_start = side_length / 2

    # Generate initial hexagonal grid
    centers = []
    radii = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j * side_length + (side_length / 2) * (i % 2)
            y = i * (np.sqrt(3) / 2) * side_length
            centers.append([x, y])
            radii.append(radius_start)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Now, refine the packing using optimization
    def objective(x):
        return -np.sum(x)  # We maximize sum of radii, so we minimize negative sum

    def constraints(x):
        # Extract current centers and radii
        new_centers = x[:2*len(centers)].reshape((len(centers), 2))
        new_radii = x[2*len(centers):]

        # Boundary constraints
        for i in range(len(new_centers)):
            x_i, y_i = new_centers[i]
            r_i = new_radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return False
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False

        # Non-overlapping constraints
        for i in range(len(new_centers)):
            for j in range(i + 1, len(new_centers)):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < new_radii[i] + new_radii[j] - 1e-12:
                    return False

        return True

    # Initial guess is the current centers and radii
    x0 = np.concatenate((centers.flatten(), radii))

    # Optimization settings
    bounds = []
    for i in range(len(centers)):
        bounds.extend([(0, 1)] * 2)  # x and y for each center
        bounds.extend([(0, 1)] * 1)  # radius for each circle

    # Run optimization
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': 1000, 'ftol': 1e-8}
    )

    # Extract results
    final_centers = result.x[:2*len(centers)].reshape((len(centers), 2))
