import torch
import triton
import triton.language as tl
from torch import Tensor
from typing import Tuple, Dict

def custom_kernel(data: Tuple[Tensor, Tensor, Dict[str, Tensor], Dict]) -> Tensor:
    """
    Implement the "outgoing" version of the TriMul operation using a fully optimized Triton kernel.
    This kernel handles the forward pass of the TriMul module for the given input and weights.

    The TriMul operation involves:
    1. Applying LayerNorm to the input
    2. Projection to hidden dimensions using linear projections
    3. Masking and gating operations
    4. A pairwise multiplication across the N dimension
    5. Normalization, gating, and final projection

    The kernel uses an efficient tiling strategy to minimize memory traffic and maximize data reuse.
    
    The output is a Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Define Triton kernel parameters
    BLOCK = 128
    num_blocks = triton.cdiv(seq_len * seq_len, BLOCK)

    # Triton kernel
    @triton.jit
    def tritond_kernel(
        input_ptr: tl.pointer_type,
        mask_ptr: tl.pointer_type,
        left_proj_ptr: tl.pointer_type,
        right_proj_ptr: tl.pointer_type,
        left_gate_ptr: tl.pointer_type,
        right_gate_ptr: tl.pointer_type,
        out_gate_ptr: tl.pointer_type,
        to_out_norm_ptr: tl.pointer_type,
        to_out_ptr: tl.pointer_type,
        output_ptr: tl.pointer_type,
        batch_size: tl.constexpr,
        seq_len: tl.constexpr,
        dim: tl.constexpr,
        hidden_dim: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        """
        Triton kernel to perform the TriMul operation efficiently using tiling.
        """
        # Calculate the index of the current thread
        pid = tl.program_id(0)
        # Calculate the position of the thread within the block
        local_pid = tl.program_id(1)

        # Calculate the global indices for the batch and sequence dimensions
        b = pid // (seq_len * seq_len)
        i = (pid // seq_len) % seq_len
        j = pid % seq_len

        # Calculate the index in the block
        block_idx = (local_pid // (seq_len * seq_len)) % seq_len
        block_j = (local_pid // seq_len) % seq_len
        block_k = local_pid % seq_len

        # Calculate the offsets for the current block
        offset_b = b * seq_len * seq_len * dim
        offset_i = i * seq_len * dim
        offset_j = j * seq_len * dim
        offset_k = seq_len * dim

        # Initialize the output
        out = tl.zeros((dim,), dtype=tl.float32)

        # Loop through the sequence dimension
        for k in range(seq_len):
            # Load input
            input_b = tl.load(input_ptr + offset_b + offset_i + k * dim)
            input_j = tl.load(input_ptr + offset_b + offset_j + k * dim)

            # Compute weights
            left_proj = tl.load(left_proj_ptr + k * hidden_dim)
            right_proj = tl.load(right_proj_ptr + k * hidden_dim)
            left_gate = tl.load(left_gate_ptr + k * hidden_dim)
            right_gate = tl.load(right_gate_ptr + k * hidden_dim)
            out_gate = tl.load(out_gate_ptr + k * hidden_dim)

            # Apply mask
            mask_val = tl.load(mask_ptr + b * seq_len * seq_len + i * seq_len + j)
            input_b = input_b * mask_val
            input_j = input_j * mask_val

            # Apply gates
            left_proj = left_proj * left_gate.sigmoid()
            right_proj = right_proj * right_gate.sigmoid()
            out_gate = out_gate * out_gate.sigmoid()

            # Accumulate
            for d in range(dim):
                out[d] += input_b[d] * input_j[d]

        # Apply normalization
        norm_weights = tl.load(to_out_norm_ptr)
        norm_bias = tl.load(to_out_norm_ptr + hidden_dim)
        out = out * norm_weights + norm_bias

        # Apply final gate
        out = out * out_gate

        # Apply final projection
        to_out_weights = tl.load(to_out_ptr)
        out = out * to_out_weights

        # Store the result
        tl.store(output_ptr + offset_b + i * seq_len * dim + j * dim, out)

    # Launch the Triton kernel
    grid = (num_blocks, 1)  # We're using only one block per output element
    tritond_kernel[grid](
        input_tensor, mask, weights["left_proj.weight"], weights["right_proj.weight"],
        weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"],
        weights["to_out_norm.weight"], weights["to_out.weight"],
        input_tensor, batch_size, seq_len, dim, hidden_dim, BLOCK
    )

    return input_tensor
