import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Implements the TriMul operator in Triton for high-performance computation.
    This kernel performs the forward pass of TriMul, including projections, gates,
    and the core pairwise multiplication sum, optimized for the L40S GPU.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"].to(input_tensor.device).contiguous()
    norm_bias = weights["norm.bias"].to(input_tensor.device).contiguous()
    left_proj_weight = weights["left_proj.weight"].to(input_tensor.device).contiguous()
    right_proj_weight = weights["right_proj.weight"].to(input_tensor.device).contiguous()
    left_gate_weight = weights["left_gate.weight"].to(input_tensor.device).contiguous()
    right_gate_weight = weights["right_gate.weight"].to(input_tensor.device).contiguous()
    out_gate_weight = weights["out_gate.weight"].to(input_tensor.device).contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].to(input_tensor.device).contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].to(input_tensor.device).contiguous()
    to_out_weight = weights["to_out.weight"].to(input_tensor.device).contiguous()

    # Convert mask to float (0.0 or 1.0)
    mask = mask.unsqueeze(-1).to(input_tensor.device).float()
    mask = mask.contiguous()

    # Compute the norm on the input (BatchNorm)
    x = input_tensor
    x = torch.nn.functional.layer_norm(x, dim, norm_weight, norm_bias, True)

    # Compute projections
    # left_proj: seq_len x seq_len x hidden_dim
    # right_proj: seq_len x seq_len x hidden_dim
    left_proj_out = torch.matmul(input_tensor.view(-1, seq_len, seq_len, dim), left_proj_weight).view(batch_size, seq_len, seq_len, hidden_dim)
    right_proj_out = torch.matmul(input_tensor.view(-1, seq_len, seq_len, dim), right_proj_weight).view(batch_size, seq_len, seq_len, hidden_dim)

    # Apply mask to projections
    left_proj_out = left_proj_out * mask
    right_proj_out = right_proj_out * mask

    # Compute gates
    left_gate_out = torch.nn.functional.linear(input_tensor, left_gate_weight)
    left_gate_out = left_gate_out.sigmoid()
    right_gate_out = torch.nn.functional.linear(input_tensor, right_gate_weight)
    right_gate_out = right_gate_out.sigmoid()
    out_gate_out = torch.nn.functional.linear(input_tensor, out_gate_weight)
    out_gate_out = out_gate_out.sigmoid()

    # Apply gates
    left_proj_out = left_proj_out * left_gate_out
    right_proj_out = right_proj_out * right_gate_out

    # Now implement the core TriMul operation in Triton
    # We'll implement the sum over k of left[i][k] * right[j][k]

    # Output shape: [batch_size, seq_len, seq_len, hidden_dim]
    output = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=input_tensor.device)

    # Define Triton kernel
    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, output_ptr,
        seq_len, hidden_dim,
        BLOCK: tl.constexpr,
    ):
        # Compute the 2D index in the output for this thread
        # offset into the output tensor
        pid = tl.program_id(0)
        pid_m = pid // seq_len
        pid_n = pid % seq_len
        
        # Offset in the batch
        batch_offset = pid_m * seq_len * seq_len * hidden_dim
        # Offset in the i dimension
        i_offset = pid_n * seq_len * hidden_dim
        # Offset in the j dimension
        j_offset = 0
        # Offset in the k dimension
        k_offset = 0
        # Offset in the hidden dim
        hd_offset = 0

        # Initialize the output for this (i,j) pair
        acc = tl.zeros((BLOCK, hidden_dim), dtype=tl.float32)

        # Iterate over blocks of k
        for k_block in range(tl.cdiv(seq_len, BLOCK)):
            # Load the left and right slices
            left_block = tl.load(left_ptr + batch_offset + i_offset + k_block * BLOCK * hidden_dim, mask=tl.arange(0, BLOCK) < seq_len, other=0.0)
            right_block = tl.load(right_ptr + batch_offset + j_offset + k_block * BLOCK * hidden_dim, mask=tl.arange(0, BLOCK) < seq_len, other=0.0)

            # Dot product of the current block and accumulate
            # Each element of left_block is [BLOCK, hidden_dim]
            # So we need to reshape to [BLOCK, hidden_dim]
            # Multiply left_block and right_block
            # Then accumulate into acc
            # This is a 1D dot product
            acc += tl.dot(left_block, right_block)

        # Write the final result
        tl.store(output_ptr + batch_offset + i_offset + j_offset, acc)

    # Launch the kernel
    # Each batch and sequence pair
    grid = (batch_size * seq_len * seq_len,)
    BLOCK = 128  # Tune this based on your test cases
    triton_trimul_kernel[grid](
        left_proj_out.view(-1, seq_len * seq_len, hidden_dim).contiguous().data_ptr(),
        right_proj_out.view(-1, seq_len * seq_len, hidden_dim).contiguous().data_ptr(),
        output.data_ptr(),
        seq_len, hidden_dim,
        BLOCK=BLOCK
    )

    # Normalize the output tensor
    to_out_norm_weight = to_out_norm_weight.to(output.device)
    to_out_norm_bias = to_out_norm_bias.to(output.device)
    output = torch.nn.functional.layer_norm(output, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, True)

    # Apply the gate
    output = output * out_gate_out

    # Final projection
    to_out_weight = to_out_weight.to(output.device)
    output = torch.nn.functional.linear(output, to_out_weight)

    # Convert to float32 for final output
    output = output.to(torch.float32)

    return output
