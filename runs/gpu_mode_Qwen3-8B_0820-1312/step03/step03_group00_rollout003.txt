import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any
import math

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Optimized Triton implementation of the outgoing version of the TriMul operator.

    Args:
        input_tensor: Input tensor of shape [bs, seq_len, seq_len, dim]
        mask: Mask tensor of shape [bs, seq_len, seq_len]
        weights: Dictionary containing model weights
        config: Dictionary containing model configuration parameters

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define kernel parameters
    BLOCK = 128
    num_blocks = triton.cdiv(seq_len * seq_len, BLOCK)

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Allocate output tensor
    output = torch.empty_like(input_tensor)

    # Compute norms
    def compute_norm(x):
        # We're not computing gradients, so we can skip this
        # But for correctness, we need to do the same as the PyTorch code
        norm = torch.nn.functional.layer_norm(x, x.shape, norm_weight, norm_bias, False)
        return norm

    # Compute projection and gate matrices
    # Left and right projections
    # Note: This is simplified as we're not doing gradients
    left = torch.matmul(input_tensor, left_proj_weight.t())
    right = torch.matmul(input_tensor, right_proj_weight.t())

    # Mask processing
    mask = mask.unsqueeze(-1)
    left = left * mask
    right = right * mask

    # Gates
    left_gate = torch.matmul(input_tensor, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(input_tensor, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(input_tensor, out_gate_weight.t()).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Implement the key part using Triton
    @triton.jit
    def tri_mul_kernel(out_ptr, left_ptr, right_ptr, mask_ptr, seq_len, dim, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        i = pid * BLOCK + tl.arange(0, BLOCK)
        i = tl.where(i < seq_len, i, tl.tensor(0, dtype=tl.int32))
        j = tl.arange(0, seq_len)
        k = tl.arange(0, seq_len)

        # Load masks
        mask = tl.load(mask_ptr + i * seq_len + j, dtype=tl.float32)

        # Load left and right values
        left = tl.load(left_ptr + i * seq_len * dim + k * dim, dtype=tl.float32)
        right = tl.load(right_ptr + j * seq_len * dim + k * dim, dtype=tl.float32)

        # Apply mask
        left = tl.where(mask == 0, 0.0, left)
        right = tl.where(mask == 0, 0.0, right)

        # Compute the output
        out = tl.dot(left, right)
        tl.store(out_ptr + i * seq_len + j, out)

    # Launch TriMul kernel
    grid = (num_blocks,)
    tri_mul_kernel[grid](output, left, right, mask, seq_len, dim, BLOCK)

    # Final steps
    out = torch.matmul(output, to_out_weight.t())
    out = torch.nn.functional.layer_norm(out, out.shape, to_out_norm_weight, to_out_norm_bias, False)

    # Apply out_gate
    output = out * out_gate

    return output
