import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Triton implementation of the TriMul operation for the TriMul module.
    This kernel implements the "outgoing" version of the TriMul operation
    with the forward pass, using optimized memory access and computation.
    """

    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Define kernel parameters
    BLOCK_SIZE = 64  # Block size for memory access and computation
    NUM_BLOCKS = triton.cdiv(seq_len, BLOCK_SIZE)
    NUM_BLOCK_SEQ = triton.cdiv(seq_len, BLOCK_SIZE)

    # Constants for tensor layout and strides
    STRIDE_B = seq_len * seq_len * dim
    STRIDE_T = seq_len * seq_len
    STRIDE_K = seq_len * dim
    STRIDE_D = dim

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Reshape input and mask
    input_reshaped = input_tensor.view(bs, seq_len, seq_len, dim)
    mask_reshaped = mask.unsqueeze(-1)  # shape: [bs, seq_len, seq_len, 1]

    # Allocate output tensor
    output = torch.zeros((bs, seq_len, seq_len, dim), dtype=torch.float32, device=input_tensor.device)

    # Define the kernel
    @triton.jit
    def tri_mul_kernel(
        input_ptr, mask_ptr, left_proj_weight_ptr, right_proj_weight_ptr,
        left_gate_weight_ptr, right_gate_weight_ptr, out_gate_weight_ptr,
        to_out_norm_weight_ptr, to_out_norm_bias_ptr, to_out_weight_ptr,
        output_ptr,
        batch_size, seq_len, dim, hidden_dim, BLOCK: tl.constexpr,
        STRIDE_B, STRIDE_T, STRIDE_K, STRIDE_D, STRIDE_T2: tl.constexpr
    ):
        pid = tl.program_id(0)
        block_idx = pid % NUM_BLOCK_SEQ
        block_seq = pid // NUM_BLOCK_SEQ

        # Compute the range for this block
        block_start_i = tl.arange(0, BLOCK) + block_idx * BLOCK
        block_start_j = tl.arange(0, BLOCK) + block_seq * BLOCK

        # Compute the offsets for this block
        i_offsets = block_start_i + tl.arange(0, BLOCK)
        j_offsets = block_start_j + tl.arange(0, BLOCK)

        # Load weights
        left_proj_weight = tl.load(left_proj_weight_ptr + tl.arange(0, hidden_dim) * dim, aligned=True)
        right_proj_weight = tl.load(right_proj_weight_ptr + tl.arange(0, hidden_dim) * dim, aligned=True)
        left_gate_weight = tl.load(left_gate_weight_ptr + tl.arange(0, hidden_dim) * dim, aligned=True)
        right_gate_weight = tl.load(right_gate_weight_ptr + tl.arange(0, hidden_dim) * dim, aligned=True)
        out_gate_weight = tl.load(out_gate_weight_ptr + tl.arange(0, hidden_dim) * dim, aligned=True)
        to_out_norm_weight = tl.load(to_out_norm_weight_ptr + tl.arange(0, hidden_dim) * hidden_dim, aligned=True)
        to_out_norm_bias = tl.load(to_out_norm_bias_ptr + tl.arange(0, hidden_dim), aligned=True)
        to_out_weight = tl.load(to_out_weight_ptr + tl.arange(0, dim) * hidden_dim, aligned=True)

        # Compute norm for input
        norm_weight = tl.load(norm_weight + tl.arange(0, dim), aligned=True)
        norm_bias = tl.load(norm_bias + tl.arange(0, dim), aligned=True)
        input_masked = input_reshaped.view(bs, seq_len, seq_len, dim)
        input_masked = input_masked * mask_reshaped
        input_norm = (input_masked - norm_bias) * norm_weight

        # Project input
        left = tl.dot(input_norm, left_proj_weight)
        right = tl.dot(input_norm, right_proj_weight)

        # Apply gates
        left_gate = tl.dot(input_norm, left_gate_weight).sigmoid()
        right_gate = tl.dot(input_norm, right_gate_weight).sigmoid()
        out_gate = tl.dot(input_norm, out_gate_weight).sigmoid()

        # Apply gates to projections
        left = left * left_gate
        right = right * right_gate

        # Compute the TriMul operation
        for k in range(seq_len):
            # Load left and right
            left_k = tl.load(input_ptr + (block_seq * seq_len + k) * STRIDE_K + i_offsets * dim, mask=(block_start_i < seq_len) & (block_start_j < seq_len), other=0.0)
            right_k = tl.load(input_ptr + (block_seq * seq_len + k) * STRIDE_K + j_offsets * dim, mask=(block_start_i < seq_len) & (block_start_j < seq_len), other=0.0)

            # Compute the product and accumulate
            for i in range(BLOCK):
                for j in range(BLOCK):
                    if (block_start_i[i] < seq_len) and (block_start_j[j] < seq_len):
                        out = left_k[i] * right_k[j]
                        tl.atomic_add(output_ptr + block_seq * seq_len * seq_len + block_start_i[i] * seq_len + block_start_j[j], out)

        # Apply the output gate and norm
        output = tl.load(output_ptr, aligned=True)
        output = (output - to_out_norm_bias) * to_out_norm_weight
        output = output * out_gate
        output = tl.dot(output, to_out_weight)
        tl.store(output_ptr, output, aligned=True)

    # Launch the kernel
    grid = (triton.cdiv(seq_len * seq_len, BLOCK_SIZE),)
    tri_mul_kernel[grid](
        input_tensor.data_ptr(),
        mask.data_ptr(),
        left_proj_weight.data_ptr(),
        right_proj_weight.data_ptr(),
        left_gate_weight.data_ptr(),
        right_gate_weight.data_ptr(),
        out_gate_weight.data_ptr(),
        to_out_norm_weight.data_ptr(),
        to_out_norm_bias.data_ptr(),
        to_out_weight.data_ptr(),
        output.data_ptr(),
        bs, seq_len, dim, hidden_dim, BLOCK_SIZE,
        STRIDE_B, STRIDE_T, STRIDE_K, STRIDE_D, seq_len * seq_len
    )

    return output
