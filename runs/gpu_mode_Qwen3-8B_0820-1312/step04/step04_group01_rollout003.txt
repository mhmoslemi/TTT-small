import torch
import triton
import triton.language as tl
import math
from typing import Tuple, Dict

def custom_kernel(data: tuple) -> torch.Tensor:
    """
    This kernel implements the outgoing version of the Triangle Multiplicative Update (TriMul)
    operator for protein structure prediction models in BioML. It uses Triton for efficient
    computation on large sequence dimensions (N), while keeping the forward pass optimized
    by using a combination of kernel fusions and proper memory access patterns.
    
    The implementation includes:
    - Weight projections and gate activations
    - Mask application
    - Efficient pairwise operations with einsum-like behavior
    - Normalization and final output projection
    - Output in float32

    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access the given weights of the model
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Cast all tensors to float32
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.to(torch.float32)

    # --- Step 1: Apply LayerNorm
    # Compute LayerNorm manually for efficiency with Triton
    mean = torch.mean(input_tensor, dim=[1, 2], keepdim=True)
    var = torch.var(input_tensor, dim=[1, 2], keepdim=True, unbiased=False)
    norm_input = (input_tensor - mean) * torch.rsqrt(var + 1e-6)
    norm_input = norm_input * norm_weight + norm_bias

    # --- Step 2: Apply projections and gates
    # Projection
    left_proj = torch.matmul(norm_input, left_proj_weight.t())  # [B, N, N, H]
    right_proj = torch.matmul(norm_input, right_proj_weight.t()) # [B, N, N, H]

    left_gate = torch.matmul(norm_input, left_gate_weight.t())   # [B, N, N, H]
    right_gate = torch.matmul(norm_input, right_gate_weight.t()) # [B, N, N, H]
    out_gate = torch.matmul(norm_input, out_gate_weight.t())     # [B, N, N, H]

    # Apply sigmoid on gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Apply mask
    mask = mask.unsqueeze(-1)  # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # --- Step 3: Compute the TriMul operation
    # We implement the "einsum" as a kernel
    out = compute_trimul_kernel(
        left, right, batch_size, seq_len, hidden_dim,
        num_warps=128,
        block_size=128  # Match the L40S GPU's efficiency
    )

    # --- Step 4: Apply LayerNorm to the output
    mean_out = torch.mean(out, dim=[1, 2], keepdim=True)
    var_out = torch.var(out, dim=[1, 2], keepdim=True, unbiased=False)
    out = (out - mean_out) * torch.rsqrt(var_out + 1e-6)
    out = out * to_out_norm_weight + to_out_norm_bias

    # Apply the output gate
    out = out * out_gate

    # Final projection
    out = torch.matmul(out, to_out_weight.t())

    return out

@triton.jit
def compute_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_size, seq_len, hidden_dim,
    BLOCK: tl.constexpr,
    num_warps: tl.constexpr,
    block_size: tl.constexpr
):
    """
    Efficient Triton kernel for the TriMul operation using a block-based approach.
    The kernel mimics the einsum operation with optimized memory access and block tiling.
    """
    pid = tl.program_id(0)
    batch_idx = pid // (seq_len * seq_len)
    i = (pid // seq_len) % seq_len
    j = pid % seq_len

    # Compute the output index
    offset = batch_idx * seq_len * seq_len * hidden_dim + i * seq_len * hidden_dim + j * hidden_dim
    out_offset = offset

    # Load the left and right data
    left = tl.zeros([BLOCK, hidden_dim], dtype=tl.float32)
    right = tl.zeros([BLOCK, hidden_dim], dtype=tl.float32)

    # Compute the result in blocks
    for k in range(0, seq_len, block_size):
        k_idx = k
        # Get block indices for k
        block_k_start = k
        block_k_end = min(k + block_size, seq_len)

        # Loop over each row in left and column in right
        for i_block in range(seq_len):
            i_start = i_block
            i_end = min(i_block + block_size, seq_len)

            # Compute the current i and j for this block
            for j_block in range(seq_len):
                j_start = j_block
                j_end = min(j_block + block_size, seq_len)

                # Load the left and right block data
                left_block = tl.load(left_ptr + i_start * seq_len * hidden_dim + k_idx * hidden_dim, 
                                    shape=(block_size, hidden_dim), 
                                    strides=(seq_len * hidden_dim, hidden_dim))
                right_block = tl.load(right_ptr + j_start * seq_len * hidden_dim + k_idx * hidden_dim, 
                                    shape=(block_size, hidden_dim), 
                                    strides=(seq_len * hidden_dim, hidden_dim))

                # Perform the matrix multiplication
                result_block = tl.dot(left_block, right_block)

                # Store the result into the output tensor
                tl.store(out_ptr + i_start * seq_len * hidden_dim + j_start * hidden_dim, 
                         result_block, 
                         shape=(block_size, hidden_dim),
                         strides=(seq_len * hidden_dim, hidden_dim))

                i_start += block_size
                j_start += block_size

    return

def compute_trimul_kernel_launcher(left, right, out, batch_size, seq_len, hidden_dim):
    """
    Launches the compute_trimul_kernel for the TriMul operation.
    """
    # Triton launch configurations
    block_size = 128
    num_warps = 128

    # Grid size
    grid = (batch_size * seq_len * seq_len,)

    # Launch the kernel
    compute_trimul_kernel[grid](
        left,
        right,
        out,
        batch_size,
        seq_len,
        hidden_dim,
        block_size,
        num_warps
    )
