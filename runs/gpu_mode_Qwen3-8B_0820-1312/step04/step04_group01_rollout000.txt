import torch
import triton
import triton.language as tl
from torch import einsum
import math

def custom_kernel(data):
    """
    Triton implementation of the TriMul module for AlphaFold3-like models.
    This kernel efficiently computes the triangular multiplicative update operation.
    
    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters
    
    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    seqlen = input_tensor.shape[1]
    batch_size = input_tensor.shape[0]

    # Convert mask to device
    mask = mask.unsqueeze(-1).to(input_tensor.device)

    # Access weights
    norm_weight = weights["norm.weight"].to(input_tensor.device)
    norm_bias = weights["norm.bias"].to(input_tensor.device)
    left_proj_weight = weights["left_proj.weight"].to(input_tensor.device)
    right_proj_weight = weights["right_proj.weight"].to(input_tensor.device)
    left_gate_weight = weights["left_gate.weight"].to(input_tensor.device)
    right_gate_weight = weights["right_gate.weight"].to(input_tensor.device)
    out_gate_weight = weights["out_gate.weight"].to(input_tensor.device)
    to_out_norm_weight = weights["to_out_norm.weight"].to(input_tensor.device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(input_tensor.device)
    to_out_weight = weights["to_out.weight"].to(input_tensor.device)

    # Norm operation (inference only)
    x = input_tensor
    x = x.to(torch.float32)
    x = x * norm_weight + norm_bias
    x = x.to(torch.bfloat16)

    # Project left and right
    left = x @ left_proj_weight
    right = x @ right_proj_weight

    # Apply mask
    left = left * mask
    right = right * mask

    # Apply gates
    left_gate = x @ left_gate_weight
    right_gate = x @ right_gate_weight
    out_gate = x @ out_gate_weight
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Compute TriMul operation
    out = triton_tri_mul(left, right, seqlen, batch_size, hidden_dim, mask)

    # Apply to_out_norm and out_gate
    out = out.to(torch.float32)
    out = out @ to_out_norm_weight + to_out_norm_bias
    out = out * out_gate
    out = out @ to_out_weight

    return out

@triton.jit
def triton_tri_mul(left, right, seqlen, batch_size, hidden_dim, mask):
    # Triton kernel for TriMul
    # left: [bs, seq_len, seq_len, hidden_dim]
    # right: [bs, seq_len, seq_len, hidden_dim]
    # out: [bs, seq_len, seq_len, hidden_dim]

    # Constants
    BLOCK = 128
    M = seqlen
    N = seqlen
    K = seqlen

    # Determine grid dimensions
    grid = (triton.cdiv(M, BLOCK), triton.cdiv(N, BLOCK), batch_size)

    # Launch kernel
    triton_tri_mul_kernel[grid](left, right, mask, out, M, N, K,
                                seqlen, batch_size, hidden_dim,
                                BLOCK=BLOCK)

    return out

@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, mask_ptr, out_ptr,
    M, N, K,
    seqlen, batch_size, hidden_dim,
    BLOCK: tl.constexpr
):
    # Each thread handles one output element
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    pid_m = tl.program_id(2)

    # Compute offset to current output element
    off_m = pid_m * BLOCK + tl.arange(0, BLOCK)
    off_n = pid_n * BLOCK + tl.arange(0, BLOCK)
    off_m = off_m < M
    off_n = off_n < N

    # Compute offset for left and right
    left = tl.load(left_ptr + pid_m * seqlen * seqlen * hidden_dim + off_m[:, None] * seqlen * hidden_dim + off_n[None, :] * hidden_dim, mask=off_m[:, None] & off_n[None, :], other=0.0)
    right = tl.load(right_ptr + pid_m * seqlen * seqlen * hidden_dim + off_m[:, None] * seqlen * hidden_dim + off_n[None, :] * hidden_dim, mask=off_m[:, None] & off_n[None, :], other=0.0)

    # Compute product
    product = tl.dot(left, right, axis=1)

    # Apply mask
    mask = tl.load(mask_ptr + pid_m * seqlen * seqlen + off_m[:, None] * seqlen + off_n[None, :], mask=off_m[:, None] & off_n[None, :], other=0.0)
    product = product * mask

    # Store result
    tl.store(out_ptr + pid_m * seqlen * seqlen * hidden_dim + off_m[:, None] * seqlen * hidden_dim + off_n[None, :] * hidden_dim, product, mask=off_m[:, None] & off_n[None, :])
