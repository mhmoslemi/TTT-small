import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Triton implementation of the "outgoing" version of the TriMul operation used in protein
    structure prediction models like AlphaFold3. The kernel computes pairwise interactions
    in a 4D tensor [B, N, N, C] using projections, gates, and matrix multiplication.

    This kernel is optimized by avoiding redundant computations and using memory-efficient
    strategies to handle large sequence dimensions (N up to 1024) on NVIDIA L40S hardware.
    """

    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # Convert input and mask to float32
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.unsqueeze(-1)  # [bs, seq_len, seq_len, 1]

    # Access weights from provided dictionary
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Define Triton kernel parameters
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_M = 128

    # Kernel to compute left, right, gate projections and perform the TriMul
    @triton.jit
    def tri_mul_kernel(
        input_ptr, left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr,
        out_gate_ptr, mask_ptr, norm_weight_ptr, norm_bias_ptr, to_out_norm_weight_ptr, to_out_norm_bias_ptr,
        to_out_weight_ptr, out_ptr,
        batch_size, seq_len, dim, hidden_dim, num_seqlen, num_batch, num_dim, num_hidden_dim,
        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_M: tl.constexpr
    ):
        # Get current program ID (representing a block in the 2D grid)
        pid = tl.program_id(0)
        # Get the current thread ID within a block
        pid_m = tl.program_id(1)
        # Get the current batch index
        batch = pid // (seq_len * seq_len)
        # Get the row index
        i = (pid // seq_len) % seq_len
        # Get the column index
        j = pid % seq_len

        # Determine the offset for the current block
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)

        # Mask tensor: [batch, seqlen, seqlen, 1]
        mask = tl.load(mask_ptr + batch * seq_len * seq_len + i * seq_len + j, dtype=tl.float32)
        if mask == 0:
            return

        # Load input for this batch
        x = tl.load(input_ptr + batch * seq_len * seq_len * dim + i * seq_len * dim + j * dim,
                    shape=(dim, ), dtype=tl.float32)

        # Apply LayerNorm
        norm_weight = tl.load(norm_weight_ptr)
        norm_bias = tl.load(norm_bias_ptr)
        x_norm = (x - norm_bias) * norm_weight

        # Compute left and right projections
        left = tl.dot(x_norm, left_proj_weight)  # [dim, hidden_dim] -> [hidden_dim]
        right = tl.dot(x_norm, right_proj_weight)  # [dim, hidden_dim] -> [hidden_dim]

        # Compute gate activations
        left_gate = tl.dot(x_norm, left_gate_weight).sigmoid()
        right_gate = tl.dot(x_norm, right_gate_weight).sigmoid()
        out_gate = tl.dot(x_norm, out_gate_weight).sigmoid()

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # Compute TriMul: C[i,j] = sum_k (A[i,k] * B[j,k])
        # We'll compute this in blocks to avoid O(N^3) memory usage
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, seq_len, BLOCK_K):
            # Load left and right
            left_block = tl.load(left_proj_ptr + batch * seq_len * hidden_dim + i * hidden_dim + k * hidden_dim,
                                 shape=(hidden_dim, ), dtype=tl.float32)
            right_block = tl.load(right_proj_ptr + batch * seq_len * hidden_dim + j * hidden_dim + k * hidden_dim,
                                 shape=(hidden_dim, ), dtype=tl.float32)

            # Compute the contribution to acc
            acc += tl.dot(left_block, right_block, transpose_b=True)

        # Apply output gate
        acc = acc * out_gate

        # Normalize output
        out_norm_weight = tl.load(to_out_norm_weight_ptr)
        out_norm_bias = tl.load(to_out_norm_bias_ptr)
        acc = (acc - out_norm_bias) * out_norm_weight

        # Apply to_out projection
        out_weight = tl.load(to_out_weight_ptr)
        out = tl.dot(acc, out_weight)

        # Write result
        tl.store(out_ptr + batch * seq_len * seq_len * dim + i * seq_len * dim + j * dim,
                 out, shape=(dim, ))

    # Launch the kernel
    # Total elements: batch_size * seq_len * seq_len * dim
    total_elements = batch_size * seq_len * seq_len * dim
    grid = (triton.cdiv(total_elements, BLOCK_N * BLOCK_M), triton.cdiv(seq_len, BLOCK_M), 1)
    tri_mul_kernel[grid](
        input_tensor,
        left_proj_weight,
        right_proj_weight,
        left_gate_weight,
        right_gate_weight,
        out_gate_weight,
        mask,
        norm_weight,
        norm_bias,
        to_out_norm_weight,
        to_out_norm_bias,
        to_out_weight,
        torch.zeros(batch_size, seq_len, seq_len, dim, device=input_tensor.device),
        batch_size, seq_len, dim, hidden_dim,
        seq_len, batch_size, dim, hidden_dim,
        BLOCK_N, BLOCK_K, BLOCK_M
    )

    return torch.zeros(batch_size, seq_len, seq_len, dim, device=input_tensor.device)
