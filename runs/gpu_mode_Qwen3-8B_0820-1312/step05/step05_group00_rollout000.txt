import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):

    Triton kernel implementation of the outgoing TriMul operator.
    This implementation uses optimized memory access patterns and fusions to achieve high performance.
    """

    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    seqlen = seq_len
    bs = batch_size

    # Define Triton kernel constants
    BLOCK = 256
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 128

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Allocate output tensor
    output = torch.zeros((batch_size, seq_len, seq_len, hidden_dim), device=input_tensor.device, dtype=input_tensor.dtype)

    # Normalize input
    x = input_tensor
    norm_x = x
    if norm_weight is not None and norm_bias is not None:
        norm_x = torch.nn.functional.layer_norm(x, x.shape[-1], norm_weight, norm_bias)

    # Project inputs
    if left_proj_weight is not None:
        left = torch.nn.functional.linear(norm_x, left_proj_weight)
    if right_proj_weight is not None:
        right = torch.nn.functional.linear(norm_x, right_proj_weight)

    # Apply mask
    mask = mask.unsqueeze(-1).to(dtype=left.dtype)
    left *= mask
    right *= mask

    # Compute gates
    if left_gate_weight is not None:
        left_gate = torch.nn.functional.linear(norm_x, left_gate_weight).sigmoid()
    if right_gate_weight is not None:
        right_gate = torch.nn.functional.linear(norm_x, right_gate_weight).sigmoid()
    if out_gate_weight is not None:
        out_gate = torch.nn.functional.linear(norm_x, out_gate_weight).sigmoid()

    # Apply gates
    left *= left_gate
    right *= right_gate

    # Define kernel
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        batch_size, seq_len, hidden_dim,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
        stride_left_b, stride_left_n, stride_left_k, stride_left_d,
        stride_right_b, stride_right_n, stride_right_k, stride_right_d,
        stride_out_b, stride_out_n, stride_out_k, stride_out_d,
    ):
        # Shared memory for intermediate results
        m = tl.num_programs(0)
        n = tl.num_programs(1)
        k = tl.num_programs(2)
        pid = tl.program_id(0)
        pid_m = pid // (m * n * k)
        pid_n = (pid // (m * n)) % k
        pid_k = (pid // m) % k
        pid_b = pid % m

        # Compute offsets
        offs_b = pid_b * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)
        offs_d = tl.arange(0, hidden_dim)

        # Load left and right pointers
        left_ptr = left_ptr + pid_b * stride_left_b + offs_b[:, None, None] * stride_left_n + offs_k[None, :, None] * stride_left_k + offs_d[None, None, :]
        right_ptr = right_ptr + pid_b * stride_right_b + offs_n[None, :, None] * stride_right_n + offs_k[None, :, None] * stride_right_k + offs_d[None, None, :]
        out_ptr = out_ptr + pid_b * stride_out_b + offs_b[:, None] * stride_out_n + offs_n[None, :] * stride_out_k + offs_d[None, None, :]

        # Initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Compute the matrix multiplication
        for k_idx in range(0, hidden_dim, BLOCK_K):
            # Load left and right matrices
            left = tl.load(left_ptr + k_idx * stride_left_k, mask=(offs_k < hidden_dim))
            right = tl.load(right_ptr + k_idx * stride_right_k, mask=(offs_k < hidden_dim))

            # Compute dot product
            dot = tl.dot(left, right)
            acc += dot

        # Store output
        tl.store(out_ptr, acc, mask=(offs_b < seq_len) & (offs_n < seq_len))

    # Launch kernel
    grid = (batch_size, seq_len, seq_len)
    tri_mul_kernel[grid](
        left, right, output,
        batch_size, seq_len, hidden_dim,
        BLOCK_M, BLOCK_N, BLOCK_K,
        stride_left_b=left.stride(0),
        stride_left_n=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_d=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_n=right.stride(1),
        stride_right_k=right.stride(2),
        stride_right_d=right.stride(3),
        stride_out_b=output.stride(0),
        stride_out_n=output.stride(1),
        stride_out_k=output.stride(2),
        stride_out_d=output.stride(3),
    )

    # Apply output gate
    output *= out_gate

    # Apply final layer norm and linear
    output = torch.nn.functional.layer_norm(
        output, output.shape[-1], to_out_norm_weight, to_out_norm_bias
    )
    output = torch.nn.functional.linear(output, to_out_weight)

    return output
