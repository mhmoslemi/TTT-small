import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Implements the outgoing version of the TriMul operator using Triton for optimized performance.
    This kernel computes the pairwise interactions by performing matrix multiplications in a
    fused and optimized way to minimize memory bandwidth and maximize throughput.
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access the weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Device and data type
    device = input_tensor.device
    dt = input_tensor.dtype  # Use device-native type

    # Apply LayerNorm
    x = input_tensor
    x = x * norm_weight + norm_bias  # LayerNorm without mean/var (assuming precomputed)

    # Compute projected tensors
    left = torch.matmul(x, left_proj_weight.t())
    right = torch.matmul(x, right_proj_weight.t())

    # Apply mask
    mask = mask.unsqueeze(-1).to(x.dtype)
    left = left * mask
    right = right * mask

    # Compute gates
    left_gate = torch.matmul(x, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(x, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(x, out_gate_weight.t()).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for kernel
    out = torch.zeros((bs, seq_len, seq_len, hidden_dim), device=device, dtype=x.dtype)
    # We'll launch our custom kernel to compute out[i, j] = sum_k left[i, k] * right[j, k]

    # Triton kernel implementation
    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        seq_len, hidden_dim,
        BLOCK: tl.constexpr
    ):
        # Identify the current thread and its index in the block
        pid = tl.program_id(0)
        block_idx = pid
        block_offset = block_idx * BLOCK
        block_start = block_offset
        block_end = block_start + BLOCK
        mask = block_end < seq_len

        # Get the block row and column indices
        block_row = block_start
        block_col = block_start

        # Load data
        left_offset = block_row * seq_len + tl.arange(0, BLOCK)
        right_offset = block_col * seq_len + tl.arange(0, BLOCK)
        out_offset = block_row * seq_len + block_col * seq_len + tl.arange(0, BLOCK)

        # Load left and right data
        left = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
        right = tl.load(right_ptr + right_offset, mask=mask, other=0.0)

        # Compute the output
        acc = tl.zeros((BLOCK,), dtype=tl.float32)
        for k in range(seq_len):
            a = left[k]
            b = right[k]
            acc += tl.sum(a * b)
        acc = acc * (1.0 / seq_len)

        # Store output
        tl.store(out_ptr + out_offset, acc, mask=mask)

    # Launch the kernel
    num_blocks = triton.cdiv(seq_len, BLOCK)
    grid = (num_blocks,)
    triton_trimul_kernel[grid](
        left, right, out,
        seq_len, hidden_dim,
        BLOCK=128
    )

    # Apply LayerNorm and gate
    out = torch.matmul(out, to_out_norm_weight.t())
    out = out + to_out_norm_bias
    out = out * out_gate

    # Final output
    output = torch.matmul(out, to_out_weight.t())
    output = output.to(torch.float32)  # Ensure final output is in float32

    return output
