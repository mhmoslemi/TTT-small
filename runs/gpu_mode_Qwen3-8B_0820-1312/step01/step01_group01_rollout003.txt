import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    A Triton implementation of the "outgoing" version of the TriMul operation used in AlphaFold3 and related models.
    The kernel computes pairwise interactions over a 4D input tensor using projections, gates, and einsum-style operations.
    The output is computed using matrix multiplication with tiling to optimize for large sequence lengths.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Extracting weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Initialize output tensor
    output = torch.zeros((batch_size, seq_len, seq_len, hidden_dim), device=input_tensor.device)

    # Define the kernel
    @triton.jit
    def tritongemm_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_weight_ptr, to_out_weight_ptr,
        output_ptr, 
        batch_size: tl.constexpr, seq_len: tl.constexpr, dim: tl.constexpr, hidden_dim: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        # Compute the batch index
        pid = tl.program_id(0)
        # Compute the row index in the output
        i = pid * BLOCK + tl.arange(0, BLOCK)
        # Compute the column index in the output
        j = tl.arange(0, seq_len)
        # Compute the sequence dimension (k)
        k = tl.arange(0, seq_len)

        # Compute the mask
        mask = tl.load(mask_ptr + (i[:, None] * seq_len + j[None, :]), mask=(i < seq_len) & (j < seq_len), other=0.0)
        mask = mask.to(tl.float32)

        # Compute the left and right vectors
        left = tl.load(left_proj_ptr + (i[:, None] * seq_len * dim + k[None, :] * dim), mask=(i < seq_len) & (k < seq_len), other=0.0)
        right = tl.load(right_proj_ptr + (j[None, :] * seq_len * dim + k[None, :] * dim), mask=(j < seq_len) & (k < seq_len), other=0.0)

        # Apply the gates
        left_gate = tl.load(left_gate_ptr + (i[:, None] * seq_len * dim + k[None, :] * dim), mask=(i < seq_len) & (k < seq_len), other=0.0)
        left_gate = tl.sigmoid(left_gate)
        right_gate = tl.load(right_gate_ptr + (j[None, :] * seq_len * dim + k[None, :] * dim), mask=(j < seq_len) & (k < seq_len), other=0.0)
        right_gate = tl.sigmoid(right_gate)
        out_gate = tl.load(out_gate_ptr + (i[:, None] * seq_len * dim + j[None, :] * dim), mask=(i < seq_len) & (j < seq_len), other=0.0)
        out_gate = tl.sigmoid(out_gate)

        # Apply the gates to left and right
        left = left * left_gate
        right = right * right_gate

        # Compute the dot product
        acc = tl.zeros((BLOCK, seq_len), dtype=tl.float32)
        for k_idx in range(0, seq_len, BLOCK):
            # Load the left and right vectors
            left_k = tl.load(left_proj_ptr + (i[:, None] * seq_len * dim + (k_idx + k)[None, :] * dim), mask=(i < seq_len) & ((k_idx + k) < seq_len), other=0.0)
            right_k = tl.load(right_proj_ptr + (j[None, :] * seq_len * dim + (k_idx + k)[None, :] * dim), mask=(j < seq_len) & ((k_idx + k) < seq_len), other=0.0)

            # Apply the mask for this k index
            mask_k = tl.load(mask_ptr + (i[:, None] * seq_len + j[None, :]), mask=(i < seq_len) & (j < seq_len), other=0.0)
            mask_k = mask_k.to(tl.float32)

            # Compute the product
            product = left_k * right_k * mask_k
            # Accumulate the product
            acc += tl.dot(left_k, right_k) * mask_k

        # Apply the output gate
        acc = acc * out_gate

        # Apply LayerNorm
        if batch_size > 1:
            norm_weight = tl.load(to_out_norm_weight_ptr + (i[:, None] * hidden_dim + j[None, :]), mask=(i < seq_len) & (j < seq_len), other=0.0)
            norm_bias = tl.load(to_out_norm_bias_ptr + (i[:, None] * hidden_dim + j[None, :]), mask=(i < seq_len) & (j < seq_len), other=0.0)
            acc = (acc - tl.mean(acc)) * norm_weight + norm_bias

        # Apply the final projection
        to_out_weight = tl.load(to_out_weight_ptr + (i[:, None] * hidden_dim + j[None, :]), mask=(i < seq_len) & (j < seq_len), other=0.0)
        output = acc * to_out_weight

        # Store the result
        output_ptr = output_ptr + (i[:, None] * seq_len + j[None, :])
        tl.store(output_ptr, output)

    # Kernel parameters
    BLOCK = 128

    # Launch the kernel
    grid = (triton.cdiv(seq_len, BLOCK),)
    tritongemm_kernel[grid](
        input_tensor, mask, left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight,
        to_out_norm_weight, to_out_weight,
        output,
        batch_size, seq_len, dim, hidden_dim, BLOCK
    )

    return output
