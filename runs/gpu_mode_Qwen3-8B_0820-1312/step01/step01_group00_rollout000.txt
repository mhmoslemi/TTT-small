import torch
import triton
import triton.language as tl
from typing import Dict, Tuple

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Implements the TriMul module as an optimized Triton kernel for protein structure prediction models.
    This kernel optimizes the computationally heavy pairwise operations that dominate the forward pass.
    """
    input_tensor, mask, weights, config = data
    B, N, N, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dtype = input_tensor.dtype

    # Extract the weights with proper data types aligned to the input
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    left_proj_weight = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)
    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)
    to_out_weight = weights["to_out.weight"].to(dtype)

    # Step 1: LayerNorm (input tensor)
    input_norm = (input_tensor * norm_weight + norm_bias).to(torch.float32)

    # Step 2: Projection and apply masks
    left = torch.matmul(input_norm, left_proj_weight.t())
    right = torch.matmul(input_norm, right_proj_weight.t())

    # Mask
    mask = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask
    right = right * mask

    # Step 3: Gate operations
    left_gate = torch.matmul(input_norm, left_gate_weight.t()).sigmoid()
    right_gate = torch.matmul(input_norm, right_gate_weight.t()).sigmoid()
    out_gate = torch.matmul(input_norm, out_gate_weight.t()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Step 4: TriMul kernel - this is the main computational kernel we implement
    # We implement this with a Triton kernel to optimize for performance

    # Define kernel to perform the TriMul matrix multiplication in 4D space
    @triton.jit
    def tritonal_tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, D,
        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
    ):
        # The kernel is optimized to process each output (i, j, d) by summing across (k)
        pid = tl.program_id(0)
        block_n = pid // (N // BLOCK_N)  # Batch and sequence (i)
        block_k = pid % (N // BLOCK_K)   # Sequence (k)
        i_offsets = tl.arange(0, BLOCK_N)
        k_offsets = tl.arange(0, BLOCK_K)
        d_offsets = tl.arange(0, BLOCK_D)

        # Convert batch and sequence indices to flat indices
        batch = block_n // N
        seq_i = block_n % N
        seq_k = k_offsets

        # Compute effective index in the flat matrix
        i_offsets_flat = seq_i + i_offsets
        k_offsets_flat = seq_k + k_offsets

        # Load left and right tensors from memory
        left = tl.load(left_ptr + i_offsets_flat[:, None] * N + k_offsets_flat[None, :], mask=(i_offsets_flat < N) & (k_offsets_flat < N), other=0.0)
        right = tl.load(right_ptr + k_offsets_flat[None, :] * N + i_offsets_flat[:, None], mask=(k_offsets_flat < N) & (i_offsets_flat < N), other=0.0)

        # Compute the dot product across k and d
        out = tl.dot(left, right, axes=[1, 1])
        tl.store(out_ptr + i_offsets_flat[:, None] * N + d_offsets[None, :], out, mask=i_offsets_flat < N)

    # Configure the grid for the kernel
    # Launch the kernel
    grid_size = (triton.cdiv(N * N, BLOCK_N * BLOCK_K))
    tritonal_tri_mul_kernel[grid_size](
        left, right, out,
        B, N, hidden_dim,
        BLOCK_N=128, BLOCK_K=128, BLOCK_D=128
    )

    # Step 5: Apply output gate and normalize
    out = out.to(torch.float32)
    out = out * out_gate
    out = torch.matmul(out, to_out_norm_weight.t()) + to_out_norm_bias
    out = torch.matmul(out, to_out_weight.t())

    return out
