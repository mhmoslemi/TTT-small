import torch
import triton
import triton.language as tl
from torch import Tensor
from typing import Tuple, Dict

def custom_kernel(data: Tuple[Tensor, Tensor, Dict[str, Tensor], Dict]) -> Tensor:
    """
    Triton kernel implementation of the TriMul operator for protein structure prediction.
    
    This kernel computes the outgoing TriMul operation using matrix multiplications with memory-efficient tiling
    strategies. It handles the pairwise operations efficiently, leveraging Triton's capabilities for high performance.
    The output tensor is cast back to float32 as required.
    
    Args:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
    
    Returns:
        output: Tensor of shape [bs, seq_len, seq_len, dim]
    """

    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Cast to correct data types
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.unsqueeze(-1).to(torch.float32)
    
    # Compute norm and projections
    x = input_tensor
    x = x.mul(norm_weight).add(norm_bias)
    
    left = x @ left_proj_weight.t()
    right = x @ right_proj_weight.t()
    left_gate = x @ left_gate_weight.t().sigmoid()
    right_gate = x @ right_gate_weight.t().sigmoid()
    out_gate = x @ out_gate_weight.t().sigmoid()
    
    # Apply masks
    left = left * mask
    right = right * mask
    
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    
    # Now perform the TriMul operation
    output = triton_tri_mul(
        left,
        right,
        mask,
        batch_size,
        seq_len,
        hidden_dim,
        dim
    )

    # Apply output norm and gate
    output = output @ to_out_norm_weight.t().add(to_out_norm_bias)
    output = output * out_gate
    output = output @ to_out_weight.t()
    
    # Cast output back to float32
    output = output.to(torch.float32)
    return output

@triton.jit
def triton_tri_mul(
    left_ptr,
    right_ptr,
    mask_ptr,
    batch_size,
    seq_len,
    hidden_dim,
    dim,
    BLOCK: tl.constexpr,
    BLOCK2: tl.constexpr
):
    pid = tl.program_id(0)
    num_batches = batch_size
    num_heads = seq_len

    batch_idx = pid // (num_heads * num_heads)
    head_i = (pid // num_heads) % num_heads
    head_j = pid % num_heads

    # Offsets for current batch, head_i, head_j
    offs_batch = batch_idx * num_heads * num_heads
    offs_i = head_i * num_heads
    offs_j = head_j * num_heads
    
    # Iterate over the sequence length for k (we will tile this)
    for k in range(0, seq_len, BLOCK):
        # Load left and right data in tile
        m = tl.arange(0, BLOCK)
        n = tl.arange(0, BLOCK)
        
        # Create block indices for left and right
        i = offs_i + m
        j = offs_j + n
        
        # Masking
        mask = tl.load(mask_ptr + batch_idx * seq_len * seq_len + i + j * seq_len, mask=0.0, other=1.0)

        # Load left and right data
        left = tl.load(left_ptr + (batch_idx * seq_len * seq_len + i * seq_len + j * seq_len) + m * seq_len, mask=(m < seq_len) & (j < seq_len))
        right = tl.load(right_ptr + (batch_idx * seq_len * seq_len + i * seq_len + j * seq_len) + n * seq_len, mask=(n < seq_len) & (i < seq_len))
        
        # Apply mask
        left = left * mask
        right = right * mask
        
        # Multiply and sum the result
        temp = tl.dot(left, right)
        
        # Store result
        tl.store(output_ptr + (batch_idx * seq_len * seq_len + i + j * seq_len), temp, mask=(i < seq_len) & (j < seq_len))
