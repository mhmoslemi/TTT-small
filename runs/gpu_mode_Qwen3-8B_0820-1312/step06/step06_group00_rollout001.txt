import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Custom Triton kernel for the "outgoing" version of the TriMul operator.
    This implementation is optimized for performance on the NVIDIA L40S GPU.
    The kernel performs the forward pass of the TriMul operation using Triton,
    including projections, gates, and the einsum operation.
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Ensure that the input is in the correct data type
    input_tensor = input_tensor.to(torch.float32)
    mask = mask.unsqueeze(-1)
    
    # Define the kernel parameters
    BLOCK = 128
    NUM_BLOCKS = triton.cdiv(seq_len * seq_len, BLOCK)
    
    # Define the kernel
    @triton.jit
    def triton_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr,
        out_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK: tl.constexpr,
    ):
        # Compute the position in the batch
        pid = tl.program_id(0)
        batch_idx = pid // (seq_len * seq_len)
        pid_in_block = pid % (seq_len * seq_len)
        i_idx = pid_in_block // seq_len
        j_idx = pid_in_block % seq_len
        
        # Load input and mask
        input = tl.load(input_ptr + batch_idx * seq_len * seq_len * dim + i_idx * seq_len * dim + j_idx * dim, mask=tl.arange(0, dim) < dim, other=0.0)
        mask_val = tl.load(mask_ptr + batch_idx * seq_len * seq_len + i_idx * seq_len + j_idx, mask=tl.arange(0, 1) < 1, other=0.0)
        
        # Linear projections
        left = tl.dot(input, left_proj_ptr, axis=-1)
        right = tl.dot(input, right_proj_ptr, axis=-1)
        
        # Apply mask
        left = left * mask_val
        right = right * mask_val
        
        # Gates
        left_gate = tl.dot(input, left_gate_ptr, axis=-1).sigmoid()
        right_gate = tl.dot(input, right_gate_ptr, axis=-1).sigmoid()
        out_gate = tl.dot(input, out_gate_ptr, axis=-1).sigmoid()
        
        # Apply gates
        left = left * left_gate
        right = right * right_gate
        
        # Compute the einsum operation
        out = tl.zeros((seq_len, seq_len, dim), dtype=tl.float32)
        for k in range(seq_len):
            a = tl.load(input_ptr + batch_idx * seq_len * seq_len * dim + i_idx * seq_len * dim + k * dim, mask=tl.arange(0, dim) < dim, other=0.0)
            b = tl.load(input_ptr + batch_idx * seq_len * seq_len * dim + j_idx * seq_len * dim + k * dim, mask=tl.arange(0, dim) < dim, other=0.0)
            out = out + tl.outer(a, b)
        
        # Normalize and apply output gate
        out = tl.dot(out, to_out_norm_ptr, axis=-1)
        out = out * out_gate
        
        # Store output
        tl.store(out_ptr + batch_idx * seq_len * seq_len * dim + i_idx * seq_len * dim + j_idx * dim, out, mask=tl.arange(0, dim) < dim)
    
    # Prepare all the weights as Triton-compatible pointers
    input_ptr = input_tensor.data_ptr()
    mask_ptr = mask.data_ptr()
    
    left_proj = weights['left_proj.weight'].contiguous()
    left_proj_ptr = left_proj.data_ptr()
    right_proj = weights['right_proj.weight'].contiguous()
    right_proj_ptr = right_proj.data_ptr()
    
    left_gate = weights['left_gate.weight'].contiguous()
    left_gate_ptr = left_gate.data_ptr()
    right_gate = weights['right_gate.weight'].contiguous()
    right_gate_ptr = right_gate.data_ptr()
    
    out_gate = weights['out_gate.weight'].contiguous()
    out_gate_ptr = out_gate.data_ptr()
    
    to_out_norm = weights['to_out_norm.weight'].contiguous()
    to_out_norm_ptr = to_out_norm.data_ptr()
    to_out = weights['to_out.weight'].contiguous()
    to_out_ptr = to_out.data_ptr()
    
    # We need to allocate output
    output = torch.empty(batch_size, seq_len, seq_len, dim, dtype=torch.float32, device=input_tensor.device)
    out_ptr = output.data_ptr()
    
    # Launch the kernel
    grid = (triton.cdiv(seq_len * seq_len, BLOCK),)
    triton_kernel[grid](
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr, out_gate_ptr,
        to_out_norm_ptr, to_out_ptr,
        out_ptr,
        batch_size, seq_len, dim, hidden_dim,
        BLOCK=BLOCK
    )
    
    return output
