import torch
import triton
import triton.language as tl
import math
import copy

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape
    
    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Step 1: Apply LayerNorm (norm)
    x = input_tensor
    x = x.to(torch.float32)

    # Compute norm
    gamma = norm_weight.to(x.device)
    beta = norm_bias.to(x.device)
    mean = x.mean(dim=(-2, -1), keepdim=True)
    var = x.var(dim=(-2, -1), keepdim=True, unbiased=False)
    x = (x - mean) * torch.rsqrt(var + 1e-6) * gamma + beta

    # Step 2: Project left and right
    # left_proj: [B, N, N, C] -> [B, N, N, hidden_dim]
    left_proj_weight = left_proj_weight.to(x.device)
    left_proj = torch.matmul(x.unsqueeze(2), left_proj_weight.t()).squeeze(2)
    left_proj = left_proj.to(torch.float32)

    # right_proj: [B, N, N, C] -> [B, N, N, hidden_dim]
    right_proj_weight = right_proj_weight.to(x.device)
    right_proj = torch.matmul(x.unsqueeze(2), right_proj_weight.t()).squeeze(2)
    right_proj = right_proj.to(torch.float32)

    # Step 3: Apply mask
    mask = mask.to(x.device)
    mask = mask.unsqueeze(-1)  # [B, N, N, 1]
    left_proj = left_proj * mask
    right_proj = right_proj * mask

    # Step 4: Compute gates
    left_gate_weight = left_gate_weight.to(x.device)
    right_gate_weight = right_gate_weight.to(x.device)
    out_gate_weight = out_gate_weight.to(x.device)

    left_gate = torch.matmul(x.unsqueeze(2), left_gate_weight.t()).squeeze(2).sigmoid()
    right_gate = torch.matmul(x.unsqueeze(2), right_gate_weight.t()).squeeze(2).sigmoid()
    out_gate = torch.matmul(x.unsqueeze(2), out_gate_weight.t()).squeeze(2).sigmoid()

    # Apply gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # Step 5: Compute TriMul (triangular matrix multiplication)
    # out = einsum("b i k d, b j k d -> b i j d", left_proj, right_proj)
    # We do this using a Triton kernel

    # Shape of output: [B, N, N, hidden_dim]
    output = torch.zeros(batch_size, seq_len, seq_len, hidden_dim, device=x.device, dtype=torch.bfloat16)

    # Launch the Triton kernel
    BLOCK = 128
    num_blocks_m = triton.cdiv(seq_len, BLOCK)
    num_blocks_n = triton.cdiv(seq_len, BLOCK)

    kernel_config = {
        "BLOCK_M": BLOCK,
        "BLOCK_N": BLOCK,
        "BLOCK_K": 128,
    }

    # Launch kernel
    kernel_launcher = triton.autotune(
        [kernel_config],
        key=["seq_len"],  # Tune according to seq_len
    )
    
    @kernel_launcher
    @triton.jit
    def trit_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        seq_len,
        BLOCK_M: tl.constexpr,
        BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr,
        hidden_dim: tl.constexpr,
        mask: tl.constexpr
    ):
        # This kernel computes the TriMul for a single batch
        # We iterate over the i and j indices (output indices)
        # For each (i, j), we compute the sum over k: left[i][k] * right[j][k]
        # We use the mask to skip indices where mask[i][j] is 0

        # We are operating on a single batch
        # Get program ID
        pid = tl.program_id(0)
        grid_m = triton.cdiv(seq_len, BLOCK_M)
        grid_n = triton.cdiv(seq_len, BLOCK_N)
        # Get the (i, j) index for this thread
        i = pid // grid_n
        j = pid % grid_n
        i = i * BLOCK_M + tl.arange(0, BLOCK_M)
        j = j * BLOCK_N + tl.arange(0, BLOCK_N)

        # Load left and right data
        # left[i, k, ...] and right[j, k, ...]
        # For each k, compute left[i, k] * right[j, k]
        # Sum over k
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.bfloat16)

        # We need to iterate over k in blocks
        # Since we can't loop in Triton, we use a "tile" approach
        for k in range(0, seq_len, BLOCK_K):
            # Load left and right data
            left_slice = left_ptr + (i[:, None] * seq_len * hidden_dim + k * hidden_dim + tl.arange(0, BLOCK_K)[:, None] * hidden_dim)
            right_slice = right_ptr + (j[None, :] * seq_len * hidden_dim + k * hidden_dim + tl.arange(0, BLOCK_K)[None, :] * hidden_dim)

            # We need to load the data, apply mask, and compute the product sum
            # Convert to bfloat16 for precision
            left_data = tl.load(left_slice, mask=(i[:, None] < seq_len) & (k + tl.arange(0, BLOCK_K)[:, None] < seq_len), other=0.0)
            right_data = tl.load(right_slice, mask=(j[None, :] < seq_len) & (k + tl.arange(0, BLOCK_K)[None, :] < seq_len), other=0.0)

            # Apply mask to left and right
            mask_left = (i[:, None] < seq_len) & (k + tl.arange(0, BLOCK_K)[:, None] < seq_len)
            mask_right = (j[None, :] < seq_len) & (k + tl.arange(0, BLOCK_K)[None, :] < seq_len)
            left_data = tl.where(mask_left, left_data, 0.0)
            right_data = tl.where(mask_right, right_data, 0.0)

            # Compute the product and sum over k
            acc += tl.dot(left_data, right_data)

        # Store the result
        out = out_ptr + i[:, None] * seq_len * hidden_dim + j[None, :] * hidden_dim
        tl.store(out, acc, mask=(i[:, None] < seq_len) & (j[None, :] < seq_len))

    # Launch the kernel for all batches
    for b in range(batch_size):
        # Get data pointers for this batch
        left_ptr = left_proj.data_ptr() + b * seq_len * seq_len * hidden_dim
        right_ptr = right_proj.data_ptr() + b * seq_len * seq_len * hidden_dim
        out_ptr = output.data_ptr() + b * seq_len * seq_len * hidden_dim

        # Launch kernel for this batch
        grid = (triton.cdiv(seq_len * seq_len, BLOCK * BLOCK),)
        trit_mul_kernel[grid](
            left_ptr, right_ptr, out_ptr,
            seq_len,
            BLOCK_M=BLOCK,
            BLOCK_N=BLOCK,
            hidden_dim=hidden_dim,
            mask=1
        )

    # Convert output to float32
    output = output.to(torch.float32)

    # Step 6: Apply LayerNorm (to_out_norm)
    to_out_norm_weight = to_out_norm_weight.to(output.device)
    to_out_norm_bias = to_out_norm_bias.to(output.device)
    mean = output.mean(dim=(-2, -1), keepdim=True)
    var = output.var(dim=(-2, -1), keepdim=True, unbiased=False)
    output = (output - mean) * torch.rsqrt(var + 1e-6) * to_out_norm_weight + to_out_norm_bias

    # Step 7: Apply out_gate
    output = output * out_gate

    # Step 8: Final projection
    # to_out: [B, N, N, hidden_dim] -> [B, N, N, dim]
    to_out_weight = to_out_weight.to(output.device)
    # We reshape output to (B, N, N * hidden_dim) to multiply with to_out_weight
    output = output.reshape(batch_size, seq_len, seq_len * hidden_dim)
    output = torch.matmul(output, to_out_weight.t())
    output = output.reshape(batch_size, seq_len, seq_len, dim)

    return output
