import torch
import triton
import triton.language as tl
import math

def custom_kernel(data) -> torch.Tensor:
    """
    Custom Triton kernel implementation of the TriMul module for protein structure prediction models.
    The kernel computes the outgoing version of the TriMul operation over a 4D input tensor.
    It includes projections, gates, and final output transformation using the provided weights.
    """
    input_tensor, mask, weights, config = data
    batch_size, seqlen, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Access weights
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    # Convert mask to device and format
    mask = mask.to(input_tensor.device).unsqueeze(-1)

    # Output tensor initialization
    output_shape = (batch_size, seqlen, seqlen, dim)
    output = torch.zeros(output_shape, device=input_tensor.device, dtype=torch.float32)

    # Define kernel parameters
    BLOCK_M = 32
    BLOCK_N = 128
    BLOCK_K = 128
    NUM_WARPS = 4
    MAX_FUSED = 512
    GROUP_SIZE_K = 16

    # Launch kernels
    # 1. LayerNorm (norm)
    @triton.jit
    def norm_kernel(x_ptr, y_ptr, n_elements, n_dim, weight_ptr, bias_ptr, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        num_blocks = tl.cdiv(n_elements, BLOCK)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < n_elements
        x = tl.load(x_ptr + offs, mask=mask)
        x = x * weight_ptr[offs] + bias_ptr[offs]
        tl.store(y_ptr + offs, x, mask=mask)

    # 2. Linear projection and gate operations
    @triton.jit
    def linear_proj_gate_kernel(x_ptr, w_ptr, y_ptr, n_elements, n_dim, n_out, scale, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        num_blocks = tl.cdiv(n_elements, BLOCK)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < n_elements
        x = tl.load(x_ptr + offs, mask=mask)
        w = tl.load(w_ptr + offs)
        y = x * w
        tl.store(y_ptr + offs, y, mask=mask)

    # 3. Element-wise operations (multiply, sigmoid, etc.)
    @triton.jit
    def elementwise_ops_kernel(x_ptr, y_ptr, scale, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        num_blocks = tl.cdiv(BLOCK, 8)
        offs = pid * 8 + tl.arange(0, 8)
        mask = offs < BLOCK
        x = tl.load(x_ptr + offs, mask=mask)
        y = x * scale
        tl.store(y_ptr + offs, y, mask=mask)

    # 4. Pairwise multiplication and summation (Einsum-like operation)
    @triton.jit
    def pairwise_kernel(x_ptr, y_ptr, z_ptr, n_batch, n_seq, n_dim, n_k, num_warps: tl.constexpr):
        pid = tl.program_id(0)
        num_blocks = tl.cdiv(n_seq * n_seq, BLOCK_M)
        offs_k = tl.arange(0, BLOCK_K)
        offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = tl.arange(0, BLOCK_N)

        # Compute local indices
        lm = (offs_m // n_seq) * n_seq + (offs_m % n_seq)
        ln = (offs_n // n_seq) * n_seq + (offs_n % n_seq)

        # Load x
        x = tl.load(x_ptr + (offs_m[:, None] * n_seq + offs_k[None, :]), mask=(offs_m < n_seq) & (offs_k < n_k))
        x = x * tl.load(mask_ptr + offs_m[:, None] * n_seq + offs_n[None, :], mask=(offs_m < n_seq) & (offs_n < n_seq))

        # Load y
        y = tl.load(y_ptr + (offs_n[:, None] * n_seq + offs_k[None, :]), mask=(offs_n < n_seq) & (offs_k < n_k))

        # Compute dot product
        acc = tl.dot(x, y, out=tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32))

        # Store result
        tl.store(z_ptr + (offs_m[:, None] * n_seq + offs_n[None, :]), acc, mask=(offs_m < n_seq) & (offs_n < n_seq))

    # Launch kernel 1: LayerNorm
    grid = (triton.cdiv(batch_size * seqlen * seqlen * dim, BLOCK_M),)
    norm_kernel[grid](input_tensor, output, batch_size * seqlen * seqlen * dim, dim, norm_weight, norm_bias, BLOCK_M)

    # Launch kernel 2: Linear projections for left and right
    # Left projection
    grid = (triton.cdiv(batch_size * seqlen * seqlen * dim, BLOCK_M),)
    linear_proj_gate_kernel[grid](output, left_proj_weight, output, batch_size * seqlen * seqlen * dim, dim, hidden_dim, 1.0, BLOCK_M)

    # Right projection
    grid = (triton.cdiv(batch_size * seqlen * seqlen * dim, BLOCK_M),)
    linear_proj_gate_kernel[grid](output, right_proj_weight, output, batch_size * seqlen * seqlen * dim, dim, hidden_dim, 1.0, BLOCK_M)

    # Apply mask
    mask_ptr = mask.data_ptr()
    grid = (triton.cdiv(batch_size * seqlen * seqlen, BLOCK_M),)
    elementwise_ops_kernel[grid](output, output, 0.0, BLOCK_M)

    # Apply left gate
    grid = (triton.cdiv(batch_size * seqlen * seqlen * hidden_dim, BLOCK_M),)
    elementwise_ops_kernel[grid](output, output, 0.0, BLOCK_M)

    # Apply right gate
    grid = (triton.cdiv(batch_size * seqlen * seqlen * hidden_dim, BLOCK_M),)
    elementwise_ops_kernel[grid](output, output, 0.0, BLOCK_M)

    # Apply out gate
    grid = (triton.cdiv(batch_size * seqlen * seqlen * hidden_dim, BLOCK_M),)
    elementwise_ops_kernel[grid](output, output, 0.0, BLOCK_M)

    # Pairwise multiplication and summation
    # Launch pairwise kernel
    grid = (triton.cdiv(n_seq * n_seq, BLOCK_M),)
    pairwise_kernel[grid](output, output, output, batch_size, seqlen, hidden_dim, seqlen, num_warps)

    # Apply layer norm and output projection
    # Apply to_out_norm
    grid = (triton.cdiv(batch_size * seqlen * seqlen * hidden_dim, BLOCK_M),)
    norm_kernel[grid](output, output, batch_size * seqlen * seqlen * hidden_dim, hidden_dim, to_out_norm_weight, to_out_norm_bias, BLOCK_M)

    # Apply output projection
    grid = (triton.cdiv(batch_size * seqlen * seqlen * hidden_dim, BLOCK_M),)
    linear_proj_gate_kernel[grid](output, to_out_weight, output, batch_size * seqlen * seqlen * hidden_dim, hidden_dim, dim, 1.0, BLOCK_M)

    return output
