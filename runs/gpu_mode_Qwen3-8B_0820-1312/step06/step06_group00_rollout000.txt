import torch
import triton
import triton.language as tl
from torch import nn, einsum

def custom_kernel(data):
    """
    Implements the outgoing version of the Triangle Multiplicative Update (TriMul) operator
    using a Triton kernel. This kernel is optimized for large sequence dimensions and
    involves pairwise multiplicative updates over the sequence dimension.

    Parameters:
        data: Tuple of (input: torch.Tensor, mask: torch.Tensor, weights: Dict[str, torch.Tensor], config: Dict)
            - input: Input tensor of shape [bs, seq_len, seq_len, dim]
            - mask: Mask tensor of shape [bs, seq_len, seq_len]
            - weights: Dictionary containing model weights
            - config: Dictionary containing model configuration parameters

    Output:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Move to device as needed
    device = input_tensor.device
    if device.type != 'cuda':
        raise RuntimeError("This kernel must run on CUDA device")

    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Perform batched linear projections
    def linear_projection(x, weight):
        shape = x.shape
        x = x.reshape(-1, shape[-1])
        x = torch.matmul(x, weight.t())
        return x.reshape(shape)

    x = input_tensor
    x = x.to(torch.float32)
    x = x * norm_weight + norm_bias

    left = linear_projection(x, left_proj_weight)
    right = linear_projection(x, right_proj_weight)

    # Apply mask
    mask = mask.unsqueeze(-1).to(torch.float32)
    left = left * mask
    right = right * mask

    left_gate = linear_projection(x, left_gate_weight)
    right_gate = linear_projection(x, right_gate_weight)
    out_gate = linear_projection(x, out_gate_weight)

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Apply the core TriMul operation: (i,k) * (j,k) -> (i,j)
    # We'll implement this with a Triton kernel

    block_size = 64
    out = torch.zeros(bs, seq_len, seq_len, hidden_dim, device=device)
    grid = (triton.cdiv(seq_len, block_size), bs)
    kernel = triton.jit(tri_mul_kernel)
    kernel(out, left, right, seq_len, hidden_dim, block_size, mask)

    out = out * out_gate
    out = torch.matmul(out, to_out_weight.t())
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias)

    # Convert to float32 for final output
    out = out.to(torch.float32)

    return out


@triton.jit
def tri_mul_kernel(out, left, right, seq_len, hidden_dim, block_size, mask):
    pid = tl.program_id(0)
    batch_idx = tl.program_id(1)
    block_start = pid * block_size
    block_idx = block_start + tl.arange(0, block_size)

    if block_start >= seq_len:
        return

    # For all i in [block_start, block_start + block_size)
    # for all j in [0, seq_len)
    # for all k in [0, seq_len)

    # Load left[i, k, :] and right[j, k, :]
    # and sum over k

    # Create indices
    indices_i = block_idx
    indices_j = tl.arange(0, seq_len)

    # We'll do the following:
    # out[batch_idx, i, j] = sum_k (left[batch_idx, i, k] * right[batch_idx, j, k])

    # Allocate accumulators
    acc = tl.zeros((block_size, seq_len), dtype=tl.float32)

    # Loop over k
    k_start = 0
    while k_start < seq_len:
        k_block_start = k_start
        k_block_end = min(k_start + block_size, seq_len)

        # Load left[i, k]
        k_indices = tl.arange(k_block_start, k_block_end)
        left_data = tl.load(left + batch_idx * seq_len * seq_len * hidden_dim
                           + indices_i[:, None] * seq_len * hidden_dim
                           + k_indices[None, :] * hidden_dim, mask=(indices_i < seq_len)[:, None] & (k_indices < seq_len))

        # Load right[j, k]
        right_data = tl.load(right + batch_idx * seq_len * seq_len * hidden_dim
                           + indices_j[None, :] * seq_len * hidden_dim
                           + k_indices[:, None] * hidden_dim, mask=(indices_j < seq_len)[None, :] & (k_indices < seq_len))

        # Multiply and accumulate
        acc += tl.dot(left_data, right_data)

        k_start += block_size

    # Apply mask
    mask_data = tl.load(mask + batch_idx * seq_len * seq_len, mask=(indices_i < seq_len)[:, None] & (indices_j < seq_len)[None, :])
    acc *= mask_data

    # Store the result
    tl.store(out + batch_idx * seq_len * seq_len * hidden_dim
             + indices_i[:, None] * seq_len * hidden_dim
             + indices_j[None, :] * hidden_dim,
             acc,
             mask=(indices_i < seq_len)[:, None] & (indices_j < seq_len)[None, :])
