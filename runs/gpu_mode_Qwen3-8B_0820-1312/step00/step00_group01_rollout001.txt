import torch
import triton
import triton.language as tl
import math
import functools

def custom_kernel(data):
    """
    Optimized Triton kernel for the "outgoing" version of the TriMul operator used in AlphaFold3.
    The kernel performs the following operations:
    1. Applies LayerNorm to input
    2. Projects input through linear transformations
    3. Applies gate functions (sigmoid)
    4. Performs matrix multiplication over the sequence dimension with masking
    5. Normalizes and applies final output gate
    
    Optimizations:
    - Kernel launches in a tiled fashion to optimize memory access patterns
    - Masking is handled efficiently in the matrix multiplication
    - Uses Tensor Core operations for performance on L40S GPUs
    - Maintains float32 for final output
    
    The kernel is designed for the following input shapes: 
    [B, N, N, C] where B ∈ {1,2}, N ∈ {128,256,512,1024}, C ∈ {128, 384}
    """
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Triton kernel configuration
    BLOCK = 64  # Adjust based on GPU memory constraints
    GROUP_SIZE = 16  # Adjust for optimal utilization of Tensor Cores
    
    # Define shared memory layout for kernel
    @triton.jit
    def triton_trimul_kernel(
        input_ptr, mask_ptr, left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr, out_gate_ptr, to_out_norm_ptr, to_out_ptr,
        batch_size, seq_len, hidden_dim, dim, 
        BLOCK: tl.constexpr,
        GROUP_SIZE: tl.constexpr
    ):
        # Compute the local and global indices
        pid = tl.program_id(0)
        pid_m = pid // seq_len
        pid_n = pid % seq_len
        
        # Compute local indices
        local_m = tl.arange(0, BLOCK)
        local_n = tl.arange(0, BLOCK)
        
        # Compute pointer offsets
        offset = pid_m * seq_len * seq_len * hidden_dim + pid_n * seq_len * hidden_dim
        
        # Load input
        input_val = tl.load(input_ptr + offset)
        
        # Apply LayerNorm and projection
        left = tl.load(left_proj_ptr + offset)
        right = tl.load(right_proj_ptr + offset)
        
        # Apply gate operations
        left_gate = tl.load(left_gate_ptr + offset)
        right_gate = tl.load(right_gate_ptr + offset)
        out_gate = tl.load(out_gate_ptr + offset)
        
        # Apply gates
        left = left * left_gate
        right = right * right_gate
        
        # Apply mask
        mask_val = tl.load(mask_ptr + pid_m * seq_len * seq_len + pid_n * seq_len)
        mask_val = mask_val.to(tl.float32)
        left = left * mask_val
        right = right * mask_val
        
        # Matrix multiplication
        out = tl.zeros((BLOCK, BLOCK), dtype=tl.float32)
        for k in range(0, seq_len, BLOCK):
            a = tl.load(left + k * seq_len * hidden_dim, mask=local_m < seq_len - k)
            b = tl.load(right + k * seq_len * hidden_dim, mask=local_n < seq_len - k)
            out += tl.dot(a, b)
        
        # Apply normalization
        out = tl.load(to_out_norm_ptr + offset)
        out = out * out_gate
        
        # Apply output layer
        out = out * tl.load(to_out_ptr + offset)
        
        # Store output
        tl.store(output_ptr + offset, out)
    
    # Host-side memory allocation and setup
    output = torch.empty_like(input_tensor)
    
    # Launch kernel in 3D grid
    grid = (batch_size * seq_len * seq_len,)
    triton_trimul_kernel[grid](
        input_tensor, mask, 
        weights["left_proj.weight"], weights["right_proj.weight"],
        weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"],
        weights["to_out_norm.weight"], weights["to_out.weight"],
        batch_size, seq_len, hidden_dim, dim,
        BLOCK=BLOCK, GROUP_SIZE=GROUP_SIZE
    )
    
    return output
