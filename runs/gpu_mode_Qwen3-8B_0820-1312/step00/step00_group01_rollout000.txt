import torch
import triton
import triton.language as tl
from torch import einsum

# Summary: This Triton kernel implements the outgoing version of the Triangle Multiplicative Update (TriMul)
# operation used in AlphaFold3 and other BioML models. It processes a 4D tensor with sequence-wise pairwise interactions
# and efficiently computes the output using fused operations and parallelism. The mask is applied to suppress zero interactions.

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    seqlen, batch_size = input_tensor.shape[1], input_tensor.shape[0]
    
    # Access the weights
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    # Output tensor
    output = torch.empty((batch_size, seqlen, seqlen, dim), dtype=torch.float32, device=input_tensor.device)

    # LayerNorm: norm
    @triton.jit
    def layer_norm(x_ptr, x_scale_ptr, x_bias_ptr, out_ptr, N, dim, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < N
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        x = x * tl.load(x_scale_ptr + offs, mask=mask, other=0.0)
        x = x + tl.load(x_bias_ptr + offs, mask=mask, other=0.0)
        tl.store(out_ptr + offs, x, mask=mask)

    grid = (triton.cdiv(seqlen, 128),)
    layer_norm[grid](input_tensor.contiguous(), norm_weight, norm_bias, input_tensor, seqlen, dim, 128)

    # Linear projections: left_proj, right_proj
    def linear_proj(x, weight, out):
        @triton.jit
        def linear(x_ptr, w_ptr, out_ptr, N, dim, BLOCK: tl.constexpr):
            pid = tl.program_id(0)
            offs = pid * BLOCK + tl.arange(0, BLOCK)
            mask = offs < N
            x = tl.load(x_ptr + offs, mask=mask, other=0.0)
            w = tl.load(w_ptr + offs, mask=mask, other=0.0)
            out = x * w
            tl.store(out_ptr + offs, out, mask=mask)

        grid = (triton.cdiv(seqlen, 128),)
        linear[x, weight, out, seqlen, dim, 128]

    linear_proj(input_tensor, left_proj_weight, left_tensor)
    linear_proj(input_tensor, right_proj_weight, right_tensor)
    left_tensor = left_tensor.contiguous()
    right_tensor = right_tensor.contiguous()

    # Apply mask
    mask = mask.unsqueeze(-1)
    left_tensor = left_tensor * mask
    right_tensor = right_tensor * mask

    # Gate activations: left_gate, right_gate, out_gate
    def apply_gate(x, weight, out):
        @triton.jit
        def sigmoid(x_ptr, w_ptr, out_ptr, N, dim, BLOCK: tl.constexpr):
            pid = tl.program_id(0)
            offs = pid * BLOCK + tl.arange(0, BLOCK)
            mask = offs < N
            x = tl.load(x_ptr + offs, mask=mask, other=0.0)
            w = tl.load(w_ptr + offs, mask=mask, other=0.0)
            x = x * w
            out = 1.0 / (1.0 + tl.exp(-x))
            tl.store(out_ptr + offs, out, mask=mask)

        grid = (triton.cdiv(seqlen, 128),)
        sigmoid[x, weight, out, seqlen, dim, 128]

    apply_gate(input_tensor, left_gate_weight, left_gate)
    apply_gate(input_tensor, right_gate_weight, right_gate)
    apply_gate(input_tensor, out_gate_weight, out_gate)
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # Apply gates
    left_tensor = left_tensor * left_gate
    right_tensor = right_tensor * right_gate

    # TriMul operation: out = einsum('... i k d, ... j k d -> ... i j d', left, right)
    # Implement this with triton kernel
    @triton.jit
    def tri_mul(x_ptr, y_ptr, out_ptr, N, dim, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        off_i = pid * BLOCK + tl.arange(0, BLOCK)
        mask = off_i < N
        i = off_i
        off_k = tl.arange(0, BLOCK)
        off_j = tl.arange(0, BLOCK)

        # Load left (x) and right (y)
        x = tl.load(x_ptr + i[:, None, None] * N + off_k[None, :, None] * N + off_j[None, None, :], mask=mask, other=0.0)
        y = tl.load(y_ptr + i[:, None, None] * N + off_k[None, :, None] * N + off_j[None, None, :], mask=mask, other=0.0)

        # Compute product and accumulate
        out = x * y
        tl.store(out_ptr + i[:, None, None] * N + off_j[None, None, :], out, mask=mask)

    # Launch tri_mul kernel
    grid = (triton.cdiv(seqlen, 128),)
    tri_mul[left_tensor, right_tensor, output, seqlen, dim, 128]

    # Apply out_gate
    output = output * out_gate

    # LayerNorm: to_out_norm
    @triton.jit
    def layer_norm(x_ptr, x_scale_ptr, x_bias_ptr, out_ptr, N, dim, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < N
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        x = x * tl.load(x_scale_ptr + offs, mask=mask, other=0.0)
        x = x + tl.load(x_bias_ptr + offs, mask=mask, other=0.0)
        tl.store(out_ptr + offs, x, mask=mask)

    grid = (triton.cdiv(seqlen, 128),)
    layer_norm[grid](output, to_out_norm_weight, to_out_norm_bias, output, seqlen, dim, 128)

    # Linear: to_out
    @triton.jit
    def linear(x_ptr, w_ptr, out_ptr, N, dim, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < N
        x = tl.load(x_ptr + offs, mask=mask, other=0.0)
        w = tl.load(w_ptr + offs, mask=mask, other=0.0)
        out = x * w
        tl.store(out_ptr + offs, out, mask=mask)

    grid = (triton.cdiv(seqlen, 128),)
    linear[grid](output, to_out_weight, output, seqlen, dim, 128)

    return output
