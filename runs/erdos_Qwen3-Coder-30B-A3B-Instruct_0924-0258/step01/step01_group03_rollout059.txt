import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to [0,1] and scale to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available

        n_points = len(h)
    else:

        h = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Initialize
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Adaptive learning rate
    lr = 0.01
    patience = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Gradient computation
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad[i] = (1 - 2 * h[j])
        
        # Update
        h = h - lr * grad
        
        # Project
        h = project_to_feasible(h, target_sum)
        
        # Evaluate
        c5 = evaluate(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            # Reduce learning rate if we're improving consistently
            if iteration % 100 == 0 and lr > 1e-5:
                lr *= 0.9
        else:
            patience += 1
            if patience > 50:
                # Add small random noise to escape local minima
                noise = np.random.uniform(-1e-3, 1e-3, n_points)
                h = h + noise
                h = project_to_feasible(h, target_sum)
                patience = 0
        
        iteration += 1
        
        # Early termination check
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
