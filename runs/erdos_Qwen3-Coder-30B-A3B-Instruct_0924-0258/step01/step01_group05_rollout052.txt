import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def generate_initial_candidates(n_candidates, n_points):
        candidates = []
        for _ in range(n_candidates):
            h = np.random.uniform(0, 1, n_points)
            h = project_feasible(h)
            candidates.append(h.copy())
        return candidates
    
    def coordinate_descent_step(h_vec, step_size=0.05):
        h_new = h_vec.copy()
        best_h = h_vec.copy()
        best_C5 = compute_C5(h_vec)
        
        # Try small steps for each element
        for i in range(len(h_vec)):
            # Try positive step
            h_trial = h_new.copy()
            h_trial[i] = min(1.0, h_trial[i] + step_size)
            h_trial = project_feasible(h_trial)
            c5_trial = compute_C5(h_trial)
            if c5_trial < best_C5:
                best_C5 = c5_trial
                best_h = h_trial.copy()
            
            # Try negative step
            h_trial = h_new.copy()
            h_trial[i] = max(0.0, h_trial[i] - step_size)
            h_trial = project_feasible(h_trial)
            c5_trial = compute_C5(h_trial)
            if c5_trial < best_C5:
                best_C5 = c5_trial
                best_h = h_trial.copy()
        
        return best_h, best_C5
    
    def local_search(h_vec, max_iter=50):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        best_h = h_current.copy()
        
        step_size = 0.05
        no_improve_count = 0
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Coordinate descent step
            h_new, c5_new = coordinate_descent_step(h_current, step_size)
            
            if c5_new < best_C5:
                h_current = h_new.copy()
                best_C5 = c5_new
                best_h = h_current.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
                
            # Adjust step size if stagnated
            if no_improve_count > 10:
                step_size = min(0.2, step_size * 1.5)
                no_improve_count = 0
                
        return best_h, best_C5
    
    # Generate initial candidates using LHS-like sampling
    candidates = generate_initial_candidates(5, n_points)
    best_h = None
    best_C5 = float('inf')
    
    for h_init in candidates:
        h_opt, c5_opt = local_search(h_init, max_iter=100)
        if c5_opt < best_C5:
            best_C5 = c5_opt
            best_h = h_opt.copy()
    
    # Main optimization loop
    max_no_improve = 100
    no_improve_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        # Run local optimization from current best
        h_opt, c5_opt = local_search(best_h, max_iter=200)
        
        if c5_opt < best_C5:
            best_C5 = c5_opt
            best_h = h_opt.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Perturb if no improvement for too long
        if no_improve_count > max_no_improve:
            # Generate new random candidate
            h_new = np.random.uniform(0, 1, n_points)
            h_new = project_feasible(h_new)
            h_opt, c5_opt = local_search(h_new, max_iter=100)
            if c5_opt < best_C5:
                best_C5 = c5_opt
                best_h = h_opt.copy()
            no_improve_count = 0
            
        # Add random perturbation occasionally
        if np.random.random() < 0.3 and no_improve_count < 20:
            # Add noise
            noise = np.random.normal(0, 0.02, n_points)
            h_perturbed = best_h + noise
            h_perturbed = project_feasible(h_perturbed)
            h_opt, c5_opt = local_search(h_perturbed, max_iter=30)
            if c5_opt < best_C5:
                best_C5 = c5_opt
                best_h = h_opt.copy()
    
    # Final validation
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
