import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def pattern_perturbation(h, strength=0.1):
    """Apply pattern-based perturbation to explore new configurations"""
    n = len(h)
    # Use a sinusoidal pattern for exploration 
    pattern = np.sin(np.linspace(0, 4*np.pi, n))
    perturbed = h + strength * pattern
    return project_to_feasible(perturbed, n)

def augmented_lagrangian_optimize(h_start, n_points, dx, time_limit):
    """Optimize using augmented Lagrangian method with pattern exploration"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initial guess for maximum overlap
    max_overlap_guess = 0.5
    
    # Alternative optimization approach: minimize the negative of the maximum overlap
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        # Use a penalty method: maximize the minimum overlap
        return -max_val * dx
    
    def constraint_func(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx - 0.3  # We want to maximize overlap, so constraint to be met
    
    # Constraints for optimization
    cons = [{'type': 'ineq', 'fun': lambda h: 0.3 - evaluate_erdos_solution(h, dx)}]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a sequence of optimization steps with different starting points
    for iteration in range(100):
        if time() - start_time > time_limit - 1:
            break
            
        # Start from a perturbed version
        h_perturbed = pattern_perturbation(best_h, 0.1 + 0.02 * np.sin(iteration))
        
        try:
            res = minimize(obj, h_perturbed, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 30, 'ftol': 1e-6})
            
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_opt, dx)
                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
                    
        except:
            pass
        
        # Occasionally increase perturbation strength if not making progress
        if iteration % 20 == 0 and iteration > 0:
            h_perturbed = pattern_perturbation(best_h, 0.15)
        else:
            h_perturbed = project_to_feasible(best_h + np.random.normal(0, 0.03, n_points), n_points)
    
    return best_h, best_c5
