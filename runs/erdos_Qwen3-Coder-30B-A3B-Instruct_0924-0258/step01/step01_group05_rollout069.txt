import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load parent solution if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values, dtype=float)
        n_points = len(h_base)
    else:
        # Default initialization
        n_points = 45
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def grid_search_level(n, num_candidates=20):
        candidates = []
        for _ in range(num_candidates):
            # Generate random feasible point
            h_candidate = np.random.rand(n)
            h_candidate = project_feasible(h_candidate)
            candidates.append(h_candidate)
        
        best_h = None
        best_C5 = float('inf')
        
        for h_cand in candidates:
            if time.time() - start_time > budget_s - 0.1:
                break
            C5 = compute_C5(h_cand)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_cand.copy()
        
        return best_h, best_C5
    
    def refine_grid(h_init, n_refine):
        # Interpolate to finer grid
        h_refined = np.interp(np.linspace(0, 1, n_refine), 
                              np.linspace(0, 1, len(h_init)), h_init)
        # Perturb slightly
        perturbation = np.random.normal(0, 0.02, n_refine)
        h_refined = h_refined + perturbation
        h_refined = project_feasible(h_refined)
        return h_refined
    
    # Grid hierarchy
    grid_levels = [5, 10, 15, 20, 25, 30, 45]
    current_h = h_base.copy()
    current_C5 = compute_C5(current_h)
    
    # Start with coarse grid search
    for i, n_level in enumerate(grid_levels):
        if n_level == 45:
            # Final level - use more candidates and depth
            candidates_per_level = 20
        else:
            candidates_per_level = 15
            
        # Perform grid search
        h_level, C5_level = grid_search_level(n_level, candidates_per_level)
        
        if C5_level < current_C5:
            current_C5 = C5_level
            current_h = h_level.copy()
            
        # Refine if not final level
        if i < len(grid_levels) - 1 and time.time() - start_time < budget_s - 1.0:
            next_n = grid_levels[i+1]
            # Refine the current best
            h_refined = refine_grid(current_h, next_n)
            h_refined = project_feasible(h_refined)
            C5_refined = compute_C5(h_refined)
            
            if C5_refined < current_C5:
                current_C5 = C5_refined
                current_h = h_refined.copy()
    
    # Final local optimization
    h_best = current_h.copy()
    C5_best = compute_C5(h_best)
    
    # Try local perturbations
    for _ in range(20):
        if time.time() - start_time > budget_s - 0.1:
            break
        h_perturbed = h_best + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_feasible(h_perturbed)
        C5_perturbed = compute_C5(h_perturbed)
        if C5_perturbed < C5_best:
            C5_best = C5_perturbed
            h_best = h_perturbed.copy()
    
    # Final projection
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
