import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range and rescale to sum to target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    def objective(h_vec):
        return compute_c5(h_vec)
    
    def solve_feasibility(h_vec, max_overlap):
        # Create bounds for h
        bounds = [(0, 1) for _ in range(n_points)]
        # Define the constraint that max overlap <= max_overlap
        def constraint_max_overlap(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            corr *= dx
            return max_overlap - np.max(corr)
        
        # Use a simple SLSQP optimization to find a feasible point
        try:
            result = minimize(objective, h_vec, method='SLSQP', bounds=bounds, 
                            constraints={'type': 'ineq', 'fun': constraint_max_overlap},
                            options={'maxiter': 50})
            return result.success, result.x
        except:
            return False, h_vec
    
    # Binary search for minimal overlap
    start_time = time.time()
    
    # Initial interval - lower bound is current known good value
    low = 0.38
    high = compute_c5(h)  # Current value
    
    # Binary search loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while time.time() - start_time < budget_s - 1 and high - low > 1e-5:
        mid = (low + high) / 2.0
        
        # Attempt to find a feasible solution with this max overlap
        success, h_new = solve_feasibility(best_h, mid)
        
        if success:
            # Try to minimize the actual overlap
            h_new = project_h(h_new)
            current_c5 = compute_c5(h_new)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
            high = mid  # Satisfiable, try for lower bound
        else:
            low = mid  # Unsatisfiable, increase lower bound
    
    return (best_h.tolist(), best_c5, n_points)
