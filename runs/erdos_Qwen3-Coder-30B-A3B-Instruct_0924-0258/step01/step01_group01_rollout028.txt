import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    resolutions = [10, 20, 40, 70]
    n_points = resolutions[0]
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Time per resolution
    time_per_res = budget_s * 0.95 / len(resolutions)
    
    for res_idx, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n_points
        
        # Interpolate if needed
        if res_idx > 0:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        
        # Optimize at this resolution
        best_in_res = h.copy()
        c5_in_res = evaluate(h, dx)
        
        # Particle swarm optimization
        particles = 100
        positions = np.random.uniform(0, 1, (particles, n_points))
        velocities = np.random.uniform(-0.05, 0.05, (particles, n_points))
        pbest = positions.copy()
        pbest_scores = np.array([evaluate(project_to_constraints(p, dx), dx) for p in positions])
        gbest_idx = np.argmin(pbest_scores)
        gbest = pbest[gbest_idx].copy()
        gbest_score = pbest_scores[gbest_idx]
        
        w = 0.7  # inertia weight
        c1 = 1.5  # cognitive parameter
        c2 = 1.5  # social parameter
        max_iter = int(time_per_res / 0.01)  # Estimate iterations
        
        iter_count = 0
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            for i in range(particles):
                # Update velocity
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (pbest[i] - positions[i]) + 
                                c2 * r2 * (gbest - positions[i]))
                # Clip velocity
                velocities[i] = np.clip(velocities[i], -0.05, 0.05)
                
                # Update position
                positions[i] += velocities[i]
                positions[i] = np.clip(positions[i], 0, 1)
                
                # Apply constraints
                positions[i] = project_to_constraints(positions[i], dx)
                
                # Evaluate
                score = evaluate(positions[i], dx)
                
                # Update personal best
                if score < pbest_scores[i]:
                    pbest_scores[i] = score
                    pbest[i] = positions[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest = positions[i].copy()
            
            iter_count += 1
            
        # Gradient descent refinement
        h = gbest.copy()
        step_size = 0.01
        max_gd_iter = 5000
        last_c5 = gbest_score
        
        for _ in range(max_gd_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Gradient: dC5/dh_i = (1 - 2*h_i) * dx
            grad = (1 - 2 * h) * dx
            
            # Update direction
            h_new = h - step_size * grad
            
            # Clip
            h_new = np.clip(h_new, 0, 1)
            
            # Project
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
            else:
                step_size *= 0.99
                
        if best_c5 < c5_in_res:
            h = best_h.copy()
            c5_in_res = best_c5
        else:
            h = best_in_res.copy()
    
    return best_h.tolist(), best_c5, len(best_h)
