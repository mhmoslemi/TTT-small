import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    n_particles = 100
    max_iter = 2000
    w_start = 0.7
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for i in range(n_particles):
        if i == 0:
            h_init = h.copy()
        else:
            h_init = h + np.random.normal(0, 0.05, n_points)
        h_init = project_feasible(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        pbest_positions.append(h_init.copy())
        pbest_values.append(compute_C5(h_init))
    
    # Global best tracking
    gbest_idx = np.argmin(pbest_values)
    gbest_value = pbest_values[gbest_idx]
    gbest_position = pbest_positions[gbest_idx].copy()
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) +
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_speed, max_speed)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate fitness
            current_value = compute_C5(particles[i])
            
            # Update particle best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < gbest_value:
                    gbest_value = current_value
                    gbest_position = particles[i].copy()
    
    # Return best solution found
    return (gbest_position.tolist(), gbest_value, n_points)
