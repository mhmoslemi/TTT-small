import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization: step function
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations and find maximum
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Ensure h in [0,1] and sum = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient for correlation at lag k_max
        n = len(h_vec)
        grad = np.zeros(n)
        
        # For lag k_max, gradient component for h[i] is:
        # - (1 - h[(i + k_max) % n]) if (i + k_max) < n
        # - h[(i - k_max) % n] if (i - k_max) >= 0

        for i in range(n):
            if i + k_max < n:
                grad[i] -= (1 - h_vec[i + k_max])
            if i - k_max >= 0:
                grad[i] -= h_vec[i - k_max]
        
        return grad
    
    def gradient_step(h_vec, alpha, k_max):
        # Compute gradient and update
        grad = compute_gradient(h_vec, k_max)
        h_new = h_vec - alpha * grad
        return project_feasible(h_new)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    alpha = 0.1
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute all overlaps
        corr = correlate(best_h, 1 - best_h, mode='full')
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (n_points - 1)
        
        # Gradient step
        h_new = gradient_step(best_h, alpha, k_max)
        new_C5 = compute_C5(h_new)
        
        if new_C5 < best_C5:
            best_h = h_new.copy()
            best_C5 = new_C5
            stagnation_count = 0
            # Reduce alpha slightly to fine-tune
            alpha *= 0.95
        else:
            stagnation_count += 1
            if stagnation_count > 50:
                # Perturb the solution
                noise = np.random.normal(0, 0.01, n_points)
                h_perturbed = best_h + noise
                h_perturbed = project_feasible(h_perturbed)
                perturbed_C5 = compute_C5(h_perturbed)
                if perturbed_C5 < best_C5:
                    best_h = h_perturbed
                    best_C5 = perturbed_C5
                # Reset stagnation count
                stagnation_count = 0
                alpha = 0.1
                
        # Occasionally reinitialize with best solution
        if np.random.random() < 0.02:
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_feasible(h_perturbed)
            perturbed_C5 = compute_C5(h_perturbed)
            if perturbed_C5 < best_C5:
                best_h = h_perturbed
                best_C5 = perturbed_C5
    
    # Final projection to ensure feasibility
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
