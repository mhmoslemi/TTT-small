import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

    # Ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    lr = 0.01
    max_iter = 10000
    patience = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Find best shift k
        corr = correlate(h, 1 - h, mode="full")
        best_k = np.argmax(corr) - (len(h) - 1)
        
        # Compute gradient
        grad = np.zeros_like(h)
        if best_k >= 0:
            for j in range(len(h) - best_k):
                grad[j] += 1 - 2*h[j + best_k]
        else:
            for j in range(-best_k, len(h)):
                grad[j] += 1 - 2*h[j + best_k]
        
        # Update step
        h = h - lr * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Adaptive learning rate
        if i % 50 == 0 and i > 0:
            lr *= 0.95
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
