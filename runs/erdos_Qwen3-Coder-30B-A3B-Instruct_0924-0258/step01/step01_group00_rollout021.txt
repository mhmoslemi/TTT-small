import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def greedy_improve(h, dx, target_c5):
    # Get the maximum overlap and corresponding shift
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    max_idx = np.argmax(corr)
    # The shift k is max_idx - (n_points - 1)
    k = max_idx - (len(h) - 1)
    
    # Identify indices that contribute to max overlap
    # corr[i] = sum h[j] * (1 - h[j + k]) for j such that j + k = i
    # but we adjust for cases near boundaries
    indices = []
    for i in range(len(h)):
        if i + k >= 0 and i + k < len(h):
            # This contributes to the correlation
            indices.append((i, i + k))
    
    # Calculate values for these pairs
    contributions = [(h[i], h[j], i, j) for i, j in indices]
    contributions.sort(key=lambda x: x[0] * (1 - x[1]), reverse=True)
    
    # Focus on top 10% for improvement
    top_n = max(1, len(contributions) // 10)
    to_adjust = contributions[:top_n]
    
    new_h = h.copy()
    
    # Decrease h for contributing indices
    adjustment_amount = min(0.1, (max_val * dx - target_c5) / len(h))
    for h_i, h_j, i, j in to_adjust:
        if i < len(new_h):
            new_h[i] = max(0, new_h[i] - adjustment_amount)
        if j < len(new_h):
            new_h[j] = max(0, new_h[j] - adjustment_amount)
    
    # Re-normalize
    new_h = project_to_feasible(new_h, len(h))
    
    return new_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construct if available

    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, 2.0 / n_points)
    best_n_points = n_points
    
    # Main loop
    iterations = 0
    local_improvement_count = 0
    max_local_count = 50
    
    while time() - start_time < budget_s * 0.95:
        iterations += 1
        
        # Try to improve with greedy step
        dx = 2.0 / n_points
        new_h = greedy_improve(h, dx, 0.38)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            local_improvement_count = 0
            if new_c5 < 0.3807:
                break  # Sufficient improvement
        else:
            local_improvement_count += 1
        
        h = new_h.copy()
        
        # Occasionally try local perturbation if no recent improvement
        if local_improvement_count > max_local_count:
            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
            local_improvement_count = 0
            # Possibly increase resolution
            if n_points < 100:
                n_points += 15
                h = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(h)), h)
                h = project_to_feasible(h, n_points)
        
        # Early termination if improvement seems slow
        if iterations > 300 and best_c5 < 0.381:
            break
        if iterations > 500:
            break
    
    return (list(best_h), best_c5, n_points)
