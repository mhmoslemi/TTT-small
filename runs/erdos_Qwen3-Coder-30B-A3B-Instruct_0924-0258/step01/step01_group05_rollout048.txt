import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient for correlation at lag k_max
        n = len(h_vec)
        grad = np.zeros(n)
        
        # For lag k_max, gradient is:
        # d/dh[i] [h[i] * (1 - h[i+k_max])] = 1 - h[i+k_max] - h[i-k_max]
        # where indices are taken modulo n
        if k_max >= 0:
            # Positive lag case: h[i] contributes to lag k_max with h[i+k_max]
            for i in range(n):
                grad[i] = 1 - h_vec[(i + k_max) % n]
                if (i - k_max) >= 0:
                    grad[i] -= h_vec[i - k_max]
                else:
                    grad[i] -= h_vec[(i - k_max) % n]
        else:
            # Negative lag case: h[i-k_max] contributes to lag k_max with h[i]
            for i in range(n):
                if (i - k_max) >= 0:
                    grad[i] = 1 - h_vec[i - k_max]
                else:
                    grad[i] = 1 - h_vec[(i - k_max) % n]
                if (i + k_max) >= 0:
                    grad[i] -= h_vec[i + k_max]
                else:
                    grad[i] -= h_vec[(i + k_max) % n]
        
        return grad
    
    def line_search(h_vec, grad, max_alpha=1.0):
        # Line search: find best alpha in step direction
        best_alpha = 0.0
        best_h_new = h_vec.copy()
        best_C5 = compute_C5(h_vec)
        alpha = max_alpha
        step_size = 0.5
        
        # Try different alphas
        while alpha > 1e-6:
            h_new = h_vec - alpha * grad
            h_new = project_feasible(h_new)
            C5_new = compute_C5(h_new)
            
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h_new = h_new.copy()
                best_alpha = alpha
            else:
                alpha *= step_size
        return best_h_new, best_C5, best_alpha
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Start gradient based optimization
    max_iter = 1000
    iteration = 0
    no_improvement_count = 0
    alpha = 0.1
    
    while iteration < max_iter and time.time() - start_time < budget_s - 0.1:
        # Compute all correlations
        corr = correlate(h_best, 1 - h_best, mode='full')
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_best) - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient(h_best, max_lag)
        
        # Perform line search along gradient direction
        h_new, C5_new, alpha_used = line_search(h_best, grad, alpha)
        
        # If there's an improvement, accept it
        if C5_new < C5_best:
            h_best = h_new
            C5_best = C5_new
            no_improvement_count = 0
            alpha = min(0.1, alpha * 1.1)  # Increase learning rate slowly
        else:
            # No improvement, try new random start or reduce learning rate
            no_improvement_count += 1
            alpha *= 0.5

        # Occasionally perturb if stuck
        if no_improvement_count > 20:
            # Add small random noise
            h_random = h_best + np.random.normal(0, 0.01, n_points)
            h_best = project_feasible(h_random)
            C5_best = compute_C5(h_best)
            no_improvement_count = 0
            alpha = 0.1  # Reset learning rate

        iteration += 1
    
    # Final projection and evaluation
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
