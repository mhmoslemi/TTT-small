import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def get_worst_shift(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    # Find the corresponding shift k
    n = len(h)
    k = max_idx - (n - 1)
    return k, max_val * dx

def greedy_adjustment(h, n_points, dx, target_c5=0.38, adjustment_fraction=0.1):
    k, c_max = get_worst_shift(h, dx)
    
    # Find indices that contribute most to the overlap for shift k
    corr = np.correlate(h, 1 - h, mode='full')
    # We actually need to compute the actual overlaps, not the correlation
    # for shift k, find indices x where h(x) * h(x+k) contributes significantly to the max
    
    # Find the actual regions of overlap for the worst shift
    total_n = 2 * n_points  # full length of correlation
    start_idx = k + n_points - 1
    indices = []
    for i in range(n_points):
        if i + k >= 0 and i + k < n_points:
            indices.append(i)
    
    # If indices are empty or not enough, just use the most contributing indices from full correlation
    if not indices:
        # Just take the areas of high correlation
        contribs = np.abs(corr)
        # pick top 10% of indices that aren't boundary cases
        top_indices = np.argsort(contribs)[-int(0.1 * len(corr)):]
        indices = []
        for idx in top_indices:
            # adjust index to reflect real overlap region
            real_x = idx - (n_points - 1)
            real_y = real_x + k
            if 0 <= real_x < n_points and 0 <= real_y < n_points:
                indices.append(real_x)
        indices = list(set(indices))[:n_points//10]
    
    # If still no valid indices, pick first few 10%
    if not indices:
        indices = list(range(n_points//10))
    
    # Update h values with some decrease for these indices
    update_amount = adjustment_fraction * (c_max - target_c5)
    if update_amount <= 0:
        update_amount = 0.001
    
    h_new = h.copy()
    for idx in indices:
        if idx < len(h_new):
            h_new[idx] = max(0, h_new[idx] - update_amount)
    
    return h_new

    resolutions = [15, 30, 45, 60, 75]
    current_res = 0
    
    max_iter_per_res = 500
    iter_per_res = 0
    
    # Keep track of improvement count
    improvement_count = 0
    no_improvement_count = 0

    while current_res < len(resolutions) and time() - start_time < budget_s * 0.95:

        # Interpolate or create h if needed

            h_interp = np.interp(np.linspace(0, 2, n_points),

        iter_per_res = 0
