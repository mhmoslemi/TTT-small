import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Default initialization
        initial_h = np.full(45, 1.0/45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial h satisfies constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    # GA Parameters
    population_size = 50
    max_generations = 1000
    mutation_rate = 0.05
    elite_ratio = 0.25
    
    # Repair function
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Fitness function
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    # Best solution tracking
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for generation in range(max_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = compute_c5(h)
            fitness_scores.append(-c5)  # Negative for maximization
            
        # Update best
        for i, h in enumerate(population):
            if -fitness_scores[i] < best_c5:
                best_c5 = -fitness_scores[i]
                best_h = h.copy()
                
        # Selection: tournament
        new_population = []
        
        # Elitism
        elite_size = int(elite_ratio * population_size)
        sorted_indices = np.argsort(fitness_scores)[-elite_size:]
        for idx in sorted_indices:
            new_population.append(population[idx].copy())
            
        # Tournament selection
        while len(new_population) < population_size:
            # Tournament size = 3
            tournament_indices = np.random.choice(population_size, 3, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitness)]
            new_population.append(population[winner_index].copy())
            
        # Crossover (uniform)
        offspring = []
        for i in range(0, len(new_population), 2):
            if i + 1 < len(new_population):
                parent1 = new_population[i]
                parent2 = new_population[i+1]
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
                # Repair children
                child1 = project_h(child1)
                child2 = project_h(child2)
                offspring.append(child1)
                offspring.append(child2)
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                offspring[i] += noise
                offspring[i] = project_h(offspring[i])
                
        # Replace population
        # Keep elites and replace rest with offspring
        population = new_population[:elite_size] + offspring[:population_size - elite_size]
        
    if best_h is None:
        # Fallback to last population if no improvement
        best_h = population[0]
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
