import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Initialize with provided construction or small random vector

    else:

    n_points = len(h_start)
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    current_h = h_start.copy()
    
    # Track improvement attempts
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Main optimization loop
    while time() - start_time < budget_s * 0.95:
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        # Compute correlation and find worst shift
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_corr_idx = np.argmax(corr)
        k_offset = max_corr_idx - (n_points - 1)
        
        # Identify overlapping indices where this maximum occurs
        # The maximum overlap happens in the correlation at shift k_offset
        # We need to adjust indices where h(x)*h(x+k) contributes most
        full_corr = corr * dx
        
        # Find all contribution positions
        contrib_positions = []
        for i in range(n_points):
            j = i + k_offset
            if 0 <= j < n_points:
                contrib_positions.append((i, j, current_h[i] * current_h[j]))
        
        # Sort by contribution
        contrib_positions.sort(key=lambda x: x[2], reverse=True)
        
        # Select top 10% of contributing positions to adjust
        n_adjust = max(1, int(len(contrib_positions) * 0.1))
        adjust_indices = set()
        for i in range(n_adjust):
            adjust_indices.add(contrib_positions[i][0])
            adjust_indices.add(contrib_positions[i][1])
        
        # Update h in affected regions
        adjustment = (current_c5 - 0.38) / (n_points * 2)

            if idx < len(new_h):
                new_h[idx] = max(0, new_h[idx] - adjustment)
        
        # Project new_h to feasibility
        new_h = project_to_feasible(new_h, n_points)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept improvement or perturb if no gain
        if new_c5 < current_c5:
            current_h = new_h.copy()
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
            # Add Gaussian noise for escape
            if no_improvement_count > max_no_improvement:
                noise = np.random.normal(0, 0.01, n_points)
                current_h = project_to_feasible(current_h + noise, n_points)
                no_improvement_count = 0

    return (list(best_h), best_c5, n_points)
