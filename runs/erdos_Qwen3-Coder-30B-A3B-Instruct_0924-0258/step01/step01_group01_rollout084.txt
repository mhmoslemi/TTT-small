import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def smooth_max(x, beta=10.0):
    # Smooth approximation to max using log-sum-exp
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

def compute_correlation(h, dx):
    # Compute the correlation integral for all lags
    corr = correlate(h, 1 - h, mode='full')
    return corr * dx

def project_to_constraints(h, dx):
    # Ensure sum(h) * dx = 1, i.e., sum(h) = n/2
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    
    return h

def evaluate(h, dx):
    corr = compute_correlation(h, dx)
    return np.max(corr)

def gradient_of_smooth_max(h, dx, beta=10.0):
    # Compute gradient of smooth max of correlations
    corr = compute_correlation(h, dx)
    n = len(h)
    max_idx = np.argmax(corr)
    # Gradient through the smooth max
    exp_corr = np.exp(beta * corr)
    Z = np.sum(exp_corr)
    # Gradient w.r.t. h
    grad = np.zeros_like(h)
    # For the correlation computation, we use backprop through FFT-based computation
    # Approximate gradient by taking differences for now
    # Better: use finite difference or symbolic gradient from correlation
    # We use finite differences for this approximation
    epsilon = 1e-5
    grad = np.zeros_like(h)
    for i in range(n):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += epsilon
        h_minus[i] -= epsilon
        corr_plus = compute_correlation(h_plus, dx)
        corr_minus = compute_correlation(h_minus, dx)
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * epsilon)
    
    return grad

def line_search_armijo(f, grad, h, alpha_max=1.0, c1=1e-4, rho=0.5):
    # Armijo backtracking line search
    alpha = alpha_max
    f_old = f(h)
    grad_norm_sq = np.dot(grad, grad)
    while alpha > 1e-10:
        h_new = h - alpha * grad
        h_new = project_to_constraints(h_new, 1.0 / len(h))  # dx=1/n for now
        f_new = f(h_new)
        if f_new <= f_old - c1 * alpha * grad_norm_sq:
            return alpha
        alpha *= rho
    return alpha

    best_h = h.copy()

    cur_c5 = best_c5
    
    max_iter = 1000
    current_iter = 0
    learning_rate = 1e-3
    patience = 0
