import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(30) * 0.5
        
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Try different resolutions
    resolutions = [30, 45, 60, 75]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if needed
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Try multiple restarts with different perturbations
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Add small random perturbation
            h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)
            
            # Define objective function to minimize
            def objective(h):
                return -evaluate_erdos_solution(h, dx)
            
            # Constraints
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            
            # Bounds
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                # Use L-BFGS-B for optimization
                res = minimize(objective, h_perturbed, method='L-BFGS-B', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_res = project_to_feasible(res.x, n_points)
                    c5 = -res.fun  # because we minimized negative
                    if c5 < best_c5:
                        best_h = h_res.copy()
                        best_c5 = c5
                        best_n_points = n_points
            except:
                pass
            
            # Follow-up refinement with SLSQP
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_res = project_to_feasible(res.x, n_points)
                    c5 = -res.fun
                    if c5 < best_c5:
                        best_h = h_res.copy()
                        best_c5 = c5
                        best_n_points = n_points
            except:
                pass
                
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
