import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute the overlap integral using convolution
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_relaxation(h, n_points, dx):
    """Use CVXPY to compute a convex relaxation of the problem"""
    # Create optimization variables
    h_var = cp.Variable(n_points)
    
    # Define the objective as a relaxation of the maximum correlation
    # We'll minimize a smooth approximation to max correlation
    # Since we want to minimize max_i correlation, we minimize sum of squares
    # of all correlations weighted by the maximum
    corr = np.correlate(h, 1 - h, mode='full')
    objective = cp.Minimize(cp.sum_squares(corr) * dx**2)
    
    # Constraints
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return h_var.value
    except:
        pass
    
    # Fallback to projection if optimization fails
    return project_to_feasible(h, n_points)

def genetic_mutation(h, n_points, mutation_rate=0.1):
    """Apply genetic mutation to solution h"""
    h_mut = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Random perturbation
            h_mut[i] += np.random.normal(0, 0.05)
            
            # Polynomial adjustment for smooth changes
            if np.random.random() < 0.5:
                # Apply small polynomial to create smoother transitions 
                h_mut[i] += 0.01 * (i - n_points/2)**2
                
    return project_to_feasible(h_mut, n_points)

def hybrid_optimization(h_start, n_points, dx, time_limit):
    """Use hybrid approach: genetic mutations followed by convex relaxation"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    pop_size = max(10, n_points // 5)
    generations = 20
    
    for gen in range(generations):
        if time() - start_time > time_limit * 0.99:
            break
            
        # Generate population
        population = [h_start.copy()]
        for _ in range(pop_size - 1):
            h_mut = genetic_mutation(h_start.copy(), n_points)
            population.append(h_mut)
            
        # Evaluate population
        results = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            results.append((c5, h))
            
        # Sort by performance
        results.sort()
        best_in_gen = results[0]
        
        if best_in_gen[0] < best_c5:
            best_h = best_in_gen[1].copy()
            best_c5 = best_in_gen[0]
            
        # Select top performers and create new generation
        top_performers = [h for _, h in results[:max(2, pop_size//3)]]
        h_start = top_performers[np.random.randint(0, len(top_performers))].copy()
        
        # Convex relaxation for exploitation
        h_refined = convex_relaxation(h_start, n_points, dx)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
            
        # Early termination heuristic
        if gen > 3 and abs(best_c5 - evaluate_erdos_solution(best_h, dx)) < 1e-5:
            break
    
    return best_h, best_c5

        if time() - start_time > budget_s * 0.95:

        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points),
