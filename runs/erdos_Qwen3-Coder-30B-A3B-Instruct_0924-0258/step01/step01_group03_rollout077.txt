import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple projection: adjust values to get correct sum
    # We'll use a simple method: scale and then clip
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    # Refined projection: adjust largest values to get the exact sum
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Adjust by moving mass
        idx = np.argsort(h)
        i = 0
        while i < len(idx) and abs(diff) > 1e-10:
            j = idx[i]
            if diff > 0 and h[j] < 1:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
            elif diff < 0 and h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
            i += 1
    return h

def gradient_descent_step(h, dx, learning_rate=0.01):
    """Perform one gradient descent step."""
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    # Max correlation occurs at shift k
    # Full correlation length = 2*n - 1
    # A[0] corresponds to shift 0, A[n-1] to shift (n-1), A[2*n-2] to shift -(n-1)
    n = len(h)
    k = max_idx - (n - 1)  # shift index
    
    # Gradient computation
    grad = np.zeros_like(h)
    if k >= 0:
        # h[i] contributes to corr[i+k] term
        for i in range(len(h) - k):
            grad[i] += (1 - h[i + k])
    else:
        # h[i] contributes to corr[i-k] term
        k_abs = -k
        for i in range(k_abs, len(h)):
            grad[i] += (1 - h[i - k_abs])
    
    # Update h
    h_new = h - learning_rate * grad
    h_new = project_to_feasible(h_new, len(h) / 2)
    return h_new
