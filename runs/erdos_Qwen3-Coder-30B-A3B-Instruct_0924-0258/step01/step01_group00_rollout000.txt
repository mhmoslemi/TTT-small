import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def adaptive_gradient_descent(h_start, n_points, dx, time_limit):
    """Adaptive gradient descent with momentum and adaptive steps"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Momentum parameters
    momentum = 0.9
    velocity = np.zeros(n_points)
    learning_rate = 0.01
    
    # Adaptive learning rate
    adaptive_lr = 0.01
    
    # For tracking large overlaps
    max_hist = []
    
    for iteration in range(1000):
        if time() - start_time > time_limit:
            break
            
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        base_val = evaluate_erdos_solution(best_h, dx)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (val_plus - base_val) / eps
        
        # Add momentum
        velocity = momentum * velocity - adaptive_lr * grad
        new_h = best_h + velocity
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            adaptive_lr = min(adaptive_lr * 1.1, 0.1)  # Increase learning rate
        else:
            adaptive_lr *= 0.9  # Decrease learning rate
            
        # Random perturbations to escape local minima
        if iteration % 10 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.02, n_points)
            new_h = project_to_feasible(best_h + perturbation, n_points)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
    
    return best_h, best_c5
