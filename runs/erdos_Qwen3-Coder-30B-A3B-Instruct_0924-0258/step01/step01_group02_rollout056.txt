import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Default initial vector
        initial_h = np.full(45, 1.0 / 45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Scale initial h to meet constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def fitness(h_vec):
        return -compute_c5(h_vec)
    
    # Genetic algorithm parameters
    population_size = 50
    num_generations = 1000
    mutation_rate = 0.05
    elite_size = population_size // 4
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    best_solution = initial_h.copy()
    best_value = compute_c5(best_solution)
    
    start_time = time.time()
    
    for generation in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Evaluate fitness
        fitness_values = [fitness(ind) for ind in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_values)
        population = [population[i] for i in sorted_indices]
        fitness_values = [fitness_values[i] for i in sorted_indices]
        
        # Track best solution
        current_best_value = -fitness_values[0]
        if current_best_value < best_value:
            best_value = current_best_value
            best_solution = population[0].copy()
        
        # Create new population
        new_population = population[:elite_size]
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_values[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitness)]
            
            # Select another parent
            tournament_indices2 = np.random.choice(population_size, tournament_size)
            tournament_fitness2 = [fitness_values[i] for i in tournament_indices2]
            winner_index2 = tournament_indices2[np.argmax(tournament_fitness2)]
            
            parent1 = population[winner_index]
            parent2 = population[winner_index2]
            
            # Uniform crossover
            child1 = np.where(np.random.rand(n_points) > 0.5, parent1, parent2)
            child2 = np.where(np.random.rand(n_points) > 0.5, parent2, parent1)
            
            # Mutation applied to both children
            for child in [child1, child2]:
                # Add Gaussian noise
                noise = np.random.normal(0, mutation_rate, n_points)
                child += noise
                child = project_h(child)
                new_population.append(child)
                
                if len(new_population) >= population_size:
                    break
        
        population = new_population[:population_size]
    
    # Final check
    final_c5 = compute_c5(best_solution)
    return (best_solution.tolist(), final_c5, n_points)
