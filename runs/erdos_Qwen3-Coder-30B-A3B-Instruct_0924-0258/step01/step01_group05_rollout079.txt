import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization - use a smooth construction
        n_points = 45
        h = np.zeros(n_points)
        # Create a piecewise constant function
        mid = n_points // 2
        h[:mid] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Small step size for local search
    step_size = 0.05
    
    def compute_C5(h_vec):
        # Compute correlations for all lags properly
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto the feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_proj = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_proj)
        if h_sum > 0:
            h_proj = h_proj * target_sum / h_sum
        return h_proj
    
    def local_descent(h_vec, max_iter=50):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
            improved = False
            
            # Try coordinate-wise changes
            for i in range(n_points):
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                # Try positive and negative steps
                h_try = h_current.copy()
                h_try[i] += step_size
                h_try = project_feasible(h_try)
                C5_try = compute_C5(h_try)
                
                if C5_try < best_C5:
                    h_current = h_try.copy()
                    best_C5 = C5_try
                    improved = True
                    
                # Try negative step
                h_try2 = h_current.copy()
                h_try2[i] -= step_size
                h_try2 = project_feasible(h_try2)
                C5_try2 = compute_C5(h_try2)
                
                if C5_try2 < best_C5:
                    h_current = h_try2.copy()
                    best_C5 = C5_try2
                    improved = True
            
            # Perform swap operations
            for _ in range(10):
                if time.time() - start_time > budget_s - 0.1:
                    break
                i, j = np.random.choice(n_points, 2, replace=False)
                h_try = h_current.copy()
                h_try[i], h_try[j] = h_try[j], h_try[i]
                h_try = project_feasible(h_try)
                C5_try = compute_C5(h_try)
                
                if C5_try < best_C5:
                    h_current = h_try.copy()
                    best_C5 = C5_try
                    improved = True
            
            if not improved:
                # Reduce step size slightly if no improvement
                step_size *= 0.95
                if step_size < 1e-6:
                    break
            else:
                # Keep best solution
                best_h = h_current.copy()
                
        return best_h, best_C5
    
    # Initialize multiple candidates with LHS sampling
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Use heuristic to not get stuck
    no_improvement_count = 0
    timeout_limit = 30  # seconds of no improvement before restart
    
    while time.time() - start_time < budget_s - 0.1:
        # Create new initializations and try searches
        for _ in range(3):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Generate a varied initial solution
            h_init = np.random.rand(n_points)
            h_init = project_feasible(h_init)
            
            # Local optimization
            h_opt, C5_opt = local_descent(h_init, max_iter=30)
            
            if C5_opt < best_C5:
                best_C5 = C5_opt
                best_h = h_opt.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Occasionally restart with new candidate if stuck
        if no_improvement_count > 20:
            h_restart = np.zeros(n_points)
            h_restart[:n_points//2] = 1.0
            h_restart = project_feasible(h_restart)
            h_opt, C5_opt = local_descent(h_restart, max_iter=30)
            if C5_opt < best_C5:
                best_C5 = C5_opt
                best_h = h_opt.copy()
            no_improvement_count = 0
            
    # Final projection to ensure feasibility
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
