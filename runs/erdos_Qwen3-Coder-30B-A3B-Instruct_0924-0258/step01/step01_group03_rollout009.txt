import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Initialize
    h = h0.copy().astype(float)
    dx = 2.0 / n_points
    
    # Gradient descent with adaptive learning rate
    learning_rate = 0.01
    stagnation_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    iterations = 0
    while time.time() - start_time < budget_s:
        iterations += 1
        
        # Compute correlation for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(corr)
        
        # Compute gradient
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = (i - max_shift) % len(h)
            gradient[i] = 1 - 2 * h[j]
        
        # Update h
        h = h - learning_rate * gradient
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum) > 1e-10:
            h = h * (n_points / 2.0) / current_sum
        
        # Evaluate
        c5 = evaluate(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adapt learning rate
        if stagnation_count > 100:
            learning_rate *= 0.9
            stagnation_count = 0
            # Add small random perturbation to escape local optima
            noise = np.random.normal(0, 1e-4, len(h))
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
        
        # Prevent NaN or Inf
        if not np.all(np.isfinite(h)):
            h = np.clip(h0, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
    
    return (best_h.tolist(), float(best_c5), n_points)
