import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def solve_sdp(n):
        # For small n, use SDP approach
        h_var = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        
        constraints = [
            X >> 0,  # Positive semidefinite
            cp.diag(X) == h_var,  # X[i,i] = h[i]
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == target_sum,
            # Constraints on overlap sums
            t <= cp.sum(X[range(n), range(n)]),  # k=0 case (not used)
        ]
        
        # Add overlap constraints for k=1 to n-1
        for k in range(1, n):
            # Wrap around indices
            indices_i = list(range(n-k))
            indices_j = list(range(k, n))
            overlap_sum = cp.sum(X[indices_i, indices_j])
            constraints.append(t <= overlap_sum)
            
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iter=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * target_sum / np.sum(h_sol)
                    return h_sol
        except Exception:
            pass
        return None

    def refine_step_function(h0, n_refine):
        # Refinement using constrained optimization
        dx_refine = 2.0 / n_refine
        h_refine = np.zeros(n_refine)
        
        # Interpolate from h0
        for i in range(n_refine):
            x = i * dx_refine
            idx = int(x / dx)
            if idx >= len(h0):
                idx = len(h0) - 1
            h_refine[i] = h0[idx]
            
        # Project to valid sum
        h_refine = h_refine * target_sum / np.sum(h_refine)
        return h_refine

    # Optimize with SDP for smaller problems
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    i = 0
    while time.time() - start_time < budget_s - 1 and i < 3:
        # Initial attempt with SDP
        h_try = solve_sdp(n_points)
        if h_try is not None:
            C5_try = compute_C5(h_try)
            if C5_try < best_C5:
                best_C5 = C5_try
                best_h = h_try.copy()
        
        # Optimization loop with gradient descent
        h_vec = best_h.copy()
        learning_rate = 0.01
        momentum = 0.9
        velocity = np.zeros_like(h_vec)
        
        inner_iter = 0
        while time.time() - start_time < budget_s - 1 and inner_iter < 500:
            inner_iter += 1
            
            # Compute all overlaps to identify contributing indices
            conv = np.correlate(h_vec, 1 - h_vec, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
            top_k = np.argsort(overlaps)[-5:][::-1]
            top_k = [k for k in top_k if k > 0]  # Exclude k=0
            
            # Approximate gradient using finite differences
            gradient = np.zeros_like(h_vec)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus)
                C_plus = compute_C5(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus)
                C_minus = compute_C5(h_minus)
                
                gradient[i] = (C_plus - C_minus) / (2 * eps)
            
            # Update with momentum
            velocity = momentum * velocity - learning_rate * gradient
            h_vec = h_vec + velocity
            
            # Project
            h_vec = project_h(h_vec)
            
            # Check if improving
            current_C5 = compute_C5(h_vec)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_vec.copy()
                learning_rate *= 1.05
            else:
                learning_rate *= 0.95
                
        # If not improving, steps smaller
        if inner_iter > 100:
            n_points = min(45, n_points + 1)  # Try increasing resolution
            if n_points > len(best_h):
                h_new = refine_step_function(best_h, n_points)
                best_h = h_new
                best_C5 = compute_C5(best_h)
            
        i += 1

    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
