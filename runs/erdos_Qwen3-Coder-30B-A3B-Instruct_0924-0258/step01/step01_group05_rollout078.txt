import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def local_descent(h_vec, max_iter=50):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        no_improvement_count = 0
        step_size = 0.05
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Coordinate descent
            improved = False
            for i in range(len(h_current)):
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                original_h = h_current[i]
                best_h_i = original_h
                best_C5_i = best_C5
                
                # Try small steps
                for delta in [-step_size, step_size]:
                    h_test = h_current.copy()
                    h_test[i] = original_h + delta
                    h_test = project_feasible(h_test)
                    new_C5 = compute_C5(h_test)
                    if new_C5 < best_C5_i:
                        best_C5_i = new_C5
                        best_h_i = h_test[i]
                
                # Update if improved
                if best_h_i != original_h:
                    h_current[i] = best_h_i
                    improved = True
                    
            # If no improvement, reduce step size or check for restart
            if not improved:
                no_improvement_count += 1
                if no_improvement_count > 10:
                    step_size *= 0.5
                    no_improvement_count = 0
                    if step_size < 0.001:
                        break
            else:
                no_improvement_count = 0
                
            new_C5 = compute_C5(h_current)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_current.copy()
            
        return best_h, best_C5
    
    def generate_latin_hypercube(n, num_samples):
        samples = np.random.random((num_samples, n))
        # Sort each column
        for i in range(n):
            samples[:, i] = np.sort(samples[:, i])
        # Normalize to [0, 1] and transform
        for i in range(n):
            samples[:, i] = samples[:, i] * (i + 1) / (n + 1)
        return samples
    
    # Generate diverse initial candidates using Latin hypercube sampling
    initial_candidates = 5
    candidates = []
    
    # Add the current h as a baseline
    candidates.append(h.copy())
    
    # Add Latin hypercube samples
    lh_samples = generate_latin_hypercube(n_points, initial_candidates - 1)
    for sample in lh_samples:
        # Scale the sample to [0, 1]
        h_sample = sample / np.sum(sample) * target_sum
        h_sample = project_feasible(h_sample)
        candidates.append(h_sample.copy())
    
    # Evaluate all candidates and select the best one
    best_C5 = float('inf')
    h_best = h.copy()
    
    for candidate in candidates:
        if time.time() - start_time > budget_s - 0.1:
            break
        # Local optimization on each candidate
        refined_h, refined_C5 = local_descent(candidate, max_iter=30)
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            h_best = refined_h.copy()
    
    # Enhanced search loop
    max_search_iterations = 1000
    for iteration in range(max_search_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Try local optimization from current best
        h_try, C5_try = local_descent(h_best, max_iter=20)
        if C5_try < best_C5:
            best_C5 = C5_try
            h_best = h_try.copy()
        
        # Occasionally introduce perturbations
        if iteration % 100 == 0 and iteration > 0:
            # Add noise
            noise = np.random.normal(0, 0.03, n_points)
            h_perturbed = h_best + noise
            h_perturbed = project_feasible(h_perturbed)
            h_try, C5_try = local_descent(h_perturbed, max_iter=10)
            if C5_try < best_C5:
                best_C5 = C5_try
                h_best = h_try.copy()
    
    # Final check
    h_best = project_feasible(h_best)
    final_C5 = compute_C5(h_best)
    
    return list(h_best), final_C5, n_points
