import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 500
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 100
    alpha = 0.01
    
    iter_count = 0
    prev_c5 = float('inf')
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        
        # Try multiple resolution levels
        for res in range(3):  # Three resolution stages
            if n_points >= n_high or time.time() - start_time > budget_s * 0.95:
                break
            
            # ADMM iteration
            for _ in range(max_iter):
                if time.time() - start_time > budget_s:
                    break
                
                # Update h by gradient descent on correlation
                corr = correlate(h, 1-h, mode='full')
                max_idx = np.argmax(corr)
                # Approximate gradient using finite differences
                h_grad = np.zeros_like(h)
                for i in range(len(h)):
                    h_copy = h.copy()
                    h_copy[i] += 1e-5
                    corr_new = correlate(h_copy, 1-h_copy, mode='full')
                    h_grad[i] = (corr_new[max_idx] - corr[max_idx]) / 1e-5
                
                h = h - alpha * h_grad
                h = project_to_constraints(h, dx)
                
                # Evaluate
                current_c5 = evaluate(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
                iter_count += 1
            
            # Try to interpolate to next resolution
            if n_points < n_high:
                n_points *= 2
                dx = 2.0 / n_points
                h_long = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, n_points//2), h)
                h = h_long
                h = project_to_constraints(h, dx)
                # Reset learning rate
                alpha = 0.01
            else:
                break
        
        # Early stopping if too slow
        if abs(prev_c5 - best_c5) < 1e-6 or iter_count > 5000:
            break
        prev_c5 = best_c5

    return best_h.tolist(), best_c5, len(best_h)
