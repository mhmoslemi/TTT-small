import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
        # Ensure clipping did not violate sum
        h = np.clip(h, 0, 1)
        # Re-adjust to exactly target sum
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-10:
            # Distribute difference
            idx = np.argsort(h)
            i = 0
            while i < len(h) and diff != 0:
                j = idx[i]
                if h[j] < 1.0 and diff > 0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
                elif h[j] > 0.0 and diff < 0:
                    update = max(diff, -h[j])
                    h[j] += update
                    diff -= update
                i += 1
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def compute_gradient(h, shift_k):
    """Compute gradient w.r.t. h for given shift k."""
    n = len(h)
    grad = np.zeros(n)
    # For shift k, h[i] contributes to h[i-k] term in correlation
    # So we're looking at the derivative of sum of h[i] * (1 - h[i+k])
    if shift_k >= 0:
        grad[shift_k:] = -h[:-shift_k] if shift_k < n else 0
    else:
        grad[:shift_k] = -h[-shift_k:] if -shift_k < n else 0
    return grad

    # Initialize h
    h = h0.copy()
    h = project_to_feasible(h, n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate(h, dx)

    # Parameters
    lr = 0.01
    max_iter = 1000
    tol = 1e-6
    stagnation_limit = 50
    
    elapsed = 0
    iteration = 0
    stagnation_counter = 0

        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        # Find the shift which produces maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(corr)
        
        # Compute gradient for this shift
        grad = compute_gradient(h, max_shift - (n_points - 1))
        
        # Update h using gradient descent
        h = h - lr * grad
        
        # Project to feasible region
        h = project_to_feasible(h, n_points / 2)

        if stagnation_counter > 10:
