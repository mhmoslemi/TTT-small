import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    """Gradient of smooth max"""
    exp_x = np.exp(alpha * x)
    return exp_x / (np.sum(exp_x) + 1e-12) * alpha

def optimize_with_perturbations(h_start, n_points, dx, time_limit):
    """Optimize using gradient-based method with multiple restarts"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use the objective function that tries to directly minimize C5
    def obj(h):
        # Convert to feasible region
        h_feasible = project_to_feasible(h, n_points)
        c5 = evaluate_erdos_solution(h_feasible, dx)
        return c5
    
    # Simple gradient-based optimization with bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    restarts = 5
    for _ in range(restarts):
        if time() - start_time > time_limit:
            break
        
        # Start with slightly perturbed version
        h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.01, n_points), n_points)
        
        try:
            res = minimize(obj, h_perturbed, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
        
        # Further perturb for next restart
        h_start = project_to_feasible(best_h + np.random.normal(0, 0.02, n_points), n_points)
    
    return best_h, best_c5

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Try a few resolutions with different patterns
    resolutions = [20, 30, 40, 50]
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary  
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time allocation based on remaining budget
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        h_opt, c5_opt = optimize_with_perturbations(h_start, n_points, dx, time_per_res / 2)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
            
        h_start = h_opt.copy()
    
    return (list(best_h), best_c5, best_n_points)
