import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations using full mode for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm parameters
    pop_size = 50
    tournament_size = 10
    elite_size = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    max_generations = 1000
    patience = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_rand = np.random.rand(n_points)
        h_rand = project_feasible(h_rand)
        population.append(h_rand)
    
    # Evaluate initial population
    fitness = [compute_C5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_C5 = fitness[best_idx]
    
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 0.1 and generation < max_generations:
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_C5:
            best_C5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= patience:
            # Reset worst individual with random feasible solution
            population[-1] = np.random.rand(n_points)
            population[-1] = project_feasible(population[-1])
            fitness[-1] = compute_C5(population[-1])
            no_improvement = 0
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.choice(tournament_size, size=1, replace=False)[0]
            parent2_idx = np.random.choice(tournament_size, size=1, replace=False)[0]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Copy one parent
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_feasible(child)
            
            # If not mutated, try reset mutation
            if np.random.rand() < mutation_rate * 0.5:
                # Reset some genes
                reset_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                child[reset_indices] = np.random.choice([0, 1], size=len(reset_indices))
                child = project_feasible(child)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
        fitness = [compute_C5(h) for h in population]
        generation += 1
        
        # Update best if needed
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_C5:
            best_C5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
