import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def project_to_feasible_with_grad(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def adaptive_hill_climb(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different strategies
    strategies = [
        (0.05, 10),  # (step_size, iterations)
        (0.02, 50),
        (0.01, 100)
    ]
    
    for step_size, iter_count in strategies:
        if time() - start_time > time_limit * 0.95:
            break
            
        current_h = best_h.copy()
        for _ in range(iter_count):
            if time() - start_time > time_limit:
                break
                
            # Add noise to current solution
            noise = np.random.normal(0, step_size, n_points)
            h_trial = current_h + noise
            
            # Project to feasible set
            h_trial = project_to_feasible(h_trial, n_points)
            
            # Evaluate trial
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                current_h = h_trial.copy()
            else:
                # Accept with probability based on temperature-like scheme
                temp = 1.0 - (time() - start_time) / time_limit
                if temp > 0 and np.random.random() < np.exp(-10 * (c5_trial - best_c5) / (best_c5 + 0.001)):
                    current_h = h_trial.copy()
    
    return best_h, best_c5

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing schedule parameters
    max_steps = 2000
    cooling_rate = 0.995
    
    step_size = 0.05
    temp = 1.0
    
    for step in range(max_steps):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution
        h_trial = best_h + np.random.normal(0, step_size, n_points)
        h_trial = project_to_feasible(h_trial, n_points)
        
        # Evaluate candidate
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        
        # Accept or reject based on metropolis criterion
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
        else:
            # Accept with probability based on temperature
            delta = c5_trial - best_c5
            if np.random.random() < np.exp(-delta / (best_c5 * temp + 1e-8)):
                best_h = h_trial.copy()
                best_c5 = c5_trial
                
        # Update temperature and step size
        temp = max(0.01, temp * cooling_rate)
        step_size = max(0.001, step_size * cooling_rate)
        
        # Occasionally fine-tune with local search
        if step % 50 == 0:

                best_c5 = local_c5
