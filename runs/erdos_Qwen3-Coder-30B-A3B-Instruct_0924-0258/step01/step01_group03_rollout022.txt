import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize h0 to desired sum
    target_sum = n_points / 2.0
    current_sum = np.sum(h0)
    h0 = h0 * target_sum / current_sum
    h0 = np.clip(h0, 0, 1)
    
    # Further normalize after clipping
    current_sum = np.sum(h0)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        idx = np.argsort(h0)
        for i in range(len(idx)-1, -1, -1):
            if diff == 0:
                break
            j = idx[i]
            if h0[j] < 1.0 and diff > 0:
                update = min(diff, 1.0 - h0[j])
                h0[j] += update
                diff -= update
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate(best_h, dx)
    
    # Initialize learning rate and parameters
    learning_rate = 0.01
    patience = 0
    patience_limit = 50
    decay_factor = 0.5
    max_iter = 1000
    iteration = 0
    
    # Gradient descent with convex relaxation
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        
        # Step 1: Compute current cross-correlation matrix to find shift that maximizes overlap
        corr_full = correlate(best_h, 1 - best_h, mode="full")
        max_shift_idx = np.argmax(corr_full)
        # Convert full conv index to actual shift value
        max_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient w.r.t. shift that maximizes overlap
        gradient = np.zeros(n_points)
        # Note: if max_shift <= 0, we compute shift indexes as (i + max_shift) or (i - abs(max_shift))
        if max_shift <= 0:
            # h[i] * (1 - h[i + max_shift]) terms: the gradient contribution is -(1 - h[i + max_shift])
            for i in range(n_points + max_shift):  # i ranges to ensure i + max_shift < n_points
                gradient[i] -= (1 - best_h[i - max_shift])
        else:
            # max_shift > 0, so i + max_shift out of bounds for some i
            for i in range(n_points - max_shift):
                gradient[i] -= (1 - best_h[i + max_shift])
        
        # Update h using gradient ascent (minimize C₅, so we descend -gradient)
        best_h = best_h - learning_rate * gradient
        
        # Projection to feasibility region [0,1] and sum = target_sum
        best_h = np.clip(best_h, 0, 1)
        current_sum = np.sum(best_h)
        if abs(current_sum - target_sum) > 1e-10:
            # Distribute correction
            diff = target_sum - current_sum
            if diff > 0:
                # Need to increase h
                idx = np.argsort(best_h)
                for i in range(len(idx)-1, -1, -1):
                    if diff <= 0:
                        break
                    j = idx[i]
                    if best_h[j] < 1.0:
                        update = min(diff, 1.0 - best_h[j])
                        best_h[j] += update
                        diff -= update
            else:
                # Need to reduce h
                idx = np.argsort(best_h)[::-1]  # descending order
                for i in range(len(idx)):
                    if diff >= 0:
                        break
                    j = idx[i]
                    if best_h[j] > 0:
                        update = max(diff, -best_h[j])
                        best_h[j] += update
                        diff -= update
                        
        # Evaluate and update best solution
        current_c5 = evaluate(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            if patience >= patience_limit:
                learning_rate *= decay_factor
                patience = 0
    
    return (best_h.tolist(), float(best_c5), n_points)
