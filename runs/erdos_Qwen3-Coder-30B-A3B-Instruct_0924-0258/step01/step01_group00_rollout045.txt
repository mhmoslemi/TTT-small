import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 30
        h = np.ones(n_points) / 2
    
    h = project_to_feasible(h, n_points)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters for greedy adjustment
    target_c5 = 0.3807
    adjustment_fraction = 0.05
    max_no_improvement = 50
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95:
        c5 = evaluate_erdos_solution(best_h, dx)
        if c5 <= target_c5:
            break
            
        # Compute correlations to find the maximum overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        
        # Shift index maps to actual shift value in [0, 2]
        shift = (max_corr_idx - (n_points - 1)) * dx
        
        # Find the positions around the worst overlap
        # This is a bit tricky because we are using correlation
        # The overlap we care about is for shift k = shift
        # So we look for h_i * (1 - h_{i+k}) where k = shift
        
        # Compute contribution of each pair to max overlap
        contributions = np.zeros(n_points)
        k_idx = int(shift / dx)
        for i in range(n_points):
            j = i + k_idx
            if 0 <= j < n_points:
                contributions[i] = best_h[i] * (1 - best_h[j])
        
        # Find top contributors
        top_contributors = np.argsort(contributions)[-max(1, n_points // 10):][::-1]
        
        # Adjust h values
        h_new = best_h.copy()
        adjustment = (c5 - target_c5) * adjustment_fraction / n_points
        
        for i in top_contributors:
            h_new[i] = max(0, h_new[i] - adjustment)
        
        # Normalize to maintenance integral constraint 
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
            # Perturb if no improvement for a while
            if no_improvement_count >= max_no_improvement:
                h_perturbed = project_to_feasible(
                    best_h + np.random.normal(0, 0.01, n_points), 
                    n_points
                )
                perturbed_c5 = evaluate_erdos_solution(h_perturbed, dx)
                if perturbed_c5 < best_c5:
                    best_h = h_perturbed
                    best_c5 = perturbed_c5
                no_improvement_count = 0
    
    return (list(best_h), best_c5, n_points)
