import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def smooth_max(x, beta=10.0):
    # Smooth approximation to max using log-sum-exp
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

def compute_correlation(h, dx):
    # Compute correlation of h with (1-h) over all lags
    corr = correlate(h, 1-h, mode='full')
    return corr * dx

def evaluate(h, dx):
    corr = compute_correlation(h, dx)
    return np.max(corr)

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx):
    # Gradient w.r.t. max correlation
    corr = compute_correlation(h, dx)
    k_max = np.argmax(corr)
    # Manual gradients of correlation with respect to h
    # More efficient to compute using FFT
    n = len(h)
    # Will use it to compute the gradient through d/dh (corr(h,1-h))
    # This is difficult to compute in closed form for max, so we'll use finite differences
    eps = 1e-6
    grad = np.zeros_like(h)
    for i in range(n):
        # Perturb h[i]
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = project_to_constraints(h_plus, dx)
        h_minus = project_to_constraints(h_minus, dx)
        
        corr_plus = compute_correlation(h_plus, dx)
        corr_minus = compute_correlation(h_minus, dx)
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    return grad

def backtracking_line_search(f, grad, h, dx, alpha=1.0, rho=0.5, c=1e-4):
    # Backtracking line search for step size
    while True:
        h_new = h - alpha * grad
        h_new = project_to_constraints(h_new, dx)
        f_new = evaluate(h_new, dx)
        f_old = evaluate(h, dx)
        if f_new <= f_old - c * alpha * np.dot(grad, grad):
            return alpha
        alpha *= rho

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start at resolution 50
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Parameters
    max_iter = 1000
    learning_rate = 1e-3
    n_refine = 0
    last_c5 = best_c5
    
    iter_count = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute gradient
        try:
            grad = compute_gradient(h, dx)
        except:
            # Fall back to finite differences for robustness
            eps = 1e-5
            grad = np.zeros_like(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = project_to_constraints(h_plus, dx)
                f_plus = evaluate(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = project_to_constraints(h_minus, dx)
                f_minus = evaluate(h_minus, dx)
                
                grad[j] = (f_plus - f_minus) / (2 * eps)
        # Update step
        alpha = learning_rate
        try:
            alpha = backtracking_line_search(last_c5, grad, h, dx, alpha=1.0)
        except:
            pass
        h_new = h - alpha * grad
        h_new = project_to_constraints(h_new, dx)
        
        c5_new = evaluate(h_new, dx)
        if c5_new < last_c5:
            h = h_new
            last_c5 = c5_new
            if last_c5 < best_c5:
                best_h = h.copy()
                best_c5 = last_c5
                
        iter_count += 1
        if iter_count % 100 == 0:
            # Optionally refine resolution
            if n_points < 200:
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), h)
                h = h_long
                n_points = 2 * n_points
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                last_c5 = evaluate(h, dx)
    
    return best_h.tolist(), best_c5, len(best_h)
