from scipy.optimize import differential_evolution, minimize
import time

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Use differential evolution for global optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Helper to compute the objective function to be minimized
    def objective(h_vec):
        h_vec = project_h(h_vec)
        return compute_C5(h_vec)
    
    # Initial best solution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Run differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=100, 
        popsize=15, 
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        callback=lambda x, info: None,
        disp=False
    )
    
    # Extract the best solution from DE
    final_h = project_h(result.x)
    final_C5 = compute_C5(final_h)
    
    # Fine-tune with Nelder-Mead if improvement is found
    if final_C5 < best_C5:
        best_h = final_h
        best_C5 = final_C5
        
        # Local refinement with Nelder-Mead
        def local_objective(h_vec):
            h_vec = project_h(h_vec)
            return compute_C5(h_vec)
            
        # Start from best solution found so far
        result_nm = minimize(
            local_objective, 
            best_h, 
            method='Nelder-Mead',

        )
        
        refined_h = project_h(result_nm.x)
        refined_C5 = compute_C5(refined_h)
