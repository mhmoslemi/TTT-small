import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def nelder_mead_improve(h_start):
        # Use Nelder-Mead on a small neighborhood to refine solution
        def obj(h_new):
            return compute_C5(project_h(h_new))
        
        res = optimize.minimize(obj, h_start, method='Nelder-Mead', 
                               options={'maxiter': 300, 'disp': False})
        return project_h(res.x)
    
    # GA parameters
    pop_size = 50
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    tournament_size = 5
    local_search_freq = 50
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(project_h(initial_h_values))
        for _ in range(pop_size - 1):
            h_new = np.random.rand(n_points) * 0.2 + 0.4  # values around 0.5
            h_new = project_h(h_new)
            population.append(h_new)
    else:
        for _ in range(pop_size):
            h_new = np.random.rand(n_points) * 0.2 + 0.4  # values around 0.5
            h_new = project_h(h_new)
            population.append(h_new)
    
    best_h = None
    best_C5 = float('inf')
    no_improvement = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h_ind in population:
            C5 = compute_C5(h_ind)
            fitness_scores.append(C5)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores.sort()
        
        # Update best solution
        if fitness_scores[0] < best_C5:
            best_C5 = fitness_scores[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early stopping
        if no_improvement > 100:
            break
            
        # Create new population
        new_pop = []
        
        # Elitism
        for i in range(elite_size):
            new_pop.append(population[i].copy())
            
        # Tournament selection + crossover + mutation
        while len(new_pop) < pop_size:
            # Tournament selection
            parents = []
            for _ in range(2):
                tournament = np.random.choice(len(population), tournament_size)
                winner = tournament[np.argmin([fitness_scores[i] for i in tournament])]
                parents.append(population[winner].copy())
                
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parents[0][:crossover_point], parents[1][crossover_point:]])
                child2 = np.concatenate([parents[1][:crossover_point], parents[0][crossover_point:]])
            else:
                child1, child2 = parents[0].copy(), parents[1].copy()
                
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.01)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.01)
                    
            new_pop.append(project_h(child1))
            if len(new_pop) < pop_size:
                new_pop.append(project_h(child2))
                
        population = new_pop
        
        # Local search every 'local_search_freq' generations
        if gen % local_search_freq == 0 and gen > 0:
            top_candidates = population[:5]
            improved_candidates = []
            for h_candidate in top_candidates:
                h_improved = nelder_mead_improve(h_candidate)
                improved_candidates.append(h_improved)
            # Replace the top with improved ones
            population[:5] = improved_candidates
            
    # Final improvement with local search
    if best_h is not None:
        best_h = nelder_mead_improve(best_h)
        best_C5 = compute_C5(best_h)
        
    # Ensure valid result
    if best_h is None:
        best_h = h
        best_C5 = compute_C5(best_h)
        
    return (best_h.tolist(), best_C5, n_points)
