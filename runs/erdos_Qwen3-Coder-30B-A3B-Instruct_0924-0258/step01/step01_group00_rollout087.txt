import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point

    else:

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Try different resolution levels
    resolutions = [15, 25, 35, 45, 60, 75]
    time_per_res = (budget_s * 0.9) / len(resolutions)
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)),

            h_start = project_to_feasible(h_interp, n_points)
        
        h_current = h_start.copy()
        
        # Multiple restarts with different perturbations
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slightly perturb the current solution
            h_perturbed = project_to_feasible(
                h_current + np.random.normal(0, 0.05, n_points), n_points)
            
            # Optimization with bounds via SLSQP
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

                res = minimize(obj, h_perturbed, method='SLSQP',
                               bounds=bounds, constraints=cons,

                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
                        
            except:
                pass
            
            # Adaptive perturbation for next restart
            h_current = project_to_feasible(
                best_h + np.random.normal(0, 0.02, n_points), n_points)
        
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
