import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm Parameters
    pop_size = 50
    tournament_size = 10
    keep_parents = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        # Create random individual
        h_ind = np.random.rand(n_points)
        h_ind = project_feasible(h_ind)
        population.append(h_ind)
    
    # Evaluate initial population
    fitness = [compute_C5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_C5 = fitness[best_idx]
    
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 0.1 and generation < max_generations:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_C5:
            best_C5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > patience:
            # Reset some individuals
            for i in range(pop_size // 5):
                if np.random.rand() < 0.5:
                    population[-(i+1)] = np.random.rand(n_points)
                    population[-(i+1)] = project_feasible(population[-(i+1)])
            no_improvement = 0
        
        # Selection - Tournament
        new_population = []
        # Keep best individuals
        for i in range(keep_parents):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Clone winner
            parent1 = population[winner_idx].copy()
            
            # Maybe crossover
            if np.random.rand() < crossover_rate:
                # Select second parent
                tournament_indices2 = np.random.choice(pop_size, tournament_size, replace=False)
                tournament_fitness2 = [fitness[i] for i in tournament_indices2]
                winner_idx2 = tournament_indices2[np.argmin(tournament_fitness2)]
                parent2 = population[winner_idx2].copy()
                
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                for i in mutate_indices:
                    if np.random.rand() < 0.5:
                        child[i] = np.random.rand()
                    else:
                        child[i] = 1.0 if child[i] < 0.5 else 0.0
                
            # Apply feasibility
            child = project_feasible(child)
            new_population.append(child)
            
        population = new_population[:pop_size]
        fitness = [compute_C5(h) for h in population]
    
    # Final check
    best_h = project_feasible(best_h)
    best_C5 = compute_C5(best_h)
    
    return list(best_h), best_C5, n_points
