```python
import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 10
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    max_iter = 10000
    learning_rate = 0.01
    patience = 500
    patience_counter = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        
        # Create perturbed version
        h_new = h + np.random.normal(0, 0.01, n_points)
        h_new = project_to_constraints(h_new, dx)
        
        # Evaluate
        c5_new = evaluate(h_new, dx)
        
        # Accept with probability based on improvement
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            patience_counter = 0
            last_improvement = iter_count
        else:
            patience_counter += 1
            if patience_counter > patience and iter_count - last_improvement > 1000:
                # Reduce learning rate
                learning_rate *= 0.5
                patience_counter = 0
        
        # Occasionally interpolate to higher resolution
        if iter_count % 1000 == 0 and n_points < 200:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_improvement = iter_count
            best_c5 = evaluate(h, dx)
            best_h = h.copy()
            patience_counter = 0
    
    return best_h.tolist(), best_c5, len(best_h)
```
