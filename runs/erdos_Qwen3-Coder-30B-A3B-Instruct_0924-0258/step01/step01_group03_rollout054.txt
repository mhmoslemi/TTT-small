import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Clamp again and re-normalize
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Distribute difference
        diff = target_sum - current_sum
        idx = np.argsort(h)
        i = 0
        while diff != 0 and i < len(h):
            j = idx[i]
            update = min(diff, 1.0 - h[j])
            if update <= 0:
                i += 1
                continue
            h[j] += update
            diff -= update
            i += 1
    return h

def compute_gradient(h, k_max, dx):
    """Compute gradient for given shift index k_max"""
    n = len(h)
    grad = np.zeros_like(h)
    for i in range(n):
        if i + k_max < n:
            grad[i] += 1.0  # Contribution to correlation at shift k_max
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Initialize h and dx
    h = h0.copy().astype(float)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    h = project_to_feasible(h, target_sum)
    
    # Parameters
    learning_rate = 0.01
    max_iter = 5000
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    # Main optimization loop
    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            # Reset with new random perturbation if stuck
            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), target_sum)
            patience_counter = 0
            continue
        
        # Find shift index that maximizes overlap
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = compute_gradient(h, k_max, dx)
        
        # Update h
        h -= learning_rate * grad
        
        # Project to feasible set
        h = project_to_feasible(h, target_sum)
        
        # Adaptive learning rate
        if it % 100 == 0:
            learning_rate *= 0.95
    
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        final_h = project_to_feasible(h, target_sum)
        final_c5 = evaluate(final_h, dx)
        return (final_h.tolist(), float(final_c5), n_points)
