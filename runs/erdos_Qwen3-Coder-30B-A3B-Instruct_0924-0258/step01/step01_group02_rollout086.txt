import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Use adaptive Adam-like optimization with gradient projection
    # State for momentum
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.01
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Gradient computation: derivative of correlation
        # For shift k_max, gradient contribution at i:
        # ∇Cₖ = 1 - 2 * h[i] for the i term in convolution
        # But this needs a more precise form
        # Let's compute gradient directly for the current k_max
        
        # Compute gradient numerically with finite differences
        # Safeguard against numerical errors
        eps = 1e-5
        grad = np.zeros(len(h))
        
        for i in range(len(h)):
            h_test_plus = h.copy()
            h_test_plus[i] += eps
            h_test_plus = project_h(h_test_plus)
            
            corr_plus = np.correlate(h_test_plus, 1 - h_test_plus, mode='full')
            c5_plus = corr_plus[k_max] * dx
            
            h_test_minus = h.copy()
            h_test_minus[i] -= eps
            h_test_minus = project_h(h_test_minus)
            
            corr_minus = np.correlate(h_test_minus, 1 - h_test_minus, mode='full')
            c5_minus = corr_minus[k_max] * dx
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update parameters with Adam-like scheme
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        
        # Update h
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Anneal learning rate
        if iteration % 50 == 0 and iteration > 0:
            lr *= 0.995
            
        # Occasionally add noise to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01, len(h))
            h = h + noise
            h = project_h(h)
            
        iteration += 1
        
        # Early termination
        if no_improvement_count > max_no_improvement:
            break
        
        if iteration > 10000:
            break

    return (best_h.tolist(), best_c5, n_points)
