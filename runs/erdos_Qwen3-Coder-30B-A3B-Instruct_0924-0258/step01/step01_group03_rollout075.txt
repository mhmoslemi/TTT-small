import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust values to meet sum without violating bounds
    if current_sum > target_sum:
        # Need to reduce
        diff = current_sum - target_sum
        # Reduce from largest values
        idxs = np.argsort(h)[::-1]
        for idx in idxs:
            if diff <= 0:
                break
            dec = min(diff, h[idx])
            h[idx] -= dec
            diff -= dec
    else:
        # Need to increase
        diff = target_sum - current_sum
        # Increase from smallest values
        idxs = np.argsort(h)
        for idx in idxs:
            if diff <= 0:
                break
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5  # Simple uniform start
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    lr = 0.01
    max_iter = 10000
    patience = 100
    no_improve_count = 0
    prev_best = best_c5
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Evaluate overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        # Correlation length is 2*n_points - 1
        # The maximum overlap happens at the shift with highest value
        max_k = np.argmax(corr)
        # The shift k ranges from 0 to 2*n_points - 2
        # We map max_k to actual shift in [0, n_points-1]
        # Since we're using full mode correlation, indices 0 to n_points-1 correspond to shifts 0 to n_points-1
        # and indices n_points to 2*n_points-2 correspond to shifts -n_points+1 to -1
        
        # Shifts are ordered: [0, n_points-1, -n_points+1, ..., -1]
        # So shift index for max_k is:
        if max_k < n_points:
            shift_idx = max_k
        else:
            shift_idx = max_k - 2 * n_points + 2  # Actually - (2*n_points - 2 - max_k)
            
        # Now compute gradient
        grad = np.zeros_like(h)
        
        # For shift_idx, compute gradient contribution
        if shift_idx >= 0:
            # h(i) contributes to overlap if (i+shift_idx) is valid
            # Overlap = sum over i of h[i] * (1 - h[i+shift_idx])
            # So the gradient w.r.t. h[i] is (1 - h[i+shift_idx]) 
            if shift_idx < n_points:
                grad[:n_points-shift_idx] += (1 - h[shift_idx:])
            else:
                # out of bounds
                pass
        else:
            # shift_idx < 0
            # h[i] contributes to overlap if (i-shift_idx) is valid
            # shift_idx = -k, k > 0
            # Overlap = sum of h[i] * (1 - h[i-k])
            # grad w.r.t. h[i] is (1 - h[i-k]) 
            if shift_idx + n_points > 0:
                grad[-shift_idx:] += (1 - h[:-shift_idx])
            else:
                pass

        # Update step
        h_new = h - lr * grad
        h_new = project_to_feasible(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        h = h_new
        
        # Adjust learning rate if needed
        if no_improve_count > 20:
            lr *= 0.9
        
        if no_improve_count > patience:
            # Introduce random perturbation to escape local minima
            h = h + np.random.normal(0, 0.001, n_points)
            h = project_to_feasible(h, target_sum)
            no_improve_count = 0
            
    return (best_h.tolist(), float(best_c5), n_points)
