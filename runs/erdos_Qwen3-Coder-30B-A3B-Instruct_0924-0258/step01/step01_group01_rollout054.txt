import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_descent_step(h, dx, learning_rate=0.01):
    n = len(h)
    # Compute gradient: dC5/dh_i = (1 - 2*h_i) * dx
    grad = (1 - 2 * h) * dx
    h_new = h - learning_rate * grad
    return project_to_constraints(h_new, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    resolutions = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Initialize
        if 'initial_h_values' in globals() and n_points == 10:
            h = np.array(initial_h_values)
            h = h[:n_points]
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_constraints(h, dx)
            
        # Optimize at current resolution
        max_iter = int(150 / (n_points / 10))  # Scale iterations with resolution
        iter_count = 0
        current_temp = 0.1
        
        while iter_count < max_iter:
            if time.time() - start_time > budget_s:
                break
                
            # Gradient descent step
            h_new = gradient_descent_step(h, dx, 0.01)
            
            # Evaluate
            c5_new = evaluate(h_new, dx)
            iter_count += 1
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            h = h_new
            
        # Simple annealing if not at final resolution
        if n_points < 70:
            # Interpolate to next resolution
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
    
    if best_h is None:
        # Fallback to initial construction
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            best_h = h
            best_c5 = evaluate(best_h, 2.0 / len(best_h))
        else:
            h = np.random.uniform(0, 1, 10)
            h = project_to_constraints(h, 0.2)
            best_h = h
            best_c5 = evaluate(best_h, 0.2)
    
    return best_h.tolist(), best_c5, len(best_h)
