import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale and re-clip
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize to target sum
    target_sum = n_points / 2.0
    h = project_h(h0, target_sum)
    dx = 2.0 / n_points
    
    # Optimization parameters
    max_iter = 10000
    lr = 0.01
    patience = 200
    best_c5 = float('inf')
    best_h = None
    patience_counter = 0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Gradient of the maximum term w.r.t. h
        # For shift k, gradient of term h[i]*(1-h[i+k]) is (1-h[i+k]) for i, -h[i-k] for i+k
        # But we compute it as part of the derivative of max_k term
        # For simplicity, we compute exact gradients from all pairs
        grad = np.zeros_like(h)
        # The maximum is at shift max_k, which means the real shift is offset = max_k - (n_points - 1)
        # For h[i], it's paired with h[i + offset] in 1-h
        offset = max_k - (n_points - 1)
        for i in range(n_points):
            j = i + offset
            if 0 <= j < n_points:
                grad[i] += -(1 - h[j])  # This is the gradient of h[i]*(1-h[j])
            # Also: h[j] contributes, add gradient of -h[i]*h[j] = -h[i]
            j2 = i - offset
            if 0 <= j2 < n_points:
                grad[i] += -h[j2]
                
        # Simple gradient ascent update (but the actual objective is -max_corr)
        # So we minimize -max_corr with respect to h, which means
        # grad = -d/dh(max_corr) = -grad_original
        # gradient ascent step on -max_corr is gradient descent on max_corr
        # However we are trying to minimize max_corr, so we have -grad
        # Need to derive correctly ...
        # Actually, consider it as we want minimize the max overlap, so minimize max_k (h[i](1-h[i+k]))
        # We're minimizing C5 = max_k corr_k = max_k sum_i h[i](1-h[i+k])
        # But the max_k is fixed for this iteration
        # The gradient of h[i] w.r.t. corr_k is (1 - h[i + k]) - h[i - k] where valid
        # Let's compute Gradients more carefully:
        # The overlap at index k is sum_{i=0}^{n-k-1} h[i] * (1 - h[i + k]) = sum h[i] - sum h[i]*h[i+k]
        # So gradient = - h[i + k] for h[i] (one term) and - h[i - k] for h[i + k] (one term)
        # Corner case handling:
        grad = np.zeros_like(h)
        for i in range(n_points):
            # h[i] appears in the correlation with h[i + k]
            j = i + offset
            if 0 <= j < n_points:
                grad[i] -= (1 - h[j])
            # h[i] also appears in the correlation with (i - k)
            j2 = i - offset
            if 0 <= j2 < n_points:
                grad[i] -= h[j2]
        
        # Update step
        h = h - lr * grad
        
        # Project back to constraints
        h = project_h(h, target_sum)
        
        # Evaluate and track best
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                # Restart with a small perturbation
                h = project_h(h + np.random.normal(0, 0.001, n_points), target_sum)
                patience_counter = 0

    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        final_h = project_h(h, target_sum)
        final_c5 = evaluate(final_h, dx)
        return (final_h.tolist(), float(final_c5), n_points)
