import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h_init = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h_init satisfies the constraint
    h_init = h_init * target_sum / np.sum(h_init)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def evaluate_population(population):
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(-c5)  # negative for maximization
        return np.array(fitness)
    
    def tournament_selection(population, fitness, k=3):
        selected = []
        for _ in range(len(population) // 2):
            idxs = np.random.choice(len(population), k)
            best_idx = idxs[np.argmax(fitness[idxs])]
            selected.append(population[best_idx].copy())
        return selected
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.random(len(parent1)) > 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        return project_h(child1), project_h(child2)
    
    def mutate(h_vec, mutation_rate=0.05):
        # Add Gaussian noise
        noise = np.random.normal(0, 0.05, len(h_vec))
        mutated = h_vec + noise
        return project_h(mutated)
    
    # Initialize population
    population_size = 75
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = evaluate_population(population)
    best_idx = np.argmax(fitness)
    best_c5 = -fitness[best_idx]
    best_h = population[best_idx].copy()
    
    start_time = time.time()
    gen = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Create new population through crossover and mutation
        new_population = [best_h.copy()]  # Elitism
        
        while len(new_population) < population_size:
            if len(selected) < 2:
                break
            
            parent1_idx = np.random.randint(len(selected))
            parent2_idx = np.random.randint(len(selected))
            
            child1, child2 = crossover(selected[parent1_idx], selected[parent2_idx])
            
            # Mutate children
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Evaluate new population
        population = new_population
        fitness = evaluate_population(population)
        
        # Update best solution
        curr_best_idx = np.argmax(fitness)
        curr_best_c5 = -fitness[curr_best_idx]
        
        if curr_best_c5 < best_c5:
            best_c5 = curr_best_c5
            best_h = population[curr_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        gen += 1
        if no_improvement_count > max_no_improvement:
            break
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
