import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec

    # Grid search approach with refinements
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Grid sizes to try
    grid_sizes = [5, 10, 15, 20, 25, 30, 45]
    grid_size_idx = 0
    
    # For each grid size, we generate and refine candidates
    while time.time() - start_time < budget_s - 0.1 and grid_size_idx < len(grid_sizes):
        current_n = grid_sizes[grid_size_idx]
        if current_n == n_points:
            # Already at target resolution, skip
            grid_size_idx += 1
            continue
            
        dx_current = 2.0 / current_n
        current_h = np.zeros(current_n)
        
        # Generate candidates for this grid
        n_candidates = 20 if current_n < 30 else 10
        candidates = []
        
        # Start with the best solution
        if current_n <= n_points:
            # Downsample the best solution
            h_downsampled = np.array([np.mean(h[i*2:(i+1)*2]) for i in range(current_n)])
            candidates.append(h_downsampled)
        
        # Add random feasible candidates
        for _ in range(n_candidates - len(candidates)):
            candidate = np.random.rand(current_n)
            candidate = project_feasible(candidate)
            candidates.append(candidate)
            
        # Evaluate all candidates
        best_in_level = None
        best_C5_in_level = float('inf')
        
        for candidate in candidates:
            C5_val = compute_C5(candidate)
            if C5_val < best_C5_in_level:
                best_C5_in_level = C5_val
                best_in_level = candidate.copy()
                
        if best_C5_in_level < best_C5:
            best_C5 = best_C5_in_level
            best_h = best_in_level.copy()
        
        # Refine: interpolating from this level to next finer level
        if grid_size_idx < len(grid_sizes) - 1:
            next_n = grid_sizes[grid_size_idx + 1]
            if next_n > current_n:
                refine_factor = next_n // current_n
                refined_candidates = []
                
                # Interpolate best solution to next level
                for _ in range(5):
                    h_refined = np.interp(
                        np.linspace(0, 2, next_n),
                        np.linspace(0, 2, current_n),
                        best_in_level
                    )
                    h_refined = project_feasible(h_refined)
                    # Add small random perturbations
                    h_refined += np.random.normal(0, 0.02, next_n)
                    h_refined = project_feasible(h_refined)
                    refined_candidates.append(h_refined)
                    
                # Evaluate refined candidates
                for candidate in refined_candidates:
                    C5_val = compute_C5(candidate)
                    if C5_val < best_C5:
                        best_C5 = C5_val
                        best_h = candidate.copy()
                        
        grid_size_idx += 1

    # Final optimization on the best solution
    refined_h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Simple gradient ascent: move towards reducing maximum correlation
        corr = correlate(refined_h, 1 - refined_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(refined_h) - 1)
        
        # Approximate gradient direction for this lag
        grad = np.zeros_like(refined_h)
        if max_lag >= 0:
            grad[max_lag:] = -refined_h[:-max_lag]
        else:
            grad[:max_lag] = -refined_h[-max_lag:]
            
        # Move along gradient
        alpha = 0.05
        new_h = refined_h - alpha * grad
        new_h = project_feasible(new_h)
        
        new_C5 = compute_C5(new_h)
        if new_C5 < best_C5:
            best_C5 = new_C5
            refined_h = new_h.copy()
        else:
            # Retry with smaller step
            alpha *= 0.5
            if alpha < 1e-6:
                break
            new_h = refined_h - alpha * grad
            new_h = project_feasible(new_h)
            if compute_C5(new_h) < best_C5:
                best_C5 = compute_C5(new_h)
                refined_h = new_h.copy()
                
    # Final best solution
    final_h = project_feasible(refined_h)
    final_C5 = compute_C5(final_h)

    return list(final_h), final_C5, len(final_h)
