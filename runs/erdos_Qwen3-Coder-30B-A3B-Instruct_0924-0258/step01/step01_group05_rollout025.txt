import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags using full mode
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def generate_initial_candidates(n_candidates, n_points):
        candidates = []
        for _ in range(n_candidates):
            # Latin hypercube sampling approach - distributed points
            h = np.random.random(n_points)
            h = project_feasible(h)
            candidates.append(h.copy())
        return candidates
    
    def coordinate_descent_step(h_vec, step_size=0.05, max_iter=20):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        improved = True
        iter_count = 0
        
        while improved and iter_count < max_iter and time.time() - start_time < budget_s - 0.1:
            improved = False
            iter_count += 1
            
            # Try each coordinate
            for i in range(len(h_current)):
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                # Try positive and negative steps
                steps = [step_size, -step_size]
                best_h = h_current.copy()
                best_C5_local = best_C5
                
                for delta in steps:
                    h_test = h_current.copy()
                    h_test[i] += delta
                    # Project
                    h_test = project_feasible(h_test)
                    new_C5 = compute_C5(h_test)
                    
                    if new_C5 < best_C5_local:
                        best_C5_local = new_C5
                        best_h = h_test.copy()
                
                if best_C5_local < best_C5:
                    h_current = best_h.copy()
                    best_C5 = best_C5_local
                    improved = True
                    
        return h_current, best_C5
    
    def swap_optimization(h_vec, num_swaps=10):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        
        for _ in range(num_swaps):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Random pair swap
            i, j = np.random.randint(0, len(h_current), 2)
            if i == j:
                continue
                
            h_test = h_current.copy()
            h_test[i], h_test[j] = h_test[j], h_test[i]
            h_test = project_feasible(h_test)
            
            new_C5 = compute_C5(h_test)
            if new_C5 < best_C5:
                h_current = h_test.copy()
                best_C5 = new_C5
                
        return h_current, best_C5
    
    candidates = generate_initial_candidates(5, n_points)
    best_h = None
    best_C5 = float('inf')
    
    # Find the best initial candidate
    for h_init in candidates:
        h_init = project_feasible(h_init)
        C5_init = compute_C5(h_init)
        if C5_init < best_C5:
            best_C5 = C5_init
            best_h = h_init.copy()
    
    # Main optimization loop
    restart_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        # Randomly choose optimization strategy with probability
        strategy = np.random.choice(['coordinate', 'swap'])
        
        if strategy == 'coordinate':
            h_opt, C5_opt = coordinate_descent_step(best_h, step_size=0.05)
            if C5_opt < best_C5:
                best_C5 = C5_opt
                best_h = h_opt.copy()
                restart_count = 0
            else:
                restart_count += 1
                
        elif strategy == 'swap':
            h_opt, C5_opt = swap_optimization(best_h, num_swaps=10)
            if C5_opt < best_C5:
                best_C5 = C5_opt
                best_h = h_opt.copy()
                restart_count = 0
            else:
                restart_count += 1
        
        # Reinitialize if no improvement for a while
        if restart_count > 100:
            # Sample new candidate
            h_new = np.random.random(n_points)
            h_new = project_feasible(h_new)
            best_h = h_new.copy()
            best_C5 = compute_C5(best_h)
            restart_count = 0
        
        # Occasionally use a more global optimization step
        if iteration % 20 == 0:
            # Add some noise to current best and re-project
            h_noise = best_h + np.random.normal(0, 0.02, n_points)
            h_noise = project_feasible(h_noise)
            C5_noise = compute_C5(h_noise)
            if C5_noise < best_C5:
                best_C5 = C5_noise
                best_h = h_noise.copy()
    
    # Final evaluation
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
