import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C_5 with respect to h for the active shift k_max."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        j = (i - k_max) % n
        grad[i] = 1 - 2 * h[j]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Normalize to required sum
    h = project_to_feasible(h, n_points / 2)
    
    # Set dx
    dx = 2.0 / n_points
    
    # Optimization parameters
    max_iter = 10000
    learning_rate = 0.01
    decay_rate = 0.9
    patience = 50
    best_c5 = float('inf')
    best_h = None
    
    # For tracking improvement
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s:
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count
            
        # Compute cross-correlation to find maximizing shift
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = compute_gradient(h, dx, k_max)
        
        # Update h
        h = h - learning_rate * grad
        
        # Project to feasible region
        h = project_to_feasible(h, n_points / 2)
        
        # Adaptive learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            learning_rate *= decay_rate
            
        # Perturbation for escape if no improvement
        if iter_count - last_improvement > patience:
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h += noise
            h = project_to_feasible(h, n_points / 2)
            last_improvement = iter_count
            
        iter_count += 1
    
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        h = project_to_feasible(h, n_points / 2)
        final_c5 = evaluate(h, dx)
        return (h.tolist(), float(final_c5), n_points)
