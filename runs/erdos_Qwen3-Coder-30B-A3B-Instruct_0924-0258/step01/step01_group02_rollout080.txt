import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction
        n_points = 45
        initial_h = np.full(n_points, 1.0 / n_points)
    
    # Parameters
    population_size = 75
    num_generations = 10000
    mutation_rate = 0.05
    crossover_rate = 0.8
    elite_size = population_size // 4
    
    # Precompute dx
    dx = 2.0 / len(initial_h)
    target_sum = len(initial_h) / 2.0
    
    def project_h(h_vec):
        """Project h to feasible region"""
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_c5(h_vec):
        """Compute C5 for given h"""
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def evaluate_population(population):
        """Evaluate all individuals in population"""
        fitness = np.array([compute_c5(ind) for ind in population])
        return -fitness  # Negative because we want to maximize fitness (minimize C5)
    
    # Initialize population
    population = []
    population.append(project_h(initial_h))
    
    # Fill rest of population with random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(len(initial_h))
        h = project_h(h)
        population.append(h)
    
    best_c5 = min([compute_c5(ind) for ind in population])
    best_individual = population[np.argmin([compute_c5(ind) for ind in population])]
    
    # Evolutionary algorithm
    start_time = time.time()
    no_improvement_count = 0
    last_best = best_c5
    
    for generation in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate population
        fitness = evaluate_population(population)
        
        # Selection (tournament selection)
        selected = []
        for _ in range(population_size):
            # Tournament of 3
            indices = np.random.choice(population_size, 3, replace=False)
            tournament_fitness = [fitness[i] for i in indices]
            winner = indices[np.argmax(tournament_fitness)]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[-elite_size:]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                # Uniform crossover
                mask = np.random.rand(len(parent1)) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                # Clone a parent
                parent = np.random.choice(selected)
                child1 = parent.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, len(child1))
                child1 = child1 + noise
                child1 = project_h(child1)
            
            new_population.append(child1)
            
            if len(new_population) < population_size:
                if 'child2' in locals() and np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.05, len(child2))
                    child2 = child2 + noise
                    child2 = project_h(child2)
                new_population.append(child2)
        
        population = new_population[:population_size]
        
        # Update best solution
        current_best = min([compute_c5(ind) for ind in population])
        if current_best < best_c5:
            best_c5 = current_best
            best_individual = population[np.argmin([compute_c5(ind) for ind in population])].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping if no improvement
        if no_improvement_count > 1000:
            break
    
    return (best_individual.tolist(), best_c5, len(best_individual))
