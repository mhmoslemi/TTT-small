import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Distribute difference
        diff = target_sum - current_sum
        if diff > 0:
            # Need to increase values
            idx = np.argsort(h)
            for i in range(len(idx)-1, -1, -1):
                if diff <= 0:
                    break
                j = idx[i]
                increase = min(diff, 1.0 - h[j])
                h[j] += increase
                diff -= increase
        else:
            # Need to decrease values
            idx = np.argsort(h)
            for i in range(len(idx)):
                if diff >= 0:
                    break
                j = idx[i]
                decrease = min(-diff, h[j])
                h[j] -= decrease
                diff += decrease
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize initial
    h0 = project_feasible(h0, n_points / 2.0)
    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = h0.copy()
    best_c5 = evaluate(best_h, dx)
    
    # Parameters
    iterations = 0
    max_iter = 1500
    eta = 0.01
    patience = 0
    patience_limit = 50
    
    h = h0.copy()
    
    while time.time() - start_time < budget_s and iterations < max_iter:
        # Solve convex relaxation approximately
        # We treat the problem as minimizing max_k int h(x)(1-h(x+k)) dx
        # Let's derive gradients from point of view of maximum value
        # Extract maximum correlation index s for current h
        corr = correlate(h, 1-h, mode="full")
        s_max = np.argmax(corr) - len(h) + 1  # align with negative indices
        
        # Gradient approximation:
        # For s_max, we expect correlation to increase with h[i] if h[i+s_max] is small
        # Simple approach: use structured gradient based on correlation of 1-h
        gradient = np.zeros_like(h)
        if s_max > 0:
            # In this case, h[i] contributes to correlation with h[i+s_max]  
            # So gradient w.r.t. h[i] ~ (1 - h[i+s_max])
            gradient[:-s_max] = 1 - h[s_max:]
        elif s_max < 0:
            # Same logic but reversed
            gradient[-s_max:] = 1 - h[:s_max]
        else:
            # Special s=0
            gradient = np.ones_like(h) - h
        
        # Update h using gradient descent
        h = h - eta * gradient
        
        # Project to feasible set
        h = project_feasible(h, n_points / 2.0)
        
        # Evaluate
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
            # Try to reduce learning rate
            if eta > 0.0001:
                eta *= 0.95
        else:
            patience += 1
            
        # Adaptively decrease learning rate to avoid jittering
        if patience > patience_limit:
            patience = 0
            eta *= 0.8
            if eta < 1e-6:
                eta = 1e-6

        iterations += 1

    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
