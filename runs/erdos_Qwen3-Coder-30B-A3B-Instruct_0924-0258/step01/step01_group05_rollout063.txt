import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Simulated Annealing
    T0 = 0.1
    tau = budget_s / np.log(10)
    T_min = 1e-5
    sigma = 0.1
    max_iter = 1000
    accept_count = 0
    total_iter = 0
    
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    
    h_best = h_current.copy()
    C5_best = C5_current
    
    # Track recent acceptance rates
    recent_acceptances = []
    max_recent = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Sample from top contributing intervals
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Create perturbation - focus on relevant intervals
        h_new = h_current.copy()
        # Add Gaussian noise to a subset
        perturb_indices = np.random.choice(n_points, size=max(1, n_points // 4), replace=False)
        noise = np.random.normal(0, sigma, len(perturb_indices))
        h_new[perturb_indices] += noise
        
        # Random flip for diversity
        if np.random.rand() < 0.1:
            flip_indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            h_new[flip_indices] = 1 - h_new[flip_indices]
            
        h_new = project_feasible(h_new)
        C5_new = compute_C5(h_new)
        
        # Calculate change in objective
        delta = C5_new - C5_current
        
        # Accept or reject
        T = max(T_min, T0 * np.exp(-iteration / tau))
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            C5_current = C5_new
            accept_count += 1
            
            if C5_current < C5_best:
                h_best = h_current.copy()
                C5_best = C5_current
        else:
            recent_acceptances.append(0)
            
        recent_acceptances.append(1 if delta < 0 else 0)
        if len(recent_acceptances) > max_recent:
            recent_acceptances.pop(0)
            
        # Adaptive step size
        if len(recent_acceptances) >= 10:
            acc_rate = np.mean(recent_acceptances[-10:])
            if acc_rate < 0.1:
                sigma = min(0.5, sigma * 1.2)
            elif acc_rate > 0.4:
                sigma = max(0.01, sigma * 0.8)
                
        total_iter += 1
        
        if iteration % 100 == 0:
            # Occasionally check progress
            if time.time() - start_time > budget_s - 0.1:
                break
    
    # Final projection and evaluation
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
