import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_descent_step(h, dx, step_size=0.01):
    n = len(h)
    # Analytic gradient for the max correlation term
    # We approximate by computing positive and negative gradients
    # and taking the steepest descent direction
    corr = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    # Compute finite difference approximation of gradient
    h_grad = np.zeros_like(h)
    eps = 1e-5
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_constraints(h_plus, dx)
        corr_plus = correlate(h_plus, 1-h_plus, mode='full')
        h_grad[i] = (corr_plus[max_idx] - corr[max_idx]) / eps
    
    h_new = h - step_size * h_grad
    h_new = project_to_constraints(h_new, dx)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution approach
    resolutions = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        if 'initial_h_values' in globals() and n_points == 10:
            h = np.array(initial_h_values)
            h = h[:n_points]
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_constraints(h, dx)
        
        # PSO parameters
        n_particles = 100
        max_iter = 5000
        w = 0.7
        c1, c2 = 1.5, 1.5
        v_max = 0.05
        iter_count = 0
        
        # Initialize particles
        particles = [h.copy() for _ in range(n_particles)]
        velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(n_particles)]
        pbest = [p.copy() for p in particles]
        pbest_vals = [evaluate(p, dx) for p in particles]
        
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        # PSO loop
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_to_constraints(particles[i], dx)
                
                # Evaluate
                val = evaluate(particles[i], dx)
                
                # Update personal best
                if val < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = val
                    
                # Update global best
                if val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = val
                    
            iter_count += 1
            
            # After PSO, try gradient descent on best particle
            if iter_count % 10 == 0:
                gbest = gradient_descent_step(gbest, dx, 0.005)
                new_val = evaluate(gbest, dx)
                if new_val < gbest_val:
                    gbest_val = new_val
                    
        if gbest_val < best_c5:
            best_c5 = gbest_val
            best_h = gbest.copy()
            
    return best_h.tolist(), best_c5, len(best_h)
