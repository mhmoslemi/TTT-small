import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.array(h, dtype=float)
    # Clamp to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Use a simple projection method: adjust values
        diff = target_sum - current_sum
        if diff > 0:
            # Need to increase values
            # Increase values proportionally in order of smallest to largest
            idx = np.argsort(h)
            for i in range(len(idx)-1, -1, -1):
                j = idx[i]
                if diff <= 0:
                    break
                possible_increase = 1.0 - h[j]
                actual_increase = min(diff, possible_increase)
                h[j] += actual_increase
                diff -= actual_increase
        else:
            # Need to decrease values
            diff = -diff
            idx = np.argsort(h)
            for i in range(len(idx)):
                j = idx[i]
                if diff <= 0:
                    break
                possible_decrease = h[j]
                actual_decrease = min(diff, possible_decrease)
                h[j] -= actual_decrease
                diff -= actual_decrease
    return h

    h = h0.copy()
    h = project_to_feasible(h, target_sum) 
    
    # Optimizer parameters
    lr = 0.01

    patience = 20
    best_c5 = float('inf')
    best_h = None
    patience_counter = 0

        if current_c5 < best_c5:
            best_c5 = current_c5

        else:
            patience_counter += 1

            patience_counter = 0

            h = project_to_feasible(h + np.random.normal(0, 0.005, n_points), target_sum)

        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)

        grad = np.zeros_like(h)
        # We compute derivative of sum h[i] * (1 - h[i + k]) for k = max_k
        n_corr = len(corr)

        # Update with gradient descent
