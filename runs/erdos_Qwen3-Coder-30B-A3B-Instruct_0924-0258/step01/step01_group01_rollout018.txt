import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 10
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)

    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Use scipy.optimize with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Track iteration count
    iter_count = 0
    max_iter_per_res = 1000
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Optimizer run with bounds based on current best
        def objective(h_values):
            h = np.array(h_values)
            h = project_to_constraints(h, dx)
            return evaluate(h, dx)
            
        def constraint_func(h_values):
            h = np.array(h_values)
            return np.sum(h) * dx - 1.0
            
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # Try a few optimization rounds with different strategies
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
                
            try:
                # Random perturb before optimization
                h_perturbed = h + np.random.normal(0, 0.02, n_points)
                h_perturbed = project_to_constraints(h_perturbed, dx)
                
                # Optimization with BFGS
                result = minimize(objective, h_perturbed, method='L-BFGS-B', 
                                bounds=bounds, constraints=cons, 
                                options={'maxiter': 100})
                
                if result.success:
                    h_new = result.x
                    h_new = project_to_constraints(h_new, dx)
                    c5_new = objective(h_new)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new
                else:
                    h_new = h + np.random.normal(0, 0.01, n_points)
                    h_new = project_to_constraints(h_new, dx)
                    c5_new = evaluate(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h = h_new
                        
                iter_count += 1
                if iter_count >= max_iter_per_res:
                    break
                    
            except:
                h_new = h + np.random.normal(0, 0.01, n_points)
                h_new = project_to_constraints(h_new, dx)
                c5_new = evaluate(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h = h_new
                iter_count += 1
                if iter_count >= max_iter_per_res:
                    break
                    
        # Adaptive resolution
        if n_points < 100 and best_c5 < 0.381:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            iter_count = 0
            
        if time.time() - start_time > budget_s * 0.95:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
