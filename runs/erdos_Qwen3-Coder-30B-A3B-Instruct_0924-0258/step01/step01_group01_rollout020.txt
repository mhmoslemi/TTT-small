import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradients(h, dx, critical_lags):
    # Compute gradient of C5 w.r.t h using critical lags
    n = len(h)
    grad = np.zeros(n)
    # For each critical lag k, compute gradient contribution
    for k in critical_lags:
        # correlation at lag k is sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
        # so gradient w.r.t. h[i] = (1-h[i+k]) if 0 <= i+k < n
        # and gradient w.r.t. h[i+k] = -h[i]
        for i in range(n):
            if i + k < n:
                grad[i] += (1 - h[i + k])
            if i >= k:
                grad[i - k] -= h[i]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    # Use `initial_h_values` if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        # Random initial
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5
    
    # Gradient descent with projection
    iter_count = 0
    max_iter_per_res = 2000
    iter_per_res = 200
    learning_rate = 0.1
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute cross correlation to find critical lag
        corr = correlate(h, 1-h, mode='full')
        max_corr_idx = np.argmax(corr)
        
        # Get top 5% as critical lags
        sorted_indices = np.argsort(-corr)
        num_critical = max(1, int(0.05 * len(corr)))
        critical_lags = sorted_indices[:num_critical] - (len(h) - 1)
        critical_lags = critical_lags[critical_lags != 0]  # remove lag 0
        
        for i in range(iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            # Compute gradient
            grad = compute_gradients(h, dx, critical_lags)
            
            # Update step
            h_new = h - learning_rate * grad
            
            # Project back to constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            
            # Evaluate new solution
            c5_new = evaluate(h_new, dx)
            
            # Accept or reject with a condition (simple greedy for speed)
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
                    
            iter_count += 1
        
        # Increase resolution if improvement and not too high
        if (n_points < n_high and 
            (iter_count % (max_iter_per_res // 2) == 0 or last_c5 < best_c5)):
            # Interpolate to make h longer
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_c5 = evaluate(h, dx)
            iter_per_res = min(1000, 2*iter_per_res)  # adjust learning rate or iter count
            learning_rate = max(learning_rate * 0.95, 0.01)  # adaptive decay
        else:
            # Decrease learning rate to stabilize
            learning_rate *= 0.999
            
        if iter_count > max_iter_per_res:
            break
        
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # Final validation
    # Ensure the final solution satisfies constraints
    best_h = project_to_constraints(best_h, dx)
    final_c5 = evaluate(best_h, dx)
    
    return best_h.tolist(), final_c5, len(best_h)
