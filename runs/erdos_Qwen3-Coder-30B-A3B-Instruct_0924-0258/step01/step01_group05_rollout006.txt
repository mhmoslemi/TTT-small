import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm parameters
    pop_size = 50
    tournament_size = 10
    keep_parents = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_rand = np.random.rand(n_points)
        h_rand = project_feasible(h_rand)
        population.append(h_rand)
    
    best_C5 = float('inf')
    best_h = None
    no_improvement_count = 0
    
    # Evaluate initial population
    fitnesses = []
    for h_vec in population:
        fitness = compute_C5(h_vec)
        fitnesses.append(fitness)
        if fitness < best_C5:
            best_C5 = fitness
            best_h = h_vec.copy()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for improvement
        if fitnesses[0] < best_C5:
            best_C5 = fitnesses[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= patience:
                # Reset worst performers
                for i in range(-5, 0):
                    population[i] = np.random.rand(n_points)
                    population[i] = project_feasible(population[i])
                no_improvement_count = 0
        
        # Create new population
        new_population = population[:keep_parents]
        
        # Tournament selection and reproduction
        while len(new_population) < pop_size:
            # Tournament selection
            selected_indices = np.random.choice(range(keep_parents), tournament_size, replace=False)
            selected_fitnesses = [fitnesses[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitnesses)]
            
            # Select two parents
            parent1 = population[winner_idx]
            parent2 = population[np.random.randint(0, keep_parents)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
                child = project_feasible(child)
            
            # Ensure we maintain the correct average
            child = project_feasible(child)
            new_population.append(child)
        
        # Reassign population
        population = new_population
        
        # Evaluate new population
        fitnesses = []
        for h_vec in population:
            fitness = compute_C5(h_vec)
            fitnesses.append(fitness)
    
    # Final check
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
