import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Compute maximum overlap for given h"""
    conv = np.correlate(h, 1 - h, mode='full')
    mid = len(conv) // 2
    overlaps = conv[mid - len(h) + 1:mid + len(h) - 1] * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback to a simple construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper functions:
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    # For SDP approach
    def solve_sdp(h_init, n_points_):
        # SDP solver
        h_var = cp.Variable(n_points_)
        X = cp.Variable((n_points_, n_points_))
        
        # Constraints
        constraints = [
            X >> 0,  # Positive semidefinite
            cp.diag(X) == h_var,  # Diagonal entries
            0 <= h_var, h_var <= 1,  # Bounded
            cp.sum(h_var) == target_sum  # Sum constraint
        ]
        
        # Objective: minimize max overlap
        # This requires reformulation - we actually want to maximize the minimum
        # But for optimization purposes, we'll use a proxy objective
        # Let's instead extract the current C5 value from a feasible point
        
        prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(X, np.ones((n_points_, n_points_)))), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = np.array(h_var.value).flatten()
                return h_sol
        except:
            pass
        return h_init.copy()
    
    # Main loop with hybrid approach
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Choose starting point based on available initial value
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Standard initialization
        h = np.ones(n_points) * 0.5
    
    # Run optimization loop with smart heuristics
    iteration = 0
    last_improvement = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Adaptive step size and exploration
        if iteration % 10 == 0:
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()
                last_improvement = iteration
                
        # 1. Local gradient-based step
        h_vec = project_h(h)
        # Compute rough gradient with finite difference
        eps = 1e-4
        gradient = np.zeros_like(h_vec)
        h_plus = h_vec.copy()
        h_minus = h_vec.copy()
        
        for i in range(n_points):
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            gradient[i] = (C_plus - C_minus) / (2 * eps)
            # Reset for next dimension
            h_plus[i] -= eps
            h_minus[i] += eps
        
        # Update step with adaptive lr
        lr = 0.01 * max(0.1, 1.0 - (iteration - last_improvement) / 100)
        h = h_vec - lr * gradient
        h = project_h(h)
        
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Occasionally try SDP for global enhancement
        if iteration % 30 == 0:
            h_try = solve_sdp(h, n_points)
            h_try = project_h(h_try)
            test_C5 = compute_C5(h_try)
            if test_C5 < best_C5:
                best_C5 = test_C5
                best_h = h_try.copy()
                h = h_try.copy()
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
