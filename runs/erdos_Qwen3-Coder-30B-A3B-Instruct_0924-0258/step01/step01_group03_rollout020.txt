import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Initialize
    h = h0.copy().astype(float)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        idx = np.argsort(h)
        # Distribute adjustment
        for i in range(len(idx)-1, -1, -1):
            if diff == 0:
                break
            j = idx[i]
            if h[j] < 1.0 and diff > 0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 50
    patience_counter = 0
    last_improvement = 0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute correlation matrix to find maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        # In correlate full mode, we get correlation at all lags
        # Find the lag that gives maximum overlap
        # corr indices go from -n+1 to n-1, so lag = max_k - (n-1)
        k = max_k - (n_points - 1)
        
        # Compute the gradient for current max_k
        grad = np.zeros(n_points)
        for idx in range(n_points):
            if idx + k >= 0 and idx + k < n_points:
                grad[idx] += 1 - 2 * h[idx + k]
        
        # Update h
        h_new = h - learning_rate * grad
        
        # Project to feasible region [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-10:
            ratio = target_sum / current_sum
            h_new = h_new * ratio
            # Ensure final projection
            h_new = np.clip(h_new, 0, 1)
            current_sum = np.sum(h_new)
            if abs(current_sum - target_sum) > 1e-10:
                diff = target_sum - current_sum
                idx = np.argsort(h_new)
                for j in range(len(idx)-1, -1, -1):
                    if diff == 0:
                        break
                    i_idx = idx[j]
                    if h_new[i_idx] < 1.0 and diff > 0:
                        update = min(diff, 1.0 - h_new[i_idx])
                        h_new[i_idx] += update
                        diff -= update
        
        h = h_new
        
        # Evaluate and track best
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
            last_improvement = i
        else:
            patience_counter += 1
            if patience_counter >= patience:
                learning_rate *= 0.9
                patience_counter = 0
        
        # Add a bit more diversity in later iterations
        if i > max_iter // 2 and np.random.random() < 0.05:
            perturbation = np.random.normal(0, 0.001, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            # Re-normalize
            current_sum = np.sum(h)
            h = h * target_sum / current_sum
            h = np.clip(h, 0, 1)

    return (best_h.tolist(), float(best_c5), n_points)
