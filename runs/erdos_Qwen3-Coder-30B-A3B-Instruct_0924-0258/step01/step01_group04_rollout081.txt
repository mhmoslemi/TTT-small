import numpy as np
from scipy import signal
import time

def compute_C5(h_vec, dx):
    """Compute the maximum overlap integral for given h vector."""
    h_vec = np.array(h_vec)
    # Compute cross-correlation of h with (1-h)
    conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
    # Extract the relevant part (corresponding to k values from 0 to 2)
    mid = len(conv) // 2
    overlaps = conv[mid - len(h_vec) + 1 : mid + len(h_vec) - 1] * dx
    return np.max(overlaps)

def project_feasible(h_vec, target_sum):
    """Project h vector to feasible region: [0,1] and sum = target_sum."""
    h_vec = np.clip(h_vec, 0, 1)
    current_sum = np.sum(h_vec)
    if current_sum > 0:
        h_vec = h_vec * target_sum / current_sum
    return h_vec

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    target_sum = n_points / 2.0

    # PSO parameters
    n_particles = 100
    max_iters = 1000
    w_start = 0.7
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_speed = 0.1

    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []

    for _ in range(n_particles):
        # Initialize near the initial construction
        h_init = h + np.random.normal(0, 0.02, n_points)
        h_init = project_feasible(h_init, target_sum)
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        pbest_positions.append(h_init.copy())
        pbest_scores.append(compute_C5(h_init, dx))

    # Global best
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    # PSO loop
    for iter_count in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * (iter_count / max_iters)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest_positions[i] - particles[i]) +
                c2 * r2 * (gbest_position - particles[i])
            )
            
            # Clamp speed
            velocities[i] = np.clip(velocities[i], -max_speed, max_speed)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i], target_sum)
            
            # Evaluate
            current_score = compute_C5(particles[i], dx)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position = particles[i].copy()

    # Return best solution found
    return (gbest_position.tolist(), gbest_score, n_points)
