import numpy as np
from scipy import optimize
import time
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Main optimization loop using a hybrid approach
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Genetic Algorithm parameters
    population_size = 50
    num_generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        individual = np.random.uniform(0.4, 0.6, n_points)
        individual = project_h(individual)
        population.append(individual)
    
    # Previous best for early stopping
    prev_best_C5 = float('inf')
    no_improvement_count = 0
    
    # GA loop
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for individual in population:
            fitness = compute_C5(individual)
            fitnesses.append(fitness)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses.sort()
        
        # Update best
        if fitnesses[0] < best_C5:
            best_C5 = fitnesses[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping
        if no_improvement_count > 100:
            break
            
        # Select elite
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            mutate(child1, mutation_rate)
            mutate(child2, mutation_rate)
            
            # Project to feasible region
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
        # Local search on top individuals periodically
        if generation % 20 == 0:
            for i in range(min(5, len(population))):
                individual = population[i].copy()
                try:
                    # Local optimization using Nelder-Mead
                    result = optimize.minimize(
                        lambda x: compute_C5(x),
                        individual,
                        method='Nelder-Mead',
                        options={'maxiter': 100, 'adaptive': True}
                    )
                    if result.success:
                        new_individual = project_h(result.x)
                        new_fitness = compute_C5(new_individual)
                        if new_fitness < fitnesses[i]:
                            population[i] = new_individual
                            fitnesses[i] = new_fitness
                except:
                    pass
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)

def tournament_selection(population, fitnesses, tournament_size=3):
    """Tournament selection for genetic algorithm"""
    selected_indices = np.random.choice(len(population), tournament_size)
    selected_fitnesses = [fitnesses[i] for i in selected_indices]
    winner_idx = selected_indices[np.argmin(selected_fitnesses)]
    return population[winner_idx]

def crossover(parent1, parent2):
    """Uniform crossover"""
    child1 = parent1.copy()
    child2 = parent2.copy()
    
    for i in range(len(parent1)):
        if random.random() < 0.5:
            child1[i], child2[i] = child2[i], child1[i]
            
    return child1, child2

def mutate(individual, mutation_rate):
    """Gaussian mutation"""
    for i in range(len(individual)):
        if random.random() < mutation_rate:
            individual[i] += np.random.normal(0, 0.05)
