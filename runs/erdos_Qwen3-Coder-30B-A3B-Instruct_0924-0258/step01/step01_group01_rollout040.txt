import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 10
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Start with lower resolution and gradually increase
    resolutions = [10, 20, 40, 70]
    current_res_idx = 0
    
    MaxIter = 1000
    
    for i_iter in range(MaxIter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Use more diverse mutation strategies
        if np.random.random() < 0.7:
            # Gaussian perturbation
            perturbation = np.random.normal(0, 0.05, n_points)
            h_new = h + perturbation
        else:
            # Random step
            h_new = np.random.uniform(0, 1, n_points)
            
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        
        c5_new = evaluate(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            # Interpolate if we get a good solution
            if current_res_idx < len(resolutions) - 1 and c5_new < best_c5 * 0.99:
                new_n = resolutions[current_res_idx + 1]
                if new_n <= 100:
                    x_old = np.linspace(0, 2, n_points)
                    x_new = np.linspace(0, 2, new_n)
                    h_new_interp = np.interp(x_new, x_old, h_new)
                    h_new = h_new_interp
                    n_points = new_n
                    dx = 2.0 / n_points
                    current_res_idx += 1
        
        h = h_new
    
    return best_h.tolist(), best_c5, len(best_h)
