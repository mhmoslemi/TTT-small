import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # k_max: the lag that produces the maximum overlap
        n = len(h_vec)
        grad = np.zeros(n)
        for i in range(n):
            # gradient contribution from h[i] to the k_max overlap
            # h[i] pairs with (1 - h[(i+k_max) % n])
            # so partial derivative w.r.t. h[i] is (1 - h[(i+k_max) % n])
            # and contribution from (1 - h[i]) paired with h[(i-k_max) % n]
            # so partial derivative w.r.t. h[i] is -h[(i-k_max) % n]
            grad[i] = (1 - h_vec[(i + k_max) % n]) - h_vec[(i - k_max) % n]
        return grad
    
    def perform_step(h_vec, alpha, k_max):
        grad = compute_gradient(h_vec, k_max)
        h_new = h_vec - alpha * grad
        return project_feasible(h_new)
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Gradient descent with adaptive step size
    alpha = 0.1
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute all correlations
        corr = correlate(h_best, 1 - h_best, mode='full')
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_best) - 1)
        
        # Perform gradient step
        h_trial = perform_step(h_best, alpha, max_lag)
        C5_trial = compute_C5(h_trial)
        
        if C5_trial < C5_best:
            h_best = h_trial.copy()
            C5_best = C5_trial
            stagnation_count = 0
            alpha = min(0.5, alpha * 1.1)  # Increase step size
        else:
            stagnation_count += 1
            if stagnation_count > 50:
                break
            alpha = max(1e-6, alpha * 0.5)  # Decrease step size
            
        # Occasionally perturb to escape local minima
        if stagnation_count % 10 == 0 and stagnation_count > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_pert = h_best + noise
            h_pert = project_feasible(h_pert)
            C5_pert = compute_C5(h_pert)
            if C5_pert < C5_best:
                h_best = h_pert
                C5_best = C5_pert
    
    # Final projection and evaluation
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
