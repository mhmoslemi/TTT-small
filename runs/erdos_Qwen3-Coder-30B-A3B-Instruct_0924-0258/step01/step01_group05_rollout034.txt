import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm parameters
    pop_size = 50
    tournament_size = 10
    elite_size = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        population.append(h_init)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Track best solution
    best_overall_h = best_h.copy()
    best_overall_C5 = best_C5
    
    # Track generations without improvement
    no_improvement = 0
    
    # Main genetic loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate fitness
        fitness = []
        for h_ind in population:
            c5 = compute_C5(h_ind)
            fitness.append(c5)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update global best
        current_best_C5 = fitness[0]
        if current_best_C5 < best_overall_C5:
            best_overall_C5 = current_best_C5
            best_overall_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early stopping
        if no_improvement >= patience:
            break
            
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parents = []
            for _ in range(2):
                tournament = np.random.choice(pop_size, tournament_size, replace=False)
                best_in_tournament = min(tournament, key=lambda i: fitness[i])
                parents.append(population[best_in_tournament])
                
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                child = np.where(np.random.rand(n_points) < 0.5, parents[0], parents[1])
            else:
                child = parents[0].copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_feasible(child)
                
            # Reset mutation (rare)
            if np.random.rand() < 0.05:
                reset_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                child[reset_indices] = np.random.choice([0, 1], size=len(reset_indices))
                child = project_feasible(child)
                
            new_population.append(child)
            
        population = new_population[:pop_size]
    
    # Final best
    final_C5 = compute_C5(best_overall_h)
    
    return list(best_overall_h), final_C5, n_points
