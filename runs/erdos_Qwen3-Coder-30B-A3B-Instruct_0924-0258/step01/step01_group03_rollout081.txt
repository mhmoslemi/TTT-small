import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while keeping values in [0,1]."""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    h = h * target_sum / current_sum
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # If still not exact, adjust carefully
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-9:
        diff = target_sum - current_sum
        # Distribute difference among elements to maintain feasibility
        idx = np.argsort(h)
        # Prefer increasing values that are less than 1
        for i in range(len(idx)-1, -1, -1):
            j = idx[i]
            if diff == 0:
                break
            if h[j] < 1.0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    return h

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    h = h0.copy()
    h = project_h(h, target_sum)
    
    # Parameters
    learning_rate = 0.01
    decay_factor = 0.9
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        # Get the shift index that gives maximum overlap
        k = np.argmax(corr)
        
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i - k) % len(h)
            grad[i] = (1 - 2 * h[j])
        
        # Update with gradient descent
        h = h - learning_rate * grad
        
        # Project to feasible set
        h = project_h(h, target_sum)
        
        # Evaluate the current solution
        c5 = evaluate(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if patience_counter > 100:
            learning_rate *= decay_factor
            patience_counter = 0
            
        # Recovery from stagnation
        if patience_counter >= patience:
            # Add small noise
            noise = np.random.uniform(-1e-3, 1e-3, len(h))
            h = h + noise
            h = project_h(h, target_sum)
            patience_counter = 0
            
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback
        final_h = project_h(h0, target_sum)
        final_c5 = evaluate(final_h, dx)
        return (final_h.tolist(), float(final_c5), n_points)
