import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization    
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def local_improve(h_vec):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        improved = True
        iterations = 0
        
        while improved and time.time() - start_time < budget_s - 0.1:
            improved = False
            iterations += 1
            
            # Coordinate descent on each interval
            for i in range(n_points):
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                orig_val = h_current[i]
                best_val = orig_val
                best_C5_local = best_C5
                
                # Try small changes
                for step in [0.02, -0.02, 0.05, -0.05]:
                    new_val = orig_val + step
                    if 0 <= new_val <= 1:
                        h_current[i] = new_val
                        new_C5 = compute_C5(h_current)
                        if new_C5 < best_C5_local:
                            best_C5_local = new_C5
                            best_val = new_val
                h_current[i] = best_val
                if best_C5_local < best_C5:
                    best_C5 = best_C5_local
                    improved = True
            
            # Pairwise swaps
            for _ in range(5):
                if time.time() - start_time > budget_s - 0.1:
                    break
                i, j = np.random.choice(n_points, 2, replace=False)
                orig_i, orig_j = h_current[i], h_current[j]
                h_current[i], h_current[j] = orig_j, orig_i
                new_C5 = compute_C5(h_current)
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    improved = True
                else:
                    h_current[i], h_current[j] = orig_i, orig_j
                    
            if iterations > 200:
                break
                
        return h_current, best_C5
    
    # Generate diverse initial candidates via Latin hypercube sampling
    initial_candidates = []
    for _ in range(5):
        h_candidate = np.random.rand(n_points)
        h_candidate = project_feasible(h_candidate)
        initial_candidates.append(h_candidate)
    
    # Evaluate all initial candidates
    best_h = h.copy()
    C5_best = compute_C5(best_h)
    
    for h_init in initial_candidates:
        h_opt, C5_opt = local_improve(h_init)
        if C5_opt < C5_best:
            C5_best = C5_opt
            best_h = h_opt.copy()
    
    # Local search loop with restarts
    restarts = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        # Perturb current best solution
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_feasible(h_perturbed)
        
        # Local optimization from perturbed point
        h_opt, C5_opt = local_improve(h_perturbed)
        
        if C5_opt < C5_best:
            C5_best = C5_opt
            best_h = h_opt.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Restart if stuck
        if no_improvement_count > 100 or restarts > 5:
            # Generate a new random candidate
            h_random = np.random.rand(n_points)
            h_random = project_feasible(h_random)
            h_opt, C5_opt = local_improve(h_random)
            if C5_opt < C5_best:
                C5_best = C5_opt
                best_h = h_opt.copy()
            no_improvement_count = 0
            restarts = 0
        else:
            restarts += 1
    
    # Final refinement
    best_h, C5_best = local_improve(best_h)
    
    return list(best_h), C5_best, n_points
