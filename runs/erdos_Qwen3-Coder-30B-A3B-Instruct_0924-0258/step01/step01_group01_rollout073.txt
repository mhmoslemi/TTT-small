import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradients(h, dx):
    corr = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    n = len(h)
    # The full correlation has 2*n-1 elements
    # The actual lag index range is from -(n-1) to (n-1) 
    actual_lag = max_idx - (n-1)
    # Now we need the gradient w.r.t. h[i] for all i
    # This will help us approximate the gradient for the maximum
    # For a single lag k, the gradient is approximately:
    # d/dh_i [ h(x) * (1 - h(x+k)) ] = delta_{i, x} * (1 - h(x+k)) - delta_{i, x+k} * h(x)
    # Let's just use heuristics based on symmetries and focus on where max correlation occurs
    
    # The gradient is numerical approximate;
    # here we aim to move in direction of decreasing correlation for max lag
    #.compute a simple gradient based on the previous result
    grad = np.zeros_like(h)
    for i in range(n):
        # df/dh_i for the maximum lag, based on 1-d discrete derivative idea
        # This is a rough gradient approximation
        # We approximate the effect of changes in h_i on the maximum correlation
        if i > 0:
            grad[i] += 1 - h[i-1]
        if i < n-1:
            grad[i] += 1 - h[i+1]
        grad[i] -= h[i]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with small resolution
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
        
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    lr = 0.01
    max_iter_per_res = 1000
    iter_count = 0
    last_c5 = best_c5

    while True:
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # For each resolution step, do some iterations
        for _ in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            iter_count += 1
            
            # Compute gradient
            grad = compute_gradients(h, dx)
            h_new = h - lr * grad
            
            # Clip
            h_new = np.clip(h_new, 0, 1)
            # Project
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
            else:
                # Adaptive learning rate decay
                lr *= 0.999
                
        # Resolution double if improvement or beyond small res
        if n_points < n_high:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                               np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_c5 = evaluate(h, dx)
            lr = 0.01
            
    return best_h.tolist(), best_c5, len(best_h)
