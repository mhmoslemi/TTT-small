import numpy as np

def run(seed=42, budget_s=600, **kwargs):

    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def optimize_step(h_vec, max_iter=200):
        # Alternating optimization between h and the constraints
        h_current = h_vec.copy()
        best_c5 = compute_c5(h_current)
        best_h = h_current.copy()
        
        for iter_ in range(max_iter):
            # Simple gradient step on the maximum correlation
            eps = 1e-5
            grad_h = np.zeros_like(h_current)
            corr_base = np.correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr_base)
            
            for i in range(len(h_current)):
                h_test = h_current.copy()
                h_test[i] += eps
                h_test = project_h(h_test)
                corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                grad_h[i] = (corr_test[k_max] - corr_base[k_max]) / eps
            
            # Gradient update
            lr = 0.01
            h_current = h_current - lr * grad_h
            h_current = project_h(h_current)
            
            # Update best so far
            new_c5 = compute_c5(h_current)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_current.copy()
                
        return best_h, best_c5
    
    # Initialize best
    best_h, best_c5 = optimize_step(h)
    
    # Binary search on the bound using convex optimization

        # is to sample a few interesting lags

        sample_lags = min(10, n // 2)

        return h_var, max_val

            break

        # Exhaustive local optimization
        current_h, current_c5 = optimize_step(best_h)

            best_c5 = current_c5
            best_h = current_h.copy()

    # Final check of projection and bounds
    best_h = project_h(best_h)
