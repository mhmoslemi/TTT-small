import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution optimization
    resolutions = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    n_points = 10
    dx = 2.0 / n_points
    
    # Start with initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
        
    # Time budget - 150s per resolution
    resolution_time = budget_s / len(resolutions)
    
    for res in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        n_points = res
        dx = 2.0 / n_points
        
        # Clone h to start new resolution
        if res == resolutions[0]:
            h_current = h.copy()
        else:
            # Interpolate from previous resolution
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h_current = np.interp(x_new, x_old, h)
        
        h_current = project_to_constraints(h_current, dx)
        
        # Particle swarm optimization
        particles = 100
        max_iter = int(resolution_time * 0.8 / 0.01)  # Rough estimate
        velocities = np.random.uniform(-0.05, 0.05, (particles, n_points))
        positions = np.random.uniform(0, 1, (particles, n_points))
        
        # Project particles to constraints
        for i in range(particles):
            positions[i] = project_to_constraints(positions[i], dx)
        
        # Initialize personal bests
        p_best = positions.copy()
        p_best_vals = [evaluate(p, dx) for p in positions]
        p_best_idx = np.argmin(p_best_vals)
        g_best = p_best[p_best_idx].copy()
        g_best_val = p_best_vals[p_best_idx]
        
        # PSO parameters
        inertia = 0.7
        c1 = 1.5
        c2 = 1.5
        w = inertia
        
        # Gradient descent refinement
        for it in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # PSO loop
            for i in range(particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (p_best[i] - positions[i]) + 
                                c2 * r2 * (g_best - positions[i]))
                
                # Clip velocities
                velocities[i] = np.clip(velocities[i], -0.05, 0.05)
                
                # Update position
                positions[i] += velocities[i]
                positions[i] = np.clip(positions[i], 0, 1)
                positions[i] = project_to_constraints(positions[i], dx)
                
                # Evaluate and update personal bests
                current_val = evaluate(positions[i], dx)
                if current_val < p_best_vals[i]:
                    p_best_vals[i] = current_val
                    p_best[i] = positions[i].copy()
                    if current_val < g_best_val:
                        g_best_val = current_val
                        g_best = positions[i].copy()
            
            # Update inertia weight
            w = inertia * (1 - it / max_iter)
            
            # Gradient step refinement
            grad_step = 0.01
            h_grad = g_best.copy()
            
            # Evaluate gradient approximation
            # dC5/dh_i = (1 - 2*h_i) * dx
            h_grad += grad_step * (1 - 2 * h_grad) * dx
            h_grad = np.clip(h_grad, 0, 1)
            h_grad = project_to_constraints(h_grad, dx)
            
            val_grad = evaluate(h_grad, dx)
            if val_grad < g_best_val:
                g_best = h_grad.copy()
                g_best_val = val_grad
            
            if g_best_val < best_c5:
                best_c5 = g_best_val
                best_h = g_best.copy()
                
        h = g_best.copy()
        
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    else:
        # Final fallback
        best_h = project_to_constraints(np.random.uniform(0, 1, n_points), dx)
        best_c5 = evaluate(best_h, dx)
        
    return best_h.tolist(), best_c5, len(best_h)
