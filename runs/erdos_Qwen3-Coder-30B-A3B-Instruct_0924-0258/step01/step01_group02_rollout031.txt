import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        # Default construction
        h_init = np.full(45, 1.0/45)
    
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    h_init = h_init * target_sum / np.sum(h_init)
    
    def compute_c5(h_vec):
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Genetic Algorithm Parameters
    population_size = 60
    num_generations = 1000
    mutation_rate = 0.05
    elite_size = population_size // 4
    tournament_size = 3
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project_h(h_rand)
        population.append(h_rand)
    
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    start_time = time.time()
    
    for generation in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5_val = compute_c5(h)
            fitness.append(-c5_val)  # Negative for maximization
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if -fitness[0] < best_c5:
            best_c5 = -fitness[0]
            best_h = population[0].copy()
        
        # Selection
        selected = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament]
            winner = tournament[np.argmax(tournament_fitness)]
            selected.append(population[winner].copy())
        
        # Crossover and Mutation
        new_population = population[:elite_size]  # Elites preserved
        
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(len(selected), 2, replace=False)
            p1, p2 = selected[parent1], selected[parent2]
            
            # Uniform crossover
            mask = np.random.rand(n_points) > 0.5
            child1 = np.where(mask, p1, p2)
            child2 = np.where(mask, p2, p1)
            
            # Mutate
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child1 += noise
                child1 = project_h(child1)
                
            if len(new_population) < population_size:
                new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        
        # Occasionally add diversity
        if generation % 50 == 0:
            for i in range(int(population_size * 0.1)):
                h_new = np.random.rand(n_points)
                h_new = project_h(h_new)
                idx = np.random.randint(population_size)
                population[idx] = h_new
    
    return (best_h.tolist(), best_c5, n_points)
