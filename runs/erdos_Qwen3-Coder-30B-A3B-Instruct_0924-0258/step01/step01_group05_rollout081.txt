import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Use convolution to compute all correlations efficiently
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm parameters
    pop_size = 50
    tournament_size = 10
    elitism = int(0.2 * pop_size)
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_init = np.random.random(n_points)
        h_init = project_feasible(h_init)
        population.append(h_init)
    
    best_C5 = float('inf')
    best_h = None
    patience_counter = 0
    
    # Evaluate initial population
    fitness = []
    for h_ind in population:
        fit = compute_C5(h_ind)
        fitness.append(fit)
        if fit < best_C5:
            best_C5 = fit
            best_h = h_ind.copy()
    
    generation = 0
    while time.time() - start_time < budget_s - 0.1:
        generation += 1
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        new_pop = population[:elitism]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            # Selection
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent1 = population[winner_idx]
            
            if np.random.random() < crossover_rate:
                # Select second parent
                tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                parent2 = population[winner_idx]
                
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_feasible(child)
            
            new_pop.append(child)
        
        # Replace population
        population = new_pop[:pop_size]
        
        # Evaluate new population
        fitness = []
        for h_ind in population:
            fit = compute_C5(h_ind)
            fitness.append(fit)
            if fit < best_C5:
                best_C5 = fit
                best_h = h_ind.copy()
        
        # Check for improvement
        if generation > 0 and generation % 10 == 0:
            patience_counter += 1
            if patience_counter > patience:
                # Reset some individuals
                for i in range(int(0.1 * pop_size)):
                    rand_idx = np.random.randint(pop_size)
                    h_init = np.random.random(n_points)
                    h_init = project_feasible(h_init)
                    population[rand_idx] = h_init
                    fitness[rand_idx] = compute_C5(h_init)
                patience_counter = 0
        
    # Final check
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
