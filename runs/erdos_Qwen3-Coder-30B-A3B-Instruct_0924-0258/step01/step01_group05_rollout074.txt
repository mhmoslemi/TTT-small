import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Gradient-based optimization
    max_iter = 1000
    alpha = 0.1
    stagnation_count = 0
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for the maximum overlap lag
        # For correlation with lag k, h[i] contributes to h[i+k] (when i+k < n)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + max_lag) % len(h)
            grad[i] = -(1 - h[j])
            j = (i - max_lag) % len(h)
            grad[i] += h[j]
        
        # Update step
        h_new = h - alpha * grad
        h_new = project_feasible(h_new)
        
        # Evaluate new solution
        new_C5 = compute_C5(h_new)
        
        # Accept improvement / update parameters
        if new_C5 < best_C5:
            h = h_new.copy()
            best_C5 = new_C5
            best_h = h.copy()
            stagnation_count = 0
            alpha = 0.1  # Reset step size
        else:
            stagnation_count += 1
            if stagnation_count > 50:
                # Reduce step size
                alpha *= 0.5
                stagnation_count = 0
                
            # Occasionally try random perturbation
            if stagnation_count % 10 == 0:
                noise = np.random.normal(0, 0.01, n_points)
                h_noisy = h + noise
                h_noisy = project_feasible(h_noisy)
                noisy_C5 = compute_C5(h_noisy)
                if noisy_C5 < best_C5:
                    h = h_noisy.copy()
                    best_C5 = noisy_C5
                    best_h = h.copy()
    
    # Final check
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
