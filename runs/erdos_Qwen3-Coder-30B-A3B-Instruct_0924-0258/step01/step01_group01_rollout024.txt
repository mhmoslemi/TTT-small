import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradients(h, dx, critical_lags):
    n = len(h)
    # Compute correlation with 1-h (forward and backward)
    corr = correlate(h, 1-h, mode='full')
    # Get the indices corresponding to the critical lags
    # Full correlation is of length 2*n-1
    full_idx = np.arange(2*n-1)
    # For each critical lag, compute the gradient contribution
    grad = np.zeros(n)
    for lag in critical_lags:
        # lag counts from 0 to 2*n-2
        # Forward correlation, lag k corresponds to
        # h[i] * (1-h[i+k]) for i = 0..n-1-k
        # Backward correlation: h[i+k] * (1-h[i]) for i = 0..n-1-k
        if lag < n:
            # Forward direction
            for i in range(n - lag):
                grad[i] += (1 - h[i + lag])
        else:
            # Backward direction
            lag_b = lag - (n - 1)
            for i in range(n - lag_b):
                grad[i + lag_b] += -h[i]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    step_size = 0.01
    decay = 0.995
    min_step_size = 1e-6
    max_iter_per_res = 2000
    iter_per_res = 0
    
    resolution_growth_threshold = 0.01
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        
        same_res_count = 0
        prev_c5 = best_c5
        
        for i in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            # Compute current C5 and identify critical lags
            c5_val = evaluate(h, dx)
            corr = correlate(h, 1-h, mode='full')
            # Get top 10% of lags 
            sorted_lag_indices = np.argsort(corr)[::-1][:max(1, int(0.1 * len(corr)))]
            critical_lags = sorted_lag_indices
            
            grad = compute_gradients(h, dx, critical_lags)
            # Take steps in gradient direction to minimize
            h_new = h - step_size * grad
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            if c5_new < c5_val:
                h = h_new
                c5_val = c5_new
                if c5_val < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_val
                    step_size *= decay
            else:
                step_size *= 0.995
                if step_size < min_step_size:
                    step_size = min_step_size
                    
            iter_per_res += 1
            
        # Check for resolution increase
        if n_points < n_high and (prev_c5 - best_c5) > resolution_growth_threshold:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            step_size = 0.01
            iter_per_res = 0
        elif n_points < n_high and iter_per_res > 1000:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            step_size = 0.01
            iter_per_res = 0
        else:
            if iter_per_res < 500:
                break

    return best_h.tolist(), best_c5, len(best_h)
