import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum correlation scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Ensure h is in [0,1] and sum equals target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def adaptive_simulated_annealing(h_vec, max_iter=500):
        h_current = h_vec.copy()
        h_best = h_current.copy()
        C5_best = compute_C5(h_current)
        
        # Annealing parameters
        T = 0.1
        T_min = 1e-5
        alpha = 0.95
        initial_T = T
        
        # Step size
        sigma = 0.1
        
        # Acceptance tracking
        accept_count = 0
        total_count = 0
        
        # History for cooling
        last_improvement = 0
        improvement_threshold = 20
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            t = iteration / max_iter
            T = max(T_min, initial_T * (alpha ** t))
            
            # Local optimization steps: sample priority intervals
            # Find intervals contributing most to max overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            max_lag = max_idx - (len(h_current) - 1)
            
            # Perturb with noise in specific locations
            h_new = h_current.copy()
            
            # Perturbation in high-contributing regions (around max_lag)
            perturb_indices = []
            if max_lag >= 0:
                # Positive lag case
                for i in range(len(h_current)):
                    if i + max_lag < len(h_current):
                        perturb_indices.append(i)
            else:
                # Negative lag case
                for i in range(len(h_current)):
                    if i - max_lag >= 0:
                        perturb_indices.append(i)
                        
            # Also try to sometimes flip some random small elements
            if np.random.rand() < 0.3 and len(perturb_indices) > 0:
                # Flip a small percentage
                flip_indices = np.random.choice(perturb_indices, size=max(1, len(perturb_indices)//20), replace=False)
                for i in flip_indices:
                    h_new[i] = 1.0 - h_new[i]
            
            # Add Gaussian noise
            noise = np.random.normal(0, sigma, len(h_current))
            h_new += noise
            
            # Apply feasibility projection
            h_new = project_feasible(h_new)
            
            # Compute new C5
            C5_new = compute_C5(h_new)
            
            # Accept or reject
            delta = C5_new - C5_best
            total_count += 1
            
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h_current = h_new.copy()
                if C5_new < C5_best:
                    C5_best = C5_new
                    h_best = h_current.copy()
                    last_improvement = iteration
                    accept_count += 1
            else:
                accept_count += 1
            
            # Adaptive sigma and cooling
            if accept_count > 0 and total_count > 0:
                accept_rate = accept_count / total_count
                if accept_rate < 0.1:
                    sigma = min(sigma * 1.2, 0.5)
                elif accept_rate > 0.4:
                    sigma *= 0.95
                    
            # If we've not improved in a while, reset or nadir
            if iteration - last_improvement > improvement_threshold:
                # Try to escape local min with a more aggressive perturbation or restart
                restart_prob = 0.02
                if np.random.rand() < restart_prob:
                    h_current = project_feasible(np.random.rand(len(h_current)))
                    C5_current = compute_C5(h_current)
                    if C5_current < C5_best:
                        C5_best = C5_current
                        h_best = h_current.copy()
                        last_improvement = iteration
                        
        return h_best, C5_best
    
    # Run multiple local optimization trials
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Use a pool-based multi-start strategy
    for _ in range(10):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Start with slightly perturbed initial solution
        h_perturbed = h + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_feasible(h_perturbed)
        
        h_opt, C5_opt = adaptive_simulated_annealing(h_perturbed, max_iter=200)
        
        if C5_opt < best_C5:
            best_C5 = C5_opt
            best_h = h_opt.copy()
            
    # Final project
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
