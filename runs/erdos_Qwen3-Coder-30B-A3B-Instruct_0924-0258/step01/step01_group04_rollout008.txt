import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    n_particles = 100
    max_iter = 2000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(n_particles):
        h_init = project_h(h + np.random.normal(0, 0.05, n_points))
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest.append(h_init.copy())
        pbest_values.append(compute_C5(h_init))
    
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_speed, max_speed)
            
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i])
            
            current_value = compute_C5(particles[i])
            
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
    
    # Final validation
    final_C5 = compute_C5(gbest)
    if final_C5 < gbest_value:
        gbest_value = final_C5
    
    return (gbest.tolist(), gbest_value, n_points)
