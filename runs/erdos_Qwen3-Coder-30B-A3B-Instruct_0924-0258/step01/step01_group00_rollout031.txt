import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def admm_optimize(h_start, n_points, dx, time_limit):
    """Use ADMM-style optimization with constraint handling"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 500
    tol = 1e-6
    
    h = h_start.copy()
    
    # For high resolution, we still do coarse optimization first to explore better regions
    if n_points > 40:
        # Reduce number of iterations for efficiency
        max_iter = 200
    
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Solve subproblem: minimize surrogate with penalty
        def obj(h):
            corr = np.correlate(h, 1 - h, mode='full')
            # Penalize violation of constraints
            max_corr = np.max(corr)
            # Simple smooth approximation to max
            penalty = 0.1 * (max_corr - 0.3)**2
            return np.sum(corr**2) * dx**2 + penalty
        
        try:
            # Use SLSQP for local optimization with current h
            res = minimize(obj, h, method='SLSQP', 
                          bounds=[(0, 1) for _ in range(n_points)],
                          options={'maxiter': 10, 'ftol': 1e-6})
            if res.success:
                h_new = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_new, dx)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                h = h_new
        except:
            pass
            
        # Simple local perturbation to escape local minima
        if iteration % 10 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed = project_to_feasible(h + noise, n_points)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
                h = h_perturbed

    return best_h, best_c5
