import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def get_convex_relaxation(h, n_points, dx):
    """Create a convex relaxation of the maximum overlap"""
    # Create correlation matrix
    corr = np.correlate(h, 1 - h, mode='full')
    # Use a smooth approximation for max
    return np.sum(corr**2) * dx**2

def genetic_mutation(h, mutation_rate=0.1):
    """Apply genetic mutation to h"""
    h_new = h.copy()
    n_points = len(h)
    # Randomly select indices to mutate
    indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    # Apply small random changes
    for i in indices:
        # Add small perturbation
        h_new[i] += np.random.normal(0, 0.02)
    return h_new

def optimize_with_cvxpy(h_start, n_points, dx, time_limit):
    """Optimize using convex optimization"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try convex relaxation approach
    # Cauchy-Schwarz or similar approaches
    
    # Simplified: use gradient-based method with adjusted objective
    def obj(h):
        return get_convex_relaxation(h, n_points, dx)
    
    def grad(h):
        eps = 1e-6
        grad = np.zeros_like(h)
        base_obj = obj(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            obj_plus = obj(h_plus)
            grad[i] = (obj_plus - base_obj) / eps
        return grad
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try to optimize
    for _ in range(3):
        if time() - start_time > time_limit:
            break
        try:
            h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)
            res = minimize(obj, h_perturbed, method='SLSQP', bounds=bounds, constraints=cons, jac=grad, options={'maxiter': 100}) 
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Determine starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction
        n_points = 30
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Use a coarse-to-fine resolution approach with genetic mutations
    resolutions = [15, 30, 45, 60, 75]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_current = h_start.copy()
        
        # Interpolation if needed
        if len(h_start) != n_points:
            h_current = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
        
        h_current = project_to_feasible(h_current, n_points)
        
        # Genetic algorithm steps
        for gen in range(30):  # 30 generations
            if time() - start_time > budget_s * 0.95:
                break
            # Create new population
            population = [h_current]
            # Add some mutations
            for _ in range(4):
                h_mut = genetic_mutation(h_current)
                h_mut = project_to_feasible(h_mut, n_points)
                population.append(h_mut)
            
            # Optimize each individual
            for h in population:
                h_opt, c5_opt = optimize_with_cvxpy(h, n_points, dx, 1.0)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n_points = n_points
            
            # Crossover between top performers
            # Select top 3
            population_scores = [(h, evaluate_erdos_solution(h, dx)) for h in population]
            population_scores.sort(key=lambda x: x[1])
            top3 = [h for h, _ in population_scores[:3]]
            
            # Simple crossover
            if len(top3) >= 2:
                h1 = top3[0]
                h2 = top3[1]
                crossover_point = n_points // 2
                h_child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                h_child = project_to_feasible(h_child, n_points)
                h_current = h_child
            
        h_start = best_h
        
    return (list(best_h), best_c5, best_n_points)
