import numpy as np
from scipy.optimize import differential_evolution
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute the overlap integral using convolution
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Start with a coarse resolution
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define objective function
    def objective(h):
        h = project_to_feasible(h, n_points)
        return evaluate_erdos_solution(h, 2.0 / n_points)
    
    # Use differential evolution for global search
    # Iterate through different resolutions
    resolutions = [30, 45, 60, 75]
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break

        dx = 2.0 / n_points
        
        # Interpolate or create coarse grid
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Optimize at current resolution
        # Use differential evolution with adaptive bounds
        try:
            res = differential_evolution(
                objective, 
                bounds, 
                maxiter=100, 
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed,
                disp=False
            )
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n_points = n_points
        except:
            pass
        
        # Early exit
        if time() - start_time > budget_s * 0.95:
            break
        
        # Update starting point for next resolution
        h_start = project_to_feasible(best_h, n_points)
    
    # Final optimization at best resolution level
    bounds = [(0, 1) for _ in range(best_n_points)]
    try:
        res = differential_evolution(
            objective, 
            bounds, 
            maxiter=200, 
            popsize=20, 
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            disp=False
        )
        if res.success:
            h_opt = project_to_feasible(res.x, best_n_points)
            c5_opt = evaluate_erdos_solution(h_opt, 2.0 / best_n_points)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
    except:
        pass
    
    return (list(best_h), best_c5, best_n_points)
