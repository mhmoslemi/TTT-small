import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        x_old = np.linspace(0, 1, len(h_vec))
        x_new = np.linspace(0, 1, target_len)
        return np.interp(x_new, x_old, h_vec)
    
    # Multi-resolution optimization
    resolutions = [5, 10, 20, 40, n_points]
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    for i in range(len(resolutions)-1):
        current_n = resolutions[i]
        next_n = resolutions[i+1]
        # Start with current h or a random one
        if current_n == 5:
            h_current = np.array([0.5, 0.5, 0.5, 0.5, 0.5])
        elif current_n == 10:
            h_current = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
        elif current_n == 20:
            h_current = np.array([0.5]*20)
        elif current_n == 40:
            h_current = np.array([0.5]*40)
        else:
            h_current = best_h.copy()
        
        # Project to start point
        h_current = project_feasible(h_current)
        
        # Adam optimizer parameters
        m = np.zeros_like(h_current)
        v = np.zeros_like(h_current)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        learning_rate = 0.01
        t = 0
        
        last_improvement = 0
        iteration = 0
        
        while time.time() - start_time < budget_s - 1 and (iteration - last_improvement) < 1000:
            iteration += 1
            t += 1
            
            # Compute overlaps to identify k contributing to max
            h_projected = project_feasible(h_current)
            conv = np.correlate(h_projected, 1 - h_projected, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
            # Get top 5 mostly contributing positions
            top_k = np.argpartition(overlaps, -5)[-5:]
            
            # Gradient approximation
            gradient = np.zeros_like(h_projected)
            eps = 1e-4
            for i_idx in range(current_n):
                h_plus = h_projected.copy()
                h_plus[i_idx] += eps
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                h_minus = h_projected.copy()
                h_minus[i_idx] -= eps
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                gradient[i_idx] = (C_plus - C_minus) / (2 * eps)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * gradient**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_current = h_current - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
            
            # Project to feasible region
            h_current = project_feasible(h_current)
            
            # Evaluate current
            current_C5 = compute_overlap(h_current)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current.copy()
                last_improvement = iteration
                # Increase learning rate
                learning_rate = min(learning_rate * 1.05, 0.1)
            else:
                # Decrease learning rate
                learning_rate = max(learning_rate * 0.99, 1e-5)
    
    # Final finer optimization at best resolution
    h_current = best_h.copy()
    max_iter = 2000
    last_improvement = 0
    iteration = 0
    final_lr = 0.005
    
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < max_iter:
        iteration += 1
        # Compute overlaps
        h_projected = project_feasible(h_current)
        conv = np.correlate(h_projected, 1 - h_projected, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        top_k = np.argpartition(overlaps, -5)[-5:]
        
        # Gradient approximation
        gradient = np.zeros_like(h_projected)
        eps = 1e-4
        for i_idx in range(len(h_projected)):
            h_plus = h_projected.copy()
            h_plus[i_idx] += eps
            h_plus = project_feasible(h_plus)
            C_plus = compute_overlap(h_plus)
            
            h_minus = h_projected.copy()
            h_minus[i_idx] -= eps
            h_minus = project_feasible(h_minus)
            C_minus = compute_overlap(h_minus)
            
            gradient[i_idx] = (C_plus - C_minus) / (2 * eps)
        
        # Update
        h_current = h_current - final_lr * gradient
        h_current = project_feasible(h_current)
        
        current_C5 = compute_overlap(h_current)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_current.copy()
            last_improvement = iteration
        if iteration - last_improvement > 200:
            final_lr *= 0.95
            if final_lr < 1e-6:
                break
    
    return (best_h.tolist(), best_C5, n_points)
