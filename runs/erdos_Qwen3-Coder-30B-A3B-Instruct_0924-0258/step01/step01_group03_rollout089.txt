import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def nelder_mead_step_simplex(points, values, alpha=1.0, beta=2.0, gamma=0.5, delta=0.5):
    """Perform one Nelder-Mead step on simplex."""
    n = len(points) - 1  # number of vertices
    # Order by function values
    indices = np.argsort(values)
    points = [points[i] for i in indices]
    values = [values[i] for i in indices]
    
    # Best, worst, second worst
    x_best = points[0]
    x_worst = points[-1]
    x_second_worst = points[-2]
    
    # Compute centroid (excluding worst)
    x_centroid = np.mean(points[:-1], axis=0)
    
    # Reflection
    x_reflect = x_centroid + alpha * (x_centroid - x_worst)
    x_reflect = project_to_feasible(x_reflect, len(x_reflect) / 2)
    
    f_reflect = evaluate(x_reflect, 2.0 / len(x_reflect))
    
    if values[0] <= f_reflect < values[-2]:
        # Accept reflection
        points[-1] = x_reflect
        values[-1] = f_reflect
    elif f_reflect < values[0]:
        # Expansion
        x_expand = x_centroid + beta * (x_reflect - x_centroid)
        x_expand = project_to_feasible(x_expand, len(x_expand) / 2)
        f_expand = evaluate(x_expand, 2.0 / len(x_expand))
        
        if f_expand < f_reflect:
            points[-1] = x_expand
            values[-1] = f_expand
        else:
            points[-1] = x_reflect
            values[-1] = f_reflect
    else:
        # Contraction
        if f_reflect < values[-1]:
            # Outside contraction
            x_contract = x_centroid + gamma * (x_reflect - x_centroid)
        else:
            # Inside contraction
            x_contract = x_centroid + gamma * (x_worst - x_centroid)
        x_contract = project_to_feasible(x_contract, len(x_contract) / 2)
        f_contract = evaluate(x_contract, 2.0 / len(x_contract))
        
        if f_contract < values[-1]:
            points[-1] = x_contract
            values[-1] = f_contract
        else:
            # Shrink
            for i in range(1, len(points)):
                points[i] = points[0] + delta * (points[i] - points[0])
                points[i] = project_to_feasible(points[i], len(points[i]) / 2)
    
    return points, values
