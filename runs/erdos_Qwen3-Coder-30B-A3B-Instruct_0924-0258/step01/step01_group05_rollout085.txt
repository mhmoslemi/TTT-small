import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h_vec, dx):
    """Compute C5 using full correlation."""
    corr = correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(corr) * dx

def project_feasible(h_vec, target_sum):
    """Project h_vec to feasible set: [0,1] and sum = target_sum."""
    h_vec = np.clip(h_vec, 0, 1)
    h_sum = np.sum(h_vec)
    if h_sum > 0:
        h_vec = h_vec * target_sum / h_sum
    return h_vec

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s - 0.1
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial evaluation
    current_C5 = evaluate_C5(h, dx)
    best_C5 = current_C5
    best_h = h.copy()
    
    # Annealing parameters
    T0 = 0.1
    cooling_rate = np.log(10) / budget_s
    T_min = 1e-5
    sigma = 0.1
    acceptance_count = 0
    total_attempts = 0

    t = 0
    while time.time() - start_time < time_limit:
        t += 1
        
        # Temperature
        T = max(T_min, T0 * np.exp(-cooling_rate * t))
        
        # Identify high-priority intervals
        current_corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(current_corr)
        max_lag = max_lag_idx - (n_points - 1)
        # For exploration, use a broader set of intervals
        high_priority = np.linspace(0, n_points-1, min(8, n_points), dtype=int)
        
        # Perturbation
        h_new = h.copy()
        # Add Gaussian noise to high priority intervals
        for i in high_priority:
            h_new[i] += np.random.normal(0, sigma)
        # Optionally flip a few intervals 
        flip_indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        for i in flip_indices:
            h_new[i] = 1 - h_new[i]
        
        # Feasibility projection
        h_new = project_feasible(h_new, target_sum)
        
        # New evaluation
        new_C5 = evaluate_C5(h_new, dx)
        total_attempts += 1
        
        # Acceptance criterion
        delta = new_C5 - current_C5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new.copy()
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
        
        # Adaptive step size
        if total_attempts > 0 and total_attempts % 20 == 0:
            acceptance_rate = acceptance_count / total_attempts if total_attempts > 0 else 0
            if acceptance_rate < 0.1:
                sigma = min(0.5, sigma * 1.2)
            elif acceptance_rate > 0.4:
                sigma = max(0.005, sigma * 0.8)
            acceptance_count = 0
            total_attempts = 0

    # Final projection and evaluation
    final_h = project_feasible(best_h, target_sum)
    final_C5 = evaluate_C5(final_h, dx)
    
    return list(final_h), final_C5, n_points
