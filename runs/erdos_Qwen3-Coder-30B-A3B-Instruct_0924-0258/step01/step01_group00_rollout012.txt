import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Start with a simple construction
        n0 = 30
        h_start = np.ones(n0) * 0.5
        # Perturb a bit to avoid degeneracy
        h_start += np.random.normal(0, 0.05, n0)
        h_start = project_to_feasible(h_start, n0)
    
    # Base resolution and time allocation
    base_resolutions = [20, 30, 40, 50]
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Sequential resolution improvement
    for n_points in base_resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate to new resolution
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Optimization with multiple restarts
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        # Use a different approach: direct minimization using a smooth approximation
        def smooth_max_approx(h, dx):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.sum(corr**2) * dx**2

        def smooth_max_grad(h, dx):
            eps = 1e-8
            grad = np.zeros_like(h)
            base_obj = smooth_max_approx(h, dx)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, len(h))
                obj_plus = smooth_max_approx(h_plus, dx)
                grad[i] = (obj_plus - base_obj) / eps
            return grad
        
        # Try multiple restarts with different initializations
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            # Random perturbations
            h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            
            try:
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Optimize with L-BFGS-B
                res = minimize(smooth_max_approx, h_perturbed, 
                               args=(dx,), method='L-BFGS-B', 
                               bounds=bounds, constraints=cons,
                               jac=smooth_max_grad, options={'maxiter': 50})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5_level:
                        best_c5_level = c5_opt
                        best_h_level = h_opt.copy()
            except:
                pass
        
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
            
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
