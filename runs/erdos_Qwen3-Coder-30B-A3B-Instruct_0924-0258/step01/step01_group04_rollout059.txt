import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    n_particles = 100
    w_start = 0.7
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    for _ in range(n_particles):
        # Initialize near the initial solution
        h_init = h + np.random.normal(0, 0.05, n_points)
        h_init = project_feasible(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        pbest_positions.append(h_init.copy())
        pbest_scores.append(compute_C5(h_init))
    
    # Global best
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Adaptive inertia
        w = w_start - (w_start - w_end) * (iteration / 1000)
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_cog = c1 * r1 * (pbest_positions[i] - particles[i])
            vel_social = c2 * r2 * (gbest_position - particles[i])
            velocities[i] = w * velocities[i] + vel_cog + vel_social
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_speed, max_speed)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            score = compute_C5(particles[i])
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
            
            # Update global best
            if score < gbest_score:
                gbest_score = score
                gbest_position = particles[i].copy()
                
        # Periodic check
        if iteration % 50 == 0:
            current_C5 = compute_C5(gbest_position)
            if current_C5 < gbest_score:
                gbest_score = current_C5
    
    # Final check
    final_C5 = compute_C5(gbest_position)
    if final_C5 < gbest_score:
        gbest_score = final_C5
    
    return (gbest_position.tolist(), gbest_score, n_points)
