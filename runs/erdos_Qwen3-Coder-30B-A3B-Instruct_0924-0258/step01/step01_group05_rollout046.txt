import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient for the k_max overlap
        n = len(h_vec)
        grad = np.zeros(n)
        # For each position j, the contribution to correlation with lag k_max
        for j in range(n):
            if k_max >= 0:
                # This h[j] is correlated with h[(j+k_max) % n]
                grad[j] -= h_vec[(j + k_max) % n]
            else:
                # This h[j] is correlated with h[(j-k_max) % n]  
                grad[j] -= h_vec[(j - k_max) % n]
        return grad
    
    h_current = h.copy()
    C5_current = compute_C5(h_current)
    best_h = h_current.copy()
    best_C5 = C5_current
    
    # Gradient descent with adaptive learning rate
    alpha = 0.1
    stagnation_count = 0
    max_stagnation = 50
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        # Compute all overlaps
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient(h_current, max_lag)
        
        # Try different step sizes
        step_sizes = [alpha, alpha*0.5, alpha*0.1]
        best_new_h = h_current.copy()
        best_new_C5 = C5_current
        best_alpha_used = 0
        
        for step_alpha in step_sizes:
            candidate_h = h_current - step_alpha * grad
            candidate_h = project_feasible(candidate_h)
            new_C5 = compute_C5(candidate_h)
            
            if new_C5 < best_new_C5:
                best_new_C5 = new_C5
                best_new_h = candidate_h.copy()
                best_alpha_used = step_alpha
                
        # Update if improved
        if best_new_C5 < C5_current:
            h_current = best_new_h.copy()
            C5_current = best_new_C5
            stagnation_count = 0
            if C5_current < best_C5:
                best_C5 = C5_current
                best_h = h_current.copy()
        else:
            stagnation_count += 1
            if stagnation_count >= max_stagnation:
                # Reduce learning rate and reset stagnation
                alpha *= 0.5
                stagnation_count = 0
                if alpha < 1e-6:
                    break
            # Add noise and reset
            noise = np.random.normal(0, 0.01, n_points)
            h_current += noise
            h_current = project_feasible(h_current)
        
        # Occasionally reinitialize with random perturbation
        if iteration % 50 == 0:
            noise = np.random.normal(0, 0.02, n_points)
            h_current += noise
            h_current = project_feasible(h_current)
    
    # Final projection and evaluation
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
