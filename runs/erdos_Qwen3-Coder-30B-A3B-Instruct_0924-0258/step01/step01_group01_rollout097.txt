import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    max_iter_per_res = 1000
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    res_iter = 0
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        
        # Adjust learning rate and iterations
        lr = 0.01 * (0.99 ** res_iter)
        for i in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            c5 = evaluate(h, dx)
            corr = correlate(h, 1-h, mode='full')
            k_max = np.argmax(corr)
            # Compute the shift where max occurs
            # full correlation length: 2*n - 1
            # actual lag = k_max - (n-1)
            lag = k_max - (n_points - 1)
            
            # Compute gradient for this lag
            # This is a simplified gradient using one lag
            # For simplicity we approximate gradient with finite differences
            grad = np.zeros_like(h)
            
            # For lag k, we want to compute d/dh_i of integral h(x)(1-h(x+k))
            # Which is 1-h(x+k) when x=i, if i+k<n (and i<n), else 0 if i+k>=n
            # so we sum over x such that x+i = k: 1-h(i)
            # loop over each element of h
            for j in range(n_points):
                # contribute to h[j] based on k
                # When h[j] contributes to integral, it's at h[j] * (1-h[j+k])
                # Gradient contribution of h[j] to max correlation
                # When doing actual gradient computation over the correlation:
                # ∂/∂h[i] ∫ h(x) (1 - h(x+k)) dx = 1 - h[i+k] if i+k < n  
                # All except when we correlate h[i+k] with h[i], if that's a valid index.
                # We want to handle the case where the integral with k=lag is the maximum.
                # Since correlation computes ∫ h(x) * (1 - h(x + k))  dx
                # For fixed k we have k values of x that contribute,
                # we need to focus only gradient to maximizer.
                # Direct computation: 
                # ∂h(x) / ∂h(j) = 1 if x=j else 0
                # ∂∫ h(x)(1-h(x+k)) / ∂h(j)  = 
                # sum_x  (1-h(x+k)) * δ(x=j) = (1-h(j+k)) if j+k < n else 0
                # For each x, it's 1-h(x+k) if x=j, so we compute the contribution.
                # We only need j and k.  This backward difference approximates:
                # j+k increments which signal the term 1-h(x)
                # Let's make 1D treatment: add +1 at j, -1 at j+k if valid
                # But we must compute proper mapping to full correlation
                #[j] contributes 1-h[j+k] to the derivative w.r.t. that correlation
                # Which equals to taking the contribution. Let's do:
                if (j + lag) >= 0 and (j + lag) < n_points:
                    grad[j] = 1 - h[j + lag]
                else:
                    grad[j] = 0
                
            # Normalize gradient for more stable update
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                grad = grad / grad_norm
            
            # Update step
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            if c5_new < c5:
                h = h_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
            iter_count += 1
            
        # Double resolution if improvement happens
        if n_points < n_high:
            h_long = np.interp(np.linspace(0, 2, 2*n_points),
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            res_iter += 1
        else:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
