import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available

        n_points = len(h0)
    else:

    # Set dx
    dx = 2.0 / n_points
    
    # Initialize h
    h = h0.copy()
    h = h * (n_points / 2) / np.sum(h)  # scale to sum = n_points/2
    h = np.clip(h, 0, 1)  # clamp to [0,1]
    
    # Gradient-based optimization with adaptive learning rate
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Parameters
    learning_rate = 0.01
    patience = 50
    patience_counter = 0
    last_improvement = 0
    n_iter = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s and n_iter < max_iter:
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        # Find max correlation index
        k_max = np.argmax(np.abs(corr))
        # Get actual shift value
        shift = k_max - (len(h) - 1)
        
        # Construct gradient
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            if shift >= 0:
                j = i + shift
                if j < len(h):
                    gradient[i] += (1 - h[j])
            else:
                j = i + shift
                if j >= 0:
                    gradient[i] += (1 - h[j])
        
        # Update h
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Project to feasible region
        current_sum = np.sum(h)
        h = h * (n_points / 2) / current_sum
        
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
            last_improvement = n_iter
        else:
            patience_counter += 1
            if patience_counter > patience:
                # Add small perturbation
                h += np.random.uniform(-1e-3, 1e-3, len(h))
                h = np.clip(h, 0, 1)
                current_sum = np.sum(h)
                h = h * (n_points / 2) / current_sum
                patience_counter = 0
                
        # Adapt learning rate
        if n_iter - last_improvement > 100:
            learning_rate *= 0.9
            last_improvement = n_iter
            
        n_iter += 1
        
        # Safety check for NaN or Inf
        if np.any(np.isnan(h)) or np.any(np.isinf(h)):
            h = best_h.copy()
            
    return (best_h.tolist(), float(best_c5), n_points)
