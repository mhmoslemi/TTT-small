import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Create a structured starting point
        h_start = np.ones(30) * 0.5
        
    n_points = len(h_start)
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points

    def objective(h):
        return -evaluate_erdos_solution(h, dx)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    # Try multiple refining strategies
    resolutions = [20, 30, 40, 50]
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_current = project_to_feasible(np.interp(np.linspace(0, 2, n_points), 
                                                  np.linspace(0, 2, len(h_start)), 
                                                  h_start), n_points)
        
        # Multiple restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.98:
                break
                
            # Add noise to starting configuration
            h_pert = project_to_feasible(h_current + np.random.normal(0, 0.05, n_points), n_points)
            
            # Local optimization with trust-constr
            try:
                res = minimize(objective, 
                               h_pert, 
                               method='trust-constr',
                               bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 200, 'verbose': 0})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5 = evaluate_erdos_solution(h_opt, dx)
                    if c5 < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5
                        best_n_points = n_points
            except:
                pass
            
            # Pattern-based perturbation
            h_pert = h_current.copy()
            # Simple pattern - increase/decrease in regions
            for i in range(0, n_points, 4):
                if i + 3 < n_points:
                    avg = np.mean(h_pert[i:i+4])
                    h_pert[i:i+4] = np.clip(h_pert[i:i+4] + np.random.normal(0, 0.02, 4), 0, 1)
            h_pert = project_to_feasible(h_pert, n_points)
            c5 = evaluate_erdos_solution(h_pert, dx)
            if c5 < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5
                best_n_points = n_points
                
        h_start = best_h.copy()
        
        # Early exit if we've found a very good solution
        if best_c5 <= 0.3807:
            break

    return (list(best_h), best_c5, best_n_points)
