import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def greedy_adjustment(h, n_points, dx, target=0.38):
    """Perform greedy adjustment to reduce maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    c_max = np.max(corr)
    if c_max <= target:
        return h.copy()

    # Find where maximum overlap occurs
    max_idx = np.argmax(corr)
    # Map back to shift value k
    k = max_idx - (n_points - 1)  # shift in [0, 2]

    # Identify top contributing positions
    # For overlap at shift k, we have h[i] * (1 - h[i+k]) in convolution
    # We want to reduce h[i] where h[i+k] is close to 1 and h[i] is large
    contributions = np.zeros(n_points)
    for i in range(n_points):
        j = i + k
        if 0 <= j < n_points:
            contributions[i] = h[i] * (1 - h[j])
    
    # Top 10% by contribution
    top_indices = np.argsort(contributions)[-max(1, n_points // 10):]
    alpha = min(0.1, (c_max - target) / n_points)
    
    # Decrease h in top contributions
    h_new = h.copy()
    for i in top_indices:
        h_new[i] = max(0, h_new[i] - alpha)
    
    # Ensure feasibility
    h_new = project_to_feasible(h_new, n_points)
    return h_new

    # Iterative greedy improvement
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time() - start_time < budget_s * 0.9:
        iterations += 1
        
        # Perform greedy adjustment
        h_new = greedy_adjustment(h, n_points, dx, target=0.38)

        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # If improved, accept changes and reset counter
        if c5_new < best_c5 - 1e-8:
            h = h_new
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally perturb to escape local minima
        if no_improvement_count > 20 and time() - start_time < budget_s * 0.8:
            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
            no_improvement_count = 0

        if iterations % 100 == 0 and n_points < 200 and time() - start_time < budget_s * 0.8:
            n_points_new = min(n_points + 15, 200)
