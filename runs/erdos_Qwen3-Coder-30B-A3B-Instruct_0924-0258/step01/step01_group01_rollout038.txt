import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def admm_update_h(h, z, lambda_z, rho, dx):
    # Simplified update for h: solve quadratic approximation
    # Minimize: ||h - (z + lambda_z/rho)||^2 + rho/2 * ||h - (1 - z)||^2
    # This is a least squares problem with linear constraints
    # Solved using projected gradient
    # 解析解: h = (z + lambda_z/rho + rho*(1 - z)) / (1 + rho)
    h_new = (z + lambda_z/rho + rho*(1 - z)) / (1 + rho)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def admm_update_z(h, z, lambda_z, rho):
    # z = 1 - h + lambda_z / rho
    z_new = 1 - h + lambda_z / rho
    return z_new

def admm_update_lagrangian(h, z, lambda_z, mu, rho, dx):
    # Update dual variables
    lambda_z += rho * (z - (1 - h))
    mu += rho * (np.sum(h) - len(h)/2)
    return lambda_z, mu

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution
    n_low = 10
    n_high = 100
    n_points = n_low
    dx = 2.0 / n_points
    max_iterations = 100
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    lambda_z = np.zeros(n_points)
    mu = 0.0
    
    # Timing and optimization loop
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
            
            # Update h using ADMM
            z = 1 - h + lambda_z / rho
            h = admm_update_h(h, z, lambda_z, rho, dx)
            h = project_to_constraints(h, dx)
            
            # Update z
            z = admm_update_z(h, z, lambda_z, rho)
            
            # Update dual variables
            lambda_z, mu = admm_update_lagrangian(h, z, lambda_z, mu, rho, dx)
            
            # Evaluate current solution
            c5 = evaluate(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
        # Resolution refinement
        if n_points < n_high:
            # Interpolate to new resolution
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points *= 2
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            # Reset ADMM variables
            lambda_z = np.zeros(n_points)
            mu = 0.0
            best_c5 = evaluate(h, dx)
        else:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
