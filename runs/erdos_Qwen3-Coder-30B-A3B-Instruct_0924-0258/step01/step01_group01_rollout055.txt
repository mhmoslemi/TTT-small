import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

    # Smooth approximation to max
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Enlarge budget slightly to catch final optimizations
    time_budget = 0.98 * budget_s

    # Start with a reasonable resolution 
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        # Interpolate the initial construction to current resolution
        h = np.interp(np.linspace(0, 2, n_points), 
                      np.linspace(0, 2, len(initial_h_values)), 
                      initial_h_values)
        h = project_to_constraints(h, dx)  
    else:
        # Random initial
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Use CVXPY for better constrained optimization
    try:
        import cvxpy as cp
        
        # Define optimization variables
        h_vars = cp.Variable(n_points)
        objective = cp.Minimize(cp.max(cp.conv(h_vars, 1 - h_vars)))
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2.0
        ]
        
        # Problem
        problem = cp.Problem(objective, constraints)
        
        max_iter = 500
        iter_count = 0
        
        while iter_count < max_iter and (time.time() - start_time) < time_budget:
            if time.time() - start_time > time_budget:
                break

            try:
                problem.solve(solver=cp.CVXOPT, verbose=False)
                if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_opt = h_vars.value
                    h_new = project_to_constraints(h_opt, dx)
                    c5_new = evaluate(h_new, dx)
                    
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            except Exception as e:
                pass
            
            # Random perturbation with clipping 
            noise = np.random.normal(0, 0.02, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            c5_new = evaluate(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new.copy()
            else:
                h = h_new.copy()
            
            iter_count += 1
            
    except ImportError:
        # Fallback to simpler optimization
        max_iter = 2000
        iter_count = 0
        T = 0.05
        decay = 0.999
        
        while iter_count < max_iter and (time.time() - start_time) < time_budget:
            if time.time() - start_time > time_budget:
                break
                
            # Perturb with noise
            noise = np.random.normal(0, T, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraints(h_new, dx)
            c5_new = evaluate(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new.copy()
            else:
                # Accept sometimes
                if np.random.rand() < np.exp(-(c5_new - best_c5) / T):
                    h = h_new.copy()
            
            iter_count += 1
            T *= decay

    return best_h.tolist(), best_c5, len(best_h)
