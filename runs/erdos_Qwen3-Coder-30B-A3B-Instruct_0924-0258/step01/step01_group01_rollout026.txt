import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.uniform(0, 1, 10)
    n_points = len(h)
    dx = 2.0 / n_points
    h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Parameters
    n_particles = 100
    max_iterations = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_max = 0.05
    step_size = 0.01
    
    # Resolution schedule
    resolutions = [10, 20, 40, 70]
    time_per_resolution = budget_s / len(resolutions) * 0.95
    
    for res in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # If we need to interpolate
        if res > n_points:
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h)
            h = h_new
            n_points = res
            dx = 2.0 / n_points
            
        # Particle Swarm Optimization
        particles = []
        velocities = []
        best_positions = []
        best_values = []
        
        for _ in range(n_particles):
            # Initialize particles with small random perturbations
            h_p = h + np.random.normal(0, 0.01, n_points)
            h_p = np.clip(h_p, 0, 1)
            h_p = project_to_constraints(h_p, dx)
            particles.append(h_p)
            velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
            val = evaluate(h_p, dx)
            best_positions.append(h_p.copy())
            best_values.append(val)
        
        # PSO main loop
        for iter_num in range(max_iterations):
            if time.time() - start_time > start_time + time_per_resolution:
                break
                
            for i in range(n_particles):
                # Update velocities
                r1, r2 = np.random.random(2)
                cognitive_component = cognitive * r1 * (best_positions[i] - particles[i])
                social_component = social * r2 * (best_positions[np.argmin(best_values)] - particles[i])
                velocities[i] = inertia * velocities[i] + cognitive_component + social_component
                
                # Clamp velocities
                velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
                
                # Update positions
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = project_to_constraints(particles[i], dx)
                
                # Evaluate
                val = evaluate(particles[i], dx)
                
                # Update personal best
                if val < best_values[i]:
                    best_values[i] = val
                    best_positions[i] = particles[i].copy()
                
                # Update global best
                if val < best_c5:
                    best_c5 = val
                    best_h = particles[i].copy()
        
        h = best_h.copy()
        n_points = len(h)
        dx = 2.0 / n_points
        
        # Gradient descent refinement (few iterations to save time)
        for _ in range(200):
            if time.time() - start_time > start_time + time_per_resolution:
                break
            grad = np.zeros(n_points)
            # Compute gradient using finite differences
            eps = 1e-6
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_constraints(h_plus, dx)
                val_plus = evaluate(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_constraints(h_minus, dx)
                val_minus = evaluate(h_minus, dx)
                
                grad[i] = (val_plus - val_minus) / (2 * eps)
            
            # Update
            h = h - step_size * grad
            h = project_to_constraints(h, dx)
            
            # Update best if improved
            new_c5 = evaluate(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h.tolist(), best_c5, len(best_h)
