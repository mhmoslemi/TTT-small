import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def grid_search(h_vec, iterations=20):
        """Better local search using coordinate descent"""
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(best_h)
        
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            improved = False
            for i in range(n_points):
                if time.time() - start_time > budget_s - 1:
                    break
                # Try small perturbations
                for pert in [-0.01, -0.005, 0.005, 0.01]:
                    h_trial = h_current.copy()
                    h_trial[i] += pert
                    h_trial = project_h(h_trial)
                    C5_trial = compute_C5(h_trial)
                    if C5_trial < best_C5:
                        best_C5 = C5_trial
                        best_h = h_trial.copy()
                        improved = True
                        h_current = h_trial.copy()
            if not improved:
                break
        return best_h, best_C5
    
    # Initialize best solution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Multi-resolution optimization
    resolutions = [5, 10, 20, 40, n_points]
    current_h = project_h(h)
    
    for i, res in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create or upsample current solution
        if i == 0:
            current_h = np.ones(res) * (target_sum / res)
        else:
            # Upsample with linear interpolation
            old_points = len(current_h)
            x_old = np.linspace(0, 2, old_points)
            x_new = np.linspace(0, 2, res)
            current_h = np.interp(x_new, x_old, current_h)
            current_h = project_h(current_h)
            
        # Grid search for local improvement
        current_h, C5 = grid_search(current_h, 30)
        if C5 < best_C5:
            best_C5 = C5
            best_h = current_h.copy()
        
        # Fine-tune with projected gradient descent
        for iteration in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Compute gradient using finite differences with top contributors
            h_vec = current_h.copy()
            conv = np.correlate(h_vec, 1 - h_vec, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
            top_k = np.argsort(overlaps)[-5:][::-1]
            
            gradient = np.zeros_like(h_vec)
            eps = 1e-4
            for idx in range(n_points):
                if time.time() - start_time > budget_s - 1:
                    break
                h_plus = h_vec.copy()
                h_plus[idx] += eps
                h_plus = project_h(h_plus)
                C_plus = compute_C5(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[idx] -= eps
                h_minus = project_h(h_minus)
                C_minus = compute_C5(h_minus)
                
                gradient[idx] = (C_plus - C_minus) / (2 * eps)
            
            # Adam-like update
            learning_rate = 0.01
            current_h = current_h - learning_rate * gradient
            current_h = project_h(current_h)
            
            # Check if we improved
            new_C5 = compute_C5(current_h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = current_h.copy()
            elif iteration % 20 == 0:  # Periodic check for early stopping
                _, new_C5 = grid_search(current_h, 2)
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = current_h.copy()
    
    # Final fine-tune
    final_h, final_C5 = grid_search(best_h, 20)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
