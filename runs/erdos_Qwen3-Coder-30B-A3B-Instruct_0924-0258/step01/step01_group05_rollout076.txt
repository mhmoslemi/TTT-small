import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def lhs_sample(n_points, n_samples):
        # Latin hypercube sampling for diverse initial points
        samples = []
        for _ in range(n_samples):
            h_sample = np.random.rand(n_points)
            # Shuffle each dimension
            for i in range(n_points):
                h_sample[i] = np.random.rand()
            samples.append(h_sample)
        return samples
    
    def coordinate_descent_step(h_vec, current_C5, step_size=0.05):
        h_best = h_vec.copy()
        C5_best = current_C5
        
        # Try small perturbations
        for i in range(len(h_vec)):
            for delta in [-step_size, step_size]:
                h_trial = h_vec.copy()
                h_trial[i] += delta
                h_trial = project_feasible(h_trial)
                C5_trial = compute_C5(h_trial)
                if C5_trial < C5_best:
                    C5_best = C5_trial
                    h_best = h_trial.copy()
        
        return h_best, C5_best
    
    def pairwise_swap(h_vec, current_C5):
        h_best = h_vec.copy()
        C5_best = current_C5
        n = len(h_vec)
        
        # Try swapping pairs of intervals
        for _ in range(10):
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            h_trial = h_vec.copy()
            h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
            h_trial = project_feasible(h_trial)
            C5_trial = compute_C5(h_trial)
            if C5_trial < C5_best:
                C5_best = C5_trial
                h_best = h_trial.copy()
        
        return h_best, C5_best
    
    # Initialize multiple candidates
    initial_h_candidates = []
    if 'initial_h_values' in globals():
        h_seed = np.array(initial_h_values, dtype=float)
        initial_h_candidates.append(h_seed)
    else:
        h_seed = np.zeros(n_points)
        h_seed[:n_points//2] = 1.0
        initial_h_candidates.append(h_seed)
        
    # Generate Latin hypercube samples
    lhs_samples = lhs_sample(n_points, 4)
    for sample in lhs_samples:
        h_proj = project_feasible(sample)
        initial_h_candidates.append(h_proj)
    
    # Find best initial candidate
    best_h = None
    C5_best = float('inf')
    for h_candidate in initial_h_candidates:
        C5 = compute_C5(h_candidate)
        if C5 < C5_best:
            C5_best = C5
            best_h = h_candidate.copy()
    
    # Optimizer loop
    last_improvement = time.time()
    max_iter = 1000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 0.1 and iter_count < max_iter:
        iter_count += 1
        current_time = time.time()
        
        # Perform local optimizations
        h_opt1, C5_opt1 = coordinate_descent_step(best_h, C5_best, step_size=0.1)
        h_opt2, C5_opt2 = pairwise_swap(best_h, C5_best)
        
        # Update best solution
        candidates = [(best_h, C5_best), (h_opt1, C5_opt1), (h_opt2, C5_opt2)]
        for h_cand, C5_cand in candidates:
            if C5_cand < C5_best:
                C5_best = C5_cand
                best_h = h_cand.copy()
                last_improvement = current_time
                
        # If no improvement in a while, try a random restart
        if current_time - last_improvement > 30:
            h_rand = np.random.rand(n_points)
            h_rand = project_feasible(h_rand)
            C5_rand = compute_C5(h_rand)
            if C5_rand < C5_best:
                C5_best = C5_rand
                best_h = h_rand
                last_improvement = current_time
                
        # Occasionally update step size
        if iter_count % 50 == 0:
            # Optionally decrease step size to refine solution
            pass
    
    # Final projection
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
