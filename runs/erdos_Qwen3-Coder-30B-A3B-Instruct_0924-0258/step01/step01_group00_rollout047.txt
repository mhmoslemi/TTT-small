import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(45) / 2
    
    n_points = len(h_start)
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    def adjust_h(h, target_c5, num_iterations=50):
        h = h.copy()
        for _ in range(num_iterations):
            c5 = evaluate_erdos_solution(h, dx)
            if c5 <= target_c5:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n_points - 1)  # shift corresponding to max correlation
            
            # Identify where most of the overlap occurs in the critical shift
            overlap_indices = []
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    overlap_indices.append(i)
            
            # Top 10% of indices by contribution
            contrib = np.array([h[i] * (1 - h[i + k]) if 0 <= i + k < n_points else 0 for i in overlap_indices])
            top_indices = np.argsort(contrib)[-max(1, len(overlap_indices) // 10):]
            
            # Reduce h to reduce overlap at these indices
            adjustment = max(0.01, (c5 - target_c5) / n_points)
            for i in overlap_indices:
                if i in overlap_indices[top_indices]:
                    h[i] = max(0, h[i] - adjustment)
            
            h = project_to_feasible(h, n_points)
            
        return h
    
    # Try multiple resolutions
    resolutions = [30, 45, 60]
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_current = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_current = project_to_feasible(h_current, n_points)
        
        # Try to improve it
        for iteration in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            h_new = adjust_h(h_current, best_c5)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                best_n_points = n_points
            h_current = h_new.copy()
            
            # Add some noise to escape local minima
            if iteration % 5 == 0:
                h_current = project_to_feasible(h_current + np.random.normal(0, 0.01, n_points), n_points)
            
    return (list(best_h), best_c5, best_n_points)
