import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_step(h, dx, step_size=0.01):
    # Analytical gradient for C5 w.r.t. h
    # dC5/dh_i = dx * (1 - 2*h_i) (approximate)
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        # The gradient is approximate at best, but we'll use it directionally
        grad[i] = 1 - 2 * h[i]
    h_new = h - step_size * grad
    return project_to_constraints(h_new, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    resolutions = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Start with initialization or random
        if 'initial_h_values' in globals() and n_points == 10:
            h = np.array(initial_h_values)
            h = h[:n_points]
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_constraints(h, dx)
            
        # Use a hybrid approach: PSO + gradient descent
        # 1) PSO phase
        num_particles = 100
        velocity_max = 0.05
        inertia = 0.7
        c1 = 0.5
        c2 = 0.5
        
        particles = []
        velocities = []
        pbest = []
        pbest_vals = []
        
        for i in range(num_particles):
            if i == 0 and 'initial_h_values' in globals() and n_points == 10:
                p = h.copy()
            else:
                p = np.random.uniform(0, 1, n_points)
                p = project_to_constraints(p, dx)
            particles.append(p.copy())
            velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
            pbest.append(p.copy())
            pbest_vals.append(evaluate(p, dx))
        
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        # PSO iterations
        max_iter_pso = 200
        for iter_pso in range(max_iter_pso):
            if time.time() - start_time > budget_s - 2:
                break
            for i in range(num_particles):
                # Update velocity
                r1, r2 = np.random.uniform(), np.random.uniform()
                velocities[i] = (inertia * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                
                # Clip velocity
                velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_to_constraints(particles[i], dx)
                
                # Evaluate fitness
                current_val = evaluate(particles[i], dx)
                
                # Update personal best
                if current_val < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = current_val
                    
                    # Update global best
                    if current_val < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = current_val
                        
        # 2) Gradient descent refinement
        max_iter_gd = 200
        h_refined = gbest.copy()
        for iter_gd in range(max_iter_gd):
            if time.time() - start_time > budget_s - 2:
                break
            h_new = gradient_step(h_refined, dx, step_size=0.01)
            c5_new = evaluate(h_new, dx)
            if c5_new < evaluate(h_refined, dx):
                h_refined = h_new
            else:
                break
                
        # 3) Interpolation if needed
        if n_points < 70:
            if len(h_refined) == n_points:
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), h_refined)
                h_refined = h_long
                n_points = 2*n_points
                dx = 2.0 / n_points
                h_refined = project_to_constraints(h_refined, dx)
        
        # Final check
        final_c5 = evaluate(h_refined, dx)
        if final_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = final_c5
            
    return best_h.tolist(), best_c5, len(best_h)
