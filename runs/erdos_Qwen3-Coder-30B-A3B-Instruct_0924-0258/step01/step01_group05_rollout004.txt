import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def latin_hypercube_sample(n, n_samples):
        samples = np.zeros((n_samples, n))
        for i in range(n):
            # For each dimension, sample uniformly from intervals
            intervals = np.linspace(0, 1, n_points + 1)
            vals = np.random.uniform(intervals[:-1], intervals[1:])
            samples[:, i] = vals
        return samples
    
    # Generate diverse initial candidates
    candidates = [h.copy()]
    lh_samples = latin_hypercube_sample(n_points, 5)
    for sample in lh_samples:
        candidate = project_feasible(sample * n_points / 2.0)
        candidates.append(candidate)
    
    # Evaluate initial candidates
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    for candidate in candidates:
        c5 = compute_C5(candidate)
        if c5 < best_C5:
            best_C5 = c5
            best_h = candidate.copy()
    
    # Local optimization
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 150
    
    while time.time() - start_time < budget_s - 0.1:
        iter_count += 1
        if no_improvement_count > max_no_improvement:
            # Restart with new random sample
            new_h = np.random.rand(n_points)
            new_h = project_feasible(new_h * n_points / 2.0)
            if compute_C5(new_h) < best_C5:
                best_h = new_h
                best_C5 = compute_C5(new_h)
            no_improvement_count = 0
        
        # Coordinate descent on each variable
        h_current = best_h.copy()
        improved = False
        
        # Try to improve each point
        for i in range(n_points):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            orig_val = h_current[i]
            best_val = orig_val
            best_c5 = best_C5
            
            # Try small variations
            steps = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
            for step in steps:
                h_current[i] = orig_val + step
                h_current = project_feasible(h_current)
                new_c5 = compute_C5(h_current)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_val = h_current[i]
                    improved = True
            
            h_current[i] = best_val
            
        if improved:
            best_C5 = best_c5
            best_h = h_current.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally do swaps
        if iter_count % 10 == 0:
            for _ in range(10):
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                i, j = np.random.choice(n_points, 2, replace=False)
                h_swap = best_h.copy()
                h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
                h_swap = project_feasible(h_swap)
                new_c5 = compute_C5(h_swap)
                if new_c5 < best_C5:
                    best_C5 = new_c5
                    best_h = h_swap.copy()
    
    # Final projection to ensure feasibility
    best_h = project_feasible(best_h)  
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
