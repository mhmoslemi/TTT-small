import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Clamp again in case of over-normalization
    h = np.clip(h, 0, 1)
    return h
