import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Scale to target sum
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    h = h0.copy()
    h = project_to_feasible(h, target_sum)
    
    # Parameters
    learning_rate = 0.01
    min_lr = 1e-6
    stagnation_count = 0
    stagnation_limit = 50
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s:
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Compute gradient manually using the active shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            # For shift k_max, we have h[i] contributing to (1-h[i+k_max])
            j = (i + k_max) % n_points
            grad[i] = 1 - 2 * h[j]
        
        # Update with gradient descent
        h = h - learning_rate * grad
        
        # Project to feasible region
        h = project_to_feasible(h, target_sum)
        
        # Check for stagnation and adjust learning rate
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count >= stagnation_limit:
                # Add small random noise
                noise = np.random.uniform(-1e-3, 1e-3, n_points)
                h = h + noise
                h = project_to_feasible(h, target_sum)
                stagnation_count = 0
                # Lower learning rate
                if learning_rate > min_lr:
                    learning_rate *= 0.9
        
        iteration += 1
        
        # Adaptive learning rate decay for better convergence
        if iteration % 100 == 0 and learning_rate > min_lr:
            learning_rate *= 0.9
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
