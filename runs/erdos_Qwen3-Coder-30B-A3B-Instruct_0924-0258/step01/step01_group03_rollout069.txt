import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    lr = 0.01
    lr_decay = 0.9
    patience = 0
    patience_limit = 50
    smoothing_window = 5
    
    # History for smoothing
    grad_history = []
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute correlation - this is the most expensive part
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Determine shift causing max correlation
        n_corr = len(corr)
        shift = k_max - (n_corr - 1) // 2
        
        # Compute gradient direction
        gradient = np.zeros_like(h)
        if shift >= 0:
            for i in range(len(h)):
                if i + shift < len(h):
                    gradient[i] += (1 - h[i + shift])
        else:  # shift < 0
            for i in range(len(h)):
                if i - shift >= 0:
                    gradient[i] += (1 - h[i - shift])
        
        # Smooth gradient
        if len(grad_history) < smoothing_window:
            grad_history.append(gradient.copy())
        else:
            grad_history.pop(0)
            grad_history.append(gradient.copy())
        
        avg_grad = np.mean(grad_history, axis=0)
        
        # Update h
        h = h - lr * avg_grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2) / h_sum
        
        # Evaluate
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        # Adapt learning rate
        if patience >= patience_limit:
            lr *= lr_decay
            patience = 0
            # Add noise
            noise = np.random.uniform(-1e-3, 1e-3, len(h))
            h = h + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum) > 1e-10:
                h = h * (n_points / 2) / h_sum
        
        # Ensure convergence
        if np.isnan(h).any():
            h = best_h.copy()
            lr *= 0.5
            patience = 0
    
    return (best_h.tolist(), float(best_c5), n_points)
