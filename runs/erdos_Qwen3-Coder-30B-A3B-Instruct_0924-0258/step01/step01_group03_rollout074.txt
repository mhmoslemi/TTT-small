import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_erdos_solution(h_values, dx):
    """Compute C5 for given h values."""
    h = np.array(h_values)
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Convex optimization helper
    def solve_convex_relaxation(h_current):
        h = cp.Variable(n_points)
        # Linear relaxation of overlap terms
        # Maximize over all lags
        objective = cp.Minimize(cp.sum(h))
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return h.value
        return h_current
    
    # Gradient based local search
    def compute_gradient(h_current, shift_max):
        g = np.zeros(n_points)
        # Compute the derivative of the overlap term at shift_max
        # This is a simplification - a more accurate gradient would consider
        # the actual contribution of each h[i] to max overlap
        for i in range(n_points):
            j = i + shift_max
            if 0 <= j < n_points:
                g[i] += (1 - 2 * h_current[j])
        return g
    
    # Main loop
    iterations = 0
    adaptive_lr = 0.01
    last_improvement = 0
    
    while time.time() - start_time < budget_s:
        iterations += 1
        
        # Periodically solve convex relaxation
        if iterations % 10 == 0:
            h_projected = solve_convex_relaxation(h)
        else:
            h_projected = h.copy()
        
        # Local refinement with gradient descent
        # Find the current best shifting
        # Compute all correlations
        corr_vals = correlate(h_projected, 1 - h_projected, mode="full")
        shift_max = np.argmax(corr_vals) - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h_projected, shift_max)
        
        # Gradient descent step
        h_new = h_projected - adaptive_lr * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new value
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = iterations
            adaptive_lr = 0.01  # Reset Learning rate on improvement
        else:
            # Reduce learning rate if no improvement
            if iterations - last_improvement > 50:
                adaptive_lr *= 0.95
        
        h = h_new.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
