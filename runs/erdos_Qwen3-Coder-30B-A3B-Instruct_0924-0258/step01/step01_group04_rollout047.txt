import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 20
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Initialize best solution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # SDP Formulation
    def solve_sdp(h_init, n_points):
        h = Variable(n_points)
        X = Variable((n_points, n_points))
        t = Variable()
        
        # Constraints
        constraints = [
            X >> 0,  # X positive semidefinite
            h >= 0,
            h <= 1,
            sum(h) == target_sum,
        ]
        
        # Add diagonal constraints
        for i in range(n_points):
            constraints.append(X[i, i] == h[i])
        
        # Add overlap constraints
        for k in range(1, n_points):
            overlap_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_sum += X[i, j]
            constraints.append(t <= overlap_sum * dx)
        
        objective = Maximize(t)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False, max_iter=100)
            if prob.status in (OPTIMAL, OPTIMAL_INACCURATE):
                return h.value, prob.value
        except:
            pass
        
        return None, None

    # Iteration with SDP improvement
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        if iteration % 5 == 0:
            # Periodic full evaluation
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()
                
        # Try SDP optimization for small problems
        if n_points <= 45:
            h_sdp, t_sdp = solve_sdp(best_h, n_points)
            if h_sdp is not None:
                h_sdp = project_h(h_sdp)
                current_C5 = compute_C5(h_sdp)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h_sdp.copy()
        
        # Local search to adjust
        h_vec = best_h.copy()
        eps = 1e-4
        gradient = np.zeros_like(h_vec)
        
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            gradient[i] = (C_plus - C_minus) / (2 * eps)
        
        # Update with gradient
        learning_rate = 0.005
        best_h = best_h - learning_rate * gradient
        best_h = project_h(best_h)
        
        # Check if we're done
        if time.time() - start_time > budget_s - 5:
            break
            
    return (best_h.tolist(), best_C5, n_points)
