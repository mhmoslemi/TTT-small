import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        idx = np.argsort(h)
        update = min(diff, 1.0 - h[idx[0]])
        h[idx[0]] += update
        diff -= update
        for i in range(len(idx)-1, -1, -1):
            if diff == 0:
                break
            j = idx[i]
            if h[j] < 1.0 and diff > 0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    h = h0.copy()
    h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.01
    patience = 0
    patience_limit = 50
    best_c5_stable = best_c5
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s:
        # Compute correlation and find shift that maximizes it
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        # Convert index back to shift value
        shift = max_shift_idx - (len(h) - 1)
        
        # Compute gradient
        grad = np.zeros_like(h)
        if shift < 0:
            # h[i] * (1 - h[i-shift]) -> gradient wrt h[i] is (1 - h[i-shift]) for i < len(h)+shift
            for i in range(len(h) + shift):
                grad[i] += 1 - h[i-shift]
            # Also consider h[i-shift] * (1 - h[i]) -> gradient wrt h[i-shift] is -h[i] for i > -shift
            for i in range(-shift):
                if i < len(h):
                    grad[i-shift] -= h[i]
        elif shift > 0:
            # h[i] * (1 - h[i+shift]) -> gradient wrt h[i] is (1 - h[i+shift]) for i < len(h)-shift
            for i in range(len(h) - shift):
                grad[i] += 1 - h[i+shift]
            # Also for h[i+shift] * (1 - h[i]) -> gradient wrt h[i+shift] is -h[i] for i < len(h)-shift
            for i in range(len(h) - shift):
                if i + shift < len(h):
                    grad[i+shift] -= h[i]
        else:
            for i in range(len(h)):
                grad[i] += 1 - 2 * h[i]
            
        # Gradient descent update with adaptive learning rate
        h = h - learning_rate * grad
        h = project_to_feasible(h, target_sum)
            
        # Compute new value and update best
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit and learning_rate > 1e-5:
                patience = 0
                learning_rate *= 0.5
        
        iteration += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
