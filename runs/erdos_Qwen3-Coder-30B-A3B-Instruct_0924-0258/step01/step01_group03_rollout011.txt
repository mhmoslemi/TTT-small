import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / current_sum
        # If clipping made sum too large, scale down
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Use initial values if available

        n_points = len(h0)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Create initial simplex for Nelder-Mead
    m = n_points + 2  # Number of vertices in simplex
    simplex = [h0.copy()]
    for i in range(m-1):
        h = h0.copy()
        # Add small random perturbation that preserves feasibility
        pert = np.random.normal(0, 0.05, n_points)
        h = h + pert
        h = project_to_feasible(h, target_sum)
        simplex.append(h)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Nelder-Mead iterations
    iteration = 0
    max_iter = budget_s * 50 if budget_s < 100 else 5000
    
    # Nelder-Mead parameters
    alpha = 1.0   # reflection
    gamma = 2.0   # expansion
    rho = 0.5     # contraction
    sigma = 0.5   # shrink
    
    while iteration < max_iter and (time.time() - start_time) < budget_s:
        # Evaluate all points in simplex
        values = [evaluate_erdos_solution(s, dx) for s in simplex]
        # Sort simplex by function values
        sorted_indices = np.argsort(values)
        simplex = [simplex[i] for i in sorted_indices]
        values = [values[i] for i in sorted_indices]
        
        # Best and worst points
        x_best = simplex[0]
        x_worst = simplex[-1]
        x_second_worst = simplex[-2]
        
        # Compute centroid (excluding worst)
        x_centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        x_reflect = x_centroid + alpha * (x_centroid - x_worst)
        x_reflect = project_to_feasible(x_reflect, target_sum)
        f_reflect = evaluate_erdos_solution(x_reflect, dx)
        
        # Check if reflection is better than second worst
        if values[0] <= f_reflect < values[-2]:
            # Replace worst with reflection
            simplex[-1] = x_reflect
            values[-1] = f_reflect
        elif f_reflect < values[0]:
            # Expansion
            x_expand = x_centroid + gamma * (x_reflect - x_centroid)
            x_expand = project_to_feasible(x_expand, target_sum)
            f_expand = evaluate_erdos_solution(x_expand, dx)
            if f_expand < f_reflect:
                simplex[-1] = x_expand
                values[-1] = f_expand
            else:
                simplex[-1] = x_reflect
                values[-1] = f_reflect
        else:
            # Contraction
            if f_reflect < values[-1]:
                # Outside contraction
                x_contract = x_centroid + rho * (x_reflect - x_centroid)
            else:
                # Inside contraction
                x_contract = x_centroid + rho * (x_worst - x_centroid)
            x_contract = project_to_feasible(x_contract, target_sum)
            f_contract = evaluate_erdos_solution(x_contract, dx)
            if f_contract < values[-1]:
                simplex[-1] = x_contract
                values[-1] = f_contract
            else:
                # Shrink
                for i in range(1, m):
                    simplex[i] = simplex[0] + sigma * (simplex[i] - simplex[0])
                    simplex[i] = project_to_feasible(simplex[i], target_sum)
        
        # Update best solution
        current_best = simplex[0]
        current_c5 = values[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
            
        # Apply random mutation every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            for i in range(m):
                if np.random.random() < 0.2:
                    h = simplex[i].copy()
                    # Add small random perturbation
                    pert = np.random.normal(0, 0.03, n_points)
                    h = h + pert
                    h = project_to_feasible(h, target_sum)
                    simplex[i] = h
                    
        iteration += 1
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
