import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.array(h, dtype=float)
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        return np.ones_like(h) * (target_sum / len(h))
    h = h * target_sum / current_sum
    # Clamp again after normalization
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Set dx
    dx = 2.0 / n_points
    
    # Initialize
    h = h0.copy()
    h = project_h(h, n_points / 2.0)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 100
    best_c5 = float('inf')
    best_h = None
    no_improvement_count = 0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Evaluate current value
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > patience:
            # Reset with small random perturbation
            h = project_h(h + np.random.normal(0, 0.001, n_points), n_points / 2.0)
            no_improvement_count = 0
            continue
            
        # Compute all correlations
        corr_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr_full)
        
        # Compute gradient for the maximum correlation
        grad = np.zeros_like(h)
        # The correlation at shift max_k is sum h[i] * (1 - h[i + max_k])
        # So for the i-th element: contribution to gradient is -h[i + max_k] if max_k >= 0, 
        # or h[i - max_k] if max_k < 0
        
        # Handle positive shift case
        if max_k < len(h):
            # The correlation involves h[i] and 1-h[i+shift]
            # d/dh[i] = -h[i+shift] for i in range(len(h)-shift)
            shift = max_k
            for idx in range(len(h) - shift):
                grad[idx] -= h[idx + shift]
                
            # Handle the cases where index goes out of bounds for h
            # By definition, we now updated few entries based on standard correlation
            # Let's recompute more carefully
            # Actually use the direct index mapping for derivation
            # corr = sum_{i} h[i] * (1 - h[i + shift])
            # d/dh[k] = sum_{i: i+shift=k} h[i] * (-1) = -h[k-shift]
            # But we must also consider the symmetric mappings
            # The exact walkthrough for gradients of correlation:
            # If we fix h[i+k], then we're evaluating h[i] * (1-h[i+k]) 
            # for some i. So for every such term:
            # d/dh[i] = (1-h[i+k])
            # d/dh[i+k] = -h[i]
            # This is equivalent to adding 1-h[i+k] to grad[i]
            # and -h[i] to grad[i+k]
            
            # Rewrite this more carefully:
            # We compute correlation with (1-h) at all shifts
            # Specific shift max_k:
            # corr = sum h[i] * (1 - h[i+max_k]) for i in [0, ..., n-max_k-1]
            # Taking derivative w.r.t. h[i]:
            # if i < n-max_k, then we get contribution from d/dh[i] = 1-h[i+max_k] 
            # if i+max_k < n, then we get contribution from d/dh[i+max_k] = -h[i]
            
            # Just go with a different approach. Actual step function of correlation has a convolution formula.
            # Alternative approach: let's compute it as corr = h * (1-h)
            # but with only the indices involved in full correlation
            # For a given shift k, the contribution to correlation sum over i of h[i] * (1 - h[i+k]):
            # gradient partial derivative of this w.r.t. h[j] is (1-h[j+k]) if j == i
            # and -h[i] if j == i + k
            
            # Use the more direct approach:
            grad = np.zeros(n_points)
            for i in range(len(h)):
                for k_idx in range(len(h)):
                    # correlation at shift k_idx is sum_h[i] * (1-h[i+k_idx])
                    corr_i = i
                    corr_k = i + k_idx
                    if corr_k < len(h):
                        # Derivative w.r.t. h[i] contributes (1 - h[i + k_idx])
                        grad[i] += (1 - h[corr_k])
                        # Derivative w.r.t. h[i + k_idx] contributes -h[i]
                        grad[corr_k] -= h[corr_i]
        
        # Simplified gradient computation using full convolution derivative properties
        # An accurate derivative includes that for i in range(n), 
        # the max k for shift where correlation is maximized:
        # For each h[i], consider the impact on the correlation component.
        # The 'max_k' corresponds to the shift where the largest correlation is achieved.
        # Let's recall for shift k: sum over all i of h[i] * (1 - h[i+k]) 
        # The gradient w.r.t. h[i] = (1 - h[i+k]) for i+i_k < n
        # and = -h[i-k] for some k for which i-k >= 0 and defines a contribution.
        # Let's simplify implementation.
        # Re-computing full correlation since it's also needed for bounds check
        corr_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr_full)
        
        # Gradient computation for full correlation as sum_{i} h[i] * (1 - h[i+k])
        # For real derivative w.r.t. h[i]: two terms:
        # 1) h[i] contributes (1 - h[i+k]) 
        # 2) h[i+k] contributes -h[i]
        # So for each position i, consider the contribution to the full correlation for shift max_k:
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + max_k < len(h):
                grad[i] += 1 - h[i + max_k]
            if i - max_k >= 0:
                grad[i - max_k] -= h[i]
        
        # Update
        h = h - learning_rate * grad
        h = project_h(h, n_points / 2.0)
        
        # Adaptive step size
        if i % 100 == 0 and i > 0:
            learning_rate *= 0.99
        
    # Final re-evaluation
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        final_c5 = evaluate(h, dx)
        return (h.tolist(), float(final_c5), n_points)
