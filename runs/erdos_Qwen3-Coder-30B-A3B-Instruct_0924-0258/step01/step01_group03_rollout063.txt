import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] with sum = target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        return np.ones_like(h) * (target_sum / len(h))
    h = h * target_sum / current_sum
    # Ensure sum constraint
    h = np.clip(h, 0, 1)
    # Final adjustment to ensure exact sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute difference
        idx = np.argsort(h)
        for i in range(len(h)):
            if abs(diff) < 1e-10:
                break
            j = idx[i]
            if h[j] < 1.0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
            elif h[j] > 0.0 and diff < 0:
                update = max(diff, -h[j])
                h[j] += update
                diff -= update
    return h
