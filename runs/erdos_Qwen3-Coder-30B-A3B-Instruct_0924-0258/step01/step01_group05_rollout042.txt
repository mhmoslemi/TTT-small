import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with n=45
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def grid_search_level(h_start, n_points, num_candidates=20):
        # Generate random candidates for current resolution
        candidates = []
        for _ in range(num_candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Randomly perturb the starting point
            h_candidate = h_start + np.random.normal(0, 0.1, n_points)
            h_candidate = project_feasible(h_candidate)
            candidates.append(h_candidate)
            
        # Evaluate candidates and find best
        best_h = h_start.copy()
        best_C5 = compute_C5(h_start)
        
        for h_cand in candidates:
            if time.time() - start_time > budget_s - 0.1:
                break
            C5_cand = compute_C5(h_cand)
            if C5_cand < best_C5:
                best_C5 = C5_cand
                best_h = h_cand.copy()
                
        return best_h, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Multi-scale grid search
    resolutions = [5, 10, 15, 20, 25, 30, 45]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Create grid with current resolution
        if res != n_points:
            # Interpolate to target resolution
            if res < n_points:
                # Coarse grid, average intervals
                h_new = np.zeros(res)
                for i in range(res):
                    start_idx = i * (n_points // res)
                    end_idx = min((i + 1) * (n_points // res), n_points)
                    h_new[i] = np.mean(h_best[start_idx:end_idx])
            else:
                # Fine grid, interpolate from current
                x_old = np.linspace(0, 2, n_points)
                x_new = np.linspace(0, 2, res)
                h_new = np.interp(x_new, x_old, h_best)
                
            # Project to feasible
            h_new = project_feasible(h_new)
        else:
            h_new = h_best.copy()
            
        # Local grid search
        h_refined, C5_refined = grid_search_level(h_new, res, num_candidates=20 if res < 45 else 5)
        
        # Update best solution
        if C5_refined < C5_best:
            h_best = h_refined.copy()
            C5_best = C5_refined
            
            # Early exit if improvement is significant
            if C5_best < 0.3807:
                break
    
    # Final adjustment with local searches
    for _ in range(10):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Random perturbation and re-evaluate
        h_perturbed = h_best + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_feasible(h_perturbed)
        C5_new = compute_C5(h_perturbed)
        if C5_new < C5_best:
            h_best = h_perturbed.copy()
            C5_best = C5_new
    
    # Final projection and computation
    h_final = project_feasible(h_best)
    C5_final = compute_C5(h_final)
    
    return list(h_final), C5_final, n_points
