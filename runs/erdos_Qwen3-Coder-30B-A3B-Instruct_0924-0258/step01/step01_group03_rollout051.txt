import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Distribute difference carefully
        diff = target_sum - current_sum
        if diff > 0:
            # Need to increase values
            idx = np.argsort(h)
            for i in range(len(idx)-1, -1, -1):
                if diff <= 0:
                    break
                j = idx[i]
                if h[j] < 1.0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need to decrease values
            idx = np.argsort(h)
            for i in range(len(idx)):
                if diff >= 0:
                    break
                j = idx[i]
                if h[j] > 0.0:
                    update = max(diff, -h[j])
                    h[j] += update
                    diff -= update
    return h

    h = h0.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to sum constraint
    h = project_to_feasible(h, target_sum)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient computation and update
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s:
        # Evaluate current state
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count
        
        # Find shift that maximizes the overlap
        corr = correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(corr)
        max_corr = corr[max_shift]
        
        # Compute gradient: contribution to current max shift
        grad = np.zeros_like(h)
        if max_shift < len(h):
            # Case k = max_shift, so we're looking at h[i] * (1 - h[i + k])
            # But since we're aligning h with (1-h), that's h[i] * (1 - h[i + k])
            # So contribution to gradient of h[i] is (1 - h[i + k]) where k = max_shift
            for i in range(len(h)):
                if (i + max_shift) < len(h):
                    grad[i] += (1 - h[i + max_shift])
        else:
            # Case where we're shifting backward
            k = max_shift - len(h) + 1
            for i in range(len(h)):
                if i - k >= 0:
                    grad[i] += h[i - k]
        
        # Update step
        h = h - learning_rate * grad
        h = project_to_feasible(h, target_sum)
        
        iter_count += 1
        
        # Adaptive learning rate
        if iter_count - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = iter_count
            if learning_rate < 1e-6:
                break
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
