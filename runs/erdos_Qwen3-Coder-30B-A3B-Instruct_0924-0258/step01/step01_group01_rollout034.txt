import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    """Calculate C5 = max_k int h(x)(1 - h(x+k)) dx"""
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    """Project h to satisfy sum(h) * dx = 1"""
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10.0):
    """Smooth approximation to max for numerical stability"""
    return np.log(np.sum(np.exp(beta * x))) / beta

def compute_correlation_grad(h, dx, beta=10.0):

    max_idx = np.argmax(corr_lag)
    max_corr = corr_lag[max_idx]

    sum_exp = np.sum(exp_terms)

def optimize_step(h, dx, lr=1e-3, iterations=50):
    """Perform a few gradient steps with given learning rate"""
    for _ in range(iterations):
        # Compute gradient - simplified direct approach
        n = len(h)
        eps = 1e-5
        grad = np.zeros_like(h)
        
        h_current = h.copy()
        base_corr = evaluate(h_current, dx)
        
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_constraints(h_plus, dx)
            corr_plus = evaluate(h_plus, dx)
            grad[i] = (corr_plus - base_corr) / eps
            
        # Gradient descent step
        h_new = h - lr * grad
        h_new = project_to_constraints(h_new, dx)
        
        # Update if improvement
        if evaluate(h_new, dx) < evaluate(h, dx):
            h = h_new
        else:
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    n_points = 20
    dx = 2.0 / n_points
    
    # Initialize with current best if available
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:

        if len(h) > n_points:

        else:
            h = h[:n_points]
    else:
        # Random but valid initialization
        h = np.random.rand(n_points)
        h = project_to_constraints(h, dx)

    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Optimization parameters
    lr = 1e-2
    max_iter = 1000
    iter_count = 0
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Try local optimization
        h_opt = optimize_step(h, dx, lr, 50)
        c5_opt = evaluate(h_opt, dx)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            
        # Update h
        h = h_opt
        iter_count += 1
        
        # Adapt learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            lr *= 0.9
            
        # Occasionally double resolution if significant decrease
        if iter_count % 200 == 0 and n_points < 100:

                n_points *= 2
                dx = 2.0 / n_points
                h = h_new
                lr = 1e-2
                
    # Final optimal refinement
    h_final = optimize_step(best_h, dx, 1e-3, 100)
    c5_final = evaluate(h_final, dx)
    
    return h_final.tolist(), c5_final, len(h_final)
