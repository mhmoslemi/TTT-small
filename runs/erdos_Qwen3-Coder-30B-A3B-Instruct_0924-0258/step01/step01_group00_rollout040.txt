import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_relaxation_step(h, n_points, dx):
    """Use CVXPY to solve a convex relaxation for improved optimization"""
    # Create variables
    h_vars = cp.Variable(n_points)
    
    # Objective: maximize the minimum of A(k) over all k
    # A(k) = sum(h[i] * h[i+k]) for i = 0 to n_points-k-1
    # But we want to minimize max_k (1 - A(k)) = min_k A(k)
    
    # We create an approximation using an SDP relaxation
    obj = cp.Minimize(cp.sum(cp.multiply(h_vars, 1 - h_vars)))
    
    # Constraints
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
    
    # Formulate problem and solve
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return np.array(h_vars.value)
    except:
        pass
    
    return h

def genetic_mutation(h, n_points, mutation_rate=0.1):
    """Apply genetic-style mutations to h"""
    h_mut = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Flip with random change
            h_mut[i] = np.random.random()
    return h_mut

def polynome_mutation(h, n_points, degree=2):
    """Apply polynomial adjustment to h"""
    x = np.linspace(0, 1, n_points)
    # Fit a polynomial to current h
    coeffs = np.polyfit(x, h, degree)
    # Perturb coefficients slightly
    coeffs += np.random.normal(0, 0.1, len(coeffs))
    # Evaluate new polynomial
    h_new = np.polyval(coeffs, x)
    return h_new

def optimize_resolution(h_start, n_points, dx, time_limit):
    """Optimize a resolution level with time constraint"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use multi-start approach with both convex and genetic methods
    start_points = []
    for _ in range(10):
        h_rand = project_to_feasible(h_start + np.random.normal(0, 0.1, n_points), n_points)
        start_points.append(h_rand)
    
    for _ in range(200):  # Number of iterations
        if time() - start_time > time_limit:
            break
            
        # Try convex relaxation to reduce objective
        h_cvx = convex_relaxation_step(best_h, n_points, dx)
        h_cvx = project_to_feasible(h_cvx, n_points)
        c5_cvx = evaluate_erdos_solution(h_cvx, dx)
        if c5_cvx < best_c5:
            best_h = h_cvx.copy()
            best_c5 = c5_cvx
            
        # Try polynomial mutation
        h_poly = polynome_mutation(best_h, n_points)
        h_poly = project_to_feasible(h_poly, n_points)
        c5_poly = evaluate_erdos_solution(h_poly, dx)
        if c5_poly < best_c5:
            best_h = h_poly.copy()
            best_c5 = c5_poly
            
        # Try genetic mutation
        h_gen = genetic_mutation(best_h, n_points)
        h_gen = project_to_feasible(h_gen, n_points)
        c5_gen = evaluate_erdos_solution(h_gen, dx)
        if c5_gen < best_c5:
            best_h = h_gen.copy()
            best_c5 = c5_gen
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point

    else:

    # Resolution levels to try
    resolutions = [30, 45, 60, 75, 90]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time limit allocation per resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 60)
        
        # Optimize at this resolution with several restarts
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        # Different strategies at each level
        if n_points <= 45:
            # Use fewer restarts at lower resolutions
            iterations = 5
        else:
            # More iterations at higher resolutions
            iterations = 10
        
        for _ in range(iterations):
            h_restart = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            h_opt, c5_opt = optimize_resolution(h_restart, n_points, dx, time_per_res / iterations)
            if c5_opt < best_c5_level:
                best_c5_level = c5_opt
                best_h_level = h_opt.copy()
        
        # If improvement, store
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
            
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
