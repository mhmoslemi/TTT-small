import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Default starting solution
        initial_h = np.full(45, 1.0/45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial solution satisfies constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        """Compute the maximum overlap integral."""
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        """Project h to feasible region: [0,1] and sum to target_sum."""
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # GA parameters
    population_size = 50
    num_generations = 1000
    mutation_rate = 0.05
    crossover_rate = 0.8
    elite_size = population_size // 4
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    # Track best solution
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    no_improvement_count = 0
    generation = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 1000:
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = compute_c5(h)
            fitness_scores.append(-c5)  # Negative because we minimize
        
        # Selection: tournament
        selected = []
        for _ in range(population_size):
            tournament_size = 3
            candidates = np.random.choice(population_size, tournament_size)
            winner_idx = candidates[np.argmax([fitness_scores[i] for i in candidates])]
            selected.append(population[winner_idx].copy())
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            offspring.extend([child1, child2])
        
        # Limit offspring to population size
        offspring = offspring[:population_size]
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                offspring[i] = offspring[i] + noise
                offspring[i] = project_h(offspring[i])
        
        # Replace population with offspring, keep elite
        sorted_indices = np.argsort(fitness_scores)[::-1][:elite_size]
        new_population = [population[i] for i in sorted_indices]
        new_population.extend(offspring[:population_size - elite_size])
        
        population = new_population
        
        # Update best solution
        current_best_idx = np.argmin([-f for f in fitness_scores])
        current_best_c5 = -fitness_scores[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)
