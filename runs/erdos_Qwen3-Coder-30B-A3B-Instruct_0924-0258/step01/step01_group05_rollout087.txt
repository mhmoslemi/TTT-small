import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_C5 = np.max(correlate(h, 1-h, mode='full')) * dx
    
    # Annealing parameters
    T0 = 0.1
    tau = budget_s / np.log(10)
    T_min = 1e-5
    sigma = 0.1
    acceptance_count = 0
    acceptance_target = 0.25
    step_count = 0
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def adaptive_perturb(h_vec, sigma_val, priority_indices=None):
        h_new = h_vec.copy()
        if priority_indices is not None:
            # Perturb priority indices with noise
            for idx in priority_indices:
                h_new[idx] += np.random.normal(0, sigma_val)
        else:
            # Perturb all positions
            h_new += np.random.normal(0, sigma_val, len(h_new))
        
        # Flip some low-priority indices randomly
        low_priority = np.argsort(h_vec)[:len(h_vec)//5]
        for idx in np.random.choice(low_priority, size=len(low_priority)//5, replace=False):
            h_new[idx] = 1 - h_new[idx]
        
        return project_feasible(h_new)
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute current C5 for comparison
        current_C5 = compute_C5(best_h)
        
        # Find top contributing intervals
        corr = correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        # Determine priority indices based on maximum correlation lag
        priority_indices = []
        if max_lag >= 0:
            # Positive lag tells us where element at i pairs with 1 - element at i+lag
            # We want to prioritize higher indices affecting the result
            priority_indices = np.arange(max_lag, len(best_h))[:len(best_h)//5]
        else:
            # Negative lag
            priority_indices = np.arange(0, len(best_h) + max_lag)[:len(best_h)//5]
        
        # Perturb with higher probability when stuck
        if step_count % 100 == 0 and sigma < 0.5:
            sigma *= 1.1
        elif step_count % 1000 == 0 and sigma > 0.01:
            sigma *= 0.95
            
        # Perturb
        h_new = adaptive_perturb(best_h, sigma, priority_indices)
        new_C5 = compute_C5(h_new)
        
        # Accept or reject
        delta_C5 = new_C5 - current_C5
        T = max(T0 * np.exp(-step_count / tau), T_min)
        acceptance_prob = np.exp(-delta_C5 / T) if T > 0 else 0.0
        
        if new_C5 < current_C5 or np.random.random() < acceptance_prob:
            best_h = h_new.copy()
            if new_C5 < best_C5:
                best_C5 = new_C5
            acceptance_count += 1
            
        # Adaptive perturbation scaling based on acceptance rate
        if step_count > 0 and step_count % 50 == 0:
            acceptance_rate = acceptance_count / 50.0
            if acceptance_rate < acceptance_target:
                sigma *= 1.1
            elif acceptance_rate > acceptance_target * 1.5:
                sigma *= 0.9
            acceptance_count = 0
            
        step_count += 1

    # Final project to ensure feasibility
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
