import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Adam optimizer implementation
    def adam_update(h, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        return h_new, m, v
    
    # Initialize Adam
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    lr = 0.01
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        t += 1
        
        if iteration % 100 == 0:
            # Periodic full evaluation
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()
                if best_C5 < 0.3807:  # Early stopping if target reached
                    break
                        
        # Compute overlaps to identify k contributing to max
        h_vec = project_h(h)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        
        # Get the index of maximum overlap
        max_k_idx = np.argmax(overlaps)
        max_k = max_k_idx - (n_points - 1)
        
        # Compute gradient for the current max_k
        gradient = np.zeros_like(h_vec)
        for i in range(n_points):
            # The derivative of int h(x)(1-h(x+k)) dx with respect to h[i]
            # is (1 - h[(i-k) mod n])
            j = (i - max_k) % n_points
            gradient[i] = 1 - h_vec[j]
        
        # Update with Adam
        h, m, v = adam_update(h, gradient, m, v, t, lr)
        
        # Project
        h = project_h(h)
        
        # Adaptive learning rate
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            # Increase learning rate slightly if improving
            lr = min(lr * 1.1, 0.1)
        else:
            # Decrease learning rate if not improving
            lr = max(lr * 0.9, 1e-5)
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
