import numpy as np

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def optimize_resolution(h_start, n_points, dx, time_limit):
    """Optimize a resolution level with time constraint"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def obj(h):
        # Objective function is the maximum correlation
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx  # Negative because we minimize
    
    def grad(h):
        # Simplified gradient approximation
        eps = 1e-6
        grad = np.zeros_like(h)
        base_obj = obj(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            obj_plus = obj(h_plus)
            grad[i] = (obj_plus - base_obj) / eps
        return grad
    
    # Bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Initial perturb
    h_init = project_to_feasible(h_start + np.random.normal(0, 0.01, n_points), n_points)
    
    # Try multiple restarts
    for _ in range(5):
        if time() - start_time > time_limit:
            break
        try:
            res = minimize(obj, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons, jac=grad, 
                           options={'maxiter': 200, 'ftol': 1e-6})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
        # Perturb for next round
        h_init = project_to_feasible(best_h + np.random.normal(0, 0.02, n_points), n_points)
    
    return best_h, best_c5
