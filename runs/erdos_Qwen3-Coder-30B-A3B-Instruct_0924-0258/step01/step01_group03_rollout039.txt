import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region maintaining sum constraint."""
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    h = h * target_sum / current_sum
    # Clamp again, then renormalize
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Fine tune by adjusting elements
        diff = target_sum - current_sum
        idx = np.argsort(h)
        # Distribute the difference
        i = 0
        while i < len(idx) and abs(diff) > 1e-10:
            j = idx[i]
            if h[j] < 1.0 and diff > 0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
            elif h[j] > 0.0 and diff < 0:
                update = max(diff, -h[j])
                h[j] += update
                diff -= update
            i += 1
    return h
