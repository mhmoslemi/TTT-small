import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try to use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback construct
        n_points = 20
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Try SDP approach
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Set up SDP problem
    def solve_sdp(h_guess, n):
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        
        # Objective: maximize the minimum overlap
        t = cp.Variable()
        
        # Constraints
        constraints = [
            X >> 0,  # Positive semidefinite
            cp.diag(X) == h,  # Diagonal elements match h
            0 <= h, h <= 1,  # Bounds on h
            cp.sum(h) == target_sum,  # Sum constraint
        ]
        
        # Add overlap constraints
        for k in range(1, n):
            # Sum of X[i, (i+k) % n] for i in range(n) 
            overlap_sum = cp.sum([X[i, (i+k) % n] for i in range(n)])
            constraints.append(t <= overlap_sum)
        
        # Objective
        prob = cp.Problem(cp.Maximize(t), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iter=100)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_sol = h.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * target_sum / np.sum(h_sol)
                    return h_sol
        except:
            pass
        return None
    
    # Try different resolutions
    for n in [20, 30, 40, 45]:
        if time.time() - start_time > budget_s - 5:
            break
            
        if n > n_points:
            # Upscale and initialize
            h_new = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_points), best_h)
        else:
            h_new = best_h[:n]
            
        # Try SDP
        h_sdp = solve_sdp(h_new, n)
        if h_sdp is not None:
            h_new = h_sdp
            n_points = n
            
        # Ensure the solution is valid
        h_new = project_h(h_new)
        
        if time.time() - start_time > budget_s - 5:
            break
            
        # Local refinement with optimization
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * target_sum / np.sum(h_vec)
            return compute_C5(h_vec)
        
        # Gradient free local search using pattern search
        current_h = h_new.copy()
        current_C5 = objective(current_h)
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Small random perturbations
            h_perturbed = current_h + np.random.normal(0, 0.01, len(current_h))
            h_perturbed = project_h(h_perturbed)
            new_C5 = objective(h_perturbed)
            
            if new_C5 < current_C5:
                current_h = h_perturbed
                current_C5 = new_C5
                
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
                
    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
