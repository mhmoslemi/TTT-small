import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def local_optimize(h_vec, max_iter=200):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        iter_no_improvement = 0
        step_size = 0.05
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Try moving each entry up and down slightly
            improved = False
            for i in range(len(h_current)):
                # Try small positive step
                h_try = h_current.copy()
                delta = step_size
                h_try[i] += delta
                h_try = project_feasible(h_try)
                C5_try = compute_C5(h_try)
                if C5_try < best_C5:
                    h_current = h_try.copy()
                    best_C5 = C5_try
                    improved = True
                    iter_no_improvement = 0
                else:
                    # Try small negative step
                    h_try = h_current.copy()
                    h_try[i] -= delta
                    h_try = project_feasible(h_try)
                    C5_try = compute_C5(h_try)
                    if C5_try < best_C5:
                        h_current = h_try.copy()
                        best_C5 = C5_try
                        improved = True
                        iter_no_improvement = 0
            
            # Swap optimization
            for _ in range(3):
                i, j = np.random.randint(0, len(h_current), 2)
                if i == j: continue
                
                h_try = h_current.copy()
                h_try[i], h_try[j] = h_try[j], h_try[i]
                h_try = project_feasible(h_try)
                C5_try = compute_C5(h_try)
                if C5_try < best_C5:
                    h_current = h_try.copy()
                    best_C5 = C5_try
                    improved = True
                    iter_no_improvement = 0
            
            if not improved:
                iter_no_improvement += 1
                if iter_no_improvement > 30:
                    # Restart with random perturbation
                    h_current += np.random.normal(0, 0.01, len(h_current))
                    h_current = project_feasible(h_current)
                    iter_no_improvement = 0
            else:
                best_h = h_current.copy()
                
            if step_size > 0.001 and iter_no_improvement > 20:
                step_size *= 0.9
                iter_no_improvement = 0
                
        return best_h, best_C5
    
    def latin_hypercube_sampling(n_samples, n_dim):
        samples = np.random.random((n_samples, n_dim))
        for i in range(n_dim):
            order = np.random.permutation(n_samples)
            samples[:,i] = np.sort(samples[:,i])
            samples[:,i] = np.take(samples[:,i], order)
        return samples
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Generate diverse initial candidates
    candidates = []
    if len(h) > 10:
        # Use latin hypercube design
        lh_samples = latin_hypercube_sampling(5, n_points)
        for i in range(5):
            sample = lh_samples[i]
            h_samp = np.zeros(n_points)
            # Distribute points to get sum = n_points/2
            cumsum = np.cumsum(sample)
            h_samp = (cumsum / cumsum[-1]) * target_sum
            candidates.append(project_feasible(h_samp))
    else:
        candidates.append(h_best)
    
    # Also include some perturbed versions of the current solution
    for _ in range(3):
        h_pert = h_best + np.random.normal(0, 0.02, n_points)
        candidates.append(project_feasible(h_pert))
    
    # Try optimization from different initial points
    for h_init in candidates:
        h_opt, C5_opt = local_optimize(h_init, max_iter=200)
        if C5_opt < C5_best:
            C5_best = C5_opt
            h_best = h_opt.copy()
    
    # Additional refinement with restarts
    for restart in range(5):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        h_rand = np.random.random(n_points)
        h_rand = project_feasible(h_rand)
        h_opt, C5_opt = local_optimize(h_rand, max_iter=150)
        
        if C5_opt < C5_best:
            C5_best = C5_opt
            h_best = h_opt.copy()
    
    # Final cleaning and validation
    h_best = project_feasible(h_best)
    final_C5 = compute_C5(h_best)
    
    return list(h_best), final_C5, n_points
