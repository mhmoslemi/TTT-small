import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Particle Swarm Optimization
    n_particles = 100
    inertia_start = 0.7
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(n_particles):
        # Start near initial solution with small perturbations
        particle = project_h(h + np.random.normal(0, 0.05, n_points))
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(particle.copy())
        pbest_values.append(compute_C5(particle))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update inertia weight
        inertia = inertia_start - (inertia_start - inertia_end) * (iteration / 1000)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_speed, max_speed)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project into feasible region
            particles[i] = project_h(particles[i])
            
            # Compute new value
            new_value = compute_C5(particles[i])
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = new_value
                
                # Update global best
                if new_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = new_value
                    
    # Final check
    final_C5 = compute_C5(gbest)
    if final_C5 < gbest_value:
        gbest_value = final_C5
        
    return (gbest.tolist(), gbest_value, n_points)
