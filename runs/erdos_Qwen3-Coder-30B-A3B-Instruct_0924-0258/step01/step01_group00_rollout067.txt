import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_relaxation(h, n_points, dx):
    # Use cvxpy for convex relaxation
    h_var = cp.Variable(n_points)
    # Convex relaxation of correlation terms
    # A good heuristic is to use gradient descent with approximations
    return np.sum(np.correlate(h, 1 - h, mode='full')**2) * dx**2

def genetic_mutation(h, mutation_rate=0.1):
    # Apply polynomial mutation (adjusting coefficients)
    mutated = h.copy()
    n_points = len(h)
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Add polynomial perturbation
            x = np.linspace(0, 2, n_points)
            poly_pert = np.polyval(np.random.randn(3), x[i] - 1) * 0.1
            mutated[i] += poly_pert
    return mutated

def genetic_crossover(h1, h2):
    # Uniform crossover
    crossover_point = np.random.randint(0, len(h1))
    child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Resolution levels
    resolutions = [30, 45, 60, 75]
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time allocation per resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 50)
        
        # Genetic algorithm with convex relaxation
        population_size = 20
        pop = [project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points) 
               for _ in range(population_size)]
        
        for gen in range(20):
            if time() - start_time > budget_s * 0.98:
                break
            
            # Evaluate fitness (inverse of C5)
            fitness = []
            for h in pop:
                c5 = evaluate_erdos_solution(h, dx)
                fitness.append(-c5)  # maximize -c5
            
            # Sort population
            sorted_idx = np.argsort(fitness)[::-1]
            pop = [pop[i] for i in sorted_idx]
            
            # Track best
            current_best = pop[0]
            c5_best = evaluate_erdos_solution(current_best, dx)
            if c5_best < best_c5:
                best_h = current_best.copy()
                best_c5 = c5_best
                best_n_points = n_points
            
            # Create next generation
            new_pop = [pop[0].copy()]  # Elitism
            
            # Crossover and mutation
            while len(new_pop) < population_size:
                # Tournament selection
                idx1, idx2 = np.random.choice(len(pop), 2, replace=False)
                parent1 = pop[idx1] if fitness[idx1] > fitness[idx2] else pop[idx2]
                idx3, idx4 = np.random.choice(len(pop), 2, replace=False)
                parent2 = pop[idx3] if fitness[idx3] > fitness[idx4] else pop[idx4]
                
                child = genetic_crossover(parent1, parent2)
                child = project_to_feasible(genetic_mutation(child), n_points)
                new_pop.append(child)
            
            pop = new_pop[:population_size]
        
        # Update starting point for next resolution
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
