import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_autocorr(h, dx):
    corr = np.correlate(h, h, mode='full')
    return corr * dx

def improve_with_gradient_descent(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Objective: maximize min_autocorr, which means minimize max_overlap
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx  # Return negative for minimization
    
    def grad(h):
        eps = 1e-6
        grad = np.zeros_like(h)
        base_obj = obj(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            obj_plus = obj(h_plus)
            grad[i] = (obj_plus - base_obj) / eps
        return grad
        
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random restarts
    for _ in range(10):
        if time() - start_time > time_limit:
            break
        try:
            h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)
            res = minimize(obj, h_perturbed, method='SLSQP', 
                          bounds=bounds, constraints=cons, jac=grad, 
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            continue
            
        # Perturb to escape local minima
        h_start = project_to_feasible(best_h + np.random.normal(0, 0.01, n_points), n_points)
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Try the initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / n_points)
    best_n_points = n_points
    
    # Try several resolutions
    resolutions = [20, 25, 30, 35, 40, 45, 50]
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate or just use the start if same size
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time allocation (this is a simulation: actual time taken is in function)
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 50)
        
        # Run optimization with gradient descent
        h_opt, c5_opt = improve_with_gradient_descent(h_start, n_points, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
            
        h_start = h_opt.copy()
    
    return (list(best_h), best_c5, best_n_points)
