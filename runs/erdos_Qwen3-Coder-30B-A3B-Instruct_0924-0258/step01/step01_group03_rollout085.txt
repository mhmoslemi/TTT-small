import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the required sum while keeping values in [0,1]."""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    h = h * target_sum / current_sum
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Re-normalize if still outside
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Distribute the difference among elements
        idx = np.argsort(h)
        diff = target_sum - current_sum
        if diff > 0:
            for i in range(len(idx)-1, -1, -1):
                if diff == 0:
                    break
                j = idx[i]
                if h[j] < 1.0 and diff > 0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
        else:
            for i in range(len(idx)):
                if diff == 0:
                    break
                j = idx[i]
                if h[j] > 0.0 and diff < 0:
                    update = max(diff, -h[j])
                    h[j] += update
                    diff -= update
    return h

def project_to_feasible(h, target_sum):
    """Project h to the feasible region with 0 <= h[i] <= 1 and sum = target_sum."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h
