import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def admm_update_h(h, z, rho, dx):
    # Minimize over h: (1/2)*||h - z||^2 + (rho/2)*||h - (1-z)||^2 + lambda * (sum(h) - n/2)
    # This is a quadratic problem with constraints. Use projected gradient descent.
    n = len(h)
    # Simplified: h = argmin (rho/2)*||h - (1-z)||^2 + lambda * (sum(h) - n/2) over h in [0,1]
    # Gradient is rho*(h - (1-z)) + lambda
    # Use fixed point iteration or closed form if possible
    
    # We solve the unconstrained version, then project
    # We treat sum constraint with Langrange multiplier: 
    #      L = sum((h_i - (1-z_i))^2 ) + lambda*(sum(h) - n/2)
    #      dL/dh = 2*(h - (1-z)) + lambda = 0 => h = (1-z) - lambda/2
    
    # Given h = (1-z) - lambda/2, sum(h) = sum(1-z) - n*lambda/2
    # So lambda = 2*(sum(1-z) - n/2) / n
    # So h = (1-z) - (sum(1-z) - n/2) / n
    
    lambda_val = 2 * (np.sum(1 - z) - n/2) / n
    new_h = (1 - z) - lambda_val / 2
    new_h = np.clip(new_h, 0, 1)
    new_h = project_to_constraints(new_h, dx)
    return new_h

def admm_update_z(h, z, rho, dx):
    # Minimize over z: ||z - (1-h)||^2 + (rho/2)*||z - (1-h)||^2
    # This is just (1 + rho) * ||z - (1-h)||^2
    # So optimal z = (1-h) if unconstrained
    # But z is projected to [0,1] and satisfies sum constraint
    return np.clip(1 - h, 0, 1)

            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
        
        dx = 2.0 / n_points
        h = project_to_constraints(h, dx)
        
        # ADMM parameters
        rho = 1.0
        max_iter = 100
        tolerance = 1e-6

            # Update h
            h_new = admm_update_h(h, 1 - h, rho, dx)
            
            # Update z = 1 - h (in projected space)
