import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Initial solution
    h_current = h.copy()
    C5_current = compute_C5(h_current)
    h_best = h_current.copy()
    C5_best = C5_current
    
    # Simulated Annealing parameters
    T0 = 0.1
    tau = budget_s / np.log(10)
    sigma = 0.1
    T_min = 1e-5
    
    # Number of iterations without improvement before restart
    patience = 200
    patience_counter = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        # Adaptive temperature
        t = iteration / 1000.0
        T = max(T_min, T0 * np.exp(-t / tau))
        
        # Priority sampling: Identify top contributors to overlap
        corr = correlate(h_current, 1 - h_current, mode='full')
        # Find indices of highest correlations
        top_indices = np.argsort(corr)[-int(n_points * 0.2):]  # Top 20%
        # Convert to lag indices
        lags = top_indices - (len(h_current) - 1)
        # Check for any positive lag
        high_contributors = set()
        for lag in lags:
            if lag >= 0:
                high_contributors.update(range(lag, len(h_current)))
            else:
                high_contributors.update(range(0, len(h_current) + lag))
        high_contributors = list(high_contributors)
        
        # Perturb the solution
        h_new = h_current.copy()
        
        # Perturb high contribution intervals with Gaussian noise
        for i in high_contributors:
            if np.random.rand() < 0.5:  # 50% chance to modify
                h_new[i] += np.random.normal(0, sigma)
        
        # Randomly flip some values to add diversity
        flip_indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
        for i in flip_indices:
            h_new[i] = 1.0 - h_new[i]
        
        # Project to feasible set
        h_new = project_feasible(h_new)
        
        # Compute new C5
        C5_new = compute_C5(h_new)
        
        # Acceptance criterion
        delta_C5 = C5_new - C5_current
        
        # Accept if better
        if delta_C5 < 0:
            h_current = h_new
            C5_current = C5_new
            if C5_new < C5_best:
                h_best = h_new.copy()
                C5_best = C5_new
                patience_counter = 0
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-delta_C5 / T):
                h_current = h_new
                C5_current = C5_new
                patience_counter = 0
            else:
                patience_counter += 1
        
        # Adjust step size based on acceptance rate
        if iteration % 100 == 0 and iteration > 100:
            # Approximate acceptance rate - not theoretically precise but good enough
            # This is a simplified version of adaptation
            if sigma > 0.001:
                sigma *= 0.98  # Gradually decrease step size for better refinement
        
        # Random restart if too many iterations without improvement
        if patience_counter > patience:
            h_current = h_best.copy()
            patience_counter = 0
            # Slightly perturb the current best to escape local plateaus
            for i in range(n_points):
                if np.random.rand() < 0.05:
                    h_current[i] += np.random.normal(0, 0.05)
            h_current = project_feasible(h_current)
            C5_current = compute_C5(h_current)
    
    # Final check and cleanup
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
