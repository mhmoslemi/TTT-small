import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        # Linear interpolation upsampling
        x_old = np.linspace(0, 1, len(h_vec))
        x_new = np.linspace(0, 1, target_len)
        return np.interp(x_new, x_old, h_vec)
    
    # Multi-resolution approach
    resolutions = [5, 10, 20, 40, n_points]
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    # Use Adam optimizer with momentum and adaptive learning rate
    def adam_update(gradients, m, v, t, lr=0.01, beta1=0.9, beta2=0.999):
        m = beta1 * m + (1 - beta1) * gradients
        v = beta2 * v + (1 - beta2) * gradients**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        return m, v, m_hat / (np.sqrt(v_hat) + 1e-8) * lr
    
    for i, res in enumerate(resolutions):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Initialize or upsample
        if i == 0:
            h_vec = h[:res].copy()
            h_vec = project_feasible(h_vec)
        else:
            h_vec = upsample_h(best_h, res)
            h_vec = project_feasible(h_vec)
            
        # Adam parameters
        m = np.zeros_like(h_vec)
        v = np.zeros_like(h_vec)
        t = 1
        lr = 0.01
        max_iter = 1000 if i == len(resolutions) - 1 else 500  # More iterations at final resolution
        
        iter_count = 0
        no_improvement_count = 0
        
        while iter_count < max_iter and time.time() - start_time < budget_s - 5:
            iter_count += 1
            t += 1
            
            # Compute overlap to find dominant k
            conv = np.correlate(h_vec, 1 - h_vec, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-res+1:mid+res-1] * dx
            # Focus on top 5 or the indices that contribute most
            top_k = np.argsort(overlaps)[-5:][::-1]
            
            # Approximate gradient with finite differences using top shifts
            gradient = np.zeros_like(h_vec)
            eps = 1e-4
            for i_point in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[i_point] += eps
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[i_point] -= eps
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                gradient[i_point] = (C_plus - C_minus) / (2 * eps)
            
            # Adam update
            m, v, step = adam_update(gradient, m, v, t, lr)
            h_vec = h_vec - step
            
            # Project to feasible region
            h_vec = project_feasible(h_vec)
            
            # Check improvement
            current_C5 = compute_overlap(h_vec)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_vec.copy()
                no_improvement_count = 0
                lr = min(lr * 1.1, 0.1)  # Increase learning rate if improving
            else:
                no_improvement_count += 1
                lr = max(lr * 0.9, 0.001)  # Decrease if not improving
                
            # Early stopping if no improvement in 500 iterations
            if no_improvement_count >= 500:
                break
                
        best_h = h_vec.copy()  # Ensure we keep the best at each level
        
    # Final validation
    final_C5 = compute_overlap(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
