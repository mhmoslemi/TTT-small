import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Default initial construction: a simple piecewise constant function
        n0 = 30
        h_start = np.ones(n0) / 2
    
    # Parameters
    n_points = len(h_start)
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Time allocations
    max_iter_time = 0.8 * budget_s
    perturb_time = 0.2 * budget_s
    
    # Start greedy improvement
    iter_count = 0
    last_improvement = 0
    target = 0.38  # target value to approach
    max_iterations = 1000
    iter_without_improvement = 0
    noise_strength = 0.01
    
    while time() - start_time < max_iter_time and iter_count < max_iterations:
        iter_count += 1
        
        # Compute current overlap
        c_max = evaluate_erdos_solution(best_h, dx)
        
        # Identify the worst shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_idx = np.argmax(corr)
        shift = k_idx - (n_points - 1)  # convert index to shift value
        
        # Find contributing indices
        # For shift k, we consider overlap at index i of h[i] * (1 - h[i+k])
        # We need to identify top contributors
        
        # Determine valid ranges
        indices = []
        # If shift is positive, we want h[i] * h[i-k]
        # If shift is negative, we want h[i] * h[i+k]
        if shift >= 0:
            for i in range(n_points - shift):
                indices.append((i, i + shift))
        else:
            for i in range(-shift, n_points):
                indices.append((i, i - shift))
        
        # Compute contribution of each overlapping pair
        contributions = []
        for i, j in indices:
            if i < n_points and j < n_points:
                contrib = best_h[i] * (1 - best_h[j])
                contributions.append((contrib, i, j))
        
        # Sort and get top 10%
        contributions.sort(reverse=True)
        top_n = max(1, len(contributions) // 10)
        
        # Modify h in top contributors to reduce overlap
        adjustment = (c_max - target) / n_points if c_max > target else 0.001 / n_points
        h_new = best_h.copy()
        for i, j in indices:
            h_new[i] = max(0, h_new[i] - adjustment)
        
        # Project to feasible set
        h_new = project_to_feasible(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Update if better, or add noise if no improvement
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_without_improvement = 0
        else:
            iter_without_improvement += 1
            if iter_without_improvement > 50:
                # Perturb slightly
                perturbation = np.random.normal(0, noise_strength, n_points)
                h_pert = project_to_feasible(best_h + perturbation, n_points)
                c5_pert = evaluate_erdos_solution(h_pert, dx)
                if c5_pert < best_c5:
                    best_h = h_pert
                    best_c5 = c5_pert
                # Reset counter
                iter_without_improvement = 0
        
    # Ensure final validity
    best_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
