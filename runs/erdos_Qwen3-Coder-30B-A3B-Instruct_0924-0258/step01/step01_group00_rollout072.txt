import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Track the last time we improved
    last_improvement = 0
    target_c5 = 0.38
    
    i = 0
    while time() - start_time < budget_s * 0.95:
        i += 1
        # Compute current overlap values
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        max_idx = np.argmax(corr)
        # The shift indices go from -n+1 to n-1, so the actual shift is
        # max_idx - (n_points - 1)
        shift = max_idx - (n_points - 1)
        
        # If we're very close to target, try to be more precise with fine tuning
        if max_overlap <= best_c5 * 1.02:
            # Identify the contributing indices
            # Indices that contribute to the correlation at max_idx
            # In full correlation at index max_idx, the correlation is 
            # sum over all valid x of h[x] * (1 - h[x + shift])
            # So identify the top contributing indices for max overlap
            # But for efficiency, we'll use a greedy approach with local decrease
            
            # Sample position indices in overlap computation
            contrib_indices = set()
            for x in range(n_points):
                y = x + shift
                if 0 <= y < n_points:
                    contrib_indices.add(x)
            
            # Very approximate: take top 10% by value
            contrib_indices = list(contrib_indices)

            sorted_indices = [i for _, i in sorted(zip(contrib_values, contrib_indices), reverse=True)]

            decrease = max(0.001, (max_overlap - target_c5) / (n_points + 1))

                    if new_h[i] > decrease:

                    else:

                    if new_h[i+shift] > decrease:
                        new_h[i+shift] -= decrease
                    else:

                    else:

            new_h = project_to_feasible(new_h, n_points)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = new_h.copy()
                last_improvement = i
            else:

                    h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
                    last_improvement = i
                else:
                    h = new_h.copy()

    return (list(best_h), best_c5, n_points)
