import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx, critical_lags):
    n = len(h)
    grad = np.zeros(n)
    # Compute correlation with (1-h) for all lags
    corr = correlate(h, 1-h, mode='full')
    # For each critical lag k, add contribution
    for k in critical_lags:
        # k represents the lag index in full correlation
        # Full corr has length 2*n - 1, lag 0 at center
        # We map k back to lag in [0, n-1] range
        if k < n:
            # Positive lag: h[i] * (1-h[i+k])
            for i in range(n-k):
                grad[i] += 1
        else:
            # Negative lag: h[i+k] * (1-h[i])
            for i in range(k-n+1, n):
                grad[i] += -1
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
        
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent with projection
    learning_rate = 0.1
    max_iter_per_res = 1000
    iter_count = 0
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        
        for iter_res in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            iter_count += 1
            
            # Evaluate current C5 and find critical lags
            c5 = evaluate(h, dx)
            
            # Get full correlation and identify top contributors
            corr = correlate(h, 1-h, mode='full')
            # Get the peaks
            peak_indices = np.argsort(-np.abs(corr))[:min(5, len(corr))]
            critical_lags = []
            for idx in peak_indices:
                # Only consider lags that make sense (not near boundary)
                if 2 < idx < 2 * n_points - 3:
                    critical_lags.append(idx)
            
            # Compute gradient
            grad = compute_gradient(h, dx, critical_lags)
            
            # Update h with gradient step
            h_new = h - learning_rate * grad
            
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Project to integral constraint
            h_new = project_to_constraints(h_new, dx)
            
            # Accept if improves
            c5_new = evaluate(h_new, dx)
            if c5_new < c5:
                h = h_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
            else:
                # Small random perturbation if no improvement
                noise = np.random.normal(0, 0.01, n_points)
                h_new = h + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = project_to_constraints(h_new, dx)
                c5_new = evaluate(h_new, dx)
                if c5_new < c5:
                    h = h_new
                    if c5_new < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_new
            
        # Adaptive resolution increase
        if n_points < n_high and iter_count > 500 and best_c5 < 0.4:
            # Interpolate to higher resolution
            h_new_full = np.interp(np.linspace(0, 2, 2*n_points),
                                  np.linspace(0, 2, n_points), h)
            h = h_new_full
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            best_c5 = evaluate(h, dx)
            best_h = h.copy()
            iter_count = 0  # Reset iteration count when resampling
        else:
            # Reduce learning rate
            learning_rate *= 0.995
            
    return best_h.tolist(), best_c5, len(best_h)
