import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def perturb(h_vec, temperature, priority_indices, sigma=0.1):
        h_new = h_vec.copy()
        # Perturb priority indices with Gaussian noise
        for i in priority_indices:
            h_new[i] += np.random.normal(0, sigma)
        # Randomly flip some non-priority indices
        non_priority = [i for i in range(n_points) if i not in priority_indices]
        flip_count = max(1, len(non_priority) // 20)
        flip_indices = np.random.choice(non_priority, flip_count, replace=False)
        for i in flip_indices:
            h_new[i] = 1.0 - h_new[i]
        return h_new
    
    def get_priority_indices(h_vec, num_top=10):
        # Find intervals that most impact maximum overlap
        # Compute influence by looking at neighbor correlations
        influence = np.zeros(n_points)
        for i in range(n_points):
            # Influence at position i: how much change in h[i] affects overlap
            influence[i] = np.abs(h_vec[i] - 0.5)  # Centered around 0.5
        # Also consider locations where h is near 0 or 1
        # Priority indices are those with highest influence
        priority_indices = np.argsort(influence)[-num_top:]
        return priority_indices
    
    # Simulated Annealing
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    
    # Initialize best solution
    h_best = h_current.copy()
    C5_best = C5_current
    
    # Algorithm parameters
    T0 = 0.1
    tau = budget_s / np.log(10)
    sigma = 0.1
    T = T0
    min_T = 1e-5
    accept_count = 0
    total_trials = 0
    last_improvement = 0
    patience = 100
    
    while time.time() - start_time < budget_s - 0.1:
        # Update temperature
        t = time.time() - start_time
        T = max(min_T, T0 * np.exp(-t / tau))
        
        # Get priority indices for perturbation
        priority_indices = get_priority_indices(h_current, num_top=max(5, n_points//9))
        
        # Perform perturbation
        h_new = perturb(h_current, T, priority_indices, sigma)
        h_new = project_feasible(h_new)
        
        # Compute new C5
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        total_trials += 1
        if C5_new < C5_current:
            h_current = h_new.copy()
            C5_current = C5_new
            accept_count += 1
            if C5_current < C5_best:
                h_best = h_current.copy()
                C5_best = C5_current
                last_improvement = 0
        else:
            # Accept with probability
            delta = C5_new - C5_current
            if np.random.rand() < np.exp(-delta / T):
                h_current = h_new.copy()
                C5_current = C5_new
                accept_count += 1
        
        # Adaptive sigma adjustment based on acceptance rate
        if total_trials > 10:
            accept_rate = accept_count / total_trials
            if accept_rate < 0.1 and sigma < 0.5:
                sigma *= 1.2
            elif accept_rate > 0.4 and sigma > 0.01:
                sigma *= 0.8
        
        # Reset patience if improved
        last_improvement += 1
        if last_improvement > patience:
            # Reset with random restart
            h_current = np.random.rand(n_points)
            h_current = project_feasible(h_current)
            C5_current = compute_C5(h_current)
            last_improvement = 0
            T = T0
    
    # Final projection
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
