import numpy as np
from scipy import optimize
import time

def compute_C5(h_vec, dx):
    """Compute C5 value for given h vector."""
    h_vec = np.array(h_vec)
    # Compute all correlations (full mode)
    conv = np.correlate(h_vec, 1 - h_vec, mode='full')
    # Take only the relevant part (max k values)
    mid = len(conv) // 2
    overlaps = conv[mid - len(h_vec) + 1:mid + len(h_vec) - 1] * dx
    return np.max(overlaps)

def project_h(h_vec, target_sum):
    """Project h vector onto feasible region."""
    h_vec = np.clip(h_vec, 0, 1)
    # Rescale to target sum
    if np.sum(h_vec) > 0:
        h_vec = h_vec * target_sum / np.sum(h_vec)
    return h_vec

    pbest_positions = [p.copy() for p in particles]
    pbest_values = [compute_C5(p, dx) for p in particles]

    global_best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[global_best_idx].copy()

        w = w_start - (w_start - w_end) * iteration / 1000

            # Update velocity
            r1, r2 = np.random.random(2)

            )

            vel_update = np.clip(vel_update, -max_speed, max_speed)
            velocities[i] = vel_update
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project into feasible region
            particles[i] = project_h(particles[i], target_sum)

            current_C5 = compute_C5(particles[i], dx)
            
            # Update personal best
            if current_C5 < pbest_values[i]:
                pbest_values[i] = current_C5
                pbest_positions[i] = particles[i].copy()

                if current_C5 < gbest_value:
