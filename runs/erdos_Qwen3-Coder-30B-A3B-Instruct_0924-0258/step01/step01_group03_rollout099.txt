import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Set dx
    dx = 2.0 / n_points
    
    # Initialize
    h = h0.copy().astype(float)
    h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Convex optimization approach
    # We'll use cvxpy to solve a relaxed version of the problem
    # and then refine with gradient descent.
    
    # Start gradient descent from best solution
    lr = 0.01
    max_iter = 10000
    patience = 50
    patience_count = 0
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute current correlation
        corr = correlate(h, 1 - h, mode="full")
        s_star = np.argmax(corr) - (n_points - 1)
        
        # Compute gradient for this s_star
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + s_star
            if 0 <= j < n_points:
                grad[i] += (1 - 2 * h[j])
        
        # Update with gradient step
        h_new = h - lr * grad
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        h = h_new.copy()
        
        # Adjust learning rate if no improvement
        if patience_count > patience:
            lr *= 0.9
            patience_count = 0
            
    return (best_h.tolist(), float(best_c5), n_points)
