import numpy as np

import time

def compute_overlap(h_vec):
    """Compute the maximum overlap integral C5."""
    h_vec = np.array(h_vec)
    dx = 2.0 / len(h_vec)
    # Compute all correlations (full mode)
    conv = np.correlate(h_vec, 1 - h_vec, mode='full')
    # Take only the relevant part (max k values)
    mid = len(conv) // 2
    overlaps = conv[mid-len(h_vec)+1:mid+len(h_vec)-1] * dx
    return np.max(overlaps)

def project_feasible(h_vec):
    """Project h_vec to feasible region: [0,1] and sum=h_target."""
    h_vec = np.clip(h_vec, 0, 1)
    target_sum = len(h_vec) / 2.0
    h_vec = h_vec * target_sum / np.sum(h_vec)
    return h_vec

def upsample_h(h_vec, target_len):
    """Upsample h_vec to target_len using linear interpolation."""
    src_indices = np.linspace(0, len(h_vec)-1, len(h_vec))
    dst_indices = np.linspace(0, len(h_vec)-1, target_len)
    return np.interp(dst_indices, src_indices, h_vec)
