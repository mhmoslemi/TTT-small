import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def evaluate_constraints(h_vec):
        # Evaluate all overlap constraints
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return corr * dx
    
    # Keep the best solution found
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 2000
    
    # Simulated Annealing approach with adaptive cooling
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    temp = 1.0
    min_temp = 1e-6
    cooling_rate = 0.995
    
    iteration = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Make a small random change to h
        h_new = current_h.copy()
        idx = np.random.randint(0, len(h_new))
        step_size = 0.05 * temp
        h_new[idx] += np.random.normal(0, step_size)
        
        # Project the new point
        h_new = project_h(h_new)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new)
        
        # Accept or reject based on simulated annealing
        if c5_new < current_c5 or np.random.rand() < np.exp((current_c5 - c5_new) / temp):
            current_h = h_new
            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Cool down temperature
        temp = max(min_temp, temp * cooling_rate)
        
        iteration += 1
        
        # Occasionally add diversity
        if iteration % 100 == 0:
            # Add some noise to avoid local minima
            noise = 0.005 * np.random.randn(len(current_h))
            current_h = current_h + noise
            current_h = project_h(current_h)
    
    return (best_h.tolist(), best_c5, n_points)
