import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(h, n_points, dx, alpha=100.0):
    """Compute smooth approximation to max function"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Soft maximum: log(sum(exp(alpha*x))) / alpha
    return np.log(np.sum(np.exp(alpha * corr * dx))) / alpha

def smooth_max_grad(h, n_points, dx, alpha=100.0):
    """Gradient of smooth max approximation"""
    eps = 1e-8
    grad = np.zeros_like(h)
    base_corr = np.correlate(h, 1 - h, mode='full')
    Z = np.sum(np.exp(alpha * base_corr * dx))
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        Z_plus = np.sum(np.exp(alpha * corr_plus * dx))
        grad[i] = (np.sum(np.exp(alpha * corr_plus * dx) * alpha * dx) / Z_plus - 
                   np.sum(np.exp(alpha * base_corr * dx) * alpha * dx) / Z) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Try different resolutions
    resolutions = [20, 30, 45, 60]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        if res != n_points:
            h_new = np.interp(np.linspace(0, 2, res), 
                              np.linspace(0, 2, n_points), h)
            h = project_to_feasible(h_new, res)
            n_points = res
            dx = 2.0 / n_points
    
    # Multi-start optimization with gradient-based method
    max_restarts = 10
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize with noise
        h_initial = project_to_feasible(
            h + np.random.normal(0, 0.05, n_points), n_points)
        
        # Gradient-based optimization with adaptive strategy
        def obj(h):
            return smooth_max(h, n_points, dx)
        
        def grad(h):
            return smooth_max_grad(h, n_points, dx)
        
        # Simple bound constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization parameters
        attempts = 0
        while attempts < 5:
            try:
                res = minimize(obj, h_initial, method='L-BFGS-B', 
                               bounds=bounds, jac=grad, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                break
            except:
                attempts += 1
                h_initial = project_to_feasible(
                    h_initial + np.random.normal(0, 0.02, n_points), n_points)
        if attempts >= 5:
            h_initial = project_to_feasible(
                h_initial + np.random.normal(0, 0.02, n_points), n_points)
    
    return (list(best_h), best_c5, n_points)
