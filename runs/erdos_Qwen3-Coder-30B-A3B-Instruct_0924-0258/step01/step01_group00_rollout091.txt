import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def generate_initial_h(n_points):
    """Generate a structured initial h"""
    if 'initial_h_values' in globals():
        return np.array(initial_h_values)
    else:
        # Create a simple initial design
        h = np.ones(n_points) * 0.5
        # Add some structure to encourage solutions that minimize overlaps
        h[0:n_points//2] = 0.7
        h[n_points//2:] = 0.3
        h = project_to_feasible(h, n_points)
        return h

def convex_relaxation(h, n_points, dx):
    """Use convex relaxation to approximate minimum overlap"""
    # Use a relaxed formulation with cvxpy for better handling
    # Minimizing max_k ∫ h(x)(1 - h(x+k)) dx
    # This becomes minimizing max_k (1 - ∫ h(x)h(x+k) dx)
    # So minimize max_k A(k) where A(k) = ∫ h(x)h(x+k) dx
    
    h_var = cp.Variable(n_points)
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points/2]
    
    # Approximate convolution constraints using linear inequalities
    # For simplicity, this is a handcrafted approximation
    # Note: full multivariable convex approach would be more sophisticated
    
    # Relaxation: minimize max of correlation terms
    obj = cp.Minimize(cp.max(cp.multiply(1 - h_var, 1 - h_var)))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        new_h = np.array(h_var.value).flatten()
        return project_to_feasible(new_h, n_points)
    except:
        return h.copy()

def genetic_mutation(h, n_points, mutation_rate=0.1):
    """Perform a simulated genetic mutation"""
    h_mut = h.copy()
    mutations = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, mutations, replace=False)
    for i in indices:
        # Gaussian mutation
        h_mut[i] += np.random.normal(0, 0.05)
    return project_to_feasible(h_mut, n_points)

def optimize_step(h_start, n_points, dx, time_limit, iteration=0):
    """Perform a step of optimization"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try a few different approaches
    for _ in range(3):
        if time() - start_time > time_limit:
            break
        
        # 1. Genetic mutation
        h_mut = genetic_mutation(best_h, n_points)
        c5 = evaluate_erdos_solution(h_mut, dx)
        if c5 < best_c5:
            best_h = h_mut.copy()
            best_c5 = c5
            
        # 2. Convex relaxation
        try:
            h_conv = convex_relaxation(best_h, n_points, dx)
            c5 = evaluate_erdos_solution(h_conv, dx)
            if c5 < best_c5:
                best_h = h_conv.copy()
                best_c5 = c5
        except:
            pass
        
        # 3. Simple random perturbation
        h_pert = project_to_feasible(best_h + np.random.normal(0, 0.02, n_points), n_points)
        c5 = evaluate_erdos_solution(h_pert, dx)
        if c5 < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Starting resolution
    resolutions = [15, 30, 45, 60, 75, 90]
    best_h = generate_initial_h(resolutions[0])
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    best_n_points = len(best_h)
    
    # Use a fine-grained resolution progression
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Adapt h to new resolution if needed
        if len(best_h) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(best_h)), 
                                best_h)
            best_h = project_to_feasible(h_interp, n_points)
        
        time_per_res = min((budget_s - (time() - start_time)) / (len(resolutions) - i), 
                           80)
        
        # Perform multiple optimization rounds
        h_current = best_h.copy()
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt, c5_opt = optimize_step(h_current, n_points, dx, 
                                        min(time_per_res / 5, 30), _)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            h_current = h_opt.copy()
    
    return (list(best_h), best_c5, best_n_points)
