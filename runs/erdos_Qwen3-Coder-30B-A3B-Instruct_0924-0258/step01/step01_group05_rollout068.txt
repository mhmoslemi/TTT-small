import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with 45 points
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Grid search with adaptive refinement
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Create different grid sizes to explore
    grid_sizes = [5, 10, 15, 20, 25, 30, 45]
    
    # For each grid size, create a batch of samples and evaluate
    for n in grid_sizes:
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Initialize current grid
        current_h = np.zeros(n)
        if n <= len(best_h):
            current_h[:n] = best_h[:n]
        else:
            # Interpolate from best solution
            x_orig = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n)
            current_h = np.interp(x_new, x_orig, best_h)
            
        # Sample multiple candidates per grid
        num_candidates = 50 if n < 30 else 20
        for _ in range(num_candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Create a variation using random perturbation
            h_candidate = current_h + np.random.normal(0, 0.1, n)
            h_candidate = project_feasible(h_candidate)
            
            # Evaluate this candidate
            C5_candidate = compute_C5(h_candidate)
            
            if C5_candidate < best_C5:
                best_C5 = C5_candidate
                best_h = h_candidate.copy()
        
        # After evaluating all candidates at this resolution, refine
        if n > 5 and n < 45:
            # Randomly sample a few around the best
            for _ in range(10):
                if time.time() - start_time > budget_s - 0.1:
                    break
                h_refined = best_h + np.random.normal(0, 0.05, n)
                h_refined = project_feasible(h_refined)
                C5_refined = compute_C5(h_refined)
                if C5_refined < best_C5:
                    best_C5 = C5_refined
                    best_h = h_refined.copy()
                    
    # Final optimization step with local search
    best_h = project_feasible(best_h)
    
    # Local search to fine-tune
    for iter in range(100):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all cross correlations
        corr = correlate(best_h, 1 - best_h, mode='full')
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        # Gradient direction
        grad = np.zeros_like(best_h)
        if max_lag >= 0:
            grad[max_lag:] = -best_h[:-max_lag]
        else:
            grad[:max_lag] = -best_h[-max_lag:]
            
        # Take step
        step_size = 0.05
        h_new = best_h - step_size * grad
        h_new = project_feasible(h_new)
        C5_new = compute_C5(h_new)
        
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h_new.copy()
    
    # Final check
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
