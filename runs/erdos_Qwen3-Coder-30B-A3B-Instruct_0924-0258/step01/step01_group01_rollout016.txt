import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 500
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent with projected updates
    max_iter = 10000
    iter_per_temp = 100
    lr = 0.01
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        base_val = evaluate(h, dx)
        
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += eps
            h_eps = project_to_constraints(h_eps, dx)
            val_eps = evaluate(h_eps, dx)
            grad[i] = (val_eps - base_val) / eps
            
        # Update step
        h_new = h - lr * grad
        h_new = project_to_constraints(h_new, dx)
        
        # Accept if better
        c5_new = evaluate(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
        elif c5_new < evaluate(h, dx) * 1.001:
            h = h_new
            
        # Rescaling
        if iter_count % iter_per_temp == 0 and n_points < n_high:
            # Interpolate to next resolution
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            best_c5 = evaluate(h, dx)
            if best_c5 < evaluate(best_h, dx):
                best_h = h.copy()
    
    return best_h.tolist(), best_c5, len(best_h)
