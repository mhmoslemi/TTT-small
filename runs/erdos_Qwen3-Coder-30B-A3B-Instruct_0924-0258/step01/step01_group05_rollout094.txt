import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def explore_neighbors(h_vec, num_neighbors=10):
        neighbors = []
        for _ in range(num_neighbors):
            # Add small random perturbations
            perturbation = np.random.normal(0, 0.02, len(h_vec))
            neighbor = h_vec + perturbation
            neighbor = project_feasible(neighbor)
            neighbors.append(neighbor)
        return neighbors
    
    def grid_search_coarse(n_vals, num_samples=50):
        candidates = []
        for n in n_vals:
            for _ in range(num_samples):
                # Random step function with binary values
                h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
                # Make sure sum is approximately target
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * target_sum / h_sum
                candidates.append(project_feasible(h))
        return candidates
    
    def grid_search_fine(h_candidates, n_points, num_samples=20):
        candidates = []
        # Take top 20% of candidates by C5 value
        c5_vals = [compute_C5(c) for c in h_candidates]
        indices = np.argsort(c5_vals)
        top_candidates = [h_candidates[i] for i in indices[:max(1, len(indices)//5)]]
        
        for candidate in top_candidates:
            # Refine around each top candidate
            for _ in range(num_samples):
                # Add small random perturbations
                perturbation = np.random.normal(0, 0.05, n_points)
                neighbor = candidate + perturbation
                neighbor = project_feasible(neighbor)
                candidates.append(neighbor)
        return candidates

    # Try several different starting grids and search strategies
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Grid search strategy with multiple restarts
    for restart in range(5):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Start with coarse grid exploration
        coarse_n_vals = [5, 10, 15]
        coarse_candidates = grid_search_coarse(coarse_n_vals, 30)
        
        # Evaluate all coarse candidates at fine resolution
        fine_candidates = []
        for candidate in coarse_candidates:
            # Evaluate at fine resolution
            if len(candidate) != n_points:
                # Interpolate to n_points resolution
                x_coarse = np.linspace(0, 2, len(candidate))
                x_fine = np.linspace(0, 2, n_points)
                h_fine = np.interp(x_fine, x_coarse, candidate)
                candidate = project_feasible(h_fine)
            fine_candidates.append(candidate)
        
        # Evaluate and select best coarse candidates
        for candidate in fine_candidates:
            c5 = compute_C5(candidate)
            if c5 < best_C5:
                best_C5 = c5
                best_h = candidate.copy()
        
        # Refine around best coarse candidate
        refined_candidates = grid_search_fine([best_h], n_points, 30)
        
        for candidate in refined_candidates:
            c5 = compute_C5(candidate)
            if c5 < best_C5:
                best_C5 = c5
                best_h = candidate.copy()
    
    # Final local search with more detailed perturbations
    current_h = best_h.copy()
    current_C5 = best_C5
    
    # Try hill climbing with various perturbations
    for _ in range(200):
        if time.time() - start_time > budget_s - 0.1:
            break
                    
        # Try a few neighbor candidates
        neighbors = explore_neighbors(current_h, 10)
        for neighbor in neighbors:
            c5 = compute_C5(neighbor)
            if c5 < current_C5:
                current_C5 = c5
                current_h = neighbor.copy()
                if c5 < best_C5:
                    best_C5 = c5
                    best_h = current_h.copy()
    
    # Final project to ensure feasibility
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
