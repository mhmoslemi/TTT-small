import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize at coarse resolution
    n_points = 10
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
    
    h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 100
    alpha = 0.01
    
    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # ADMM iterations
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Update h by minimizing with respect to current z
            # We approximate the max_k integral with a fixed k approach
            # Since exact gradient not easily available for max operation,
            # we use gradient descent approximation with small steps
            
            # Compute the gradient of correlation with respect to h
            z = 1 - h
            c = correlate(h, z, mode='full')
            k_max = np.argmax(c)
            
            # Approximate gradient by finite differences
            eps = 1e-4
            grad_h = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_constraints(h_plus, dx)
                c_plus = correlate(h_plus, 1-h_plus, mode='full')
                grad_h[i] = (c_plus[k_max] - c[k_max]) / eps
            
            # Update h with gradient descent
            h_new = h - alpha * grad_h
            h_new = project_to_constraints(h_new, dx)
            
            # Update z = 1 - h_new (ADMM constraint)
            z_new = 1 - h_new
            
            # Update multipliers
            # Dual variable update for constraint z = 1 - h
            # For simplicity, just use a fixed update step for multipliers
            
            # Update h
            h = h_new
            iter_count += 1
            
            current_c5 = evaluate(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
        # Resolution refinement
        if n_points < 100:  # Upper bound
            if time.time() - start_time > budget_s * 0.8:  # Leave some time
                break
            old_c5 = evaluate(h, dx)
            
            # Interpolate to higher resolution
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            n_points *= 2
            dx = 2.0 / n_points
            h = project_to_constraints(h_long, dx)
            
            current_c5 = evaluate(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            
        else:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
