import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial h values from parent solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(30) * 0.5
    
    n_points = len(h_start)
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 5000
    target = 0.38
    adjustment_fraction = 0.1
    perturb_threshold = 50
    perturb_strength = 0.01
    
    iter_count = 0
    no_improve_count = 0
    current_h = h_start.copy()
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        iter_count += 1
        
        # Compute overlap and find maximum
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_val = np.max(corr)
        max_idx = np.argmax(corr)
        
        # Find the shift k corresponding to max
        k = max_idx - (len(current_h) - 1)
        
        # Determine the indices where overlap is largest (using a sliding window approach)
        # This isn't exactly the same as focusing on the 'top 10%', but 
        # we approximate by considering all non-zero correlation terms 
        # and focusing on where they occur.
        
        # We look for regions that contribute significantly
        contributions = []
        for i in range(len(current_h)):
            # Shift k indices and compute h[i] * (1 - h[i+k])
            if 0 <= i + k < len(current_h):
                contributions.append((i, current_h[i] * (1 - current_h[i+k])))
        
        # Sort by contribution descending and take top portion
        contributions.sort(key=lambda x: x[1], reverse=True)
        top_contributors = [idx for idx, _ in contributions[:max(1, len(contributions)//10)]]
        
        # Adjust h values in best contributors
        h_new = current_h.copy()
        adjustment = adjustment_fraction * (max_val - target) / n_points
        
        for i in top_contributors:
            h_new[i] = max(0, h_new[i] - adjustment)
        
        # Normalize to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Perturb solution if no improvement
            if no_improve_count > perturb_threshold:
                h_perturbed = project_to_feasible(
                    current_h + np.random.normal(0, perturb_strength, n_points), 
                    n_points
                )
                perturbed_c5 = evaluate_erdos_solution(h_perturbed, dx)
                if perturbed_c5 < best_c5:
                    best_h = h_perturbed.copy()
                    best_c5 = perturbed_c5
                current_h = h_perturbed
                no_improve_count = 0
            else:
                current_h = h_new

    return (list(best_h), best_c5, n_points)
