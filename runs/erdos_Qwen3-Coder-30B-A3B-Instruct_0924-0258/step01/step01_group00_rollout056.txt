import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    else:

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Multi-resolution scheme with adaptive time allocation
    resolutions = [15, 20, 25, 30, 35, 40, 45]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        # Interpolate if needed
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time budget for this resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 60)
        
        # Multiple restarts with perturbed initialization
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Perturb initial point
            if restart == 0:
                h_perturbed = h_start.copy()
            else:
                # Pattern-based perturbation
                noise = np.random.normal(0, 0.05, n_points)
                h_perturbed = project_to_feasible(h_start + noise, n_points)
                
            # Optimization using L-BFGS-B to avoid constraints issues
            def obj(h):
                # Use a penalty approach to minimize max overlap
                corr = np.correlate(h, 1 - h, mode='full')
                return -np.max(corr) * dx  # We want to maximize min overlap (minimize max)
            
            def grad(h):
                # Gradient of the objective (simplified)
                corr = np.correlate(h, 1 - h, mode='full')
                max_idx = np.argmax(corr)
                # Gradient is more complex due to convolution
                # We'll use a finite difference approximation
                eps = 1e-6
                grad = np.zeros_like(h)
                for i in range(len(h)):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    grad[i] = -(corr_plus[max_idx] - corr[max_idx]) / eps
                return grad
            
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Solve with L-BFGS-B method
            try:
                res = minimize(obj, h_perturbed, method='L-BFGS-B', bounds=bounds, 
                               jac=grad, options={'maxiter': 100})

                    h_opt = project_to_feasible(res.x, n_points)

            except:
                pass
            
            # Subsequently use current best for next restart
            h_start = best_h.copy()
        
    return (list(best_h), best_c5, best_n_points)
