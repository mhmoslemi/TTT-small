import numpy as np
from scipy.optimize import minimize
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def ga_step(population, fitnesses):
        # Tournament selection
        def tournament_select(pop, fit, k=3):
            idxs = np.random.choice(len(pop), k)
            best = np.argmin([fit[i] for i in idxs])
            return pop[idxs[best]]
        
        # Create new population
        new_pop = []
        # Elitism: keep top 10%
        elite_idx = np.argsort(fitnesses)[:int(0.1*len(population))]
        for i in elite_idx:
            new_pop.append(population[i].copy())
        
        # Generate offspring
        while len(new_pop) < len(population):
            parent1 = tournament_select(population, fitnesses)
            parent2 = tournament_select(population, fitnesses)
            
            # Crossover
            if np.random.rand() < 0.8:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < 0.1:
                    child2[i] += np.random.normal(0, 0.05)
            
            new_pop.append(child1)
            if len(new_pop) < len(population):
                new_pop.append(child2)
        
        return new_pop[:len(population)]
    
    # Initialize population
    population_size = 50
    population = []
    for _ in range(population_size):
        h_init = h + np.random.normal(0, 0.05, n_points)
        h_init = project_h(h_init)
        population.append(h_init)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Genetic algorithm with local search
    for gen in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for individual in population:
            fitness = compute_C5(individual)
            fitnesses.append(fitness)
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_C5:
            best_C5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Periodic local search (Nelder-Mead)
        if gen % 50 == 0 and gen > 0:
            local_pop = [population[min_fitness_idx].copy()]
            # Add some nearby individuals to the local search
            for _ in range(4):
                idx = np.random.randint(0, population_size)
                neighbor = population[idx].copy()
                neighbor += np.random.normal(0, 0.02, n_points)
                neighbor = project_h(neighbor)
                local_pop.append(neighbor)
            
            # Local search with Nelder-Mead
            for i, candidate in enumerate(local_pop):
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Create a wrapper for scipy minimize
                def obj(h_local):
                    return compute_C5(h_local)
                
                try:
                    result = minimize(obj, candidate, method='Nelder-Mead', 
                            options={'maxiter': 100, 'disp': False})
                    if result.success:
                        candidate = result.x
                        candidate = project_h(candidate)
                        local_pop[i] = candidate
                except:
                    pass
            
            # Replace worst individuals with local search results
            worst_idx = np.argsort(fitnesses)[-len(local_pop):]
            for i, idx in enumerate(worst_idx):
                population[idx] = local_pop[i]
        else:
            # Evolutionary step
            population = ga_step(population, fitnesses)
        
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
