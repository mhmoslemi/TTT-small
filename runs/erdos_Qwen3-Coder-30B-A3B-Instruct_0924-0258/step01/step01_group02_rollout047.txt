import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure sum constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Simulated Annealing
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Initial temperature
    current_c5 = compute_c5(h)
    T = 10 * current_c5
    
    start_time = time.time()
    max_iter = 100000
    iter_count = 0
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Perturb solution
        h_new = h.copy()
        # Randomly adjust a few elements
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            # Add small random perturbation
            h_new[i] += np.random.normal(0, 0.01)
        
        h_new = project_h(h_new)
        
        # Compute new energy
        new_c5 = compute_c5(h_new)
        
        # Metropolis criterion
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            # Accept with probability based on temperature
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
                current_c5 = new_c5
        
        # Cool down
        T *= 0.9999
        if T < 1e-6:
            T = 1e-6
            
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
