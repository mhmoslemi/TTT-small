import numpy as np
from scipy import optimize
import time

def compute_overlap(h_vec):
    """Compute the maximum overlap integral for given h vector."""
    h_vec = np.array(h_vec)
    conv = np.correlate(h_vec, 1 - h_vec, mode='full')
    mid = len(conv) // 2
    overlaps = conv[mid - len(h_vec) + 1:mid + len(h_vec) - 1]
    return np.max(overlaps)

def project_feasible(h_vec, target_sum):
    """Project h_vec to feasible region: [0,1] and sum to target_sum."""
    h_vec = np.clip(h_vec, 0, 1)
    h_vec = h_vec * target_sum / np.sum(h_vec)
    return h_vec

    elite_size = pop_size // 5

    # Initialize population
    population = []
    for _ in range(pop_size):
        h_candidate = np.random.uniform(0.4, 0.6, n_points)
        h_candidate = project_feasible(h_candidate, target_sum)
        population.append(h_candidate)

    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for individual in population:
            f = compute_overlap(individual)
            fitness.append(f)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check if we found a better solution
        if fitness[0] < best_C5:
            best_C5 = fitness[0]
            best_h = population[0].copy()
        
        # Selection: top 20% as parents
        parents = population[:pop_size // 5]
        
        # Generate new population
        new_population = population[:elite_size]  # Keep elites
        
        # Local search with Nelder-Mead on top individuals
        if gen % 50 == 0:
            for i in range(5):
                if time.time() - start_time > budget_s - 1:
                    break
                x0 = population[i].copy()
                result = optimize.minimize(
                    lambda x: compute_overlap(project_feasible(x, target_sum)),
                    x0,
                    method='Nelder-Mead',
                    options={"maxiter": 20, "disp": False}
                )
                if result.success:
                    new_ind = project_feasible(result.x, target_sum)
                    if compute_overlap(new_ind) < compute_overlap(new_population[i]):
                        new_population[i] = new_ind
                        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if time.time() - start_time > budget_s - 1:
                break
                
            # Tournament selection
            parent1 = parents[np.random.randint(len(parents))]
            parent2 = parents[np.random.randint(len(parents))]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)

    # Final evaluation
    final_C5 = compute_overlap(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5

    return (best_h.tolist(), best_C5, n_points)
