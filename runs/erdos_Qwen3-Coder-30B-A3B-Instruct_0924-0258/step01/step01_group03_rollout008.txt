import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = target_sum."""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * target_sum / len(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available, else create simple uniform start
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5  # Simple uniform start
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize learning rate and iteration tracking
    learning_rate = 0.01
    last_improvement = 0
    stagnation_counter = 0
    
    # Keep track of best solution
    best_h = None
    best_c5 = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s:
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        k_best = np.argmax(corr)
        
        # Compute gradient
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = (i - k_best) % len(h)
            gradient[i] = 1 - 2 * h[j]
        
        # Update h
        h = h - learning_rate * gradient
        
        # Project to feasible region
        h = project_to_feasible(h, target_sum)
        
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Random perturbation if stagnated
        if stagnation_counter > 50:
            noise = np.random.uniform(-1e-3, 1e-3, len(h))
            h = h + noise
            h = project_to_feasible(h, target_sum)
            stagnation_counter = 0
        
        iteration += 1
        
        # Prevent infinite loops
        if iteration > 10000:
            break
    
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback to initial h
        final_h = project_to_feasible(h, target_sum)
        final_c5 = evaluate(final_h, dx)
        return (final_h.tolist(), float(final_c5), n_points)
