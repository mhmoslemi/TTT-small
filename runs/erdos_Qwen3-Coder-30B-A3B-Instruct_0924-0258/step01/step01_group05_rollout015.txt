import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with n=45
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def generate_grid_candidates(n_points, num_candidates):
        candidates = []
        for _ in range(num_candidates):
            # Generate random feasible steps
            h_candidate = np.random.rand(n_points)
            h_candidate = project_feasible(h_candidate)
            candidates.append(h_candidate)
        return candidates
    
    def refine_candidates(candidates, n_points):
        refined = []
        candidates.sort(key=compute_C5)
        top_candidates = candidates[:max(1, len(candidates)//5)]
        for cand in top_candidates:
            # Add some perturbation
            pert = np.random.normal(0, 0.05, n_points)
            perturbed = cand + pert
            perturbed = project_feasible(perturbed)
            refined.append(perturbed)
        return refined
    
    # Start with a coarse grid
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Grid levels from coarse to fine
    grid_levels = [5, 10, 15, 20, 25, 30, 35, 40, 45]
    
    # Try increasing resolutions
    for i, n in enumerate(grid_levels):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        if n == n_points:
            continue
            
        # Generate candidates for this resolution
        current_candidates = []
        if i == 0:
            # Start with random candidates
            current_candidates = generate_grid_candidates(n, 50)
        else:
            # Refine from previous best
            current_candidates = refine_candidates(previous_candidates, n)
            # Add some diversity
            current_candidates.extend(generate_grid_candidates(n, 20))
        
        # Evaluate candidates
        for candidate in current_candidates:
            if time.time() - start_time > budget_s - 0.1:
                break
            candidate = project_feasible(candidate)
            c5 = compute_C5(candidate)
            if c5 < best_C5:
                best_C5 = c5
                best_h = candidate.copy()
        
        previous_candidates = current_candidates
    
    # Final optimization by local descent around current best
    def local_descent(h_start, max_iter=30):
        h_current = h_start.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Compute correlations to find worst lag
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            # Create gradient vector for this lag
            grad = np.zeros_like(h_current)
            if max_lag >= 0:
                grad[max_lag:] = -h_current[:-max_lag]
            else:
                grad[:max_lag] = -h_current[-max_lag:]
            
            # Try gradient descent steps
            best_alpha = 0.0
            best_new_C5 = best_C5
            best_new_h = h_current.copy()
            for alpha in [0.01, 0.05, 0.1, 0.2]:
                candidate_h = h_current - alpha * grad
                candidate_h = project_feasible(candidate_h)
                new_C5 = compute_C5(candidate_h)
                if new_C5 < best_new_C5:
                    best_new_C5 = new_C5
                    best_new_h = candidate_h.copy()
                    best_alpha = alpha
            
            if best_new_C5 < best_C5:
                h_current = best_new_h.copy()
                best_C5 = best_new_C5
                best_h = h_current.copy()
            else:
                # No improvement, add noise to escape local optima
                noise = np.random.normal(0, 0.02, len(h_current))
                h_current += noise
                h_current = project_feasible(h_current)
        
        return best_h, best_C5
    
    # Finish with local descent
    final_h, final_C5 = local_descent(best_h, 20)
    
    return list(final_h), final_C5, len(final_h)
