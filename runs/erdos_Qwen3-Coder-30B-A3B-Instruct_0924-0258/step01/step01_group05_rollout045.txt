import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, max_lag):
        # Compute gradient for the specific lag that maximizes correlation
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        if max_lag >= 0:  # Positive lag
            # For lag k: grad[i] = (1-h[i+k]) - h[i-k] 
            grad[max_lag:] = -(1 - h_vec[:-max_lag])
            grad[:-max_lag] += h_vec[max_lag:]
        else:  # Negative lag
            # For lag k: grad[i] = (1-h[i+k]) - h[i-k] 
            grad[:max_lag] = -(1 - h_vec[-max_lag:])
            grad[-max_lag:] += h_vec[:max_lag]
        return grad
    
    def line_search(h_vec, grad, alpha_max=1.0, max_iter=10):
        best_h = h_vec.copy()
        best_C5 = compute_C5(h_vec)
        best_alpha = 0.0
        
        # Test a sequence of alpha values
        step_sizes = [alpha_max * (0.5 ** i) for i in range(max_iter)]
        
        for alpha in step_sizes:
            if time.time() - start_time > budget_s - 0.1:
                break
            candidate_h = h_vec - alpha * grad
            candidate_h = project_feasible(candidate_h)
            new_C5 = compute_C5(candidate_h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = candidate_h.copy()
                best_alpha = alpha
        return best_h, best_C5, best_alpha
    
    # Main optimization loop
    h_current = h.copy()
    C5_current = compute_C5(h_current)
    best_h = h_current.copy()
    best_C5 = C5_current
    alpha = 0.1
    stagnation_count = 0
    max_stagnation = 50
    
    # Use a simple gradient descent approach with line search
    while time.time() - start_time < budget_s - 0.1:
        # Compute all maximal correlations
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient for this lag
        grad = compute_gradient(h_current, max_lag)
        
        # Do line search along this gradient
        new_h, new_C5, found_alpha = line_search(h_current, grad, alpha_max=alpha, max_iter=8)
        
        # If we didn't improve, reduce alpha and try again
        if new_C5 >= C5_current:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                # Perturb slightly
                noise = np.random.normal(0, 0.02, n_points)
                h_current = project_feasible(h_current + noise)
                stagnation_count = 0
                alpha = 0.1
            else:
                alpha *= 0.5  # Reduce step size
        else:
            h_current = new_h
            C5_current = new_C5
            stagnation_count = 0
            alpha = 0.1  # Reset step size on improvement
            
            if C5_current < best_C5:
                best_C5 = C5_current
                best_h = h_current.copy()
    
    # Final projection and evaluation
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
