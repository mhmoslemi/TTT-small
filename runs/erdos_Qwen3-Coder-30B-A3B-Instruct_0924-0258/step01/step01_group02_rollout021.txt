import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 10 * best_c5
    cooling_rate = 0.95
    min_T = 1e-6
    iterations_per_temp = 500
    
    iteration = 0
    while T > min_T:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        for _ in range(iterations_per_temp):
            # Create a neighbor solution
            h_new = h.copy()
            
            # Randomly choose indices to perturb
            num_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            
            # Perturb selected values
            for i in indices:
                # Add random delta in [-0.1, 0.1]
                delta = np.random.uniform(-0.1, 0.1)
                h_new[i] += delta
            
            # Project to feasible set
            h_new = project_h(h_new)
            
            # Compute energies
            c5_old = compute_c5(h)
            c5_new = compute_c5(h_new)
            
            # Annealing acceptance criterion
            if c5_new < c5_old or np.random.rand() < np.exp(-(c5_new - c5_old) / T):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            
            iteration += 1
            
            # Early termination if no improvement for a while
            if iteration % 1000 == 0:
                elapsed = time.time() - start_time
                if elapsed > budget_s:
                    break
        
        T *= cooling_rate
    
    return (best_h.tolist(), best_c5, n_points)
