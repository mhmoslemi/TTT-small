import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx, critical_lags):
    # compute gradient w.r.t. each h[i] based on critical lags
    n = len(h)
    grad = np.zeros(n)
    for k in critical_lags:
        # correlation at lag k is sum_{i} h[i] * (1 - h[i+k])
        # so d/dh[i] term is (1 - h[i+k]) if k >= 0 and i+k < n
        # and d/dh[i+k] term is -h[i] if k < 0 and i >= -k
        if k >= 0:
            # for h[i], contribution from h[i+k] term (with 1-h[i+k])
            if i+k < n:
                grad[i] += (1 - h[i+k])
        else:
            # k < 0: contribution to h[i+k] (from h[i])
            if i >= -k:
                grad[i+k] -= h[i]
    return grad

def line_search(h, grad, step_size, dx):
    # Line search along gradient direction
    # Project to upper bound
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_constraints(h_new, dx)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():

        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    max_iter_per_res = 1000
    max_iter = max_iter_per_res * 10
    iter_count = 0
    
    # Optimization parameters
    learning_rate = 0.05
    decay = 0.99
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s * 0.95:
            break
        
        # Get current maximum correlation values and critical lags
        corr = correlate(h, 1-h, mode='full')
        k_values = np.arange(-len(h)+1, len(h))
        # Global maximum lag
        max_k = k_values[np.argmax(corr)]
        # Top 5% lags with highest values
        top_indices = np.argsort(corr)[-max(1, len(corr)//20):][::-1]
        critical_lags = k_values[top_indices]
        
        # Adjust learning rate
        if iter_count > 0 and iter_count % 100 == 0:
            learning_rate *= decay
            
        # Compute approximate gradient
        grad = np.zeros(n_points)
        for k_idx, k in enumerate(critical_lags):
            # Handle boundary cases for indices
            if k >= 0:
                # diagonal offset interaction
                for i in range(n_points - k):
                    grad[i] += (1 - h[i+k])
            else:
                # negative lag, so contribution from i+k to i
                k_abs = -k
                for i in range(k_abs, n_points):
                    grad[i] -= h[i-k_abs]
        
        # Normalize gradient
        max_grad = np.max(np.abs(grad))
        if max_grad > 1e-12:
            grad /= max_grad
            
        # Gradient step
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        
        c5_new = evaluate(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Update h
        h = h_new
        iter_count += 1
        
        # resolution adaptation
        if n_points < n_high and iter_count % (max_iter_per_res // 4) == 0:
            # If better solution, interpolate
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            iter_count = 0  # reset counter
    
    return best_h.tolist(), best_c5, len(best_h)
