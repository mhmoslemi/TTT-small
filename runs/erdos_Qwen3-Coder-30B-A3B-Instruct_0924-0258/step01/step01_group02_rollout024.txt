import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction with 45 points
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure the constraint ∫h = 1 is met (sum(h) * dx = 1)
    h = h * target_sum / np.sum(h)

    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range and rescale to satisfy the sum constraint
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Start with current h
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Simulated Annealing settings
    init_temp = 10 * current_c5
    temperature = init_temp
    cooling_rate = 0.95
    min_temp = 1e-6
    iterations_per_temp = 500
    
    start_time = time.time()
    
    while temperature > min_temp:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        for _ in range(iterations_per_temp):
            # Make a perturbation
            h_new = current_h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.02)
            h_new[idx] += delta
            
            # Apply projection to satisfy constraints
            h_new = project_h(h_new)
            
            # Compute new overlap
            c5_new = compute_c5(h_new)
            
            # Metropolis criterion
            if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temperature):
                current_h = h_new
                current_c5 = c5_new
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        
        # Cool down
        temperature *= cooling_rate

    return (best_h.tolist(), best_c5, n_points)
