import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def coordinate_descent_step(h_vec, step_size=0.05):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        for i in range(len(h_current)):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Try small positive and negative steps
            for delta in [-step_size, step_size]:
                h_candidate = h_current.copy()
                h_candidate[i] += delta
                h_candidate = project_feasible(h_candidate)
                new_C5 = compute_C5(h_candidate)
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h_candidate.copy()
            
            h_current = best_h.copy()
        
        return best_h, best_C5
    
    def swap_optimization(h_vec, num_swaps=10):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        best_h = h_current.copy()
        
        for _ in range(num_swaps):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            i, j = np.random.randint(0, len(h_current), 2)
            if i == j:
                continue
                
            h_candidate = h_current.copy()
            h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]
            h_candidate = project_feasible(h_candidate)
            new_C5 = compute_C5(h_candidate)
            
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_candidate.copy()
                h_current = best_h.copy()
        
        return best_h, best_C5
    
    def simplex_search(h_vec, simplex_size=3):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        best_h = h_current.copy()
        
        # Generate simplex around current point
        simplex = [h_current.copy()]
        for _ in range(simplex_size - 1):
            perturbation = np.random.normal(0, 0.02, len(h_current))
            h_perturbed = h_current + perturbation
            h_perturbed = project_feasible(h_perturbed)
            simplex.append(h_perturbed.copy())
        
        # Evaluate and find best
        simplex_values = [compute_C5(h) for h in simplex]
        best_idx = np.argmin(simplex_values)
        if simplex_values[best_idx] < best_C5:
            best_C5 = simplex_values[best_idx]
            best_h = simplex[best_idx].copy()
        
        return best_h, best_C5
    
    # Generate diverse initial candidates
    initial_candidates = []
    for _ in range(5):
        if time.time() - start_time > budget_s - 0.1:
            break
        h_candidate = np.random.rand(n_points)
        h_candidate = project_feasible(h_candidate)
        initial_candidates.append(h_candidate)
    
    # Add existing solution if available
    initial_candidates.append(h.copy())
    
    # Find best initial candidate
    best_h = initial_candidates[0]
    best_C5 = compute_C5(initial_candidates[0])
    
    for candidate in initial_candidates[1:]:
        C5_candidate = compute_C5(candidate)
        if C5_candidate < best_C5:
            best_C5 = C5_candidate
            best_h = candidate.copy()
    
    # Local optimization loop
    no_improvement_count = 0
    last_improvement_time = time.time()
    
    while time.time() - start_time < budget_s - 0.1:
        # Try coordinate descent
        h_opt, C5_opt = coordinate_descent_step(best_h, step_size=0.05)
        if C5_opt < best_C5:
            best_C5 = C5_opt
            best_h = h_opt.copy()
            no_improvement_count = 0
            last_improvement_time = time.time()
        else:
            no_improvement_count += 1
            
        # Try swap optimization
        h_opt, C5_opt = swap_optimization(best_h, num_swaps=10)
        if C5_opt < best_C5:
            best_C5 = C5_opt
            best_h = h_opt.copy()
            no_improvement_count = 0
            last_improvement_time = time.time()
            
        # Try simplex search
        h_opt, C5_opt = simplex_search(best_h, simplex_size=3)
        if C5_opt < best_C5:
            best_C5 = C5_opt
            best_h = h_opt.copy()
            no_improvement_count = 0
            last_improvement_time = time.time()
            
        # If no improvement for a while, try random perturb
        if no_improvement_count > 20:
            noise = np.random.normal(0, 0.02, n_points)
            h_perturbed = best_h + noise
            h_perturbed = project_feasible(h_perturbed)
            C5_perturbed = compute_C5(h_perturbed)
            if C5_perturbed < best_C5:
                best_C5 = C5_perturbed
                best_h = h_perturbed.copy()
            no_improvement_count = 0
            
        # Reset if stuck too long
        if time.time() - last_improvement_time > 30:
            # Generate new random candidate
            h_new = np.random.rand(n_points)
            h_new = project_feasible(h_new)
            C5_new = compute_C5(h_new)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
            last_improvement_time = time.time()
    
    # Final validation
    h_best = project_feasible(best_h)
    final_C5 = compute_C5(h_best)
    
    return list(h_best), final_C5, n_points
