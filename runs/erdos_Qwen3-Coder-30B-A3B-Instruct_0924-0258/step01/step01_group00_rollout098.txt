import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Greedy iterative improvement
    max_iter = 1000
    iter_count = 0
    improvement_count = 0
    last_improvement = 0
    target_c5 = 0.38
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        if iter_count - last_improvement > 50:
            # Add noise if no improvement in a while
            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points, dx)
        
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            improvement_count += 1
            last_improvement = iter_count
        
        # Find the maximum correlation index
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        # Full correlation length is 2*n_points-1, so we shift k = max_idx - (n_points-1)
        k_shift = max_idx - (n_points - 1)
        
        # We can be more precise about identifying conflicting regions
        # The peak at k_shift overlaps the most
        # Find regions where h(x) and h(x+k) both contribute significantly
        
        # Determine which positions are contributing the most to overlap at current peak
        # Extract the shift k from the full correlation
        # corr has length 2*n_points-1, index 0 to 2*n_points-2
        # The actual shift k is k_shift = max_idx - (n_points-1)
        
        # Check it in terms of valid x-intervals
        # Here we target high overlap regions by adjusting h
        weights = h * (1 - h)  # This roughly estimates where both h(x) and 1-h(x+k) have a positive presence
        weights = np.convolve(weights, np.ones(abs(k_shift) + 1), mode='valid')
        
        # For simplicity, assume we adjust both at highest y values
        # Find top 10% contribution indices
        n_top = max(1, int(n_points * 0.1))
        top_indices = np.argpartition(weights, -n_top)[-n_top:]
        
        # Create perturbation
        perturbation = np.zeros_like(h)
        if len(top_indices) > 0:
            # Adjust h indices, increase 1-h(x) component
            for i in top_indices:
                # boundaries and handling of regions
                if i < n_points and i + k_shift >= 0 and i + k_shift < n_points:
                    # we can increase h(i) to reduce overlap
                    # Detail handling - major care needed
                    if h[i] < 1:
                        reduction = min(0.02, (c5 - target_c5) / n_points)

                            perturbation[i] += reduction
                        if h[i + k_shift] > 0:
                            perturbation[i + k_shift] += -min(reduction, h[i + k_shift])

    c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), c5, n_points)
