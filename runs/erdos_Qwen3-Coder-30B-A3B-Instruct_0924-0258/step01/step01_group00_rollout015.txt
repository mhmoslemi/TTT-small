import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_relaxation(h, n_points, dx):
    # Create input for convex optimization to minimize max correlation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.sum(corr**2) * dx**2

def gradient_of_convex(h, n_points, dx):
    eps = 1e-8
    grad = np.zeros_like(h)
    base_obj = convex_relaxation(h, n_points, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        obj_plus = convex_relaxation(h_plus, n_points, dx)
        grad[i] = (obj_plus - base_obj) / eps
    return grad

def genetic_mutation(h, n_points, mutation_rate=0.1):
    # Simple genetic mutation: perturb some elements
    h_mut = h.copy()
    indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    for i in indices:
        h_mut[i] += np.random.normal(0, 0.05)
    return project_to_feasible(h_mut, n_points)

def refine_with_convex(h, n_points, dx, time_limit):
    # Refine using convex optimization
    start_time = time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Constraints: sum(h) = n_points/2 and 0 <= h_i <= 1
    h_vars = cp.Variable(n_points)
    constraints = [cp.sum(h_vars) == n_points / 2,
                   h_vars >= 0, h_vars <= 1]
    
    # Objective: minimize max correlation
    corr = np.correlate(h_vars.value, 1 - h_vars.value, mode='full')
    obj = cp.Minimize(cp.max(corr))
    
    try:
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_opt = h_vars.value
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
    except:
        pass
    
    return best_h, best_c5

    best_n_points = len(h_start)
    
    for i, n_points in enumerate(resolution_levels):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)),

            h_start = project_to_feasible(h_interp, n_points)
        
        # Track time for current resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolution_levels), 50)
        
        # Store best solution for this resolution
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        # Genetic algorithm for this resolution
        population_size = 20
        population = [project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)

        for gen in range(50):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Evaluate fitness
            fitness = [evaluate_erdos_solution(h, dx) for h in population]
            sorted_indices = np.argsort(fitness)
            
            # Select best
            top_k = population_size // 4
            elite = [population[i] for i in sorted_indices[:top_k]]
            
            # Create new population
            new_population = elite.copy()
            while len(new_population) < population_size:
                parent1 = elite[np.random.randint(0, len(elite))]
                parent2 = elite[np.random.randint(0, len(elite))]

                child = genetic_mutation(child, n_points, mutation_rate=0.3)
                new_population.append(child)
            
            population = new_population[:]
            
            # Keep track of best
            current_best = min(fitness)
            if current_best < best_c5_level:
                best_c5_level = current_best
                idx = np.argmin(fitness)
                best_h_level = population[idx].copy()
                
        # Use convex refinement
        h_refined, c5_refined = refine_with_convex(best_h_level, n_points, dx, time_per_res / 10)
        if c5_refined < best_c5_level:
            best_c5_level = c5_refined
            best_h_level = h_refined.copy()
            
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
            
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
