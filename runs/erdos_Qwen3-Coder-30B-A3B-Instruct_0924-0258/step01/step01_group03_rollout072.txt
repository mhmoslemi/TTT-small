import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize initial h
    h0 = h0 * (n_points / 2.0) / np.sum(h0)
    h0 = np.clip(h0, 0, 1)
    
    # Initialize
    h = h0.copy()
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    no_improve_count = 0
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        sum_h = np.sum(h_vec)
        if abs(sum_h) > 1e-10:
            h_vec = h_vec * (n_points / 2.0) / sum_h
        return np.clip(h_vec, 0, 1)
    
    def compute_gradient(h_vec, s_max):
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        for i in range(n):
            if 0 <= i + s_max < n:
                grad[i] += 1 - 2 * h_vec[i + s_max]
            if 0 <= i - s_max < n:
                grad[i] -= 1 - 2 * h_vec[i - s_max]
        return grad
    
    iter_count = 0
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Find shift that maximizes cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        s_max = np.argmax(corr) - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, s_max)
        
        # Update h
        h = h - learning_rate * grad
        h = project_h(h)
        
        # Evaluate
        c5_val = evaluate(h, dx)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > patience:
            learning_rate *= 0.5
            no_improve_count = 0
            
        iter_count += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
