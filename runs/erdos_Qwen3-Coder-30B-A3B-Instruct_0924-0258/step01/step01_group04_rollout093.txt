import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

        n_points = len(h)

        # Fallback: create a simple initial guess

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def upsample_h(h_vec, target_len):
        x_old = np.linspace(0, 1, len(h_vec))
        x_new = np.linspace(0, 1, target_len)
        return np.interp(x_new, x_old, h_vec)

    # Multi-resolution approach
    resolution_levels = [5, 10, 20, 40, n_points]
    current_h = h.copy()
    current_C5 = compute_overlap(current_h)
    
    for i, level in enumerate(resolution_levels):
        # Set up optimization variables
        if level < n_points:
            current_h = upsample_h(current_h, level)
        else:
            current_h = current_h[:n_points]
            
        n_current = len(current_h)
        dx_current = 2.0 / n_current
        target_sum_current = n_current / 2.0
        
        # Adam optimizer parameters
        m = np.zeros_like(current_h)
        v = np.zeros_like(current_h)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        alpha = 0.01
        
        best_current_h = current_h.copy()
        best_C5 = compute_overlap(best_current_h)
        
        # Early stopping counter
        no_improvement = 0
        max_no_improvement = 1000
        
        for iteration in range(10000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient using finite differences
            current_C5 = compute_overlap(current_h)
            gradient = np.zeros_like(current_h)
            eps = 1e-5
            
            for j in range(n_current):
                h_plus = current_h.copy()
                h_plus[j] += eps
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                h_minus = current_h.copy()
                h_minus[j] -= eps
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                gradient[j] = (C_plus - C_minus) / (2 * eps)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * (gradient ** 2)
            m_hat = m / (1 - beta1 ** (iteration + 1))
            v_hat = v / (1 - beta2 ** (iteration + 1))
            
            current_h = current_h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
            current_h = project_feasible(current_h)
            
            # Check for improvement
            new_C5 = compute_overlap(current_h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_current_h = current_h.copy()
                no_improvement = 0
                alpha = min(alpha * 1.05, 0.1)  # Adaptive learning rate
            else:
                no_improvement += 1
                alpha = max(alpha * 0.95, 1e-6)  # Decrease if not improving
                
            if no_improvement > max_no_improvement:
                break
                
        current_h = best_current_h.copy()
        
    final_C5 = compute_overlap(current_h)
    if final_C5 < current_C5:
        current_C5 = final_C5
        current_h = current_h.copy()
        
    return (current_h.tolist(), current_C5, n_points)
