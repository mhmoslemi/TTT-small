import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm parameters
    pop_size = 50
    tournament_size = 10
    elite_size = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    max_gens = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_rand = np.random.rand(n_points)
        h_rand = project_feasible(h_rand)
        population.append(h_rand)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 0.1 and generation < max_gens:
        # Evaluate fitness of all individuals
        fitness = [compute_C5(ind) for ind in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_C5:
            best_C5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            # Reset some individuals if stuck
            for i in range(len(population)):
                if np.random.rand() < 0.3:
                    population[i] = np.random.rand(n_points)
                    population[i] = project_feasible(population[i])
            no_improvement_count = 0
        
        # Selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size, replace=False)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        new_population = [population[i].copy() for i in elite_indices]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate:
                parent1, parent2 = np.random.choice(pop_size, 2, replace=False)
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, selected[parent1], selected[parent2])
            else:
                # Select one parent for mutation only
                parent = np.random.choice(pop_size)
                child = selected[parent].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_feasible(child)
            elif np.random.rand() < 0.1:  # 10% chance for reset mutation
                # Reset some genes
                reset_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                child[reset_indices] = np.random.choice([0, 1], size=len(reset_indices))
                child = project_feasible(child)
            
            new_population.append(child)
        
        # Keep population size correct
        population = new_population[:pop_size]
        generation += 1
        
        # Occasionally try local improvements
        if generation % 5 == 0:
            for i in range(len(population)):
                if time.time() - start_time > budget_s - 0.1:
                    break
                # Local perturbation
                perturb = population[i] + np.random.normal(0, 0.01, n_points)
                perturb = project_feasible(perturb)
                local_C5 = compute_C5(perturb)
                if local_C5 < fitness[i]:
                    population[i] = perturb
                    fitness[i] = local_C5

    # Final check
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
