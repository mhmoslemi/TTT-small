import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Precompute the correlation indices for efficiency
    def compute_correlation_indices():
        indices = []
        for k in range(1, n_points):
            indices.append(k)
        return indices

    # Genetic Algorithm with Local Search
    def ga_with_local_search():
        population_size = 50
        generations = 500
        crossover_rate = 0.8
        mutation_rate = 0.1
        elite_size = 5
        
        # Initialize population
        population = []
        for _ in range(population_size):
            h_init = np.random.uniform(0.4, 0.6, n_points)
            h_init = project_h(h_init)
            population.append(h_init)
        
        best_h = None
        best_C5 = float('inf')
        
        for gen in range(generations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate population
            fitness = []
            for h_ind in population:
                fit = compute_C5(h_ind)
                fitness.append(fit)
                
            # Sort by fitness
            sorted_indices = np.argsort(fitness)
            population = [population[i] for i in sorted_indices]
            fitness = [fitness[i] for i in sorted_indices]
            
            # Update best solution
            if fitness[0] < best_C5:
                best_C5 = fitness[0]
                best_h = population[0].copy()
                
            # Apply elitism
            new_population = population[:elite_size]
            
            # Generate offspring
            for _ in range(population_size - elite_size):
                # Tournament selection
                parent1 = tournament_selection(population, fitness, 3)
                parent2 = tournament_selection(population, fitness, 3)
                
                # Crossover
                if np.random.random() < crossover_rate:
                    child = crossover(parent1, parent2)
                else:
                    child = parent1.copy()
                
                # Mutation
                if np.random.random() < mutation_rate:
                    child = mutate(child)
                
                # Project to feasible region
                child = project_h(child)
                
                new_population.append(child)
                
            population = new_population
            
            # Local search every 50 generations
            if gen % 50 == 0:
                current_best = population[0]
                for i in range(min(5, len(population))):
                    candidate = population[i]
                    # Nelder-Mead local search
                    try:
                        res = minimize(compute_C5, candidate, method='Nelder-Mead', 
                                       options={'maxiter': 20}, callback=lambda x: None)
                        if res.fun < compute_C5(current_best):
                            current_best = res.x
                    except:
                        pass
                # Re-evaluate and update
                if compute_C5(current_best) < best_C5:
                    best_C5 = compute_C5(current_best)
                    best_h = current_best.copy()
        
        return best_h, best_C5
    
    def tournament_selection(population, fitness, k):
        indices = np.random.choice(len(population), k, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.random(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        return child
    
    def mutate(h_vec):
        # Add Gaussian noise to some elements
        noise_level = 0.05
        mask = np.random.random(n_points) < 0.2
        h_vec = h_vec + np.random.normal(0, noise_level, n_points) * mask
        return h_vec
    
    # Run GA with local search
    best_h, best_C5 = ga_with_local_search()
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        
    return (best_h.tolist(), best_C5, n_points)
