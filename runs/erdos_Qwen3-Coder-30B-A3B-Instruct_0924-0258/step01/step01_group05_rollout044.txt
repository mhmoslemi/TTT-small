import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with 45 points
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def grid_search_level(n_points_level, num_candidates=50):
        best_h = None
        best_C5 = float('inf')
        
        for _ in range(num_candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Generate random feasible vector
            h_candidate = np.random.rand(n_points_level)
            h_candidate = project_feasible(h_candidate)
            
            C5 = compute_C5(h_candidate)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_candidate.copy()
        
        return best_h, best_C5
    
    def refine_candidates(candidates, level):
        new_candidates = []
        num_to_refine = max(1, len(candidates) // 5)
        candidates_sorted = sorted(candidates, key=lambda x: x[1])
        to_refine = candidates_sorted[:num_to_refine]
        
        for h, C5 in to_refine:
            if time.time() - start_time > budget_s - 0.1:
                break
            # Add noise and refine
            h_refined = h + np.random.normal(0, 0.05, len(h))
            h_refined = project_feasible(h_refined)
            new_candidates.append((h_refined, compute_C5(h_refined)))
            
        return new_candidates
    
    # Start with coarse grid search
    levels = [5, 10, 15, 20, 25, 30, 45]
    current_candidates = []
    
    # Level 1 (coarse)
    h_level1, c5_level1 = grid_search_level(5, 50)
    if h_level1 is not None:
        current_candidates.append((h_level1, c5_level1))
    
    # Level 2 (coarse) 
    h_level2, c5_level2 = grid_search_level(10, 50)
    if h_level2 is not None:
        current_candidates.append((h_level2, c5_level2))
        
    # Level 3 (middle)
    h_level3, c5_level3 = grid_search_level(15, 50)
    if h_level3 is not None:
        current_candidates.append((h_level3, c5_level3))
        
    # Refine at medium levels
    refined_candidates = refine_candidates(current_candidates, 15)
    current_candidates.extend(refined_candidates)
    
    # Level 4 (fine)
    h_level4, c5_level4 = grid_search_level(20, 20)
    if h_level4 is not None:
        current_candidates.append((h_level4, c5_level4))
        
    # Level 5 (fine)
    h_level5, c5_level5 = grid_search_level(25, 20)
    if h_level5 is not None:
        current_candidates.append((h_level5, c5_level5))
        
    # Level 6 (finer)
    h_level6, c5_level6 = grid_search_level(30, 20)
    if h_level6 is not None:
        current_candidates.append((h_level6, c5_level6))
        
    # Refine at fine levels
    refined_candidates = refine_candidates(current_candidates, 30)
    current_candidates.extend(refined_candidates)
    
    # Final level (highest resolution)
    h_level7, c5_level7 = grid_search_level(45, 20)
    if h_level7 is not None:
        current_candidates.append((h_level7, c5_level7))
    
    # Extract best solution from all candidates
    best_solution = min(current_candidates, key=lambda x: x[1])
    h_best = best_solution[0]
    C5_best = best_solution[1]
    
    # Further local refinement around the best
    max_refinement = 10
    for _ in range(max_refinement):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Perturb slightly
        h_perturb = h_best + np.random.normal(0, 0.02, len(h_best))
        h_perturb = project_feasible(h_perturb)
        
        C5_perturb = compute_C5(h_perturb)
        if C5_perturb < C5_best:
            h_best = h_perturb
            C5_best = C5_perturb
    
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
