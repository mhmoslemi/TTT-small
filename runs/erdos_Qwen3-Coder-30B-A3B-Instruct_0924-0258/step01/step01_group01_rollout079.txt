import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_step(h, dx, step_size=0.01):
    n = len(h)
    # Compute gradient of C5 w.r.t. h
    # C5 = max_k int h(x)(1-h(x+k)) dx
    # Gradient approximated by local correlation
    corr_h = correlate(h, 1-h, mode='full')
    # We want to find the gradient wrt each entry
    # Approximate using finite differences / correlations
    # This adjustment depends on all positions due to overlap
    # Assume gradient for term h_i is related to sum of correlations
    # Let's compute a local approximation based on current h
    # We use the fact that perturbing h_i causes small changes in overlap
    # We're looking for a direction to decrease C5
    # Gradient is approximated by approximating d/dh_i of the max of correlations
    
    # Gradient of h_i term in overlap integral:
    # dh_i / dh_i = 1, but (1-h_i) changes with h_i
    # Total contribution to C5 of h_i is (1-2*h_i) * dx
    # A better estimate involves differentiating the convolution
    # But in practice, we build a gradient estimate by local evaluation
    # Do local gradient descent by computing finite differences
    
    grad = np.zeros(n)
    eps = 1e-5 * step_size
    # For each element, compute finite diff
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] = min(h[i] + eps, 1.0)
        h_minus = h.copy()
        h_minus[i] = max(h[i] - eps, 0.0)
        c_plus = evaluate(project_to_constraints(h_plus, dx), dx)
        c_minus = evaluate(project_to_constraints(h_minus, dx), dx)
        grad[i] = (c_plus - c_minus) / (2*eps)
    
    return grad
