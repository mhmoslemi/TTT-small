import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    # Parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute correlations for all shifts
        corr = correlate(h, 1 - h, mode="full")
        # Find the shift that maximizes overlap
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient (simplified computation)
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Forward shift case (simplified for key shift)
            j = (i + k_max - (n_points - 1)) % n_points
            if j < n_points:
                grad[i] += (1 - h[j])
        
        # Apply gradient descent
        h = h - learning_rate * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Evaluate and update best
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if patience_counter >= patience:
            learning_rate *= 0.9
            patience_counter = 0
            
            # Perturb slightly if stuck
            noise = np.random.normal(0, 1e-4, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            
    return (best_h.tolist(), float(best_c5), n_points)
