import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        """Compute maximum overlap integral C5"""
        h_vec = np.array(h_vec)
        # Compute correlations for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        """Project h_vec onto feasible set: 0 <= h <= 1, sum(h) = target_sum"""
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm Parameters
    pop_size = 50
    tournament_size = 10
    elite_size = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_candidate = np.random.rand(n_points)
        h_candidate = project_feasible(h_candidate)
        population.append(h_candidate)
    
    # Evaluate initial population
    fitness = [compute_C5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_C5 = fitness[best_idx]
    
    # Track best for early stopping
    no_improvement_count = 0
    prev_best_C5 = best_C5
    
    # GA loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_C5:
            best_C5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            # Reset some individuals to maintain diversity
            for i in range(pop_size // 5):
                if np.random.rand() < 0.5:
                    idx = np.random.randint(pop_size // 2, pop_size)
                    population[idx] = np.random.rand(n_points)
                    population[idx] = project_feasible(population[idx])
                    fitness[idx] = compute_C5(population[idx])
            no_improvement_count = 0
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        # Selection, crossover, mutation
        while len(new_population) < pop_size:
            # Tournament selection
            selected = []
            for _ in range(2):
                tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                selected.append(population[winner_idx])
            
            parent1, parent2 = selected
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                # Copy parents
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                # Gaussian mutation
                mutate = np.random.rand(n_points) < mutation_rate
                if np.any(mutate):
                    noise = np.random.normal(0, 0.05, n_points)
                    child += noise * mutate
                    child = project_feasible(child)
                
                # Optional reset mutation (5% of genes)
                if np.random.rand() < 0.3:
                    reset_genes = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                    child[reset_genes] = np.random.choice([0, 1], size=len(reset_genes))
                    child = project_feasible(child)
            
            new_population.extend([child1, child2])
            
        # Keep only pop_size individuals
        population = new_population[:pop_size]
        
        # Re-evaluate fitness
        fitness = [compute_C5(h) for h in population]
    
    # Final validation
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
