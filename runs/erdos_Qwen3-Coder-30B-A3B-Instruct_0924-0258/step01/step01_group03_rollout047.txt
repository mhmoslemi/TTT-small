import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize h0 to have sum = n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    h = h0.copy()
    
    dx = 2.0 / n_points
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent with adaptive learning rate
    max_iter = 10000
    learning_rate = 0.01
    patience = 50
    best_iter = 0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Recompute correlation and find maximizing shift
        corr_full = correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(corr_full) - (len(h) - 1)
        
        # Compute gradient for this shift
        grad = np.zeros_like(h)
        for j in range(len(h)):
            k = j + max_shift
            if 0 <= k < len(h):
                grad[j] += (1 - 2 * h[k])
        
        # Update with gradient
        h_new = h - learning_rate * grad
        
        # Project to feasible region [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Evaluate current solution
        c5 = evaluate(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_iter = i
            learning_rate = max(0.001, learning_rate)  # decrease learning rate more slowly when improving
        else:
            if i - best_iter > patience:
                learning_rate *= 0.9  # Reduce learning rate when no improvement
                best_iter = i
    
    return (best_h.tolist(), float(best_c5), n_points)
