import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient for the maximum overlap at lag k_max
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        for i in range(n):
            # Contribution to correlation at lag k_max
            # h[i] * (1 - h[(i + k_max) % n])
            # Gradient w.r.t. h[i] is (1 - h[(i + k_max) % n])
            grad[i] += 1 - h_vec[(i + k_max) % n]
            # Contribution to correlation at lag -k_max (backward)
            # h[(i - k_max) % n] * (1 - h[i])
            # Gradient w.r.t. h[i] is -h[(i - k_max) % n]
            grad[i] -= h_vec[(i - k_max) % n]
        return grad
    
    def line_search(h_vec, grad, max_alpha=1.0):
        best_h = h_vec.copy()
        best_C5 = compute_C5(h_vec)
        best_alpha = 0.0
        for alpha in [0.1, 0.25, 0.5, 1.0]:
            if alpha > max_alpha:
                continue
            candidate_h = h_vec - alpha * grad
            candidate_h = project_feasible(candidate_h)
            new_C5 = compute_C5(candidate_h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = candidate_h.copy()
                best_alpha = alpha
        return best_h, best_C5, best_alpha
    
    h_current = h.copy()
    C5_current = compute_C5(h_current)
    best_h = h_current.copy()
    best_C5 = C5_current
    alpha = 0.1
    stagnation_count = 0
    
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all correlations to find maximum
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Compute gradient for this maximum lag
        grad = compute_gradient(h_current, max_lag)
        
        # Line search for optimal step size
        h_new, C5_new, alpha_used = line_search(h_current, grad, alpha)
        
        # If no improvement, try different alpha
        if C5_new >= C5_current:
            alpha *= 0.5
            alpha = max(alpha, 0.001)
            # Try again with smaller alpha
            h_new, C5_new, _ = line_search(h_current, grad, alpha)
            if C5_new >= C5_current:
                # If still no improvement, perturb
                h_new = h_current + np.random.normal(0, 0.01, n_points)
                h_new = project_feasible(h_new)
                C5_new = compute_C5(h_new)
                alpha = 0.1
                stagnation_count += 1
            else:
                stagnation_count = 0
        else:
            stagnation_count = 0
            alpha = min(alpha * 1.2, 1.0)
            
        h_current = h_new
        C5_current = C5_new
        
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h_current.copy()
            
        if stagnation_count > 50:
            # Add random perturbation to escape local minima
            h_current += np.random.normal(0, 0.02, n_points)
            h_current = project_feasible(h_current)
            best_C5 = compute_C5(h_current)
            best_h = h_current.copy()
            stagnation_count = 0
            alpha = 0.1
            
    # Final projection to ensure feasibility
    best_h = project_feasible(best_h)
    best_C5 = compute_C5(best_h)
    
    return list(best_h), best_C5, n_points
