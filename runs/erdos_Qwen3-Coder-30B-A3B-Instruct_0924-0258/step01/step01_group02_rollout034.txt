import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # Default starting vector
        n_points = 45
        h0 = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial h satisfies constraints
    h0 = h0 * target_sum / np.sum(h0)
    
    # GA parameters
    pop_size = 75
    max_generations = 1000
    mutation_rate = 0.05
    elite_size = pop_size // 4
    
    def compute_c5(h_vec):
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    start_time = time.time()
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s:
        generation += 1
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = compute_c5(h)
            fitnesses.append(-c5)  # Negative because we want to maximize fitness (minimize c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)[::-1]
        population = [population[i] for i in sorted_indices]
        fitnesses.sort(reverse=True)
        
        # Update best solution
        current_best_c5 = -fitnesses[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Terminate if no improvement for too long
        if no_improvement_count > 1000:
            break
            
        # Selection: keep top 50%
        selected = population[:pop_size//2]
        
        # Crossover and mutation
        new_population = selected[:elite_size]  # Elite preservation
        
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_select(selected, fitnesses)
            parent2 = tournament_select(selected, fitnesses)
            
            # Crossover
            child1, child2 = uniform_crossover(parent1, parent2)
            
            # Mutation
            mutate(child1, mutation_rate)
            mutate(child2, mutation_rate)
            
            # Repair
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
        
    return (best_h.tolist(), best_c5, n_points)

def tournament_select(population, fitnesses, k=3):
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmax([fitnesses[i] for i in idxs])]
    return population[best_idx]

def uniform_crossover(parent1, parent2):
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, rate):
    mask = np.random.rand(len(h)) < rate
    noise = np.random.normal(0, 0.05, len(h))
    h += noise * mask
