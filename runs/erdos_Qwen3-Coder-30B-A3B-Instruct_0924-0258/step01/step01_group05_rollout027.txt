import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def generate_latin_hypercube(n, k):
        samples = np.random.random((n, k))
        for i in range(k):
            samples[:, i] = np.sort(samples[:, i])
            samples[:, i] += np.random.random(n) * (1.0/n)
        return samples
    
    def local_optimize(h_vec, max_iter=100):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        no_improvement_count = 0
        step_size = 0.1
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Try coordinate descent
            for i in range(n_points):
                if time.time() - start_time > budget_s - 0.1:
                    break
                original_h = h_current[i]
                best_h_i = original_h
                best_C5_i = best_C5
                
                # Try moving up and down
                for delta in [step_size, -step_size]:
                    h_current[i] = original_h + delta
                    h_current = project_feasible(h_current)
                    new_C5 = compute_C5(h_current)
                    if new_C5 < best_C5_i:
                        best_C5_i = new_C5
                        best_h_i = h_current[i]
                
                h_current[i] = best_h_i
                h_current = project_feasible(h_current)
                if best_C5_i < best_C5:
                    best_C5 = best_C5_i
                    best_h = h_current.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            
            # Pair swap optimization
            for _ in range(10):
                if time.time() - start_time > budget_s - 0.1:
                    break
                i, j = np.random.choice(n_points, 2, replace=False)
                h_save = h_current.copy()
                h_current[i], h_current[j] = h_current[j], h_current[i]
                h_current = project_feasible(h_current)
                new_C5 = compute_C5(h_current)
                if new_C5 < best_C5:
                    best_C5 = new_C5
                    best_h = h_current.copy()
                    no_improvement_count = 0
                else:
                    h_current = h_save
            
            if no_improvement_count > 20:
                step_size = min(0.5, step_size * 1.2)
                no_improvement_count = 0
            elif no_improvement_count > 5:
                no_improvement_count = 0
            
            # Restart if stuck
            if iteration > 100 and best_C5 > 0.383:
                break
                
        return best_h, best_C5
    
    # Generate diverse initial candidates
    candidates = []
    n_candidates = 5
    if n_points >= 10:
        for _ in range(n_candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Latin hypercube sampling
            lhc = generate_latin_hypercube(1, n_points)[0]
            candidate = lhc * 2  # scale to [0,2]
            candidate = project_feasible(candidate)
            candidates.append(candidate)
    
    # Start with the given h or initial construction if available
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Add the current solution to candidates
    if len(candidates) == 0 or not np.allclose(candidates[0], h):
        candidates.append(h_best)
    
    # Local optimization loop
    optimization_round = 0
    while time.time() - start_time < budget_s - 0.1:
        optimization_round += 1
        for i in range(len(candidates)):
            if time.time() - start_time > budget_s - 0.1:
                break
            candidate = candidates[i]
            improved_h, improved_C5 = local_optimize(candidate, max_iter=20)
            if improved_C5 < C5_best:
                C5_best = improved_C5
                h_best = improved_h.copy()
        
        # Occasionally generate a new random candidate
        if optimization_round % 5 == 0 and len(candidates) < 10:
            if time.time() - start_time > budget_s - 0.1:
                break
            lhc = generate_latin_hypercube(1, n_points)[0]
            new_candidate = lhc * 2
            new_candidate = project_feasible(new_candidate)
            candidates.append(new_candidate)
    
    # Final projection
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
