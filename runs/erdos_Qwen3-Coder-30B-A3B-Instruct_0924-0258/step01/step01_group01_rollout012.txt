import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 10
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
    
    h = project_to_constraints(h, dx)
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 100
    tol = 1e-6
    
    # Iteration counter
    iter_num = 0
    last_improvement = 0
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # ADMM iterations
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            h_old = h.copy()
            
            # Update h: minimize with respect to current z and multipliers
            # We approximate the max_k term by computing all correlations and taking the max
            # This uses a potentially approximate approach for the ADMM subproblem
            
            # Simple gradient ascent attempt on objective
            # But since objective is non-smooth, better to make perturbations
            
            # Generate candidate perturbations
            candidates = []
            n_candidates = 8
            for _ in range(n_candidates):
                h_new = h + np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = project_to_constraints(h_new, dx)
                candidates.append(h_new)
            
            # Evaluate candidates
            c5_vals = [evaluate(h_cand, dx) for h_cand in candidates]
            best_idx = np.argmin(c5_vals)
            
            h = candidates[best_idx]
            c5 = c5_vals[best_idx]
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                last_improvement = iter_num
                
            iter_num += 1
            
        # Resolution refinement
        if n_points < 64 and iter_num - last_improvement > 20:
            if time.time() - start_time > budget_s * 0.9:
                break
            # Interpolate to double resolution
            n_points *= 2
            dx = 2.0 / n_points
            h_long = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, n_points//2), h)
            h = h_long
            h = project_to_constraints(h, dx)
            last_improvement = iter_num
            
        if time.time() - start_time > budget_s * 0.95:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
