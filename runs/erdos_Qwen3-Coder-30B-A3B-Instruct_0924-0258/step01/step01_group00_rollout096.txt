import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(45) / 2
    
    n_points = len(h_start)
    dx = 2.0 / n_points
    h = h_start.copy()
    h = project_to_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Greedy adjustment parameters
    adjustment_fraction = 0.01
    perturbation_strength = 0.01
    max_no_improvement = 50
    no_improvement_count = 0
    target = 0.38
    
    # Main optimization loop
    while time() - start_time < budget_s * 0.95:
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Add some noise to escape local minimum
            h = project_to_feasible(h + np.random.normal(0, perturbation_strength, n_points), n_points)
            no_improvement_count = 0
            continue
            
        # Compute correlation to find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        shift = max_idx - (len(h) - 1)
        
        # Find overlapping intervals
        # The overlap is h(x)(1 - h(x + k)) for some shift k
        # Find all indices where this product is large
        h_vals = h
        # Compute overlap at each point
        overlaps = np.zeros(len(h))
        # Fast computation of overlaps by convolution
        # We will adjust values that contribute most to correlation
        
        if shift > 0:
            # h(x) * (1 - h(x + shift)) 
            for i in range(len(h) - shift):
                overlaps[i] += h_vals[i] * (1 - h_vals[i + shift])
        else:
            # shift is negative, handle accordingly
            for i in range(-shift, len(h)):
                overlaps[i] += h_vals[i] * (1 - h_vals[i - shift])
                
        # Find top contributors
        top_indices = np.argsort(overlaps)[-max(1, len(h) // 10):]  # Top 10%
        
        # Reduce h values in these top contributors
        adjusted = False
        delta = adjustment_fraction * (c5 - target) / len(h)
        for i in top_indices:
            if h[i] > delta:
                h[i] -= delta
                adjusted = True
            else:
                h[i] = 0.0
                
        h = project_to_feasible(h, n_points)
        if not adjusted:
            h = project_to_feasible(h + np.random.normal(0, perturbation_strength, n_points), n_points)

    return (list(best_h), best_c5, n_points)
