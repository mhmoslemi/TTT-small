import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=10.0):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def objective_with_smooth_max(h, n_points, dx):
    """Objective using smooth max to approximate the maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Use smooth max to penalize large overlaps
    max_val = smooth_max(corr, alpha=50.0)
    # Add regularization term to keep values bounded
    reg = 0.01 * np.sum((h - 0.5)**2)  # favor balanced distribution
    return max_val * dx + reg

def gradient_of_objective(h, n_points, dx):
    """Compute numerical gradient of the smooth objective"""
    eps = 1e-6
    grad = np.zeros_like(h)
    base_obj = objective_with_smooth_max(h, n_points, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        obj_plus = objective_with_smooth_max(h_plus, n_points, dx)
        grad[i] = (obj_plus - base_obj) / eps
    return grad

def optimize_resolution(h_start, n_points, dx, time_limit):
    """Optimize a resolution level with time constraint"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def obj(h):
        return objective_with_smooth_max(h, n_points, dx)
    
    def grad(h):
        return gradient_of_objective(h, n_points, dx)
    
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try several restarts with perturbed initial values
    for _ in range(3):
        if time() - start_time > time_limit:
            break
            
        h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.03, n_points), n_points)
        
        try:
            res = minimize(obj, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons, jac=grad, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(30) / 2  # start with uniform distribution
    
    # Resolution levels to try
    resolutions = [15, 20, 25, 30, 35, 40, 45, 50, 55, 60, 65, 70, 75]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time limit allocation per resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 40)
        
        # Optimize at this resolution with several restarts
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        for _ in range(4):  # More restarts
            h_restart = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            h_opt, c5_opt = optimize_resolution(h_restart, n_points, dx, time_per_res / 4)
            if c5_opt < best_c5_level:
                best_c5_level = c5_opt
                best_h_level = h_opt.copy()
        
        # If improvement, store
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
            
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
