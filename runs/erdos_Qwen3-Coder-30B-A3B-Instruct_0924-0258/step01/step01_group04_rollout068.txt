import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Multi-resolution optimization
    resolutions = [5, 10, 20, 40, n_points]
    current_h = None
    best_C5 = float('inf')
    best_h = None
    
    # Start from coarse resolution
    for res_idx, res in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
        # Initialize or upscale
        if current_h is None:
            current_h = np.ones(res) * 0.5
        else:
            # From previous resolution, upscale
            current_h = np.interp(np.linspace(0, 1, res), np.linspace(0, 1, len(current_h)), current_h)
        
        # Project to feasible range
        current_h = project_feasible(current_h)
        
        # Optimization iteration
        for iteration in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            current_C5 = compute_overlap(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
                
            # Approximate gradient with finite differences
            gradient = np.zeros_like(current_h)
            eps = 1e-4
            for i in range(len(current_h)):
                h_plus = current_h.copy()
                h_plus[i] += eps
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                h_minus = current_h.copy()
                h_minus[i] -= eps
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                gradient[i] = (C_plus - C_minus) / (2 * eps)
            
            # Adam-style update
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            m = np.zeros_like(current_h)
            v = np.zeros_like(current_h)
            
            # Update moving averages
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * (gradient ** 2)
            
            # Bias correction
            m_hat = m / (1 - beta1 ** (iteration + 1))
            v_hat = v / (1 - beta2 ** (iteration + 1))
            
            # Update step
            lr = 0.01
            current_h = current_h - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            
            # Projection
            current_h = project_feasible(current_h)
            
    # Final evaluation
    final_C5 = compute_overlap(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        
    return (best_h.tolist(), best_C5, n_points)
