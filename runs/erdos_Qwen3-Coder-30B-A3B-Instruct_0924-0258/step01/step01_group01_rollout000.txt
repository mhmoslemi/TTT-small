import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points_list = [10, 20, 40, 70]
    max_iter_per_res = 2000
    n_particles = 100
    inertia = 0.7
    inertia_decay = 0.999
    max_vel = 0.05
    step_size = 0.01
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.uniform(0, 1, 10)
    h = project_to_constraints(h, 2.0/len(h))
    
    best_h = h.copy()
    best_c5 = evaluate(h, 2.0/len(h))
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = project_to_constraints(h, dx)
        
        # Particle swarm optimization
        particles = np.array([h.copy() for _ in range(n_particles)])
        velocities = np.random.uniform(-max_vel, max_vel, (n_particles, n_points))
        particle_best = [h.copy() for h in particles]
        particle_best_c5 = [evaluate(h, dx) for h in particles]
        global_best_idx = np.argmin(particle_best_c5)
        global_best_h = particle_best[global_best_idx].copy()
        global_best_c5 = particle_best_c5[global_best_idx]
        
        for iter_count in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
            
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                pbest = particle_best[i]
                gbest = global_best_h
                
                velocities[i] = (inertia * velocities[i] + 
                               0.5 * r1 * (pbest - particles[i]) + 
                               0.5 * r2 * (gbest - particles[i]))
                
                # Clip velocity
                velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_to_constraints(particles[i], dx)
                
                # Evaluate
                c5 = evaluate(particles[i], dx)
                if c5 < particle_best_c5[i]:
                    particle_best[i] = particles[i].copy()
                    particle_best_c5[i] = c5
                
                if c5 < global_best_c5:
                    global_best_h = particles[i].copy()
                    global_best_c5 = c5
                    
            inertia *= inertia_decay
            if global_best_c5 < best_c5:
                best_h = global_best_h.copy()
                best_c5 = global_best_c5
                
        h = global_best_h.copy()
    
    return best_h.tolist(), best_c5, len(best_h)
