import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to stay within [0,1] and sum to target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient for the maximum overlap term."""
    n = len(h)
    grad = np.zeros(n)
    # For each position compute contribution to gradient
    for i in range(n):
        # The shift k_max gives the maximum overlap,
        # gradient contribution from h[i] is (1 - 2*h[(i-k_max)%n])
        j = (i - k_max) % n
        grad[i] = 1 - 2 * h[j]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h = project_to_feasible(h, n_points / 2)
    
    # Set dx
    dx = 2.0 / n_points
    
    # Optimization parameters
    learning_rate = 0.01
    patience = 50  # For early stopping
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s:
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        # Find maximum overlap
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = compute_gradient(h, dx, k_max)
        
        # Update h
        h = h - learning_rate * grad
        
        # Project back to feasible set
        h = project_to_feasible(h, n_points / 2)
        
        # Evaluate
        c5 = evaluate(h, dx)
        
        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if patience_counter > 100:
            learning_rate *= 0.9
            patience_counter = 0
            
        # Stochastic perturbation to escape local minima
        if patience_counter > patience:
            perturbation = np.random.uniform(-1e-3, 1e-3, n_points)
            h = h + perturbation
            h = project_to_feasible(h, n_points / 2)
            patience_counter = 0
            
        iteration += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
