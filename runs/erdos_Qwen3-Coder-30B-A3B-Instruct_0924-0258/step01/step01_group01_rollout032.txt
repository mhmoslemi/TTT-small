import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # This requires computing correlation of h with (1-h) and taking maximum
    corr = correlate(h, 1-h, mode='full')
    # The full correlation length is 2*n-1
    # Each entry is scaled by dx
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    # So sum(h) = 1/dx = n/2
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10.0):
    # Smooth approximation to max using log-sum-exp
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

def smooth_max_derivative(x, beta=10.0):
    # Derivative of smooth max
    exp_x = np.exp(beta * x)
    return exp_x / np.sum(exp_x)

def compute_correlation_gradients(h, dx):
    # Compute gradient w.r.t. h for smooth max of correlations
    n = len(h)
    # Compute full correlation (n-1 negative lags, 0 lag, n-1 positive lags)
    corr = correlate(h, 1-h, mode='full') * dx
    # Apply smooth max to get differentiable objective
    C5_smooth = smooth_max(corr, beta=10.0)
    # Derivative of smooth max with respect to corr
    dcorr = smooth_max_derivative(corr, beta=10.0)
    
    # Compute gradient using chain rule with correlation
    # This is a complex operation requiring reverse mode
    # We use numerical approximation for simplicity
    grad = np.zeros_like(h)
    eps = 1e-6
    for i in range(n):
        h_pert = h.copy()
        h_pert[i] += eps
        h_pert = project_to_constraints(h_pert, dx)
        corr_pert = correlate(h_pert, 1-h_pert, mode='full') * dx
        C5_pert = smooth_max(corr_pert, beta=10.0)
        grad[i] = (C5_pert - C5_smooth) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with resolution
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, len(initial_h)), initial_h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5

    # Optimization parameters
    lr = 1e-3
    max_iter = int(1e5)
    iter_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        iter_count += 1
        # Compute gradient
        try:
            grad = compute_correlation_gradients(h, dx)
            # Update step
            h_new = h - lr * grad
            h_new = project_to_constraints(h_new, dx)
            # Evaluate
            c5_new = evaluate(h_new, dx)
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
            # Adaptive learning rate
            if iter_count % 500 == 0:
                lr *= 0.5
        except Exception as e:
            # Fallback to simple SA if optimization fails
            if iter_count % 50 == 0:
                T = 0.1 * np.exp(-iter_count / 1000)
                h_new = h + np.random.normal(0, T, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = project_to_constraints(h_new, dx)
                c5_new = evaluate(h_new, dx)
                if c5_new < last_c5:
                    h = h_new
                    last_c5 = c5_new
                    if last_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = last_c5
    
    # Final projection to make sure constraint is exact
    final_h = project_to_constraints(best_h, dx)
    final_c5 = evaluate(final_h, dx)
    
    return final_h.tolist(), final_c5, len(final_h)
