import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with step function
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def perturb_solution(h_vec, sigma=0.1):
        # Create a copy to perturb
        h_new = h_vec.copy()
        
        # Gaussian noise to high priority intervals
        # Priority based on where overlaps are largest
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_vec) - 1)
        
        # Perturb the elements that align with the strongest overlap
        if max_lag >= 0:  # Positive lag case
            h_new[max_lag:] += np.random.normal(0, sigma, len(h_new[max_lag:]))
        else:  # Negative lag case
            h_new[:max_lag] += np.random.normal(0, sigma, len(h_new[:max_lag]))
            
        # Ensure some randomness still
        indices = np.random.choice(len(h_vec), size=max(1, len(h_vec)//10), replace=False)
        h_new[indices] += np.random.normal(0, sigma, len(indices))
        
        # Keep it feasible
        h_new = project_feasible(h_new)
        return h_new

    # Initialize best solution
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Simulated Annealing parameters
    T = 0.1  # Initial temperature
    T_min = 1e-5  # Minimum temperature
    cooling_rate = np.log(10) / budget_s  # Cooling rate
    sigma = 0.1  # Perturbation scale
    acceptance_count = 0
    total_iterations = 0
    no_improvement_cycles = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 0.1:
        total_iterations += 1
        
        # Generate new solution
        h_new = perturb_solution(h_best, sigma)
        C5_new = compute_C5(h_new)
        
        # Accept or reject based on simulated annealing
        delta = C5_new - C5_best
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_best = h_new
            C5_best = C5_new
            acceptance_count += 1
            no_improvement_cycles = 0
        else:
            no_improvement_cycles += 1
            
        # Adjust temperature
        T = max(T_min, T * np.exp(-cooling_rate))
        
        # Modify perturbation scale based on acceptance rate
        if total_iterations % 10 == 0:
            acceptance_rate = acceptance_count / 10.0
            if acceptance_rate < 0.1 and sigma < 0.5:
                sigma *= 1.2  # Increase perturbation
            elif acceptance_rate > 0.4 and sigma > 0.01:
                sigma *= 0.8  # Decrease perturbation
            acceptance_count = 0
            
        # Early stopping if no improvement for too long
        if no_improvement_cycles > max_no_improvement:
            break
    
    # Final projection to ensure feasibility
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
