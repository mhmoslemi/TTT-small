import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available (n=45 samples)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Initialize population
    pop_size = 50
    population = []
    for _ in range(pop_size):
        h_temp = np.random.random(n_points)
        h_temp = project_feasible(h_temp)
        population.append(h_temp.copy())
    
    # Fitness evaluation for population
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            fit = compute_C5(h)
            fitness.append(fit)
        return np.array(fitness)
    
    # Genetic algorithm
    max_generations = 1000
    tournament_size = 10
    elitism_count = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    no_improvement_count = 0
    
    best_solution = None
    best_fitness = float('inf')
    
    # Evaluate initial population
    fitness = evaluate_population(population)
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_fitness:
        best_fitness = fitness[best_idx]
        best_solution = population[best_idx].copy()
    
    generation = 0
    while generation < max_generations and (time.time() - start_time) < budget_s - 0.1:
        # Selection: Tournament
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elitism_count]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            selected_indices = np.random.choice(pop_size, tournament_size, replace=False)
            selected_fitness = [fitness[i] for i in selected_indices]
            parent1_idx = selected_indices[np.argmin(selected_fitness)]
            
            selected_indices = np.random.choice(pop_size, tournament_size, replace=False)
            selected_fitness = [fitness[i] for i in selected_indices]
            parent2_idx = selected_indices[np.argmin(selected_fitness)]
            
            parent1 = population[parent1_idx].copy()
            parent2 = population[parent2_idx].copy()
            
            # Crossover
            if np.random.random() < crossover_rate:
                # Uniform crossover
                mask = np.random.random(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                # Two-point crossover
                point1, point2 = sorted(np.random.choice(n_points, 2, replace=False))
                child1 = np.concatenate([parent1[:point1], parent2[point1:point2], parent1[point2:]])
                child2 = np.concatenate([parent2[:point1], parent1[point1:point2], parent2[point2:]])
            
            # Mutation
            if np.random.random() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child1 += noise
                child1 = project_feasible(child1)
            
            if np.random.random() < mutation_rate:
                # Reset mutation
                reset_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                child2[reset_indices] = np.random.choice([0, 1], size=len(reset_indices))
                child2 = project_feasible(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        # Keep population size consistent
        population = new_population[:pop_size]
        
        # Evaluate new population
        fitness = evaluate_population(population)
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_fitness:
            best_fitness = fitness[best_idx]
            best_solution = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            # Perturb best solution
            best_solution += np.random.normal(0, 0.01, n_points)
            best_solution = project_feasible(best_solution)
            no_improvement_count = 0
            
        generation += 1
    
    # Final check with deterministic optimization step
    h_best = project_feasible(best_solution)
    C5_best = compute_C5(h_best)
    
    # Try local optimization near best solution
    for _ in range(10):
        if (time.time() - start_time) > budget_s - 0.1:
            break
        # Small random perturbations
        h_perturbed = h_best + np.random.normal(0, 0.005, n_points)
        h_perturbed = project_feasible(h_perturbed)
        C5_perturbed = compute_C5(h_perturbed)
        if C5_perturbed < C5_best:
            C5_best = C5_perturbed
            h_best = h_perturbed
    
    return list(h_best), C5_best, n_points
