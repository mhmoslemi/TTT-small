import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_optimize(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define variables
    h = cp.Variable(n_points)
    
    # Objective: minimize max correlation
    max_corr = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
    
    # Create convolved terms (simplified approach)
    corr_terms = []
    for k in range(2 * n_points - 1):
        # For small resolution, approximate max via soft constraints
        if k < n_points:
            corr_val = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
        else:
            corr_val = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
        corr_terms.append(corr_val)
    
    for i, term in enumerate(corr_terms):
        constraints.append(term <= max_corr)
    
    prob = cp.Problem(cp.Minimize(max_corr), constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=100)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            h_sol = project_to_feasible(h_sol, n_points)
            c5 = evaluate_erdos_solution(h_sol, dx)
            if c5 < best_c5:
                best_h = h_sol
                best_c5 = c5
    except:
        pass
    
    return best_h, best_c5

def genetic_mutate(h, m_rate=0.1):
    h_mut = h.copy()
    for i in range(len(h)):
        if np.random.random() < m_rate:
            h_mut[i] += np.random.normal(0, 0.05)
    return h_mut

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Start with initial_h_values or fallback
    if 'initial_h_values' in globals() and len(initial_h_values) > 10:
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(30) * 0.5
    
    # Always start with 30
    n_points = 30
    dx = 2.0 / n_points
    h_current = project_to_feasible(h_start, n_points)
    c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = c5
    best_n_points = n_points
    
    # Multi-resolution search
    resolutions = [15, 30, 45, 60]
    
    for res_idx, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
        
        dx = 2.0 / n_points
        if len(h_current) != n_points:
            h_current = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, len(h_current)), 
                                  h_current)
            h_current = project_to_feasible(h_current, n_points)
        
        # Genetic algorithm part
        population_size = 20
        population = [project_to_feasible(h_current + np.random.normal(0, 0.03, n_points), n_points)
                      for _ in range(population_size)]
        
        for gen in range(10):
            if time() - start_time > budget_s * 0.98:
                break
            
            # Evaluate fitness
            fitness = []
            for h in population:
                c5_val = evaluate_erdos_solution(h, dx)
                fitness.append(c5_val)
            
            # Sort by fitness
            sorted_indices = np.argsort(fitness)
            best_in_gen = population[sorted_indices[0]]
            c5_in_gen = fitness[sorted_indices[0]]
            
            if c5_in_gen < best_c5:
                best_h = best_in_gen.copy()
                best_c5 = c5_in_gen
                best_n_points = n_points
            
            # Select top 50%
            top_half = [population[i] for i in sorted_indices[:population_size//2]]
            
            # Generate new population
            new_population = top_half[:]
            while len(new_population) < population_size:
                # Crossover
                a, b = np.random.choice(top_half, size=2, replace=False)
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([a[:crossover_point], b[crossover_point:]])
                child = project_to_feasible(genetic_mutate(child), n_points)
                new_population.append(child)
            
            population = new_population
            
            # Occasionally use convex optimization
            if gen % 3 == 0:
                h_sample = population[0]
                h_convex, c5_convex = convex_optimize(h_sample, n_points, dx, budget_s / 10)
                if c5_convex < best_c5:
                    best_h = h_convex.copy()
                    best_c5 = c5_convex
                    best_n_points = n_points
        
        # Next resolution
        h_current = best_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
