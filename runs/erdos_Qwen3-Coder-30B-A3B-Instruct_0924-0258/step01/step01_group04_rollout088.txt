import numpy as np
from scipy import optimize
import time
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Genetic Algorithm parameters
    pop_size = 50
    num_generations = 1000
    elite_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 3
    
    # Initialize population
    def initialize_population():
        population = []
        for _ in range(pop_size):
            h_init = np.random.uniform(0.4, 0.6, n_points)
            h_init = project_h(h_init)
            population.append(h_init)
        return population
    
    # Tournament selection
    def tournament_selection(population, fitnesses):
        selected_idx = random.randint(0, len(population) - 1)
        for _ in range(tournament_size - 1):
            candidate_idx = random.randint(0, len(population) - 1)
            if fitnesses[candidate_idx] < fitnesses[selected_idx]:
                selected_idx = candidate_idx
        return population[selected_idx]
    
    # Crossover operation
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    # Mutation operation
    def mutate(h_vec):
        mutated = h_vec.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] = random.uniform(0, 1)
        return mutated
    
    # Main GA loop with local refinement
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    population = initialize_population()
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitnesses = [compute_C5(ind) for ind in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_C5:
            best_C5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = project_h(mutate(child1))
            child2 = project_h(mutate(child2))
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:pop_size]
        
        # Every 50 generations, perform local refinement on top individuals
        if gen % 50 == 0:
            # Rename h to restore adjustability
            for i in range(min(5, len(population))):
                ind = population[i]
                def objective(x):
                    h_test = project_h(x)
                    return compute_C5(h_test)
                
                try:
                    # Local optimization
                    result = optimize.minimize(objective, ind, method='Nelder-Mead',
                                             options={'maxiter': 100, 'disp': False})
                    
                    if result.success:
                        refined_h = project_h(result.x)
                        refined_C5 = compute_C5(refined_h)
                        if refined_C5 < best_C5:
                            best_C5 = refined_C5
                            best_h = refined_h.copy()
                except:
                    pass
    
    return (best_h.tolist(), best_C5, n_points)
