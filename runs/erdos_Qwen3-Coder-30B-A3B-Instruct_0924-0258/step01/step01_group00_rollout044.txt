import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Refined search with different approaches
    resolutions = [n_points, 40, 50, 60]
    
    for res_idx, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if needed
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / (len(resolutions) - res_idx), 60)
        
        # Try several optimization strategies
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        # Strategy 1: Local optimization with projected gradient
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            try:
                h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def grad(h):
                    eps = 1e-6
                    g = np.zeros_like(h)
                    base_val = obj(h)
                    for i in range(len(h)):
                        h_p = h.copy()
                        h_p[i] += eps
                        h_p = project_to_feasible(h_p, n_points)
                        g[i] = (obj(h_p) - base_val) / eps
                    return g
                
                cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Use SLSQP
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, jac=grad, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_res = project_to_feasible(res.x, n_points)
                    c5 = evaluate_erdos_solution(h_res, dx)
                    if c5 < best_c5_level:
                        best_c5_level = c5
                        best_h_level = h_res.copy()
            except:
                pass
        
        # Strategy 2: Genetic algorithm on top of current solution
        if res_idx > 0 and n_points > 30:
            population_size = min(20, max(5, n_points // 3))
            pop = [project_to_feasible(best_h_level + np.random.normal(0, 0.03, n_points), n_points) 
                   for _ in range(population_size)]
            for gen in range(10):
                if time() - start_time > budget_s * 0.95:
                    break
                scores = [evaluate_erdos_solution(h, dx) for h in pop]
                best_idx = np.argmin(scores)
                if scores[best_idx] < best_c5_level:
                    best_c5_level = scores[best_idx]
                    best_h_level = pop[best_idx].copy()
                
                # Selection
                sorted_indices = np.argsort(scores)
                elite = [pop[i] for i in sorted_indices[:population_size//3]]
                
                # Crossover and mutation
                new_pop = elite.copy()
                while len(new_pop) < population_size:
                    parent1, parent2 = np.random.choice(sorted_indices[:population_size//2], 2, replace=False)
                    parent1, parent2 = pop[parent1], pop[parent2]
                    crossover_point = np.random.randint(0, n_points)
                    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                    mutation_strength = 0.01 + 0.03 * (1 - gen/10)
                    child += np.random.normal(0, mutation_strength, n_points)
                    child = project_to_feasible(child, n_points)
                    new_pop.append(child)
                
                pop = new_pop[:population_size]
        
        # If improvement at this resolution, update global best
        if best_c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
        
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
