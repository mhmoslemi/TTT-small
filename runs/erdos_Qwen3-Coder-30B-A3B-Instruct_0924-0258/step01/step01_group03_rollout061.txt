import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and bounds [0,1]."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to sum=target_sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Ensure sum constraint exactly
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Adjust smallest elements to make up the difference
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff == 0:
                break
            j = idx[i]
            if h[j] < 1 and diff > 0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
            elif h[j] > 0 and diff < 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    return h

def nelder_mead_optimize(initial_h, target_sum, dx, budget_s):
    """Optimize using Nelder-Mead with projection."""
    n_points = len(initial_h)
    tol = 1e-6
    
    # Set up the simplex
    simplex_size = n_points + 2
    simplex = [initial_h.copy()]
    for i in range(simplex_size - 1):
        h = initial_h + np.random.normal(0, 0.01, n_points)
        h = project_to_feasible(h, target_sum)
        simplex.append(h)
    
    best_h = None
    best_c5 = float('inf')
    iter_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Evaluate all simplex points
        values = []
        for h in simplex:
            if np.isnan(np.sum(h)):
                values.append(float('inf'))
            else:
                val = evaluate(h, dx)
                values.append(val)
        
        # Order the simplex
        idx = np.argsort(values)
        simplex = [simplex[i] for i in idx]
        values = [values[i] for i in idx]
        
        # Update best
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
        
        # If converged or best improvement is small, mutate
        if iter_count % 100 == 0:
            # Mutate some points
            for i in range(1, min(5, len(simplex))):
                h = simplex[i] + np.random.uniform(-0.05, 0.05, n_points)
                simplex[i] = project_to_feasible(h, target_sum)
        
        # Nelder-Mead operations
        # Centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        alpha = 1.0
        r = centroid + alpha * (centroid - simplex[-1])
        r = project_to_feasible(r, target_sum)
        r_val = evaluate(r, dx)
        
        if values[0] <= r_val < values[-2]:
            simplex[-1] = r
            continue
        elif r_val < values[0]:
            # Expansion
            beta = 2.0
            e = centroid + beta * (r - centroid)
            e = project_to_feasible(e, target_sum)
            e_val = evaluate(e, dx)
            if e_val < r_val:
                simplex[-1] = e
            else:
                simplex[-1] = r
        else:
            # Contraction
            gamma = 0.5
            if r_val < values[-1]:
                c = centroid + gamma * (r - centroid)
            else:
                c = centroid + gamma * (simplex[-1] - centroid)
            c = project_to_feasible(c, target_sum)
            c_val = evaluate(c, dx)
            if c_val < values[-1]:
                simplex[-1] = c
            else:
                # Reduction
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project_to_feasible(simplex[i], target_sum)
    
    # Final evaluation
    final_values = [evaluate(h, dx) for h in simplex]
    min_idx = np.argmin(final_values)
    if final_values[min_idx] < best_c5:
        best_c5 = final_values[min_idx]
        best_h = simplex[min_idx].copy()
    
    return best_h, best_c5
