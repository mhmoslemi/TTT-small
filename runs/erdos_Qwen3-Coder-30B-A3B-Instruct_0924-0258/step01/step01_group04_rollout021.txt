import numpy as np
from scipy import optimize
import time
try:
    from cvxpy import *
except ImportError:
    # Fallback to a simpler method if CVXPY not available
    pass

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Try SDP approach with limited size
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Simple gradient-based method with adaptive step size
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        if iteration % 50 == 0:
            # Periodic full evaluation
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()
                
        # Compute gradient using finite differences
        gradient = np.zeros_like(h)
        eps = 1e-4
        h_vec = project_h(h)
        
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            gradient[i] = (C_plus - C_minus) / (2 * eps)
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * gradient
        h = h + velocity
        
        # Project to feasible region
        h = project_h(h)
        
        # Adaptive step size
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            learning_rate *= 1.1
        else:
            learning_rate *= 0.9
            
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
