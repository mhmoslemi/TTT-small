import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_analytical_gradient(h_vec):
        # Compute correlations and find the maximum
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max = np.argmax(corr)
        n = len(h_vec)
        
        # Initialize gradient
        grad = np.zeros(n)
        
        # Gradient for k_max shift
        for i in range(n):
            if i + k_max < n:
                grad[i] += (1 - h_vec[i + k_max]) * dx
            if i - k_max >= 0:
                grad[i] -= h_vec[i - k_max] * dx
                
        return grad
    
    # Adam optimizer variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute gradient
        grad = compute_analytical_gradient(h)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        # Update h
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible set
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Periodic perturbation
        if iteration % 500 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h)
            
        # Early termination if no improvement
        if no_improvement_count > max_no_improvement:
            break
            
        iteration += 1
        
        # Anneal learning rate
        if iteration % 100 == 0:
            lr *= 0.995
    
    return (best_h.tolist(), best_c5, n_points)
