import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10.0):
    return np.log(np.sum(np.exp(beta * x))) / beta

def compute_corr_gradient(h, dx):
    n = len(h)
    # Compute correlation of h with (1-h) 
    corr = correlate(h, 1-h, mode='full')
    # We want gradient of max_k corr_k
    # Using smooth max approximation to make gradient smooth
    max_k = np.argmax(corr)
    # Generate gradient of correlation with respect to h
    # We'll compute it numerically for now, but could be optimized
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        corr_plus = correlate(h_plus, 1-h_plus, mode='full')
        corr_minus = correlate(h_minus, 1-h_minus, mode='full')
        grad[i] = (corr_plus[max_k] - corr_minus[max_k]) / (2*eps) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with resolution 50
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize with initial values or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.random.uniform(0, 1, n_points)
    
    h = project_to_constraints(h, dx)
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 1e-3
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient of objective
        try:
            grad = compute_corr_gradient(h, dx)
        except Exception:
            # fallback to finite differences if gradient not available
            eps = 1e-5
            grad = np.zeros_like(h)
            base_c5 = evaluate(h, dx)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_constraints(h_plus, dx)
                c5_plus = evaluate(h_plus, dx)
                grad[i] = (c5_plus - base_c5) / eps
                
        h = h - learning_rate * grad
        h = project_to_constraints(h, dx)
        
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Adaptive learning rate reduction
        if iter_count % 500 == 0 and iter_count > 0:
            learning_rate *= 0.5
            
    return best_h.tolist(), best_c5, len(best_h)
