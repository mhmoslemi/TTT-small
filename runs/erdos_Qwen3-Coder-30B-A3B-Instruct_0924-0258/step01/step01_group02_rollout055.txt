import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback: create a simple initial guess
        n_points = 45
        initial_h = np.full(n_points, 1.0 / n_points)
    
    def compute_c5(h):
        # Compute correlation with 1-h
        corr = signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / len(h))
    
    def project_h(h):
        # Ensure h is in [0,1] and sum to target
        h = np.clip(h, 0, 1)
        target_sum = len(h) / 2.0
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        return h

    def fitness(h):
        return -compute_c5(h)

    # GA parameters
    pop_size = 60
    num_generations = 1000
    mutation_rate = 0.05
    crossover_rate = 0.8
    tournament_size = 3
    elite_size = 10

    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize population
    population = [project_h(initial_h.copy())]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    no_improvement_count = 0

    start_time = time.time()

    for gen in range(num_generations):
        if time.time() - start_time > budget_s:
            break

        # Evaluate fitness for current population
        fitness_vals = [fitness(h) for h in population]
        
        # Track best
        current_best_idx = np.argmax(fitness_vals)
        current_best_h = population[current_best_idx]
        current_best_c5 = -fitness_vals[current_best_idx]

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count > 1000:
            break

        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            winner_idx = tournament[np.argmax([fitness_vals[i] for i in tournament])]
            selected.append(population[winner_idx].copy())

        # Crossover and Mutation
        new_pop = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % pop_size] if i+1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutate
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child1 += noise
                child1 = project_h(child1)
                
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child2 += noise
                child2 = project_h(child2)
                
            new_pop.extend([child1, child2])

        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness_vals)[-elite_size:]
        elite = [population[i] for i in elite_indices]
        population = elite + new_pop[:pop_size - elite_size]

    return (best_h.tolist(), best_c5, n_points)
