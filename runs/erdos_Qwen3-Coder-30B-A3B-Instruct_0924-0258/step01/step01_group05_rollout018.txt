import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def perturb(h_vec, eps=0.1):
        # Add small random noise
        noise = np.random.uniform(-eps, eps, len(h_vec))
        return project_feasible(h_vec + noise)
    
    def grid_search_level(n_level, n_candidates, coarse=False):
        best_h = None
        best_C5 = float('inf')
        candidates = []
        
        # Generate candidates
        for _ in range(n_candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Start with random feasible solution
            h_temp = np.random.rand(n_level)
            h_temp = project_feasible(h_temp)
            
            # Add some structure for diversity
            if not coarse and np.random.random() < 0.3:
                # Add a few peaks
                peaks = np.random.randint(0, n_level, 3)
                for i in peaks:
                    h_temp[i] = min(1.0, h_temp[i] + 0.3)
                h_temp = project_feasible(h_temp)
                
            candidates.append(h_temp)
            
        # Evaluate all candidates
        for h_cand in candidates:
            if time.time() - start_time > budget_s - 0.1:
                break
            C5 = compute_C5(h_cand)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_cand.copy()
                
        return best_h, best_C5
    
    # Multi-level search
    # Coarse levels
    coarse_levels = [5, 10, 15]
    fine_levels = [20, 25, 30]
    
    # Initialize with coarse search
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Try coarse grid search
    for n_level in coarse_levels:
        if time.time() - start_time > budget_s - 0.1:
            break
        h_cand, C5_cand = grid_search_level(n_level, 50, coarse=True)
        if C5_cand < best_C5:
            best_C5 = C5_cand
            best_h = h_cand.copy()
            
    # Refine with finer grids
    for n_level in fine_levels:
        if time.time() - start_time > budget_s - 0.1:
            break
        # Start with best from previous level, refine with perturbations
        h_cand, C5_cand = grid_search_level(n_level, 20, coarse=False)
        if C5_cand < best_C5:
            best_C5 = C5_cand
            best_h = h_cand.copy()
    
    # Final refinement using local search from current best
    if time.time() - start_time < budget_s - 0.1:
        # Perturb current best and try gradient-based descent
        h_current = best_h.copy()
        last_improvement = 0
        iteration = 0
        
        # Simple local refinement: try gradient-style steps
        while iteration < 100 and time.time() - start_time < budget_s - 0.1:
            iteration += 1
            if iteration - last_improvement > 20:
                # If no progress, perturb more aggressively
                h_current = perturb(h_current, 0.05)
                last_improvement = iteration
            else:
                # Try gradient ascent direction (maximize C5)
                # But actually, we solve a minimization, so we want to minimize C5
                # So we minimize by decreasing C5 - which corresponds to 
                # finding the max cut in reciprocal space
                
                # Actually, a different way: first, compute current max correlation
                corr = correlate(h_current, 1 - h_current, mode='full')
                max_lag_idx = np.argmax(corr)
                max_lag = max_lag_idx - (len(h_current) - 1)
                
                # Approximate gradient direction (this is crude but effective)
                # For lag k, the gradient points to reduce values where they influence
                grad = np.zeros_like(h_current)
                n = len(h_current)
                for i in range(n):
                    j = (i + max_lag) % n
                    if max_lag >= 0:
                        grad[i] = -h_current[j]
                    else:
                        grad[i] = -h_current[j]
                
                # Gradient descent step
                step_size = 0.01
                h_trial = h_current - step_size * grad
                h_trial = project_feasible(h_trial)
                new_C5 = compute_C5(h_trial)
                
                if new_C5 < best_C5:
                    h_current = h_trial.copy()
                    best_C5 = new_C5
                    last_improvement = iteration
                else:
                    # Try perturbing to escape
                    h_current = perturb(h_current, 0.02)
                    # After perturbing check again
                    new_C5 = compute_C5(h_current)
                    if new_C5 < best_C5:
                        best_C5 = new_C5
                        last_improvement = iteration
    
    # Final projection to ensure feasibility
    h_final = project_feasible(h_current)
    final_C5 = compute_C5(h_final)
    
    return list(h_final), final_C5, n_points
