import numpy as np
import time
from scipy.signal import correlate

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx):
    # Compute gradient for the maximum correlation term
    n = len(h)
    # Compute all correlations
    corr = correlate(h, 1-h, mode='full')
    # Find critical lags
    max_corr = np.max(corr)
    # Identify indices that achieve this max
    max_indices = np.where(np.abs(corr - max_corr) < 1e-10)[0]
    
    # Take first few critical indices
    critical_indices = max_indices[:min(10, len(max_indices))]
    
    grad = np.zeros(n)
    
    for i in critical_indices:
        # For correlation index i:
        # In full convolution, corresponding to shift (i - (n-1))
        shift = i - (n - 1)
        if shift >= 0:
            # h[j] * (1 - h[j+shift]) contributes to this term
            # So we have partial derivative of h[j] = (1 - h[j+shift]) 
            # and partial derivative of h[j+shift] = -h[j]
            # We only look at positive shifts to make it easier
            for j in range(n - shift):
                grad[j] += (1 - h[j + shift]) * dx
                if j + shift < n:
                    grad[j + shift] -= h[j] * dx
        else:
            # shift is negative, so h[j+shift] term
            shift = -shift
            for j in range(n - shift):
                grad[j] += (1 - h[j + shift]) * dx
                if j + shift < n:
                    grad[j + shift] -= h[j] * dx

    # Normalize gradient
    grad = -grad / (np.max(np.abs(grad)) + 1e-10)
    return grad

    np.random.seed(seed)
    start_time = time.time()

    n_low = 10
    n_high = 70
    n_points = n_low

    else:

        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = float('inf')
    
    # See what we get from initial
    current_c5 = max(correlate(h, 1-h, mode='full')) * dx
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
        
    # AdaGrad-style optimization
    learning_rate = 0.01
    accumulated_grad = np.zeros(n_points)
    
    # Optimization loop
    iter_count = 0
    last_improvement = 0
    
    while iter_count < 10000:
        if time.time() - start_time > budget_s * 0.97:
            break
            
        # Evaluate current
        corr = correlate(h, 1-h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Compute gradient
        grad = compute_gradient(h, dx)
        
        # AdaGrad update
        accumulated_grad += grad * grad
        step_size = learning_rate / (np.sqrt(accumulated_grad) + 1e-8)
        h_new = h - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        
        # Evaluate
        corr_new = correlate(h_new, 1-h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept if better or random
        if new_c5 < current_c5 or np.random.rand() < 0.1:
            h = h_new
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count

        iter_count += 1
        
        # Resolution adaptation
        if n_points < n_high and iter_count - last_improvement > 2000:
            # Interpolate to higher resolution
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_improvement = iter_count
            
    return best_h.tolist(), best_c5, len(best_h)
