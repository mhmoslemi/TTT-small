import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
        # Clip again after scaling
        h = np.clip(h, 0, 1)
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def nelder_mead_optimize(initial_h, dx, budget_s):
    """Optimize using Nelder-Mead with constraint handling."""
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    
    # Initialize simplex
    def create_simplex(start_point, size=1e-3):
        simplex = [start_point]
        for i in range(n_points):
            perturbed = start_point.copy()
            perturbed[i] += np.random.uniform(-size, size)
            perturbed = project_to_feasible(perturbed, target_sum)
            simplex.append(perturbed)
        return simplex
    
    simplex = create_simplex(initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Nelder-Mead parameters
    alpha = 1.0  # reflection
    beta = 2.0   # expansion
    gamma = 0.5  # contraction
    delta = 0.5  # reduction
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Evaluate all points
        values = []
        for point in simplex:
            val = evaluate(point, dx)
            values.append(val)
        
        # Sort by value
        sorted_indices = np.argsort(values)
        simplex = [simplex[i] for i in sorted_indices]
        values = [values[i] for i in sorted_indices]
        
        # Update best
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
        
        # Check for convergence
        if iteration % 100 == 0:
            if np.std(values) < 1e-6:
                # Try random restart
                simplex = create_simplex(best_h if best_h is not None else initial_h)
        
        # Mutation every 50 iterations
        if iteration % 50 == 0 and best_h is not None:
            for i in range(1, len(simplex)):
                if np.random.random() < 0.3:
                    simplex[i] = project_to_feasible(
                        simplex[i] + np.random.normal(0, 0.01, n_points), 
                        target_sum
                    )
        
        # Nelder-Mead operations
        x0 = simplex[0]  # best
        x1 = simplex[1]  # second best
        
        # Centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        xr = (1 + alpha) * centroid - alpha * simplex[-1]
        xr = project_to_feasible(xr, target_sum)
        val_r = evaluate(xr, dx)
        
        if values[0] <= val_r < values[-2]:
            simplex[-1] = xr
            continue
            
        # Expansion
        if val_r < values[0]:
            xe = (1 + beta) * centroid - beta * simplex[-1]
            xe = project_to_feasible(xe, target_sum)
            val_e = evaluate(xe, dx)
            if val_e < val_r:
                simplex[-1] = xe
            else:
                simplex[-1] = xr
            continue
            
        # Contraction
        if val_r >= values[-2]:
            xc = (1 + gamma) * centroid - gamma * simplex[-1]
            xc = project_to_feasible(xc, target_sum)
            val_c = evaluate(xc, dx)
            if val_c < values[-1]:
                simplex[-1] = xc
                continue
                
            # Reduction
            for i in range(1, len(simplex)):
                simplex[i] = simplex[0] + delta * (simplex[i] - simplex[0])
                simplex[i] = project_to_feasible(simplex[i], target_sum)
                
    return best_h, best_c5

    final_c5 = evaluate(best_h, dx)
