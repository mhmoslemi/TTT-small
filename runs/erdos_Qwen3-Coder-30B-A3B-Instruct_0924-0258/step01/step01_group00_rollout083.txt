import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def get_correlations(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def compute_min_correlation(h, dx):
    corr = get_correlations(h, dx)
    return np.min(corr)

def augmented_lagrangian_objective(h, n_points, dx, alpha, lambda_val, mu):
    h = project_to_feasible(h, n_points)
    corr = get_correlations(h, dx)
    min_corr = np.min(corr)
    penalty = mu * max(0, alpha - min_corr)**2
    return -min_corr + penalty

def augmented_lagrangian_gradient(h, n_points, dx, alpha, lambda_val, mu):
    h = project_to_feasible(h, n_points)
    corr = get_correlations(h, dx)
    min_corr = np.min(corr)
    min_idx = np.argmin(corr)
    
    eps = 1e-6
    grad = np.zeros_like(h)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        corr_plus = get_correlations(h_plus, dx)
        min_corr_plus = np.min(corr_plus)
        grad[i] = (min_corr_plus - min_corr) / eps
    
    # Augmented Lagrangian gradient
    lagrangian_grad = -grad
    if min_corr < alpha:
        lagrangian_grad += 2 * mu * (alpha - min_corr) * grad
    
    return lagrangian_grad

def optimize_with_augmented_lagrangian(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_alpha = compute_min_correlation(h_start, dx)
    
    lambda_val = 0.0
    mu = 1.0
    alpha = -1.0
    
    for iteration in range(500):
        if time() - start_time > time_limit:
            break
            
        # Update alpha based on current best
        current_alpha = compute_min_correlation(best_h, dx)
        alpha = max(0.0, current_alpha)
        
        # Objective with current alpha
        def obj(h):
            return augmented_lagrangian_objective(h, n_points, dx, alpha, lambda_val, mu)
            
        def grad(h):
            return augmented_lagrangian_gradient(h, n_points, dx, alpha, lambda_val, mu)
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, 
                          constraints=cons, jac=grad, options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_new = project_to_feasible(res.x, n_points)
                new_alpha = compute_min_correlation(h_new, dx)
                if new_alpha > best_alpha:
                    best_h = h_new.copy()
                    best_alpha = new_alpha
        except:
            pass
        
        # Random perturbation with decreased strength over time
        strength = 0.1 * (1.0 - (iteration / 500.0))
        h_pert = best_h + np.random.normal(0, strength, n_points)
        h_pert = project_to_feasible(h_pert, n_points)
        pert_alpha = compute_min_correlation(h_pert, dx)
        if pert_alpha > best_alpha:
            best_h = h_pert.copy()
            best_alpha = pert_alpha
    
    return best_h, best_alpha

    np.random.seed(seed)

    # Initial starting point
    if 'initial_h_values' in globals():

        h_start = np.zeros(n0)
        # Place a few peaks and valleys
        h_start[5:10] = 0.8
        h_start[15:20] = 0.2
        h_start[25:30] = 0.6
        # Normalize
        h_start = h_start * (n0 / 2) / np.sum(h_start)
    
    resolutions = [30, 45, 60]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Multi-resolution optimization
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time allocation per resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple restarts
        best_h_level = h_start.copy()
        best_alpha_level = compute_min_correlation(h_start, dx)
        
        for _ in range(3):  # 3 restarts
            # Add structured perturbations
            h_pert = h_start + np.random.normal(0, 0.05, n_points)
            h_pert = project_to_feasible(h_pert, n_points)
            
            h_opt, alpha_opt = optimize_with_augmented_lagrangian(
                h_pert, n_points, dx, time_per_res/3)
            
            if alpha_opt > best_alpha_level:
                best_alpha_level = alpha_opt
                best_h_level = h_opt.copy()
        
        # Convert to C5 bound: C5 = 1 - min_overlap
        c5_level = 1.0 - best_alpha_level
        
        if c5_level < best_c5:
            best_h = best_h_level.copy()
            best_c5 = c5_level
            best_n_points = n_points
            
        h_start = best_h_level.copy()
    
    return (list(best_h), best_c5, best_n_points)
