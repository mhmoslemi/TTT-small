import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_analytical_gradient(h_vec, k_max):
        # Analytical gradient for the correlation at shift k_max
        # Gradient[i] = d/dh[i] (h ⋆ (1 - h))[k_max] 
        # = sum_{j: j+k_max=i} (1 - 2 * h[i]) + sum_{j: j=i-k_max} (-h[j])
        # But more directly for the actual shift:
        # For h[i] to contribute to the correlation at k_max:
        # correlation[k_max] = sum_{j} h[j] * (1 - h[j + k_max])
        # Gradient w.r.t. h[i] = 1 - 2*h[i] + (if i-k_max >= 0 and i-k_max < n) then -h[i-k_max] else 0
        # Actually, let's compute it properly using finite difference with better precision.
        # But for optimization we'll do coordinate-wise updates

        # Let's compute exact gradient at k_max, can write this more carefully for
        # simple correlation derivative with a shift
        grad = np.zeros_like(h_vec)
        
        # For correlation at shift k_max:
        # corr[k_max] = sum_{i} h[i] * (1 - h[i + k_max])  
        # if i+k_max >= 0 and i+k_max < n
        # d/dh[i] of corr[k_max] = (1 - h[i+k_max]) - h[i]*0 for the first term 
        # + 0  for the second term (since h[i] is not multiplied by h[i])  
        # But we're computing the derivative with respect to h[i] of INTEGRAL over k
        # kernel k is fixed (ours is h ⋆ (1-h) at shift k_max)
        # corr = \sum_{i} h[i] * (1 - h[i + k_max])
        # grad[i] = d/dh[i] [h[i] * (1 - h[i + k_max])] = 1 - h[i + k_max] - h[i] * 0 = 1 - h[i + k_max]
        # Only non-zero for i+ k_max < n, i >= 0. So we compute first:
        # We can represent the full kernel being used for that shift k_max:
        # All positions j, such that i = j + k_max, which means j = i - k_max
        # For each such i in 0..n-1, we look at the term h[i - k_max] in the kernel
        
        # The full integral of correlation with shift k_max can be seen as:
        # J = sum_{i=0}^{n-1} h[i] * (1 - h[i + k_max]) = sum_{i} h[i] - h[i] * h[i + k_max]
        # Compute the derivative w.r.t h[i]:
        # dJ/dh[i] = 1 - (h[i + k_max] + h[i] * 0) = 1 - h[i + k_max] for i+k_max < n
        # But note there's also dependence when we look at h[i - k_max] for k_max term. 
        # But again: look at specifically i-th element in h, contribution to k_max correlation:
        # The derivative is: 1 - h[i + k_max] 
        # why? because we have h[i] * (1 - h[i+k_max]) -> gain w.r.t. h[i] is 1 - h[i+k_max]
        # The h[i] term is independent in the sense it's not another h[i] -> it is a function of same h vector, but not itself.
        
        # This approach: define correlation[k_max] = sum_{i} h[i] * (1 - h[i + k_max])
        # d/dh[i] corr[k_max] = 1 - h[i + k_max] if i + k_max in [0,n-1], 0 otherwise.
        # There is a subtlety: h is indexed in [0, n-1], so zero index shifted is h[i + k_max] valid if i+k_max < n and i >= 0
        # So grad[i] = 1 - h[i+k_max] if i+k_max < n  
        
        if k_max >= 0 and k_max <= n_points-1:
            for i in range(n_points):
                if i + k_max < n_points:
                    grad[i] = 1 - h_vec[i + k_max]
                    
        return grad

    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    max_iter = 5000
    stagnation_count = 0
    last_best_c5 = best_c5
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Compute analytical gradient for the k_max correlation
        grad = compute_analytical_gradient(h, k_max)
        
        # Gradient update
        h = h - lr * grad
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Anneal learning rate
        lr *= 0.999
        
        # Occasionally add noise
        if iteration % 500 == 0:
            p = 0.05  # Noise to 5% of elements
            noise_indices = np.random.choice(n_points, size=int(p * n_points), replace=False)
            noise = 0.01 * np.random.uniform(-1, 1, len(noise_indices))
            h[noise_indices] += noise
            h = project_h(h)
            
        # Early termination if not improving much
        if stagnation_count > 1000:
            break

        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
