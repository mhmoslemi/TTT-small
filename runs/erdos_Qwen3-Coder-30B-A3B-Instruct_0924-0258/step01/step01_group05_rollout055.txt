import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with fewer points to allow faster optimization
        n_points = 40
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute cross-correlations for all lags in full mode
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(len(parent1)) > 0.5
        child = np.where(mask, parent1, parent2)
        return project_feasible(child)
    
    def mutate(h_vec, rate=0.1):
        # Gaussian mutation and reset mutation
        h_mut = h_vec.copy()
        # Gaussian mutation
        mask = np.random.rand(len(h_vec)) < rate
        h_mut[mask] += np.random.normal(0, 0.05, np.sum(mask))
        # Random reset mutation
        reset_mask = np.random.rand(len(h_vec)) < 0.05
        h_mut[reset_mask] = np.random.choice([0, 1], size=np.sum(reset_mask))
        return project_feasible(h_mut)
    
    def tournament_selection(population, fitnesses, k=10):
        # Tournament selection
        indices = np.random.choice(len(population), size=k, replace=False)
        tournament_fitness = [fitnesses[i] for i in indices]
        winner_idx = indices[np.argmin(tournament_fitness)]
        return population[winner_idx]
    
    # Genetic Algorithm parameters
    pop_size = 50
    n_generations = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_temp = np.random.rand(n_points)
        h_temp = project_feasible(h_temp)
        population.append(h_temp)
    
    # Evaluate initial population
    fitnesses = [compute_C5(h) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_C5 = fitnesses[best_idx]
    
    patience = 0
    patience_limit = 100
    
    # Main GA loop
    for gen in range(n_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for improvement
        if sorted_fitnesses[0] < best_C5:
            best_C5 = sorted_fitnesses[0]
            best_h = sorted_population[0].copy()
            patience = 0
        else:
            patience += 1
            
        # Early stopping if no improvement
        if patience >= patience_limit:
            break
            
        # Create new population
        new_population = [sorted_population[0], sorted_population[1]]  # Elitism
        
        # Generate offspring
        while len(new_population) < pop_size:
            if time.time() - start_time > budget_s - 0.1:
                break
                
            parent1 = tournament_selection(sorted_population, sorted_fitnesses)
            parent2 = tournament_selection(sorted_population, sorted_fitnesses)
            
            if np.random.rand() < crossover_rate:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
                
            child = mutate(child, mutation_rate)
            new_population.append(child)
        
        # Update population
        population = new_population[:pop_size]
        fitnesses = [compute_C5(h) for h in population]
    
    # Final check
    final_fitnesses = [compute_C5(h) for h in population]
    final_best_idx = np.argmin(final_fitnesses)
    if final_fitnesses[final_best_idx] < best_C5:
        best_h = population[final_best_idx].copy()
        best_C5 = final_fitnesses[final_best_idx]
    
    return list(best_h), best_C5, n_points
