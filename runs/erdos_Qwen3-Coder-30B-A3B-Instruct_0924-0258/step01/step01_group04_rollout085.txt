from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Constraint function for bounds
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global search with local refinement
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    # Try to improve the initial solution using local optimization
    if time.time() - start_time < budget_s - 2:
        try:
            result = differential_evolution(
                lambda x: compute_overlap(project_feasible(x)),
                bounds,
                maxiter=100,
                popsize=15,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=seed
            )
            if result.success:
                h_candidate = project_feasible(result.x)
                candidate_C5 = compute_overlap(h_candidate)
                if candidate_C5 < best_C5:
                    best_C5 = candidate_C5
                    best_h = h_candidate.copy()
        except:
            pass

    if time.time() - start_time < budget_s - 2:
        for _ in range(5):

                result = minimize(
                    lambda x: compute_overlap(project_feasible(x)),
                    best_h,
                    method='Nelder-Mead',
                    options={'maxiter': 200, 'disp': False}
                )
                if result.success:
                    h_candidate = project_feasible(result.x)
                    candidate_C5 = compute_overlap(h_candidate)

            elites = population[:5]

                                fitness[i] = candidate_C5

            new_pop = elites.copy()

                # Tournament selection
                i1 = np.random.randint(0, n_pop//2)
                i2 = np.random.randint(0, n_pop//2)
                
                parent1 = population[i1] if fitness[i1] < fitness[i2] else population[i2]

                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

                if np.random.rand() < 0.1:
                    mut_idx = np.random.randint(0, n_points)
                    child[mut_idx] = np.random.rand()

            population = new_pop.copy()
        
        # Final check
        final_fitness = [compute_overlap(ind) for ind in population]

            min_idx = np.argmin(final_fitness)
            best_C5 = final_fitness[min_idx]
            best_h = population[min_idx].copy()
    
    # Verify final solution
    final_C5 = compute_overlap(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
