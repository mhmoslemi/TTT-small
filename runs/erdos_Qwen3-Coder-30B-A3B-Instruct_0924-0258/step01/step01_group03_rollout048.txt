import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Parameters
    max_iter = 10000
    lr = 0.01
    patience = 50
    patience_counter = 0
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute correlation to find best shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        shift = max_idx - (len(h) - 1)
        
        # Compute gradient
        gradient = np.zeros_like(h)
        if shift >= 0:
            # h[i] contributes to overlap with (1-h[i+shift])
            gradient[:-shift] -= 2 * h[shift:]
        else:
            # shift < 0
            gradient[-shift:] -= 2 * h[:shift]
        
        # Update h
        h = h - lr * gradient
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Evaluate
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            lr *= 0.5
            patience_counter = 0
    
    return (best_h.tolist(), float(best_c5), n_points)
