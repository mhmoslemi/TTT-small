import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute cross-correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to meet the sum constraint while keeping values in [0,1]."""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    h = h * target_sum / current_sum
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Re-normalize if still outside
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Distribute the difference among elements intelligently
        diff = target_sum - current_sum
        idx = np.argsort(h)
        if diff > 0:
            # Need to increase values
            for i in range(len(idx)-1, -1, -1):
                if diff <= 0:
                    break
                j = idx[i]
                if h[j] < 1.0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need to decrease values
            for j in idx:
                if diff >= 0:
                    break
                if h[j] > 0.0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    return h
