def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

        lr = min(0.05, 1.0 / (1+iter_count*0.001))

        h_next = h + lr * noise
        h_next = np.clip(h_next, 0, 1)
        h_next = project_to_constraints(h_next, dx)
        
        c5_new = evaluate(h_next, dx)
        if c5_new < last_c5:
            h = h_next
            last_c5 = c5_new
            if last_c5 < best_c5:
                best_h = h.copy()
                best_c5 = last_c5
            refine_count += 1
            iter_count += 1
        else:
            iter_count += 1
            # Keep stepping by different nearby elevation
            if np.random.rand() < 0.05:
                h_new = h + (0.02 * np.random.rand(n_points) - 0.01)
                h_new = np.clip(h_new, 0, 1)
                h_new = project_to_constraints(h_new, dx)
                c5_new = evaluate(h_new, dx)
                if c5_new < last_c5:
                    h = h_new
                    last_c5 = c5_new
                    if last_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = last_c5
                iter_count += 1

            h_interp = np.interp(np.linspace(0, 2, 2*n_points),
