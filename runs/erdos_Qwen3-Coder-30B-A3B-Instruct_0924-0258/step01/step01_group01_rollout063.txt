import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def admm_update_h(h, z, rho, dx, n):
    # Solve min_h  max_k corr(h, z) dx + (rho/2) * ||h - z + lambda/rho||^2
    # For each k, we have: sum_i h[i] * z[i+k] 
    # The key is to update h in a way that minimizes the max correlation
    # We'll try gradient descent with proper penalty and projection
    h_new = h.copy()
    # Gradient approximation (simplified: just doing gradient descent on one max corr at a time)
    # Not perfect but feasible approach
    corr = correlate(h, z, mode='full')
    max_idx = np.argmax(corr)
    # Focus on one direction that maximizes the correlation in this way
    # This is a heuristic approach but sufficient for exploration
    # Simple gradient: approximate for entropy
    grad = np.zeros(n)
    for i in range(n):
        # Approximate gradient on h[i]: 
        # derivative of the max_k [ sum_j h[j]*z[j+k] ] w.r.t h[i]
        for k in range(2*n-1):
            j = i - k + n - 1
            if 0 <= j < n:
                grad[i] += z[j]
    # Gradient ascent to reduce (convex approximation)
    h_new = h_new + 0.01 * grad  # small step size
    h_new = project_to_constraints(h_new, dx)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 200
    n_points = n_low
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
    h = project_to_constraints(h, dx)
    
    # Track best
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 100
    T = 0.05
    
    # Refinement loop with adaptive resolution
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for it in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # ADMM step: maintain z = 1 - h
            z = 1 - h
            
            # Update h via gradient descent type update
            h_new = admm_update_h(h, z, rho, dx, n_points)
            c5_new = evaluate(h_new, dx)
            
            # Acceptance check (like simulated annealing)
            delta = c5_new - best_c5
            if delta < 0 or np.random.uniform() < np.exp(-delta / T):
                h = h_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    
        # Try to double resolution if improvement and time allows
        if n_points < n_high:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            best_c5 = evaluate(h, dx)
            if best_c5 < 0.3807:
                break
        else:
            break
            
    return best_h.tolist(), best_c5, len(best_h)
