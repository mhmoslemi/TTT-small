import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, max_lag):
        # Compute gradient for the max lag
        grad = np.zeros_like(h_vec)
        if max_lag >= 0:  # Positive lag
            # h[i] contributes to correlation with lag k if h[i+k] is present
            # So gradient for h[i] is -(1 - h[i+k]) 
            for i in range(len(h_vec)):
                j = (i + max_lag) % len(h_vec)
                grad[i] -= (1 - h_vec[j])
        else:  # Negative lag
            max_lag = -max_lag
            for i in range(len(h_vec)):
                j = (i - max_lag) % len(h_vec)
                grad[i] -= h_vec[j]
        return grad
    
    # Main optimization loop
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    
    best_h = h_current.copy()
    best_C5 = C5_current
    
    alpha = 0.1
    stagnation_count = 0
    max_stagnation = 50
    
    # Time check 
    while time.time() - start_time < budget_s - 0.1:
        # Compute all correlations
        corr = correlate(h_current, 1 - h_current, mode='full')
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        max_val = corr[max_lag_idx]
        
        # Compute gradient
        grad = compute_gradient(h_current, max_lag)
        
        # Try multiple step sizes
        step_sizes = [alpha, alpha / 2, alpha / 4]
        best_new_h = h_current.copy()
        best_new_C5 = C5_current
        best_alpha = 0.0
        
        for step_size in step_sizes:
            h_candidate = h_current - step_size * grad
            h_candidate = project_feasible(h_candidate)
            new_C5 = compute_C5(h_candidate)
            if new_C5 < best_new_C5:
                best_new_C5 = new_C5
                best_new_h = h_candidate.copy()
                best_alpha = step_size
                
        if best_new_C5 < C5_current:
            h_current = best_new_h.copy()
            C5_current = best_new_C5
            best_h = h_current.copy()
            best_C5 = C5_current
            stagnation_count = 0
            alpha = min(alpha * 1.2, 1.0)  # Increase learning rate
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                # Add noise and restart with new randomization
                noise = np.random.normal(0, 0.02, n_points)
                h_current = h_current + noise
                h_current = project_feasible(h_current)
                C5_current = compute_C5(h_current)
                stagnation_count = 0
                alpha = max(alpha * 0.5, 0.005)  # Decrease learning rate

    # Final validation
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
