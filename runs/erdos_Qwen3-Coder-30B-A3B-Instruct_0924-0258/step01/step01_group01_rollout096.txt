import numpy as np
import time
from scipy.signal import correlate

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    # The full correlation length is 2*n-1
    # Each entry is scaled by dx
    return np.max(corr) * dx

def compute_gradients(h, dx):
    # For each possible lag k, compute the gradient of the corresponding correlation term
    n = len(h)
    # corr = correlate(h, 1-h, mode='full')
    # For efficiency we'll compute the gradient manually for the maximum
    corr = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    # Find the corresponding lag
    lag = max_idx - (n - 1)
    # Equivalent to the cross-correlation with 1-h at this lag
    # Gradient of h[i] w.r.t. the max 1d lag corr[i + lag]
    # We'll focus on around 10% of the largest correlation values
    max_corr = np.max(corr)
    # Identify critical lags
    threshold = max_corr * 0.95
    critical_lags = np.where(corr >= threshold)[0]
    
    # Build gradient vector
    grad = np.zeros(n)
    for idx in critical_lags:
        # idx corresponds to lag = idx - (n-1) in correlation indexing
        # Correlation shifts h to the right by idx - (n-1)
        k = idx - (n-1)
        # The contribution to corr_k = sum(h[i] * (1 - h[i+k]))
        # Gradient w.r.t. h[i] = 1 - h[i+k] (in range)
        # Gradient w.r.t. h[i+k] = -h[i] (in range)
        # so we compute for all i where i+k is valid in h:
        i_start = max(0, -k)
        i_end = min(n, n-k)
        if k >= 0:
            # h[i] * (1-h[i+k])  
            for i in range(i_start, i_end):
                grad[i] += 1 - h[i+k]
                if i+k < n:
                    grad[i+k] -= h[i]
        else:
            # h[i] * (1-h[i+k])
            # k < 0, i+k < i
            for i in range(i_start, i_end):
                grad[i] += 1 - h[i+k]
                if i+k >= 0:
                    grad[i+k] -= h[i]
    
    # Normalize
    norm = np.linalg.norm(grad)
    if norm > 1e-12:
        grad = grad / norm
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    # Use initial or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)

    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5

    # - Start with low resolution at n=10, 
    # - Make 500–1000 gradient steps per level, 
    # - Cool down learning rate, 
    # - Then adaptively double resolution.
    
    T = 0.1
    cooling = 0.99
    iter_per_temp = 1000

    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break

        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s:
                break
            
            # Compute gradient
            grad = compute_gradients(h, dx)
            # Gradient descent step
            h_new = h - T * grad
            # Clip to [0, 1]
            h_new = np.clip(h_new, 0, 1)
            # Project to satisfy integral constraint
            h_new = project_to_constraints(h_new, dx)
            
            # Evaluate new
            c5_new = evaluate(h_new, dx)
            
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
                    
            iter_count += 1

        # Cooling
        T *= cooling
        
        # Adaptively increase resolution if stable and time permits
        if n_points < n_high and T < 1e-5:
            # Interpolate to make h longer
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                               np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_c5 = evaluate(h, dx)
            T = 0.1
            iter_per_temp = 1000  # reset

        # Early break if too low temperature
        if T < 1e-6:
            break

    return best_h.tolist(), best_c5, len(best_h)
