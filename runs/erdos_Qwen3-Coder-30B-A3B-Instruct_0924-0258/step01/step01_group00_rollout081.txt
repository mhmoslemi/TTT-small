import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial h from parent construction or fallback
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Simple starting construction: uniform distribution
        h_start = np.full(30, 0.5)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Configurations to try: different resolution levels
    resolutions = [25, 30, 35, 40, 45, 50]
    max_iter = 500
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.9: 
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Perturb initial h to create diversification
        h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
        h_current = h_perturbed.copy()
        
        # Gradient-based optimization
        for i in range(max_iter):
            if time() - start_time > budget_s:
                break
                
            # Objective function: negative max correlation (minimize overlap)
            def obj(h):
                corr = np.correlate(h, 1 - h, mode='full')
                return -np.max(corr) * dx
            
            # Gradient: computed numerically
            def grad(h):
                eps = 1e-6
                g = np.zeros_like(h)
                base_obj = obj(h)
                for j in range(len(h)):
                    h_p = h.copy()
                    h_p[j] += eps
                    h_p = project_to_feasible(h_p, n_points)
                    obj_p = obj(h_p)
                    g[j] = (obj_p - base_obj) / eps
                return g
            
            # Constraints for optimization
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj, h_current, method='SLSQP',
                               bounds=bounds, constraints=cons, jac=grad,
                               options={'maxiter': 10, 'ftol': 1e-8})
                if res.success:
                    h_new = project_to_feasible(res.x, n_points)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        best_n_points = n_points
                    h_current = h_new
                else:
                    # Fallback: perturb and reinitialize
                    h_current = project_to_feasible(h_current + np.random.normal(0, 0.02, n_points), n_points)
            except:
                h_current = project_to_feasible(h_current + np.random.normal(0, 0.02, n_points), n_points)
                
            # Occasionally perturb with larger amplitude
            if i % 50 == 0:
                h_current = project_to_feasible(h_current + np.random.normal(0, 0.03, n_points), n_points)
                
        h_start = best_h.copy()
        
    return (list(best_h), best_c5, best_n_points)
