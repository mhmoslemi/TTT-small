import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    resolutions = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    
    for res in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        n_points = res
        dx = 2.0 / n_points
        time_remaining = budget_s - (time.time() - start_time)
        time_per_res = time_remaining * 0.25
        
        # Initialize
        if 'initial_h_values' in globals() and res == 10:
            h = np.array(initial_h_values)
            h = h[:n_points]
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_constraints(h, dx)
            
        # PSO parameters
        n_particles = 100
        max_iter = 500
        w = 0.7
        c1, c2 = 1.5, 1.5
        v_max = 0.05
        
        # Initialize particles
        particles = [h.copy() for _ in range(n_particles)]
        velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(n_particles)]
        pbest = [p.copy() for p in particles]
        pbest_vals = [evaluate(p, dx) for p in particles]
        
        # Global best
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        # Optimizer loop
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[i] = (w * velocities[i] + 
                               c1 * r1 * (pbest[i] - particles[i]) + 
                               c2 * r2 * (gbest - particles[i]))
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = project_to_constraints(particles[i], dx)
                
                # Evaluate fitness
                fitness = evaluate(particles[i], dx)
                
                # Update personal best
                if fitness < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = fitness
                    
                    # Update global best
                    if fitness < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = fitness
                        
            if gbest_val < best_c5:
                best_h = gbest.copy()
                best_c5 = gbest_val
                
        h = gbest.copy()
        
        # Gradient descent
        max_gd_iter = 1000
        step_size = 0.01
        for _ in range(max_gd_iter):
            if time.time() - start_time > budget_s:
                break
                
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Compute gradient using finite differences to approximate dC5/dh_i
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = project_to_constraints(h_plus, dx)
                val_plus = evaluate(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                h_minus = project_to_constraints(h_minus, dx)
                val_minus = evaluate(h_minus, dx)
                
                grad[i] = (val_plus - val_minus) / (2e-5)
            
            # Update step
            h_new = h - step_size * grad
            h_new = project_to_constraints(h_new, dx)
            
            # Accept if better
            c5_new = evaluate(h_new, dx)
            if c5_new < evaluate(h, dx):
                h = h_new
            else:
                # Step size reduction
                step_size *= 0.95
                if step_size < 1e-6:
                    break
                    
        # Final update of best
        if evaluate(h, dx) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate(h, dx)
            
    return best_h.tolist(), best_c5, len(best_h)
