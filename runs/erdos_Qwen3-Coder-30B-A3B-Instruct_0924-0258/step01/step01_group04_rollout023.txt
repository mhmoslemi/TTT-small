import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create initial guess
        n_points = 20
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Try SDP approach for small dimensions
    def solve_sdp(n_points_local):
        # Estimate duration for SDP solve
        if (time.time() - start_time) > budget_s - 100:
            return None
            
        h_local = np.ones(n_points_local) * 0.5
        try:
            # Use warm start if possible
            if 'h_sdp' in globals():
                h_local = h_sdp[:n_points_local]
            
            # CVXPY SDP setup
            h_var = cp.Variable(n_points_local)
            X = cp.Variable((n_points_local, n_points_local), symmetric=True)
            t = cp.Variable()
            
            # Constraints
            constraints = [
                cp.constraints.PSD(X),
                cp.diag(X) == h_var,
                0 <= h_var,
                h_var <= 1,
                cp.sum(h_var) == target_sum,
                t <= cp.sum(X[:, :n_points_local])  # Simple constraint to avoid vectorized indexing issues
            ]
            
            # Simple version with enumerated constraints
            for k in range(1, n_points_local):
                constraint = t <= cp.sum(X[range(n_points_local-k), range(k, n_points_local)])
                constraints.append(constraint)
            
            # Objective: maximize t
            objective = cp.Maximize(t)
            prob = cp.Problem(objective, constraints)
            
            # Try to solve with MOSEK or another efficient solver
            prob.solve(solver=cp.MOSEK, verbose=False, max_iter=100)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_result = h_var.value
                return h_result
        except:
            pass
        return None

    # Main optimization loop with alternating heuristics
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    max_iter = 10000
    iter_count = 0
    n_points_current = n_points
    last_improvement = 0

    while (time.time() - start_time) < budget_s - 5 and iter_count < max_iter:
        iter_count += 1
        
        # Every 200 iterations, try SDP if resolvable and time allows
        if iter_count % 200 == 0 and n_points_current <= 40 and (time.time() - start_time) < budget_s - 100:
            sdp_result = solve_sdp(n_points_current)
            if sdp_result is not None:
                sdp_h = project_h(sdp_result)
                sdp_C5 = compute_C5(sdp_h)
                if sdp_C5 < best_C5:
                    best_C5 = sdp_C5
                    best_h = sdp_h.copy()
                    last_improvement = iter_count
                    
        # Local search with gradient estimation
        h_vec = project_h(best_h)
        opt_h = h_vec.copy()
        
        # Finite difference gradient estimation
        grad = np.zeros_like(h_vec)
        eps = 1e-4
        for i in range(n_points_current):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            grad[i] = (C_plus - C_minus) / (2 * eps)
            
        # Step descent with adaptive learning rate
        learning_rate = 0.01
        opt_h -= learning_rate * grad
        opt_h = project_h(opt_h)
        
        # Update if better
        opt_C5 = compute_C5(opt_h)
        if opt_C5 < best_C5:
            best_C5 = opt_C5
            best_h = opt_h.copy()
            last_improvement = iter_count
            # Increase learning rate for faster convergence if good
            learning_rate *= 1.1
        else:
            # Decrease learning rate if no improvement
            learning_rate *= 0.9
            
        # Periodic evaluation
        if iter_count % 500 == 0:
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()

        # If stalled, add small random noise
        if iter_count - last_improvement > 1000 and iter_count > 2000:
            noise_magnitude = 0.01
            best_h += np.random.normal(0, noise_magnitude, n_points_current)
            best_h = project_h(best_h)
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                last_improvement = iter_count

    # Final evaluation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points_current)
