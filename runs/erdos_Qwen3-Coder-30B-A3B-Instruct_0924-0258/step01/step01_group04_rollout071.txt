import numpy as np
from scipy.optimize import minimize
import time
try:
    from cvxpy import *
except ImportError:
    # Fallback to a simpler approach if CVXPY is not available
    pass

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the provided construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback to simple construction
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Use scipy.optimize for direct optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    # Start with a more aggressive adaptation
    def objective(h_vec):
        return compute_C5(h_vec)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Perturb initial h to ensure exploring different regions
    n_iter = 0
    max_iter = int(2e4)
    
    while time.time() - start_time < budget_s - 1 and n_iter < max_iter:
        n_iter += 1
        
        # Add small noise to escape local optima
        noise = np.random.normal(0, 0.005, n_points)
        trial_h = best_h + noise
        trial_h = project_h(trial_h)
        
        result = minimize(objective, trial_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, 
                          options={'ftol': 1e-8, 'gtol': 1e-6, 'maxiter': 50})
        
        if result.success:
            current_h = project_h(result.x)
            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_h = current_h.copy()
                best_C5 = current_C5
                # Increase learning rate adaptively
                # In this simple version, we just use better exploration
        else:
            # Revert to best
            current_h = best_h
            current_C5 = best_C5
        
        # Early stopping if limited by time
        if n_iter % 1000 == 0:
            if time.time() - start_time > budget_s - 10:
                break
    
    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
