import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Initialize h
    h = h0.copy().astype(float)
    dx = 2.0 / n_points
    # Normalize initial h to meet sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Parameters
    eta = 0.01
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 50
    smoothing_window = 5
    
    # Gradient history for smoothing
    grad_history = []
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Compute all correlations to find the worst case shift
        corr = correlate(h, 1 - h, mode="full")
        # Find the shift index that maximizes correlation
        max_idx = np.argmax(np.abs(corr))
        k_max = max_idx - (len(h) - 1)
        
        # Compute the gradient based on k_max
        gradient = np.zeros_like(h)
        if k_max >= 0:
            # h[i] contributes to gradient of h[i + k_max]
            gradient[k_max:] += (1 - h[:-k_max])
        else:
            # h[i] contributes to gradient of h[i - k_max] where k_max < 0
            k_abs = -k_max
            gradient[:-k_abs] += (1 - h[k_abs:])
        
        # Smooth gradient
        if len(grad_history) >= smoothing_window:
            grad_history.pop(0)
        grad_history.append(gradient)
        
        if len(grad_history) >= 2:
            avg_grad = np.mean(grad_history, axis=0)
        else:
            avg_grad = gradient

        # Update h
        h = h - eta * avg_grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        # Re-normalize to maintain integral constraint
        h = h * (n_points / 2.0) / current_sum
        
        # Evaluate current solution
        current_c5 = evaluate(h, dx)
        
        # Update best if found
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adaptive learning rate
        if no_improvement_count > 100:
            eta *= 0.9
            no_improvement_count = 0
        
        # Perturbation if no improvement for a while
        if no_improvement_count >= max_no_improvement:
            # Add small random perturbation
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * (n_points / 2.0) / current_sum
            no_improvement_count = 0
        
        # Occasionally re-evaluate best to avoid stagnation
        if iter_count % 1000 == 0:
            current_c5 = evaluate(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
