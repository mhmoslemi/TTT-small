import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization: approximate solution from literature
        n_points = 45
        h = np.zeros(n_points)
        # Place 1's in first half with some smoothing
        h[:n_points//2] = 1.0
        # Apply a bit of smoothing
        h = np.convolve(h, np.array([0.25, 0.5, 0.25]), mode='same')
        h = np.clip(h, 0, 1)
        # Adjust to sum constraint
        h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags using scipy
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm
    pop_size = 50
    tournament_size = 10
    elitism = int(0.2 * pop_size)
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    max_generations = 1000
    
    # Initialize population
    def init_population(size, n):
        pop = []
        for _ in range(size):
            h = np.random.rand(n)
            h = project_feasible(h)
            pop.append(h)
        return pop
    
    population = init_population(pop_size, n_points)
    
    def evaluate_population(pop):
        fitnesses = []
        for ind in pop:
            fitness = compute_C5(ind)
            fitnesses.append(fitness)
        return np.array(fitnesses)
    
    # Tournament selection
    def tournament_selection(pop, fitnesses, k=tournament_size):
        indices = np.random.choice(len(pop), size=k, replace=False)
        best_idx = indices[np.argmin(fitnesses[indices])]
        return pop[best_idx]
    
    # Uniform crossover
    def uniform_crossover(parent1, parent2):
        mask = np.random.rand(len(parent1)) > 0.5
        child = np.where(mask, parent1, parent2)
        return project_feasible(child)
    
    # Two-point crossover
    def two_point_crossover(parent1, parent2):
        point1 = np.random.randint(1, len(parent1) - 1)
        point2 = np.random.randint(point1 + 1, len(parent1))
        child = parent1.copy()
        child[point1:point2] = parent2[point1:point2]
        return project_feasible(child)
    
    # Gaussian mutation
    def gaussian_mutation(individual, rate=mutation_rate, sigma=0.05):
        mask = np.random.rand(len(individual)) < rate
        noise = np.random.normal(0, sigma, len(individual))
        child = individual + noise * mask
        return project_feasible(child)
    
    # Reset mutation
    def reset_mutation(individual, rate=0.05):
        mask = np.random.rand(len(individual)) < rate
        child = individual.copy()
        child[mask] = np.random.choice([0, 1], size=np.sum(mask))
        return project_feasible(child)
    
    # Main evolutionary loop
    best_fitness = float('inf')
    best_individual = h.copy()
    no_improvement_count = 0
    
    # Evaluate initial population
    fitnesses = evaluate_population(population)
    current_best_idx = np.argmin(fitnesses)
    if fitnesses[current_best_idx] < best_fitness:
        best_fitness = fitnesses[current_best_idx]
        best_individual = population[current_best_idx].copy()
        
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitnesses[i] for i in sorted_indices]
        
        # Elitism: keep best individuals
        new_pop = sorted_pop[:elitism]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            if np.random.rand() < crossover_rate:
                parent1 = tournament_selection(population, fitnesses)
                parent2 = tournament_selection(population, fitnesses)
                if np.random.rand() < 0.5:
                    child = uniform_crossover(parent1, parent2)
                else:
                    child = two_point_crossover(parent1, parent2)
            else:
                parent = tournament_selection(population, fitnesses)
                child = parent.copy()
            
            # Mutation step
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    child = gaussian_mutation(child)
                else:
                    child = reset_mutation(child)
            
            new_pop.append(child)
        
        population = new_pop[:pop_size]
        fitnesses = evaluate_population(population)
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping if no improvement
        if no_improvement_count > patience:
            break
            
    # Final check and return
    final_h = project_feasible(best_individual)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
