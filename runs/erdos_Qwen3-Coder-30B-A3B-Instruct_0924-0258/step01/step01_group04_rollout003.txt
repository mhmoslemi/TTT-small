import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    learning_rate = 0.01
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % 100 == 0:
            # Periodic full evaluation
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = best_h.copy()
        
        # Compute the current overlap for all shifts
        h_vec = project_h(h)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        
        # Find the dominant shift k
        k_max = np.argmax(overlaps)
        
        # Compute gradient for this shift
        gradient = np.zeros_like(h_vec)
        for i in range(n_points):
            # The derivative of integral h(x)(1-h(x+k)) w.r.t. h[i] is (1-h[(i-k)%n])
            j = (i - k_max) % n_points
            gradient[i] = 1 - h_vec[j]
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * gradient**2
        m_hat = m / (1 - beta1**(iteration))
        v_hat = v / (1 - beta2**(iteration))
        
        # Update parameters
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project
        h = project_h(h)
            
        # Check if we improved
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
    
    # Final check
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
