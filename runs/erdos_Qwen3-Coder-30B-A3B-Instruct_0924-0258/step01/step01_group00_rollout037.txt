import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

        n_points = len(h_start)

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / n_points)
    
    # Try a variety of resolutions to avoid local minima
    resolutions = [20, 25, 30, 35, 40, 45, 50]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.9:
            break
            
        dx = 2.0 / res
        h_current = project_to_feasible(np.interp(np.linspace(0, 2, res), 
                                                 np.linspace(0, 2, len(h_start)), 
                                                 h_start), res)
        
        # Try multiple restarts with different perturbations
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb the current solution
            h_perturbed = project_to_feasible(h_current + np.random.normal(0, 0.03, res), res)
            
            # Use a simple local optimization approach
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint(h):
                return np.sum(h) - res / 2
                
            cons = {'type': 'eq', 'fun': constraint}
            bounds = [(0, 1) for _ in range(res)]
            
            try:
                result = minimize(objective, h_perturbed, method='SLSQP',
                                bounds=bounds, constraints=cons,
