import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default initialization
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def solve_feasibility(h0, max_overlap):
        # Construct the convex optimization problem for feasibility check
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            if k == 0:
                # Special case
                expr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                expr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            constraints.append(expr <= max_overlap * dx)
            
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return True, h.value
            else:
                return False, None
        except:
            return False, None
    
    # Binary search parameters
    low = 0.375
    high = compute_c5(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    
    # Binary search to find minimal possible C5
    while time.time() - start_time < budget_s - 10 and high - low > 1e-4:
        mid = (low + high) / 2
        feasible, new_h = solve_feasibility(best_h, mid)
        
        if feasible:
            high = mid
            if new_h is not None:
                best_h = project_h(new_h)
            best_c5 = compute_c5(best_h)
        else:
            low = mid
    
    # Final refinement using gradient-based local optimization
    def objective(h_vec):
        return compute_c5(h_vec)
    
    # Local refinement
    h_refined = best_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    try:
        res = minimize(objective, h_refined, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-6})
        if res.success:
            h_refined = project_h(res.x)
            c5_refined = compute_c5(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
