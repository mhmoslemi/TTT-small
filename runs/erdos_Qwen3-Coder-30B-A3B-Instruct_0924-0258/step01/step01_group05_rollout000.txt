import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with 45 points
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Local optimization with coordinate descent and swaps
    def local_optimize(h_vec, max_iter=100):
        h_current = h_vec.copy()
        h_current = project_feasible(h_current)
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        # Coordinate descent
        step_size = 0.1
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            improved = False
            for i in range(n_points):
                # Save original value
                orig_val = h_current[i]
                
                # Try moving up and down in small steps
                for direction in [-1, 1]:
                    # Try different step sizes
                    for step in [0.05, 0.1]:
                        candidate_h = h_current.copy()
                        candidate_h[i] = orig_val + direction * step
                        
                        # Project and check
                        candidate_h = project_feasible(candidate_h)
                        new_C5 = compute_C5(candidate_h)
                        
                        if new_C5 < best_C5:
                            h_current = candidate_h.copy()
                            best_C5 = new_C5
                            improved = True
                            if new_C5 < best_C5:
                                best_h = h_current.copy()
                                best_C5 = new_C5
                            
            # Occasionally try swapping intervals
            if np.random.random() < 0.1:
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                candidate_h = h_current.copy()
                candidate_h[idx1], candidate_h[idx2] = candidate_h[idx2], candidate_h[idx1]
                candidate_h = project_feasible(candidate_h)
                new_C5 = compute_C5(candidate_h)
                if new_C5 < best_C5:
                    h_current = candidate_h
                    best_C5 = new_C5
                    improved = True
                    best_h = h_current.copy()
            
            # Reset step size if no improvement for a while
            if not improved:
                step_size = min(step_size * 1.1, 0.5)
            else:
                step_size = 0.1  # Reset to normal
                
            if not improved and iteration > 0 and iteration % 20 == 0:
                # Random perturb for escape
                for i in range(n_points//5):
                    idx = np.random.randint(n_points)
                    h_current[idx] += np.random.normal(0, 0.02)
                h_current = project_feasible(h_current)
                
        return best_h, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Initialize multiple candidates using Latin hypercube sampling
    # Will stay with the best solution
    candidates = []
    for _ in range(5):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Generate random sample
        random_h = np.random.rand(n_points)
        random_h = project_feasible(random_h)
        candidates.append(random_h)

    # Add the current best as one candidate
    candidates.append(h_best)
    
    # Evaluate and keep best initial candidate
    best_initial_index = 0
    for i, candidate in enumerate(candidates):
        C5 = compute_C5(candidate)
        if C5 < compute_C5(candidates[best_initial_index]):
            best_initial_index = i
    h_best = candidates[best_initial_index].copy()
    C5_best = compute_C5(h_best)
    
    # Main optimization loop
    no_improvement_count = 0
    while time.time() - start_time < budget_s - 0.1:
        # Optimize each candidate
        for i, candidate in enumerate(candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
            h_opt, C5_opt = local_optimize(candidate, max_iter=30)
            if C5_opt < C5_best:
                C5_best = C5_opt
                h_best = h_opt.copy()
                
        # Occasionally create new candidates
        if np.random.random() < 0.3 and time.time() - start_time < budget_s - 10:
            for _ in range(2):
                # Create a new candidate part way through
                h_new = np.random.rand(n_points)
                h_new = project_feasible(h_new)
                candidates.append(h_new)
        
        # Trim candidates list to avoid growth
        if len(candidates) > 10:
            # Keep the best ones
            candidate_scores = [compute_C5(c) for c in candidates]
            best_indices = np.argsort(candidate_scores)[:5]
            candidates = [candidates[i] for i in best_indices]

    # Final validation
    h_best = project_feasible(h_best)
    final_C5 = compute_C5(h_best)
    
    return list(h_best), final_C5, n_points
