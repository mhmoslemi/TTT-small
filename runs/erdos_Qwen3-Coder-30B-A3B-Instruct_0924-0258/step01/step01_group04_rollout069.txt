import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: start with a simple construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def upsample_h(h_vec, target_len):
        x_orig = np.linspace(0, 1, len(h_vec))
        x_target = np.linspace(0, 1, target_len)
        return np.interp(x_target, x_orig, h_vec)

    # Multi-resolution optimization
    resolutions = [5, 10, 20, 40, n_points]
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    for i, res in enumerate(resolutions[:-1]):
        current_points = res
        if i == 0:
            h_current = np.ones(current_points) * 0.5
        else:
            h_current = upsample_h(best_h, current_points)
        
        # Adam optimizer parameters
        m = np.zeros_like(h_current)
        v = np.zeros_like(h_current)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.01
        
        # Optimization loop
        for k in range(1000):
            if time.time() - start_time > budget_s - 2:
                break
                
            h_current = project_feasible(h_current)
            conv = np.correlate(h_current, 1 - h_current, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-current_points+1:mid+current_points-1] * dx
            top_k = np.argsort(overlaps)[-5:][::-1]
            
            # Compute gradient with respect to the maximum overlap
            gradient = np.zeros_like(h_current)
            eps_val = 1e-4
            for j in range(current_points):
                h_plus = h_current.copy()
                h_plus[j] += eps_val
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                h_minus = h_current.copy()
                h_minus[j] -= eps_val
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                gradient[j] = (C_plus - C_minus) / (2 * eps_val)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * gradient**2
            m_hat = m / (1 - beta1**(k+1))
            v_hat = v / (1 - beta2**(k+1))
            h_current = h_current - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project to feasible region
            h_current = project_feasible(h_current)
            
            # Check for improvement
            current_C5 = compute_overlap(h_current)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current.copy()
    
    # Final high-resolution optimization
    h_final = best_h.copy()
    for i in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_final = project_feasible(h_final)
        conv = np.correlate(h_final, 1 - h_final, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        top_k = np.argsort(overlaps)[-5:][::-1]
        
        gradient = np.zeros_like(h_final)
        eps_val = 1e-4
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps_val
            h_plus = project_feasible(h_plus)
            C_plus = compute_overlap(h_plus)
            
            h_minus = h_final.copy()
            h_minus[j] -= eps_val
            h_minus = project_feasible(h_minus)
            C_minus = compute_overlap(h_minus)
            
            gradient[j] = (C_plus - C_minus) / (2 * eps_val)
        
        # Adam update (simplified version)
        m = 0.9 * m + 0.1 * gradient
        v = 0.999 * v + 0.001 * gradient**2
        h_final = h_final - 0.01 * m / (np.sqrt(v) + 1e-8)
        h_final = project_feasible(h_final)
        
        current_C5 = compute_overlap(h_final)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_final.copy()
    
    return (best_h.tolist(), best_C5, n_points)
