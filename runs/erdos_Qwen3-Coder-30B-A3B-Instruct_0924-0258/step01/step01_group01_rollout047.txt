import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def compute_gradient(h, dx, critical_lags):
    n = len(h)
    grad = np.zeros(n)
    # For each critical lag k, compute contribution to gradient
    for k in critical_lags:
        # Compute derivative of correlation with respect to h[i]
        # This is tricky with cross-correlation, so we do finite differences
        # But since we're maximizing the correlation, we can use the fact
        # that it's the same as the correlation of h with (1-h) shifted
        # For lag k, the contribution to grad at position i is (1 - h[i+k]) if k >= 0
        # Let's compute it more carefully using convolution trick
        # We build gradient using the fact that d/dh_i of int h(x)(1-h(x+i))dx 
        # is (1 - h[i+k]) for k where lag exists
        # But the derivative of correlation with h is tricky...
        # So we take difference of correlation when we change h[i] by epsilon
        if k >= 0 and k < n:
            for i in range(n):
                if i + k < n:
                    grad[i] += (1.0 - h[i + k]) * dx
    return grad

def gradient_descent_step(h, dx, lr, critical_lags):
    grad = compute_gradient(h, dx, critical_lags)
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_constraints(h_new, dx)
    return h_new

def find_critical_lags(h, dx):
    corr = correlate(h, 1-h, mode='full')
    # Get indices of top 5% contributions to max
    abs_corr = np.abs(corr)
    top_k = max(1, int(len(corr) * 0.05))
    top_indices = np.argpartition(abs_corr, -top_k)[-top_k:]
    # But we want the actual lag values for computing gradient
    # Full correlation has length 2*n - 1, centered
    # correlation at index i means lag i - (n-1) 
    lags = np.arange(len(corr)) - (len(h) - 1)
    top_lags = lags[top_indices]
    # Only take meaningful lags (exclude padding)
    valid_lags = top_lags[np.abs(top_lags) < len(h)]
    return list(set(valid_lags))  # unique lags

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter_per_res = 1000
    patience = 100
    patience_counter = 0
    
    # Track last improvement
    last_c5 = best_c5
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            return best_h.tolist(), best_c5, len(best_h)
            
        # At each resolution, optimize with gradient descent
        iter_count = 0
        for _ in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            # Find critical lags (most influential for the max)
            crit_lags = find_critical_lags(h, dx)
            
            # Take one step
            h_new = gradient_descent_step(h, dx, learning_rate, crit_lags)
            c5_new = evaluate(h_new, dx)
            
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                patience_counter = 0
            else:
                patience_counter += 1
                
            iter_count += 1
            if patience_counter > patience:
                break
                
        # Adaptively increase resolution if improvement is found
        if n_points < n_high and last_c5 < best_c5*0.999:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_c5 = evaluate(h, dx)
            patience_counter = 0
            # Decrease learning rate for finer resolution
            learning_rate = min(0.01, learning_rate * 0.8)
        else:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
