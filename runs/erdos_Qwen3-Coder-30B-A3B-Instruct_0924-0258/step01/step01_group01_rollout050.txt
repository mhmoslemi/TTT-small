import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    resolutions = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    
    for i, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Initialize
        if i == 0 and 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = h[:n_points]
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_constraints(h, dx)
            
        # Particle swarm optimization + gradient descent
        n_particles = 100
        velocities = np.random.uniform(-0.05, 0.05, (n_particles, n_points))
        particle_positions = np.random.uniform(0, 1, (n_particles, n_points))
        particle_positions = np.array([project_to_constraints(p, dx) for p in particle_positions])
        particle_best_positions = particle_positions.copy()
        particle_best_values = np.array([evaluate(p, dx) for p in particle_positions])
        
        inertia = 0.7
        w1, w2 = 1.5, 1.5
        max_iter = 5000
        
        local_best_idx = np.argmin(particle_best_values)
        global_best_position = particle_best_positions[local_best_idx].copy()
        global_best_value = particle_best_values[local_best_idx]
        
        # Limit time for each resolution
        resolution_time = budget_s / len(resolutions) * 0.9
        resolution_start = time.time()
        
        for it in range(max_iter):
            if time.time() - resolution_start > resolution_time:
                break
                
            # Update particles
            for p in range(n_particles):
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[p] = (inertia * velocities[p] + 
                                w1 * r1 * (particle_best_positions[p] - particle_positions[p]) +
                                w2 * r2 * (global_best_position - particle_positions[p]))
                velocities[p] = np.clip(velocities[p], -0.05, 0.05)
                particle_positions[p] += velocities[p]
                particle_positions[p] = project_to_constraints(particle_positions[p], dx)
                
                # Evaluate and update best
                current_value = evaluate(particle_positions[p], dx)
                if current_value < particle_best_values[p]:
                    particle_best_values[p] = current_value
                    particle_best_positions[p] = particle_positions[p].copy()
                    
                if current_value < global_best_value:
                    global_best_value = current_value
                    global_best_position = particle_positions[p].copy()
                    
            # Gradient-based refinement for the best particle
            if np.random.random() < 0.1 and time.time() - resolution_start < resolution_time * 0.8:
                # Compute gradient analytically
                h_ref = global_best_position.copy()
                grad = np.zeros_like(h_ref)
                # Numerical gradient for max_k int h(x)(1-h(x+k))dx
                # For simplicity, use a simple gradient descent step
                step_size = 0.01
                h_new = h_ref + step_size * grad
                h_new = project_to_constraints(h_new, dx)
                new_value = evaluate(h_new, dx)
                if new_value < global_best_value:
                    global_best_value = new_value
                    global_best_position = h_new
                    
        if global_best_value < best_c5:
            best_c5 = global_best_value
            best_h = global_best_position.copy()
            
    return best_h.tolist(), best_c5, len(best_h)
