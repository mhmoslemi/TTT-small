import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and 0 <= h <= 1."""
    h = np.array(h, dtype=float)
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to correct sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Adjust by redistributing excess/deficit
    diff = target_sum - current_sum
    if diff > 0:
        # Need to add diff, can only add to values < 1
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] < 1:
                add = min(diff, 1.0 - h[j])
                h[j] += add
                diff -= add
    else:
        # Need to subtract |diff|, can only subtract from values > 0
        diff = -diff
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] > 0:
                sub = min(diff, h[j])
                h[j] -= sub
                diff -= sub
    
    return h

    target_sum = n_points / 2.0
    
    # Initialize best solution
    best_h = project_to_feasible(h0, target_sum)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    elapsed = 0

        h_start += noise

                h_start[i] = np.clip(h_start[i] + np.random.normal(0, 0.1), 0, 1)
        
        h_start = project_to_feasible(h_start, target_sum)
        
        # Run local optimization on this point using Nelder-Mead
        def obj(h_vec):
            h_proj = project_to_feasible(h_vec, target_sum)
            return evaluate_erdos_solution(h_proj, dx)
        
        # Run nelder-mead
        res = minimize(obj, h_start, method='Nelder-Mead', options={'maxiter': 500, 'xtol': 1e-6, 'ftol': 1e-6})
        
        if res.success:
            h_opt = project_to_feasible(res.x, target_sum)
            c5_opt = evaluate_erdos_solution(h_opt, dx)

                best_h = h_opt

            c5_mut = evaluate_erdos_solution(h_mut, dx)

                best_h = h_mut
                best_c5 = c5_mut
