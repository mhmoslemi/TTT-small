import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint and stay in [0,1]."""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    h = h * target_sum / current_sum
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Re-normalize if still outside
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Distribute difference carefully
        diff = target_sum - current_sum
        idx = np.argsort(h)
        if diff > 0:
            # Need to increase values
            for i in range(len(idx)-1, -1, -1):
                if diff <= 0:
                    break
                j = idx[i]
                if h[j] < 1.0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need to decrease values
            for i in range(len(idx)):
                if diff >= 0:
                    break
                j = idx[i]
                if h[j] > 0.0:
                    update = max(diff, -h[j])
                    h[j] += update
                    diff -= update
    return h

    # Gradient descent parameters
    learning_rate = 0.01
