import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # PSO parameters
    n_particles = 100
    max_iter = 1000
    inertia_start = 0.7
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_speed = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(n_particles):
        # Start near the initial solution with small random perturbations
        h_p = h + np.random.normal(0, 0.05, n_points)
        h_p = project_h(h_p)
        particles.append(h_p.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest_positions.append(h_p.copy())
        pbest_values.append(compute_C5(h_p))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iteration
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            new_velocity = (inertia * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) +
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Clamp velocity
            new_velocity = np.clip(new_velocity, -max_speed, max_speed)
            
            # Update position
            new_position = particles[i] + new_velocity
            
            # Project into feasible region
            new_position = project_h(new_position)
            
            # Evaluate new position
            new_value = compute_C5(new_position)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest_values[i] = new_value
                pbest_positions[i] = new_position.copy()
                
                # Update global best
                if new_value < gbest_value:
                    gbest_value = new_value
                    gbest_position = new_position.copy()
            
            # Update particle
            particles[i] = new_position
            velocities[i] = new_velocity
    
    return (gbest_position.tolist(), gbest_value, n_points)
