import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def optimize_with_local_search(h_start, n_points, dx, time_limit):
    """Optimize using a hybrid with local search and perturbation"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Constraints and bounds for scipy optimization
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Sample size for perturbation
    sample_size = min(20, n_points)
    
    for _ in range(1000):  # Loop with early termination
        if time() - start_time > time_limit:
            break
            
        # Random sample for local search
        indices = np.random.choice(n_points, sample_size, replace=False)
        h_local = best_h.copy()
        h_local[indices] += np.random.normal(0, 0.02, sample_size)
        
        # Project and optimize
        h_local = project_to_feasible(h_local, n_points)
        
        # Some gradient-based optimization
        try:
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                h_local,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
    
    return best_h, best_c5

    best_n_points = len(h_start)
    dx = 2.0 / best_n_points
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Resolution to try
    resolutions = [30, 45, 60, 75]
    
    # Perform optimizations at different resolutions
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate initial h_start to new resolution if needed
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time allocation for each resolution
        remaining_time = budget_s - (time() - start_time)
