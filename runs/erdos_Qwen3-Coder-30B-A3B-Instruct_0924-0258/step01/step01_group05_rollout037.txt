import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        # Shuffle to avoid bias
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        """Compute the maximum overlap integral."""
        h_vec = np.array(h_vec)
        # Compute all cross-correlations
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        """Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum."""
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Initialize parameters for simulated annealing
    T = 0.1  # Initial temperature
    T_min = 1e-5  # Minimum temperature
    cooling_rate = np.log(10) / budget_s  # Cooling schedule
    
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    best_h = current_h.copy()
    best_C5 = current_C5
    
    # Track acceptance rate for adaptive step size
    accept_count = 0
    total_attempts = 0
    adaptive_sigma = 0.1
    
    # Main annealing loop
    while time.time() - start_time < budget_s - 0.1:
        # Sample top 20% intervals by influence on maximum overlap
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_lag_idx = np.argmax(corr)
        # Identify indices that most contribute to this max overlap
        # This is a simple heuristic to identify "critical" intervals
        # In a real implementation, one might compute the gradient or
        # identify which terms make up the maximal cross-correlation
        
        # Perturb the solution: Gaussian noise to some indices
        new_h = current_h.copy()
        
        # Adaptive perturbation with selective indices
        # Modify top 25% of intervals by importance
        num_perturb = max(1, n_points // 4)
        idxs = np.random.choice(n_points, size=num_perturb, replace=False)
        
        # Perturb selected indices with adaptive Gaussian noise
        noise = np.random.normal(0, adaptive_sigma, num_perturb)
        new_h[idxs] += noise
        
        # Apply feasibility projection
        projected_h = project_feasible(new_h)
        
        # Compute new overlap value
        new_C5 = compute_C5(projected_h)
        
        # Acceptance probability
        delta_C5 = new_C5 - current_C5
        total_attempts += 1

            current_C5 = new_C5
            accept_count += 1
            
            # Update best solution
            if new_C5 < best_C5:
                best_h = projected_h.copy()
                best_C5 = new_C5
        
        # Update temperature
        T = T * np.exp(-cooling_rate)
        
        # Adaptive step size adjustment
        if total_attempts > 100:
            accept_rate = accept_count / total_attempts
            if accept_rate < 0.1:
                adaptive_sigma *= 1.2
            elif accept_rate > 0.4:
                adaptive_sigma *= 0.8
            accept_count = 0
            total_attempts = 0

        if np.random.rand() < 0.05:
            flip_idx = np.random.randint(0, n_points)
            new_h[flip_idx] = 1 - new_h[flip_idx]
            new_h = project_feasible(new_h)
            new_C5_alt = compute_C5(new_h)
            if new_C5_alt < current_C5:

                current_C5 = new_C5_alt
                if new_C5_alt < best_C5:
                    best_h = new_h.copy()
                    best_C5 = new_C5_alt

    # Final projection to ensure feasibility
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
