import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple uniform start
    
    # Initialize
    h = h0.copy().astype(float)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Optimization parameters
    learning_rate = 0.01
    stagnation_count = 0
    best_c5 = float('inf')
    best_h = None
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 10000:
        # Evaluate current state
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute correlation to find max shift
        corr = correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(corr)
        
        # Compute gradient w.r.t. h at the maximizing shift
        grad = np.zeros_like(h)
        n = len(h)
        for i in range(n):
            j = (i - max_shift) % n  # wrap around
            grad[i] = 1 - 2 * h[j]
        
        # Update h
        h = h - learning_rate * grad
        
        # Project to feasible set
        h = project_to_feasible(h, target_sum)
        
        # Adaptive learning rate
        if stagnation_count > 50:
            learning_rate *= 0.9
            stagnation_count = 0
            # Add some random noise
            noise = np.random.uniform(-1e-3, 1e-3, n)
            h = h + noise
            h = project_to_feasible(h, target_sum)
        else:
            stagnation_count += 1
            
        iteration += 1
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback
        h_final = project_to_feasible(h0, target_sum)
        final_c5 = evaluate(h_final, dx)
        return (h_final.tolist(), float(final_c5), n_points)
