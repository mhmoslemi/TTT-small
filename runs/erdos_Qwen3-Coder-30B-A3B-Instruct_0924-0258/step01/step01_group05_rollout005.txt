import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        """Compute the maximum overlap integral."""
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        """Project h onto feasible set: 0 <= h <= 1, sum(h) = target_sum."""
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # GA parameters
    pop_size = 50
    tournament_size = 10
    elitism = int(0.2 * pop_size)
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        population.append(h_init)
    
    best_C5 = float('inf')
    best_h = None
    no_improvement_count = 0
    
    generation = 0
    while time.time() - start_time < budget_s - 0.1:
        generation += 1
        # Evaluate fitness
        fitness = [compute_C5(h) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_C5:
            best_C5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            # Reset worst individual
            worst_idx = np.argmax(fitness)
            population[worst_idx] = np.random.rand(n_points)
            population[worst_idx] = project_feasible(population[worst_idx])
            no_improvement_count = 0
            
        # Selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size, replace=False)
            winner = min(tournament, key=lambda i: fitness[i])
            selected.append(population[winner].copy())
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness)[:elitism]
        new_population = [population[i].copy() for i in elite_indices]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if len(new_population) >= pop_size:
                break
            # Select two parents
            parent1, parent2 = np.random.choice(pop_size, 2, replace=False)
            parent1 = selected[parent1]
            parent2 = selected[parent2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutate_indices1 = np.random.rand(n_points) < mutation_rate
            mutate_indices2 = np.random.rand(n_points) < mutation_rate
            
            child1[mutate_indices1] += np.random.normal(0, 0.05, np.sum(mutate_indices1))
            child2[mutate_indices2] += np.random.normal(0, 0.05, np.sum(mutate_indices2))
            
            # Try reset mutation as fallback
            if np.random.rand() < 0.1:
                reset_indices1 = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                reset_indices2 = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                child1[reset_indices1] = np.random.choice([0, 1], size=len(reset_indices1))
                child2[reset_indices2] = np.random.choice([0, 1], size=len(reset_indices2))
            
            # Project to feasible space
            child1 = project_feasible(child1)
            child2 = project_feasible(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
            
        population = new_population[:pop_size]
        
    # Final evaluation
    if best_h is None:
        best_h = population[0]
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
