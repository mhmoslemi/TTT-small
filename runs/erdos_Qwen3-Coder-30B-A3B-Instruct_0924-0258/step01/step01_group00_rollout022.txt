import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) / 2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    target = 0.38
    
    # Main optimization loop
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 50
    
    while time() - start_time < budget_s * 0.95:
        iter_count += 1
        if no_improve_count >= max_no_improve:
            # Perturb slightly
            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
            no_improve_count = 0
        
        # Evaluate current state
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Get correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Determine shift k for max overlap
        k = max_idx - (len(h) - 1)
        
        # Build regions contributing to max overlap
        # correlation is for shift k, positions from 0 to 2*dx step dx
        # for shift k, we're computing int h(x)*(1-h(x+k)) dx
        # for this shift, overlap occurs for x where both h(x) and h(x+k) are > 0
        # actual correlation for index i is int h(x)*h(x+k) dx where shift k is i - (n-1) 
        # shift k = i - (n-1)
        # if max_idx is at index i, shift k = i - (n-1)
        
        # Regions where both h[x] and h[x+k] are potentially non-zero
        # Convert correlation index back to actual shift k
        shift_k = max_idx - (len(h) - 1)
        
        # for shift k, we consider how h(x) contributes with h(x+k) with integral over x
        # in the convolution it's sum (h[i] * (1 - h[i+k]))
        # so we want to bring h[x] down where these terms occur
        
        if abs(shift_k) > len(h):
            break
            
        # Identify most influential parts for overlap
        # We look at all indices where h[i] * h[i+shift_k] contributes to max overlap
        # (overlap occurs when both values are significant)
        contrib_indices = []
        if shift_k == 0:
            # i: h[i](1-h[i]) contributes
            contrib_indices = list(range(len(h)))
        elif shift_k > 0:
            # h[i] * h[i-shift_k] contributes for i >= shift_k and i < len(h)
            contrib_indices = list(range(shift_k, len(h)))
        else:
            # h[i] * h[i-shift_k] contributes for i >= 0 and i < len(h) + shift_k
            contrib_indices = list(range(0, len(h) + shift_k))
            
        # For each i in contrib_indices, calculate h[i] * h[i+k] for overlap
        # But we want to look at actual contributor to C5
        # That's in the correlation: sum i [h[i] * (1-h[i+k])]
        # We will look at which i contribute most to all terms which are max_val
        
        # Find indices contributing most to the mismatch (max correlation)
        raw_contrib = []
        for i in contrib_indices:
            if shift_k >= 0:
                # h[i] contributes to overlap with h[i-shift_k]
                raw_contrib.append(h[i] * (1 - h[i - shift_k]))
            else:
                # h[i] contributes to overlap with h[i-shift_k]
                raw_contrib.append(h[i] * (1 - h[i - shift_k]))
        
        # Normalize and identify top contributors
        if len(raw_contrib) > 0:
            top_indices = np.argsort(raw_contrib)[-int(0.1 * len(raw_contrib)):]
            
            # Decrease h at these places
            adjustment_amount = min((c5 - target) / len(h), 0.1)
            for i in top_indices:
                if i < len(h) and shift_k >= 0:
                    if i - shift_k >= 0:
                        h[i] = max(0, h[i] - adjustment_amount)
                elif i < len(h) and shift_k < 0:
                    if i - shift_k < len(h):
                        h[i] = max(0, h[i] - adjustment_amount)
            h = project_to_feasible(h, n_points)
        else:
            h = project_to_feasible(h + np.random.normal(0, 0.005, n_points), n_points)
            
    # Ensure final solution has correct integral
    h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(h, dx)
    return (list(h), final_c5, n_points)
