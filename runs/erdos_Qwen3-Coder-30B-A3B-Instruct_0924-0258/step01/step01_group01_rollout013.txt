import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 100
    alpha = 0.01
    tol = 1e-6
    
    # Refinement
    resolutions = [10, 20, 40, 70]  # Max resolution based on memory concerns
    current_res = 0
    
    while current_res < len(resolutions) and time.time() - start_time < budget_s * 0.95:
        n_points = resolutions[current_res]
        dx = 2.0 / n_points
        
        if current_res > 0:
            # Interpolate
            h_long = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, n_points//2), h)
            h = h_long
        
        h = project_to_constraints(h, dx)
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            # Update h using gradient descent on the max correlation
            # We approximate by using a linearized version or direct search step
            
            h_prev = h.copy()
            # Simple gradient step on max correlation 
            h_grad = np.zeros_like(h)
            # Compute correlation of h with (1-h)
            corr = correlate(h, 1-h, mode='full')

            # We can compute an approximate gradient by perturbing h
            epsilon = 1e-4

            c5_plus = evaluate(h_plus, dx)
            c5_base = evaluate(h, dx)
            if abs(c5_plus - c5_base) > 1e-10:
                h_grad = (c5_plus - c5_base) / epsilon * (h_plus - h)
            else:
                # Fallback use central difference
                h_plus = h + epsilon
                h_plus = project_to_constraints(h_plus, dx)
                h_minus = h - epsilon
                h_minus = project_to_constraints(h_minus, dx)
                c5_plus = evaluate(h_plus, dx)
                c5_minus = evaluate(h_minus, dx)
                h_grad = (c5_plus - c5_minus) / (2*epsilon)
            
            h = h - alpha * h_grad
            h = project_to_constraints(h, dx)
            
            c5_new = evaluate(h, dx)
            if c5_new < best_c5:

                best_c5 = c5_new

    final_c5 = evaluate(best_h, dx)
