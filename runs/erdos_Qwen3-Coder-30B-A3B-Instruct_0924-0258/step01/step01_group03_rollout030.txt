import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to [0,1] and normalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Set dx
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize parameters
    learning_rate = 0.01
    patience = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent optimization
    for iter_num in range(100000):  # Prevent infinite loop
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        
        # Gradient computation
        gradient = np.zeros_like(h)
        n = len(h)
        
        # For shift k that maximizes correlation
        shift = max_k - (n - 1)
        for i in range(n):
            j = (i - shift) % n
            gradient[i] = 1 - 2 * h[j]
        
        # Update step
        h = h - learning_rate * gradient
        
        # Project to feasible region
        h = project_to_feasible(h, target_sum)
        
        # Evaluate
        current_c5 = evaluate(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
            if best_c5 < 0.3807:
                break
        else:
            patience += 1
            if patience > 50:
                # Add noise to escape local minima
                noise = np.random.uniform(-1e-3, 1e-3, n)
                h = h + noise
                h = project_to_feasible(h, target_sum)
                patience = 0
            if patience % 100 == 0 and patience > 0:
                learning_rate *= 0.9
        
        # Optionally reduce learning rate periodically
        if iter_num % 100 == 0 and iter_num > 0:
            learning_rate *= 0.95
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
