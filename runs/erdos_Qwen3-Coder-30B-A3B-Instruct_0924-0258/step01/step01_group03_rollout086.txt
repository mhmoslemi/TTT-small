import numpy as np
import time
from scipy.optimize import fmin
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Adjust to meet sum constraint
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values - find smallest elements
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] < 1:
                can_increase = 1 - h[j]
                increase = min(can_increase, diff)
                h[j] += increase
                diff -= increase
    else:
        # Need to decrease values - find largest elements
        idx = np.argsort(h)[::-1]
        for i in range(len(idx)):
            if diff >= 0:
                break
            j = idx[i]
            if h[j] > 0:
                can_decrease = h[j]
                decrease = max(-can_decrease, diff)
                h[j] += decrease
                diff -= decrease
    return h

    target_sum = n_points / 2.0
    
    # Project initial to feasible
    h0 = project_to_feasible(h0, target_sum)
    
    # Define objective function for optimization (minimize negative of C5)
    def objective(h_vec):
        h = project_to_feasible(h_vec, target_sum)
        return -evaluate(h, dx)
    
    # Initial simplex using the initial solution with small perturbations
    m = n_points + 2  # Size of simplex
    simplex = [h0.copy()]
    
    for i in range(1, m):
        pert = np.random.normal(0, 0.05, n_points)
        h_try = h0 + pert
        h_try = project_to_feasible(h_try, target_sum)
        simplex.append(h_try)
    
    # Nelder-Mead optimization
    best_h = None
    best_c5 = float('inf')
    
    # Main loop for time control
    max_iter = 10000
    iter_count = 0
    
    # Track the best value found
    best_value = float('inf')
    best_simplex = None
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Evaluate all simplex points
        values = []
        for h in simplex:
            values.append(objective(h))
        values = np.array(values)
        
        # Sort simplex by values (ascending)
        idxs = np.argsort(values)
        simplex = [simplex[i] for i in idxs]
        values = [values[i] for i in idxs]
        
        # Update best
        if values[0] < best_value:
            best_value = values[0]
            best_simplex = simplex[0].copy()
            
        # Stop if we have a good solution already
        if -best_value < 0.3807:
            break
            
        # Nelder-Mead operations
        xbar = np.mean(simplex[:-1], axis=0)
        xr = xbar + (xbar - simplex[-1])  # Reflection
        xr = project_to_feasible(xr, target_sum)
        
        xr_val = objective(xr)
        
        if values[0] <= xr_val < values[-2]:
            # Replace worst
            simplex[-1] = xr
        elif xr_val < values[0]:
            # Expansion
            xe = xbar + 2 * (xbar - simplex[-1])
            xe = project_to_feasible(xe, target_sum)
            xe_val = objective(xe)
            if xe_val < xr_val:
                simplex[-1] = xe
            else:
                simplex[-1] = xr
        else:
            # Contraction
            xc = xbar + 0.5 * (simplex[-1] - xbar)
            xc = project_to_feasible(xc, target_sum)
            xc_val = objective(xc)

                # Reduction
                for i in range(1, m):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])

        final_h = project_to_feasible(best_simplex, target_sum)
        final_c5 = evaluate(final_h, dx)

    else:
        # Fallback to original solution
        return (h0.tolist(), float(evaluate(h0, dx)), n_points)
