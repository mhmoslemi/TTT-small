import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_autocorr(h):
    """Compute autocorrelation A(k) = ∫ h(x)h(x + k) dx"""
    corr = np.correlate(h, h, mode='full')
    return corr

def optimize_step(h, n_points, dx, time_limit):
    """Optimize using gradient-based method with simulated annealing fallback"""
    start_time = time()
    
    # Objective function that penalizes small autocorrelations
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        # Add regularizer to make it more convex-like
        return max_overlap + 0.1 * np.sum((corr - np.mean(corr))**2)
    
    # Gradient computation via finite differences
    def grad(h):
        eps = 1e-6
        grad = np.zeros_like(h)
        base_obj = obj(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            obj_plus = obj(h_plus)
            grad[i] = (obj_plus - base_obj) / eps
        return grad

    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Try multiple restarts with different perturbations
    for restart in range(5):
        if time() - start_time > time_limit:
            break
            
        h_restart = project_to_feasible(h + np.random.normal(0, 0.05, n_points), n_points)
        
        try:
            res = minimize(obj, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons, jac=grad, 
                           options={'maxiter': 30, 'ftol': 1e-6})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
    
    # Simulated annealing as fallback
    if time() - start_time < time_limit * 0.8:
        T = 1.0
        cooling_rate = 0.95
        min_T = 1e-5
        iterations = 0
        
        while T > min_T and time() - start_time < time_limit:
            new_h = best_h.copy()
            # Make small random change
            i = np.random.randint(0, n_points)
            new_h[i] += np.random.normal(0, 0.02)
            new_h = project_to_feasible(new_h, n_points)
            
            old_c5 = evaluate_erdos_solution(best_h, dx)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            
            if new_c5 < old_c5 or np.random.rand() < np.exp((old_c5 - new_c5) / T):
                best_h = new_h
                best_c5 = new_c5
                
            T *= cooling_rate
            iterations += 1
    
    return best_h, best_c5

        h_start[5:10] = 0.8
        h_start[20:25] = 0.3
    
    # Use different resolutions
    common_resolutions = [20, 30, 40, 50]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Perform optimization
    for n_points in common_resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        if len(h_start) != n_points:
            # Interpolate
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Compute time budget for this resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(common_resolutions), 40)
        
        h_opt, c5_opt = optimize_step(h_start, n_points, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
        
        h_start = h_opt.copy()
    
    return (list(best_h), best_c5, best_n_points)
