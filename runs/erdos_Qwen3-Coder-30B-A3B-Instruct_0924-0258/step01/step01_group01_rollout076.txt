import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

    time_per_res = budget_s * 0.25  # 150s per resolution

    best_c5 = float('inf')

        # Particle swarm optimization with local descent
        particles = 100
        velocities = np.random.uniform(-0.05, 0.05, (particles, n_points))
        positions = np.random.uniform(0, 1, (particles, n_points))
        for i in range(particles):
            positions[i] = project_to_constraints(positions[i], dx)
        
        personal_best = positions.copy()
        personal_best_scores = [evaluate(p, dx) for p in positions]
        global_best_idx = np.argmin(personal_best_scores)
        global_best = personal_best[global_best_idx].copy()
        global_best_score = personal_best_scores[global_best_idx]
        
        # Inertia weight
        w = 0.7
        c1, c2 = 1.5, 1.5
        iter_count = 0
        res_start = time.time()
        
        while time.time() - start_time < budget_s and time.time() - res_start < time_per_res:
            iter_count += 1
            for i in range(particles):
                # Velocity update
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (personal_best[i] - positions[i]) + 
                                c2 * r2 * (global_best - positions[i]))
                velocities[i] = np.clip(velocities[i], -0.05, 0.05)
                positions[i] += velocities[i]
                positions[i] = np.clip(positions[i], 0, 1)
                positions[i] = project_to_constraints(positions[i], dx)
                
                # Update personal best
                score = evaluate(positions[i], dx)
                if score < personal_best_scores[i]:
                    personal_best_scores[i] = score
                    personal_best[i] = positions[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best = positions[i].copy()
                    
            # Local descent using gradient with small steps
            # This is a simple steepest descent approximation
            trial_h = global_best + np.random.normal(0, 0.001, n_points)
            trial_h = np.clip(trial_h, 0, 1)
            trial_h = project_to_constraints(trial_h, dx)
            trial_score = evaluate(trial_h, dx)
            if trial_score < global_best_score:
                global_best = trial_h
                global_best_score = trial_score
                
        # Final check of convergence
        h = global_best.copy()
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
    return best_h.tolist(), best_c5, len(best_h)
