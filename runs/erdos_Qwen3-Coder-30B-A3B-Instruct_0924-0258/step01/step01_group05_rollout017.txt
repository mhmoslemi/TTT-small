import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def grid_search_level(h_start, n_points, num_candidates=20, max_iter=3):
        best_h = h_start.copy()
        best_C5 = compute_C5(best_h)
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            candidates = []
            
            # Generate candidates for this level
            for _ in range(num_candidates):
                # Start from current best and slightly perturb
                h_candidate = best_h + np.random.normal(0, 0.05, n_points)
                h_candidate = project_feasible(h_candidate)
                candidates.append(h_candidate)
            
            # Evaluate all candidates
            for h_candidate in candidates:
                if time.time() - start_time > budget_s - 0.1:
                    break
                C5_candidate = compute_C5(h_candidate)
                if C5_candidate < best_C5:
                    best_C5 = C5_candidate
                    best_h = h_candidate.copy()
            
        return best_h, best_C5
    
    # Begin with initial solution
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Grid search with refinement
    grid_sizes = [5, 10, 15, 20, 25, 30, 45]
    
    # Start from successively larger grid sizes
    for n in grid_sizes:
        if n > n_points:
            continue
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Resample or keep solution
        if n < n_points:
            # Downsample
            h_resampled = np.array([np.mean(h_best[i::n_points//n]) for i in range(n_points//n)])
            h_resampled = project_feasible(h_resampled)
        else:
            # The best solution we have
            h_resampled = h_best.copy()
            
        # Local refinement at this resolution
        h_refined, C5_refined = grid_search_level(h_resampled, n, num_candidates=20, max_iter=2)
        
        if C5_refined < C5_best:
            C5_best = C5_refined
            h_best = h_refined.copy()
            
    # Do a final thorough local search
    for _ in range(5):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        h_perturbed = h_best + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_feasible(h_perturbed)
        
        # Simple gradient-like improvement step
        # Compute gradient estimate of C5 with respect to h
        corr = correlate(h_perturbed, 1 - h_perturbed, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_perturbed) - 1)
        
        grad = np.zeros_like(h_perturbed)
        if max_lag >= 0:  # Positive lag
            grad[max_lag:] = -h_perturbed[:-max_lag]
        else:  # Negative lag
            grad[:max_lag] = -h_perturbed[-max_lag:]
            
        # Try small step
        h_new = h_perturbed - 0.05 * grad
        h_new = project_feasible(h_new)
        C5_new = compute_C5(h_new)
        
        if C5_new < C5_best:
            C5_best = C5_new
            h_best = h_new.copy()
            
    # Final projection and evaluation
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
