import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h_vec, dx):
    """Compute C5 for given h vector."""
    h_vec = np.array(h_vec)
    corr = correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(corr) * dx

def project_feasible(h_vec, target_sum):
    """Project h_vec onto feasible set: 0 <= h <= 1, sum(h) = target_sum."""
    h_vec = np.clip(h_vec, 0, 1)
    h_sum = np.sum(h_vec)
    if h_sum > 0:
        h_vec = h_vec * target_sum / h_sum
    return h_vec

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Grid search approach
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h, dx)
    
    # Grid levels to try
    grid_sizes = [5, 10, 15, 20, 25, 30, 45]
    
    for level, n in enumerate(grid_sizes):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Only continue if we're increasing resolution or at final resolution
        if n <= n_points:
            continue
            
        # Generate candidates at this resolution
        candidates = []
        num_candidates = 20 if n > 30 else 50
        
        for _ in range(num_candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Generate random feasible h vector
            h_candidate = np.random.rand(n)
            h_candidate = project_feasible(h_candidate, target_sum)
            candidates.append(h_candidate)
            
        # Evaluate candidates
        for candidate in candidates:
            if time.time() - start_time > budget_s - 0.1:
                break
            C5 = evaluate_C5(candidate, 2.0/n)
            if C5 < best_C5:
                best_C5 = C5
                # Interpolate to higher resolution if needed
                if n < n_points:
                    # Resample to higher resolution
                    ratio = n_points / n
                    h_resampled = np.repeat(candidate, int(ratio))
                    h_resampled = h_resampled[:n_points]
                    # Project to ensure constraint
                    h_resampled = project_feasible(h_resampled, target_sum)
                    best_h = h_resampled.copy()
                else:
                    best_h = candidate.copy()

    # Local refinement around best solution
    h_current = best_h.copy()
    C5_current = best_C5
    
    # Try different perturbations
    for iter in range(100):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Create perturbed solution
        h_perturbed = h_current + np.random.normal(0, 0.02, len(h_current))
        h_perturbed = project_feasible(h_perturbed, target_sum)
        
        C5_new = evaluate_C5(h_perturbed, dx)
        
        if C5_new < C5_current:
            h_current = h_perturbed.copy()
            C5_current = C5_new
            
        # Occasionally jump to a completely new candidate if stuck
        if iter % 20 == 0 and np.random.random() < 0.3:
            h_perturbed = np.random.rand(n_points)
            h_perturbed = project_feasible(h_perturbed, target_sum)
            C5_new = evaluate_C5(h_perturbed, dx)
            if C5_new < C5_current:
                h_current = h_perturbed.copy()
                C5_current = C5_new

    # Final evaluation and ensure feasibility
    final_h = project_feasible(h_current, target_sum)
    final_C5 = evaluate_C5(final_h, dx)
    
    return list(final_h), final_C5, n_points
