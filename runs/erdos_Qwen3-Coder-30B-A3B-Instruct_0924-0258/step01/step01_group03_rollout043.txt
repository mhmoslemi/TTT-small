import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to maintain sum constraint and [0,1] bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to exact sum
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    h = project_h(h, target_sum)
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Hyperparameters
    learning_rate = 0.01
    decay_rate = 0.9
    patience = 50
    patience_counter = 0
    
    # Gradient descent loop
    iteration = 0
    while time.time() - start_time < budget_s:
        # Compute correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient for the dominating shift
        gradient = np.zeros_like(h)
        n_corr = len(corr)
        # The cross-correlation length is 2*n_points - 1
        # k = 0 also corresponds to shift 0
        # For shift k_max, the overlap involves
        # elements h[i] * (1 - h[(i+k_max) % n_points])
        # But we need to go back and avoid wrapping issues
        
        # k_max can be interpreted differently for different indices in 'full' mode
        # So we need to interpret from the reverse
        # In 'full' mode, result is 2n-1 elements
        # corr[0] corresponds to shift = -n+1
        # corr[n-1] corresponds to shift = 0
        # corr[2*n-2] corresponds to shift = n-1
        
        shift = k_max - (n_points - 1)  # Convert index to actual shift
        
        # We compute gradient for the shift that gives the maximum overlap
        for i in range(n_points):
            if shift >= 0:
                j = (i + shift) % n_points
            else:
                j = (i - (-shift)) % n_points
            gradient[i] += (1 - h[j])
        
        # Update h
        h = h - learning_rate * gradient
        h = project_h(h, target_sum)
        
        # Evaluate current
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        if patience_counter >= patience:
            # Add perturbation if stuck
            noise = np.random.uniform(-1e-3, 1e-3, n_points)
            h = h + noise
            h = project_h(h, target_sum)
            patience_counter = 0
            learning_rate *= decay_rate
        
        iteration += 1
        if iteration % 100 == 0:
            learning_rate *= decay_rate
    
    return (best_h.tolist(), float(best_c5), n_points)
