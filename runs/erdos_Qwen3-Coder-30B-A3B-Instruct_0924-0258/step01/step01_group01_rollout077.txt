import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    resolutions = [10, 20, 40, 70]
    n_points = resolutions[0]
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Parameters
    particle_count = 100
    max_iter = 500
    inertia = 0.7
    c1 = 1.5
    c2 = 1.5
    velocity_limit = 0.05
    
    for res_idx, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        if res_idx > 0:
            # Interpolate from previous resolution
            h = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, len(h)), h)
        
        h = project_to_constraints(h, dx)
        best_in_res = evaluate(h, dx)
        best_h_in_res = h.copy()
        
        # Particle Swarm Optimization
        particles = np.random.uniform(0, 1, (particle_count, n_points))
        particles = np.array([project_to_constraints(p, dx) for p in particles])
        velocities = np.random.uniform(-velocity_limit, velocity_limit, (particle_count, n_points))
        pbest = particles.copy()
        pbest_scores = np.array([evaluate(p, dx) for p in particles])
        gbest_idx = np.argmin(pbest_scores)
        gbest = pbest[gbest_idx].copy()
        gbest_score = pbest_scores[gbest_idx]
        
        iter_count = 0
        while iter_count < max_iter:
            if time.time() - start_time > budget_s:
                break
                
            for i in range(particle_count):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (inertia * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = project_to_constraints(particles[i], dx)
                
                # Update personal best
                score = evaluate(particles[i], dx)
                if score < pbest_scores[i]:
                    pbest[i] = particles[i].copy()
                    pbest_scores[i] = score
                    if score < gbest_score:
                        gbest = particles[i].copy()
                        gbest_score = score
                        
            # Update global best in this resolution
            if gbest_score < best_in_res:
                best_in_res = gbest_score
                best_h_in_res = gbest.copy()
            
            iter_count += 1
            
        # Apply gradient descent to refine
        h = best_h_in_res.copy()
        step_size = 0.01
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            grad = np.zeros(n_points)
            for i in range(n_points):
                grad[i] = (1 - 2 * h[i]) * dx
            h -= step_size * grad
            h = np.clip(h, 0, 1)
            h = project_to_constraints(h, dx)
            score = evaluate(h, dx)
            if score < best_in_res:
                best_in_res = score
                best_h_in_res = h.copy()
                
        h = best_h_in_res.copy()
        if best_in_res < best_c5:
            best_c5 = best_in_res
            best_h = h.copy()
    
    return best_h.tolist(), best_c5, len(best_h)
