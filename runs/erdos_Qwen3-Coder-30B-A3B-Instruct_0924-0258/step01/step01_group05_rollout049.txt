import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, max_lag):
        # Compute gradient for the maximum lag case
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        if max_lag >= 0:  # Positive lag
            # For each position i, the contribution to correlation with lag k
            # depends on h[i] and (1 - h[i+k])
            for i in range(n):
                if i + max_lag < n:
                    grad[i] = -(1 - h_vec[i + max_lag])
                if i - max_lag >= 0:
                    grad[i] -= h_vec[i - max_lag]
        else:  # Negative lag
            max_lag = -max_lag
            for i in range(n):
                if i - max_lag >= 0:
                    grad[i] = -(1 - h_vec[i - max_lag])
                if i + max_lag < n:
                    grad[i] -= h_vec[i + max_lag]
        return grad
    
    def optimize_step(h_vec, alpha=0.1, max_iter=20):
        h_current = h_vec.copy()
        best_C5 = compute_C5(h_current)
        no_improvement_count = 0
        
        # Get current best overlap
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Compute gradient
            grad = compute_gradient(h_current, max_lag)
            
            # Try different step sizes
            alpha_candidates = [0.1, 0.05, 0.01]
            best_alpha = 0.0
            best_new_h = h_current.copy()
            best_new_C5 = best_C5
            
            for alpha_cand in alpha_candidates:
                candidate_h = h_current - alpha_cand * grad
                candidate_h = project_feasible(candidate_h)
                new_C5 = compute_C5(candidate_h)
                if new_C5 < best_new_C5:
                    best_new_C5 = new_C5
                    best_new_h = candidate_h.copy()
                    best_alpha = alpha_cand
            
            # If not improved, maybe perturb
            if best_new_C5 >= best_C5:
                no_improvement_count += 1
                if no_improvement_count > 20:
                    # Add noise and re-normalize
                    noise = np.random.normal(0, 0.05, n_points)
                    h_current += noise
                    h_current = project_feasible(h_current)
                    no_improvement_count = 0
                    continue
            else:
                h_current = best_new_h.copy()
                best_C5 = best_new_C5
                no_improvement_count = 0
                
            # Recompute lag
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
        return h_current, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Local optimization loop
    iterations = 0
    max_iterations = 1000
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 0.1 and iterations < max_iterations:
        # Try several random perturbations
        for _ in range(10):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Perturb current point
            h_perturbed = h_best + np.random.normal(0, 0.02, n_points)
            h_perturbed = project_feasible(h_perturbed)
            
            # Optimize from this point
            h_opt, C5_opt = optimize_step(h_perturbed, alpha=0.1, max_iter=30)
            if C5_opt < C5_best:
                C5_best = C5_opt
                h_best = h_opt.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
        iterations += 1
        if no_improvement > 50:
            break
    
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
