import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Use np.correlate for full convolution
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # GA parameters
    pop_size = 50
    tournament_size = 10
    keep_parents = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    max_generations = 1000
    patience = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        individual = np.random.rand(n_points)
        individual = project_feasible(individual)
        population.append(individual)
    
    best_C5 = float('inf')
    best_individual = None
    
    generation = 0
    no_improvement = 0
    
    while generation < max_generations and time.time() - start_time < budget_s - 0.1:
        # Evaluate fitness of all individuals
        fitness = [compute_C5(ind) for ind in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        current_best_C5 = fitness[current_best_idx]
        if current_best_C5 < best_C5:
            best_C5 = current_best_C5
            best_individual = population[current_best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Check for early stopping
        if no_improvement > patience:
            # Reset some individuals to avoid local minimum
            for i in range(pop_size // 4):
                if time.time() - start_time > budget_s - 0.1:
                    break
                idx = np.random.randint(0, pop_size)
                population[idx] = np.random.rand(n_points)
                population[idx] = project_feasible(population[idx])
            no_improvement = 0
        
        # Selection with tournament
        selected = []
        for _ in range(pop_size):
            if time.time() - start_time > budget_s - 0.1:
                break
            tournament = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament]
            winner_idx = tournament[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population
        new_population = selected[:keep_parents]  # Elitism
        
        while len(new_population) < pop_size:
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Selection for crossover
            parent1_idx = np.random.randint(0, pop_size)
            parent2_idx = np.random.randint(0, pop_size)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point1 = np.random.randint(1, n_points - 1)
                crossover_point2 = np.random.randint(crossover_point1, n_points)
                
                child1 = np.concatenate([
                    selected[parent1_idx][:crossover_point1],
                    selected[parent2_idx][crossover_point1:crossover_point2],
                    selected[parent1_idx][crossover_point2:]
                ])
                child2 = np.concatenate([
                    selected[parent2_idx][:crossover_point1],
                    selected[parent1_idx][crossover_point1:crossover_point2],
                    selected[parent2_idx][crossover_point2:]
                ])
            else:
                child1 = selected[parent1_idx].copy()
                child2 = selected[parent2_idx].copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Project feasible
            child1 = project_feasible(child1)
            child2 = project_feasible(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population[:pop_size]
        generation += 1
    
    # Final evaluation
    if best_individual is None:
        best_individual = population[0]
    final_C5 = compute_C5(best_individual)
    
    return list(best_individual), final_C5, n_points
