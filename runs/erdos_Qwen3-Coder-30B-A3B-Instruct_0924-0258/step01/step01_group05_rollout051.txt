import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def evaluate_candidate(h_vec):
        h_vec = project_feasible(h_vec)
        return compute_C5(h_vec), h_vec.copy()
    
    # Initialize several candidates using Latin hypercube-like sampling
    num_candidates = 5
    candidates = []
    for _ in range(num_candidates):
        # Sample from a distribution that explores the space
        h_sample = np.random.rand(n_points)
        h_sample = project_feasible(h_sample)
        C5_val, h_opt = evaluate_candidate(h_sample)
        candidates.append((C5_val, h_opt))
    
    # Sort by C5 value
    candidates.sort(key=lambda x: x[0])
    h_best = candidates[0][1].copy()
    C5_best = candidates[0][0]
    
    # Local optimization with different strategies
    iteration = 0
    no_improve_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        
        # Try coordinate descent on current best
        improved = False
        for i in range(n_points):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Try small step in both directions
            alpha = 0.05
            h_trial = h_best.copy()
            h_trial[i] = min(1.0, max(0.0, h_trial[i] + alpha))
            C5_new, _ = evaluate_candidate(h_trial)
            if C5_new < C5_best:
                h_best = h_trial.copy()
                C5_best = C5_new
                improved = True
                continue
                
            h_trial = h_best.copy()
            h_trial[i] = min(1.0, max(0.0, h_trial[i] - alpha))
            C5_new, _ = evaluate_candidate(h_trial)
            if C5_new < C5_best:
                h_best = h_trial.copy()
                C5_best = C5_new
                improved = True
        
        # Swap random pairs
        for _ in range(10):
            if time.time() - start_time > budget_s - 0.1:
                break
            i, j = np.random.randint(0, n_points, 2)
            h_trial = h_best.copy()
            h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
            C5_new, _ = evaluate_candidate(h_trial)
            if C5_new < C5_best:
                h_best = h_trial.copy()
                C5_best = C5_new
                improved = True
        
        # Random perturbation step
        if not improved:
            no_improve_count += 1
            if no_improve_count > 10:
                # Add noise to encourage exploration
                noise = np.random.normal(0, 0.02, n_points)
                h_best += noise
                h_best = project_feasible(h_best)
                C5_best = compute_C5(h_best)
                no_improve_count = 0
        else:
            no_improve_count = 0
            
        # Generate new random candidate occasionally
        if iteration % 100 == 0:
            h_new = np.random.rand(n_points)
            h_new = project_feasible(h_new)
            C5_new, _ = evaluate_candidate(h_new)
            if C5_new < C5_best:
                h_best = h_new.copy()
                C5_best = C5_new
    
    # Final check
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
