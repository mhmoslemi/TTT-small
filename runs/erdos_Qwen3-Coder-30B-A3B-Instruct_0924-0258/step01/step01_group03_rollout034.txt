import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.full_like(h, target_sum / len(h))
    else:
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

    # Normalize to sum = n_points / 2
    h = project_h(h, n_points / 2)
    
    dx = 2.0 / n_points
    
    # Parameters
    learning_rate = 0.01
    max_iter = 2000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        k_opt = np.argmax(corr)
        
        # Compute gradient
        grad = np.zeros_like(h)
        n = len(h)
        # For shift k_opt, compute gradient
        k_shift = k_opt - (n - 1)  # shift in range [-n+1, n-1]
        for j in range(n):
            j_shifted = (j - k_shift) % n
            grad[j] = 1 - 2 * h[j_shifted]
        
        # Update h
        h = h - learning_rate * grad
        
        # Project to feasible set
        h = project_h(h, n_points / 2)
        
        # Check for improvement
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 50  # reset patience
        else:
            patience -= 1
            if patience <= 0:
                # Add small random noise to escape local minima
                noise = np.random.uniform(-1e-3, 1e-3, n_points)
                h = h + noise
                h = project_h(h, n_points / 2)
                patience = 50
                learning_rate *= 0.95  # decay learning rate
        
        # Periodic sanity check
        if i % 100 == 0:
            if np.any(np.isnan(h)) or np.any(np.isinf(h)):
                h = project_h(np.random.random(n_points) * 0.5, n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
