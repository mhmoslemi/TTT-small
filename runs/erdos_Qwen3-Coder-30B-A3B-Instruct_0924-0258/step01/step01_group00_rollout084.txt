import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def improve_step_function(h, dx, n_points):
    """Improve step function using gradient-based local optimization"""
    # Create a smooth version of h to make it differentiable
    # Use a small quadratic penalty to encourage low overlap
    def penalty(h):
        # This computes sum of squared correlations to encourage small overlaps
        corr = np.correlate(h, 1-h, mode='full') 
        return np.sum(corr**2) * dx**2
    
    def grad_penalty(h):
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros_like(h)
        base_obj = penalty(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            obj_plus = penalty(h_plus)
            grad[i] = (obj_plus - base_obj) / eps
        return grad
    
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(penalty, h, method='SLSQP', 
                       bounds=bounds, constraints=cons, jac=grad_penalty,
                       options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            h = project_to_feasible(res.x, n_points)
    except:
        pass
    
    return h

def pattern_perturb(h, n_points, strength=0.02):
    """Perturb with structured pattern to escape local minima"""
    pert = np.random.normal(0, strength, n_points)
    # Add some cosine components to encourage exploration
    x = np.linspace(0, 2, n_points)
    pert += 0.5 * strength * np.sin(2 * np.pi * x) 
    h_new = h + pert
    return project_to_feasible(h_new, n_points)

    # Try different resolutions
    resolutions = [15, 20, 25, 30, 35, 40, 45, 50] 
    current_h = h_start.copy()
    current_n = len(current_h)
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate if needed
        if n_points != current_n:
            x_orig = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_orig, current_h)
            current_h = project_to_feasible(h_interp, n_points)
            current_n = n_points
        
        dx = 2.0 / current_n
        best_h_level = current_h.copy()
        best_c5_level = evaluate_erdos_solution(current_h, dx)
        
        # Multiple restarts
        for restart in range(3):
            if time() - start_time > budget_s * 0.9:
                break
            
            # Perturb
            h_perturbed = pattern_perturb(current_h, current_n, 0.01 + 0.01 * restart)
            
            # Local optimization
            h_opt = improve_step_function(h_perturbed, dx, current_n)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            
            if c5_opt < best_c5_level:
                best_c5_level = c5_opt
                best_h_level = h_opt.copy()
        
        # Check if we improved the global best
        if best_c5_level < best_c5:
            best_c5 = best_c5_level
            best_h = best_h_level.copy()

        best_n_points = len(best_h)
        
    return (list(best_h), best_c5, best_n_points)
