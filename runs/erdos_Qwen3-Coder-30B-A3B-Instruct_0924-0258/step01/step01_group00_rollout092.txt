import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def adaptive_convex_relaxation(h, n_points, dx, num_shifts=20):
    """Use CVXPY to solve a convex relaxation of the problem"""
    # Create variables for the correlation values
    correlations = cp.Variable(num_shifts)
    
    # Build matrix of h values for correlation calculations
    h_extended = np.pad(h, (0, n_points-1), mode='constant')
    
    # Compute shifted correlations explicitly
    shifts = np.linspace(0, 2, num_shifts, endpoint=False)
    # Use all correlations to build constraints
    corr_vals = []
    for k in range(num_shifts):
        # Compute correlation for shift k
        shift = int(round(k * n_points / 2))
        corr_val = np.sum(h * h_extended[shift:shift+n_points])
        corr_vals.append(corr_val)
    
    # Minimize the most negative correlation (which is the maximum of 1- correlation)
    # We proxy this by maximizing sum of correlations
    objective = cp.Minimize(cp.sum(correlations))
    
    # Set up constraints
    constraints = [
        correlations[i] <= corr_vals[i] for i in range(num_shifts)
    ]
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return -np.sum(correlations.value) * dx
    except:
        pass
    return np.max([1 - c for c in corr_vals]) * dx

def genetic_mutation(h, mutation_rate=0.1):
    """Apply genetic mutations to h including crossover and polynomial adjustment"""
    h_mut = h.copy()
    mask = np.random.random(len(h)) < mutation_rate
    h_mut[mask] = np.random.random(np.sum(mask))
    
    # Add small polynomial perturbation
    x = np.linspace(0, 2, len(h))
    poly = np.poly1d(np.random.randn(3))
    h_mut += 0.01 * poly(x)
    
    return np.clip(h_mut, 0, 1)
