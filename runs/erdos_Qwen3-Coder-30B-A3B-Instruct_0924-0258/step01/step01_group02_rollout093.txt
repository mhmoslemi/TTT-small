import numpy as np

def run(seed=42, budget_s=600, **kwargs):

    # Initialize from the provided initial construction

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Binary search bounds
    low = 0.38
    high = compute_c5(h)
    
    start_time = time.time()
    
    # Try to improve using gradient-free optimization
    # We'll use a modified Nelder-Mead approach with smarter initialization
    
    def optimization_step(h_vec, max_iter):
        # Simple local search: for each h[i], try perturbations
        best_h = h_vec.copy()
        best_c5 = compute_c5(best_h)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Evaluate all local changes
            best_improvement = 0
            best_idx = -1
            best_delta = 0
            
            for i in range(len(h_vec)):
                if h_vec[i] <= 0.99:  # Check if we can increase
                    pert = 0.001
                    h_test = h_vec.copy()
                    h_test[i] += pert
                    h_test = project_h(h_test)
                    c5_new = compute_c5(h_test)
                    if c5_new < best_c5 - 1e-8:
                        improvement = best_c5 - c5_new
                        if improvement > best_improvement:
                            best_improvement = improvement
                            best_idx = i
                            best_delta = pert
            
            if best_idx >= 0:
                h_vec[best_idx] += best_delta
                h_vec = project_h(h_vec)
                new_c5 = compute_c5(h_vec)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_vec.copy()
                else:
                    break
            else:
                break
        
        return best_h, best_c5
    
    # Initial optimization
    current_h, current_c5 = optimization_step(h, 100)
    
    # Binary search for better bound
    while high - low > 1e-5:
        if time.time() - start_time > budget_s - 1:
            break
            
        mid = (low + high) / 2
        feasibility_flag = True
        
        # Try to construct a feasible solution that achieves mid
        # We use a relaxation approach here to find a feasible solution
        
        # Create approximate solution using projection
        h_feasible = current_h.copy()
        h_feasible = project_h(h_feasible)
        
        # Try cap at the desired bound
        # This is more effective than trying to solve a convex program 
        # for each iteration as it's too computationally intensive
        trial_h = h_feasible.copy()
        c5_trial = compute_c5(trial_h)
        
        # Simple heuristic: project to tighten the constraint
        # This might still be too aggressive, do temporary descriptive test
        if c5_trial <= mid:
            high = mid
        else:
            low = mid
            
        # Occasionally refine best solution if still within bounds
        if c5_trial < current_c5:
            current_c5 = c5_trial
            current_h = trial_h.copy()
            
        # Occasionally adjust and refine with local search
        if time.time() - start_time > 0.5 * budget_s:
            current_h, current_c5 = optimization_step(current_h, 50)
    
    return (current_h.tolist(), current_c5, n_points)
