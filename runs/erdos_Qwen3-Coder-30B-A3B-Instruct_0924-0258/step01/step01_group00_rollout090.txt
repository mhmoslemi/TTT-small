import numpy as np
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def conv_expansion(h, n_points, dx):
    """Convex expansion of the maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    return np.sum(corr**2) * dx**2 + 0.01 * (max_val - 0.3)**2

def conv_grad(h, n_points, dx):
    """Numerical gradient of the convex expansion"""
    eps = 1e-8
    grad = np.zeros_like(h)
    base_obj = conv_expansion(h, n_points, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        obj_plus = conv_expansion(h_plus, n_points, dx)
        grad[i] = (obj_plus - base_obj) / eps
    return grad

def optimize_resolution(h_start, n_points, dx, time_limit):
    """Optimize a resolution level with time constraint"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

        return conv_expansion(h, n_points, dx)

        return conv_grad(h, n_points, dx)
    
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    for i in range(300):  # Limit iterations
        if time() - start_time > time_limit:
            break
        
        # Perturb for randomness
        h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)
        
        try:
            res = minimize(obj, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons, jac=grad, 
                           options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
        
        # Update starting point
        h_start = best_h.copy()
    
    return best_h, best_c5

def geometric_progression(start, ratios):
    """Generate geometric progression."""
    res = [start]
    for r in ratios:
        res.append(res[-1] * r)
    return res
