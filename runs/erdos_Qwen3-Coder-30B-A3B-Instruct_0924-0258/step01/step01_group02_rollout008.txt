import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.full(45, 1.0 / 45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial h satisfies the constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def fitness(h_vec):
        return -compute_c5(h_vec)
    
    # Genetic Algorithm parameters
    population_size = 50
    num_generations = 1000
    mutation_rate = 0.05
    elite_size = population_size // 4
    
    # Initialize population
    population = []
    population.append(initial_h)
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness for all individuals
        fitness_scores = [fitness(h) for h in population]
        
        # Track best solution
        current_best_idx = np.argmax(fitness_scores)
        current_best = population[current_best_idx]
        current_c5 = -fitness_scores[current_best_idx]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Create new population
        new_population = population[:elite_size]
        
        # Tournament selection plus crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            
            parent1 = population[winner_idx]
            
            # Select second parent
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            
            parent2 = population[winner_idx]
            
            # Crossover (uniform)
            child = np.where(np.random.rand(n_points) > 0.5, parent1, parent2)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = project_h(child)
            
            new_population.append(child)
        
        population = new_population
        generation += 1
        
        # Early stopping if no improvement for many generations
        if generation > 1000:
            break
    
    return (best_h.tolist(), best_c5, n_points)
