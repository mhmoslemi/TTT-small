import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    max_iter_per_res = 1000
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Gradient descent step
        corr = correlate(h, 1-h, mode='full')
        max_idx = np.argmax(corr)
        # The correlation is of length 2*n-1, so the actual offset index is max_idx - n + 1
        k_offset = max_idx - n_points + 1
        # The gradient of correlation w.r.t. h at lag k can be computed by shifting
        # and reflecting appropriately. Use the convolution theorem to compute this
        # directly: gradient at location i w.r.t. h[i] comes from (1-h) shifted by k
        
        # Create shifted versions to compute gradient for the k_offset
        h_reflected = 1 - h
        if k_offset >= 0:
            # h_reflected[k_offset:] contributes to gradient of h[:n_points-k_offset]
            grad = np.concatenate([h_reflected[k_offset:], np.zeros(k_offset)])
        else:
            # h_reflected[:k_offset] (which is actually [k_offset:]) contributes to gradient of h[-k_offset:]
            grad = np.concatenate([np.zeros(-k_offset), h_reflected[:n_points+k_offset]])
            
        # Further, we are only interested in dC5/dh_i = max_k d/dh_i [int h(x) (1-h(x+k)) dx] 
        # = max_k [1 - h(x+k) if x = i, 0 otherwise]  = 1-h(i+k)  
        # This is exactly what correlate gives us (convolution of h with (1-h))
        
        # Let's compute gradient properly for saddle point descent
        # For cross-correlation, d/dh_i [h(x) * (1-h(x+k))] = 
        # 1_{x=i} * (1-h(i+k)) - (1-h(x-k)) * 1_{x=i}  
        # So, if we know the max_k, we can compute dC5/dh_i to descend faster
        
        # Better approach: using steepest descent on C5
        # dC5/dh_i = d/dh_i integral of h(x)(1-h(x+k)) for all k:
        # = \sum_k (1-h(x+k)) * delta_{i,x} - h(x-k) * delta_{i,x-k} (which is better)
        # = \sum_k (1-h(x+k)) * delta_{i,x} where x = i, and [h(x-k)*delta_{i,x-k}]
        # = those k where x = i so (1-h(i+k)) and k where i = x+k so (h(i-k))
        # This explains the convolution. We can directly do once the max is known
        
        # Option: use joint gradient descent on all h values
        # Compute moment-based gradient directly:
        # gradient matrix: for (i, k) where i+k=j, it's (1-h[j]) if i<0, etc.
        # For now, simple scaling
        eta = 0.01 * (1.0 + 0.05 * np.random.rand())
        h = h - eta * grad
        h = np.clip(h, 0, 1)
        h = project_to_constraints(h, dx)
        
        iter_count += 1
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Adapt resolution
        if n_points < n_high and iter_count % (max_iter_per_res // 2) == 0:
            if iter_count >= max_iter_per_res:
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), h)
                h = h_long
                n_points = 2 * n_points
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                iter_count = 0
        if time.time() - start_time > budget_s:
            break
            
    return best_h.tolist(), best_c5, len(best_h)
