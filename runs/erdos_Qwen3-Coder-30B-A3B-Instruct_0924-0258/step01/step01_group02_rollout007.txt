import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.full(45, 1.0/45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Genetic algorithm params
    population_size = 50
    generations = 1000
    mutation_rate = 0.05
    crossover_rate = 0.5
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    best_c5 = compute_c5(initial_h)
    best_h = initial_h.copy()
    
    start_time = time.time()
    last_improvement = 0
    gen = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(-c5)  # Negative because we maximize fitness (minimize C5)
        
        # Tournament selection
        new_population = []
        indices = list(range(len(population)))
        np.random.shuffle(indices)
        
        for i in range(0, len(indices), 2):
            if i+1 >= len(indices):
                break
            idx1, idx2 = indices[i], indices[i+1]
            if fitness[idx1] > fitness[idx2]:
                new_population.append(population[idx1])
            else:
                new_population.append(population[idx2])
        
        # Crossover
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                idx1, idx2 = np.random.choice(len(new_population), 2, replace=False)
                h1, h2 = new_population[idx1], new_population[idx2]
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
                
                child1 = project_h(child1)
                child2 = project_h(child2)
                
                new_population.extend([child1, child2])
            else:
                # Just copy one
                idx = np.random.randint(len(new_population))
                new_population.append(new_population[idx])
        
        # Trim to population size
        new_population = new_population[:population_size]
        
        # Mutation
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                h = new_population[i]
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = project_h(h)
                new_population[i] = h
        
        # Update population
        population = new_population
        
        # Check for new best
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = gen
        
        gen += 1
        
        # Early termination
        if gen - last_improvement > 1000:
            break
    
    return (best_h.tolist(), best_c5, n_points)
