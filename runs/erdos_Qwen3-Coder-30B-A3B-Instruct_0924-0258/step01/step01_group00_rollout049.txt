import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute the overlap integral using convolution
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def greedy_improve(h, n_points, dx, target=0.38):
    """Greedy adjustment to reduce maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    max_idx = np.argmax(corr)
    
    # Find the shift corresponding to the maximum overlap
    shift = max_idx - (n_points - 1)
    
    # Efficient adjustment: identify top contributing indices
    indices = np.argsort(corr)[-int(0.1 * len(corr)):]
    
    # Create adjustment mask
    adjustment = np.zeros_like(h)
    for i in indices:
        if i < n_points:
            adjustment[i] += 1.0
        if i >= n_points + shift and i < 2 * n_points + shift - 1:
            adjustment[i - n_points - shift] += 1.0
    
    # Apply adjustment proportional to deficit from target
    adjustment_force = (max_val * dx - target) / n_points
    adjustment *= adjustment_force
    
    # Apply adjustment and project to feasible set
    h_new = h - adjustment
    h_new = project_to_feasible(h_new, n_points)
    
    return h_new
