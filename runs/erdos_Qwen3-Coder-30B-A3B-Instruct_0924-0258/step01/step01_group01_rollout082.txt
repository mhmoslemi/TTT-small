import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(a, beta=10.0):
    # Smooth approximation of max using log-sum-exp
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * a)))

def compute_correlations(h):
    # Returns all correlations for different lags
    corr = correlate(h, 1-h, mode='full')
    return corr

def compute_gradient(h, corr_full, dx):
    # Analytical gradient computation
    n = len(h)
    grad = np.zeros_like(h)
    
    # For each position i, compute gradient contribution
    # Contribution from lag k is: d/dh_i [corr_k] where corr_k is correlation at lag k
    # corr_k = sum_{j} h[j] * (1 - h[j+k]) 
    # For fixed k:
    # d/dh_i (h[j] * (1 - h[j+k])) = 0 if i != j and i != (j+k)
    #                    = (1 - h[j+k]) if i == j
    #                    = -h[j] if i == (j+k)
    
    # Efficient computation for each h[i]
    # ignore the 'full' style correlate indexes for now
    # Bias to use internal code analysis directly using full array
    # Let's make a wrapper that computes the gradient of corr wrt h
    # Simplification: use finite differences to compute the gradient
    eps = 1e-5
    h_plus = h.copy()
    h_minus = h.copy()
    
    # The gradient of max over correlations
    # This is complex. We'll compute numerically
    grad = np.zeros_like(h)
    for i in range(n):
        h_plus[i] += eps
        h_minus[i] -= eps
        corr_plus = compute_correlations(h_plus)
        corr_minus = compute_correlations(h_minus)
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2*eps)
        h_plus[i] = h[i]
        h_minus[i] = h[i]
    
    return grad

def backtracking_armijo(learning_rate, grad, h, dx):
    # Simple backtracking line search with Armijo condition
    c1 = 1e-4
    rho = 0.5
    alpha = learning_rate
    f_old = evaluate(h, dx)
    
    for _ in range(20):  # maximum backtracking steps
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        f_new = evaluate(h_new, dx)
        if f_new <= f_old - c1 * alpha * np.dot(grad, grad):
            return alpha
        alpha *= rho
    return alpha

    best_c5 = evaluate(h, dx)

    for iter_id in range(max_iter):
