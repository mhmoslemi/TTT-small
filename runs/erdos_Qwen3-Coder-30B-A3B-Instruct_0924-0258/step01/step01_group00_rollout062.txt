import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction as starting point if available

    else:
        h_start = np.ones(30) * 0.5
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Try different resolutions
    resolutions = [20, 30, 40, 50]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if needed
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Use a more direct approach with multiple restarts
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)
        
        # Multiple restarts with different random perturbations
        for restart in range(5):
            if time() - start_time > budget_s * 0.98:
                break
                
            # Start with perturbed version
            h_perturbed = project_to_feasible(
                h_start + np.random.normal(0, 0.03, n_points), n_points)
            
            # Optimize using scipy minimize with L-BFGS-B (more efficient for large-scale)
            def obj(h):
                h = project_to_feasible(h, n_points)
                return evaluate_erdos_solution(h, dx)
                
            def grad(h):
                # Numerical gradient for approximating gradient of overlap integral
                eps = 1e-6
                grad_vals = np.zeros_like(h)
                base_obj = obj(h)
                h = project_to_feasible(h, n_points)
                
                for i in range(len(h)):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    obj_plus = evaluate_erdos_solution(h_plus, dx)
                    grad_vals[i] = (obj_plus - base_obj) / eps
                return grad_vals
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturbed, method='L-BFGS-B',
                               bounds=bounds, jac=grad,
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_res = project_to_feasible(res.x, n_points)
                    c5 = evaluate_erdos_solution(h_res, dx)
                    if c5 < best_c5_level:
                        best_c5_level = c5
                        best_h_level = h_res.copy()

                pass
            
            # Update current best if we've improved

            best_h = best_h_level.copy()
            best_c5 = best_c5_level
            best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
