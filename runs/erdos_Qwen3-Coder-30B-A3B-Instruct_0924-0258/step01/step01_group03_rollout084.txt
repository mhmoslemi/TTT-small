import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        return np.ones_like(h) * (target_sum / len(h))
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    return h

    # Normalize initial h
    h = project_to_feasible(h, target_sum)
    
    # Gradient descent with adaptive learning rate
    learning_rate = 0.01
    stagnation_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()

        # Compute gradient for this maximizer
        grad = np.zeros_like(h)

        h = h - learning_rate * grad

        c5 = evaluate(h, dx)
