import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def perturb(h, temp, dx):
    n = len(h)
    noise = np.random.normal(0, temp, n)
    h_new = h + noise
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_constraints(h_new, dx)
    return h_new

def compute_gradient(h, dx, k_opt):
    # For specific k, compute gradient of correlation w.r.t h
    # We only care about the direction, not exact value
    n = len(h)
    grad = np.zeros(n)
    # grad[i] = d/dh[i] int h(x)(1 - h(x+k)) dx
    # = (1 - h[i+k]) - h[i-k] 
    # But we only care about the sign for direction of descent
    for i in range(n):
        if i+k_opt < n:
            grad[i] -= (1 - h[i+k_opt]) 
        if i-k_opt >= 0:
            grad[i] -= h[i-k_opt] 
    
    # Use simpler approach: 
    # Gradient is just 1-h[k] + h[-k] at each position
    # But let's use FFT perspective
    return grad 

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 150
    n_points = n_low
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Use alternating approach: gradient steps with simulated annealing steps 
    iter_count = 0
    T = 0.1
    cooling = 0.99
    iter_per_temp = 500
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Adaptive resolution strategy:
        if n_points < n_high and iter_count > 0 and iter_count % 2000 == 0:
            if best_c5 < 0.381:
                # Try to double resolution 
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), h)
                h = h_long
                n_points = 2 * n_points
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                best_c5 = evaluate(h, dx)
                best_h = h.copy()
                
        # Local descent step
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            # Simple gradient-like update based on correlation peaks
            c5_val = evaluate(h, dx)
            corr = correlate(h, 1-h, mode='full')
            k_max = np.argmax(corr)
            max_correl = corr[k_max] 
            
            # Adjustment based on gradient sign
            n = len(h)
            # k_max ranges from 0 to 2*n-2, mapping to temporal lag from -n+1 to n-1 
            lag = k_max - (n-1)
            
            h_new = h.copy()
            # Try modifying values to reduce C5 at lag
            # We modify elements around the peak impact
            for i in range(max(0, -lag), min(n, n-lag)):
                if i + lag >= 0 and i + lag < n:
                    # h[i] interacts with h[i+lag] in integral
                    if h[i] > 0.01:
                        h_new[i] = max(0, h[i] - 0.01 * (1 - h[i+lag]) * 0.5)
                    if h[i+lag] < 0.99:
                        h_new[i+lag] = min(1, h[i+lag] + 0.01 * h[i] * 0.5)
            
            h_new = project_to_constraints(h_new, dx)    
            c5_new = evaluate(h_new, dx)
            
            # Accept with SA acceptance probability
            delta = c5_new - c5_val
            if delta < 0 or np.random.uniform() < np.exp(-delta / T):
                h = h_new
                c5_val = c5_new
                
            if c5_val < best_c5:
                best_h = h.copy()
                best_c5 = c5_val
                
            iter_count += 1
            
        # Cool down
        T *= cooling
        if T < 1e-6:
            break
                
    return best_h.tolist(), best_c5, len(best_h)
