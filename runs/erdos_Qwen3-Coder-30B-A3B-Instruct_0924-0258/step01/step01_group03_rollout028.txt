import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    if current_sum > target_sum:
        # Reduce values
        scale = target_sum / current_sum
        h = h * scale
        # Clip again and renormalize
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        # Distribute difference
        diff = target_sum - current_sum
        if diff > 1e-10:
            idx = np.argsort(h)
            for i in range(len(idx)-1, -1, -1):
                if diff <= 0:
                    break
                j = idx[i]
                if h[j] < 1.0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
    else:
        # Increase values
        diff = target_sum - current_sum
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] > 0:
                update = min(diff, h[j])
                h[j] -= update
                diff -= update
        # Now increase smaller ones
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] < 1.0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    return h
