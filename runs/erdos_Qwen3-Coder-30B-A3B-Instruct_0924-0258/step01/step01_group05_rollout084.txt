import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Genetic Algorithm parameters
    pop_size = 50
    tournament_size = 10
    elite_size = pop_size // 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    patience = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_init = np.random.rand(n_points)
        h_init = project_feasible(h_init)
        population.append(h_init)
    
    best_solution = None
    best_C5 = float('inf')
    patience_counter = 0
    
    generation = 0
    while time.time() - start_time < budget_s - 0.1:
        generation += 1
        # Evaluate fitness
        fitness_scores = []
        for h_ind in population:
            score = compute_C5(h_ind)
            fitness_scores.append(score)
        
        # Track best solution
        min_score_idx = np.argmin(fitness_scores)
        if fitness_scores[min_score_idx] < best_C5:
            best_C5 = fitness_scores[min_score_idx]
            best_solution = population[min_score_idx].copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Early stopping if no improvement
        if patience_counter >= patience:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        elites = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population
        new_population = elites.copy()
        while len(new_population) < pop_size:
            # Selection
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            parent2 = population[np.random.choice(pop_size)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Two-point crossover
                point1, point2 = sorted(np.random.choice(n_points, 2, replace=False))
                child = np.concatenate([parent1[:point1], parent2[point1:point2], parent1[point2:]])
            
            # Mutate
            if np.random.rand() < mutation_rate:
                # Gaussian mutation
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
                child = project_feasible(child)
            elif np.random.rand() < mutation_rate:
                # Reset mutation
                reset_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                child[reset_indices] = np.random.choice([0, 1], size=len(reset_indices))
                child = project_feasible(child)
            
            new_population.append(child)
        
        # Keep population size consistent
        population = new_population[:pop_size]
    
    # Final validation
    if best_solution is None:
        best_solution = h.copy()
    
    best_solution = project_feasible(best_solution)
    final_C5 = compute_C5(best_solution)
    
    return list(best_solution), final_C5, n_points
