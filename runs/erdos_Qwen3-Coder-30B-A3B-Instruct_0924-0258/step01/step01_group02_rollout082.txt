import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Default initialization
        initial_h = np.full(45, 1.0/45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Scale initial h to satisfy constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    # Population-based approach using genetic algorithm
    population_size = 75
    mutation_rate = 0.05
    elite_size = population_size // 4
    generations = 2000
    
    def compute_c5(h_vec):
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def create_individual():
        # Generate random feasible individual
        h = np.random.rand(n_points)
        h = project_h(h)
        return h
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    best_c5 = compute_c5(initial_h)
    best_h = initial_h.copy()
    
    start_time = time.time()
    no_improvement_count = 0
    generation = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness for entire population
        fitness_scores = []
        for h in population:
            c5 = compute_c5(h)
            fitness_scores.append(-c5)  # Negative for maximization
        
        # Selection (tournament)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness_scores)[-elite_size:]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring via crossover and mutation
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(0, population_size)]
            parent2 = selected[np.random.randint(0, population_size)]
            
            # Uniform crossover
            mask = np.random.rand(n_points) > 0.5
            offspring = np.where(mask, parent1, parent2)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                offspring += noise
                offspring = project_h(offspring)
            
            new_population.append(offspring)
        
        population = new_population[:population_size]
        
        # Track best solution
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
        # Early termination if no improvement for many iterations
        if no_improvement_count > 1000:
            break
            
        generation += 1
        
        # Occasionally add diversity with random individuals
        if generation % 50 == 0:
            for _ in range(5):
                population[np.random.randint(0, population_size)] = create_individual()
    
    return (best_h.tolist(), best_c5, n_points)
