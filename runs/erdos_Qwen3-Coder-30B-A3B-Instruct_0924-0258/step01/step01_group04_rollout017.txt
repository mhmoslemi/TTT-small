import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess

        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        x_orig = np.linspace(0, 1, len(h_vec))
        x_target = np.linspace(0, 1, target_len)
        return np.interp(x_target, x_orig, h_vec)
    
    # Multi-resolution approach
    resolutions = [5, 10, 20, 40, n_points]
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    for i, current_n in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize or upscale
        if i == 0:
            h_current = h[:current_n].copy()
        else:
            h_current = upsample_h(best_h, current_n)
        
        # Adam optimizer parameters
        m = np.zeros(current_n)
        v = np.zeros(current_n)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.01
        t = 0
        
        # Optimization for current resolution
        for iteration in range(10000):
            if time.time() - start_time > budget_s - 1:
                break
                
            t += 1
            h_current = project_feasible(h_current)
            current_C5 = compute_overlap(h_current)
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current.copy()
            
            # Compute gradient using finite differences
            gradient = np.zeros(current_n)
            eps_val = 1e-4
            
            # Sample dominant k shifts
            conv = np.correlate(h_current, 1 - h_current, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-current_n+1:mid+current_n-1] * dx
            top_k = np.argsort(overlaps)[-5:][::-1]
            
            for j in range(current_n):
                h_plus = h_current.copy()
                h_plus[j] += eps_val
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                h_minus = h_current.copy()
                h_minus[j] -= eps_val
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                gradient[j] = (C_plus - C_minus) / (2 * eps_val)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * gradient**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_current -= lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Early stopping if no improvement
            if iteration % 1000 == 0:
                if iteration > 0 and compute_overlap(h_current) >= best_C5 * 1.01:
                    break
                
    return (best_h.tolist(), best_C5, n_points)
