import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10.0):
    # Smooth approximation to max
    return np.log(np.sum(np.exp(beta * x))) / beta

def compute_correlation_and_grad(h, dx):
    # Compute correlation matrix using FFT
    n = len(h)
    h1 = 1 - h
    # Full correlation of h with (1-h)
    corr_full = correlate(h, h1, mode='full')
    # Positions for lags from -(n-1) to (n-1)
    max_corr = np.max(corr_full)
    return max_corr * dx

def compute_smooth_max_corr(h, dx, beta=10.0):
    # Compute smooth max correlation
    n = len(h)
    h1 = 1 - h
    corr_full = correlate(h, h1, mode='full')
    # Convert to per-lag values (extract mid part)
    corr_values = corr_full[len(corr_full)//2 - (n-1):len(corr_full)//2 + n]
    return smooth_max(corr_values * dx, beta)

def gradient_h(h, dx, beta=10.0):
    # numerically compute gradient of smooth max correlation w.r.t. h
    # We use finite differences for demonstration only
    delta = 1e-6
    grad = np.zeros_like(h)
    base_val = compute_smooth_max_corr(h, dx, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += delta
        h_plus = project_to_constraints(h_plus, dx)
        val_plus = compute_smooth_max_corr(h_plus, dx, beta)
        grad[i] = (val_plus - base_val) / delta
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution to be fast
    n_low = 50
    n_high = 200
    n_points = n_low
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) > n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        else:
            h = np.resize(h, n_points)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5
    
    # Optimizer ingredient
    def obj_func(h):
        h = project_to_constraints(h, dx)
        return compute_smooth_max_corr(h, dx, beta=10.0)
    
    def grad_func(h):
        return gradient_h(h, dx, beta=10.0)
    
    iter_count = 0
    max_iter = 2000
    lr = 1e-3
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        h = project_to_constraints(h, dx)
        obj_val = compute_smooth_max_corr(h, dx, beta=10.0)
        grad = grad_func(h)
        
        # Gradient descent step
        h_new = h - lr * grad
        h_new = project_to_constraints(h_new, dx)
        new_c5 = evaluate(h_new, dx)
        
        if new_c5 < last_c5:
            h = h_new
            last_c5 = new_c5
            if last_c5 < best_c5:
                best_h = h.copy()
                best_c5 = last_c5
        
        iter_count += 1
        if iter_count % 100 == 0:
            # adaptive learning rate
            lr *= 0.95
            if lr < 1e-6:
                lr = 1e-6
                
        if n_points < n_high and i > 100:
            if last_c5 < best_c5 * 0.995:
                # interpolate to higher resolution
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), h)
                h = h_long
                n_points = 2 * n_points
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                last_c5 = evaluate(h, dx)
                lr = 1e-3
                i = 0  # reset iteration for new resolution
    
    return best_h.tolist(), best_c5, len(best_h)
