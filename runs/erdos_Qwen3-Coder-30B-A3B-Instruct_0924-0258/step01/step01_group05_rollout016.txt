import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent or default
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values, dtype=float)
        n_points = len(h_base)
    else:
        n_points = 45
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def evaluate_candidates(candidates, n_points):
        best_h = None
        best_C5 = float('inf')
        for candidate in candidates:
            candidate = project_feasible(candidate)
            C5 = compute_C5(candidate)
            if C5 < best_C5:
                best_C5 = C5
                best_h = candidate.copy()
        return best_h, best_C5
    
    # Start with a coarse search
    n_levels = 3
    base_n = 5
    candidates = []
    n_samples = 50
    
    # Generate candidates at coarse level
    for _ in range(n_samples):
        if time.time() - start_time > budget_s - 0.1:
            break
        h = np.random.rand(base_n)
        h = project_feasible(h)
        candidates.append(h)
    
    best_h, best_C5 = evaluate_candidates(candidates, base_n)
    
    # Refine the search
    for level in range(n_levels):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Interpolate the best solution to the next level
        n_new = base_n * (2**(level+1))
        if n_new > n_points:
            n_new = n_points
        new_candidates = []
        
        # Generate candidates by refining around current best
        for _ in range(20):  # Refine top candidates
            if time.time() - start_time > budget_s - 0.1:
                break
            # Start from best and perturb
            h_ref = best_h.copy()
            h_ref = np.interp(np.linspace(0, 1, n_new), np.linspace(0, 1, len(h_ref)), h_ref)
            noise = np.random.normal(0, 0.1, n_new)
            h_pert = h_ref + noise
            h_pert = project_feasible(h_pert)
            new_candidates.append(h_pert)
            
        # Add some random candidates
        for _ in range(10):
            if time.time() - start_time > budget_s - 0.1:
                break
            h = np.random.rand(n_new)
            h = project_feasible(h)
            new_candidates.append(h)
        
        if new_candidates:
            best_h, best_C5 = evaluate_candidates(new_candidates, n_new)
    
    # Final refinement at highest resolution (n_points)
    if time.time() - start_time < budget_s - 0.1:
        final_candidates = []
        max_iter = 1000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Start with best and slightly perturb
            h_ref = best_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h_pert = h_ref + noise
            h_pert = project_feasible(h_pert)
            final_candidates.append(h_pert)
            
            # Try variations of best solution (simpler local search)
            if i % 100 == 0:
                h_var = best_h.copy()
                # Shift function to explore
                shift_idx = max(1, int(np.random.rand() * 5))
                shifted = np.roll(h_var, shift_idx)
                shifted = project_feasible(shifted)
                final_candidates.append(shifted)
        
        if final_candidates:
            best_h, best_C5 = evaluate_candidates(final_candidates, n_points)
    
    # Final projection to ensure feasibility
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
