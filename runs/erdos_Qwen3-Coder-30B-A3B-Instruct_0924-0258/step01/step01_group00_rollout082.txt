import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mean_correlation(h, dx):
    """Compute mean of correlations"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.mean(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():

    else:
        h = np.ones(30) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    max_iter = 500
    patience = 20
    patience_counter = 0
    best_c5_stable = best_c5
    
    # Iterative optimization
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Small perturbations to escape local optima
        perturbation_strength = max(0.05, 0.1 * np.exp(-iteration / 50))
        h_perturbed = project_to_feasible(h + np.random.normal(0, perturbation_strength, n_points), n_points)
        
        # Local optimization using trust-constr
        def obj(h_vec):
            # Minimize negative of mean correlation to maximize it
            return -mean_correlation(h_vec, dx)
        
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple restarts
        best_local_h = h_perturbed
        best_local_c5 = evaluate_erdos_solution(h_perturbed, dx)
        
        for _ in range(3):
            try:
                res = minimize(obj, h_perturbed, method='trust-constr',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 30, 'gtol': 1e-6})
                if res.success:
                    h_candidate = project_to_feasible(res.x, n_points)
                    c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                    if c5_candidate < best_local_c5:
                        best_local_c5 = c5_candidate
                        best_local_h = h_candidate
            except:
                pass
            
            # Random perturbation for diversity
            h_perturbed = project_to_feasible(h_perturbed + np.random.normal(0, 0.02, n_points), n_points)
        
        # Accept improvement
        if best_local_c5 < best_c5:
            h = best_local_h.copy()
            best_c5 = best_local_c5
            patience_counter = 0
            best_c5_stable = best_c5
        else:
            patience_counter += 1
            # Gradually increase the tolerance
            tolerance_factor = 1.0 + 0.001 * patience_counter
            if patience_counter > patience and best_c5 > best_c5_stable * tolerance_factor:
                # Restart with a more diverse configuration
                h = np.random.uniform(0, 1, n_points)
                h = project_to_feasible(h, n_points)
                patience_counter = 0
        
        # Progress tracking
        if iteration % 50 == 0:
            elapsed = time() - start_time
            if elapsed > budget_s * 0.9:

    for _ in range(20):

        h_new = h + np.random.normal(0, 0.005, n_points)
        h_new = project_to_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
    
    return (list(h), best_c5, n_points)
