import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) / 2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Early iterations use larger steps
    c5_target = 0.38
    adjustment_fraction = 0.1
    max_iterations = 2000
    improvement_count = 0
    no_improvement_count = 0
    perturbation_strength = 0.01
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Compute current overlap and find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        max_idx = np.argmax(corr)
        # The shift k corresponding to max_index
        k = max_idx - (n_points - 1)
        
        # Find top contributing indices
        # Compute overlap for each x: h(x)(1-h(x+k))
        # Look at indices where this product is large
        indices = []
        # This is a simplified heuristic - for each point that contributes to maximum,
        # we'll target those areas for adjustment
        for j in range(n_points):
            left_idx = j
            right_idx = (j + k) % n_points if k >= 0 else (j + k + n_points) % n_points
            if left_idx < n_points and right_idx < n_points:
                overlap_val = h[left_idx] * (1 - h[right_idx])
                indices.append((j, overlap_val))
        
        # Sort by overlap value and get top 10%
        indices.sort(key=lambda x: x[1], reverse=True)
        top_indices = [idx for idx, _ in indices[:max(1, int(0.1 * n_points))]]
        
        # Create a copy of h to adjust
        h_new = h.copy()
        
        # Decrement values in top contributing indices
        for idx in top_indices:
            if h_new[idx] > 0:
                # Adjust by fraction of the difference
                adjustment = min(adjustment_fraction * (max_overlap - c5_target) / n_points, h_new[idx])
                h_new[idx] -= adjustment
                
        # Re-project to maintain constraints
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate the new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            improvement_count += 1
            no_improvement_count = 0
            h = h_new.copy()
        else:
            no_improvement_count += 1
            # Perturb if no improvement in many iterations
            if no_improvement_count > 50:
                h = project_to_feasible(h + np.random.normal(0, perturbation_strength, n_points), n_points)
                no_improvement_count = 0
                perturbation_strength = min(perturbation_strength * 1.05, 0.1)
                
    return (list(best_h), best_c5, n_points)
