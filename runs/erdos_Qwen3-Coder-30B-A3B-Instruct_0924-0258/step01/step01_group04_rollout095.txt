def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Fallback: create a simple initial guess if no initial_h_values available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Start with a coarse resolution
        n_points = 20
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    best_h = h.copy()
    best_C5 = compute_C5(best_h)

    # Try SDP optimization
    try:
        # Construct matrix
        h_vec = Variable(n_points)
        X = Variable((n_points, n_points), symmetric=True)
        
        # Constraints: X is Gram matrix with diagonal entries equal to h
        constraints = [X >> 0]  # Positive semidefinite
        for i in range(n_points):
            constraints.append(X[i, i] == h_vec[i])
            
        # Bounds & sum constraint
        constraints += [0 <= h_vec, h_vec <= 1]
        constraints.append(sum_entries(h_vec) == target_sum)
        
        # Objective - maximize min overlap
        # We'll solve the dual later by approximating the overlap with a feasible sum
        # Instead: we'll use a more direct optimization approach
        
        # Try restart optimization with projected gradient descent
        n_restarts = 5
        for restart in range(n_restarts):
            # Random perturbation
            h_restart = project_h(h + np.random.normal(0, 0.01, n_points))
            
            # Wrapper for optimization
            def optimization_function(h_vec):
                h_vec = project_h(h_vec)
                return compute_C5(h_vec)
            
            # Optimizer
            result = minimize(optimization_function, 
                            h_restart, 
                            method='L-BFGS-B', 
                            bounds=[(0, 1) for _ in range(n_points)],
                            options={'maxiter': 500})
            
            if result.success:
                h_final = project_h(result.x)
                C5_final = compute_C5(h_final)
                if C5_final < best_C5:
                    best_C5 = C5_final
                    best_h = h_final.copy()
            
            if time.time() - start_time > budget_s - 1:
                break
            
            # Early termination if good solution
            if best_C5 < 0.38085:
                break
                
    except Exception as e:
        # Fallback to gradient descent if SDP not available
        # This continues the earlier algorithm in spiral form
    
    # New improved gradient descent approach with adaptive learning rates
    max_iter = 10000
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_vec = project_h(h)
        current_C5 = compute_C5(h_vec)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_vec.copy()
            
        # Compute gradient via finite differences
        gradient = np.zeros_like(h_vec)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            C_minus = compute_C5(h_minus)
            
            gradient[i] = (C_plus - C_minus) / (2 * eps)
            
        # Update with momentum
        velocity = momentum * velocity - learning_rate * gradient
        h = h + velocity
        
        # Project to feasible region
        h = project_h(h)
        
        # Adaptive learning rate
        if current_C5 < best_C5:
            learning_rate *= 1.1  # Increase if improving
        else:
            learning_rate *= 0.9  # Decrease if not improving
            
        # Early termination
        if best_C5 < 0.38085:
            break

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()

    return (best_h.tolist(), best_C5, n_points)
