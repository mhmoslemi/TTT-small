import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 10.0 * best_c5
    cooling_rate = 0.95
    min_temp = 1e-6
    iterations_per_temp = 500
    
    # Simulated Annealing loop
    while T > min_temp:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        for _ in range(iterations_per_temp):
            # Create a new candidate solution by perturbing h
            h_new = h.copy()
            
            # Randomly select indices to perturb
            n_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, size=n_perturb, replace=False)
            
            # Perturb selected values
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.05)
            
            # Project the new solution to satisfy constraints
            h_new = project_h(h_new)
            
            # Compute new C5
            c5_new = compute_c5(h_new)
            
            # Accept or reject the new solution
            delta = c5_new - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
        
        # Cool down
        T *= cooling_rate
    
    return (best_h.tolist(), best_c5, n_points)
