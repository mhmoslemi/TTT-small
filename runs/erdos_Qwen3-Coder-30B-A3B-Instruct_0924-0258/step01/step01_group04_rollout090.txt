import numpy as np
from scipy import optimize
import time

def compute_overlap(h_vec):
    """Compute C5 overlap value for given h vector."""
    h_vec = np.array(h_vec)
    dx = 2.0 / len(h_vec)
    # Compute full cross-correlation of h with (1 - h)
    conv = np.correlate(h_vec, 1 - h_vec, mode='full')
    # Extract relevant portion (overlaps for k in [0,2])
    mid = len(conv) // 2
    overlaps = conv[mid - len(h_vec) + 1 : mid + len(h_vec) - 1] * dx
    return np.max(overlaps)

def project_feasible(h_vec):
    """Project h to feasible region: [0,1] and sum = n/2."""
    h_vec = np.clip(h_vec, 0, 1)
    target_sum = len(h_vec) / 2.0
    current_sum = np.sum(h_vec)
    if current_sum > 0:
        h_vec = h_vec * target_sum / current_sum
    return h_vec

def upsample_h(h_vec, target_len):
    """Linearly upsample h vector to target length."""
    x_orig = np.linspace(0, 1, len(h_vec))
    x_target = np.linspace(0, 1, target_len)
    return np.interp(x_target, x_orig, h_vec)
