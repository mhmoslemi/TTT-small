import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization: step function
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value scaled by dx
        return np.max(corr) * dx

    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec

    def compute_gradient(h_vec, k_max):
        # Compute gradient for maximum overlap at lag k_max
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        for i in range(n):
            # Derivative w.r.t. h[i]: 
            # contributes to overlap when h[i] multiplies (1-h[i+k_max])
            if i + k_max < n:
                grad[i] -= (1 - h_vec[i + k_max])
            # also depends on h[i-k_max] in the reverse direction
            if i - k_max >= 0:
                grad[i] -= h_vec[i - k_max]
        return grad

    def line_search(h_vec, grad, max_alpha=1.0):
        # Simple line search to find good step size
        best_h = h_vec.copy()
        best_C5 = compute_C5(h_vec)
        best_alpha = 0.0
        
        # Try several alpha values
        alphas = [max_alpha * 0.1, max_alpha * 0.5, max_alpha]
        for alpha in alphas:
            candidate_h = h_vec - alpha * grad
            candidate_h = project_feasible(candidate_h)
            new_C5 = compute_C5(candidate_h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = candidate_h.copy()
                best_alpha = alpha
        return best_h, best_C5, best_alpha

    h_curr = h.copy()
    C5_curr = compute_C5(h_curr)
    best_h = h_curr.copy()
    best_C5 = C5_curr
    alpha = 0.1
    stagnation_count = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        if stagnation_count > 50:
            break
            
        # Compute all overlap sums
        conv = correlate(h_curr, 1 - h_curr, mode='full')
        # Find index of maximum overlap
        max_idx = np.argmax(conv)
        k_max = max_idx - (len(h_curr) - 1)
        
        # Compute gradient for this overlap
        grad = compute_gradient(h_curr, k_max)
        
        # Line search to determine step size
        new_h, new_C5, step_alpha = line_search(h_curr, grad, alpha)
        
        if new_C5 < C5_curr:
            h_curr = new_h
            C5_curr = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h_curr.copy()
            stagnation_count = 0
            alpha = min(1.0, alpha * 1.1)  # Increase step size
        else:
            # No improvement
            stagnation_count += 1
            h_curr = project_feasible(h_curr + np.random.normal(0, 0.005, len(h_curr)))
            C5_curr = compute_C5(h_curr)
            alpha = max(0.01, alpha * 0.5)  # Decrease step size

    # Final projection
    best_h = project_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return list(best_h), final_C5, n_points
