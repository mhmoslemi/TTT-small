import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Simulated Annealing parameters
    T0 = 0.1
    tau = budget_s / np.log(10)
    sigma = 0.1
    T_min = 1e-5
    
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    
    h_best = h_current.copy()
    C5_best = C5_current
    
    # Cooling schedule
    def temperature(t):
        T = T0 * np.exp(-t / tau)
        return max(T, T_min)
    
    # Adaptive step size
    acceptance_count = 0
    acceptance_target = 0.25
    last_acceptance_time = start_time
    
    while time.time() - start_time < budget_s - 0.1:
        # Update temperature
        t_elapsed = time.time() - start_time
        T = temperature(t_elapsed)
        
        # Create candidate solution
        h_candidate = h_current.copy()
        
        # Perturb high contribution intervals
        corr_full = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr_full)
        max_lag = max_lag_idx - (len(h_current) - 1)
        
        # Identify priority intervals based on correlation
        if max_lag >= 0:
            priority_indices = np.arange(max_lag, len(h_current))
        else:
            priority_indices = np.arange(len(h_current) + max_lag)
        
        # Add some random indices for exploration
        all_indices = np.arange(len(h_current))
        np.random.shuffle(all_indices)
        priority_indices = np.unique(np.concatenate([priority_indices, all_indices[:max(1, len(all_indices)//5)]])[:len(h_current)//3])
        
        # Apply Gaussian noise to priority indices
        noise = np.random.normal(0, sigma, len(h_candidate))
        h_candidate[priority_indices] += noise[priority_indices]
        
        # Apply small random flips
        flip_indices = np.random.choice(len(h_candidate), size=max(1, len(h_candidate)//20), replace=False)
        h_candidate[flip_indices] = 1 - h_candidate[flip_indices]
        
        # Project to feasible set
        h_candidate = project_feasible(h_candidate)
        
        # Compute cost
        C5_candidate = compute_C5(h_candidate)
        
        # Acceptance criterion
        delta_C5 = C5_candidate - C5_current
        if delta_C5 < 0:
            accept = True
        else:
            if T > 0:
                accept = np.random.random() < np.exp(-delta_C5 / T)
            else:
                accept = False
        
        if accept:
            h_current = h_candidate.copy()
            C5_current = C5_candidate
            if C5_current < C5_best:
                h_best = h_current.copy()
                C5_best = C5_current
                
        # Adaptive step size
        acceptance_count += 1
        if acceptance_count >= 10:
            # Estimate acceptance rate
            acceptance_rate = 1.0  # Just for initialization
            
            # Periodically adjust step size
            if time.time() - last_acceptance_time > 1.0:
                if acceptance_rate < acceptance_target * 0.8:
                    sigma = min(sigma * 1.1, 0.5)
                elif acceptance_rate > acceptance_target * 1.2:
                    sigma = max(sigma * 0.9, 0.01)
                last_acceptance_time = time.time()
                acceptance_count = 0
    
    # Final projection
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
