import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    # Project h to satisfy sum(h) * dx = 1
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution
    n_low = 10
    n_high = 100
    n_points = n_low
    dx = 2.0 / n_points
    
    # Use initial construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
    
    # Ensure constraints
    h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 100
    alpha = 0.01  # Learning rate
    
    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # ADMM iterations for current resolution
        for it in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Update h by minimizing: ||h - (z + lambda/rho)||^2
            # Where z = 1 - h (constraint) but we approximate via gradient descent
            # Using a simplified gradient approach
            old_h = h.copy()
            
            # Simple gradient descent step on C5
            corr = correlate(h, 1-h, mode='full')
            k_max = np.argmax(corr)
            # Approximate gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_constraints(h_plus, dx)
                corr_plus = correlate(h_plus, 1-h_plus, mode='full')
                grad[i] = (corr_plus[k_max] - corr[k_max]) / eps
                
            h = h - alpha * grad
            h = project_to_constraints(h, dx)
            
            c5_new = evaluate(h, dx)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                
            iter_count += 1
            
        # Try to refine resolution
        if n_points < n_high and time.time() - start_time < budget_s * 0.9:
            # Interpolate
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            if best_c5 < 0.382:  # More aggressive refinement if already good
                max_iter = 100
            else:
                max_iter = 50
        else:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
