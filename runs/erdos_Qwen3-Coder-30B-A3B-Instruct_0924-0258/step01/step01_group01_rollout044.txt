import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx"""
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    """Project h to satisfy sum(h) * dx = 1"""
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Find largest element and increment it
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution
    n_points = 10
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        # Random initial
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Optimized gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 20
    best_iter = 0
    
    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Minor optimization with local search
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Perturb h by gradient descent
            # We evaluate the derivative numerically
            eps = 1e-4
            h_pert = h.copy()
            grad = np.zeros_like(h)
            
            # Compute approximate gradient
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] = min(1.0, h_plus[j] + eps)
                h_plus = project_to_constraints(h_plus, dx)
                c5_plus = evaluate(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] = max(0.0, h_minus[j] - eps)
                h_minus = project_to_constraints(h_minus, dx)
                c5_minus = evaluate(h_minus, dx)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update h
            h_new = h - learning_rate * grad
            h_new = project_to_constraints(h_new, dx)
            
            # Evaluate
            c5_new = evaluate(h_new, dx)
            
            # Accept if better
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_iter = i
                best_h = h.copy()
            elif i - best_iter > patience:
                break
                
            iter_count += 1
        
        # Double resolution if beneficial
        if n_points < 100:  # Limit resolution if needed
            # Interpolate to make h longer, then reproject
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                             np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            best_c5 = evaluate(h, dx)
            best_iter = 0
            learning_rate *= 0.9  # Reduce learning rate
        else:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
