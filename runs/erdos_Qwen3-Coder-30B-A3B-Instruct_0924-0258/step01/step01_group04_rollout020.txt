import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: start with a simple construction
        n_points = 20
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Use CVXPY to solve SDP
    def solve_sdp(h_init, n):
        try:
            h = cp.Variable(n)
            X = cp.Variable((n, n), symmetric=True)
            t = cp.Variable()
            
            # Constraints
            constraints = [
                cp.trace(X) == n/2,  # sum(h) = n/2
                h >= 0,
                h <= 1,
                X >> 0,  # Positive semidefinite
            ]
            
            # X_ij = h_i * h_j
            for i in range(n):
                constraints.append(X[i, i] == h[i])
                
            # For each lag k, constraint on overlap
            for k in range(1, n):
                # Sum of h[i] * h[i+k] for valid i
                overlap_sum = 0
                for i in range(n):
                    j = (i + k) % n
                    overlap_sum += X[i, j]
                constraints.append(t <= overlap_sum)
                
            # Objective
            objective = cp.Maximize(t)
            prob = cp.Problem(objective, constraints)
            
            # Solve with MOSEK
            prob.solve(solver=cp.MOSEK, verbose=False, max_iter=100)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:
                h_solution = h.value
                return h_solution
            else:
                return None
        except:
            return None
            
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Optimization in multiple phases
    # Phase 1: Grid search and SDP initialization
    iterations = 0
    n_current = n_points
    
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        # Try SDP approach
        h_sdp = solve_sdp(best_h, n_current)
        if h_sdp is not None:
            h_sdp = project_h(h_sdp)
            c5_sdp = compute_C5(h_sdp)
            if c5_sdp < best_C5:
                best_C5 = c5_sdp
                best_h = h_sdp.copy()
        
        # Local optimization
        h_local = best_h.copy()
        def obj_func(h_vec):
            return compute_C5(project_h(h_vec))
            
        try:
            res = optimize.minimize(obj_func, h_local, method='L-BFGS-B', 
                                  bounds=[(0, 1) for _ in range(n_current)],
                                  options={'maxiter': 10})
            if res.success:
                h_trial = project_h(res.x)
                c5_trial = compute_C5(h_trial)
                if c5_trial < best_C5:
                    best_C5 = c5_trial
                    best_h = h_trial.copy()
        except:
            pass
            
        # Up-scale if possible
        if time.time() - start_time < budget_s - 100:
            if n_current < 45:  # try larger n
                n_current = min(45, n_current * 2)
                h_new = np.interp(np.linspace(0, 1, n_current), 
                                 np.linspace(0, 1, len(best_h)), best_h)
                best_h = h_new
                best_C5 = compute_C5(best_h)
                
        # Periodic evaluation
        if iterations % 100 == 0:
            current_C5 = compute_C5(best_h)
            if current_C5 < best_C5:
                best_C5 = current_C5

    # Final validation
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        
    return (best_h.tolist(), best_C5, n_points)
