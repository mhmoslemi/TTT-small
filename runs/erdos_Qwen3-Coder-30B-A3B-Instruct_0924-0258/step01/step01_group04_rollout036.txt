    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Use differential evolution on a subset of parameters
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Simple wrapper for the optimization
    def objective(h_vec):
        h_proj = project_h(h_vec)
        return compute_C5(h_proj)
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Try different optimization approaches
    num_attempts = 0
    max_attempts = 300
    
    while time.time() - start_time < budget_s - 1 and num_attempts < max_attempts:

        # Random restart concentrated around current best
        restart_h = best_h + np.random.normal(0, 0.02, n_points)

        try:
            res = minimize(objective, restart_h, method='Nelder-Mead',
