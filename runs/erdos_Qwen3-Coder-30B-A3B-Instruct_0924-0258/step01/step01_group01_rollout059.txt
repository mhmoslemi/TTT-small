import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def smooth_max(x, beta=10.0):
    return 1.0/beta * np.log(np.sum(np.exp(beta * x)))

def smooth_max_grad(x, beta=10.0):
    exp_x = np.exp(beta * x)
    return exp_x / np.sum(exp_x)

def compute_correlation_gradient(h, dx):
    n = len(h)
    # Compute the gradient of the max correlation with respect to h
    # This uses the fact that grad of max is proportional to exp(beta * corr) / sum(exp(beta * corr))
    corr_full = correlate(h, 1-h, mode='full') * dx 
    corr_indices = np.arange(-n+1, n)
    # Clamp large values to prevent overflow
    corr_full = np.clip(corr_full, -50, 50)
    # Smooth max approach for differentiable max
    max_corr = np.max(corr_full)
    exp_corr = np.exp(10 * (corr_full - max_corr))
    # The soft maximum gradient is the normalized exponential of correlations
    # but we are interested in effect of h on the max of cross-correlations
    
    # More precise: compute the actual gradient of the max correlation
    # analytically compute the gradient in frequency domain
    h1 = np.concatenate([h, np.zeros(n-1)])
    h2 = np.concatenate([1-h, np.zeros(n-1)])
    
    # Forward FFTs
    H1 = fft(h1)
    H2 = fft(h2)
    # Cross correlations in frequency domain
    Corr_FFT = H1 * np.conj(H2)
    corr_time = ifft(Corr_FFT).real[:2*n-1]
    
    # If we wanted gradient from this signal, we'd need to compute:
    # gradient w.r.t h = derivative of corr(h, 1-h) w.r.t h
    # which involves convolution derivative rules
    # Instead, we'll use finite differences
    
    grad = np.zeros_like(h)
    eps = 1e-6
    for i in range(len(h)):
        h_eps = h.copy()
        h_eps[i] += eps
        corr_plus = correlate(h_eps, 1-h_eps, mode='full')
        corr_minus = correlate(h, 1-h, mode='full')
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / eps
    
    return grad * dx

def armijo_line_search(h, grad, f, f_grad, alpha_init=1.0, rho=0.5, c=1e-4):
    alpha = alpha_init
    while True:
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, 2.0/len(h))
        f_new = evaluate(h_new, 2.0/len(h))
        if f_new <= f - c * alpha * np.dot(grad, grad):
            return alpha
        alpha *= rho
        if alpha < 1e-10:
            return alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with resolution 70 for good tradeoff
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Optimization parameters
    lr = 0.005
    max_iter = 1000
    iter_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        iter_count += 1
        # Compute current value and gradient
        c5_current = evaluate(h, dx)
        
        # Compute approximate gradient
        # Using finite differences for derivative
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(len(h)):
            h_pert = h.copy()
            h_pert[j] += eps
            c5_plus = evaluate(h_pert, dx)
            grad[j] = (c5_plus - c5_current) / eps
        
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_constraints(h_new, dx)
        
        # Accept or reject based on improvement
        c5_new = evaluate(h_new, dx)
        if c5_new < c5_current:
            h = h_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        else:
            # Try a smaller learning rate if no improvement
            lr *= 0.9
        
        # Adaptive learning rate
        if iter_count % 500 == 0:
            lr *= 0.5
    
    return best_h.tolist(), best_c5, len(best_h)
