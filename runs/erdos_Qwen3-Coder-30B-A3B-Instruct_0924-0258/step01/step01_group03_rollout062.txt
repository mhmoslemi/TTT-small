import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute difference
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] < 1:
                increase = min(diff, 1 - h[j])
                h[j] += increase
                diff -= increase
    else:
        # Need to decrease values
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff >= 0:
                break
            j = idx[i]
            if h[j] > 0:
                decrease = min(-diff, h[j])
                h[j] -= decrease
                diff += decrease
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # dx
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize simplex
    m = n_points + 2  # number of points in simplex
    simplex = [h0.copy()]
    for _ in range(m - 1):
        pert = np.random.normal(0, 0.05, n_points)
        h = h0 + pert
        h = project_to_feasible(h, target_sum)
        simplex.append(h)
    
    # Nelder-Mead parameters
    alpha = 1.0  # reflection
    gamma = 2.0  # expansion
    rho = 0.5    # contraction
    sigma = 0.5  # reduction
    
    best_h = None
    best_c5 = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s:
        # Evaluate all points in simplex
        values = []
        for h in simplex:
            val = evaluate(h, dx)
            values.append(val)
        
        # Sort by value
        idxs = np.argsort(values)
        simplex = [simplex[i] for i in idxs]
        values = [values[i] for i in idxs]
        
        # Track best
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
        
        # Check for convergence
        if values[-1] - values[0] < 1e-8:
            # Try mutation
            if iteration % 50 == 0:
                for i in range(1, len(simplex)):
                    if np.random.random() < 0.3:
                        pert = np.random.normal(0, 0.05, n_points)
                        simplex[i] = simplex[i] + pert
                        simplex[i] = project_to_feasible(simplex[i], target_sum)
        
        # Nelder-Mead operations
        x0 = simplex[0]  # best
        x1 = simplex[-1]  # worst
        x2 = simplex[-2]  # second worst
        
        # Compute centroid (excluding worst)
        xc = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        xr = xc + alpha * (xc - x1)
        xr = project_to_feasible(xr, target_sum)
        vr = evaluate(xr, dx)
        
        if values[0] <= vr < values[-2]:
            simplex[-1] = xr
            continue
        
        # Expansion
        if vr < values[0]:
            xe = xc + gamma * (xr - xc)
            xe = project_to_feasible(xe, target_sum)
            ve = evaluate(xe, dx)
            if ve < vr:
                simplex[-1] = xe
            else:
                simplex[-1] = xr
            continue
        
        # Contraction
        if vr >= values[-2]:
            if vr < values[-1]:
                # outside contraction
                xc2 = x1 + rho * (xc - x1)
            else:
                # inside contraction
                xc2 = xc + rho * (x1 - xc)
            xc2 = project_to_feasible(xc2, target_sum)
            vc2 = evaluate(xc2, dx)
            if vc2 < values[-1]:
                simplex[-1] = xc2
                continue
        
        # Reduction
        first = simplex[0]
        for i in range(1, len(simplex)):
            simplex[i] = first + sigma * (simplex[i] - first)
            simplex[i] = project_to_feasible(simplex[i], target_sum)
        
        iteration += 1
    
    if best_h is None:
        # Fallback
        h = project_to_feasible(h0, target_sum)
        c5 = evaluate(h, dx)
        return (h.tolist(), float(c5), n_points)
    
    c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(c5), n_points)
