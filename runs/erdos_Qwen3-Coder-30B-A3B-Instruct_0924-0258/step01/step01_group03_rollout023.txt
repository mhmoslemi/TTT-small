import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Use linear projection
    scaling = target_sum / current_sum
    h = h * scaling
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Set target sum
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # Initialize
    h = h0.copy()
    h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iteration = 0
    learning_rate = 0.01
    
    # Main optimization loop
    while time.time() - start_time < budget_s:
        if iteration % 50 == 0 and iteration > 0:
            learning_rate *= 0.5
        
        # Find the shift that maximizes overlap for current h
        corr = correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(corr)
        shift = max_shift - (len(h) - 1)
        
        # Compute gradient approximated by difference in cross-correlation
        # This is approximate gradient for max_{s} sum h[i] * (1 - h[i+s])
        grad = np.zeros_like(h)
        if shift >= 0:
            for i in range(len(h) - shift):
                grad[i] += 1 - 2 * h[i + shift]
        else:
            for i in range(-shift, len(h)):
                grad[i] += 1 - 2 * h[i + shift]
        
        h = h - learning_rate * grad
        
        # Project to feasible region
        h = project_to_feasible(h, target_sum)
        
        # Evaluate and possibly update best solution
        current_c5 = evaluate(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        iteration += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
