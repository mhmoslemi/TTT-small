import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_step(h, dx, step_size=0.01):
    # Analytical gradient for C5 = max_k int h(x)(1 - h(x+k)) dx
    n = len(h)
    grad = np.zeros(n)
    # For each position, compute gradient contribution
    for i in range(n):
        # Hessian approximation: second derivative of h(1-h) is -2
        # If we're at i, contribution to C5 from local terms
        # But we should take full correlation and get its derivative
        # Simplified: we'll sample local gradients
        # Actually, use direct gradient computation from derivative of correlation
        # Approx: using forward difference or see the problem structure
        # The expression has a nickname max_k integral of h*(1-h_k)
        # Gradient w.r.t h_i is approximately sum over k of (1 - h_i - h_{i+k})
        # Final grad = (1 - 2*h) * dx  from the convexity analysis
        # We'll see it in practice: approximategradient
        # But it's clearer to treat it as per Correlation maximum
        # For now, we use a surrogate: the gradient is negated because we're minimizing
        # and at stationary point of the corr conv product, taking derivative of h^2 term is:
        # So dC5_dh = (2*h - 1) * dx  or just (1 - 2*h) * dx for our scaled setup
        if i == 0:
            grad[i] = (1 - 2*h[i]) * dx
        else:
            # For numerical stability, add small shift
            grad[i] = (1 - 2*h[i]) * dx
            
    # Gradient descent step
    h_new = h - step_size * grad
    h_new = project_to_constraints(h_new, dx)
    return h_new

    np.random.seed(seed)
    start_time = time.time()

        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n_points
        
        # Initialize h for this resolution

        else:
            h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
        
        # Optimization loop for this resolution
        local_time_limit = budget_s / len(n_points_list) * 0.9
        
        # Track best solution for this resolution
        local_best_h = h.copy()
        local_best_c5 = evaluate(h, dx)
        
        # DYNA_STEP_SIZE = 0.01
        # Basic gradient descent
        max_iter = 3000
        iter_count = 0
        while iter_count < max_iter and time.time() - start_time < budget_s * 0.95:
            h = gradient_step(h, dx, step_size=0.01)

            iter_count += 1
            
            if c5 < local_best_c5:
