import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Simulated Annealing setup
    T0 = 0.1
    tau = budget_s / np.log(10) if budget_s > 0 else 1.0
    sigma = 0.1
    T_min = 1e-5
    
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    
    h_best = h_current.copy()
    C5_best = C5_current
    
    iteration = 0
    accept_count = 0
    total_trials = 0
    
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        T = max(T_min, T0 * np.exp(-iteration / tau))
        
        # Priority sampling: identify top contributing intervals
        # Compute current overlap for all lags
        corr_current = correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr_current)
        # Find 20% of intervals contributing most (using the current pattern)
        # Focus on central portion of correlations
        mid_idx = len(corr_current) // 2
        max_lag = max_lag_idx - mid_idx
        # Perturb a small subset of indices around those contributing to maximum
        indices_to_perturb = set()
        for i in range(n_points):
            for k in [-1, 0, 1]:
                if 0 <= i+k < n_points:
                    indices_to_perturb.add(i)
                    indices_to_perturb.add((i+k) % n_points)
                    
        # Convert to sorted list for reproducible sampling
        indices_to_perturb = sorted(list(indices_to_perturb))
        
        # Create perturbed solution
        h_new = h_current.copy()
        # Apply Gaussian noise to sampled indices
        # For top 20% indices
        n_perturb = max(1, len(indices_to_perturb) // 5)
        indices_sampled = np.random.choice(indices_to_perturb, size=n_perturb, replace=False)
        
        # Add Gaussian noise
        noise = np.random.normal(0, sigma, len(indices_sampled))
        for i, idx in enumerate(indices_sampled):
            h_new[idx] += noise[i]
            
        # Maybe flip a few small values to 0/1 to diversify
        if np.random.rand() < 0.3:
            flip_indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            for idx in flip_indices:
                h_new[idx] = 1 - h_new[idx]
        
        # Project to feasible set
        h_new = project_feasible(h_new)
        
        # Compute new overlap
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        total_trials += 1
        if C5_new < C5_current:
            h_current = h_new
            C5_current = C5_new
            accept_count += 1
            if C5_new < C5_best:
                h_best = h_new.copy()
                C5_best = C5_new
        else:
            # Accept with probability based on temperature
            delta = C5_new - C5_current
            if np.random.rand() < np.exp(-delta / T):
                h_current = h_new
                C5_current = C5_new
                accept_count += 1
                
        # Adaptively adjust step size based on acceptance
        if total_trials > 10:
            accept_rate = accept_count / total_trials
            if accept_rate < 0.1:
                sigma = min(0.5, sigma * 1.1)
            elif accept_rate > 0.4:
                sigma *= 0.9
            accept_count = 0
            total_trials = 0
                
    # Final cleanup and return
    h_final = project_feasible(h_best)
    C5_final = compute_C5(h_final)
    
    return list(h_final), C5_final, n_points
