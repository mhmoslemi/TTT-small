import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    max_iter = 5000
    stagnation_count = 0
    last_improvement = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Compute analytical gradient for k_max
        # For correlation term (h * (1-h))[k_max]
        # Gradient w.r.t h[i] = (1 - 2*h[i]) if h[i+k_max] exists
        # or -h[i-k_max] if h[i-k_max] exists
        grad = np.zeros_like(h)
        
        # For every i, when we consider shift k_max: 
        #  grad[i] = (1 - 2*h[i]) if (i+k_max < n_points)  
        #  or -h[i-k_max] if (i-k_max >= 0)
        
        for i in range(len(h)):
            if i + k_max < n_points:
                grad[i] += (1 - 2 * h[i]) * dx
            if i - k_max >= 0:
                grad[i] -= h[i - k_max] * dx
        
        # Update h using gradient
        h = h - lr * grad
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Anneal learning rate
        lr *= 0.999
        
        # Perturbation to escape local optima
        if iteration % 500 == 0 and iteration > 0:
            noise_indices = np.random.choice(len(h), size=int(0.05 * len(h)), replace=False)
            noise = 0.005 * np.random.normal(0, 1, len(noise_indices))
            h[noise_indices] += noise
            h = project_h(h)
        
        # Early termination if no improvement for long
        if iteration - last_improvement > 1000:
            break
            
        iteration += 1
        
        # Reduce learning rate if stagnating
        if stagnation_count > 500:
            lr *= 0.9
            
        if lr < 1e-6:
            lr = 1e-6
            
        if iteration > max_iter:
            break
    
    return (best_h.tolist(), best_c5, n_points)
