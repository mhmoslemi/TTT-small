import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    # Helper to compute overlap integral
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    # Helper to project to feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Binary search for minimal C5
    def verify_feasibility(h_vec, max_overlap):
        # Setup convex variables
        h_var = Variable(n_points)
        
        # Constraint 1: h in [0,1]
        constraints = [h_var >= 0, h_var <= 1]
        
        # Constraint 2: sum = target_sum
        constraints.append(sum(h_var) == target_sum)
        
        # Constraint 3: overlaps <= max_overlap
        for k in range(n_points):
            # correlation at lag k
            corr_k = 0
            for i in range(n_points - k):
                corr_k += h_var[i] * (1 - h_var[i + k])
            corr_k *= dx
            constraints.append(corr_k <= max_overlap)
        
        # Solve the optimization
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            return True
        except:
            return False
    
    # Binary search bounds
    low = 0.38  # Known lower bound
    high = compute_c5(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    
    # Binary search until convergence or timeout
    while high - low > 1e-5:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        mid = (low + high) / 2
        if verify_feasibility(best_h, mid):
            high = mid
            # Try to improve h
            def obj(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode='full')
                corr *= dx
                return np.max(corr)
            
            # Run optimization with current mid as target
            def constrain_sum(h_vec):
                return np.sum(h_vec) - target_sum
                
            res = minimize(
                obj,
                best_h.copy(),
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constrain_sum},
                options={'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if res.success:
                h_candidate = project_h(res.x)
                c5_candidate = obj(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
                    
        else:
            low = mid
            
        # Occasionally improve current solution
        if low < 0.3807 or (high - low < 1e-4):
            # Local optimization with gradient-based approach
            def obj(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode='full')
                corr *= dx
                return np.max(corr)
            
            # Add small perturbation occasionally
            if np.random.random() < 0.2:
                h_perturb = best_h + 0.01 * np.random.normal(0, 1, n_points)
                h_perturb = project_h(h_perturb)
                c5_perturb = obj(h_perturb)
                if c5_perturb < best_c5:
                    best_h = h_perturb.copy()
                    best_c5 = c5_perturb
    
    # Final refinement
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.8:
        def obj(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            corr *= dx
            return np.max(corr)
        
        def constrain_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        # Use L-BFGS-B optimization for final improvement with current best guess
        res = minimize(
            obj,
            best_h.copy(),
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constrain_sum},
            options={'ftol': 1e-10, 'gtol': 1e-10}
        )
        
        if res.success:
            h_final = project_h(res.x)
            c5_final = obj(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
                
    return (best_h.tolist(), best_c5, n_points)
