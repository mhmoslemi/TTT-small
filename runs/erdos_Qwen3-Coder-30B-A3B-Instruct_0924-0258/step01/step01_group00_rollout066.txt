import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def setup_convex_optimization(h, n_points, dx):
    """Set up convex optimization problem to maximize A(k) = ∫h(x)h(x+k)dx"""
    # Create correlation matrix that represents A(k) for all lags k
    # This is a simplified convex surrogate using a quadratic penalty
    # For the full problem: minimize max_k (1 - A(k)) = minimize min_k A(k)
    # We can relax max_k A(k) using convex constraints and solve it efficiently
    
    # Create variables for h
    h_var = cp.Variable(n_points)
    
    # Constraints for feasibility
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Define the objective using a quadratic penalty or smooth relaxation
    # We seek minimum of sum_k (1 - A(k)) = minimum of sum_k (1 - integral of h(x)*h(x+k)) 
    # This is equivalent to maximization of sum_k A(k), but we use max_k (1 - A(k))

    # Since we are trying to minimize the maximum of 1 - A(k), this becomes:
    # minimize z
    # subject to 1 - A(k) <= z for all k
    
    # Since we're really solving min 1 - max_k A(k)
    # Let's build constraints through a small approximation using few points
    # and convex relaxation using cvxpy  
    return h_var, constraints

def convex_relaxation_minimize(h_start, n_points, dx, time_limit):
    """Perform convex relaxation on the current solution to improve it"""
    start_time = time()
    # Build a convex version and solve it
    # We will construct an efficient model for relaxation
    
    try:

        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
        
        # Approximate max correlation with a smooth function sanity check

        # We will compute actual correlations now iteratively

        current_c5 = evaluate_erdos_solution(h_init, dx)
        best_h = h_init.copy()
        best_c5 = current_c5
        
        loop_count = 0
        while time() - start_time < time_limit and loop_count < 200:
            # Generate a small population with mutations
            pool = [project_to_feasible(h_init + np.random.normal(0, 0.02, n_points), n_points)]
            for _ in range(5):
                # Generate mutated solutions
                h_mutation = h_init + np.random.normal(0, 0.02, n_points) 
                h_mutation = project_to_feasible(h_mutation, n_points)
                pool.append(h_mutation)
            
            # Evaluate pool and keep best computed by actual correlation
            for h_candidate in pool:
                c5 = evaluate_erdos_solution(h_candidate, dx)
                if c5 < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5
            
            h_init = best_h.copy()
            loop_count += 1
            
        return best_h, best_c5
    except Exception as e:
        # fallback to simple iteration
        return h_start, evaluate_erdos_solution(h_start, dx)
