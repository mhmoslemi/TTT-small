import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

    else:
        # Create a default initial vector if none provided
        n_points = 45

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    max_iter = 5000
    stagnation_counter = 0
    stagnation_threshold = 1000
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlap and find k_max
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Compute analytical gradient for k_max
        grad = np.zeros_like(h)
        # Gradient of correlation term h[i] * (1 - h[i + k_max]) w.r.t. h[i]:
        # For i such that i + k_max < n: grad[i] = 1 - 2*h[i]
        # For i such that i + k_max >= n: grad[i] = 0 (due to convolution)
        for i in range(len(h)):
            if i + k_max < len(h):
                grad[i] = 1 - 2 * h[i]
        
        # Update h using gradient descent
        h = h - lr * grad
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Anneal learning rate
        lr *= 0.999
        
        # Occasionally add noise to escape local minima
        if iteration % 500 == 0 and iteration > 0:
            noise_indices = np.random.choice(len(h), size=int(0.05 * len(h)), replace=False)
            noise = 0.01 * np.random.normal(0, 1, len(noise_indices))
            h[noise_indices] += noise
            h = project_h(h)
        
        # Early termination if no improvement for a long time
        if stagnation_counter > stagnation_threshold:
            break
        
        iteration += 1
        
        # Occasionally restart with different initialization
        if iteration % 1000 == 0 and iteration > 0:
            h = np.copy(best_h)
            # Add small perturbation to restart
            noise = 0.005 * np.random.uniform(-1, 1, len(h))
            h = h + noise
            h = project_h(h)
    
    return (best_h.tolist(), best_c5, n_points)
