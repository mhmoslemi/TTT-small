import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_relaxation(h, n_points, dx):
    """Use convex optimization to minimize max overlap"""
    # Build correlation matrix for constraint relaxation
    corr = np.correlate(h, 1 - h, mode='full')
    C5_relaxed = np.max(corr) * dx
    
    # Better: use CVXPY for proper convex relaxation
    # Variables
    h_vars = cp.Variable(n_points)
    # Objective: minimize max correlation term
    obj = cp.Minimize(cp.max(corr))
    
    # Constraints
    constraints = [
        cp.sum(h_vars) == n_points / 2,
        h_vars >= 0,
        h_vars <= 1
    ]
    
    # Solve problem
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT)
        return prob.value
    except:
        # fallback to original estimate if optimization fails
        return C5_relaxed

def genetic_mutation(h, n_points):
    """Apply genetic operators to create new solution"""
    # crossover
    h1 = h.copy()
    h2 = np.random.rand(n_points)
    crossover_point = np.random.randint(1, n_points-1)
    h1[crossover_point:] = h2[crossover_point:]
    
    # mutation: add polynomial noise
    x = np.linspace(0, 2, n_points)
    noise = np.polyval(np.random.randn(3), x) * 0.05
    h1 += noise
    return project_to_feasible(h1, n_points)

def optimize_with_restarts(h_start, n_points, dx, time_limit):
    """Perform optimization with time constraint and restart attempts"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different algorithms
    methods = ['L-BFGS', 'BFGS', 'CG']
    
    for restart in range(10):
        if time() - start_time > time_limit:
            break
            
        h_try = h_start.copy()
        
        # Apply genetic perturbation
        h_try = genetic_mutation(h_try, n_points)
        
        # Apply convex relaxation
        try:
            # Convert to CVX problem
            h_vars = cp.Variable(n_points)
            obj = cp.Minimize(cp.max(np.correlate(h_vars, 1-h_vars, mode='full') * dx))
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            new_h = h_vars.value
            if new_h is not None:
                h_try = project_to_feasible(new_h, n_points)
        except:
            pass
        
        # Local optimization
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        try:
            res = minimize(obj, h_try, method='L-BFGS', options={'maxiter': 10})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
        
        # Update start for next iteration
        h_start = best_h.copy()
        
    return best_h, best_c5
