import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def optimize_with_perturbations(h_start, n_points, dx, time_limit):
    """Optimize using a combination of gradient steps and perturbations"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with projected steps
    for iteration in range(200):
        if time() - start_time > time_limit:
            break
            
        # Small random perturbation
        perturbation = np.random.normal(0, 0.02, n_points)
        h_perturbed = project_to_feasible(best_h + perturbation, n_points)
        
        # Try to improve with local optimization
        try:
            # Simple gradient estimate
            eps = 1e-6
            grad = np.zeros_like(h_perturbed)
            base_c5 = evaluate_erdos_solution(h_perturbed, dx)
            for i in range(n_points):
                h_plus = h_perturbed.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (c5_plus - base_c5) / eps
            
            # Gradient step
            new_h = h_perturbed - 0.1 * grad
            new_h = project_to_feasible(new_h, n_points)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        except:
            pass
            
        # Occasionally reset with better solution 
        if np.random.random() < 0.05 and iteration % 10 == 0:
            h_perturbed = project_to_feasible(best_h + np.random.normal(0, 0.05, n_points), n_points)
            c5_candidate = evaluate_erdos_solution(h_perturbed, dx)
            if c5_candidate < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_candidate
    
    return best_h, best_c5

    # Multiple restarts
    for restart in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start with perturbed initial
        h_restart = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
