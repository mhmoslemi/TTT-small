import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with fewer points for faster optimization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations properly for all lags
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def grid_search(n_grid, num_samples=20):
        """Perform grid search at given resolution"""
        best_h = None
        best_C5 = float('inf')
        
        for _ in range(num_samples):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Generate random feasible point
            h_candidate = np.random.rand(n_grid)
            h_candidate = project_feasible(h_candidate)
            
            C5 = compute_C5(h_candidate)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_candidate.copy()
        
        return best_h, best_C5
    
    # Start with coarse grid
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Try grid search at different resolutions
    grid_sizes = [10, 15, 25, 35, 45]
    for n_grid in grid_sizes:
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Perform search at this resolution
        h_candidate, C5_candidate = grid_search(n_grid, 10)
        if C5_candidate < C5_best:
            C5_best = C5_candidate
            h_best = h_candidate.copy()
            
        # Refine around current best, but only if we have time
        if time.time() - start_time > budget_s - 1.0:
            break
    
    # Final fine-tuning with local search
    def local_descent(h_initial, max_iter=30):
        h_current = h_initial.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Try to improve by perturbing each component
            h_new = h_current.copy()
            idx = np.random.randint(0, len(h_current))
            perturb = np.random.uniform(-0.05, 0.05) 
            h_new[idx] += perturb
            h_new = project_feasible(h_new)
            
            C5_new = compute_C5(h_new)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
                h_current = h_new.copy()
        
        return best_h, best_C5
    
    # Do several local searches from current best
    for _ in range(10):
        if time.time() - start_time > budget_s - 0.1:
            break
        h_new, C5_new = local_descent(h_best, max_iter=20)
        if C5_new < C5_best:
            C5_best = C5_new
            h_best = h_new.copy()
    
    h_best = project_feasible(h_best)
    C5_best = compute_C5(h_best)
    
    return list(h_best), C5_best, n_points
