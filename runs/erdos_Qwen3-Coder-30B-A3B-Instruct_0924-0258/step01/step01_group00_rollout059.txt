import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def augmented_lagrangian(h, n_points, dx, alpha):
    """Augmented Lagrangian for maximizing alpha subject to A(k) >= alpha"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Maximize alpha subject to correlation constraints
    violation = np.minimum(0, alpha - corr[:n_points])  # Only consider first half of corr
    penalty = 100 * np.sum(violation**2)
    return -alpha + penalty

def solve_subproblem(h, n_points, dx, alpha):
    """Solve subproblem to minimize max correlation"""
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx  # Negative because we want to maximize
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 30, 'ftol': 1e-6})
        if res.success:
            h_res = project_to_feasible(res.x, n_points)
            return h_res, -obj(h_res)
    except:
        pass
    return h, evaluate_erdos_solution(h, dx)

def pattern_perturb(h, strength=0.1):
    """Perturb pattern of h with some noise"""
    noise = np.random.normal(0, strength, len(h))
    h_pert = h + noise
    return np.clip(h_pert, 0, 1)
