import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def optimized_initial_h(n_points):
    """Construct better initial points with a more targeted approach"""
    # Use piecewise constant construction with optimized step positions
    h = np.ones(n_points) * 0.5
    
    # Add more structure: two pulses
    pulse_width = n_points // 3
    if pulse_width > 0:
        h[:pulse_width] = 1.0
        h[-pulse_width:] = 1.0
    
    # Normalize to sum = n_points/2
    h = project_to_feasible(h, n_points)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Reference initial construction if exists
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dt = 2.0 / n_points
        c5 = evaluate_erdos_solution(h_start, dt)
        best_h, best_c5, best_n_points = h_start.copy(), c5, n_points
    else:
        # Default construction
        n_points = 30
        h_start = optimized_initial_h(n_points)
        dt = 2.0 / n_points
        c5 = evaluate_erdos_solution(h_start, dt)
        best_h, best_c5, best_n_points = h_start.copy(), c5, n_points

    # Resolutions to try
    resolutions = [25, 30, 35, 40, 45]
    
    # Time spent so far
    elapsed = time() - start_time
    
    for n_points in resolutions:
        if elapsed > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        # Interpolate to target size
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Mutate to get fresh start
        h_perturbed = project_to_feasible(h_start + np.random.normal(0, 0.03, n_points), n_points)
        
        # Multiple restarts
        for _ in range(3):
            # First try direct optimization if we have time
            try:
                # SLSQP with gradient (using a soft approximation)
                def obj(h):
                    corr = np.correlate(h, 1 - h, mode='full')
                    # Want to minimize maximum correlation
                    return -np.max(corr) * dx  # Negative because minimize
                
                # Same object but add equality constraint
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(obj, h_perturbed, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 100, 'ftol': 1e-8})
                
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
            except Exception as e:
                pass
                
            # Perturb for next restart
            h_perturbed = project_to_feasible(best_h + np.random.normal(0, 0.05, n_points), n_points)
            
            elapsed = time() - start_time
            if elapsed > budget_s * 0.95:
                break
                
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
