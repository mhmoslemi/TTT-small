import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    np.random.seed(seed)
    
    start_time = time()

    resolutions = [15, 30, 45, 60, 75]
    
    # Start with initial construction or a simple construction

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    for i, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate if necessary
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time limit allocation per resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 50)
        
        # Define objective and constraints for this resolution
        def obj(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return -np.max(corr)  # Maximize the maximum overlap (minimize -max)

        def constraint_func(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr)  # We want to maximize this, so constraint is >= some value
        
        # Initialize with adaptive perturbations
        best_h_level = h_start.copy()
        best_c5_level = evaluate_erdos_solution(h_start, dx)

        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Local perturbation
            h_perturbed = project_to_feasible(best_h_level + np.random.normal(0, 0.1, n_points) * np.exp(-restart), n_points)
            
            # No bounds or constraints in scipy minimize - handle them via projected update
            try:
                res = minimize(obj, h_perturbed, method='L-BFGS-B', 
                               bounds=[(0, 1) for _ in range(n_points)], 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5_level:
                        best_c5_level = c5_opt
                        best_h_level = h_opt.copy()
            except Exception:
                pass
            
            # Additional random perturbation for exploration
            h_random = project_to_feasible(best_h_level + np.random.normal(0, 0.03, n_points), n_points)
            c5_random = evaluate_erdos_solution(h_random, dx)
            if c5_random < best_c5_level:

            best_c5 = best_c5_level

    return (list(best_h), best_c5, best_n_points)
