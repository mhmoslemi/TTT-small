import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_relaxation(h, n_points, dx):
    """Use convex relaxation to approximate the maximum overlap"""
    # Compute correlation matrix for all lags
    corr_vals = np.correlate(h, 1 - h, mode='full')
    # Use max relaxation as least squares approximation
    return np.sum(corr_vals**2) * dx**2

def genetic_mutation(h, n_points, mutation_rate=0.1):
    """Generate a mutated version of h"""
    h_new = h.copy()
    # Flip bit with probability mutation_rate
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            # Gaussian perturbation
            h_new[i] += np.random.normal(0, 0.1)
    return project_to_feasible(h_new, n_points)

def solve_convex_optimization(h, n_points, dx):
    """Solve convex relaxation to improve h"""
    h = project_to_feasible(h, n_points)
    # Create variables
    h_vars = cp.Variable(n_points)
    # Constraints
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2.0]
    # Objective: minimize quadratic relaxation of max correlation
    corr_vals = np.correlate(h, 1 - h, mode='full')
    obj = cp.sum_squares(corr_vals) * dx**2
    problem = cp.Problem(cp.Minimize(obj), constraints)
    try:
        problem.solve(solver=cp.CVXOPT, verbose=False)
        return h_vars.value
    except:
        return h

    best_c5 = evaluate_erdos_solution(h_base, dx)
    
    # Multi-resolution search with 3 levels
    resolutions = [15, 30, 45]
    for res_level, n_points_new in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_new = 2.0 / n_points_new
        
        # Interpolate if necessary
        if n_points_new != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_new), 
                                np.linspace(0, 2, n_points), 
                                h_base)
            h = project_to_feasible(h_interp, n_points_new)
        else:
            h = h_base.copy()
        
        # Genetic algorithm with convex optimization improvements
        population_size = 20
        population = []
        for _ in range(population_size):
            h_pop = genetic_mutation(h, n_points_new, 0.15)
            population.append(h_pop)

            fitness = []

            sorted_indices = np.argsort(fitness)
