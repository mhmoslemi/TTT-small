import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute difference carefully
    diff = target_sum - current_sum
    if diff > 0:  # Need to increase
        idx = np.argsort(h)
        for i in range(len(idx)-1, -1, -1):
            if diff <= 0:
                break
            j = idx[i]
            if h[j] < 1.0:
                update = min(diff, 1.0 - h[j])
                h[j] += update
                diff -= update
    else:  # Need to decrease
        idx = np.argsort(h)
        for i in range(len(idx)):
            if diff >= 0:
                break
            j = idx[i]
            if h[j] > 0.0:
                update = max(diff, -h[j])
                h[j] += update
                diff -= update
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Set dx
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize best solution
    best_h = h0.copy()
    best_c5 = evaluate(best_h, dx)
    
    # Initialize simplex for Nelder-Mead
    simplex_size = n_points + 2
    simplex = [h0.copy()]
    for i in range(simplex_size - 1):
        pert = np.random.normal(0, 0.05, n_points)
        h_new = h0 + pert
        h_new = project_to_feasible(h_new, target_sum)
        simplex.append(h_new)
    
    # Helper function for Nelder-Mead
    def obj_func(h):
        h = project_to_feasible(h, target_sum)
        return evaluate(h, dx)
    
    # Nelder-Mead optimization
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all simplex points
        f_vals = [obj_func(s) for s in simplex]
        # Order by function values
        order = np.argsort(f_vals)
        simplex = [simplex[i] for i in order]
        f_vals = [f_vals[i] for i in order]
        
        # Update best if improved
        if f_vals[0] < best_c5:
            best_c5 = f_vals[0]
            best_h = simplex[0].copy()
            last_improvement = iteration
        
        # Stop if no improvement for some time
        if iteration - last_improvement > 200:
            break
        
        # Nelder-Mead reflection/expansion/contraction
        x_bar = np.mean(simplex[:-1], axis=0)
        x_r = x_bar + (x_bar - simplex[-1])
        f_r = obj_func(x_r)
        
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r
        elif f_r < f_vals[0]:
            x_e = x_bar + 2 * (x_bar - simplex[-1])
            f_e = obj_func(x_e)
            if f_e < f_r:
                simplex[-1] = x_e
            else:
                simplex[-1] = x_r
        else:
            x_c = x_bar + 0.5 * (simplex[-1] - x_bar)
            f_c = obj_func(x_c)
            if f_c < f_vals[-1]:
                simplex[-1] = x_c
            else:
                # Reduction
                x0 = simplex[0]
                for i in range(1, len(simplex)):
                    simplex[i] = x0 + 0.5 * (simplex[i] - x0)
        
        # Occasionally add mutation
        if iteration % 50 == 0:
            for i in range(1, len(simplex)):
                if np.random.random() < 0.4:
                    pert = np.random.normal(0, 0.01, n_points)
                    simplex[i] = simplex[i] + pert
                    simplex[i] = project_to_feasible(simplex[i], target_sum)
    
    # Final evaluation of best
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
