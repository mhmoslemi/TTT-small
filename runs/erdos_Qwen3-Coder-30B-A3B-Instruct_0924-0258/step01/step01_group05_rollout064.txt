import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        # Compute correlations for all lags and return maximum
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    # Initialize
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    
    best_h = h_current.copy()
    best_C5 = C5_current
    
    # Annealing parameters
    T0 = 0.1
    cooling_rate = np.log(10) / budget_s
    T = T0
    min_T = 1e-5
    
    # Adaptive parameters
    sigma = 0.1
    acceptance_count = 0
    total_trials = 0
    
    # Main Simulated Annealing loop
    while time.time() - start_time < budget_s - 0.1:
        # Random perturbation
        h_new = h_current.copy()
        
        # Perturb top 20% of intervals by Gaussian noise
        # Compute which intervals contribute most to overlap
        corr = correlate(h_current, 1 - h_current, mode='full')
        # Find top contributing indices based on absolute correlation values
        # Not using max correlation directly, but based on large correlation values
        abs_corr = np.abs(corr)
        # Get indices where correlation is above some threshold (approximate)
        top_indices = np.argsort(abs_corr)[-int(0.2 * len(h_current)):][::-1]
        
        if len(top_indices) > 0:
            # Identify most significant intervals to perturb
            indices_to_change = np.random.choice(len(h_current), 
                                               size=min(5, len(h_current)//2), 
                                               replace=False)
            for i in indices_to_change:
                # Add noise to top contributing intervals
                if np.random.rand() < 0.8:
                    # Gaussian noise
                    h_new[i] += np.random.normal(0, sigma)
                else:
                    # Binary flip
                    h_new[i] = 1 - h_new[i]
        
        # Ensure h_new remains feasible
        h_new = project_feasible(h_new)
        
        # Compute new C5
        C5_new = compute_C5(h_new)
        
        # Accept or reject
        delta = C5_new - C5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, min_T)):
            h_current = h_new.copy()
            C5_current = C5_new
            acceptance_count += 1
            
            if C5_current < best_C5:
                best_C5 = C5_current
                best_h = h_current.copy()
        
        total_trials += 1
        
        # Update temperature
        if total_trials % 10 == 0:
            T = T0 * np.exp(-cooling_rate * (time.time() - start_time))
            
            # Adjust step size based on acceptance rate
            if total_trials > 0:
                acceptance_rate = acceptance_count / total_trials
                if acceptance_rate < 0.1:
                    sigma *= 1.1
                elif acceptance_rate > 0.4:
                    sigma *= 0.9
                acceptance_count = 0
                total_trials = 0
        
        # Occasionally restart with slightly different h
        if np.random.rand() < 0.01:
            h_current = project_feasible(h_current + np.random.normal(0, 0.05, n_points))
            C5_current = compute_C5(h_current)
            if C5_current < best_C5:
                best_C5 = C5_current
                best_h = h_current.copy()
    
    # Final validation
    final_h = project_feasible(best_h)
    final_C5 = compute_C5(final_h)
    
    return list(final_h), final_C5, n_points
