import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Initialize with a simple construction
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    # Try different resolutions
    resolutions = [30, 45, 60]
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        # Interpolate to new resolution
        if len(h_start) != n_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_orig, h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Multiple restarts with different perturbations
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slightly perturb the starting point
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = project_to_feasible(h_start + perturbation, n_points)
            
            # Optimization with SLSQP
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint(h):
                return np.sum(h) - n_points / 2
            
            cons = [{'type': 'eq', 'fun': constraint}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
            except:
                pass
            
            # Update the starting point for next restart
            h_start = project_to_feasible(best_h + np.random.normal(0, 0.02, n_points), n_points)
    
    return (list(best_h), best_c5, best_n_points)
