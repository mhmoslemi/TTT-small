import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def optimize_at_resolution(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Target to maximize the minimal correlation
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.min(corr) * dx  # Negative because we minimize
    
    def constraint_func(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.min(corr) * dx
    
    # Constraints: integral = 1 and h in [0,1]
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Gradient computation with finite differences
    def gradient(h):
        eps = 1e-6
        grad = np.zeros_like(h)
        base_obj = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            obj_plus = objective(h_plus)
            grad[i] = (obj_plus - base_obj) / eps
        return grad
    
    # Perturbation based restart
    restarts = 5
    for _ in range(restarts):
        if time() - start_time > time_limit:
            break
        # Start with a perturbed version of best
        h_perturbed = project_to_feasible(best_h + np.random.normal(0, 0.05, n_points), n_points)
        
        try:
            # Optimizer: SLSQP for better handling of constraints
            res = minimize(objective, h_perturbed, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           jac=gradient, options={'ftol': 1e-8, 'maxiter': 100})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
        
        # Perturb for next restart
        h_perturbed = project_to_feasible(best_h + np.random.normal(0, 0.02, n_points), n_points)
    
    return best_h, best_c5

    best_h = h_start.copy()

    # Use first resolution as basis for fine graining

    for i, n_points in enumerate(resolutions):

        # Interpolate if necessary
