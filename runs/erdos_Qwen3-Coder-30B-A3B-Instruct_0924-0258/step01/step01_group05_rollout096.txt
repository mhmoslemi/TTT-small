import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient for the k_max shift
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        for i in range(n):
            # Contribution to overlap when h[i] is at position i
            # and (1-h[i+k_max]) is at position i+k_max
            if i + k_max < n:
                grad[i] -= (1 - h_vec[i + k_max])
            if i - k_max >= 0:
                grad[i] -= h_vec[i - k_max]
        return grad
    
    def optimize_step(h_vec, max_iter=100):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        alpha = 0.1
        stagnation_count = 0
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Compute all overlaps
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_lag_idx = np.argmax(corr)
            max_lag = max_lag_idx - (len(h_current) - 1)
            
            # Compute gradient for the critical overlap
            grad = compute_gradient(h_current, abs(max_lag))
            
            # Try to update with current alpha
            candidate_h = h_current - alpha * grad
            candidate_h = project_feasible(candidate_h)
            new_C5 = compute_C5(candidate_h)
            
            if new_C5 < best_C5:
                h_current = candidate_h.copy()
                best_C5 = new_C5
                stagnation_count = 0
                alpha = 0.1  # Reset step size
            else:
                stagnation_count += 1
                if stagnation_count >= 10:
                    # Reduce step size
                    alpha *= 0.5
                    # Random perturbation
                    noise = np.random.normal(0, 0.02, len(h_current))
                    h_current += noise
                    h_current = project_feasible(h_current)
                    stagnation_count = 0
                    
        return h_current, best_C5
    
    h_best = h.copy()
    C5_best = compute_C5(h_best)
    
    # Main optimization loop
    for iter_count in range(200):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Perturb the current solution and optimize
        h_perturbed = h_best + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_feasible(h_perturbed)
        
        h_opt, C5_opt = optimize_step(h_perturbed, max_iter=100)
        
        if C5_opt < C5_best:
            C5_best = C5_opt
            h_best = h_opt.copy()
    
    # Final projection
    h_best = project_feasible(h_best)
    final_C5 = compute_C5(h_best)
    
    return list(h_best), final_C5, n_points
