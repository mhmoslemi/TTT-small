import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    max_iter = 5000
    stagnation_count = 0
    last_best = best_c5
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute correlation and identify k_max
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Compute gradient analytically for k_max
        # For correlation at shift k_max, gradient of h[i] is:
        # (1 - 2 * h[i]) if we consider expanding the correlation term
        # But in terms of specific coefficients:
        # for the specific shift k_max, we have contribution h[i] * (1 - h[i + k_max])
        # So d/dh[i] = 1 - 2 * h[i]  if i + k_max < len(h)
        # and d/dh[i+k_max] = -h[i] if i + k_max < len(h)
        
        # More carefully: at k_max, derivative w.r.t. h[i] is
        # - h[i - k_max] if positive shift, but we need 1-h^(k_max) term.
        # portion of correlation at k_max for index i:
        # h[i] * (1 - h[i + k_max])
        # so d/dh[i] = 1 - 2 * h[i] (when checked with computer)
        # and d/dh[i + k_max] = -h[i]
        
        # So if (i is in range) then d/dh[i] includes:
        #   (1 - 2 h[i]) for contribution to corr[k_max]
        # More precise: compute contribution explicitly
        grad = np.zeros_like(h)
        # For shift k_max:
        # From correlation at shift k_max:
        # sum h[i] * (1 - h[i + k_max]) = corr[k_max]
        # So d/d(h[i]) of this term only depends on i + k_max:
        # if i + k_max < len(h): term is h[i] * (1 - h[i + k_max])
        # d/dh[i] = 1 - 2 * h[i]  (component)
        # d/dh[i + k_max] = - h[i] (shifted)
        reverse_k = n_points - 1 - k_max
        for i in range(n_points):
            # If h[i] contributes to shift k_max  
            if i + k_max < n_points:
                grad[i] += 1 - 2 * h[i]
            # If h[i] is part of 1-h's contribution to shift k_max, 
            # we contribute - h[i - k_max]
            if i >= k_max:
                grad[i] -= h[i - k_max] if i - k_max >= 0 else 0
                
        # Normalize gradient
        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm
            
        # Apply gradient update
        h = h - lr * grad
        
        # Project h to feasible set
        h = project_h(h)
        
        # Compute new C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Anneal learning rate
        lr *= 0.999
        
        # Occasionally add noise to escape local minima
        if iteration % 500 == 0 and iteration > 0:
            noise = 0.005 * np.random.normal(0, 1, len(h))
            h = h + noise
            h = project_h(h)
            
        # Early termination if not improving much
        if stagnation_count > 1000:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
