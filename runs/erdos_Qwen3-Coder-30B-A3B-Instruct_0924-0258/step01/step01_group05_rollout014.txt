import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    T0 = 0.1
    cooling_rate = np.log(10) / budget_s
    sigma = 0.1
    min_temp = 1e-5
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def evaluate_solution(h_vec):
        h_vec = project_feasible(h_vec)
        return compute_C5(h_vec), h_vec
    
    # Initial solution
    best_C5, h_best = evaluate_solution(h)
    
    # Annealing loop
    T = T0
    time_left = budget_s - (time.time() - start_time)
    
    while time_left > 0.1:
        # Perturb solution
        h_new = h_best.copy()
        # Focus on top contributors
        corr = correlate(h_best, 1 - h_best, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h_best) - 1)
        
        # Random perturbation
        noise = np.random.normal(0, sigma, n_points)
        h_new += noise
        h_new = project_feasible(h_new)
        
        # Evaluate new solution
        new_C5, h_new = evaluate_solution(h_new)
        delta = new_C5 - best_C5
        
        # Accept or reject
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, min_temp)):
            h_best = h_new
            if new_C5 < best_C5:
                best_C5 = new_C5
        
        # Cooling schedule
        T = max(T0 * np.exp(-cooling_rate * (time.time() - start_time)), min_temp)
        
        # Adjust step size based on acceptance rate
        if T < T0 * 0.1:
            sigma = max(sigma * 0.95, 0.01)
        elif T > T0 * 0.5:
            sigma = min(sigma * 1.05, 0.5)
            
        time_left = budget_s - (time.time() - start_time)

    # Final check and return
    h_best = project_feasible(h_best)
    final_C5 = compute_C5(h_best)
    
    return list(h_best), final_C5, n_points
