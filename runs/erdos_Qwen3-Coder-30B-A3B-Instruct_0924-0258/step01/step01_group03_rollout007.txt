import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
        # Final adjustment if needed
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            diff = target_sum - current_sum
            # Distribute difference among elements
            idx = np.argsort(h)
            for i in range(len(idx)-1, -1, -1):
                if abs(diff) < 1e-12:
                    break
                j = idx[i]
                if h[j] < 1.0:
                    update = min(diff, 1.0 - h[j])
                    h[j] += update
                    diff -= update
    return h
