import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def admm_update_h(h, z, rho, dx):
    # H is updated by minimizing the augmented Lagrangian
    # We solve: min_h max_k corr(h, z) + (rho/2) ||h - z + u||^2
    # Since max_k corr(h, z) is difficult to optimize, we approximate it
    # with a surrogate or run optimization on a sampled cost.
    # We derive updated h via gradient descent on a proximal form
    # For simplicity, we use a subgradient approach with fixed step size.
    n = len(h)
    # Use a small gradient step to maintain Lipschitz condition
    alpha = 0.001 / rho if rho > 0 else 0.001
    # Analytic gradient of the correlation sum |corr(h, z)|
    # We can approximate this with finite differences or directly use a surrogate.
    # However, for efficiency, we perform simple gradient ascent approximation
    # For now, just do one gradient step on a simple cross term
    # Compute cross-correlation of h with z to get approximate gradient
    correlation = correlate(h, z, 'full')
    # This is a simplified version - full ADMM would require more careful handling
    h_new = h - alpha * (correlation[:n] - correlation[n-1::-1])
    h_new = np.clip(h_new, 0, 1)
    return h_new

def admm_update_z(h, z, u, rho, dx):
    # z = argmin_z max_k corr(h, z) + (rho/2) ||z - (1-h) + u||^2
    # This approximated by setting z = (1-h) - u (which is a closed form if we ignore max_k)
    # This is ATOMICITY in ADMM for this simple case
    z_new = (1 - h) - u
    return z_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_low = 10
    n_high = 100
    n_points = n_low
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)

    # Initialize other variables for ADMM
    z = 1 - h
    u = np.zeros_like(h)
    rho = 1.0
    T = 0.1
    cooling = 0.99
    iter_per_temp = 500
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5
    
    # ADMM iteration counter
    iter_count = 0
    max_iter_per_res = 200
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # ADMM iterations at current resolution
        for _ in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            # Update h using gradient ascent to minimize correlation with z  
            h_new = admm_update_h(h, z, rho, dx)
            # Update z using closed form constraint
            z_new = admm_update_z(h_new, z, u, rho, dx)
            # Update dual variables
            u_new = u + (h_new - z_new)
            
            # Project h to constraints
            h_new = project_to_constraints(h_new, dx)
            
            # Evaluate
            c5_new = evaluate(h_new, dx)
            
            # Accept change if better or probabilistically
            delta = c5_new - last_c5
            if delta < 0 or np.random.uniform() < np.exp(-delta / T):
                h = h_new
                z = z_new
                u = u_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
            
            iter_count += 1
            
        # Cool down
        T *= cooling
        
        # Resolution refinement
        if n_points < n_high and time.time() - start_time < budget_s * 0.9:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            z = 1 - h
            u = np.zeros_like(h)
            last_c5 = evaluate(h, dx)
            # Reset temperature for new resolution, but faster convergence
            T = 0.1
            
        elif T < 1e-6:
            break
    
    return best_h.tolist(), best_c5, len(best_h)
