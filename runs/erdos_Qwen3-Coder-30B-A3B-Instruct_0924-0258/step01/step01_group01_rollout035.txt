import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with lower resolution
    n_low = 10
    n_high = 70
    n_points = n_low
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best ever
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # ADMM params
    rho = 1.0
    max_iter = 100
    alpha = 0.01
    
    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # ADMM iterations
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Update h (minimize with fixed z)
            # This requires solving a least squares problem with constraints
            # The penalty term in the objective is 
            # 0.5 * rho * ||h - (1-z) + lambda/rho||^2  
            # So, h = (1-z + lambda/rho) / (1 + rho)
            
            # Expected z from current h (1-h)
            z = 1 - h
            
            # ADMM update for h
            h_new = project_to_constraints(1-z, dx)
            h_new = np.clip(h_new, 0, 1)
            
            # For better convergence, we do gradient descent type update
            h_new = h + alpha * (1 - z) 
            h_new = project_to_constraints(h_new, dx)
            h_new = np.clip(h_new, 0, 1)
            
            # Update z 
            z_new = 1 - h_new
            # Multiplier update
            h = h_new
            z = z_new
            
            # Evaluate
            c5_new = evaluate(h, dx)
            
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                
            iter_count += 1
        
        # Check resolution refinement  
        if n_points < n_high:
            if best_c5 < 0.9 * evaluate(best_h, dx):  # Improve significantly?
                h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                                  np.linspace(0, 2, n_points), best_h)
                h = h_long
                n_points = 2 * n_points
                dx = 2.0 / n_points
                h = project_to_constraints(h, dx)
                best_c5 = evaluate(h, dx)
            else:
                break  
        else:
            break

    return best_h.tolist(), best_c5, len(best_h)
