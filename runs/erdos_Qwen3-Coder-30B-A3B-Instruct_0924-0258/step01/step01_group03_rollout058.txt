import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy bounds [0,1] and sum constraint."""
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial projection
    h = project_to_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    lr = 0.01
    stagnation_count = 0
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s:
        # Evaluate
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient
        grad = np.zeros_like(h)
        n = len(h)
        if k_max < n:
            # h[i] contributes to term h[i](1-h[i+k])
            for i in range(n):
                j = (i + k_max) % n
                grad[i] = 1 - 2 * h[j]
        else:
            # h[i] contributes to term h[i](1-h[i-(n-(k_max-n))])
            k_shift = k_max - n
            for i in range(n):
                j = (i - k_shift) % n
                grad[i] = 1 - 2 * h[j]
        
        # Update
        h = h - lr * grad
        
        # Project to feasible region
        h = project_to_feasible(h, target_sum)
        
        # Evaluate
        current_c5 = evaluate(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count >= 50:
                # Add small noise
                noise = np.random.uniform(-1e-3, 1e-3, n_points)
                h = h + noise
                h = project_to_feasible(h, target_sum)
                stagnation_count = 0
        
        # Adaptive learning rate
        if iteration - last_improvement > 100:
            lr *= 0.9
            last_improvement = iteration
            
        iteration += 1
        
        # Check for numerical issues
        if np.any(np.isnan(h)) or np.any(np.isinf(h)):
            h = project_to_feasible(np.ones(n_points) * 0.5, target_sum)
    
    return (best_h.tolist(), float(best_c5), n_points)
