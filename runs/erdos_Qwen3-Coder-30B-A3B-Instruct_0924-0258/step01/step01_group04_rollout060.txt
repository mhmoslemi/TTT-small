import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.random.uniform(0.4, 0.6, n_points)
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def genetic_algorithm():
        population_size = 50
        generations = 1000
        mutation_rate = 0.1
        crossover_rate = 0.8
        elite_count = 10
        
        # Initialize population
        population = []
        for _ in range(population_size):
            ind = np.random.uniform(0.4, 0.6, n_points)
            ind = project_feasible(ind)
            population.append(ind)
        
        best_h = h.copy()
        best_C5 = compute_overlap(best_h)
        
        for gen in range(generations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate fitness
            fitness = []
            for ind in population:
                C5 = compute_overlap(ind)
                fitness.append(C5)
            
            # Sort by fitness (lower is better)
            sorted_indices = np.argsort(fitness)
            population = [population[i] for i in sorted_indices]
            fitness = [fitness[i] for i in sorted_indices]
            
            # Check for improvement
            if fitness[0] < best_C5:
                best_C5 = fitness[0]
                best_h = population[0].copy()
            
            # Selection (tournament)
            def tournament_selection(tournament_size=3):
                tournament = np.random.choice(len(population), tournament_size)
                winner = tournament[np.argmin([fitness[i] for i in tournament])]
                return population[winner].copy()
            
            new_population = population[:elite_count]  # Elitism
            
            # Generate offspring
            while len(new_population) < population_size:
                parent1 = tournament_selection()
                parent2 = tournament_selection()
                
                if np.random.rand() < crossover_rate:
                    # Single point crossover
                    crossover_point = np.random.randint(1, n_points)
                    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                else:
                    child1 = parent1.copy()
                    child2 = parent2.copy()
                
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child1[i] += np.random.normal(0, 0.05)
                    if np.random.rand() < mutation_rate:
                        child2[i] += np.random.normal(0, 0.05)
                
                # Project to feasible region
                child1 = project_feasible(child1)
                child2 = project_feasible(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            
            population = new_population
            
            # Occasionally run local search on top individuals
            if gen % 50 == 0:
                top_individuals = population[:5]
                for ind in top_individuals:
                    try:
                        # Local optimization around this point
                        bounds = [(0, 1) for _ in range(n_points)]
                        res = minimize(
                            lambda x: compute_overlap(x),
                            ind,
                            method='L-BFGS-B',
                            bounds=bounds,
                            options={'maxiter': 20}
                        )
                        if res.success:
                            local_h = project_feasible(res.x)
                            local_C5 = compute_overlap(local_h)
                            if local_C5 < best_C5:
                                best_C5 = local_C5
                                best_h = local_h.copy()
                    except:
                        pass
                        
        return best_h, best_C5

    best_h, best_C5 = genetic_algorithm()
    return (best_h.tolist(), best_C5, n_points)
