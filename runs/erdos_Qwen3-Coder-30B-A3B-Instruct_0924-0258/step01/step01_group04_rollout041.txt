import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Helper to compute C5
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Multi-resolution optimization
    resolutions = [5, 10, 20, 40, n_points]
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Gradient-based optimization
    def adam_optimizer(h_vec, max_iter=1000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        m = np.zeros_like(h_vec)
        v = np.zeros_like(h_vec)
        for t in range(1, max_iter + 1):
            if time.time() - start_time > budget_s - 1:
                break
            # Compute overlap and identify top contributing indices
            conv = np.correlate(h_vec, 1 - h_vec, mode='full')
            mid = len(conv) // 2
            overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
            top_k = np.argsort(overlaps)[-5:][::-1]
            
            # Approximate gradient based on dominant overlaps
            gradient = np.zeros_like(h_vec)
            eps_val = 1e-4
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps_val
                h_plus = project_h(h_plus)
                C_plus = compute_C5(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[i] -= eps_val
                h_minus = project_h(h_minus)
                C_minus = compute_C5(h_minus)
                
                gradient[i] = (C_plus - C_minus) / (2 * eps_val)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * gradient
            v = beta2 * v + (1 - beta2) * gradient**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_vec = h_vec - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            h_vec = project_h(h_vec)
            
            # Check for improvement
            current_C5 = compute_C5(h_vec)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_vec.copy()
                
        return h_vec
    
    # Optimization at each resolution level
    current_h = h.copy()
    for i, res in enumerate(resolutions[:-1]):
        if time.time() - start_time > budget_s - 1:
            break
        # Initialize low-res h
        current_h = np.interp(np.linspace(0, 1, res), np.linspace(0, 1, len(current_h)), current_h)
        current_h = project_h(current_h)
        # Optimize at this resolution
        current_h = adam_optimizer(current_h, max_iter=200, lr=0.01)
        # Upscale to next resolution
        if i < len(resolutions) - 2:
            next_h = np.interp(np.linspace(0, 1, resolutions[i+1]), np.linspace(0, 1, len(current_h)), current_h)
            current_h = project_h(next_h)
    
    # Final high-res optimization
    current_h = adam_optimizer(current_h, max_iter=2000, lr=0.01)
    
    final_C5 = compute_C5(current_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = current_h.copy()
    
    return (best_h.tolist(), best_C5, n_points)
