import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization with more structure
        n_points = 45
        h = np.zeros(n_points)
        # Place a step in the middle to make it more structured
        mid = n_points // 2
        h[:mid] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec

    def perturb_h(h_vec, amount=0.05):
        noise = np.random.normal(0, amount, len(h_vec))
        return project_feasible(h_vec + noise)

    def grid_search_level(h_base, n, max_candidates=50):
        candidates = []
        for _ in range(max_candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
            h_candidate = perturb_h(h_base, 0.1)
            c5 = compute_C5(h_candidate)
            candidates.append((c5, h_candidate))
        candidates.sort(key=lambda x: x[0])
        return candidates[:max(int(max_candidates * 0.2), 1)]

    # Start with the given solution
    h_best = h.copy()
    C5_best = compute_C5(h_best)

    # Try more refined search
    grid_levels = [5, 10, 15, 20, 25, 30, 35, 40, 45]
    # Current working point
    current_h = h_best.copy()
    current_C5 = C5_best

    for level_n in grid_levels:
        if time.time() - start_time > budget_s - 0.1:
            break
            
        if level_n <= 15:
            num_candidates = 50
            perturb_amount = 0.2
        elif level_n <= 30:
            num_candidates = 30
            perturb_amount = 0.1
        else:
            num_candidates = 20
            perturb_amount = 0.05
            
        # Generate candidates
        candidates = []
        for _ in range(num_candidates):
            if time.time() - start_time > budget_s - 0.1:
                break
            candidate_h = perturb_h(current_h, perturb_amount)
            c5 = compute_C5(candidate_h)
            candidates.append((c5, candidate_h))
        
        if candidates:
            candidates.sort(key=lambda x: x[0])
            # Keep the best ones
            for c5, h_cand in candidates[:min(10, len(candidates))]:
                if c5 < current_C5:
                    current_C5 = c5
                    current_h = h_cand.copy()
                    if c5 < C5_best:
                        C5_best = c5
                        h_best = h_cand.copy()
                        
        # Refine with a more focused approach
        for _ in range(3):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Perturb to create a new candidate
            h_new = perturb_h(current_h, 0.05)
            c5_new = compute_C5(h_new)
            if c5_new < current_C5:
                current_C5 = c5_new
                current_h = h_new.copy()
                if c5_new < C5_best:
                    C5_best = c5_new
                    h_best = h_new.copy()
    
    # Final projection and evaluation
    h_final = project_feasible(h_best)
    C5_final = compute_C5(h_final)
    
    return list(h_final), C5_final, n_points
