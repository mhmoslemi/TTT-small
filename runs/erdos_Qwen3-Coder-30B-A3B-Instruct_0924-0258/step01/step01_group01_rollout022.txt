import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_step(h, dx, lr=1e-3):
    n = len(h)
    # Compute cross-correlation to find critical lag
    corr = correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    # The actual lag is max_idx - (n-1)
    lag = max_idx - (n-1)
    
    # Approximate gradient based on the most critical lag
    # For lag k, the contribution to the objective is int h(x)(1-h(x+k)) dx
    pad = n-1
    h_padded = np.pad(h, pad, mode='constant', constant_values=0)
    grad = np.zeros(n)
    
    for i in range(n):
        # Contribution at lag -k
        if i + lag >= 0 and i + lag < n:
            grad[i] = 1 - 2 * h[i + lag]
        else:
            grad[i] = 0
            
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_constraints(h_new, dx)
    return h_new

    start_time = time.time()

    n_high = 70
    n_points = n_low

    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    iter_count = 0
    lr = 1e-3
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perform gradient steps
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            h_new = gradient_step(h, dx, lr)
            c5_new = evaluate(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                lr = min(lr * 1.1, 1e-1)
            else:
                lr = max(lr * 0.99, 1e-6)
                
            h = h_new
            iter_count += 1
            
        # Adaptive refinement
        if n_points < n_high and time.time() - start_time < budget_s * 0.9:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), 
                              np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            lr = 1e-3
        else:
            break
            
    return best_h.tolist(), best_c5, len(best_h)
