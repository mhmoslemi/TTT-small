import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Precompute indices for fast correlation computation
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value over all lags, scaled by dx
        return np.max(corr) * dx
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1, sum(h) = target_sum
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * target_sum / h_sum
        return h_vec
    
    def local_search(h_vec, max_iter=100):
        h_current = h_vec.copy()
        best_h = h_current.copy()
        best_C5 = compute_C5(h_current)
        
        improvement_count = 0
        stagnation_count = 0
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            h_old = h_current.copy()
            h_new = h_current.copy()
            
            # Try coordinate descent for each dimension
            for i in range(n_points):
                if time.time() - start_time > budget_s - 0.1:
                    break
                    
                # Try small positive and negative perturbations
                perturbations = [0.05, -0.05, 0.1, -0.1]
                best_perturbed_h = h_current.copy()
                best_perturbed_C5 = best_C5
                
                for delta in perturbations:
                    test_h = h_current.copy()
                    test_h[i] += delta
                    test_h = project_feasible(test_h)
                    test_C5 = compute_C5(test_h)
                    
                    if test_C5 < best_perturbed_C5:
                        best_perturbed_C5 = test_C5
                        best_perturbed_h = test_h.copy()
                
                # Accept the best improvement
                if best_perturbed_C5 < best_C5:
                    h_current = best_perturbed_h.copy()
                    best_C5 = best_perturbed_C5
                    improvement_count += 1
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
                # Check if algorithm is stuck
                if stagnation_count > 30:
                    # Escalate to random perturbation
                    h_current = project_feasible(h_current + np.random.normal(0, 0.02, n_points))
                    stagnation_count = 0
                
            # If no significant improvement, do a random restart
            if np.allclose(h_current, h_old, atol=1e-5):
                h_current = project_feasible(h_current + np.random.normal(0, 0.05, n_points))
                stagnation_count = 0
                
            if best_C5 < compute_C5(best_h):
                best_h = h_current.copy()
            
            # Occasionally restart if too many iterations without improvement
            if improvement_count == 0 and _ > 10:
                h_current = project_feasible(h_current + np.random.normal(0, 0.05, n_points))
                
        return best_h, best_C5
    
    # Initialize candidates using Latin hypercube sampling
    candidates = []
    n_candidates = 5
    
    for i in range(n_candidates):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Generate initial candidate using LHS sampling
        h_candidate = np.random.rand(n_points)
        # Adjust to sum to target_sum
        h_candidate = h_candidate * target_sum / np.sum(h_candidate)
        # Clip to [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        candidates.append(h_candidate)
        
    # Evaluate all candidates
    best_C5 = float('inf')
    h_best = h.copy()
    
    for i, candidate in enumerate(candidates):
        candidate = project_feasible(candidate)
        C5_candidate = compute_C5(candidate)
        if C5_candidate < best_C5:
            best_C5 = C5_candidate
            h_best = candidate.copy()
    
    # Local optimization starting from initial solution
    h_current = h.copy()
    h_current = project_feasible(h_current)
    C5_current = compute_C5(h_current)
    
    if C5_current < best_C5:
        best_C5 = C5_current
        h_best = h_current.copy()
    
    # Main optimization loop
    max_iter = 1000
    improvement_threshold = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Try several local searches from current best point
        for _ in range(2):
            if time.time() - start_time > budget_s - 0.1:
                break
            # Add slight noise and perform local search
            h_perturbed = h_best + np.random.normal(0, 0.02, n_points)
            h_perturbed = project_feasible(h_perturbed)
            h_opt, C5_opt = local_search(h_perturbed, max_iter=50)
            
            if C5_opt < best_C5:
                best_C5 = C5_opt
                h_best = h_opt.copy()
                
        # Periodically restart with random initialization
        if iteration % 20 == 0:
            h_random = project_feasible(np.random.rand(n_points))
            h_rand_opt, C5_rand = local_search(h_random, max_iter=30)
            if C5_rand < best_C5:
                best_C5 = C5_rand
                h_best = h_rand_opt.copy()
    
    h_best = project_feasible(h_best)
    final_C5 = compute_C5(h_best)
    
    return list(h_best), final_C5, n_points
