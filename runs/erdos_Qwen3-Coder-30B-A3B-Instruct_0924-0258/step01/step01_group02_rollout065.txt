import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default: uniform distribution scaled to sum to n/2
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    # Ensure h sums to n/2
    dx = 2.0 / len(h)
    target_sum = len(h) / 2.0
    h = h * target_sum / np.sum(h)
    
    # Compute current C5 value
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Project h to feasible set: clip to [0,1] and rescale to sum to target_sum
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Function to compute overlap at specific lag k
    def overlap_at_lag(h_vec, k):
        # This is the contribution to overlap integral at lag k
        n = len(h_vec)
        if k >= n:
            return 0.0
        # Convolution integral component over continuous domain
        # Using discrete approximation: 2/n factors  
        if k == 0:
            # When k=0: h[i] * (1-h[i]) for all i
            # But only for first n-k terms since the kernel is flipped by definition
            return np.sum(h_vec[:n-k] * (1 - h_vec[:n-k])) * dx
        else:
            # When k>0, we take h[i]* (1-h[i+k]) for valid i
            # This is the dot product of h[i]*1-h[i+k] for i in range(0, n-k)
            return np.sum(h_vec[:n-k] * (1 - h_vec[k:])) * dx
    
    # Update to include projection in the optimization step
    def evaluate_overlap(h_vec):
        max_overlap = 0.0
        n = len(h_vec)
        # Go through all lags (from 0 to n-1) 
        for k in range(n):
            overlap = overlap_at_lag(h_vec, k)
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # New approach: Use projected gradient descent, with a smarter procedure
    # Simple optimization to refine the solution:
    # Refine with scipy minimize, using SLSQP method
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    n_points = len(h)
    max_iter = 2000
    iteration = 0
    
    # Use a smoother cost for gradient-based method
    def smooth_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') 
        return np.max(corr) * dx
    
    def smooth_c5_with_regularization(h_vec, gamma=1e-6):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')  
        max_val = np.max(corr) * dx
        # Add regularization to encourage sparsity and stability
        regularization = gamma * np.sum(h_vec * (1 - h_vec))
        return max_val + regularization
    
    # The optimization variables (projected to [0,1] and unit sum)
    def obj_func(h_vec):
        h_vec = project_h(h_vec) 
        return smooth_c5(h_vec)
    
    # Use scipy.optimize with bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    ]
    
    try:
        res = minimize(
            obj_func,
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-10, 'gtol': 1e-8},
            callback=lambda x: None
        )
        if res.success and res.fun < best_c5:
            h = project_h(res.x)
            best_c5 = smooth_c5(h)
            best_h = h.copy()
    except Exception as e:
        # Fallback to projection
        pass
    
    # Refinement step: also project gradient
    lr = 1e-3
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute gradient approximatively or use bounded projected steps
        # For now, apply a few simple iterations of projected gradient descent
        # while keeping all points feasible
        
        # Better refinement: a local search in neighborhood
        corr0 = np.correlate(h, 1-h, mode='full')
        max_k = np.argmax(corr0)
        
        # Simple exploration around maximum deviation
        pert_h = h.copy()
        eps = 1e-3 * np.random.rand(n_points)
        
        # Add a small noise and project
        pert_h += eps
        pert_h = project_h(pert_h)
        
        c5_test = smooth_c5(pert_h)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = pert_h.copy()
        
        # Adaptively adjusting step size
        lr *= 0.9995
        
        # Occasionally add more noise to escape local minima
        if i % 50 == 0:
            noise = 0.01 * np.random.uniform(-1, 1, len(h))
            h = h + noise
            h = project_h(h)
            
        iteration += 1
    
    # Final compute of C5 using high-precision match to prior solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
