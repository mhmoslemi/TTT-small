import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_descent_step(h, dx, step_size=0.01):
    n = len(h)
    # compute gradient analytically for max_k int h(x)(1-h(x+k)) dx
    # this avoids computing the full convolution
    # gradient dC5/dh_i = (1 - 2*h_i) * dx (approximately)
    grad = (1 - 2 * h) * dx
    h_new = h - step_size * grad
    return project_to_constraints(h_new, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points_list = [10, 20, 40, 70]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Initialize with initial_h_values or random
        if 'initial_h_values' in globals() and n_points <= len(initial_h_values):
            h = np.array(initial_h_values[:n_points])
        else:
            h = np.random.uniform(0, 1, n_points)
            
        h = project_to_constraints(h, dx)
        
        # Gradient descent iterations
        max_iter = 5000
        iter_count = 0
        last_c5 = evaluate(h, dx)
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            h_new = gradient_descent_step(h, dx, step_size=0.01)
            c5_new = evaluate(h_new, dx)
            
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                
            iter_count += 1
            
        # Also attempt a few more gradient steps on the final result
        for i in range(100):
            if time.time() - start_time > budget_s:
                break
            h_new = gradient_descent_step(h, dx, step_size=0.005)
            c5_new = evaluate(h_new, dx)
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
        
    return best_h.tolist(), best_c5, len(best_h)
