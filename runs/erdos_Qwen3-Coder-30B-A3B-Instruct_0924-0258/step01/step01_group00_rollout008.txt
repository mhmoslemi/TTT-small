import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_correlations(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def augmented_lagrangian_objective(h, alpha, constraints, penalty_param=10.0):
    obj = -alpha
    # Add penalty for constraint violations
    for constr_val in constraints:
        if constr_val < 0:
            obj += penalty_param * constr_val**2
    return obj

def optimize_at_resolution(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Precompute all correlation values for current h
    def compute_all_constraints(h):
        corr = compute_correlations(h, dx)
        # Only consider valid shifts - overlapping regions
        valid_corr = corr[len(corr)//2:]  # Left to right shifts
        return valid_corr
    
    def objective(h):
        corr = compute_all_constraints(h)
        return -np.min(corr)
    
    def constraint_func(h):
        corr = compute_all_constraints(h)
        return np.min(corr)
    
    cons = [{'type': 'ineq', 'fun': lambda h: constraint_func(h) - 0.3}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use SLSQP optimizer
    max_iter = 200
    for i in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        try:
            res = minimize(objective, h_start, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
            
        # Slight perturbation for restarts
        h_start = project_to_feasible(best_h + np.random.normal(0, 0.01, n_points), n_points)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    resolutions = [30, 45, 60]
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        n0 = 30
        h_start = np.ones(n0) / 2
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate starting point
        if len(h_start) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 150)
        
        # Multiple restarts
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
        
            h_restart = h_start.copy()
            # Add controlled noise for exploration
            noise_scale = 0.05
            if restart > 0:
                h_restart = project_to_feasible(h_start + np.random.normal(0, noise_scale, n_points), n_points)
            
            h_opt, c5_opt = optimize_at_resolution(h_restart, n_points, dx, time_per_res / 3)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
        
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
