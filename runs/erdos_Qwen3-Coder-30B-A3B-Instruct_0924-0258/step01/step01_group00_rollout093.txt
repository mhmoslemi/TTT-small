import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def convex_maximize_A(h, n_points, dx):
    # Build correlation matrix for all shifts
    corr = np.correlate(h, h, mode='full')
    # Corresponding indices
    full_len = 2 * n_points - 1
    center = n_points - 1
    # Get all A(k)
    A_values = corr[center:]  # Only shifts k >= 0
    # We want to maximize min(A_k)
    # Convert to CVXPY problem
    # We'll use a penalty approach to approximate max constraint
    
    # Maximize a lower bound on the minimum A(k)
    # We'll use a quadratic penalty for the minimum
    min_val = cp.Minimize(-cp.sum(A_values) / len(A_values))  # approximate -min(A_values)
    return min_val

def genetic_mutation(h, n_points):
    # Two-point crossover + Gaussian mutation
    h_mut = h.copy()
    if np.random.random() < 0.7:  # Crossover
        i1, i2 = sorted(np.random.choice(n_points, 2, replace=False))
        h_mut[i1:i2] = h[i1:i2]
    # Mutation
    noise = np.random.normal(0, 0.05, n_points)
    h_mut += noise
    return h_mut

def polynomial_fit_mutation(h, n_points):
    # Generate polynomial fit to adapt shape
    x = np.linspace(0, 2, n_points)
    coeffs = np.polyfit(x, h, 3)  # cubic fit
    h_fit = np.polyval(coeffs, x)
    # Add noise to fit
    h_fit += np.random.normal(0, 0.01, n_points)
    return h_fit

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Initialize with simple pattern
        n_points = 30
        h_start = np.ones(n_points) * 0.5
        # Apply a gentle saw-tooth pattern
        x_vals = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(2 * np.pi * x_vals)
        h_start = np.clip(h_start, 0, 1)
    
    # Adaptive optimization with coarse-to-fine approach
    # Resolution levels
    res_levels = [15, 30, 45, 60]
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / len(h_start))
    best_n_points = len(h_start)
    
    for res_n in res_levels:
        if time() - start_time > 0.95 * budget_s:
            break
        dx = 2.0 / res_n
        
        # Interpolate
        if len(h_start) != res_n:
            h_interp = np.interp(np.linspace(0, 2, res_n),
                                np.linspace(0, 2, len(h_start)),
                                h_start)
            h_start = project_to_feasible(h_interp, res_n)
        
        # Genetic algorithm with hill climbing
        population_size = 10
        population = []
        for i in range(population_size):
            h_temp = project_to_feasible(h_start + np.random.normal(0, 0.05, res_n), res_n)
            population.append(h_temp)
        
        # GA iterations
        ga_iters = 70
        for iter in range(ga_iters):
            if time() - start_time > 0.95 * budget_s:
                break
            # Evaluate each solution
            fitness = []
            for h in population:
                c5 = evaluate_erdos_solution(h, dx)
                fitness.append(c5)
            
            # Sort by fitness (lower is better)
            sorted_indices = np.argsort(fitness)
            sorted_pop = [population[i] for i in sorted_indices]
            sorted_fit = [fitness[i] for i in sorted_indices]
            
            if sorted_fit[0] < best_c5:
                best_h = sorted_pop[0].copy()
                best_c5 = sorted_fit[0]
                best_n_points = res_n
            
            # Select top 30%
            elite_size = population_size // 3
            elite = sorted_pop[:elite_size]
            
            # Generate offspring
            new_population = elite.copy()
            
            for i in range(population_size - elite_size):
                parent1 = elite[np.random.randint(0, elite_size)]
                parent2 = elite[np.random.randint(0, elite_size)]
                
                # Crossover and mutation
                child = np.zeros(res_n)
                start, end = sorted(np.random.choice(res_n, 2, replace=False))
                child[:start] = parent1[:start]
                child[start:end] = parent2[start:end]
                child[end:] = parent1[end:]
                
                # Apply mutation
                if np.random.random() < 0.7:
                    child = genetic_mutation(child, res_n)
                else:
                    child = polynomial_fit_mutation(child, res_n)
                
                # Project to feasible
                child = project_to_feasible(child, res_n)
                new_population.append(child)
            
            population = new_population
        
        h_start = best_h.copy()

    return (list(best_h), best_c5, best_n_points)
