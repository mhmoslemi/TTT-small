import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def smooth_max(x, beta=10.0):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(beta * x))) / beta

def compute_correlations(h, dx):
    """Compute all correlations between h and (1-h)"""
    corr = correlate(h, 1-h, mode='full')
    # Normalize by dx
    return corr * dx

def evaluate_smooth(h, dx, beta=10.0):
    """Evaluate smooth approximation to C5"""
    corr = compute_correlations(h, dx)
    # Only consider valid lags (non-zero)
    # From correlate with mode='full', we have 2*n-1 elements
    n = len(h)
    # Extract the main correlation values that we need
    # This corresponds to lags from -(n-1) to (n-1) 
    # But we just take the middle peak which is our maximum
    return smooth_max(corr, beta)

def project_to_constraints(h, dx):
    """Project to satisfy integration constraint with clipping"""
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    
    # Scale to have the right sum
    h = h * target_sum / current_sum
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust again to ensure correct sum exactly
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Increase the smallest value
        idx = np.argmin(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def gradient_smooth(h, dx, beta=10.0):
    """Gradient of smooth max of correlations"""
    n = len(h)
    # Compute the correlations
    corr_vals = compute_correlations(h, dx)
    # Compute exp(beta * corr_vals)
    exp_corr = np.exp(beta * corr_vals)
    # Softmax weights
    weights = exp_corr / np.sum(exp_corr)
    # Now compute gradient of the smooth max wrt h
    # This is much simpler using the fact that
    # d/dh smooth_max(corr) = sum_i weights_i * d/dh corr_i
    # We compute d/dh corr_i using convolution
    # We need d/dh ([h]*(1-h)) where * is convolution  
    # The gradient w.r.t. h of corr(x, y) at position k 
    # is sum_j dh[k-j] * d(1-h[j])/dh[k-j] at k = j -- no, wrong
    # Actually we have for correlation with y:
    # d/dx [x*y](k) = sum_j d/dx[x(j)*y(k-j)] = y(k-j) 
    # So partial of corr with respect to h_i = [1-h]_{k-i}
    # That's just another correlation, just shift that
    # Let's use FFT and direct computation for accuracy
    # For each element h_i, we compute the gradient of value
    # grad is sum_{k_i} weights_k * \frac{\partial}{\partial h_i} corr_k = weights_k * (1-h_{k-i}) if i is valid ...
    # Let me slightly redefine by computing derivative directly.
    
    # Efficient FFT-based gradient computation for correlation
    # Correlation of h and (1-h) is conv(h, 1-h)
    # The gradient is a bit tricky.
    # corr = corr vals (2*n-1 elements)
    # grad_i relates to partial derivative of corr with respect to h_i
    grad = np.zeros_like(h)
    
    # For each element i of h,
    # \frac{\partial}{\partial h_i} corr_k = coefficient of h_i in correlation sum
    # The derivative of ( h_i * (1-h_{i-k}) ) = (1-h_{i-k}) or (1-h_{i+k}) but correctly placed
    
    # More explicitly, let's compute it by shifting for code clarity:
    W = weights
    # We need to compute weighted avg of gradients of different lags
    
    # Let's compute it in frequency domain 
    # if we are computing conv(h, 1-h)
    # and want grad w.r.t to h_i 
    # cross-correlation (1-h) @ h shifted by i will affect it
    # This is messy just compute for now exactly linear gradients
    
    # Start from definition
    # Let's approach by finite differences for stable computation
    grad = np.zeros_like(h)
    
    # Compute by finite differences
    eps = 1e-6
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_minus = h.copy()
        h_minus[i] -= eps
        val_plus = evaluate_smooth(h_plus, dx, beta)
        val_minus = evaluate_smooth(h_minus, dx, beta)
        grad[i] = (val_plus - val_minus) / (2 * eps)
    
    return grad

def optimized_evaluate(h, dx, beta=10.0):
    """Fast evaluation that avoids recomputation"""
    corr = compute_correlations(h, dx)
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with low resolution
    n_points = 50  # good balance of small enough for optimization but fine enough 
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        # Interpolate
        h_initial = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n_points), 
                      np.linspace(0, 2, len(h_initial)), h_initial)
    else:
        # Initialize with some structure
        h = np.random.uniform(0, 1, n_points)
        
    # Ensure initial integrate constraint
    h = project_to_constraints(h, dx)
    
    best_h = h.copy()
    best_c5 = optimized_evaluate(h, dx)
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 1e-3
    
    # Loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Get current gradient
        try:
            grad = gradient_smooth(h, dx, beta=10.0)
        except:
            grad = np.random.normal(0, 1e-6, len(h))
            
        # Update h
        h = h - learning_rate * grad
        h = project_to_constraints(h, dx)
        
        # Evaluate result
        c5_new = optimized_evaluate(h, dx)
        
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new

        # Adaptive learning rate decay
        if iteration > 0 and iteration % 500 == 0:
            learning_rate *= 0.5
            
    return best_h.tolist(), best_c5, len(best_h)
