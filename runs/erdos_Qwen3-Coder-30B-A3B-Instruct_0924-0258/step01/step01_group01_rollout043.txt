import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, dx):
    n = len(h)
    current_sum = np.sum(h)
    target_sum = n / 2.0
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    h = h * target_sum / current_sum
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(h)
        h[idx] = min(h[idx] + diff, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_low = 10
    n_high = 500
    n_points = n_low
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]
    else:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_constraints(h, dx)
    
    # Track best
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_c5 = best_c5
    
    # Gradient descent approach with adaptive steps
    iter_count = 0
    max_iter_per_res = 1000
    resolution_doubling_threshold = 0.1
    
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Optimization for current resolution
        improvement_count = 0
        for _ in range(max_iter_per_res):
            if time.time() - start_time > budget_s:
                break
                
            # Compute gradient information
            c5 = evaluate(h, dx)
            
            # Approximate gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] = min(h[i] + eps, 1.0)
                h_plus = project_to_constraints(h_plus, dx)
                c5_plus = evaluate(h_plus, dx)
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update step
            step_size = 0.01
            h_new = h - step_size * grad
            
            # Project to valid space
            h_new = project_to_constraints(h_new, dx)
            
            c5_new = evaluate(h_new, dx)
            
            if c5_new < last_c5:
                h = h_new
                last_c5 = c5_new
                if last_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = last_c5
                improvement_count += 1
            elif np.random.random() < 0.05:  # Random acceptance with low probability
                h = h_new
                last_c5 = c5_new
                
            iter_count += 1
            
        # Adaptive refinement
        if n_points < n_high and improvement_count > max_iter_per_res * resolution_doubling_threshold:
            h_long = np.interp(np.linspace(0, 2, 2*n_points), np.linspace(0, 2, n_points), h)
            h = h_long
            n_points = 2 * n_points
            dx = 2.0 / n_points
            h = project_to_constraints(h, dx)
            last_c5 = evaluate(h, dx)
    
    return best_h.tolist(), best_c5, len(best_h)
